def conv_full(a, b):
    n = len(a) + len(b) - 1
    size = 1 << int(np.ceil(np.log2(n)))
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]

import time, math, numpy as np

# ---------- basic utilities ----------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y

def random_feasible_vector(n, rng):
    x = rng.random(n)
    return project_box_sum(x, n / 2.0)

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def compute_c5(h):
    n = h.size
    # linear correlation using FFT for speed
    size = 1 << int(np.ceil(np.log2(2 * n)))
    fh = np.fft.rfft(h, size)
    fg = np.fft.rfft((1.0 - h)[::-1], size)
    corr = np.fft.irfft(fh * fg, size)[: 2 * n - 1]
    return np.max(corr) * (2.0 / n)

def compute_c5_and_idx(h):
    n = h.size
    size = 1 << int(np.ceil(np.log2(2 * n)))
    fh = np.fft.rfft(h, size)
    fg = np.fft.rfft((1.0 - h)[::-1], size)
    corr = np.fft.irfft(fh * fg, size)[: 2 * n - 1]
    idx = int(np.argmax(corr))
    return corr[idx] * (2.0 / n), idx

# ---------- convolution helpers ----------
def conv_full(a, b):
    n = len(a) + len(b) - 1
    size = 1 << int(np.ceil(np.log2(n)))
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]

def corr_full(h):
    n = h.size
    size = 1 << int(np.ceil(np.log2(2 * n)))
    fh = np.fft.rfft(h, size)
    fg = np.fft.rfft((1.0 - h)[::-1], size)
    return np.fft.irfft(fh * fg, size)[: 2 * n - 1]

# ---------- smooth gradient descent ----------
def smooth_grad_descent(h, beta, lr, max_iter, deadline):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        c = corr_full(h)                     # length 2n-1
        bc = beta * c
        max_bc = np.max(bc)
        w = np.exp(bc - max_bc)
        w /= w.sum()
        conv1 = conv_full(w, h)              # length 3n-2
        conv2 = conv_full(w[::-1], h)
        start = n - 1
        grad = 1.0 - (conv1[start:start + n] + conv2[start:start + n])
        h_new = h - lr * grad
        h_new = project_box_sum(np.clip(h_new, 0.0, 1.0), n / 2.0)
        new_c = compute_c5(h_new)
        if new_c < best_c - 1e-12:
            best_c, best_h = new_c, h_new.copy()
        else:
            lr *= 0.5
        h = h_new
        if lr < 1e-7:
            break
    return best_h, best_c

# ---------- hill climbing (targeted) ----------
def targeted_move(h, rng, k, scale=0.2):
    n = h.size
    if k <= 0 or k >= n:
        return perturb_h(h, rng, scale)
    low_mask = (h[: n - k] < 0.5) & (h[k:] < 0.5)
    high_mask = (h[: n - k] > 0.5) & (h[k:] > 0.5)
    if not low_mask.any() or not high_mask.any():
        return perturb_h(h, rng, scale)
    i_low = rng.choice(np.flatnonzero(low_mask))
    i_high = rng.choice(np.flatnonzero(high_mask))
    max_eps = min(
        0.5 - h[i_low],
        0.5 - h[i_low + k],
        h[i_high] - 0.5,
        h[i_high + k] - 0.5,
    )
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i_low] += eps
    nh[i_low + k] += eps
    nh[i_high] -= eps
    nh[i_high + k] -= eps
    return nh

def hill_climb_targeted(h_start, rng, max_iters=500, init_step=0.4, n_trials=4, p_target=0.25):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    n = h.shape[0]

    for it in range(max_iters):
        _, idx = compute_c5_and_idx(h)
        k_star = idx - (n - 1)
        improved = False
        for _ in range(n_trials):
            if rng.random() < p_target:
                cand = targeted_move(h, rng, k_star, scale=step)
            else:
                cand = perturb_h(h, rng, scale=step)
            c = compute_c5(cand)
            if c < best_c - 1e-12:
                best_c, h = c, cand
                improved = True
                break
        if not improved and rng.random() < 0.02:
            cand = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(cand)
            if c < best_c:
                best_c, h = c, cand
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return h, best_c

# ---------- main driver ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # ----- choose resolution -----
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        n0 = init.size
        target_n = max(720, n0 * 2)
        target_n = min(target_n, 1008)  # keep moderate
        if n0 != target_n:
            old_x = (np.arange(n0) + 0.5) * (2.0 / n0)
            new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
            h = np.interp(new_x, old_x, init)
        else:
            h = init.copy()
        h = np.clip(h, 0.0, 1.0)
        h = project_box_sum(h, target_n / 2.0)
    else:
        target_n = 720
        h = random_feasible_vector(target_n, rng)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    beta_schedule = [20, 80, 300, 1000, 3000, 10000, 30000, 80000]
    lr = 0.5

    # ----- main stochastic search -----
    while time.time() - start_time < budget_s * 0.88:
        # start from a perturbed best or fresh random point
        if rng.random() < 0.2:
            h0 = random_feasible_vector(target_n, rng)
        else:
            h0 = perturb_h(best_h, rng, scale=0.3)
        cur_h = h0.copy()
        # sweep beta values
        for beta in beta_schedule:
            deadline = start_time + budget_s * 0.88
            cur_h, cur_c = smooth_grad_descent(cur_h, beta, lr, max_iter=200, deadline=deadline)
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
                lr = max(lr * 0.9, 0.01)
            lr = max(lr * 0.95, 0.01)
            if time.time() - start_time >= budget_s * 0.88:
                break
        # quick targeted hill‑climb for polishing
        if time.time() - start_time < budget_s * 0.88:
            h_hc, c_hc = hill_climb_targeted(best_h, rng, max_iters=300, init_step=0.2)
            if c_hc < best_c:
                best_c, best_h = c_hc, h_hc.copy()

    # ----- final projection and evaluation -----
    best_h = project_box_sum(np.clip(best_h, 0.0, 1.0), best_h.size / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
