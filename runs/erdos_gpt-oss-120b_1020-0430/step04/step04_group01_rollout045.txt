def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    # Simple proportional scaling to reduce diff while staying within [0,1]
    # Use iterative redistribution like before
    for _ in range(10000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def parametric_search(n, rng, deadline):
    best_h = None

    # coarse grid

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n/2.0, rng)
    else:
        n = 720
        h0 = random_feasible_vector(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # parametric trap search
    trap_deadline = start_time + 0.2 * budget_s
    trap_h, trap_c, trap_w, trap_p = parametric_search(n, rng, trap_deadline)
    if trap_c < best_c:
        best_h, best_c = trap_h, trap_c

    # main optimization loop
    beta_list = [10, 50, 200, 800, 2000, 5000]
    while time.time() < deadline:
        # random start or current best
        if rng.random() < 0.7:
            start = perturb_h(best_h, rng, scale=0.3)
        else:
            start = random_feasible_vector(n, rng)
        # SLSQP ladder
        for beta in beta_list:
            if time.time() >= deadline:
                break
            try:
                h_opt = slsqp_optimize(start, beta, max_iter=300)
                c_opt = compute_c5(h_opt)
                if c_opt < best_c:
                    best_h, best_c = h_opt.copy(), c_opt
                start = h_opt
            except Exception:
                continue
        # hill climb
        if time.time() >= deadline:
            break
        h_hc, c_hc = hill_climb(best_h, rng, max_iters=2000, start_step=0.2)
        if c_hc < best_c:
            best_h, best_c = h_hc.copy(), c_hc

    # final cleanup
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size/2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)

import time
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Helper utilities
# --------------------------------------------------------------
def enforce_sum(h, target, rng):
    """Adjust h so that Σh == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def correlate_full(h):
    """Full linear correlation of h with (1‑h) using FFT (zero‑padding)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)


# --------------------------------------------------------------
# Smooth surrogate for SLSQP (log‑sum‑exp)
# --------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = correlate_full(x)
    bc = CURRENT_BETA * corr
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        x = enforce_sum(x, CURRENT_TARGET, rng)
    return x

def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Greedy hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h.copy()
            h = new_h
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


# --------------------------------------------------------------
# Parametric trapezoid shape
# --------------------------------------------------------------
def make_trapezoid(w, p, n):
    """Trapezoid with top width w, left foot at p, area = 1."""
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx
    a = 1.0 / (1.0 + w / 2.0)  # height guaranteeing area 1
    # slopes
    left_slope = a / p if p > 1e-12 else 0.0
    right_len = 2.0 - (p + w)
    right_slope = a / right_len if right_len > 1e-12 else 0.0
    h = np.empty_like(xs)
    # left ascent
    mask_left = xs < p
    h[mask_left] = left_slope * xs[mask_left]
    # top plateau
    mask_top = (xs >= p) & (xs <= p + w)
    h[mask_top] = a
    # right descent
    mask_right = xs > p + w
    h[mask_right] = a - right_slope * (xs[mask_right] - (p + w))
    # clip for safety
    np.clip(h, 0.0, 1.0, out=h)
    return h


def parametric_search(n, rng, deadline):
    """Search the two‑parameter trapezoid family."""
    best_h = None
    best_c = float("inf")
    best_w = 0.0
    best_p = 0.0
    # coarse grid
    for w in np.linspace(0.0, 1.5, 31):
        p_center = (2.0 - w) / 2.0
        h = make_trapezoid(w, p_center, n)
        h = enforce_sum(h, n / 2.0, rng)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
            best_w, best_p = w, p_center
    # local random refinement
    it = 0
    while time.time() < deadline and it < 2000:
        w = best_w + rng.normal(scale=0.07)
        w = np.clip(w, 0.0, 1.5)
        p = best_p + rng.normal(scale=0.07)
        p = np.clip(p, 0.0, 2.0 - w)
        h = make_trapezoid(w, p, n)
        h = enforce_sum(h, n / 2.0, rng)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
            best_w, best_p = w, p
        else:
            if rng.random() < 0.02:
                best_w = rng.uniform(0.0, 1.5)
                best_p = rng.uniform(0.0, 2.0 - best_w)
        it += 1
    return best_h, best_c, best_w, best_p
