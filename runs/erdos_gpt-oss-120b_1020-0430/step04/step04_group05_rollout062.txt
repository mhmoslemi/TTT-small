import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic helpers ----------
def enforce_sum(h, target, rng, max_iter=2000):
    """Clip to [0,1] and adjust to meet exact sum `target`."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # final tiny correction
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

def compute_c5(h):
    """Exact C5 value for a vector h."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def compute_corr(h):
    """Full correlation needed for incremental updates."""
    return np.correlate(h, 1 - h, mode="full")

# ---------- hill climbing ----------
def hill_climb(h, corr, rng, deadline,
               delta_start=0.2, delta_decay=0.9999,
               top_frac=0.02, improve_only=True):
    """
    Fast O(n) hill climb that conserves the sum.
    Returns best vector and its max correlation value.
    """
    n = h.size
    mid = n - 1
    cur_h = h.copy()
    cur_corr = corr.copy()
    cur_max = cur_corr.max()
    best_h = cur_h.copy()
    best_max = cur_max
    delta = delta_start
    one_minus = 1.0 - cur_h

    while time.time() < deadline:
        # worst lag
        lag = int(np.argmax(cur_corr) - mid)
        # contributions for this lag
        if lag >= 0:
            length = n - lag
            i_cand = np.arange(length)
            contrib = cur_h[:length] * (1.0 - cur_h[lag:])
        else:
            l = -lag
            length = n - l
            i_cand = np.arange(l, n)
            contrib = cur_h[l:] * (1.0 - cur_h[:length])
        if length == 0:
            # degenerate, pick random i
            i = rng.integers(0, n)
        else:
            k = max(1, int(len(i_cand) * top_frac))
            top_idx = i_cand[np.argpartition(-contrib, k - 1)[:k]]
            i = int(rng.choice(top_idx))
        # pick a target j different from i
        attempts = 0
        while True:
            j = rng.integers(0, n)
            if j != i and cur_h[j] < 1.0:
                break
            attempts += 1
            if attempts > 10:
                j = (i + 1) % n
                break
        # allowed move size
        max_move = min(delta, cur_h[i], 1.0 - cur_h[j])
        if max_move <= 1e-12:
            delta *= delta_decay
            continue
        eps = max_move * rng.random()
        if eps < 1e-14:
            delta *= delta_decay
            continue

        # propose new correlation (approximate, ignores O(eps^2) cross terms)
        new_corr = cur_corr.copy()
        new_corr[mid - i:mid - i + n] += (-eps) * one_minus
        new_corr[mid - j:mid - j + n] += eps * one_minus
        new_max = new_corr.max()

        accept = False
        if improve_only:
            accept = new_max < cur_max - 1e-12
        else:
            # simple simulated annealing
            if new_max < cur_max:
                accept = True
            else:
                T = 0.01 * (new_max - cur_max + 1e-12)
                if rng.random() < np.exp(-(new_max - cur_max) / max(T, 1e-12)):
                    accept = True

        if accept:
            # apply real update
            cur_h[i] -= eps
            cur_h[j] += eps
            one_minus = 1.0 - cur_h
            cur_corr = new_corr
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = cur_h.copy()
        # else discard changes automatically (new_corr is temporary)

        delta *= delta_decay
    return best_h, best_max

# ---------- SLSQP smooth max ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize_one_beta(h0, beta, rng, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # repair possible sum drift
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x

# ---------- main run ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    best_c5 = best_max * (2.0 / best_h.size)

    # ---------- optional up‑sampling ----------

    remaining = budget_s - (mid_time - t0)

            up_h = enforce_sum(up_h, n2 / 2.0, rng)

            for beta in beta_list:
