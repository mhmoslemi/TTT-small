def gd_max_corr(h, target, deadline, initial_step=0.5):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    step = initial_step
    iter = 0
    while time.time() < deadline:
        # compute correlation
        conv = correlate_full(h)
        max_val = conv.max()
        # early exit if max <= best_c? can only improve.
        # find all max idx
        max_idxs = np.flatnonzero(conv == max_val)
        # average gradient from those shifts
        g = np.zeros_like(h)
        for idx in max_idxs:
            # shift = idx - (n - 1) (since full correlation length 2n-1)
            shift = idx - (n - 1)
            # compute gradient for shift
            if shift >= 0:
                # add h[i+shift] for i < n-shift
                g[:n-shift] += h[shift:]
                # add h[i-shift] for i >= shift => i-shift >=0 -> i >= shift
                g[shift:] += h[:n-shift]
            else:
                s = -shift
                # shift negative: similar
                g[s:] += h[:n-s]
                g[:n-s] += h[s:]
        g = g / len(max_idxs)  # average

        return h0
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter':max_iter,'ftol':1e-12,'disp':False})

def binary_sa(start_bin, rng, deadline):
    h = start_bin.astype(float).copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    # parameters
    T0 = best_c * 0.02 + 1e-6
    Tf = 1e-7
    max_iters = 200000  # large
    it = 0
    while time.time() < deadline and it < max_iters:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if h[i] == h[j]: # both same (0/0 or 1/1)
            it+=1
            continue
        # ensure i has 1 and j has 0
        if h[i] < h[j]:
            i, j = j, i
        # swap
        h[i] = 0.0
        h[j] = 1.0
        c = compute_c5(h)
        frac = it / max_iters
        T = T0 * ((Tf/T0) ** frac)
        delta = c - best_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T,1e-12)):
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
        it += 1
    return best_h, best_c

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0,n)
    j = rng.integers(0,n)
    if i==j:
        return h.copy()
    # max epsilon allowed from i to j to stay within bounds
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random()*2-1) * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def project_onto_bounds_and_sum(v, target):
    # ensure within [0,1] initially
    v = np.clip(v, 0.0, 1.0)
    # compute diff
    diff = target - v.sum()
    # iterate
    # Use while loop
    # maintain mask of free indices (where we can adjust in direction of diff)
    max_iter = 10000
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = v < 1.0 - 1e-12
        else:
            free = v > 1e-12
        if not np.any(free):
            # cannot adjust further
            break
        # distribution amount
        inc = diff / free.sum()
        # apply inc but keep within bounds
        if diff > 0:
            # add, limited by 1
            add = np.minimum(inc, 1.0 - v[free])
            v[free] += add
            diff -= add.sum()
        else:
            # subtract, limited by 0
            sub = np.minimum(-inc, v[free])
            v[free] -= sub
            diff += sub.sum()
    # final clip
    return np.clip(v, 0.0, 1.0)

def correlate_full(h):
    n = h.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1-h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[:2*n-1]
    return conv

import time
import numpy as np
from scipy.optimize import minimize

# ---------- projection onto [0,1] and sum constraint ----------
def project_onto_bounds_and_sum(v, target):
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    # water‑filling style fix
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = v < 1.0 - 1e-12
            if not np.any(free):
                break
            inc = diff / free.sum()
            add = np.minimum(inc, 1.0 - v[free])
            v[free] += add
            diff -= add.sum()
        else:
            free = v > 1e-12
            if not np.any(free):
                break
            dec = -diff / free.sum()
            sub = np.minimum(dec, v[free])
            v[free] -= sub
            diff += sub.sum()
    return np.clip(v, 0.0, 1.0)


# ---------- basic feasible vector ----------
def random_feasible_vector(n, rng):
    target = n / 2.0
    v = rng.random(n)
    v = v / v.sum() * target
    v = project_onto_bounds_and_sum(v, target)
    return v


# ---------- small perturbation (preserves sum) ----------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() * 2.0 - 1.0) * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


# ---------- correlation utilities ----------
def correlate_full(h):
    """Linear correlation of h with (1‑h) (full length)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1.0 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Discrete C5 for step function h (dx = 2/n)."""
    return correlate_full(h).max() * (2.0 / h.size)


# ---------- gradient of the maximal overlap ----------
def max_conv_gradient(h):
    """Sub‑gradient of max_k Σ h_i (1‑h_{i+k}) w.r.t. h."""
    n = h.size
    conv = correlate_full(h)
    max_val = conv.max()
    idxs = np.flatnonzero(conv == max_val)
    g = np.zeros_like(h)
    for idx in idxs:
        shift = idx - (n - 1)          # can be negative
        grad = np.ones_like(h)
        if shift >= 0:
            grad[: n - shift] -= h[shift:]
            grad[shift:] -= h[: n - shift]
        else:
            s = -shift
            grad[s:] -= h[: n - s]
            grad[: n - s] -= h[s:]
        g += grad
    return g / len(idxs)


# ---------- smooth surrogate for SLSQP ----------
CURRENT_TARGET = 0.0
CURRENT_BETA = 0.0


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    conv = correlate_full(x)
    bc = CURRENT_BETA * conv
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter, deadline):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    if time.time() >= deadline:
        return h0
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_bounds_and_sum(x, CURRENT_TARGET)
    return x


# ---------- binary simulated annealing ----------
def binary_sa(start_bin, rng, deadline):
    h = start_bin.astype(float).copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    T0 = best_c * 0.02 + 1e-6
    Tf = 1e-7
    max_iters = 250000
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if h[i] == h[j]:
            continue
        if h[i] < h[j]:
            i, j = j, i
        h[i] = 0.0
        h[j] = 1.0
        c = compute_c5(h)
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        delta = c - best_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c
