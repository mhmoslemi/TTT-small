def compute_c5_fft(h):
    n = h.size
    L = 2*n-1
    M = 1 << (L-1).bit_length()
    # compute convolution of h and (1-h) reversed
    # correlation: a ⊛ b: (a * reverse(b)) padded.
    # So compute FFT of a and reverse(b):
    a = np.fft.rfft(h, M)
    b = np.fft.rfft((1-h)[::-1], M)
    corr = np.fft.irfft(a * b, M)[:L]
    return np.max(corr) * (2.0 / n)

def enforce_sum(h, target, rng):
    # Clip h to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # Compute diff
    diff = target - h.sum()
    # If diff near zero, return
    if abs(diff) < 1e-12:
        return h
    # Use simple waterfilling: If diff >0 (need increase), fill up to 1 proportionally
    # else if diff<0 (need decrease), reduce down to 0 proportionally.
    # This ensures sum exactly while staying within bounds.
    # But need careful to avoid overshoot. We'll implement iterative approach:
    max_iter = 1000
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # candidates where h < 1
            candidates = np.where(h < 1.0 - 1e-12)[0]
            if candidates.size == 0:
                break
            # distribute diff uniformly among candidates (or random)
            # a simple approach: increase each candidate by min(diff/len(candidates), 1-h)
            inc = diff / candidates.size
            # but cap at 1-h
            increase = np.minimum(inc, 1.0 - h[candidates])
            # if some couldn't take full inc, we need to recompute diff
            h[candidates] += increase
            diff = target - h.sum()
        else:
            candidates = np.where(h > 1e-12)[0]
            if candidates.size == 0:
                break
            dec = -diff / candidates.size
            decrease = np.minimum(dec, h[candidates])
            h[candidates] -= decrease
            diff = target - h.sum()
    # final fallback: adjust one random element
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

        frac = elapsed / (time_budget * 0.95)

def gradient_shift(h, s):
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[:n - s] += (1 - h[s:])
        grad[s:] -= h[:n - s]
    else:
        s_abs = -s
        grad[s_abs:] += (1 - h[:n - s_abs])
        grad[:n - s_abs] -= h[s_abs:]
    return grad

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utilities ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def enforce_sum(h, target, rng):
    """Clip to [0,1] and adjust to make sum exactly target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # simple water‑filling iteration
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            add = diff / idx.size
            inc = np.minimum(add, 1.0 - h[idx])
            h[idx] += inc
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            sub = (-diff) / idx.size
            dec = np.minimum(sub, h[idx])
            h[idx] -= dec
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h


def random_feasible_vector(n, rng):
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)


def get_fft_params(n):
    L = 2 * n - 1
    M = 1 << (L - 1).bit_length()  # next power of two
    return L, M


def compute_c5_fft(h):
    """C5 using FFT (fast)."""
    n = h.size
    L, M = get_fft_params(n)
    a = np.fft.rfft(h, M)
    b = np.fft.rfft((1 - h)[::-1], M)
    corr = np.fft.irfft(a * b, M)[:L]
    return np.max(corr) * (2.0 / n)


def worst_shift_info(h):
    """Return worst shift s, index, and correlation value."""
    n = h.size
    L, M = get_fft_params(n)
    a = np.fft.rfft(h, M)
    b = np.fft.rfft((1 - h)[::-1], M)
    corr = np.fft.irfft(a * b, M)[:L]
    idx = np.argmax(corr)
    s = idx - (n - 1)
    return s, idx, corr[idx]


def gradient_shift(h, s):
    """Gradient of corr_s w.r.t. h."""
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[: n - s] += (1 - h[s:])
        grad[s:] -= h[: n - s]
    else:
        s_abs = -s
        grad[s_abs:] += (1 - h[: n - s_abs])
        grad[: n - s_abs] -= h[s_abs:]
    return grad


def greedy_optimize2(h, max_steps=5000):
    """Deterministic improvement using full‑epsilon transfers."""
    n = h.size
    for _ in range(max_steps):
        s, _, cur_corr = worst_shift_info(h)
        grad = gradient_shift(h, s)
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-12:
            break
        # try full epsilon
        nh = h.copy()
        nh[i] -= max_eps
        nh[j] += max_eps
        if compute_c5_fft(nh) + 1e-12 < cur_corr:
            h = nh
            continue
        # try half epsilon
        eps = max_eps * 0.5
        nh = h.copy()
        nh[i] -= eps
        nh[j] += eps
        if compute_c5_fft(nh) + 1e-12 < cur_corr:
            h = nh
            continue
        break
    return h


def perturb_random(h, rng, scale=0.2):
    """Move a random epsilon from i to j."""
    n = h.size
    i = rng.integers(n)
    j = rng.integers(n - 1)
    if j >= i:
        j += 1
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


# ---------- SLSQP ----------
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=1200):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x
