def subgradient_optimize(h0, rng, max_iters, step0=0.5, decay=0.999, eps=1e-9):
    h = h0.copy()
    n = h.shape[0]
    step = step0
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iters):
        # compute conv
        conv = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
        max_val = conv.max()
        # tolerance for max selection; select all values within 1e-9 relative
        # use tolerance = max(1e-12, max_val * 1e-9)
        tol = max(1e-12, max_val * 1e-9)
        mask = conv >= max_val - tol
        # get shift indices
        idxs = np.nonzero(mask)[0]
        shifts = idxs - (n - 1)
        # compute sum of shifted h
        sum_shifted = np.zeros_like(h)
        for k in shifts:
            # shift left
            if k >= 0:
                a_plus = np.concatenate((h[k:], np.zeros(k)))
                a_minus = np.concatenate((np.zeros(k), h[:n - k]))
            else:
                k2 = -k
                a_plus = np.concatenate((np.zeros(k2), h[:n - k2]))
                a_minus = np.concatenate((h[k2:], np.zeros(k2)))
            sum_shifted += a_plus + a_minus
        lenK = len(shifts)
        g = lenK - sum_shifted
        # step
        h = h - step * g
        # project
        h = np.clip(h, 0.0, 1.0)
        # enforce sum
        target = n / 2.0
        diff = target - h.sum()
        # using simple waterfilling
        if abs(diff) > 1e-12:
            # capacity to add or remove per free index
            # find free
            free = (h > 1e-12) & (h < 1 - 1e-12)
            free_idx = np.where(free)[0]
            if free_idx.size:
                per = diff / free_idx.size
                h[free_idx] += per
                h = np.clip(h, 0.0, 1.0)
            else:
                # If no free, distribute to all with tolerance
                h += diff / n
                h = np.clip(h, 0.0, 1.0)
            # re-evaluate diff after clipping
            diff = target - h.sum()
            if abs(diff) > 1e-12:
                # fallback iterative
                for _ in range(100):
                    if abs(diff) < 1e-14:
                        break
                    i = rng.integers(0, n)
                    add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                    h[i] += add
                    diff = target - h.sum()
        # evaluate
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        # decay step
        step *= decay
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global variables for the smooth surrogate used by SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    """Equality constraint: sum(x) == CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """Smooth approximation of max_k correlation using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def compute_c5(h):
    """Exact discrete C5 for a step function sampled on [0,2]."""
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())          # next power of two >= L
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)

def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h so that Σh = target while keeping 0≤h≤1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-9:
        free = (h > 1e-12) & (h < 1 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            per = diff / idx.size
            h[idx] += per
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with Σh = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.5
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

def subgradient_optimize(h0, rng, max_iters, step0, decay, end_time):
    """Sub‑gradient descent on the exact max‑overlap objective."""
    h = h0.copy()
    n = h.shape[0]
    step = step0
    best_h = h.copy()
    best_c = compute_c5(h)
    target = n / 2.0
    while max_iters > 0:
        if time.time() >= end_time:
            break
        # correlation and max value
        conv = np.correlate(h, 1 - h, mode="full") * (2.0 / n)
        max_val = conv.max()
        tol = max(max_val * 1e-12, 1e-12)
        mask = conv >= max_val - tol
        shifts_all = np.nonzero(mask)[0] - (n - 1)
        # limit number of shifts to keep cost low
        if shifts_all.size > 12:
            shifts = rng.choice(shifts_all, size=12, replace=False)
        else:
            shifts = shifts_all
        # accumulate shifted sums
        sum_shifted = np.zeros_like(h)
        for k in shifts:
            if k >= 0:
                a_plus = np.concatenate((h[k:], np.zeros(k)))
                a_minus = np.concatenate((np.zeros(k), h[:n - k]))
            else:
                k2 = -k
                a_plus = np.concatenate((np.zeros(k2), h[:n - k2]))
                a_minus = np.concatenate((h[k2:], np.zeros(k2)))
            sum_shifted += a_plus + a_minus
        lenK = shifts.size
        g = lenK - sum_shifted  # sub‑gradient of the max
        # gradient step
        h -= step * g
        # project onto box and sum hyperplane
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            free = (h > 1e-12) & (h < 1 - 1e-12)
            idx = np.where(free)[0]
            if idx.size:
                per = diff / idx.size
                h[idx] += per
                h = np.clip(h, 0.0, 1.0)
            else:
                h += diff / n
                h = np.clip(h, 0.0, 1.0)
        # evaluate
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        step *= decay
        max_iters -= 1
    return best_h, best_c

def slsqp_optimize(h0, beta, max_iter=300):
    """SLSQP refinement using the smooth log‑sum‑exp surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # drift correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

        candidates.append(h0)

    # ------------------------------------------------------------------
    # Pick the best from the pool
    # ------------------------------------------------------------------
    best_h = None
    best_c = np.inf
    for cand in candidates:
        c = compute_c5(cand)
        if c < best_c:
            best_c, best_h = c, cand.copy()

    # ------------------------------------------------------------------
    # Phase 1 – sub‑gradient descent with random restarts
    # ------------------------------------------------------------------
    phase1_end = start_time + budget_s * 0.55
    while time.time() < phase1_end:
        start_vec = best_h.copy() if rng.random() < 0.7 else random_feasible_vector(n, rng)
        # generous iteration cap; loop will stop early due to time
        h_opt, c_opt = subgradient_optimize(
            start_vec,
            rng,
            max_iters=50000,
            step0=0.5,
            decay=0.9995,
            end_time=phase1_end,
        )
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()

    # ------------------------------------------------------------------
    # Phase 2 – up‑sampling and refinement (if time permits)
    # ------------------------------------------------------------------
    elapsed = time.time() - start_time
    if budget_s - elapsed > budget_s * 0.15:
        target_n = min(720, max(n * 2, n + 120))
        if target_n != n:
            old_x = (np.arange(n) + 0.5) * (2.0 / n)
            new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
            up_h = np.interp(new_x, old_x, best_h)
            up_h = enforce_sum(up_h, target_n / 2.0, rng)
            # sub‑gradient on finer grid
            up_end = start_time + budget_s * 0.95
            up_h, c_up = subgradient_optimize(
                up_h,
                rng,
                max_iters=80000,
                step0=0.4,
                decay=0.9997,
                end_time=up_end,
            )
            if c_up < best_c:
                best_c, best_h, n = c_up, up_h.copy(), target_n

    # ------------------------------------------------------------------
    # Phase 3 – final smooth SLSQP fine‑tuning (if time remains)
    # ------------------------------------------------------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 5.0:
        for beta in (100, 500, 2000, 8000):
            if time.time() - start_time > budget_s - 4.0:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, max_iter=800)
                c_ref = compute_c5(h_ref)
                if c_ref < best_c:
                    best_c, best_h = c_ref, h_ref.copy()
            except Exception:
                continue

    # ------------------------------------------------------------------
    # Final clean‑up
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
