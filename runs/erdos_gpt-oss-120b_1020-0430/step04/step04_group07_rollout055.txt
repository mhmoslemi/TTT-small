def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None; best_c = np.inf; best_n = None

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0,0,1)
        h0 = enforce_sum(h0, rng)
        best_h = h0; best_n = h0.size; best_c = compute_c5(best_h)

    # define possible n
    n_list = [256, 320, 384, 512]

    # main loop
    while time.time() - start < 0.85 * budget_s:
        # pick n
        n = int(rng.choice(n_list))
        # start with best known but resized or new random
        if best_h is not None and rng.random() < 0.7:
            if best_n != n:

import time
import numpy as np
from scipy.optimize import minimize

# globals for the soft‑max SLSQP optimizer
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector with entries in [0,1] and sum=n/2."""
    h = rng.random(n)
    target = n / 2.0
    h = h * (target / h.sum())
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def hill_climb(h: np.ndarray, rng: np.random.Generator,
               max_iters: int = 1200, step_scale: float = 0.8):
    """Steepest‑descent hill climbing using the worst‑shift gradient."""
    n = h.size
    # initial score
    corr = np.correlate(h, 1 - h, mode="full")
    best_c = np.max(corr) * (2.0 / n)
    best_h = h.copy()
    scale = step_scale
    decay = max(1, max_iters // 5)

    for it in range(max_iters):
        # correlation & worst shift for current h
        corr = np.correlate(h, 1 - h, mode="full")
        c = np.max(corr) * (2.0 / n)
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)

        # gradient of the worst shift
        if k == 0:
            grad = 1.0 - 2.0 * h
        else:
            grad = np.ones_like(h)
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]

        nh = guided_perturb(h, grad, rng, scale=scale)
        # evaluate candidate
        corr_nh = np.correlate(nh, 1 - nh, mode="full")
        c_nh = np.max(corr_nh) * (2.0 / n)

        if c_nh < best_c:
            best_c, best_h, h = c_nh, nh.copy(), nh

        if (it + 1) % decay == 0:
            scale = max(scale * 0.7, 1e-5)

    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 8) -> np.ndarray:
    """Fine‑tune using the worst‑shift gradient with decreasing step size."""
    n = h.size
    for it in range(max_iter):
        # worst‑shift gradient
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            g = np.ones_like(h)
            g[: n - k] -= h[k:]
            g[k:] -= h[: n - k]

        h = h - step * (0.5 ** it) * g
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h


def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 800) -> np.ndarray:
    """Soft‑max SLSQP optimisation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta

    def sum_eq(x):
        return np.sum(x) - CURRENT_TARGET

    def softmax_obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = CURRENT_BETA * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x


def symmetrize(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force mirror symmetry about the centre of the interval."""
    hs = (h + h[::-1]) * 0.5
    return enforce_sum(hs, rng)

        h_up, c_up = hill_climb(h_up, rng, max_iters=1800, step_scale=0.7)

        for beta in (20.0, 200.0, 2000.0, 8000.0, 20000.0):
            if time.time() - start > budget_s:
                break
            h_up = slsqp_optimize(h_up, beta, rng, max_iter=600)
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, best_n = h_up.copy(), c_up, n_up

        h_up = gradient_refine(h_up, rng, step=0.006, max_iter=10)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up.copy(), c_up, n_up

        # prepare for possible further up‑sampling
        best_h, best_n = h_up, n_up

    # --- stochastic tweaking until time runs out --------------------
    while time.time() - start < budget_s * 0.999 and best_h is not None:
        grad = (1.0 - 2.0 * best_h) if best_h.size == 1 else None  # placeholder
        # compute worst‑shift gradient for the current best
        corr = np.correlate(best_h, 1 - best_h, mode="full")
        idx = np.argmax(corr) - (best_h.size - 1)
        k = abs(idx)
        if k == 0:
            grad = 1.0 - 2.0 * best_h
        else:
            grad = np.ones_like(best_h)
            grad[: best_h.size - k] -= best_h[k:]
            grad[k:] -= best_h[: best_h.size - k]

        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # final symmetry enforcement (often beneficial) and sanity check
    best_h = symmetrize(best_h, rng)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
