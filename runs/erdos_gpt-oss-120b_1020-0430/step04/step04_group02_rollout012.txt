def hill_climb(h_start, rng, max_iters=1500, init_step=0.2, decay=0.9, decay_interval=200):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    n = h.size

    for iter in range(max_iters):
        # compute worst shift
        corr = compute_corr(h)  # raw correlation (not multiplied by 2/n)
        idx = int(np.argmax(corr))
        k_star = idx - (n - 1)
        # propose move
        move_type = rng.random()
        if move_type < 0.4:
            cand = perturb_transfer(h, rng, scale=step)
        elif move_type < 0.7:
            cand = targeted_move(h, rng, k_star, scale=step)
        else:
            cand = swap_move(h, rng)

        c = compute_c5(cand)
        if c < best_c - 1e-12:
            best_c = c
            h = cand
        # decays step
        if (iter+1) % decay_interval == 0:
            step *= decay
            step = max(step, 1e-4)
    return h, best_c

def perturb_transfer(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(n)
    j = rng.integers(n)
    while j == i:
        j = rng.integers(n)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def swap_move(h, rng):
    n = h.size
    i, j = rng.choice(n, size=2, replace=False)
    if h[i] == h[j]:
        return h.copy()
    new_h = h.copy()
    new_h[i], new_h[j] = new_h[j], new_h[i]

def targeted_move(h, rng, k, scale=0.2):
    n = h.size
    if k <= 0 or k >= n:
        return perturb_transfer(h, rng, scale)
    # low mask where both <0.5
    low_mask = (h[: n - k] < 0.5) & (h[k:] < 0.5)
    high_mask = (h[: n - k] > 0.5) & (h[k:] > 0.5)
    if not low_mask.any() or not high_mask.any():
        return perturb_transfer(h, rng, scale)
    i_low = rng.choice(np.flatnonzero(low_mask))
    i_high = rng.choice(np.flatnonzero(high_mask))
    # compute max epsilon such that both low can increase up to 0.5 and high can decrease down to 0.5
    max_eps = min(
        0.5 - h[i_low],
        0.5 - h[i_low + k],
        h[i_high] - 0.5,
        h[i_high + k] - 0.5,
    )
    if max_eps <= 0.0:
        return perturb_transfer(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i_low] += eps
    new_h[i_low + k] += eps
    new_h[i_high] -= eps
    new_h[i_high + k] -= eps
    return new_h

import time

# ---------- global for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None

# ---------- utilities ----------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y


def random_feasible_vector(n, rng):
    x = rng.random(n)
    return project_box_sum(x, n / 2.0)


def compute_corr(h):
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    H = np.fft.fft(h, N)
    G = np.fft.fft(1.0 - h, N)
    corr = np.fft.ifft(H * np.conj(G)).real[: 2 * n - 1]
    return corr


def compute_c5(h):
    n = h.shape[0]
    corr = compute_corr(h)
    return np.max(corr) * (2.0 / n)


def compute_c5_and_idx(h):
    n = h.shape[0]
    corr = compute_corr(h)
    idx = int(np.argmax(corr))
    return corr[idx] * (2.0 / n), idx


# ---------- move operators ----------
def perturb_transfer(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(n)
    j = rng.integers(n)
    while j == i:
        j = rng.integers(n)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def swap_move(h, rng):
    n = h.size
    i, j = rng.choice(n, size=2, replace=False)
    if h[i] == h[j]:
        return h.copy()
    new_h = h.copy()
    new_h[i], new_h[j] = new_h[j], new_h[i]
    return new_h


def targeted_move(h, rng, k, scale=0.2):
    n = h.size
    if k <= 0 or k >= n:
        return perturb_transfer(h, rng, scale)
    low_mask = (h[: n - k] < 0.5) & (h[k:] < 0.5)
    high_mask = (h[: n - k] > 0.5) & (h[k:] > 0.5)
    if not low_mask.any() or not high_mask.any():
        return perturb_transfer(h, rng, scale)
    i_low = rng.choice(np.flatnonzero(low_mask))
    i_high = rng.choice(np.flatnonzero(high_mask))
    max_eps = min(
        0.5 - h[i_low],
        0.5 - h[i_low + k],
        h[i_high] - 0.5,
        h[i_high + k] - 0.5,
    )
    if max_eps <= 0.0:
        return perturb_transfer(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i_low] += eps
    new_h[i_low + k] += eps
    new_h[i_high] -= eps
    new_h[i_high + k] -= eps
    return new_h

def hill_climb(h_start, rng, max_iters=2000, init_step=0.2, decay=0.9, decay_interval=200):
