def compute_gradient_logmax(h, beta):
    n = h.size
    total = h.sum()
    autocorr = np.correlate(h, h, mode='full')
    O = total - autocorr  # shape 2n-1

    Omax = O.max()

def next_pow2(x):
    return 1 << (x-1).bit_length()

def compute_gradient_logmax(h, beta):
    n = h.size
    total = h.sum()
    # autocorr via FFT (linear correlation)
    # compute autocorr = np.correlate(h, h, mode='full')
    # Use FFT for speed:
    m = next_pow2(2*n-1)
    fh = np.fft.rfft(h, n=m)
    autocorr = np.fft.irfft(fh * np.conj(fh), n=m)[:2*n-1]
    O = total - autocorr
    Omax = O.max()
    # compute w
    exp_term = np.exp(beta * (O - Omax))
    w = exp_term / exp_term.sum()
    # convolution w*h using FFT
    # w length 2n-1, h length n => conv length 3n-2
    m2 = next_pow2((2*n-1) + n - 1)
    fw = np.fft.rfft(w, n=m2)
    fh2 = np.fft.rfft(h, n=m2)  # same h, could reuse but fine
    conv_wh = np.fft.irfft(fw * fh2, n=m2)[:3*n-2]
    # extract middle n entries starting at n-1
    conv_mid = conv_wh[n-1 : n-1 + n]
    grad = 1.0 - 2.0 * conv_mid
    return grad

import time, numpy as np

# ---------- utilities ----------
def next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()


def project_box_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Projection onto {0≤v≤1, sum(v)=target} using bisection on λ."""
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """C5 for a discrete vector h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def worst_shift_info(h: np.ndarray):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    s = idx - (n - 1)
    return s, idx, corr[idx]


def gradient_shift(h: np.ndarray, s: int) -> np.ndarray:
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[: n - s] += (1.0 - h[s:])
        grad[s:] -= h[: n - s]
    else:
        s_abs = -s
        grad[s_abs:] += (1.0 - h[: n - s_abs])
        grad[: n - s_abs] -= h[s_abs:]
    return grad


def greedy_optimize(h: np.ndarray, max_steps: int = 2000) -> np.ndarray:
    """Deterministic improvement for the worst shift."""
    for _ in range(max_steps):
        s, _, cur = worst_shift_info(h)
        grad = gradient_shift(h, s)
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-12:
            break
        # try full transfer
        nh = h.copy()
        nh[i] -= max_eps
        nh[j] += max_eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        # try half
        eps = max_eps * 0.5
        nh[i] -= eps
        nh[j] += eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        break
    return h


def perturb_random(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    if h[i] < 1e-12 or h[j] > 1.0 - 1e-12:
        return h.copy()
    eps = min(h[i], 1.0 - h[j]) * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def compute_gradient_logmax(h: np.ndarray, beta: float) -> np.ndarray:
    """Gradient of (1/β) log‑sum‑exp of O(s)."""
    n = h.size
    tot = h.sum()
    # autocorrelation via FFT (linear, not circular)
    m = next_pow2(2 * n - 1)
    fh = np.fft.rfft(h, n=m)
    autoc = np.fft.irfft(fh * np.conj(fh), n=m)[: 2 * n - 1]
    O = tot - autoc                     # O(s) = total - autocorr(s)
    Omax = O.max()
    w = np.exp(beta * (O - Omax))
    w /= w.sum()
    # convolution w * h (linear)
    m2 = next_pow2((2 * n - 1) + n - 1)
    fw = np.fft.rfft(w, n=m2)
    fh2 = np.fft.rfft(h, n=m2)
    conv = np.fft.irfft(fw * fh2, n=m2)[: 3 * n - 2]
    conv_mid = conv[n - 1 : n - 1 + n]   # length n
    grad = 1.0 - 2.0 * conv_mid
    return grad


def solve_one_start(h0: np.ndarray, target: float, time_limit: float):
    """Gradient‑descent + greedy refinement starting from h0."""
    h = h0.copy()
    best_h = h
    best_c = compute_c5(h)
    beta_sched = [1.0, 5.0, 15.0, 50.0, 150.0, 500.0, 1500.0, 5000.0]
    step0 = 0.5
    for beta in beta_sched:
        step = step0
        for _ in range(300):
            if time.time() > time_limit:
                break
            grad = compute_gradient_logmax(h, beta)
            cand = project_box_sum(h - step * grad, target)
            c_cand = compute_c5(cand)
            if c_cand < best_c - 1e-12:
                best_c, best_h, h = c_cand, cand, cand
                step = min(step * 1.05, 1.0)
            else:
                step *= 0.7
                if step < 1e-8:
                    break
        # short greedy polishing after each β
        if time.time() < time_limit:
            h2 = greedy_optimize(h, max_steps=600)
            c2 = compute_c5(h2)
            if c2 < best_c:
                best_c, best_h, h = c2, h2, h2
    return best_h, best_c


def run(seed: int = 42, budget_s: int = 1100, **kwargs):

    start_time = time.time()
    # ----- initialise -----

    best_h = h.copy()

    # ----- main optimisation loop -----
    deadline = start_time + budget_s * 0.90
    while time.time() < deadline:

    # ----- clean‑up -----
