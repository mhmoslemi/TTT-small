def apply_swap(A, i, j, corr):
    # assume A[i]==1, A[j]==0
    n = A.shape[0]
    # delta values
    delta_a_i = -1
    delta_b_i = +1
    delta_a_j = +1
    delta_b_j = -1
    # handle contributions for each shift index k
    # We'll compute updates via loops
    # let's precompute offset
    offset = n - 1
    for k in range(-(n-1), n):
        idx = k + offset
        # contributions from i:
        # A[i] term: only if -i <= k <= n-1-i
        if -i <= k <= n-1-i:
            b = 1 - A[i + k]  # B[i+k] before flip
            corr[idx] += delta_a_i * b
        # B[i] term: only if i - (n-1) <= k <= i
        if i - (n-1) <= k <= i:
            a = A[i - k]    # A[i - k] before flip
            corr[idx] += delta_b_i * a

        # contributions from j (similar)
        if -j <= k <= n-1-j:
            b = 1 - A[j + k]
            corr[idx] += delta_a_j * b
        if j - (n-1) <= k <= j:
            a = A[j - k]
            corr[idx] += delta_b_j * a
    # Now actually swap bits
    A[i] = 0
    A[j] = 1

def compute_corr_fft(A):
    n = A.shape[0]
    L = 2*n
    # use real FFT for speed
    # Using np.fft.rfft on real input gives length L//2+1 complex numbers; compute product then irfft.
    # Let's implement with np.fft.fft for simplicity.
    FA = np.fft.fft(A, L)
    FB = np.fft.fft(1 - A, L)
    conv = np.fft.ifft(FA * FB).real[:2*n-1]
    return conv

    # n must be power of two

    # Using bit_count if available (Python 3.8+). Use vectorized: we can use np.unpackbits, but that works for uint8 only.
    # Simpler: compute popcount using built-in int bit_count within list comp. For n up to 1024, list comprehension fine.
    seq = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=np.int8)

def greedy_search(A, time_limit, rng):
    n = A.shape[0]
    offset = n - 1
    # initial correlation using FFT
    corr = compute_corr_fft(A)
    corr[offset] = 0.0
    best_A = A.copy()
    best_corr_max = corr.max()
    best_c = best_corr_max * (2.0 / n)

    start = time.time()
    while (time.time() - start) < time_limit:
        cur_max = corr.max()
        if cur_max * (2.0 / n) < best_c - 1e-12:
            best_c = cur_max * (2.0 / n)
            best_A = A.copy()
            best_corr_max = cur_max
        # find shift index
        idx = int(np.argmax(corr))
        k = idx - offset
        # compute violating indices for this shift
        if k >= 0:
            i_range = np.arange(0, n - k)
            mask = (A[i_range] == 1) & (A[i_range + k] == 0)
            viol = i_range[mask]
        else:
            kk = -k
            i_range = np.arange(kk, n)
            mask = (A[i_range] == 1) & (A[i_range - kk] == 0)
            viol = i_range[mask]
        if viol.size == 0:
            # fallback random i with A[i]==1
            ones_idx = np.flatnonzero(A == 1)
            if ones_idx.size == 0:
                break
            i = int(rng.choice(ones_idx))
        else:
            i = int(rng.choice(viol))

        zero_idx = np.flatnonzero(A == 0)
        if zero_idx.size == 0:
            break
        # candidate js: sample up to 5 random zeros
        max_candidates = 5
        if zero_idx.size > max_candidates:
            js = rng.choice(zero_idx, size=max_candidates, replace=False)
        else:
            js = zero_idx

        B = 1 - A
        A_rev = A[::-1]

        # evaluate swaps
        best_new_max = cur_max
        best_corr_candidate = None
        best_j = None
        for j in js:
            if j == i:
                continue
            corr_candidate = corr.copy()
            apply_swap_corr(corr_candidate, B, A_rev, i, int(j))
            new_max = corr_candidate.max()
            if new_max < best_new_max - 1e-12:
                best_new_max = new_max
                best_corr_candidate = corr_candidate
                best_j = int(j)
        if best_j is not None:
            # accept best swap
            A[i] = 0
            A[best_j] = 1
            corr = best_corr_candidate
            # continue
            continue
        # else maybe perform random swap with small probability
        if rng.random() < 0.02:
            # random i,j
            i_rand = int(rng.choice(np.flatnonzero(A == 1)))
            j_rand = int(rng.choice(np.flatnonzero(A == 0)))
            corr_candidate = corr.copy()
            apply_swap_corr(corr_candidate, B, A_rev, i_rand, j_rand)
            new_max = corr_candidate.max()
            if new_max < cur_max - 1e-12:
                A[i_rand] = 0
                A[j_rand] = 1
                corr = corr_candidate
        # else continue loop
    return best_A, best_c

import time
import numpy as np

def thue_morse(n: int) -> np.ndarray:
    """Binary Thue‑Morse sequence of length n (n must be a power of two)."""
    return np.array([bin(i).count("1") % 2 for i in range(n)], dtype=np.int8)

def random_balanced_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random binary vector with exactly n/2 ones."""
    a = np.zeros(n, dtype=np.int8)
    a[: n // 2] = 1
    rng.shuffle(a)
    return a

def compute_corr_fft(a: np.ndarray) -> np.ndarray:
    """Full correlation of a with (1‑a) using FFT. Length = 2·n‑1."""
    n = a.shape[0]
    L = 2 * n
    fa = np.fft.fft(a, L)
    fb = np.fft.fft(1 - a, L)
    conv = np.fft.ifft(fa * fb).real[: 2 * n - 1]
    return conv

def apply_swap_corr(corr: np.ndarray, b: np.ndarray, a_rev: np.ndarray, i: int, j: int):
    """
    Incremental update of correlation when swapping a[i]==1 and a[j]==0.
    b = 1‑a, a_rev = a[::-1] are taken from the *pre‑swap* state.
    """
    n = b.shape[0]
    off = n - 1
    corr[off - i : off - i + n] += -1.0 * b
    corr[i : i + n] += 1.0 * a_rev
    corr[off - j : off - j + n] += 1.0 * b
    corr[j : j + n] += -1.0 * a_rev
    corr[off] = 0.0  # enforce zero at shift 0

def greedy_search(a: np.ndarray, time_limit: float, rng: np.random.Generator):
    """
    Hill‑climbing on binary vectors.
    Returns the best vector seen and its C5 value.
    """
    n = a.shape[0]
    off = n - 1
    corr = compute_corr_fft(a)
    corr[off] = 0.0
    best_a = a.copy()
    best_c = corr.max() * (2.0 / n)

    start = time.time()
    while time.time() - start < time_limit:
        cur_max = corr.max()
        if cur_max * (2.0 / n) < best_c - 1e-12:
            best_c = cur_max * (2.0 / n)
            best_a = a.copy()
        idx = int(np.argmax(corr))
        k = idx - off

        # indices where a[i]=1 and a[i+k]=0 (violating pair for the worst shift)
        if k >= 0:
            i_range = np.arange(0, n - k)
            mask = (a[i_range] == 1) & (a[i_range + k] == 0)
            viol = i_range[mask]
        else:
            kk = -k
            i_range = np.arange(kk, n)
            mask = (a[i_range] == 1) & (a[i_range - kk] == 0)
            viol = i_range[mask]

        if viol.size == 0:
            ones = np.flatnonzero(a == 1)
            if ones.size == 0:
                break
            i = int(rng.choice(ones))
        else:
            i = int(rng.choice(viol))

        zeros = np.flatnonzero(a == 0)
        if zeros.size == 0:
            break
        # sample up to 5 candidate zeros
        cand = zeros if zeros.size <= 5 else rng.choice(zeros, size=5, replace=False)

        b = 1 - a
        a_rev = a[::-1]

        best_new = cur_max
        best_corr = None
        best_j = None
        for j in cand:
            if j == i:
                continue
            corr_try = corr.copy()
            apply_swap_corr(corr_try, b, a_rev, i, int(j))
            m = corr_try.max()
            if m < best_new - 1e-12:
                best_new = m
                best_corr = corr_try
                best_j = int(j)

        if best_j is not None:
            a[i] = 0
            a[best_j] = 1
            corr = best_corr
            continue

        # occasional random swap hoping for escape
        if rng.random() < 0.02:
            i2 = int(rng.choice(np.flatnonzero(a == 1)))
            j2 = int(rng.choice(np.flatnonzero(a == 0)))
            corr_try = corr.copy()
            apply_swap_corr(corr_try, b, a_rev, i2, j2)
            if corr_try.max() < cur_max - 1e-12:
                a[i2] = 0
                a[j2] = 1
                corr = corr_try

    return best_a, best_c

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 1024                      # power of two gives clean FFT lengths
    target = n // 2
    seeds = []

    # deterministic Thue‑Morse start
    seeds.append(thue_morse(n).copy())

    # a few random starts
    for _ in range(3):
        seeds.append(random_balanced_binary(n, rng))

    # up‑sample any provided initial construction
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)
        m = init.size
        xs_old = np.linspace(0.0, 2.0, m, endpoint=False)
        xs_new = np.linspace(0.0, 2.0, n, endpoint=False)
        interp = np.interp(xs_new, xs_old, init, period=2.0)
        a0 = (interp > 0.5).astype(np.int8)
        cur = a0.sum()
        if cur > target:
            ones = np.flatnonzero(a0 == 1)
            a0[rng.choice(ones, size=cur - target, replace=False)] = 0
        elif cur < target:
            zeros = np.flatnonzero(a0 == 0)
            a0[rng.choice(zeros, size=target - cur, replace=False)] = 1
        seeds.append(a0)

    best_vec = None
    best_c = float("inf")
    total = len(seeds)

    for s_idx, vec in enumerate(seeds):
        elapsed = time.time() - start
        remaining = budget_s - elapsed
        if remaining < 5:
            break
        # divide remaining time equally among the yet‑to‑process seeds
        per = remaining / (total - s_idx)
        a = vec.copy()
        a = (a > 0).astype(np.int8)
        cur = a.sum()
        if cur > target:
            ones = np.flatnonzero(a == 1)

            zeros = np.flatnonzero(a == 0)

        if c_opt < best_c:

            best_vec = a_opt.copy()

    # recompute exact C5

    final_corr[n - 1] = 0.0
    final_c = final_corr.max() * (2.0 / n)
