def slsqp_optimize(h0, beta, rng, max_iter=500):
    n = h0.size
    target = n / 2.0
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        # log-sum-exp stable
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target},)
    res = opt.minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                        options={'maxiter': max_iter, 'ftol':1e-12, 'disp':False})
    x = np.clip(res.x, 0.0, 1.0)
    # correct sum exactly

def greedy_swap_improve(h, rng, deadline, step=0.01, inner_iters=30, max_outer=100):
    n = h.size
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h)
    outer = 0
    while time.time() < deadline and outer < max_outer:
        full_corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = int(np.argmax(full_corr))
        k = max_idx - (n - 1)
        cur_c = full_corr[max_idx] * dx
        if cur_c < best_c - 1e-12:
            best_c = cur_c
            best_h = h.copy()
        improved = False
        # try inner swaps
        for _ in range(inner_iters):
            # compute partners and product for shift k
            if k >= 0:
                length = n - k
                if length <= 0:
                    break
                p = h[:length] * h[k:]
                i_range = np.arange(length)
                i_max_local = int(np.argmax(p))
                i_max = i_range[i_max_local]
                # partner values for all positions
                partner = np.empty(n)
                mask = np.arange(n) + k < n
                partner[mask] = h[np.arange(n)[mask] + k]
                partner[~mask] = 0.0
            else:
                k_abs = -k
                length = n - k_abs
                if length <= 0:
                    break
                p = h[k_abs:] * h[:length]
                i_range = np.arange(k_abs, n)
                i_max_local = int(np.argmax(p))
                i_max = i_range[i_max_local]
                partner = np.empty(n)
                mask = np.arange(n) - k_abs >= 0
                partner[mask] = h[np.arange(n)[mask] - k_abs]
                partner[~mask] = 0.0
            # feasible target indices
            feas = h < 1.0 - 1e-12
            if not np.any(feas):
                break
            # Choose l with minimal partner
            cand = np.where(feas)[0]
            l_min = cand[np.argmin(partner[cand])]
            if partner[l_min] >= partner[i_max] - 1e-12:
                break
            eps = min(step, h[i_max], 1.0 - h[l_min])
            if eps <= 1e-12:
                break
            h[i_max] -= eps
            h[l_min] += eps
            improved = True
        # recompute after inner
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        new_c = new_corr[np.argmax(new_corr)] * dx
        if new_c < best_c - 1e-12:
            best_c = new_c
            best_h = h.copy()
        # if no improvement, maybe break
        if not improved or abs(new_c - cur_c) < 1e-12:
            break
        outer += 1
    return best_h, best_c

def compute_c5(h):
    n = h.size
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))

def adjust_sum_exact(h, rng):
    target = h.size / 2.0
    diff = target - h.sum()
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, h.size)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    h[:] = np.clip(h, 0.0, 1.0)
    return h

def slsqp_optimize(h0, beta, rng, max_iter=600):
    n = h0.size
    target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    res = minimize(
        obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
