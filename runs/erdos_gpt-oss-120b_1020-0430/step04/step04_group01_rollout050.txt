        mid = (lo + hi) / 2
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()

            lo = mid
        else:
            hi = mid

    # Due to rounding, adjust any small diff
    diff = S - x.sum()
    if abs(diff) > 1e-12:
        # adjust by distributing diff across non-bound elements
        # find indices with 0 < x < 1
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if np.any(mask):
            # distribute equally
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def gradient_smooth(h, beta):
    n = h.shape[0]
    # conv = correlate_full(h)
    conv = correlate_full(h) # length 2n-1
    # w weights
    # to avoid overflow, use shift: bc = beta * conv; subtract max.
    bc = beta * conv
    max_bc = bc.max()
    w = np.exp(bc - max_bc)
    w_sum = w.sum()
    w = w / w_sum
    # prepare for convolution
    # w length 2n-1
    w_rev = w[::-1]
    # pad length for convolution: len= (2n-1) + n -1 = 3n-2
    L = 1 << ((3*n - 2).bit_length())
    # FFT of w and w_rev
    fw = np.fft.rfft(w, L)
    fw_rev = np.fft.rfft(w_rev, L)
    # also compute FFT for (1 - h) and h
    f_one_minus_h = np.fft.rfft(1 - h, L)
    f_h = np.fft.rfft(h, L)
    # term1 = conv(w, (1 - h)) slice
    term1_full = np.fft.irfft(fw * f_one_minus_h, L)  # length L
    term2_full = np.fft.irfft(fw_rev * f_h, L)
    # slice central part: indices n-1 : n-1 + n
    start = n - 1
    term1 = term1_full[start:start + n]
    term2 = term2_full[start:start + n]
    grad = term1 - term2
    return grad

import time, numpy as np

# ---------- basic utilities ----------
def proj_capped(v, S):
    """Projection of v onto {0≤x≤1, sum x = S} (capped simplex)."""
    v = np.clip(v, 0.0, 1.0)
    if abs(v.sum() - S) < 1e-12:
        return v
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        if mask.any():
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)
    return x


def random_feasible(n, rng):
    return proj_capped(rng.random(n), n / 2.0)


# ---------- correlation ----------
def correlate_full(h):
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    return correlate_full(h).max() * (2.0 / h.size)


# ---------- smooth gradient ----------
def gradient_smooth(h, beta):
    n = h.size
    conv = correlate_full(h)                     # length 2n-1
    bc = beta * conv
    m = bc.max()
    w = np.exp(bc - m)
    w /= w.sum()                                 # weights, sum=1
    w_rev = w[::-1]

    L = 1 << ((3 * n - 2).bit_length())         # for conv of (2n-1) with n
    fw = np.fft.rfft(w, L)
    fw_rev = np.fft.rfft(w_rev, L)
    f_one = np.fft.rfft(1 - h, L)
    f_h = np.fft.rfft(h, L)

    term1 = np.fft.irfft(fw * f_one, L)[n - 1:n - 1 + n]
    term2 = np.fft.irfft(fw_rev * f_h, L)[n - 1:n - 1 + n]
    return term1 - term2


def smooth_descent(h0, beta, iters, step0):
    h = h0.copy()
    S = h.size / 2.0
    for t in range(iters):
        grad = gradient_smooth(h, beta)
        step = step0 / np.sqrt(t + 1.0)
        h = proj_capped(h - step * grad, S)
    return h


# ---------- binary simulated annealing ----------
def binary_sa(start_bin, rng, deadline):
    h = start_bin.astype(float).copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)

    T0 = best_c * 0.02 + 1e-6
    Tf = 1e-6
    max_iters = 20000
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if h[i] == h[j]:
            continue
        if h[i] < h[j]:
            i, j = j, i
        h[i] = 0.0
        h[j] = 1.0
        c = compute_c5(h)
        delta = c - best_c
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            if c < best_c:
                best_c, best_h = c, h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c

        max_iters = int(min(150000, max(1000,
                         time_left * 0.5 / est_iter_time)))
        step0 = 0.5 * (2.0 / h.size)      # scale roughly with dx
        h = smooth_descent(h, beta, max_iters, step0)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        # occasional jitter to shake out of shallow basins
        if rng.random() < 0.15:
            h = proj_capped(h + rng.normal(scale=0.02, size=n), n / 2.0)

    # ---------- optional up‑sampling ----------
    if time.time() < deadline and best_h.size * 2 <= 1440:
        h_f = np.repeat(best_h, 2)
        h_f = proj_capped(h_f, h_f.size / 2.0)
        # short high‑beta polish
        h_f = smooth_descent(h_f, 5000.0, 5000,
                             0.5 * (2.0 / h_f.size))
        c_f = compute_c5(h_f)
        if c_f < best_c:
            best_c, best_h = c_f, h_f.copy()

    # ---------- binary simulated annealing ----------
    if time.time() < deadline:
        idx = np.argsort(-best_h)[: best_h.size // 2]
        bin_start = np.zeros(best_h.size)
        bin_start[idx] = 1.0
        bin_h, bin_c = binary_sa(bin_start, rng, deadline)
        if bin_c < best_c:
            best_c, best_h = bin_c, bin_h.copy()

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = proj_capped(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
