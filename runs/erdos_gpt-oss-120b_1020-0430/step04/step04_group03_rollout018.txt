import time
import numpy as np
from scipy.optimize import minimize

# ---------- helpers ----------
def enforce_sum(h, target, rng):
    """Clip to [0,1] and adjust to make sum exactly target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 5000
    it = 0
    # simple water‑filling style adjustments
    while abs(diff) > 1e-12 and it < max_iter:
        # indices that can still move
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            add = min(diff / idx.size, 1.0 - h[idx])
            h[idx] += add
            diff = target - h.sum()
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            sub = min(-diff / idx.size, h[idx])
            h[idx] -= sub
            diff = target - h.sum()
        it += 1
    # fallback if still off
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    """Uniform random vector with values in [0,1] and sum = n/2."""
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)

def perturb_random(h, rng, scale=0.2):
    """Move a small random amount from one coordinate to another."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def compute_c5(h):
    """Discrete C5 for a vector h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def worst_shift_info(h):
    """Return shift with maximal overlap (index, value)."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    s = idx - (h.size - 1)
    return s, idx, corr[idx]

def gradient_shift(h, s):
    """Gradient of overlap w.r.t. h for a fixed shift s."""
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[: n - s] += (1.0 - h[s:])
        grad[s:] -= h[: n - s]
    else:
        s_abs = -s
        grad[s_abs:] += (1.0 - h[: n - s_abs])
        grad[: n - s_abs] -= h[s_abs:]
    return grad

def greedy_optimize(h, max_steps=2000):
    """Deterministic pairwise transfer improvement."""
    for _ in range(max_steps):
        s, _, cur_corr = worst_shift_info(h)
        grad = gradient_shift(h, s)
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-12:
            break
        # try full epsilon
        nh = h.copy()
        nh[i] -= max_eps
        nh[j] += max_eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        # try half epsilon
        eps = max_eps * 0.5
        nh[i] -= eps
        nh[j] += eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        break
    return h

# ---------- soft‑max objective ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """Soft‑max of the overlap (to be maximized)."""
    corr = np.correlate(x, 1.0 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=600):
    """SLSQP refinement using the soft‑max objective."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    rng = np.random.default_rng()
    x = enforce_sum(x, CURRENT_TARGET, rng)
    return x

# ---------- soft‑max on autocorrelation ----------
def softmax_O_and_grad(h, beta):
    """Return soft‑max of -autocorr (objective) and its gradient."""
    a = np.correlate(h, h, mode='full')
    a_min = a.min()
    exp_term = np.exp(-beta * (a - a_min))
    Z = exp_term.sum()
    O = -a_min + (1.0 / beta) * np.log(Z)
    w = exp_term / Z
    grad = -2.0 * np.convolve(h, w, mode='full')[len(h) - 1: len(h) - 1 + len(h)]
    return O, grad

def softmax_O(h, beta):
    a = np.correlate(h, h, mode='full')
    a_min = a.min()
    exp_term = np.exp(-beta * (a - a_min))
    Z = exp_term.sum()
    return -a_min + (1.0 / beta) * np.log(Z)

# ---------- gradient descent ----------
def gradient_descent(start, target_sum, rng, time_budget):
    """Simple gradient descent on the soft‑max objective."""
    h = start.copy()
    n = h.size
    beta_seq = [5, 20, 100, 500, 2000, 5000]
    eta = 0.1  # step‑size base
    t0 = time.time()
    for beta in beta_seq:
        for _ in range(40):
            if time.time() - t0 > time_budget:
                return h
            O, grad = softmax_O_and_grad(h, beta)
            gn = np.linalg.norm(grad)
            if gn < 1e-12:
                break
            step = eta / (gn + 1e-12)
            h_new = h - step * grad
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = enforce_sum(h_new, target_sum, rng)
            O_new = softmax_O(h_new, beta)
            if O_new < O - 1e-12:
                h = h_new
            else:
                eta *= 0.5
                if eta < 1e-7:
                    break
        eta = 0.1
    return h
