        # take top half as ones
        idx = np.argpartition(interp, -base_n//2)[-base_n//2:]

def binary_sa(n, rng, deadline, init_bin=None):
    # n even
    target_ones = n // 2
    # initial
    if init_bin is not None:
        h = init_bin.copy()
    else:
        h = np.zeros(n, dtype=float)
        h[rng.choice(n, target_ones, replace=False)] = 1.0
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c

    # initial temperature
    T = cur_c * 0.5 + 1e-6
    decay = 0.9995
    iter = 0
    last_improve_iter = 0
    # Precompute indices of ones and zeros for speed; update after swap
    ones = np.flatnonzero(h == 1)
    zeros = np.flatnonzero(h == 0)

    while time.time() < deadline:
        # if no ones or zeros (should not happen)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        # update indices lists (temporarily)
        # compute new c
        new_c = compute_c5(h)
        delta = new_c - cur_c
        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur_c = new_c
            # update indices lists properly
            # replace i with new zeros and j with new ones
            # but we can recompute ones,zeros each iteration for simplicity (cost O(n)), but okay.
            ones = np.flatnonzero(h == 1)
            zeros = np.flatnonzero(h == 0)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
                last_improve_iter = iter
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
            # indices unchanged
        # cooling
        T = max(T * decay, 1e-12)
        iter += 1
        if iter - last_improve_iter > 200000:  # no improve for many steps, restart
            # restart random binary but keep best_h
            h = np.zeros(n, dtype=float)
            h[rng.choice(n, target_ones, replace=False)] = 1.0
            cur_c = compute_c5(h)
            ones = np.flatnonzero(h == 1)
            zeros = np.flatnonzero(h == 0)
            T = cur_c * 0.5 + 1e-6
            last_improve_iter = iter
    return best_h, best_c

def adjust_sum_exact(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # Iteratively adjust at random indices
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()

import time
import numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()


def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            up = n - k
            g[:up] += 1.0 - h[k:]          # + (1-h[i+k])
            g[k:] -= h[:up]                # - h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x


def binary_sa(n, rng, deadline, init_bin=None):
    target_ones = n // 2
    if init_bin is not None:
        h = init_bin.copy()
    else:
        h = np.zeros(n, dtype=float)
        h[rng.choice(n, target_ones, replace=False)] = 1.0
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    T = cur_c * 0.5 + 1e-12
    decay = 0.9995
    it = 0
    last_imp = 0
    while time.time() < deadline:
        i = rng.choice(np.flatnonzero(h == 1))
        j = rng.choice(np.flatnonzero(h == 0))
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h)
        delta = new_c - cur_c
        accept = delta < 0 or rng.random() < np.exp(-delta / T)
        if accept:
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
                last_imp = it
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T * decay, 1e-12)
        it += 1
        if it - last_imp > 200000:
            h[:] = 0.0
            h[rng.choice(n, target_ones, replace=False)] = 1.0
            cur_c = compute_c5(h)
            T = cur_c * 0.5 + 1e-12
            last_imp = it
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_deadline = start + budget_s * 0.99

    # ---- Stage 1: SA on n=180 ----
    n1 = 180
    deadline1 = start + (total_deadline - start) * 0.20
    init1 = None
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
        xt = np.linspace(0.0, 2.0, n1, endpoint=False)
        interp = np.interp(xt, xs, parent)
        idx = np.argpartition(interp, -n1 // 2)[-n1 // 2:]
        init1 = np.zeros(n1, dtype=float)
        init1[idx] = 1.0
    h1, c1 = binary_sa(n1, rng, deadline1, init1)

    best_h = h1.copy()
    best_c = c1
    best_n = n1

    # ---- Stage 2: Upsample to 360, binary SA ----
    n2 = 360
    xs = np.linspace(0.0, 2.0, n1, endpoint=False)
    xt = np.linspace(0.0, 2.0, n2, endpoint=False)
    h_interp = np.interp(xt, xs, h1)
    idx = np.argpartition(h_interp, -n2 // 2)[-n2 // 2:]
    h2_init = np.zeros(n2, dtype=float)
    h2_init[idx] = 1.0
    deadline2 = deadline1 + (total_deadline - deadline1) * 0.30
    h2, c2 = binary_sa(n2, rng, deadline2, h2_init)
    if c2 < best_c:
        best_h, best_c, best_n = h2.copy(), c2, n2

    # ---- Stage 3: Continuous refinement (SLSQP + gradient) ----
    h_cont = h2.copy()
    deadline3 = deadline2 + (total_deadline - deadline2) * 0.40
    betas = (5, 20, 80, 200, 500, 1000, 2000, 5000)
    for beta in betas:
        if time.time() > deadline3:
            break
        max_iter = 400 if beta < 200 else 600
        if beta >= 2000:
            max_iter = 800
        try:
            h_cont = slsqp_optimize(h_cont, beta, rng, max_iter=max_iter)
        except Exception:
            continue
    if time.time() < deadline3:
        h_cont = gradient_refine(h_cont, rng, step=0.02, max_iter=12)
    c_cont = compute_c5(h_cont)
    if c_cont < best_c:
        best_h, best_c, best_n = h_cont.copy(), c_cont, n2

    # ---- Stage 4: Optional upsample to 720 if time permits ----
    if time.time() < total_deadline and best_n * 2 <= 720:
        n3 = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n3, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        np.clip(h_up, 0.0, 1.0, out=h_up)
        adjust_sum_exact(h_up, rng)
        # refinement cascade
        for beta in betas:
            if time.time() > total_deadline:
                break
            max_iter = 400 if beta < 200 else 600
            if beta >= 2000:
                max_iter = 800
            try:
                h_up = slsqp_optimize(h_up, beta, rng, max_iter=max_iter)
            except Exception:
                continue
        if time.time() < total_deadline:
            h_up = gradient_refine(h_up, rng, step=0.02, max_iter=12)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up.copy(), c_up, n3

    # final safety
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
