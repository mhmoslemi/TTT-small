def worst_shift_gradient(h):
    n = h.size
    corr = np.correlate(h, 1-h, mode='full')
    idx = np.argmax(corr) # 0..2n-2
    k = idx - (n-1) # shift
    g = np.zeros_like(h)
    if k == 0:
        g = 1 - 2*h
    else:
        # For shift k>0:
        # corr = sum_{i=0}^{n-1-k} h_i * (1 - h_{i+k})
        # gradient w.r.t. h_i for i in [0..n-1] includes:
        # if i <= n-1-k: derivative of term i: 1 - h_{i+k}
        # if i >= k: derivative of term i-k: -h_{i-k}
        # Combine
        upper = n - k
        g[:upper] += 1 - h[k:]
        g[k:] -= h[:upper]
    return g
```

def continuous_sa_optimize(h0, rng, time_deadline):
    n = h0.size
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    start_time = time.time()
    # schedule parameters
    # Cooling schedule: exponential in fraction of elapsed time
    while time.time() < time_deadline:
        # fraction of time left
        elapsed = time.time() - start_time
        frac = elapsed / (time_deadline - start_time)
        # temperature decreasing from T0 to Tf
        T0 = cur_c * 0.3
        Tf = cur_c * 0.001
        T = T0 * (Tf/T0) ** frac  # exponential
        # propose move
        i, j = rng.choice(n, size=2, replace=False)
        # compute max feasible delta
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            continue
        # scale delta
        delta = rng.random() * max_delta
        # apply
        cur_h[i] -= delta
        cur_h[j] += delta
        new_c = compute_c5(cur_h)
        dE = new_c - cur_c
        accept = False
        if dE <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / T):
                accept = True
        if accept:
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] += delta
            cur_h[j] -= delta
    return best_h, best_c

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def worst_shift_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst (max) correlation shift w.r.t. h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    t = idx - (n - 1)            # shift (can be negative)
    g = np.zeros_like(h)
    if t >= 0:
        # first term: (1 - h[i+t]) for i = 0..n-1-t
        g[: n - t] += 1.0 - h[t:]
        # second term: -h[i-t] for i = t..n-1
        g[t:] -= h[: n - t]
    else:
        l = -t
        # first term: (1 - h[i-l]) for i = l..n-1
        g[l:] += 1.0 - h[: n - l]
        # second term: -h[i+l] for i = 0..n-1-l
        g[: n - l] -= h[l:]
    return g


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    steps: int = 5, step: float = 0.1) -> np.ndarray:
    """Few projected‑gradient steps against the current worst shift."""
    n = h.shape[0]
    target = n / 2.0
    for _ in range(steps):
        g = worst_shift_gradient(h)
        h = h - step * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h[rng.integers(0, n)] += diff
            h = np.clip(h, 0.0, 1.0)
    return h


def continuous_sa_optimize(h0: np.ndarray, rng: np.random.Generator,
                           deadline: float) -> (np.ndarray, float):
    """Simulated‑annealing on the continuous domain (mass transfers)."""
    n = h0.shape[0]
    target = n / 2.0
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c

    start = time.time()
    while time.time() < deadline:
        # cooling schedule (exponential)
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        T0 = max(cur_c * 0.3, 1e-8)
        Tf = max(cur_c * 1e-4, 1e-9)
        T = T0 * (Tf / T0) ** frac

        # pick a feasible source (h>0) and sink (h<1)
        src_idxs = np.where(cur_h > 1e-12)[0]
        sink_idxs = np.where(cur_h < 1.0 - 1e-12)[0]
        if src_idxs.size == 0 or sink_idxs.size == 0:
            break
        i = rng.choice(src_idxs)
        j = rng.choice(sink_idxs)
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta
        # propose move
        cur_h[i] -= delta
        cur_h[j] += delta
        new_c = compute_c5(cur_h)
        dE = new_c - cur_c
        accept = False
        if dE <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / T):
                accept = True
        if accept:
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] += delta
            cur_h[j] -= delta
    # final tiny gradient polishing
    best_h = gradient_refine(best_h, rng, steps=3, step=0.05)
    best_c = compute_c5(best_h)
    return best_h, best_c
