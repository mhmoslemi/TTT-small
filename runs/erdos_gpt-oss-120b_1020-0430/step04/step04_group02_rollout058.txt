class CorrTracker:
    def __init__(self, h):
        self.n = h.size
        self.h = h.astype(np.float64)
        self.u = 1.0 - self.h
        # compute full linear correlation
        self.corr = np.correlate(self.h, self.u, mode='full')
        self.offset = self.n - 1  # index of lag zero

    def current_c5(self):
        return np.max(self.corr) * (2.0 / self.n)

    def worst_shift(self):
        # Return k where corr max occurs
        idx = int(np.argmax(self.corr))
        k = idx - self.offset
        return k, self.corr[idx]

    def apply_move(self, i, j, eps):
        # eps added to h[i], subtracted from h[j]
        n = self.n
        offset = self.offset
        # old references
        # i contributions
        # part A: eps * u[i+k]
        k_start_A = -i
        k_end_A = n - 1 - i
        idx_start_A = offset + k_start_A
        idx_end_A = offset + k_end_A + 1
        # u slice:
        self.corr[idx_start_A:idx_end_A] += eps * self.u[i + np.arange(k_start_A, k_end_A+1)]

        # part B: -eps * h[i - k]
        k_start_B = i - (n - 1)
        k_end_B = i
        idx_start_B = offset + k_start_B
        idx_end_B = offset + k_end_B + 1
        k_vals = np.arange(k_start_B, k_end_B+1)
        self.corr[idx_start_B:idx_end_B] -= eps * self.h[i - k_vals]

        # j contributions (delta_h = -eps)
        # part A: -eps * u[j+k] (since delta_h[j] = -eps)
        k_start_Aj = -j
        k_end_Aj = n - 1 - j
        idx_start_Aj = offset + k_start_Aj
        idx_end_Aj = offset + k_end_Aj + 1
        self.corr[idx_start_Aj:idx_end_Aj] -= eps * self.u[j + np.arange(k_start_Aj, k_end_Aj+1)]

        # part B: +eps * h[j - k] (since -eps * (-h) = +eps)
        k_start_Bj = j - (n - 1)
        k_end_Bj = j
        idx_start_Bj = offset + k_start_Bj
        idx_end_Bj = offset + k_end_Bj + 1
        k_vals_j = np.arange(k_start_Bj, k_end_Bj+1)
        self.corr[idx_start_Bj:idx_end_Bj] += eps * self.h[j - k_vals_j]

        # now update h and u
        self.h[i] += eps
        self.h[j] -= eps
        self.u[i] -= eps
        self.u[j] += eps

import time
import numpy as np
from scipy.optimize import minimize

# ---------- helper functions ----------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y


def random_feasible_vector(n, rng):
    x = rng.random(n)
    return project_box_sum(x, n / 2.0)


def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def targeted_move(h, rng, k, scale=0.2):
    n = h.size
    if k <= 0 or k >= n:
        return perturb_h(h, rng, scale)
    low_mask = (h[: n - k] < 0.5) & (h[k:] < 0.5)
    high_mask = (h[: n - k] > 0.5) & (h[k:] > 0.5)
    if not low_mask.any() or not high_mask.any():
        return perturb_h(h, rng, scale)
    i_low = rng.choice(np.flatnonzero(low_mask))
    i_high = rng.choice(np.flatnonzero(high_mask))
    max_eps = min(
        0.5 - h[i_low],
        0.5 - h[i_low + k],
        h[i_high] - 0.5,
        h[i_high + k] - 0.5,
    )
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i_low] += eps
    new_h[i_low + k] += eps
    new_h[i_high] -= eps
    new_h[i_high + k] -= eps
    return new_h


def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def compute_c5_and_idx(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return corr[idx] * (2.0 / h.size), idx


# ---------- hill climbing with simulated annealing ----------
def hill_climb_sa(
    h_start, rng, steps, init_step, temp0, temp_end, p_target=0.25
):
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    n = h.size
    for t in range(steps):
        # exponential temperature schedule
        T = temp0 * (temp_end / temp0) ** (t / steps)
        # worst shift
        _, idx = compute_c5_and_idx(h)
        k_star = idx - (n - 1)
        # propose move
        if rng.random() < p_target and 0 < k_star < n:
            cand = targeted_move(h, rng, k_star, scale=init_step)
        else:
            cand = perturb_h(h, rng, scale=init_step)
        cand_c = compute_c5(cand)
        delta = cand_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = cand
            cur_c = cand_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    return best_h, best_c


# ---------- hill climbing with targeted moves (deterministic) ----------
def hill_climb_targeted(
    h_start, rng, max_iters=800, init_step=0.5, n_trials=4, p_target=0.25
):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    n = h.shape[0]

    for it in range(max_iters):
        _, idx = compute_c5_and_idx(h)
        k_star = idx - (n - 1)  # shift corresponding to max correlation
        improved = False
        for _ in range(n_trials):
            if rng.random() < p_target:
                cand = targeted_move(h, rng, k_star, scale=step)
            else:
                cand = perturb_h(h, rng, scale=step)
            c = compute_c5(cand)
            if c < best_c - 1e-12:
                best_c, h = c, cand
                improved = True
                break
        if not improved and rng.random() < 0.02:
            cand = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(cand)
            if c < best_c:
                best_c, h = c, cand
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return h, best_c


# ---------- SLSQP surrogate minimisation ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)
