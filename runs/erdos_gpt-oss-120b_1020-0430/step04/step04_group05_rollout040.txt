def compute_corr_fft(h):
    n = h.size
    m = 1 << ((2*n - 1).bit_length())
    # zero pad
    H = np.fft.rfft(h, n=m)
    G = np.fft.rfft(1 - h, n=m)
    conv = np.fft.irfft(H * G, n=m)  # length m
    # take first 2n-1 values
    return conv[:2*n - 1]

def targeted_perturb(h, lag, rng, scale=0.5):
    n = h.size
    if lag >= 0:
        # compute array of positions i where i+lag < n
        valid = np.arange(n - lag)
        contrib = h[valid] * (1.0 - h[valid + lag])
        if contrib.size == 0:
            return perturb_pair_random(h, rng, scale)
        # select i with max contrib
        i = valid[np.argmax(contrib)]
        j = i + lag
    else:
        lag_abs = -lag
        valid = np.arange(lag_abs, n)
        # For negative lag, shift indices: i = valid, i+lag = i - lag_abs
        contrib = h[valid] * (1.0 - h[valid - lag_abs])
        if contrib.size == 0:
            return perturb_pair_random(h, rng, scale)
        i = valid[np.argmax(contrib)]
        j = i - lag_abs
    # compute max epsilon we can transfer from i to j (decrease h[i], increase h[j])
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        # fallback random
        return perturb_pair_random(h, rng, scale)
    eps = rng.random() * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def compute_corr_fft(h):
    n = h.shape[0]
    m = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, n=m)
    G = np.fft.rfft(1 - h, n=m)
    conv = np.fft.irfft(H * G, n=m)
    return conv[:2*n - 1]

def targeted_perturb(h, lag, rng, scale=0.5):
    n = h.size
    if lag >= 0:
        valid = np.arange(n - lag)
        h_valid = h[valid]
        h_shift = h[valid + lag]
    else:
        lag_abs = -lag
        valid = np.arange(lag_abs, n)
        h_valid = h[valid]
        h_shift = h[valid - lag_abs]
    # contributions
    contrib = h_valid * (1.0 - h_shift)
    if contrib.size == 0:
        return random_perturb(h, rng, scale)
    # idx of max contribution
    idx = np.argmax(contrib)
    i = valid[idx]
    j = i + lag if lag >= 0 else i - lag_abs
    # max epsilon moving from i to j
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return random_perturb(h, rng, scale)
    eps = rng.random() * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

import time
import numpy as np
from scipy.optimize import minimize

# ----- basic utilities -----
def enforce_sum(h, target, rng, max_iter=5000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-14:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)

def upsample(h, n_new):
    n_old = h.size
    old_x = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    new_x = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return np.interp(new_x, old_x, h)

# ----- convolution via FFT -----
def _fft_len(n):
    return 1 << ((2 * n - 1).bit_length())

def compute_corr_fft(h):
    n = h.shape[0]
    m = _fft_len(n)
    H = np.fft.rfft(h, n=m)
    G = np.fft.rfft(1 - h, n=m)
    conv = np.fft.irfft(H * G, n=m)
    return conv[:2 * n - 1]

def compute_corr_and_c5(h):
    corr = compute_corr_fft(h)
    c = np.max(corr) * (2.0 / h.size)
    return corr, c

def compute_c5_fast(h):
    corr = compute_corr_fft(h)
    return np.max(corr) * (2.0 / h.size)

# ----- perturbations -----
def random_perturb(h, rng, scale=0.5):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    # try move i -> j
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0:
            return h.copy()
        eps = rng.random() * max_eps * scale
        new_h = h.copy()
        new_h[j] -= eps
        new_h[i] += eps
        return new_h
    eps = rng.random() * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def targeted_perturb(h, lag, rng, scale=0.5):
    n = h.size
    if lag >= 0:
        valid = np.arange(n - lag)
        hv = h[valid]
        hs = h[valid + lag]
    else:
        lag = -lag
        valid = np.arange(lag, n)
        hv = h[valid]
        hs = h[valid - lag]
    contrib = hv * (1.0 - hs)
    if contrib.size == 0:
        return random_perturb(h, rng, scale)
    idx = np.argmax(contrib)
    i = valid[idx]
    j = i + lag if lag >= 0 else i - lag
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return random_perturb(h, rng, scale)
    eps = rng.random() * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

# ----- simulated annealing -----
def anneal(h_start, rng, end_time):
    cur_h = h_start.copy()
    corr, cur_c = compute_corr_and_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    n = cur_h.size
    it = 0
    last_imp = 0
    max_iters = 5_000_000
    T0, Tf = 0.02, 1e-6
    while time.time() < end_time:
        T = T0 * (Tf / T0) ** (it / max_iters)
        if rng.random() < 0.7:
            lag = np.argmax(corr) - (n - 1)
            new_h = targeted_perturb(cur_h, lag, rng, scale=0.5)
        else:
            new_h = random_perturb(cur_h, rng, scale=0.5)
        new_c = compute_c5_fast(new_h)
        dE = new_c - cur_c
        if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            cur_h, cur_c = new_h, new_c
            corr = compute_corr_fft(cur_h)
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
                last_imp = it
        if it - last_imp > 20000:
            cur_h = best_h + rng.normal(scale=0.01, size=n)
            cur_h = enforce_sum(cur_h, n / 2.0, rng)
            corr = compute_corr_fft(cur_h)
            cur_c = np.max(corr) * (2.0 / n)
            last_imp = it
        it += 1
    return best_h, best_c

# ----- local hill‑climb -----
def hill_climb(h_start, rng, end_time):
    cur_h = h_start.copy()
    cur_c = compute_c5_fast(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    step = 0.5

            if new_c < best_c:

    return best_h, best_c

    res = minimize(
