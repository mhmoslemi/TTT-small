def enforce_sum(h, target, rng):
    """Adjust h so that Σh == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    # simple random adjustment loop
    for _ in range(10000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add == 0.0:
            continue
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h = enforce_sum(h, target, rng)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

# Correlation utilities (FFT based)
# --------------------------------------------------------------
def correlate_full(h):
    """Full linear correlation of h with (1-h) using FFT (zero‑padding)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)

def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):

    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


# --------------------------------------------------------------
# Projected sub‑gradient optimizer for the max‑overlap objective
# --------------------------------------------------------------
def subgradient_optimize(h0, rng, deadline):
    """Iterative sub‑gradient descent on the max‑overlap (no smoothing)."""
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    it = 0
    eta0 = 0.5
    while True:
        it += 1
        if time.time() >= deadline:
            break
        conv = correlate_full(h)
        idx_max = int(conv.argmax())
        shift = idx_max - (n - 1)  # integer shift causing the max
        # gradient of the selected shift
        grad = np.zeros(n)
        if shift >= 0:
            k = shift
            grad[: n - k] += 1.0 - h[k:]
            grad[k:] -= h[: n - k]
        else:
            k = -shift
            grad[k:] += 1.0 - h[: n - k]
            grad[: n - k] -= h[k:]
        eta = eta0 / np.sqrt(it)
        h_new = h - eta * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        # enforce exact sum
        diff = target - h_new.sum()
        if abs(diff) > 1e-12:
            h_new = enforce_sum(h_new, target, rng)
        h = h_new
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
    return best_h, best_c


# --------------------------------------------------------------
# Smooth surrogate + SLSQP (for fine polishing)
# --------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    conv = correlate_full(x)
    bc = CURRENT_BETA * conv
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        x = enforce_sum(x, CURRENT_TARGET, rng)
    return x


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    budget = budget_s

    # ----------------------------------------------------------
    # 1. Base resolution and initial candidates
    # ----------------------------------------------------------
    n = 720
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        h0 = random_feasible_vector(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # a few handcrafted seeds
    candidates = [h0]

    # block half‑ones
    block = np.zeros(n)
    block[: n // 2] = 1.0
    candidates.append(enforce_sum(block, n / 2.0, rng))

    # alternating 0/1
    alt = np.tile([0.0, 1.0], n // 2)
    candidates.append(enforce_sum(alt, n / 2.0, rng))

    # triangular (tent) shape
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx
    tri = 1.0 - np.abs(xs - 1.0)
    candidates.append(enforce_sum(np.clip(tri, 0.0, 1.0), n / 2.0, rng))

    # sine / cosine families
    for f in range(1, 4):
        s = 0.5 + 0.5 * np.sin(np.pi * f * xs)
        candidates.append(enforce_sum(np.clip(s, 0.0, 1.0), n / 2.0, rng))
        c = 0.5 + 0.5 * np.cos(np.pi * f * xs)
        candidates.append(enforce_sum(np.clip(c, 0.0, 1.0), n / 2.0, rng))

    # evaluate simple pool
    for cand in candidates:
        c = compute_c5(cand)
        if c < best_c:
            best_c, best_h = c, cand.copy()

    # ----------------------------------------------------------
    # 2. Main stochastic search using sub‑gradient + hill‑climb
    # ----------------------------------------------------------
    phase_end = start_time + 0.85 * budget
    while time.time() < phase_end:
        # pick a start point
        if best_h is not None and rng.random() < 0.5:
            start = perturb_h(best_h, rng, scale=0.3)
        else:
            start = random_feasible_vector(n, rng)

        # allocate a per‑restart budget
        remaining = phase_end - time.time()
        per_restart = min(30.0, remaining * 0.5)
        deadline = time.time() + per_restart

        # sub‑gradient descent
        h_opt, c_opt = subgradient_optimize(start, rng, deadline)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()

        # brief hill‑climb refinement if time permits
        if time.time() < deadline:
            h_hc, c_hc = hill_climb(best_h, rng, max_iters=2000, start_step=0.2)
            if c_hc < best_c:
                best_c, best_h = c_hc, h_hc.copy()

    # ----------------------------------------------------------
    # 3. Final polishing with smooth surrogate + SLSQP
    # ----------------------------------------------------------
    final_deadline = start_time + budget
    if time.time() < final_deadline:
        try:
            for beta in [20.0, 100.0, 500.0, 2000.0]:
                if time.time() >= final_deadline:
                    break
                h_pol = slsqp_optimize(best_h, beta, max_iter=250)
                c_pol = compute_c5(h_pol)
                if c_pol < best_c:
                    best_c, best_h = c_pol, h_pol.copy()
        except Exception:
            pass

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
