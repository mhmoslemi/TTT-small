def greedy_improve(h, rng, max_iters=2000):
    n = h.shape[0]
    for it in range(max_iters):
        # compute min overlap and candidate shifts
        O_full = np.correlate(h, h, mode='full')
        O = O_full[n-1:]
        minO = O.min()
        cand_shifts = np.where(O == minO)[0]
        # If many cand_shifts, pick one random
        # try to find a shift with pos diff
        improved = False
        for attempt in range(5):
            if cand_shifts.size == 0:
                break
            k = rng.choice(cand_shifts)
            if k == 0:
                continue
            region_len = n - k
            diff = h[:region_len] - h[k:]
            pos = np.where(diff > 1e-12)[0]
            if pos.size > 0:
                # perform a move
                i = rng.choice(pos)
                a = h[i]
                b = h[i+k]
                max_eps = min(a - b, 1.0 - b, a)  # a > b
                if max_eps > 1e-12:
                    eps = max_eps * rng.random()
                    h[i] -= eps
                    h[i+k] += eps
                    improved = True
                    break
            # else try another shift
            # remove this shift from cand_shifts for attempts
            cand_shifts = np.setdiff1d(cand_shifts, [k], assume_unique=True)
        if not improved:
            # if cannot improve any min shift, break
            break
    return h
```

import time
import numpy as np
from scipy.optimize import minimize

# ---- global variables for SLSQP ----
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1-h(x+k)) dx for discrete h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator) -> None:
    """Redistribute tiny mass to enforce Σh = n/2 exactly (preserves bounds)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.02, max_iter: int = 12) -> np.ndarray:
    """Projected gradient descent against the currently worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]      # + (1 - h[i+k])
            g[k:] -= h[:upper]            # - h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


def sum_eq_constraint(x: np.ndarray) -> float:
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x: np.ndarray) -> float:
    """Log‑sum‑exp surrogate for max_k corr(k)."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 300) -> np.ndarray:
    """SLSQP on the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # final exact sum correction
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x


def greedy_improve(h: np.ndarray, rng: np.random.Generator,
                   max_iters: int = 2000,
                   time_deadline: float = None) -> np.ndarray:
    """
    Simple monotone improvement: repeatedly pick the shift with minimal overlap
    and move a little mass from larger to smaller entries in the overlapping region.
    """
    n = h.shape[0]
    for _ in range(max_iters):
        if time_deadline is not None and time.time() > time_deadline:
            break
        # overlap O_k for k>=0
        O_full = np.correlate(h, h, mode="full")
        O = O_full[n - 1:]                      # O[0] ... O[n-1]
        minO = O.min()
        ks = np.where(O == minO)[0]
        improved = False
        # try each candidate shift (random order)
        rng.shuffle(ks)
        for k in ks:
            if k == 0:
                continue
            region = n - k
            diff = h[:region] - h[k:]
            pos = np.where(diff > 1e-12)[0]
            if pos.size == 0:
                continue
            i = rng.choice(pos)
            a = h[i]
            b = h[i + k]
            # maximal epsilon that still increases the product a*b
            max_eps = min((a - b) / 2.0, 1.0 - b, a)
            if max_eps <= 1e-12:
                continue
            eps = max_eps * rng.random()
            h[i] = a - eps
            h[i + k] = b + eps
            improved = True
            break
        if not improved:
            break
    return h

                    break
                try:
                    h = slsqp_optimize(h, beta, rng, max_iter=iters)
                except Exception:
                    continue
                # small gradient sweep after each SLSQP
                h = gradient_refine(h, rng, step=0.02, max_iter=6)

            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

    # ----- optional up‑sampling of the best solution -----
    if best_h is not None and time.time() < deadline and best_n * 2 <= 720:
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        up_h = np.interp(xt, xs, best_h)
        adjust_sum_exact(up_h, rng)

        up_h = greedy_improve(up_h, rng, max_iters=3000, time_deadline=deadline)
        up_h = gradient_refine(up_h, rng, step=0.02, max_iter=12)

        for beta, iters in ((20.0, 300), (200.0, 400), (2000.0, 500)):
            if time.time() > deadline:
                break
            up_h = slsqp_optimize(up_h, beta, rng, max_iter=iters)
        up_h = gradient_refine(up_h, rng, step=0.02, max_iter=8)

        c_up = compute_c5(up_h)
        if c_up < best_c:
            best_h, best_c, best_n = up_h, c_up, n_up

    # ----- fallback (should never be needed) -----
    if best_h is None:
        n = 180
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = n
    else:
        best_h = np.clip(best_h, 0.0, 1.0)
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    return best_h.tolist(), float(best_c), int(best_n)
