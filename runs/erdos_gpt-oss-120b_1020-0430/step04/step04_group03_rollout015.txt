def softmax_corr(h, beta):
    corr = np.correlate(h, 1 - h, mode='full')
    eta = beta * corr
    max_eta = eta.max()
    exp_eta = np.exp(eta - max_eta)
    p = exp_eta / exp_eta.sum()
    return corr, p

def perturb_random(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    # compute max eps such that 0 <= h[i]-eps and h[j]+eps <=1
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def worst_shift_info(h):
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    s = idx - (h.size - 1)
    return s, idx, corr[idx]

def gradient_shift(h, s):
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[:n - s] += (1 - h[s:])
        grad[s:] -= h[:n - s]
    else:
        s_abs = -s
        grad[s_abs:] += (1 - h[:n - s_abs])
        grad[:n - s_abs] -= h[s_abs:]
    return grad

def greedy_optimize(h, max_steps=2000):
    for _ in range(max_steps):
        s, _, cur_corr = worst_shift_info(h)
        grad = gradient_shift(h, s)
        # find pos and neg indices as in original code
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-12:
            break
        # try full epsilon
        nh = h.copy()
        nh[i] -= max_eps
        nh[j] += max_eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        # try half epsilon
        eps = max_eps * 0.5
        nh[i] -= eps
        nh[j] += eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        break
    return h

import numpy as np
import time

# ---------- projection ----------
def project_onto_bounds_and_sum(v, target):
    """Euclidean projection onto {0<=x<=1, sum(x)=target}."""
    low = v.min() - 1.0
    high = v.max()
    for _ in range(50):
        mid = (low + high) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    return np.clip(v - (low + high) * 0.5, 0.0, 1.0)

# ---------- basic utilities ----------
def random_feasible_vector(n, rng):
    v = rng.random(n)
    return project_onto_bounds_and_sum(v, n / 2.0)

def perturb_random(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

# ---------- greedy local optimizer ----------
def worst_shift_info(h):
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr)
    s = idx - (h.size - 1)
    return s, idx, corr[idx]

def gradient_shift(h, s):
    n = h.size
    g = np.zeros(n)
    if s >= 0:
        g[: n - s] += (1 - h[s:])
        g[s:] -= h[: n - s]
    else:
        a = -s
        g[a:] += (1 - h[: n - a])
        g[: n - a] -= h[a:]
    return g

def greedy_optimize(h, max_steps=2000):
    for _ in range(max_steps):
        s, _, _ = worst_shift_info(h)
        g = gradient_shift(h, s)
        pos = np.where((g > 1e-12) & (h > 1e-12))[0]
        neg = np.where((g < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(g[pos])]
        j = neg[np.argmin(g[neg])]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-12:
            break
        # try full
        nh = h.copy()
        nh[i] -= max_eps
        nh[j] += max_eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        # half
        eps = max_eps * 0.5
        nh[i] -= eps
        nh[j] += eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        break
    return h

# ---------- smoothed gradient ----------
def grad_softmax(h, beta):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    eta = beta * corr
    m = eta.max()
    exp_eta = np.exp(eta - m)
    p = exp_eta / exp_eta.sum()
    g = np.zeros_like(h)
    for idx, pi in enumerate(p):
        if pi == 0.0:
            continue
        s = idx - (n - 1)
        if s >= 0:
            if s == 0:
                g += pi * (1 - 2 * h)
            else:
                i = slice(0, n - s)
                i_s = slice(s, n)
                g[i] += pi * (1 - h[i_s])
                g[i_s] -= pi * h[i]
        else:
            a = -s
            i = slice(a, n)
            i_s = slice(0, n - a)
            g[i] += pi * (1 - h[i_s])
            g[i_s] -= pi * h[i]
    return g

def projected_gradient_descent(h0, beta, max_iter=80, step=1.0, target=None):
    n = h0.size
    target = n / 2.0 if target is None else target
    h = project_onto_bounds_and_sum(np.clip(h0, 0.0, 1.0), target)
    for _ in range(max_iter):
        g = grad_softmax(h, beta)
        h_new = h - step * g
        h_new = project_onto_bounds_and_sum(h_new, target)
        c_new = compute_c5(h_new)
        c_cur = compute_c5(h)
        if c_new + 1e-12 < c_cur:
            h = h_new
            step = min(step * 1.1, 2.0)
        else:
            step *= 0.5
            if step < 1e-6:
                break
    return h

# ---------- multiscale ----------
def upscale(h_coarse, n_fine):
    n_coarse = h_coarse.size
    xs_c = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    xs_f = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    h_f = np.interp(xs_f, xs_c, h_coarse)
    return project_onto_bounds_and_sum(h_f, n_fine / 2.0)

def coarse_search(rng, n_coarse, time_limit):
    t0 = time.time()
    best = random_feasible_vector(n_coarse, rng)
    best = greedy_optimize(best, max_steps=2000)
    best_c = compute_c5(best)
    while time.time() - t0 < time_limit:
        h = random_feasible_vector(n_coarse, rng)
        h = greedy_optimize(h, max_steps=2000)
        c = compute_c5(h)
        if c < best_c:
            best, best_c = h, c
    return best

def fine_optimize(start_h, rng, time_limit, target):
    beta_seq = [200, 800, 2000, 5000]
    h = start_h.copy()
    t0 = time.time()
    for beta in beta_seq:
        h = projected_gradient_descent(h, beta, max_iter=60, step=1.0, target=target)
        h = greedy_optimize(h, max_steps=500)
        if time.time() - t0 > time_limit:
            break
    return h

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_all = time.time()

    # fine resolution (use parent if present)

    best_coarse = coarse_search(rng, n_coarse, coarse_budget)

    upscaled = upscale(best_coarse, n_fine)

    candidates = [upscaled]

    best_fine = None

    t_fine_start = time.time()

        remaining = fine_budget - (time.time() - t_fine_start)

        h_opt = fine_optimize(cand, rng, remaining, target_fine)
        c_opt = compute_c5(h_opt)
        if c_opt < best_c:

    final_c = compute_c5(best_fine)

    return best_fine.tolist(), float(final_c), int(n_fine)
