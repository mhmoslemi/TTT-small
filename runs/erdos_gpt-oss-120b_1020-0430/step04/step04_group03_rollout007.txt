import time, numpy as np, cvxpy as cp

# ---------- utilities ----------
def enforce_sum(h, target, rng):
    """Clip h to [0,1] and adjust to make sum exactly target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:  # need to increase
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i]) * rng.random()
            h[i] += add
            diff -= add
        else:  # need to decrease
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i]) * rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)

def perturb_random(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def compute_c5(h):
    """Discrete C5 for a vector h."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def worst_shift_info(h):
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    s = idx - (h.size - 1)
    return s, idx, corr[idx]

def gradient_shift(h, s):
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[: n - s] += (1 - h[s:])
        grad[s:] -= h[: n - s]
    else:
        s_abs = -s
        grad[s_abs:] += (1 - h[: n - s_abs])
        grad[: n - s_abs] -= h[s_abs:]
    return grad

def greedy_optimize(h, max_steps=2000):
    """Deterministic improvement using full‑epsilon transfers."""
    for _ in range(max_steps):
        s, _, cur_corr = worst_shift_info(h)
        grad = gradient_shift(h, s)
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-12:
            break
        # full epsilon
        nh = h.copy()
        nh[i] -= max_eps
        nh[j] += max_eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        # half epsilon
        eps = max_eps * 0.5
        nh[i] -= eps
        nh[j] += eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        break
    return h

def compute_G(h):
    """Linear coefficients of correlation change: G @ delta + corr ≈ new corr."""
    n = h.size
    rows = 2 * n - 1
    G = np.zeros((rows, n), dtype=float)
    for s in range(-n + 1, n):
        r = s + n - 1
        if s >= 0:
            # first term (1 - h[i+s]) for i=0..n-1-s
            i = np.arange(n - s)
            G[r, i] = 1.0 - h[i + s]
            # second term -h[i] for i=s..n-1 (affects column i)
            j = np.arange(s, n)
            G[r, j] -= h[j - s]
        else:
            s_abs = -s
            # first term (1 - h[i+s]) for i=s_abs..n-1
            i1 = np.arange(s_abs, n)
            G[r, i1] = 1.0 - h[i1 - s_abs]
            # second term -h[i] for i=s_abs..n-1, column j = i - s_abs
            j2 = np.arange(0, n - s_abs)
            G[r, j2] -= h[j2 + s_abs]
    return G

    # ----- iterative refinement (LP + occasional random restarts) -----
    while time.time() - start_time < budget_s * 0.95:
        # stop if already under target

        corr_full = np.correlate(best_h, 1 - best_h, mode='full')

        try:
