def compute_c5_fft(h):
    n = h.shape[0]
    q_rev = (1 - h)[::-1]
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=L)
    Q = np.fft.rfft(q_rev, n=L)
    conv = np.fft.irfft(H * Q, n=L)
    O_max = conv[:2 * n - 1].max()
    return O_max * (2.0 / n)

def binary_anneal(n, time_limit, rng):
    # initialize h as balanced random binary
    h = np.zeros(n, dtype=np.int8)
    ones_idx = rng.choice(n, n//2, replace=False)
    h[ones_idx] = 1
    cur_c = compute_c5_fft(h)
    best_h = h.copy()
    best_c = cur_c
    # schedule: start T = cur_c * 0.01 maybe
    T0 = cur_c * 0.02
    # compute number of iterations possible roughly? But we'll just loop while time left.
    start = time.time()
    while time.time() - start < time_limit:
        # propose swap:
        # choose i where h[i]==1, j where h[j]==0
        i = rng.choice(np.flatnonzero(h == 1))
        j = rng.choice(np.flatnonzero(h == 0))
        # swap
        h[i], h[j] = 0, 1
        new_c = compute_c5_fft(h)
        delta = new_c - cur_c
        # acceptance
        if delta < 0 or rng.random() < np.exp(-delta / T0):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = 1, 0
        # temperature schedule: gradually reduce T0 maybe using factor per iteration? Could compute fraction elapsed.
        # We'll simple linear cooling:
        elapsed = time.time() - start
        T = T0 * (1 - elapsed / time_limit)
        if T < 1e-6:
            T = 1e-6
        T0 = T

def find_swap_pair(h, s):
    n = h.size
    if s >= 0:
        # i where h[i] = 1 and h[i+s] = 0
        bad = np.where((h[:n-s] == 1) & (h[s:] == 0))[0]
        # good zeros where h[k] = 0 and h[k+s] = 1
        good = np.where((h[:n-s] == 0) & (h[s:] == 1))[0]
        # note indices of good are same base range; they are potential k.
    else:
        s_abs = -s
        bad = np.where((h[s_abs:] == 1) & (h[:n-s_abs] == 0))[0] + s_abs
        good = np.where((h[s_abs:] == 0) & (h[:n-s_abs] == 1))[0] + s_abs
    if bad.size == 0 or good.size == 0:
        return None
    # pick random from each
    i = bad[rng.integers(bad.size)]
    k = good[rng.integers(good.size)]
    return i, k

import time, math, numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def enforce_sum(h, target, rng):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    it = 0
    while abs(diff) > 1e-12 and it < 10000:
        if diff > 0:
            idx = np.where(h < 1 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1 - h[i]) * rng.random()
            h[i] += add
            diff -= add
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i]) * rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)

def perturb_random(h, rng, scale=0.2):
    i, j = rng.integers(0, h.size, size=2)
    if i == j:
        return h.copy()
    eps = min(h[i], 1.0 - h[j]) * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

# ---------- overlap evaluation ----------
def compute_c5(h):
    """Exact discrete C5 using np.correlate."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def compute_c5_fft(h):
    """Fast C5 via FFT (returns same value)."""
    n = h.shape[0]
    hf = h.astype(np.float64)
    q_rev = (1.0 - hf)[::-1]
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(hf, n=L)
    Q = np.fft.rfft(q_rev, n=L)
    conv = np.fft.irfft(H * Q, n=L)[:2 * n - 1]
    return conv.max() * (2.0 / n)

def worst_shift_and_value(h):
    """Return (O_max, shift) where O_max = max_i h_i(1-h_{i+s})."""
    n = h.shape[0]
    hf = h.astype(np.float64)
    q_rev = (1.0 - hf)[::-1]
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(hf, n=L)
    Q = np.fft.rfft(q_rev, n=L)
    conv = np.fft.irfft(H * Q, n=L)[:2 * n - 1]
    idx = int(conv.argmax())
    Omax = int(round(conv[idx]))
    s = (n - 1) - idx
    return Omax, s

def find_swap_pair(h, s, rng):
    """Return a pair (i_bad, i_good) that reduces the overlap for shift s,
    or None if none exists."""
    n = h.size
    if s >= 0:
        bad = np.where((h[:n - s] == 1) & (h[s:] == 0))[0]
        good = np.where((h[:n - s] == 0) & (h[s:] == 1))[0]
    else:
        sabs = -s
        bad = np.where((h[sabs:] == 1) & (h[:n - sabs] == 0))[0] + sabs
        good = np.where((h[sabs:] == 0) & (h[:n - sabs] == 1))[0] + sabs
    if bad.size == 0 or good.size == 0:
        return None
    i = int(rng.choice(bad))
    j = int(rng.choice(good))
    return i, j

# ---------- binary local search ----------
def binary_local_search_one_start(h, target_O, rng, deadline):
    """Reduce O_max greedily; if stuck, perform short annealing."""
    while True:
        Omax, s = worst_shift_and_value(h)
        if Omax <= target_O:
            return h, Omax
        pair = find_swap_pair(h, s, rng)
        if pair is None:
            break
        i, j = pair
        h[i], h[j] = 0, 1
    # annealing phase (max 4 s or until deadline)
    cur_h = h.copy()
    cur_O, _ = worst_shift_and_value(cur_h)
    T0 = max(1.0, cur_O - target_O)
    anneal_limit = min(deadline, time.time() + 4.0)
    while time.time() < anneal_limit and time.time() < deadline:
        # pick random 1↔0 swap
        ones = np.flatnonzero(cur_h == 1)
        zeros = np.flatnonzero(cur_h == 0)
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        cur_h[i], cur_h[j] = 0, 1
        new_O, _ = worst_shift_and_value(cur_h)
        d = new_O - cur_O
        elapsed = time.time() - (anneal_limit - 4.0)
        T = max(1e-8, T0 * (1 - elapsed / 4.0))
        if d < 0 or rng.random() < math.exp(-d / T):
            cur_O = new_O
        else:
            cur_h[i], cur_h[j] = 1, 0
    return cur_h, cur_O

def binary_global_search(n, target_O, rng, time_deadline):
    """Repeated random starts, keep best binary vector."""
    best_h = None
    best_O = 10**9
    while time.time() < time_deadline:
        h = np.zeros(n, dtype=np.int8)
        h[rng.choice(n, n // 2, replace=False)] = 1
        h, O = binary_local_search_one_start(h, target_O, rng, time_deadline)
        if O < best_O:
            best_O, best_h = O, h.copy()
            if best_O <= target_O:
                break
    return best_h, best_O

# ---------- fractional (gradient/ SLSQP) ----------
def gradient_shift(h, s):
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[:n - s] += (1 - h[s:])
        grad[s:] -= h[:n - s]
    else:
        s = -s
        grad[s:] += (1 - h[:n - s])
        grad[:n - s] -= h[s:]
    return grad

def greedy_optimize(h, max_steps=2000):
    for _ in range(max_steps):
        _, s = worst_shift_and_value(h)
        grad = gradient_shift(h, s)
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        eps = min(h[i], 1.0 - h[j])
        if eps < 1e-12:
            break
        nh = h.copy()
        nh[i] -= eps
        nh[j] += eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
    return h

CURRENT_TARGET = None
CURRENT_BETA = None
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=600):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    rng = np.random.default_rng()
    x = enforce_sum(x, CURRENT_TARGET, rng)
    return x

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s

    # problem size
    n = len(initial_h_values) if 'initial_h_values' in globals() else 720
    target_O = int(math.floor(0.3808 * n / 2.0))

    best_h = None
    best_c = 1e9
    best_O = 10**9

    # evaluate any provided construction
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        c0 = compute_c5(h0)
        O0 = int(round(c0 * n / 2.0))
        best_h, best_c, best_O = h0, c0, O0

    # ----- binary search phase -----
    bin_deadline = start + budget_s * 0.65
    h_bin, O_bin = binary_global_search(n, target_O, rng, bin_deadline)
    if h_bin is not None:
        c_bin = O_bin * (2.0 / n)
        if c_bin < best_c:
            best_h = h_bin.astype(float)
            best_c = c_bin
            best_O = O_bin

    # ----- fractional refinement phase -----
    frac_deadline = start + budget_s * 0.95
    # starting point: either best found so far (as float) or random
    h_cur = best_h.copy() if best_h is not None else random_feasible_vector(n, rng)
    h_cur = enforce_sum(np.clip(h_cur, 0.0, 1.0), n / 2.0, rng)
    best_c = compute_c5(h_cur)
    best_O = int(round(best_c * n / 2.0))
    best_h = h_cur.copy()

    while time.time() < frac_deadline:
        # random perturbation / restart
        if rng.random() < 0.5:
            start_vec = perturb_random(best_h, rng, scale=0.5)
        else:
            start_vec = random_feasible_vector(n, rng)
        # deterministic greedy
        local = greedy_optimize(start_vec, max_steps=3000)
        c_loc = compute_c5(local)
        if c_loc < best_c:
            best_h, best_c = local, c_loc
            best_O = int(round(best_c * n / 2.0))
        # occasional SLSQP polishing
        if rng.random() < 0.3:
            for beta in (200, 800, 2000, 5000):
                if time.time() >= frac_deadline:
                    break
                refined = slsqp_optimize(best_h, beta, max_iter=800)
                c_ref = compute_c5(refined)
                if c_ref < best_c:
                    best_h, best_c = refined, c_ref
                    best_O = int(round(best_c * n / 2.0))

    # final tiny perturbations
    final_deadline = start + budget_s * 0.99
    while time.time() < final_deadline:
        cand = perturb_random(best_h, rng, scale=0.02)
        c_cand = compute_c5(cand)
        if c_cand < best_c:
            best_h, best_c = cand, c_cand
            best_O = int(round(best_c * n / 2.0))

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
