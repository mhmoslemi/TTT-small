import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global variables for the smooth surrogate (softmax) used in SLSQP
# ----------------------------------------------------------------------
CURRENT_TARGET = 0.0
CURRENT_BETA = 0.0


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1-x_{i+k}) using log-sum-exp."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h so that Σh = target while staying in [0,1]."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-14:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-14:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-14:
            for i in range(n):
                if abs(diff) < 1e-14:
                    break
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random feasible h with Σh = n/2 and 0≤h≤1."""
    u = rng.random(n)
    scale = (n / 2.0) / u.sum()
    h = u * scale
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def slsqp_optimize(h0, beta, max_iter=800):
    """Refine h using SLSQP on the smooth surrogate with given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        n = x.size
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def max_corr_and_shift(h):
    """Return max correlation value and the associated shift index (signed)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    max_val = float(corr[idx])
    shift = idx - (h.size - 1)
    return max_val, shift


def gradient_descent_max_shift(h0, rng, target, max_iters=5000,
                               step0=0.15, decay=0.999, time_limit=None):
    """Projected gradient descent targeting the shift with maximal overlap."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step0
    n = h.size
    start = time.time()
    for _ in range(max_iters):
        if time_limit is not None and (time.time() - start) > time_limit:
            break
        max_val, shift = max_corr_and_shift(h)
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c, best_h = cur_c, h.copy()
        k = shift
        if k >= 0:
            forward = np.empty_like(h)
            forward[:n - k] = h[k:]
            forward[n - k:] = 0.0
            backward = np.empty_like(h)
            backward[k:] = h[:n - k]
            backward[:k] = 0.0
        else:
            m = -k
            forward = np.empty_like(h)
            forward[m:] = h[:n - m]
            forward[:m] = 0.0
            backward = np.empty_like(h)
            backward[:n - m] = h[m:]
            backward[n - m:] = 0.0
        grad = 1.0 - forward - backward
        h_new = h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = enforce_sum(h_new, target, rng)
        new_c = compute_c5(h_new)
        # accept only improvements
        if new_c < cur_c:
            h = h_new
            step = max(step * 0.9, 1e-6)
        else:
            step = max(step * decay, 1e-6)
    return best_h, best_c


def pair_transfer_optimize(h0, rng, max_iters=8000, time_limit=None,
                          init_factor=0.5, decay=0.999):
    """Move mass from i to j where the current maximal shift concentrates."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    factor = init_factor
    for _ in range(max_iters):
        if time_limit is not None and (time.time() - start) > time_limit:
            break
        max_val, shift = max_corr_and_shift(h)
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c, best_h = cur_c, h.copy()
        k = shift
        if k >= 0:
            i_range = np.arange(n - k)
            j_range = i_range + k
        else:
            i_range = np.arange(-k, n)
            j_range = i_range + k
        if i_range.size == 0:
            factor = init_factor
            continue
        prod = h[i_range] * (1.0 - h[j_range])
        idx_max = int(np.argmax(prod))
        i = int(i_range[idx_max])
        j = int(j_range[idx_max])
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            factor = init_factor
            continue
        eps = factor * max_eps
        if eps <= 0.0:
            factor = init_factor
            continue
        h[i] -= eps
        h[j] += eps
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
        factor = max(1e-4, factor * decay)
    return best_h, best_c
