def compute_c5(h):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2*n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)
```

def perturb(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1 - h[j])
    if max_eps <= 0:
        return h
    eps = (rng.random() - 0.5) * 2 * max_eps * scale
    if eps > max_eps: eps = max_eps
    if eps < -max_eps: eps = -max_eps
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h
```

def simulated_annealing(h0, rng, time_limit, init_temp=0.01, final_temp=1e-6, init_scale=0.5, final_scale=0.01):
    start_t = time.time()
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    cur_h = h0.copy()
    cur_c = best_c
    it = 0
    while time.time() - start_t < time_limit:
        it += 1
        t_frac = (time.time() - start_t) / time_limit
        T = init_temp * (final_temp / init_temp) ** t_frac
        scale = init_scale * (final_scale / init_scale) ** t_frac
        new_h = perturb(cur_h, rng, scale=scale)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_h = new_h
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        # Optionally restart if stuck
        if it % 20000 == 0 and cur_c > best_c * 1.01:
            cur_h = best_h.copy()
            cur_c = best_c
    return best_h, best_c

def slsqp_optimize(h0, beta, max_iter=800):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - CURRENT_TARGET}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                    options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # adjust sum
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        # distribute diff equally among free variables
        free_idx = np.where((x > 1e-12) & (x < 1-1e-12))[0]
        if free_idx.size:
            per = diff / free_idx.size
            x[free_idx] += per
            x = np.clip(x, 0.0, 1.0)
    # final enforce sum
    x = enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x
```

def perturb_worst_shift(h, s, rng, scale=0.2):
    n = h.size
    # pick a pair (i,i+s) both in range
    i = rng.integers(0, n - s)
    j = i + s
    # donors a,b
    a = rng.integers(0, n)
    b = rng.integers(0, n)
    # ensure donors not same as i or j, and have some mass
    # we aim to increase h[i] and h[j] each by eps (max limited by 1 - current)
    max_eps_i = 1.0 - h[i]
    max_eps_j = 1.0 - h[j]
    max_eps = min(max_eps_i, max_eps_j)
    # donors capacity
    max_eps_a = h[a]
    max_eps_b = h[b]
    max_eps = min(max_eps, max_eps_a, max_eps_b)

    new_h[j] += eps

    new_h[b] -= eps

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global variables for SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σx = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max correlation using log‑sum‑exp."""
    n = x.shape[0]
    rev = (1 - x)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())          # next power of two
    H = np.fft.rfft(x, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    bc = CURRENT_BETA * conv
    bmax = np.max(bc)
    # log‑sum‑exp, numerically stable
    return (bmax + np.log(np.sum(np.exp(bc - bmax)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for a step function sampled on [0,2]."""
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)


def enforce_sum(v, target, rng):
    """Project v onto [0,1] with Σv = target (exact up to 1e-12)."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(10):
        s = v.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        free = (v > 1e-12) & (v < 1 - 1e-12)
        idx = np.where(free)[0]
        if idx.size == 0:
            break
        per = diff / idx.size
        v[idx] += per
        v = np.clip(v, 0.0, 1.0)

                break
            add = min(diff, 1.0 - v[i]) if diff > 0 else max(diff, -v[i])
            v[i] += add
            diff = target - v.sum()
    return v


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with Σ = n/2."""
    v = rng.random(n)
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    v = enforce_sum(v, target, rng)
    return v


def perturb(h, rng, scale=0.2):
    """Transfer ε from i to j while preserving sum."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def hill_climb(h0, rng, max_iters=2000, step=0.5):
    """Simple greedy hill climbing."""
    h = h0.copy()
    best_c = compute_c5(h)
    for _ in range(max_iters):
        nh = perturb(h, rng, scale=step)
        c = compute_c5(nh)
        if c < best_c:
            h, best_c = nh, c
    return h, best_c


def simulated_annealing_timebound(h0, rng, time_budget,
                                  init_temp=0.05, final_temp=1e-6,
                                  init_scale=0.5, final_scale=0.02):
    """Simulated annealing using exact C5, limited by a time budget."""

    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c

    # approximate iterations count for loop control

        # occasional restart if stuck

    return best_h, best_c


def slsqp_optimize(h0, beta, max_iter=400):
    """SLSQP refinement using the smooth surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size

    diff = CURRENT_TARGET - x.sum()

    return x

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
