```python
def project_box_sum(x, target, lower=0.0, upper=1.0):
    # Project onto [lower, upper] and sum constraint sum(x)=target
    # Use bisection on lambda.
    lo = -1e6
    hi = 1e6
    for _ in range(50):
        lam = (lo+hi)/2.0
        xp = np.clip(x - lam, lower, upper)
        s = xp.sum()
        if s < target:
            hi = lam
        else:
            lo = lam
    return np.clip(x - (lo+hi)/2.0, lower, upper)
```

```python
def gradient_logsumexp(h, beta):
    n = h.shape[0]
    # correlation
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    corr = np.fft.irfft(H * R, m)[:L]
    # weights
    max_c = np.max(corr)
    w = np.exp(beta * (corr - max_c))
    w_sum = w.sum()
    w = w / w_sum
    # compute conv w * h and conv w_rev * h
    # w * h
    W = np.fft.rfft(w, m)
    conv_wh = np.fft.irfft(W * H, m)[:L]
    # w_rev * h
    w_rev = w[::-1]
    WR = np.fft.rfft(w_rev, m)
    conv_wrev_h = np.fft.irfft(WR * H, m)[:L]
    # compute gradient for i in 0..n-1
    idx = np.arange(n) + (n - 1)
    grad = 1.0 - conv_wrev_h[idx] - conv_wh[idx]
    return grad
```

def project_box_sum(x, target, lower=0.0, upper=1.0, max_iter=50):
    # Project onto box and sum constraint via bisection on lambda
    lo, hi = -1.0, 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        xp = np.clip(x - lam, lower, upper)
        s = xp.sum()
        if s < target:
            hi = lam
        else:
            lo = lam
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, lower, upper)

def gradient_logsumexp(h, beta):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())

    R = np.fft.rfft(rev, m)
    corr = np.fft.irfft(H * R, m)[:L]
    max_corr = np.max(corr)
    w = np.exp(beta * (corr - max_corr))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # w * h convolution
    W = np.fft.rfft(w, m)
    conv_wh = np.fft.irfft(W * H, m)[:L]
    w_rev = w[::-1]
    WR = np.fft.rfft(w_rev, m)
    conv_wrev_h = np.fft.irfft(WR * H, m)[:L]
    idx = np.arange(n) + (n - 1)
    grad = 1.0 - conv_wrev_h[idx] - conv_wh[idx]
    return grad

    grad = gradient_logsumexp(h, beta)
    new_h = h - step * grad
    new_h = project_box_sum(new_h, target, 0.0, 1.0)

    new_h[j] += eps

def hill_climb_multi(h_start, rng, max_iters=3000, init_step=0.5, n_trials=8):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
                improved = True
                break
        if not improved and rng.random() < 0.02:
            new_h = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return h, best_c

def simulated_annealing(h0, rng, max_iter=8000, init_temp=1e-2, final_temp=1e-5,
                        init_scale=0.5, final_scale=0.02):
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    cur_h = h0.copy()
    cur_c = best_c
    for i in range(max_iter):
        t = i / max_iter
        temp = init_temp * (final_temp / init_temp) ** t if init_temp > 0 else 0.0
        scale = init_scale * (final_scale / init_scale) ** t if init_scale > 0 else 0.0
        new_h = perturb_h(cur_h, rng, scale=scale)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or (temp > 0 and rng.random() < np.exp(-delta / temp)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
    return best_h, best_c

def slsqp_optimize(h0, beta, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x

def project_box_sum(x, target, lower=0.0, upper=1.0, max_iter=50):
    lo, hi = -2.0, 2.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        xp = np.clip(x - lam, lower, upper)
        s = xp.sum()
        if s < target:
            hi = lam
        else:
            lo = lam
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, lower, upper)

def gradient_logsumexp(h, beta):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    corr = np.fft.irfft(H * R, m)[:L]
    max_corr = np.max(corr)
    w = np.exp(beta * (corr - max_corr))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    W = np.fft.rfft(w, m)
    conv_wh = np.fft.irfft(W * H, m)[:L]
    w_rev = w[::-1]
    WR = np.fft.rfft(w_rev, m)
    conv_wrev_h = np.fft.irfft(WR * H, m)[:L]
    idx = np.arange(n) + (n - 1)
    grad = 1.0 - conv_wrev_h[idx] - conv_wh[idx]
    return grad

def gradient_refine(h, beta, step, target):
    grad = gradient_logsumexp(h, beta)
    new_h = h - step * grad
    return project_box_sum(new_h, target, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n0 = 360
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n0 / 2.0, rng)
    else:
        h0 = random_feasible_vector(n0, rng)
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # quick random search
    while time.time() - start < budget_s * 0.15:
        cand = random_feasible_vector(n0, rng)
        c = compute_c5(cand)
        if c < best_c:
            best_c, best_h = c, cand.copy()

    restarts = 0
    while time.time() - start < budget_s * 0.5 and restarts < 5:
        start_vec = perturb_h(best_h, rng, scale=0.5) if rng.random() < 0.5 else random_feasible_vector(n0, rng)

        h_hc, c_hc = hill_climb_multi(start_vec, rng, max_iters=4000, init_step=0.5, n_trials=8)
        if c_hc < best_c:
            best_c, best_h = c_hc, h_hc.copy()

        h_sa, c_sa = simulated_annealing(best_h, rng, max_iter=12000,
                                         init_temp=1e-2, final_temp=1e-5,
                                         init_scale=0.5, final_scale=0.02)
        if c_sa < best_c:
            best_c, best_h = c_sa, h_sa.copy()

        for beta in (30, 100, 300, 1000, 3000, 10000, 30000, 50000):
            if time.time() - start > budget_s * 0.5:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, max_iter=800)
            except Exception:
                continue
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_c, best_h = c_ref, h_ref.copy()

        if time.time() - start < budget_s * 0.5:
            for step in (0.05, 0.02, 0.01, 0.005, 0.002):
                best_h = gradient_refine(best_h, 50000, step, n0 / 2.0)
                best_c = compute_c5(best_h)

        restarts += 1

    elapsed = time.time() - start
    if budget_s - elapsed > budget_s * 0.15:
        n1 = 1080
        old_x = (np.arange(n0) + 0.5) * (2.0 / n0)
        new_x = (np.arange(n1) + 0.5) * (2.0 / n1)
        up_h = np.interp(new_x, old_x, best_h)
        up_h = enforce_sum(up_h, n1 / 2.0, rng)

        up_h, _ = hill_climb_multi(up_h, rng, max_iters=5000, init_step=0.4, n_trials=6)
        up_h, _ = simulated_annealing(up_h, rng, max_iter=15000,
                                      init_temp=5e-3, final_temp=1e-5,
                                      init_scale=0.4, final_scale=0.02)

        for beta in (100, 300, 1000, 3000, 10000, 30000, 50000):
            if time.time() - start > budget_s:
                break
            try:
                up_h = slsqp_optimize(up_h, beta, max_iter=600)
            except Exception:
                continue
            c_up = compute_c5(up_h)
            if c_up < best_c:
                best_c, best_h, n0 = c_up, up_h.copy(), n1

        if time.time() - start < budget_s:
            for step in (0.05, 0.02, 0.01, 0.005):
                best_h = gradient_refine(best_h, 50000, step, n0 / 2.0)
                best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n0 / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n0)
