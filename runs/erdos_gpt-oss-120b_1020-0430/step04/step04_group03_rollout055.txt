def project_box_sum(v, target):
    lo = -np.max(v) - 1.0 # safe low: ensure v+lo <=0
    hi = 1 - np.min(v) + 1.0 # safe high: ensure v+hi >=1
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

        return False

    h[i] -= eps
    h[j] += eps
    # Ensure bounds (just in case due to float)
    h[i] = max(0.0, min(1.0, h[i]))
    h[j] = max(0.0, min(1.0, h[j]))
    return True

def gradient_shift(h, s):
    n = h.size
    grad = np.zeros(n)
    if s >= 0:

import time
import numpy as np
from scipy.optimize import minimize

# --------- basic utilities ---------------------------------
def project_box_sum(v, target):
    """Projection of v onto the box [0,1] with sum(v)=target via bisection."""
    lo = -np.max(v) - 1.0  # guarantee all v+lo <= 0
    hi = 1.0 - np.min(v) + 1.0  # guarantee all v+hi >= 1
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def compute_corr_raw(h):
    """Unscaled correlation vector of h with (1-h)."""
    return np.correlate(h, 1.0 - h, mode="full")


def compute_c5(h):
    """Discrete C5 for vector h."""
    n = h.size
    return np.max(compute_corr_raw(h)) * (2.0 / n)


def gradient_shift(h, s):
    """Gradient of correlation at shift s w.r.t. h."""
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[: n - s] += 1.0 - h[s:]
        grad[s:] -= h[: n - s]
    else:
        s_abs = -s
        grad[s_abs:] += 1.0 - h[: n - s_abs]
        grad[: n - s_abs] -= h[s_abs:]
    return grad


# --------- simple greedy improvement ------------------------
def simple_greedy(h, max_iters=1500):
    """Deterministic pairwise transfer greedily reducing C5."""
    n = h.size
    for _ in range(max_iters):
        corr_raw = compute_corr_raw(h)
        idx = np.argmax(corr_raw)
        shift = idx - (n - 1)
        grad = gradient_shift(h, shift)

        pos_mask = (grad > 1e-12) & (h > 1e-12)
        neg_mask = (grad < -1e-12) & (h < 1.0 - 1e-12)
        if not np.any(pos_mask) or not np.any(neg_mask):
            break

        i_cand = np.where(pos_mask)[0]
        i = i_cand[np.argmax(grad[i_cand])]
        j_cand = np.where(neg_mask)[0]
        j = j_cand[np.argmin(grad[j_cand])]

        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 1e-12:
            break

        # try full epsilon
        h_new = h.copy()
        h_new[i] -= max_eps
        h_new[j] += max_eps
        if compute_c5(h_new) + 1e-12 < compute_c5(h):
            h[:] = h_new
            continue

        # try half epsilon
        eps = max_eps * 0.5
        h_new[i] -= eps
        h_new[j] += eps
        if compute_c5(h_new) + 1e-12 < compute_c5(h):
            h[:] = h_new
            continue
        break
    return h


# --------- level-step (equalising top two overlaps) -------
def level_step(h):
    """Transfer mass to equalise the two largest correlations."""
    n = h.size
    corr_raw = compute_corr_raw(h)
    if corr_raw.size < 2:
        return False

    top2 = np.argpartition(corr_raw, -2)[-2:]
    order = np.argsort(corr_raw[top2])[::-1]
    idx_s, idx_t = top2[order[0]], top2[order[1]]
    g_s, g_t = corr_raw[idx_s], corr_raw[idx_t]
    if g_s - g_t < 1e-12:
        return False

    shift_s = idx_s - (n - 1)
    shift_t = idx_t - (n - 1)
    grad_s = gradient_shift(h, shift_s)
    grad_t = gradient_shift(h, shift_t)

    pos_mask = (grad_s > 1e-12) & (h > 1e-12)
    neg_mask = (grad_s < -1e-12) & (h < 1.0 - 1e-12)
    if not np.any(pos_mask) or not np.any(neg_mask):
        return False

    i_cand = np.where(pos_mask)[0]
    i = i_cand[np.argmax(grad_s[i_cand])]
    j_cand = np.where(neg_mask)[0]
    j = j_cand[np.argmin(grad_s[j_cand])]

    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 1e-12:
        return False

    denom = (grad_s[i] - grad_s[j]) - (grad_t[i] - grad_t[j])
    if denom > 1e-12:
        eps_star = (g_s - g_t) / denom
        eps = max(0.0, min(max_eps, eps_star))
    else:
        eps = max_eps

    if eps <= 1e-12:
        return False

    h[i] -= eps
    h[j] += eps
    h[i] = min(1.0, max(0.0, h[i]))
    h[j] = min(1.0, max(0.0, h[j]))
    return True


# --------- SLSQP polishing ---------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=600):
    """Smooth‑max SLSQP refinement."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    if abs(x.sum() - CURRENT_TARGET) > 1e-9:
        x = project_box_sum(x, CURRENT_TARGET)
    return x


# --------- main entry ---------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----- initialise ------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        h = np.clip(h, 0.0, 1.0)
        n = h.size
        target_sum = n / 2.0
        if abs(h.sum() - target_sum) > 1e-12:
            h = project_box_sum(h, target_sum)
    else:
        n = 720
        target_sum = n / 2.0
        h = rng.random(n)
        h = project_box_sum(h, target_sum)

    # ----- quick greedy clean‑up ------------------------------------
    h = simple_greedy(h, max_iters=1500)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ----- main level‑step loop ------------------------------------
    no_improve = 0
    while time.time() - start_time < budget_s * 0.80:
        if not level_step(h):
            no_improve += 1
            if no_improve >= 200:
                # random restart
                h = rng.random(n)
                h = project_box_sum(h, target_sum)
                no_improve = 0
            continue
        no_improve = 0
        cur_c = compute_c5(h)
        if cur_c < best_c - 1e-12:
            best_c, best_h = cur_c, h.copy()
        # occasional tiny random transfer to escape flat regions
        if rng.random() < 0.01:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                eps = min(h[i], 1.0 - h[j]) * rng.random() * 0.02
                h[i] -= eps
                h[j] += eps

    # ----- SLSQP polishing ------------------------------------------
    while time.time() - start_time < budget_s * 0.97:
        for beta in (200, 800, 2000, 5000):
            if time.time() - start_time > budget_s * 0.97:
                break
            try:
                refined = slsqp_optimize(best_h, beta, max_iter=500)
            except Exception:
                continue
            c_ref = compute_c5(refined)
            if c_ref < best_c - 1e-12:
                best_c, best_h = c_ref, refined.copy()

    # ----- final clean‑up -------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    if abs(best_h.sum() - target_sum) > 1e-9:
        best_h = project_box_sum(best_h, target_sum)

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
