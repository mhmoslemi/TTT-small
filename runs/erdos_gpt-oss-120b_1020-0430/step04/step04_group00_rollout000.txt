def continuous_sa_search(h, rng, deadline, max_iters=10000):
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    it = 0
    T0 = cur_c * 0.05 # starting temperature relative
    # or set T0 = 0.01
    while time.time() < deadline and it < max_iters:
        # choose i where h>0, j where h<1
        i_candidates = np.where(cur_h > 1e-8)[0]
        j_candidates = np.where(cur_h < 1-1e-8)[0]
        if len(i_candidates)==0 or len(j_candidates)==0:
            break
        i = rng.choice(i_candidates)
        j = rng.choice(j_candidates)
        max_delta = min(cur_h[i], 1 - cur_h[j], 0.03) # step size
        if max_delta <= 0:
            it += 1
            continue
        delta = rng.random() * max_delta
        # propose
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        # evaluate
        new_c = compute_c5(new_h)
        # temperature schedule
        T = T0 * np.exp(-it / 2000) # slower cooling
        accept = False
        if new_c < cur_c:
            accept = True
        else:
            if rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-12)):
                accept = True
        if accept:
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
        it += 1
    return best_h, best_c

def gradient_refine(h, rng, step=0.03, max_iter=15):
    n = h.shape[0]
    target = n/2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr) - (n-1)
        # k shift magnitude (abs idx)
        k = abs(idx)
        # compute gradient
        g = np.zeros_like(h)
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            # For i from 0 to n-k-1: contribution from pair (i,i+k):
            # derivative w.r.t h[i]: (1 - h[i+k]), derivative w.r.t h[i+k]: -h[i]
            # So add to g[i] + (1 - h[i+k]), g[i+k] - h[i]
            g[:n-k] += 1.0 - h[k:]
            g[k:] -= h[:n-k]
        # step size decays
        step_sz = step * (0.7 ** it)
        h = h - step_sz * g
        # clip
        h = np.clip(h,0.0,1.0)
        # Adjust sum exactly
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # distribute diff using random entries
            for _ in range(200):
                i = rng.integers(0,n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n/2.0
    CURRENT_BETA = beta
    bounds = [(0.0,1.0)]*n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x,0.0,1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0,n)
        add = min(diff,1.0 - x[i]) if diff >0 else max(diff,-x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    bc = CURRENT_BETA * corr
    # compute softmax of bc: logsumexp
    m = np.max(bc)
    # compute logsumexp in stable way
    lse = m + np.log(np.sum(np.exp(bc - m)))
    return lse / CURRENT_BETA

def binary_sa_hill_climb(h_bin, rng, time_deadline):
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    # init temperature
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    last_improve = 0
    while time.time() < time_deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        T = init_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
                last_improve = it
        else:
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
        if it - last_improve > 5000:
            break
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    lse = m + np.log(np.sum(np.exp(bc - m)))
    return lse / CURRENT_BETA

def adjust_sum_exact(h, rng):
    target = h.size / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, h.size)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def gradient_refine(h, rng, step=0.03, max_iter=15):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        step_sz = step * (0.7 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def binary_sa_hill_climb(h_bin, rng, deadline):
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    init_T = max(0.01 * cur_c, 1e-4)
    it = last_imp = 0
    while time.time() < deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        T = init_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c, best_h, last_imp = c, cur_h.copy(), it
        else:
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
        if it - last_imp > 5000:
            break
    return best_h, best_c

def continuous_sa_search(h, rng, deadline, max_iters=20000):
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    n = cur_h.size
    it = 0
    T0 = cur_c * 0.05 + 1e-6
    while time.time() < deadline and it < max_iters:
        i_cand = np.where(cur_h > 1e-8)[0]
        j_cand = np.where(cur_h < 1 - 1e-8)[0]
        if i_cand.size == 0 or j_cand.size == 0:
            break
        i = rng.choice(i_cand)
        j = rng.choice(j_cand)
        max_delta = min(cur_h[i], 1.0 - cur_h[j], 0.03)
        if max_delta <= 0:
            it += 1
            continue
        delta = rng.random() * max_delta
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_c = compute_c5(new_h)
        T = T0 * np.exp(-it / 5000.0) + 1e-12
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / T):
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
        it += 1
    adjust_sum_exact(cur_h, rng)
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    return best_h, best_c
