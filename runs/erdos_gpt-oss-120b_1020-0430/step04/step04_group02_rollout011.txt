def compute_c5(h):
    n = h.shape[0]
    g = 1.0 - h
    # convolution (full) of h and g
    size = 2*n-1
    # choose power-of-two size for FFT:
    N = 1 << (size-1).bit_length()
    H = np.fft.rfft(h, n=N)
    G = np.fft.rfft(g, n=N)
    conv = np.fft.irfft(H*G, n=N)[:size]
    max_val = conv.max()
    return max_val * (2.0 / n)

def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=50):
    # find lambda such that sum(clip(x+lam, lo, hi)) = target
    lo_lambda, hi_lambda = -2.0, 2.0
    for _ in range(max_iter):
        mid = (lo_lambda + hi_lambda) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            hi_lambda = mid
        else:
            lo_lambda = mid
    y = np.clip(x + (lo_lambda + hi_lambda) / 2.0, lo, hi)
    # Correction if small error and any interior values:
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y
```

    _, max_val, shift = corr_and_max(h)
    c = max_val * (2.0 / h.size)
    best_h = h.copy()

    start = time.time()
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= time_budget:
            break
        # temperature schedule: exponential decay from 0.5 to 1e-4
        frac = elapsed / time_budget
        T = 0.5 * (1e-4 / 0.5) ** frac  # = 0.5 * (0.0002) ** frac

            h = cand
            c = c_c
            shift = shift_c
            if c < best_c:
                best_c = c
                best_h = h.copy()

import time

# ---------- projection ----------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -5.0, 5.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y
