def continuous_sa(h0, rng, deadline):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = len(h)
    T0 = best_c * 0.02 + 1e-6
    Tf = 1e-7
    max_iters = 200000  # maybe adjust based on time
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # propose move
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # epsilon limited by bounds
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps  # random in [-max_eps, max_eps]
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        c_new = compute_c5(h_new)
        delta = c_new - best_c
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_new
            if c_new < best_c:
                best_c, best_h = c_new, h_new.copy()
        # else revert (h unchanged)
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# global random generator (set in run)
GLOBAL_RNG = None

def enforce_sum(h, target):
    """Adjust h so that Σh == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = GLOBAL_RNG.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n):
    """Random vector in [0,1]^n with sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = GLOBAL_RNG.integers(0, n)
        j = GLOBAL_RNG.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (GLOBAL_RNG.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h = enforce_sum(h, target)
    return h

def perturb_h(h, scale=0.2):
    """Transfer a small amount ε from i to j (preserves the sum)."""
    n = h.size
    i = GLOBAL_RNG.integers(0, n)
    j = GLOBAL_RNG.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (GLOBAL_RNG.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def correlate_full(h):
    """Full linear correlation of h with (1‑h) using FFT."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv

def compute_c5(h):
    """Discrete C₅ for step function h (dx = 2/len(h))."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)

def softmax_obj_full(x, repeat, beta):
    """Log‑sum‑exp smoothing of max_k Σ_i x_i (1‑x_{i+k}) for tiled vector."""
    h = np.tile(x, repeat)
    corr = correlate_full(h)
    bc = beta * corr
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta

def sum_eq_constraint_full(x, target):
    return np.sum(x) - target

def slsqp_optimize_base(x0, repeat, target, beta, max_iter=800):
    """SLSQP on the smooth surrogate for a periodic base vector."""
    bounds = [(0.0, 1.0)] * len(x0)
    cons = {"type": "eq", "fun": sum_eq_constraint_full, "args": (target,)}
    res = minimize(
        softmax_obj_full,
        x0,
        args=(repeat, beta),
        method="SLSQP",
        bounds=bounds,
        constraints=[cons],
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        n = len(x)
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = GLOBAL_RNG.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x

def binary_sa(start_bin, deadline):
    """Swap‑based simulated annealing on a binary vector (sum = n/2)."""
    h = start_bin.astype(float).copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    T0 = best_c * 0.02 + 1e-6
    Tf = 1e-6
    max_iters = 200000
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        i = GLOBAL_RNG.integers(0, n)
        j = GLOBAL_RNG.integers(0, n)
        if h[i] == h[j]:
            continue
        if h[i] < h[j]:
            i, j = j, i
        h[i] = 0.0
        h[j] = 1.0
        c = compute_c5(h)
        delta = c - best_c
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        if delta < 0 or GLOBAL_RNG.random() < np.exp(-delta / max(T, 1e-12)):
            if c < best_c:
                best_c, best_h = c, h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c

def continuous_sa(h0, deadline):
    """Continuous simulated annealing preserving total mass."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    T0 = best_c * 0.02 + 1e-6
    Tf = 1e-7
    max_iters = 200000
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        i = GLOBAL_RNG.integers(0, n)
        j = GLOBAL_RNG.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (GLOBAL_RNG.random() - 0.5) * 2.0 * max_eps
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        c_new = compute_c5(h_new)
        delta = c_new - best_c
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        if delta < 0 or GLOBAL_RNG.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_new
            if c_new < best_c:
                best_c, best_h = c_new, h_new.copy()
    return best_h, best_c

def hill_climb(h_start, max_iters=5000, start_step=0.2):
    """Greedy hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    global GLOBAL_RNG
    GLOBAL_RNG = np.random.default_rng(seed)
    start_time = time.time()
    budget = budget_s
    n = 720  # fixed grid size
    best_h = None
    best_c = float('inf')

    # 0) Baseline from provided initial_h_values if any
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0)
        c0 = compute_c5(h0)
        best_h, best_c = h0, c0

    # 1) One‑parameter flat‑top family (w in [0,1.8])
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx
    def shape_h(w):
        if w >= 1.999999:
            h = np.full(n, 0.5)
        else:
            a = 1.0 - w / 2.0
            v = 1.0 / (1.0 + w / 2.0)
            d = np.abs(xs - 1.0)
            h_raw = v * (1.0 - d) / a
            h = np.clip(h_raw, 0.0, 1.0)
        h = enforce_sum(h, n / 2.0)
        return h

    w_vals = np.linspace(0.0, 1.8, 1801)
    for w in w_vals:
        if time.time() - start_time > budget * 0.15:
            break
        h_cand = shape_h(w)
        c = compute_c5(h_cand)
        if c < best_c:
            best_c, best_h = c, h_cand.copy()

    # 2) Low‑dim periodic optimisation (up to ~55% budget)
    candidate_ms = [360, 240, 180, 144, 120, 90, 80, 72,
                   60, 48, 40, 36, 30, 24, 20, 15]
    for m in candidate_ms:
        repeat = n // m
        target_m = m / 2.0
        for restart in range(3):
            if time.time() - start_time > budget * 0.55:
                break
            b0 = random_feasible_vector(m)
            for beta in [10.0, 50.0, 200.0, 800.0, 2000.0]:
                if time.time() - start_time > budget * 0.55:
                    break
                b_opt = slsqp_optimize_base(b0, repeat, target_m, beta, max_iter=800)
                h_full = np.tile(b_opt, repeat)
                c = compute_c5(h_full)
                if c < best_c:
                    best_c, best_h = c, h_full.copy()
                b0 = b_opt  # warm‑start next beta

    # 3) Binary simulated annealing (up to ~85% budget)
    if time.time() - start_time < budget * 0.85:
        idx = np.argsort(-best_h)[: n // 2]
        bin_start = np.zeros(n)
        bin_start[idx] = 1.0
        deadline = start_time + budget * 0.85
        bin_h, bin_c = binary_sa(bin_start, deadline)
        if bin_c < best_c:
            best_c, best_h = bin_c, bin_h.copy()

    # 4) Continuous simulated annealing (up to ~95% budget)
    if time.time() - start_time < budget * 0.95:
        deadline = start_time + budget * 0.95
        cont_h, cont_c = continuous_sa(best_h, deadline)
        if cont_c < best_c:
            best_c, best_h = cont_c, cont_h.copy()

    # 5) Final greedy hill‑climb (till budget exhausted)
    if time.time() - start_time < budget:
        hc_h, hc_c = hill_climb(best_h, max_iters=5000, start_step=0.2)
        if hc_c < best_c:
            best_c, best_h = hc_c, hc_h.copy()

    # final cleanup
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
