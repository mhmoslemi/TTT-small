def compute_c5_fft(h):
    n = h.size
    L = 1 << (2*n-1).bit_length()  # next power of 2 >= 2n
    # compute FFT of h and (1-h) padded
    fh = np.fft.rfft(h, n=L)
    fg = np.fft.rfft(1-h, n=L)
    corr = np.fft.irfft(fh * np.conj(fg), n=L)
    # relevant part length 2n-1
    corr = corr[:2*n-1]
    return np.max(corr) * (2.0 / n)
```

def anneal_optimize(h0, cur_c0, start_time, max_duration, rng):
    h = h0.copy()
    cur_c = cur_c0
    best_h = h.copy()
    best_c = cur_c0
    n = h.size
    # initial temperature
    T0 = cur_c * 0.02  # small; adjust
    while time.time() - start_time < max_duration:
        # progress
        elapsed = time.time() - start_time
        progress = elapsed / max_duration
        T = T0 * (1 - progress)
        # propose move
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1  # ensure j != i
        # max feasible epsilon
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        # step size scaled with T
        # compute scale factor: as T reduces, reduce eps
        scale = max(0.01, T / (cur_c + 1e-12))  # example
        eps = max_eps * rng.random() * scale
        # ensure eps positive
        if eps <= 0:
            continue
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        c_new = compute_c5(h_new)
        delta = c_new - cur_c
        if delta <= 0 or T > 0 and rng.random() < np.exp(-delta / T):
            h = h_new
            cur_c = c_new
            if cur_c < best_c:
                best_h = h.copy()
                best_c = cur_c
    return best_h, best_c

def enforce_sum(h, target, rng):
    """Clip h to [0,1] and adjust to achieve exact sum = target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:  # need to increase total
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i]) * rng.random()
            h[i] += add
            diff -= add
        else:  # need to decrease total
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i]) * rng.random()
            h[i] -= sub
            diff += sub
        it += 1

        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
