def enforce_sum_fast(h, target, rng):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    # Uniform correction
    h += diff / n
    # Clip again
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # If still off, adjust via random picks
    if abs(diff) > 1e-12:
        for _ in range(1000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

def gradient_descent_optimize(h0, target, rng, time_limit=30.0, max_iters=20000, init_lr=0.1, lr_decay=0.9995, init_beta=0.5, beta_factor=1.05, beta_max=5000.0):
   h = h0.copy()
   best_h = h.copy()
   best_c = compute_c5(h)
   start = time.time()
   beta = init_beta
   lr = init_lr
   n = h.size
   while time.time() - start < time_limit and max_iters > 0:
        # compute corr
        corr = np.correlate(h, 1.0 - h, mode='full')
        # compute softmax weights
        maxc = np.max(beta * corr)
        exp_vals = np.exp(beta * corr - maxc)
        w = exp_vals / np.sum(exp_vals)
        # compute gradient
        w_rev = w[::-1]
        g1 = np.convolve(w_rev, 1.0 - h, mode='full')[n-1:n-1+n]
        g2 = np.convolve(w, h, mode='full')[n-1:n-1+n]
        grad = g1 - g2
        # update
        h = h - lr * grad
        # clip and enforce sum
        h = enforce_sum_fast(h, target, rng)
        # evaluate
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        # schedule
        lr *= lr_decay
        beta = min(beta * beta_factor, beta_max)
        max_iters -= 1
   return best_h, best_c

import time
import numpy as np

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def max_overlap_and_shift(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return float(corr[idx]), idx - (h.size - 1)

def enforce_sum_fast(h, target, rng, max_iter=1000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(max_iter):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

def random_binary_vector(n, rng):
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, size=n//2, replace=False)
    h[ones] = 1.0
    return h

def mix_parents(p1, p2, rng):
    mask = rng.random(p1.size) < 0.5
    child = np.where(mask, p1, p2)
    return enforce_sum_fast(child, p1.size / 2.0, rng)

def pair_transfer_optimize(h0, rng, max_iters=5000, time_limit=30.0,
                          init_factor=0.5, decay=0.999):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    factor = init_factor
    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        max_val, shift = max_overlap_and_shift(h)
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            i_range = np.arange(-shift, n)
            j_range = i_range + shift
        if i_range.size == 0:
            factor = init_factor
            continue
        prod = h[i_range] * (1.0 - h[j_range])
        idx_max = int(np.argmax(prod))
        i = int(i_range[idx_max])
        j = int(j_range[idx_max])
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            factor = init_factor
            continue
        eps = factor * max_eps
        if eps <= 0.0:
            factor = init_factor
            continue
        h[i] -= eps
        h[j] += eps
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
        factor = max(1e-4, factor * decay)
    final_c = compute_c5(best_h)
    return best_h, final_c

def gradient_descent_optimize(h0, target, rng, time_limit=30.0,
                              max_iters=20000, init_lr=0.5,
                              lr_decay=0.9995, init_beta=0.5,
                              beta_factor=1.03, beta_max=2000.0):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    start = time.time()
    lr = init_lr
    beta = init_beta
    it = 0
    while it < max_iters and time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode="full")
        maxc_corr = np.max(corr)
        exp_vals = np.exp(beta * (corr - maxc_corr))
        w = exp_vals / np.sum(exp_vals)
        g1 = np.convolve(w[::-1], 1.0 - h, mode='full')[n-1:n-1+n]
        g2 = np.convolve(w, h, mode='full')[n-1:n-1+n]
        grad = g1 - g2
        h = h - lr * grad
        h = enforce_sum_fast(h, target, rng)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        lr = max(lr * lr_decay, 1e-6)
        beta = min(beta * beta_factor, beta_max)
        it += 1
    return best_h, best_c

    start_time = time.time()

    population.sort(key=lambda x: x[1])
    best_h, best_c = population[0]
    # evolutionary loop
    while time.time() - start_time < budget_s * 0.85:

        child = mix_parents(p1, p2, rng)
