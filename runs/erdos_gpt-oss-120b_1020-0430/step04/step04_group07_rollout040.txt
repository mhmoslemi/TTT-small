def simulated_annealing(h, rng, max_iter, start_temp, end_temp):
    n = h.size
    target = n//2
    # Ensure integer sum = target (since binary).
    current = h.copy()
    current_c = compute_c5(current)
    best = current.copy()
    best_c = current_c
    for it in range(max_iter):
        # temperature schedule
        t = start_temp * ((end_temp/start_temp) ** (it / max_iter))
        # propose swap
        # select a 1 index and a 0 index
        ones = np.where(current == 1)[0]
        zeros = np.where(current == 0)[0]
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        current[i], current[j] = 0, 1
        new_c = compute_c5(current)
        delta = new_c - current_c
        # accept if better or with probability exp(-delta/t) (delta>0)
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-12)):
            current_c = new_c
            if new_c < best_c:
                best_c = new_c
                best = current.copy()
        else:
            # revert swap
            current[i], current[j] = 1, 0
    return best, best_c

def fractional_refine(h, rng, max_iters, eps_scale):
    n = h.size
    target = n / 2
    cur_c = compute_c5(h)
    for _ in range(max_iters):
        i = rng.integers(n)
        j = rng.integers(n)
        while i == j:
            j = rng.integers(n)
        # propose adjusting h[i] by delta and h[j] by -delta
        # delta max limited by [0,1] constraints:
        max_inc_i = 1 - h[i]
        max_dec_i = h[i]
        max_inc_j = 1 - h[j]
        max_dec_j = h[j]
        # generate delta randomly from [-eps, eps]
        eps = eps_scale * rng.random()
        # choose sign based on capacity
        # For delta positive: need inc i and dec j; must have eps <= max_inc_i and eps <= max_dec_j
        # For delta negative: inc j and dec i.
        # We'll try both signs random.
        if rng.random() < 0.5:
            delta = min(eps, max_inc_i, max_dec_j)
            if delta == 0:
                continue
            nh = h.copy()
            nh[i] += delta
            nh[j] -= delta
        else:
            delta = min(eps, max_inc_j, max_dec_i)
            if delta == 0:
                continue
            nh = h.copy()
            nh[i] -= delta
            nh[j] += delta
        # no need to enforce sum; it's already preserved
        new_c = compute_c5(nh)
        if new_c < cur_c:
            h = nh
            cur_c = new_c
    return h, cur_c

from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # fix sum exactly (quick redistribution)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def random_cyclic_binary(n, rng):
    step = rng.integers(1, n)
    # ensure step is coprime with n to get permutation; if not, just use step.
    # Not crucial.
    idx = (np.arange(n) * step) % n
    h = np.zeros(n, dtype=float)
    h[idx < n // 2] = 1.0
    return h

def interpolate_h(old_h, new_n):
    x_old = np.linspace(0.0, 2.0, old_h.size, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h = np.interp(x_new, x_old, old_h)
    h = np.clip(h, 0.0, 1.0)
    # project to satisfy sum exactly
    target = new_n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # simple redistribution
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = np.random.randint(0, new_n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

def hill_climb_binary(h, rng, max_iters):
    n = h.size
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    for _ in range(max_iters):
        # pick random 1 and 0 positions
        i = rng.integers(n)
        while cur_h[i] != 1:
            i = rng.integers(n)
        j = rng.integers(n)
        while cur_h[j] != 0:
            j = rng.integers(n)
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_c = compute_c5(cur_h)
        if new_c < cur_c:
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] = 1.0
            cur_h[j] = 0.0
    return best_h, best_c

def fractional_refine(h, rng, max_iters, eps_scale=0.02):
    n = h.size
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    for _ in range(max_iters):
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)
        eps = eps_scale * rng.random()
        # try move from i to j
        if rng.random() < 0.5:
            delta = min(eps, 1 - cur_h[i], cur_h[j])
            if delta <= 0:
                continue
            nh = cur_h.copy()
            nh[i] += delta
            nh[j] -= delta
        else:
            delta = min(eps, cur_h[i], 1 - cur_h[j])
            if delta <= 0:
                continue
            nh = cur_h.copy()
            nh[i] -= delta
            nh[j] += delta
        new_c = compute_c5(nh)
        if new_c < cur_c:
            cur_h = nh
            cur_c = new_c
    return cur_h, cur_c
