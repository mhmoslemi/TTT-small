def targeted_perturb(h, lag, rng, top_frac=0.15, scale=0.4):
    n = h.size
    start = max(0, -lag)
    end = min(n, n - lag)
    if start >= end:
        return perturb(h, rng, scale) (just random)
    contributions = h[start:end] * (1.0 - h[start+lag:end+lag])
    idx_range = np.arange(start, end)
    k = max(1, int(len(idx_range) * top_frac))
    # top indices
    top_idx = idx_range[np.argpartition(-contributions, k-1)[:k]]
    # bottom indices
    bot_idx = idx_range[np.argpartition(contributions, k-1)[:k]]
    # pick i from top and j from bot (maybe j can be any else)
    i = rng.choice(top_idx)
    # choose j from full range but avoid same polarity
    # we could also choose j from bot_idx or random elsewhere
    if rng.random() < 0.6:
        j = rng.choice(bot_idx)
    else:
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)
    # compute max epsilon that can be transferred without violating bounds
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def simulated_annealing(h_start, rng, max_iters, time_end):
    h = h_start.copy()
    cur_c, cur_corr = compute_c5_corr(h)
    best_h, best_c = h.copy(), cur_c
    n = h.size
    T0 = cur_c * 0.02  # scale
    Tf = cur_c * 1e-5
    for it in range(max_iters):
        if time.time() >= time_end:
            break
        T = T0 * (Tf / T0) ** (it / max_iters)
        lag = np.argmax(cur_corr) - (n - 1)
        # Use targeted perturb more often maybe
        if rng.random() < 0.7:
            new_h = targeted_perturb(h, lag, rng, top_frac=0.15, scale=0.4)
        else:
            new_h = perturb(h, rng, scale=0.4)
        new_c, new_corr = compute_c5_corr(new_h)
        dE = new_c - cur_c
        if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            h, cur_c, cur_corr = new_h, new_c, new_corr
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
    return best_h, best_c

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize_one_beta(h0, beta, rng, max_iter=200):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        # fix sum by adjusting small values
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x
```

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
