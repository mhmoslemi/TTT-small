```
def anneal(h0, rng, budget, start_time):
    h = h0.copy()
    best = h.copy()
    best_c = compute_c5(h)
    T0 = 0.1  # initial temperature
    T_min = 1e-5
    # estimate number of steps based on time left
    steps = 0
    while time.time() - start_time < budget:
        # linear schedule based on elapsed fraction
        t = (time.time() - start_time) / budget
        T = T0 * (1 - t) + T_min * t
        # propose random move
        i, j = rng.integers(0, h.size, size=2)
        if i==j: continue
        max_eps = min(h[i], 1 - h[j])
        if max_eps == 0: continue
        eps = max_eps * rng.random()
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        c_new = compute_c5(h_new)
        delta = c_new - best_c
        if delta < 0 or rng.random() < np.exp(-delta / T):
            h = h_new
            if c_new < best_c:
                best = h_new.copy()
                best_c = c_new
        steps += 1
        # perhaps adjust T decay by steps number
    return best
```

import time, numpy as np
from scipy.optimize import minimize

def enforce_sum(h, target, rng):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0: break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i]) * rng.random()
            h[i] += add
            diff -= add
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0: break
            i = rng.choice(idx)
            sub = min(-diff, h[i]) * rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    # produce binary vector with exactly half ones, then add tiny noise
    h = np.zeros(n)
    # assign half ones
    idx = rng.permutation(n)[:n//2]
    h[idx] = 1.0
    # optional small random perturbation to create fractional values
    # we keep binary for stability
    return h

def perturb_random(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def worst_shift_info(h):
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    s = idx - (h.size - 1)
    return s, idx, corr[idx]

def gradient_shift_corr(h, s):
    n = h.size
    grad = np.ones(n)
    if s >= 0:
        grad[:n - s] -= h[s:]
        grad[s:] -= h[:n - s]
    else:
        s_abs = -s
        grad[s_abs:] -= h[:n - s_abs]
        grad[:n - s_abs] -= h[s_abs:]
    return grad

def greedy_optimize_fixed(h0, max_steps=2000, rng=None):
    h = h0.copy()
    for _ in range(max_steps):
        s, _, cur_corr = worst_shift_info(h)
        grad = gradient_shift_corr(h, s)
        # indices that can be decreased (positive gradient) and increased (negative)
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-12:
            break
        # try full transfer
        nh = h.copy()
        nh[i] -= max_eps
        nh[j] += max_eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        # try half transfer
        eps = max_eps * 0.5
        nh[i] -= eps
        nh[j] += eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        break
    return h

# global variables for SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    # x is the vector h
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=600):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum precisely
    rng = np.random.default_rng()
    x = enforce_sum(x, CURRENT_TARGET, rng)
    return x

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # setup initial vector
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        # ensure feasibility
        h0 = enforce_sum(np.clip(h0, 0.0, 1.0), n/2.0, rng)
    else:
        n = 720
        h0 = random_feasible_vector(n, rng)
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # main search loop
    while time.time() - start_time < budget_s * 0.80:
        # pick start point
        if rng.random() < 0.6 and best_h is not None:
            start = perturb_random(best_h, rng, scale=0.5)
        else:
            start = random_feasible_vector(n, rng)
        start = enforce_sum(np.clip(start, 0.0, 1.0), n/2.0, rng)
        # greedy local improvement
        local = greedy_optimize_fixed(start, max_steps=2500, rng=rng)
        c_local = compute_c5(local)
        if c_local + 1e-12 < best_c:
            best_h, best_c = local, c_local
        # occasional SLSQP polishing
        if rng.random() < 0.2:
            for beta in (500, 2000, 5000, 10000):
                if time.time() - start_time > budget_s * 0.95:
                    break
                try:
                    refined = slsqp_optimize(best_h, beta, max_iter=800)
                except Exception:
                    continue
                c_ref = compute_c5(refined)
                if c_ref + 1e-12 < best_c:
                    best_h, best_c = refined, c_ref

    # final fine perturbations
    while time.time() - start_time < budget_s * 0.99:
        cand = perturb_random(best_h, rng, scale=0.02)
        c_cand = compute_c5(cand)
        if c_cand + 1e-12 < best_c:
            best_h, best_c = cand, c_cand

    best_h = enforce_sum(np.clip(best_h, 0.0, 1.0), n/2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
