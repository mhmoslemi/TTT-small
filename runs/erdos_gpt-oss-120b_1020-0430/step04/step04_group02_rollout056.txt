def binary_anneal(h_bin, n, time_limit, rng):
    kk = np.arange(-(n-1), n, dtype=int)
    h = h_bin.copy()
    # padded
    h_padded = np.concatenate((np.zeros(n, dtype=int), h, np.zeros(n, dtype=int)))
    v = np.correlate(h, 1 - h, mode='full')
    cur_c = v.max() * (2.0 / n)
    best_h = h.copy()
    best_c = cur_c
    # precompute list of indices
    # Using arrays for ones and zeros
    ones = np.flatnonzero(h)
    zeros = np.flatnonzero(1 - h)
    start = time.time()
    T0 = 0.002
    Tmin = 1e-6
    while time.time() - start < time_limit:
        # compute worst shift
        idx_max = v.argmax()
        k_star = idx_max - (n - 1)
        # try targeted swap
        # compute shifted pattern
        if k_star >= 0:
            shifted = np.empty_like(h)
            shifted[:n-k_star] = h[k_star:]
            shifted[n-k_star:] = 0
        else:
            shifted = np.empty_like(h)
            shifted[-k_star:] = h[:n+k_star]
            shifted[: -k_star] = 0
        first_bad = np.flatnonzero((h == 1) & (shifted == 0))
        second_bad = np.flatnonzero((h == 0) & (shifted == 1))
        if first_bad.size and second_bad.size and rng.random() < 0.7:
            i = int(rng.choice(first_bad))
            j = int(rng.choice(second_bad))
        else:
            # random swap
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
        # compute term_i, term_j
        term_i = (1 - h_padded[i + n + kk]) - h_padded[i + n - kk]
        term_j = (1 - h_padded[j + n + kk]) - h_padded[j + n - kk]
        cand_v = v - term_i + term_j
        cand_c = cand_v.max() * (2.0 / n)
        # temperature
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = max(Tmin, T0 * (1 - frac))
        if cand_c < cur_c - 1e-12 or rng.random() < np.exp(-(cand_c - cur_c) / T):
            # accept
            v = cand_v
            cur_c = cand_c
            # swap in h and padded
            h[i], h[j] = h[j], h[i]  # will be 0 and 1 swapped
            h_padded[i + n] = h[i]
            h_padded[j + n] = h[j]
            # update ones/zeros lists (recompute perhaps)
            # recompute ones and zeros each iteration for simplicity
            ones = np.flatnonzero(h)
            zeros = np.flatnonzero(1 - h)

import time
import numpy as np
from scipy.optimize import minimize

# ------- projection onto box with exact sum -------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y

# ------- C5 evaluation -------
def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / h.size)

# ------- softmax surrogate for SLSQP -------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)

# ------- Thue–Morse binary sequence -------
def thue_morse(n):
    seq = np.zeros(n, dtype=int)
    for i in range(n):
        seq[i] = bin(i).count("1") & 1
    return seq

# ------- initialise binary h -------
def init_binary_h(n, rng):
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size != n:
            old_x = (np.arange(h0.size) + 0.5) * (2.0 / h0.size)
            new_x = (np.arange(n) + 0.5) * (2.0 / n)
            h0 = np.interp(new_x, old_x, h0)
        h_bin = (h0 >= 0.5).astype(int)
    else:
        h_bin = thue_morse(n)
    target = n // 2
    cur = h_bin.sum()
    if cur > target:
        excess = cur - target
        idx = rng.choice(np.flatnonzero(h_bin), size=excess, replace=False)
        h_bin[idx] = 0
    elif cur < target:
        deficit = target - cur
        idx = rng.choice(np.flatnonzero(h_bin == 0), size=deficit, replace=False)
        h_bin[idx] = 1
    return h_bin

# ------- binary simulated annealing -------
def binary_anneal(h_start, n, time_limit, rng):
    kk = np.arange(-(n - 1), n, dtype=int)          # shape (2n-1,)
    h = h_start.copy()
    h_padded = np.concatenate((np.zeros(n, dtype=int), h, np.zeros(n, dtype=int)))
    v = np.correlate(h, 1 - h, mode="full")
    cur_c = v.max() * (2.0 / n)
    best_h = h.copy()
    best_c = cur_c
    start = time.time()
    T0 = 0.002
    Tmin = 1e-6
    while time.time() - start < time_limit:
        # worst shift
        idx_max = int(v.argmax())
        k_star = idx_max - (n - 1)
        # shifted version for targeted move
        if k_star >= 0:
            shifted = np.empty_like(h)
            shifted[: n - k_star] = h[k_star:]
            shifted[n - k_star:] = 0
        else:
            shifted = np.empty_like(h)
            shifted[-k_star:] = h[: n + k_star]
            shifted[: -k_star] = 0
        bad1 = np.flatnonzero((h == 1) & (shifted == 0))
        bad2 = np.flatnonzero((h == 0) & (shifted == 1))
        if bad1.size and bad2.size and rng.random() < 0.7:
            i = int(rng.choice(bad1))
            j = int(rng.choice(bad2))
        else:
            ones = np.flatnonzero(h)
            zeros = np.flatnonzero(1 - h)
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
        # incremental update
        term_i = (1 - h_padded[i + n + kk]) - h_padded[i + n - kk]
        term_j = (1 - h_padded[j + n + kk]) - h_padded[j + n - kk]
        cand_v = v - term_i + term_j
        cand_c = cand_v.max() * (2.0 / n)
        # temperature
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = max(Tmin, T0 * (1 - frac))
        if cand_c < cur_c - 1e-12 or rng.random() < np.exp(-(cand_c - cur_c) / T):
            v = cand_v
            cur_c = cand_c
            h[i], h[j] = h[j], h[i]          # swap values
            h_padded[i + n] = h[i]
            h_padded[j + n] = h[j]
            if cur_c < best_c - 1e-12:
                best_c = cur_c
                best_h = h.copy()
    return best_h, best_c

# ------- main entry point -------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # choose resolution
    n = 720
    if n % 2:
        n += 1

    # initialise binary candidate
    h_bin = init_binary_h(n, rng)

    # binary annealing (≈60% of time)
    time_for_anneal = budget_s * 0.60
    best_bin, best_c = binary_anneal(h_bin, n, time_for_anneal, rng)

    # continuous refinement (remaining time)
    h_float = best_bin.astype(float)
    beta_list = [10, 50, 200, 500, 1000, 2000, 5000, 10000]
    for beta in beta_list:
        if time.time() - start_time > budget_s * 0.95:
            break
        try:
            h_ref = slsqp_optimize(h_float, beta, max_iter=400)
        except Exception:
            continue
        c_ref = compute_c5(h_ref)
        if c_ref < best_c:
            best_c = c_ref
            h_float = h_ref

    # final clean‑up
    h_final = np.clip(h_float, 0.0, 1.0)
    h_final = project_box_sum(h_final, n / 2.0)
    final_c = compute_c5(h_final)

    return h_final.tolist(), float(final_c), int(n)
