def pgd_optimize(h0, rng, max_iters=200, step0=0.1, top_m=5, beta_weights=False):
    h = h0.copy()
    n = h.size
    target = n / 2.0
    dx = 2.0 / n
    best_c = compute_c5(h)
    step = step0
    for it in range(max_iters):
        # time check optionally
        corr = np.correlate(h, 1.0 - h, mode='full')
        C = corr * dx
        # gradient
        if beta_weights:
            # full softmax
            m = C.max()
            exp_vals = np.exp(beta_weights * (C - m))
            w = exp_vals / exp_vals.sum()
            # compute gradient across all k via loops weighted
            g = np.zeros(n)
            # iterate over all k
            ks = np.arange(-n+1, n)
            for idx_k, k in enumerate(ks):
                wk = w[idx_k]
                if wk == 0:
                    continue
                if k >= 0:
                    if k < n:
                        g[:n-k] += wk * (1.0 - h[k:])
                        g[k:] -= wk * h[:n-k]
                else:
                    p = -k
                    g[p:] += wk * (1.0 - h[:n-p])
                    g[:n-p] -= wk * h[p:]
        else:
            # select top_m shifts
            idxs = np.argpartition(C, -top_m)[-top_m:]
            w = np.full_like(idxs, 1.0 / top_m, dtype=float)
            g = np.zeros(n)
            for idx_k, wk in zip(idxs, w):
                k = idx_k - (n - 1)
                if k >= 0:
                    if k < n:
                        g[:n-k] += wk * (1.0 - h[k:])
                        g[k:] -= wk * h[:n-k]
                else:
                    p = -k
                    g[p:] += wk * (1.0 - h[:n-p])
                    g[:n-p] -= wk * h[p:]
        # take step
        h_new = project_box_sum(h - step * g, target)
        c_new = compute_c5(h_new)
        if c_new < best_c - 1e-12:
            h, best_c = h_new, c_new
            # optional increase step slightly
            step = step * 1.02
        else:
            step *= 0.7
            if step < 1e-6:
                break

import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic helpers ----------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y


def random_feasible_vector(n, rng):
    x = rng.random(n)
    return project_box_sum(x, n / 2.0)


def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


# ---------- PGD optimizer ----------
def pgd_optimize(h0, max_iters, step0, top_m):
    h = h0.copy()
    n = h.size
    target = n / 2.0
    dx = 2.0 / n
    best_c = compute_c5(h)
    step = step0
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        C = corr * dx
        # select top shifts
        idxs = np.argpartition(C, -top_m)[-top_m:]
        weight = 1.0 / top_m
        g = np.zeros(n)
        for idx in idxs:
            k = idx - (n - 1)
            if k >= 0:
                if k < n:
                    g[: n - k] += weight * (1.0 - h[k:])
                    g[k:] -= weight * h[: n - k]
            else:
                p = -k
                g[p:] += weight * (1.0 - h[: n - p])
                g[: n - p] -= weight * h[p:]
        h_new = project_box_sum(h - step * g, target)
        c_new = compute_c5(h_new)
        if c_new < best_c - 1e-12:
            h, best_c = h_new, c_new
            step *= 1.02
        else:
            step *= 0.7
            if step < 1e-6:
                break
    return h, best_c


# ---------- SLSQP surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)


# ---------- main entry ----------

    rng = np.random.default_rng(seed)
    start_time = time.time()
    TARGET_BOUND = 0.3808

    # initial resolution
