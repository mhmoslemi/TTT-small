def fast_corr(h):
    n = h.size
    a = h
    b = 1 - h
    b_rev = b[::-1]
    n_fft = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, n_fft)
    fb = np.fft.rfft(b_rev, n_fft)
    conv = np.fft.irfft(fa * fb, n_fft)[:2*n-1]
    return conv

def worst_shift_and_grad(h):
    n = h.shape[0]
    corr = full_corr(h)
    idx = np.argmax(corr)
    s = idx - (n-1)  # shift can be negative

        grad[:n-s] -= h[s:]

def projected_subgradient_descent(h0, target, max_iters, time_start, time_budget):
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5_fast(h)
    n = h.size
    step0 = 0.2  # initial step (maybe bigger)
    for it in range(1, max_iters+1):
        if time.time() - time_start > time_budget:
            break
        # compute worst shift and gradient
        max_corr, grad, c_val, s = worst_shift_and_grad(h)
        # step size schedule
        alpha = step0 / np.sqrt(it)
        # gradient direction
        h_new = h - alpha * grad
        # project onto feasible
        h_new = project_feasible(h_new, target, rng)
        # compute value
        c_new = compute_c5_fast(h_new)
        if c_new < best_val:
            best_val = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_val

# ---------------------- feasibility helpers ----------------------
def enforce_sum(h, target, rng, max_iter=5000):
    """Clip to [0,1] and adjust to meet exact sum `target`."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum=n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:

            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

def perturb_h(h, rng, scale=0.2):
    """Transfer a small ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def project_feasible(v, target, rng):

    v = enforce_sum(v, target, rng)
    return v

def fast_corr(h):
    """Full correlation of h with (1-h) using FFT (mode='full')."""
    n = h.shape[0]
    b_rev = (1 - h)[::-1]
    n_fft = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(h, n=n_fft)
    fb = np.fft.rfft(b_rev, n=n_fft)
    conv = np.fft.irfft(fa * fb, n=n_fft)[: 2 * n - 1]
    return conv

def compute_c5_fast(h):
    """C5 value for a discretised step function."""
    n = h.shape[0]
    corr = fast_corr(h)
    return np.max(corr) * (2.0 / n)

def worst_shift_and_grad(h):
    """Return (max_corr, gradient_of_max_corr, C5, shift) for worst shift."""
    n = h.shape[0]
    corr = fast_corr(h)
    idx = int(np.argmax(corr))
    s = idx - (n - 1)      # signed shift
    grad = np.zeros(n)
    if s >= 0:
        grad[: n - s] += 1 - h[s:]
        grad[s:] -= h[: n - s]
    else:
        shift = -s
        grad[shift:] += 1 - h[: n - shift]
        grad[: n - shift] -= h[shift:]
    max_corr = corr[idx]
    c_val = max_corr * (2.0 / n)
    return max_corr, grad, c_val, s

def projected_subgradient_descent(
    h0, target, max_iters, time_start, time_end, rng, step0=0.2
):
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5_fast(h)
    for it in range(1, max_iters + 1):
        if time.time() > time_end:
            break
        _, grad, c_val, _ = worst_shift_and_grad(h)
        alpha = step0 / np.sqrt(it)
        h_new = h - alpha * grad
        h_new = project_feasible(h_new, target, rng)
        new_val = compute_c5_fast(h_new)
        if new_val < best_val:
            best_val = new_val
            best_h = h_new.copy()
        h = h_new
    return best_h, best_val

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize_one_beta(h0, beta, rng, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x
