def fractional_sa_hill_climb(h, rng, time_deadline):
    cur = h.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    n = cur.size
    # define max delta as 0.1 maybe but adapt to bound
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    last_improve = 0
    while time.time() < time_deadline:
        i, j = rng.choice(n, size=2, replace=False)
        # Determine max possible transfer from i to j
        # direction: move from i (source) to j (sink)
        # Can either increase j and decrease i, or opposite.
        # Let's pick direction randomly
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_transfer = min(cur[src], 1.0 - cur[dst])
        if max_transfer <= 0:
            continue
        # Choose delta uniformly up to max_transfer
        delta = rng.random() * max_transfer
        cur[src] -= delta
        cur[dst] += delta
        c = compute_c5(cur)
        T = init_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            # accept
            cur_c = c
            # update best
            if c < best_c:
                best_c = c
                best = cur.copy()
                last_improve = it
        else:
            # reject, revert
            cur[src] += delta
            cur[dst] -= delta
        it += 1
        if it - last_improve > 10000:
            break
    return best, best_c

import time
import numpy as np
from scipy.optimize import minimize

# Global placeholders for SLSQP constraints
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    """Maximum overlap for step function samples h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """Smooth max via log‑sum‑exp, scaled by CURRENT_BETA."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=600):
    """SLSQP on the smooth max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction (simple redistribution)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def adjust_sum_exact(h, rng):
    """Redistribute mass to satisfy Σh = n/2 exactly (maintaining bounds)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_refine(h, rng, step=0.02, max_iter=12):
    """Projected gradient descent against the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]   # + (1-h[i+k])
            g[k:] -= h[:upper]        # - h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def binary_sa_hill_climb(h_bin, rng, time_deadline):
    """Swap‑based simulated annealing on binary vectors."""
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    last_improve = 0
    while time.time() < time_deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        T = init_T * (0.9995 ** it) + 1e-12
        accept = c < cur_c or rng.random() < np.exp(-(c - cur_c) / T)
        if accept:
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
                last_improve = it
        else:
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
        if it - last_improve > 8000:
            break
    return best_h, best_c

def fractional_sa_hill_climb(h, rng, time_deadline):
    """Transfer‑based simulated annealing on continuous vectors."""
    n = h.size
    cur = h.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    last_improve = 0
    while time.time() < time_deadline:
        i, j = rng.choice(n, size=2, replace=False)
        src, dst = (i, j) if rng.random() < 0.5 else (j, i)
        max_transfer = min(cur[src], 1.0 - cur[dst])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        cur[src] -= delta
        cur[dst] += delta
        c = compute_c5(cur)
        T = init_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            if c < best_c:
                best_c = c
                best = cur.copy()
                last_improve = it
        else:
            cur[src] += delta
            cur[dst] -= delta
        it += 1
        if it - last_improve > 12000:
            break
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    parent = np.array(initial_h_values, dtype=float) if 'initial_h_values' in globals() else None
    base_ns = [360, 240, 180]

    for n in base_ns:
        if time.time() > deadline:
            break

        start_vecs = []

        if parent is not None:
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            start_vecs.append(interp)
            idx = np.argpartition(interp, -n // 2)[-n // 2:]
            bin_vec = np.zeros(n)
            bin_vec[idx] = 1.0
            start_vecs.append(bin_vec)

        # random binary
        bin_rand = np.zeros(n)
        bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
        start_vecs.append(bin_rand)

        # random fractional (uniform)
        rand_frac = rng.random(n)
        target = n / 2.0
        rand_frac *= target / rand_frac.sum()
        rand_frac = np.clip(rand_frac, 0.0, 1.0)
        adjust_sum_exact(rand_frac, rng)
        start_vecs.append(rand_frac)

        # random fractional (biased)
        rand_frac2 = rng.random(n) ** 2
        rand_frac2 *= target / rand_frac2.sum()
        rand_frac2 = np.clip(rand_frac2, 0.0, 1.0)
        adjust_sum_exact(rand_frac2, rng)
        start_vecs.append(rand_frac2)

        for h0 in start_vecs:
            if time.time() > deadline:
                break
            h = np.clip(h0, 0.0, 1.0)
            adjust_sum_exact(h, rng)

            # binary SA if applicable
            if np.all((h == 0) | (h == 1)):
                sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.35)
                h_bin, c_bin = binary_sa_hill_climb(h.copy(), rng, sub_dead)
                if c_bin < best_c:
                    best_h, best_c, best_n = h_bin.copy(), c_bin, n
                h = h_bin.astype(float)

            # fractional SA
            sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.30)
            h_frac, c_frac = fractional_sa_hill_climb(h.copy(), rng, sub_dead)
            if c_frac < best_c:
                best_h, best_c, best_n = h_frac.copy(), c_frac, n
            h = h_frac

            # gradient refinement
            h = gradient_refine(h, rng, step=0.03, max_iter=12)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

            # SLSQP cascade
            betas = [5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0, 10000.0]
            for beta in betas:
                if time.time() > deadline:
                    break
                max_iter = 600 if beta < 200 else 900
                if beta >= 5000:
                    max_iter = 1200
                try:
                    h = slsqp_optimize(h, beta, rng, max_iter=max_iter)
                except Exception:
                    continue
                c = compute_c5(h)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n

            # final gradient step
            h = gradient_refine(h, rng, step=0.02, max_iter=15)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

    # Upsampling refinement (if time permits)
    if best_h is not None and time.time() < deadline and best_n * 2 <= 720:
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        adjust_sum_exact(h_up, rng)

        # gradient + fractional SA cycle
        cycles = 2
        for _ in range(cycles):
            if time.time() > deadline:
                break
            h_up = gradient_refine(h_up, rng, step=0.03, max_iter=18)

            sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.25)
            h_up, _ = fractional_sa_hill_climb(h_up, rng, sub_dead)

        # optional mild SLSQP pass
        if time.time() < deadline:
            try:
                h_up = slsqp_optimize(h_up, beta=2000.0, rng=rng, max_iter=800)
            except Exception:
                pass

        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, n_up

    # Fallback
    if best_h is None:
        n = 180
        best_n = n
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
