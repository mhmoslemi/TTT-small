def adam_optimize(h_init, beta_seq, iters_per_beta, lr, target_sum, rng, start_time, budget):
    h = h_init.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    t = 0
    best_h = h.copy()
    best_c = compute_c5(h)
    for beta in beta_seq:
        for _ in range(iters_per_beta):
            if time.time() - start_time > budget * 0.99:
                return best_h, best_c
            t += 1
            grad = gradient_softmax(h, beta)
            # Adam update
            m = 0.9 * m + 0.1 * grad
            v = 0.999 * v + 0.001 * (grad * grad)
            m_hat = m / (1 - 0.9**t)
            v_hat = v / (1 - 0.999**t)
            step = lr / (np.sqrt(v_hat) + 1e-8)
            # maybe scale step with beta (beta large -> smaller step)
            # combine
            h = project_box_sum(h - step, target_sum)
            c = compute_c5(h)
            if c < best_c:
                best_c, best_h = c, h.copy()
    return best_h, best_c

def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    # clip and adjust by shift lambda s.t. sum(clip(x + s)) = target
    # binary search s
    lo_shift, hi_shift = -2.0, 2.0
    for _ in range(max_iter):
        mid = (lo_shift + hi_shift) / 2
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            hi_shift = mid
        else:
            lo_shift = mid
    y = np.clip(x + (lo_shift + hi_shift)/2, lo, hi)
    # adjust small diff
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        # distribute diff uniformly over non-boundary entries
        mask = (y > lo) & (y < hi)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y

def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    """Shift then clip x so that sum equals target and lo<=x<=hi."""
    lo_s, hi_s = -2.0, 2.0
    for _ in range(max_iter):
        mid = (lo_s + hi_s) * 0.5
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            hi_s = mid
        else:
            lo_s = mid
    y = np.clip(x + (lo_s + hi_s) * 0.5, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo) & (y < hi)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y


def random_feasible_vector(n, rng):
    return project_box_sum(rng.random(n), n / 2.0)


def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:

    eps = np.clip(eps, -max_move, max_move)

    nh[i] -= eps
    nh[j] += eps
    return nh


def compute_c5(h):

def compute_c5_and_idx(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return corr[idx] * (2.0 / h.size), idx

def targeted_move(h, rng, k, scale=0.2):
    n = h.size
    if k <= 0 or k >= n:
        return perturb_h(h, rng, scale)
    low_mask = (h[: n - k] < 0.5) & (h[k:] < 0.5)
    high_mask = (h[: n - k] > 0.5) & (h[k:] > 0.5)
    if not low_mask.any() or not high_mask.any():
        return perturb_h(h, rng, scale)
    i_low = rng.choice(np.flatnonzero(low_mask))
    i_high = rng.choice(np.flatnonzero(high_mask))
    max_eps = min(
        0.5 - h[i_low],
        0.5 - h[i_low + k],
        h[i_high] - 0.5,
        h[i_high + k] - 0.5,
    )
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = np.clip(eps, -max_eps, max_eps)
    nh = h.copy()
    nh[i_low] += eps
    nh[i_low + k] += eps
    nh[i_high] -= eps
    nh[i_high + k] -= eps
    return nh


def hill_climb_targeted(h_start, rng, max_iters=800, init_step=0.5,
                        n_trials=4, p_target=0.25):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    n = h.shape[0]

    for it in range(max_iters):
        _, idx = compute_c5_and_idx(h)
        k_star = idx - (n - 1)
        improved = False
        for _ in range(n_trials):
            cand = targeted_move(h, rng, k_star, scale=step) if rng.random() < p_target else perturb_h(h, rng, scale=step)
            c = compute_c5(cand)
            if c < best_c - 1e-12:

                improved = True
                break

            if c < best_c:

        if (it + 1) % decay_interval == 0:

    return h, best_c


# ----- softmax gradient (smooth max) ----------------------------------
def gradient_softmax(h, beta):
    """Gradient of (1/beta) log sum exp(beta*c) with c = correlation."""
    n = h.size
    g = 1.0 - h
    c = np.convolve(h, g, mode='full')                 # length 2n-1
    max_c = np.max(c)
    diff = c - max_c                                   # non‑positive
    exp_z = np.exp(beta * diff)                        # safe
    p = exp_z / np.sum(exp_z)                          # probabilities, sum=1

    # term1 = Σ_s p_s * (1 - h_{k+s})
    conv_rev = np.convolve(p[::-1], g, mode='full')
    term1 = conv_rev[np.arange(n) + (2 * n - 2)]

    # term2 = Σ_s p_s * h_{k-s}
    conv2 = np.convolve(p, h, mode='full')
    term2 = conv2[np.arange(n) + (n - 1)]

    return term1 - term2

def adam_optimize(h0, beta_seq, iters_seq, lr, target_sum, rng,
                  start_time, budget):
    h = h0.copy()
    n = h.size
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    best_h = h.copy()
    best_c = compute_c5(h)

    for beta, iters in zip(beta_seq, iters_seq):
        for _ in range(iters):
            if time.time() - start_time > budget:
                return best_h, best_c
            grad = gradient_softmax(h, beta)
            # Adam update (bias correction omitted for speed)
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            step = (lr / np.sqrt(beta)) / (np.sqrt(v) + eps)
            h = project_box_sum(h - step, target_sum)

            # occasional small noise to escape plateaus
            if rng.random() < 0.01:
                h = project_box_sum(h + rng.normal(scale=0.003, size=n), target_sum)

            c = compute_c5(h)
            if c < best_c:
                best_c, best_h = c, h.copy()
    return best_h, best_c

        c = compute_c5(cand)
        if c < best_c:
            best_c, best_h = c, cand.copy()

    # ----- smooth‑max Adam optimisation (low resolution) ----------
    beta_seq = [5, 15, 30, 60, 120, 240, 480, 960, 1920, 3840, 7680]
    iters_seq = [250, 250, 250, 250, 200, 200, 150, 150, 120, 120, 80]
    h_opt, c_opt = adam_optimize(best_h, beta_seq, iters_seq, lr=0.05,
                                 target_sum=n0 / 2.0, rng=rng,
                                 start_time=start_time,
                                 budget=budget_s * 0.70)
    if c_opt < best_c:
        best_c, best_h = c_opt, h_opt.copy()

    # ----- occasional random restarts --------------------------------
    while time.time() - start_time < budget_s * 0.70:
        h_start = perturb_h(best_h, rng, scale=0.25)
        h_opt, c_opt = adam_optimize(h_start, beta_seq, iters_seq, lr=0.05,
                                     target_sum=n0 / 2.0, rng=rng,
                                     start_time=start_time,
                                     budget=budget_s * 0.70)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()
        # break if time is tight
        if time.time() - start_time > budget_s * 0.68:
            break

    # ----- short targeted hill‑climb ---------------------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 5.0:
        hc_iters = int(min(800, remaining * 5))
        best_h, best_c = hill_climb_targeted(best_h, rng,
                                             max_iters=hc_iters,
                                             init_step=0.3)

    # ----- up‑sample if time allows ----------------------------------
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 30.0:
        n_up = min(720, max(540, n0 * 2))
        if n_up != n0:
            old_x = (np.arange(n0) + 0.5) * (2.0 / n0)
            new_x = (np.arange(n_up) + 0.5) * (2.0 / n_up)
            h_up = np.interp(new_x, old_x, best_h)
            h_up = project_box_sum(h_up, n_up / 2.0)

            beta_seq_up = [5, 15, 30, 60, 120, 240, 480, 960, 1920, 3840]
            iters_up = [300, 300, 300, 300, 250, 250, 200, 200, 150, 150]
            h_opt, c_opt = adam_optimize(h_up, beta_seq_up, iters_up,
                                         lr=0.04, target_sum=n_up / 2.0,
                                         rng=rng,
                                         start_time=time.time(),
                                         budget=remaining * 0.90)
            if c_opt < best_c:
                best_c, best_h, n0 = c_opt, h_opt.copy(), n_up

            remaining2 = budget_s - (time.time() - start_time)
            if remaining2 > 5.0:
                hc_iters2 = int(min(1200, remaining2 * 3))
                best_h, best_c = hill_climb_targeted(best_h, rng,
                                                     max_iters=hc_iters2,
                                                     init_step=0.2)

    # ----- final cleaning ---------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
