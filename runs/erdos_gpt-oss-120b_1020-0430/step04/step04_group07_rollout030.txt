def enforce_sum_bounds(h, target, rng):
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # While loop with small tolerance
    max_iter = 1000
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        # find candidates that can be increased/decreased
        if diff > 0:
            candidates = np.where(h < 1)[0]
            if len(candidates) == 0:
                break
            # capacity per candidate = 1 - h[i]
            caps = 1 - h[candidates]
            total_cap = caps.sum()
            if total_cap < diff - 1e-12:
                # can't satisfy, just fill all
                h[candidates] = 1
                diff = target - h.sum()
                continue
            # distribute diff proportionally to caps (or uniformly)
            # Use uniform distribution: each gets diff/len(candidates) up to cap
            inc = min(diff / len(candidates), caps.max())
            # but better to allocate equally but not exceed caps
            # We'll allocate min(diff/len, cap) and if some reach cap, repeat
            # Simpler: find amount = diff / len(candidates)
            amt = diff / len(candidates)
            # Add min(amt, cap) to each and recompute diff
            add = np.minimum(amt, caps)
            h[candidates] += add
            diff = target - h.sum()
        else:
            candidates = np.where(h > 0)[0]
            if len(candidates) == 0:
                break
            caps = h[candidates]
            total_cap = caps.sum()
            if total_cap < -diff - 1e-12:
                h[candidates] = 0
                diff = target - h.sum()
                continue
            amt = diff / len(candidates)  # diff negative
            sub = np.minimum(-amt, caps)
            h[candidates] -= sub
            diff = target - h.sum()
    return h

def apply_transfer(h, corr, a, b, eps):
    # eps: amount moved from a to b (eps positive)
    # Ensure eps <= h[a] and eps <= 1 - h[b]
    delta_a = -eps
    delta_b = eps
    n = h.shape[0]
    idx = np.arange(n)
    # use np.take with wrap indexing
    # compute shifted values
    ha_pos = np.take(h, (a + idx) % n)
    ha_neg = np.take(h, (a - idx) % n)

    hb_neg = np.take(h, (b - idx) % n)

    k2 = (a - b) % n
    cross = delta_a * delta_b

    if k2 != k1:

def compute_delta_corr(h, a, b, eps):
    delta_a = -eps
    delta_b = eps
    n = h.shape[0]
    idx = np.arange(n)
    ha_pos = np.take(h, (a + idx) % n)
    ha_neg = np.take(h, (a - idx) % n)
    hb_pos = np.take(h, (b + idx) % n)
    hb_neg = np.take(h, (b - idx) % n)
    delta_corr = delta_a * (ha_pos + ha_neg) + delta_b * (hb_pos + hb_neg)
    k1 = (b - a) % n
    k2 = (a - b) % n
    cross = delta_a * delta_b
    delta_corr[k1] += cross
    if k2 != k1:
        delta_corr[k2] += cross
    sq = delta_a * delta_a + delta_b * delta_b
    delta_corr[0] += sq
    return delta_corr

import time
import numpy as np
from scipy.optimize import minimize

# ---------- global variables for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def autocorr_cyclic(h: np.ndarray) -> np.ndarray:
    """Cyclic autocorrelation Σ_i h_i h_{i+k} for k=0..n-1."""
    ft = np.fft.fft(h)
    corr = np.real(np.fft.ifft(np.abs(ft) ** 2))
    return corr


def compute_c5_from_corr(corr: np.ndarray, n: int) -> float:
    """C5 = 1 - (2/n) * min_k corr[k]."""
    dx = 2.0 / n
    return 1.0 - dx * np.min(corr)


def compute_delta_corr(h: np.ndarray, a: int, b: int, eps: float) -> np.ndarray:
    """Delta of cyclic autocorrelation when moving eps from a to b."""
    n = h.shape[0]
    delta_a = -eps
    delta_b = eps
    idx = np.arange(n)
    ha_pos = np.take(h, (a + idx) % n)
    ha_neg = np.take(h, (a - idx) % n)
    hb_pos = np.take(h, (b + idx) % n)
    hb_neg = np.take(h, (b - idx) % n)

    delta = delta_a * (ha_pos + ha_neg) + delta_b * (hb_pos + hb_neg)

    k1 = (b - a) % n
    k2 = (a - b) % n
    cross = delta_a * delta_b
    delta[k1] += cross
    if k2 != k1:
        delta[k2] += cross

    sq = delta_a * delta_a + delta_b * delta_b
    delta[0] += sq
    return delta


def random_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    """Binary vector with exactly n/2 ones."""
    if n % 2:
        n += 1
    h = np.zeros(n, dtype=np.float64)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h


def interpolate_h(old_h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a step function to a new resolution."""
    old_n = old_h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    new_h = np.clip(new_h, 0.0, 1.0)
    target = new_n / 2.0
    diff = target - new_h.sum()
    # simple redistribution of the small residual
    while abs(diff) > 1e-12:
        idx = rng.integers(0, new_n)
        if diff > 0:
            add = min(diff, 1.0 - new_h[idx])
        else:
            add = max(diff, -new_h[idx])
        new_h[idx] += add
        diff = target - new_h.sum()
    return new_h


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Exact sum correction while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def hill_climb_with_time(h0: np.ndarray, n: int, rng: np.random.Generator,
                         start_time: float, time_budget: float) -> tuple:
    """Targeted hill‑climb aiming to raise the minimal autocorrelation."""
    end_time = start_time + time_budget
    cur_h = h0.copy()
    cur_corr = autocorr_cyclic(cur_h)
    best_h = cur_h.copy()
    best_c5 = compute_c5_from_corr(cur_corr, n)

    step_scale = 0.5
    it = 0
    max_iters = 2_000_000
    while it < max_iters and time.time() < end_time:
        # worst lag and its gradient
        k_min = int(np.argmin(cur_corr))
        g = np.roll(cur_h, k_min) + np.roll(cur_h, -k_min)
        i = int(np.argmin(g))
        j = int(np.argmax(g))
        if i == j:
            i, j = rng.choice(n, size=2, replace=False)

        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0:
            # try opposite direction
            max_eps = min(cur_h[j], 1.0 - cur_h[i])
            if max_eps > 0:
                i, j = j, i
            else:
                i, j = rng.choice(n, size=2, replace=False)
                max_eps = min(cur_h[i], 1.0 - cur_h[j])
                if max_eps <= 0:
                    it += 1
                    continue

        eps = rng.random() * max_eps * step_scale
        if eps < 1e-12:
            it += 1
            continue

        delta = compute_delta_corr(cur_h, i, j, eps)
        new_corr = cur_corr + delta
        new_c5 = compute_c5_from_corr(new_corr, n)

        if new_c5 < best_c5 - 1e-13:
            # accept improvement
            cur_h[i] -= eps
            cur_h[j] += eps
            cur_corr = new_corr
            best_h = cur_h.copy()
            best_c5 = new_c5
        else:
            # occasional probabilistic acceptance to escape plateaus
            prob = 0.001 * max(0.0, 1.0 - it / 500_000)
            if rng.random() < prob:
                cur_h[i] -= eps
                cur_h[j] += eps
                cur_corr = new_corr

        if (it + 1) % 2000 == 0:
            step_scale = max(step_scale * 0.9, 1e-5)
        it += 1

    return best_h, best_c5


# ---------- SLSQP refinement (soft‑max) ----------
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of the max‑overlap objective."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 500) -> np.ndarray:
    """One SLSQP pass with a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        # quick redistribution
        while abs(diff) > 1e-12:
            idx = rng.integers(0, n)
            add = min(diff, 1.0 - x[idx]) if diff > 0 else max(diff, -x[idx])
            x[idx] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ---------- main driver ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    best_h = None
    best_c5 = np.inf
    best_n = None

    # use parent if available
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)

    n_candidates = [96, 128, 160, 192, 224, 256, 288, 320, 352, 384]

    # ---------- multi‑restart hill‑climb ----------
    while time.time() - start < 0.78 * budget_s:
        n = int(rng.choice(n_candidates))
        fresh = rng.random() < 0.18

        if not fresh and best_h is not None:

                h_start = best_h.copy()
            else:
                h_start = interpolate_h(best_h, n)
        else:
            h_start = random_binary(n, rng)

        # run a focused hill‑climb
        time_slice = 0.015 * budget_s  # ~15 s per start (adjustable)
        h_opt, c5_opt = hill_climb_with_time(h_start, n, rng, start, time_slice)

        if c5_opt < best_c5 - 1e-6:
            best_h, best_c5, best_n = h_opt.copy(), c5_opt, n

    # ---------- up‑sampling & refinement ----------
    while time.time() - start < 0.94 * budget_s:
        if best_h is None:
            break
        n_up = best_n * 2
        if n_up > 384:
            break
        h_up = interpolate_h(best_h, n_up)
        time_slice = 0.008 * budget_s
        h_opt, c5_opt = hill_climb_with_time(h_up, n_up, rng, start, time_slice)
        if c5_opt < best_c5 - 1e-6:

    # ---------- final SLSQP polishing ----------

        best_c5 = compute_c5_from_corr(autocorr_cyclic(best_h), best_n)

    # ---------- sanity enforcement ----------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c5 = compute_c5_from_corr(autocorr_cyclic(best_h), best_n)

    return best_h.tolist(), float(final_c5), int(best_n)
