```
def grad_smooth_max(h, beta):
    n = h.shape[0]
    u = 1.0 - h
    corr = np.correlate(h, u, mode='full')
    maxc = np.max(corr)
    expb = np.exp(beta*(corr - maxc))
    w = expb / np.sum(expb)  # length 2n-1
    # conv = np.convolve(h, w)    # length 3n-2
    conv = np.convolve(h, w, mode='full')
    conv_rev = np.convolve(h, w[::-1], mode='full')
    # compute gradient
    # g_i = 1 - conv_rev[i + 2*n - 2] - conv[i + n - 1]
    idx1 = np.arange(n) + (2*n - 2)
    idx2 = np.arange(n) + (n - 1)
    g = 1.0 - conv_rev[idx1] - conv[idx2]
    return g
```

```python
def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng, max_iter=2000):
    target = h.size / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, h.size)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            if add == 0:
                continue
            h[i] += add
        else:
            sub = max(diff, -h[i])
            if sub == 0:
                continue
            h[i] += sub
        diff = target - h.sum()
    return h
```

```python
def grad_smooth_max(h, beta):
    n = h.size
    u = 1.0 - h
    corr = np.correlate(h, u, mode="full")
    maxc = np.max(corr)
    expb = np.exp(beta * (corr - maxc))
    w = expb / np.sum(expb)
    conv = np.convolve(h, w, mode='full')
    conv_rev = np.convolve(h, w[::-1], mode='full')
    offset = n - 1
    conv_slice = conv[offset:offset + n]
    conv_rev_slice = conv_rev[offset:offset + n]
    g = 1.0 - conv_rev_slice - conv_slice
    return g
```

```python
def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]  # + (1 - h[i+k])
            g[k:] -= h[:upper]        # - h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # adjust sum by random redistribution
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h
```

def binary_sa_hill_climb(h_bin, rng, time_deadline):
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    last_improve = 0
    while time.time() < time_deadline:
        # pick random swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # perform swap
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        T = init_T * (0.9995 ** it) + 1e-12
        accept = False
        if c < cur_c:
            accept = True
        else:
            if rng.random() < np.exp(-(c - cur_c) / T):
                accept = True
        if accept:
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
                last_improve = it
        else:
            # revert swap
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
        if it - last_improve > 5000:
            break
    return best_h, best_c

        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def binary_sa_hill_climb(h_bin, rng, time_deadline):
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    last_improve = 0
    while time.time() < time_deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        T = init_T * (0.9995 ** it) + 1e-12
        accept = False
        if c < cur_c:
            accept = True
        else:
            if rng.random() < np.exp(-(c - cur_c) / T):
                accept = True
        if accept:
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
                last_improve = it
        else:
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
        if it - last_improve > 5000:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.98

    best_h = None
    best_c = np.inf
    best_n = None

    parent = np.array(initial_h_values, dtype=float) if 'initial_h_values' in globals() else None
    n_base = 360
    target = n_base / 2.0

    if parent is not None:
        xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_base, endpoint=False)
        parent_interp = np.interp(xt, xs, parent)
        parent_interp = np.clip(parent_interp, 0.0, 1.0)
        adjust_sum_exact(parent_interp, rng)
    else:
        parent_interp = None

    betas = [0.5, 1, 2, 5, 10, 20, 40, 80, 160, 320, 640, 1280, 2560]
    iters_per_beta = 150
    restarts = 8

    for r in range(restarts):
        if time.time() > deadline:
            break

        if r == 0 and parent_interp is not None:
            h = parent_interp.copy()
        else:
            h = np.zeros(n_base)
            h[rng.choice(n_base, n_base // 2, replace=False)] = 1.0

        adjust_sum_exact(h, rng)
        h = np.clip(h + rng.normal(scale=0.03, size=n_base), 0.0, 1.0)
        adjust_sum_exact(h, rng)

        for beta in betas:
            if time.time() > deadline:
                break
            step0 = 0.6 / (1.0 + beta ** 0.4)
            for it in range(iters_per_beta):
                if time.time() > deadline:
                    break
                g = grad_smooth_max(h, beta)
                step = step0 * (0.94 ** it)
                h = h - step * g
                h = np.clip(h, 0.0, 1.0)
                adjust_sum_exact(h, rng)
                if it % 30 == 0:
                    c = compute_c5(h)
                    if c < best_c:
                        best_h, best_c, best_n = h.copy(), c, n_base
            h = np.clip(h + rng.normal(scale=0.01 / (beta + 1), size=n_base), 0.0, 1.0)
            adjust_sum_exact(h, rng)

        h = gradient_refine(h, rng, step=0.02, max_iter=12)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n_base

        idx = np.argpartition(h, -n_base // 2)[-n_base // 2:]
        h_bin = np.zeros(n_base)
        h_bin[idx] = 1.0
        sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.4)
        h_bin_opt, c_bin = binary_sa_hill_climb(h_bin, rng, sub_dead)
        if c_bin < best_c:
            best_h, best_c, best_n = h_bin_opt.copy(), c_bin, n_base

    if best_h is not None and time.time() < deadline and best_n * 2 <= 720:
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        adjust_sum_exact(h_up, rng)
        for beta in [0.5, 1, 2, 5, 10, 20, 40, 80]:
            if time.time() > deadline:
                break
            step = 0.5 / (1.0 + beta ** 0.3)
            for _ in range(80):
                if time.time() > deadline:
                    break
                g = grad_smooth_max(h_up, beta)
                h_up = np.clip(h_up - step * g, 0.0, 1.0)
                adjust_sum_exact(h_up, rng)
        h_up = gradient_refine(h_up, rng, step=0.02, max_iter=8)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, n_up

    if best_h is None:
        n_f = 360
        best_n = n_f
        best_h = np.zeros(n_f)
        best_h[: n_f // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
