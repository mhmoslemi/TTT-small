def anneal_optimize(h0, rng, time_limit):
    n = h0.size
    target = n/2.0
    h = np.clip(h0,0.0,1.0).copy()
    # enforce sum exactly using enforce_sum
    h = enforce_sum(h, target, rng)
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    start = time.time()
    # set schedule
    max_iters = int(1e7)  # big bound, break via time
    # initial temperature can be maybe 0.01
    T0 = 0.02
    T_min = 0.0001
    # linear exponential schedule
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        t = it / max_iters
        T = T0 * (T_min / T0) ** t  # exponential decay
        # generate move
        # pick i in [0,n), j in [0,n) distinct
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # ensure variable bounds after move
        # compute max amount that can be transferred from i to j while staying in [0,1]
        if h[i] <= 0 or h[j] >= 1:
            # maybe transfer opposite direction from j to i
            continue
        max_eps = min(h[i], 1.0 - h[j])
        eps = rng.random() * max_eps  # uniform transfer amount
        # apply move
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        c_new = compute_c5(h_new)
        delta = c_new - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / T):
            h = h_new
            cur_c = c_new
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    return best_h, best_c
```

    corr = np.correlate(x, 1.0 - x, mode='full')

    max_bc = np.max(bc)

    # compute gradient
    grad = np.zeros_like(x)
    # iterate over k offsets
    shift_range = np.arange(- (n - 1), n)  # length 2n-1
    for idx, k in enumerate(shift_range):

        # compute contributions

            grad[:n - k] += w * (1.0 - x[k:])   # (1 - x_{j+k})

    # objective: softmax + mu * diff^2

import time, numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Exact C5 for step function h on [0,2] using dx=2/n."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h, target, rng, max_iter=5000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-14:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-14:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-14:
            for i in range(n):
                if abs(diff) < 1e-14:
                    break
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:

        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        for _ in range(2000):
            if abs(diff) < 1e-14:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

                improved = True
                break
        if not improved and rng.random() < 0.02:
            new_h = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
                improved = True
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return h, best_c

def max_overlap_and_shift(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return float(corr[idx]), idx - (h.size - 1)

def pair_transfer_optimize(h0, rng, max_iters=5000, time_limit=30.0,
                          init_factor=0.5, decay=0.999):
    h = h0.copy()
    n = h.size
    best_h, best_c = h.copy(), compute_c5(h)
    start = time.time()
    factor = init_factor
    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        max_val, shift = max_overlap_and_shift(h)
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c, best_h = cur_c, h.copy()
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            i_range = np.arange(-shift, n)
            j_range = i_range + shift
        if i_range.size == 0:
            factor = init_factor
            continue
        prod = h[i_range] * (1.0 - h[j_range])
        idx_max = int(np.argmax(prod))
        i, j = int(i_range[idx_max]), int(j_range[idx_max])
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            factor = init_factor
            continue
        eps = factor * max_eps
        if eps <= 0.0:
            factor = init_factor
            continue
        h[i] -= eps
        h[j] += eps
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
        factor = max(1e-4, factor * decay)
    final_c = compute_c5(best_h)
    return best_h, final_c

def softmax_obj_and_grad(x, beta, mu, target):
    n = x.size
    corr = np.correlate(x, 1.0 - x, mode='full')
    bc = beta * corr
    max_bc = np.max(bc)
    exp_bc = np.exp(bc - max_bc)
    sum_exp = np.sum(exp_bc)
    softmax = (max_bc + np.log(sum_exp)) / beta
    w = exp_bc / sum_exp
    grad = np.zeros_like(x)
    for idx in range(2 * n - 1):
        k = idx - (n - 1)
        wk = w[idx]
        if wk == 0.0:
            continue
        if k >= 0:
            grad[: n - k] += wk * (1.0 - x[k:])
            grad[k:] -= wk * x[: n - k]
        else:
            kp = -k
            grad[kp:] += wk * (1.0 - x[: n - kp])
            grad[: n - kp] -= wk * x[kp:]
    diff = x.sum() - target
    grad += 2.0 * mu * diff
    obj = softmax + mu * diff * diff
    return obj, grad

def softmax_penalized_obj(x, beta, mu, target):
    obj, _ = softmax_obj_and_grad(x, beta, mu, target)
    return obj

def softmax_penalized_grad(x, beta, mu, target):
    _, grad = softmax_obj_and_grad(x, beta, mu, target)
    return grad

def gradient_optimize(h0, rng, max_time, beta_list):
    n = h0.size
    target = n / 2.0
    best_h, best_c = h0.copy(), compute_c5(h0)
    start = time.time()
    for beta in beta_list:
        if time.time() - start > max_time:
            break
        mu = beta * 10.0
        res = minimize(
            softmax_penalized_obj,
            best_h,
            args=(beta, mu, target),
            jac=softmax_penalized_grad,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * n,
            options={'maxiter': 200, 'ftol': 1e-12, 'disp': False},
        )
        x = np.clip(res.x, 0.0, 1.0)
        x = enforce_sum(x, target, rng)
        c = compute_c5(x)
        if c < best_c:
            best_c, best_h = c, x.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    best_h, best_c = h0.copy(), compute_c5(h0)

    # quick hill‑climb (small budget)
    if time.time() - start < 0.1 * budget_s:
        hc_h, hc_c = hill_climb_multi(best_h, rng, max_iters=1000, init_step=0.4, n_trials=4)
        if hc_c < best_c:
            best_h, best_c = hc_h, hc_c

    # gradient refinement (≈60% of remaining time)
    remaining = budget_s - (time.time() - start)
    grad_time = remaining * 0.6
    beta_seq = [5, 20, 80, 200, 500, 1000, 2000, 5000]
    grad_h, grad_c = gradient_optimize(best_h, rng, grad_time, beta_seq)
    if grad_c < best_c:
        best_h, best_c = grad_h, grad_c

    # final pair‑transfer (use whatever time left)
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0:
        pt_h, pt_c = pair_transfer_optimize(best_h, rng, max_iters=20000, time_limit=remaining)
        if pt_c < best_c:
            best_h, best_c = pt_h, pt_c

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
