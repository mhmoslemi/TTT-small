import time
import numpy as np
from scipy.optimize import minimize

# ---------- utility functions ----------
def enforce_sum(h, target, rng, max_iter=2000):
    """Clip to [0,1] and adjust to meet exact sum `target` (float)."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # final uniform correction if needed
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    """Return C5 for a discrete step function h on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

# ---------- SLSQP refinement ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    # approximate max correlation using log‑sum‑exp
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize_one_beta(h0, beta, rng, max_iter=200):
    """Run SLSQP on the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum (possible tiny drift)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, CURRENT_TARGET, rng, max_iter=2000)
    return x

# ---------- main driver ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.time()
    total_budget = budget_s

    # ----- choose grid size (720 gives a good balance) -----
    target_n = 720
    target_sum = target_n // 2

    # ----- build initial binary vector -----
    if "initial_h_values" in globals():
        h_float = np.array(initial_h_values, dtype=float)
        n0 = h_float.size
        if n0 != target_n:
            old_x = (np.arange(n0) + 0.5) * (2.0 / n0)
            new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
            h_float = np.interp(new_x, old_x, h_float)
        # threshold to binary
        h_bin = (h_float > 0.5).astype(float)
        cur_sum = int(h_bin.sum())
        diff = target_sum - cur_sum
        if diff > 0:  # need more ones
            zeros = np.where(h_bin == 0)[0]
            # pick zeros with largest original value
            order = zeros[np.argsort(-h_float[zeros])]
            h_bin[order[:diff]] = 1.0
        elif diff < 0:  # too many ones
            ones = np.where(h_bin == 1)[0]
            order = ones[np.argsort(h_float[ones])]
            h_bin[order[: -diff]] = 0.0
    else:
        # pure random binary with exact sum
        h_bin = np.zeros(target_n, dtype=float)
        idx = rng.choice(target_n, size=target_sum, replace=False)
        h_bin[idx] = 1.0

    # ensure feasibility
    h_bin = enforce_sum(h_bin, target_sum, rng)

    # ----- simulated annealing on binary vectors -----
    main_budget = total_budget * 0.85
    t_end_main = t_start + main_budget

    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c

    # parameters for temperature schedule
    T0 = 0.015
    Tf = 1e-6

    while time.time() < t_end_main:
        # progress fraction for temperature
        frac = (time.time() - t_start) / main_budget
        T = max(Tf, T0 * ((Tf / T0) ** frac))

        # worst lag for current configuration
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        lag = np.argmax(corr) - (target_n - 1)

        # choose a swap that targets the worst lag
        if lag >= 0:
            bad = np.where((cur_h[:target_n - lag] == 1) &
                           (cur_h[lag:] == 0))[0]
            good = np.where((cur_h[:target_n - lag] == 0) &
                            (cur_h[lag:] == 1))[0]
            # shift good indices to match original position
            if good.size:
                good = good + lag
        else:
            k = -lag
            bad = np.where((cur_h[k:] == 1) &
                           (cur_h[:target_n - k] == 0))[0] + k
            good = np.where((cur_h[k:] == 0) &
                            (cur_h[:target_n - k] == 1))[0] + k

        if bad.size and good.size:
            i = rng.choice(bad)
            j = rng.choice(good)
        else:
            # fallback random 1↔0 swap
            ones = np.where(cur_h == 1)[0]
            zeros = np.where(cur_h == 0)[0]
            i = rng.choice(ones)
            j = rng.choice(zeros)

        new_h = cur_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c

    # ----- greedy hill climbing (pure improvement) -----
    hill_budget = total_budget * 0.03
    t_end_hill = t_start + main_budget + hill_budget
    while time.time() < t_end_hill:
        # try a few random swaps, accept only if improve
        improved = False
        for _ in range(200):
            ones = np.where(best_h == 1)[0]
            zeros = np.where(best_h == 0)[0]
            i = rng.choice(ones)
            j = rng.choice(zeros)
            trial = best_h.copy()
            trial[i] = 0.0
            trial[j] = 1.0
            trial_c = compute_c5(trial)
            if trial_c < best_c:
                best_h, best_c = trial, trial_c
                improved = True
                break
        if not improved:
            break  # no simple improvement left

    # ----- SLSQP continuous refinement -----
    beta_list = [10, 50, 200, 1000, 5000]
    t_end_slsqp = t_start + total_budget
    for beta in beta_list:
        if time.time() >= t_end_slsqp:

        try:

        refined_c = compute_c5(refined)
        if refined_c < best_c:
            best_h, best_c = refined, refined_c

    # ----- final clean‑up -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target_sum, rng)
    final_c = compute_c5(best_h)
