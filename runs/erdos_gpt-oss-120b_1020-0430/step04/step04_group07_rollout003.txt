def hill_climb(h, rng, max_iters, step_scale):
    best_h = h.copy()
    best_c = compute_c5(best_h)
    scale = step_scale
    decay = max(1, max_iters // 5)
    temp = 0.01 # initial temperature relative to best_c
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=scale)
        nc = compute_c5(nh)
        # accept if better or with prob
        if nc < best_c or rng.random() < np.exp(-(nc - best_c) / temp):
            h = nh
            if nc < best_c:
                best_c, best_h = nc, nh.copy()
        if (it+1) % decay == 0:
            scale = max(scale * 0.7, 1e-4)
            temp *= 0.95
    return best_h, best_c

# global variables used by the SLSQP soft‑max objective
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector with entries in [0,1] and sum=n/2."""
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()

            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the current worst‑shift overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def hill_climb(h: np.ndarray, rng: np.random.Generator,
               max_iters: int = 2000, step_scale: float = 0.6) -> tuple:
    """Hill‑climb with occasional simulated‑annealing acceptance."""
    best_h = h.copy()
    best_c = compute_c5(best_h)
    scale = step_scale
    decay = max(1, max_iters // 5)
    temp = best_c * 0.02 + 1e-8  # initial temperature
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=scale)
        nc = compute_c5(nh)
        if nc < best_c:
            best_c, best_h, h = nc, nh.copy(), nh
        else:
            # annealing acceptance
            if rng.random() < math.exp(-(nc - best_c) / temp):
                h = nh
        if (it + 1) % decay == 0:
            scale = max(scale * 0.7, 1e-4)
            temp *= 0.95
    return best_h, best_c


def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 500) -> np.ndarray:
    """Soft‑max SLSQP optimisation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta

    def sum_eq_constraint(x):
        return np.sum(x) - CURRENT_TARGET

    def softmax_obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = CURRENT_BETA * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # exact sum correction (quick redistribution)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
