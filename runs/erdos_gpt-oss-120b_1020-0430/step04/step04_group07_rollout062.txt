compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * (2.0 / n)

    def run(seed=42, budget_s=1100, **kwargs):
        rng = np.random.default_rng(seed)
        start = time.time()
        best_h = None; best_c = np.inf; best_n = None

        # Check initial
        if 'initial_h_values' in globals():
            h0 = np.array(initial_h_values, dtype=float)
            h0 = np.clip(h0,0,1)
            h0 = enforce_sum(h0, rng)
            c0 = compute_c5(h0)
            best_h, best_c, best_n = h0.copy(), c0, h0.size

        # Stage: Binary MILP attempts
        # candidate ns for MILP
        milp_ns = [96, 120, 144]
        for n in milp_ns:
            elapsed = time.time() - start
            remaining = budget_s - elapsed

        constraints.append(w[:, i] <= h[i])
    # w >= h_i + h_j - 1

            pass

def project_to_constraints(v, target):
    # v is array (any real)
    # returns y with 0<=y_i<=1 and sum y_i = target
    # solve min ||y - v||^2 s.t. sum y = target, 0<=y_i<=1.
    # Equivalent to projecting onto capped simplex.
    # Can be done with water-filling: find lambda such that sum clip(v_i - lambda, 0, 1) = target.
    # Use binary search on lambda.
    lo = -1.0
    hi = 1.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        y = np.clip(v - lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return y

CURRENT_BETA = None

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def project_to_sum_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the box [0,1]^n with sum == target."""
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        y = np.clip(v - lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def enforce_sum(h: np.ndarray, rng=None) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    return project_to_sum_box(h, target)

def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector in [0,1] with sum = n/2."""
    v = rng.random(n)
    return enforce_sum(v)

def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k = max_k ∑ h_i (1‑h_{i+k})."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h: np.ndarray, grad: np.ndarray, rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])

        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

    """Hill‑climb using gradient‑directed transfers."""
    n = h.size

    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """Smooth approximation of max_k O_k."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int = 600) -> np.ndarray:
    """Soft‑max SLSQP optimisation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n

    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = enforce_sum(x)
    return x

def gradient_refine(h: np.ndarray, rng: np.random.Generator, step: float = 0.015, max_iter: int = 8) -> np.ndarray:
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size

        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)

        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h)
    return h

                if c_up < best_c:
                    best_h, best_c, best_n = h_up.copy(), c_up, n_up

            h_up = gradient_refine(h_up, rng, step=0.01, max_iter=6)
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, best_n = h_up.copy(), c_up, n_up

    # --- final simulated‑annealing style tweaking -------------------------
    while time.time() - start < budget_s * 0.999 and best_h is not None:
        n = best_h.size
        grad = worst_gradient(best_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if best_h[i] > 0 and best_h[j] < 1:
            eps = min(best_h[i], 1 - best_h[j]) * rng.random() * 0.03
            nh = best_h.copy()
            nh[i] -= eps
            nh[j] += eps
            c_new = compute_c5(nh)
            if c_new < best_c:
                best_h, best_c = nh, c_new
        else:
            i, j = rng.choice(n, size=2, replace=False)
            if best_h[i] > 0 and best_h[j] < 1:
                eps = min(best_h[i], 1 - best_h[j]) * rng.random() * 0.01
                nh = best_h.copy()
                nh[i] -= eps
                nh[j] += eps
                c_new = compute_c5(nh)
                if c_new < best_c:
                    best_h, best_c = nh, c_new

    # --- final sanity check ------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
