    diff = target - h.sum()
    n = h.size
    # while diff not zero, adjust random indices.
    rng = np.random.default_rng()
    for _ in range(10000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0,n)
        # max increase if diff>0 else max decrease
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0:
            h[i] += add
            diff = target - h.sum()
    # final correction: if still diff small, add uniform diff/n
    if abs(diff) > 1e-12:

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    n = x.size
    m = 2 * n
    H = np.fft.rfft(x, n=m)
    G = np.fft.rfft(1.0 - x, n=m)
    conv = np.fft.irfft(H * np.conj(G), n=m)
    corr = conv[: 2 * n - 1] * (2.0 / n)
    beta = CURRENT_BETA
    mx = np.max(corr)
    # smooth max = mx + (1/β) log Σ exp(β (c_i - mx))
    val = mx + (1.0 / beta) * np.log(np.sum(np.exp(beta * (corr - mx))))
    return val


def slsqp_optimize_one_beta(h0, beta, rng, max_iter=200):

    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x += diff / n
        x = np.clip(x, 0.0, 1.0)
        diff = CURRENT_TARGET - x.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
                x[i] += add
                diff = CURRENT_TARGET - x.sum()
    return x


# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    # ----- initial vector -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    best_h, best_c = h0.copy(), compute_c5_fast(h0)

    # ----- main stochastic search -----
    main_deadline = t0 + budget_s * 0.85
    beta_vals = [10, 50, 200, 800, 3000, 10000]

    while time.time() < main_deadline:
        # random restart bias
        if rng.random() < 0.6:
            start = perturb_h(best_h, rng, scale=0.5)
        else:
            start = random_feasible_vector(n, rng)

        # Simulated annealing
        sa_h, sa_c = simulated_annealing(start, rng, main_deadline)
        if sa_c < best_c:
            best_h, best_c = sa_h, sa_c

        # SLSQP refinement ladder
        for beta in beta_vals:
            if time.time() >= main_deadline:
                break
            refined = slsqp_optimize_one_beta(best_h, beta, rng, max_iter=200)
            ref_c = compute_c5_fast(refined)
            if ref_c < best_c:
                best_h, best_c = refined, ref_c

    # ----- optional up‑sampling -----
    remaining = budget_s - (time.time() - t0)
    if remaining > budget_s * 0.1:
        n_up = max(720, n * 2)
        old_x = (np.arange(n) + 0.5) * (2.0 / n)
        new_x = (np.arange(n_up) + 0.5) * (2.0 / n_up)
        up_h = np.interp(new_x, old_x, best_h)
        up_h = enforce_sum(up_h, n_up / 2.0, rng)

        up_deadline = time.time() + remaining * 0.6
        up_h, up_c = simulated_annealing(up_h, rng, up_deadline)
        for beta in beta_vals:
            if time.time() - t0 >= budget_s:
                break
            up_h = slsqp_optimize_one_beta(up_h, beta, rng, max_iter=200)
            up_c = compute_c5_fast(up_h)
            if up_c < best_c:
                best_h, best_c, n = up_h.copy(), up_c, n_up

    # ----- final quick hill‑climb -----
    cur_h, cur_c = best_h.copy(), best_c
    step = 0.4
    for _ in range(4000):
        if time.time() - t0 > budget_s:
            break
        nh = perturb_h(cur_h, rng, scale=step)
        nc = compute_c5_fast(nh)
        if nc < cur_c:
            cur_h, cur_c = nh, nc
            if nc < best_c:
                best_h, best_c = nh.copy(), nc
        step *= 0.9995

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5_fast(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
