import time, numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    # compute correlation with smoothing parameter CURRENT_BETA
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log-sum-exp stable
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def compute_c5(h):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)

def enforce_sum(h, target, rng, max_iter=5000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-9:
        free = (h > 1e-12) & (h < 1 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            per = diff / idx.size
            h[idx] += per
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def coeffs_to_h(coeffs, x, rng):
    m = len(coeffs) // 2
    a = coeffs[:m]
    b = coeffs[m:]
    # compute cos and sin matrices
    i = np.arange(1, m + 1)[:, None]  # shape (m,1)
    cos_mat = np.cos(np.pi * i * x)   # (m,n)
    sin_mat = np.sin(np.pi * i * x)
    h = 0.5 + a @ cos_mat + b @ sin_mat
    h = enforce_sum(h, x.size / 2.0, rng)
    return h

def local_optimize(coeffs, x, rng, max_iter=300):
    def obj(c):
        h = coeffs_to_h(c, x, rng)
        return compute_c5(h)
    res = minimize(obj, coeffs, method='Nelder-Mead',
                   options={'maxiter': max_iter, 'fatol': 1e-9, 'xatol': 1e-9, 'disp': False})
    h_opt = coeffs_to_h(res.x, x, rng)
    c_opt = compute_c5(h_opt)
    return h_opt, c_opt, res.x

def slsqp_optimize(h0, beta, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # correct sum drift
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng_local = np.random.default_rng()
        n = x.size
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng_local.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # Use medium resolution
    n = 720
    x = (np.arange(n) + 0.5) * (2.0 / n)

    best_h = None
    best_c = np.inf
    best_coeffs = None

    # If a parent vector exists, adapt it
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size != n:
            old_x = (np.arange(h0.size) + 0.5) * (2.0 / h0.size)
            h0 = np.interp(x, old_x, h0)
        h0 = enforce_sum(h0, n / 2.0, rng)
        c0 = compute_c5(h0)
        best_h = h0
        best_c = c0

    # Random search over Fourier coefficients
    dim = 16
    max_random = 2500
    for it in range(max_random):
        if time.time() - start > budget_s * 0.45:
            break
        coeffs = rng.uniform(-0.3, 0.3, size=dim)
        h = coeffs_to_h(coeffs, x, rng)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
            best_coeffs = coeffs.copy()

    # Local refinement on the best coefficient vector
    if best_coeffs is not None:
        # allocate up to 20% of remaining time
        while time.time() - start < budget_s * 0.65:
            h_ref, c_ref, new_coeffs = local_optimize(best_coeffs, x, rng, max_iter=300)
            if c_ref < best_c:
                best_c = c_ref
                best_h = h_ref.copy()
                best_coeffs = new_coeffs.copy()
                # continue refining from new coefficients
            else:
                # no improvement, break
                break

    # Final refinement via SLSQP on full grid
    # allocate remaining time (up to 90% of budget)
    beta_list = [50, 200, 800, 2000, 5000, 10000]
    for beta in beta_list:
        if time.time() - start > budget_s * 0.9:
            break
        try:
            h_ref = slsqp_optimize(best_h, beta, max_iter=500)
        except Exception:
            continue
        c_ref = compute_c5(h_ref)
        if c_ref < best_c:
            best_c = c_ref
            best_h = h_ref.copy()

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)

import time
import numpy as np
from scipy.optimize import minimize

# globals for the smooth surrogate used by SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint: sum(x) == CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k correlation using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for a step function sampled on [0,2]."""
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h to satisfy Σh = target while staying in [0,1]."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-9:
        free = (h > 1e-12) & (h < 1 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            per = diff / idx.size
            h[idx] += per
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def coeffs_to_h(coeffs, x, rng):
    """Construct h from Fourier‑type coefficients and enforce feasibility."""
    m = len(coeffs) // 2
    a = coeffs[:m]
    b = coeffs[m:]
    i = np.arange(1, m + 1)[:, None]          # (m,1)
    cos_mat = np.cos(np.pi * i * x)           # (m,n)
    sin_mat = np.sin(np.pi * i * x)           # (m,n)
    h = 0.5 + a @ cos_mat + b @ sin_mat       # (n,)
    h = enforce_sum(h, x.size / 2.0, rng)
    return h


def local_optimize(coeffs, x, rng, max_iter=300):
    """Refine coefficients with Nelder‑Mead."""
    def obj(c):
        h = coeffs_to_h(c, x, rng)
        return compute_c5(h)

    res = minimize(
        obj,
        coeffs,
        method="Nelder-Mead",
        options={"maxiter": max_iter, "fatol": 1e-9, "xatol": 1e-9, "disp": False},
    )
    h_opt = coeffs_to_h(res.x, x, rng)
    c_opt = compute_c5(h_opt)
    return h_opt, c_opt, res.x


def slsqp_optimize(h0, beta, max_iter=400):
    """SLSQP refinement of the smooth max‑approximation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny drift correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng_local = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng_local.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Base resolution
    # ------------------------------------------------------------------
    n = 720
    x = (np.arange(n) + 0.5) * (2.0 / n)

    best_h = None
    best_c = np.inf
    best_coeffs = None

    # ------------------------------------------------------------------
    # Parent vector (if supplied)
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size != n:
            old_x = (np.arange(h0.size) + 0.5) * (2.0 / h0.size)
            h0 = np.interp(x, old_x, h0)
        h0 = enforce_sum(h0, n / 2.0, rng)
        c0 = compute_c5(h0)
        best_h, best_c = h0, c0

    # ------------------------------------------------------------------
    # Random search over Fourier coefficients
    # ------------------------------------------------------------------
    dim = 16                     # 8 cosine + 8 sine terms
    max_random = 8000
    for it in range(max_random):
        if time.time() - start > budget_s * 0.45:
            break
        coeffs = rng.uniform(-0.30, 0.30, size=dim)
        h = coeffs_to_h(coeffs, x, rng)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
            best_coeffs = coeffs.copy()

    # ------------------------------------------------------------------
    # Local refinement on the best coefficient vector
    # ------------------------------------------------------------------
    if best_coeffs is not None:
        while time.time() - start < budget_s * 0.65:
            h_ref, c_ref, new_coeffs = local_optimize(best_coeffs, x, rng, max_iter=300)
            if c_ref < best_c:
                best_c = c_ref
                best_h = h_ref.copy()
                best_coeffs = new_coeffs.copy()
                # continue refining from the improved coefficients
            else:
                break

    # ------------------------------------------------------------------
    # Final refinement with SLSQP on the full grid
    # ------------------------------------------------------------------
    beta_list = [50, 200, 800, 2000, 5000, 10000]
    for beta in beta_list:
        if time.time() - start > budget_s * 0.9:
            break
        try:
            h_ref = slsqp_optimize(best_h, beta, max_iter=500)
        except Exception:
            continue
        c_ref = compute_c5(h_ref)
        if c_ref < best_c:
            best_c = c_ref
            best_h = h_ref.copy()

    # ------------------------------------------------------------------
    # Final clean‑up
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
