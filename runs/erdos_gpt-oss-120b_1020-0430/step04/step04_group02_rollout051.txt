def subgradient_descent(h0, rng, max_iters, init_step):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5_fft(h)
    step = init_step
    n = len(h)
    for it in range(max_iters):
        # compute correlation using FFT
        n2 = 1 << ((2*n-1).bit_length())
        H = np.fft.rfft(h, n2)
        B = np.fft.rfft((1-h)[::-1], n2)
        corr = np.fft.irfft(H*B, n2)[:2*n-1]
        max_corr = corr.max()
        c_val = max_corr * (2.0 / n)
        # update best
        if c_val < best_c - 1e-12:
            best_c = c_val
            best_h = h.copy()
        # compute gradient subgradient
        # find indices attaining max within tolerance
        # tolerance relative maybe 1e-12
        idxs = np.where(np.abs(corr - max_corr) < 1e-12)[0]  # tolerance may need adapt

def max_corr_fft(h):
    n = h.shape[0]
    n2 = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, n2)
    B = np.fft.rfft((1 - h)[::-1], n2)
    corr = np.fft.irfft(H * B, n2)[:2*n-1]
    return corr

def compute_c5_fft(h):
    n = h.shape[0]
    corr = max_corr_fft(h)
    return np.max(corr) * (2.0 / n)

import time, numpy as np, scipy.optimize as opt

def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    lo, hi = lo, hi
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low+high)/2.0
        y = np.clip(x+mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x+(low+high)/2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y

def compute_c5_fft(h):
    n = h.shape[0]
    n2 = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, n2)
    B = np.fft.rfft((1 - h)[::-1], n2)
    corr = np.fft.irfft(H * B, n2)[:2*n-1]
    return np.max(corr) * (2.0 / n), corr

def gradient_shift(h, k):
    n = h.shape[0]
    g = np.ones_like(h)
    if k >= 0:
        if k < n:
            g[:n - k] -= h[k:]
            g[k:] -= h[:n - k]
    else:
        k = -k
        if k < n:
            g[:n - k] -= h[k:]
            g[k:] -= h[:n - k]
    return g

def subgradient_descent(h0, rng, max_iters=2000, init_step=0.5):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c, _ = compute_c5_fft(h)
    step = init_step
    # precompute n2 for FFT
    n2 = 1 << ((2*n-1).bit_length())
    # For schedule we can precompute decay steps
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        H = np.fft.rfft(h, n2)
        B = np.fft.rfft((1 - h)[::-1], n2)
        corr = np.fft.irfft(H * B, n2)[:2*n-1]
        max_corr = corr.max()
        c_val = max_corr * (2.0 / n)
        if c_val < best_c - 1e-12:
            best_c = c_val
            best_h = h.copy()
        # indices where corr close to max
        # use relative tolerance
        eps = 1e-12
        idxs = np.where(np.abs(corr - max_corr) <= eps)[0]
        grad = np.zeros_like(h)
        for idx in idxs:
            k = idx - (n - 1)
            grad += gradient_shift(h, k)
        if len(idxs) > 0:
            grad /= len(idxs)
        else:
            # fallback to random gradient?
            grad = np.zeros_like(h)
        # gradient descent step: move opposite direction
        h_new = h - step * grad
        # enforce bounds before projection
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_box_sum(h_new, n/2.0)
        h = h_new
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-6)
    # final compute best_c if needed
    final_c, _ = compute_c5_fft(best_h)
    return best_h, final_c

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(n)
    j = rng.integers(n)
    if i == j:
        return h.copy()
    # move mass from i to j within bounds
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

# SLSQP surrogate functions similar to prior

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    from scipy.optimize import minimize
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)

    corr = np.fft.irfft(H * B, n2)[: 2 * n - 1]
    return corr


def compute_c5_fft(h):
    n = h.shape[0]
    corr = max_corr_fft(h)
    return np.max(corr) * (2.0 / n), corr


# ---------- gradient ----------
def gradient_shift(h, k):
    n = h.shape[0]
    g = np.ones_like(h)
    if k >= 0:
        if k < n:
            g[: n - k] -= h[k:]
            g[k:] -= h[: n - k]
    else:
        k = -k
        if k < n:
            g[: n - k] -= h[k:]
            g[k:] -= h[: n - k]
    return g


# ---------- perturbation ----------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


# ---------- sub‑gradient descent ----------
def subgradient_descent(h0, rng, max_iters=2500, init_step=0.5):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c, _ = compute_c5_fft(best_h)
    step = init_step
    n2 = 1 << ((2 * n - 1).bit_length())
    decay_interval = max(1, max_iters // 10)

    for it in range(max_iters):
        # correlation
        H = np.fft.rfft(h, n2)
        B = np.fft.rfft((1.0 - h)[::-1], n2)
        corr = np.fft.irfft(H * B, n2)[: 2 * n - 1]
        max_corr = corr.max()
        c_val = max_corr * (2.0 / n)

        if c_val < best_c - 1e-12:
            best_c, best_h = c_val, h.copy()

        # sub‑gradient of the active shifts
        idxs = np.where(np.abs(corr - max_corr) <= 1e-12)[0]
        grad = np.zeros_like(h)
        for idx in idxs:
            k = idx - (n - 1)
            grad += gradient_shift(h, k)
        if idxs.size:
            grad /= idxs.size

        # step and projection
        h_new = h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h = project_box_sum(h_new, n / 2.0)

        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-6)

    final_c, _ = compute_c5_fft(best_h)
    return best_h, final_c


# ---------- SLSQP surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)


# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # --- initialise ---
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_box_sum(h0, n0 / 2.0)
    else:
        n0 = 360
        h0 = rng.random(n0)
        h0 = project_box_sum(h0, n0 / 2.0)

    # decide resolution
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    target_n = 720
    if remaining > budget_s * 0.6:
        target_n = 1440

    if target_n != n0:
        old_x = (np.arange(n0) + 0.5) * (2.0 / n0)
        new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
        h = np.interp(new_x, old_x, h0)
        h = project_box_sum(h, target_n / 2.0)
    else:
        h = h0.copy()

    best_h = h.copy()
    best_c, _ = compute_c5_fft(best_h)

    n = best_h.size
    beta_list = [5, 20, 80, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000]

    # --- optimisation loop ---
    while True:
        if time.time() - start > budget_s * 0.85:
            break

        # start point for this trial
        if rng.random() < 0.2:
            start_vec = rng.random(n)
            start_vec = project_box_sum(start_vec, n / 2.0)
        else:
            start_vec = perturb_h(best_h, rng, scale=0.3)

        # sub‑gradient descent
        iters = int(2000 + 1000 * rng.random())
        h_opt, c_opt = subgradient_descent(start_vec, rng, max_iters=iters, init_step=0.5)

        if c_opt < best_c - 1e-12:
            best_c, best_h = c_opt, h_opt.copy()
            # SLSQP refinement with increasing β
            for beta in beta_list:
                if time.time() - start > budget_s * 0.95:
                    break
                try:
                    h_ref = slsqp_optimize(best_h, beta, max_iter=400)
                except Exception:
                    continue
                c_ref, _ = compute_c5_fft(h_ref)
                if c_ref < best_c - 1e-12:
                    best_c, best_h = c_ref, h_ref.copy()

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c, _ = compute_c5_fft(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
