def enforce_sum(h, target, rng):
    # Clip
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # While absolute diff > tolerance:
    tries=0
    while abs(diff) > 1e-12 and tries < 1000:
        if diff > 0:
            # Need to increase sum
            # Find indices where h < 1
            idx = np.where(h < 1 - 1e-12)[0]
            if len(idx) == 0:
                break
            # Distribute diff equally among them
            inc = diff / len(idx)
            # Ensure we don't exceed 1
            inc = min(inc, np.min(1 - h[idx]))
            h[idx] += inc
        else:
            idx = np.where(h > 1e-12)[0]
            if len(idx) == 0:
                break
            dec = diff / len(idx)  # diff negative -> dec negative
            # Ensure not go below 0
            dec = max(dec, -np.min(h[idx]))
            h[idx] += dec
        diff = target - h.sum()
        tries += 1
    # If still not exact due to rounding, do small random adjustments
    if abs(diff) > 1e-9:
        # random small adjustments
        for _ in range(100):
            i = rng.integers(0, len(h))
            add = min(diff, 1 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, h, mode='full')
    min_corr = np.min(corr[n-1:])  # positive lags
    return 1.0 - (2.0 / n) * min_corr

def random_binary_vector(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n//2, replace=False)
    h[idx] = 1.0
    return h

def enforce_sum(h, target, rng):
    # Implementation as described earlier.
    # Ensure sum is close to target.
    # For binary h, sum=target exactly if binary. For real h, adjust.
    # We'll reuse earlier function but adapt.
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    max_iter = 1000
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        # Choose indices where we can adjust
        if diff > 0:
            idx = np.where(h < 1 - 1e-12)[0]
            if idx.size == 0: break
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0: break
        i = rng.choice(idx)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # final correction if needed
    if abs(diff) > 1e-9:
        # distribute small amount uniformly
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def gradient_ascent(h0, max_iters, rng):
    h = h0.copy()
    n = len(h)
    step = 0.4
    for it in range(max_iters):
        # compute correlation h with 1-h to get worst lag
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        # Determine worst lag(s) where corr is max
        eps = 1e-12
        worst_idxs = np.where(np.abs(corr - max_corr) < eps)[0]
        # compute gradient contributions based on h*h min overlap? But we can approximate gradient from min_corr or max_corr.
        # Use min_corr approach as earlier.
        # Compute positive lags min value
        corr_hh = np.correlate(h, h, mode='full')
        pos_corr = corr_hh[n-1:]
        min_val = pos_corr.min()
        worst_lags = np.where(np.abs(pos_corr - min_val) < eps)[0]
        grads = np.zeros_like(h)
        for L in worst_lags:
            if L == 0:
                grads += 2 * h
            else:
                grads[:n-L] += h[L:]
                grads[L:] += h[:n-L]
        # Normalize grads to max absolute 1
        max_abs = np.max(np.abs(grads))
        if max_abs > 0:
            grads = grads / max_abs
        # Update
        h = h + step * grads
        # Clip and enforce sum
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, n/2.0, rng)
        # Decrease step
        step *= 0.9995
    return h

def enforce_sum(h, target, rng):
    """Clip to [0,1] and adjust to meet exact sum `target`."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if idx.size == 0:
            break
        i = rng.choice(idx)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-9:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def random_binary_vector(n, rng):
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def compute_c5(h):
    """Exact C5 using correlation with 1‑h."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def gradient_ascent(h0, max_iters, rng):
    """Maximise the minimal self‑overlap (equivalently minimise C5)."""
    h = h0.copy()
    n = h.size
    step = 0.4
    eps = 1e-12
    for it in range(max_iters):
        # correlation of h with itself
        corr = np.correlate(h, h, mode="full")
        pos = corr[n - 1 :]               # lags 0 … n‑1
        min_val = pos.min()
        worst = np.where(np.abs(pos - min_val) < eps)[0]

        grads = np.zeros_like(h)
        for L in worst:
            if L == 0:
                grads += 2.0 * h
            else:
                grads[: n - L] += h[L:]
                grads[L:] += h[: n - L]

        m = np.max(np.abs(grads))
        if m > 0:
            grads /= m
        h = h + step * grads
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, n / 2.0, rng)
        step *= 0.9995
    return h


def simulated_annealing_refine(h0, rng, max_iters, time_start, time_budget):
    """SA that targets the lag giving the current worst C5."""
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    n = h.size
    T0, Tf = 0.02, 1e-5
    eps = 1e-12
    for it in range(max_iters):
        if time.time() - time_start > time_budget:
            break
        T = T0 * (Tf / T0) ** (it / max_iters)
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        worst = np.where(np.abs(corr - max_corr) < eps)[0]
        lag_idx = rng.choice(worst)
        lag = lag_idx - (n - 1)

        # targeted perturbation
        if rng.random() < 0.7:
            if lag >= 0:
                i = rng.integers(0, n - lag)
                j = rng.integers(0, n)
            else:
                L = -lag
                i = rng.integers(L, n)
                j = rng.integers(0, n)
        else:
            i = rng.integers(0, n)
            j = rng.integers(0, n)

        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            new_h = h.copy()
        else:
            eps_move = (rng.random() - 0.5) * 2.0 * max_eps * 0.5
            new_h = h.copy()
            new_h[i] -= eps_move
            new_h[j] += eps_move

        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = enforce_sum(new_h, n / 2.0, rng)
        new_c = compute_c5(new_h)
        dE = new_c - cur_c
        if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            h, cur_c = new_h, new_c
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
    return best_h, best_c
