import time

def enforce_sum(h, target, rng, max_iter=2000):
    """Clip h to [0,1] and adjust to have sum == target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        free = (h > 1e-12) & (h < 1 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            per = diff / idx.size
            h[idx] += per
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum == n/2."""
    h = np.full(n, 0.5)

        j = rng.integers(0, n)

        h[j] += eps

    target = n / 2.0

def perturb_h(h, rng, scale=0.2):
    """Transfer a small ε from i to j (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def compute_corr(h):
    """Full correlation of h with (1-h) (raw sum, not scaled)."""
    n = h.shape[0]
    rev = (1.0 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return conv

def compute_c5(h):
    """Exact discrete C5 for sampled h."""
    n = h.shape[0]
    conv = compute_corr(h)
    return np.max(conv) * (2.0 / n)

        prod = h[i_start:i_end] * (1.0 - h[i_start - s:i_end - s])
        i_local = int(np.argmax(prod))
        i = i_start + i_local
        j = i - s
    eps_max = min(h[i], 1.0 - h[j])
    if eps_max <= 0.0:
        return h.copy()
    prod_val = h[i] * (1.0 - h[j])
    eps = min(eps_max, prod_val * 0.5) * (0.5 + 0.5 * rng.random())
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb_targeted(h_start, rng, max_iters=2000, step_scale=0.4):
    """Stochastic hill‑climb mixing random perturbations and targeted moves."""
    h = h_start.copy()
    best_c = compute_c5(h)
    for it in range(max_iters):
        if rng.random() < 0.2:
            new_h = perturb_h(h, rng, scale=step_scale)
        else:
            new_h = targeted_move(h, rng)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, h = c, new_h
        elif rng.random() < 0.01:
            h = new_h
    return h, best_c

# ---------- SLSQP with smooth surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    conv = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * conv
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny drift correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        n = x.size
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

# ---------- main run ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # initial vector
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # Phase 1: targeted hill‑climb on base resolution
    while time.time() - start_time < budget_s * 0.3:
        if rng.random() < 0.7:
            start_vec = perturb_h(best_h, rng, scale=0.4)
        else:
            start_vec = random_feasible_vector(n, rng)
        h_new, c_new = hill_climb_targeted(start_vec, rng, max_iters=2000, step_scale=0.4)
        if c_new < best_c:
            best_h, best_c = h_new, c_new

    # Phase 2: up‑sample and refine
    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.55:
        target_n = min(1080, max(n * 2, n + 180))
        old_x = (np.arange(n) + 0.5) * (2.0 / n)
        new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
        up_h = np.interp(new_x, old_x, best_h)
        up_h = enforce_sum(up_h, target_n / 2.0, rng)
        up_h, up_c = hill_climb_targeted(up_h, rng, max_iters=3000, step_scale=0.3)
        if up_c < best_c:
            best_h, best_c = up_h, up_c
            n = target_n

    # Phase 3: continuation SLSQP
    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.85:
        for beta in (30, 120, 500, 2000, 8000):
            if time.time() - start_time > budget_s * 0.85:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, max_iter=800)
                c_ref = compute_c5(h_ref)
                if c_ref < best_c:
                    best_h, best_c = h_ref, c_ref
                    n = best_h.size
            except Exception:
                continue

    # Phase 4: final brief local search
    if time.time() - start_time < budget_s - 2:
        h_fin, c_fin = hill_climb_targeted(best_h, rng, max_iters=2000, step_scale=0.2)
        if c_fin < best_c:
            best_h, best_c = h_fin, c_fin

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
