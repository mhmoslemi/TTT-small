Implement compute_c5 fast:

def compute_c5(h):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    # compute next power-of-two size for FFT
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)

def softmax_obj(x, beta):
    # compute conv as above
    rev = (1 - x)[::-1]
    n = x.size
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(x, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    bc = beta * conv
    max_bc = np.max(bc)
    # avoid overflow: subtract max
    return (max_bc + np.log(np.sum(np.exp(bc - max_bc)))) / beta

def slsqp_optimize(h0, beta, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    constraints = {"type": "eq", "fun": lambda x: np.sum(x) - CURRENT_TARGET}
    # Use minimize with method='SLSQP', options: maxiter,
    # ftol maybe small.
    res = minimize(lambda x: softmax_obj(x, CURRENT_BETA), h0, method='SLSQP',
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - np.sum(x)
    if abs(diff) > 1e-9:
        # correct sum using small adjustments like before
        rng = np.random.default_rng()
        # distribute diff across free indices
        free = (x > 1e-12) & (x < 1 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            per = diff / idx.size
            x[idx] += per
            x = np.clip(x, 0.0, 1.0)
        else:
            # distribute globally
            x += diff / n
            x = np.clip(x, 0.0, 1.0)
    return x

def binary_hill_climb(h, rng, max_iters=2000):
    # h is binary 0/1
    best_h = h.copy()
    best_c = compute_c5(best_h)
    n = h.size
    # precompute indices
    ones = np.flatnonzero(best_h == 1)
    zeros = np.flatnonzero(best_h == 0)
    for it in range(max_iters):
        # pick random swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        new_h = best_h.copy()
        new_h[i] = 0
        new_h[j] = 1

            zeros = np.flatnonzero(best_h == 0)
        # optionally early stop if no improvements else continue
        # Could also have a probability acceptance for worse solutions but not needed.
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# Global vars used by the smooth surrogate
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint for SLSQP: sum(x) == CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max correlation using log‑sum‑exp."""
    n = x.size
    rev = (1.0 - x)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(x, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    bc = CURRENT_BETA * conv
    max_bc = np.max(bc)
    # log‑sum‑exp in a numerically stable way
    return (max_bc + np.log(np.sum(np.exp(bc - max_bc)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for a sampled step function."""
    n = h.shape[0]
    rev = (1.0 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h so ∑h = target while preserving 0≤h≤1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-9:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            per = diff / idx.size
            h[idx] += per
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_continuous(h, rng, max_iters=2000, init_scale=0.5):
    """Simple stochastic hill‑climb on the exact C5."""
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    scale = init_scale
    decay = (0.01) ** (1.0 / max_iters)
    for _ in range(max_iters):
        new_h = perturb_h(cur_h, rng, scale=scale)
        c = compute_c5(new_h)
        if c < cur_c:
            cur_h, cur_c = new_h, c
        scale = max(scale * decay, 1e-6)
    return cur_h, cur_c


def random_binary_vector(n, rng):
    """Generate a binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h


def binary_from_continuous(h):
    """Turn a continuous vector into a binary one with half the entries =1."""
    n = h.size
    k = n // 2
    idx = np.argpartition(-h, k)[:k]
    bin_h = np.zeros(n, dtype=float)
    bin_h[idx] = 1.0
    return bin_h


def binary_hill_climb(h, rng, max_iters=4000, init_temp=0.02, final_temp=0.0):
    """Hill‑climb on binary vectors with a tiny annealing schedule."""
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    n = h.size
    ones = [i for i in range(n) if cur_h[i] == 1.0]
    zeros = [i for i in range(n) if cur_h[i] == 0.0]
    for it in range(max_iters):
        t = it / max_iters
        temp = init_temp * (1.0 - t) + final_temp * t
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # propose swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        c = compute_c5(cur_h)
        delta = c - cur_c
        accept = False
        if delta < 0:
            accept = True
        elif temp > 0 and rng.random() < np.exp(-delta / temp):
            accept = True
        if accept:
            cur_c = c
            # update index lists
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
        else:
            # revert
            cur_h[i] = 1.0
            cur_h[j] = 0.0
    return cur_h, cur_c


def slsqp_optimize(h0, beta, max_iter=500):
    """SLSQP refinement of the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # correct any tiny drift in the sum
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x
