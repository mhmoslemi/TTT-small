def subgradient_descent(h, rng, max_iters=500, init_eta=0.1, decay=0.99):
    n = len(h)
    target = n/2
    eta = init_eta
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iters):
        grad = worst_gradient(h) # gradient of O_k*
        # take step
        y = h - eta * grad
        # project onto [0,1] with sum target
        h = project_onto_box_sum(y, target)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        eta *= decay
        # optionally early break if no improvement for many iterations
    return best_h, best_c
```

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Utility functions ----------

def autocorr_full(h):
    """Linear autocorrelation of h with itself (full length)."""
    n = h.size
    if n >= 256:
        size = 2 * n
        H = np.fft.rfft(h, n=size)
        corr = np.fft.irfft(H * np.conj(H), n=size)[: 2 * n - 1]
    else:
        corr = np.correlate(h, h, mode="full")
    return corr


def compute_c5(h):
    """Exact discrete C5 for a step function on [0,2]."""
    n = h.size
    target = n / 2.0
    dx = 2.0 / n
    corr = autocorr_full(h)
    min_corr = np.min(corr)  # corresponds to the worst overlap shift
    Omax = target - min_corr
    return Omax * dx


def project_onto_box_sum(y, target, tol=1e-12):
    """Projection of y onto {0<=x<=1, sum x = target}."""
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        s = np.sum(np.clip(y + mid, 0.0, 1.0))
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(y + (low + high) * 0.5, 0.0, 1.0)


def enforce_sum(h):
    """Clamp to [0,1] and enforce Σh = len(h)/2 exactly."""
    n = h.size
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    h = project_onto_box_sum(h, target)
    return h


def random_feasible_vector(n, rng):
    """Random vector in [0,1]^n with Σ = n/2."""
    h = rng.random(n)
    h *= (n / 2.0) / h.sum()
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h)
    return h


# ---------- Gradient & hill‑climb ----------

def worst_gradient(h):
    """Gradient of the worst (max) overlap O_k = ∫h(1−h(·+k))."""
    n = h.size
    corr = autocorr_full(h)
    idx = np.argmin(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h, grad, rng, scale=0.5):
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h, rng, max_iters=1400, step_scale=0.7):
    """Hill‑climb using gradient‑directed transfers."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

def subgradient_descent(h, rng, max_iters=300, init_eta=0.3, decay=0.99):
    """Projected sub‑gradient method on the max‑overlap objective."""

# ---------- Soft‑max SLSQP refinement ----------

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=800):
    """SLSQP optimisation of the smooth max (soft‑max) objective."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,

    )
    x = np.clip(res.x, 0.0, 1.0)

    x = enforce_sum(x)
    return x

                c_up = compute_c5(h_up)
                if c_up < best_c:
                    best_h, best_c, best_n = h_up.copy(), c_up, n_up

            # sub‑gradient tidy‑up
            h_up, c_up = subgradient_descent(h_up, rng, max_iters=150, init_eta=0.2, decay=0.97)
            if c_up < best_c:
                best_h, best_c, best_n = h_up.copy(), c_up, n_up

    # ---------- Final stochastic tweaking ----------
    while time.time() - start < budget_s * 0.999 and best_h is not None:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # ---------- Sanity and return ----------
    best_h = enforce_sum(best_h)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
