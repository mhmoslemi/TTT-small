    i = rng.integers(n)
    j = rng.integers(n-1)
    if j >= i:
        j += 1
    # compute delta sign and max
    delta_up_i = 1.0 - h[i]

def simulated_annealing(h_initial, rng, deadline, init_temp=0.02, decay=0.9995, step_max=0.05):
    h = h_initial.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    temp = init_temp
    it = 0
    last_improve_iter = 0
    while time.time() < deadline:
        # propose move
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        # allowed adjustments
        delta_up_i = 1.0 - h[i]
        delta_down_i = h[i]
        delta_up_j = 1.0 - h[j]
        delta_down_j = h[j]
        max_pos = min(delta_up_i, delta_down_j)
        max_neg = min(delta_down_i, delta_up_j)
        if max_pos <= 0 and max_neg <= 0:
            # improbable but just continue
            continue
        # choose sign
        if max_pos > 0 and max_neg > 0:
            sign = 1 if rng.random() < 0.5 else -1
        elif max_pos > 0:
            sign = 1
        else:
            sign = -1
        max_abs = max_pos if sign == 1 else max_neg
        # step magnitude; scale with temperature
        delta = rng.random() * min(step_max * (temp / init_temp), max_abs)
        if sign == -1:
            delta = -delta

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a discrete step function."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force sum(h) = len(h)/2 exactly by moving mass between random entries."""
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    attempts = 0
    # simple random redistribution; stops when diff is negligible
    while abs(diff) > 1e-12 and attempts < n * 2000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) > 1e-15:
            h[i] += add
            diff = target - h.sum()
        attempts += 1
    return h


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.02, max_iter: int = 20) -> np.ndarray:
    """Projected gradient descent against the current worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]      # + (1 - h[i+k])
            g[k:] -= h[:upper]           # - h[i]
        step_sz = step * (0.8 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        # fix the integral exactly
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                if abs(add) > 1e-15:
                    h[i] += add
                    diff = target - h.sum()
                    if abs(diff) < 1e-12:
                        break
    return h


def simulated_annealing(h_start: np.ndarray, rng: np.random.Generator,
                        deadline: float, init_temp: float = 0.02,
                        decay: float = 0.9995, step_max: float = 0.05) -> tuple:
    """Pairwise‐swap annealing that preserves the sum exactly."""
    h = h_start.copy()
    n = h.size
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    temp = init_temp
    it = 0
    last_improve = 0

    while time.time() < deadline:
        # pick two distinct indices
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # allowed transfer from j→i (positive) or i→j (negative)
        max_pos = min(1.0 - h[i], h[j])      # increase i, decrease j
        max_neg = min(h[i], 1.0 - h[j])      # decrease i, increase j
        if max_pos <= 0 and max_neg <= 0:
            continue

        # choose direction
        if max_pos > 0 and max_neg > 0:
            sign = 1 if rng.random() < 0.5 else -1
        elif max_pos > 0:
            sign = 1
        else:
            sign = -1
        max_abs = max_pos if sign == 1 else max_neg

        # step size adapts with temperature
        cur_step = max(step_max * (temp / init_temp), 1e-6)
        delta = rng.random() * min(cur_step, max_abs)
        if sign == -1:
            delta = -delta

        # apply move
        h[i] += delta
        h[j] -= delta
        new_c = compute_c5(h)

        # acceptance test
        accept = False
        if new_c < cur_c:
            accept = True
        else:
            if temp > 1e-12:
                prob = np.exp(-(new_c - cur_c) / temp)
                if rng.random() < prob:
                    accept = True
        if accept:
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
                last_improve = it
        else:
            # revert
            h[i] -= delta
            h[j] += delta

        # temperature update
        temp = max(temp * decay, 1e-12)
        it += 1

        # occasional diversification if stuck
        if it - last_improve > 20000:
            # small random redistribution on a few indices
            num = max(1, n // 100)
            idxs = rng.choice(n, size=2 * num, replace=False)
            for p in range(num):
                a, b = idxs[2 * p], idxs[2 * p + 1]
                max_swap = min(h[a], 1.0 - h[b])
                if max_swap > 0:
                    d = rng.random() * max_swap
                    h[a] -= d
                    h[b] += d
            cur_c = compute_c5(h)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
                last_improve = it

    return best_h, best_c
