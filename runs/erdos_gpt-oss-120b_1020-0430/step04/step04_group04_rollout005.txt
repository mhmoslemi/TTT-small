def compute_corr_raw(h):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return conv

def enforce_sum(h, target, rng, max_iter=5000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        free = (h > 1e-12) & (h < 1 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            per = diff / idx.size
            h[idx] += per
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.5
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

# ------------------- correlation helpers ------------------- #
def compute_corr_raw(h):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return conv

def compute_c5(h):
    n = h.shape[0]
    conv = compute_corr_raw(h)
    return conv.max() * (2.0 / n)

# ------------------- gradient descent ------------------- #
def gradient_at_max_shift(h, dx):
    n = h.size
    corr = compute_corr_raw(h)
    max_val = corr.max()
    idxs = np.where(corr >= max_val - 1e-12)[0]
    grad = np.zeros_like(h, dtype=float)
    count = 0
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            forward = np.empty_like(h)
            forward[:n - k] = h[k:]
            forward[n - k:] = 0.0
            backward = np.empty_like(h)
            backward[k:] = h[:n - k]
            backward[:k] = 0.0
        else:
            kk = -k
            forward = np.empty_like(h)
            forward[kk:] = h[:n - kk]
            forward[:kk] = 0.0
            backward = np.empty_like(h)
            backward[:n - kk] = h[kk:]
            backward[n - kk:] = 0.0
        grad += (1.0 - forward) - backward
        count += 1
    if count:
        grad /= count
    return grad * dx

def gradient_descent(h, target, rng, max_iters=150, init_lr=0.3, decay=0.96):
    n = h.size
    dx = 2.0 / n
    for i in range(max_iters):
        lr = init_lr * (decay ** i)
        grad = gradient_at_max_shift(h, dx)
        if np.all(np.abs(grad) < 1e-12):
            break
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = enforce_sum(h_new, target, rng)
        if compute_c5(h_new) < compute_c5(h):
            h = h_new
    return h

# ------------------- local heuristics ------------------- #
def hill_climb_multi(h_start, rng, max_iters=2000, init_step=0.5, n_trials=6):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
                improved = True
                break
        if not improved and rng.random() < 0.02:
            new_h = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return h, best_c

def simulated_annealing(h0, rng, max_iter, init_temp, final_temp,
                        init_scale=0.5, final_scale=0.05):
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    cur_h, cur_c = best_h.copy(), best_c
    for i in range(max_iter):
        t = i / max_iter
        temp = init_temp * (final_temp / init_temp) ** t if init_temp > 0 else 0.0
        scale = init_scale * (final_scale / init_scale) ** t if init_scale > 0 else 0.0
        new_h = perturb_h(cur_h, rng, scale=scale)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or (temp > 0 and rng.random() < np.exp(-delta / temp)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
    return best_h, best_c

# ------------------- SLSQP surrogate ------------------- #
CURRENT_TARGET = None
CURRENT_BETA = None
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET
def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

# ------------------- main driver ------------------- #
def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # ---- initial base vector ---- #
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = enforce_sum(np.clip(h0, 0.0, 1.0), n / 2.0, rng)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    candidates = [h0]
    xs = (np.arange(n) + 0.5) * (2.0 / n)

    # analytic seeds
    tri = np.where(xs <= 1, xs, 2 - xs)
    candidates.append(enforce_sum(tri, n / 2.0, rng))
    cosine = (1 + np.cos(np.pi * (xs - 1))) / 2.0
    candidates.append(enforce_sum(cosine, n / 2.0, rng))
    lin = xs / 2.0
    candidates.append(enforce_sum(lin, n / 2.0, rng))
    candidates.append(enforce_sum(1 - lin, n / 2.0, rng))

    for _ in range(6):
        candidates.append(random_feasible_vector(n, rng))

    best_h = None
    best_c = np.inf
    for cand in candidates:
        c = compute_c5(cand)
        if c < best_c:
            best_c, best_h = c, cand.copy()
    target_sum = n / 2.0

    def try_update(h):
        nonlocal best_h, best_c, target_sum
        c = compute_c5(h)
        if c < best_c - 1e-12:
            best_c, best_h = c, h.copy()
            target_sum = best_h.size / 2.0
            return True
        return False

    # ---- main optimisation loop (coarse) ---- #
    while time.time() - start < budget_s * 0.55:
        start_vec = perturb_h(best_h, rng, scale=0.4) if rng.random() < 0.6 else random_feasible_vector(n, rng)

        h_hc, c_hc = hill_climb_multi(start_vec, rng, max_iters=2500, init_step=0.6, n_trials=6)
        try_update(h_hc)

        h_sa, _ = simulated_annealing(best_h, rng,
                                       max_iter=8000,
                                       init_temp=1e-2,
                                       final_temp=1e-5,
                                       init_scale=0.5,
                                       final_scale=0.02)
        try_update(h_sa)

        h_gd = gradient_descent(best_h, target_sum, rng,
                                max_iters=150, init_lr=0.3, decay=0.96)
        try_update(h_gd)

        for beta in (100, 500, 2000, 5000):
            if time.time() - start > budget_s * 0.55:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, max_iter=400)
                try_update(h_ref)
            except Exception:
                pass

    # ---- optional up‑sampling (fine) ---- #
    elapsed = time.time() - start
    if budget_s - elapsed > budget_s * 0.15:
        target_n = min(960, max(n * 2, n + 240))
        old_x = (np.arange(n) + 0.5) * (2.0 / n)
        new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
        up_h = np.interp(new_x, old_x, best_h)
        up_h = enforce_sum(up_h, target_n / 2.0, rng)
        n = target_n
        target_sum = n / 2.0

        h_hc, _ = hill_climb_multi(up_h, rng, max_iters=2000, init_step=0.5, n_trials=6)
        try_update(h_hc)

        h_sa, _ = simulated_annealing(up_h, rng,
                                      max_iter=6000,
                                      init_temp=5e-3,
                                      final_temp=1e-5,
                                      init_scale=0.4,
                                      final_scale=0.02)
        try_update(h_sa)

        h_gd = gradient_descent(up_h, target_sum, rng,
                                max_iters=120, init_lr=0.25, decay=0.95)
        try_update(h_gd)

        for beta in (200, 800, 2000, 5000):
            if time.time() - start > budget_s:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, max_iter=300)
                try_update(h_ref)
            except Exception:
                pass

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
