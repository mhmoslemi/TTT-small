import time
import numpy as np

# ---------- basic helpers ----------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    """Project x onto [lo,hi] with exact sum = target using bisection on a shift."""
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    # tiny correction in case of rounding errors
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y


def random_feasible_vector(n, rng):
    x = rng.random(n)
    return project_box_sum(x, n / 2.0)

        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def targeted_move(h, rng, k, scale=0.2):
    n = h.size
    if k <= 0 or k >= n:
        return perturb_h(h, rng, scale)
    low_mask = (h[: n - k] < 0.5) & (h[k:] < 0.5)
    high_mask = (h[: n - k] > 0.5) & (h[k:] > 0.5)
    if not low_mask.any() or not high_mask.any():
        return perturb_h(h, rng, scale)
    i_low = rng.choice(np.flatnonzero(low_mask))
    i_high = rng.choice(np.flatnonzero(high_mask))
    max_eps = min(
        0.5 - h[i_low],
        0.5 - h[i_low + k],
        h[i_high] - 0.5,
        h[i_high + k] - 0.5,
    )
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i_low] += eps
    new_h[i_low + k] += eps
    new_h[i_high] -= eps
    new_h[i_high + k] -= eps
    return new_h


def hill_climb_targeted(h_start, rng, max_iters=800, init_step=0.5, n_trials=4, p_target=0.25):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    n = h.shape[0]

    for it in range(max_iters):
        _, idx = compute_c5_and_idx(h)
        k_star = idx - (n - 1)
        improved = False
        for _ in range(n_trials):
            if rng.random() < p_target:
                cand = targeted_move(h, rng, k_star, scale=step)
            else:
                cand = perturb_h(h, rng, scale=step)
            c = compute_c5(cand)
            if c < best_c - 1e-12:
                best_c, h = c, cand
                improved = True
                break
        if not improved and rng.random() < 0.02:
            cand = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(cand)
            if c < best_c:
                best_c, h = c, cand
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return h, best_c


# ---------- SLSQP refinement ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)


# ---------- sub‑gradient descent ----------
def subgradient_step(h, k):
    N = h.size
    grad = np.zeros(N)
    if k >= 0:
        grad[: N - k] += 1.0 - h[k:]
        grad[k:] -= h[: N - k]
    else:
        p = -k
        grad[p:] += 1.0 - h[: N - p]
        grad[: N - p] -= h[p:]
    return grad


def subgradient_descent(h0, rng, max_iters=400, init_step=0.4, decay=0.9):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = init_step
    N = h.size
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        k_star = idx - (N - 1)
        grad = subgradient_step(h, k_star)
        new_h = project_box_sum(h - step * grad, N / 2.0)
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c, best_h = new_c, new_h.copy()
        h = new_h
        if (it + 1) % decay_interval == 0:
            step *= decay
    return best_h, best_c


# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # --- initialise base resolution ---
    base_n = 360
    if "initial_h_values" in globals():
        base_n = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = random_feasible_vector(base_n, rng)

    # we aim for a higher resolution (double) if time permits
    target_n = base_n * 2
    if target_n > 1440:
        target_n = 1440
    # upsample (linear) to target_n
    old_x = (np.arange(base_n) + 0.5) * (2.0 / base_n)
    new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
    h0 = np.interp(new_x, old_x, h0)
    h0 = project_box_sum(np.clip(h0, 0.0, 1.0), target_n / 2.0)

    N = h0.size
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # a few deterministic seeds (sinusoids) for diversity
    xs = (np.arange(N) + 0.5) * (2.0 / N)
    for f in (1, 2, 3):
        cand = 0.5 + 0.5 * np.sin(np.pi * f * xs)
        cand = project_box_sum(np.clip(cand, 0.0, 1.0), N / 2.0)
        c = compute_c5(cand)
        if c < best_c:
            best_c, best_h = c, cand.copy()

    beta_list = [20, 80, 200, 500, 1000, 2000, 5000, 10000, 20000]

    # main optimisation loop
    while time.time() - start_time < budget_s * 0.78:
        # choose start point
        if rng.random() < 0.6:
            start_vec = project_box_sum(best_h + rng.normal(scale=0.03, size=N), N / 2.0)
        else:
            start_vec = random_feasible_vector(N, rng)

        # sub‑gradient descent
        h_sg, c_sg = subgradient_descent(start_vec, rng, max_iters=500,
                                         init_step=0.6, decay=0.92)
        if c_sg < best_c:
            best_c, best_h = c_sg, h_sg.copy()

            # a brief targeted hill‑climb
            h_hc, c_hc = hill_climb_targeted(best_h, rng, max_iters=400,
                                             init_step=0.3, n_trials=3, p_target=0.35)
            if c_hc < best_c:
                best_c, best_h = c_hc, h_hc.copy()

            # SLSQP refinement with increasing β
            for beta in beta_list:
                if time.time() - start_time > budget_s * 0.78:
                    break
                try:
                    h_ref = slsqp_optimize(best_h, beta, max_iter=500)
                except Exception:
                    continue
                c_ref = compute_c5(h_ref)
                if c_ref < best_c:
                    best_c, best_h = c_ref, h_ref.copy()

    # --- optional fine up‑sampling if time remains ---
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > budget_s * 0.12:
        finer_n = min(1440, N * 2)
        if finer_n != N:
            old_x = (np.arange(N) + 0.5) * (2.0 / N)
            new_x = (np.arange(finer_n) + 0.5) * (2.0 / finer_n)
            h_up = np.interp(new_x, old_x, best_h)
            h_up = project_box_sum(h_up, finer_n / 2.0)

            # sub‑gradient on finer grid
            h_sg, c_sg = subgradient_descent(h_up, rng, max_iters=800,
                                             init_step=0.5, decay=0.93)
            if c_sg < best_c:
                best_c, best_h = c_sg, h_sg.copy()
                N = finer_n

            # quick hill‑climb
            h_hc, c_hc = hill_climb_targeted(best_h, rng, max_iters=600,
                                             init_step=0.3, n_trials=3, p_target=0.3)
            if c_hc < best_c:
                best_c, best_h = c_hc, h_hc.copy()
                N = finer_n

            # final SLSQP passes
            for beta in beta_list:
                if time.time() - start_time > budget_s:
                    break
                try:
                    h_ref = slsqp_optimize(best_h, beta, max_iter=500)
                except Exception:
                    continue
                c_ref = compute_c5(h_ref)
                if c_ref < best_c:
                    best_c, best_h = c_ref, h_ref.copy()
                    N = finer_n

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
