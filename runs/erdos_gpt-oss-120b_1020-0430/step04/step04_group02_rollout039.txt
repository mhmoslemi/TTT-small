def smooth_obj_and_grad(h, beta):
    sum_h = h.sum()
    q = np.correlate(h, h, mode='full')
    corr = sum_h - q
    m = np.max(beta * corr)
    exp_val = np.exp(beta * corr - m)
    Z = np.sum(exp_val)
    obj = (np.log(Z) + m) / beta
    w_arr = exp_val / Z
    # compute gradient
    n = h.size
    conv_valid = np.convolve(w_arr, np.ones(n), mode='valid')
    A = conv_valid[::-1]
    B = np.convolve(w_arr, h, mode='full')[ (n-1) : (n-1)+n ]
    w_rev = w_arr[::-1]
    C = np.convolve(w_rev, h, mode='full')[ (n-1) : (n-1)+n ]
    grad = A - B - C
    return obj, grad
```

def smooth_obj_and_grad(h, beta, L):
    n = h.size
    # compute q = correlation h with h
    H = np.fft.rfft(h, n=L)
    q_full = np.fft.irfft(H * np.conj(H), n=L)[:2*n-1]
    sum_h = h.sum()
    corr = sum_h - q_full
    # compute w
    bc = beta * corr
    max_bc = np.max(bc)
    exp_bc = np.exp(bc - max_bc)
    Z = np.sum(exp_bc)
    w = exp_bc / Z
    # compute gradient components via FFT
    # A via convolution with ones
    ones = np.ones(n)
    W = np.fft.rfft(w, n=L)
    O = np.fft.rfft(ones, n=L)
    conv_full_ones = np.fft.irfft(W * O, n=L)[:3*n-2]  # full length
    conv_valid = conv_full_ones[n-1: n-1 + n]  # length n
    A = conv_valid[::-1]  # reverse
    # B = w * h convolution
    conv_full_wh = np.fft.irfft(W * H, n=L)[:3*n-2]
    B = conv_full_wh[n-1: n-1 + n]
    # C = w_rev * h convolution
    w_rev = w[::-1]
    W_rev = np.fft.rfft(w_rev, n=L)
    conv_full_wrev_h = np.fft.irfft(W_rev * H, n=L)[:3*n-2]
    C = conv_full_wrev_h[n-1: n-1 + n]
    grad = A - B - C
    obj = (np.log(Z) + max_bc) / beta
    return obj, grad
```

def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y

    # ----- gradient‑descent refinements -----
    for cand in candidates:
        if time.time() - start_time > budget_s * 0.78:
            break
        h_opt, c_opt = grad_descent_start(cand, rng, start_time, budget_s)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()

    # ----- optional up‑sampling -----
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > budget_s * 0.12:
        target_n = min(720, max(540, n * 2))
        if target_n != n:
            old_x = (np.arange(n) + 0.5) * (2.0 / n)
            new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
            h_up = np.interp(new_x, old_x, best_h)
            h_up = project_box_sum(h_up, target_n / 2.0)

            h_opt, c_opt = grad_descent_start(h_up, rng, start_time, budget_s)
            if c_opt < best_c:
                best_c, best_h, n = c_opt, h_opt.copy(), target_n

    # ----- final local search -----
    if time.time() - start_time < budget_s * 0.99:
        h_opt, c_opt = final_hill_climb(best_h, rng, start_time, budget_s)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()

    # ----- clean‑up and return -----
    best_h = project_box_sum(np.clip(best_h, 0.0, 1.0), best_h.size / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
