def enforce_sum_and_bounds(x, target):
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    # tolerances
    it = 0
    while abs(diff) > 1e-12 and it < 1000:
        if diff > 0:
            idx = np.where(x < 1.0)[0]
            if idx.size == 0:
                break
            # distribute diff equally among underfull
            add = diff / idx.size
            # cap at 1
            inc = np.minimum(add, 1.0 - x[idx])
            x[idx] += inc
        else:
            idx = np.where(x > 0.0)[0]
            if idx.size == 0:
                break
            sub = diff / idx.size
            dec = np.minimum(-sub, x[idx])
            x[idx] -= dec
        diff = target - x.sum()
        it += 1
    # final clip
    x = np.clip(x, 0.0, 1.0)
    return x

def gradient_descent_h(h0, target, time_left, beta_schedule, max_iter_per_beta=200):
    h = np.copy(h0)
    n = h.size
    start = time.time()
    for beta in beta_schedule:
        for it in range(max_iter_per_beta):
            now = time.time()
            if now - start > time_left:
                return h
            f = compute_f(h)  # shape m
            # compute softmax objective for potential early stopping maybe
            max_f = np.max(f)
            # compute weights
            exp_terms = np.exp(beta * (f - max_f))
            w = exp_terms / np.sum(exp_terms)
            # compute gradients for each shift
            grads = compute_gradients(h)  # shape (m,n)
            grad_obj = np.dot(w, grads)  # shape (n,)
            # step size selection: simple diminishing steps
            # step scale = 0.5 * (1 / np.sqrt(it+1))
            step_size = 0.5 / np.sqrt(it + 1)
            # attempt line search
            for ls in range(10):
                h_candidate = enforce_sum_and_bounds(h - step_size * grad_obj, target)
                f_candidate = compute_f(h_candidate)
                # compute softmax objective value for candidate and current
                max_f_cand = np.max(f_candidate)
                obj_cand = (max_f_cand + np.log(np.sum(np.exp(beta * (f_candidate - max_f_cand))))) / beta
                max_f_curr = max_f
                obj_curr = (max_f_curr + np.log(np.sum(np.exp(beta * (f - max_f_curr))))) / beta
                if obj_cand < obj_curr - 1e-12:
                    h = h_candidate
                    f = f_candidate
                    max_f = max_f_cand
                    break
                else:
                    step_size *= 0.5
            # else if not improved, break out to next beta
            else:
                # No improvement after line search; break out of iteration loop
                break
    return h

def resize_h(h, new_n):
    n = h.size
    if new_n == n:
        return h.copy()
    # positions are midpoints of intervals: xi = (i+0.5)*2/n
    xs_old = (np.arange(n) + 0.5) * (2.0 / n)
    xs_new = (np.arange(new_n) + 0.5) * (2.0 / new_n)

def weighted_gradient(h, w):
    n = h.size
    grad = np.zeros(n)
    a = 1.0 - h
    # positive shifts including zero
    w_pos = w[n-1:]  # length n
    for s in range(n):
        ws = w_pos[s]
        if ws == 0.0:
            continue
        # add ws * a[i+s] to grad[i] for i up to n-s-1
        grad[:n - s] += ws * a[s:]
    # negative shifts
    w_neg = w[:n-1]  # length n-1, correspond to s = -(n-1)..-1
    # t = 1..n-1
    for t in range(1, n):
        ws = w_neg[n-1 - t]
        if ws == 0.0:
            continue
        grad[t:] -= ws * h[:n - t]
    return grad

def weighted_gradient(h, w):
    n = h.size
    grad = np.zeros(n)
    a = 1.0 - h
    # positive shifts s = 0..n-1
    w_pos = w[n-1:]  # length n

def enforce_sum_and_bounds(x, target):

    """Array of all shift overlaps f_s = ∫h(x)(1-h(x+s))dx (discrete)."""

def weighted_gradient(h, w):
    """Gradient of soft‑max objective: sum_s w_s * grad f_s."""
    n = h.size
    grad = np.zeros(n)
    a = 1.0 - h
    # positive shifts (including zero)
    w_pos = w[n - 1 :]  # length n, s = 0..n-1
    for s in range(n):
        ws = w_pos[s]
        if ws == 0.0:
            continue
        grad[: n - s] += ws * a[s:]
    # negative shifts s = -t, t=1..n-1
    w_neg = w[: n - 1]  # corresponds to s = -(n-1)..-1
    for t in range(1, n):
        ws = w_neg[n - 1 - t]
        if ws == 0.0:
            continue
        grad[t:] -= ws * h[: n - t]
    return grad

                return h

        h_cur = resize_h(h_init, cur_n)
    else:
        h_cur = h_init.copy()

    best_h = h_cur.copy()
    best_c = compute_c5(best_h)

    total_levels = len(levels)

    # ----- optimisation per level -----
    for lvl_idx, lvl_n in enumerate(levels):
        # upscale/downscale to current level
        if best_h.size != lvl_n:
            best_h = resize_h(best_h, lvl_n)

        target_sum = lvl_n / 2.0
        elapsed = time.time() - start_time
        time_left = budget_s - elapsed
        if time_left <= 0:
            break

        remaining_levels = total_levels - lvl_idx
        level_budget = time_left / remaining_levels

        level_start = time.time()
        beta_schedule = [5, 20, 100, 500, 2000]

        # repeat restarts while we have time
        while time.time() - level_start < 0.9 * level_budget:
            # starting point
            if rng.random() < 0.4:
                start_h = perturb_random(best_h, rng, scale=0.3)
            else:
                start_h = random_feasible_vector(lvl_n, rng)

            # gradient descent (time‑limited)
            gd_time = 0.7 * (level_budget - (time.time() - level_start))
            if gd_time <= 0:
                break
            h_opt = gradient_descent(
                start_h,
                target_sum,
                beta_schedule,
                max_iter_per_beta=5000,
                time_limit=gd_time,
            )

            # deterministic greedy polishing
            h_opt = greedy_optimize(h_opt, max_steps=800)

            c_opt = compute_c5(h_opt)
            if c_opt < best_c:
                best_h, best_c = h_opt, c_opt

        # final greedy pass on the level's best
        best_h = greedy_optimize(best_h, max_steps=1200)
        best_c = compute_c5(best_h)

        if time.time() - start_time > 0.95 * budget_s:
            break

    # ----- tiny random search in remaining budget -----
    while time.time() - start_time < 0.99 * budget_s:
        cand = perturb_random(best_h, rng, scale=0.02)
        c_cand = compute_c5(cand)
        if c_cand < best_c:
            best_h, best_c = cand, c_cand

    # final cleanup
    best_h = enforce_sum_and_bounds(np.clip(best_h, 0.0, 1.0), best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
