def enforce_sum(h, target):
    h = np.clip(h, 0, 1)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # distribute diff proportionally to free capacity
    # capacity for increasing: 1 - h
    inc_cap = 1 - h
    dec_cap = h
    if diff > 0:
        # need to add diff, increase where inc_cap > 0
        total_cap = inc_cap.sum()
        if total_cap > 0:
            h += diff * inc_cap / total_cap
    else:
        # need to subtract diff (diff negative)
        total_cap = dec_cap.sum()
        if total_cap > 0:
            h += diff * dec_cap / total_cap
    # clipping may violate exact sum slightly, final adjust by small uniform
    h = np.clip(h, 0, 1)
    # final correction
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # adjust uniformly across all indices that are not at bound
        free = np.where((h > 0) & (h < 1))[0]
        if len(free) > 0:
            h[free] += diff / len(free)
            h = np.clip(h, 0, 1)
    return h
```

def descent_optimize(h_start, rng, time_start, time_budget):
    n = len(h_start)
    target = n/2.0
    best_h = h_start.copy()
    best_c = compute_c5(best_h)
    h = best_h.copy()
    cur_c = best_c
    cur_corr = np.correlate(h, 1 - h, mode='full')
    step_factor = 0.5
    no_improve = 0
    while time.time() - time_start < time_budget:
        lag = np.argmax(cur_corr) - (n - 1)
        # compute overlapping indices
        if lag >= 0:
            i = np.arange(n - lag)
            j = i + lag
        else:
            i = np.arange(-lag, n)
            j = i + lag
        contrib = h[i] * (1.0 - h[j])
        max_idx_local = np.argmax(contrib)
        src = i[max_idx_local]
        dst = j[max_idx_local]
        max_eps = min(h[src], 1.0 - h[dst])
        if max_eps <= 1e-12:
            # fallback random perturb
            new_h = perturb_h(h, rng, scale=0.1)
            new_c = compute_c5(new_h)
            if new_c < cur_c:
                h, cur_c = new_h, new_c
                cur_corr = np.correlate(h, 1 - h, mode='full')
                if new_c < best_c:
                    best_c, best_h = new_c, new_h.copy()
                no_improve = 0
                continue
            else:
                # random small move leads nowhere, break or continue
                no_improve += 1
                if no_improve > 200:
                    # random jitter
                    h = perturb_h(best_h, rng, scale=0.2)
                    cur_c = compute_c5(h)
                    cur_corr = np.correlate(h, 1 - h, mode='full')
                    no_improve = 0
                continue
        eps = max_eps * step_factor
        new_h = h.copy()
        new_h[src] -= eps
        new_h[dst] += eps
        new_c = compute_c5(new_h)
        if new_c < cur_c:
            h, cur_c = new_h, new_c
            cur_corr = np.correlate(h, 1 - h, mode='full')
            no_improve = 0
            # update step factor maybe increase when success
            step_factor = min(step_factor * 1.02, 0.8)
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
        else:
            # no improvement: reduce step_factor

import time
import numpy as np
from scipy.optimize import minimize

# ---------- globals for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k correlation."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def enforce_sum(h, target, rng, max_iter=5000):
    """Clip to [0,1] and adjust to meet exact sum `target`."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-13:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-13:
        # final uniform correction on a free index
        free = np.where((h > 0) & (h < 1))[0]
        if free.size:
            h[free[0]] += diff
            h = np.clip(h, 0.0, 1.0)
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum=n/2."""
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, target, rng)
    return h


def compute_c5(h):
    """Return C5 for a given vector."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def compute_c5_corr(h):
    """Return (C5, full correlation array)."""
    corr = np.correlate(h, 1 - h, mode="full")
    c = np.max(corr) * (2.0 / h.size)
    return c, corr


def perturb_h(h, rng, scale=0.2):
    """Transfer a random ε from i to j (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        # try opposite direction
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0:
            return h.copy()
        eps = max_eps * scale * rng.random()
        new_h = h.copy()
        new_h[j] -= eps
        new_h[i] += eps
        return new_h
    eps = max_eps * scale * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def descent_optimize(h_start, rng, start_time, time_budget):
    """Iterative greedy descent targeting the worst lag."""
    n = len(h_start)
    target = n / 2.0
    best_h = h_start.copy()
    best_c = compute_c5(best_h)
    h = best_h.copy()
    cur_c, cur_corr = compute_c5_corr(h)
    step_factor = 0.5
    no_improve = 0

    while time.time() - start_time < time_budget:
        lag = np.argmax(cur_corr) - (n - 1)  # convert index to signed lag
        if lag >= 0:
            i = np.arange(n - lag)
            j = i + lag
        else:
            i = np.arange(-lag, n)
            j = i + lag
        contrib = h[i] * (1.0 - h[j])
        if contrib.size == 0:
            # degenerate case, random perturb
            new_h = perturb_h(h, rng, scale=0.1)
            new_c = compute_c5(new_h)
            if new_c < cur_c:
                h, cur_c, cur_corr = new_h, new_c, np.correlate(new_h, 1 - new_h, mode="full")
                if new_c < best_c:
                    best_c, best_h = new_c, new_h.copy()
            continue

        worst_local = np.argmax(contrib)
        src = i[worst_local]
        dst = j[worst_local]

        max_eps = min(h[src], 1.0 - h[dst])
        if max_eps <= 1e-12:
            # fallback random perturb
            new_h = perturb_h(h, rng, scale=0.1)
            new_c = compute_c5(new_h)
            if new_c < cur_c:
                h, cur_c, cur_corr = new_h, new_c, np.correlate(new_h, 1 - new_h, mode="full")
                if new_c < best_c:
                    best_c, best_h = new_c, new_h.copy()
            no_improve += 1
            if no_improve > 300:
                h = perturb_h(best_h, rng, scale=0.2)
                cur_c, cur_corr = compute_c5_corr(h)
                no_improve = 0
            continue

        eps = max_eps * step_factor
        new_h = h.copy()
        new_h[src] -= eps
        new_h[dst] += eps
        new_c = compute_c5(new_h)

        if new_c < cur_c:
            h, cur_c, cur_corr = new_h, new_c, np.correlate(new_h, 1 - new_h, mode="full")
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
            step_factor = min(step_factor * 1.02, 0.8)
            no_improve = 0
        else:
            step_factor = max(step_factor * 0.95, 0.01)
            no_improve += 1
            if no_improve > 500:
                h = perturb_h(best_h, rng, scale=0.2)
                cur_c, cur_corr = compute_c5_corr(h)
                no_improve = 0

    return best_h, best_c


def slsqp_optimize_one_beta(h0, beta, rng, max_iter=400):
    """SLSQP refinement with softmax surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = enforce_sum(x, CURRENT_TARGET, rng)
    return x


def upsample_h(h, target_n):
    n = h.size
    old_x = (np.arange(n) + 0.5) * (2.0 / n)
    new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
    new_h = np.interp(new_x, old_x, h)
    return new_h
