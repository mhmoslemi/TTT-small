def enforce_sum_fast(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 10
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-12) & (h < 1-1e-12)
        cnt = np.count_nonzero(free)
        if cnt == 0:

import time
import numpy as np

# --------------------------------------------------------------
# Basic utilities
# --------------------------------------------------------------
def compute_c5(h):
    """Exact discrete C5 for a step function sampled on [0,2]."""
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)


def enforce_sum_fast(h, target):
    """Project h onto [0,1] and enforce Σh = target (simple iterative scheme)."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-12) & (h < 1 - 1e-12)
        cnt = int(free.sum())
        if cnt == 0:
            adjust = diff / h.size
            h += adjust
            h = np.clip(h, 0.0, 1.0)
        else:
            adjust = diff / cnt
            h[free] += adjust
            h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with Σh = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 4):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.5
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum_fast(h, n / 2.0)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


# --------------------------------------------------------------
# Gradient‑descent on a smooth max surrogate
# --------------------------------------------------------------
def gradient_descent_optimize(h0, rng, time_budget, n):
    """Smooth‑max gradient descent with annealed β and step‑size."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    dx = 2.0 / n
    target = n / 2.0

    beta = 10.0
    beta_max = 5000.0
    lr = 200.0          # raw step multiplier (applies to gradient ≲dx)
    iteration = 0
    eval_every = 100
    start = time.time()

    while time.time() - start < time_budget:
        # correlation (scaled)
        corr = np.correlate(h, 1 - h, mode="full") * dx

        # soft‑max weights (log‑sum‑exp trick)
        shift = np.max(beta * corr)
        w = np.exp(beta * corr - shift)
        w /= w.sum()

        # weighted shifted sums via convolution
        w_rev = w[::-1]
        conv_minus = np.convolve(h, w, mode="full")[:n]
        conv_plus = np.convolve(h, w_rev, mode="full")[:n]

        grad = dx * (1.0 - (conv_minus + conv_plus))

        # gradient step
        h = h - lr * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum_fast(h, target)

        # periodic evaluation
        if iteration % eval_every == 0:
            c = compute_c5(h)
            if c < best_c:
                best_c, best_h = c, h.copy()

        # schedule updates
        beta = min(beta * 1.02, beta_max)
        lr = max(lr * 0.999, 10.0)

        # occasional random jitter to escape plateaus
        if iteration % 500 == 0 and iteration > 0:
            h = perturb_h(h, rng, scale=0.5)
            h = enforce_sum_fast(h, target)

        iteration += 1
        if iteration > 20000:
            break

    return best_h, best_c

def hill_climb_refine(h0, rng, time_budget, n):
    """Local stochastic hill‑climb using tiny perturbations."""
    h = h0.copy()
    best_c = compute_c5(h)
    target = n / 2.0
    start = time.time()
    while time.time() - start < time_budget:
        cand = perturb_h(h, rng, scale=0.1)
        cand = enforce_sum_fast(cand, target)
        c = compute_c5(cand)
        if c < best_c:
            best_c, h = c, cand.copy()
    return h, best_c

    rng = np.random.default_rng(seed)
    overall_start = time.time()

    # ---- initialise ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum_fast(h0, n / 2.0)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    # ---- phase 1 : smooth‑max gradient descent ----
    time_left = budget_s - (time.time() - overall_start)
    phase1_time = time_left * 0.7
    best_h, best_c = gradient_descent_optimize(h0, rng, phase1_time, n)

    # ---- optional up‑sampling if enough budget remains ----
    time_left = budget_s - (time.time() - overall_start)
    if time_left > 0.2 * budget_s and n < 720:
        target_n = min(720, n * 2)
        old_x = (np.arange(n) + 0.5) * (2.0 / n)

        up_h = np.interp(new_x, old_x, best_h)
        up_h = enforce_sum_fast(up_h, target_n / 2.0)
        # re‑run gradient descent on finer grid with remaining time
        best_h, best_c = gradient_descent_optimize(up_h, rng, time_left * 0.6, target_n)
        n = target_n

    # ---- phase 2 : hill‑climb refinement on the (possibly up‑sampled) grid ----
    time_left = budget_s - (time.time() - overall_start)

    # ---- final clean‑up ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum_fast(best_h, n / 2.0)
    final_c = compute_c5(best_h)
