import time
import numpy as np

# ---------- core utilities ----------
def compute_c5_full(h):
    """Return C5 value and full correlation array for binary/float h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n)), corr


def attempt_swap(h, shift, cur_c, rng):
    """Try to find a 1↔0 swap that lowers the current C5.
    Returns (new_h, new_c, new_corr) on success, else None."""
    n = h.shape[0]
    idx = np.arange(n)
    s = shift

    if s >= 0:
        k = s
        condA = (h == 1) & (idx + k < n) & (h[idx + k] == 0)
        condB = (h == 0) & (idx + k < n) & (h[idx + k] == 1)
        condA_good = condA & (idx >= k) & (h[idx - k] == 0)
        condB_good = condB & (idx >= k) & (h[idx - k] == 1)
    else:
        k = -s
        condA = (h == 1) & (idx >= k) & (h[idx - k] == 0)
        condB = (h == 0) & (idx >= k) & (h[idx - k] == 1)
        condA_good = condA & (idx + k < n) & (h[idx + k] == 0)
        condB_good = condB & (idx + k < n) & (h[idx + k] == 1)

    A = np.flatnonzero(condA)
    B = np.flatnonzero(condB)
    if A.size == 0 or B.size == 0:
        return None

    # --- try delta = 2 swaps (most promising) ---
    A_good = np.flatnonzero(condA_good)
    B_good = np.flatnonzero(condB_good)
    if A_good.size and B_good.size:
        # a few random attempts among the good sets
        attempts = min(5, A_good.size * B_good.size)
        for _ in range(attempts):
            a = int(rng.choice(A_good))
            b = int(rng.choice(B_good))
            if a == b:
                continue
            new_h = h.copy()
            new_h[a] = 0
            new_h[b] = 1
            new_c, new_corr = compute_c5_full(new_h)
            if new_c < cur_c - 1e-12:
                return new_h, new_c, new_corr

    # --- fall back to random pairs from A×B ---
    for _ in range(10):
        a = int(rng.choice(A))
        b = int(rng.choice(B))
        if a == b:
            continue
        new_h = h.copy()
        new_h[a] = 0
        new_h[b] = 1
        new_c, new_corr = compute_c5_full(new_h)
        if new_c < cur_c - 1e-12:
            return new_h, new_c, new_corr
    return None


def local_opt_binary(h, rng, deadline):
    """Iterated improvement focusing on the current worst shift."""
    cur_c, cur_corr = compute_c5_full(h)
    improved = True
    n = h.shape[0]
    while improved and time.time() < deadline:
        # worst shift for the current vector
        max_idx = cur_corr.argmax()
        shift = max_idx - (n - 1)
        res = attempt_swap(h, shift, cur_c, rng)
        if res is None:
            break
        new_h, new_c, new_corr = res
        # accept only if we really improved
        if new_c < cur_c - 1e-12:
            h, cur_c, cur_corr = new_h, new_c, new_corr
        else:
            break
    return h, cur_c
