def hill_climb(h_start, rng, deadline, max_iters=5000, step_init=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step_init
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c; best_h = new_h.copy(); h = new_h
        if (it+1) % decay_interval == 0:
            step = max(step * 0.7, 1e-6)
    return best_h, best_c

    h = h_bin.copy()
    n = h.size
    current_c = compute_c5(h)
    best_h = h.copy()
    best_c = current_c
    # temperature schedule
    T0 = best_c * 0.02 + 1e-6
    Tf = 1e-6
    # estimate total iterations based on time left
    # We can simply compute alpha using a large fixed iteration count
    est_iters = 200000
    alpha = -np.log(Tf / T0) / est_iters
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # choose i where h[i]==1

        while h[i] != 1.0:

                    best_h = h.copy()

import time, math, numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Utility functions
# --------------------------------------------------------------
def enforce_sum(h, target, rng):
    """Adjust h to have sum=target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    """Uniform random point in [0,1]^n with sum=n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)

def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount between two entries."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def correlate_full(h):
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    fr = np.fft.rfft((1 - h)[::-1], L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv

def compute_c5(h):
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)

# --------------------------------------------------------------
# Smooth surrogate for SLSQP
# --------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = correlate_full(x)
    bc = CURRENT_BETA * corr
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        x = enforce_sum(x, CURRENT_TARGET, rng)
    return x

# --------------------------------------------------------------
# Hill climbing (continuous)
# --------------------------------------------------------------
def hill_climb(h_start, rng, deadline, max_iters=8000, step_init=0.3):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step_init
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-6)
    return best_h, best_c

# --------------------------------------------------------------
# Binary simulated annealing
# --------------------------------------------------------------
def binary_sa(h_bin, rng, deadline):
    h = h_bin.copy()
    n = h.size
    current_c = compute_c5(h)
    best_h = h.copy()
    best_c = current_c
    T0 = best_c * 0.02 + 1e-6
    Tf = 1e-6
    est_iters = 200000
    alpha = -math.log(Tf / T0) / est_iters
    it = 0
    while time.time() < deadline:
        it += 1
        i = rng.integers(0, n)
        while h[i] != 1.0:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while h[j] != 0.0:
            j = rng.integers(0, n)
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h)
        if new_c < current_c:
            current_c = new_c
            if new_c < best_c:
                best_c, best_h = new_c, h.copy()
        else:
            T = T0 * math.exp(-alpha * it)
            if rng.random() < math.exp(-(new_c - current_c) / max(T, 1e-12)):
                current_c = new_c
                if new_c < best_c:
                    best_c, best_h = new_c, h.copy()
            else:
                h[i] = 1.0
                h[j] = 0.0
    return best_h, best_c
