def simulated_annealing(h, target_sum, time_limit):
    best_h = h
    best_c = compute_c5(h)
    cur_h = h.copy()
    cur_c = best_c
    start = time.time()
    # initial temperature based on best_c maybe 0.01
    T0 = 0.01
    # schedule: T = T0 * (1 - t / total)
    total = time_limit
    while time.time() - start < time_limit:
        t = time.time() - start
        T = T0 * max(0.0, 1 - t/total)
        # propose random perturb
        # choose i, j random
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j: continue
        max_eps = min(cur_h[i], 1 - cur_h[j])
        if max_eps == 0:
            continue
        # epsilon random with scale maybe max_eps * rng.random()
        eps = max_eps * rng.random()
        cand = cur_h.copy()
        cand[i] -= eps
        cand[j] += eps
        # we can avoid projection because sum unchanged & bounds satisfied
        cand_c = compute_c5(cand)
        # acceptance
        if cand_c < cur_c or rng.random() < np.exp(-(cand_c - cur_c)/ (T + 1e-12)):
            cur_h, cur_c = cand, cand_c
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
    return best_h

def proj_capped_simplex(v, target):
    # ensure v is numpy array
    v = np.asarray(v, dtype=np.float64)
    # Clip initial to [0,1] maybe not needed
    # Determine lower and upper bound for lambda
    # As lambda -> -inf, w_i -> 1 for all => sum = n >= target
    # As lambda -> +inf, w_i -> 0 => sum = 0 <= target
    low = -1.0  # might be too high? Actually we need ensure w all 1; for v_i - low >=1 => low <= v_i - 1 for all i => low <= min(v) - 1.
    low = np.min(v) - 1.0 - 1.0  # value sufficiently low
    high = np.max(v) + 1.0 + 1.0  # value sufficiently high
    for _ in range(60):
        mid = (low + high) / 2.0
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            low = mid
        else:
            high = mid
    w = np.clip(v - (low + high) / 2.0, 0.0, 1.0)
    # adjust exact sum if slight rounding
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        # find index to adjust (largest capacity left)
        # add diff to any w_i not at bound
        idxs = np.where((w > 0) & (w < 1))[0]
        if idxs.size > 0:
            i = idxs[0]
            w[i] = np.clip(w[i] + diff, 0.0, 1.0)
        else:
            # all at bounds: adjust first element within bounds
            i = np.argmax(w) if diff > 0 else np.argmin(w)
            w[i] = np.clip(w[i] + diff, 0.0, 1.0)
    return w

def gradient_shift(h, s):
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        # i from 0 .. n-1-s: term (1 - h[i+s])
        grad[:n - s] += (1 - h[s:])
        # i from s .. n-1: term -h[i-s]
        grad[s:] -= h[:n - s]
    else:
        s_abs = -s
        # i from s_abs .. n-1: (1 - h[i - s_abs])?? Actually s negative, so i sees shift s = -s_abs => i + s = i - s_abs
        grad[s_abs:] += (1 - h[:n - s_abs])
        # i from 0 .. n - s_abs - 1: -h[i + s_abs]
        grad[:n - s_abs] -= h[s_abs:]
    return grad

def anneal_optimize(h0, target_sum, time_limit, rng):
    n = h0.size
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()

    # Use typical initial T as maybe 0.005
    T0 = 0.005

        t = time.time() - start
        frac = t / time_limit
        T = T0 * (1 - frac)  # linear cooling
        # propose random epsilon transfer
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # determine max transfer
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0.0:

    rng = np.random.default_rng(seed)

        c_a = compute_c5(h_anneal)
        if c_a < best_c:

    # Phase 3: final local search: greedy + small perturb

        # produce candidate by small perturb and greedy refinement
        cand = perturb_random(best_h, rng, scale=rng.random()*0.1+0.05)

    best_h = proj_capped_simplex(best_h, target_sum)
    final_c = compute_c5(best_h)

import time
import numpy as np
from scipy.optimize import minimize

# ------------------- core utilities ------------------- #

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for vector h (integral over [0,2] approximated)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def gradient_shift(h: np.ndarray, s: int) -> np.ndarray:
    """Gradient of the correlation at shift s w.r.t. h."""
    n = h.size
    grad = np.zeros(n, dtype=np.float64)
    if s >= 0:
        grad[: n - s] += (1.0 - h[s:])
        grad[s:] -= h[: n - s]
    else:
        s_abs = -s
        grad[s_abs:] += (1.0 - h[: n - s_abs])
        grad[: n - s_abs] -= h[s_abs:]
    return grad


def proj_capped_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Projection of v onto the intersection of the hypercube [0,1]^n
    with the affine hyperplane sum(v) = target.
    Uses a simple binary search on the dual variable λ.
    """
    v = np.asarray(v, dtype=np.float64)
    low = np.min(v) - 2.0  # guarantees all entries clipped to 1
    high = np.max(v) + 2.0  # guarantees all entries clipped to 0
    for _ in range(60):
        mid = (low + high) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            low = mid
        else:
            high = mid
    w = np.clip(v - (low + high) * 0.5, 0.0, 1.0)
    # minor correction for numerical drift
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        free = np.where((w > 0.0) & (w < 1.0))[0]
        if free.size > 0:
            i = free[0]
        else:
            i = np.argmax(w) if diff > 0 else np.argmin(w)
        w[i] = np.clip(w[i] + diff, 0.0, 1.0)
    return w


def greedy_optimize(h0: np.ndarray, max_steps: int = 2000) -> np.ndarray:
    """Deterministic improvement by moving mass from a positive‑gradient index to a negative‑gradient index."""
    h = h0.copy()
    n = h.size
    for _ in range(max_steps):
        # worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr)
        s = idx - (n - 1)
        # gradient at that shift
        grad = gradient_shift(h, s)
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-12:
            break
        # try full transfer
        nh = h.copy()
        nh[i] -= max_eps
        nh[j] += max_eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        # try half transfer
        eps = max_eps * 0.5
        nh[i] -= eps
        nh[j] += eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        break
    return h


def perturb_random(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a random amount of mass from one coordinate to another."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def anneal_optimize(h0: np.ndarray, target_sum: float, time_limit: float, rng: np.random.Generator) -> np.ndarray:
    """Simple simulated‑annealing on the feasible set."""
    n = h0.size
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    start = time.time()
    T0 = 0.005  # initial temperature
    while time.time() - start < time_limit:
        frac = (time.time() - start) / time_limit
        T = T0 * (1.0 - frac)  # linear cooling
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0.0:
            continue
        eps = max_eps * rng.random()
        cand = cur_h.copy()
        cand[i] -= eps
        cand[j] += eps
        cand_c = compute_c5(cand)
        delta = cand_c - cur_c
        if delta < 0.0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            cur_h, cur_c = cand, cand_c
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
    return best_h


def slsqp_optimize(h0: np.ndarray, beta: float, max_iter: int = 800) -> np.ndarray:
    """Polish a feasible vector using SLSQP with a log‑sum‑exp surrogate."""
    n = h0.size
    target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    res = minimize(
        obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return proj_capped_simplex(x, target)


# ------------------- main entry point ------------------- #

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- initialise feasible h ----
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_default = 720
        h = np.full(n_default, 0.5, dtype=float)
    n = h.size
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    h = proj_capped_simplex(h, target_sum)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ---- Phase 1: greedy refinement (quick) ----
    phase1_end = start + budget_s * 0.05
    if time.time() < phase1_end:
        h_g = greedy_optimize(best_h, max_steps=5000)
        c_g = compute_c5(h_g)
        if c_g < best_c:
            best_h, best_c = h_g, c_g

    # ---- Phase 2: simulated annealing (major work) ----
    remaining = budget_s - (time.time() - start)
    anneal_time = min(remaining * 0.70, budget_s * 0.65)
    if anneal_time > 5.0:
        h_a = anneal_optimize(best_h, target_sum, anneal_time, rng)
        c_a = compute_c5(h_a)
        if c_a < best_c:
            best_h, best_c = h_a, c_a

    # ---- Phase 3: small‑scale stochastic improvements ----
    while time.time() - start < budget_s * 0.95:
        cand = perturb_random(best_h, rng, scale=rng.random() * 0.1 + 0.05)
        # occasional quick local search
        if rng.random() < 0.3:
            cand = greedy_optimize(cand, max_steps=500)
        c_cand = compute_c5(cand)
        if c_cand < best_c:
            best_h, best_c = cand, c_cand

    # ---- Phase 4: final SLSQP polish (if time permits) ----
    if time.time() - start < budget_s * 0.99:
        try:
            refined = slsqp_optimize(best_h, beta=20000.0, max_iter=1000)
            c_ref = compute_c5(refined)
            if c_ref < best_c:
                best_h, best_c = refined, c_ref
        except Exception:
            pass

    # ---- clean up and return ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = proj_capped_simplex(best_h, target_sum)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
