def simulated_annealing(h0, rng, max_iter, time_limit):
    h = h0.copy()
    c = compute_c5(h)
    best_h, best_c = h.copy(), c
    n = h.size
    start = time.time()
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        # compute worst shift and gradient
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        # gradient
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            g = np.ones(n)
            g[:n - k] -= h[k:]
            g[k:] -= h[:n - k]
        # choose move
        pos = np.where(g > 1e-12)[0]
        neg = np.where(g < -1e-12)[0]
        if pos.size == 0 or neg.size == 0:
            # maybe random move
            i, j = rng.integers(0, n, size=2)
        else:
            i = rng.choice(pos)
            j = rng.choice(neg)
        # epsilon max
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        # epsilon scaling based on iteration
        eps_scale = 0.5 * (1 - it / max_iter) + 0.01  # decays from 0.5 to 0.01
        eps = max_eps * rng.random() * eps_scale
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        c_new = compute_c5(h_new)
        # temperature decline
        T0 = 1e-4
        T = T0 * (1 - it / max_iter)
        if c_new < c or rng.random() < np.exp(-(c_new - c) / (T + 1e-12)):
            h, c = h_new, c_new
            if c < best_c:
                best_h, best_c = h.copy(), c
        # occasionally random move
        if it % 1000 == 0 and rng.random() < 0.2:
            # random i,j epsilon
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0:
                continue
            eps = max_eps * rng.random() * 0.2
            h_rand = h.copy()
            h_rand[i] -= eps
            h_rand[j] += eps
            c_rand = compute_c5(h_rand)
            if c_rand < best_c:
                best_h, best_c = h_rand.copy(), c_rand
                h, c = h_rand, c_rand
    return best_h, best_c
```

    corr = np.correlate(h, 1 - h, mode='full')

    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

                   constraints=cons,

import time, numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k O_k."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp (stable)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 500) -> np.ndarray:
    """SLSQP refinement using a soft‑max objective."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:

            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

    """Generate a random vector in [0,1] with Σh = n/2."""
    h = rng.random(n)

            h[i] += add
            diff = target - h.sum()
    return h


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()

        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])

        diff = target - h.sum()
    return h

                        max_iter: int, time_limit: float):
    """SA that respects the sum constraint by transferring mass between indices."""
    h = h0.copy()

    c = compute_c5(h)
    best_h, best_c = h.copy(), c

        else:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue

        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps_scale = 0.5 * (1.0 - it / max_iter) + 0.01
        eps = max_eps * rng.random() * eps_scale

        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        c_new = compute_c5(h_new)

        # temperature schedule
        T0 = 5e-4
        T = T0 * (1.0 - it / max_iter)
        if c_new < c or rng.random() < np.exp(-(c_new - c) / (T + 1e-15)):
            h, c = h_new, c_new
            if c < best_c:
                best_h, best_c = h.copy(), c

        # occasional random perturbation to diversify
        if it % 500 == 0 and rng.random() < 0.15:
            i2 = rng.integers(0, n)
            j2 = rng.integers(0, n)
            if i2 != j2:
                max_eps2 = min(h[i2], 1.0 - h[j2])
                if max_eps2 > 0:
                    eps2 = max_eps2 * rng.random() * 0.2
                    h2 = h.copy()
                    h2[i2] -= eps2
                    h2[j2] += eps2
                    c2 = compute_c5(h2)
                    if c2 < best_c:
                        best_h, best_c = h2.copy(), c2
                        h, c = h2, c2

    return best_h, best_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    best_h = None
    best_c = np.inf
    best_n = None

    # -----------------------------------------------------------------
    # initial solution from parent (if available)
    # -----------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        best_h, best_c = h0.copy(), compute_c5(h0)
        best_n = h0.size

    # -----------------------------------------------------------------
    # Phase 1 – many random restarts at n = 256
    # -----------------------------------------------------------------
    n0 = 256
    phase1_deadline = start + budget_s * 0.45
    while time.time() < phase1_deadline:
        h_start = random_feasible_vector(n0, rng)
        remaining = phase1_deadline - time.time()
        if remaining <= 0:
            break
        # 120k iterations fit comfortably in the remaining slice
        h_opt, c_opt = simulated_annealing(
            h_start, rng, max_iter=120000, time_limit=remaining
        )
        if c_opt < best_c:
            best_h, best_c, best_n = h_opt.copy(), c_opt, n0

    # -----------------------------------------------------------------
    # Phase 2 – up‑sample to n = 384 and longer SA
    # -----------------------------------------------------------------
    if best_h is not None:
        n_up = 384
        x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
        x_up = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(x_up, x_old, best_h)
        h_up = np.clip(h_up, 0.0, 1.0)
        h_up = enforce_sum(h_up, rng)

        phase2_deadline = start + budget_s * 0.85
        remaining = phase2_deadline - time.time()
        if remaining > 0:
            h_up_opt, c_up_opt = simulated_annealing(
                h_up, rng, max_iter=300000, time_limit=remaining
            )
            if c_up_opt < best_c:
                best_h, best_c, best_n = h_up_opt.copy(), c_up_opt, n_up

    # -----------------------------------------------------------------
    # Phase 3 – final SLSQP refinement (if time permits)
    # -----------------------------------------------------------------
    if best_h is not None and time.time() < start + budget_s * 0.97:
        for beta in (20.0, 200.0, 2000.0, 8000.0):
            if time.time() > start + budget_s * 0.99:
                break
            best_h = slsqp_optimize(best_h, beta, rng, max_iter=800)
            best_c = compute_c5(best_h)

    # -----------------------------------------------------------------
    # Sanity check and final return
    # -----------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
