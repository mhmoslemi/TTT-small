def proj_grad_descent(h0, beta, max_iters, step0):
    h = h0.copy()
    target = h.size / 2.0
    step = step0
    best_c = compute_c5(h)
    for it in range(max_iters):
        grad = gradient_softmax(h, beta)
        # direction: we want to minimize, so move opposite grad
        h_new = h - step * grad
        h_new = project_box_sum(h_new, target)
        new_c = compute_c5(h_new)
        # accept if improves else reduce step
        if new_c < best_c - 1e-12:
            h = h_new
            best_c = new_c
            step *= 1.1  # increase step
        else:
            step *= 0.5
        # optionally break if step too small
        if step < 1e-12:
            break
    return h, best_c

import time
import numpy as np

# ---------- basic utilities ----------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    """Project x onto the box [lo,hi] while enforcing sum(x)=target."""
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y


def random_binary(n, rng):
    arr = np.zeros(n, dtype=float)
    arr[: n // 2] = 1.0
    rng.shuffle(arr)
    return arr


def compute_c5(h):
    """return C5 for the discretised step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


# ---------- gradient of soft‑max surrogate ----------
def gradient_softmax(h, beta):
    """gradient of (1/beta) logsumexp(beta * c_i) where c_i = corr_i."""
    n = h.size
    c = np.correlate(h, 1.0 - h, mode="full")
    # stable soft‑max weights
    maxb = np.max(beta * c)
    w = np.exp(beta * c - maxb)
    w /= w.sum()

    shifts = np.arange(-(n - 1), n)               # length 2n‑1
    K = np.arange(n).reshape(1, n)               # (1,n)
    T = shifts.reshape(-1, 1)                    # (2n‑1,1)

    idx_plus = K + T
    mask_plus = (idx_plus >= 0) & (idx_plus < n)
    idx_plus_clipped = np.clip(idx_plus, 0, n - 1)
    term_a = (w[:, None] * h[idx_plus_clipped] * mask_plus).sum(axis=0)

    idx_minus = K - T
    mask_minus = (idx_minus >= 0) & (idx_minus < n)
    idx_minus_clipped = np.clip(idx_minus, 0, n - 1)
    term_b = (w[:, None] * h[idx_minus_clipped] * mask_minus).sum(axis=0)

    grad = np.ones_like(h) - (term_a + term_b)
    return grad


# ---------- projected gradient descent on the surrogate ----------
def proj_grad_descent(h0, beta, max_iters, step0):
    """minimise the soft‑max surrogate for a fixed beta."""
    h = h0.copy()
    target = h.size / 2.0
    step = step0
    best_c = compute_c5(h)
    for _ in range(max_iters):
        grad = gradient_softmax(h, beta)
        h_new = h - step * grad
        h_new = project_box_sum(h_new, target)
        new_c = compute_c5(h_new)
        if new_c < best_c - 1e-12:
            h, best_c = h_new, new_c
            step = min(step * 1.2, 1.0)
        else:
            step *= 0.5
        if step < 1e-8:
            break
    return h, best_c


# ---------- simple hill‑climb for final polishing ----------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def compute_c5_and_shift(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (h.size - 1)
    return corr[idx] * (2.0 / h.size), shift


def hill_climb_targeted(h_start, rng, max_iters=600, init_step=0.4, p_target=0.25):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    n = h.size
    for it in range(max_iters):
        _, k_star = compute_c5_and_shift(h)
        improved = False
        for _ in range(5):
            if rng.random() < p_target:
                cand = targeted_move(h, rng, k_star, scale=step)
            else:
                cand = perturb_h(h, rng, scale=step)
            c = compute_c5(cand)
            if c < best_c - 1e-12:
                best_c, h = c, cand
                improved = True
                break
        if not improved and rng.random() < 0.02:
            cand = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(cand)
            if c < best_c:
                best_c, h = c, cand
        step = max(step * 0.95, 1e-5)
    return h, best_c


def targeted_move(h, rng, k, scale=0.2):
    n = h.size
    if k <= 0 or k >= n:
        return perturb_h(h, rng, scale)
    low_mask = (h[: n - k] < 0.5) & (h[k:] < 0.5)
    high_mask = (h[: n - k] > 0.5) & (h[k:] > 0.5)
    if not low_mask.any() or not high_mask.any():
        return perturb_h(h, rng, scale)
    i_low = rng.choice(np.flatnonzero(low_mask))
    i_high = rng.choice(np.flatnonzero(high_mask))
    max_eps = min(
        0.5 - h[i_low],
        0.5 - h[i_low + k],
        h[i_high] - 0.5,
        h[i_high + k] - 0.5,
    )
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i_low] += eps
    nh[i_low + k] += eps
    nh[i_high] -= eps
    nh[i_high + k] -= eps
    return nh

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Initialise
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_box_sum(h0, n / 2.0)
    else:
        n = 360
        h0 = random_binary(n, rng)

    # Build a modest pool of starts
    starts = [h0]
    for _ in range(6):
        starts.append(random_binary(n, rng))

    xs = (np.arange(n) + 0.5) * (2.0 / n)
    for f in (1, 2, 3, 5):
        cand = 0.5 + 0.5 * np.sin(np.pi * f * xs)
        cand = project_box_sum(np.clip(cand, 0.0, 1.0), n / 2.0)
        starts.append(cand)
        cand = 0.5 + 0.5 * np.cos(np.pi * f * xs)
        cand = project_box_sum(np.clip(cand, 0.0, 1.0), n / 2.0)
        starts.append(cand)

    # optimisation parameters
    beta_list = [5, 20, 80, 200, 500, 1000, 2000, 5000, 10000]
    max_iters_per_beta = 250
    best_h = None
    best_c = np.inf

    for h_start in starts:
        if time.time() - start_time > budget_s * 0.85:
            break
        h = h_start.copy()
        step0 = 0.5
        for beta in beta_list:
            if time.time() - start_time > budget_s * 0.85:
                break
            h, _ = proj_grad_descent(h, beta, max_iters_per_beta, step0)
            step0 = max(step0 * 0.9, 1e-4)

        # final tidy hill‑climb
        h, c = hill_climb_targeted(h, rng, max_iters=500, init_step=0.3)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # optional up‑sampling and refinement
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > budget_s * 0.12 and best_h is not None:
        target_n = max(720, n * 2)
        target_n = min(target_n, 960)
        if target_n != n:
            old_x = (np.arange(n) + 0.5) * (2.0 / n)
            new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
            h_up = np.interp(new_x, old_x, best_h)
            h_up = project_box_sum(h_up, target_n / 2.0)

            step0 = 0.5
            for beta in beta_list:
                if time.time() - start_time > budget_s:
                    break
                h_up, _ = proj_grad_descent(h_up, beta, max_iters_per_beta, step0)
                step0 = max(step0 * 0.9, 1e-4)

            h_up, c_up = hill_climb_targeted(h_up, rng, max_iters=800, init_step=0.3)
            if c_up < best_c:
                best_c, best_h, n = c_up, h_up.copy(), target_n

    # final projection and result
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
