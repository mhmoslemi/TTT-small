def continuous_sa(h0, rng, time_limit):
    h = h0.copy()
    cur_c = compute_c5_fast(h)
    T0 = 0.05 * cur_c + 1e-6
    it = 0
    while time.time() < time_limit:
        # choose two indices i (any) and j (any, j!=i)
        i = rng.integers(0,n)
        j = rng.integers(0,n-1)
        if j >= i: j+=1
        # delta drawn uniformly from [-delta_max, delta_max], delta_max decays over time
        delta_max = 0.2 * (0.9995**it)  # decays
        delta = rng.uniform(-delta_max, delta_max)
        # proposed new values
        hi = h[i] + delta
        hj = h[j] - delta
        # ensure bounds
        if hi < 0 or hi > 1 or hj < 0 or hj > 1:
            continue
        old_i = h[i]; old_j = h[j]
        h[i] = hi; h[j] = hj
        c_new = compute_c5_fast(h)
        # acceptance
        if c_new < cur_c or rng.random() < np.exp(-(c_new-cur_c)/ (T0 * (0.9999**it) + 1e-12)):
            cur_c = c_new
        else:
            h[i] = old_i; h[j] = old_j
        it += 1
        # maybe adjust temperature or early break if it huge
    return h, cur_c

def continuous_sa(h0, rng, time_deadline):
    h = h0.copy()
    cur_c = compute_c5_fast(h)
    best_h = h.copy()
    best_c = cur_c
    n = h.shape[0]
    T0 = max(0.02 * cur_c, 1e-5)
    it = 0
    last_improve = 0
    while time.time() < time_deadline:
        # select two distinct indices
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        # delta scale decays
        delta_max = 0.2 * (0.9995 ** it)
        delta = rng.uniform(-delta_max, delta_max)
        hi = h[i] + delta
        hj = h[j] - delta
        if hi < 0.0 or hi > 1.0 or hj < 0.0 or hj > 1.0:
            # reject move
            continue
        # apply
        old_i, old_j = h[i], h[j]
        h[i], h[j] = hi, hj
        c = compute_c5_fast(h)
        T = T0 * (0.9995 ** it) + 1e-12
        accept = False
        if c < cur_c:
            accept = True
        else:
            if rng.random() < np.exp(-(c - cur_c) / T):
                accept = True
        if accept:
            cur_c = c
            if c < best_c:
                best_c, best_h = c, h.copy()
                last_improve = it
        else:
            h[i], h[j] = old_i, old_j
        it += 1
        if it - last_improve > 10000:
            # stagnation break
            break
    return best_h, best_c

def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]        # + (1 - h[i+k])
            g[k:] -= h[:upper]               # - h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

import time, math, numpy as np
from scipy.optimize import minimize

# ---------- utilities ----------
def compute_c5_fast(h: np.ndarray) -> float:
    n = h.shape[0]
    rev = h[::-1]
    b = 1.0 - h
    nfft = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(rev, nfft)
    B = np.fft.rfft(b, nfft)
    conv = np.fft.irfft(H * B, nfft)
    maxv = np.max(conv[: 2 * n - 1])
    return float(maxv * (2.0 / n))

def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator) -> None:
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()

                   step: float = 0.02, max_iter: int = 12) -> np.ndarray:
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            up = n - k
            g[:up] += 1.0 - h[k:]
            g[k:] -= h[:up]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

# ---------- binary simulated annealing ----------
def binary_sa_hill_climb(h_bin: np.ndarray, rng: np.random.Generator,
                         deadline: float) -> tuple:
    cur = h_bin.copy()
    cur_c = compute_c5_fast(cur)
    best_h, best_c = cur.copy(), cur_c
    n = cur.size
    ones = list(np.flatnonzero(cur == 1))
    zeros = list(np.flatnonzero(cur == 0))
    T0 = max(0.01 * cur_c, 1e-4)
    it = last_improve = 0
    while time.time() < deadline:
        if not ones or not zeros:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur[i] = 0.0
        cur[j] = 1.0
        c = compute_c5_fast(cur)
        T = T0 * (0.9995 ** it) + 1e-12
        accept = c < cur_c or rng.random() < math.exp(-(c - cur_c) / T)
        if accept:
            cur_c = c
            ones.remove(i); ones.append(j)
            zeros.remove(j); zeros.append(i)
            if c < best_c:
                best_c, best_h, last_improve = c, cur.copy(), it
        else:
            cur[i] = 1.0
            cur[j] = 0.0
        it += 1
        if it - last_improve > 5000:
            break
    return best_h, best_c

# ---------- continuous simulated annealing ----------
def continuous_sa(h0: np.ndarray, rng: np.random.Generator,
                 deadline: float) -> tuple:
    h = h0.copy()
    cur_c = compute_c5_fast(h)
    best_h, best_c = h.copy(), cur_c
    n = h.size
    T0 = max(0.02 * cur_c, 1e-5)
    it = last_improve = 0
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta_max = 0.2 * (0.9995 ** it)
        delta = rng.uniform(-delta_max, delta_max)
        hi, hj = h[i] + delta, h[j] - delta
        if not (0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0):
            continue
        old_i, old_j = h[i], h[j]
        h[i], h[j] = hi, hj
        c = compute_c5_fast(h)
        T = T0 * (0.9995 ** it) + 1e-12
        accept = c < cur_c or rng.random() < math.exp(-(c - cur_c) / T)
        if accept:
            cur_c = c
            if c < best_c:
                best_c, best_h, last_improve = c, h.copy(), it
        else:
            h[i], h[j] = old_i, old_j
        it += 1
        if it - last_improve > 10000:
            break
    return best_h, best_c

# ---------- SLSQP smooth‑max optimizer ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 400) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

# ---------- main driver ----------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    parent = globals().get("initial_h_values", None)
    parent_n = len(parent) if isinstance(parent, (list, np.ndarray)) else None

    # candidate grid sizes (all <=720)
    ns = [180, 240, 360, 540, 720]
    if parent_n and parent_n not in ns:
        ns = [parent_n] + ns
    ns = [n for n in ns if n <= 720]

    best_h = None
    best_c = np.inf
    best_n = None

    for idx, n in enumerate(ns):
        if time.time() > deadline:
            break
        # allocate roughly equal time to each remaining size
        remaining = len(ns) - idx
        time_left = deadline - time.time()
        sub_deadline = time.time() + time_left / remaining

        # initial vector for this size
        if parent_n == n:
            h0 = np.array(parent, dtype=float)
        elif best_h is not None and best_n != n:
            xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(xt, xs, best_h)
        else:
            h0 = np.zeros(n)
            h0[rng.choice(n, n // 2, replace=False)] = 1.0
        adjust_sum_exact(h0, rng)

        cur_c = compute_c5_fast(h0)
        if cur_c < best_c:
            best_h, best_c, best_n = h0.copy(), cur_c, n

        # repeated restarts within the allocated window
        while time.time() < sub_deadline - 0.5:
            # random binary start
            rnd = rng.random(n)
            idx_top = np.argpartition(rnd, -n // 2)[-n // 2:]
            h_bin = np.zeros(n)
            h_bin[idx_top] = 1.0

            # binary SA (≈40% of remaining time)
            t_bin = min((sub_deadline - time.time()) * 0.4, 6.0)
            h_bin_opt, _ = binary_sa_hill_climb(h_bin, rng, time.time() + t_bin)

            # short gradient smoothing
            h_bin_opt = gradient_refine(h_bin_opt, rng, step=0.02, max_iter=4)

            # continuous SA with the rest of the budget for this restart
            t_cont = max(0.0, sub_deadline - time.time() - 0.1)
            h_cont, _ = continuous_sa(h_bin_opt, rng, time.time() + t_cont)

            # final gradient touch
            h_cont = gradient_refine(h_cont, rng, step=0.02, max_iter=6)
            c_final = compute_c5_fast(h_cont)

            if c_final < best_c:
                best_h, best_c, best_n = h_cont.copy(), c_final, n
                if best_c < 0.3808:  # reached target, can stop early
                    break
        # optional fine‑tuning with SLSQP if time permits
        if time.time() < deadline - 5.0 and best_h is not None:
            for beta in (20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0):
                if time.time() > deadline - 2.0:
                    break
                try:
                    best_h = slsqp_optimize(best_h, beta, rng, max_iter=300)
                except Exception:
                    continue
                best_c = compute_c5_fast(best_h)
                if best_c < 0.3808:
                    break

        adjust_sum_exact(best_h, rng)
        if best_c < 0.3808:
            break

    # final safety check
    if best_h is None:
        n = 180
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5_fast(best_h)
        best_n = n
    else:
        best_h = np.clip(best_h, 0.0, 1.0)
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5_fast(best_h)
        best_n = best_h.size

    return best_h.tolist(), float(best_c), int(best_n)
