def smooth_gradient_descent(h, beta, iters, step):
    n = h.shape[0]
    target_sum = n/2.0
    for _ in range(iters):
        # auto correlation compute
        auto = np.correlate(h, h, mode='full')
        S = h.sum()
        overlap = S - auto  # shape 2n-1
        # compute weights
        max_ov = overlap.max()
        exp_vals = np.exp(beta * (overlap - max_ov))
        w = exp_vals / exp_vals.sum()
        # combine weights
        w0 = w[n-1]
        g = np.zeros_like(h)
        g -= 2 * w0 * h
        # loop for m
        for m in range(1, n):
            wk = w[n-1 + m] + w[n-1 - m]
            if wk != 0.0:
                # update slices
                g[:n-m] -= wk * h[m:]
                g[m:]    -= wk * h[:n-m]
        # gradient step
        h = h - step * g
        # ensure feasibility
        h = np.clip(h, 0.0, 1.0)
        h = project_box_sum(h, target_sum)
    return h

def smooth_gd(h, beta, iters, step):
    n = h.size
    target = n/2.0
    for _ in range(iters):
        auto = np.correlate(h, h, mode='full')
        S = h.sum()
        overlap = S - auto  # shape 2n-1
        # compute weights
        max_ov = overlap.max()
        # shift to prevent overflow:
        exp_vals = np.exp(beta * (overlap - max_ov))
        w = exp_vals / exp_vals.sum()
        # combine symmetric weights
        w0 = w[n-1]
        g = np.zeros_like(h)
        g -= 2.0 * w0 * h
        # loop for m
        for m in range(1, n):
            wk = w[n-1 + m] + w[n-1 - m]
            if wk != 0.0:
                # slices
                g[:n-m] -= wk * h[m:]
                g[m:]    -= wk * h[:n-m]
        # update
        h = h - step * g
        # projection
        h = np.clip(h, 0.0, 1.0)
        h = project_box_sum(h, target)
    return h

import time

# ---------- basic utilities ----------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y


def random_feasible_vector(n, rng):
    return project_box_sum(rng.random(n), n / 2.0)


def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


# ---------- smooth gradient descent ----------
def smooth_gd(h, beta, iters, step):
    n = h.size
    target = n / 2.0
    for _ in range(iters):
        auto = np.correlate(h, h, mode="full")
        S = h.sum()
        overlap = S - auto                       # length 2n-1
        max_ov = overlap.max()
        exp_vals = np.exp(beta * (overlap - max_ov))
        w = exp_vals / exp_vals.sum()            # softmax weights
        w0 = w[n - 1]
        g = np.zeros_like(h)
        g -= 2.0 * w0 * h
        for m in range(1, n):
            wk = w[n - 1 + m] + w[n - 1 - m]     # combined symmetric weight
            if wk != 0.0:
                g[: n - m] -= wk * h[m:]
                g[m:]      -= wk * h[: n - m]
        h = h - step * g
        h = np.clip(h, 0.0, 1.0)
        h = project_box_sum(h, target)
    return h
