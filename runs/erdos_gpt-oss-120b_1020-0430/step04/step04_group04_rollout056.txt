def slsqp_beta_schedule(v0, beta_list, monotonic, maxiter):
    v = v0.copy()
    best_c = np.inf
    best_v = v.copy()
    bounds = [(0,1)]*len(v)
    cons = []
    if monotonic:
        cons.append({'type':'ineq', 'fun': lambda x, *args: x[:-1] - x[1:]})
    for beta in beta_list:
        res = minimize(smooth_max_obj, v, args=(beta,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter':maxiter, 'ftol':1e-9, 'disp':False})
        v = np.clip(res.x, 0.0, 1.0)
        c = compute_c5_from_v(v)
        if c < best_c:
            best_c = c
            best_v = v.copy()
    return best_v, best_c

import time
import numpy as np
from scipy.optimize import minimize

# -------------------- core utilities --------------------
def construct_h(v):
    """Full symmetric step function from half-vector v (non‑increasing)."""
    return np.concatenate((v, 1 - v[::-1]))


def full_correlation(h):
    """Correlation of h with (1‑h) via FFT, without scaling."""
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return conv


def compute_c5_from_v(v):
    """Exact C₅ for a half‑vector v (full h = construct_h(v))."""
    h = construct_h(v)
    conv = full_correlation(h)
    # dx = 2 / len(h) = 1 / len(v)
    return np.max(conv) * (1.0 / v.shape[0])


def smooth_max_obj(v, beta):
    """Smooth approximation of max_k correlation (log‑sum‑exp)."""
    h = construct_h(v)
    conv = full_correlation(h) * (2.0 / h.size)   # scale by dx
    m = conv.max()
    # stable log‑sum‑exp
    shift = conv - m
    return m + np.log(np.sum(np.exp(beta * shift))) / beta


def monotonic_constraint(v, *args):
    """Enforces v[i] ≥ v[i+1] (non‑increasing)."""
    return v[:-1] - v[1:]


def slsqp_beta_schedule(v0, betas, monotonic, maxiter, rng):
    """Run SLSQP on a sequence of β values."""
    v = np.clip(v0, 0.0, 1.0)
    best_v = v.copy()
    best_c = np.inf
    bounds = [(0.0, 1.0)] * v.size
    cons = [{'type': 'ineq', 'fun': monotonic_constraint}] if monotonic else []
    for beta in betas:
        try:
            res = minimize(
                smooth_max_obj,
                v,
                args=(beta,),
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False},
            )
        except Exception:
            continue
        v = np.clip(res.x, 0.0, 1.0)
        c = compute_c5_from_v(v)
        if c < best_c:
            best_c, best_v = c, v.copy()
    return best_v, best_c


def random_monotonic(m, rng):
    """Random non‑increasing vector in [0,1]^m."""
    return np.sort(rng.random(m))[::-1]


def random_unconstrained(m, rng):
    """Random vector in [0,1]^m (no order)."""
    return rng.random(m)


def hill_climb_local(v, rng, max_iters, scale):
    """Simple projected hill‑climb (maintains monotonicity)."""
    best_v = np.clip(v, 0.0, 1.0)
    best_c = compute_c5_from_v(best_v)
    m = best_v.size
    for _ in range(max_iters):
        dv = (rng.random(m) - 0.5) * scale
        cand = np.clip(best_v + dv, 0.0, 1.0)
        cand = np.sort(cand)[::-1]               # enforce monotonicity
        c = compute_c5_from_v(cand)
        if c < best_c:
            best_c, best_v = c, cand.copy()
    return best_v, best_c


# -------------------- main routine --------------------
