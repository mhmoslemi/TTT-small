```
def targeted_perturb(h, lag, rng, top_frac=0.2, scale=0.5):
    n = h.size
    if lag >=0:
        valid = np.arange(n - lag)
        prod = h[valid] * (1 - h[valid + lag])
    else:
        lag = -lag
        valid = np.arange(lag, n)
        prod = h[valid] * (1 - h[valid - lag])
    # select donor i from top of prod; choose j from bottom of prod (maybe different indices)
    k = max(1, int(len(valid)*top_frac))
    top_idx = valid[np.argpartition(-prod, k-1)[:k]]
    bot_idx = valid[np.argpartition(prod, k-1)[:k]]
    i = rng.choice(top_idx)
    # For donor i, which coordinate? It's i (the first component in product) for positive lag; for negative lag it's i (which is the second? Wait negative case we set valid = range(lag, n) and product = h[valid] * (1 - h[valid - lag]), where i = valid, partner = i - lag. So donor index is i.)
    # For receiver j, pick from bottom.
    j = rng.choice(bot_idx)
    if i == j:
        # pick random alternative
        j = rng.integers(0, n)
        if i == j:
            j = (j+1)%n
    # ensure epsilon feasible
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h
```

```
def compute_c5_corr_fft(h):
    n = h.size
    # length for FFT: next power of 2 >= 2*n
    L = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    corr_full = np.fft.irfft(H * G, n=L)[:2*n-1]

    c = np.max(corr_full) * (2.0 / n)
    return c, corr_full
```

```python
        if valid_len <= 0:
            return perturb_h(h, rng, scale)
        idx = np.arange(lag, n)
```

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize_one_beta(h0, beta, rng, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x

# ---------- Simulated annealing ----------
def anneal(start_h, rng, time_end, target_iters=200000):
    """Time‑aware SA returning best found (h, C5)."""
    h_cur = start_h.copy()
    c_cur, corr_cur = compute_c5_corr(h_cur)
    best_h, best_c = h_cur.copy(), c_cur
    n = h_cur.size
    it = 0
    while time.time() < time_end and it < target_iters:
        T = 0.02 * (1e-5 / 0.02) ** (it / target_iters)  # exponential cooling
        lag = np.argmax(corr_cur) - (n - 1)
        if rng.random() < 0.7:
            new_h = targeted_perturb(h_cur, lag, rng, top_frac=0.2, scale=0.5)
        else:
            new_h = perturb_h(h_cur, rng, scale=0.5)
        new_c, new_corr = compute_c5_corr(new_h)
        dE = new_c - c_cur
        if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            h_cur, c_cur, corr_cur = new_h, new_c, new_corr
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
        it += 1
    return best_h, best_c

# ---------- main run ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    # ---- initial vector ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
    else:
        n0 = 360
        h0 = random_feasible_vector(n0, rng)

    target_n = max(720, n0)
    if target_n != n0:
        old_x = (np.arange(n0) + 0.5) * (2.0 / n0)
        new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
        h = np.interp(new_x, old_x, h0)
        h = enforce_sum(h, target_n / 2.0, rng)
        n = target_n
    else:
        h = enforce_sum(h0, n0 / 2.0, rng)
        n = n0

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ---- main optimization (≈90% of budget) ----
    main_end = t0 + budget_s * 0.9
    beta_list = [5, 20, 80, 200, 500, 1000, 2000, 5000]

    while time.time() < main_end:
        # start point
        if rng.random() < 0.8:
            start = perturb_h(best_h, rng, scale=0.4)
        else:
            start = random_feasible_vector(n, rng)

        # Simulated annealing (≈60% of remaining time)
        now = time.time()
        rem = main_end - now
        if rem <= 0:
            break
        sa_end = now + rem * 0.6
        max_sa_iters = max(5000, int(200000 * (rem / (budget_s * 0.9))))
        sa_h, sa_c = anneal(start, rng, sa_end, target_iters=max_sa_iters)
        if sa_c < best_c:
            best_h, best_c = sa_h.copy(), sa_c

        # SLSQP refinement (use the leftover ~40% of this chunk)
        for beta in beta_list:
            if time.time() >= main_end:
                break
            try:
                ref_h = slsqp_optimize_one_beta(best_h, beta, rng, max_iter=300)
            except Exception:
                continue
            ref_c = compute_c5(ref_h)
            if ref_c < best_c:
                best_h, best_c = ref_h.copy(), ref_c

    # ---- final local hill‑climb (remaining time) ----
    final_end = t0 + budget_s
    h_cur = best_h.copy()
    cur_c = best_c
    step = 0.3
    while time.time() < final_end:
        new_h = perturb_h(h_cur, rng, scale=step)
        new_c = compute_c5(new_h)
        if new_c < cur_c:
            cur_c, h_cur = new_c, new_h
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
        step *= 0.9995

    # ---- final clean‑up ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
