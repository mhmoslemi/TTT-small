def project_onto_sum_and_box(v, target, tol=1e-12, max_iter=100):
    lo = -1.0
    hi = 1.0
    # expand bounds to ensure sum after clipping at extremes covers target.
    # At λ = -inf, clip(v_i-λ,0,1) -> 1 => sum = n.
    # At λ = +inf, -> 0 => sum =0.
    # So we can set lo = -max(v) maybe -1, hi = max(v) + 1.
    lo = -1.0
    hi = 2.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) < tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)
```

def multi_transfer(h, grads, rng, top_k=5, scale=0.5):
    # Choose top_k indices with highest grads where h>0
    eligible_pos = np.where((grads > 0) & (h > 1e-12))[0]
    if eligible_pos.size == 0:
        return h.copy()
    eligible_neg = np.where((grads < 0) & (h < 1 - 1e-12))[0]
    if eligible_neg.size == 0:
        return h.copy()
    pos_idx = eligible_pos[np.argsort(-grads[eligible_pos])[:top_k]]
    neg_idx = eligible_neg[np.argsort(grads[eligible_neg])[:top_k]]
    # compute total possible
    total_pos = h[pos_idx].sum()
    total_neg = (1 - h[neg_idx]).sum()
    max_transfer = min(total_pos, total_neg)
    if max_transfer <= 0:
        return h.copy()
    transfer = max_transfer * scale * rng.random()
    # allocate proportionally to grads magnitude
    pos_weights = grads[pos_idx]
    neg_weights = -grads[neg_idx]
    pos_weights = pos_weights / pos_weights.sum()
    neg_weights = neg_weights / neg_weights.sum()
    # compute amount taken from each pos and added to each neg
    nh = h.copy()
    for i, w in zip(pos_idx, pos_weights):
        take = w * transfer
        nh[i] -= take
    for j, w in zip(neg_idx, neg_weights):
        add = w * transfer
        nh[j] += add
    return nh

def slsqp_optimize(h0, beta, max_iter=600):
    n = h0.size
    target = n/2.0
    def sum_eq_constraint(x):
        return np.sum(x) - target
    def softmax_obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        # compute (m + log(sum(exp(bc - m))))/beta
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    bounds = [(0.0,1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    x = project_sum_and_box(x, target)
    return x

import time
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# Constraint handling: projection onto [0,1] and sum = target
def project_sum_and_box(v, target, tol=1e-12, max_iter=60):
    """Euclidean projection of v onto {x: 0<=x<=1, sum(x)=target}."""
    # First clip to the box; if sum already correct we are done.
    x = np.clip(v, 0.0, 1.0)
    s = x.sum()
    if abs(s - target) <= tol:
        return x
    # Binary search for Lagrange multiplier λ
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Uniform random point in the box, then projected to the hyperplane."""
    v = rng.random(n)
    return project_sum_and_box(v, n / 2.0)


# ------------------------------------------------------------
# Core problem utilities
def compute_c5(h):
    """Exact discrete C5 for a vector h."""
    corr = np.correlate(h, 1 - h, mode="full")
    return corr.max() * (2.0 / h.size)


def gradient_shift(h, s):
    """Gradient d/dh of the overlap for a single shift s."""
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[: n - s] += 1 - h[s:]
        grad[s:] -= h[: n - s]
    else:
        s = -s
        grad[s:] += 1 - h[: n - s]
        grad[: n - s] -= h[s:]
    return grad


def perturb_random(h, rng, scale=0.2):
    """Transfer a random amount from one coordinate to another."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


# ------------------------------------------------------------
# Projected (sub)gradient descent on the max‑overlap objective
def pgd_optimize(
    h0,
    max_iter=1200,
    init_alpha=0.7,
    target=None,
    rng=None,
    time_limit=None,
):
    """Sub‑gradient step with projection. Returns (h, C5)."""
    n = h0.size
    if target is None:
        target = n / 2.0
    h = h0.copy()
    cur_c = compute_c5(h)
    alpha = init_alpha
    no_improve = 0
    start = time.time()
    for _ in range(max_iter):
        # optional external time cap
        if time_limit is not None and time.time() - start > time_limit:
            break

        # worst shift(s) and summed gradient
        corr = np.correlate(h, 1 - h, mode="full")
        max_val = corr.max()
        idxs = np.where(np.abs(corr - max_val) < 1e-8)[0]
        shifts = idxs - (n - 1)
        grad = np.zeros_like(h)
        for s in shifts:
            grad += gradient_shift(h, s)

        d = -grad
        if np.linalg.norm(d) < 1e-12:
            break

        step = alpha
        improved = False
        for _ in range(12):  # line search
            h_new = project_sum_and_box(h + step * d, target)
            new_c = compute_c5(h_new)
            if new_c < cur_c - 1e-12:
                h = h_new
                cur_c = new_c
                improved = True
                step *= 1.2  # try to enlarge a bit next iteration
                break
            step *= 0.5
        if not improved:
            no_improve += 1
            alpha *= 0.5
            if alpha < 1e-6 or no_improve >= 30:
                break
        else:
            no_improve = 0
            alpha = step
    return h, cur_c


# ------------------------------------------------------------
# SLSQP polishing (soft‑max surrogate of the max)
def slsqp_optimize(h0, beta, max_iter=800):
    n = h0.size
    target = n / 2.0

    def sum_eq(x):
        return np.sum(x) - target

    def softmax_obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        # log‑sum‑exp trick
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_sum_and_box(x, target)
