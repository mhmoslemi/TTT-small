import time
import numpy as np
from scipy.optimize import minimize

# -------------------- basic utilities --------------------
def compute_c5(h):
    """Return C5 for a discrete h (array)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(arr, rng):
    """Adjust arr (in place) so that sum(arr) == len(arr)/2 exactly."""
    target = arr.size / 2.0
    diff = target - arr.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, arr.size)
        if diff > 0:
            add = min(diff, 1.0 - arr[i])
        else:
            add = max(diff, -arr[i])
        arr[i] += add
        diff = target - arr.sum()

def gradient_refine(h, rng, step=0.02, max_iter=12):
    """Projected gradient descent against the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]        # + (1 - h[i+k])
            g[k:] -= h[:upper]               # - h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    """SLSQP on the smooth‑max surrogate for a full‑size h."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def build_L(n, p):
    """Return interpolation matrix L (n x p) mapping monotone knot values to a symmetric h."""
    # positions (midpoints)
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    ds = np.abs(xs - 1.0)                     # distance to centre, in [0,1]
    t = np.linspace(0.0, 1.0, p)              # knot locations
    scale = (p - 1)
    pos = ds * scale
    lower = np.floor(pos).astype(int)
    lower = np.clip(lower, 0, p - 2)
    s = pos - lower
    L = np.zeros((n, p))
    L[np.arange(n), lower] = 1.0 - s
    L[np.arange(n), lower + 1] = s
    return L

def local_random_search(h, rng, deadline):
    """Small random swap hill‑climb on the full vector."""
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    step = 0.015
    n = h.size
    while time.time() < deadline:
        i, j = rng.choice(n, 2, replace=False)
        delta = (rng.random() - 0.5) * step
        if not (0.0 <= cur_h[i] - delta <= 1.0 and 0.0 <= cur_h[j] + delta <= 1.0):
            continue
        cur_h[i] -= delta
        cur_h[j] += delta
        c = compute_c5(cur_h)
        if c < best_c:
            best_c, best_h = c, cur_h.copy()
        if c < cur_c:
            cur_c = c
        else:
            # revert
            cur_h[i] += delta
            cur_h[j] -= delta
        step *= 0.99999
    return best_h, best_c
