def simulated_annealing(h0, rng, max_time, init_T=0.1, final_T=1e-4):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    T = init_T
    start = time.time()
    # define cooling schedule: exponential based on remaining time
    while time.time() - start < max_time:
        # update temperature based on elapsed fraction
        elapsed = time.time() - start
        frac = elapsed / max_time
        T = init_T * (final_T / init_T) ** frac  # exponential cooling
        # propose move
        new_h = perturb_h(h, rng, scale=0.5)  # maybe bigger moves early
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = new_h
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Helper utilities ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth max of the overlap using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for a step function."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)


def project_to_feasible(v, target):
    """Project onto {0≤h≤1, Σh=target} by shifting and clipping."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Random vector with 0≤h≤1 and Σh=n/2."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_to_feasible(v, n / 2.0)


def perturb_h(h, rng, scale=0.2):
    """Transfer ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.3):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay = max(1, max_iters // 8)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h, h = c, new_h.copy(), new_h
        if (it + 1) % decay == 0:
            step = max(step * 0.85, 1e-4)
    return best_h, best_c


def simulated_annealing(h0, rng, max_time, init_T=0.1, final_T=1e-4):
    """Simple SA with exponential cooling."""
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    start = time.time()
    while time.time() - start < max_time:
        elapsed = time.time() - start
        frac = elapsed / max_time
        T = init_T * (final_T / init_T) ** frac
        new_h = perturb_h(h, rng, scale=0.5)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h, cur_c = new_h, new_c
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
    return best_h, best_c


def slsqp_optimize(h0, beta, rng, max_iter=500):
    """SLSQP refinement using soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # Project back onto the exact hyperplane
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def de_optimize_custom(n, rng, max_time, upsample_func=None):
    """Differential Evolution with projection onto the feasible set."""
    target = n / 2.0
    popsize = max(30, int(0.15 * n))
    pop = []

    if upsample_func is not None:
        vec = upsample_func()
        vec = project_to_feasible(vec, target)
        pop.append(vec)

    while len(pop) < popsize:
        pop.append(random_feasible_vector(n, rng))

    pop = np.array(pop)
    scores = np.array([compute_c5(ind) for ind in pop])
    best_idx = scores.argmin()
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]

    start = time.time()
    while time.time() - start < max_time:
        for i in range(popsize):
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, 3, replace=False)
            F = 0.5 + rng.random() * 0.5          # 0.5–1.0
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            mutant = project_to_feasible(mutant, target)

            CR = 0.9
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(0, n)] = True
            trial = np.where(cross, mutant, pop[i])
            trial = project_to_feasible(trial, target)

            trial_score = compute_c5(trial)
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_h = trial.copy()
        # optional early break (not used)
    return best_h, best_score


# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    total_budget = budget_s

    # Choose discretisation size
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        upsample_initial = None  # not needed; we already have size n
    else:
        n = 240

    # Time allocations (approximate)
    de_time = total_budget * 0.55
    slsqp_time = total_budget * 0.25
    sa_time = total_budget * 0.15
    hill_time = total_budget - de_time - slsqp_time - sa_time - 2.0
    if hill_time < 1.0:
        hill_time = 1.0

    # ----- Differential Evolution -----
    def upsample_func():
        # only used when we decide to upsample a smaller initial vector;
        # here we already have the right size, so return it directly.
        return np.array(initial_h_values, dtype=float)

    h_de, c_de = de_optimize_custom(
        n,
        rng,
        de_time,
        upsample_func if ("initial_h_values" in globals()) else None,
    )
    best_h, best_c = h_de.copy(), c_de

    # ----- SLSQP refinement (soft‑max) -----
    slsqp_start = time.time()
    betas = (20.0, 100.0, 500.0, 2000.0)
    for beta in betas:
        if time.time() - start_time > total_budget * 0.95:
            break
        h_s = slsqp_optimize(best_h, beta, rng, max_iter=400)
        c_s = compute_c5(h_s)
        if c_s < best_c:
            best_h, best_c = h_s.copy(), c_s
        # time guard
        if time.time() - slsqp_start > slsqp_time:
            break

    # ----- Simulated Annealing -----
    if time.time() - start_time < total_budget * 0.975:
        h_sa, c_sa = simulated_annealing(best_h, rng, sa_time)
        if c_sa < best_c:
            best_h, best_c = h_sa.copy(), c_sa

    # ----- Deterministic hill‑climb refinement -----
    if time.time() - start_time < total_budget * 0.99:
        h_hc, c_hc = hill_climb(best_h, rng, max_iters=3000, start_step=0.2)
        if c_hc < best_c:
            best_h, best_c = h_hc.copy(), c_hc

    # ----- Final feasibility correction -----
    best_h = np.clip(best_h, 0.0, 1.0)
    target = n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
