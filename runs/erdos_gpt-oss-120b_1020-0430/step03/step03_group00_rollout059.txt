def simulated_annealing(h0, time_limit, rng):
    # returns best_h, best_c found within time_limit seconds
    n = h0.size
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    cur_h = h0.copy()
    cur_c = best_c
    # parameters
    T0 = 0.02   # Since c differences maybe 0.001-0.01, could start 0.02
    Tmin = 0.0001
    alpha = 0.97
    T = T0
    # compute gradient once for targeted moves each step
    start_time = time.time()
    while time.time() - start_time < time_limit:
        # propose move
        if rng.random() < 0.7:
            # targeted move
            # compute gradient for current worst shift
            # using gradient_worst_shift function
            grad = gradient_worst_shift(cur_h)
            # indices
            pos = np.where((grad > 1e-9) & (cur_h > 1e-12))[0]
            neg = np.where((grad < -1e-9) & (cur_h < 1 - 1e-12))[0]
            if pos.size > 0 and neg.size > 0:
                i = rng.choice(pos)
                j = rng.choice(neg)
                max_eps = min(cur_h[i], 1 - cur_h[j])
                if max_eps > 0:
                    eps = max_eps * 0.3 * rng.random()
                    new_h = cur_h.copy()
                    new_h[i] -= eps
                    new_h[j] += eps
                else:
                    new_h = perturb_random(cur_h, rng, scale=0.3)
            else:
                new_h = perturb_random(cur_h, rng, scale=0.3)
        else:
            new_h = perturb_random(cur_h, rng, scale=0.3)

        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < math.exp(-delta / T):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
        # cooling schedule
        # maybe after fixed number of proposals per temperature

def enforce_sum(h, target, rng):
    """Clip to [0,1] and adjust to make sum exactly target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # if diff small -> done
    if abs(diff) < 1e-12:
        return h
    # adjust by picking random indices
    # Use loop but not too many iterations.
    max_iter = 5000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:
            # need increase some entries
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i])
            # scaled random fraction to avoid overshooting perhaps
            add *= rng.random()
            h[i] += add
            diff -= add
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i])
            sub *= rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    # fallback if still not satisfied:
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        if diff > 0:
            h[i] = min(1.0, h[i] + diff)
        else:
            h[i] = max(0.0, h[i] + diff)
    return h

def perturb_random(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    # ensure they are distinct and possible transfer
    max_attempts = 10
    for _ in range(max_attempts):
        if i != j and h[i] > 0 and h[j] < 1:
            max_eps = min(h[i], 1 - h[j])
            if max_eps > 0:
                eps = max_eps * scale * rng.random()
                new_h = h.copy()
                new_h[i] -= eps
                new_h[j] += eps
                return new_h
        i = rng.integers(0, n)
        j = rng.integers(0, n)
    # fallback no change
    return h.copy()

def perturb_targeted(h, grad, rng, scale=0.2):
    pos = np.where((grad > 1e-9) & (h > 1e-12))[0]
    neg = np.where((grad < -1e-9) & (h < 1 - 1e-12))[0]
    if pos.size == 0 or neg.size == 0:
        return perturb_random(h, rng, scale)
    i = rng.choice(pos)
    j = rng.choice(neg)
    max_eps = min(h[i], 1 - h[j])
    if max_eps <= 0:
        return perturb_random(h, rng, scale)
    eps = max_eps * scale * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def softmax_obj(x):
    # x is vector
    corr = np.correlate(x, 1 - x, mode='full')
    # multiply by beta
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # compute log-sum-exp for stability
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    # Use minimize from scipy.optimize
    res = minimize(
        softmax_obj,
        h0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False}
    )
    x = np.clip(res.x, 0.0, 1.0)
    # fix sum to target
    x = enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

import time
import math
import numpy as np
from scipy.optimize import minimize

# ---------- utility functions ----------
def enforce_sum(h, target, rng):
    """Clip to [0,1] and adjust entries so that sum equals target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    max_iter = 5000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:  # need to increase
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i])
            add *= rng.random()
            h[i] += add
            diff -= add
        else:        # need to decrease
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i])
            sub *= rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        if diff > 0:
            h[i] = min(1.0, h[i] + diff)
        else:
            h[i] = max(0.0, h[i] + diff)
    return h

def random_feasible_vector(n, rng):
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)

def compute_c5(h):
    """Discrete C₅ for a step function h on [0,2] with n samples."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

# ---------- perturbations ----------
def perturb_random(h, rng, scale=0.2):
    n = h.size
    for _ in range(10):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        if h[i] <= 0 or h[j] >= 1:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = max_eps * scale * rng.random()
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        return new_h
    return h.copy()

def gradient_worst_shift(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr)
    k = idx - (n - 1)
    grad = np.ones(n)
    if k >= 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    else:
        kk = -k
        grad[kk:] -= h[: n - kk]
        grad[: n - kk] -= h[kk:]
    return grad, k

def perturb_targeted(h, grad, rng, scale=0.2):
    pos = np.where((grad > 1e-9) & (h > 1e-12))[0]
    neg = np.where((grad < -1e-9) & (h < 1.0 - 1e-12))[0]
    if pos.size == 0 or neg.size == 0:
        return perturb_random(h, rng, scale)
    i = rng.choice(pos)
    j = rng.choice(neg)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_random(h, rng, scale)
    eps = max_eps * scale * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

# ---------- hill climbing ----------
def hill_climb_targeted(h_start, rng, max_iters, start_step, target_sum):
    h = h_start.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)

    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr)
        max_corr = corr[idx]
        c_current = max_corr * (2.0 / n)
        k = idx - (n - 1)

        grad = np.ones(n)
        if k >= 0:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
        else:
            kk = -k
            grad[kk:] -= h[: n - kk]
            grad[: n - kk] -= h[kk:]

        new_h = perturb_targeted(h, grad, rng, scale=step)
        new_h = enforce_sum(new_h, target_sum, rng)
        c_new = compute_c5(new_h)

        if c_new < best_c:
            best_c = c_new
            best_h = new_h.copy()
        if c_new < c_current:
            h = new_h

        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)

    best_h = enforce_sum(best_h, target_sum, rng)
    return best_h, best_c

# ---------- SLSQP refinement ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

# ---------- simulated annealing ----------
def simulated_annealing(h0, time_limit, rng):
    n = h0.size
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    cur_h = h0.copy()
    cur_c = best_c

    T = 0.02
    Tmin = 1e-4
    alpha = 0.96
    start = time.time()
    while time.time() - start < time_limit:
        if rng.random() < 0.6:
            grad, _ = gradient_worst_shift(cur_h)
            new_h = perturb_targeted(cur_h, grad, rng, scale=0.3)
        else:
            new_h = perturb_random(cur_h, rng, scale=0.3)

        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
        T = max(T * alpha, Tmin)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    return best_h, best_c

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # initial solution (may be provided)
    if "initial_h_values" in globals():
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = random_feasible_vector(180, rng)
    best_h = enforce_sum(np.clip(best_h, 0.0, 1.0), best_h.size / 2.0, rng)
    best_c = compute_c5(best_h)

    n_initial = best_h.size
    # progressive resolutions (always end at 720 if time permits)
    resolutions = []
    if n_initial <= 180:
        resolutions = [180, 360, 720]
    elif n_initial <= 360:
        resolutions = [n_initial, 720] if n_initial != 720 else [720]
    else:
        resolutions = [n_initial]
    resolutions = sorted(set(resolutions))

    for n_target in resolutions:
        if best_h.size != n_target:
            # up‑sample by linear interpolation
            xs_old = (np.arange(best_h.size) + 0.5) * (2.0 / best_h.size)
            xs_new = (np.arange(n_target) + 0.5) * (2.0 / n_target)
            best_h = np.interp(xs_new, xs_old, best_h)
            best_h = enforce_sum(np.clip(best_h, 0.0, 1.0), n_target / 2.0, rng)
            best_c = compute_c5(best_h)

        # parameters depending on resolution
        if n_target == 180:
            restarts, max_iters, start_step = 4, 3000, 0.5
        elif n_target == 360:
            restarts, max_iters, start_step = 3, 3000, 0.4
        else:  # 720
            restarts, max_iters, start_step = 2, 4000, 0.3

        beta_schedule = [5.0, 20.0, 80.0, 200.0, 800.0, 2000.0]

        for r in range(restarts):
            if time.time() - start_time > budget_s * 0.96:
                break
            start_vec = best_h.copy() if r == 0 else perturb_random(best_h, rng, scale=0.3)
            h_local, c_local = hill_climb_targeted(
                start_vec, rng, max_iters=max_iters,
                start_step=start_step, target_sum=n_target / 2.0
            )
            if c_local < best_c:
                best_h, best_c = h_local.copy(), c_local

            for beta in beta_schedule:
                if time.time() - start_time > budget_s * 0.985:
                    break
                try:
                    h_ref = slsqp_optimize(best_h, beta, max_iter=500)
                except Exception:
                    continue
                c_ref = compute_c5(h_ref)
                if c_ref < best_c:
                    best_h, best_c = h_ref.copy(), c_ref

            # short simulated‑annealing burst
            time_left = budget_s - (time.time() - start_time)
            if time_left > 1.0:
                sa_time = min(3.0, time_left * 0.3)
                if sa_time > 0.2:
                    h_sa, c_sa = simulated_annealing(best_h, sa_time, rng)
                    if c_sa < best_c:
                        best_h, best_c = h_sa.copy(), c_sa

        # final local refinement for this resolution if time permits
        if time.time() - start_time < budget_s * 0.995:
            extra_iters = int(1500 * (budget_s - (time.time() - start_time)) / budget_s)
            if extra_iters > 0:
                h_extra, c_extra = hill_climb_targeted(
                    best_h, rng, max_iters=extra_iters,
                    start_step=start_step * 0.5,
                    target_sum=n_target / 2.0,
                )
                if c_extra < best_c:
                    best_h, best_c = h_extra.copy(), c_extra

    # final micro‑search
    while time.time() - start_time < budget_s * 0.999:
        cand = perturb_random(best_h, rng, scale=0.02)
        c_cand = compute_c5(cand)
        if c_cand < best_c:
            best_h, best_c = cand, c_cand

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
