def simulated_annealing(h, rng, time_limit):
    n = h.size
    target = n//2
    best_h = h.copy()
    best_c = compute_c5_fft(best_h)
    cur_h = h.copy()
    cur_c = best_c
    start = time.time()
    iter = 0
    # schedule parameters
    init_temp = best_c  # maybe start_temp as best_c 0.38
    final_temp = 1e-4
    # we can compute a decay factor such that after max_iters it reaches final_temp.
    # we don't know max_iters; we can approximate based on time.

    # We'll plan for say 200k iterations max. compute decay = (final_temp / init_temp) ** (1 / max_iters)

    max_iters = 300000
    decay = (final_temp / init_temp) ** (1.0 / max_iters)

    temp = init_temp

    while time.time() - start < time_limit and iter < max_iters:
        # propose swap between 1 and 0
        # random choose index of 1
        ones = np.where(cur_h > 0.5)[0]  # indices of 1s
        zeros = np.where(cur_h < 0.5)[0]
        # Choose random 1 and 0 to swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        new_h = cur_h.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]
        new_c = compute_c5_fft(new_h)
        delta = new_c - cur_c
        if delta <= 0 or rng.random() < np.exp(-delta / temp):
            cur_h = new_h
            cur_c = new_c
            # update best
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        # schedule update
        temp *= decay
        iter += 1

    return best_h, best_c

def compute_c5_fft(h):
    n = h.shape[0]
    # length of full correlation
    m = 2*n - 1
    N = 1 << (m-1).bit_length()
    hf = np.fft.rfft(h, N)
    gf = np.fft.rfft(1.0 - h, N)
    cor = np.fft.irfft(np.conj(hf) * gf, N)[:m]
    return np.max(cor) * (2.0 / n)

def make_binary_feasible(v, target, rng):
    # round to 0 or 1
    h = np.rint(v).astype(int)
    diff = int(target - h.sum())
    if diff > 0:
        # need to set diff zeros to 1
        zeros = np.where(h == 0)[0]
        if zeros.size < diff:
            # fallback: random, but shouldn't happen
            zeros = np.arange(h.size)
        idx = rng.choice(zeros, size=diff, replace=False)
        h[idx] = 1
    elif diff < 0:
        ones = np.where(h == 1)[0]
        idx = rng.choice(ones, size=-diff, replace=False)
        h[idx] = 0
    return h.astype(float)

def hill_climb_binary(h, rng, iterations):
    best = h.copy()
    best_c = compute_c5_fft(best)
    cur = h.copy()
    cur_c = best_c
    n = h.size
    for _ in range(iterations):
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i: j += 1
        if cur[i] == cur[j]:
            continue
        # swap
        new = cur.copy()
        new[i], new[j] = new[j], new[i]
        new_c = compute_c5_fft(new)
        if new_c < cur_c:
            cur = new
            cur_c = new_c
            if new_c < best_c:
                best = new.copy()
                best_c = new_c
    return best, best_c

        j = rng.integers(0, n-1)
        if j >= i: j += 1
        if h[i] == h[j]:
            continue
        # compute max allowed epsilon
        max_pos = min(h[i], 1.0 - h[j])
        max_neg = min(h[j], 1.0 - h[i])
        if max_pos + max_neg == 0:

        eps *= step

            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

def slsqp_optimize(h0, beta, max_iter=200):
    from scipy.optimize import minimize
    n = h0.size
    target = n / 2.0
    def sum_eq(x):
        return np.sum(x) - target
    def softmax_obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0,1.0)
    # enforce sum exactly
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # adjust a random index
        for _ in range(1000):
            if abs(diff) < 1e-12:
                break
            i = np.random.randint(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Fast correlation using FFT (linear, non‑circular)
# ----------------------------------------------------------------------
def compute_c5_fast(h: np.ndarray) -> float:
    n = h.size
    m = 2 * n - 1
    N = 1 << (m - 1).bit_length()
    hf = np.fft.rfft(h, N)
    gf = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(np.conj(hf) * gf, N)[:m]
    return np.max(corr) * (2.0 / n)


# ----------------------------------------------------------------------
# Exact C₅ (used for the final reported bound)
# ----------------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


# ----------------------------------------------------------------------
# Project a vector back onto the feasible simplex (0≤h≤1, Σh=n/2)
# ----------------------------------------------------------------------
def enforce_sum(v: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    diff = target - v.sum()
    n = v.size
    # greedy redistribution
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - v[i])
        else:
            add = max(diff, -v[i])
        if abs(add) < 1e-12:
            continue
        v[i] += add
        diff = target - v.sum()
    # tiny cleanup if needed
    if abs(diff) > 1e-12:
        for _ in range(1000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - v[i])
            else:
                add = max(diff, -v[i])
            if abs(add) < 1e-12:
                continue
            v[i] += add
            diff = target - v.sum()
    return np.clip(v, 0.0, 1.0)

# Random feasible (continuous) vector
# ----------------------------------------------------------------------
def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    target = n / 2.0
    v = rng.exponential(1.0, size=n)
    v = v / v.sum() * target
    v = np.clip(v, 0.0, 1.0)
    return enforce_sum(v, target, rng)


# ----------------------------------------------------------------------
# Small two‑coordinate transfer; preserves the sum exactly
# ----------------------------------------------------------------------
def perturb_h_transfer(h: np.ndarray, rng: np.random.Generator, scale: float = 0.3) -> np.ndarray:
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if j >= i:
        j += 1
    max_pos = min(h[i], 1.0 - h[j])
    max_neg = min(h[j], 1.0 - h[i])
    if max_pos + max_neg == 0:
        return h.copy()
    eps = (rng.random() * (max_pos + max_neg)) - max_neg
    eps *= scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return np.clip(new_h, 0.0, 1.0)


# ----------------------------------------------------------------------
# Simple hill‑climb (only improving moves)
# ----------------------------------------------------------------------
def hill_climb_continuous(start: np.ndarray, rng: np.random.Generator,
                         max_iters: int = 20000, start_step: float = 0.4):
    h = start.copy()
    best_h = h.copy()
    best_c = compute_c5_fast(h)
    step = start_step
    n = h.size
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        if h[i] == h[j]:
            continue
        max_pos = min(h[i], 1.0 - h[j])
        max_neg = min(h[j], 1.0 - h[i])
        if max_pos + max_neg == 0:
            continue
        eps = (rng.random() * (max_pos + max_neg)) - max_neg
        eps *= step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5_fast(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


# ----------------------------------------------------------------------
# SLSQP on the smooth log‑sum‑exp surrogate (continuation in β)
# ----------------------------------------------------------------------
def slsqp_optimize(h0: np.ndarray, beta: float, max_iter: int = 200) -> np.ndarray:
    n = h0.size
    target = n / 2.0

    def sum_eq(x):
        return np.sum(x) - target

    def softmax_obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # tiny exact‑sum repair
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            if abs(add) < 1e-12:
                continue
            x[i] += add
            diff = target - x.sum()
    return x


# ----------------------------------------------------------------------
# Build an initial population (continuous, respects the constraints)
# ----------------------------------------------------------------------
def initial_population(n: int, rng: np.random.Generator, size: int = 30):
    pop = []
    for _ in range(size):
        h = random_feasible_vector(n, rng)
        pop.append((h, compute_c5_fast(h)))
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size != n:
            # linear interpolation to required size
            xs_old = np.linspace(0, 2, h0.size, endpoint=False)
            xs_new = np.linspace(0, 2, n, endpoint=False)
            h0 = np.interp(xs_new, xs_old, h0)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
        pop.append((h0, compute_c5_fast(h0)))
    pop.sort(key=lambda x: x[1])
    return pop[:size]


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # start with 180 points; later we may double resolution
    n = 180
    target = n / 2.0

    # ------------------------------------------------------------------
    # Evolutionary search on the 180‑point grid
    # ------------------------------------------------------------------
    pop = initial_population(n, rng, size=30)
    best_h, best_c_fast = pop[0][0].copy(), pop[0][1]

    while time.time() - start < budget_s * 0.85:
        # tournament selection of a parent
        idxs = rng.choice(len(pop), size=3, replace=False)
        parent = min((pop[i] for i in idxs), key=lambda x: x[1])[0]

        # mutation with a random step scale
        child = perturb_h_transfer(parent, rng, scale=rng.uniform(0.1, 0.5))
        child_c_fast = compute_c5_fast(child)

        # replace worst individual if improvement
        if child_c_fast < pop[-1][1]:
            pop[-1] = (child, child_c_fast)
            pop.sort(key=lambda x: x[1])

        # occasional recombination of two elite members
        if rng.random() < 0.15:
            i1 = rng.integers(0, len(pop) // 2)
            i2 = rng.integers(0, len(pop) // 2)
            p1, p2 = pop[i1][0], pop[i2][0]
            child = (p1 + p2) / 2.0
            child = np.clip(child, 0.0, 1.0)
            child = enforce_sum(child, target, rng)
            child_c_fast = compute_c5_fast(child)
            if child_c_fast < pop[-1][1]:
                pop[-1] = (child, child_c_fast)
                pop.sort(key=lambda x: x[1])

        if child_c_fast < best_c_fast:
            best_c_fast, best_h = child_c_fast, child.copy()

    # ------------------------------------------------------------------
    # Intensive local refinement (hill‑climb)
    # ------------------------------------------------------------------
    time_left = budget_s - (time.time() - start)
    if time_left > 5.0:
        iters = int(min(80000, time_left * 3000))
        best_h, best_c_fast = hill_climb_continuous(best_h, rng,
                                                    max_iters=iters,
                                                    start_step=0.4)

    # ------------------------------------------------------------------
    # SLSQP continuation (smooth surrogate)
    # ------------------------------------------------------------------
    for beta in (10.0, 40.0, 120.0, 400.0, 800.0):
        if time.time() - start > budget_s * 0.95:
            break
        best_h = slsqp_optimize(best_h, beta, max_iter=200)

    # ------------------------------------------------------------------
    # Optional up‑sampling to a finer grid if time permits
    # ------------------------------------------------------------------
    if time.time() - start < budget_s * 0.90 and n * 2 <= 360:
        up_h = np.repeat(best_h, 2)
        n_up = up_h.size
        target_up = n_up / 2.0
        up_h = enforce_sum(up_h, target_up, rng)

        # quick local search on the finer grid
        iters = int(min(60000, (budget_s - (time.time() - start)) * 2500))
        up_h, _ = hill_climb_continuous(up_h, rng, max_iters=iters, start_step=0.25)

        # final SLSQP passes on the up‑sampled candidate
        for beta in (10.0, 40.0, 120.0, 400.0, 800.0):
            if time.time() - start > budget_s * 0.99:
                break
            up_h = slsqp_optimize(up_h, beta, max_iter=150)

        # evaluate the refined fine‑grid solution
        final_c_up = compute_c5(up_h)
        final_c_cur = compute_c5(best_h)
        if final_c_up < final_c_cur:
            best_h, n = up_h, n_up

    # ------------------------------------------------------------------
    # Final clean‑up and exact evaluation
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
