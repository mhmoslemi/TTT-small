def project_onto_bounds_sum(v, target):
    # v is array
    # want min ||x - v||^2 such that 0<=x<=1, sum=target.
    # Equivalent to x_i = clip(v_i + lambda, 0, 1) where lambda chosen to satisfy sum.

        s = x.sum()
        if s > target:
            high = mid
        else:
            low = mid
    return np.clip(v + (low+high)/2, 0, 1)

import time
import math
import numpy as np
from scipy.optimize import minimize

# ---------- core utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function h (dx = 2 / len(h))."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)


def project_to_feasible(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of v onto {0≤x≤1, sum x = target}."""
    low = -np.max(v) - 1.0
    high = 1.0 - np.min(v) + 1.0
    for _ in range(50):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            high = mid
        else:
            low = mid
    return np.clip(v + (low + high) * 0.5, 0.0, 1.0)


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random feasible vector (0≤h≤1, sum = n/2)."""
    v = rng.random(n)
    return project_to_feasible(v, n / 2.0)


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.1) -> np.ndarray:
    """Add Gaussian noise and re‑project onto feasible set."""
    v = h + rng.normal(scale=scale, size=h.shape)
    return project_to_feasible(v, h.size / 2.0)


# ---------- SLSQP with smooth max ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp trick
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(x0: np.ndarray, beta: float, max_iter: int = 200) -> np.ndarray:
    """Run a few SLSQP iterations on a smooth approximation of max."""
    global CURRENT_TARGET, CURRENT_BETA
    n = x0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        _softmax_obj,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_to_feasible(x, CURRENT_TARGET)


# ---------- gradient‑based search ----------
def gradient_of_shifts(h: np.ndarray, shifts: np.ndarray) -> np.ndarray:
    """Gradient of max‑correlation (without the dx factor)."""
    n = h.size
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            if n - k > 0:
                grad[: n - k] += 1 - h[k:]
                grad[k:] -= h[: n - k]
        else:
            kk = -k
            if n - kk > 0:
                grad[kk:] += 1 - h[: n - kk]
                grad[: n - kk] -= h[kk:]
    return grad
