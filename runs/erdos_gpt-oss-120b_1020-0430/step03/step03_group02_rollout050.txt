def project_box_sum(h, target, rng):
    h = np.clip(h,0.0,1.0)
    diff = target - h.sum()
    # while diff not small
    # Using vectorized approach: find indices where we can add or subtract.
    # We'll loop but limited iterations.
    for _ in range(100):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # can increase entries where h < 1
            mask = h < 1 - 1e-12
            if not np.any(mask):
                break
            # compute capacity = (1 - h) for each; total capacity sumcap
            cap = 1 - h[mask]
            total_cap = cap.sum()
            if total_cap < diff - 1e-12:
                # fill all to 1; diff remains
                h[mask] = 1.0
                diff = target - h.sum()
                continue
            # distribute proportionally
            # For simplicity: add diff * cap / total_cap to each.
            add = diff * cap / total_cap
            h[mask] += add
            diff = target - h.sum()
        else:
            mask = h > 1e-12
            if not np.any(mask):
                break
            cap = h[mask]  # amount we can subtract
            total_cap = cap.sum()
            if total_cap < -diff - 1e-12:
                h[mask] = 0.0
                diff = target - h.sum()
                continue
            sub = (-diff) * cap / total_cap
            h[mask] -= sub
            diff = target - h.sum()
    return h

def enforce_sum(h, target, rng=None):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # iterative projection
    for _ in range(100):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            cap = 1.0 - h[mask]
            total = cap.sum()
            if total <= diff:
                h[mask] = 1.0
                diff = target - h.sum()
                continue
            add = diff * cap / total
            h[mask] += add
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            cap = h[mask]
            total = cap.sum()
            if total <= -diff:
                h[mask] = 0.0
                diff = target - h.sum()
                continue
            sub = -diff * cap / total
            h[mask] -= sub
        diff = target - h.sum()
    # final small correction with RNG if available
    if rng is not None and abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, len(h))
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h
```

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]          # subtract h_{i+k}
        grad[k:]       -= h[: n - k]    # subtract h_{i‑k}
    return grad

def softmax_obj_and_grad(h, beta):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    # compute weights safely
    w = np.exp(beta * (corr - max_corr))
    sum_w = np.sum(w)
    obj = max_corr + np.log(sum_w) / beta
    # compute weighted gradient via convolution
    # w is length 2n-1
    # term1 = conv(1-h, w)
    term1 = np.convolve(1 - h, w, mode='full')
    # term2 = conv(h, w_rev)
    term2 = np.convolve(h, w[::-1], mode='full')
    offset = len(w) - 1  # 2n-2
    grad = (term1[offset:offset + n] - term2[offset:offset + n]) / sum_w
    return obj, grad
```

import time, numpy as np

def compute_c5(h):
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng=None):
    """Project h onto 0≤h≤1 and Σh = target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # vectorised redistribution
    for _ in range(100):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            cap = 1.0 - h[mask]
            total = cap.sum()
            if total <= diff:
                h[mask] = 1.0
                diff = target - h.sum()
                continue
            add = diff * cap / total
            h[mask] += add
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            cap = h[mask]
            total = cap.sum()
            if total <= -diff:
                h[mask] = 0.0
                diff = target - h.sum()
                continue
            sub = -diff * cap / total
            h[mask] -= sub
        diff = target - h.sum()
    # tiny random fix (if needed)
    if rng is not None and abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, h.size)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Random feasible h (0≤h≤1, Σh=n/2)."""
    h = rng.random(n)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    h *= target / h.sum()
    h = enforce_sum(h, target, rng)
    return h


def worst_gradient(h):
    """Gradient of the worst‑shift overlap O_k = Σ h_i(1‑h_{i+k})."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h, grad, rng, scale=0.5):
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h_start, rng, max_iters=2000, step_scale=0.5):
    """Hill‑climb using gradient‑directed transfers."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def softmax_obj_and_grad(h, beta):
    """Smooth approximation of max_k corr_k and its gradient."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    w = np.exp(beta * (corr - max_corr))
    sum_w = np.sum(w)
    obj = max_corr + np.log(sum_w) / beta
    # weighted gradient via convolutions
    term1 = np.convolve(1 - h, w, mode="full")
    term2 = np.convolve(h, w[::-1], mode="full")
    off = len(w) - 1
    grad = (term1[off:off + n] - term2[off:off + n]) / sum_w
    return obj, grad
