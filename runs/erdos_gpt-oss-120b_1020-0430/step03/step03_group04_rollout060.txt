def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def project_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    # If diff small, just ignore
    if abs(diff) < 1e-12:
        return h
    # use loop up to maybe 1000 attempts
    for _ in range(10000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible(n, rng):
    # sample uniform
    h = rng.random(n)
    target = n / 2.0
    # scale to target sum
    total = h.sum()
    if total == 0:
        h[:] = target / n
    else:
        h *= target / total
    # clip & fix sum
    h = np.clip(h, 0.0, 1.0)
    h = project_sum(h, target, rng)
    return h

def worst_shift_and_grad(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n - 1)
    # compute gradient
    grad = np.zeros_like(h)
    if shift >= 0:
        i = np.arange(0, n - shift)
        j = i + shift
    else:
        i = np.arange(-shift, n)
        j = i + shift
    # i and j arrays
    # grad[i] += 1 - h[j]
    # grad[j] -= h[i]
    grad[i] += 1.0 - h[j]
    grad[j] -= h[i]
    return shift, grad, corr

def subgrad_optimize(h0, rng, deadline):
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    it = 0

        shift, grad, corr = worst_shift_and_grad(h)
        # step size schedule
        step = 0.5 / np.sqrt(it + 1)

        # clip
        np.clip(h, 0.0, 1.0, out=h)
        h = project_sum(h, target, rng)

            if c < best_c:
                best_c = c
                best_h = h.copy()
        it += 1

        if it > 20000:

                break

    if c < best_c:
        best_c = c
        best_h = h.copy()
    return best_h, best_c

def hill_climb(h0, rng, max_iters=2000, step_scale=0.2):
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step_scale
    for it in range(max_iters):
        # randomly select i,j and transfer epsilon
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2 * max_eps * step
        # no need clamp as eps already bounded
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps

            best_h = new_h.copy()
            h = new_h
        # gradually decay step
        if (it + 1) % (max_iters // 5) == 0:

import time
import numpy as np

# -------------------------------------------------------------
# basic utilities
# -------------------------------------------------------------
def compute_c5(h):
    """Exact discrete C5 for step function h on [0,2] with uniform spacing."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def project_sum(h, target, rng):
    """Adjust h in-place so that sum(h) == target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    # simple uniform redistribution, then random fixes if needed
    if abs(diff) > 1e-12:
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        # final random fix (few iterations)
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def random_feasible(n, rng):
    """Uniform random vector in [0,1] with exact sum n/2."""
    h = rng.random(n)
    target = n / 2.0
    total = h.sum()
    if total == 0.0:
        h.fill(target / n)
    else:
        h *= target / total
    h = np.clip(h, 0.0, 1.0)
    h = project_sum(h, target, rng)
    return h


def worst_shift_and_grad(h):
    """Return worst shift (max overlap) and gradient of that overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    shift = idx - (n - 1)               # shift can be negative
    grad = np.zeros_like(h)

    if shift >= 0:
        i = np.arange(0, n - shift)
        j = i + shift
    else:
        i = np.arange(-shift, n)
        j = i + shift                     # j < i
    # gradient contribution from each pair (i,j)
    grad[i] += 1.0 - h[j]
    grad[j] -= h[i]
    return shift, grad, corr

    n = h.size

    best_h = h.copy()
    best_c = compute_c5(h)
    it = 0
    while time.time() < deadline:

        step = 0.5 / np.sqrt(it + 1)        # decreasing step size

        np.clip(h, 0.0, 1.0, out=h)
        h = project_sum(h, target, rng)

    if c < best_c:
        best_c = c
        best_h = h.copy()
    return best_h, best_c

    h = h0.copy()
    n = h.size
