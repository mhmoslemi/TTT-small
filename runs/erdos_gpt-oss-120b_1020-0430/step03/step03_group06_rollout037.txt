def compute_c5(h):
    n = h.size
    # Compute convolution via FFT of length m = 1 << (2*n-1).bit_length() for speed.
    m = 1 << ((2*n-1).bit_length())
    # compute FFT of h and (1-h)
    H = np.fft.rfft(h, n=m)
    G = np.fft.rfft(1 - h, n=m)
    conv = np.fft.irfft(H * G, n=m)
    # linear conv length 2n-1: conv[0:2n-1]
    conv_lin = conv[: 2*n-1]
    # shift zero at index n-1
    max_val = conv_lin[n-1:].max()
    return max_val * (2.0 / n)

def propose_move(h, rng, max_step=0.2):
    n = h.size
    i, j = rng.choice(n, size=2, replace=False)
    hi, hj = h[i], h[j]
    # allowed delta range:
    max_pos = min(1.0 - hi, hj)
    max_neg = min(hi, 1.0 - hj)
    if max_pos + max_neg == 0:
        return None, None, None  # no move possible
    # sample delta uniform in [-max_neg, max_pos] scaled by step_factor
    delta = (rng.random() * (max_pos + max_neg) - max_neg) * max_step
    # ensure after scaling delta within allowed range
    delta = max(-max_neg, min(max_pos, delta))
    if abs(delta) < 1e-12:
        return None, None, None
    new_h = h.copy()
    new_h[i] = hi + delta
    new_h[j] = hj - delta
    return new_h, i, j, delta

    for it in range(max_iters):

        k_min = np.argmin(overlaps)

            break

        g = np.zeros_like(h)

            g[:len(h)-k] += h[k:]
            g[k:] += h[:len(h)-k]
        else:

def slsqp_refine(h0, beta_list, max_iter, rng):
    n = h0.size
    target = n / 2.0
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    # define constraints and bounds inside
    bounds = [(0.0, 1.0)] * n

        return np.sum(x) - target
    eq_cons = {'type':'eq', 'fun': sum_constraint}
    for beta in beta_list:
        # define objective using closure
        def obj(x, beta=beta):

            bc = beta * corr
            m = np.max(bc)
            # stable log-sum-exp
            return (m + np.log(np.sum(np.exp(bc - m)))) / beta

                       constraints=eq_cons,

                    x[i] += diff
        c = compute_c5(x)
        if c < best_c:
            best_c, best_h = c, x
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Helper functions
# --------------------------------------------------------------
def enforce_sum_simple(h, target, rng):
    """Adjust h so that its sum equals target while keeping 0≤h≤1."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        cand = np.where(h < 1 - 1e-12)[0]
    else:
        cand = np.where(h > 1e-12)[0]
    if cand.size == 0:
        return h
    i = rng.choice(cand)
    h[i] += diff
    h[i] = min(max(h[i], 0.0), 1.0)
    return h


def compute_overlap_fft(h):
    """Linear autocorrelation of h (overlap for shifts 0..n-1)."""
    n = h.size
    m = 1 << ((2 * n - 1).bit_length())  # next power of two
    H = np.fft.rfft(h, n=m)
    conv = np.fft.irfft(H * np.conj(H), n=m)
    conv = conv[: 2 * n - 1]               # linear part only
    overlaps = conv[n - 1 :]               # shifts 0..n-1
    return overlaps


def compute_c5(h):
    """Exact C5 for the discrete step function h."""
    n = h.size
    total = h.sum()
    overlaps = compute_overlap_fft(h)
    min_ov = overlaps.min()
    if min_ov < 0:
        min_ov = 0.0
    return (total - min_ov) * (2.0 / n)


# --------------------------------------------------------------
# Local optimisation primitives
# --------------------------------------------------------------
def gradient_ascent(h_start, max_iters, step, rng, target):
    """Increase the minimal overlap using a simple projected gradient ascent."""
    h = h_start.copy()
    best_h, best_c = h.copy(), compute_c5(h)
    n = h.size
    for _ in range(max_iters):
        overlaps = compute_overlap_fft(h)
        k = int(np.argmin(overlaps))          # worst shift
        g = np.zeros_like(h)
        if k > 0:
            g[: n - k] += h[k:]
            g[k:] += h[: n - k]
        else:                                 # k==0  -> derivative of sum h_i^2
            g = 2.0 * h
        g -= g.mean()                         # keep sum zero direction
        delta = step * g
        # respect box constraints
        max_pos = 1.0 - h
        max_neg = -h
        delta = np.where(delta > 0,
                         np.minimum(delta, max_pos),
                         np.maximum(delta, max_neg))
        if np.all(np.abs(delta) < 1e-12):
            break
        h_new = h + delta
        h_new = enforce_sum_simple(h_new, target, rng)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()
            h = h_new
        else:
            # occasional uphill move to escape plateaus
            if rng.random() < 0.001:
                h = h_new
                best_c = c_new
        step *= 0.9995
        if step < 1e-4:
            break
    return best_h, best_c


def hill_climb_once(h_start, steps, step, rng):
    """Greedy hill‑climb: accept only improving two‑coordinate transfers."""
    h = h_start.copy()
    best_h, best_c = h.copy(), compute_c5(h)
    n = h.size
    for _ in range(steps):
        i, j = rng.choice(n, size=2, replace=False)
        hi, hj = h[i], h[j]
        max_pos = min(1.0 - hi, hj)
        max_neg = min(hi, 1.0 - hj)
        if max_pos + max_neg == 0:
            continue
        delta = (rng.random() * (max_pos + max_neg) - max_neg) * step
        new_h = h.copy()
        new_h[i] = hi + delta
        new_h[j] = hj - delta
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h, h = c, new_h.copy(), new_h
    return best_h, best_c


def slsqp_refine(h0, beta_list, max_iter, rng):
    """Final smooth‑surrogate refinement with SLSQP."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def sum_eq(x):
        return np.sum(x) - target

    eq_cons = {"type": "eq", "fun": sum_eq}
    best_h, best_c = h0.copy(), compute_c5(h0)

    for beta in beta_list:
        def obj(x, beta=beta):
            corr = np.correlate(x, 1 - x, mode="full")
            bc = beta * corr
            m = np.max(bc)
            return (m + np.log(np.sum(np.exp(bc - m)))) / beta

        res = minimize(
            obj,
            best_h,
            method="SLSQP",
            bounds=bounds,
            constraints=eq_cons,
            options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
        )
        x = np.clip(res.x, 0.0, 1.0)
        x = enforce_sum_simple(x, target, rng)
        c = compute_c5(x)
        if c < best_c:
            best_c, best_h = c, x
    return best_h, best_c

# Main entry point
# --------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----- initialise problem size and a first feasible vector -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size * 2 <= 360:          # up‑sample if we have room
            h0 = np.repeat(h0, 2)
        n = h0.size
        target = n / 2.0
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum_simple(h0, target, rng)
    else:
        n = 360
        target = n / 2.0
        h0 = np.zeros(n)
        h0[: n // 2] = 1.0
        rng.shuffle(h0)

    best_h, best_c = h0.copy(), compute_c5(h0)

    # ----- a few diverse deterministic starts -----
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    for f in range(1, 4):
        for fun in (np.sin, np.cos):
            h = 0.5 + 0.5 * fun(np.pi * f * xs)
            h = np.clip(h, 0.0, 1.0)
            h = enforce_sum_simple(h, target, rng)
            c = compute_c5(h)
            if c < best_c:
                best_c, best_h = c, h.copy()
    # linear ramps
    for coeff in (0.5, -0.5):
        h = xs * coeff + 0.5
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum_simple(h, target, rng)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # ----- random binary starts -----
    for _ in range(30):
        h = np.zeros(n)
        h[: n // 2] = 1.0
        rng.shuffle(h)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # ----- main optimisation loop (gradient + hill‑climb) -----
    step = 0.2
    while time.time() - start_time < budget_s * 0.78:
        start = best_h.copy() if rng.random() < 0.7 else np.zeros(n)
        if start.size != n:
            start = np.zeros(n)
        if start.sum() == 0:
            start[: n // 2] = 1.0
            rng.shuffle(start)

        # gradient ascent on the worst shift
        h_g, c_g = gradient_ascent(start, max_iters=2000, step=step,
                                   rng=rng, target=target)
        if c_g < best_c:
            best_c, best_h = c_g, h_g.copy()

        # short greedy hill‑climb for fine tuning
        h_hc, c_hc = hill_climb_once(best_h, steps=2000, step=0.1, rng=rng)
        if c_hc < best_c:
            best_c, best_h = c_hc, h_hc.copy()

        step = max(0.05, step * 0.995)

    # ----- final smooth surrogate refinement (if time permits) -----
    if time.time() - start_time < budget_s * 0.95:
        beta_seq = [10, 40, 120, 400, 800, 1600, 3200]
        h_ref, c_ref = slsqp_refine(best_h, beta_seq, max_iter=1000, rng=rng)
        if c_ref < best_c:
            best_c, best_h = c_ref, h_ref

    # ----- clean‑up and return -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum_simple(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
