from scipy.optimize import minimize

# ----- global state for surrogate objective -----
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    """Equality Σx_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    """Log‑sum‑exp surrogate for max_k Σ_i x_i (1‑x_{i+k})."""
    corr = _full_corr_fft(x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def _full_corr_fft(h):
    """Full correlation of h and (1‑h) via FFT."""
    n = h.size
    nfft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(h, nfft)
    B = np.fft.rfft(1.0 - h, nfft)
    corr = np.fft.irfft(A * np.conj(B), nfft)[: 2 * n - 1]
    return corr


def compute_c5(h):
    """Exact discrete C₅ for a step function."""
    n = h.size
    nfft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(h, nfft)
    B = np.fft.rfft(1.0 - h, nfft)
    corr = np.fft.irfft(A * np.conj(B), nfft)[: 2 * n - 1]
    return np.max(corr) * (2.0 / n)

    diff = target - h.sum()

    """Simple pairwise transfer hill‑climb."""
    h = h_start.copy()
    best_h = h.copy()

                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                if abs(add) < 1e-12:
                    continue
                h[i] += add
                diff = CURRENT_TARGET - h.sum()
    return h


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----- problem size -----
    n = 300                      # try a finer grid
    target = n / 2.0
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx

    # ----- initial vector (interpolate if a parent is present) -----
    if "initial_h_values" in globals():
        old = np.asarray(initial_h_values, dtype=float)
        old_n = old.size
        old_xs = (np.arange(old_n) + 0.5) * (2.0 / old_n)
        h0 = np.interp(xs, old_xs, old)
    else:
        h0 = np.full(n, 0.5)
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = enforce_sum(h0, target, rng)

    # ----- build a pool of seeds -----
    candidates = [h0]
    for f in range(1, 5):
        for fun in (np.sin, np.cos):
            h = 0.5 + 0.5 * fun(np.pi * f * xs)
            candidates.append(enforce_sum(np.clip(h, 0, 1), target, rng))
    candidates.append(enforce_sum(xs / 2.0, target, rng))          # ramp up
    candidates.append(enforce_sum(1.0 - xs / 2.0, target, rng))   # ramp down
    for _ in range(8):
        candidates.append(random_feasible_vector(n, rng))

    # ----- pick the best seed -----
    best_h = None
    best_c = float("inf")
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    global_best_h, global_best_c = best_h, best_c

    # ----- main optimisation loop -----
    beta_seq = [5, 15, 40, 100, 300, 800, 2000]
    while time.time() - start_time < budget_s * 0.85:
        # start point: perturb current best or fresh random
        if rng.random() < 0.7:
            start = perturb_h(global_best_h, rng, scale=0.4)
        else:
            start = random_feasible_vector(n, rng)

        # local hill‑climb
        h_opt, c_opt = hill_climb(start, rng, max_iters=3000, step_start=0.6)

        # if improvement, run continuation SLSQP
        if c_opt < global_best_c:
            h_opt = slsqp_optimize_continuation(h_opt, beta_seq, max_iter=350)
            c_opt = compute_c5(h_opt)

        if c_opt < global_best_c:
            global_best_h, global_best_c = h_opt.copy(), c_opt

        # occasional fine‑grained hill‑climb after a good jump
        if time.time() - start_time < budget_s * 0.70 and c_opt < global_best_c:
            hf, cf = hill_climb(global_best_h, rng, max_iters=5000, step_start=0.3)
            if cf < global_best_c:
                global_best_h, global_best_c = hf.copy(), cf

    # ----- final clean‑up -----
    global_best_h = np.clip(global_best_h, 0.0, 1.0)
    global_best_h = enforce_sum(global_best_h, target, rng)
    final_c = compute_c5(global_best_h)

    return global_best_h.tolist(), float(final_c), int(n)
