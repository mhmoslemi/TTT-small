def gradient_transfer(h, k, step, min_step=1e-5, decay=0.995):
    n = h.size
    # compute g
    g = np.zeros_like(h)
    # indexes where i+k < n
    idx = np.arange(n - k)
    g[idx] = 1.0 - h[idx + k]
    # indexes where i >= k
    idx2 = np.arange(k, n)
    g[idx2] -= h[idx2 - k]
    # now choose i_max, i_min
    i_max = np.argmax(g)
    i_min = np.argmin(g)
    max_transfer = min(h[i_max], 1.0 - h[i_min])
    eps = max_transfer * step
    if eps > 0:
        h[i_max] -= eps
        h[i_min] += eps
    return h, step * decay
```

def gradient_transfer_optimize(h0, rng, max_iters=20000, start_step=0.5, decay=0.999, min_step=1e-6):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        # compute correlation
        corr = np.correlate(h, 1-h, mode='full')
        c = np.max(corr) * (2.0/n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        # find index of max correlation (full array)
        # need shift index relative to 0: k = idx - (n-1) . shifts negative allowed
        k = np.argmax(corr) - (n-1)
        # compute gradient g as described
        g = np.zeros_like(h)
        if k >= 0:
            # i + k < n condition: i from 0..n-1-k
            i_range = np.arange(n - k)
            g[i_range] = 1.0 - h[i_range + k]
            # i >= k condition: i from k..n-1
            i_range2 = np.arange(k, n)
            g[i_range2] -= h[i_range2 - k]
        else:
            # k negative: shift left.
            k_pos = -k
            # for negative shift, expression symmetrical: C_{-k} = Σ_i h_{i+k} (1 - h_i)
            # compute g similarly: the gradient with respect to h_i:
            # dC_{-k}/dh_i = (1 - h_{i-k}) - h_{i+k}
            # For k negative, let k = -k_pos
            # So:
            # term (1 - h_{i-k}) for i - (-k_pos) = i + k_pos < n => i < n - k_pos
            i_range = np.arange(n - k_pos)
            g[i_range] = 1.0 - h[i_range + k_pos]   # 1 - h_{i + k_pos}
            # term -h_{i+k} for i + (-k_pos) = i - k_pos >= 0 => i >= k_pos
            i_range2 = np.arange(k_pos, n)
            g[i_range2] -= h[i_range2 - k_pos]
        # choose i_max, i_min
        i_max = np.argmax(g)
        i_min = np.argmin(g)
        # compute transfer
        max_transfer = min(h[i_max], 1.0 - h[i_min])
        if max_transfer <= 0:

            break

        if eps < 1e-12:
            break
        h[i_max] -= eps
        h[i_min] += eps

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global helpers for the smooth surrogate used by SLSQP
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


# ----------------------------------------------------------------------
# Problem utilities
# ----------------------------------------------------------------------
def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        if abs(add) < 1e-12:
            continue
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(1000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            if abs(add) < 1e-12:
                continue
            h[i] += add
            diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n, rng):
    v = rng.exponential(1.0, size=n)
    v = v / v.sum() * (n / 2.0)
    h = np.clip(v, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def perturb_h(h, rng, scale=0.5):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    while j == i:
        j = rng.integers(0, n)
    max_pos = min(h[i], 1.0 - h[j])
    max_neg = min(h[j], 1.0 - h[i])
    if max_pos + max_neg == 0:
        return h.copy()
    eps = (rng.random() * (max_pos + max_neg)) - max_neg
    eps *= scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return np.clip(new_h, 0.0, 1.0)


def hill_climb(h_start, rng, max_iters=2500, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)
        max_pos = min(h[i], 1.0 - h[j])
        max_neg = min(h[j], 1.0 - h[i])
        if max_pos + max_neg == 0:
            continue
        eps = (rng.random() * (max_pos + max_neg)) - max_neg
        eps *= step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            if abs(add) < 1e-12:
                continue
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def gradient_optimize(h0, rng, max_iters=20000, start_step=0.5, decay=0.9995, min_step=1e-6):
    h = h0.copy()
    n = h.shape[0]
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for _ in range(max_iters):
        corr = np.correlate(h, 1 - h, mode="full")
        c = np.max(corr) * (2.0 / n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        idx = np.argmax(corr)
        k = idx - (n - 1)
        g = np.zeros_like(h)
        if k >= 0:
            if k < n:
                g[: n - k] = 1.0 - h[k:]
                g[k:] -= h[: n - k]
        else:
            l = -k
            if l < n:
                g[l:] = 1.0 - h[: n - l]
                g[: n - l] -= h[l:]
        i_max = np.argmax(g)
        i_min = np.argmin(g)
        max_transfer = min(h[i_max], 1.0 - h[i_min])
        if max_transfer <= 0:
            step = max(step * decay, min_step)
            continue
        eps = max_transfer * step
        if eps < 1e-12:
            break
        h[i_max] -= eps
        h[i_min] += eps
        step = max(step * decay, min_step)
    best_h = enforce_sum(best_h, target, rng)
    return best_h, best_c
