def softmax_weights(corr, beta):
    # compute probabilities proportional to exp(beta * corr)
    max_corr = np.max(beta * corr)
    exp_vals = np.exp(beta * corr - max_corr)
    sum_exp = np.sum(exp_vals)
    p = exp_vals / sum_exp
    return p

import time
import numpy as np

# ---------- basic utilities ----------
def compute_c5(h):
    """Maximum overlap integral for discrete h on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_simple(z, target):
    """Clip to [0,1] and renormalise to hit the exact sum `target`."""
    proj = np.clip(z, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        # capacity for moving the diff
        if diff > 0:
            cap = 1.0 - proj
        else:
            cap = proj
        cap_sum = cap.sum()
        if cap_sum > 0:
            proj += diff * cap / cap_sum
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def perturb_random(h, rng, scale=0.1):
    """Transfer a small amount ε from one index to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


# ---------- worst‑shift gradient search ----------
def worst_grad_search(h0, rng, target, max_iters=2000, step0=0.5):
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step0
    stagn = 0
    for _ in range(max_iters):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr)
        c_curr = corr[idx] * (2.0 / n)
        if c_curr < best_c:
            best_c = c_curr
            best_h = h.copy()
        k = idx - (n - 1)

        # gradient of the worst shift (scaled by dx)
        grad = np.ones(n)
        if k >= 0:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
        else:
            kk = -k
            grad[: n - kk] -= h[kk:]
            grad[kk:] -= h[: n - kk]
        grad *= (2.0 / n)

        new_h = h - step * grad
        new_h = project_simple(new_h, target)
        c_new = compute_c5(new_h)
        if c_new < c_curr:
            h = new_h
            step = min(step * 1.1, step0)
            stagn = 0
        else:
            step *= 0.5
            stagn += 1
            if step < 1e-6:
                break
            if stagn >= 30:
                h = perturb_random(h, rng, scale=0.1)
                step = step0
                stagn = 0
    return best_h, best_c


# ---------- smooth‑max gradient ----------
def softmax_grad(h, beta):
    """Gradient of the soft‑max approximation of max_k C_k."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    # probabilities proportional to exp(beta * corr)
    maxbeta = np.max(beta * corr)
    ex = np.exp(beta * corr - maxbeta)
    probs = ex / ex.sum()
    offset = n - 1
    grad = np.full(n, 1.0)  # the constant part from ∂/∂h_i Σh_i

    for kk in range(- (n - 1), n):
        p = probs[kk + offset]
        if p == 0.0:
            continue
        if kk >= 0:
            grad[: n - kk] -= p * h[kk:]
            grad[kk:] -= p * h[: n - kk]
        else:
            k = -kk
            grad[: n - k] -= p * h[k:]
            grad[k:] -= p * h[: n - k]
    grad *= (2.0 / n)
    return grad


def softmax_descent(h0, beta, rng, target, max_iters=500, step0=0.3):
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step0
    for _ in range(max_iters):
        grad = softmax_grad(h, beta)
        new_h = h - step * grad
        new_h = project_simple(new_h, target)
        c_new = compute_c5(new_h)
        if c_new < best_c:
            best_c = c_new
            best_h = new_h.copy()
        if c_new < compute_c5(h):
            h = new_h
            step = min(step * 1.05, step0)
        else:
            step *= 0.5
            if step < 1e-6:
                break
    return best_h, best_c


# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def elapsed():
        return time.time() - start

    # target resolution (increase if budget is generous)
    n_target = 720
    if budget_s < 200:
        n_target = 540

    # build a pool of initial vectors
    init_vecs = []
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        m = h0.size
        if m != n_target:
            xs0 = (np.arange(m) + 0.5) * (2.0 / m)
            xs = (np.arange(n_target) + 0.5) * (2.0 / n_target)
            h_up = np.interp(xs, xs0, h0)
            h_up = project_simple(h_up, n_target / 2.0)
            init_vecs.append(h_up)
        else:
            init_vecs.append(project_simple(h0, n_target / 2.0))
    # random continuous starts
    for _ in range(4):
        r = rng.random(n_target)
        init_vecs.append(project_simple(r, n_target / 2.0))
    # random binary starts (exactly half ones)
    for _ in range(2):
        idx = rng.choice(n_target, size=n_target // 2, replace=False)
        b = np.zeros(n_target)
        b[idx] = 1.0
        init_vecs.append(b)

    best_h = None
    best_c = np.inf
    target_sum = n_target / 2.0

    # ----- phase 1: worst‑shift local optimisation -----
    for h_start in init_vecs:
        if elapsed() > budget_s * 0.90:
            break
        h_loc, c_loc = worst_grad_search(h_start, rng, target_sum,
                                          max_iters=1500, step0=0.5)
        if c_loc < best_c:
            best_c, best_h = c_loc, h_loc.copy()

    # ----- phase 2: smooth‑max refinement with increasing β -----
    betas = [5.0, 20.0, 80.0, 200.0, 800.0, 2000.0]
    for beta in betas:
        if elapsed() > budget_s * 0.96:
            break
        h_ref, c_ref = softmax_descent(best_h, beta, rng, target_sum,
                                       max_iters=400, step0=0.3)
        if c_ref < best_c:
            best_c, best_h = c_ref, h_ref.copy()

    # ----- phase 3: tiny random tweaks until time runs out -----
    while elapsed() < budget_s * 0.99:
        cand = perturb_random(best_h, rng, scale=0.02)
        c_cand = compute_c5(cand)
        if c_cand < best_c:
            best_c, best_h = c_cand, cand.copy()

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_simple(best_h, target_sum)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n_target)
