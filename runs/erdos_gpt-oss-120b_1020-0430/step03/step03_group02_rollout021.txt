def stochastic_descent(h, rng, time_limit):
    start = time.time()
    n = h.shape[0]
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    iteration = 0
    T0 = 0.005
    while time.time() - start < time_limit:
        grad = worst_gradient(cur_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        max_eps = min(cur_h[i], 1 - cur_h[j])
        if max_eps <= 0:
            # random pick different i/j
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            while i == j:
                j = rng.integers(0, n)
            max_eps = min(cur_h[i], 1 - cur_h[j])
        eps = max_eps * rng.random() * 0.5  # maybe 0.5 factor
        nh = cur_h.copy()
        nh[i] -= eps
        nh[j] += eps

        c_new = compute_c5(nh)
        iteration += 1
        T = T0 * (0.9995**iteration)
        if c_new < cur_c or rng.random() < np.exp(-(c_new - cur_c) / max(T, 1e-12)):

import time, numpy as np

def compute_c5(h):
    """C5 for discrete step function h on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, rng):
    """Adjust h in place so that sum(h)=len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 10000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
        attempts += 1
    return h


def random_feasible_vector(n, rng):
    """Generate a random h with 0≤h≤1 and Σh=n/2."""
    h = rng.random(n)
    target = n / 2.0
    h = h * target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 10000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
        attempts += 1
    return h


def worst_gradient(h):
    """Gradient of the worst shift O_k = Σ h_i (1‑h_{i+k})."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h, grad, rng, scale=0.5):
    """Transfer ε from a high‑gradient index to a low‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h_start, rng, max_iters=3000, step_scale=0.5):
    """Hill‑climb using worst‑shift gradient."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # initial vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        n = int(len(initial_h_values))
        h = np.array(initial_h_values, dtype=float)
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    else:
        n = 192
        h = random_feasible_vector(n, rng)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ------------------------------------------------------------------
    # quick local improvement
    # ------------------------------------------------------------------
    h_local, c_local = guided_hill_climb(best_h, rng, max_iters=2000, step_scale=0.5)
    if c_local < best_c:
        best_h, best_c = h_local.copy(), c_local

    # ------------------------------------------------------------------
    # simulated annealing on continuous variables
    # ------------------------------------------------------------------
    cur_h = best_h.copy()
    cur_c = best_c
    iter_cnt = 0
    T0 = 0.01
    anneal_end = start + budget_s * 0.80   # 80 % of total budget

    while time.time() < anneal_end:
        grad = worst_gradient(cur_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0.0:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            max_eps = min(cur_h[i], 1.0 - cur_h[j])
            if max_eps <= 0.0:
                continue
        eps = max_eps * rng.random() * 0.5
        nh = cur_h.copy()
        nh[i] -= eps
        nh[j] += eps
        c_new = compute_c5(nh)
        iter_cnt += 1

        # linear temperature schedule
        remaining = max(anneal_end - time.time(), 0.0)
        total = max(anneal_end - start, 1e-9)
        frac = remaining / total
        T = T0 * frac

        if c_new < cur_c or rng.random() < np.exp(-(c_new - cur_c) / max(T, 1e-12)):
            cur_h, cur_c = nh, c_new
            if c_new < best_c:
                best_h, best_c = nh.copy(), c_new

        # occasional random jitter to keep diversity
        if iter_cnt % 5000 == 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                max_eps = min(cur_h[i], 1.0 - cur_h[j])
                if max_eps > 0:
                    eps = max_eps * rng.random() * 0.1
                    cur_h[i] -= eps
                    cur_h[j] += eps
                    cur_c = compute_c5(cur_h)
                    if cur_c < best_c:
                        best_h, best_c = cur_h.copy(), cur_c

    # ------------------------------------------------------------------
    # final polish with deterministic hill‑climb
    # ------------------------------------------------------------------
    if time.time() < start + budget_s * 0.95:
        h_final, c_final = guided_hill_climb(best_h, rng,
                                             max_iters=1500, step_scale=0.3)
        if c_final < best_c:
            best_h, best_c = h_final.copy(), c_final

    # ------------------------------------------------------------------
    # optional up‑sampling refinement if time permits
    # ------------------------------------------------------------------
    if time.time() < start + budget_s * 0.99:
        if n * 2 <= 240:
            h_up = np.repeat(best_h, 2)
            n_up = h_up.size
            h_up, c_up = guided_hill_climb(h_up, rng,
                                           max_iters=1200, step_scale=0.3)
            if c_up < best_c:
                best_h, best_c, n = h_up.copy(), c_up, n_up

    # ------------------------------------------------------------------
    # final feasibility enforcement and return
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
