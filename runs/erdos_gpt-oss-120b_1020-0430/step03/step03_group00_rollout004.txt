def enforce_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # compute free indices
    # We'll adjust all entries proportionally while respecting bounds.
    # We'll iteratively distribute diff.
    for _ in range(100):
        free = (h > 1e-12) & (h < 1-1e-12)
        if not np.any(free):
            break
        total_free = free.sum()
        delta = diff / total_free
        # apply delta
        keep = np.where(free, h + delta, h)
        # clip to bounds
        keep = np.clip(keep, 0.0, 1.0)
        new_diff = target - keep.sum()
        if abs(new_diff - diff) < 1e-12:
            h = keep
            break
        h = keep
        diff = new_diff
    # final small fix if needed:
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # distribute to any index not at bound
        idx = np.argmax(np.minimum(h, 1 - h))  # index with most slack
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h
```

def perturb_transfer(h, rng, scale=0.1):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h
    max_transfer = min(h[i], 1.0 - h[j])
    if max_transfer <= 0:
        return h
    eps = max_transfer * scale * rng.random()
    h_new = h.copy()
    h_new[i] -= eps
    h_new[j] += eps
    return h_new

def perturb_targeted(h, grad, rng, scale=0.15):
    pos = np.where((grad > 1e-9) & (h > 1e-12))[0]
    neg = np.where((grad < -1e-9) & (h < 1.0 - 1e-12))[0]

def slsqp_refine(h0, beta, max_iter=500):
    n = h0.size
    target_sum = n/2.0
    # define objective
    def obj(x):
        corr = np.fft.ifft(np.fft.fft(x, 2*n) * np.conj(np.fft.fft(1 - x, 2*n))).real
        bc = beta * corr[:2*n-1]
        m = np.max(bc)
        # log-sum-exp
        lse = m + np.log(np.sum(np.exp(bc - m)))
        return lse / beta
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol':1e-12, 'disp': False})
    x = np.clip(res.x,0.0,1.0)
    x = enforce_sum(x, target_sum)
    return x
```

        m = np.max(bc)

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utilities ----------
def enforce_sum(h, target):
    """Clip to [0,1] and adjust to make the sum exactly target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    for _ in range(100):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            slack = 1.0 - h
        else:
            slack = h
        total = slack.sum()
        if total == 0:
            break
        delta = diff * slack / total
        h += delta
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        # one‑off correction
        if diff > 0:
            slack = 1.0 - h
        else:
            slack = h
        idx = np.argmax(slack)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h


def compute_c5(h):
    """Discrete C5 = max correlation * 2/n."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def worst_shift_gradient(h):
    """Gradient of the worst (max) shift term."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    k = idx - (n - 1)  # shift
    grad = np.ones(n)
    if k >= 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    else:
        kk = -k
        grad[: n - kk] -= h[kk:]
        grad[kk:] -= h[: n - kk]
    return grad, k


def perturb_transfer(h, rng, scale=0.2):
    """Transfer a small ε from i to j (preserves sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    h2 = h.copy()
    h2[i] -= eps
    h2[j] += eps
    return h2


def perturb_targeted(h, grad, rng, scale=0.15):
    """Move mass from positive‑gradient to negative‑gradient coordinates."""
    pos = np.where((grad > 1e-9) & (h > 1e-12))[0]
    neg = np.where((grad < -1e-9) & (h < 1.0 - 1e-12))[0]
    if pos.size == 0 or neg.size == 0:
        return perturb_transfer(h, rng, scale)
    i = rng.choice(pos)
    j = rng.choice(neg)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_transfer(h, rng, scale)
    eps = max_eps * scale * rng.random()
    h2 = h.copy()
    h2[i] -= eps
    h2[j] += eps
    return h2


def hill_climb(h_start, rng, max_iters, step_scale, target_sum):
    """Targeted hill‑climbing on the worst shift."""
    h = h_start.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    n = h.size
    step = step_scale
    decay_interval = max(1, max_iters // 5)
    for it in range(max_iters):
        grad, _ = worst_shift_gradient(h)
        cand = perturb_targeted(h, grad, rng, scale=step)
        cand = enforce_sum(cand, target_sum)
        val = compute_c5(cand)
        if val < best_val:
            best_val, best_h = val, cand.copy()
        # accept if improvement, otherwise keep current
        if val < compute_c5(h):
            h = cand
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_val


def slsqp_refine(h0, beta, max_iter=400):
    """Smooth‐max SLSQP refinement."""
    n = h0.size
    target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        # log‑sum‑exp, stabilized
        lse = m + np.log(np.sum(np.exp(bc - m)))
        return lse / beta

    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            obj,
            h0,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
        )
        x = np.clip(res.x, 0.0, 1.0)
        x = enforce_sum(x, target)
        return x
    except Exception:
        return h0.copy()

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # ---- initial vector -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = rng.random(180)
    n0 = h0.size
    target_n = 720
    if n0 != target_n:
        xs0 = (np.arange(n0) + 0.5) * (2.0 / n0)
        xs = (np.arange(target_n) + 0.5) * (2.0 / target_n)
        h = np.interp(xs, xs0, h0)
    else:
        h = h0.copy()
    h = enforce_sum(np.clip(h, 0.0, 1.0), target_n / 2.0)

    best_h = h.copy()
    best_c = compute_c5(best_h)
    target_sum = target_n / 2.0

    # ---- early low‑beta SLSQP passes ------------------------------------
    low_beta = [0.5, 2.0, 10.0, 50.0]
    for beta in low_beta:
        if time.time() - start > budget_s * 0.75:
            break
        cand = slsqp_refine(best_h, beta, max_iter=300)
        c = compute_c5(cand)
        if c < best_c:
            best_h, best_c = cand, c

    # ---- main stochastic search -----------------------------------------
    high_beta = [200.0, 800.0, 2000.0]
    while time.time() - start < budget_s * 0.94:
        # random start for hill‑climb
        if rng.random() < 0.6:
            start_vec = perturb_transfer(best_h, rng, scale=rng.uniform(0.1, 0.4))
        else:
            start_vec = rng.random(target_n)
            start_vec = enforce_sum(start_vec, target_sum)

        # hill‑climb
        max_iters = int(rng.integers(800, 1800))
        step = rng.uniform(0.2, 0.6)
        h_loc, c_loc = hill_climb(start_vec, rng, max_iters, step, target_sum)
        if c_loc < best_c:
            best_h, best_c = h_loc, c_loc

        # occasional high‑beta refinement
        if rng.random() < 0.25:
            beta = rng.choice(high_beta)
            h_ref = slsqp_refine(best_h, beta, max_iter=500)
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_h, best_c = h_ref, c_ref

    # ---- final fine‑grained random tweaks --------------------------------
    while time.time() - start < budget_s * 0.99:
        cand = perturb_transfer(best_h, rng, scale=0.02)
        c = compute_c5(cand)
        if c < best_c:
            best_h, best_c = cand, c

    # ---- clean‑up --------------------------------------------------------
    best_h = enforce_sum(np.clip(best_h, 0.0, 1.0), target_sum)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
