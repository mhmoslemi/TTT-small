def improve_worst_shift(h, rng):
    # compute corr
    c = np.correlate(h, 1 - h, mode='full')
    max_val = c.max()
    # find shift(s) with max_val
    idxs = np.where(c == max_val)[0]
    # choose first shift
    idx = idxs[0]
    shift = idx - (len(h) - 1)
    shift = abs(shift)
    if shift == 0:
        return False
    # find i where h[i]=1 and i+shift < n and h[i+shift]=0
    i_candidates = np.where((h[:len(h)-shift] == 1) & (h[shift:] == 0))[0]
    # find j where h[j]=0 and j+shift < n and h[j+shift]=1
    j_candidates = np.where((h[:len(h)-shift] == 0) & (h[shift:] == 1))[0]
    if i_candidates.size == 0 or j_candidates.size == 0:
        return False
    # pick random i, j
    i = rng.choice(i_candidates)
    j = rng.choice(j_candidates)
    # swap
    h[i], h[j] = 0.0, 1.0
    return True

def adjust_to_target(v, target):

    idx = np.argpartition(-v, target-1)[:target]  # fastest selection
    out = np.zeros_like(v, dtype=float)

    return out

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Helper functions ----------
def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_to_feasible(v, target):
    """Projection of v onto {0≤h≤1, Σh = target} via bisection on a shift."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_binary_vector(n, rng):
    """Return a binary vector of length n with exactly n//2 ones."""
    idx = rng.choice(n, size=n // 2, replace=False)
    out = np.zeros(n, dtype=float)
    out[idx] = 1.0
    return out


def adjust_to_target(v, target):
    """Binary vector with exactly target ones – keep the largest values."""
    v = np.asarray(v, dtype=float)
    idx = np.argpartition(-v, target - 1)[:target]
    out = np.zeros_like(v, dtype=float)
    out[idx] = 1.0
    return out


def thue_morse_binary(n):
    """First n terms of the Thue‑Morse sequence (0/1)."""
    t = np.fromiter((i.bit_count() & 1 for i in range(n)), dtype=np.uint8, count=n)
    return t.astype(float)


def upsample_initial(to_n):
    """Linear interpolation of the global initial_h_values to length to_n."""
    if "initial_h_values" not in globals():
        return None
    init = np.array(initial_h_values, dtype=float)
    m = init.size
    xs_old = np.linspace(0.0, 2.0, m, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, to_n, endpoint=False)
    new = np.interp(xs_new, xs_old, init)
    return new


# ---------- Simulated annealing on binary vectors ----------
def anneal_binary(h0, rng, time_limit):
    """Binary simulated annealing (swap a 1 with a 0)."""
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    start = time.time()
    T0 = cur_c * 0.2 + 1e-4  # initial temperature

    while time.time() - start < time_limit:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        new_c = compute_c5(h)
        delta = new_c - cur_c
        frac = (time.time() - start) / time_limit
        T = T0 * (1 - frac)  # linear cooling
        accept = delta <= 0 or (T > 0 and rng.random() < np.exp(-delta / T))
        if accept:
            cur_c = new_c
            if new_c < best_c:
                best_c, best_h = new_c, h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_c


# ---------- Targeted local improvement ----------
def targeted_improve(h, rng):
    """Swap a mismatching pair for the current worst shift, if it lowers max."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    max_val = corr.max()
    idx = np.where(corr == max_val)[0][0]
    shift = idx - (n - 1)
    shift = abs(shift)
    if shift == 0:
        return False
    # positions where h[i]=1 and h[i+shift]=0
    i_cand = np.where((h[:n - shift] == 1) & (h[shift:] == 0))[0]
    # positions where h[j]=0 and h[j+shift]=1
    j_cand = np.where((h[:n - shift] == 0) & (h[shift:] == 1))[0]
    if i_cand.size == 0 or j_cand.size == 0:
        return False
    i = rng.choice(i_cand)
    j = rng.choice(j_cand)
    # perform swap
    h[i], h[j] = 0.0, 1.0
    new_max = np.max(np.correlate(h, 1 - h, mode="full"))
    if new_max < max_val:
        return True
    # revert
    h[i], h[j] = 1.0, 0.0
    return False


# ---------- Main optimisation routine ----------
def run(seed=42, budget_s=1100, **kwargs):

    start_total = time.time()

    # Choose discretisation size

    # ---- Main search loop (restarts) ------------------------------------
    while True:
        elapsed = time.time() - start_total
        remaining = budget_s - elapsed
        if remaining < 12.0:  # keep a small safety margin
            break

        # allocate time for a single restart (30 s or a fraction)
        restart_time = min(30.0, remaining * 0.25)

        # start from a fresh random binary vector
        h_start = random_binary_vector(n, rng)

            best_h[to_flip] = 0.0

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
