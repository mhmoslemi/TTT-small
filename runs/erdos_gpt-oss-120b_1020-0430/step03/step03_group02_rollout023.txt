CURRENT_TARGET = None; CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # softmax: log-sum-exp
    # return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
    # Actually objective we want to minimize max(corr). Softmax approximates max.
    # There's a factor 1/BETA outside; it's okay.
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=800):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # adjust sum in case of slight diff
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:

import time, numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h):
    """Discrete C5: max_k ∫ h(x)(1-h(x+k))dx."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    it = 0
    while abs(diff) > 1e-12 and it < 5000:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
        it += 1
    return h


def random_feasible_vector(n, rng):
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    it = 0
    while abs(diff) > 1e-12 and it < 5000:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
        it += 1
    return h


def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h_start, rng, max_iters=3000, step_scale=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    decay = max(1, max_iters // 5)
    cur_scale = step_scale
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=cur_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            cur_scale = max(cur_scale * 0.7, 1e-4)
    return best_h, best_c


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=600):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    it = 0
    while abs(diff) > 1e-12 and it < 2000:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        if add != 0.0:
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
        it += 1
    return x


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h, best_c, best_n = None, np.inf, None

    # use parent if present
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0, c0, h0.size

    candidates = [256, 240, 224, 192, 168, 144, 120]
    if best_n is not None:
        candidates = [best_n] + [n for n in candidates if n != best_n]

    for n in candidates:
        if time.time() - start > budget_s * 0.85:
            break

        # initialise
        if best_n == n and best_h is not None:
            h = best_h.copy()
        elif best_h is not None:
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h = np.interp(x_new, x_old, best_h)
            h = np.clip(h, 0.0, 1.0)
            h = enforce_sum(h, rng)
        elif "initial_h_values" in globals():
            parent = np.array(initial_h_values, dtype=float)
            x_old = np.linspace(0, 2, parent.size, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h = np.interp(x_new, x_old, parent)
            h = np.clip(h, 0.0, 1.0)
            h = enforce_sum(h, rng)
        else:
            h = random_feasible_vector(n, rng)

        # basic hill‑climb
        h, c = guided_hill_climb(h, rng, max_iters=2500, step_scale=0.5)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

        # SLSQP schedule
        for beta in (5, 20, 80, 200, 500, 1000, 2000, 5000):
            if time.time() - start > budget_s * 0.95:
                break
            mx = 800 if beta >= 500 else 600
            h = slsqp_optimize(h, beta, rng, max_iter=mx)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

        # symmetry enforcement and refinement
        if time.time() - start < budget_s * 0.97:
            sym = (h + h[::-1]) / 2.0
            sym = enforce_sum(sym, rng)
            sym, c = guided_hill_climb(sym, rng, max_iters=2000, step_scale=0.4)
            if c < best_c:
                best_h, best_c, best_n = sym.copy(), c, n

        # final local climb
        if time.time() - start < budget_s * 0.98:
            h, c = guided_hill_climb(h, rng, max_iters=1500, step_scale=0.3)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

    # annealing on the best candidate
    if best_h is not None and time.time() - start < budget_s * 0.99:
        T = 0.02
        iters = 5000
        h = best_h.copy()
        c = best_c
        n = h.size
        for _ in range(iters):
            if time.time() - start > budget_s * 0.999:
                break
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * 0.5
            nh = h.copy()
            nh[i] -= eps
            nh[j] += eps
            nc = compute_c5(nh)
            if nc < c or rng.random() < np.exp(-(nc - c) / T):
                h, c = nh, nc
                if c < best_c:
                    best_h, best_c = h.copy(), c
            T *= 0.9999

    # final feasibility and return
    if best_h is None:
        best_n = 96
        best_h = np.full(best_n, 0.5)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
