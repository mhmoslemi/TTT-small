def targeted_perturb(h, k0, rng, scale):
    n = h.size
    # Find possible i such that both i and i+k0 in [0,n)
    if k0 >= 0:
        i_min = 0
        i_max = n - k0 - 1
    else:
        i_min = -k0
        i_max = n - 1
    if i_max < i_min:
        # can't target, fallback generic
        return perturb_h(h, rng, scale)
    i = rng.integers(i_min, i_max+1)
    j = i + k0

    # Choose two source indices distinct from i and j
    # We'll select them randomly with possible repeats.
    # First source for i:
    source_i = rng.integers(0, n)
    while source_i == i or source_i == j:
        source_i = rng.integers(0, n)
    # second source for j:
    source_j = rng.integers(0, n)
    while source_j == i or source_j == j or source_j == source_i:
        source_j = rng.integers(0, n)

    # compute max transfer amounts
    max_eps_i = min(h[source_i], 1.0 - h[i])
    max_eps_j = min(h[source_j], 1.0 - h[j])
    # Choose epsilon drawn randomly up to scale * max_eps
    if max_eps_i <= 0 or max_eps_j <= 0:
        return h.copy()
    eps_i = (rng.random() - 0.5) * 2.0 * max_eps_i * scale
    eps_j = (rng.random() - 0.5) * 2.0 * max_eps_j * scale
    # ensure direction: we want to increase i and j, i.e., eps_i > 0 and eps_j > 0.
    # So we can just take positive random up to max_eps * scale.
    # Let's take eps_i = rng.random() * max_eps_i * scale
    # Similarly for eps_j. That yields positive.
    eps_i = rng.random() * max_eps_i * scale
    eps_j = rng.random() * max_eps_j * scale

    # apply
    new_h = h.copy()
    new_h[source_i] -= eps_i
    new_h[i] += eps_i
    new_h[source_j] -= eps_j
    new_h[j] += eps_j
    # Clip just in case
    new_h = np.clip(new_h, 0.0, 1.0)
    return new_h

def enforce_sum(h, target, rng, max_iter=2000):
    """Clip to [0,1] and adjust to exact sum target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-14:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-14:  # fallback uniform correction
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    h = rng.random(n)
    h = h / h.mean() * 0.5  # scale to average 0.5
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def perturb_h(h, rng, scale=0.2):
    """Transfer a fraction from one coordinate to another (preserves sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new


def targeted_perturb(h, k0, rng, scale):
    """Move mass into a pair (i,i+k0) that currently hurts the max overlap."""
    n = h.size
    # find feasible i where both i and i+k0 are inside the array
    if k0 >= 0:
        i_min, i_max = 0, n - k0 - 1
    else:
        i_min, i_max = -k0, n - 1
    if i_max < i_min:
        return perturb_h(h, rng, scale)
    i = rng.integers(i_min, i_max + 1)
    j = i + k0
    # pick sources distinct from targets
    src_i = rng.integers(0, n)
    while src_i in (i, j):
        src_i = rng.integers(0, n)
    src_j = rng.integers(0, n)
    while src_j in (i, j, src_i):
        src_j = rng.integers(0, n)

    max_eps_i = min(h[src_i], 1.0 - h[i])
    max_eps_j = min(h[src_j], 1.0 - h[j])
    if max_eps_i <= 0 or max_eps_j <= 0:
        return perturb_h(h, rng, scale)

    eps_i = rng.random() * max_eps_i * scale
    eps_j = rng.random() * max_eps_j * scale

    new = h.copy()
    new[src_i] -= eps_i
    new[i] += eps_i
    new[src_j] -= eps_j
    new[j] += eps_j
    return new


def hill_climb_targeted(h_start, rng, max_iters=2500, init_step=0.5,
                       target_prob=0.7):
    """Hill‑climb using generic and targeted perturbations."""
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        # choose move type
        if rng.random() < target_prob:
            # find lag with worst overlap
            corr = np.correlate(h, 1 - h, mode="full")
            idx = np.argmax(corr)
            k0 = idx - (h.size - 1)
            if k0 == 0:
                new_h = perturb_h(h, rng, scale=step)
            else:
                new_h = targeted_perturb(h, k0, rng, step)
        else:
            new_h = perturb_h(h, rng, scale=step)

        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            h = new_h
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)
    return h, best_c


# ---------- SLSQP surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=400):
    """Smooth surrogate optimization with SLSQP for a given beta."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum (tiny drift)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    budget = budget_s

    # ---------- initial vector ----------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    # ---------- generate diverse seeds ----------
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    candidates = [h0]

    # sinusoids
    for f in (1, 2, 3, 4, 5):
        for base in (np.sin, np.cos):
            h = 0.5 + 0.5 * base(np.pi * f * xs)
            candidates.append(enforce_sum(np.clip(h, 0.0, 1.0), n / 2.0, rng))

    # linear ramps
    candidates.append(enforce_sum(xs / 2.0, n / 2.0, rng))
    candidates.append(enforce_sum(1.0 - xs / 2.0, n / 2.0, rng))

    # block masks
    mask = np.zeros(n)
    mask[: n // 2] = 1.0
    candidates.append(enforce_sum(mask, n / 2.0, rng))
    mask = np.zeros(n)
    start = n // 4
    mask[start : start + n // 2] = 1.0
    candidates.append(enforce_sum(mask, n / 2.0, rng))
    mask = np.arange(n) % 2
    candidates.append(enforce_sum(mask, n / 2.0, rng))

    # random vectors
    for _ in range(6):
        candidates.append(random_feasible_vector(n, rng))

    # ---------- select best starter ----------
    best_h = None
    best_c = float("inf")
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    global_h, global_c = best_h, best_c

    # ---------- main optimisation loop ----------
    beta_schedule = [5, 20, 80, 200, 500, 1000, 2000]
    max_hc_iters = 2500

    while time.time() - start_time < budget * 0.70:
        # pick start vector
        if rng.random() < 0.7:
            start_vec = perturb_h(global_h, rng, scale=0.4)
        else:
            start_vec = random_feasible_vector(n, rng)

        # hill‑climb with targeting
        h_opt, c_opt = hill_climb_targeted(
            start_vec,
            rng,
            max_iters=max_hc_iters,
            init_step=0.5,
            target_prob=0.75,
        )
        if c_opt < global_c:
            global_h, global_c = h_opt.copy(), c_opt

        # smooth SLSQP refinements
        for beta in beta_schedule:
            if time.time() - start_time > budget * 0.70:
                break
            try:
                h_ref = slsqp_optimize(global_h, beta, max_iter=400)
            except Exception:
                continue
            c_ref = compute_c5(h_ref)
            if c_ref < global_c:
                global_h, global_c = h_ref.copy(), c_ref

    # ---------- up‑sampling if time permits ----------
    elapsed = time.time() - start_time
    remaining = budget - elapsed
    if remaining > budget * 0.15:
        target_n = min(720, max(480, n * 2))
        if target_n != n:
            old_x = (np.arange(n) + 0.5) * (2.0 / n)
            new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
            h_up = np.interp(new_x, old_x, global_h)
            h_up = enforce_sum(np.clip(h_up, 0.0, 1.0), target_n / 2.0, rng)

            # hill‑climb on finer grid
            h_up_opt, c_up_opt = hill_climb_targeted(
                h_up,
                rng,
                max_iters=3000,
                init_step=0.3,
                target_prob=0.75,
            )
            # SLSQP ladder on fine grid
            for beta in beta_schedule:
                if time.time() - start_time > budget:
                    break
                try:
                    h_up_opt = slsqp_optimize(h_up_opt, beta, max_iter=400)
                except Exception:
                    pass
                c_up_opt = compute_c5(h_up_opt)
                if c_up_opt < global_c:
                    global_h, global_c, n = h_up_opt.copy(), c_up_opt, target_n

    # ---------- final cleanup ----------
    global_h = np.clip(global_h, 0.0, 1.0)
    global_h = enforce_sum(global_h, n / 2.0, rng)
    final_c = compute_c5(global_h)

    return global_h.tolist(), float(final_c), int(n)
