def gradient_refine(h, rng, max_iters, step):
    n = h.shape[0]
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    for it in range(max_iters):
        corr = np.correlate(cur_h, 1 - cur_h, mode='full')
        max_c = np.max(corr)
        k = np.argmax(corr) - (n - 1)
        grad = np.ones(n)
        # compute contributions
        # i+k < n:
        mask = np.arange(n) + k
        in_range = (mask >= 0) & (mask < n)
        grad[in_range] -= cur_h[mask[in_range]]
        # i-k >=0:
        mask2 = np.arange(n) - k
        in_range2 = (mask2 >= 0) & (mask2 < n)
        grad[in_range2] -= cur_h[mask2[in_range2]]
        # pick pos/neg indices
        pos = np.where(grad > 1e-8)[0]
        neg = np.where(grad < -1e-8)[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(cur_h[i], 1 - cur_h[j])
        if max_eps <= 0:
            break
        eps = min(max_eps, step)
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        # compute new max_c
        new_c = compute_c5(new_h)
        if new_c <= cur_c + 1e-12:  # improvement or neutral
            cur_h = new_h
            cur_c = new_c
            # maybe reduce step size gradually?
        else:
            # try smaller epsilon
            eps2 = eps * 0.5
            while eps2 > 1e-6:
                if eps2 > max_eps:
                    eps2 = max_eps
                new_h2 = cur_h.copy()
                new_h2[i] -= eps2
                new_h2[j] += eps2
                new_c2 = compute_c5(new_h2)
                if new_c2 <= cur_c + 1e-12:

                    break
                eps2 *= 0.5

import time, math, numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# basic utilities
# --------------------------------------------------------------
def compute_c5(h):
    """Discrete C5 for step function h on [0,2] with dx=2/n."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / h.size)

def enforce_sum(h, target, rng):
    """Adjust h in‑place so that sum(h)=target while 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    """Random h with 0≤h≤1 and sum=n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    enforce_sum(h, n / 2.0, rng)
    return h

def perturb_h(h, rng, scale=0.2):
    """Transfer ε between two coordinates (preserves sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

def biased_perturb_h(h, rng, scale=0.2):
    """Bias transfer away from positions contributing most to the max overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    k = np.argmax(corr) - (n - 1)
    i0 = max(0, -k)
    i1 = min(n, n - k)
    idx = np.arange(i0, i1)
    prod = h[idx] * (1.0 - h[idx + k])
    if prod.size == 0:
        return perturb_h(h, rng, scale)
    thr_hi = np.quantile(prod, 0.70)
    thr_lo = np.quantile(prod, 0.30)
    high = idx[prod > thr_hi]
    low = idx[prod < thr_lo]
    if high.size == 0 or low.size == 0:
        return perturb_h(h, rng, scale)
    i = rng.choice(high)
    j = rng.choice(low)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

# --------------------------------------------------------------
# simulated annealing
# --------------------------------------------------------------
def simulated_annealing(h0, rng, max_iter, init_temp, final_temp):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    n = h.size
    log_ratio = math.log(final_temp / init_temp) if init_temp > 0 else 0.0
    for it in range(1, max_iter + 1):
        T = init_temp * math.exp(log_ratio * it / max_iter)
        if rng.random() < 0.7:
            cand = biased_perturb_h(h, rng, scale=0.25)
        else:
            cand = perturb_h(h, rng, scale=0.25)
        cand_c = compute_c5(cand)
        diff = cand_c - cur_c
        if diff < 0 or rng.random() < math.exp(-diff / max(T, 1e-12)):
            h, cur_c = cand, cand_c
            if cand_c < best_c:
                best_c, best_h = cand_c, cand.copy()
    return best_h, best_c

# smooth max approximation via SLSQP
# --------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=250):
    """SLSQP on a smooth approximation of max overlap."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {"type": "eq", "fun": sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

# gradient‑based local refinement (focus on worst shift)
# --------------------------------------------------------------
def gradient_for_shift(h, k):
    """∂L_k/∂h_i = 1 - h_{i+k} - h_{i-k} with zero padding."""
    n = h.size
    grad = np.ones(n)
    if k >= 0:
        if k < n:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
    else:
        kk = -k
        if kk < n:
            grad[kk:] -= h[: n - kk]
            grad[: n - kk] -= h[kk:]
    return grad

def gradient_refine(h, rng, max_iters=5000, step_init=0.12):
    """Iteratively move mass from positive‑gradient to negative‑gradient positions
    for the currently worst shift."""
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    step = step_init
    n = cur_h.size
    for _ in range(max_iters):
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        max_c = np.max(corr)
        k = np.argmax(corr) - (n - 1)
        grad = gradient_for_shift(cur_h, k)
        pos = np.where(grad > 1e-9)[0]
        neg = np.where(grad < -1e-9)[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 1e-12:
            # fallback random pair
            i = rng.choice(pos)
            j = rng.choice(neg)
            max_eps = min(cur_h[i], 1.0 - cur_h[j])
            if max_eps <= 1e-12:
                break
        eps = min(step, max_eps)
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        if new_c <= cur_c + 1e-12:
            cur_h, cur_c = new_h, new_c
        else:
            # try smaller eps
            improved = False
            eps2 = eps * 0.5
            while eps2 > 1e-8:
                if eps2 > max_eps:
                    eps2 = max_eps
                new_h2 = cur_h.copy()
                new_h2[i] -= eps2
                new_h2[j] += eps2
                new_c2 = compute_c5(new_h2)
                if new_c2 <= cur_c + 1e-12:
                    cur_h, cur_c = new_h2, new_c2
                    improved = True
                    break
                eps2 *= 0.5
            if not improved:
                step *= 0.5
        step *= 0.9995  # slow decay
    return cur_h, cur_c
