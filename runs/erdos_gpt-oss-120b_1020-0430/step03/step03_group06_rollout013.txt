def simulated_annealing(start_h, rng, time_limit):
    best_h = start_h.copy()
    best_c = compute_c5_fast(best_h)
    cur_h = start_h.copy()
    cur_c = best_c
    start = time.time()
    # temperature schedule: T_start = 0.002
    T_start = 0.002
    # compute total iterations target based on time remaining? We'll just rely on time.
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        progress = elapsed / time_limit
        T = T_start * (1 - progress) + 1e-6  # linear decay down to near zero
        # generate neighbor
        # To speed, we choose random ones and zeros
        ones_idx = np.flatnonzero(cur_h)
        zeros_idx = np.flatnonzero(1 - cur_h)
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        new_h = cur_h.copy()
        new_h[i] = 0
        new_h[j] = 1
        new_c = compute_c5_fast(new_h)
        d = new_c - cur_c
        if d < 0 or rng.random() < np.exp(-d / T):
            cur_h = new_h
            cur_c = new_c
            if new_c < best_c:
                best_h = new_h
                best_c = new_c
    return best_h, best_c

import time
import numpy as np

def compute_c5_fast(h):
    n = h.shape[0]
    a = h.astype(np.float64)
    b = 1.0 - a
    b_rev = b[::-1]
    m = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, m)
    fb = np.fft.rfft(b_rev, m)
    conv = np.fft.irfft(fa * fb, m)[:2*n - 1]
    return conv.max() * (2.0 / n)

def random_binary_vector(n, w, rng):
    v = np.zeros(n, dtype=np.int8)
    v[rng.choice(n, w, replace=False)] = 1
    return v

def hill_climb_binary(h, rng, max_no_improve=2000):
    cur_c = compute_c5_fast(h)
    no_improve = 0
    n = h.size
    while no_improve < max_no_improve:
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        new_h = h.copy()
        new_h[i] = 0
        new_h[j] = 1
        new_c = compute_c5_fast(new_h)
        if new_c < cur_c - 1e-12:
            h = new_h
            cur_c = new_c
            no_improve = 0
        else:
            no_improve += 1
    return h, cur_c

def simulated_annealing(start_h, rng, time_limit):
    best_h = start_h.copy()
    best_c = compute_c5_fast(best_h)
    cur_h = start_h.copy()
    cur_c = best_c
    start = time.time()
    T_initial = 0.002  # tune
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        # linear cooling
        progress = elapsed / time_limit
        T = T_initial * (1.0 - progress) + 1e-8
        # propose swap
        ones = np.flatnonzero(cur_h)
        zeros = np.flatnonzero(1 - cur_h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        new_h = cur_h.copy()
        new_h[i] = 0
        new_h[j] = 1
        new_c = compute_c5_fast(new_h)
        d = new_c - cur_c
        if d < 0 or rng.random() < np.exp(-d / T):
            cur_h = new_h
            cur_c = new_c
            if new_c < best_c:
                best_h = new_h
                best_c = new_c
    return best_h, best_c

import time
import numpy as np

# ------------------------------------------------------------
# Core utilities
# ------------------------------------------------------------
def compute_c5_fast(h):
    """Exact discrete C5 using FFT convolution."""
    n = h.shape[0]
    a = h.astype(np.float64)
    b = 1.0 - a
    b_rev = b[::-1]
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, m)
    fb = np.fft.rfft(b_rev, m)
    conv = np.fft.irfft(fa * fb, m)[: 2 * n - 1]
    return conv.max() * (2.0 / n)


def random_binary_vector(n, w, rng):
    """Uniform random binary vector with exactly w ones."""
    v = np.zeros(n, dtype=np.int8)
    v[rng.choice(n, w, replace=False)] = 1
    return v


def enforce_sum_continuous(h, target, rng):
    """Adjust h to exact sum while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-12:
            continue
        h[i] += add
        diff = target - h.sum()
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(1000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            if abs(add) < 1e-12:
                continue
            h[i] += add
            diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def hill_climb_binary(h, rng, max_no_improve=2000):
    """Greedy hill‑climb on binary vectors (swap 1↔0)."""
    cur_c = compute_c5_fast(h)
    no_imp = 0
    n = h.size
    while no_imp < max_no_improve:
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        new_h = h.copy()
        new_h[i] = 0
        new_h[j] = 1
        new_c = compute_c5_fast(new_h)
        if new_c < cur_c - 1e-12:
            h, cur_c = new_h, new_c
            no_imp = 0
        else:
            no_imp += 1
    return h, cur_c


def simulated_annealing(start_h, rng, time_limit):
    """Simple SA on binary vectors (swap 1↔0) with linear cooling."""
    best_h = start_h.copy()
    best_c = compute_c5_fast(best_h)
    cur_h = start_h.copy()
    cur_c = best_c
    start = time.time()
    T0 = 0.002  # initial temperature
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        prog = elapsed / time_limit
        T = T0 * (1.0 - prog) + 1e-9
        ones = np.flatnonzero(cur_h)
        zeros = np.flatnonzero(1 - cur_h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        new_h = cur_h.copy()
        new_h[i] = 0
        new_h[j] = 1
        new_c = compute_c5_fast(new_h)
        d = new_c - cur_c
        if d < 0 or rng.random() < np.exp(-d / T):
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h, new_c
    return best_h, best_c

def slsqp_optimize_continuous(h0, rng, beta_list=None, max_iter=300):
    if beta_list is None:
        beta_list = [10.0, 40.0, 120.0, 400.0, 800.0]
    from scipy.optimize import minimize

    n = h0.size
    target = n / 2.0
    global _CURRENT_TARGET, _CURRENT_BETA
    _CURRENT_TARGET = target

    def _soft_obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = _CURRENT_BETA * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / _CURRENT_BETA

    def _sum_eq(x):
        return np.sum(x) - _CURRENT_TARGET

    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq}
    best_x = h0.astype(float).copy()
    best_c = compute_c5_fast(best_x)

    for beta in beta_list:
        _CURRENT_BETA = beta
        res = minimize(
            _soft_obj,
            best_x,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
        )
        x = np.clip(res.x, 0.0, 1.0)
        # tiny sum repair
        diff = target - x.sum()
        if abs(diff) > 1e-12:
            for _ in range(2000):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
                if abs(add) < 1e-12:
                    continue
                x[i] += add
                diff = target - x.sum()
        best_x = x
        cur_c = compute_c5_fast(best_x)
        if cur_c < best_c:
            best_c = cur_c
    return best_x, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ---- initialise -------------------------------------------------
    if "initial_h_values" in globals():
        h0_cont = np.array(initial_h_values, dtype=float)
        n = h0_cont.size
        w = n // 2
        # binary start: take the w largest entries
        idx = np.argsort(-h0_cont)
        h_bin = np.zeros(n, dtype=np.int8)
        h_bin[idx[:w]] = 1
        cur_h = h_bin
        cont_start = enforce_sum_continuous(h0_cont, n / 2.0, rng)
    else:
        n = 180
        w = n // 2
        cur_h = random_binary_vector(n, w, rng)
        cont_start = cur_h.astype(float)

    # ---- initial local improvement ---------------------------------
    cur_h, cur_c = hill_climb_binary(cur_h, rng, max_no_improve=2000)
    best_h, best_c = cur_h, cur_c

    # ---- main SA / hill‑climb loop ---------------------------------
    safety = 2.0  # seconds reserved for final polishing
    while True:
        elapsed = time.time() - start_time
        time_left = budget_s - elapsed - safety
        if time_left <= 0.2:
            break
        slice_time = min(5.0, time_left)

        # random restart probability
        if rng.random() < 0.3:
            start_vec = random_binary_vector(n, w, rng)
        else:
            start_vec = best_h

        cand_h, cand_c = simulated_annealing(start_vec, rng, slice_time)

        if cand_c < best_c:
            best_h, best_c = cand_h, cand_c

        # occasional focused hill‑climb
        if rng.random() < 0.25:
            cand_h2, cand_c2 = hill_climb_binary(best_h, rng, max_no_improve=1000)
            if cand_c2 < best_c:
                best_h, best_c = cand_h2, cand_c2

    # ---- final deterministic hill‑climb ----------------------------
    best_h, best_c = hill_climb_binary(best_h, rng, max_no_improve=5000)

    # ---- continuous refinement (both binary and original start) -----
    remaining = budget_s - (time.time() - start_time)
    if remaining > 0.5:
        for start_vec in (best_h.astype(float), cont_start):
            h_opt, c_opt = slsqp_optimize_continuous(
                start_vec, rng, beta_list=[120.0, 400.0, 800.0], max_iter=200
            )
            h_opt = enforce_sum_continuous(h_opt, n / 2.0, rng)
            c_opt = compute_c5_fast(h_opt)
            if c_opt < best_c:
                best_h, best_c = h_opt, c_opt

    # ---- final clean‑up --------------------------------------------
    final_h = np.clip(best_h, 0.0, 1.0)
    final_h = enforce_sum_continuous(final_h, n / 2.0, rng)
    final_c = compute_c5_fast(final_h)

    return final_h.tolist(), float(final_c), int(n)
