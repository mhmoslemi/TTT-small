import time, numpy as np

# --------------------------------------------------------------
# Helper functions
# --------------------------------------------------------------
def compute_c5(h):
    """Discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, rng):
    """Project h onto {0≤h_i≤1, sum h_i = n/2}."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # proportional redistribution respecting bounds
    for _ in range(5):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            cap = 1.0 - h
            tot = cap.sum()
            if tot < 1e-12:
                break
            add = diff * cap / tot
            h = h + add
        else:
            cap = h
            tot = cap.sum()
            if tot < 1e-12:
                break
            sub = diff * cap / tot   # diff negative
            h = h + sub
        diff = target - h.sum()
    # tiny random fix‑up if needed
    if abs(diff) > 1e-12:
        n = h.size
        for _ in range(1000):
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h


def worst_gradient(h):
    """Gradient of the worst‑shift overlap O_k = Σ h_i(1‑h_{i+k})."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        return 1.0 - 2.0 * h
    grad = np.ones_like(h)
    grad[: n - k] -= h[k:]
    grad[k:] -= h[: n - k]
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    best_h = None
    best_c = np.inf
    best_n = None

    # Load initial construction if provided
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)

    # ------------------------------------------------------------------
    # Candidate discretisations

    candidate_ns = [84, 96, 108, 120, 144, 168, 192, 216, 240, 256]

    # ------------------------------------------------------------------
    # Core optimisation primitives
    # ------------------------------------------------------------------
    def smooth_gradient_step(h, beta):
        """One gradient‑descent step on the β‑smoothed max."""
        n = h.size
        corr = np.correlate(h, 1.0 - h, mode="full")
        maxc = np.max(corr)
        # soft‑max weights (stable)
        exps = beta * (corr - maxc)          # ≤ 0
        exps = np.clip(exps, -700, 0)        # avoid overflow
        w = np.exp(exps)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum

        # gradient = 1 − (A + B)
        pad = np.concatenate([np.zeros(n - 1), h, np.zeros(n - 1)])
        A = np.correlate(pad, w, mode="valid")
        B = np.correlate(pad, w[::-1], mode="valid")
        g = 1.0 - (A + B)

        # adaptive step size (max change ≈ 0.5)
        max_abs = np.max(np.abs(g))
        step = 0.5 / max_abs if max_abs > 0 else 0.0
        h_new = h - step * g

        h_new = enforce_sum(h_new, rng)
        return h_new

    def gradient_descent_schedule(h, rng):

        beta_sched = [5, 20, 80, 200, 500, 1000, 2000, 5000]

            for _ in range(steps):

    for n in candidate_ns:

            elif best_h is not None and r == 0:

                h = enforce_sum(h, rng)

            # ---- smooth‑gradient descent ------------------------------------
            h = gradient_descent_schedule(h, rng)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), float(c), n

            # ---- quick hill‑climb -----------------------------------------
            h = quick_hill_climb(h, rng)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), float(c), n

    # ------------------------------------------------------------------
    # Final intensive refinement on the global best (if time permits)
    # ------------------------------------------------------------------
    if best_h is not None and time.time() - start < 0.99 * budget_s:
        h = best_h.copy()
        # deeper gradient descent with higher β and more steps
        beta_sched = [5000, 10000, 20000]
        for beta in beta_sched:
            for _ in range(30):
                if time.time() - start > 0.998 * budget_s:
                    break
                h = smooth_gradient_step(h, beta)
        # final hill‑climb
        h = quick_hill_climb(h, rng)

        if final_c < best_c:

    return best_h.tolist(), float(best_c), int(best_n)
