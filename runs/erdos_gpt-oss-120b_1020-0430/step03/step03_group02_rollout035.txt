def binary_hill_climb(h, rng, max_iters, temp_schedule):
    best_h = h.copy()
    best_c = compute_c5(best_h)
    c = best_c
    n = len(h)
    for it in range(max_iters):
        # select random swap
        i = rng.choice(np.where(best_h==1)[0])
        j = rng.choice(np.where(best_h==0)[0])
        # Swap
        h[i], h[j] = 0, 1
        new_c = compute_c5(h)
        # Accept if better, else accept with probability
        T = temp_schedule(it)
        if new_c < c or rng.random() < np.exp((c - new_c)/T):
            c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = 1, 0
    return best_h, best_c

def mismatched_sets(h, k):
    n = len(h)
    if k >= 0:
        idx_i = np.arange(n-k)
        idx_j = idx_i + k
    else:
        k2 = -k
        idx_i = np.arange(k2, n)
        idx_j = idx_i - k2
    h_i = h[idx_i]
    h_j = h[idx_j]
    A_local = np.where((h_i == 1) & (h_j == 0))[0]  # positions in idx_i
    B_local = np.where((h_i == 0) & (h_j == 1))[0]
    # map to global indices
    A = idx_i[A_local]
    B = idx_i[B_local]
    return A, B

def binary_local_search(h, rng, max_iters, time_limit=None):
    n = len(h)
    best_h = h.copy()
    best_c = compute_c5(h)
    c = best_c
    start = time.time()
    for it in range(max_iters):
        if time_limit is not None and time.time() - start > time_limit:
            break
        # Find worst shift
        corr = np.correlate(h, 1-h, mode='full')
        idx = np.argmax(corr)
        k = idx - (n - 1)

        new_c = compute_c5(h)
        # Adapt acceptance: simple greedy: accept if new_c <= c

def binary_local_search(h, rng, max_iters, time_limit=None):
    n= len(h)
    best_h = h.copy()
    best_c = compute_c5(h)
    c = best_c
    start = time.time()

def mismatched_sets(h, k):
    n = h.shape[0]
    if k >= 0:
        i = np.arange(n - k)
        j = i + k
        hi = h[i]
        hj = h[j]
    else:
        k = -k
        i = np.arange(k, n)
        j = i - k
        hi = h[i]
        hj = h[j]
    # sets
    A = i[(hi == 1) & (hj == 0)]
    B = i[(hi == 0) & (hj == 1)]
    return A, B
```

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------- basic utilities ---------------------------------
def compute_c5(h):
    """Discrete C5 for a step function on [0,2] using zero extension."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, rng):
    """Adjust vector to satisfy Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


# ----------------------- SLSQP refinement --------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log-sum-exp stabilized
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=400):
    """Soft‑max SLSQP refinement for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction (quick redistribution)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ----------------------- binary helpers ---------------------------------
def mismatched_sets(h, k):
    """Return indices i (global) where h_i=1,h_{i+k}=0 (set A) and
    where h_i=0,h_{i+k}=1 (set B) for a given shift k."""
    n = h.shape[0]
    if k >= 0:
        i = np.arange(n - k)
        j = i + k
    else:
        kk = -k
        i = np.arange(kk, n)
        j = i - kk
    hi = h[i]
    hj = h[j]
    A = i[(hi == 1) & (hj == 0)]
    B = i[(hi == 0) & (hj == 1)]
    return A, B


def binary_local_search(h, rng, max_iters=5000, time_limit=None):
    """Greedy hill‑climb on binary vectors using the worst‑shift mismatch."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    start = time.time()
    no_improve = 0

    for it in range(max_iters):
        if time_limit is not None and time.time() - start > time_limit:
            break

        # worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr)
        k = idx - (n - 1)

        A, B = mismatched_sets(h, k)
        if A.size and B.size:
            i = rng.choice(A)
            j = rng.choice(B)
        else:
            i = rng.choice(np.where(h == 1)[0])
            j = rng.choice(np.where(h == 0)[0])

        # swap
        h[i], h[j] = 0.0, 1.0
        new_c = compute_c5(h)

        if new_c < cur_c:
            cur_c = new_c
            no_improve = 0
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = 1.0, 0.0
            no_improve += 1

        if no_improve >= 800:
            break

    return best_h, best_c
