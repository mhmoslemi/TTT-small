def project_h(h, target):
    # clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # adjust sum to target by distributing diff across free (not at bounds)
    # produce free mask
    for _ in range(10):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 0.0 + 1e-12) & (h < 1.0 - 1e-12)
        n_free = np.count_nonzero(free)
        if n_free == 0:
            # all at bounds, cannot adjust; fallback: add diff equally, then clip
            h += diff / h.size
            h = np.clip(h, 0.0, 1.0)
            continue
        delta = diff / n_free
        # only apply to free entries and clip after.
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final small correction using loop adjusting one entry at a time
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # adjust iteratively
        for i in range(h.size):
            if diff == 0:

            if diff > 0 and h[i] < 1.0:

                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
    return h

def project_simplex_bounds(x, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=100):
    # find lambda such that sum(clip(x - lambda, lower, upper)) = target
    # monotonic decreasing as lambda increases (since clip(x - lambda) decreases)
    # binary search between low and high bounds
    lo = -1e6
    hi = 1e6
    for _ in range(max_iter):
        mid = (lo+hi)/2.0
        proj = np.clip(x - mid, lower, upper)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo+hi)/2.0, lower, upper)

def pgd_smooth(h0, target, beta, eta0=0.2, max_iter=2000, decay=0.995):
    h = np.copy(h0)
    eta = eta0
    best_h, best_c = h.copy(), compute_c5(h)
    for t in range(max_iter):
        # compute gradient
        grad = compute_grad_smooth(h, beta)

        # project onto bounds and sum
        h_new = project_simplex_bounds(h_new, target, 0.0, 1.0)
        # evaluate c5
        c = compute_c5(h_new)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
            # maybe increase step size a bit
        # optionally adjust step size based on improvement
        if c < compute_c5(h):
            eta *= 1.02
        else:
            eta *= decay
        # update h for next iteration
        h = h_new

```python
def compute_grad_smooth(h, beta):
    n = h.size
    conv = np.correlate(h, 1 - h, mode='full')
    # Compute weights w (softmax)
    scaled = beta * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / np.sum(exp_scaled)
    grad = np.zeros_like(h)
    # shift indices
    offset = n - 1
    for idx in range(2 * n - 1):
        wi = w[idx]
        shift = idx - offset
        if wi == 0.0:
            continue
        if shift >= 0:
            # i in 0..n-shift-1
            end = n - shift
            if end > 0:
                grad[:end] += wi * (1 - h[shift:])  # h[i+shift]
            # i in shift..n-1
            if shift < n:
                grad[shift:] -= wi * h[:n - shift]
        else:
            # shift < 0, let s = -shift
            s = -shift
            # i in s..n-1 has i+shift >=0
            if s < n:
                grad[s:] += wi * (1 - h[:n - s])
            # i in 0..n-s-1 has i-shift <= n-1
            end = n - s
            if end > 0:
                grad[:end] -= wi * h[s:]
    return grad
```

Check slices: For shift >=0: i from 0..n-1-shift, we use h[i+shift] = h[shift:n]; slice grad[:n-shift] and add wi*(1-h[shift:]). Good. For second term: i from shift..n-1, h[i-shift] = h[:n-shift]; we subtract wi*h[:n-shift] from grad[shift:].

For shift <0 (shift negative): shift = -s. i+shift >=0 => i >= s, i <= n-1. For i in s..n-1: h[i+shift] = h[i-s]; that's h[:n-s]. So grad[s:] += wi*(1 - h[:n-s]). Good. Second term: i-shift <= n-1 => i + s <= n-1 => i <= n-1-s => i in 0..n-1-s. For i in that range: h[i-shift] = h[i+s]; that's h[s:]. So grad[:n-s] -= wi * h[s:].

import time
import numpy as np

def compute_c5(h):
    """Exact discrete C5 for step function h on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_simplex_bounds(x, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=100):
    """Project x onto the set { lower <= y_i <= upper, sum(y)=target }."""
    lo, hi = -1e6, 1e6
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, lower, upper)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, lower, upper)


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    x = rng.random(n)
    return project_simplex_bounds(x, n / 2.0, 0.0, 1.0)


def compute_grad_smooth(h, beta):
    """Gradient of the log‑sum‑exp surrogate of max_k C(k)."""
    n = h.size
    conv = np.correlate(h, 1 - h, mode="full")
    scaled = beta * conv
    m = np.max(scaled)
    exp_scaled = np.exp(scaled - m)
    w = exp_scaled / exp_scaled.sum()            # weights for each shift
    grad = np.zeros_like(h)

    offset = n - 1
    for idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        shift = idx - offset
        if shift >= 0:
            end = n - shift
            if end > 0:
                grad[:end] += wi * (1 - h[shift:])
            grad[shift:] -= wi * h[:end]
        else:
            s = -shift
            end = n - s
            if end > 0:
                grad[s:] += wi * (1 - h[:end])
                grad[:end] -= wi * h[s:]
    return grad


def pgd_smooth(h0, target, beta, eta0=0.2, max_iter=600, decay=0.98):
    """Projected gradient descent on the smooth surrogate."""
    h = h0.copy()
    eta = eta0
    last_c = compute_c5(h)
    for it in range(max_iter):
        grad = compute_grad_smooth(h, beta)
        h_new = h - eta * grad
        h_new = project_simplex_bounds(h_new, target, 0.0, 1.0)

        if (it + 1) % 50 == 0:
            c_new = compute_c5(h_new)
            if c_new > last_c:
                eta *= decay
            else:
                eta *= 1.02
                last_c = c_new
        h = h_new
    return h


def hill_climb(h, target, rng, max_iters=2000, init_step=0.5):
    """Greedy local improvement via pairwise mass transfers."""
    best = h.copy()
    best_c = compute_c5(best)
    step = init_step
    n = best.size
    for _ in range(max_iters):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(best[i], 1.0 - best[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * step
        new_h = best.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c = compute_c5(new_h)
        if c < best_c:
            best, best_c = new_h, c
            step = min(step * 1.02, 0.5)
        else:
            step *= 0.99
    best = project_simplex_bounds(best, target, 0.0, 1.0)
    return best, best_c


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s

    # ---- base dimension and initial vector ---------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_simplex_bounds(h0, n / 2.0, 0.0, 1.0)
    else:
        n = 240
        h0 = random_feasible_vector(n, rng)

    target = n / 2.0

    # ---- assemble a diverse pool -------------------------------------------
    candidates = [h0]
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    for f in (1, 2, 3, 4, 5):
        cand = 0.5 + 0.5 * np.sin(np.pi * f * xs)
        candidates.append(project_simplex_bounds(np.clip(cand, 0, 1), target, 0, 1))
        cand = 0.5 + 0.5 * np.cos(np.pi * f * xs)
        candidates.append(project_simplex_bounds(np.clip(cand, 0, 1), target, 0, 1))
    for _ in range(8):
        candidates.append(random_feasible_vector(n, rng))

    # ---- pick the best among the pool --------------------------------------
    best_h = None
    best_c = float("inf")
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # ---- main optimisation loop --------------------------------------------
    betas = [5, 20, 80, 200, 500, 1000]
    pgd_iters = 600
    while time.time() < deadline * 0.85:
        # start from a perturbation of the current best or a fresh random point
        if rng.random() < 0.6:
            start_vec = best_h + rng.normal(scale=0.05, size=n)
        else:
            start_vec = random_feasible_vector(n, rng)
        start_vec = project_simplex_bounds(start_vec, target, 0.0, 1.0)

        h_cur = start_vec
        for beta in betas:
            if time.time() >= deadline * 0.85:
                break
            h_cur = pgd_smooth(h_cur, target, beta, eta0=0.2,
                               max_iter=pgd_iters, decay=0.98)

        c_cur = compute_c5(h_cur)
        if c_cur < best_c:
            best_c, best_h = c_cur, h_cur.copy()

        # occasional hill‑climb refines the discrete max
        if time.time() < deadline * 0.90:
            hc_h, hc_c = hill_climb(best_h, target, rng,
                                    max_iters=1500, init_step=0.4)
            if hc_c < best_c:
                best_c, best_h = hc_c, hc_h.copy()

    # ---- final clean‑up ----------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_simplex_bounds(best_h, target, 0.0, 1.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
