def worst_shift_and_gradient(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    max_corr = corr[idx]
    k = idx - (n - 1)
    grad = np.ones(n)
    if k >= 0:
        grad[:n-k] -= h[k:]
        grad[k:] -= h[:n-k]
    else:
        kk = -k
        grad[:n-kk] -= h[kk:]
        grad[kk:] -= h[:n-kk]
    return k, grad
```

def perturb_targeted_full(h, rng):
    n = h.size
    k, grad = worst_shift_and_gradient(h)
    pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
    neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
    if pos.size == 0 or neg.size == 0:
        return perturb_random(h, rng, scale=0.2)
    # Choose the most positive gradient and most negative
    i = pos[np.argmax(grad[pos])]
    j = neg[np.argmin(grad[neg])]
    max_eps = min(h[i], 1.0 - h[j])
    # Move full possible epsilon
    eps = max_eps
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h
```

def hill_climb_dynamic(h_start, rng, max_iters, start_step, target_sum):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    dec_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        # compute worst shift and gradient
        k, grad = worst_shift_and_gradient(h)
        # choose move type
        if rng.random() < 0.7: # most of the time use targeted full move
            new_h = perturb_targeted_full(h, rng)
        else:
            new_h = perturb_targeted(h, grad, k, rng, scale=step)
        # Optionally random small moves
        if rng.random() < 0.1:
            new_h = perturb_random(new_h, rng, scale=step*0.5)
        # Ensure feasibility (clip and sum)
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = enforce_sum(new_h, target_sum, rng)
        c_new = compute_c5(new_h)
        if c_new < best_c:
            best_c = c_new
            best_h = new_h.copy()
        # Accept if improvement; else reject or accept with small probability maybe
        if c_new <= compute_c5(h) or rng.random() < 0.01:
            h = new_h
        # Step decay
        if (it+1) % dec_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ----------------- constraints helpers -----------------
CURRENT_TARGET = None
CURRENT_BETA = None

def enforce_sum(h, target, rng):
    """Clip to [0,1] and adjust to make sum exactly target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 5000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:      # need to increase
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i])
            add *= rng.random()
            h[i] += add
            diff -= add
        else:             # need to decrease
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i])
            sub *= rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    # final tiny correction if needed
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        if diff > 0:
            h[i] = min(1.0, h[i] + diff)
        else:
            h[i] = max(0.0, h[i] + diff)
    return h

def random_feasible_vector(n, rng):
    h = rng.random(n)
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, n / 2.0, rng)
    return h

# ----------------- objective -----------------
def compute_c5(h):
    """Exact discrete C5 for vector h."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def worst_shift_and_gradient(h):
    """Return worst shift index k (relative to zero) and gradient of its correlation."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr)
    k = idx - (n - 1)
    grad = np.ones(n)
    if k >= 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    else:
        kk = -k
        grad[: n - kk] -= h[kk:]
        grad[kk:] -= h[: n - kk]
    return k, grad, corr[idx] * (2.0 / n)

# ----------------- perturbations -----------------
def perturb_random(h, rng, scale=0.2):
    """Transfer ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def perturb_targeted(h, grad, scale=0.2):
    """Move ε from a positive‑gradient index to a negative‑gradient one."""
    pos = np.where((grad > 1e-9) & (h > 1e-12))[0]
    neg = np.where((grad < -1e-9) & (h < 1.0 - 1e-12))[0]
    if pos.size == 0 or neg.size == 0:
        return perturb_random(h, np.random.default_rng(), scale)
    i = np.random.choice(pos)
    j = np.random.choice(neg)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return perturb_random(h, np.random.default_rng(), scale)
    eps = max_eps * scale * np.random.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def perturb_targeted_full(h, rng):
    """Transfer the maximal possible ε from the most positive to the most negative gradient entry."""
    k, grad, _ = worst_shift_and_gradient(h)
    pos = np.where((grad > 1e-9) & (h > 1e-12))[0]
    neg = np.where((grad < -1e-9) & (h < 1.0 - 1e-12))[0]
    if pos.size == 0 or neg.size == 0:
        return perturb_random(h, rng, scale=0.2)
    i = pos[np.argmax(grad[pos])]
    j = neg[np.argmin(grad[neg])]
    max_eps = min(h[i], 1.0 - h[j])
    new_h = h.copy()
    new_h[i] -= max_eps
    new_h[j] += max_eps
    return new_h

# ----------------- local search -----------------
def hill_climb_dynamic(start, rng, max_iters, start_step, target_sum):
    """Targeted hill climbing with occasional full‑gradient moves."""
    h = start.copy()
    n = h.size
    # initial correlation
    _, _, c_current = worst_shift_and_gradient(h)
    best_h = h.copy()
    best_c = c_current
    step = start_step
    decay_interval = max(1, max_iters // 10)
    stagnation = 0
    for it in range(max_iters):
        # compute worst shift gradient
        k, grad, c_current = worst_shift_and_gradient(h)

        # choose move type
        if rng.random() < 0.7:
            new_h = perturb_targeted_full(h, rng)
        else:
            new_h = perturb_targeted(h, grad, scale=step)

        if rng.random() < 0.1:
            new_h = perturb_random(new_h, rng, scale=step * 0.5)

        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = enforce_sum(new_h, target_sum, rng)
        c_new = compute_c5(new_h)

        if c_new < best_c:
            best_c = c_new
            best_h = new_h.copy()
            stagnation = 0
        else:
            stagnation += 1

        if c_new <= c_current or rng.random() < 0.01:
            h = new_h
            c_current = c_new

        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)

        if stagnation > 200:
            break
    return best_h, best_c

# ----------------- SLSQP refinement -----------------
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    rng = np.random.default_rng()
    x = enforce_sum(x, CURRENT_TARGET, rng)
    return x

    best_h = h.copy()

    # ----- main optimisation loop -----
    while time.time() - start_time < budget_s * 0.85:

        else:

    while time.time() - start_time < budget_s * 0.99:
        # temperature decays roughly quadratically
        elapsed = (time.time() - start_time) / budget_s
        T = max(0.02 * (1 - elapsed) ** 2, 1e-6)

    best_h = np.clip(best_h, 0.0, 1.0)
