CURRENT_TARGET = None
CURRENT_BETA = None
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET
def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log-sum-exp scaled
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
def slsqp_optimize(h0, beta, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n/2.0
    CURRENT_BETA = beta
    cons = {"type":"eq","fun":sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter':max_iter, 'ftol':1e-12, 'disp':False})
    x = np.clip(res.x, 0, 1)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

import time, math, numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h):
    """Discrete C5 value for step function h on [0,2]."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)


def compute_c5_and_shift(h):
    """Return (C5 value, optimal shift k, full correlation array)."""
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr)
    shift = idx - (h.size - 1)          # shift k such that corr corresponds to lag k
    val = corr[idx] * (2.0 / h.size)
    return val, shift, corr


def enforce_sum(h, target, rng):
    """Adjust h in‑place so that sum(h)=target while respecting 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate random h with 0≤h≤1 and sum=n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    enforce_sum(h, n / 2.0, rng)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer ε between two coordinates (preserves sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new


def deterministic_move(h, shift, eps_max, rng):
    """Move mass from a high‑product index to a low‑product index for the worst shift."""
    n = h.size
    if shift >= 0:
        idx = np.arange(0, n - shift)
        if idx.size == 0:
            return h, False
        prod = h[idx] * (1.0 - h[idx + shift])
        i = idx[np.argmax(prod)]
        j = idx[np.argmin(prod)]
        max_eps = min(h[i], 1.0 - h[j], eps_max)
        if max_eps <= 0:
            return h, False
        eps = max_eps * rng.random()
        if eps <= 0:
            return h, False
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        return new_h, True
    else:
        s = -shift
        idx = np.arange(s, n)
        if idx.size == 0:
            return h, False
        prod = h[idx] * (1.0 - h[idx - s])
        i = idx[np.argmax(prod)]
        j = idx[np.argmin(prod)]
        max_eps = min(h[i], 1.0 - h[j], eps_max)
        if max_eps <= 0:
            return h, False
        eps = max_eps * rng.random()
        if eps <= 0:
            return h, False
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        return new_h, True


# ---------- SLSQP surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp scaled by beta
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=500):
    """Refine h using a smooth approximation of the max correlation."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {"type": "eq", "fun": sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x


# ---------- main optimisation ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # initialise
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.size
        h = np.clip(h, 0.0, 1.0)
        enforce_sum(h, n / 2.0, rng)
    else:
        n = 180
        h = random_feasible_vector(n, rng)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    cur_c, cur_shift, _ = compute_c5_and_shift(h)

    # annealing parameters
    init_temp = 0.02
    final_temp = 1e-6
    eps_max = 0.05
    it = 0
    # main loop (approx. 85 % of budget)
    while time.time() - start < budget_s * 0.85:
        it += 1
        # optional epsilon decay
        if it % 5000 == 0:
            eps_max = max(eps_max * 0.5, 0.001)

        # temperature based on elapsed fraction
        elapsed = time.time() - start
        frac = elapsed / (budget_s * 0.85)
        T = init_temp * (final_temp / init_temp) ** frac

        # deterministic improvement on the worst shift
        new_h, moved = deterministic_move(h, cur_shift, eps_max, rng)
        if moved:
            new_c = compute_c5(new_h)
            if new_c < cur_c or rng.random() < math.exp(-(new_c - cur_c) / max(T, 1e-12)):
                h = new_h
                cur_c = new_c
                if cur_c < best_c:
                    best_c, best_h = cur_c, h.copy()
                cur_c, cur_shift, _ = compute_c5_and_shift(h)
                continue

        # fallback random perturbation
        cand = perturb_h(h, rng, scale=0.2)
        cand_c = compute_c5(cand)
        if cand_c < cur_c or rng.random() < math.exp(-(cand_c - cur_c) / max(T, 1e-12)):
            h = cand
            cur_c = cand_c
            if cur_c < best_c:
                best_c, best_h = cur_c, h.copy()
            cur_c, cur_shift, _ = compute_c5_and_shift(h)

        # occasional full sum re‑enforcement to avoid drift
        if it % 10000 == 0:
            enforce_sum(h, n / 2.0, rng)

    # ---------- final polishing with SLSQP ----------
    for beta in (200, 800, 2000, 8000, 20000):
        if time.time() - start > budget_s * 0.98:
            break
        try:
            refined = slsqp_optimize(best_h, beta, max_iter=800)
            rc = compute_c5(refined)
            if rc < best_c:
                best_c, best_h = rc, refined.copy()
        except Exception:
            continue

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
