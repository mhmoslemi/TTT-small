def anneal(h, rng, max_iters, init_temp, final_temp, max_step_scale):
    n = h.size
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    # compute temperature schedule
    for it in range(max_iters):
        t = init_temp * (final_temp/init_temp)**(it/max_iters)  # exponential decay
        # propose move:
        if cur_h.dtype == np.int64 or np.all(np.isin(cur_h, [0,1])):
            # binary swap
            ones = np.where(cur_h > 0.5)[0]
            zeros = np.where(cur_h < 0.5)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            nh = cur_h.copy()
            nh[i] = 0
            nh[j] = 1
        else:
            # real move
            # pick i to reduce (positive gradient) approx by picking index with high h maybe?
            # we could just pick random i with h[i] > 0, random j with h[j] < 1
            i = rng.integers(0, n)
            while cur_h[i] <= 1e-12:
                i = rng.integers(0, n)
            j = rng.integers(0, n)
            while cur_h[j] >= 1-1e-12:
                j = rng.integers(0, n)
            max_eps = min(cur_h[i], 1.0 - cur_h[j])
            eps = rng.random() * max_eps * max_step_scale
            nh = cur_h.copy()
            nh[i] -= eps
            nh[j] += eps
        nc = compute_c5(nh)
        if nc < cur_c or rng.random() < np.exp(-(nc - cur_c) / t):
            cur_h, cur_c = nh, nc
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
        # optionally reduce max_step_scale gradually
        max_step_scale *= 0.9995

    return best_h, best_c

def climb_one_step(h, grad, rng, scale):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps > 0:
        eps = rng.random() * max_eps * scale
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
    else:
        # binary swap
        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
    return h_new

def simulated_annealing(h, rng, max_iters, init_temp, final_temp, scale):

    n = h.size

    cur_h = h.copy()
    cur_c = compute_c5(cur_h)

    best_h, best_c = cur_h.copy(), cur_c

    for it in range(max_iters):
        # temperature schedule
        t = init_temp * ((final_temp / init_temp) ** (it / max_iters))

        # compute worst gradient for targeted shift (most overlap) to direct move.
        grad = worst_gradient(cur_h)
        # propose move:
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps > 1e-12:
            eps = rng.random() * max_eps * scale
            nh = cur_h.copy()
            nh[i] -= eps
            nh[j] += eps
        else:
            # binary swap (if both are near extremes)
            # find another candidate j2 for swap
            # We'll just swap i and j directly if they are 0/1 complementary
            nh = cur_h.copy()
            # swap values
            temp = nh[i]
            nh[i] = nh[j]
            nh[j] = temp
        nc = compute_c5(nh)
        delta = nc - cur_c

            cur_h, cur_c = nh, nc
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
        # gradually reduce scale
        scale *= 0.9995
    return best_h, best_c

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] (linear correlation)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while keeping 0≤h≤1."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_binary_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector in [0,1] with exact sum n/2."""
    h = rng.random(n)
    target = n / 2.0
    h = h * target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break

        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst (max) overlap shift."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray, rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    nh = h.copy()
    if max_eps > 1e-12:
        eps = rng.random() * max_eps * scale
        nh[i] -= eps
        nh[j] += eps
    else:
        # binary‑type swap
        nh[i], nh[j] = nh[j], nh[i]
    return nh


def guided_hill_climb(h0: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.5) -> tuple:
    """Hill‑climb using the worst‑shift gradient."""
    h = h0.copy()
    best_h, best_c = h.copy(), compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

# SLSQP with soft‑max surrogate
# --------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp, scaled back
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 400) -> np.ndarray:
    """SLSQP minimisation of soft‑max surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def simulated_annealing(h0: np.ndarray, rng: np.random.Generator,
                        max_iters: int = 2000,
                        init_temp: float = 0.001,
                        final_temp: float = 1e-6,
                        scale: float = 0.5) -> tuple:

    cur = h0.copy()
    cur_c = compute_c5(cur)
    best_h, best_c = cur.copy(), cur_c
    for it in range(max_iters):
        t = init_temp * ((final_temp / init_temp) ** (it / max_iters))
        grad = worst_gradient(cur)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        max_eps = min(cur[i], 1.0 - cur[j])
        if max_eps > 1e-12:
            eps = rng.random() * max_eps * scale
            nh = cur.copy()
            nh[i] -= eps
            nh[j] += eps
        else:
            # binary‑type swap
            ones = np.where(cur > 0.5)[0]
            zeros = np.where(cur < 0.5)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i2 = rng.choice(ones)
            j2 = rng.choice(zeros)
            nh = cur.copy()
            nh[i2] = 0.0
            nh[j2] = 1.0
        nc = compute_c5(nh)
        delta = nc - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-12)):
            cur, cur_c = nh, nc
            if cur_c < best_c:
                best_h, best_c = cur.copy(), cur_c
        scale *= 0.9995
    return best_h, best_c


# --------------------------------------------------------------
# main driver
# --------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # ------------------------------------------------------------------
    # start from parent construction if present
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0.copy(), c0, h0.size

    # candidate discretisations (even sizes)
    cand = [84, 96, 108, 120, 144, 168, 192, 216, 240, 288, 336, 384, 432, 480]
    if best_n is not None and best_n in cand:
        cand = [best_n] + [n for n in cand if n != best_n]

    # ------------------------------------------------------------------
    # optimisation loop over discretisations
    # ------------------------------------------------------------------
    for n in cand:
        if time.time() - start > 0.92 * budget_s:
            break

        # initialise reference shape for this n
        if best_h is not None and best_n != n:
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h_ref = np.interp(x_new, x_old, best_h)
            h_ref = np.clip(h_ref, 0.0, 1.0)
            h_ref = enforce_sum(h_ref, rng)
        else:
            h_ref = best_h.copy() if best_h is not None else None

        # number of independent trials (time‑adaptive)
        remaining = budget_s - (time.time() - start)
        max_trials = min(25, int(max(1, remaining / 0.04)))  # crude budget split

        for _ in range(max_trials):
            if time.time() - start > 0.94 * budget_s:
                break

            # random start – mix binary / real and optionally blend with reference
            if rng.random() < 0.5:
                h = random_binary_vector(n, rng)
            else:
                h = random_feasible_vector(n, rng)

            if h_ref is not None and rng.random() < 0.3:
                t = rng.random()
                h = t * h_ref + (1.0 - t) * h
                h = np.clip(h, 0.0, 1.0)
                h = enforce_sum(h, rng)

            # simulated annealing phase
            h_sa, c_sa = simulated_annealing(
                h, rng, max_iters=2000, init_temp=1e-3, final_temp=1e-6, scale=0.5
            )
            if c_sa < best_c:
                best_h, best_c, best_n = h_sa.copy(), c_sa, n

            # short guided hill‑climb polish
            h_gc, c_gc = guided_hill_climb(h_sa, rng, max_iters=1500, step_scale=0.4)
            if c_gc < best_c:
                best_h, best_c, best_n = h_gc.copy(), c_gc, n

        # local SLSQP refinement for the current best (if size matches)
        if best_h is not None and best_n == n:
            for beta in (10.0, 50.0, 200.0, 800.0, 2000.0):
                if time.time() - start > 0.96 * budget_s:
                    break
                best_h = slsqp_optimize(best_h, beta, rng, max_iter=300)
                best_c = compute_c5(best_h)

    # ------------------------------------------------------------------
    # final up‑sampling and fine optimisation (if time permits)
    # ------------------------------------------------------------------
    if best_h is not None and time.time() - start < 0.98 * budget_s:
        # up‑sample up to ~480 points (if beneficial)
        up_n = best_n * 2
        if up_n <= 480:
            h_up = np.repeat(best_h, 2)
            h_up = enforce_sum(h_up, rng)
            h_up, c_up = simulated_annealing(
                h_up, rng, max_iters=3000, init_temp=1e-3, final_temp=1e-6, scale=0.3
            )
            if c_up < best_c:
                best_h, best_c, best_n = h_up.copy(), c_up, up_n
            # brief SLSQP polish on the enlarged version
            for beta in (20.0, 80.0, 200.0):
                if time.time() - start > budget_s:
                    break
                best_h = slsqp_optimize(best_h, beta, rng, max_iter=200)
                best_c = compute_c5(best_h)

    # ------------------------------------------------------------------
    # final tiny random tweaks while time remains
    # ------------------------------------------------------------------
    while time.time() - start < 0.999 * budget_s and best_h is not None:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # ------------------------------------------------------------------
    # final feasibility enforcement & return
    # ------------------------------------------------------------------
    if best_h is None:
        best_n = 84
        best_h = np.full(best_n, 0.5)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
