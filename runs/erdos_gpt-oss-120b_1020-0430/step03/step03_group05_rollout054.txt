def simulated_annealing(h0, rng, max_time):
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    cur_val = best_val
    target = h.size / 2.0
    T0 = 0.2  # initial step magnitude
    # schedule: T = T0 * (1 - (t / max_time))
    while time.time() - start < max_time:
        elapsed = time.time() - start
        T = max(0.01, T0 * (1 - elapsed / max_time))
        # propose perturbation (transfer epsilon)
        i, j = rng.integers(0, h.size, size=2)
        if i == j:
            continue
        # compute max eps based on bounds
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2 * max_eps * T
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        # evaluate
        new_val = compute_c5(new_h)
        d = new_val - cur_val
        if d < 0 or rng.random() < np.exp(-d / (T + 1e-9)):
            h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
    return best_h, best_val

import time
import math
import numpy as np
from scipy.optimize import minimize

# ------- Global placeholders for soft-max surrogate (used by SLSQP) -------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) using log-sum-exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_to_feasible(v, target):
    """Projection onto {0≤h≤1, Σh = target} via a bisection on the shift."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Fast random feasible vector (0≤h≤1, Σh=n/2)."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_to_feasible(v, n / 2.0)


def random_binary_vector(n, rng):
    """Random binary vector with exactly n/2 ones (n assumed even)."""
    half = n // 2
    vec = np.zeros(n, dtype=float)
    ones = rng.choice(n, size=half, replace=False)
    vec[ones] = 1.0
    # if n is odd, adjust the remaining half‑unit
    if n % 2:
        diff = (n / 2.0) - vec.sum()
        if diff > 0:
            i = rng.integers(0, n)
            vec[i] = diff
        elif diff < 0:
            i = rng.integers(0, n)
            vec[i] = max(0.0, vec[i] + diff)
    return vec


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.3):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


def slsqp_optimize(h0, beta, rng, max_iter=300):
    """Run a single SLSQP optimisation with the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction (project onto feasible hyperplane)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def upsample_initial(to_n):
    """Linear interpolation of the global initial_h_values to length to_n."""
    init = np.array(initial_h_values, dtype=float)
    m = init.size
    xs_old = np.linspace(0.0, 2.0, m, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, to_n, endpoint=False)
    return np.interp(xs_new, xs_old, init)


def simulated_annealing(h0, rng, max_time):
    """Temperature‑controlled SA that preserves the sum constraint."""
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    cur_val = best_val
    n = h.size
    while time.time() - start < max_time:
        elapsed = time.time() - start
        # linear cooling from 0.2 to 0.01
        T = max(0.01, 0.2 * (1.0 - elapsed / max_time))
        new_h = perturb_h(h, rng, scale=T)
        new_val = compute_c5(new_h)
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
    return best_h, best_val


def optimise_n(n, rng, time_budget):
    """Optimization pipeline for a fixed discretisation size n."""
    target = n / 2.0
    best_h = None
    best_val = np.inf
    start = time.time()

    # ---- generate diverse starting points ---------------------------------
    starts = []

    if "initial_h_values" in globals():
        init = upsample_initial(n)
        starts.append(project_to_feasible(init, target))

    starts.append(random_binary_vector(n, rng).astype(float))
    while len(starts) < 5:
        starts.append(random_feasible_vector(n, rng))

    # ---- explore each start -------------------------------------------------
    for init_h in starts:
        if time.time() - start > time_budget * 0.95:
            break

        # local deterministic hill climb
        h1, v1 = hill_climb(init_h, rng, max_iters=1200, start_step=0.3)
        if v1 < best_val:
            best_val, best_h = v1, h1.copy()

        # simulated annealing (continuous)
        remaining = time_budget - (time.time() - start)
        if remaining <= 0:
            break
        h2, v2 = simulated_annealing(best_h, rng, max_time=min(remaining, 20.0))
        if v2 < best_val:
            best_val, best_h = v2, h2.copy()

        # SLSQP refinement with growing beta
        for beta in (10.0, 50.0, 200.0, 800.0, 2000.0):
            if time.time() - start > time_budget * 0.99:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, rng, max_iter=300)
                v_ref = compute_c5(h_ref)
                if v_ref < best_val:
                    best_val, best_h = v_ref, h_ref.copy()
            except Exception:
                continue

    # final polish if time permits
    if time.time() - start < time_budget * 0.99:
        h_fin, v_fin = hill_climb(best_h, rng, max_iters=2000, start_step=0.2)
        if v_fin < best_val:
            best_val, best_h = v_fin, h_fin.copy()

    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):

    Optimise a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    best_h = None
    best_c = np.inf
    best_n = None

    # candidate discretisations – larger ones first (more time while budget is high)
    candidate_ns = [300, 240, 168, 120, 84, 60]
    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        if n0 not in candidate_ns:
            candidate_ns = [n0] + candidate_ns

    for n in candidate_ns:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.98:
            break
        remaining = budget_s - elapsed
        # allocate a modest slice of the remaining budget; ensure at least 5 s
        time_for_n = min(remaining * 0.6, max(5.0, n * 0.25))
        h_n, c_n = optimise_n(n, rng, time_for_n)
        if c_n < best_c:
            best_c, best_h, best_n = c_n, h_n.copy(), n

    # fallback in the unlikely event nothing was found
    if best_h is None:
        best_n = 60
        best_h = np.full(best_n, 0.5)

    # final feasibility enforcement (tiny numeric corrections)
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, best_n)
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
