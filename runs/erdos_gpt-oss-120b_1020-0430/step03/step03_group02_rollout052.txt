def anneal(h, budget_s, start_time):
   best_h = h.copy()
   best_c = compute_c5(h)
   temperature = 0.01  # start
   # Steps parameters:
   step_scale = 0.5
   while time.time() - start_time < budget_s:
       # compute worst lag and gradient
       corr = linear_correlation(h)
       idx = int(np.argmax(corr))
       lag = idx - (h.size - 1)
       # compute gradient for worst lag
       grad = compute_gradient(h, lag)  # gradient vector
       # choose i (positive g) and j (negative g)
       pos_idx = np.where(grad > 1e-12)[0]
       neg_idx = np.where(grad < -1e-12)[0]
       if pos_idx.size == 0 or neg_idx.size == 0:
           break
       i = rng.choice(pos_idx)
       j = rng.choice(neg_idx)
       max_eps = min(h[i], 1.0 - h[j])
       # also limit size by gradient magnitude
       eps = max_eps * step_scale * rng.random()
       # if eps very small, reduce step_scale gradually
       h_new = h.copy()
       h_new[i] -= eps
       h_new[j] += eps
       # compute new C5
       c_new = compute_c5(h_new)
       # acceptance
       if c_new < best_c:
           best_c = c_new
           best_h = h_new.copy()
       if c_new < compute_c5(h) or rng.random() < np.exp((compute_c5(h) - c_new) / (temperature + 1e-12)):
           h = h_new
           # maybe update temperature gradually:
       # schedule
       elapsed = time.time() - start_time
       # degrade temperature and step_scale with time
       temperature = max(0.001, temperature * (1 - 0.0005))  # example
       step_scale = max(0.01, step_scale * (1 - 0.0002))
   return best_h, best_c

def worst_shift_and_gradient(h):
    n = h.shape[0]
    corr = compute_corr_fft(h)
    idx = int(np.argmax(corr))
    lag = idx - (n - 1)
    grad = np.ones(n)

        k = lag
        grad[:n - k] -= h[k:]
        grad[k:] -= h[:n - k]
    else:
        k = -lag
        grad[k:] -= h[:n - k]
        grad[:n - k] -= h[k:]
    return lag, grad, corr

        elapsed = time.time() - start_time
        frac = elapsed / max_seconds
        temperature = temp_start * (temp_end / temp_start) ** frac  # exponential decay
        # Also step scale decays

import time, numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

# -------------------------------------------------------------
def enforce_sum(h, target, rng, max_iter=10000):
    diff = target - h.sum()
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:
            cand = np.where(h < 1 - 1e-12)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            add = min(diff, 1 - h[i])
            h[i] += add
        else:
            cand = np.where(h > 1e-12)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            add = max(diff, -h[i])
            h[i] += add
        diff = target - h.sum()
        it += 1
    if abs(diff) > 1e-12:
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h[0] += diff
    return h

def random_feasible(n, rng):
    target = n / 2.0
    h = rng.random(n)
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, target, rng)

# -------------------------------------------------------------
def compute_corr_fft(h):
    n = h.shape[0]
    m = 1 << ((2 * n - 1).bit_length())          # next power of two ≥ 2n‑1
    H = np.fft.rfft(np.r_[h, np.zeros(m - n)])
    G = np.fft.rfft(np.r_[1 - h, np.zeros(m - n)])
    corr = np.fft.irfft(H * np.conj(G), m)[: 2 * n - 1]
    return corr

def compute_c5_fft(h):
    n = h.shape[0]
    return np.max(compute_corr_fft(h)) * (2.0 / n)

def worst_shift_and_gradient(h):
    n = h.size
    corr = compute_corr_fft(h)
    idx = int(np.argmax(corr))
    lag = idx - (n - 1)                     # shift k
    grad = np.ones(n)
    if lag == 0:
        grad -= 2 * h
    elif lag > 0:
        k = lag
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    else:
        k = -lag
        grad[k:] -= h[: n - k]
        grad[: n - k] -= h[k:]
    return lag, grad, corr

# -------------------------------------------------------------
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = compute_corr_fft(x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj, h0, method="SLSQP", bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) <= 1e-12:
                break
    return x

def anneal_h(h0, rng, max_seconds, start_time):
    h = enforce_sum(h0.copy(), h0.size / 2.0, rng)
    n = h.size
    best_h = h.copy()
    best_c = compute_c5_fft(h)
    temp_start, temp_end = 0.02, 0.0002
    while time.time() - start_time < max_seconds:
        lag, grad, corr = worst_shift_and_gradient(h)
        cur_c = np.max(corr) * (2.0 / n)

        elapsed = time.time() - start_time
        frac = (elapsed) / max_seconds
        temperature = temp_start * (temp_end / temp_start) ** frac
        step_scale = 0.5 * (0.3 ** frac)

        pos = np.where(grad > 1e-12)[0]
        neg = np.where(grad < -1e-12)[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = rng.choice(pos)
        j = rng.choice(neg)
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = max_eps * step_scale * rng.random()
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps

        new_c = compute_c5_fft(h_new)
        if new_c < best_c:
            best_c, best_h = new_c, h_new.copy()

        accept = False
        if new_c < cur_c:
            accept = True
        else:
            prob = np.exp((cur_c - new_c) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            h = h_new
    return best_h, best_c
