def enforce_sum_vec(x, target):
    # Clip
    x = np.clip(x, 0, 1)
    s = target - x.sum()
    if abs(s) < 1e-12:
        return x
    # Determine indices that are not at bound
    idx = np.where((x > 0) & (x < 1))[0]
    if idx.size == 0:
        # fallback to random adjustments similar to enforce_sum
        # We'll just adjust random elements with capacity.
        rng = np.random.default_rng()
        for _ in range(1000):
            i = rng.integers(0, len(x))
            capacity = 1 - x[i] if s > 0 else -x[i]
            if capacity == 0: continue
            delta = np.sign(s) * min(abs(s), capacity) * 0.5 # half?
            x[i] += delta
            s = target - x.sum()
            if abs(s) < 1e-12:
                break
        return x
    # Distribute s evenly among idx, but ensure you do not exceed bounds
    # We'll iterate adjusting until leftover s is small
    while True:
        nidx = idx.size
        if nidx == 0 or abs(s) < 1e-12:
            break
        delta = s / nidx
        # Clip per element to not exceed bounds
        for i in idx:
            if delta > 0:
                cap = 1 - x[i]
                add = min(delta, cap)
            else:
                cap = -x[i]
                add = max(delta, cap)  # delta negative
            x[i] += add
            s = target - x.sum()
        # recompute idx
        idx = np.where((x > 0) & (x < 1))[0]
    return x

def biased_perturb_h(h, rng, scale=0.2):
    n = h.size
    # compute correlation
    corr = np.correlate(h, 1 - h, mode="full")
    k = np.argmax(corr)
    shift = k - (n - 1)
    i0 = max(0, -shift)
    i1 = min(n, n - shift)
    idx = np.arange(i0, i1)
    if idx.size == 0:
        return perturb_h(h, rng, scale)
    prod = h[idx] * (1 - h[idx + shift])
    # choose high product source i (among idx)
    # choose low product destination j (any index maybe)
    # Let's pick high among idx and low among complement
    # Actually destination can be any index; choose random if low not exist.
    thr_hi = np.quantile(prod, 0.7)
    high = idx[prod > thr_hi]
    if high.size == 0:
        high = idx
    i = rng.choice(high)
    # Destination j: choose index with low product (if any) else random
    # Compute product for all positions? Not needed.
    # Choose j not equal i
    # To bias away from high product: choose j where product smaller => maybe use low in same idx elements.
    # Let's compute product at all positions with shift?
    # We'll compute full product array:
    prod_full = np.zeros(n)
    prod_full[idx] = prod
    thr_lo = np.quantile(prod_full, 0.3)
    low_candidates = np.where(prod_full < thr_lo)[0]
    # Remove i if present.
    low_candidates = low_candidates[low_candidates != i]
    if low_candidates.size == 0:
        # choose random j
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1

    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:

    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps

import time, math, numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h):
    """Discrete C5 for step function h with dx = 2/n."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / h.size)


def enforce_sum(h, target, rng):
    """Adjust h in‑place so that sum(h)=target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random feasible vector with sum=n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    enforce_sum(h, n / 2.0, rng)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer ε between two coordinates (preserves sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new


def biased_perturb_h(h, rng, scale=0.2):
    """Perturb guided by the shift that currently yields maximal overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    k = np.argmax(corr)
    shift = k - (n - 1)
    i0 = max(0, -shift)
    i1 = min(n, n - shift)
    idx = np.arange(i0, i1)
    if idx.size == 0:
        return perturb_h(h, rng, scale)

    prod = h[idx] * (1.0 - h[idx + shift])
    thr_hi = np.quantile(prod, 0.7)
    high = idx[prod > thr_hi]
    if high.size == 0:
        high = idx

    i = rng.choice(high)

    # build a full product array to spot low‑product positions
    prod_full = np.zeros(n)
    prod_full[idx] = prod
    thr_lo = np.quantile(prod_full, 0.3)
    low_candidates = np.where(prod_full < thr_lo)[0]
    low_candidates = low_candidates[low_candidates != i]
    if low_candidates.size == 0:
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
    else:
        j = rng.choice(low_candidates)

    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable soft‑max
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=200):
    """Smooth‑max SLSQP refinement."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(_softmax_obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x


# ---------- Differential Evolution ----------
def de_optimize(h0, rng, time_limit, pop_size=30, F=0.8, CR=0.9):
    """Simple DE with feasibility projection; stops after time_limit seconds."""
    n = h0.size
    # initialise population: include h0
    pop = np.empty((pop_size, n))
    pop[0] = h0.copy()
    for i in range(1, pop_size):
        pop[i] = random_feasible_vector(n, rng)

    fitness = np.empty(pop_size)
    for i in range(pop_size):
        fitness[i] = compute_c5(pop[i])

    best_idx = np.argmin(fitness)
    best_h = pop[best_idx].copy()
    best_f = fitness[best_idx]

    start = time.time()
    while time.time() - start < time_limit:
        for i in range(pop_size):
            if time.time() - start >= time_limit:
                break
            # mutation: 50% rand/1, 50% best/1
            idxs = list(range(pop_size))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, 3, replace=False)
            if rng.random() < 0.5:
                mutant = pop[a] + F * (pop[b] - pop[c])          # rand/1
            else:
                mutant = best_h + F * (pop[a] - pop[b])          # best/1
            mutant = np.clip(mutant, 0.0, 1.0)
            enforce_sum(mutant, n / 2.0, rng)

            # crossover (binomial)
            trial = pop[i].copy()
            jrand = rng.integers(0, n)
            mask = rng.random(n) < CR
            mask[jrand] = True
            trial[mask] = mutant[mask]

            trial = np.clip(trial, 0.0, 1.0)
            enforce_sum(trial, n / 2.0, rng)

            f = compute_c5(trial)
            if f <= fitness[i]:
                pop[i] = trial
                fitness[i] = f
                if f < best_f:
                    best_f = f
                    best_h = trial.copy()
        # optional early stopping if no improvement for a while
    return best_h, best_f

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----- initialise -----

    # ----- Differential Evolution -----
    de_fraction = 0.65                     # allocate ~65% of time to DE
    de_time = budget_s * de_fraction
    # use a modest population; may increase if time permits

    best_h, best_c = de_optimize(best_h, rng, de_time, pop_size=pop_size,

    # ----- SLSQP polishing -----
