CURRENT_TARGET = None

def _eq_sum(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def _smooth_obj(x, beta):
    """Log‑sum‑exp smooth approximation of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1.0 - x, mode="full")

    m = np.max(bc)

def compute_c5(h):
    """Exact discrete C₅ for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng):
    """Adjust h so that Σh = target while 0≤h≤1 (tiny random repairs)."""
    diff = target - h.sum()
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-12:
            continue
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_binary_vector(n, rng):
    """Generate a random binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h


def slsqp_optimize(h0, beta, max_iter=500):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _eq_sum}
    res = minimize(lambda x: _smooth_obj(x, beta), h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            if abs(add) < 1e-12:
                continue
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    n = 180                     # base resolution
    target = n / 2.0

    # ------------------------------------------------------------------
    # Build a modest pool of diverse starting vectors (binary)
    # ------------------------------------------------------------------
    candidates = []

    # use the parent if supplied
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size == n:
            h0 = np.clip(h0, 0.0, 1.0)
            h0 = enforce_sum(h0, target, rng)
            candidates.append(h0)

    # random binary vectors
    for _ in range(4):
        candidates.append(random_binary_vector(n, rng))

    # Legendre‑type sequence (prime p=179)
    p = 179
    leg = np.empty(p, dtype=float)
    leg[0] = 1.0                     # flip to help the half‑sum
    for i in range(1, p):
        ls = pow(i, (p - 1) // 2, p)
        leg[i] = 1.0 if ls == 1 else 0.0
    # adjust to exact half‑sum
    diff = int(target - leg.sum())
    if diff > 0:
        zeros = np.where(leg == 0)[0]
        leg[zeros[:diff]] = 1.0
    elif diff < 0:
        ones = np.where(leg == 1)[0]
        leg[ones[: -diff]] = 0.0
    # pad / truncate to length n
    if p < n:
        leg_seq = np.concatenate([leg, np.zeros(n - p, dtype=float)])
    else:
        leg_seq = leg[:n]
    leg_seq = enforce_sum(leg_seq, target, rng)
    candidates.append(leg_seq)

    # Thue‑Morse binary sequence
    tm = np.fromiter((bin(i).count("1") & 1 for i in range(n)), dtype=float)
    diff = int(target - tm.sum())
    if diff > 0:
        zeros = np.where(tm == 0)[0]
        tm[zeros[:diff]] = 1.0
    elif diff < 0:
        ones = np.where(tm == 1)[0]
        tm[ones[: -diff]] = 0.0
    candidates.append(tm)

    # sine / cosine based binary patterns
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    for f in (1, 2, 3):
        for fun in (np.sin, np.cos):
            h = 0.5 + 0.5 * fun(np.pi * f * xs)
            h_bin = np.rint(h)            # round to 0/1
            h_bin = enforce_sum(h_bin, target, rng)
            candidates.append(h_bin)

    # ------------------------------------------------------------------
    # Pick the best among the pool
    # ------------------------------------------------------------------
    best_h = None
    best_c = float("inf")
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # ------------------------------------------------------------------
    # Simulated annealing on a binary representation
    # ------------------------------------------------------------------
    cur_h = (best_h > 0.5).astype(float)
    cur_h = enforce_sum(cur_h, target, rng)
    cur_c = compute_c5(cur_h)

    best_h = cur_h.copy()
    best_c = cur_c

    max_sa_steps = 200_000
    max_T = 5e-3
    min_T = 1e-6
    step = 0

    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.70:
            break
        fraction = step / max_sa_steps
        if fraction > 1.0:
            fraction = 1.0
        T = max_T * (1 - fraction) + min_T * fraction

        ones_idx = np.where(cur_h > 0.5)[0]
        zeros_idx = np.where(cur_h < 0.5)[0]
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)

        new_h = cur_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_c = compute_c5(new_h)

        if new_c < cur_c or (T > 0 and rng.random() < math.exp(-(new_c - cur_c) / T)):
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
        step += 1

    # ------------------------------------------------------------------
    # Continuous refinement with SLSQP (smooth surrogate)
    # ------------------------------------------------------------------
    betas = [10, 40, 120, 400, 800, 1600]
    h_opt = best_h.copy()
    for beta in betas:
        if time.time() - start_time > budget_s * 0.90:
            break
        h_opt = slsqp_optimize(h_opt, beta, max_iter=500)
        c_val = compute_c5(h_opt)
        if c_val < best_c:
            best_h, best_c = h_opt.copy(), c_val

    # ------------------------------------------------------------------
    # Optional up‑sampling (double resolution) if time permits
    # ------------------------------------------------------------------
    if time.time() - start_time < budget_s * 0.97 and n * 2 <= 360:
        h_up = np.repeat(best_h, 2)
        n_up = h_up.size
        target_up = n_up / 2.0
        h_up = enforce_sum(h_up, target_up, rng)

        # brief SA on up‑sampled vector
        cur_h = (h_up > 0.5).astype(float)
        cur_h = enforce_sum(cur_h, target_up, rng)
        cur_c = compute_c5(cur_h)
        best_h_up, best_c_up = cur_h.copy(), cur_c
        max_sa_up = 100_000
        step_up = 0
        while True:
            if time.time() - start_time > budget_s * 0.95:
                break
            fraction = step_up / max_sa_up
            if fraction > 1.0:
                fraction = 1.0
            T = max_T * (1 - fraction) + min_T * fraction
            ones_idx = np.where(cur_h > 0.5)[0]
            zeros_idx = np.where(cur_h < 0.5)[0]
            if ones_idx.size == 0 or zeros_idx.size == 0:
                break
            i = rng.choice(ones_idx)
            j = rng.choice(zeros_idx)
            new_h = cur_h.copy()
            new_h[i] = 0.0
            new_h[j] = 1.0
            new_c = compute_c5(new_h)
            if new_c < cur_c or (T > 0 and rng.random() < math.exp(-(new_c - cur_c) / T)):
                cur_h, cur_c = new_h, new_c
                if new_c < best_c_up:
                    best_h_up, best_c_up = new_h.copy(), new_c
            step_up += 1

        # continuous refinement on up‑sampled best
        n = n_up
        target = n / 2.0
        CURRENT_TARGET = target
        h_opt = best_h_up.copy()
        for beta in betas:
            if time.time() - start_time > budget_s:
                break
            h_opt = slsqp_optimize(h_opt, beta, max_iter=500)
            c_val = compute_c5(h_opt)
            if c_val < best_c_up:
                best_h_up, best_c_up = h_opt.copy(), c_val
        best_h, best_c = best_h_up, best_c_up

    # ------------------------------------------------------------------
    # Final clean‑up
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
