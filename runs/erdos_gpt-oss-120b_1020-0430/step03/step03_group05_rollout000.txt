def de_optimize(n, rng, max_time):
    target = n/2.0
    popsize = min(200, max(30, int(n*0.8)))
    # initial population: include oversampled initial_h if present, and random feasible vectors.
    pop = []
    if 'initial_h_values' in globals():
        init = upsample_initial(n)
        pop.append(project_to_feasible(init, target))
    while len(pop) < popsize:
        pop.append(random_feasible_vector(n, rng))
    pop = np.array(pop)
    scores = np.array([compute_c5_fft(ind) for ind in pop])
    best_idx = scores.argmin()
    best = pop[best_idx].copy()
    best_score = scores[best_idx]
    F = 0.8
    CR = 0.9
    start = time.time()
    while time.time() - start < max_time:
        for i in range(popsize):
            # mutate
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, 3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            # clip and project
            # apply crossover
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(0, n)] = True
            trial = np.where(cross, mutant, pop[i])
            # enforce bounds 0-1
            trial = np.clip(trial, 0.0, 1.0)
            trial = project_to_feasible(trial, target)
            trial_score = compute_c5_fft(trial)
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best = trial.copy()
        # optional early stop if no improvement after many generations.
    return best, best_score

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Helper functions ----------
def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_to_feasible(v, target, iters=60):
    """Project v onto {0≤h≤1, Σh=target} by bisection on a shift."""
    lo, hi = -2.0, 2.0
    for _ in range(iters):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_to_feasible(v, n / 2.0)


def upsample_initial(to_n):
    """Linear interpolation of global initial_h_values to length to_n."""
    init = np.array(initial_h_values, dtype=float)
    m = init.size
    xs_old = np.linspace(0.0, 2.0, m, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, to_n, endpoint=False)
    return np.interp(xs_new, xs_old, init)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one index to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def swap_h(h, rng):
    """Swap two entries (keeps the sum exactly)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    new_h = h.copy()
    new_h[i], new_h[j] = h[j], h[i]
    return new_h


# ---------- Differential Evolution ----------
def de_optimize(n, rng, max_time):
    """Simple DE with projection onto sum & box constraints."""
    target = n / 2.0
    popsize = min(200, max(30, int(n * 0.8)))
    pop = []

    if "initial_h_values" in globals():
        init = upsample_initial(n)
        pop.append(project_to_feasible(init, target))

    while len(pop) < popsize:
        pop.append(random_feasible_vector(n, rng))

    pop = np.array(pop)
    scores = np.array([compute_c5(ind) for ind in pop])
    best_idx = scores.argmin()
    best = pop[best_idx].copy()
    best_score = scores[best_idx]

    F = 0.8
    CR = 0.9
    start = time.time()

    while time.time() - start < max_time:
        for i in range(popsize):
            if time.time() - start >= max_time:
                break
            # mutation
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, 3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            # crossover
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(0, n)] = True
            trial = np.where(cross, mutant, pop[i])
            trial = np.clip(trial, 0.0, 1.0)
            trial = project_to_feasible(trial, target)
            trial_score = compute_c5(trial)
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best = trial.copy()
        # optional early‑stop if no improvement for many generations
    return best, best_score


# ---------- SLSQP refinement ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=800):
    """Single SLSQP run on the smooth max‑log‑sum‑exp surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # exact sum correction (project onto feasible hyperplane)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ---------- Main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Choose a reasonably large discretisation.
    # Use 360 points (or fallback to 300 if up‑sampling not wanted).
    if "initial_h_values" in globals():
        n = 360
    else:
        n = 300

    # Total time split: ~70% DE, ~20% SLSQP, remaining hill‑climb.
    elapsed = time.time() - start_time
    remaining = max(0.0, budget_s - elapsed)
    if remaining < 30.0:  # too little budget – fallback to tiny search
        n = 120
        remaining = max(0.0, budget_s - (time.time() - start_time))

    de_time = remaining * 0.68
    slsqp_time = remaining * 0.20
    hill_time = remaining - de_time - slsqp_time

    # ---------- Differential Evolution ----------
    h_cur, c_cur = de_optimize(n, rng, de_time)
    best_h, best_c, best_n = h_cur.copy(), c_cur, n

    # ---------- SLSQP refinement ----------
    slsqp_start = time.time()
    betas = [10.0, 100.0, 500.0, 2000.0, 8000.0]
    for beta in betas:
        if time.time() - start_time > start_time + de_time + slsqp_time:
            break
        h_new = slsqp_optimize(best_h, beta, rng, max_iter=800)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()
    # ---------- Simple hill‑climb & swaps ----------
    hill_start = time.time()
    h_local = best_h.copy()
    c_local = best_c
    step_scale = 0.25
    while time.time() - hill_start < hill_time:
        # perturb move
        cand = perturb_h(h_local, rng, scale=step_scale)
        c_cand = compute_c5(cand)
        if c_cand < c_local:
            h_local, c_local = cand, c_cand
            if c_cand < best_c:
                best_c, best_h = c_cand, cand.copy()
            step_scale = min(0.5, step_scale * 1.05)
        else:
            step_scale = max(0.05, step_scale * 0.97)

        # occasional swap move (preserves sum exactly)
        if rng.random() < 0.3:
            cand2 = swap_h(h_local, rng)
            c_cand2 = compute_c5(cand2)
            if c_cand2 < c_local:
                h_local, c_local = cand2, c_cand2
                if c_cand2 < best_c:
                    best_c, best_h = c_cand2, cand2.copy()

    # ---------- Final feasibility enforcement ----------
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, best_n)
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
