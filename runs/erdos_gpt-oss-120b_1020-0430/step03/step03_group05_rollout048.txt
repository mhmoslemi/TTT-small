def greedy_descent(h, rng, time_limit):
    start = time.time()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    no_improve_iters = 0
    while time.time() - start < time_limit:
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        k = max_idx - (n - 1)
        # compute epsilon max per i
        idx_forward = (np.arange(n) + k) % n
        eps_max = np.minimum(h, 1 - h[idx_forward])
        # improvement estimate: eps_max * ((1 - h[idx_forward]) + h)
        improve_est = eps_max * ((1 - h[idx_forward]) + h)
        # filter zeros
        candidates = np.where(eps_max > 1e-12)[0]
        if candidates.size == 0:
            break
        # choose best candidate
        best_i = candidates[np.argmax(improve_est[candidates])]
        eps = eps_max[best_i] * 0.5  # half of max
        if eps < 1e-12:
            break
        # apply move
        j = idx_forward[best_i]
        h[best_i] -= eps
        h[j] += eps
        # enforce bound due to numeric errors
        # (should be okay)
        # compute new c5
        c = compute_c5(h)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            no_improve_iters = 0
        else:
            no_improve_iters += 1
        # optionally random move if stuck
        if no_improve_iters >= 200:
            # random perturb
            i_rand = rng.integers(0, n)
            j_rand = rng.integers(0, n)
            if i_rand != j_rand:
                max_eps = min(h[i_rand], 1 - h[j_rand])
                if max_eps > 0:
                    eps_rand = max_eps * rng.random() * 0.3
                    h[i_rand] -= eps_rand
                    h[j_rand] += eps_rand
            no_improve_iters = 0
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# Basic utilities
# ------------------------------------------------------------
def compute_c5(h):
    """Exact discrete C5 for step function h on [0,2] (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_to_feasible(v, target):
    """Projection onto {0≤x≤1, Σx = target} via bisection on a shift."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def upsample_parent_to_new_size(parent, new_n):
    """Linear interpolation of a parent vector to a new length."""
    old_n = parent.size
    xs_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(xs_new, xs_old, parent)


# ------------------------------------------------------------
# Smooth surrogate used by SLSQP
# ------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=400):
    """One SLSQP refinement step on the log‑sum‑exp surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # fix possible sum drift
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x


# ------------------------------------------------------------
# Greedy descent based on the worst shift
# ------------------------------------------------------------
def greedy_descent(h_start, rng, time_limit):
    """Iteratively move mass from a 1→0 mismatch of the worst shift."""
    h = h_start.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    no_improve = 0
    start = time.time()

    while time.time() - start < time_limit:
        # correlation and worst shift
        corr = np.correlate(h, 1 - h, mode="full")
        max_idx = int(np.argmax(corr))
        k = max_idx - (n - 1)            # integer shift (may be negative)
        # vector of forward indices i -> i+k (mod n)
        idx_fwd = (np.arange(n) + k) % n

        # maximal transferable epsilon for each i
        eps_max = np.minimum(h, 1.0 - h[idx_fwd])
        feasible = eps_max > 1e-12
        if not np.any(feasible):
            break

        # first‑order improvement estimate for a full transfer
        improve_est = eps_max * ((1.0 - h[idx_fwd]) + h)
        i = np.argmax(improve_est * feasible)   # pick best feasible i
        eps = eps_max[i] * 0.5                  # move half of what is allowed
        if eps < 1e-12:
            break
        j = idx_fwd[i]
        h[i] -= eps
        h[j] += eps

        # occasional re‑projection to counter floating‑point drift
        if (no_improve + 1) % 250 == 0:
            h = project_to_feasible(h, n / 2.0)

        c = compute_c5(h)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # if stuck, try a random small transfer
        if no_improve >= 300:
            i_r = rng.integers(0, n)
            j_r = rng.integers(0, n)
            if i_r != j_r:
                eps_r = min(h[i_r], 1.0 - h[j_r]) * rng.random() * 0.3
                if eps_r > 1e-12:
                    h[i_r] -= eps_r
                    h[j_r] += eps_r
            no_improve = 0

    return best_h, best_c

# Main optimisation driver
# ------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    best_h = None
    best_c = np.inf
    best_n = None

    # candidate discretisations (order from coarse to fine)
    candidate_ns = [60, 84, 120, 168, 240, 300]
    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        if n0 not in candidate_ns:
            candidate_ns = [n0] + candidate_ns

    for n in candidate_ns:
        # time bookkeeping
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining < 8.0:
            break

        # allocate a portion of remaining time to this n
        # (more for larger n)
        alloc = min(remaining * 0.5, max(30.0, n * 0.25))

        # ----- build an initial feasible vector -----
        if "initial_h_values" in globals() and n == len(initial_h_values):
            h0 = np.array(initial_h_values, dtype=float).copy()
        elif "initial_h_values" in globals():
            h0 = upsample_parent_to_new_size(
                np.array(initial_h_values, dtype=float), n
            )
        else:
            h0 = rng.random(n)
        h0 = np.clip(h0, 0.0, 1.0)
        target = n / 2.0
        # project onto feasible set (box + sum)
        h0 = project_to_feasible(h0, target)

        # ----- greedy descent -----
        t_greedy = alloc * 0.6
        h_g, c_g = greedy_descent(h0, rng, t_greedy)
        if c_g < best_c:
            best_c, best_h, best_n = c_g, h_g.copy(), n

        # ----- optional SLSQP polishing -----
        if time.time() - start_time < budget_s * 0.95:
            # a few increasing beta values
            for beta in (20.0, 80.0, 300.0, 1200.0):
                if time.time() - start_time > budget_s * 0.98:
                    break
                h_pol = slsqp_optimize(h_g, beta, rng, max_iter=350)
                c_pol = compute_c5(h_pol)
                if c_pol < best_c:
                    best_c, best_h, best_n = c_pol, h_pol.copy(), n
                h_g = h_pol  # continue polishing from last result

    # fallback if nothing succeeded
    if best_h is None:
        n = 120
        best_h = np.full(n, 0.5)
        best_n = n
        best_c = compute_c5(best_h)

    # final feasibility correction
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            i = rng.integers(0, best_n)
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()
            if abs(diff) < 1e-12:
                break

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
