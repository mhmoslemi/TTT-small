def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # choose n
    n = 180  # default if parent not present; else use parent n
    if 'initial_h_values' in globals():
        init_h = np.array(initial_h_values, dtype=float)
        n = init_h.size
        # convert to binary by threshold 0.5
        binary_h = (init_h > 0.5).astype(int)
        # adjust to sum exactly n/2 (if needed)
        current_sum = binary_h.sum()
        target_sum = n//2
        # if sum too high, turn some ones to zeros at random; else turn zeros to ones
        while current_sum > target_sum:
            idx = rng.choice(np.where(binary_h==1)[0])
            binary_h[idx] = 0
            current_sum -= 1
        while current_sum < target_sum:
            idx = rng.choice(np.where(binary_h==0)[0])
            binary_h[idx] = 1
            current_sum += 1
        h_current = binary_h.astype(float)
    else:
        # random binary with half ones
        h_current = np.zeros(n, dtype=float)
        ones_idx = rng.choice(n, size=n//2, replace=False)
        h_current[ones_idx] = 1.0

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global helpers for the smooth surrogate used by SLSQP
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    """Smooth log‑sum‑exp surrogate of max_{k} Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


# ----------------------------------------------------------------------
# Problem‑specific utilities
# ----------------------------------------------------------------------
def compute_c5(h):
    """Exact discrete C₅ for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng):
    """Adjust h so that Σh = target while 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-12:
            continue
        h[i] += add
        diff = target - h.sum()
    # tiny fallback
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(1000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            if abs(add) < 1e-12:
                continue
            h[i] += add
            diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    v = rng.exponential(1.0, size=n)
    v = v / v.sum() * (n / 2.0)
    h = np.clip(v, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def hill_climb(h_start, rng, max_iters=2500, start_step=0.5):
    """Simple hill‑climb: keep only improving moves."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)
        max_pos = min(h[i], 1.0 - h[j])
        max_neg = min(h[j], 1.0 - h[i])
        if max_pos + max_neg == 0:
            continue
        eps = (rng.random() * (max_pos + max_neg)) - max_neg
        eps *= step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            if abs(add) < 1e-12:
                continue
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ----------------------------------------------------------------------
# Fast incremental correlation updates
# ----------------------------------------------------------------------
def _apply_delta(r, h, i, eps, offset):
    """Update full correlation r after changing h[i] by eps."""
    n = h.size
    # term eps * (1 - h[i + k])
    r[offset - i : offset + (n - i)] += eps * (1.0 - h)
    # term -eps * h[i - k]  (using reversed h)
    r[offset - (n - 1 - i) : offset + i + 1] -= eps * h[::-1]
    # quadratic term at zero shift
    r[offset] -= eps * eps


def _apply_move(h, r, i, j, eps, offset):
    """Transfer eps from i to j (eps may be negative)."""
    # i loses eps, j gains eps
    _apply_delta(r, h, i, -eps, offset)
    h[i] -= eps
    _apply_delta(r, h, j, eps, offset)
    h[j] += eps


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    total_budget = budget_s

    # --------------------------------------------------------------
    # Determine dimension and initial vector
    # --------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 180
        h0 = np.full(n, 0.5)
        h0 = enforce_sum(h0, n / 2.0, rng)

    target = n / 2.0
    offset = n - 1

    # --------------------------------------------------------------
    # Initialise correlation and SA parameters
    # --------------------------------------------------------------
    cur_h = h0.copy()
    cur_r = np.correlate(cur_h, 1 - cur_h, mode="full")
    cur_max = cur_r.max()
    best_h = cur_h.copy()
    best_max = cur_max

    # simulated‑annealing phase (≈70% of the budget)
    time_sa = total_budget * 0.70
    T0 = cur_max * 0.5 + 1e-9
    it = 0
    while time.time() - start_time < time_sa:
        elapsed = time.time() - start_time
        t_frac = elapsed / time_sa
        T = T0 * (1.0 - t_frac)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        max_pos = min(cur_h[i], 1.0 - cur_h[j])
        max_neg = min(cur_h[j], 1.0 - cur_h[i])
        if max_pos + max_neg == 0:
            continue
        eps = (rng.random() * (max_pos + max_neg)) - max_neg

        # propose move
        _apply_move(cur_h, cur_r, i, j, eps, offset)
        new_max = cur_r.max()
        delta = new_max - cur_max

        if delta <= 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
            # accept
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = cur_h.copy()
        else:
            # revert
            _apply_move(cur_h, cur_r, i, j, -eps, offset)

        # periodic exact recomputation to avoid drift
        if it % 2000 == 0:
            cur_r = np.correlate(cur_h, 1 - cur_h, mode="full")
            cur_max = cur_r.max()
        it += 1

    # --------------------------------------------------------------
    # Post‑SA refinement (hill‑climb + occasional SLSQP)
    # --------------------------------------------------------------
    best_h = enforce_sum(best_h, target, rng)

    time_remain = total_budget - (time.time() - start_time)
    # hill‑climb
    if time_remain > 30:
        max_iters = int(min(20000, time_remain * 4000))
        h_hc, c_hc = hill_climb(best_h, rng, max_iters=max_iters, start_step=0.5)
        if c_hc < compute_c5(best_h):
            best_h = h_hc

    # smoothing via SLSQP (few β values)
    time_left = total_budget - (time.time() - start_time)
    if time_left > 20:
        for beta in [20.0, 80.0, 200.0, 800.0]:
            if time.time() - start_time > total_budget * 0.98:
                break
            try:
                h_opt = slsqp_optimize(best_h, beta, max_iter=500)
            except Exception:
                continue
            c_opt = compute_c5(h_opt)
            if c_opt < compute_c5(best_h):
                best_h = h_opt

    # --------------------------------------------------------------
    # Final cleanup
    # --------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
