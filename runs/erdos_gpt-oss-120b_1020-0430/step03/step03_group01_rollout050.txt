# enforce_sum:

def enforce_sum(h, target, rng, max_iter=2000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    # try to distribute diff using greedy steps
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # final correction
    if abs(diff) > 1e-12:
        # distribute evenly with clipping
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
    return h

Now compute_c5:

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

Now perturb_h:

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    # ensure within bounds (though eps limited)
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

Now random_feasible_vector:

def random_feasible_vector(n, rng):
    # start with uniform random values in [0,1]
    h = rng.random(n)
    target = n/2.0
    # scale to meet sum
    total = h.sum()
    if total == 0:
        h = np.full(n, 0.5)
        total = h.sum()
    scale = target / total
    h *= scale
    # clip and repair
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, target, rng)
    return h

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # compute log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Helper functions ----------

def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for step function h on [0,2] (dx = 2/n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / h.size))


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator,
                max_iter: int = 2000) -> np.ndarray:
    """Clip to [0,1] and adjust to meet Σh = target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector with 0≤h_i≤1 and Σh_i = n/2."""
    target = n / 2.0
    h = rng.random(n)
    total = h.sum()
    if total == 0:
        h = np.full(n, 0.5)
        total = h.sum()
    h *= target / total
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, target, rng)
    return h


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def targeted_perturb(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """
    Move mass from an index that hurts the worst shift to one that helps it.
    Uses the shift that maximises the overlap integral (i.e. the worst shift).
    """
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = int(np.argmax(corr) - (n - 1))          # integer shift (may be negative)
    shift = k_star % n                               # treat cyclically for the heuristic
    idx = np.arange(n)
    inc_vals = (1.0 - h[(idx + shift) % n]) - h[(idx - shift) % n]
    idx_inc = int(np.argmax(inc_vals))
    idx_dec = int(np.argmin(inc_vals))
    max_eps = min(1.0 - h[idx_inc], h[idx_dec])
    if max_eps <= 0:
        return perturb_h(h, rng, scale)
    eps = max_eps * scale * rng.random()
    new_h = h.copy()
    new_h[idx_inc] += eps
    new_h[idx_dec] -= eps
    return new_h


def hill_climb_multi(h_start: np.ndarray, rng: np.random.Generator,
                     max_iters: int = 2000, init_step: float = 0.5,
                     n_trials: int = 5) -> tuple[np.ndarray, float]:
    """Simple hill‑climb that tries several random perturbations per step."""
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
                improved = True
                break
        if not improved and rng.random() < 0.02:
            new_h = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
                improved = True
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return h, best_c


# ---------- SLSQP surrogate optimisation ----------

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x: np.ndarray) -> float:
    """Equality constraint Σx_i = CURRENT_TARGET."""
    return float(np.sum(x) - CURRENT_TARGET)


def softmax_obj(x: np.ndarray) -> float:
    """Log‑sum‑exp surrogate of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = float(np.max(bc))
    # log‑sum‑exp in a numerically stable way
    return float((m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA)


def slsqp_optimize(h0: np.ndarray, beta: float, max_iter: int = 300) -> np.ndarray:
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # repair tiny drift in the sum
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        x = enforce_sum(x, CURRENT_TARGET, rng)
    return x


# ---------- Main optimisation routine ----------

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Optimise a step function h on [0,2] to minimise the Erdős C₅ bound."""
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----- initial vector -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ----- simulated annealing with targeted moves -----------------------
    T = 0.02                # initial temperature
    Tmin = 1e-6
    decay_T = 0.9995
    step_scale = 0.5
    decay_step = 0.9999

    cur_h = best_h.copy()
    cur_c = best_c
    iter_cnt = 0
    max_iters = 500000  # safety cap; real stop is time‑based

    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.75:
            break
        if iter_cnt >= max_iters:
            break

        # propose a move (80 % targeted, otherwise random)
        if rng.random() < 0.8:
            new_h = targeted_perturb(cur_h, rng, scale=step_scale)
        else:
            new_h = perturb_h(cur_h, rng, scale=step_scale)

        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        accept = False
        if delta < 0:
            accept = True
        else:
            prob = np.exp(-delta / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c

        # schedule temperature and step size
        T = max(T * decay_T, Tmin)
        step_scale = max(step_scale * decay_step, 1e-4)
        iter_cnt += 1

        # occasional random restart
        if iter_cnt % 20000 == 0 and rng.random() < 0.5:
            cur_h = random_feasible_vector(n, rng)
            cur_c = compute_c5(cur_h)

        # occasional brief hill‑climb (helps escape plateaus)
        if iter_cnt % 50000 == 0:
            hc_h, hc_c = hill_climb_multi(
                cur_h, rng, max_iters=1500, init_step=step_scale, n_trials=6
            )
            cur_h, cur_c = hc_h, hc_c
            if hc_c < best_c:
                best_h, best_c = hc_h.copy(), hc_c

    # ----- medium‑scale hill‑climb from the best found --------------------
    if time.time() - start_time < budget_s * 0.92:
        hc_h, hc_c = hill_climb_multi(
            best_h, rng, max_iters=4000, init_step=0.3, n_trials=8
        )
        if hc_c < best_c:
            best_h, best_c = hc_h, hc_c

    # ----- continuation with SLSQP (increasing β) ------------------------
    beta_schedule = [5, 20, 80, 200, 500, 1000, 2000, 5000]
    for beta in beta_schedule:
        if time.time() - start_time > budget_s * 0.98:
            break
        try:
            refined = slsqp_optimize(best_h, beta, max_iter=500)
        except Exception:
            continue
        refined_c = compute_c5(refined)
        if refined_c < best_c:
            best_h, best_c = refined, refined_c

    # ----- optional up‑sampling + fine hill‑climb if time permits -------
    remaining = budget_s - (time.time() - start_time)
    if remaining > budget_s * 0.12 and n < 720:
        target_n = min(720, n * 2)
        old_x = (np.arange(n) + 0.5) * (2.0 / n)
        new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
        h_up = np.interp(new_x, old_x, best_h)
        h_up = enforce_sum(h_up, target_n / 2.0, rng)

        # a moderate hill‑climb on the finer grid
        hc_h, hc_c = hill_climb_multi(
            h_up, rng, max_iters=3000, init_step=0.3, n_trials=6
        )
        if hc_c < best_c:
            best_h, best_c, n = hc_h, hc_c, target_n

        # final SLSQP refinement on the fine grid
        for beta in beta_schedule:
            if time.time() - start_time > budget_s:
                break
            try:
                refined = slsqp_optimize(best_h, beta, max_iter=500)
            except Exception:
                continue
            refined_c = compute_c5(refined)
            if refined_c < best_c:
                best_h, best_c = refined, refined_c

    # ----- final clean‑up -------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, np.random.default_rng())
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
