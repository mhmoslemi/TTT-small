def softmax_obj(x):
    # compute correl terms (C(k) before scaling)
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    # use log-sum-exp stable
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds,

    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation to max_k∑h[i](1‑h[i+k]) using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """Refine h0 with SLSQP on the smooth max‑approximation."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x


# ---------- main driver ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- initialise -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        enforce_sum(h0, n / 2.0, rng)
    else:
        n = 180
        h0 = random_symmetric_feasible(n, rng)

    # ----- first greedy pass -----
    best_h = greedy_reduce_max_overlap(h0.copy())
    best_c = compute_c5(best_h)

    # ----- random‑restart greedy search (≈85% of time) -----
    restart_deadline = start + budget_s * 0.85
    while time.time() < restart_deadline:
        if rng.random() < 0.5:
            # perturb current best slightly
            h = best_h.copy()
            for _ in range(5):
                i, j = rng.integers(0, n, size=2)
                if i == j:
                    continue
                eps = min(h[i], 1.0 - h[j]) * rng.random()
                if eps <= 0:
                    continue
                h[i] -= eps
                h[j] += eps
        else:
            # fresh random symmetric start
            h = random_symmetric_feasible(n, rng)

        h = greedy_reduce_max_overlap(h)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h, c

    # ----- final polishing with SLSQP (use remaining budget) -----
    for beta in (200, 800, 3200, 12800, 51200):
        if time.time() > start + budget_s:
            break
        try:
            refined = slsqp_optimize(best_h, beta, max_iter=300)
            refined = greedy_reduce_max_overlap(refined)
            c = compute_c5(refined)
            if c < best_c:
                best_h, best_c = refined, c
        except Exception:
            continue

    # ----- final cleanup -----
    best_h = np.clip(best_h, 0.0, 1.0)
    enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
