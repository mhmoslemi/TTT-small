import time, numpy as np, math, sys
from scipy.optimize import minimize

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def perturb_move(h, rng, scale):
    n = h.shape[0]
    i = rng.integers(0, n)
    j = rng.integers(0, n-1)
    if i == j:
        return h
    # maximum epsilon that do not exceed bounds
    max_eps = min(h[i], 1 - h[j]) if h[i] > 0 else 0
    max_eps_neg = min(h[j], 1 - h[i])  # complementary
    # We'll select direction uniformly.
    if rng.random() < 0.5:
        # transfer from i to j
        eps_max = max_eps
        if eps_max <= 0:
            return h
        eps = (rng.random() * 2 - 1) * scale * eps_max
        eps = max(-eps_max, min(eps, eps_max))
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        return new_h
    else:
        # transfer from j to i
        eps_max = max_eps_neg
        if eps_max <= 0:
            return h
        eps = (rng.random() * 2 - 1) * scale * eps_max
        eps = max(-eps_max, min(eps, eps_max))
        new_h = h.copy()

def simulated_annealing(h0, rng, deadline):
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    # Temperature initial based on current c
    T0 = max(0.02, 0.2 * cur_c) # some scaling
    it = 0
    while time.time() < deadline:
        # adaptive scale based on temperature
        scale = max(0.01, T0 * (0.9995 ** it))
        cand_h = perturb_move(h, rng, scale)
        cand_c = compute_c5(cand_h)
        if cand_c < cur_c:
            h, cur_c = cand_h, cand_c
            if cand_c < best_c:
                best_h, best_c = cand_h.copy(), cand_c
        else:
            # temp schedule
            T = T0 * (0.9995 ** it) + 1e-12
            if rng.random() < math.exp(-(cand_c - cur_c) / T):
                h, cur_c = cand_h, cand_c
        it += 1
        # occasional gradient refinement
        if it % 5000 == 0:
            h = gradient_refine(h, rng, step=0.02, max_iter=5)
            cur_c = compute_c5(h)
            if cur_c < best_c:
                best_h, best_c = h.copy(), cur_c
    return best_h, best_c
