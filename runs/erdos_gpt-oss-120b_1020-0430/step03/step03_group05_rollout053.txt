def targeted_perturb(h, rng, scale=0.1):
    n = h.size
    # compute correlation
    corr = np.correlate(h, 1-h, mode='full')
    i_center = n-1
    max_val = np.max(corr)
    max_idxs = np.where(corr == max_val)[0]
    # choose one shift at random from max_idxs
    shift_idx = rng.choice(max_idxs)
    k = shift_idx - i_center  # shift value (i->i+k)
    # compute contribution array length n
    # For i such that 0 <= i < n and 0 <= i+k < n
    contrib = np.zeros(n)
    if k >= 0:
        i_range = np.arange(0, n - k)
        contrib[i_range] = h[i_range] * (1 - h[i_range + k])
    else:
        i_range = np.arange(-k, n)
        contrib[i_range] = h[i_range] * (1 - h[i_range + k])  # since k negative
    # select i where contrib large
    # probability weighted by contrib
    # if sum zero, fallback to random
    if contrib.sum() > 0:
        probs = contrib / contrib.sum()
        i = rng.choice(n, p=probs)
    else:
        i = rng.integers(0, n)
    # select j where contribution small (maybe opposite)
    # we want to move mass from i to j
    # choose j where (h[j] * (1 - h[j+k])) is minimal or (h[j] high to increase complement)
    # But we can just choose random
    j = rng.integers(0, n-1)
    if j >= i:
        j += 1
    # ε limited by h[i] and 1-h[j]
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    eps = scale * max_eps * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h
```

def hill_climb_targeted(h_start, rng, max_iters=2000, scale=0.1):
    h = h_start.copy()
    best_c = compute_c5(h)
    best_h = h.copy()
    for it in range(max_iters):
        if rng.random() < 0.7:
            new_h = targeted_perturb(h, rng, scale=scale)
        else:
            new_h = perturb_h(h, rng, scale=scale)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        # optional early step size reduction
        if (it+1) % 500 == 0:
            scale *= 0.7
    return best_h, best_c

def de_optimize(n, rng, max_time):
    target = n / 2.0
    popsize = max(30, int(0.2 * n))  # maybe 30-60
    # Initialize population
    pop = np.array([random_feasible_vector(n, rng) for _ in range(popsize)])
    scores = np.array([compute_c5(ind) for ind in pop])
    best_idx = np.argmin(scores)
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]
    F = 0.8
    CR = 0.9
    start = time.time()
    # while time left
    while time.time() - start < max_time:
        for i in range(popsize):
            # choose distinct indices r1,r2,r3
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, 3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            # clip + project sum
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = project_to_feasible(mutant, target)
            # crossover
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(0, n)] = True
            trial = np.where(cross, mutant, pop[i])
            trial = project_to_feasible(trial, target)
            trial_score = compute_c5(trial)
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_h = trial.copy()
        # optional check for convergence break if no improvement for many generations
        # but keep simple
    return best_h, best_score

def slsqp_softmax_opt(h0, beta, rng, max_iter=300):
    n = h0.size
    target = n / 2.0
    # Set global in closure? Use nested function or define sum_eq_constraint using partial
    def sum_eq(x):
        return np.sum(x) - target
    def softmax_obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        bc = beta * corr
        # stable log sum exp
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # correct sum if needed
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # iterative addition
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x
```

import time
import numpy as np
from scipy.optimize import minimize

# -------------------------------------------------
# Utility functions
# -------------------------------------------------
def compute_c5(h):
    """Exact discrete C5 for step function h on [0,2] with dx=2/len(h)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_to_feasible(v, target, iters=60):
    """Project v onto {0<=x<=1, sum(x)=target} via bisection on a shift."""
    lo, hi = -2.0, 2.0
    for _ in range(iters):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def adjust_sum(v, target, rng, max_iter=2000):
    """Fast correction of the sum while staying in [0,1]."""
    diff = target - v.sum()
    n = v.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - v[i])
        else:
            add = max(diff, -v[i])
        if add != 0.0:
            v[i] += add
            diff = target - v.sum()
    return v


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_to_feasible(v, n / 2.0)


def perturb_h(h, rng, scale=0.2):
    """Move a random amount from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    while j == i:
        j = rng.integers(0, n)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = max_eps * scale * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def targeted_perturb(h, rng, scale=0.2):
    """Perturb focusing on the shift that currently attains the maximum."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    i_center = n - 1
    max_val = np.max(corr)
    max_idxs = np.where(corr == max_val)[0]
    shift_idx = rng.choice(max_idxs)
    k = shift_idx - i_center  # shift amount

    # contribution of each i to this shift
    contrib = np.zeros(n)
    if k >= 0:
        idx = np.arange(0, n - k)
        contrib[idx] = h[idx] * (1.0 - h[idx + k])
    else:
        idx = np.arange(-k, n)
        contrib[idx] = h[idx] * (1.0 - h[idx + k])

    if contrib.sum() > 0.0:
        probs = contrib / contrib.sum()
        i = rng.choice(n, p=probs)
    else:
        i = rng.integers(0, n)

    j = rng.integers(0, n)
    while j == i:
        j = rng.integers(0, n)

    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = max_eps * scale * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_targeted(h_start, rng, max_iters=3000, scale=0.2):
    """Hill climbing that mixes generic and targeted moves."""
    h = h_start.copy()
    best_c = compute_c5(h)
    best_h = h.copy()
    for it in range(max_iters):
        # 70% targeted moves, 30% generic
        if rng.random() < 0.7:
            nh = targeted_perturb(h, rng, scale)
        else:
            nh = perturb_h(h, rng, scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c = c
            best_h = nh.copy()
            h = nh
        # slowly reduce step size
        if (it + 1) % 600 == 0:
            scale *= 0.7
    return best_h, best_c

# Differential Evolution (simple, with projection)
# -------------------------------------------------
def de_optimize(n, rng, max_time):
    target = n / 2.0
    popsize = max(30, int(0.2 * n))
    pop = np.array([random_feasible_vector(n, rng) for _ in range(popsize)])
    scores = np.array([compute_c5(ind) for ind in pop])
    best_idx = scores.argmin()
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]

    F = 0.8
    CR = 0.9
    start = time.time()
    while time.time() - start < max_time:
        for i in range(popsize):
            # mutation indices
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, 3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = adjust_sum(mutant, target, rng)
            # crossover
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(0, n)] = True
            trial = np.where(cross, mutant, pop[i])
            trial = adjust_sum(trial, target, rng)
            trial_score = compute_c5(trial)
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_h = trial.copy()
    return best_h, best_score

# SLSQP with smooth max (log‑sum‑exp) surrogate
# -------------------------------------------------
def slsqp_softmax_opt(h0, beta, rng, max_iter=400):
    n = h0.size
    target = n / 2.0

    def sum_eq(x):
        return np.sum(x) - target

    def softmax_obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # final exact sum correction
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = adjust_sum(x, target, rng)
    return x
