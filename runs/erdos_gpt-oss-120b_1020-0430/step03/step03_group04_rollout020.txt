def subgradient_optimize(h0, target, rng, max_iters=1000, lr=0.05, decay=0.0, beta1=0.9, beta2=0.999, eps=1e-8):
    n = h0.shape[0]
    h = project(h0.copy(), target)
    m = np.zeros(n)
    v = np.zeros(n)
    best_h = h.copy()
    best_c = compute_c5(h)
    for t in range(1, max_iters+1):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr)
        k = idx - (n - 1)
        # compute gradient
        g = np.ones(n)
        if k >= 0:
            g[:n - k] -= h[k:]
            g[k:] -= h[:n - k]
        else:
            kk = -k
            # shift negative: h_i h_{i - kk}
            # Equivalent to swapping roles: C_{-k} = sum_i h_{i+kk} (1 - h_i) for i where i+kk < n
            # Derivative should be: gradient = 1 - h_{i-kk} - h_{i+kk}? Actually symmetric.
            # We can just use absolute k as same formula because correlation for -k yields same terms reversed
            # Use same as positive k: treat shift = kk
            g[:n - kk] -= h[kk:]
            g[kk:] -= h[:n - kk]
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)

        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def project(v, target, max_iter=200):
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        excess = v.sum() - target
        if abs(excess) <= 1e-12:
            break
        # mask free
        free = (v > 0) & (v < 1)
        cnt = free.sum()
        if cnt == 0:
            # cannot adjust free, break
            break
        delta = excess / cnt
        v[free] -= delta
        # clip again
        v = np.clip(v, 0.0, 1.0)
    # final small arbitrary fix for residual
    diff = target - v.sum()
    if abs(diff) > 1e-12:
        # distribute to any index with capacity
        for i in range(len(v)):
            if diff > 0 and v[i] < 1.0:
                add = min(diff, 1.0 - v[i])
                v[i] += add
                diff -= add
            elif diff < 0 and v[i] > 0.0:
                sub = min(-diff, v[i])
                v[i] -= sub
                diff += sub
            if abs(diff) <= 1e-12:
                break
    return v

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    while i == j:
        j = rng.integers(0, n)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps == 0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def subgradient_optimize(h0, target, rng, max_iters=800, lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    n = h0.size
    h = project(h0.copy(), target)
    m = np.zeros(n)
    v = np.zeros(n)
    best_h = h.copy()
    best_c = compute_c5(h)
    for t in range(1, max_iters + 1):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr)
        k = idx - (n - 1)
        p = abs(k)
        g = np.ones(n)
        if p > 0:
            g[:n - p] -= h[p:]
            g[p:] -= h[:n - p]
        # Adam
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)

        step_vec = lr * m_hat / (np.sqrt(v_hat) + eps)
        h = h - step_vec
        h = project(h, target)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

import time

# ---------- basic utilities ----------
def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def project(v, target, max_iter=200):
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        excess = v.sum() - target
        if abs(excess) <= 1e-12:
            break
        free = (v > 0) & (v < 1)
        cnt = free.sum()
        if cnt == 0:
            break
        delta = excess / cnt
        v[free] -= delta
        v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    if abs(diff) > 1e-12:
        for i in range(v.size):
            if diff > 0 and v[i] < 1.0:
                add = min(diff, 1.0 - v[i])
                v[i] += add
                diff -= add
            elif diff < 0 and v[i] > 0.0:
                sub = min(-diff, v[i])
                v[i] -= sub
                diff += sub
            if abs(diff) <= 1e-12:
                break
    return v


def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    while i == j:
        j = rng.integers(0, n)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    target = h.size / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-5)
    best_h = project(best_h, target)
    return best_h, best_c

# ---------- subgradient (Adam) ----------
def subgradient_optimize(h0, target, rng, max_iters=800, lr=0.12,
                         beta1=0.9, beta2=0.999, eps=1e-8):
    n = h0.size
    h = project(h0.copy(), target)
    m = np.zeros(n)
    v = np.zeros(n)
    best_h = h.copy()
    best_c = compute_c5(h)
    for t in range(1, max_iters + 1):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr)
        k = idx - (n - 1)
        p = abs(k)
        if p == 0:
            g = 1.0 - 2.0 * h
        else:
            g = np.ones(n)
            g[:n - p] -= h[p:]
            g[p:] -= h[:n - p]
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step_vec = lr * m_hat / (np.sqrt(v_hat) + eps)
        h = h - step_vec
        h = project(h, target)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c
