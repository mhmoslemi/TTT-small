def hill_climb(h, rng, max_iters, step_scale_start=0.5, step_decay=0.9995, min_step=1e-4):
    step_scale = step_scale_start
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_h, best_c = nh.copy(), c
            h = nh
        else:
            # occasionally accept if random and maybe better diff (simulated annealing)
            if rng.random() < 0.01: # 1% chance to move
                h = nh
        step_scale = max(step_scale * step_decay, min_step)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # starting from initial if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0, c0, h0.size

    # define list of ns to explore
    ns = [192, 288, 384] # maybe also 240, 480? Keep within budget.
    # ensure best_n first
    if best_n is not None:
        ns = [best_n] + [n for n in ns if n != best_n]

    for n in ns:
        # check time
        if time.time() - start > 0.95*budget_s:
            break
        # initialize h
        if best_n == n and best_h is not None:
            h = best_h.copy()
        else:
            # interpolate from best_h if exists else random
            if best_h is not None:
                x_old = np.linspace(0, 2, best_n, endpoint=False)
                x_new = np.linspace(0, 2, n, endpoint=False)
                h = np.interp(x_new, x_old, best_h)
                h = np.clip(h, 0.0, 1.0)
                h = enforce_sum(h, rng)
            else:
                h = random_feasible_vector(n, rng)

        # Phase: hill climb
        # allocate iterations based on time left and n
        # let iters = int(min(3000, 20000/n))? Not sure.
        # We'll set max_iters = 3000 for n <= 192 else 2000.
        max_iters = 3000 if n <= 192 else 2000
        h, c = hill_climb(h, rng, max_iters=max_iters, step_scale_start=0.5)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

        # Phase: SLSQP schedule
        # beta sequence gradually increase.
        betas = [5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0]
        for beta in betas:
            if time.time() - start > 0.97*budget_s:
                break
            maxit = 800 if beta >= 200 else 600
            h = slsqp_optimize(h, beta, rng, max_iter=maxit)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

        # Phase: fine gradient refinement
        if time.time() - start < 0.98*budget_s:
            h = gradient_refine(h, rng, step=0.02, max_iter=12)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

    # local near end: random perturbation with guided gradient for best solution
    while time.time() - start < budget_s * 0.999 and best_h is not None:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # final feasibility enforcement and return
    if best_h is None:
        best_n = 192
        best_h = np.full(best_n, 0.5)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)

def simulated_annealing(h, rng, max_steps, init_temp):
    temp = init_temp
    for t in range(max_steps):
        # choose two indices i, j (i != j) randomly
        i, j = rng.choice(len(h), size=2, replace=False)
        # propose moving eps from i to j. choose eps random between [-delta, delta] but respecting bounds.
        max_eps = min(h[i], 1 - h[j]) if h[i] > 0 else 0
        # Random epsilon maybe scaled by temperature.
        eps = (rng.random() - 0.5) * 2 * temp * max_eps
        # Actually we want positive move to j maybe, but we can randomly flip sign.
        # Simpler: choose move from i to j (positive epsilon up to max_eps)
        eps = rng.random() * max_eps
        nh = h.copy()
        nh[i] -= eps
        nh[j] += eps
        # enforce sum automatically satisfied.
        nd = compute_c5(nh)
        if nd < compute_c5(h) or rng.random() < np.exp(-(nd - compute_c5(h))/temp):
            h = nh
        temp *= 0.9995
    return h

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utility functions ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple random water‑filling
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random point in the feasible polytope."""
    h = rng.random(n)
    h *= (n / 2.0) / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, rng)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst (max) overlap O_k = Σ h_i(1‑h_{i+k})."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray, rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def hill_climb(h: np.ndarray, rng: np.random.Generator,
               max_iters: int = 2500,
               step_start: float = 0.5,
               step_decay: float = 0.9995,
               min_step: float = 1e-4) -> tuple[np.ndarray, float]:
    """Gradient‑directed hill climb with occasional random moves."""
    n = h.size
    step = step_start
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c

    for _ in range(max_iters):
        grad = worst_gradient(cur_h)
        nh = guided_perturb(cur_h, grad, rng, scale=step)

        # occasional fully random transfer to escape plateaus
        if rng.random() < 0.05:
            i, j = rng.choice(n, size=2, replace=False)
            max_eps = min(cur_h[i], 1.0 - cur_h[j])
            if max_eps > 0:
                eps = rng.random() * max_eps * step
                rand_nh = cur_h.copy()
                rand_nh[i] -= eps
                rand_nh[j] += eps
                if compute_c5(rand_nh) < compute_c5(nh):
                    nh = rand_nh

        c = compute_c5(nh)
        if c < cur_c:
            cur_h, cur_c = nh, c
            if c < best_c:
                best_h, best_c = nh.copy(), c
        step = max(step * step_decay, min_step)

    return best_h, best_c


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ h_i(1‑h_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp in a numerically stable way
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 400) -> np.ndarray:
    """Soft‑max SLSQP refinement for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction (quick redistribution)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.02, max_iter: int = 12) -> np.ndarray:
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] = 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # ---- initialise from parent if present ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0, c0, h0.size

    # ---- focus on n = 192 (the size of the parent) ----
    target_n = 192
    if best_n != target_n:
        best_n = target_n
        best_h = random_feasible_vector(target_n, rng)
        best_c = compute_c5(best_h)

    # ---- multiple random restarts with hill‑climb ----
    restart = 0
    MAX_RESTARTS = 30
    while time.time() - start < 0.80 * budget_s and restart < MAX_RESTARTS:
        h = random_feasible_vector(target_n, rng)
        h, c = hill_climb(h, rng, max_iters=2000, step_start=0.6)
        # quick soft‑max refinement (few β values, low iterations)
        for beta in (5.0, 20.0, 80.0):
            if time.time() - start > 0.84 * budget_s:
                break
            h = slsqp_optimize(h, beta, rng, max_iter=300)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h.copy(), c
        restart += 1

    # ---- deep refinement of the current best ----
    if time.time() - start < 0.90 * budget_s:
        h = best_h.copy()
        # extensive soft‑max schedule
        for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0, 10000.0):
            if time.time() - start > 0.94 * budget_s:
                break
            maxit = 800 if beta >= 200 else 600
            h = slsqp_optimize(h, beta, rng, max_iter=maxit)
        # fine gradient refinement
        if time.time() - start < 0.96 * budget_s:
            h = gradient_refine(h, rng, step=0.02, max_iter=16)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h.copy(), c

    # ---- final local tweaks with tiny guided moves ----
    while time.time() - start < 0.999 * budget_s:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.015)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # ---- enforce feasibility and return ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
