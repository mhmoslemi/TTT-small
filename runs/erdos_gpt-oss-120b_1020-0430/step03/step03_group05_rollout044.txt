import time
import math
import numpy as np

# ---------------------- helper utilities ---------------------- #

def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_to_feasible(v, target):
    """Projection of v onto {0≤h≤1, Σh = target} via bisection on a shift."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Random feasible vector (0≤h≤1, Σh = n/2)."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_to_feasible(v, n / 2.0)


def perturb_transfer(h, rng, scale=0.2):
    """Transfer ε from i to j (preserves sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


# ---------------------- optimization phases ---------------------- #

def simulated_annealing(h0, rng, time_limit):
    """Simple SA that preserves feasibility via transfer moves."""
    n = h0.size
    target = n / 2.0
    cur = h0.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c

    start = time.time()
    # SA schedule
    T0 = 1e-3
    T_min = 1e-8
    decay = 0.99995

    T = T0
    it = 0
    while time.time() - start < time_limit:
        it += 1
        # proposal
        cand = perturb_transfer(cur, rng, scale=0.2)
        cand_c = compute_c5(cand)
        delta = cand_c - cur_c
        if delta < 0 or rng.random() < math.exp(-delta / max(T, T_min)):
            cur, cur_c = cand, cand_c
            if cur_c < best_c:
                best, best_c = cur.copy(), cur_c
        # cool
        T = max(T_min, T * decay)
    return best, best_c


def hill_climb(h_start, rng, max_iters=2000, start_step=0.3):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        cand = perturb_transfer(h, rng, scale=step)
        c = compute_c5(cand)
        if c < best_c:
            best_c, best_h, h = c, cand.copy(), cand
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c

    """
    Optimise a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # --- Choose discretisation size ---

        if cand_c < best_c:

    final_c = compute_c5(best_h)
