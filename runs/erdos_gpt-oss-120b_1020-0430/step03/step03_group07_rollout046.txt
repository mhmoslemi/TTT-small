import time, math, numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def enforce_sum(h, target, rng):
    """Adjust vector h in-place so that sum(h) == target and 0<=h<=1."""
    diff = target - h.sum()
    n = h.size
    # simple random walk corrections
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    """Random feasible vector in [0,1]^n with sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 6):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j: continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0: continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.4
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)

def perturb_h(h, rng, scale=0.2):
    """Transfer ε from i to j, preserving the sum."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j: return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0: return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def correlate_full(h):
    """Full linear correlation of h with (1‑h) using FFT."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    fr = np.fft.rfft((1.0 - h)[::-1], L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv

def compute_c5(h):
    """Discrete C5 for step function h on [0,2] (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)

# ----------------------------------------------------------------------
# Smooth max surrogate used by SLSQP
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = correlate_full(x)
    bc = CURRENT_BETA * corr
    m = bc.max()
    # log‑sum‑exp (stable)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=400):
    """SLSQP on smooth surrogate with given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def hill_climb(h_start, rng, deadline, max_iters=5000, start_step=0.4):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        nh = perturb_h(h, rng, scale=step)
        c = compute_c5(nh)
        if c < best_c:
            best_c = c
            best_h = nh.copy()
            h = nh
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c

def simulated_annealing(start_h, rng, deadline,
                        T0=0.08, Tf=1e-6, max_iters=6000):
    cur_h = start_h.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        nh = perturb_h(cur_h, rng, scale=0.45)
        nc = compute_c5(nh)
        delta = nc - cur_c
        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            cur_h, cur_c = nh, nc
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
    return best_h, best_c

        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 180
        h0 = np.full(n, 0.5)
        h0 = enforce_sum(h0, n / 2.0, rng)

    # ---- generate diverse seeds -----------------------------------------
    candidates = [h0]

    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx

    # sinusoids
    for f in range(1, 5):
        s = 0.5 + 0.5 * np.sin(np.pi * f * xs)
        candidates.append(enforce_sum(np.clip(s, 0, 1), n / 2.0, rng))
        c = 0.5 + 0.5 * np.cos(np.pi * f * xs)
        candidates.append(enforce_sum(np.clip(c, 0, 1), n / 2.0, rng))

    # linear ramps
    candidates.append(enforce_sum(np.clip(xs / 2.0, 0, 1), n / 2.0, rng))
    candidates.append(enforce_sum(np.clip(1.0 - xs / 2.0, 0, 1), n / 2.0, rng))

    # smoothed random binary
    for _ in range(3):
        bin_vec = np.zeros(n)
        bin_vec[: n // 2] = 1.0
        rng.shuffle(bin_vec)
        # small moving‑average smoothing
        ker = np.array([0.2, 0.6, 0.2])
        smooth = np.convolve(bin_vec, ker, mode="same")
        candidates.append(enforce_sum(np.clip(smooth, 0, 1), n / 2.0, rng))

    # random feasible vectors
    for _ in range(4):
        candidates.append(random_feasible_vector(n, rng))

    # ---- best of the pool ------------------------------------------------
    best_h = None
    best_c = float("inf")
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # --------------------------------------------------------------------
    beta_series = [20, 80, 200, 800, 3200]
    max_n = 720  # upper resolution we allow

    # ---- multiscale optimisation loop -----------------------------------
    while time.time() < deadline:
        # decide whether to upsample
        if best_h.size * 2 <= max_n and time.time() + 8.0 < deadline:
            # upsample by simple repeat
            new_n = best_h.size * 2
            up = np.repeat(best_h, 2)
            up = enforce_sum(up, new_n / 2.0, rng)
            best_h, best_c = up, compute_c5(up)

        # allocate a short horizon for this resolution
        sub_dead = min(deadline, time.time() + 30.0)

        # pick a start point
        if rng.random() < 0.7:
            start = perturb_h(best_h, rng, scale=0.35)
        else:
            start = random_feasible_vector(best_h.size, rng)

        # simulated annealing
        cand_h, cand_c = simulated_annealing(start, rng, sub_dead)
        if cand_c < best_c:
            best_h, best_c = cand_h.copy(), cand_c

        # hill climbing refinement
        hc_h, hc_c = hill_climb(best_h, rng, sub_dead, max_iters=4000, start_step=0.45)
        if hc_c < best_c:
            best_h, best_c = hc_h.copy(), hc_c

        # SLSQP ladder
        for beta in beta_series:
            if time.time() >= sub_dead:
                break
            try:
                opt_h = slsqp_optimize(best_h, beta, max_iter=500)
                opt_c = compute_c5(opt_h)
                if opt_c < best_c:
                    best_h, best_c = opt_h.copy(), opt_c
            except Exception:
                continue

        # occasional random restart if progress stalls
        if rng.random() < 0.12:
            alt = random_feasible_vector(best_h.size, rng)
            alt_c = compute_c5(alt)
            if alt_c < best_c:
                best_h, best_c = alt, alt_c

    # ---- final clean‑up --------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
