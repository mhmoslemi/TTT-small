import time
import numpy as np

# ---------- utility functions ----------
def compute_c5(h):
    """Discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, rng):
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
        it += 1
    h[:] = np.clip(h, 0.0, 1.0)
    return h


def random_feasible_vector(n, rng):
    """Random feasible h (0<=h<=1, Σh=n/2)."""
    target = n / 2.0
    h = rng.random(n)
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, rng)


def worst_gradient(h):
    """Gradient of the worst shift overlap O_k = Σ h_i (1‑h_{i+k})."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h, grad, rng, scale=0.5):
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h, rng, max_iters=2000, step_scale=0.5, time_limit=None):
    """Greedy hill‑climb driven by the worst‑shift gradient."""
    best_h = h.copy()
    best_c = compute_c5(best_h)
    cur = h.copy()
    n = h.size
    decay_interval = max(1, max_iters // 5)
    for it in range(max_iters):
        if time_limit is not None and time.time() > time_limit:
            break
        grad = worst_gradient(cur)
        nh = guided_perturb(cur, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, cur = c, nh.copy(), nh
        # decay step scale periodically
        if (it + 1) % decay_interval == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h, rng, step=0.02, max_iter=10):
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h


def thue_morse_seq(n):
    """Binary Thue‑Morse sequence of length n (power of two)."""
    i = np.arange(n, dtype=np.uint64)
    bits = np.unpackbits(i.view(np.uint8).reshape(-1, 8), axis=1)
    parity = bits.sum(axis=1) & 1
    return parity.astype(float)


def maybe_smooth(h, rng, prob=0.4):
    """Occasionally smooth h with a small 3‑point kernel."""
    if rng.random() < prob:
        kernel = np.array([0.25, 0.5, 0.25])
        hs = np.convolve(h, kernel, mode="same")
        hs = np.clip(hs, 0.0, 1.0)
        hs = enforce_sum(hs, rng)
        return hs
    return h
