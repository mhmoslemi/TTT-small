def run(seed=42,budget_s=1100,**kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # candidate n list
    n_candidates = [128, 160, 192, 224, 256]
    # We prioritize larger n
    for n in n_candidates:
        if time.time() - start > 0.94 * budget_s:
            break

        # initial best for this n if from parent
        if 'initial_h_values' in globals():
            h0 = np.array(initial_h_values, dtype=float)
            if h0.size != n:
                # interpolate
                x_old = np.linspace(0,2,h0.size,endpoint=False)
                x_new = np.linspace(0,2,n,endpoint=False)
                h0 = np.interp(x_new,x_old,h0)
            h0 = np.clip(h0,0,1)
            h0 = enforce_sum(h0,rng)
        else:
            h0 = random_feasible_vector(n, rng)

        # compute initial best
        c0 = compute_c5(h0)
        if c0 < best_c:
            best_h, best_c, best_n = h0.copy(), c0, n

        # GA parameters
        pop_size = 30
        elite_size = 2
        tournament_size = 3
        max_gen = 1000  # can be limited by time
        mut_rate = 0.2
        mut_range = 0.15  # this will decay over generations

        # initialize population
        population = [h0]
        while len(population) < pop_size:
            population.append(random_feasible_vector(n, rng))

        # evaluate scores
        scores = np.array([compute_c5(ind) for ind in population])
        # update best
        idx_best = np.argmin(scores)
        if scores[idx_best] < best_c:
            best_h, best_c = population[idx_best].copy(), scores[idx_best]
            best_n = n

        gen = 0
        while gen < max_gen and time.time() - start < 0.96 * budget_s:
            # sort population by score ascending
            sorted_idx = np.argsort(scores)
            new_population = [population[i] for i in sorted_idx[:elite_size]]
            # gradually decay mutation range
            cur_mut_range = mut_range * (0.99 ** gen)

            while len(new_population) < pop_size:
                # tournament selection
                def select():
                    candidates = rng.integers(0, pop_size, size=tournament_size)
                    cand_scores = scores[candidates]
                    return population[candidates[np.argmin(cand_scores)]]
                p1 = select()
                p2 = select()
                alpha = rng.uniform(-0.2,1.2)  # some extrapolation
                child = (1-alpha)*p1 + alpha*p2
                # mutation
                mask = rng.random(n) < mut_rate
                delta = rng.uniform(-cur_mut_range, cur_mut_range, size=n)
                child[mask] += delta[mask]
                # clip
                np.clip(child,0,1, out=child)
                # enforce sum
                child = enforce_sum(child, rng)
                new_population.append(child)

            # evaluate new population
            population = new_population
            scores = np.array([compute_c5(ind) for ind in population])
            # update best
            idx_best = np.argmin(scores)
            if scores[idx_best] < best_c:
                best_h, best_c = population[idx_best].copy(), scores[idx_best]
                best_n = n

            gen += 1

        # after GA, perform local refinement on best_h (the best for this n)
        if time.time() - start < 0.98 * budget_s:
            refined_h, refined_c = guided_hill_climb(best_h.copy(), rng, max_iters=3000, step_scale=0.5)
            if refined_c < best_c:
                best_h, best_c = refined_h, refined_c
                best_n = n

            # additional fine gradient
            if time.time() - start < 0.99 * budget_s:
                best_h = gradient_refine(best_h, rng, step=0.02, max_iter=8)
                best_c = compute_c5(best_h)

    # final random perturbation attempts until budget
    while time.time() - start < budget_s * 0.999 and best_h is not None:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh, nc

    # final clip and enforce sum
    best_h = np.clip(best_h,0,1)
    best_h = enforce_sum(best_h,rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]          # subtract h_{i+k}
        grad[k:]       -= h[: n - k]    # subtract h_{i‑k}
    return grad

global CURRENT_TARGET, CURRENT_BETA
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0,beta,rng,max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n/2.0
    CURRENT_BETA = beta
    bounds = [(0.0,1.0)]*n
    cons = {"type":"eq","fun":sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter':max_iter,'ftol':1e-12,'disp':False})
    x = np.clip(res.x,0,1)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0,n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic helpers ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function h on [0,2]."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h to satisfy Σh = len(h)/2 while staying within [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 5000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
        attempts += 1
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector in [0,1]^n with sum n/2."""
    h = rng.random(n)
    h *= (n / 2.0) / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, rng)


# ---------- gradient / local search ----------
def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray, rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h_start: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2500, step_scale: float = 0.5) -> tuple[np.ndarray, float]:
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.02, max_iter: int = 8) -> np.ndarray:
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h


# ---------- SLSQP (soft‑max) ----------
CURRENT_TARGET = 0.0
CURRENT_BETA = 0.0


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 400) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ---------- GA ----------
def tournament_select(pop, scores, rng, tsize=3):
    idx = rng.integers(0, len(pop), size=tsize)
    best = idx[np.argmin(scores[idx])]
    return pop[best]


def blend_crossover(p1, p2, rng, alpha_range=(-0.2, 1.2)):
    a = rng.uniform(*alpha_range)
    child = (1.0 - a) * p1 + a * p2
    np.clip(child, 0.0, 1.0, out=child)
    return child


def mutate(child, rng, mut_rate, mut_range):
    mask = rng.random(child.size) < mut_rate
    delta = rng.uniform(-mut_range, mut_range, size=child.size)
    child[mask] += delta[mask]
    np.clip(child, 0.0, 1.0, out=child)
    return child


# ---------- Main entry ----------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # candidate discretisations (larger n first)
    n_candidates = [256, 224, 192, 160, 128]
    for n in n_candidates:
        if time.time() - start > 0.94 * budget_s:
            break

        # ----- initial individual -----
        if "initial_h_values" in globals():
            h0 = np.array(initial_h_values, dtype=float)
            if h0.size != n:
                x_old = np.linspace(0.0, 2.0, h0.size, endpoint=False)
                x_new = np.linspace(0.0, 2.0, n, endpoint=False)
                h0 = np.interp(x_new, x_old, h0)
            h0 = np.clip(h0, 0.0, 1.0)
            h0 = enforce_sum(h0, rng)
        else:
            h0 = random_feasible_vector(n, rng)

        # ----- population -----
        pop_size = 30
        elite_size = 2
        population = [h0]
        while len(population) < pop_size:
            population.append(random_feasible_vector(n, rng))

        scores = np.array([compute_c5(ind) for ind in population])
        idx_best = int(np.argmin(scores))
        if scores[idx_best] < best_c:
            best_h, best_c, best_n = population[idx_best].copy(), scores[idx_best], n

        # ----- GA loop -----
        max_gen = 500
        mut_rate = 0.2
        mut_range0 = 0.15
        tournament_sz = 3
        gen = 0
        while gen < max_gen and time.time() - start < 0.96 * budget_s:
            # sort for elitism
            order = np.argsort(scores)
            new_pop = [population[i] for i in order[:elite_size]]
            cur_mut_range = mut_range0 * (0.99 ** gen)

            while len(new_pop) < pop_size:
                p1 = tournament_select(population, scores, rng, tournament_sz)
                p2 = tournament_select(population, scores, rng, tournament_sz)
                child = blend_crossover(p1, p2, rng)
                child = mutate(child, rng, mut_rate, cur_mut_range)
                child = enforce_sum(child, rng)
                new_pop.append(child)

            population = new_pop
            scores = np.array([compute_c5(ind) for ind in population])
            idx_best = int(np.argmin(scores))
            if scores[idx_best] < best_c:
                best_h, best_c, best_n = population[idx_best].copy(), scores[idx_best], n

            gen += 1

        # ----- local refinement on best of this n -----
        if time.time() - start < 0.98 * budget_s:
            refined_h, refined_c = guided_hill_climb(best_h.copy(), rng, max_iters=3000, step_scale=0.5)
            if refined_c < best_c:
                best_h, best_c, best_n = refined_h, refined_c, n

            # soft‑max SLSQP passes
            for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0):
                if time.time() - start > 0.99 * budget_s:
                    break
                maxit = 400 if beta < 200 else 200
                best_h = slsqp_optimize(best_h, beta, rng, max_iter=maxit)
                best_c = compute_c5(best_h)

            # final gradient polish
            if time.time() - start < 0.999 * budget_s:
                best_h = gradient_refine(best_h, rng, step=0.02, max_iter=8)
                best_c = compute_c5(best_h)

    # ----- remaining time: tiny random guided moves -----
    while time.time() - start < budget_s * 0.999 and best_h is not None:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh, nc

    # final sanity
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
