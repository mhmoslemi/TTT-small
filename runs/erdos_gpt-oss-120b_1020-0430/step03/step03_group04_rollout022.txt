def project_onto_box_sum(v, target, lo=-1e5, hi=1e5, tol=1e-9):
    # find lambda such that sum(clamp(v - lambda, 0, 1)) = target
    # Work with double
    low = lo
    high = hi
    # adjust lo, hi to safe bounds
    # lower bound: lambda such that all values are at 1 -> sum=n >= target. we can set low = -max(v) - 1
    low = -np.max(v) - 1.0
    # upper bound: lambda such that all values are at 0 -> sum=0 <= target.
    high = np.max(v) + 1.0
    for _ in range(60):
        lam = (low + high) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            low = lam
        else:
            high = lam
    x = np.clip(v - (low + high) / 2.0, 0.0, 1.0)
    # due to rounding ensure sum exactly target by distributing leftover small diff across non-bound entries
    diff = target - x.sum()
    if abs(diff) > 1e-8:
        # adjust by adding/subtracting small eps equally to entries not at bounds
        free = (x > 0.0 + 1e-12) & (x < 1.0 - 1e-12)
        if np.any(free):
            # distribute diff equally
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def perturb_h(h, rng, scale=0.1):
   n = len(h)
   i = rng.integers(n)
   j = rng.integers(n)
   if i==j: return h
   max_eps = min(h[i], 1.0 - h[j])  # we can move at most this from i to j
   if max_eps <= 0: return h
   eps = (rng.random()*2 - 1) * max_eps * scale
   # ensure within [-max_eps, max_eps]
   eps = max(-max_eps, min(eps, max_eps))
   new_h = h.copy()
   new_h[i] -= eps
   new_h[j] += eps
   return new_h

    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def compute_c5(h):
    """Exact discrete C5 for step function h on [0,2] with uniform spacing."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1] with sum exactly n/2."""
    target = n / 2.0
    h = rng.random(n)
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def perturb_h(h, rng, scale=0.2):
    """Move a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() * 2.0 - 1.0) * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=1500, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements; simple step decay)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


def project_onto_box_sum(v, target, tol=1e-9):
    """
    Euclidean projection of v onto the set {x | 0<=x_i<=1, sum x_i = target}.
    Uses binary search on the Lagrange multiplier.
    """
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # tiny correction in case of rounding errors
    diff = target - x.sum()
    if abs(diff) > tol:
        free = (x > 0.0 + 1e-12) & (x < 1.0 - 1e-12)
        if np.any(free):
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
    return x


def subgradient_descent(
    h_init,
    target,
    max_iters,
    start_step,
    rng,
    start_time,
    deadline,
):
    """
    Sub‑gradient descent on the worst shift.
    Stops early if the shared deadline is passed.
    """
    h = h_init.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    n = h.size
    decay = 0.999

    for it in range(max_iters):
        if time.time() > deadline:
            break

        # worst shift
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)

        # sub‑gradient for that shift
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]      # + (1 - h[i+k])
            g[k:] -= h[:upper]            # - h[i]

        step_sz = start_step * (decay ** it)
        h_new = h - step_sz * g
        h_new = project_onto_box_sum(h_new, target)
        c_new = compute_c5(h_new)

        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()

        # simple acceptance rule (allow occasional uphill moves)
        if c_new < cur_c:
            h, cur_c = h_new, c_new
        else:
            T = step_sz + 1e-12
            if rng.random() < np.exp(-(c_new - cur_c) / T):
                h, cur_c = h_new, c_new

    return best_h, best_c
