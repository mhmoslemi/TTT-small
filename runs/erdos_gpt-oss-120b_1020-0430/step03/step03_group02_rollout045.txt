def project_capsimplex(v, target, max_iter=50, eps=1e-12):
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(max_iter):
        mid = (lo+hi)/2
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if abs(s - target) <= eps:
            return w
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2, 0.0, 1.0)

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global variables used by the soft‑max SLSQP optimizer
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k = Σ h_i (1‑h_{i+k})."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def project_capsimplex(v: np.ndarray, target: float, max_iter: int = 50, eps: float = 1e-12) -> np.ndarray:
    """
    Euclidean projection of v onto the capped simplex
    { w | 0 ≤ w_i ≤ 1, Σ w_i = target }.
    """
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if abs(s - target) <= eps:
            return w
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random point in the capped simplex."""
    v = rng.random(n)
    return project_capsimplex(v, n / 2.0)

# Soft‑max objective for SLSQP
def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int = 500) -> np.ndarray:
    """Refine using SLSQP on a soft‑max approximation of the max‑overlap."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum (projection is cheap and safe)
    if abs(x.sum() - CURRENT_TARGET) > 1e-12:
        x = project_capsimplex(x, CURRENT_TARGET)
    return x


# ----------------------------------------------------------------------
def guided_hill_climb(h0: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.5) -> tuple:
    """Hill‑climb that moves mass from a positive‑gradient index to a negative‑gradient one."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0.0:
                eps = rng.random() * max_eps * step_scale
                h[i] -= eps
                h[j] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-5)
    return best_h, best_c


def pgd_optimize(h0: np.ndarray, target: float,
                 max_iter: int = 2000, step0: float = 0.5,
                 decay: float = 0.99, eval_every: int = 200) -> tuple:
    """Projected gradient descent on the worst‑shift overlap."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step0
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - step * grad
