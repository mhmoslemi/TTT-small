def proj_capped_simplex(y, target, max_val=1.0, tol=1e-10, max_iter=100):
    lo = np.min(y) - max_val
    hi = np.max(y)
    for _ in range(max_iter):
        mid = (lo + hi) / 2
        x = np.clip(y - mid, 0, max_val)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            lo = mid
        else:
            hi = mid
    # after loop return x
    return np.clip(y - (lo+hi)/2, 0, max_val)
```

def grad_smooth_max(h, beta):
    corr = np.correlate(h, 1 - h, mode='full')

    g = conv1[n-1:n-1+n] - conv2[n-1:n-1+n]

def proj_capped_simplex(v, target, max_val=1.0, tol=1e-9, max_iter=100):
    lo = np.min(v) - max_val
    hi = np.max(v)
    for _ in range(max_iter):
        mid = (lo + hi) / 2
        x = np.clip(v - mid, 0, max_val)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2, 0, max_val)

def subgradient_descent(h0, target, max_iters, init_step, rng):
    h = h0.copy()
    step = init_step
    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        L = max_idx - (h.size - 1)
        # compute gradient g
        n = h.size
        g = np.zeros(n)
        if L >= 0:
            # part where k+L < n: k in [0, n-L-1]
            g[:n-L] = 1 - h[L:]
            # part where k-L >=0: k in [L, n-1]
            g[L:] -= h[:n-L]
        else:
            Lp = -L
            # k+L >=0 => k in [Lp, n-1]
            g[Lp:] = 1 - h[:n-Lp]
            # k-L < n => k in [0, n-Lp-1]
            g[:n-Lp] -= h[Lp:]
        h = h - step * g
        h = proj_capped_simplex(h, target)
        # decay step
        step = init_step * (1 - (it+1)/max_iters)
        # optionally early break if step small
    return h

import time
import numpy as np

# ---------- utilities ----------

def proj_capped_simplex(v, target, max_val=1.0, tol=1e-9, max_iter=100):
    """Euclidean projection of v onto {x: sum(x)=target, 0<=x<=max_val}."""
    lo = np.min(v) - max_val
    hi = np.max(v)
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, max_val)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, max_val)


def random_feasible_vector(n, rng):
    """Uniform random point on the capped simplex."""
    v = rng.random(n)
    return proj_capped_simplex(v, n / 2.0)


def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_multi(h_start, rng, max_iters=2000, init_step=0.5, n_trials=5):
    """Simple hill‑climb that tries random pairwise transfers."""
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c = c
                h = new_h
                improved = True
                break
        if not improved and rng.random() < 0.02:
            new_h = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(new_h)
            if c < best_c:
                best_c = c
                h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return h, best_c


# ---------- main optimisation ----------

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_limit = budget_s

    # ----- initial dimension and vector -----
    if "initial_h_values" in globals():

        n = h0.size
        target = n / 2.0
        h0 = proj_capped_simplex(np.clip(h0, 0.0, 1.0), target)
    else:
        n = 240
        target = n / 2.0
        h0 = random_feasible_vector(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ----- construct a pool of starting vectors -----
    starts = [best_h]
    xs = (np.arange(n) + 0.5) * (2.0 / n)

    # random vectors
    for _ in range(8):
        starts.append(random_feasible_vector(n, rng))

    # perturbed copies of current best
    for _ in range(5):

    # sinusoidal families
    for f in [1, 2, 3, 4, 5]:

        starts.append(proj_capped_simplex(np.clip(h, 0.0, 1.0), target))
        h = 0.5 + 0.5 * np.cos(np.pi * f * xs)
        starts.append(proj_capped_simplex(np.clip(h, 0.0, 1.0), target))

    # linear ramps
    starts.append(proj_capped_simplex(xs / 2.0, target))
    starts.append(proj_capped_simplex(1.0 - xs / 2.0, target))

    # binary masks
    mask = np.zeros(n)

    starts.append(proj_capped_simplex(mask, target))
    mask = np.zeros(n)

    mask[start : start + n // 2] = 1.0
    starts.append(proj_capped_simplex(mask, target))
    mask = np.arange(n) % 2
    starts.append(proj_capped_simplex(mask, target))

    # ----- subgradient descent on the true max (C5) -----
    # allocate about 80% of the time budget for this phase
    phase_time = time_limit * 0.8

    final_c = compute_c5(best_h)
