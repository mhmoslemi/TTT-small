import time
import numpy as np
from scipy.optimize import minimize

# Global placeholders for constraints and surrogate
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) via log‑sum‑exp."""
    bc = CURRENT_BETA * _corr_full_fft(x)
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def _corr_full_fft(h):
    """Correlation of h with (1‑h) using FFT (full mode)."""
    n = h.shape[0]
    b = (1.0 - h)[::-1]
    L = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(h, L)
    fb = np.fft.rfft(b, L)
    c = np.fft.irfft(fa * fb, L)[: 2 * n - 1]
    return c


def compute_c5(h):
    """Exact discrete C5 for a step function h."""
    n = h.shape[0]
    return np.max(_corr_full_fft(h)) * (2.0 / n)


def project_to_feasible(v, target):
    """Project onto {0≤v≤1, Σv = target} by shifting and clipping."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Fast random feasible vector (0≤h≤1, Σh=n/2)."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_to_feasible(v, n / 2.0)


def upsample_initial(to_n):
    """Linear interpolation of the global initial_h_values (if present)."""
    init = np.array(initial_h_values, dtype=float)
    m = init.size
    xs_old = np.linspace(0.0, 2.0, m, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, to_n, endpoint=False)
    new = np.interp(xs_new, xs_old, init)
    return new


def perturb_h(h, rng, scale=0.2):
    """Transfer a random ε between two coordinates while keeping the sum."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=3000, start_step=0.5):
    """Deterministic hill‑climb; accept only improvements."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


def slsqp_optimize(h0, beta, rng, max_iter=500):
    """SLSQP refinement using the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def binary_hill_climb(h_bin, rng, max_iters=20000, patience=2000):
    """Hill‑climb on binary vectors (swap a 1 and a 0)."""
    n = h_bin.size
    best_h = h_bin.copy()
    best_c = compute_c5(best_h)
    no_improve = 0
    for _ in range(max_iters):
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # perform swap
        best_h[i] = 0.0
        best_h[j] = 1.0
        c = compute_c5(best_h)
        if c < best_c:
            best_c = c
            no_improve = 0
        else:
            # revert
            best_h[i] = 1.0
            best_h[j] = 0.0
            no_improve += 1
            if no_improve >= patience:
                break
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise the Erdős C₅ overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # Candidate discretisations (increase gradually)
    candidate_ns = [120, 180, 240, 300, 400, 500, 600]
    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        if n0 not in candidate_ns:
            candidate_ns = [n0] + candidate_ns

    for n in candidate_ns:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.96:
            break
        # ----- initialise -----
        if "initial_h_values" in globals():
            if n == len(initial_h_values):
                h0 = np.array(initial_h_values, dtype=float)
            else:
                h0 = upsample_initial(n)
        else:
            h0 = random_feasible_vector(n, rng)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_to_feasible(h0, n / 2.0)

        # ----- SLSQP refinement with increasing beta -----
        h_cur = h0
        for beta in (10.0, 50.0, 200.0, 800.0, 2000.0, 8000.0):
            if time.time() - start_time > budget_s * 0.94:
                break
            try:
                h_cur = slsqp_optimize(h_cur, beta, rng, max_iter=500)
            except Exception:
                continue
            h_cur = np.clip(h_cur, 0.0, 1.0)
            h_cur = project_to_feasible(h_cur, n / 2.0)
            c_cur = compute_c5(h_cur)
            if c_cur < best_c:
                best_c, best_h, best_n = c_cur, h_cur.copy(), n

        # ----- Continuous hill‑climb -----
        if time.time() - start_time < budget_s * 0.96:
            h_cur, c_cur = hill_climb(h_cur, rng, max_iters=5000, start_step=0.4)
            if c_cur < best_c:
                best_c, best_h, best_n = c_cur, h_cur.copy(), n

        # ----- Binary rounding + binary hill‑climb -----
        if time.time() - start_time < budget_s * 0.98:
            # round to nearest binary vector keeping exactly n/2 ones
            idx_desc = np.argsort(-h_cur)
            h_bin = np.zeros(n, dtype=float)
            h_bin[idx_desc[: n // 2]] = 1.0
            h_bin, c_bin = binary_hill_climb(h_bin, rng, max_iters=20000, patience=3000)
            if c_bin < best_c:
                best_c, best_h, best_n = c_bin, h_bin.copy(), n

    # Fallback if nothing succeeded
    if best_h is None:
        best_n = 120
        best_h = np.full(best_n, 0.5)
        best_h = project_to_feasible(best_h, best_n / 2.0)
        best_c = compute_c5(best_h)

    # Final feasibility enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, best_n)
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
