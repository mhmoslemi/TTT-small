def gradient_softmax(h, beta):
    # compute correlation and softmax probabilities
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    # compute shifted exponentials for stability
    exp_vals = np.exp(beta * (corr - max_corr))
    p = exp_vals / exp_vals.sum()
    # compute gradient via convolution
    # p array length 2n-1; reorder? p corresponds to lags ℓ from -(n-1) to (n-1)
    # conv of p with (1-h)
    term1 = np.convolve(p, 1 - h, mode='full')
    term2 = np.convolve(p, h, mode='full')
    # Align to h length: offset = n-1
    start = n - 1

import time
import numpy as np

# ---------- utility functions ----------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for step function h (dx = 2 / n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_capped_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Projection of v onto {x | 0 <= x_i <= 1, sum x_i = target}.
    Uses simple bisection on the shift λ.
    """
    lo, hi = -2.0, 2.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random h with entries in [0,1] and sum = n/2."""
    v = rng.random(n)
    return project_onto_capped_simplex(v, n / 2.0)


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h: np.ndarray, max_iters: int, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Simple hill‑climb using transfer moves."""
    best = h.copy()
    best_c = compute_c5(best)
    n = best.size
    for _ in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(best[i], 1.0 - best[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.2
        cand = best.copy()
        cand[i] -= eps
        cand[j] += eps
        c = compute_c5(cand)
        if c < best_c:
            best, best_c = cand, c
    return best, best_c


def gradient_softmax(h: np.ndarray, beta: float):
    """
    Gradient of the smooth max objective fβ(h) = (1/β) log Σexp(β·cℓ)
    where cℓ = Σ_i h_i (1 − h_{i+ℓ}).
    Returns gradient, softmax probabilities and raw correlations.
    """
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    exp_vals = np.exp(beta * (corr - m))
    p = exp_vals / exp_vals.sum()
    # convolution to obtain Σ pℓ (1−h_{i+ℓ}) and Σ pℓ h_{i−ℓ}
    term1 = np.convolve(p, 1 - h, mode="full")
    term2 = np.convolve(p, h, mode="full")
    start = n - 1
    grad = term1[start:start + n] - term2[start:start + n]
    return grad, p, corr


def gradient_descent_h(
    h0: np.ndarray,
    beta_schedule,
    lr_factor: float,
    iters_per_beta: int,
    rng: np.random.Generator,
):
    """Projected gradient descent on the smooth surrogate."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)

    for beta in beta_schedule:
        for _ in range(iters_per_beta):
            g, _, _ = gradient_softmax(h, beta)
            norm = np.linalg.norm(g)
            if norm > 0:
                step = lr_factor / (norm * (beta + 1.0))
                h = project_onto_capped_simplex(h - step * g, target)
            c = compute_c5(h)
            if c < best_c:
                best_c, best_h = c, h.copy()
    return best_h, best_c

    start_time = time.time()

    # ---- initialise ----
    if "initial_h_values" in globals():

        n = h0.size

    else:

        h0 = random_feasible_vector(n, rng)

    best_h = h0
    best_c = compute_c5(best_h)

    # parameters
    beta_sched = [5, 20, 80, 200, 500, 1000, 2000]
    iters_beta = 120
    lr_factor = 0.20
    hill_iters = 300

    # ---- optimisation loop ----
    while time.time() - start_time < budget_s * 0.92:

        if rng.random() < 0.65:

            start_h, beta_sched, lr_factor, iters_beta, rng
        )
        if c_gd < best_c:

        h_hc, c_hc = hill_climb(best_h, hill_iters, rng)

            best_c = compute_c5(best_h)

    best_h = project_onto_capped_simplex(best_h, n / 2.0)
