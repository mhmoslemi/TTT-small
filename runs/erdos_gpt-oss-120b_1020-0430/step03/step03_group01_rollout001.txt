def hill_climb(h0, rng, max_iters=2000, scale_start=0.5, trials_per_iter=5, decay=0.8):
    h = h0.copy()
    best_c = compute_c5(h)
    scale = scale_start

        improved = False
        for _ in range(trials_per_iter):
            new_h = perturb_transfer(h, rng, scale=scale)
            new_c = compute_c5(new_h)
            if new_c < best_c:
                best_c = new_c
                h = new_h
                improved = True
                break
        if not improved:

            if rng.random() < 0.01:
                new_h = perturb_transfer(h, rng, scale=scale*2)
                new_c = compute_c5(new_h)
                if new_c < best_c:
                    best_c = new_c
                    h = new_h
        # decay scale periodically
        if (it+1) % (max_iters//10) == 0:
            scale = max(scale*decay, 1e-5)
    return h, best_c

def enforce_sum(h, target, rng):
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    diff = target - total
    n = h.size
    # if diff is small, just return
    if abs(diff) < 1e-12:
        return h
    # try to adjust by adding diff/n to all items, then reclip; repeat a few times
    for _ in range(10):
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        total = h.sum()
        diff = target - total
        if abs(diff) < 1e-12:
            break
    # final iterative fill if still diff
    if abs(diff) > 1e-12:
        # adjust greedily
        for _ in range(1000):
            if abs(diff) < 1e-12:
                break
            # choose random index
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1-x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)


def enforce_sum(h, target, rng):
    """Project h onto {0≤h≤1, Σh = target}."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    # simple redistribution
    if abs(diff) > 1e-12:
        for _ in range(10):
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    # final greedy fix if needed
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with Σ = n/2."""
    h = rng.random(n)
    h = h / h.sum() * (n / 2.0)
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def perturb_transfer(h, rng, scale=0.2):
    """Move a small ε from one coordinate to another, preserving the sum."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h0, rng, max_iters=2500, scale_start=0.5, trials_per_iter=6):
    """Simple stochastic hill‑climb on the exact C5."""
    h = h0.copy()
    best_c = compute_c5(h)
    scale = scale_start
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(trials_per_iter):
            new_h = perturb_transfer(h, rng, scale=scale)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
                improved = True
                break
        if not improved and rng.random() < 0.02:
            new_h = perturb_transfer(h, rng, scale=scale * 2.0)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
        if (it + 1) % decay_every == 0:
            scale = max(scale * 0.7, 1e-5)
    return h, best_c


def slsqp_optimize(h0, beta, max_iter=400):
    """Refine h using SLSQP on the smooth surrogate with given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    constraints = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options={"maxiter": max_iter, "ftol": 1e-10, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny drift correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    main_budget = budget_s * 0.75

    # ------------------------------------------------------------------
    # 1. Base vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    best_h, best_c = h0, compute_c5(h0)

    # ------------------------------------------------------------------
    # 2. Small pool of diverse seeds
    # ------------------------------------------------------------------
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    for f in (1, 2, 3, 5, 7):
        s = 0.5 + 0.5 * np.sin(np.pi * f * xs)
        s = np.clip(s, 0.0, 1.0)
        s = enforce_sum(s, n / 2.0, rng)
        c = compute_c5(s)
        if c < best_c:
            best_h, best_c = s, c
    # linear ramps
    ramp = xs / 2.0
    ramp = enforce_sum(ramp, n / 2.0, rng)
    c = compute_c5(ramp)
    if c < best_c:
        best_h, best_c = ramp, c
    # reversed ramp
    ramp2 = 1.0 - xs / 2.0
    ramp2 = enforce_sum(ramp2, n / 2.0, rng)
    c = compute_c5(ramp2)
    if c < best_c:
        best_h, best_c = ramp2, c

    # ------------------------------------------------------------------
    # 3. Main iterative search
    # ------------------------------------------------------------------
    beta_seq = [10, 50, 200, 800, 2000, 5000]
    while time.time() - start < main_budget:
        # choose start point
        if rng.random() < 0.7:
            start_vec = perturb_transfer(best_h, rng, scale=0.3)
        else:
            start_vec = random_feasible_vector(n, rng)

        # hill climb
        hc_h, hc_c = hill_climb(start_vec, rng, max_iters=3000,
                                scale_start=0.5, trials_per_iter=8)
        if hc_c < best_c:
            best_h, best_c = hc_h, hc_c

        # SLSQP refinement ladder
        for beta in beta_seq:
            if time.time() - start >= main_budget:
                break
            try:
                ref_h = slsqp_optimize(best_h, beta, max_iter=500)
            except Exception:
                continue
            ref_c = compute_c5(ref_h)
            if ref_c < best_c:
                best_h, best_c = ref_h, ref_c

        # occasional crossover between two good solutions
        if rng.random() < 0.15:
            other = random_feasible_vector(n, rng)
            other, _ = hill_climb(other, rng, max_iters=2000,
                                  scale_start=0.4, trials_per_iter=6)
            mix = (best_h + other) / 2.0
            mix = enforce_sum(mix, n / 2.0, rng)
            mix, mix_c = hill_climb(mix, rng, max_iters=2500,
                                    scale_start=0.4, trials_per_iter=6)
            if mix_c < best_c:
                best_h, best_c = mix, mix_c

    # ------------------------------------------------------------------
    # 4. Upsampling & fine refinement if time permits
    # ------------------------------------------------------------------
    remain = budget_s - (time.time() - start)
    if remain > budget_s * 0.12:
        target_n = min(720, n * 2)
        if target_n != n:
            old_x = (np.arange(n) + 0.5) * (2.0 / n)
            new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
            h_up = np.interp(new_x, old_x, best_h)
            h_up = enforce_sum(h_up, target_n / 2.0, rng)

            h_up, up_c = hill_climb(h_up, rng, max_iters=4000,
                                    scale_start=0.3, trials_per_iter=8)
            # SLSQP ladder on upsampled grid
            for beta in beta_seq[1:]:   # skip tiny beta
                if time.time() - start >= budget_s:
                    break
                try:
                    h_up = slsqp_optimize(h_up, beta, max_iter=500)
                except Exception:
                    pass
            up_c = compute_c5(h_up)
            if up_c < best_c:
                best_h, best_c, n = h_up, up_c, target_n

    # ------------------------------------------------------------------
    # 5. Final clean‑up
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
