def simulated_annealing(h_start, rng, time_limit):

    n = h_start.size

    cur = objective_val(h)

            best_h = h.copy()

    # ------------------------------------------------------------------
    # Phase 1 – diversified short SA runs (exploration)
    # ------------------------------------------------------------------
    time_elapsed = time.time() - start_time
    time_left = budget_s - time_elapsed
    # allocate roughly half of remaining budget to this phase
    phase1_budget = time_left * 0.5
    per_run = 20.0                       # seconds per short run
    while time.time() - start_time < start_time + phase1_budget:
        if time.time() - start_time > budget_s * 0.95:
            break
        run_time = min(per_run, budget_s - (time.time() - start_time))
        # pick a random start (binary or continuous)
        if rng.random() < 0.5:
            h_start = random_binary_vector(n, rng)
        else:
            v = rng.exponential(1.0, size=n)
            v = v / v.sum() * target
            h_start = np.clip(v, 0.0, 1.0)
            h_start = enforce_sum(h_start, target, rng)
        h_sa, _ = simulated_annealing(h_start, rng, run_time)
        c_exact = compute_c5_exact(h_sa)
        if c_exact < best_c:
            best_c = c_exact
            best_h = h_sa.copy()

    # ------------------------------------------------------------------
    # Phase 2 – intensive SA on current best (exploitation)
    # ------------------------------------------------------------------
    time_elapsed = time.time() - start_time
    time_left = budget_s - time_elapsed
    if time_left > 5.0:   # keep a few seconds for final polishing
        h_sa, _ = simulated_annealing(best_h, rng, time_left * 0.6)
        c_exact = compute_c5_exact(h_sa)
        if c_exact < best_c:
            best_c = c_exact
            best_h = h_sa.copy()

    # ------------------------------------------------------------------
    # Phase 3 – greedy hill‑climb fine‑tuning
    # ------------------------------------------------------------------
    time_elapsed = time.time() - start_time
    time_left = budget_s - time_elapsed
    if time_left > 1.0:
        iters = int(time_left * 2000)   # about 2000 moves per second
        h_hc, _ = greedy_hill_climb(best_h, rng, iters)
        c_exact = compute_c5_exact(h_hc)
        if c_exact < best_c:
            best_c = c_exact
            best_h = h_hc.copy()

    # ------------------------------------------------------------------
    # Final safety‑check and return
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_c = compute_c5_exact(best_h)

    return best_h.tolist(), float(final_c), int(n)
