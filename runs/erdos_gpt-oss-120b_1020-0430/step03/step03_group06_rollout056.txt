def hill_climb_opt(h_start, rng, max_iters=2000, start_step=0.2):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)
        max_pos = min(h[i], 1.0 - h[j])

            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.8, 1e-5)
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global helpers for the SLSQP surrogate (log‑sum‑exp of the overlaps)
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    """Smooth log‑sum‑exp surrogate of max_{k} Σ_i x_i (1‑x_{i+k})."""
    beta = CURRENT_BETA
    corr = np.correlate(x, 1 - x, mode="full")
    bc = beta * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta


# ----------------------------------------------------------------------
# Problem‑specific utilities
# ----------------------------------------------------------------------
def compute_c5(h):
    """Exact discrete C₅ for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(vec, target, rng):
    """Adjust vec so that Σ vec = target while 0≤vec≤1."""
    diff = target - vec.sum()
    n = vec.size
    for _ in range(20000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - vec[i])
        else:
            add = max(diff, -vec[i])
        if abs(add) < 1e-12:
            continue
        vec[i] += add
        diff = target - vec.sum()
    return np.clip(vec, 0.0, 1.0)


def random_feasible(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    v = rng.exponential(1.0, size=n)
    v = v / v.sum() * (n / 2.0)
    v = np.clip(v, 0.0, 1.0)
    return enforce_sum(v, n / 2.0, rng)


def grad_log_sum_exp(h, beta):
    """Gradient of the log‑sum‑exp surrogate."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    bc = beta * corr
    m = np.max(bc)
    w = np.exp(bc - m)
    w /= np.sum(w)                       # probabilities p_k
    # conv with w (original) and reversed w
    conv_orig = np.convolve(w, h, mode="full")
    conv_rev = np.convolve(w[::-1], h, mode="full")
    offset = len(w) - 1                  # = 2*n-2
    grad = 1.0 - conv_orig[offset : offset + n] - conv_rev[offset : offset + n]
    return grad


def project_box_sum(v, target, rng, iters=60):
    """Project v onto {0≤x≤1, Σx=target} using a simple binary search on a shift."""
    lo, hi = -1.0, 1.0

    def f(lam):
        return np.sum(np.clip(v + lam, 0.0, 1.0))

    # enlarge bounds if needed
    while f(lo) > target:
        lo *= 2.0
    while f(hi) < target:
        hi *= 2.0
    for _ in range(iters):
        mid = (lo + hi) * 0.5
        s = f(mid)
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(v + lam, 0.0, 1.0)


def hill_climb(h_start, rng, max_iters=3000, start_step=0.5):
    """Local hill‑climb using pairwise transfers (preserves sum)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)
        max_pos = min(h[i], 1.0 - h[j])
        max_neg = min(h[j], 1.0 - h[i])
        if max_pos + max_neg == 0.0:
            continue
        eps = (rng.random() * (max_pos + max_neg)) - max_neg
        eps *= step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


def slsqp_refine(h0, betas, rng, max_iter=200):
    """Run a short SLSQP refinement for a list of β values."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    h = np.clip(h0, 0.0, 1.0)
    for beta in betas:
        CURRENT_BETA = beta
        try:
            res = minimize(
                _softmax_obj,
                h,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
            )
            h = np.clip(res.x, 0.0, 1.0)
            # tiny repair of the sum
            diff = CURRENT_TARGET - h.sum()
            if abs(diff) > 1e-12:
                h = enforce_sum(h, CURRENT_TARGET, rng)
        except Exception:
            pass
    return h

# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----- initialise -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 240
        h0 = np.full(n, 0.5)
        h0 = enforce_sum(h0, n / 2.0, rng)

    target = n / 2.0
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ----- main stochastic search -----
    while time.time() - start_time < budget_s * 0.78:
        # start point: either perturb current best or fresh random
        if rng.random() < 0.6:
            start = best_h.copy()
            # small random transfer moves
            for _ in range(8):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                max_pos = min(start[i], 1.0 - start[j])
                max_neg = min(start[j], 1.0 - start[i])
                if max_pos + max_neg == 0.0:
                    continue
                eps = (rng.random() * (max_pos + max_neg)) - max_neg
                eps *= 0.15
                start[i] -= eps
                start[j] += eps
        else:
            start = random_feasible(n, rng)

        # ----- annealed gradient descent on the log‑sum‑exp surrogate -----
        h = start.copy()
        betas = [1.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0, 640.0]
        for b_idx, beta in enumerate(betas):
            iters = 120
            step = 0.25 / (1.0 + b_idx)  # gradually shrinking step
            for _ in range(iters):
                grad = grad_log_sum_exp(h, beta)
                h = project_box_sum(h - step * grad, target, rng)
                # occasional cheap evaluation
                if rng.random() < 0.03:
                    c = compute_c5(h)
                    if c < best_c:
                        best_c = c
                        best_h = h.copy()
                # time guard
                if time.time() - start_time > budget_s * 0.78:
                    break
            if time.time() - start_time > budget_s * 0.78:
                break

        # ----- local hill‑climb refinement -----
        h_loc, c_loc = hill_climb(h, rng, max_iters=2000, start_step=0.2)
        if c_loc < best_c:
            best_c = c_loc
            best_h = h_loc.copy()

        # ----- occasional SLSQP refinement when a new record is found -----
        if best_c == c_loc:  # we just improved
            h_ref = slsqp_refine(best_h, [10.0, 40.0, 120.0, 400.0, 800.0], rng, max_iter=150)
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_c = c_ref
                best_h = h_ref.copy()

    # ----- optional up‑sampling for a finer grid -----
    if time.time() - start_time < budget_s * 0.93 and n * 2 <= 480:
        h_up = np.repeat(best_h, 2)
        n_up = h_up.size
        target_up = n_up / 2.0
        h_up = enforce_sum(h_up, target_up, rng)

        # brief gradient descent on the finer grid
        betas_fine = [80.0, 160.0, 320.0, 640.0]
        h = h_up.copy()
        for beta in betas_fine:
            for _ in range(150):
                grad = grad_log_sum_exp(h, beta)
                h = project_box_sum(h - 0.07, target_up, rng)  # fixed small step
                h = enforce_sum(h, target_up, rng)
        # final hill‑climb on the up‑sampled grid
        h_fine, c_fine = hill_climb(h, rng, max_iters=3000, start_step=0.1)
        if c_fine < best_c:
            best_c = c_fine
            best_h = h_fine.copy()
            n = n_up

    # ----- final clean‑up -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
