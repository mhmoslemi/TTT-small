def de_optimize(h0, rng, time_budget):
    n = h0.size
    target_sum = n/2
    popsize = max(10, min(30, n//6)) # ~30
    # initialize population
    pop = np.empty((popsize, n), dtype=float)
    pop[0] = h0.copy()
    for i in range(1, popsize):
        pop[i] = random_feasible_vector(n, rng)
    # evaluate fitness
    fitness = np.empty(popsize)
    for i in range(popsize):
        fitness[i] = compute_c5(pop[i])
    # track best
    best_idx = np.argmin(fitness)
    best_h = pop[best_idx].copy()
    best_f = fitness[best_idx]
    
    start = time.time()
    # parameters
    F = 0.8
    CR = 0.9
    # loop
    gen = 0
    while time.time() - start < time_budget:
        for i in range(popsize):
            # mutation
            idxs = list(range(popsize))
            idxs.remove(i)
            a,b,c = rng.choice(idxs, 3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            # crossover
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(0,n)] = True
            trial = np.where(cross, mutant, pop[i])
            # clip and enforce sum
            trial = np.clip(trial, 0.0, 1.0)
            enforce_sum(trial, target_sum, rng)
            # compute fitness
            f = compute_c5(trial)
            if f < fitness[i]:
                pop[i] = trial
                fitness[i] = f
                if f < best_f:
                    best_f = f
                    best_h = trial.copy()
        # adapt parameters? maybe occasionally adjust F, CR.
        gen += 1
        # maybe adjust F and CR based on progress.
        # break if time near out
        if time.time() - start > time_budget * 0.99:
            break
    return best_h, best_f
```

def softmax_obj(x, beta):
    corr = np.correlate(x, 1-x, mode='full')
    bc = beta * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta

def slsqp_optimize(h0, beta, max_iter=200):
    n = h0.size
    target = n / 2.0
    bounds = [(0.0,1.0)]*n
    cons = {'type':'eq', 'fun': lambda x: np.sum(x) - target}
    res = minimize(lambda x: softmax_obj(x,beta), h0, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter':max_iter,'ftol':1e-12,'disp':False})
    x = np.clip(res.x,0.0,1.0)
    enforce_sum(x, target, np.random.default_rng())
    return x
```

def enforce_sum_one_pass(h, target):
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # find index with capacity
    if diff > 0:
        # add
        for i in range(h.size):
            add = min(diff, 1.0 - h[i])
            if add > 1e-12:
                h[i] += add
                diff -= add
                if abs(diff) < 1e-12:
                    break
    else:
        diff = -diff
        for i in range(h.size):
            sub = min(diff, h[i])
            if sub > 1e-12:
                h[i] -= sub
                diff -= sub
                if diff < 1e-12:
                    break
    return h

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    # tolerance huge loops not needed, just adjust with two indices
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 10:
        if diff > 0:
            # need to add diff: find idx where h[i] < 1
            idx = rng.choice(np.where(h < 1.0)[0])
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff = target - h.sum()
        else:
            idx = rng.choice(np.where(h > 0.0)[0])
            sub = min(-diff, h[idx])
            h[idx] -= sub
            diff = target - h.sum()
        attempts += 1
    return h

def random_feasible_vector(n, rng):
    h = np.full(n, 0.5)
    # random transfers
    for _ in range(n * 8):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])

        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.5
        h[i] -= eps
        h[j] += eps
    enforce_sum(h, n/2.0, rng)

import time
import math
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h):
    """Discrete C5 for step function h (dx = 2/n)."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)


def enforce_sum(h, target):
    """Adjust h in‑place so that sum(h)=target while keeping 0≤h≤1."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        for i in range(h.size):
            cap = 1.0 - h[i]
            if cap <= 0.0:
                continue
            add = min(diff, cap)
            h[i] += add
            diff -= add
            if diff <= 1e-12:
                break
    else:
        diff = -diff
        for i in range(h.size):
            cap = h[i]
            if cap <= 0.0:
                continue
            sub = min(diff, cap)
            h[i] -= sub
            diff -= sub
            if diff <= 1e-12:
                break
    return h


def random_feasible_vector(n, rng):
    """Generate a random h with 0≤h≤1 and sum=n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 8):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.5
        h[i] -= eps
        h[j] += eps
    enforce_sum(h, n / 2.0)
    return h


# ---------- soft‑max objective for SLSQP ----------
def softmax_obj(x, beta):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = beta * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta


def slsqp_optimize(h0, beta, max_iter=200):
    """Local refinement using a smooth max‑approximation."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    res = minimize(lambda x: softmax_obj(x, beta), h0,
                   method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, target)
    return x


# ---------- Differential Evolution ----------
def de_optimize(h0, rng, time_budget):
    n = h0.size
    target = n / 2.0
    popsize = max(10, min(30, n // 6))          # 10–30 individuals
    pop = np.empty((popsize, n), dtype=float)
    pop[0] = np.clip(h0, 0.0, 1.0)
    enforce_sum(pop[0], target)
    for i in range(1, popsize):
        pop[i] = random_feasible_vector(n, rng)

    fitness = np.empty(popsize, dtype=float)
    for i in range(popsize):
        fitness[i] = compute_c5(pop[i])

    best_idx = int(np.argmin(fitness))
    best_h, best_f = pop[best_idx].copy(), fitness[best_idx]

    start = time.time()
    F = 0.8
    CR = 0.9

    while time.time() - start < time_budget:
        for i in range(popsize):
            # mutation: select three distinct others
            idxs = list(range(popsize))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, 3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])

            # crossover (binomial)
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(0, n)] = True
            trial = np.where(cross, mutant, pop[i])

            # respect bounds and the sum constraint
            trial = np.clip(trial, 0.0, 1.0)
            enforce_sum(trial, target)

            # evaluate
            f = compute_c5(trial)
            if f < fitness[i]:
                pop[i] = trial
                fitness[i] = f
                if f < best_f:
                    best_f, best_h = f, trial.copy()
        # quick time check to avoid overrunning the budget
        if time.time() - start > time_budget * 0.99:
            break
    return best_h, best_f


# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- initial vector ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)

    else:
        n = 180
        h0 = random_feasible_vector(n, rng)

    # ---- Phase 1: Differential Evolution ----
    de_budget = budget_s * 0.78
    best_h, best_c = de_optimize(h0, rng, de_budget)

    # ---- Phase 2: SLSQP refinements ----
    remaining = budget_s - (time.time() - start)
    betas = [200, 800, 2000, 8000, 20000]
    for beta in betas:
        if time.time() - start >= budget_s:
            break
        refined = slsqp_optimize(best_h, beta, max_iter=200)
        refined_c = compute_c5(refined)
        if refined_c < best_c:
            best_h, best_c = refined, refined_c

    # ---- final cleanup ----
    best_h = np.clip(best_h, 0.0, 1.0)
    enforce_sum(best_h, n / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
