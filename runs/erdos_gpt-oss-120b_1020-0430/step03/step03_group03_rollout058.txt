def compute_overlap_fft(h):
    n = h.size
    n2 = 1 << (2 * n - 1).bit_length()  # next power of two >= 2n-1
    # compute linear convolution h * (1 - h)
    f1 = np.fft.rfft(h, n2)
    f2 = np.fft.rfft(1 - h, n2)
    conv = np.fft.irfft(f1 * f2, n2)[:2*n-1]
    return conv * (2.0 / n)

def compute_weighted_convolutions(w, h):

    lo = -np.max(v - lower)  # when λ = lo, all v_i - λ >= lower (no lower bound active)

import time, numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Exact discrete C5 for step function h (dx=2/n)."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)

def enforce_sum(h, target, rng):
    """Adjust h in‑place so that sum(h)=target while respecting 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

# ----- projection onto box ∩ hyperplane -----
def project_onto_feasible(v, target, max_iter=70):
    lo, hi = -1e5, 1e5
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) < 1e-12:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)

# ----- weighted forward/backward convolutions -----
def weighted_fwd_bwd(w, h):
    n = h.size
    forward = np.zeros(n)
    backward = np.zeros(n)
    off = n - 1
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        k = idx - off
        if k >= 0:
            forward[: n - k] += wk * h[k:]
            backward[k:] += wk * h[: n - k]
        else:
            s = -k
            forward[s:] += wk * h[: n - s]
            backward[: n - s] += wk * h[s:]
    return forward, backward

# ----- SLSQP surrogate (smooth max) -----
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full") * (2.0 / x.size)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=200):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {"type": "eq", "fun": sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

# ----- gradient descent with smooth max -----
def gd_optimize(h, target, betas, etas, iters, rng):
    n = h.size
    dx = 2.0 / n
    for beta, eta, steps in zip(betas, etas, iters):
        for _ in range(steps):
            O = np.correlate(h, 1 - h, mode="full") * dx
            m = O.max()
            w = np.exp(beta * (O - m))
            s = w.sum()
            if s == 0:
                w.fill(1.0 / w.size)
            else:
                w /= s
            fwd, bwd = weighted_fwd_bwd(w, h)
            grad = 1.0 - (fwd + bwd)
            h = h - eta * grad
            h = project_onto_feasible(h, target)
    return h

# ----- main entry point -----
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # initialise h
    if "initial_h_values" in globals():

    final_c = compute_c5(best_h)
