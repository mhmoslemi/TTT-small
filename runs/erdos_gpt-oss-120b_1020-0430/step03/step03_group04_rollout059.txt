def simulated_annealing_binary(start_h, rng, deadline):
    n = start_h.size
    cur_h = start_h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c

    # Build lists
    ones = [int(i) for i in np.flatnonzero(cur_h > 0.5)]
    zeros = [int(i) for i in np.flatnonzero(cur_h < 0.5)]

    # temperature schedule
    T0 = cur_c * 0.2 + 1e-6
    decay = 5e-6  # we can compute based on max_iters approximate

    it = 0
    while time.time() < deadline:
        if not ones or not zeros:
            break
        # randomly select a pair
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        # perform swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        cand_c = compute_c5(cur_h)
        # compute T
        T = max(T0 * np.exp(-decay * it), 1e-12)
        accept = False
        if cand_c < cur_c:
            accept = True
        else:
            if rng.random() < np.exp(-(cand_c - cur_c) / T):
                accept = True
        if accept:
            cur_c = cand_c
            # update lists
            ones[i_idx] = j
            zeros[j_idx] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i] = 1.0
            cur_h[j] = 0.0
        it += 1
    return best_h, best_c

def gradient_refine(h, rng, step=0.02, max_iter=20):
    # uses algorithm in previous code, but we might increase iterations and adjust step schedule
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]  # derivative w.r.t h[i] decreasing C
            g[k:] -= h[:upper]       # derivative w.r.t h[i+k] also
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        # clip
        np.clip(h, 0.0, 1.0, out=h)
        # enforce sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # adjust diff by adding to random entries
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

import time

# ------------------- basic utilities ------------------- #
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for step function h on [0,2] with uniform spacing."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def random_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.02, max_iter: int = 20) -> np.ndarray:
    """Continuous refinement moving against gradient of the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]      # ∂/∂h[i] of term -h[i]h[i+k]
            g[k:] -= h[:upper]            # ∂/∂h[i+k]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


def simulated_annealing_binary(start_h: np.ndarray,
                               rng: np.random.Generator,
                               deadline: float) -> tuple[np.ndarray, float]:
    """Binary simulated‑annealing (swap a 1 with a 0) respecting the sum constraint."""
    n = start_h.size
    cur_h = start_h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c

    # maintain mutable Python lists of indices
    ones = [int(i) for i in np.flatnonzero(cur_h > 0.5)]
    zeros = [int(i) for i in np.flatnonzero(cur_h < 0.5)]

    T0 = cur_c * 0.5 + 1e-12          # initial temperature
    decay = 5e-6                     # exponential decay factor
    it = 0
    while time.time() < deadline:
        if not ones or not zeros:
            break
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]

        cur_h[i] = 0.0
        cur_h[j] = 1.0
        cand_c = compute_c5(cur_h)

        T = max(T0 * np.exp(-decay * it), 1e-12)
        accept = cand_c < cur_c or rng.random() < np.exp(-(cand_c - cur_c) / T)

        if accept:
            cur_c = cand_c
            # update index lists
            ones[i_idx] = j
            zeros[j_idx] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i] = 1.0
            cur_h[j] = 0.0
        it += 1
    return best_h, best_c


# ------------------- main driver ------------------- #
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    best_h = None
    best_c = np.inf
    best_n = None

    # parent construction, if supplied
    parent = None
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)

    def parent_binary_for_n(n):
        """Interpolate parent to length n, then take top n/2 values as ones."""
        x_src = np.linspace(0.0, 2.0, parent.size, endpoint=False)
        x_tgt = np.linspace(0.0, 2.0, n, endpoint=False)
        interp = np.interp(x_tgt, x_src, parent)
        idx = np.argsort(-interp)[: n // 2]
        bin_h = np.zeros(n, dtype=float)
        bin_h[idx] = 1.0
        return bin_h

    candidate_ns = [180, 144, 240, 120, 96]  # ordered by decreasing size

    for n in candidate_ns:
        if time.time() - start_time > budget_s * 0.95:
            break

        remaining = budget_s - (time.time() - start_time)
        max_n_time = min(remaining * 0.6, 300.0)

        # initialise start vectors
        starts = []

        # alternating 1‑0 pattern
        alt = np.zeros(n, dtype=float)
        alt[::2] = 1.0
        starts.append(alt)

        # single block
        blk = np.zeros(n, dtype=float)
        blk[: n // 2] = 1.0
        starts.append(blk)

        # parent derived binary
        if parent is not None:
            try:
                starts.append(parent_binary_for_n(n))
            except Exception:
                pass

        # a random binary vector
        starts.append(random_binary(n, rng))

        per_start = max_n_time / len(starts)

        for base in starts:
            # guarantee exact n/2 ones
            if int(base.sum()) != n // 2:
                idx = np.argsort(-base)[: n // 2]
                h0 = np.zeros(n, dtype=float)
                h0[idx] = 1.0
            else:
                h0 = base.copy()

            deadline_sa = time.time() + per_start
            bh, bc = simulated_annealing_binary(h0, rng, deadline_sa)
            if bc < best_c:
                best_h, best_c, best_n = bh.copy(), bc, n

            # continuous refinement of the binary result
            bh_cont = gradient_refine(bh.copy(), rng, step=0.02, max_iter=20)
            bc_cont = compute_c5(bh_cont)
            if bc_cont < best_c:
                best_h, best_c, best_n = bh_cont.copy(), bc_cont, n

    # final continuous polishing on the overall best
    if best_h is not None:
        best_h = gradient_refine(best_h.copy(), rng, step=0.01, max_iter=30)
        final_c = compute_c5(best_h)
        best_c = final_c

    # optional up‑sampling refinement (if time permits)
    if best_h is not None and best_n * 2 <= 240:
        time_left = budget_s - (time.time() - start_time)
        if time_left > 10.0:
            n_up = best_n * 2
            x_src = np.linspace(0.0, 2.0, best_n, endpoint=False)
            x_up = np.linspace(0.0, 2.0, n_up, endpoint=False)
            interp = np.interp(x_up, x_src, best_h)
            idx = np.argsort(-interp)[: n_up // 2]
            h_up = np.zeros(n_up, dtype=float)
            h_up[idx] = 1.0
            deadline_up = time.time() + time_left * 0.9
            bh_up, bc_up = simulated_annealing_binary(h_up, rng, deadline_up)
            bh_up = gradient_refine(bh_up, rng, step=0.02, max_iter=20)
            bc_up = compute_c5(bh_up)
            if bc_up < best_c:
                best_h, best_c, best_n = bh_up, bc_up, n_up

    # safety checks
    if best_h is None:
        best_n = 180
        best_h = np.full(best_n, 0.5, dtype=float)
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            i = rng.integers(0, best_n)
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()
            if abs(diff) < 1e-12:
                break

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
