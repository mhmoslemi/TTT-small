def project_sum_bounds(h, target):
    # clip to [0,1]; then adjust to meet sum via simple redistribution
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # if diff >0 add to entries with room
    # allocate proportionally to slack = 1 - h_i
    if diff > 0:
        slack = 1.0 - h
        total_slack = slack.sum()
        if total_slack > 0:
            h += diff * slack / total_slack
    elif diff < 0:
        excess = h
        total_excess = excess.sum()
        if total_excess > 0:
            h += diff * excess / total_excess
    # final clip
    h = np.clip(h, 0.0, 1.0)
    # Iterate few times to ensure sum close enough

    return h

def compute_c5(h):
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    # start with uniform 0.5 and shuffle around
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    enforce_sum(h, n / 2.0, rng)
    return h

def random_binary_vector(n, rng):
    h = np.zeros(n)

    h[ones] = 1.0
    return h

CURRENT_TARGET = None
CURRENT_BETA = None
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET
def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")

def slsqp_optimize(h0, beta, max_iter=200):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta

    bounds = [(0.0, 1.0)] * n
    res = minimize(softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,

def anneal_descent(h0, target, rng, start_time, budget_s):
    n = h0.size
    betas = [5, 20, 80, 320, 1280, 5120]
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    for beta in betas:
        # step size scales with beta
        lr = 0.3 / np.sqrt(beta)
        # allocate iterations based on remaining time
        remaining = max(0.0, budget_s - (time.time() - start_time))
        # aim for up to 3000 iters per beta if time permits
        max_iters = int(min(3000, remaining * 1500 / len(betas)))
        max_iters = max(200, max_iters)
        for _ in range(max_iters):
            # time guard
            if time.time() - start_time > budget_s:
                break
            corr = np.correlate(h0, 1 - h0, mode="full")
            # softmax weights
            max_bc = np.max(beta * corr)
            exp_vals = np.exp(beta * corr - max_bc)
            w = exp_vals / exp_vals.sum()
            # gradient
            g = np.full(n, 1.0)   # sum_k w_k * 1 = 1
            # loop over shifts
            for s in range(-n + 1, n):
                wk = w[s + n - 1]
                if wk == 0.0:
                    continue
                if s >= 0:
                    g[:n - s] -= wk * h0[s:]
                    g[s:] -= wk * h0[:n - s]
                else:
                    kk = -s
                    g[kk:] -= wk * h0[:n - kk]
                    g[:n - kk] -= wk * h0[kk:]
            # update

            np.clip(h0, 0.0, 1.0, out=h0)
            # enforce sum (small correction)
            diff = target - h0.sum()
            if abs(diff) > 1e-12:
                if diff > 0:

                    if total > 0:
                        h0 += diff * slack / total

                    excess = h0
                    total = excess.sum()
                    if total > 0:
                        h0 += diff * excess / total
                np.clip(h0, 0.0, 1.0, out=h0)
            # occasional update of best
            cur_c = compute_c5(h0)
            if cur_c < best_c:
                best_c, best_h = cur_c, h0.copy()
        # after beta
        # re-evaluate best
    return best_h, best_c
