def smooth_max_c5(h, beta):
    corr = np.correlate(h, 1 - h, mode='full')
    bc = beta * corr
    m = np.max(bc)
    exp_bc = np.exp(bc - m)
    sum_exp = np.sum(exp_bc)
    f = (m + np.log(sum_exp)) / beta
    w = exp_bc / sum_exp  # shape L (2n-1)
    # compute gradient
    # w index offset 0..L-1 corresponds to lag ℓ = -(n-1) .. (n-1)
    # w as array.
    # compute conv_rev and conv_fwd:
    w_rev = w[::-1]

    conv_fwd = np.convolve(w, h, mode='full')

def project_to_feasible(v, target, tol=1e-12, max_iter=100):
    lo = -1.0  # lower bound for λ such that sum > target, start with low λ to increase sum.
    hi = 1.0   # high bound for λ such that sum < target.
    # Actually for λ too negative, v_i - λ becomes large, clip -> 1, sum = n. For λ too positive, clip -> 0, sum = 0.
    # So we need lo, hi sufficiently wide. Use lo = -max(v) - 1, hi = max(v) + 1.
    lo = -np.max(v) - 2.0
    hi = 1.0 - np.min(v) + 2.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
        if hi - lo < tol:
            break
    x = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    # After binary search, adjust slight residual due to tolerance:
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        # distribute diff to non-bound entries proportionally
        # find indices where 0 < x < 1
        free = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        # if no free, then adjust any.
        if free.size > 0:
            adjust = diff / free.size
            x[free] += adjust
            # clip again
            x = np.clip(x, 0.0, 1.0)
        else:
            # fallback: add diff uniformly then clip
            x += diff / v.size
            x = np.clip(x, 0.0, 1.0)
    return x

import time
import numpy as np

def compute_c5(h):
    """Exact discrete C₅ for step function h on [0,2] (dx = 2/n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def smooth_max_val(h, beta):
    """Smooth approximation of max_k ∑ h_i(1-h_{i+k}) (without gradient)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    bc = beta * corr
    m = np.max(bc)
    sum_exp = np.sum(np.exp(bc - m))
    return (m + np.log(sum_exp)) / beta


def smooth_max_c5(h, beta):
    """
    Smooth max value and its gradient.
    Returns (value, gradient) where gradient has shape h.shape.
    """
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    bc = beta * corr
    m = np.max(bc)
    exp_bc = np.exp(bc - m)
    sum_exp = np.sum(exp_bc)
    f = (m + np.log(sum_exp)) / beta
    w = exp_bc / sum_exp                     # weights, sum to 1

    # gradient of the smooth max:
    # grad = Σ w_k * (1 - h_{j+k} - h_{j-k})
    # compute the two convolutions efficiently.
    w_rev = w[::-1]
    conv_rev = np.convolve(w_rev, h, mode="full")
    conv_fwd = np.convolve(w, h, mode="full")
    offset = n - 1
    conv1 = conv_rev[2 * offset : 2 * offset + n]   # Σ w_k h_{j+k}
    conv2 = conv_fwd[offset : offset + n]          # Σ w_k h_{j-k}
    grad = 1.0 - (conv1 + conv2)
    return f, grad


def project_to_feasible(v, target, tol=1e-12, max_iter=80):
    """
    Euclidean projection of vector v onto the set
    {x | 0 ≤ x_i ≤ 1, Σ x_i = target}.
    Uses binary search on the Lagrange multiplier λ.
    """
    lo = -np.max(v) - 2.0
    hi = 1.0 - np.min(v) + 2.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
        if hi - lo < tol:
            break
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        cnt = np.count_nonzero(free)
        if cnt:
            x[free] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
        else:
            x += diff / x.size
            x = np.clip(x, 0.0, 1.0)
    return x


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    v = rng.random(n)
    return project_to_feasible(v, n / 2.0)


def gradient_descent_start(h0, beta_list, rng, time_budget, start_time):
    """
    Run a continuation of projected gradient descents over the
    list of β values. Returns the best (h, C5) pair found.
    """
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)

    for beta in beta_list:
        step = 0.5
        max_iters = 2000
        for it in range(max_iters):
            if time.time() - start_time > time_budget * 0.95:
                return best_h, best_c

            f, grad = smooth_max_c5(h, beta)

            # simple backtracking line search (max 5 tries)
            step_i = step
            for _ in range(5):
                cand = project_to_feasible(h - step_i * grad, target)
                f_new = smooth_max_val(cand, beta)
                if f_new < f:
                    break
                step_i *= 0.5
            else:
                # no improvement
                step_i = step * 0.5

            if f_new < f:
                h = cand
                f = f_new
                step = step_i * 1.1
            else:
                step *= 0.5

            # occasional jitter to escape shallow minima
            if (it + 1) % 120 == 0 and rng.random() < 0.35:
                h = project_to_feasible(h + rng.normal(scale=0.015, size=n), target)

            # track true objective
            cur_c = compute_c5(h)
            if cur_c < best_c:
                best_c, best_h = cur_c, h.copy()

    return best_h, best_c

    rng = np.random.default_rng(seed)
    start_time = time.time()
    # ------------------------------------------------------------------
    # 1. Prepare initial vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)

        h0 = random_feasible_vector(n, rng)

    # ------------------------------------------------------------------
    # 2. Global best tracking
    # ------------------------------------------------------------------
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ------------------------------------------------------------------
    # 3. Main optimisation loop with restarts
    # ------------------------------------------------------------------
    beta_schedule = [5, 20, 80, 200, 500, 1000]
    remaining = budget_s - (time.time() - start_time)

    # first run from the provided start
    best_h, best_c = gradient_descent_start(best_h, beta_schedule, rng, budget_s, start_time)

    # additional random restarts if time permits
    while time.time() - start_time < budget_s * 0.85:
        h_restart = random_feasible_vector(n, rng)

        if c_opt < best_c:
            best_h, best_c = h_opt.copy(), c_opt
        # break if very little time left
        if time.time() - start_time > budget_s * 0.9:
            break

    # ------------------------------------------------------------------
    # 4. Optional up‑sampling (increase resolution) if time remains
    # ------------------------------------------------------------------
    elapsed = time.time() - start_time
    if budget_s - elapsed > budget_s * 0.15:
        target_n = min(720, max(360, n * 2))
        if target_n != n:
            # linear interpolation to higher resolution
            old_x = (np.arange(n) + 0.5) * (2.0 / n)
            new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
            h_up = np.interp(new_x, old_x, best_h)
            h_up = project_to_feasible(h_up, target_n / 2.0)

            # optimisation on finer grid
            beta_up = [5, 20, 80, 200, 500, 1000]
            h_up_opt, c_up_opt = gradient_descent_start(

            if c_up_opt < best_c:

    # ------------------------------------------------------------------
    # 5. Final repair and return
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_feasible(best_h, n / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
