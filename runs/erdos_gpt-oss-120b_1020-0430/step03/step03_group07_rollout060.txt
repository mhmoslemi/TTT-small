def enforce_sum(vec, target, rng):
    """Adjust vec so that sum(vec) == target while keeping 0 ≤ vec ≤ 1."""

        if diff > 0:

    return vec


def random_feasible_vector(n, rng):
    """Random vector in [0,1]^n with prescribed sum n/2."""
    # start from 0.5 and perform many small transfers that preserve the sum
    vec = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)

        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        vec[i] -= eps
        vec[j] += eps
    vec = np.clip(vec, 0.0, 1.0)
    target = n / 2.0
    diff = target - vec.sum()
    if abs(diff) > 1e-12:
        vec = enforce_sum(vec, target, rng)
    return vec


# ----------------------------------------------------------------------
# Correlation utilities (full linear correlation of h with 1‑h)
# ----------------------------------------------------------------------
def correlate_full(h):
    """FFT‑based full linear correlation of h with (1‑h) reversed."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fh = np.fft.rfft(h, L)
    rev = (1.0 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)


# ----------------------------------------------------------------------
# Projection onto the capped simplex (0 ≤ x ≤ 1, Σx = target)
# ----------------------------------------------------------------------
def project_sum(v, target, max_iter=10):
    """Project v onto {0≤x≤1, Σx = target} using simple water‑filling."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        diff = v.sum() - target
        if abs(diff) < 1e-12:
            break
        # free variables are those not at a bound
        free = (v > 0.0 + 1e-12) & (v < 1.0 - 1e-12)
        if not np.any(free):
            break
        v[free] -= diff / free.sum()
        v = np.clip(v, 0.0, 1.0)
    # final tiny correction
    diff = v.sum() - target
    if abs(diff) > 1e-9:
        idx = np.argmax(v) if diff > 0 else np.argmin(v)
        v[idx] -= diff
        v = np.clip(v, 0.0, 1.0)
    return v


# ----------------------------------------------------------------------
# Gradient of the smoothed max   fβ(h) = (1/β) log Σ exp(β·corr_k)
# ----------------------------------------------------------------------
def compute_gradient(h, w):
    """Gradient of fβ given weights w_k (∝ exp(β·corr_k))."""
    n = h.size
    mid = n - 1
    grad = np.zeros(n)
    shifts = np.arange(-(n - 1), n, dtype=int)
    active = w > 1e-15
    for wk, s in zip(w[active], shifts[active]):
        if s >= 0:
            # first part: (1‑h[i‑s]) for i≥s
            grad[s:] += wk * (1.0 - h[: n - s])
            # second part: –h[i+s] for i < n‑s
            grad[: n - s] -= wk * h[s:]
        else:
            t = -s
            grad[: n - t] += wk * (1.0 - h[t:])
            grad[t:] -= wk * h[: n - t]
    return grad


# ----------------------------------------------------------------------
# Gradient descent with continuation on β
# ----------------------------------------------------------------------
def smooth_max_and_weights(h, beta):
    """Return (smooth_max, weights) for given β."""
    corr = correlate_full(h)
    # stable log‑sum‑exp
    max_beta_corr = (beta * corr).max()
    lse = logsumexp(beta * corr - max_beta_corr) + max_beta_corr
    smooth = lse / beta
    # weights = softmax(β·corr)
    exp_vals = np.exp(beta * corr - max_beta_corr)
    w = exp_vals / exp_vals.sum()
    return smooth, corr, w


def grad_descent_beta(h, beta, max_iters, lr, rng):
    target = h.size / 2.0
    for it in range(max_iters):
        # compute smooth max and weights
        _, _, w = smooth_max_and_weights(h, beta)
        # gradient step
        g = compute_gradient(h, w)
        step = lr * (0.98 ** it)          # slight decay
        h_new = h - step * g
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_sum(h_new, target)
        h = h_new
    return h


# ----------------------------------------------------------------------
# Simple hill‑climb (used for final polishing)
# ----------------------------------------------------------------------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=1500, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


# ----------------------------------------------------------------------
# SLSQP refinement (smooth surrogate of max, same as previous code)
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = correlate_full(x)
    bc = CURRENT_BETA * corr
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.96

    # ------------------------------------------------------------------
    # 1. Choose resolution (720 points gives finer freedom)
    # ------------------------------------------------------------------
    base_n = 720
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        # up‑sample if needed
        if h0.size != base_n:
            factor = base_n // h0.size
            h0 = np.repeat(h0, factor)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, base_n / 2.0, rng)
    else:
        h0 = random_feasible_vector(base_n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ------------------------------------------------------------------
    # 2. Gradient‑descent continuation on β
    # ------------------------------------------------------------------
    beta_schedule = [
        (5.0, 800, 0.25),
        (20.0, 800, 0.20),
        (80.0, 800, 0.15),
        (200.0, 800, 0.10),
        (800.0, 600, 0.07),
        (2000.0, 400, 0.05),
    ]

    for beta, iters, lr in beta_schedule:
        if time.time() >= deadline:
            break
        best_h = grad_descent_beta(best_h, beta, iters, lr, rng)
        c = compute_c5(best_h)
        if c < best_c:
            best_c, best_h = c, best_h.copy()

    # ------------------------------------------------------------------
    # 3. Local hill‑climb polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        h_hc, c_hc = hill_climb(best_h, rng, max_iters=2500, start_step=0.12)
        if c_hc < best_c:
            best_c, best_h = c_hc, h_hc.copy()

    # ------------------------------------------------------------------
    # 4. SLSQP refinement (same continuation as before)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        for beta in [5.0, 20.0, 80.0, 200.0, 800.0, 2000.0]:
            if time.time() >= deadline:
                break
            try:
                h_opt = slsqp_optimize(best_h, beta, max_iter=400)
                c_opt = compute_c5(h_opt)
                if c_opt < best_c:
                    best_c, best_h = c_opt, h_opt.copy()
            except Exception:
                continue

    # ------------------------------------------------------------------
    # 5. Final clean‑up (ensure feasibility)
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
