def perturb_h(h, rng, scale):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    while j == i:
        j = rng.integers(0, n)
    max_eps_pos = min(h[i], 1.0 - h[j])
    max_eps_neg = min(h[j], 1.0 - h[i])
    if max_eps_pos + max_eps_neg == 0: return h.copy()
    eps = (rng.random() * (max_eps_pos + max_eps_neg)) - max_eps_neg
    eps *= scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return np.clip(new_h, 0.0, 1.0)

def enforce_sum(h, target, rng):

    # assume target is reachable (target <= len(h) and >=0)
    diff = target - h.sum()
    n = h.size

CURRENT_TARGET = None
CURRENT_BETA = None

def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def _softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
```

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type':'eq','fun':_sum_eq_constraint}
    res = minimize(_softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter':max_iter,'ftol':1e-12,'disp':False})
    x = np.clip(res.x, 0.0, 1.0)
    # repair sum if needed
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            if abs(add) < 1e-12:
                continue
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x

import time
import math
import numpy as np
from scipy.optimize import minimize

# ---------- Global variables for the SLSQP surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    """Smooth log-sum-exp surrogate of max_k Σ_i x_i (1-x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


# ---------- Problem specific utilities ----------
def compute_c5(h):
    """Exact discrete C₅ for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng):
    """Adjust h so that Σh = target while 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    # greedy correction loop
    for _ in range(10000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-12:
            continue
        h[i] += add
        diff = target - h.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(1000):
            i = rng.integers(0, n)
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
            elif diff < 0 and h[i] > 0.0:
                add = max(diff, -h[i])
            else:
                continue
            h[i] += add
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h


def random_binary_vector(n, rng):
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h


def perturb_h(h, rng, scale=0.5):
    """Transfer a small amount between two coordinates (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    while j == i:
        j = rng.integers(0, n)
    max_pos = min(h[i], 1.0 - h[j])
    max_neg = min(h[j], 1.0 - h[i])
    if max_pos + max_neg == 0:
        return h.copy()
    eps = (rng.random() * (max_pos + max_neg)) - max_neg
    eps *= scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            if abs(add) < 1e-12:
                continue
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x


# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- initialise ----
    if "initial_h_values" in globals():

    best_h = h0.copy()

    # ---- Simulated annealing on the original grid ----
    T = 0.5                # initial temperature (controls step size)

    curr_h = best_h.copy()
    curr_c = best_c
    it = 0
    re_enforce = 2000

            curr_h, curr_c = new_h, new_c
