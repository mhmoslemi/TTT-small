def compute_corr_h_1mh_full(h):
    # using FFT
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv

def compute_autocorr_full(h):
    # compute convolution h * rev_h (where rev_h = h[::-1])
    n = h.shape[0]

    rev = h[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv

    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    conv = correlate_full(x)
    bc = CURRENT_BETA * conv
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=500):
    """Run SLSQP on the smooth surrogate for a given beta."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = opt.minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # repair possible tiny sum drift
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    main_deadline = start_time + budget_s * 0.9
    final_deadline = start_time + budget_s

    # ------------------------------------------------------------------
    # Initial feasible vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.size
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, n / 2.0, rng)
    else:
        n = 200
        h = random_feasible_vector(n, rng)

    best_h = h.copy()
    best_val = compute_c5(best_h)
    last_improve = start_time

    # ------------------------------------------------------------------
    # Parameters
    # ------------------------------------------------------------------
    beta_list = [10.0, 30.0, 100.0, 300.0, 1000.0, 3000.0]
    slsqp_iters = 500
    greedy_segment = 0.22  # seconds per greedy block
    restart_delay = 5.0    # seconds without improvement before a restart

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < main_deadline:
        # --- Greedy improvement on the current best ---
        deadline_g = time.time() + greedy_segment
        best_h, best_val = greedy_improve(best_h, rng, deadline_g, best_val)

        # --- SLSQP refinement (smooth surrogate) ---
        for beta in beta_list:
            if time.time() >= main_deadline:
                break
            try:
                h_opt = slsqp_optimize(best_h, beta, max_iter=slsqp_iters)
                c_opt = compute_c5(h_opt)
                if c_opt < best_val - 1e-12:
                    best_h, best_val = h_opt, c_opt
                    last_improve = time.time()
            except Exception:
                continue

        # --- Random restart if stuck for a while ---
        if time.time() - last_improve > restart_delay:
            h_restart = random_feasible_vector(n, rng)
            val_restart = compute_c5(h_restart)
            if val_restart < best_val - 1e-12:
                best_h, best_val = h_restart, val_restart
                last_improve = time.time()
            else:
                # give the restart a short greedy burn‑in
                deadline_r = time.time() + greedy_segment
                h_restart, val_restart = greedy_improve(h_restart, rng, deadline_r, val_restart)
                if val_restart < best_val - 1e-12:
                    best_h, best_val = h_restart, val_restart
                    last_improve = time.time()

    # ------------------------------------------------------------------
    # Final polishing (high‑beta SLSQP if time permits)
    # ------------------------------------------------------------------
    if time.time() + 1.5 < final_deadline:
        try:
            h_opt = slsqp_optimize(best_h, beta=8000.0, max_iter=800)
            c_opt = compute_c5(h_opt)
            if c_opt < best_val:
                best_h, best_val = h_opt, c_opt
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Clean‑up and return
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
