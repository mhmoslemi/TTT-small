def project_sum(x, target, tol=1e-10, max_iter=50):
    x = np.clip(x,0,1)
    # compute current sum:
    s = x.sum()
    if abs(s-target) <= tol:
        return x
    # we need to find delta such that sum(clipped(x+delta)) = target
    lo = -1.0
    hi = 1.0
    for _ in range(max_iter):
        mid = (lo+hi)/2.0
        y = np.clip(x+mid,0,1)
        s = y.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x+(lo+hi)/2.0,0,1)
    # after projection, we may still have small mismatch due to tolerance; adjust minimal diff by adding to a non-saturated element
    diff = target - y.sum()
    if abs(diff) > tol:
        # assign diff to first non-bound element if any else adjust by small epsilon
        idx = np.where((y > 0) & (y < 1))[0]
        if idx.size > 0:
            y[idx[0]] += diff
            y = np.clip(y,0,1)
        else:
            # all at bounds; cannot adjust; return as is
            pass
    return y

import time, math, numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h):
    """Exact discrete C5 for step function h (dx=2/n)."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)

def project_sum(x, target, tol=1e-10, max_iter=20):
    """Project x onto [0,1]^n with sum equal to target (bisection on uniform shift)."""
    x = np.clip(x, 0.0, 1.0)
    s = x.sum()
    if abs(s - target) <= tol:
        return x
    lo, hi = -1.0, 1.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny residual adjustment if needed
    diff = target - y.sum()
    if abs(diff) > tol:
        free = np.where((y > 0.0) & (y < 1.0))[0]
        if free.size:
            y[free[0]] += diff
            y = np.clip(y, 0.0, 1.0)
    return y

def random_feasible_vector(n, rng):
    """Uniform random point in [0,1]^n projected to sum=n/2."""
    v = rng.random(n)
    return project_sum(v, n / 2.0)

def tournament_select(pop, fitness, tsize, rng):
    """Select one individual by tournament."""
    idxs = rng.choice(len(pop), size=tsize, replace=False)
    best = idxs[0]
    best_f = fitness[best]
    for i in idxs[1:]:
        f = fitness[i]
        if f < best_f:
            best, best_f = i, f
    return pop[best]

# ---------- SLSQP surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # smooth max (log-sum-exp) scaled back
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on a smooth approximation of the max overlap."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {"type": "eq", "fun": sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    return project_sum(x, CURRENT_TARGET)

# ---------- main optimisation ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- problem size -----
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        base = np.array(initial_h_values, dtype=float)
        base = np.clip(base, 0.0, 1.0)
        base = project_sum(base, n / 2.0)
    else:
        n = 180
        base = random_feasible_vector(n, rng)

    target_sum = n / 2.0

    # ----- population initialisation -----
    pop_size = 120
    population = [base]
    while len(population) < pop_size:
        population.append(random_feasible_vector(n, rng))

    fitness = np.array([compute_c5(ind) for ind in population])
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c = fitness[best_idx]

    # ----- GA parameters -----
    ga_time = budget_s * 0.78
    sigma_start, sigma_end = 0.30, 0.02
    elite_frac = 0.05
    tournament_size = 5
    generation = 0

    # ----- main GA loop -----
    while time.time() - start < ga_time:
        fraction = (time.time() - start) / ga_time
        current_sigma = sigma_start * (1.0 - fraction) + sigma_end * fraction

        elite_cnt = max(2, int(pop_size * elite_frac))
        order = np.argsort(fitness)
        new_pop = [population[i].copy() for i in order[:elite_cnt]]

        while len(new_pop) < pop_size:
            p1 = tournament_select(population, fitness, tournament_size, rng)
            p2 = tournament_select(population, fitness, tournament_size, rng)

            mask = rng.random(n) < 0.5
            child = np.where(mask, p1, p2)

            child += rng.normal(scale=current_sigma, size=n)
            child = project_sum(child, target_sum)

            new_pop.append(child)

        population = new_pop
        fitness = np.array([compute_c5(ind) for ind in population])

        cur_best_idx = np.argmin(fitness)
        cur_best_c = fitness[cur_best_idx]
        if cur_best_c < best_c:
            best_c = cur_best_c
            best_h = population[cur_best_idx].copy()

        generation += 1

        # occasional SLSQP polishing of the current best
        if generation % 25 == 0 and time.time() - start < ga_time:
            try:
                refined = slsqp_optimize(best_h, beta=5000, max_iter=200)
                rc = compute_c5(refined)
                if rc < best_c:
                    best_c, best_h = rc, refined.copy()
            except Exception:
                pass

    # ----- final refinement phase -----
    remaining = budget_s - (time.time() - start)
    for beta in (2000, 8000, 20000, 50000):
        if time.time() - start > budget_s:
            break
        try:
            refined = slsqp_optimize(best_h, beta=beta, max_iter=250)
            rc = compute_c5(refined)
            if rc < best_c:
                best_c, best_h = rc, refined.copy()
        except Exception:
            continue

    # final safety clipping & sum enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum(best_h, target_sum)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
