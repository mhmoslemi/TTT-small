def de_optimize(n, rng, max_time, init_vec=None):

   popsize = min(150, max(20, n//2))  # for n=300, popsize=150

   pop = []

   if init_vec is not None:
        pop.append(init_vec)

   while len(pop) < popsize:
        pop.append(random_feasible_vector(n, rng))

   pop = np.array(pop)
   scores = np.array([compute_c5(v) for v in pop])
   best_idx = scores.argmin()
   best_h = pop[best_idx].copy()
   best_score = float(scores[best_idx])

   F = 0.8
   CR = 0.7

   start = time.time()
   while time.time() - start < max_time:
        # one generation
        for i in range(popsize):

            idxs = list(range(popsize))
            idxs.remove(i)

def simulated_annealing(h_start, rng, max_time, init_temp=0.01, cooling=0.9995):

   h = h_start.copy()
   best_h = h.copy()
   best_c = compute_c5(h)
   cur_c = best_c
   T = init_temp
   start = time.time()
   n = h.size
   while time.time() - start < max_time:
        # propose move
        i = rng.integers(0,n)
        j = rng.integers(0,n)
        if i == j: continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        # sample epsilon uniformly between -max_eps and max_eps? Actually sign negative means moving opposite direction.
        eps = (rng.random() - 0.5) * 2.0 * max_eps
        # make move
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        # ensure within bounds (should be)
        c_new = compute_c5(h_new)
        delta = c_new - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / (T+1e-12)):
            h = h_new
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        # update temperature
        T *= cooling
        # optionally reset T if too small or after certain steps? 
   return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Utility functions ----------
def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_to_feasible(v, target):
    """Projection of v onto {0≤h≤1, Σh = target} via bisection on a shift."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_to_feasible(v, n / 2.0)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=3000, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


# ---------- SLSQP with smooth max ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def de_optimize(n, rng, max_time, init_vec=None):
    target = n / 2.0
    popsize = min(150, max(20, n // 2))
    pop = []
    if init_vec is not None:
        pop.append(project_to_feasible(init_vec, target))
    while len(pop) < popsize:
        pop.append(random_feasible_vector(n, rng))
    pop = np.array(pop, dtype=float)
    scores = np.array([compute_c5(v) for v in pop])
    best_idx = scores.argmin()
    best_h = pop[best_idx].copy()
    best_score = float(scores[best_idx])

    F = 0.8
    CR = 0.7
    start = time.time()
    while time.time() - start < max_time:
        for i in range(popsize):
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, size=3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(0, n)] = True
            trial = np.where(cross, mutant, pop[i])

            c = compute_c5(trial)
            if c < scores[i]:

                scores[i] = c
                if c < best_score:
                    best_score = c
                    best_h = trial.copy()
        if time.time() - start >= max_time:
            break
    return best_h, best_score
