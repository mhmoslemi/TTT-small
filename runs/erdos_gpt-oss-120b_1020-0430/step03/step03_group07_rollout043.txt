def gradient_descent_h(h, rng, max_iters=5000, init_step=0.05, step_decay=0.99):
   best_h = h.copy()
   best_c = compute_c5(h)
   step = init_step
   for it in range(max_iters):
       # compute autocorrelation
       corr = corr_full_hh(best_h)
       min_val = corr.min()
       # find shift indices where min_val occurs
       # shift index mapping
       # conv indices 0..2n-2 correspond to shift = -(n-1) .. (n-1)
       shifts = np.where(np.isclose(corr, min_val, atol=1e-12))[0]
       ks = shifts - (len(best_h) - 1)
       # choose first k
       k_star = ks[0]
       # compute neighbor sum s = shift_forward + shift_backward for k_star
       n = len(best_h)
       s = np.zeros(n)
       # forward
       if k_star >= 0:
           fwd = np.concatenate([best_h[k_star:], np.zeros(k_star)])
           back = np.concatenate([np.zeros(k_star), best_h[:n - k_star]])
       else:
           # negative shift: forward/backward swapped
           k = -k_star
           fwd = np.concatenate([np.zeros(k), best_h[:n - k]])   # because shift forward by negative k => shift backward
           back = np.concatenate([best_h[k:], np.zeros(k)])
       s = fwd + back
       # identify donor/receiver
       donor_candidates = np.where(best_h > 1e-8)[0]
       receiver_candidates = np.where(best_h < 1 - 1e-8)[0]
       if donor_candidates.size == 0 or receiver_candidates.size == 0:
           break
       donor = donor_candidates[np.argmin(s[donor_candidates])]
       receiver = receiver_candidates[np.argmax(s[receiver_candidates])]
       eps = min(step, best_h[donor], 1 - best_h[receiver])
       if eps < 1e-10:
           step *= step_decay
           continue
       new_h = best_h.copy()
       new_h[donor] -= eps
       new_h[receiver] += eps
       new_c = compute_c5(new_h)
       if new_c < best_c - 1e-12:
           best_h, best_c = new_h, new_c
       else:
           # maybe try random donor/receiver if fails
           # random attempt
           for _ in range(5):
               i = rng.integers(0, n)

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)

            best_h = new_h.copy()
            h = new_h

    return best_h, best_c

        eps = (rng.random() - 0.5) * 2 * max_eps * 0.5
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0, 1)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h = enforce_sum(h, target, rng)
    return h

# ------------------------------------------------------------
# Correlation utilities
# ------------------------------------------------------------
def correlate_full_h1mh(h):
    """Full correlation of h with (1‑h) (zero‑padded)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv

def corr_full_hh(h):
    """Full autocorrelation of h (zero‑padded)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    fr = np.fft.rfft(h[::-1], L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv

def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    return correlate_full_h1mh(h).max() * (2.0 / h.size)

# ------------------------------------------------------------
# Simple perturb / hill‑climb
# ------------------------------------------------------------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_int = max(1, max_iters // 10)
    for it in range(max_iters):
        nh = perturb_h(h, rng, scale=step)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay_int == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c

# ------------------------------------------------------------
# Gradient‑descent on the worst shift
# ------------------------------------------------------------
def gradient_descent(h_start, rng, max_iters=2500, init_step=0.1, decay=0.995):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = init_step
    n = h.size
    for it in range(max_iters):
        if step < 1e-6:
            break
        auto = corr_full_hh(h)[n - 1:]          # k = 0 .. n‑1
        min_val = auto.min()
        ks = np.where(auto == min_val)[0]
        k = int(ks[0])                         # worst shift
        if k == 0:
            s = 2 * h
        else:
            fwd = np.concatenate([h[k:], np.zeros(k)])
            back = np.concatenate([np.zeros(k), h[: n - k]])
            s = fwd + back
        donor_mask = h > 1e-12
        recv_mask = h < 1 - 1e-12
        if not donor_mask.any() or not recv_mask.any():
            break
        donors = np.where(donor_mask)[0]
        recvs = np.where(recv_mask)[0]
        donor = donors[np.argmin(s[donors])]
        recv = recvs[np.argmax(s[recvs])]
        eps = min(step, h[donor], 1 - h[recv])
        if eps <= 0:
            step *= decay
            continue
        nh = h.copy()
        nh[donor] -= eps
        nh[recv] += eps
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        else:
            # occasional random trial
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                eps2 = min(step, h[i], 1 - h[j])
                if eps2 > 0:
                    nh2 = h.copy()
                    nh2[i] -= eps2
                    nh2[j] += eps2
                    c2 = compute_c5(nh2)
                    if c2 < best_c:
                        best_c, best_h, h = c2, nh2.copy(), nh2
        step *= decay
    return best_h, best_c

# ------------------------------------------------------------
# Smooth surrogate + SLSQP
# ------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = correlate_full_h1mh(x)
    bc = CURRENT_BETA * corr
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # fix tiny drift in the sum
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(1000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    main_deadline = start_time + 0.85 * budget_s

    # ---- initial vector ------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 360
        h0 = np.full(n, 0.5)
        h0 = enforce_sum(h0, n / 2.0, rng)

    best_h, best_c = h0.copy(), compute_c5(h0)

    # ---- diverse starting pool -----------------------------------------
    cand = [h0]
    xs = np.arange(n) / n
    for f in (1, 2, 3):
        for func in (np.sin, np.cos):
            h = 0.5 + 0.5 * func(np.pi * f * xs)
            cand.append(enforce_sum(np.clip(h, 0, 1), n / 2.0, rng))
    # binary step
    step = np.zeros(n)
    step[: n // 2] = 1.0
    cand.append(enforce_sum(step, n / 2.0, rng))
    # alternating
    alt = np.tile([0.0, 1.0], n // 2)
    if alt.size < n:
        alt = np.append(alt, 0.0)
    cand.append(enforce_sum(alt, n / 2.0, rng))
    # random few
    for _ in range(4):
        cand.append(random_feasible_vector(n, rng))

    for h in cand:
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # ---- main optimisation loop -----------------------------------------
    while time.time() < main_deadline:
        # start point
        if rng.random() < 0.7:
            start = perturb_h(best_h, rng, scale=0.35)
        else:
            start = random_feasible_vector(n, rng)

        # gradient descent focused on the worst shift
        gd_h, gd_c = gradient_descent(start, rng, max_iters=2500,
                                      init_step=0.12, decay=0.996)
        if gd_c < best_c:
            best_c, best_h = gd_c, gd_h.copy()

        # short hill‑climb refinement
        hc_h, hc_c = hill_climb(best_h, rng, max_iters=1500, start_step=0.25)
        if hc_c < best_c:
            best_c, best_h = hc_c, hc_h.copy()

        # SLSQP ladder (smooth surrogate)
        for beta in (20, 80, 200, 800, 2000):
            if time.time() >= main_deadline:
                break
            try:
                opt_h = slsqp_optimize(best_h, beta, max_iter=250)
                opt_c = compute_c5(opt_h)
                if opt_c < best_c:
                    best_c, best_h = opt_c, opt_h.copy()
            except Exception:
                continue

    # ---- optional up‑sampling -------------------------------------------
    remaining = start_time + budget_s - time.time()
    if remaining > 30.0 and best_h.size * 2 <= 720:
        up_h = np.repeat(best_h, 2)
        up_n = up_h.size
        up_h = enforce_sum(up_h, up_n / 2.0, rng)

        gd_h, gd_c = gradient_descent(up_h, rng, max_iters=3000,
                                      init_step=0.09, decay=0.998)
        if gd_c < best_c:
            best_c, best_h = gd_c, gd_h.copy()

        for beta in (200, 800, 2000, 5000):
            if time.time() >= start_time + budget_s:
                break
            try:
                opt_h = slsqp_optimize(best_h, beta, max_iter=260)
                opt_c = compute_c5(opt_h)
                if opt_c < best_c:
                    best_c, best_h = opt_c, opt_h.copy()
            except Exception:
                continue

    # ---- final clean‑up -------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
