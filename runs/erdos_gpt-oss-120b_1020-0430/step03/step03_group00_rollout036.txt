- compute_worst_shift_and_gradient(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr)
    max_corr = corr[idx]
    k = idx - (n - 1)
    # compute gradient as earlier
    grad = np.ones(n)
    if k >= 0:
        grad[:n - k] -= h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[:n - kk] -= h[kk:]
        grad[kk:] -= h[:n - kk]
    return k, max_corr, grad

- greedy_transfer(h, grad):
    pos = np.where(grad > 1e-12)[0]
    neg = np.where(grad < -1e-12)[0]
    if len(pos)==0 or len(neg)==0:
        return h
    # get sort indices
    pos_sorted = pos[np.argsort(-grad[pos])]  # descending grad
    neg_sorted = neg[np.argsort(grad[neg])]   # ascending (most negative)
    new_h = h.copy()
    i_pos = 0
    i_neg = 0
    while i_pos < len(pos_sorted) and i_neg < len(neg_sorted):
        i = pos_sorted[i_pos]
        j = neg_sorted[i_neg]
        # amount we can transfer from i to j
        max_eps = min(new_h[i], 1.0 - new_h[j])
        if max_eps <= 0:
            # either saturated, move pointer
            if new_h[i] <= 1e-12:
                i_pos += 1
                continue
            if new_h[j] >= 1.0 - 1e-12:
                i_neg += 1
                continue
        # we can transfer some amount; maybe move the full max_eps
        # But maybe we could limit to small fraction to avoid overshoot.
        # Use factor = 0.5? But we can do full move.
        eps = max_eps
        new_h[i] -= eps

        if new_h[i] <= 1e-12:

def greedy_transfer(h, grad):
    pos = np.where(grad > 1e-12)[0]
    neg = np.where(grad < -1e-12)[0]
    if pos.size == 0 or neg.size == 0:
        return h.copy()
    # sort
    pos_order = pos[np.argsort(-grad[pos])]
    neg_order = neg[np.argsort(grad[neg])]
    new_h = h.copy()
    i = 0
    j = 0
    while i < len(pos_order) and j < len(neg_order):
        ii = pos_order[i]
        jj = neg_order[j]
        # max transferable
        max_eps = min(new_h[ii], 1.0 - new_h[jj])
        if max_eps <= 0:
            # skip saturated
            if new_h[ii] <= 1e-12:
                i += 1
                continue
            if new_h[jj] >= 1.0 - 1e-12:
                j += 1
                continue
        # transfer full amount
        eps = max_eps
        new_h[ii] -= eps
        new_h[jj] += eps
        # update pointers
        if new_h[ii] <= 1e-12:
            i += 1
        if new_h[jj] >= 1.0 - 1e-12:
            j += 1
    # slight sum error fix
    # maybe adjust for tiny diff
    diff = new_h.size/2.0 - new_h.sum()
    if abs(diff) > 1e-12:
        # adjust a random index with capacity
        idx = np.random.randint(0, new_h.size)
        if diff > 0:
            new_h[idx] = min(1.0, new_h[idx] + diff)
        else:
            new_h[idx] = max(0.0, new_h[idx] + diff)
    return new_h
