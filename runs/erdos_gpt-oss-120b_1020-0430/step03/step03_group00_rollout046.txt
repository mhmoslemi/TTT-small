def hill_climb_targeted_improved(h_start, rng, max_iters, init_step, target_sum):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(best_h)
    step = init_step
    no_improve = 0
    for it in range(max_iters):
        # compute worst shift
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr)
        max_corr = corr[idx]
        c_current = max_corr * (2.0/h.size)
        # compute gradient
        n = h.size
        k = idx - (n-1)
        grad = np.ones(n)
        if k >= 0:
            grad[:n-k] -= h[k:]
            grad[k:] -= h[:n-k]
        else:
            kk = -k
            grad[:n-kk] -= h[kk:]
            grad[kk:] -= h[:n-kk]
        # propose new
        new_h = perturb_targeted(h, grad, k, rng, scale=step)
        c_new = compute_c5(new_h)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = new_h.copy()
            no_improve = 0
        if c_new < c_current:
            h = new_h
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= max(10, max_iters//20):
            step = max(step * 0.5, 1e-5)
            no_improve = 0
    return best_h, best_c

def simulated_annealing(init_h, rng, time_limit):
    start = time.time()
    h = init_h.copy()
    best_h = h.copy()
    best_c = compute_c5(best_h)
    # SA parameters
    T = 5.0
    T_min = 1e-3
    alpha = 0.995  # cooling rate per iteration
    iter = 0
    while time.time() - start < time_limit and T > T_min:
        # generate multiple candidates per temperature?
        # For simplicity, one
        # choose move type: targeted gradient or random
        if rng.random() < 0.7:
            # make gradient-targeted move with scale relative to T (higher T => larger epsilon)
            scale = min(0.5, max(0.001, T/5))
            # compute worst shift to get gradient and k
            corr = np.correlate(h, 1 - h, mode='full')
            idx = np.argmax(corr)
            k = idx - (h.size - 1)
            grad = np.ones(h.size)
            n = h.size
            if k >= 0:
                grad[:n - k] -= h[k:]
                grad[k:] -= h[:n - k]
            else:
                kk = -k
                grad[:n - kk] -= h[kk:]
                grad[kk:] -= h[:n - kk]
            new_h = perturb_targeted(h, grad, k, rng, scale=scale)
        else:
            new_h = perturb_random(h, rng, scale=0.2)

                best_h = new_h.copy()
        # cooling schedule
        T *= alpha
        iter += 1
        # occasionally reset T to maintain ability to explore?

            T = max(T, 1.0)  # ensure not too low

def slsqp_refine(h0, beta_list, rng, max_iter_per_beta=800):
    n = h0.size
    target = n/2.0
    bounds = [(0.0, 1.0)]*n
    def sum_eq(x):
        return np.sum(x) - target
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    for beta in beta_list:
        # define objective
        def soft_obj(x):
            corr = np.correlate(x, 1 - x, mode='full')
            bc = beta * corr
            m = np.max(bc)
            # log-sum-exp
            return (m + np.log(np.sum(np.exp(bc - m)))) / beta
        cons = {'type': 'eq', 'fun': sum_eq}
        try:
            res = minimize(soft_obj, best_h, method='SLSQP', bounds=bounds,
                constraints=cons, options={'maxiter': max_iter_per_beta, 'ftol':1e-12, 'disp':False})
        except Exception:
            continue
        x = np.clip(res.x,0.0,1.0)
        x = enforce_sum(x, target, rng)
        c = compute_c5(x)
        if c < best_c:
            best_c = c
            best_h = x
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utilities ----------
def enforce_sum(h, target, rng):
    """Clip to [0,1] and adjust entries so that sum equals target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # simple iterative redistribution
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i])
            add *= rng.random()
            h[i] += add
            diff -= add
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i])
            sub *= rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)

def perturb_random(h, rng, scale=0.2):
    """Transfer a small amount between two coordinates (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def perturb_targeted(h, grad, shift, rng, scale=0.2):
    """Move mass from positive‑gradient indices to negative‑gradient ones."""
    pos = np.where((grad > 1e-9) & (h > 1e-12))[0]
    neg = np.where((grad < -1e-9) & (h < 1.0 - 1e-12))[0]
    if pos.size == 0 or neg.size == 0:
        return perturb_random(h, rng, scale)
    i = rng.choice(pos)
    j = rng.choice(neg)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return perturb_random(h, rng, scale)
    eps = max_eps * scale * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def compute_c5(h):
    """Discrete C5 value for a vector h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

# ---------- hill climbing ----------
def hill_climb_targeted_improved(h_start, rng, max_iters, init_step, target_sum):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(best_h)
    step = init_step
    no_imp = 0
    n = h.size
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr)
        max_corr = corr[idx]
        c_cur = max_corr * (2.0 / n)
        k = idx - (n - 1)

        # gradient of the worst shift
        grad = np.ones(n)
        if k >= 0:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
        else:
            kk = -k
            grad[: n - kk] -= h[kk:]
            grad[kk:] -= h[: n - kk]

        new_h = perturb_targeted(h, grad, k, rng, scale=step)
        c_new = compute_c5(new_h)

        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = new_h.copy()
            no_imp = 0

        if c_new < c_cur:
            h = new_h
            no_imp = 0
        else:
            no_imp += 1

        if no_imp >= max(10, max_iters // 20):
            step = max(step * 0.5, 1e-5)
            no_imp = 0

    return best_h, best_c

# ---------- simulated annealing ----------
def simulated_annealing(init_h, rng, time_limit):
    start = time.time()
    h = init_h.copy()
    best_h = h.copy()
    best_c = compute_c5(best_h)

    T = 5.0
    T_min = 1e-3
    alpha = 0.995
    iter_cnt = 0

    n = h.size
    while time.time() - start < time_limit and T > T_min:
        # decide move type
        if rng.random() < 0.7:
            # gradient‑guided move
            corr = np.correlate(h, 1.0 - h, mode="full")
            idx = np.argmax(corr)
            k = idx - (n - 1)
            grad = np.ones(n)
            if k >= 0:
                grad[: n - k] -= h[k:]
                grad[k:] -= h[: n - k]
            else:
                kk = -k
                grad[: n - kk] -= h[kk:]
                grad[kk:] -= h[: n - kk]
            scale = min(0.5, max(1e-4, T / 5.0))
            new_h = perturb_targeted(h, grad, k, rng, scale=scale)
        else:
            new_h = perturb_random(h, rng, scale=0.2)

        c_new = compute_c5(new_h)
        delta = c_new - best_c
        if delta < 0 or rng.random() < np.exp(-delta / T):
            h = new_h
            if c_new < best_c:
                best_c = c_new
                best_h = new_h.copy()

        T *= alpha
        iter_cnt += 1
        if iter_cnt % 2000 == 0 and T < 0.5:
            T = max(T, 0.5)  # keep some exploration power

    return best_h, best_c

# ---------- SLSQP refinement ----------
def slsqp_refine(h0, beta_list, rng, max_iter_per_beta=800):
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def sum_eq(x):
        return np.sum(x) - target

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    for beta in beta_list:
        def soft_obj(x):
            corr = np.correlate(x, 1.0 - x, mode="full")
            bc = beta * corr
            m = np.max(bc)
            # log‑sum‑exp for smooth max
            return (m + np.log(np.sum(np.exp(bc - m)))) / beta

        cons = {"type": "eq", "fun": sum_eq}
        try:
            res = minimize(
                soft_obj,
                best_h,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": max_iter_per_beta, "ftol": 1e-12, "disp": False},
            )
        except Exception:
            continue
        x = np.clip(res.x, 0.0, 1.0)
        x = enforce_sum(x, target, rng)
        c = compute_c5(x)
        if c < best_c:
            best_c = c
            best_h = x
    return best_h, best_c

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # initial vector
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = random_feasible_vector(180, rng)

    # up‑sample to double resolution
    n0 = h0.size
    n = 2 * n0
    xs0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.interp(xs, xs0, h0)
    h = enforce_sum(np.clip(h, 0.0, 1.0), n / 2.0, rng)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # Phase 1: targeted hill climbing (≈30% of budget)
    while time.time() - start_time < budget_s * 0.30:
        start_vec = perturb_random(best_h, rng, scale=0.3)
        h_loc, c_loc = hill_climb_targeted_improved(
            start_vec, rng, max_iters=2500, init_step=0.5, target_sum=n / 2.0
        )
        if c_loc < best_c:
            best_h, best_c = h_loc, c_loc

    # Phase 2: simulated annealing (≈40% of budget)
    while time.time() - start_time < budget_s * 0.70:
        remaining = budget_s - (time.time() - start_time)
        # give each SA run up to 25 s, but not more than half the remaining time
        sa_time = min(25.0, remaining * 0.5)
        h_sa, c_sa = simulated_annealing(best_h, rng, time_limit=sa_time)
        if c_sa < best_c:
            best_h, best_c = h_sa, c_sa

    # Phase 3: SLSQP refinement (≈20% of budget)
    while time.time() - start_time < budget_s * 0.95:
        beta_list = [5, 20, 80, 200, 800, 2000, 5000]
        h_ref, c_ref = slsqp_refine(best_h, beta_list, rng, max_iter_per_beta=800)
        if c_ref < best_c:
            best_h, best_c = h_ref, c_ref
        else:
            break  # no improvement, exit early

    # Final micro‑search (last few percent)
    while time.time() - start_time < budget_s * 0.99:
        new_h = perturb_random(best_h, rng, scale=0.02)
        c_new = compute_c5(new_h)
        if c_new < best_c:
            best_h, best_c = new_h, c_new

    # Clean‑up and final evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
