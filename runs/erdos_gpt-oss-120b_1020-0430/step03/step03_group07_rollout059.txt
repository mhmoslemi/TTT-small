import time
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# Basic feasibility helpers

def enforce_sum(h, target, rng):
    """Adjust h so that Σh == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h = enforce_sum(h, target, rng)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


# ------------------------------------------------------------
# Correlation utilities
# ------------------------------------------------------------
def correlate_full(h):
    """Full linear correlation of h with (1-h) using FFT (zero‑padding)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)


# ------------------------------------------------------------
# SLSQP surrogate optimizer (unchanged)
# ------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Log‑sum‑exp smoothing of max_k Σ_i x_i (1‑x_{i+k})."""
    conv = correlate_full(x)
    bc = CURRENT_BETA * conv
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ------------------------------------------------------------
# Hill‑climb (simple greedy)
# ------------------------------------------------------------
def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


# ------------------------------------------------------------
# Subgradient descent focusing on the current worst shift
# ------------------------------------------------------------
def subgradient_descent(h0, rng, deadline, max_iters=5000,
                        init_step=0.1, decay=0.99, top_frac=0.02):
    """Greedy subgradient moves that reduce the maximal overlap."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = init_step
    n = h.size
    top_k = max(1, int(top_frac * n))
    idxs = np.arange(n)

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = correlate_full(h) * (2.0 / n)
        k_idx = int(np.argmax(conv))
        shift = k_idx - (n - 1)

        # compute w_i = sum of neighbours involved in the worst shift
        if shift >= 0:
            mask1 = idxs + shift < n
            mask2 = idxs - shift >= 0
            w = np.zeros(n)
            w[mask1] += h[idxs[mask1] + shift]
            w[mask2] += h[idxs[mask2] - shift]
        else:
            s = -shift
            mask1 = idxs - s >= 0
            mask2 = idxs + s < n
            w = np.zeros(n)
            w[mask1] += h[idxs[mask1] - s]
            w[mask2] += h[idxs[mask2] + s]

        # choose source from high‑w and sink from low‑w
        top_inds = np.argpartition(-w, top_k - 1)[:top_k]
        low_inds = np.argpartition(w, top_k - 1)[:top_k]
        i_high = rng.choice(top_inds)
        i_low = rng.choice(low_inds)

        eps = step
        eps = min(eps, h[i_high])
        eps = min(eps, 1.0 - h[i_low])
        if eps > 0:
            h[i_high] -= eps
            h[i_low] += eps
        else:
            h = perturb_h(h, rng, scale=step)

        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if (it + 1) % 200 == 0:
            step = max(step * decay, 1e-6)
    return best_h, best_c


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    main_deadline = start_time + budget_s * 0.88  # reserve some slack

    # ---- initial vector -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 360
        h0 = np.full(n, 0.5)
        h0 = enforce_sum(h0, n / 2.0, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # a few random starts
    for _ in range(3):
        cand = random_feasible_vector(n, rng)
        c = compute_c5(cand)
        if c < best_c:
            best_c, best_h = c, cand.copy()

    # ---- iterative search ------------------------------------------------
    while time.time() < main_deadline:
        # pick start point
        if rng.random() < 0.6:
            start = best_h.copy()
            # small jitter to diversify
            start = perturb_h(start, rng, scale=0.05)
        else:
            start = random_feasible_vector(n, rng)

        # subgradient descent
        sub_h, sub_c = subgradient_descent(
            start, rng, main_deadline, max_iters=2500,
            init_step=0.12, decay=0.995, top_frac=0.025
        )
        if sub_c < best_c:
            best_c, best_h = sub_c, sub_h.copy()

        # hill‑climb refinement
        hc_h, hc_c = hill_climb(best_h, rng, max_iters=1800, start_step=0.3)
        if hc_c < best_c:
            best_c, best_h = hc_c, hc_h.copy()

        # occasional SLSQP polishing (increasing smoothness)
        for beta in (200, 800, 2000):
            if time.time() >= main_deadline:
                break
            try:
                sl_h = slsqp_optimize(best_h, beta, max_iter=250)
                sl_c = compute_c5(sl_h)
                if sl_c < best_c:
                    best_c, best_h = sl_c, sl_h.copy()
            except Exception:
                continue

    # ---- optional up‑sampling for finer resolution -----------------------
    up_deadline = start_time + budget_s * 0.985
    if time.time() < up_deadline and best_h.size * 2 <= 720:
        up_factor = 2
        h_up = np.repeat(best_h, up_factor)
        n_up = h_up.size
        h_up = enforce_sum(h_up, n_up / 2.0, rng)

        # refine on finer grid
        sub_h, sub_c = subgradient_descent(
            h_up, rng, up_deadline, max_iters=4000,
            init_step=0.09, decay=0.997, top_frac=0.02
        )
        if sub_c < best_c:
            best_c, best_h = sub_c, sub_h.copy()

        hc_h, hc_c = hill_climb(best_h, rng, max_iters=2500, start_step=0.2)
        if hc_c < best_c:
            best_c, best_h = hc_c, hc_h.copy()

        for beta in (200, 800, 2000, 5000):
            if time.time() >= start_time + budget_s:
                break
            try:
                sl_h = slsqp_optimize(best_h, beta, max_iter=200)
                sl_c = compute_c5(sl_h)
                if sl_c < best_c:
                    best_c, best_h = sl_c, sl_h.copy()
            except Exception:
                continue

    # ---- final clean‑up -------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
