def reduce_max_lag(h, rng, max_iters=2000, init_step=0.1):
    best = h.copy()
    best_c = compute_c5(best)
    n = len(h)
    for it in range(max_iters):
        corr = np.correlate(best, 1-best, mode='full')
        k = np.argmax(corr)  # index in 0..2n-2
        lag = k - (n-1)  # shift (negative allowed). Since h i + lag must be in range
        if lag >= 0:
            # i from 0 to n-1-lag
            i_max = n - lag - 1
            # choose i where best[i] > best[i+lag]
            # sample random i satisfying condition or fallback
            idx_candidates = np.where(best[:i_max+1] > best[lag:lag+i_max+1] + 1e-12)[0]
        else:
            # lag negative => shift left; i from -lag to n-1
            lag_abs = -lag
            i_start = lag_abs
            # choose i where best[i] > best[i+lag] (i+lag = i - lag_abs)
            idx_candidates = np.where(best[i_start:] > best[:n-lag_abs] + 1e-12)[0] + i_start
        if idx_candidates.size == 0:
            # fallback random pair
            i = rng.integers(0, n)
            j = rng.integers(0, n)
        else:
            i = rng.choice(idx_candidates)
            # j is i+lag (the target)
            j = i + lag
        # ensure i != j
        # transfer epsilon
        max_up = 1.0 - best[j]  # we will increase h[j]
        max_down = best[i]       # we will decrease h[i]
        epsilon = init_step * rng.random() * min(max_up, max_down)
        if epsilon <= 0:
            continue
        new = best.copy()
        new[i] -= epsilon
        new[j] += epsilon
        c = compute_c5(new)
        if c < best_c:
            best = new
            best_c = c
            # optionally increase step? not.
    return best, best_c

def perturb_pairwise(h, rng, scale=0.2):
    n = h.shape[0]
    i = rng.integers(n)
    j = rng.integers(n)
    if i == j:
        return h.copy()

    eps = scale * rng.random() * max_eps
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps

            max_i = n - lag - 1
            if max_i < 0:
                # lag too large, skip
                continue
            # choose i where h[i] > h[i+lag]
            indices = np.where(h[:max_i+1] > h[lag:lag+max_i+1] + 1e-12)[0]
        else:
            lag_abs = -lag
            # i such that i >= lag_abs and i-lag_abs in range
            # choose i where h[i] > h[i+lag] => i+lag = i-lag_abs
            # So compare h[i] > h[i-lag_abs]
            indices = np.where(h[lag_abs:] > h[:n-lag_abs] + 1e-12)[0] + lag_abs
        if indices.size == 0:
            # fallback to random pair
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            # ensure we can transfer some mass
        else:
            i = rng.choice(indices)
            j = i - lag  # since lag may be negative
        # ensure i != j
        if i == j:
            continue
        # transfer epsilon from i to j (decrease i, increase j)
        max_up = 1.0 - h[j]
        max_down = h[i]
        max_eps = min(max_up, max_down)
        if max_eps <= 0:
            continue
        eps = step * rng.random() * max_eps
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c = compute_c5(new_h)
        if c < best_c:
            h = new_h
            best_c = c
            # maybe reset step larger? keep
        # decay step periodically
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return h, best_c

import time
import numpy as np
import scipy.optimize as opt

# ---------- helper functions ----------
def compute_c5(h):
    """Exact discrete C5 for step function h on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_to_sum_box(v, target, max_iter=60):
    """
    Project vector v onto the box [0,1] with the linear equality sum(v)=target.
    Uses bisection on the shift tau in v-tau, then clipping.
    """
    lo, hi = -1.0, 1.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) * 0.5
    return np.clip(v - tau, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum=n/2."""
    v = rng.random(n)
    return project_to_sum_box(v, n / 2.0)


def perturb_pairwise(h, rng, scale=0.2):
    """Transfer a small amount from one coordinate to another (preserves sum)."""
    n = h.shape[0]
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_up = 1.0 - h[j]
    max_down = h[i]
    max_eps = min(max_up, max_down)
    if max_eps <= 0.0:
        return h.copy()
    eps = scale * rng.random() * max_eps
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def pairwise_hill_climb(h_start, rng, max_iters=2000, step=0.1):
    """Simple hill climbing using random pairwise transfers."""
    h = h_start.copy()
    best_c = compute_c5(h)
    n = h.size
    cur_step = step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        max_up = 1.0 - h[j]
        max_down = h[i]
        max_eps = min(max_up, max_down)
        if max_eps <= 0.0:
            continue
        eps = cur_step * rng.random() * max_eps
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c = compute_c5(new_h)
        if c < best_c:
            h, best_c = new_h, c
        if (it + 1) % decay_interval == 0:
            cur_step = max(cur_step * 0.8, 1e-4)
    return h, best_c


def reduce_max_lag_hill(h_start, rng, max_iters=3000, init_step=0.05):
    """
    Target the lag that currently attains the maximum overlap
    and move mass from earlier to later positions (or vice‑versa)
    to decrease that specific term.
    """
    h = h_start.copy()
    best_c = compute_c5(h)
    n = h.size
    step = init_step
    decay_interval = max(1, max_iters // 10)

    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode="full")
        k = np.argmax(corr)
        lag = k - (n - 1)          # shift = lag (can be negative)

        if lag >= 0:
            max_i = n - lag - 1
            if max_i < 0:
                continue
            idx = np.where(h[: max_i + 1] > h[lag : lag + max_i + 1] + 1e-12)[0]
        else:
            lag_abs = -lag
            idx = np.where(h[lag_abs:] > h[: n - lag_abs] + 1e-12)[0] + lag_abs

        if idx.size == 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
        else:
            i = rng.choice(idx)
            j = i - lag  # because lag = i - j

        if i == j:
            continue

        max_up = 1.0 - h[j]
        max_down = h[i]
        max_eps = min(max_up, max_down)
        if max_eps <= 0.0:
            continue

        eps = step * rng.random() * max_eps
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c = compute_c5(new_h)
        if c < best_c:
            h, best_c = new_h, c

        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-6)

    return h, best_c


def lbfgsb_refine(h0, beta, lam=200.0, max_iter=300):
    """Refine using L‑BFGS‑B on a smooth surrogate of the max‑overlap."""
    n = h0.size
    target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        smooth = (m + np.log(np.sum(np.exp(bc - m)))) / beta
        penalty = lam * (np.sum(x) - target) ** 2
        return smooth + penalty

    bounds = [(0.0, 1.0)] * n
    res = opt.minimize(
        obj,
        h0,
        method="L-BFGS-B",
        bounds=bounds,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum
    x = project_to_sum_box(x, target)
    return x
