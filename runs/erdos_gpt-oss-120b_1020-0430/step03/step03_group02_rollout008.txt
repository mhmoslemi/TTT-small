    start = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # Use parent if present
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0, 1)
        h0 = enforce_sum(h0, rng)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0.copy(), c0, h0.size

    candidate_ns = [96, 120, 144, 168, 192, 216, 240, 256]

            break

            h_start = best_h.copy()
        elif best_h is not None:

            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h_start = np.interp(x_new, x_old, best_h)

import time, numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h, rng):
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple random redistribution
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h≤1, Σh=n/2)."""
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h = enforce_sum(h, rng)
    return h

def worst_gradient(h):
    """Gradient of the worst shift overlap."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h_start, rng, max_iters=2500, step_scale=0.5):
    """Hill‑climb using gradient‑directed transfers."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    decay = max(1, max_iters // 5)
    step = step_scale
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

    return np.sum(x) - CURRENT_TARGET

    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:

    return x

    # up‑sampling and further local optimisation if time permits
    # ------------------------------------------------------------------
    while best_h is not None and time.time() - start < 0.96 * budget_s and best_n < 300:
        n_up = best_n * 2
        if n_up > 300:
            break
        x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = np.clip(h_up, 0.0, 1.0)
        h_up = enforce_sum(h_up, rng)

        h_up, c_up = guided_hill_climb(h_up, rng, max_iters=3000, step_scale=0.5)
        for beta in (5, 20, 80, 200):
            if time.time() - start > budget_s:
                break
            h_up = slsqp_optimize(h_up, beta, rng, max_iter=400)
        c_up = compute_c5(h_up)

        if c_up < best_c:
            best_h, best_c, best_n = h_up.copy(), c_up, n_up
        else:
            break

    # ------------------------------------------------------------------
    # final simulated‑annealing style random tweaking
    # ------------------------------------------------------------------
    if best_h is not None:
        h = best_h.copy()
        c = best_c
        n = best_n
        temp = 0.02
        while time.time() - start < budget_s * 0.999:
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * 0.5
            nh = h.copy()
            nh[i] -= eps
            nh[j] += eps
            nc = compute_c5(nh)
            delta = nc - c
            if delta < 0 or rng.random() < np.exp(-delta / temp):
                h, c = nh, nc
                if c < best_c:
                    best_h, best_c = h.copy(), c
            temp *= 0.9999

    # ------------------------------------------------------------------
    # final feasibility enforcement and return
    # ------------------------------------------------------------------
    if best_h is None:
        best_n = 96
        best_h = np.full(best_n, 0.5)
        best_c = compute_c5(best_h)
    else:
        best_h = enforce_sum(np.clip(best_h, 0.0, 1.0), rng)
        best_c = compute_c5(best_h)

    return best_h.tolist(), float(best_c), int(best_h.size)
