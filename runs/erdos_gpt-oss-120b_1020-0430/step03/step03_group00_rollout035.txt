    corr = np.correlate(x, 1 - x, mode='full')
    max_val = np.max(beta * corr)
    # but for stable compute exp
    expb = np.exp(beta * corr - max_val)
    Z = np.sum(expb)
    obj = (np.log(Z) + max_val) / beta

    # compute weights
    w = expb / Z  # same length as corr
    # compute gradient as sum_k w_k * grad_i_k
    grad = np.zeros_like(x)
    n = x.size
    # For each shift k (index in corr array)
    # shift index offset = k - (n - 1) => actual shift value shift = k - (n-1)
    for idx, w_k in enumerate(w):
        shift = idx - (n - 1)
        if w_k == 0:
            continue

        # grad_i = w_k * (1 - x_{i+shift} - x_{i-shift}) with zero padding.
        # we can vectorize using slicing.
        if shift >= 0:

def project_onto_bounds_and_sum(y, target):
    # Clip y to [0,1] after shifting by lambda, using bisection on lambda.
    # Solve for lambda s.t sum( clip(y - lambda, 0, 1) ) = target.
    # Binary search lambda.
    n = y.size
    # Lower bound: lambda_low = max(y - 1) (such that y_i - lambda <= 1)
    # Upper bound: lambda_high = min(y) (such that y_i - lambda >= 0)
    lo = np.max(y - 1.0)  # may be negative
    hi = np.min(y)  # may be > target? But hi >= lo.
    for _ in range(50):
        lam = (lo + hi) / 2.0
        x = np.clip(y - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    # final x
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    # Might be off by rounding; adjust small diff by adding/removing from non-bound indices.
    diff = target - x.sum()
    # If diff small (<1e-8) okay.
    if abs(diff) > 1e-8:

            x[free] += adj
            # clip again
            x = np.clip(x, 0.0, 1.0)
            # final correction
            x = enforce_sum(x, target, np.random.default_rng())
    return x

    start_time = time.time()

        h0 = random_feasible_vector(360, rng)
    # Option for upsampling to higher n
    n_target = 720
    if h0.size < n_target:
        # Use linear interpolation to oversample
        xs0 = (np.arange(h0.size) + 0.5) * (2.0 / h0.size)
        xs = (np.arange(n_target) + 0.5) * (2.0 / n_target)
        h = np.interp(xs, xs0, h0)
    else:
        h = h0.copy()
        n_target = h0.size
    # ensure feasible
    h = enforce_sum(np.clip(h,0,1), n_target/2.0, rng)
    best_h = h.copy()
    best_c = compute_c5(best_h)

    # Define a function to perform SLSQP with beta schedule on given h
    def slsqp_refine(h_start):
        nonlocal best_h, best_c, start_time
        n = h_start.size
        target = n / 2.0
        bounds = [(0.0,1.0)] * n
        # equality constraint
        cons = {"type":"eq", "fun": lambda x: np.sum(x) - target}
        # Beta schedule
        for beta in [5, 20, 80, 200, 800, 2000]:
            if time.time() - start_time > budget_s * 0.95:
                break
            # define objective
            def fun(x, beta=beta):
                corr = np.correlate(x, 1 - x, mode="full")
                # stable softmax
                maxb = np.max(beta * corr)
                expb = np.exp(beta * corr - maxb)
                Z = np.sum(expb)
                obj = (np.log(Z) + maxb) / beta
                # gradient
                w = expb / Z
                grad = np.zeros_like(x)
                n = x.size
                for idx, wk in enumerate(w):
                    if wk == 0:
                        continue
                    shift = idx - (n - 1)
                    if shift >= 0:
                        grad[:n - shift] += wk * (-x[shift:])
                        grad[shift:] += wk * (-x[:n - shift])
                    else:
                        kk = -shift
                        grad[:n - kk] += wk * (-x[kk:])
                        grad[kk:] += wk * (-x[:n - kk])
                    grad += wk
                # return obj, grad
                return obj, grad
            # minimize
            try:
                res = minimize(fun, h_start, method='SLSQP', jac=True,
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 500, 'ftol':1e-12, 'disp':False})
            except Exception:
                continue
            x = np.clip(res.x,0,1)
            # enforce sum exactly using projection
            x = enforce_sum(x, target, rng)
            c = compute_c5(x)
            if c < best_c:
                best_h, best_c = x.copy(), c
            # set start for next beta as current best
            h_start = x.copy()
        return h_start

    # Perform multiple restarts within time
    while time.time() - start_time < budget_s * 0.78:
        # Random start
        if rng.random() < 0.5:
            start = random_feasible_vector(n_target, rng)
        else:
            # perturb best a bit
            start = perturb_random(best_h, rng, scale=0.2)
        # local hill-climb targeted (increase iterations)
        h_local, c_local = hill_climb_targeted(start, rng, max_iters=3000, start_step=0.5, target_sum=n_target/2.0)
        if c_local < best_c:
            best_h, best_c = h_local.copy(), c_local
        # SLSQP refinement
        h_ref = slsqp_refine(best_h)
        # maybe also refine h_local
        h_ref2 = slsqp_refine(h_local)
        # pick best among them
        for cand in (best_h, h_local, h_ref, h_ref2):
            if cand is None:
                continue
            c = compute_c5(cand)
            if c < best_c:
                best_h, best_c = cand.copy(), c

    # final micro-search
    while time.time() - start_time < budget_s * 0.99:
        new_h = perturb_random(best_h, rng, scale=0.02)
        c_new = compute_c5(new_h)
        if c_new < best_c:
            best_h, best_c = new_h.copy(), c_new

    # final cleanup
    best_h = np.clip(best_h,0,1)
    best_h = enforce_sum(best_h, best_h.size/2.0, rng)

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)

def enforce_sum(h, target, rng):
    """Clip to [0,1] and adjust to make sum exactly target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 20000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:  # need to increase some entries
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i])
            add *= rng.random()
            h[i] += add
            diff -= add
        else:  # need to decrease
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i])
            sub *= rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        if diff > 0:
            h[i] = min(1.0, h[i] + diff)
        else:
            h[i] = max(0.0, h[i] + diff)
    return h

def random_feasible_vector(n, rng):
    """Start from uniform random values and repair to satisfy Σh = n/2."""
    h = rng.random(n)
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, n / 2.0, rng)
    return h

def perturb_random(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb_targeted(h_start, rng, max_iters, start_step, target_sum):
    h = h_start.copy()
    best_c = compute_c5(h)
    best_h = h.copy()
    step = start_step
    decay_interval = max(1, max_iters // 10)
    n = h.size
    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr)
        max_corr = corr[idx]
        c_current = max_corr * (2.0 / n)
        k = idx - (n - 1)

        # gradient of the worst shift
        grad = np.ones(n)
        if k >= 0:
            grad[:n - k] -= h[k:]
            grad[k:] -= h[:n - k]
        else:
            kk = -k
            grad[:n - kk] -= h[kk:]
            grad[kk:] -= h[:n - kk]

        # propose a move
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i != j:
            max_eps = min(h[i] if grad[i] > 0 else (1 - h[i]), h[j] if grad[j] < 0 else (1 - h[j]))
            if max_eps > 0:

                new_h[j] += eps

                if c_new < best_c:

                if c_new < c_current:
                    h = new_h

        if (it + 1) % decay_interval == 0:

# Smooth approximation of max (log‑sum‑exp) with gradient

def softmax_obj_grad(x, beta):
    n = x.size
    corr = np.correlate(x, 1 - x, mode="full")
    maxb = np.max(beta * corr)
    expb = np.exp(beta * corr - maxb)
    Z = np.sum(expb)
    f = (np.log(Z) + maxb) / beta
    w = expb / Z  # weights sum to 1
    grad = np.ones(n)  # constant term from sum w = 1
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        shift = idx - (n - 1)
        if shift >= 0:
            end = n - shift
            grad[:end] -= wk * x[shift:]
            grad[shift:] -= wk * x[:end]
        else:
            kk = -shift
            end = n - kk
            grad[:end] -= wk * x[kk:]
            grad[kk:] -= wk * x[:end]
    return f, grad
