def project_to_hyperplane_box(x, target, tol=1e-10, max_iter=100):
    lo, hi = -1.0, 1.0
    for _ in range(max_iter):
        mid = (lo+hi)/2
        proj = np.clip(x + mid, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) < tol:
            return proj
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo+hi)/2, 0.0, 1.0)

def obj_and_grad(h, beta):

    # Softmax weights
    exp_w = np.exp(beta * (r - r_max))
    w = exp_w / exp_w.sum()
    # compute convs
    conv1 = np.convolve(w[::-1], h, mode='full')
    conv2 = np.convolve(w, h, mode='full')
    n = h.size
    s1 = conv1[n-1:n-1+n]
    s2 = conv2[n-1:n-1+n]

def obj_and_grad(h, beta):
    r = correlate_full(h)
    r_max = r.max()
    # softmax weights
    exp_term = np.exp(beta * (r - r_max))
    s = exp_term.sum()
    if s == 0.0:
        w = np.zeros_like(r)
        w[np.argmax(r)] = 1.0
    else:
        w = exp_term / s
    # convolutions for gradient
    n = h.size
    conv_plus = np.convolve(w[::-1], h, mode='full')
    conv_minus = np.convolve(w, h, mode='full')
    s1 = conv_plus[n-1:n-1+n]
    s2 = conv_minus[n-1:n-1+n]
    grad = 1.0 - (s1 + s2)
    # objective value
    obj = (r_max + np.log(s)) / beta
    return obj, grad

import time
import numpy as np

# --------------------------------------------------------------
# Projection onto the intersection of the hyperplane Σh = target
# and the box 0 ≤ h ≤ 1.  Uses a simple bisection on the Lagrange
# multiplier λ in h_i ← clip(h_i + λ,0,1).
# --------------------------------------------------------------
def project_to_hyperplane_box(x, target, tol=1e-12, max_iter=30):
    lo, hi = -1.0, 1.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        proj = np.clip(x + mid, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)


# --------------------------------------------------------------
# Linear correlation of h with (1-h) using FFT (full mode)
# --------------------------------------------------------------
def correlate_full(h):
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1.0 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


# --------------------------------------------------------------
# Exact discrete C5 for a step function (dx = 2 / n)
# --------------------------------------------------------------
def compute_c5(h):
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)


# --------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient.
# Gradient: ∇f = 1 - Σ_ℓ w_ℓ (h_{i+ℓ}+h_{i-ℓ})
# where w = softmax(β·r) and r = correlation vector.
# --------------------------------------------------------------
def obj_and_grad(h, beta):
    r = correlate_full(h)
    r_max = r.max()
    exp_term = np.exp(beta * (r - r_max))
    s = exp_term.sum()
    if s == 0.0:                     # extreme β → only the max matters
        w = np.zeros_like(r)
        w[np.argmax(r)] = 1.0
    else:
        w = exp_term / s
    n = h.size
    # convolution to obtain Σ_ℓ w_ℓ h_{i+ℓ}  (w reversed) and Σ_ℓ w_ℓ h_{i-ℓ}
    conv_plus = np.convolve(w[::-1], h, mode="full")
    conv_minus = np.convolve(w, h, mode="full")
    s1 = conv_plus[n - 1 : n - 1 + n]
    s2 = conv_minus[n - 1 : n - 1 + n]
    grad = 1.0 - (s1 + s2)
    obj = (r_max + np.log(s)) / beta
    return obj, grad


# --------------------------------------------------------------
# Small random transfer that preserves the sum.
# --------------------------------------------------------------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    out = h.copy()
    out[i] -= eps
    out[j] += eps
    return out


# --------------------------------------------------------------
# Simple greedy hill climb (accept only improvements).
# --------------------------------------------------------------
def hill_climb(h_start, rng, max_iters=2000, start_scale=0.5):
    best = h_start.copy()
    best_c = compute_c5(best)
    scale = start_scale
