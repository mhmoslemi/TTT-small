def hill_climb(h, rng, max_iters, step_scale):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(best_h)
    cur_h = h.copy()
    cur_c = best_c
    for it in range(max_iters):
        # compute worst shift gradient
        corr = np.correlate(cur_h, 1 - cur_h, mode='full')
        idx = np.argmax(corr)
        k = idx - (n - 1)
        # gradient
        grad = np.ones(n)
        if k >= 0:
            grad[:n - k] -= cur_h[k:]
            grad[k:] -= cur_h[:n - k]
        else:
            kk = -k
            grad[:n - kk] -= cur_h[kk:]
            grad[kk:] -= cur_h[:n - kk]
        pos = np.where((grad > 1e-9) & (cur_h > 1e-12))[0]
        neg = np.where((grad < -1e-9) & (cur_h < 1.0 - 1e-12))[0]
        if pos.size > 0 and neg.size > 0:
            i = rng.choice(pos)
            j = rng.choice(neg)
            max_eps = min(cur_h[i], 1.0 - cur_h[j])
            eps = max_eps * step_scale * rng.random()
            new_h = cur_h.copy()
            new_h[i] -= eps
            new_h[j] += eps
        else:
            new_h = perturb_transfer(cur_h, step_scale, rng)
        new_h = enforce_sum(np.clip(new_h,0,1), n/2, rng)
        new_c = compute_c5(new_h)
        if new_c < cur_c:
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
        else:
            # optional occasional acceptance
            if rng.random() < 0.001:
                cur_h, cur_c = new_h, new_c
        # decrease step scale gradually
        step_scale *= 0.9995
    return best_h, best_c

def hill_climb(h0, rng, iters, step_scale):
    h = h0.copy()
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    best_h = h.copy()
    best_c = np.max(corr) * (2.0 / n)
    for it in range(iters):
        # compute worst shift gradient
        k, grad, corr = worst_shift_and_gradient(h)
        pos = np.where((grad > 1e-9) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-9) & (h < 1.0 - 1e-12))[0]
        if pos.size > 0 and neg.size > 0:
            i = rng.choice(pos)
            j = rng.choice(neg)
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0:
                continue
            eps = max_eps * step_scale * rng.random()
            new_h = h.copy()
            new_h[i] -= eps
            new_h[j] += eps
        else:
            new_h = perturb_transfer(h, step_scale, rng)
        new_h = enforce_sum(np.clip(new_h, 0.0, 1.0), n/2.0, rng)
        c_new = compute_c5(new_h)
        if c_new < best_c:
            best_c, best_h = c_new, new_h.copy()
        # accept if improvement, else keep old h
        if c_new < np.max(corr) * (2.0 / n):
            h = new_h
        # optionally reduce step_scale over time
        step_scale = max(step_scale * 0.9995, 1e-4)
    return best_h, best_c

def simulated_annealing(h0, rng, time_limit):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(best_h)
    cur_c = best_c
    T = 0.05  # initial temperature
    T_min = 1e-5
    alpha = 0.995
    # We'll also decide step scale dynamic
    step_scale = 0.2
    start = time.time()
    while time.time() - start < time_limit:
        # propose move with targeted gradient with probability p
        if rng.random() < 0.7:
            k, grad, _ = worst_shift_and_gradient(h)
            pos = np.where((grad > 1e-9) & (h > 1e-12))[0]
            neg = np.where((grad < -1e-9) & (h < 1.0 - 1e-12))[0]
            if pos.size > 0 and neg.size > 0:
                i = rng.choice(pos)
                j = rng.choice(neg)
                max_eps = min(h[i], 1.0 - h[j])
                eps = max_eps * step_scale * rng.random()
                new_h = h.copy()
                new_h[i] -= eps
                new_h[j] += eps
            else:
                new_h = perturb_transfer(h, step_scale, rng)
        else:
            new_h = perturb_transfer(h, step_scale, rng)
        new_h = enforce_sum(np.clip(new_h, 0.0, 1.0), n/2.0, rng)
        c_new = compute_c5(new_h)
        # Metropolis acceptance
        if c_new < cur_c or rng.random() < math.exp(-(c_new - cur_c) / max(T, 1e-12)):
            h = new_h
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = new_h.copy()
        # cooling
        T = max(T * alpha, T_min)
        # optionally adjust step_scale with T
        step_scale = max(0.05 * (T / 0.05), 0.02)  # small scale near low T
    return best_h, best_c

def enforce_sum(h, target, rng):
    """Clip to [0,1] and adjust to make sum exactly target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:                     # need to increase some entries
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i])
            add *= rng.random()
            h[i] += add
            diff -= add
        else:                            # need to decrease some entries
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i])
            sub *= rng.random()
            h[i] -= sub
            diff += sub

        else:

            j = rng.choice(neg)
            max_eps = min(h[i], 1.0 - h[j])
            eps = max_eps * step_scale * rng.random()
            nh = h.copy()
            nh[i] -= eps
            nh[j] += eps
        else:
            nh = perturb_transfer(h, step_scale, rng)
        # sum is already preserved, simply clip for safety
        nh = np.clip(nh, 0.0, 1.0)
        c_new = compute_c5(nh)
        if c_new < best_c:
            best_c, best_h = c_new, nh.copy()
        # accept only if it improves the current worst shift
        c_cur = np.max(corr) * (2.0 / n)
        if c_new < c_cur:
            h = nh
        step_scale = max(step_scale * 0.9995, 1e-4)
    return best_h, best_c


def simulated_annealing(h0, rng, time_limit):
    """Simple SA that may accept temporary worsenings."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(best_h)
    cur_c = best_c
    T = 0.05
    T_min = 1e-5
    alpha = 0.995
    step_scale = 0.2
    start = time.time()
    while time.time() - start < time_limit:
        if rng.random() < 0.7:
            k, grad, _ = worst_shift_and_gradient(h)
            pos = np.where((grad > 1e-9) & (h > 1e-12))[0]
            neg = np.where((grad < -1e-9) & (h < 1.0 - 1e-12))[0]
            if pos.size and neg.size:
                i = rng.choice(pos)
                j = rng.choice(neg)
                max_eps = min(h[i], 1.0 - h[j])
                eps = max_eps * step_scale * rng.random()
                nh = h.copy()
                nh[i] -= eps
                nh[j] += eps
            else:
                nh = perturb_transfer(h, step_scale, rng)
        else:
            nh = perturb_transfer(h, step_scale, rng)

        nh = np.clip(nh, 0.0, 1.0)
        c_new = compute_c5(nh)
        if c_new < cur_c or rng.random() < math.exp(-(c_new - cur_c) / max(T, 1e-12)):
            h, cur_c = nh, c_new
            if c_new < best_c:
                best_c, best_h = c_new, nh.copy()
        T = max(T * alpha, T_min)
        # shrink step size as we cool
        step_scale = max(0.02, step_scale * alpha)
    return best_h, best_c


def resample_vector(h, n_target, rng):
    """Linear interpolation to a different number of points."""
    n_src = h.size
    if n_target == n_src:
        return h.copy()
    xs_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    xs_tgt = (np.arange(n_target) + 0.5) * (2.0 / n_target)
    return np.interp(xs_tgt, xs_src, h)

# Main entry point
# ------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    total_budget = budget_s

    # --- obtain an initial construction ---------------------------------
    if "initial_h_values" in globals():
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = random_feasible_vector(180, rng)

    best_h = None
    best_c = float("inf")

    # we will touch three resolutions, ending with the finest we can afford
    for n in (180, 360, 720):
        # stop if we are out of time
        if time.time() - start_time > total_budget * 0.98:
            break

        # start point for this resolution
        if best_h is not None:
            cur_start = resample_vector(best_h, n, rng)
        else:
            cur_start = resample_vector(h_init, n, rng)
        cur_start = enforce_sum(np.clip(cur_start, 0.0, 1.0), n / 2.0, rng)

        # allocate a slice of the remaining budget to this resolution
        remaining = total_budget - (time.time() - start_time)
        phase_budget = remaining * 0.3
        phase_start = time.time()

        # --- multiple hill‑climb attempts --------------------------------
        while time.time() - phase_start < phase_budget:
            # pick a start vector (often a perturbed best)
            if best_h is not None and rng.random() < 0.6:
                start = perturb_transfer(best_h, 0.3, rng)
                if start.size != n:
                    start = resample_vector(best_h, n, rng)
            else:
                start = random_feasible_vector(n, rng)

            # determine a safe number of iterations given remaining time
            time_left = phase_budget - (time.time() - phase_start)
            est_per_iter = 0.001  # generous upper bound
            max_iters = max(1, int(time_left / (est_per_iter * 20)))  # 20 ops per iter
            iters = min(2000, max_iters)

            h_loc, c_loc = hill_climb(start, rng, iters, step_scale=0.5)
            if c_loc < best_c:
                best_h, best_c = h_loc.copy(), c_loc

            if time.time() - start_time > total_budget * 0.94:
                break

        # --- simulated annealing on the current best --------------------
        remaining = total_budget - (time.time() - start_time)
        if remaining > 5.0 and best_h is not None and best_h.size == n:
            sa_time = min(remaining * 0.2, 120.0)
            h_sa, c_sa = simulated_annealing(best_h, rng, sa_time)
            if c_sa < best_c:
                best_h, best_c = h_sa.copy(), c_sa

        # prepare for next (finer) resolution
        if best_h is not None:
            # keep the shape but resample later
            pass

    # ---------- final micro‑search (tiny random transfers) ----------
    while time.time() - start_time < total_budget * 0.997:
        if best_h is None:
            break
        nh = perturb_transfer(best_h, 0.02, rng)
        nh = enforce_sum(np.clip(nh, 0.0, 1.0), best_h.size / 2.0, rng)
        c_new = compute_c5(nh)
        if c_new < best_c:
            best_h, best_c = nh.copy(), c_new

    # final clean‑up
    if best_h is None:
        best_h = np.full(180, 0.5)
    best_h = enforce_sum(np.clip(best_h, 0.0, 1.0), best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
