def project_h(h, target):
    # Solve for lambda such that sum clip(h_i + lambda, 0, 1) = target
    # Binary search for lambda in [-1, 1] (wider maybe).
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo+hi)/2.0, 0.0, 1.0)

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    x = project_h(x, CURRENT_TARGET)
    return x
```

import time
import numpy as np
from scipy.optimize import minimize

# ---------- projection onto [0,1]∩{sum=target} ----------
def project_h(h, target):
    """Euclidean projection of h onto the hyperplane sum=h and the box [0,1]."""
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h + lam, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Uniform random vector projected to the feasible set."""
    h = rng.random(n)
    return project_h(h, n / 2.0)


# ---------- correlation based objective ----------
def compute_c5(h):
    """Discrete C5 value for a step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


# ---------- perturbations ----------
def perturb_random(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def perturb_targeted(h, grad, rng, scale=0.2):
    """Move mass from positive‑gradient indices to negative‑gradient ones."""
    pos = np.where((grad > 1e-9) & (h > 1e-12))[0]
    neg = np.where((grad < -1e-9) & (h < 1.0 - 1e-12))[0]
    if pos.size == 0 or neg.size == 0:
        return perturb_random(h, rng, scale)
    i = rng.choice(pos)
    j = rng.choice(neg)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return perturb_random(h, rng, scale)
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


# ---------- hill climbing on the worst shift ----------
def hill_climb_targeted(h_start, rng, max_iters, start_step, target_sum):
    h = h_start.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr)
        max_corr = corr[idx]
        c_cur = max_corr * (2.0 / n)
        k = idx - (n - 1)

        # gradient of the worst shift
        grad = np.ones(n)
        if k >= 0:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
        else:
            kk = -k
            grad[: n - kk] -= h[kk:]
            grad[kk:] -= h[: n - kk]

        new_h = perturb_targeted(h, grad, rng, scale=step)
        c_new = compute_c5(new_h)

        if c_new < best_c:
            best_c, best_h = c_new, new_h.copy()
        if c_new < c_cur:
            h = new_h
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


# ---------- smooth approximation for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log-sum-exp stability
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_h(x, CURRENT_TARGET)
    return x


# ---------- main driver ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ---- base resolution ----
    if "initial_h_values" in globals():
        base_n = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=float)
    else:
        base_n = 360
        h0 = random_feasible_vector(base_n, rng)
    h0 = project_h(h0, base_n / 2.0)

    # ---- upsample to high resolution (factor 2) ----
    high_n = base_n * 2
    xs_low = (np.arange(base_n) + 0.5) * (2.0 / base_n)
    xs_high = (np.arange(high_n) + 0.5) * (2.0 / high_n)
    best_h = np.interp(xs_high, xs_low, h0)
    best_h = project_h(best_h, high_n / 2.0)
    best_c = compute_c5(best_h)

    # ---- low‑resolution exploration (25% of budget) ----
    low_budget = budget_s * 0.25
    beta_sched_low = [5, 20, 80, 200, 800]
    attempt = 0
    while time.time() - start_time < low_budget and attempt < 12:
        if attempt == 0:
            h = h0.copy()
        else:
            h = random_feasible_vector(base_n, rng)

        for beta in beta_sched_low:
            if time.time() - start_time >= low_budget:
                break
            h = slsqp_optimize(h, beta, max_iter=250)

        # upsample and evaluate
        h_up = np.interp(xs_high, xs_low, h)
        h_up = project_h(h_up, high_n / 2.0)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_c, best_h = c_up, h_up.copy()
        attempt += 1

    # ---- high‑resolution refinement (65% of budget) ----
    beta_sched_high = [5, 20, 80, 200, 800, 3000, 8000]
    while time.time() - start_time < budget_s * 0.90:
        # start from a perturbation of the current best or a fresh point
        if rng.random() < 0.7:
            start_h = perturb_random(best_h, rng, scale=0.2)
        else:
            start_h = random_feasible_vector(high_n, rng)

        # targeted hill climb
        h_loc, c_loc = hill_climb_targeted(
            start_h, rng, max_iters=1200, start_step=0.5, target_sum=high_n / 2.0
        )
        if c_loc < best_c:
            best_c, best_h = c_loc, h_loc.copy()

        # SLSQP refinements with increasing beta
        for beta in beta_sched_high:
            if time.time() - start_time >= budget_s * 0.95:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, max_iter=400)
            except Exception:
                continue
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_c, best_h = c_ref, h_ref.copy()

        # early exit if time is running short
        if time.time() - start_time > budget_s * 0.88:
            break

    # ---- final tiny random search (remaining budget) ----
    while time.time() - start_time < budget_s:
        new_h = perturb_random(best_h, rng, scale=0.02)
        c_new = compute_c5(new_h)
        if c_new < best_c:
            best_c, best_h = c_new, new_h.copy()

    # ---- final clean‑up ----
    best_h = project_h(np.clip(best_h, 0.0, 1.0), best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
