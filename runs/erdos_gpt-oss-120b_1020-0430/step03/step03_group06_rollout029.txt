def hill_climb_targeted(h0):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        # compute correlation
        corr = np.correlate(h, 1 - h, mode="full")
        # compute max shift index from corr
        idx_max = np.argmax(corr)
        shift = idx_max - (n - 1)  # shift range from -(n-1) to (n-1)
        # define permissible i range
        if shift >= 0:
            i_min = 0
            i_max = n - shift  # exclusive
        else:
            i_min = -shift
            i_max = n
        # random i
        i = rng.integers(i_min, i_max)
        j = i + shift
        # compute max epsilon possible
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            # try another random i up to some attempts
            continue
        eps = rng.random() * max_eps * step
        # new vector
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        # compute new C5
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
            h = new_h
        else:
            # maybe keep new h with some probability (simulated annealing)
            # compute delta = new_c - best_c

            # we could just ignore (revert)
            pass
        # step decay
        step *= decay
        # ensure step doesn't go below tiny
        if step < 1e-6:
            step = 1e-6
    return best_h, best_c

def random_swap(h):
    i, j = rng.integers(0, n, size=2)
    while j == i:
        j = rng.integers(0, n)
    new_h = h.copy()
    new_h[i], new_h[j] = new_h[j], new_h[i]

def targeted_hill_climb(h0, rng, max_iters, start_step=0.5, decay=0.9999, prob_swap=0.1, prob_random_transfer=0.2):
   h = h0.copy()
   best_h = h.copy()
   best_c = compute_c5(h)
   step = start_step
   n = h.size
   for it in range(max_iters):
       if rng.random() < prob_swap:
           # swap
           i, j = rng.integers(0, n, size=2)
           while j == i:
               j = rng.integers(0, n)
           new_h = h.copy()
           new_h[i], new_h[j] = new_h[j], new_h[i]
           new_c = compute_c5(new_h)
           if new_c < best_c:
               best_c, best_h, h = new_c, new_h.copy(), new_h
           continue
       # compute correlation and max shift
       corr = np.correlate(h, 1 - h, mode="full")
       idx_max = np.argmax(corr)
       shift = idx_max - (n - 1)
       if shift == 0:
           # no beneficial move from shift zero, fall back to random transfer
           use_random = True
       else:
           use_random = rng.random() < prob_random_transfer
       if use_random:
           # random transfer i->j
           i = rng.integers(0, n)
           j = rng.integers(0, n)
           while j == i:
               j = rng.integers(0, n)
           # choose direction that reduces product if possible
           # compute possible eps moving from i to j
           max_eps = min(h[i], 1.0 - h[j])
           # also consider reverse
           rev_eps = min(h[j], 1.0 - h[i])
           # decide direction
           if max_eps >= rev_eps:
               eps_max = max_eps
               from_i, to_j = i, j
           else:
               eps_max = rev_eps
               from_i, to_j = j, i
           if eps_max <= 0:
               continue
           eps = rng.random() * eps_max * step
           new_h = h.copy()
           new_h[from_i] -= eps
           new_h[to_j] += eps
           new_c = compute_c5(new_h)
           if new_c < best_c:
               best_c, best_h, h = new_c, new_h.copy(), new_h
           continue
       # targeted transfer along max shift
       # determine allowed i range
       if shift > 0:
           i_min = 0
           i_max = n - shift
       else:
           i_min = -shift
           i_max = n
       if i_max <= i_min:
           continue
       i = rng.integers(i_min, i_max)
       j = i + shift
       max_eps = min(h[i], 1.0 - h[j])
       if max_eps <= 0:
           # maybe try reverse? moving from j to i would increase product though
           continue
       eps = rng.random() * max_eps * step
       new_h = h.copy()
       new_h[i] -= eps
       new_h[j] += eps
       new_c = compute_c5(new_h)
       if new_c < best_c:
           best_c, best_h, h = new_c, new_h.copy(), new_h
       # decay step
       step = max(step * decay, 1e-6)
   return best_h, best_c

    h = h_bin.copy()
    best_c = compute_c5(h)
    n = len(h)
    for it in range(max_iters):
        # compute correlation and max shift
        corr = np.correlate(h, 1 - h, mode="full")
        idx_max = np.argmax(corr)
        shift = idx_max - (n - 1)
        if shift != 0:
            # indices i that are 1 and i+shift < n (or > -shift)
            if shift > 0:
                i_candidates = np.where(h[:n - shift] == 1)[0]
                if i_candidates.size == 0:
                    continue
                i = rng.choice(i_candidates)
                j = i + shift
                # h[j] is likely 0; if not, we can find a zero elsewhere to swap
                if h[j] == 0:
                    # swap i (1) and j (0)
                    h[i] = 0
                    h[j] = 1
                else:
                    # find a zero anywhere else
                    zero_positions = np.where(h == 0)[0]
                    if zero_positions.size == 0:
                        continue
                    j = rng.choice(zero_positions)
                    h[i] = 0
                    h[j] = 1
            else: # shift < 0
                shift_abs = -shift
                i_candidates = np.where(h[shift_abs:] == 1)[0] + shift_abs  # actual i (the 'first' index)
                if i_candidates.size == 0:
                    continue
                i = rng.choice(i_candidates)
                j = i + shift
                if h[j] == 0:
                    h[i] = 0
                    h[j] = 1
                else:
                    zero_positions = np.where(h == 0)[0]
                    j = rng.choice(zero_positions)
                    h[i] = 0
                    h[j] = 1
            # compute new C5
            new_c = compute_c5(h)
            if new_c < best_c:
                best_c = new_c
            else:
                # revert
                h[i] = 1
                h[j] = 0
        else:
            # shift zero: random swap
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]

def binary_adjust(bits, target, rng):
    current = bits.sum()
    diff = int(target - current)
    if diff > 0:
        # need to turn diff zeros into 1
        zeros = np.where(bits == 0)[0]
        chosen = rng.choice(zeros, size=diff, replace=False)
        bits[chosen] = 1
    elif diff < 0:
        # need to turn -diff ones into 0
        ones = np.where(bits == 1)[0]
        chosen = rng.choice(ones, size=-diff, replace=False)
        bits[chosen] = 0
    return bits

    """Smooth log‑sum‑exp surrogate of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


# ----------------------------------------------------------------------
# Problem‑specific utilities
# ----------------------------------------------------------------------
def compute_c5(h):
    """Exact discrete C₅ for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng):
    """Adjust h so that Σh = target while 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    # simple stochastic correction
    for _ in range(1200):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-12:
            continue
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    v = rng.exponential(1.0, size=n)
    v = v / v.sum() * (n / 2.0)
    h = np.clip(v, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def binary_random_vector(n, rng):
    """Random binary vector with exactly n/2 ones."""
    arr = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    arr[idx] = 1.0
    return arr


def perturb_h(h, rng, scale=0.5):
    """Transfer a small amount between two coordinates (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    while j == i:
        j = rng.integers(0, n)
    max_pos = min(h[i], 1.0 - h[j])
    max_neg = min(h[j], 1.0 - h[i])
    if max_pos + max_neg == 0:
        return h.copy()
    eps = (rng.random() * (max_pos + max_neg)) - max_neg
    eps *= scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return np.clip(new_h, 0.0, 1.0)


def slsqp_optimize(h0, beta, max_iter=500):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny exact‑sum repair
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(1500):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            if abs(add) < 1e-12:
                continue
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def targeted_hill_climb(h0, rng, max_iters=20000,
                       start_step=0.5, decay=0.9999,
                       prob_swap=0.07, prob_random=0.18):
    """Hill‑climb using transfers along the worst shift, occasional swaps."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    for it in range(max_iters):
        # occasional swap
        if rng.random() < prob_swap:
            i, j = rng.integers(0, n, size=2)
            while j == i:
                j = rng.integers(0, n)
            new_h = h.copy()
            new_h[i], new_h[j] = new_h[j], new_h[i]
            new_c = compute_c5(new_h)
            if new_c < best_c:
                best_c, best_h, h = new_c, new_h.copy(), new_h
            continue

        # correlation and worst shift
        corr = np.correlate(h, 1 - h, mode="full")
        idx_max = np.argmax(corr)
        shift = idx_max - (n - 1)

        # random transfer (not necessarily along worst shift)
        if rng.random() < prob_random:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)
            # choose direction that can move mass
            eps_max_fwd = min(h[i], 1.0 - h[j])
            eps_max_rev = min(h[j], 1.0 - h[i])
            if eps_max_fwd >= eps_max_rev:
                eps_max = eps_max_fwd
                frm, to = i, j
            else:
                eps_max = eps_max_rev
                frm, to = j, i
            if eps_max <= 0:
                continue
            eps = rng.random() * eps_max * step
            new_h = h.copy()
            new_h[frm] -= eps
            new_h[to] += eps
            new_c = compute_c5(new_h)
            if new_c < best_c:
                best_c, best_h, h = new_c, new_h.copy(), new_h
            continue

        # targeted transfer along the worst shift
        if shift != 0:
            if shift > 0:
                i_min, i_max = 0, n - shift
            else:
                i_min, i_max = -shift, n
            if i_max <= i_min:
                continue
            i = rng.integers(i_min, i_max)
            j = i + shift
            eps_max = min(h[i], 1.0 - h[j])
            if eps_max <= 0:
                continue
            eps = rng.random() * eps_max * step
            new_h = h.copy()
            new_h[i] -= eps
            new_h[j] += eps
            new_c = compute_c5(new_h)
            if new_c < best_c:
                best_c, best_h, h = new_c, new_h.copy(), new_h
        # decay step size
        step = max(step * decay, 1e-6)
    return best_h, best_c


def binary_targeted_hill_climb(h0, rng, max_iters=40000):
    """Binary hill‑climb: swap a 1 with a 0, biasing the worst shift."""
    h = h0.astype(int).copy()
    best_h = h.copy()
    best_c = compute_c5(h.astype(float))
    n = h.size
    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode="full")
        idx_max = np.argmax(corr)
        shift = idx_max - (n - 1)

        if shift != 0:
            if shift > 0:
                ones = np.where(h[:n - shift] == 1)[0]
                if ones.size == 0:
                    continue
                i = rng.choice(ones)
                j = i + shift
                if h[j] == 0:
                    # perform the swap
                    h[i] = 0
                    h[j] = 1
                else:
                    zeros = np.where(h == 0)[0]
                    if zeros.size == 0:
                        continue
                    j = rng.choice(zeros)
                    h[i] = 0
                    h[j] = 1
            else:  # shift < 0
                s = -shift
                ones = np.where(h[s:] == 1)[0] + s
                if ones.size == 0:
                    continue
                i = rng.choice(ones)
                j = i + shift
                if h[j] == 0:
                    h[i] = 0
                    h[j] = 1
                else:
                    zeros = np.where(h == 0)[0]
                    if zeros.size == 0:
                        continue
                    j = rng.choice(zeros)
                    h[i] = 0
                    h[j] = 1
        else:
            # shift zero: random swap
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                continue
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h[i] = 0
            h[j] = 1

        new_c = compute_c5(h.astype(float))
        if new_c < best_c:
            best_c = new_c
            best_h = h.copy()
        else:
            # revert to previous best configuration
            h = best_h.copy()
    return best_h.astype(float), best_c


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # --------------------------------------------------------------
    # Determine dimension and initial vector
    # --------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 180
        h0 = np.full(n, 0.5, dtype=float)
        h0 = enforce_sum(h0, n / 2.0, rng)

    target = n / 2.0

    # --------------------------------------------------------------
    # Build a varied pool of starting points
    # --------------------------------------------------------------
    candidates = [h0]

    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx

    # Sinusoidal shapes
    for f in range(1, 5):
        for fun in (np.sin, np.cos):
            h = 0.5 + 0.5 * fun(np.pi * f * xs)
            h = np.clip(h, 0.0, 1.0)
            candidates.append(enforce_sum(h, target, rng))

    # Linear ramps
    candidates.append(enforce_sum(xs / 2.0, target, rng))
    candidates.append(enforce_sum(1.0 - xs / 2.0, target, rng))

    # Random feasible vectors
    for _ in range(5):
        candidates.append(random_feasible_vector(n, rng))

    # Binary candidates
    for _ in range(4):
        candidates.append(binary_random_vector(n, rng))

    # Simple block (contiguous ones)
    block = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)])
    candidates.append(enforce_sum(block, target, rng))

    # Alternating pattern
    alt = np.tile([1.0, 0.0], n // 2)
    if n % 2:
        alt = np.append(alt, 1.0)
    candidates.append(enforce_sum(alt, target, rng))

    # --------------------------------------------------------------
    # Pick the best among the pool
    # --------------------------------------------------------------
    best_h = None
    best_c = float("inf")
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    global_best_h, global_best_c = best_h, best_c

    # --------------------------------------------------------------
    # Quick SLSQP refinement on the current best
    # --------------------------------------------------------------
    beta_list = [10.0, 50.0, 200.0, 800.0, 2000.0, 5000.0]
    for beta in beta_list:
        if time.time() - start_time > budget_s * 0.55:
            break
        global_best_h = slsqp_optimize(global_best_h, beta, max_iter=600)
        global_best_c = compute_c5(global_best_h)

    # --------------------------------------------------------------
    # Continuous targeted hill‑climb (moderate effort)
    # --------------------------------------------------------------
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 0:
        max_iters = int(min(25000, remaining * 30))
        h_tmp, c_tmp = targeted_hill_climb(
            global_best_h, rng,
            max_iters=max_iters,
            start_step=0.5,
            decay=0.9995,
            prob_swap=0.06,
            prob_random=0.15,
        )
        if c_tmp < global_best_c:
            global_best_h, global_best_c = h_tmp, c_tmp

    # --------------------------------------------------------------
    # Binary conversion and binary hill‑climb
    # --------------------------------------------------------------
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 0:
        bin_h = (global_best_h > 0.5).astype(float)
        # enforce exact half‑ones (integer target)
        cur_sum = int(bin_h.sum())
        need = int(target) - cur_sum
        if need > 0:
            zeros = np.where(bin_h == 0)[0]
            if zeros.size >= need:
                idx = rng.choice(zeros, size=need, replace=False)
                bin_h[idx] = 1.0
        elif need < 0:
            ones = np.where(bin_h == 1)[0]
            need = -need
            if ones.size >= need:
                idx = rng.choice(ones, size=need, replace=False)
                bin_h[idx] = 0.0
        # binary hill‑climb
        max_iters_bin = int(min(50000, remaining * 30))
        h_bin_opt, c_bin_opt = binary_targeted_hill_climb(bin_h, rng, max_iters=max_iters_bin)
        if c_bin_opt < global_best_c:
            global_best_h, global_best_c = h_bin_opt, c_bin_opt

    # --------------------------------------------------------------
    # Final fine‑tune (continuous) on the possibly improved binary solution
    # --------------------------------------------------------------
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 0:
        max_iters = int(min(20000, remaining * 30))
        h_final, c_final = targeted_hill_climb(
            global_best_h, rng,
            max_iters=max_iters,
            start_step=0.3,
            decay=0.9997,
            prob_swap=0.05,
            prob_random=0.12,
        )
        if c_final < global_best_c:
            global_best_h, global_best_c = h_final, c_final

    # --------------------------------------------------------------
    # Optional up‑sampling (if time permits)
    # --------------------------------------------------------------
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 0 and n * 2 <= 360:
        # repeat each value twice
        h_up = np.repeat(global_best_h, 2)
        n_up = h_up.size
        target_up = n_up / 2.0
        h_up = enforce_sum(h_up, target_up, rng)
        # up‑sampled refinement
        max_iters_up = int(min(15000, remaining * 30))
        h_up_opt, c_up_opt = targeted_hill_climb(
            h_up, rng,
            max_iters=max_iters_up,
            start_step=0.4,
            decay=0.9998,
            prob_swap=0.04,
            prob_random=0.1,
        )
        if c_up_opt < global_best_c:
            global_best_h, global_best_c, n = h_up_opt, c_up_opt, n_up

    # --------------------------------------------------------------
    # Final clean‑up (clip + exact sum enforcement)
    # --------------------------------------------------------------
    global_best_h = np.clip(global_best_h, 0.0, 1.0)
    global_best_h = enforce_sum(global_best_h, n / 2.0, rng)
    final_c = compute_c5(global_best_h)

    return global_best_h.tolist(), float(final_c), int(n)
