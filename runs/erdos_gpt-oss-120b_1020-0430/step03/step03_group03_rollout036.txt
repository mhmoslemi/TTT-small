def slsqp_optimize(h0, beta, max_iter=500):
    n = h0.size
    target = n / 2.0
    def sum_eq(x):
        return np.sum(x) - target
    def softmax_obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    cons = {'type':'eq','fun': sum_eq}

def project_onto_bounds_and_sum(x, target):
    # Clip to [0,1]
    x = np.clip(x, 0.0, 1.0)
    diff = target - np.sum(x)
    # If diff is zero, done
    if abs(diff) < 1e-12:
        return x
    # Determine direction: need to add diff (>0) or subtract (<0)
    # Find indices that can be adjusted
    if diff > 0:
        # can increase at indices where x < 1
        idx = np.where(x < 1 - 1e-12)[0]
        if idx.size == 0:
            return x
        add = diff / idx.size

        limited = add > max_add

            diff = target - x.sum()

def project_h(x, target):

    diff = target - np.sum(x)
    # Simple adjustment: if diff > 0, add diff to first element not 1
    # but we need keep within [0,1].

        if diff > 0:
            idx = np.where(x < 1 - 1e-12)[0]
            if idx.size == 0:
                break
            # distribute equally among idx
            delta = diff / idx.size
            # apply limited
            # compute permissible addition per idx
            max_add = 1 - x[idx]
            # if delta <= min(max_add) then done
            if delta <= max_add.min():
                x[idx] += delta
                diff = 0.0
            else:
                # set those to 1
                over = delta > max_add
                x[idx[over]] = 1.0
                diff -= np.sum(max_add[over])
                # continue with remaining indices
        else:
            idx = np.where(x > 1e-12)[0]
            if idx.size == 0:
                break
            delta = (-diff) / idx.size
            max_sub = x[idx]
            if delta <= max_sub.min():
                x[idx] -= delta
                diff = 0.0
            else:
                over = delta > max_sub
                x[idx[over]] = 0.0
                diff += np.sum(max_sub[over])
    return x

def local_swap_improve(h, rng, max_iters):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(best_h)
    cur_h = h.copy()
    cur_c = best_c
    for _ in range(max_iters):
        corr = np.correlate(cur_h, 1 - cur_h, mode='full')
        max_corr = np.max(corr)
        cur_c = max_corr * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = cur_h.copy()
        idx_max = np.argmax(corr)
        shift = idx_max - (n - 1)
        # compute i_range etc.
        if shift >= 0:
            i_range = np.arange(0, n - shift)
            hi = cur_h[i_range]
            hj = cur_h[i_range + shift]
        else:
            s = -shift
            i_range = np.arange(s, n)
            hi = cur_h[i_range]
            hj = cur_h[i_range - s]
        prod = hi * (1 - hj)
        i_max_local = np.argmax(prod)

        mask = (hj > 0.5) & (hi < 0.5)
        if np.any(mask):
            hi_c = hi[mask]
            i_cand = i_range[mask]
            j_local = np.argmin(hi_c)
            j = i_cand[j_local]
        else:
            j_local = np.argmin(hi)
            j = i_range[j_local]
        if i_max == j:
            other = i_range[i_range != i_max]
            if other.size == 0:

            j = rng.choice(other)
        max_eps = min(cur_h[i_max], 1 - cur_h[j])

        if new_c < cur_c - 1e-12 or rng.random() < 0.001:

        while diff > 1e-12:
            idx = np.where(x < 1 - 1e-12)[0]
            if idx.size == 0:
                break
            add = diff / idx.size
            max_add = 1 - x[idx]
            if add <= max_add.min():
                x[idx] += add
                diff = 0.0
            else:
                over = max_add < add
                x[idx[over]] = 1.0
                diff -= max_add[over].sum()
    else:  # decrease
        diff = -diff
        while diff > 1e-12:
            idx = np.where(x > 1e-12)[0]
            if idx.size == 0:
                break
            sub = diff / idx.size
            max_sub = x[idx]
            if sub <= max_sub.min():
                x[idx] -= sub
                diff = 0.0
            else:
                over = max_sub < sub
                x[idx[over]] = 0.0
                diff -= max_sub[over].sum()
    return x

# ------------------ SLSQP polishing ------------------
def slsqp_optimize(h0, beta, max_iter=400):
    n = h0.size
    target = n / 2.0
    def sum_eq(x):
        return np.sum(x) - target
    def softmax_obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    cons = {'type': 'eq', 'fun': sum_eq}
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        softmax_obj, h0, method='SLSQP',
        bounds=bounds, constraints=cons,
        options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False}
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_h(x, target)

# ------------------ local swap improvement ------------------
def local_swap_improve(h_start, rng, max_iters):
    n = h_start.size
    best_h = h_start.copy()
    best_c = compute_c5(best_h)
    cur_h = h_start.copy()
    cur_c = best_c
    for _ in range(max_iters):
        corr = np.correlate(cur_h, 1 - cur_h, mode='full')
        max_corr = np.max(corr)
        cur_c = max_corr * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = cur_h.copy()
        idx_max = np.argmax(corr)
        shift = idx_max - (n - 1)
        if shift >= 0:
            i_range = np.arange(0, n - shift)
            hi = cur_h[i_range]
            hj = cur_h[i_range + shift]
        else:
            s = -shift
            i_range = np.arange(s, n)
            hi = cur_h[i_range]
            hj = cur_h[i_range - s]
        prod = hi * (1 - hj)
        i_max_local = np.argmax(prod)
        i_max = i_range[i_max_local]
        mask = (hj > 0.5) & (hi < 0.5)
        if np.any(mask):
            hi_c = hi[mask]
            i_cand = i_range[mask]
            j = i_cand[np.argmin(hi_c)]
        else:
            j = i_range[np.argmin(hi)]
        if i_max == j:
            other = i_range[i_range != i_max]
            if other.size == 0:
                continue
            j = int(rng.choice(other))
        max_eps = min(cur_h[i_max], 1 - cur_h[j])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps * 0.2
        if eps < 1e-12:
            continue
        h_new = cur_h.copy()
        h_new[i_max] -= eps
        h_new[j] += eps
        new_c = compute_c5(h_new)
        if new_c < cur_c - 1e-12 or rng.random() < 0.001:
            cur_h, cur_c = h_new, new_c
    return best_h, best_c

# ------------------ gradient descent on smooth max ------------------
def gradient_descent_smooth(h0, rng, time_limit):
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(best_h)
    beta = 5.0
    it = 0
    start = time.time()
    while time.time() - start < time_limit:
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        cur_c = max_corr * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
            if best_c <= 0.3808:
                break
        # softmax weights
        w = np.exp(beta * corr - beta * max_corr)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        w_rev = w[::-1]
        conv1 = np.convolve(w_rev, 1 - h, mode='full')
        conv2 = np.convolve(w_rev, h, mode='full')
        grad = conv1[n - 1:n - 1 + n] - conv2[n - 1:n - 1 + n]
        eta = 0.2 / math.sqrt(it + 1)
        h_new = project_h(h - eta * grad, target)
        h = h_new
        it += 1
        if it % 2000 == 0:
            beta = min(beta * 2.0, 5000.0)
    return best_h, best_c

# ------------------ main entry point ------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # ---- initialise ---------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        np.clip(h0, 0.0, 1.0, out=h0)
        # enforce exact sum n/2
        target0 = h0.size / 2.0
        h0 = project_h(h0, target0)
    else:
        n = 180
        h0 = random_binary_vector(n, rng)
    n = h0.size
    target = n / 2.0

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ---- phase 1: gradient descent on smooth max ----
    phase1_time = budget_s * 0.55
    gd_h, gd_c = gradient_descent_smooth(best_h, rng, phase1_time)
    if gd_c < best_c:
        best_h, best_c = gd_h, gd_c

    # ---- phase 2: targeted swap refinement ----------
    phase2_time = budget_s * 0.25
    while time.time() - start_time < phase1_time + phase2_time:
        # small random perturbation of current best as start
        h_start = best_h.copy()
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)
        max_eps = min(h_start[i], 1 - h_start[j])
        if max_eps > 0:
            eps = rng.random() * max_eps * 0.1
            h_start[i] -= eps
            h_start[j] += eps
        local_h, local_c = local_swap_improve(h_start, rng, max_iters=1500)
        if local_c < best_c:
            best_h, best_c = local_h, local_c
        if best_c <= 0.3808:
            break

    # ---- phase 3: SLSQP polishing ------------------
    for beta in (50, 200, 800, 2000, 8000, 20000):
        if time.time() - start_time > budget_s:
            break
        try:
            best_h = slsqp_optimize(best_h, beta, max_iter=500)
            best_c = compute_c5(best_h)
        except Exception:
            continue
        if best_c <= 0.3808:
            break

    # final safety projection & result
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_h(best_h, target)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
