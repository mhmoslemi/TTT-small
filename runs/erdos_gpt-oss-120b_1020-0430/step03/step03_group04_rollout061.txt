def sa_optimize(h0, rng, time_limit):
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c

    n = h.size
    target = n/2
    T = cur_c*0.1 + 1e-8   # initial temperature

    it = 0
    while time.time() < time_limit:
        # Determine worst shift
        corr_full = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr_full)
        k = max_idx - (n - 1)
        # guided or random move
        if rng.random() < 0.6 and abs(k) < n:  # guided
            # choose random i in valid range
            if k >= 0:
                i = rng.integers(0, n - k)
                j = i + k
            else:
                i = rng.integers(-k, n)
                j = i + k
            max_eps = min(h[i], 1 - h[j])
            if max_eps > 0:
                eps = (rng.random() - 0.5) * 2 * max_eps * 0.5
                # ensure eps allows staying within bounds
                eps = max(-max_eps, min(eps, max_eps))
                new_h = h.copy()
                new_h[i] -= eps
                new_h[j] += eps
            else:
                new_h = h  # no change
        else:
            # random transfer
            new_h = perturb_h(h, rng, scale=0.3)

        new_c = compute_c5(new_h)
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-12)):
            h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, h.copy()
        # cool
        T *= 0.9995
        it += 1
    return best_h, best_c
```

import time
import numpy as np

# --------------------------------------------------------------
# basic utilities
# --------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 = max_k ∫ h(x)(1‑h(x+k))dx."""
    n = h.size
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector in [0,1] with exact sum n/2."""
    h = rng.random(n)
    target = n / 2.0
    scale = target / h.sum()
    h *= scale
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    out = h.copy()
    out[i] -= eps
    out[j] += eps
    return out


def guided_transfer(h: np.ndarray, rng: np.random.Generator, k: int, scale: float = 0.5) -> np.ndarray:
    """Move a small amount along the worst shift direction."""
    n = h.size
    if k >= 0:
        i = rng.integers(0, n - k)
        j = i + k
    else:
        i = rng.integers(-k, n)
        j = i + k
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    out = h.copy()
    out[i] -= eps
    out[j] += eps
    return out


def hill_climb(
    h_start: np.ndarray,
    rng: np.random.Generator,
    max_iters: int = 2000,
    start_step: float = 0.6,
    decay: float = 0.995,
) -> tuple[np.ndarray, float]:
    """Simple hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        cand = perturb_h(h, rng, scale=step)
        c = compute_c5(cand)
        if c < best_c:
            best_c = c
            best_h = cand.copy()
            h = cand
        if (it + 1) % decay_every == 0:
            step = max(step * decay, 1e-5)
    return best_h, best_c


def gradient_refine(
    h: np.ndarray,
    rng: np.random.Generator,
    step: float = 0.02,
    max_iter: int = 20,
) -> np.ndarray:
    """Iteratively move against the gradient of the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            up = n - k
            g[:up] += 1.0 - h[k:]
            g[k:] -= h[:up]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


def simulated_annealing(
    h0: np.ndarray,
    rng: np.random.Generator,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """SA with guided moves that respect the sum constraint."""
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    n = h.size
    T = max(0.01 * cur_c, 1e-4)
    it = 0
    no_improve = 0
    while time.time() < deadline:
        # worst shift
        corr_full = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr_full)
        k = max_idx - (n - 1)

        # choose move
        if rng.random() < 0.6:
            cand = guided_transfer(h, rng, k, scale=0.5)
        else:
            cand = perturb_h(h, rng, scale=0.3)

        cand_c = compute_c5(cand)
        if cand_c < cur_c:
            accept = True
        else:
            accept = rng.random() < np.exp(-(cand_c - cur_c) / max(T, 1e-12))

        if accept:
            h, cur_c = cand, cand_c
            if cur_c < best_c:
                best_c, best_h = cur_c, h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # cooling / occasional reheating
        it += 1
        T *= 0.9995
        if no_improve > 5000:
            T = max(0.01 * cur_c, 1e-4)
            no_improve = 0
    return best_h, best_c

        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c = h_up.copy(), c_up
            n = n_up

    # --------------------------------------------------------------
    # final sanity check and return
    # --------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_h.shape[0] / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            i = rng.integers(0, best_h.shape[0])
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()
            if abs(diff) < 1e-12:
                break

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.shape[0])
