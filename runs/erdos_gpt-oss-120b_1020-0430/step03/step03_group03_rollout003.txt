import time, math, numpy as np
from scipy.optimize import minimize

# ---------- basic helpers ----------
def compute_c5(h):
    """Discrete C5 for step function h on [0,2] (dx = 2/len)."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / h.size)

def enforce_sum(h, target, rng):
    """Adjust h in‑place so that sum(h)=target while respecting 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            if add <= 0:
                continue
            h[i] += add
            diff = target - h.sum()
        else:
            sub = min(-diff, h[i])
            if sub <= 0:
                continue
            h[i] -= sub
            diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h≤1, sum=n/2)."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    enforce_sum(h, n / 2.0, rng)
    return h

def perturb_h(h, rng, scale=0.2):
    """Transfer ε between two coordinates (preserves sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

def guided_perturb(h, rng, lag, grad, scale=0.2):
    """Move mass from low‑gradient to high‑gradient positions (focus on worst lag)."""
    n = h.size
    low_thr = np.quantile(grad, 0.30)
    high_thr = np.quantile(grad, 0.70)
    low_idx = np.where(grad <= low_thr)[0]
    high_idx = np.where(grad >= high_thr)[0]
    if low_idx.size == 0 or high_idx.size == 0:
        return perturb_h(h, rng, scale)
    i = rng.choice(low_idx)
    j = rng.choice(high_idx)
    if i == j:
        return perturb_h(h, rng, scale)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

# ---------- SLSQP surrogates ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # smooth max: log-sum-exp / beta
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=200):
    """SLSQP on smooth max‑approximation (preserves constraints)."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {"type": "eq", "fun": sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

# ---------- main optimisation ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # initial vector
    if "initial_h_values" in globals():
        h0 = np.clip(np.array(initial_h_values, dtype=float), 0.0, 1.0)
        n = h0.size
        enforce_sum(h0, n / 2.0, rng)
    else:
        n = 180
        h0 = random_feasible_vector(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # time allocation
    sa_budget = budget_s * 0.78
    target_c = 0.3808  # early‑stop threshold

    while time.time() - start < sa_budget:
        # choose start point for this SA run
        if rng.random() < 0.6:
            cur = best_h.copy()
        else:
            cur = random_feasible_vector(n, rng)
        cur_c = compute_c5(cur)

        # estimate remaining iterations (roughly 400 it/s)
        remaining = max(0.0, sa_budget - (time.time() - start))
        max_iter = int(min(25000, remaining * 400))
        if max_iter < 2000:
            max_iter = 2000

        T0, Tf = 0.02, 1e-6
        log_ratio = math.log(Tf / T0) if T0 > 0 else 0.0

        for it in range(1, max_iter + 1):
            # temperature
            T = T0 * math.exp(log_ratio * it / max_iter)

            # compute lag and gradient for the worst overlap
            corr_full = np.correlate(cur, 1.0 - cur, mode="full")
            idx_max = np.argmax(corr_full)
            lag = idx_max - (n - 1)

            grad = np.zeros(n, dtype=float)
            if lag >= 0:
                left_len = n - lag
                grad[:left_len] += cur[lag:]
                grad[lag:] += cur[:left_len]
            else:
                s = -lag
                left_len = n - s
                grad[s:] += cur[:left_len]
                grad[:left_len] += cur[s:]

            # propose a move
            if rng.random() < 0.15:
                cand = perturb_h(cur, rng, scale=0.3)
            else:
                cand = guided_perturb(cur, rng, lag, grad, scale=0.3)

            cand_c = compute_c5(cand)
            diff = cand_c - cur_c
            if diff < 0 or rng.random() < math.exp(-diff / max(T, 1e-12)):
                cur, cur_c = cand, cand_c
                if cand_c < best_c:
                    best_c, best_h = cand_c, cand.copy()
                    if best_c < target_c:
                        # good enough, break early
                        break

        if best_c < target_c:
            break

        # occasional smooth‑max refinement
        for beta in (200, 800, 2000):
            if time.time() - start > sa_budget:
                break
            try:
                refined = slsqp_optimize(best_h, beta, max_iter=150)
                rc = compute_c5(refined)
                if rc < best_c:
                    best_c, best_h = rc, refined.copy()
                    if best_c < target_c:
                        break
            except Exception:
                continue
        if best_c < target_c:
            break

    # final high‑beta polishing
    for beta in (5000, 15000, 50000):
        if time.time() - start > budget_s:
            break
        try:
            best_h = slsqp_optimize(best_h, beta, max_iter=200)
            best_c = compute_c5(best_h)
            if best_c < target_c:
                break
        except Exception:
            continue

    best_h = np.clip(best_h, 0.0, 1.0)
    enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
