def simulated_annealing(h0, rng, max_iters, init_temp, final_temp):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    for it in range(max_iters):
        t = init_temp * ( (final_temp / init_temp) ** (it / max_iters) )
        # propose move
        i, j = rng.integers(0, n, size=2, endpoint=False)
        while j == i:
            j = rng.integers(0, n)
        # random direction
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_delta = min(h[src], 1 - h[dst])
        if max_delta <= 1e-12:
            continue
        # scale step based on temperature
        # Use step = max_delta * min(1, t) maybe.
        # We can define step_factor = t / init_temp  (goes from 1 to small)
        factor = t / init_temp
        delta = rng.random() * max_delta * factor
        # apply
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        # compute C5
        c_new = compute_c5(h_new)
        # accept if better, else with probability exp(-(c_new - cur_c)/t)
        if c_new < cur_c or rng.random() < np.exp(-(c_new - cur_c) / (t + 1e-12)):
            h = h_new
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
    return best_h, best_c

def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_interval = max(1, max_iters // 20)
    for it in range(max_iters):
        i, j = rng.integers(0, n, size=2)
        while j == i:
            j = rng.integers(0, n)
        max_pos = min(h[i], 1.0 - h[j])
        max_neg = min(h[j], 1.0 - h[i])
        max_delta = max_pos + max_neg
        if max_delta <= 0:
            continue
        delta = (rng.random() * max_delta) - max_neg
        delta *= step
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        # clip just in case
        new_h = np.clip(new_h, 0.0, 1.0)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        # optionally accept if c < current best? we used improvement only.
        # step decay
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.8, 1e-4)
    return best_h, best_c

def simulated_annealing(h_start, rng, max_iters, init_temp, final_temp):
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    n = h.size
    for it in range(max_iters):
        t = init_temp * ((final_temp / init_temp) ** (it / max_iters))
        i, j = rng.integers(0, n, size=2)
        while j == i:
            j = rng.integers(0, n)
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_delta = min(h[src], 1.0 - h[dst])
        if max_delta <= 0:
            continue
        # delta proportion to temperature
        delta = rng.random() * max_delta * (t / init_temp)
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        # clip is not required because delta respects bounds
        c_new = compute_c5(h_new)
        if c_new < cur_c or rng.random() < np.exp(-(c_new - cur_c) / (t + 1e-12)):
            h = h_new
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
    return best_h, best_c

def enforce_sum(h, target, rng):
    # ensure sum exactly target while staying within bounds.
    # We'll compute diff = target - sum(h)
    diff = target - h.sum()
    n = h.size
    # We adjust by adding diff to random indices while obeying bounds.
    # We'll repeat until diff magnitude small.
    for _ in range(5000):
        if abs(diff) < 1e-10:
            break
        # Choose random index
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-12:
            continue
        h[i] += add
        diff = target - h.sum()
    # final clip produce sum approx target; adjust residual by distributing to first few indices.
    if abs(diff) > 1e-10:
        # adjust first element
        i = 0
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def perturb_h(h, rng, scale=0.3):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    while j == i:
        j = rng.integers(0, n)
    max_pos = min(h[i], 1.0 - h[j])
    max_neg = min(h[j], 1.0 - h[i])
    max_delta = max_pos + max_neg
    if max_delta == 0:
        return h.copy()
    delta = (rng.random() * max_delta) - max_neg
    delta *= scale
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return np.clip(new_h, 0.0, 1.0)

CURRENT_TARGET = None
CURRENT_BETA = None
def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET
def _softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type':'eq', 'fun':_sum_eq_constraint}
    res = minimize(_softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter':max_iter, 'ftol':1e-12, 'disp':False})
    x = np.clip(res.x, 0.0, 1.0)
    # repair sum 
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            if abs(add) < 1e-12:
                continue
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper functions (global for the soft‑max surrogate used by SLSQP)
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def _eq_sum(x):
    return np.sum(x) - CURRENT_TARGET


def _soft_max(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact C5 for a discrete step function."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng):
    """Adjust h so that Σh = target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(4000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        if abs(add) < 1e-12:
            continue
        h[i] += add
        diff = target - h.sum()
    # tiny final correction (if needed)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):

            if abs(add) < 1e-12:
                continue
            h[i] += add
            diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n, rng):
    v = rng.exponential(1.0, size=n)
    v = v / v.sum() * (n / 2.0)
    h = np.clip(v, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def perturb_h(h, rng, scale=0.3):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    while j == i:
        j = rng.integers(0, n)
    max_pos = min(h[i], 1.0 - h[j])
    max_neg = min(h[j], 1.0 - h[i])
    max_delta = max_pos + max_neg
    if max_delta == 0:
        return h.copy()
    delta = (rng.random() * max_delta) - max_neg
    delta *= scale
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return np.clip(new_h, 0.0, 1.0)


def hill_climb(h_start, rng, max_iters=3000, start_step=0.5):
    """Simple greedy hill‑climb (preserves sum)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay = max(1, max_iters // 15)
    for it in range(max_iters):
        i, j = rng.integers(0, n, size=2)
        while j == i:
            j = rng.integers(0, n)
        max_pos = min(h[i], 1.0 - h[j])
        max_neg = min(h[j], 1.0 - h[i])
        max_delta = max_pos + max_neg
        if max_delta == 0:
            continue
        delta = (rng.random() * max_delta) - max_neg
        delta *= step
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_h = np.clip(new_h, 0.0, 1.0)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def simulated_annealing(h_start, rng, max_iters, init_temp=0.005, final_temp=0.0001):
    """SA that preserves the sum exactly."""
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    n = h.size
    for it in range(max_iters):
        t = init_temp * ((final_temp / init_temp) ** (it / max_iters))
        i, j = rng.integers(0, n, size=2)
        while j == i:
            j = rng.integers(0, n)
        # random direction
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_delta = min(h[src], 1.0 - h[dst])
        if max_delta <= 0:
            continue
        # scale move with temperature
        delta = rng.random() * max_delta * (t / init_temp)
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        c_new = compute_c5(h_new)
        if c_new < cur_c or rng.random() < np.exp(-(c_new - cur_c) / (t + 1e-12)):
            h, cur_c = h_new, c_new
            if c_new < best_c:
                best_c, best_h = c_new, h_new.copy()
    return best_h, best_c


def slsqp_optimize(h0, beta, max_iter=400):
    """SLSQP on the smooth soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _eq_sum}
    res = minimize(
        _soft_max,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny exact‑sum repair
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):

            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            if abs(add) < 1e-12:
                continue
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x
