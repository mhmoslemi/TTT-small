def smooth_obj(x, beta, target, lam):
    # x is in [0,1], shape (n,)
    corr = np.correlate(x, 1 - x, mode='full')
    # use scaling factor dx later?
    bc = beta * corr
    m = np.max(bc)
    t = m + np.log(np.sum(np.exp(bc - m))) / beta
    penalty = lam * (np.sum(x) - target) ** 2
    return t + penalty

def smooth_obj(x, beta, target, lam):
    # x may be outside [0,1], but we will clip
    x_clipped = np.clip(x, 0.0, 1.0)
    corr = np.correlate(x_clipped, 1 - x_clipped, mode='full')
    zeb = beta * corr
    m = np.max(zeb)
    expz = np.exp(zeb - m)
    sum_exp = np.sum(expz)
    soft = expz / sum_exp
    # smooth max approx: (m + np.log(sum_exp)) / beta
    smooth = (m + np.log(sum_exp)) / beta
    penalty = lam * (np.sum(x_clipped) - target) ** 2
    return smooth + penalty

def smooth_grad(x, beta, target, lam):
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    corr = np.correlate(x, 1 - x, mode='full')
    zeb = beta * corr
    m = np.max(zeb)
    expz = np.exp(zeb - m)
    sum_exp = np.exp(zeb - m).sum()
    weights = expz / sum_exp  # shape (2n-1)

    grad = np.zeros_like(x)
    # for each shift
    for shift in range(-n+1, n):
        w = weights[shift + n - 1]
        # first term: indices i where i+shift in range
        i_start = max(0, -shift)
        i_end = min(n-1, n-1 - shift)
        if i_start <= i_end:
            idx = slice(i_start, i_end+1)
            grad[idx] += w * (1 - x[idx + shift])
        # second term: indices i where i-shift in range
        i_start2 = max(0, shift)
        i_end2 = min(n-1, n-1 + shift)
        if i_start2 <= i_end2:
            idx2 = slice(i_start2, i_end2+1)
            grad[idx2] -= w * x[idx2 - shift]

    # penalty gradient
    grad += 2 * lam * (np.sum(x) - target)
    return grad

def hill_climb(h_start, rng, max_iters=1000, start_step=0.5):
    h = h_start.copy()
    best_c = compute_c5(h)
    best_h = h.copy()

            best_h = new_h.copy()
            h = new_h
        if (it+1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c

def gradient_refine(h, rng, step=0.02, max_iter=20):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = idx
        # compute gradient as prior
        # Actually previous gradient algorithm computed g for worst shift:
        # If k==0: g = 1 - 2 h
        # else: for i < n - k: g[i] += 1 - h[i+k]; for i>=k: g[i] -= h[i-k]
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            shift = k
            if shift > 0:
                g[:n-shift] += 1.0 - h[shift:]
                g[shift:] -= h[:n-shift]
            else:
                shift = -k
                # shift negative means correlation with left shift
                g[shift:] += 1.0 - h[:n-shift]
                g[:n-shift] -= h[shift:]
        # step size
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        # clip
        h = np.clip(h, 0.0, 1.0)
        # fix sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def simulated_annealing(h0, rng, deadline):
    n = h0.size
    target = n/2.0
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    T0 = max(0.01 * cur_c, 1e-4)
    it = 0
    while time.time() < deadline and it < 5000:
        step = max(0.5 * (0.999 ** it), 1e-4)
        cand_h = perturb_h(cur_h, rng, scale=step)
        cand_c = compute_c5(cand_h)
        if cand_c < cur_c:
            cur_h, cur_c = cand_h, cand_c
            if cand_c < best_c:
                best_h, best_c = cand_h.copy(), cand_c
        else:
            T = T0 * (0.999 ** it) + 1e-12
            if rng.random() < np.exp(-(cand_c - cur_c) / T):
                cur_h, cur_c = cand_h, cand_c
        it += 1
    return best_h, best_c

# ---------- basic utilities ----------

def compute_c5(h):
    """Exact discrete C5 for step function h on [0,2] with uniform spacing."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=1000, start_step=0.5):
    """Simple hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c

# ---------- smooth surrogate for max ----------

def smooth_obj_wrap(x, beta, target, lam):
    """Smooth approximation of max_k Σ_i x_i (1‑x_{i+k}) plus quadratic sum penalty."""
    x = np.clip(x, 0.0, 1.0)
    corr = np.correlate(x, 1 - x, mode="full")
    zeb = beta * corr
    m = np.max(zeb)
    expz = np.exp(zeb - m)
    sum_exp = expz.sum()
    smooth = (m + np.log(sum_exp)) / beta
    penalty = lam * (np.sum(x) - target) ** 2
    return smooth + penalty


def smooth_grad_wrap(x, beta, target, lam):
    """Gradient of the smooth objective."""
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    corr = np.correlate(x, 1 - x, mode="full")
    zeb = beta * corr
    m = np.max(zeb)
    expz = np.exp(zeb - m)
    sum_exp = expz.sum()
    weights = expz / sum_exp  # shape (2n‑1)
    grad = np.zeros_like(x)

    for shift in range(-n + 1, n):
        w = weights[shift + n - 1]
        i_start = max(0, -shift)
        i_end = min(n - 1, n - 1 - shift)
        if i_start <= i_end:
            idx = slice(i_start, i_end + 1)
            grad[idx] += w * (1 - x[idx + shift])
        i_start2 = max(0, shift)
        i_end2 = min(n - 1, n - 1 + shift)
        if i_start2 <= i_end2:
            idx2 = slice(i_start2, i_end2 + 1)
            grad[idx2] -= w * x[idx2 - shift]

    grad += 2.0 * lam * (np.sum(x) - target)
    return grad
