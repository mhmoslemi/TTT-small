    corr = np.correlate(h, h, mode='full')

def binary_hill_climb(h, rng, max_iters=5000, temp_start=0.1, temp_end=1e-4):
    best = h.copy()
    best_c = compute_c5(best)
    cur = h.copy()
    cur_c = best_c
    n = len(h)
    # Precompute indices of 1s and 0s for faster sampling
    # We'll update them after each swap.
    # Use arrays of indices for ones and zeros.
    ones = np.flatnonzero(cur == 1)
    zeros = np.flatnonzero(cur == 0)
    for it in range(max_iters):
        # Temperature schedule
        t = temp_start + (temp_end - temp_start) * (it / max_iters)
        # Propose swap
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # create new vector by swapping
        cur[i] = 0.0
        cur[j] = 1.0
        # Update indices
        # Remove i from ones, add to zeros; remove j from zeros, add to ones
        ones = ones[ones != i]  # remove i
        zeros = np.append(zeros, i)  # i now zero
        zeros = zeros[zeros != j]  # remove j
        ones = np.append(ones, j)   # j now one
        # compute c
        c_new = compute_c5(cur)
        # decide accept
        if c_new < cur_c or rng.random() < np.exp((cur_c - c_new) / (t + 1e-12)):
            # accept
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best = cur.copy()
        else:
            # reject: revert swap
            cur[i] = 1.0
            cur[j] = 0.0
            # revert indices
            ones = ones[ones != j]
            zeros = np.append(zeros, j)
            zeros = zeros[zeros != i]
            ones = np.append(ones, i)
            # Actually we can just recompute ones/zeros after revert.
            # Simpler: recompute
            ones = np.flatnonzero(cur == 1)
            zeros = np.flatnonzero(cur == 0)
    return best, best_c

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb(h_start, rng, max_iters=1000, start_step=0.3):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it+1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c

import time
import math
import numpy as np
from scipy.optimize import minimize

# ---------- Utility functions ----------
def compute_c5(h):
    """Exact discrete C5 for a step function h on [0,2] (dx = 2/len(h))."""
    n = h.size
    # autocorrelation of h with itself (linear, full mode)
    corr = np.correlate(h, h, mode="full")
    autocorr = corr[n - 1 :]            # shifts k = 0..n-1
    # ignore k=0 (gives sum(h)), we need the minimal overlap for k>0
    if n > 1:
        min_autocorr = np.min(autocorr[1:])
    else:
        min_autocorr = autocorr[0]
    max_overlap = np.sum(h) - min_autocorr
    return max_overlap * (2.0 / n)


def project_to_feasible(v, target):
    """Projection of v onto {0≤h≤1, Σh = target} via bisection on a shift."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Generate a random feasible vector (0≤h≤1, Σh=n/2)."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_to_feasible(v, n / 2.0)


def random_binary_vector(n, rng):
    """Generate a binary feasible vector with exactly n/2 ones."""
    ones = rng.choice(n, n // 2, replace=False)
    h = np.zeros(n, dtype=np.float64)
    h[ones] = 1.0
    return h


# ---------- Binary hill‑climb (simulated annealing) ----------
def binary_hill_climb(h0, rng, max_iters, time_limit):
    n = h0.size
    cur = h0.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    start = time.time()
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        # linear temperature schedule (0.1 → 0)
        t = 0.1 * (1.0 - it / max_iters)
        # propose a swap 1↔0
        i = rng.integers(n)
        while cur[i] < 0.5:      # need a 1
            i = rng.integers(n)
        j = rng.integers(n)
        while cur[j] > 0.5:      # need a 0
            j = rng.integers(n)
        cur[i] = 0.0
        cur[j] = 1.0
        c_new = compute_c5(cur)
        if c_new < cur_c or (t > 0 and rng.random() < math.exp((cur_c - c_new) / t)):
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best = cur.copy()
        else:
            # revert
            cur[i] = 1.0
            cur[j] = 0.0
    return best, best_c


# ---------- Continuous local search ----------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.3):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


# ---------- Soft‑max surrogate + SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    # smooth approx of max_k Σ_i x_i (1 - x_{i+k})
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # fix sum exactly (simple projection)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x
