        h[indices] = 1.0
        return h

                h[flip] = 1

                flip = rng.choice(one_idxs, size=-diff, replace=False)

    final_c = compute_c5(best_h)

import time
import numpy as np

def _c5_of(h):
    """Exact C₅ for a step function on a uniform grid."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def _thue_morse(n):
    """Binary Thue‑Morse word (0/1) of length n."""
    seq = np.fromiter(((i.bit_count() & 1) for i in range(n)), dtype=np.int8)
    return seq

    n = 200                     # even, half‑ones = 100
    half = n // 2
    target_O = int(np.floor(0.3808 * half))   # we need O_max ≤ this

    # --------------------------------------------------------------
    # Build a small pool of starting binary vectors
    # --------------------------------------------------------------
    starters = []

    # use the parent construction if it exists (convert to binary)
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0_bin = (h0 >= 0.5).astype(np.int8)
        cur = h0_bin.sum()
        while cur < half:
            z = np.where(h0_bin == 0)[0]
            h0_bin[rng.choice(z)] = 1
            cur += 1
        while cur > half:
            o = np.where(h0_bin == 1)[0]
            h0_bin[rng.choice(o)] = 0
            cur -= 1
        starters.append(h0_bin)

    # Thue‑Morse (balanced for even n)
    tm = _thue_morse(n)
    if tm.sum() != half:
        # adjust the count by flipping the smallest‑indexed bits
        diff = half - tm.sum()
        if diff > 0:
            zeros = np.where(tm == 0)[0]
            tm[rng.choice(zeros, size=diff, replace=False)] = 1
        else:
            ones = np.where(tm == 1)[0]
            tm[rng.choice(ones, size=-diff, replace=False)] = 0
    starters.append(tm)

    # a few random feasible vectors
    for _ in range(4):
        vec = np.zeros(n, dtype=np.int8)
        vec[rng.choice(n, size=half, replace=False)] = 1
        starters.append(vec)

    # pick the best starter
    best_h = None
    best_O = n + 1
    for h in starters:
        O = np.max(np.correlate(h, 1 - h, mode="full"))
        if O < best_O:
            best_O = O
            best_h = h.copy()

    cur_h = best_h.copy()
    cur_O = best_O

    # --------------------------------------------------------------
    # Simulated‑annealing / greedy descent driven by the worst lag
    # --------------------------------------------------------------
    no_improve = 0
    max_no_improve = 4000
    while time.time() - start < budget_s * 0.95:
        # update global best
        if cur_O < best_O:
            best_O, best_h = cur_O, cur_h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if best_O <= target_O:
            break

        # locate the lag giving the current maximum overlap
        corr = np.correlate(cur_h, 1 - cur_h, mode="full")
        idx = corr.argmax()
        lag = idx - (n - 1)
        if lag == 0:                     # degenerate – random perturbation
            i, j = rng.choice(n, size=2, replace=False)
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
            cur_O = np.max(np.correlate(cur_h, 1 - cur_h, mode="full"))
            continue

        L = abs(lag)
        # mismatched (1,0) pairs for the direction of the lag
        if lag > 0:
            cand = np.where((cur_h[: n - L] == 1) & (cur_h[L:] == 0))[0]
        else:
            cand = np.where((cur_h[: n - L] == 0) & (cur_h[L:] == 1))[0]

        if cand.size == 0:               # should not happen, but safeguard
            i = rng.integers(0, n - L)
        else:
            i = rng.choice(cand)
        j = i + L

        # propose the swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_corr = np.correlate(cur_h, 1 - cur_h, mode="full")
        new_O = new_corr.max()

        # acceptance rule (annealing)
        delta = new_O - cur_O
        accept = False
        if delta <= 0:
            accept = True
        else:
            elapsed = time.time() - start
            frac = elapsed / (budget_s * 0.95)
            T = max(0.05, (1.0 - frac) * 4.0)   # start ~4, cool to 0.05
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur_O = new_O
        else:
            # revert the swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

        # occasional random restart if stuck
        if no_improve >= max_no_improve:
            cur_h[:] = 0
            cur_h[rng.choice(n, size=half, replace=False)] = 1
            cur_O = np.max(np.correlate(cur_h, 1 - cur_h, mode="full"))
            no_improve = 0

    # final polishing with a short greedy hill‑climb
    def _hill_climb(h, steps=3000):
        O = np.max(np.correlate(h, 1 - h, mode="full"))
        for _ in range(steps):
            i, j = rng.choice(n, size=2, replace=False)
            if h[i] == h[j]:
                continue
            h[i], h[j] = h[j], h[i]
            new_O = np.max(np.correlate(h, 1 - h, mode="full"))
            if new_O <= O:
                O = new_O
            else:
                h[i], h[j] = h[j], h[i]
        return h, O

    best_h, best_O = _hill_climb(best_h, steps=4000)

    final_c5 = _c5_of(best_h.astype(float))
    return best_h.astype(float).tolist(), float(final_c5), int(n)
