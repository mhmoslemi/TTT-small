def project_onto_simplex_box(v, target):
    # want to find x = clip(v + lam, 0, 1) such that sum x = target
    # solve for lam with bisection because function f(lam)=sum clip(v+lam,0,1) is monotonic increasing in lam.
    # Use bisection between lo=-max(v) (so all entries <=0 after shift) and hi=1 - min(v) (so all entries >=1 after shift). Then binary search until diff small.
    # Complexity O(n log epsilon).
    lo = -np.max(v) - 1.0  # ensure v+lo <= 0
    hi = 1.0 - np.min(v) + 1.0  # ensure v+hi >= 1
    for _ in range(50):
        lam = (lo + hi) / 2.0
        x = v + lam
        np.clip(x, 0.0, 1.0, out=x)
        s = x.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    x = v + (lo + hi) / 2.0
    np.clip(x, 0.0, 1.0, out=x)
    return x

def project_onto_simplex_box(v, target):
    # find lam via bisection
    # lower bound lam_low such that sum clip(v + lam_low, 0, 1) <= target
    # upper bound lam_high such that sum >= target
    # We can start with lam_low = -np.max(v) - 1 (ensures all v+lam <= 0 => sum 0)
    # lam_high = (1 - np.min(v)) + 1 (ensures all v+lam >=1 => sum n)
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        projected = v + lam
        np.clip(projected, 0.0, 1.0, out=projected)
        s = projected.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    projected = v + (lo + hi) / 2.0
    np.clip(projected, 0.0, 1.0, out=projected)
    return projected

def softmax_pgd(h, rng, target, betas, iter_per_beta, step0):
   h_cur = h.copy()
   for beta, iters in zip(betas, iter_per_beta):
       for it in range(iters):
           # compute O, p, gradient
           O = np.correlate(h_cur, 1 - h_cur, mode='full')
           maxO = O.max()
           exp_scaled = np.exp(beta * (O - maxO))
           p = exp_scaled / exp_scaled.sum()
           grad_full = np.correlate(p, 1 - h_cur, mode='full')
           g = grad_full[len(h_cur)-1 : len(h_cur)-1 + len(h_cur)]
           # step size may be decreasing with it 
           step = step0 * (0.95 ** it)
           h_new = h_cur - step * g
           # project
           h_cur = project_onto_simplex_box(h_new, target)
   return h_cur

import time
import numpy as np

# ------------------- basic utilities ------------------- #
def compute_c5(h):
    """Exact discrete C5 for step function h on [0,2] with uniform spacing."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


def random_feasible_vector(n, rng):
    """Uniform random vector in [0,1] with exact sum n/2."""
    h = rng.random(n)
    target = n / 2.0
    h = h * (target / h.sum())
    # project onto box + equality (simple bisection)
    return _project_onto_simplex_box(h, target)


def random_binary_vector(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h


# ------------------- projection ------------------- #
def _project_onto_simplex_box(v, target, max_iter=60):
    """
    Project v onto the set {x : 0<=x_i<=1, sum x_i = target}.
    Uses bisection on the common shift lambda.
    """
    lo = -np.max(v) - 1.0      # ensures v+lo <= 0
    hi = 1.0 - np.min(v) + 1.0  # ensures v+hi >= 1
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        x = v + lam
        np.clip(x, 0.0, 1.0, out=x)
        s = x.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    x = v + (lo + hi) * 0.5
    np.clip(x, 0.0, 1.0, out=x)
    return x


# ------------------- soft‑max PGD ------------------- #
def _softmax_pgd(h, target, betas, iters_per_beta, step0_dict, rng):
    """Projected gradient descent on the soft‑max surrogate of max_k O(k)."""
    n = h.size
    for beta, nit in zip(betas, iters_per_beta):
        step0 = step0_dict.get(beta, 0.2)
        for it in range(nit):
            # correlation O(k)
            O = np.correlate(h, 1.0 - h, mode="full")
            maxO = O.max()
            # soft‑max weights
            exp_scaled = np.exp(beta * (O - maxO))
            p = exp_scaled / exp_scaled.sum()
            # gradient = corr(p, 1 - h) evaluated at shifts 0..n-1
            grad_full = np.correlate(p, 1.0 - h, mode="full")
            g = grad_full[n - 1 : n - 1 + n]
            # decaying step size
            step = step0 * (0.95 ** it)
            h_new = h - step * g
            h = _project_onto_simplex_box(h_new, target)
    return h


# ------------------- simulated annealing ------------------- #
def simulated_annealing(h0, rng, deadline):
    """Simple SA that respects the sum constraint."""
    n = h0.size
    target = n / 2.0
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    T0 = max(0.01 * cur_c, 1e-4)
    it = 0
    while time.time() < deadline and it < 8000:
        step = max(0.5 * (0.999 ** it), 5e-4)
        cand_h = perturb_h(cur_h, rng, scale=step)
        cand_c = compute_c5(cand_h)
        accept = False
        if cand_c < cur_c:
            accept = True
        else:
            T = T0 * (0.999 ** it) + 1e-12
            if rng.random() < np.exp(-(cand_c - cur_c) / T):
                accept = True
        if accept:
            cur_h, cur_c = cand_h, cand_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
        it += 1
    return best_h, best_c

        if up_n <= 300:
            x_src = np.linspace(0.0, 2.0, best_n, endpoint=False)
            x_up = np.linspace(0.0, 2.0, up_n, endpoint=False)
            h_up = np.interp(x_up, x_src, best_h)
            h_up = _project_onto_simplex_box(h_up, up_n / 2.0)

            # a brief PGD on the up‑sampled grid
            h_up = _softmax_pgd(
                h_up,
                up_n / 2.0,
                betas[:5],               # fewer betas for speed
                [200, 160, 120, 80, 60],
                {b: step0_dict[b] for b in betas[:5]},
                rng,
            )
            # final local refinement
            h_up, c_up = hill_climb(h_up, rng, max_iters=1200, start_step=0.4)
            if c_up < best_c:
                best_h, best_c, best_n = h_up, c_up, up_n

    # ----- final safety checks ----- #
    if best_h is None:
        best_n = 120
        best_h = np.full(best_n, 0.5)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_onto_simplex_box(best_h, best_n / 2.0)

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
