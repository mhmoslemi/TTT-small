def simulated_annealing(h0, target_sum, time_limit, rng, start_temp=0.001, alpha=0.9995, min_temp=1e-7, scale=0.2):
    n = h0.size
    h = h0.copy()
    c_cur = compute_c5_fast(h)
    best_h = h.copy()
    best_c = compute_c5_exact(h)
    T = start_temp
    iter_count = 0
    while time.time() < time_limit:
        # random move
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # max epsilon
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = max_eps * rng.random() * scale
        # apply move
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        # clip just in case
        # Not needed as eps ensures bounds but rounding may cause slight violation
        # compute new objective using fast
        c_new = compute_c5_fast(h_new)
        delta = c_new - c_cur
        # acceptance
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_new
            c_cur = c_new
        # update best if exact is better
        if time.time() < time_limit:  # compute if needed
            c_exact = compute_c5_exact(h)
            if c_exact < best_c:
                best_c = c_exact
                best_h = h.copy()
        # temperature schedule
        T = max(T * alpha, min_temp)
        iter_count += 1
    return best_h, best_c

def enforce_sum(h, target, rng):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # adjust using random index; if diff>0 add to some underfull, if diff<0 subtract from some overfull.
    # We'll loop limited times to satisfy.
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
            if idx.size == 0: break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        else:
            idx = np.where(h > 0.0)[0]
            if idx.size == 0: break
            i = rng.choice(idx)
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    # final correction if needed:
    if abs(diff) > 1e-9:
        i = rng.integers(0, h.size)
        h[i] = min(1.0, max(0.0, h[i] + diff))
    # ensure precision
    return h

def hill_climb_targeted(h, rng, max_iters, step):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5_exact(h)
    cur_h = h.copy()
    cur_c = best_c
    for it in range(max_iters):
        corr = np.correlate(cur_h, 1 - cur_h, mode='full')
        idx = np.argmax(corr)
        max_corr = corr[idx]
        c_cur = max_corr * (2.0 / n)
        k = idx - (n - 1)
        # gradient
        grad = np.ones(n)
        if k >= 0:
            # pairwise: h[0:n-k] vs h[k:n] both contributions
            grad[:n - k] -= cur_h[k:]
            grad[k:] -= cur_h[:n - k]
        else:
            kk = -k
            grad[:n - kk] -= cur_h[kk:]
            grad[kk:] -= cur_h[:n - kk]
        # pick i from positive gradients where h>0, j from negative gradients where h<1
        pos_idx = np.where((grad > 1e-9) & (cur_h > 1e-12))[0]
        neg_idx = np.where((grad < -1e-9) & (cur_h < 1.0 - 1e-12))[0]
        if pos_idx.size == 0 or neg_idx.size == 0:
            # fallback random move
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            max_eps = min(cur_h[i], 1.0 - cur_h[j])
            if max_eps <= 0:

import time
import numpy as np

# ---------- basic utilities ----------
def enforce_sum(h, target, rng):
    """Clip to [0,1] and adjust to make Σh exactly target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # simple greedy repair
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        else:
            idx = np.where(h > 0.0)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    if abs(diff) > 1e-9:  # final tiny correction
        i = rng.integers(0, h.size)
        h[i] = min(1.0, max(0.0, h[i] + diff))
    return h


def random_feasible_vector(n, rng):
    """Uniform random vector in [0,1]^n with Σh = n/2."""
    h = rng.random(n)
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def compute_c5_exact(h):
    """C5 using the reference np.correlate implementation."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def compute_c5_fast(h):
    """C5 using FFT (linear correlation). Faster for many evaluations."""
    n = h.shape[0]
    g = 1.0 - h
    m = 2 * n - 1
    L = 1 << (m - 1).bit_length()          # next power of two
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(g, L)
    corr = np.fft.irfft(H * np.conj(G), L)[:m]
    return np.max(corr) * (2.0 / n)


# ---------- simulated annealing ----------
def simulated_annealing(h0, target_sum, time_limit, rng,
                       start_temp=1e-3, alpha=0.9995, min_temp=1e-8,
                       scale=0.2, check_interval=200):
    n = h0.size
    h = h0.copy()
    cur_c = compute_c5_fast(h)
    best_h = h.copy()
    best_c = compute_c5_exact(h)
    T = start_temp
    it = 0
    while time.time() < time_limit:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = max_eps * rng.random() * scale
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        new_c = compute_c5_fast(h_new)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_new
            cur_c = new_c
        if it % check_interval == 0:
            h = enforce_sum(h, target_sum, rng)
            exact = compute_c5_exact(h)
            if exact < best_c:
                best_c = exact
                best_h = h.copy()
        T = max(T * alpha, min_temp)
        it += 1
    best_h = enforce_sum(best_h, target_sum, rng)
    best_c = compute_c5_exact(best_h)
    return best_h, best_c


# ---------- targeted hill climbing ----------
def hill_climb_targeted(h0, rng, time_limit):
    n = h0.size
    best_h = h0.copy()
    best_c = compute_c5_exact(best_h)
    cur_h = h0.copy()
    cur_c = best_c
    step = 0.5
    while time.time() < time_limit:
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        idx = np.argmax(corr)
        k = idx - (n - 1)

        # gradient of the worst shift
        grad = np.ones(n)
        if k >= 0:
            grad[: n - k] -= cur_h[k:]
            grad[k:] -= cur_h[: n - k]
        else:
            kk = -k
            grad[: n - kk] -= cur_h[kk:]
            grad[kk:] -= cur_h[: n - kk]

        pos = np.where((grad > 1e-9) & (cur_h > 1e-12))[0]
        neg = np.where((grad < -1e-9) & (cur_h < 1.0 - 1e-12))[0]

        if pos.size and neg.size:
            i = rng.choice(pos)
            j = rng.choice(neg)
        else:
            i = rng.integers(0, n)
            j = rng.integers(0, n)

        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0:
            continue
        eps = max_eps * step * rng.random()
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps

        new_c = compute_c5_exact(new_h)
        if new_c < cur_c:
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
        step = max(step * 0.99, 0.001)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # ---- initial vector ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        h0 = enforce_sum(np.clip(h0, 0.0, 1.0), n0 / 2.0, rng)
    else:
        n0 = 360
        h0 = random_feasible_vector(n0, rng)

    best_h, best_c = h0, compute_c5_exact(h0)

    # ---- Phase 1: SA at base resolution ----
    phase1_end = start + budget_s * 0.45
    h_sa, c_sa = simulated_annealing(best_h, best_h.size / 2.0,
                                     phase1_end, rng,
                                     start_temp=1e-3, alpha=0.9995,
                                     scale=0.2, check_interval=200)
    if c_sa < best_c:
        best_h, best_c = h_sa, c_sa

    # ---- Phase 2: Upsample and SA ----
    now = time.time()
    if now < start + budget_s * 0.80:
        n_up = best_h.size * 2
        xs_old = (np.arange(best_h.size) + 0.5) * (2.0 / best_h.size)
        xs_up = (np.arange(n_up) + 0.5) * (2.0 / n_up)
        h_up = np.interp(xs_up, xs_old, best_h)
        h_up = enforce_sum(np.clip(h_up, 0.0, 1.0), n_up / 2.0, rng)
        phase2_end = start + budget_s * 0.80
        h_sa2, c_sa2 = simulated_annealing(h_up, n_up / 2.0,
                                           phase2_end, rng,
                                           start_temp=1e-3, alpha=0.9995,
                                           scale=0.2, check_interval=200)
        if c_sa2 < best_c:
            best_h, best_c = h_sa2, c_sa2

    # ---- Phase 3: Targeted hill climbing ----
    now = time.time()
    if now < start + budget_s * 0.95:
        hc_end = start + budget_s * 0.95
        h_hc, c_hc = hill_climb_targeted(best_h, rng, hc_end)
        if c_hc < best_c:
            best_h, best_c = h_hc, c_hc

    # ---- Phase 4: Tiny random refinements ----
    while time.time() < start + budget_s * 0.99:
        i = rng.integers(0, best_h.size)
        j = rng.integers(0, best_h.size)
        if i == j:
            continue
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0:
            continue
        eps = max_eps * rng.random() * 0.02
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        c_cand = compute_c5_exact(cand)
        if c_cand < best_c:
            best_h, best_c = cand, c_cand

    # final clean‑up
    best_h = enforce_sum(np.clip(best_h, 0.0, 1.0), best_h.size / 2.0, rng)
    final_c = compute_c5_exact(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
