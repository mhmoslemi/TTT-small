def enforce_sum(h, target, rng):
    """Adjust h in-place so that sum(h) == target while keeping 0 <= h <= 1."""
    diff = target - h.sum()
    n = h.size
    # simple greedy correction
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    # start from uniform 0.5 and perform a random walk that preserves the sum
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    enforce_sum(h, n / 2.0, rng)
    return h

def improve_worst_lag(h, rng, scale=0.1):
    """Move mass from a high spot to the pair that hurts the worst lag."""
    n = h.size
    corr = full_correlation(h)
    k = int(np.argmax(corr) - (n - 1))        # lag with maximal overlap (worst)
    # indices participating in that lag (non‑circular)
    idx = np.arange(n)
    if k >= 0:
        mask = idx + k < n
        i_vals = idx[mask]
        j_vals = i_vals + k
    else:
        kp = -k
        mask = idx + kp < n
        j_vals = idx[mask]
        i_vals = j_vals + kp
    if i_vals.size == 0:
        return h.copy()
    prod = h[i_vals] * h[j_vals]
    worst_pair = np.argmin(prod)            # smallest product -> most to improve
    i = i_vals[worst_pair]
    j = j_vals[worst_pair]

    # donor: a location with relatively large value not equal to i,j
    top = np.argsort(-h)[:12]               # 12 largest entries
    donor = None
    for cand in top:
        if cand not in (i, j) and h[cand] > 1e-12:
            donor = cand
            break
    if donor is None:
        return h.copy()

    # how much can we raise i and j?
    eps_i = 1.0 - h[i]
    eps_j = 1.0 - h[j]
    eps = min(eps_i, eps_j, h[donor] / 2.0) * scale
    if eps <= 0.0:
        return h.copy()

    nh = h.copy()
    nh[i] += eps
    nh[j] += eps
    nh[donor] -= 2.0 * eps
    nh = np.clip(nh, 0.0, 1.0)

    # tiny sum correction
    target = n / 2.0
    diff = target - nh.sum()
    if abs(diff) > 1e-12:

        for _ in range(5):
            p = rng.integers(0, n)
            if p in (i, j, donor):
                continue
            add = min(diff, 1.0 - nh[p]) if diff > 0 else max(diff, -nh[p])
            nh[p] += add
            diff = target - nh.sum()
            if abs(diff) < 1e-12:
                break

def hill_climb(h_start, rng, max_iters=2000, start_step=0.4):
    """Greedy hill‑climb that only accepts improvements."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h

            step = max(step * 0.7, 1e-6)
    return best_h, best_c


def simulated_annealing(start_h, rng, deadline, T0=0.03, Tf=1e-5, max_iters=1800):
    cur_h = start_h.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        new_h = perturb_h(cur_h, rng, scale=0.35)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
    return best_h, best_c

# Smooth max surrogate + SLSQP
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """β‑smoothed max of the overlap values."""
    n = x.size
    # compute correlation using FFT (fast)
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(x, L)
    fr = np.fft.rfft((1.0 - x)[::-1], L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    bc = CURRENT_BETA * conv
    m = bc.max()
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=400):
    """Run SLSQP on the smoothed objective for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # final sum correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ----------------------------------------------------------------------
# Upsampling utilities
# ----------------------------------------------------------------------
def upsample_linear(h, factor, rng):
    n = h.size
    new_n = n * factor
    xs_old = (np.arange(n) + 0.5) * (2.0 / n)
    xs_new = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    h_new = np.interp(xs_new, xs_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    enforce_sum(h_new, new_n / 2.0, rng)
    return h_new


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # reserve a small margin for final clean‑up
    main_deadline = start_time + budget_s * 0.86
    upsample_deadline = start_time + budget_s * 0.96

    # ------------------------------------------------------------------
    # 1. Initial vector (use parent if present)
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        enforce_sum(h0, n / 2.0, rng)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ------------------------------------------------------------------
    # 2. Main stochastic optimisation loop
    # ------------------------------------------------------------------
    beta_schedule = [5, 20, 80, 200, 800, 2000]

    while time.time() < main_deadline:
        # ---------- choose a starting point ----------
        if rng.random() < 0.6:
            start = perturb_h(best_h, rng, scale=0.3)
        else:
            start = random_feasible_vector(n, rng)

        # ---------- quick targeted tweak ----------
        start = improve_worst_lag(start, rng, scale=0.07)

        # ---------- simulated annealing ----------
        cand_h, cand_c = simulated_annealing(
            start, rng, main_deadline, T0=0.04, Tf=1e-5, max_iters=2000
        )
        if cand_c < best_c:
            best_h, best_c = cand_h, cand_c

        # ---------- hill‑climb refinement ----------
        hc_h, hc_c = hill_climb(best_h, rng, max_iters=2500, start_step=0.35)
        if hc_c < best_c:
            best_h, best_c = hc_h, hc_c

        # ---------- SLSQP ladder ----------
        for beta in beta_schedule:
            if time.time() >= main_deadline:
                break
            try:
                opt_h = slsqp_optimize(best_h, beta, max_iter=450)
                opt_c = compute_c5(opt_h)
                if opt_c < best_c:
                    best_h, best_c = opt_h, opt_c
            except Exception:
                continue

        # ---------- extra targeted tries ----------
        for _ in range(3):
            tw_h = improve_worst_lag(best_h, rng, scale=0.05)
            tw_c = compute_c5(tw_h)
            if tw_c < best_c:
                best_h, best_c = tw_h, tw_c
                break

    # ------------------------------------------------------------------
    # 3. Optional up‑sampling to a finer grid
    # ------------------------------------------------------------------
    if time.time() < upsample_deadline and n * 2 <= 720:
        up_factor = 2
        h_up = upsample_linear(best_h, up_factor, rng)
        n_up = h_up.size

        # refine on the finer grid
        hc_up_h, hc_up_c = hill_climb(h_up, rng, max_iters=5000, start_step=0.25)
        if hc_up_c < best_c:
            best_h, best_c, n = hc_up_h, hc_up_c, n_up

        for beta in beta_schedule:
            if time.time() >= start_time + budget_s:
                break
            try:
                opt_up = slsqp_optimize(best_h, beta, max_iter=450)
                opt_up_c = compute_c5(opt_up)
                if opt_up_c < best_c:
                    best_h, best_c, n = opt_up, opt_up_c, opt_up.size
            except Exception:
                continue

    # ------------------------------------------------------------------
    # 4. Final clean‑up
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
