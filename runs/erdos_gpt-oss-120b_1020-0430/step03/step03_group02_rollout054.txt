def weighted_gradient(h, beta):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    maxc = corr.max()
    expdiff = np.exp(beta * (corr - maxc))
    probs = expdiff / expdiff.sum()
    G = np.zeros_like(h)
    # Loop over shift indices
    for idx, w in enumerate(probs):
        if w == 0:
            continue
        shift = idx - (n - 1)
        if shift == 0:
            G += w * (1 - 2 * h)
        elif shift > 0:
            k = shift
            G[: n - k] -= w * h[k:]
            G[k:] -= w * h[: n - k]
        else:
            k = -shift
            G[: n - k] -= w * h[k:]
            G[k:] -= w * h[: n - k]
    return G, corr

import time
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# basic utilities
# --------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_to_feasible(v: np.ndarray, target: float) -> np.ndarray:
    """Projection of v onto {0 ≤ x ≤ 1, sum x = target} via bisection on a shift."""
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector projected onto the feasible set."""
    v = rng.random(n)
    return project_to_feasible(v, n / 2.0)


# --------------------------------------------------------------
# gradients
# --------------------------------------------------------------
def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap C_k = Σ h_i(1‑h_{i+k})."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones(n)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def weighted_gradient(h: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weighted gradient of max_k C_k."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    # soft‑max probabilities (avoid overflow)
    expdiff = np.exp(beta * (corr - m))
    probs = expdiff / expdiff.sum()
    G = np.zeros_like(h)

    for i, w in enumerate(probs):
        if w == 0.0:
            continue
        shift = i - (n - 1)
        if shift == 0:
            G += w * (1.0 - 2.0 * h)
        else:
            k = abs(shift)
            # both positive and negative shifts give the same contribution
            G[: n - k] -= w * h[k:]
            G[k:] -= w * h[: n - k]
    return G


# --------------------------------------------------------------
# small random transfer used in the final polishing stage
# --------------------------------------------------------------
def random_transfer(h: np.ndarray, rng: np.random.Generator, scale: float = 0.02) -> np.ndarray:
    grad = worst_gradient(h)
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


# --------------------------------------------------------------
# SLSQP refinement (soft‑max objective)
# --------------------------------------------------------------
def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int = 250) -> np.ndarray:
    n = h0.size
    target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        m = np.max(corr)
        exp_part = np.exp(beta * (corr - m))
        return (m + np.log(np.sum(exp_part))) / beta

    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # exact sum correction (quick redistribution)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x

                h = h - step * grad
                h = project_to_feasible(h, n / 2.0)
                c = compute_c5(h)
                if c < best_c:
                    best_c, best_h = c, h.copy()
            if time.time() - start_time > budget_s * 0.99:
                break

        # final cheap hill‑climb using the exact worst‑shift gradient
        for it in range(2000):
            if time.time() - start_time > budget_s * 0.99:
                break
            g = worst_gradient(h)
            i = int(np.argmax(g))
            j = int(np.argmin(g))
            if i == j:
                break
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0.0:
                break
            eps = max_eps * 0.3
            h[i] -= eps
            h[j] += eps
            c = compute_c5(h)
            if c < best_c:
                best_c, best_h = c, h.copy()
    return (best_h, best_c)


# --------------------------------------------------------------
# main entry point
# --------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    best_h = None
    best_c = np.inf
    best_n = None

    # initial solution from parent, if any
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        n0 = h0.size
        h0 = project_to_feasible(h0, n0 / 2.0)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0, c0, n0

    # candidate discretisations (prefer smaller first)
    cand_ns = [84, 96, 108, 120, 144, 168, 192, 216, 240]
    if best_n is not None:
        cand_ns = [best_n] + [n for n in cand_ns if n != best_n]

    # optimise each size while we have time
    for n in cand_ns:
        if time.time() - start > budget_s * 0.94:
            break

        # initialise for this n
        init_h = None
        if best_h is not None:
            if best_n == n:
                init_h = best_h.copy()
            else:
                x_old = np.linspace(0, 2, best_n, endpoint=False)
                x_new = np.linspace(0, 2, n, endpoint=False)
                init_h = np.interp(x_new, x_old, best_h)
                init_h = np.clip(init_h, 0.0, 1.0)
                init_h = project_to_feasible(init_h, n / 2.0)

        h_n, c_n = optimise_for_n(n, init_h, rng, start, budget_s)
        if h_n is not None and c_n < best_c:
            best_h, best_c, best_n = h_n, c_n, n

    # final SLSQP polishing if budget permits
    if best_h is not None and time.time() - start < budget_s * 0.90:
        best_h = slsqp_optimize(best_h, beta=2000.0, rng=rng, max_iter=300)
        best_c = compute_c5(best_h)

    # tiny random transfers until time runs out
    while time.time() - start < budget_s * 0.999 and best_h is not None:
        nh = random_transfer(best_h, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh, nc

    # final feasibility enforcement
    if best_h is None:
        best_n = 96
        best_h = np.full(best_n, 0.5)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_feasible(best_h, best_n / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
