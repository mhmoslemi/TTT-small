def gradient_biased_perturb(h, rng, scale=0.2, top_frac=0.2, bottom_frac=0.2):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    # identify shifts within 0.5% of max
    eps_rel = 0.001
    worst_shifts = np.where(corr >= max_corr * (1 - eps_rel))[0]
    # compute aggregated product per left index
    prod = np.zeros(n)
    for k_idx in worst_shifts:
        shift = k_idx - (n - 1)
        if shift >= 0:
            i_range = np.arange(0, n - shift)
            prod[i_range] += h[i_range] * (1 - h[i_range + shift])
        else:
            i_range = np.arange(-shift, n)
            prod[i_range] += h[i_range] * (1 - h[i_range + shift])
    if prod.max() == prod.min():
        # fallback to simple perturb
        return perturb_transfer(h, rng, scale)
    # determine thresholds
    high_thr = np.quantile(prod, 1 - top_frac)
    low_thr = np.quantile(prod, bottom_frac)
    high_idx = np.where(prod >= high_thr)[0]
    low_idx = np.where(prod <= low_thr)[0]
    if high_idx.size == 0 or low_idx.size == 0:
        return perturb_transfer(h, rng, scale)
    i = rng.choice(high_idx)
    j = rng.choice(low_idx)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_transfer(h, rng, scale)
    eps = max_eps * scale * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h
```

def simulated_annealing_phase(h0, rng, max_iter, init_temp, final_temp, bias_prob=0.8):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    n = h.size
    log_ratio = math.log(final_temp / init_temp) if init_temp > 0 else 0.0
    for it in range(1, max_iter + 1):
        T = init_temp * math.exp(log_ratio * it / max_iter)
        if rng.random() < bias_prob:
            cand = gradient_biased_perturb(h, rng, scale=0.2)
        else:
            cand = perturb_transfer(h, rng, scale=0.3)
        cand_c = compute_c5(cand)
        diff = cand_c - cur_c
        if diff < 0 or rng.random() < math.exp(-diff / max(T, 1e-12)):
            h, cur_c = cand, cand_c
            if cand_c < best_c:
                best_c = cand_c
                best_h = cand.copy()
    return best_h, best_c

import time, math, numpy as np
from scipy.optimize import minimize

# --------- global vars for SLSQP ---------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    # smooth approximation of max_{k} corr_k
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log-sum-exp in a stable way
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def enforce_sum(h, target, rng):
    """Adjust h in‑place so that sum(h)=target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    # simple random redistribution loop
    for _ in range(10000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            if add <= 0:
                continue
            h[i] += add
        else:
            sub = min(-diff, h[i])
            if sub <= 0:
                continue
            h[i] -= sub
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector satisfying 0≤h_i≤1 and sum = n/2."""
    h = np.full(n, 0.5, dtype=float)
    # perform many small random transfers that keep the sum unchanged
    for _ in range(5 * n):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # choose a direction uniformly
        if rng.random() < 0.5:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0:
                continue
            eps = max_eps * rng.random() * 0.5
            h[i] -= eps
            h[j] += eps
        else:
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0:
                continue
            eps = max_eps * rng.random() * 0.5
            h[i] += eps
            h[j] -= eps
    enforce_sum(h, n / 2.0, rng)
    return h


def compute_c5(h):
    """Exact discrete C5 for step function h (dx=2/n)."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / h.size)


def perturb_transfer(h, rng, scale=0.2):
    """Random signed transfer between two coordinates (preserves sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    if rng.random() < 0.5:
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            return h.copy()
        eps = max_eps * scale * rng.random()
        new = h.copy()
        new[i] -= eps
        new[j] += eps
    else:
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0:
            return h.copy()
        eps = max_eps * scale * rng.random()
        new = h.copy()
        new[i] += eps
        new[j] -= eps
    return new


def gradient_biased_perturb(h, rng, scale=0.2, top_frac=0.2, low_frac=0.2):
    """Bias transfer away from positions that contribute most to the worst overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    # consider shifts whose correlation is within 1 % of the maximum
    eps_rel = 0.01
    worst_idxs = np.where(corr >= max_corr * (1.0 - eps_rel))[0]
    prod = np.zeros(n, dtype=float)
    for idx in worst_idxs:
        shift = idx - (n - 1)
        if shift >= 0:
            i_range = np.arange(0, n - shift)
            prod[i_range] += h[i_range] * (1.0 - h[i_range + shift])
        else:
            i_range = np.arange(-shift, n)
            prod[i_range] += h[i_range] * (1.0 - h[i_range + shift])
    if prod.max() == prod.min():
        return perturb_transfer(h, rng, scale)
    high_thr = np.quantile(prod, 1.0 - top_frac)
    low_thr = np.quantile(prod, low_frac)
    high = np.where(prod >= high_thr)[0]
    low = np.where(prod <= low_thr)[0]
    if high.size == 0 or low.size == 0:
        return perturb_transfer(h, rng, scale)
    i = rng.choice(high)
    j = rng.choice(low)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_transfer(h, rng, scale)
    eps = max_eps * scale * rng.random()
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on smooth max‑approximation."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {"type": "eq", "fun": sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x


def simulated_annealing_phase(h0, rng, max_iter, init_temp, final_temp, bias_prob=0.85):
    """Simple SA using biased and uniform perturbations."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    log_ratio = math.log(final_temp / init_temp) if init_temp > 0 else 0.0
    for it in range(1, max_iter + 1):
        T = init_temp * math.exp(log_ratio * it / max_iter)
        if rng.random() < bias_prob:
            cand = gradient_biased_perturb(h, rng, scale=0.2)
        else:
            cand = perturb_transfer(h, rng, scale=0.3)
        cand_c = compute_c5(cand)
        diff = cand_c - cur_c
        if diff < 0 or rng.random() < math.exp(-diff / max(T, 1e-12)):
            h, cur_c = cand, cand_c
            if cand_c < best_c:
                best_c = cand_c
                best_h = cand.copy()
    return best_h, best_c

    start = time.time()

    target_n = 360
    if "initial_h_values" in globals():
        base = np.array(initial_h_values, dtype=float)
        n0 = base.size
        if n0 * 2 == target_n:
            h0 = np.repeat(base, 2)
        else:
            # simple linear interpolation to target_n
            x_old = np.linspace(0, 2, n0, endpoint=False)
            x_new = np.linspace(0, 2, target_n, endpoint=False)
            h0 = np.interp(x_new, x_old, base)
        n = target_n
        enforce_sum(h0, n / 2.0, rng)
    else:
        n = target_n
        h0 = random_feasible_vector(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ----- simulated annealing phase (≈78 % of budget) ----- #
    sa_budget = budget_s * 0.78
    while time.time() - start < sa_budget:
        # choose a start point
        if rng.random() < 0.65:
            start_vec = perturb_transfer(best_h, rng, scale=0.25)

            start_vec,
            rng,

                    break
                try:
                    refined = slsqp_optimize(best_h, beta, max_iter=350)
                    rc = compute_c5(refined)
                    if rc < best_c:
                        best_h, best_c = refined, rc
                except Exception:
                    continue

    # ----- final polishing (remaining budget) ----- #
    for beta in (2000, 8000, 20000):
        if time.time() - start > budget_s:
            break
        try:
            best_h = slsqp_optimize(best_h, beta, max_iter=800)
            best_c = compute_c5(best_h)
        except Exception:
            continue

    # final safety clamp
    best_h = np.clip(best_h, 0.0, 1.0)
    enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
