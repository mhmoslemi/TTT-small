def local_refine(h, rng, max_time, start_step=0.3):
    start_time = time.time()
    step = start_step
    best_c = compute_c5(h)
    n = len(h)
    while time.time() - start_time < max_time:
        # compute correlation
        corr = np.correlate(h, 1 - h, mode="full")
        k_max = np.argmax(corr)
        shift = k_max - (n - 1)  # shift offset; can be negative
        # Determine feasible i range
        if shift >= 0:
            i_range = range(0, n - shift)
        else:
            i_range = range(-shift, n)  # i = i+shift within 0..n-1
        # random trial attempts
        improved = False
        for _ in range(20):
            i = rng.integers(i_range.start, i_range.stop)
            j = i + shift
            if step == 0:
                break
            max_eps = min(h[i], 1 - h[j])
            if max_eps <= 0:
                continue
            eps = step * max_eps * rng.random()
            # if eps too low, skip
            if eps < 1e-12:
                continue
            new_h = h.copy()
            new_h[i] -= eps
            new_h[j] += eps
            new_c = compute_c5(new_h)
            if new_c < best_c:
                best_c = new_c
                h = new_h
                improved = True
                break
        if not improved:
            step *= 0.7
            if step < 1e-5:
                break
    return h, best_c

import time
import numpy as np
from scipy.optimize import minimize, Bounds

# globals used by the smoothed max objective
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_to_feasible(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto {0≤h≤1, Σh = target} by a simple bisection on the shift."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def upsample_initial(to_n: int) -> np.ndarray:
    """Linear interpolation of the global initial_h_values to length to_n."""
    if "initial_h_values" not in globals():
        raise RuntimeError("initial_h_values not available")
    init = np.array(initial_h_values, dtype=float)
    m = init.size
    xs_old = np.linspace(0.0, 2.0, m, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, to_n, endpoint=False)
    return np.interp(xs_new, xs_old, init)


def sum_eq_constraint(x: np.ndarray) -> float:
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x: np.ndarray) -> float:
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 800) -> np.ndarray:
    """Refine h0 with SLSQP using a smoothed max objective."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = Bounds(0.0, 1.0)
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum (projective correction)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def de_optimize(n: int, rng: np.random.Generator, max_time: float):
    """Simple Differential Evolution with projection onto the feasible set."""
    target = n / 2.0
    popsize = max(30, n // 3)
    pop = []

    # seed with a mixture of binary and fractional vectors
    while len(pop) < popsize:
        if rng.random() < 0.5:
            idx = rng.choice(n, size=n // 2, replace=False)
            v = np.zeros(n)
            v[idx] = 1.0
        else:
            v = rng.random(n)
            v = v / v.sum() * target
            v = project_to_feasible(v, target)
        pop.append(v)
    pop = np.array(pop)
    scores = np.array([compute_c5(ind) for ind in pop])
    best_idx = scores.argmin()
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]

    F = 0.7
    CR = 0.9
    start = time.time()
    while time.time() - start < max_time:
        for i in range(popsize):
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, 3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            trial = np.where(rng.random(n) < CR, mutant, pop[i])
            trial = project_to_feasible(trial, target)
            trial_score = compute_c5(trial)
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_h = trial.copy()
    return best_h, best_score


def local_refine(h: np.ndarray, rng: np.random.Generator,
                 max_time: float, start_step: float = 0.3):
    """Targeted hill‑climb that moves mass along the worst‑overlap shift."""
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(best_h)
    step = start_step
    start = time.time()
    while time.time() - start < max_time:
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        k_max = int(np.argmax(corr))
        shift = k_max - (n - 1)                     # signed shift
        if shift >= 0:
            i_start, i_end = 0, n - shift
        else:
            i_start, i_end = -shift, n
        improved = False
        for _ in range(30):
            i = rng.integers(i_start, i_end)
            j = i + shift
            max_eps = min(best_h[i], 1.0 - best_h[j])
            if max_eps <= 0.0:
                continue
            eps = step * max_eps * rng.random()
            if eps < 1e-12:
                continue
            new_h = best_h.copy()
            new_h[i] -= eps
            new_h[j] += eps
            new_c = compute_c5(new_h)
            if new_c < best_c:
                best_c = new_c
                best_h = new_h
                improved = True
                break
        if not improved:
            step *= 0.7
            if step < 1e-5:
                break
    # final feasibility projection
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_total = time.time()

    # -------- Coarse global search (DE) ----------
    n_coarse = 120
    time_coarse = min(300.0, (budget_s - (time.time() - start_total)) * 0.3)
    h_coarse, _ = de_optimize(n_coarse, rng, time_coarse)

    # -------- Upsample to fine resolution ----------
    n_fine = 300
    xs_coarse = np.linspace(0.0, 2.0, n_coarse, endpoint=False)
    xs_fine = np.linspace(0.0, 2.0, n_fine, endpoint=False)
    h_fine = np.interp(xs_fine, xs_coarse, h_coarse)

    # blend with the provided initial solution if available
    if "initial_h_values" in globals():
        h_init = upsample_initial(n_fine)
        h_fine = (h_fine + h_init) / 2.0

    h_fine = project_to_feasible(h_fine, n_fine / 2.0)

    # -------- SLSQP refinement schedule ----------
    beta_vals = [5, 20, 50, 200, 800, 2000, 8000]
    for beta in beta_vals:
        if time.time() - start_total > budget_s * 0.93:
            break
        h_fine = slsqp_optimize(h_fine, beta, rng, max_iter=800)

    # -------- Local refinement with targeted hill‑climb ----------
    remaining = budget_s - (time.time() - start_total)
    if remaining > 5.0:
        h_fine, _ = local_refine(h_fine, rng, max_time=remaining, start_step=0.3)

    # -------- Multiple short random restarts (if time permits) ----------
    while time.time() - start_total < budget_s * 0.98:
        remaining = budget_s - (time.time() - start_total)
        if remaining < 4.0:
            break
        # small random perturbation of current best
        trial = h_fine.copy()
        for _ in range(5):
            i, j = rng.integers(0, n_fine), rng.integers(0, n_fine)
            if i == j:
                continue
            max_eps = min(trial[i], 1.0 - trial[j])
            if max_eps <= 0.0:
                continue
            eps = max_eps * rng.random() * 0.05
            trial[i] -= eps
            trial[j] += eps
        trial = project_to_feasible(trial, n_fine / 2.0)
        # allocate a short refinement window
        sub_time = min(remaining * 0.25, 7.0)
        trial, _ = local_refine(trial, rng, max_time=sub_time, start_step=0.2)
        if compute_c5(trial) < compute_c5(h_fine):
            h_fine = trial

    # -------- Final feasibility enforcement ----------
    h_fine = np.clip(h_fine, 0.0, 1.0)
    target = n_fine / 2.0
    diff = target - h_fine.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n_fine)
            add = min(diff, 1.0 - h_fine[i]) if diff > 0 else max(diff, -h_fine[i])
            h_fine[i] += add
            diff = target - h_fine.sum()

    final_c = compute_c5(h_fine)

    return h_fine.tolist(), float(final_c), int(n_fine)
