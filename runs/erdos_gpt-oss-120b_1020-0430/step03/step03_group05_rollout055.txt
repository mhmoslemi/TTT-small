def improve_max_shift(h, max_iters, rng):
    n = len(h)
    best_h = h.copy()
    best_c = compute_c5(best_h)
    for iter in range(max_iters):
        # compute correlation and find shift of max
        corr_full = np.correlate(best_h, 1 - best_h, mode='full')
        # find index of max; could be multiple; take first
        imax = np.argmax(corr_full)
        # shift s
        s = imax - (n - 1)
        # overlapping indices
        if s >= 0:
            i_start = 0
            i_end = n - s
            i_range = np.arange(i_start, i_end)
            j_range = i_range + s
        else:
            s_abs = -s
            i_start = s_abs
            i_end = n
            i_range = np.arange(i_start, i_end)
            j_range = i_range + s
        # difference hi - hj
        diffs = best_h[i_range] - best_h[j_range]
        # want positive diffs for moving from hi to hj
        # pick index with max diff > 0
        positive = diffs > 1e-12
        if not np.any(positive):
            break
        # choose among top few
        idxs = np.where(positive)[0]
        # choose randomly among top 5 diffs
        # compute diff values
        diff_vals = diffs[positive]
        # select candidates with largest diff
        sel = np.argsort(-diff_vals)[:min(5, len(diff_vals))]
        chosen_i = i_range[positive][sel[rng.integers(len(sel))]]
        chosen_j = j_range[positive][sel[rng.integers(len(sel))]]
        # epsilon transfer
        max_eps = min(best_h[chosen_i], 1.0 - best_h[chosen_j])
        if max_eps <= 0:
            continue
        eps = max_eps * 0.5 * (rng.random())
        if eps <= 0:
            continue
        new_h = best_h.copy()
        new_h[chosen_i] -= eps
        new_h[chosen_j] += eps
        new_c = compute_c5(new_h)
        if new_c < best_c - 1e-9:
            best_h = new_h
            best_c = new_c
        # else maybe continue exploring
    return best_h, best_c

def simulated_annealing(h0, rng, max_iters, time_limit=None, start_temp=0.01, end_temp=0.0001):
    n = len(h0)
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    # log schedule: T = start_temp * (end_temp/start_temp)^{i/max_iters}
    for it in range(max_iters):
        # optionally break if time exceed
        if time_limit is not None and time.time() > time_limit:
            break
        # propose move: choose two indices i,j (i != j)
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0:
            continue
        eps = max_eps * rng.random()
        # if eps small (less than 1e-12), continue
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        # temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / max_iters)
        accept = False
        if new_c < cur_c:
            accept = True
        else:
            # Metropolis criterion
            if rng.random() < np.exp(-(new_c - cur_c) / T):
                accept = True
        if accept:
            cur_h = new_h
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
    return best_h, best_c

def simulated_annealing(h, rng, max_iters=5000, time_limit=None, start_temp=0.01, end_temp=0.0001):
    n = len(h)
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    for it in range(max_iters):
        if time_limit is not None and time.time() > time_limit:
            break
        # propose move
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        # compute max epsilon for transfer from i to j (i -> j)
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0:
            continue
        eps = max_eps * rng.random()
        if eps <= 0:
            continue
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        # temperature schedule linear in exponent:
        T = start_temp * (end_temp / start_temp) ** (it / max_iters)
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-12)):
            cur_h = new_h
            cur_c = new_c
            if cur_c < best_c:
                best_h = cur_h.copy()
                best_c = cur_c
    return best_h, best_c

def de_optimize(n, rng, max_time):

    # popsize
    popsize = max(15, min(80, n // 4))
    # initialize pop

        init_vec = project_to_feasible(init_vec, target)
        pop.append(init_vec)
    # fill rest
    while len(pop) < popsize:
        pop.append(random_feasible_vector(n, rng))
    pop = np.array(pop)
    scores = np.array([compute_c5(ind) for ind in pop])
    best_idx = scores.argmin()
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]
    # DE parameters
    F = 0.7
    CR = 0.9
    # time loop
    start = time.time()
    while time.time() - start < max_time:
        for i in range(popsize):
            # mutation
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, 3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            # crossover
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(0, n)] = True
            trial = np.where(cross, mutant, pop[i])
            # project feasible
            trial = project_to_feasible(trial, target)
            trial_score = compute_c5(trial)
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_h = trial.copy()
        # optional early break if no improvement after some cycles
        # but we can loop until time.
    return best_h, best_score

import time
import numpy as np

def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx


def project_to_feasible(v, target):
    """Project v onto {0≤h≤1, Σh = target} via bisection on a shift."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Generate a random feasible vector (0≤h≤1, Σh=n/2)."""
    v = rng.random(n)
    target = n / 2.0
    v = v / v.sum() * target
    return project_to_feasible(v, target)


def upsample_initial(to_n):
    """Linear interpolation of the global initial_h_values to length to_n."""
    init = np.array(initial_h_values, dtype=float)
    m = init.size
    if m == to_n:
        return init.copy()
    xs_old = np.linspace(0.0, 2.0, m, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, to_n, endpoint=False)
    return np.interp(xs_new, xs_old, init)


def de_optimize(n, rng, max_time):
    """Simple Differential Evolution with projection onto feasibility."""
    target = n / 2.0
    popsize = max(15, min(80, n // 4))
    pop = []

    # seed with upsampled initial if available
    if "initial_h_values" in globals():
        init_vec = upsample_initial(n)
        init_vec = project_to_feasible(init_vec, target)
        pop.append(init_vec)

    while len(pop) < popsize:
        pop.append(random_feasible_vector(n, rng))

    pop = np.array(pop)
    scores = np.array([compute_c5(ind) for ind in pop])
    best_idx = scores.argmin()
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]

    F = 0.7
    CR = 0.9
    start = time.time()
    while time.time() - start < max_time:
        for i in range(popsize):
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, 3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(0, n)] = True
            trial = np.where(cross, mutant, pop[i])
            trial = project_to_feasible(trial, target)
            trial_score = compute_c5(trial)
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_h = trial.copy()
    return best_h, best_score


def greedy_reduce_max_shift(h, rng, max_iters=2000):
    """Iteratively reduce the current worst‑shift contribution."""
    best_h = h.copy()
    best_c = compute_c5(best_h)
    n = best_h.size
    for _ in range(max_iters):
        corr_full = np.correlate(best_h, 1 - best_h, mode="full")
        imax = np.argmax(corr_full)
        s = imax - (n - 1)  # shift corresponding to the maximum
        if s >= 0:
            i_range = np.arange(0, n - s)
            j_range = i_range + s
        else:
            i_range = np.arange(-s, n)
            j_range = i_range + s
        diffs = best_h[i_range] - best_h[j_range]
        pos = np.where(diffs > 1e-12)[0]
        if pos.size == 0:
            break
        pos_diffs = diffs[pos]
        if pos.size > 5:
            top = np.argpartition(-pos_diffs, 5)[:5]
        else:
            top = np.arange(pos.size)
        chosen = pos[top[rng.integers(len(top))]]
        i = i_range[chosen]
        j = j_range[chosen]
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0:
            continue
        eps = max_eps * 0.5 * rng.random()
        if eps <= 0:
            continue
        new_h = best_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        if new_c + 1e-12 < best_c:
            best_h = new_h
            best_c = new_c
    return best_h, best_c


def simulated_annealing(h0, rng, max_iters=5000, start_temp=0.01, end_temp=0.0001):
    """Metropolis annealing using simple transfer moves."""
    n = h0.size
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    for it in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0:
            continue
        eps = max_eps * rng.random()
        if eps <= 0:
            continue
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        T = start_temp * (end_temp / start_temp) ** (it / max_iters)
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-12)):
            cur_h = new_h
            cur_c = new_c
            if cur_c < best_c:
                best_h = cur_h.copy()
                best_c = cur_c
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    best_h = None
    best_c = np.inf
    best_n = None

    # candidate discretisations (coarse → fine)
    candidate_ns = [60, 120, 180, 240, 300, 420, 540, 660]
    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        if n0 not in candidate_ns:
            candidate_ns = [n0] + candidate_ns

    for n in candidate_ns:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining < 8.0:
            break

        # allocate time for this n
        time_for_n = min(remaining * 0.5, max(8.0, n * 0.15))

        # --- Global search (DE) ---
        de_time = time_for_n * 0.5
        h_de, _ = de_optimize(n, rng, de_time)

        # --- Greedy local improvement ---
        h_g, _ = greedy_reduce_max_shift(h_de, rng, max_iters=2000 + n // 5)

        # --- Simulated annealing fine‑tuning ---
        h_s, _ = simulated_annealing(h_g, rng,
                                     max_iters=3000 + n // 3,
                                     start_temp=0.01,
                                     end_temp=0.0001)

        final_c = compute_c5(h_s)
        if final_c < best_c:
            best_c = final_c
            best_h = h_s.copy()
            best_n = n

        if time.time() - start_time > budget_s * 0.96:
            break

    # --- Upsampling refinement if time permits ---
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 5.0 and best_h is not None and best_n < 800:
        new_n = min(best_n * 2, 800)
        xs_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xs_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
        h_up = np.interp(xs_new, xs_old, best_h)
        h_up = project_to_feasible(h_up, new_n / 2.0)

        h_up_g, _ = greedy_reduce_max_shift(h_up, rng, max_iters=1500 + new_n // 5)
        h_up_s, _ = simulated_annealing(h_up_g, rng,
                                        max_iters=2000 + new_n // 3,
                                        start_temp=0.008,
                                        end_temp=0.00005)
        final_up_c = compute_c5(h_up_s)
        if final_up_c < best_c:
            best_c = final_up_c
            best_h = h_up_s.copy()
            best_n = new_n

    # --- Final feasibility enforcement ---
    if best_h is None:
        best_n = 60
        best_h = np.full(best_n, 0.5)

    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, best_n)
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
