def hill_climb_multi(h_start, rng, max_iters=2000, init_step=0.5, n_trials=8):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    # schedule step reducing every (max_iters//10)
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c - 1e-12:
                best_c = c
                h = new_h
                improved = True
                break
        if not improved and rng.random() < 0.05:
            new_h = perturb_h(h, rng, scale=step*2)
            c = compute_c5(new_h)
            if c < best_c:
                best_c = c
                h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return h, best_c

def smooth_obj(x, beta, target):
    # x in [0,1], penalize sum deviation
    corr = np.correlate(x, 1-x, mode='full')
    bc = beta * corr
    # smooth max via log-sum-exp
    m = np.max(bc)
    sm = (m + np.log(np.sum(np.exp(bc - m)))) / beta
    # penalty
    penalty = 1e3 * (np.sum(x) - target) ** 2
    return sm + penalty

def enforce_sum(h, target, rng, max_iter=5000):
    # Clip
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    if abs(diff) < 1e-12:
        return h
    # We'll adjust in loops:
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        # Choose random index where we can add/subtract
        i = rng.integers(0, n)
        if diff > 0:
            # need to increase sum
            room = 1.0 - h[i]
            if room <= 0:
                continue
            add = min(diff, room)
            h[i] += add
            diff -= add
        else:
            # need to decrease sum
            room = h[i]
            if room <= 0:
                continue
            sub = min(-diff, room)
            h[i] -= sub
            diff += sub
    # After loop, maybe small leftover diff due to rounding; distribute evenly
    if abs(diff) > 1e-12:
        # Distribute to all coordinates with capacity
        # Compute adjustable indices
        # For diff>0, increase where h<1
        # For diff<0, decrease where h>0
        must = diff
        # create array of capacity
        if must > 0:
            caps = 1.0 - h
        else:
            caps = h
        total_cap = caps.sum()
        if total_cap > 0:
            # distribute proportionally
            h += caps * (must / total_cap)
            h = np.clip(h, 0.0, 1.0)
    # final clamp
    h = np.clip(h, 0.0, 1.0)
    return h

import time
import numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h, target, rng, max_iter=5000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    if abs(diff) < 1e-12:
        return h
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            room = 1.0 - h[i]
            if room <= 0:
                continue
            add = min(diff, room)
            h[i] += add
            diff -= add
        else:
            room = h[i]
            if room <= 0:
                continue
            sub = min(-diff, room)
            h[i] -= sub
            diff += sub
    # small residual correction by proportional distribution
    if abs(diff) > 1e-12:
        if diff > 0:
            caps = 1.0 - h
        else:
            caps = h
        total_cap = caps.sum()
        if total_cap > 0:
            h += caps * (diff / total_cap)
    h = np.clip(h, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    h = rng.random(n)
    target = n / 2.0
    h = enforce_sum(h, target, rng)
    return h

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    # ensure it's within limits
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def slsqp_optimize(h0, beta, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # adjust randomly
        rng = np.random.default_rng()
        i = rng.integers(0, n)
        x[i] += diff
        x = np.clip(x, 0.0, 1.0)
    return x

def anneal_optimize(h0, rng, end_time):
    h = h0.copy()
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    n = h.size
    target = n / 2.0
    # schedule parameters
    T_start = 0.015
    T_end = 0.0005
    max_steps = 200000
    step = 0
    while time.time() < end_time and step < max_steps:
        progress = step / max_steps
        T = T_start * (1 - progress) + T_end * progress
        scale = 0.5 * (1 - progress) + 0.02 * progress
        new_h = perturb_h(h, rng, scale=scale)
        new_c = compute_c5(new_h)
        delta = new_c - c
        if delta < 0 or rng.random() < np.exp(-delta / T):
            h = new_h
            c = new_c
            if c < best_c:
                best_c = c
                best_h = h.copy()
        if step % 2500 == 0:
            diff = target - h.sum()
            if abs(diff) > 1e-12:
                h = enforce_sum(h, target, rng)
                c = compute_c5(h)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
        step += 1
    return best_h, best_c

def upsample_h(h, new_n):
    n = h.size
    old_x = (np.arange(n) + 0.5) * (2.0 / n)
    new_x = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    return np.interp(new_x, old_x, h)

def run(seed=42, budget_s=1100, **kwargs):

    start_time = time.time()
    # initial vector
    if 'initial_h_values' in globals():

import time
import numpy as np
from scipy.optimize import minimize

# globals for the SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth log‑sum‑exp surrogate of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=5000):
    """Clip to [0,1] and adjust to hit the exact target sum."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    if abs(diff) < 1e-12:
        return h
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            room = 1.0 - h[i]
            if room <= 0:
                continue
            add = min(diff, room)
            h[i] += add
            diff -= add
        else:
            room = h[i]
            if room <= 0:
                continue
            sub = min(-diff, room)
            h[i] -= sub
            diff += sub
    # tiny residual fix by proportional distribution
    if abs(diff) > 1e-12:
        caps = (1.0 - h) if diff > 0 else h
        tot = caps.sum()
        if tot > 0:
            h += caps * (diff / tot)
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def slsqp_optimize(h0, beta, max_iter=400):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny correction of the sum
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        i = rng.integers(0, n)
        x[i] += diff
        x = np.clip(x, 0.0, 1.0)
    return x


def anneal_optimize(h0, rng, end_time):
    """Simple simulated‑annealing optimiser (preserves sum by construction)."""
    h = h0.copy()
    c = compute_c5(h)
    best_h, best_c = h.copy(), c
    n = h.size
    target = n / 2.0

    T_start, T_end = 0.03, 0.0005
    max_steps = 200_000
    step = 0
    while time.time() < end_time and step < max_steps:
        prog = step / max_steps
        T = T_start * (1 - prog) + T_end * prog
        scale = 0.6 * (1 - prog) + 0.02 * prog
        h_new = perturb_h(h, rng, scale=scale)
        c_new = compute_c5(h_new)
        delta = c_new - c
        if delta < 0 or rng.random() < np.exp(-delta / T):
            h, c = h_new, c_new
            if c < best_c:
                best_c, best_h = c, h.copy()
        if step % 2500 == 0:
            diff = target - h.sum()
            if abs(diff) > 1e-12:
                h = enforce_sum(h, target, rng)
                c = compute_c5(h)
                if c < best_c:
                    best_c, best_h = c, h.copy()
        step += 1
    return best_h, best_c


def upsample_h(h, new_n):
    """Linear interpolation to a finer grid."""
    n = h.size
    old_x = (np.arange(n) + 0.5) * (2.0 / n)
    new_x = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    return np.interp(new_x, old_x, h)
