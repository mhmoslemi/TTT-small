def subgradient(h, active_shifts):
    n = len(h)
    grad = np.zeros(n)
    for k in active_shifts:
        # term1 (positive)
        # indices i where i+k in [0,n-1]
        low = max(0, -k)
        high = min(n, n - k)  # i < n - k
        # for those i, term1 = 1 - h[i+k]
        grad[low:high] += 1.0 - h[low + k : high + k]  # careful: i + k = (low:k) + k.
        # For i outside that range (i where i+k out of bounds), term1 = 1.
        # That's i in [0, low) and i in [high, n)
        grad[:low] += 1.0
        grad[high:] += 1.0
        # term2 (negative)
        # indices i where i - k in [0, n-1]
        low2 = max(0, k)
        high2 = min(n, n + k)  # i < n + k
        # for those i, term2 = -h[i - k]
        grad[low2:high2] -= h[low2 - k : high2 - k]
        # for i where i - k out of bounds, term2 = 0 (no change)
    return grad

    # Build initial h0:

def subgradient(h, shifts):
    n = h.size
    grad = np.zeros(n)
    # For each shift k
    for k in shifts:
        # term1: (1 - h[i+k]) if i+k in range else 1
        # define i range where i+k is in [0,n-1]
        low = max(0, -k)
        high = min(n, n - k)  # exclusive upper bound
        # For i in [low, high):
        grad[low:high] += 1.0 - h[low + k : high + k]
        # For i outside this range (left part and right part) add 1
        if low > 0:
            grad[:low] += 1.0
        if high < n:
            grad[high:] += 1.0
        # term2: -h[i - k] if i - k in range else 0
        low2 = max(0, k)
        high2 = min(n, n + k)  # exclusive
        # For i in [low2, high2):
        grad[low2:high2] -= h[low2 - k : high2 - k]
        # No addition for out-of-range portion
    return grad

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    n = h.size
    if diff > 0:
        cap = 1.0 - h
        total_cap = cap.sum()
        if total_cap > 0:
            add = diff * cap / total_cap
            h += add
        else:
            # no capacity, cannot add, so equal distribution maybe clamped
            pass
    else:
        need = -diff
        removable = h
        total_removable = removable.sum()
        if total_removable > 0:
            sub = need * removable / total_removable
            h -= sub
        else:
            pass
    # Clip again
    np.clip(h, 0.0, 1.0, out=h)
    # small residual fix
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # fallback simple loop
        for _ in range(1000):
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

import time, math, numpy as np
from scipy.optimize import minimize

# ---------- global for smooth max ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

# ---------- basic helpers ----------
def compute_c5(h):
    """Discrete C5 for step function h on [0,2] with dx=2/n."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)

def enforce_sum(h, target, rng):
    """Adjust h in‑place so that sum(h)=target while staying in [0,1]."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    n = h.size
    if diff > 0:                     # need to add mass
        cap = 1.0 - h
        total = cap.sum()
        if total > 0:
            h += diff * cap / total
    else:                            # need to remove mass
        need = -diff
        total = h.sum()
        if total > 0:
            h -= need * h / total
    np.clip(h, 0.0, 1.0, out=h)
    # tiny residual correction
    diff = target - h.sum()
    for _ in range(200):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def random_feasible_vector(n, rng):
    """Start from 0.5 everywhere and shuffle mass."""
    h = np.full(n, 0.5)
    for _ in range(n * 8):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps
        h[i] -= eps
        h[j] += eps
    return h

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

def biased_perturb_h(h, rng, scale=0.2):
    """Move mass away from positions that give the largest product."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    k = np.argmax(corr) - (n - 1)
    i0 = max(0, -k)
    i1 = min(n, n - k)
    idx = np.arange(i0, i1)
    prod = h[idx] * (1 - h[idx + k])
    if prod.size == 0:
        return perturb_h(h, rng, scale)
    thr_hi = np.quantile(prod, 0.70)
    thr_lo = np.quantile(prod, 0.30)
    high = idx[prod > thr_hi]
    low = idx[prod < thr_lo]
    if high.size == 0 or low.size == 0:
        return perturb_h(h, rng, scale)
    i = rng.choice(high)
    j = rng.choice(low)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

def subgradient(h, shifts):
    """Sub‑gradient of max‑overlap w.r.t. h for the active shifts."""
    n = h.size
    g = np.zeros(n)
    for k in shifts:
        # term (1 - h[i+k]) with out‑of‑range giving 1
        lo = max(0, -k)
        hi = min(n, n - k)
        g[lo:hi] += 1.0 - h[lo + k:hi + k]
        if lo > 0:
            g[:lo] += 1.0
        if hi < n:
            g[hi:] += 1.0
        # term -h[i-k] with out‑of‑range giving 0
        lo2 = max(0, k)
        hi2 = min(n, n + k)
        g[lo2:hi2] -= h[lo2 - k:hi2 - k]
    return g

def slsqp_optimize(h0, beta, max_iter=200):
    """Smooth max via log‑sum‑exp and SLSQP."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {"type": "eq", "fun": sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

def simulated_annealing(h0, rng, max_iter, init_temp, final_temp):
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    n = h.size
    for it in range(1, max_iter + 1):
        T = init_temp * (final_temp / init_temp) ** (it / max_iter)
        cand = biased_perturb_h(h, rng, scale=0.3) if rng.random() < 0.6 else perturb_h(h, rng, scale=0.3)
        cand_c = compute_c5(cand)
        diff = cand_c - cur_c
        if diff < 0 or rng.random() < math.exp(-diff / max(T, 1e-12)):
            h = cand
            cur_c = cand_c
            if cand_c < best_c:
                best_c = cand_c
                best_h = cand.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 360                     # higher resolution gives a tighter bound
    target = n / 2.0

    # initialise h (interpolate if a parent solution exists)
    if "initial_h_values" in globals():
        old = np.array(initial_h_values, dtype=float)
        old_n = old.size

        x_new = np.linspace(0, 2, n, endpoint=False)

        h0 = np.clip(h0, 0.0, 1.0)

    cur_c = best_c
    no_imp = 0
    it = 0
    while time.time() - start < grad_budget:
        corr = np.correlate(h, 1 - h, mode="full")
        max_val = corr.max()
        active = (np.where(corr == max_val)[0] - (n - 1)).tolist()
        g = subgradient(h, active)
        d = g - np.mean(g)                # projection onto sum‑zero hyperplane
        eta = 0.4 / (1.0 + it * 0.001)     # decreasing step size

        new_h = np.clip(new_h, 0.0, 1.0)

            no_imp += 1

        if cur_c < best_c:

            no_imp = 0

            no_imp = 0

    # ---------- simulated annealing + SLSQP polishing ----------

    sa_budget = refine_budget * 0.65
    while time.time() - start < start + sa_budget:
        start_h = perturb_h(best_h, rng, scale=0.3) if rng.random() < 0.5 else random_feasible_vector(n, rng)
        time_left = budget_s - (time.time() - start)
        if time_left <= 0:
            break
        max_iter = int(min(20000, max(2000, time_left * 1500)))

    enforce_sum(best_h, target, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
