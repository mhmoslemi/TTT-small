def softmax_obj_and_grad(h, beta, target, mu):
    n = h.size
    S = np.sum(h)
    # compute autocorrelation A = np.correlate(h, h, mode='full')
    A = np.correlate(h, h, mode='full')
    # compute C = S - A   (C length = 2n-1)
    C = S - A
    # compute stable p
    # compute x = beta * C
    x = beta * C
    m = np.max(x)
    # w = exp(x - m)
    w = np.exp(x - m)
    Z = np.sum(w)
    # p = w / Z
    p = w / Z
    # f_soft = (m + np.log(Z)) / beta - (beta? Actually original: (m + log Z)/beta
    f_soft = (m + np.log(Z)) / beta
    # compute gradient:
    # conv_p_h = np.convolve(p, h, mode='full')
    conv1 = np.convolve(p, h, mode='full')
    # conv_rev = np.convolve(p[::-1], h, mode='full')
    conv2 = np.convolve(p[::-1], h, mode='full')
    # Extract slice
    start = n - 1
    end = start + n
    term1 = conv1[start:end]   # sum_k p_k * h_{j - k}
    term2 = conv2[start:end]   # sum_k p_k * h_{j + k}
    grad = 1.0 - term1 - term2
    # penalty gradient
    diff = S - target
    if mu != 0.0:
        grad = grad + 2.0 * mu * diff * np.ones_like(h)
        # add penalty to f
        f_soft = f_soft + mu * diff * diff
    return f_soft, grad

def enforce_sum(h, target, rng, max_iter=5000):
    """Clamp to [0,1] and adjust to hit the exact sum."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])

        # small global correction
        h += diff / n
        h = np.clip(h, 0.0, 1.0)

def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from a random 1‑coordinate to a random 0‑coordinate."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def compute_c5(h):
    """Exact discrete C₅ for step function h on a uniform grid."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) with analytic gradient
# ----------------------------------------------------------------------
def softmax_obj_and_grad(h, beta, target, mu):
    """Return smooth surrogate value and its gradient."""
    n = h.size
    S = np.sum(h)
    A = np.correlate(h, h, mode="full")          # autocorrelation
    C = S - A                                   # C(k) = Σ h_i (1‑h_{i+k})
    x = beta * C
    m = np.max(x)
    w = np.exp(x - m)                           # scaled exponentials
    Z = np.sum(w)
    p = w / Z                                   # normalized weights
    f = (m + np.log(Z)) / beta                  # log‑sum‑exp value
    diff = S - target
    if mu != 0.0:
        f += mu * diff * diff

    # gradient: 1 − Σ p_k h_{j+k} − Σ p_k h_{j‑k}
    conv1 = np.convolve(p, h, mode="full")
    conv2 = np.convolve(p[::-1], h, mode="full")
    start = n - 1
    end = start + n
    term1 = conv1[start:end]    # Σ p_k h_{j‑k}
    term2 = conv2[start:end]    # Σ p_k h_{j+k}
    grad = 1.0 - term1 - term2
    if mu != 0.0:
        grad += 2.0 * mu * diff * np.ones_like(h)
    return f, grad


def softmax_obj_fun(h, beta, target, mu):
    f, _ = softmax_obj_and_grad(h, beta, target, mu)
    return f


def softmax_obj_grad(h, beta, target, mu):
    _, g = softmax_obj_and_grad(h, beta, target, mu)
    return g

def optimize_with_beta_schedule(h0, beta_list, target, mu, rng, maxiter_stage=200):
    """Successively run L‑BFGS‑B for a list of β values."""
    h = h0.copy()
    n = h.size
    for beta in beta_list:
        bounds = [(0.0, 1.0)] * n
        args = (beta, target, mu)

            fun=softmax_obj_fun,
            x0=h,

            jac=softmax_obj_grad,
            method="L-BFGS-B",
            bounds=bounds,

        )
        h = np.clip(res.x, 0.0, 1.0)
        h = enforce_sum(h, target, rng)
    return h

    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > budget_s * 0.2:
        target_n = min(720, n * 2)
        if target_n != n:
            best_h = upsample_and_optimize(best_h, target_n, beta_list, mu, rng)
            n = target_n
            target = n / 2.0
            best_c = compute_c5(best_h)

    # ------------------------------------------------------------------
    # 4. Final clean‑up and return
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
