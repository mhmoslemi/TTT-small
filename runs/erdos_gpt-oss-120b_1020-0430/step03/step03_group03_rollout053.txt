def softmax_max_corr_grad(h, beta):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    # maybe we don't need to subtract max to compute softmax safely.
    x = beta * corr
    m = np.max(x)
    e = np.exp(x - m)
    w = e / e.sum()
    # reverse w
    w_rev = w[::-1]
    b1 = 1 - h
    conv1 = np.convolve(w_rev, b1, mode='full')
    conv2 = np.convolve(w_rev, h, mode='full')
    shift = n - 1
    term1 = conv1[shift:shift + n]  # length n
    term2 = conv2[shift:shift + n]
    grad = term1 - term2
    # This gradient is d/dh (softmax expectation)
    return grad

Algorithm to project onto bounded simplex: first clip to [0,1]; then compute sum = sum(h). If sum > target, need to reduce mass from entries > 0 (non-zero). We can subtract equally (or proportionally) from all entries > 0 until sum equals target, ensuring not dropping below 0. Similarly if sum < target, add to entries <1.

Implementation: while sum != target: compute diff = target - sum. Identify indices where we can increase (h < 1) if diff>0 else decrease (h > 0). Let n_avail = count of such indices. delta = diff / n_avail. Then add delta to those indices, clamp again to 0/1, recalc sum. This loops a few times but converges quickly. Use numpy to vectorize.

Write function project_onto_bounds_and_sum(h, target, eps=1e-12):

    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    while abs(diff) > eps:
        if diff > 0:
            idx = h < 1 - eps
        else:
            idx = h > eps
        n_idx = idx.sum()
        if n_idx == 0:
            break
        delta = diff / n_idx
        # apply with clipping to bounds
        h[idx] += delta
        # Clip again
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
    return h

def softmax_weights(corr, beta):
    x = beta * corr
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()

def softmax_gradient(h, beta):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    w = softmax_weights(corr, beta)
    w_rev = w[::-1]
    b1 = 1 - h
    conv1 = np.convolve(w_rev, b1, mode='full')
    conv2 = np.convolve(w_rev, h, mode='full')
    shift = n - 1
    term1 = conv1[shift:shift + n]
    term2 = conv2[shift:shift + n]
    grad = term1 - term2
    return grad

import time, math, numpy as np

# -------------------- helpers --------------------
def compute_c5(h):
    """Exact discrete C5 for step function h (dx=2/n)."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)


def project_bounds_and_sum(h, target, max_iter=10, eps=1e-12):
    """Clip to [0,1] and then adjust the sum to exactly target."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        diff = target - h.sum()
        if abs(diff) <= eps:
            break
        if diff > 0:
            idx = h < 1.0 - eps
        else:
            idx = h > eps
        n_idx = idx.sum()
        if n_idx == 0:
            break
        delta = diff / n_idx
        h[idx] += delta
        h = np.clip(h, 0.0, 1.0)
    return h


def random_feasible_vector(n, rng):
    """Generate a random h with 0≤h≤1 and sum=n/2."""
    # start from uniform 0.5 and shuffle mass
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    return project_bounds_and_sum(h, n / 2.0, max_iter=200)


def perturb_h(h, rng, scale=0.2):
    """Transfer a random ε between two entries (preserves sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def softmax_weights(corr, beta):
    x = beta * corr
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


def softmax_gradient(h, beta):
    """Gradient of (log‑sum‑exp(β·corr))/β with respect to h."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    w = softmax_weights(corr, beta)          # length 2n‑1
    w_rev = w[::-1]
    conv1 = np.convolve(w_rev, 1 - h, mode="full")
    conv2 = np.convolve(w_rev, h, mode="full")
    shift = n - 1
    term1 = conv1[shift:shift + n]
    term2 = conv2[shift:shift + n]
    return term1 - term2


# -------------------- optimisation --------------------
def adam_optimize(h0, target, beta_schedule, steps_per_beta,
                  lr_base=0.05, rng=None, time_limit=None, start_time=None,
                  budget=None):
    """Adam optimisation with projection after every step."""
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    beta1_pow, beta2_pow = 1.0, 1.0
    t = 0

    for beta, steps in zip(beta_schedule, steps_per_beta):
        # step size shrinks a little with larger beta
        lr = lr_base / (np.log2(beta + 1) + 1)
        for _ in range(steps):
            t += 1
            beta1_pow *= beta1
            beta2_pow *= beta2
            grad = softmax_gradient(h, beta)
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1_pow)
            v_hat = v / (1 - beta2_pow)
            step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            h = h - step
            h = np.clip(h, 0.0, 1.0)
            h = project_bounds_and_sum(h, target)

            # occasional small jitter to escape plateaus
            if rng is not None and rng.random() < 0.005:
                h += (rng.random(h.shape) - 0.5) * 0.02
                h = np.clip(h, 0.0, 1.0)
                h = project_bounds_and_sum(h, target)

            if time_limit is not None and (time.time() - start_time) > time_limit:
                return h
    return h

    start_time = time.time()
    # ------------------- initialise -------------------

    # schedule for the smooth max approximation
    beta_sched = [5, 20, 100, 500, 2000, 8000]

    # ------------------- main search -------------------
    # keep restarting while we have budget
    while True:

        h_opt = adam_optimize(
            h0,
            target,
            beta_sched,
            steps_sched,
            lr_base=0.07,
            rng=rng,
            time_limit=budget_s * 0.88,
            start_time=start_time,
        )
        c_val = compute_c5(h_opt)
        if c_val < best_c:
            best_h, best_c = h_opt.copy(), c_val

    # ------------------- final polishing -------------------
    # a higher‑beta short run to fine‑tune the current best
    final_beta = [2000, 8000, 20000]
    final_steps = [400, 400, 400]
    h_polish = best_h.copy()

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_bounds_and_sum(best_h, target)

    return best_h.tolist(), float(best_c), int(n)
