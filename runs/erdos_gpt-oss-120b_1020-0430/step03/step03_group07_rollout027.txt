def propose_move(h, rng, temp, T0):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j: return None
    # compute allowed epsilon range
    max_eps = min(h[i], 1 - h[j])
    min_eps = -min(h[j], 1 - h[i])
    if max_eps <= 0 and min_eps >= 0:
        return None
    # choose epsilon around 0 to encourage small moves
    # Let sigma = (temp / T0) * (max_eps - min_eps) * 0.5 as typical magnitude
    sigma = (temp / T0) * (max_eps - min_eps) * 0.5
    # sample epsilon from uniform [-sigma, sigma] truncated to allowed range
    low = max(min_eps, -sigma)
    high = min(max_eps, sigma)
    if low >= high:
        # fallback to full allowed range
        low, high = min_eps, max_eps
    eps = low + rng.random() * (high - low)
    return i, j, eps

def simulated_annealing_full(start_h, rng, deadline, T0=0.03, Tf=1e-6, max_iters=200000):

    cur_h = start_h.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c

    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        # propose move
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if i == j:
            continue
        # allowed epsilon
        max_eps = min(cur_h[i], 1 - cur_h[j])
        min_eps = -min(cur_h[j], 1 - cur_h[i])
        if max_eps - min_eps <= 0:
            continue
        # sample epsilon: maybe uniform in [min_eps, max_eps]
        eps = min_eps + rng.random() * (max_eps - min_eps)
        # scale epsilon by temperature factor (optional)
        # Actually we can set eps = eps * (T / T0)   # small at low T
        eps = eps * (T / T0)
        # Apply move
        cur_h[i] -= eps
        cur_h[j] += eps
        new_c = compute_c5(cur_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] += eps
            cur_h[j] -= eps
    return best_h, best_c

    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)

def enforce_sum(h, target, rng):

        else:

def hill_climb(h_start, rng, max_iters=20000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-6)
    return best_h, best_c

def simulated_annealing(start_h, rng, deadline, T0=0.02, Tf=1e-6, max_iters=200000):
    cur_h = start_h.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    n = cur_h.size
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # temperature schedule
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)  # exponential cooling
        # propose move
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if i == j:
            continue
        max_eps = min(cur_h[i], 1 - cur_h[j])
        min_eps = -min(cur_h[j], 1 - cur_h[i])
        if max_eps - min_eps <= 0:
            continue
        eps = min_eps + rng.random() * (max_eps - min_eps)  # full range
        # Apply
        cur_h[i] -= eps
        cur_h[j] += eps
        new_c = compute_c5(cur_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] += eps
            cur_h[j] -= eps
    return best_h, best_c

import time

from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global state for the smooth surrogate used by SLSQP
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def enforce_sum(h, target, rng):
    """Adjust h so that Σh == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    # simple stochastic correction
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Random 0‑1 vector with exactly n/2 ones, then smooth a bit."""
    h = np.zeros(n)
    idx = rng.permutation(n)[: n // 2]
    h[idx] = 1.0
    # small random fluid moves
    for _ in range(n * 2):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        min_eps = -min(h[j], 1.0 - h[i])
        if max_eps - min_eps <= 0:
            continue
        eps = min_eps + rng.random() * (max_eps - min_eps) * 0.02
        h[i] -= eps
        h[j] += eps
    return enforce_sum(h, n / 2.0, rng)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    min_eps = -min(h[j], 1.0 - h[i])
    if max_eps - min_eps <= 0:
        return h.copy()
    eps = min_eps + rng.random() * (max_eps - min_eps) * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def correlate_full(h):
    """Full linear correlation of h with (1‑h) using FFT."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())  # next power of two
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)

# Smooth surrogate for SLSQP
# ----------------------------------------------------------------------
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Log‑sum‑exp smoothing of max_k Σ_i x_i (1‑x_{i+k})."""
    conv = correlate_full(x)
    bc = CURRENT_BETA * conv
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ----------------------------------------------------------------------
# Local search primitives
# ----------------------------------------------------------------------
def hill_climb(h_start, rng, max_iters=20000, start_step=0.5):
    """Greedy hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-6)
    return best_h, best_c


def simulated_annealing(start_h, rng, deadline, T0=0.03, Tf=1e-6, max_iters=200000):
    """Long SA with occasional full‑transfer moves."""
    cur_h = start_h.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    n = cur_h.size
    target = n / 2.0

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        # exponential cooling
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)

        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if i == j:
            continue

        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        min_eps = -min(cur_h[j], 1.0 - cur_h[i])
        if max_eps - min_eps <= 0:
            continue

        # 5 % chance of a full extreme move
        u = rng.random()
        if u < 0.05 and max_eps > 0:
            eps = max_eps
        elif u < 0.10 and min_eps < 0:
            eps = min_eps
        else:
            eps = min_eps + rng.random() * (max_eps - min_eps)

        # apply move
        cur_h[i] -= eps
        cur_h[j] += eps
        new_c = compute_c5(cur_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] += eps
            cur_h[j] -= eps

        # occasional exact‑sum repair (tiny drift)
        if abs(cur_h.sum() - target) > 1e-9:
            cur_h = enforce_sum(cur_h, target, rng)

    return best_h, best_c


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # reserve a small margin for final clean‑up
    deadline_main = start_time + budget_s * 0.90
    deadline_up = start_time + budget_s * 0.96

    # ------------------------------------------------------------------
    # 1. Initialise base vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ------------------------------------------------------------------
    # 2. Quick hill‑climb from a few random starts
    # ------------------------------------------------------------------
    for _ in range(3):
        h_rand = random_feasible_vector(n, rng)
        h_opt, c_opt = hill_climb(h_rand, rng, max_iters=8000, start_step=0.5)
        if c_opt < best_c:
            best_h, best_c = h_opt.copy(), c_opt

    # ------------------------------------------------------------------
    # 3. Long simulated annealing (time‑bounded)
    # ------------------------------------------------------------------
    h_sa, c_sa = simulated_annealing(best_h, rng, deadline_main,
                                     T0=0.03, Tf=1e-6, max_iters=200000)
    if c_sa < best_c:
        best_h, best_c = h_sa.copy(), c_sa

    # occasional extra SA restarts while time permits
    while time.time() < deadline_main - 5.0:
        h_start = random_feasible_vector(n, rng)
        h_sa2, c_sa2 = simulated_annealing(h_start, rng, deadline_main,
                                           T0=0.03, Tf=1e-6, max_iters=200000)
        if c_sa2 < best_c:
            best_h, best_c = h_sa2.copy(), c_sa2

    # ------------------------------------------------------------------
    # 4. Deterministic hill‑climb refinement
    # ------------------------------------------------------------------
    h_hc, c_hc = hill_climb(best_h, rng, max_iters=25000, start_step=0.4)
    if c_hc < best_c:
        best_h, best_c = h_hc.copy(), c_hc

    # ------------------------------------------------------------------
    # 5. SLSQP smoothing ladder
    # ------------------------------------------------------------------
    beta_list = [20.0, 80.0, 200.0, 800.0]
    for beta in beta_list:
        if time.time() >= deadline_main:
            break
        h_opt = slsqp_optimize(best_h, beta, max_iter=150)
        c_opt = compute_c5(h_opt)
        if c_opt < best_c:
            best_h, best_c = h_opt.copy(), c_opt

    # ------------------------------------------------------------------
    # 6. Optional up‑sampling to a finer grid (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline_up and best_h.size * 2 <= 800:
        up_factor = 2
        h_up = np.repeat(best_h, up_factor)
        n_up = h_up.size
        h_up = enforce_sum(h_up, n_up / 2.0, rng)

        # hill‑climb on finer grid
        h_up_opt, c_up_opt = hill_climb(h_up, rng,
                                        max_iters=30000, start_step=0.3)
        if c_up_opt < best_c:
            best_h, best_c, n = h_up_opt.copy(), c_up_opt, n_up

        # final SLSQP on up‑sampled grid
        for beta in [80.0, 200.0]:
            if time.time() >= deadline_up:
                break
            h_opt = slsqp_optimize(best_h, beta, max_iter=120)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c:
                best_h, best_c, n = h_opt.copy(), c_opt, h_opt.size

    # ------------------------------------------------------------------
    # 7. Final clean‑up
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
