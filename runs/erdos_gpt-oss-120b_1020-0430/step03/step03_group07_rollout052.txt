def simulated_annealing(start_h, rng, deadline, T0=0.02, Tf=1e-5, max_iters=1500, prob_target=0.5):
    cur_h = start_h.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        if rng.random() < prob_target:
            new_h = targeted_perturb(cur_h, rng, scale=0.2)
        else:
            new_h = perturb_h(cur_h, rng, scale=0.3)
        new_c = compute_c5(new_h)
        # cooling schedule exponential
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-12)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
    return best_h, best_c

def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_c = compute_c5(h)
    best_h = h.copy()
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        # Use targeted perturb with probability 0.3
        if rng.random() < 0.3:
            new_h = targeted_perturb(h, rng, scale=step)
        else:
            new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        # update step decay
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c

def correlate_full(h):
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv

    diff = target - h.sum()
    n = h.size

# ----------------------------------------------------------------------
# Basic feasibility utilities
# ----------------------------------------------------------------------
def enforce_sum(h, target, rng):
    """Adjust h so that Σh == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(10000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0.0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # final tiny correction
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h


def random_feasible_vector(n, rng):
    """Random vector in [0,1]^n with sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

# Correlation utilities (FFT based)
# ----------------------------------------------------------------------
def correlate_full(h):
    """Full correlation of h with (1‑h)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1.0 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def autocorr(h):
    """Full autocorrelation of h with itself."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    fr = np.fft.rfft(h[::-1], L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Discrete C₅ for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)

# Targeted perturbation (focus on the worst lag)
# ----------------------------------------------------------------------
def targeted_perturb(h, rng, scale=0.2):
    """Increase product at the lag giving minimum self‑correlation."""
    n = h.size
    sc = autocorr(h)
    min_idx = np.argmin(sc)
    lag = min_idx - (n - 1)          # can be negative
    # search for a pair (i,i+lag) where we can increase both
    if lag >= 0:
        i_start, i_end = 0, n - lag
    else:
        i_start, i_end = -lag, n
    best_i = -1
    best_sum = -1.0
    for i in range(i_start, i_end):
        j = i + lag
        inc_possible = min(1.0 - h[i], 1.0 - h[j])
        if inc_possible <= 1e-12:
            continue
        s = h[i] + h[j]
        if s > best_sum:
            best_sum = s
            best_i = i
    if best_i == -1:
        return perturb_h(h, rng, scale)   # fallback
    j = best_i + lag
    inc_possible = min(1.0 - h[best_i], 1.0 - h[j])
    eps = scale * inc_possible
    eps = min(eps, inc_possible)
    # need to subtract the same total elsewhere (two positions)
    dec_idxs = []
    attempts = 0
    while len(dec_idxs) < 2 and attempts < 30:
        d = rng.integers(0, n)
        if d not in (best_i, j) and h[d] > eps:
            dec_idxs.append(d)
        attempts += 1
    if len(dec_idxs) < 2:
        return perturb_h(h, rng, scale)   # fallback
    new_h = h.copy()
    new_h[best_i] += eps
    new_h[j] += eps
    new_h[dec_idxs[0]] -= eps
    new_h[dec_idxs[1]] -= eps
    new_h = np.clip(new_h, 0.0, 1.0)
    target = n / 2.0
    diff = target - new_h.sum()
    if abs(diff) > 1e-12:
        new_h = enforce_sum(new_h, target, rng)
    return new_h

def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        if rng.random() < 0.3:
            new_h = targeted_perturb(h, rng, scale=step)
        else:
            new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c

def simulated_annealing(start_h, rng, deadline, T0=0.02, Tf=1e-5,
                       max_iters=1500, prob_target=0.6):
    cur_h = start_h.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        if rng.random() < prob_target:
            new_h = targeted_perturb(cur_h, rng, scale=0.2)
        else:
            new_h = perturb_h(cur_h, rng, scale=0.3)
        new_c = compute_c5(new_h)
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-12)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
    return best_h, best_c

# Smooth surrogate + SLSQP
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = correlate_full(x)
    bc = CURRENT_BETA * corr
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on smoothed objective for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # correct possible drift in the sum
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x
