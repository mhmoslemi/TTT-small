def improve_hill_climb(h, rng, time_limit, start_temp=0.0, temp_decay=0.9999):
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_h = h.copy()
    cur_c = best_c
    n = len(h)
    # Precompute indices for 1 and 0 sets? Maybe we can just random choose two indices and swap if they have opposite values else ignore.

    # For binary case: we can sample i,j until they differ.

    start_time = time.time()
    T = start_temp
    it = 0
    while time.time() - start_time < time_limit:
        # random i, j different
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur_h[i] == cur_h[j]:
            continue
        # swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_c = compute_c5(cur_h)
        delta = new_c - cur_c
        if delta <= 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        # temperature schedule
        if T > 0:
            T *= temp_decay
        it += 1
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global variables for the smooth surrogate used by SLSQP
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    """Smooth log‑sum‑exp surrogate of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=400):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny exact‑sum repair
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            if abs(add) < 1e-12:
                continue
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def compute_c5(h):
    """Exact discrete C₅ for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng):
    """Adjust h so that Σ h_i = target while 0≤h_i≤1."""
    diff = target - h.sum()
    n = h.size
    # stochastic adjustments
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-12:
            continue
        h[i] += add
        diff = target - h.sum()
    # deterministic sweep if needed
    if abs(diff) > 1e-12:
        for i in range(n):
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            if abs(add) < 1e-12:
                continue
            h[i] += add
            diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_binary_vector(n, target_sum, rng):
    """Generate a binary vector of length n with exactly target_sum ones."""
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, size=int(target_sum), replace=False)
    h[ones] = 1.0
    return h


def improve_hill_climb_binary(h_start, rng, time_limit, start_temp=0.0, temp_decay=1.0):
    """
    Greedy / simulated‑annealing hill climb on a binary vector.
    Returns the best vector seen and its C₅ value.
    """
    start = time.time()
    cur_h = h_start.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    T = start_temp
    n = cur_h.size

    while time.time() - start < time_limit:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur_h[i] == cur_h[j]:
            continue
        # swap a 1 and a 0
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_c = compute_c5(cur_h)
        delta = new_c - cur_c
        accept = False
        if delta <= 0:
            accept = True
        elif T > 0.0 and rng.random() < np.exp(-delta / T):
            accept = True
        if accept:
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        if T > 0.0:
            T *= temp_decay
    return best_h, best_c
