def local_search(h, rng, time_limit):
    best = h.copy()
    best_c = compute_c5(best)
    start = time.time()
    # Stage 1: greedy improvement loop (deterministic)
    h = best.copy()
    while True:
        c_cur = compute_c5(h)
        # compute worst shift idx
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr)
        s = idx - (h.size - 1)
        # gradient
        grad = compute_gradient_shift(h, s)
        # positive positions (grad>0) with h>0
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-12:
            break
        eps = max_eps  # take full
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c_new = compute_c5(new_h)
        if c_new + 1e-12 < c_cur:
            h = new_h
            if c_new < best_c:
                best_c = c_new
                best = new_h.copy()
        else:
            # can't improve further with full epsilon
            # try smaller epsilon
            eps = max_eps * 0.5
            new_h[i] -= eps
            new_h[j] += eps
            c_new = compute_c5(new_h)
            if c_new + 1e-12 < c_cur:
                h = new_h
                if c_new < best_c:
                    best_c = c_new
                    best = new_h.copy()
            else:
                break
    # Stage 2: random perturbation hill climbing
    while time.time() - start < time_limit:

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def worst_shift_info(h):
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    max_corr = corr[idx]
    s = idx - (h.size - 1)
    return s, idx, max_corr

def gradient_shift(h, s):
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        # part for (1 - h[t+s]) for t in [0, n-1-s]
        grad[:n - s] += (1 - h[s:])
        # part for -h[t-s] for t in [s, n-1]
        grad[s:] -= h[:n - s]
    else:
        s_abs = -s
        # (1 - h[t - s_abs]) for t >= s_abs
        grad[s_abs:] += (1 - h[:n - s_abs])
        # -h[t + s_abs] for t in [0, n - s_abs -1]
        grad[:n - s_abs] -= h[s_abs:]
    return grad

def sum_eq_constraint(x): return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log-sum-exp: (m + log(sum(exp(bc-m)))) / beta
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

import time, numpy as np
from scipy.optimize import minimize

# ---------- helper utilities ----------
def enforce_sum(h, target, rng):
    """Clip to [0,1] and adjust to make sum exactly target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:  # need to increase
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0: break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i]) * rng.random()
            h[i] += add
            diff -= add
        else:  # need to decrease
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0: break
            i = rng.choice(idx)
            sub = min(-diff, h[i]) * rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    h = rng.random(n)
    h = enforce_sum(h, n / 2.0, rng)
    return h

def perturb_random(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def compute_c5(h):
    """Exact discrete C5 for a vector h."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def worst_shift_info(h):
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    s = idx - (h.size - 1)
    return s, idx, corr[idx]

def gradient_shift(h, s):
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[:n - s] += (1 - h[s:])
        grad[s:] -= h[:n - s]
    else:
        s_abs = -s
        grad[s_abs:] += (1 - h[:n - s_abs])
        grad[:n - s_abs] -= h[s_abs:]
    return grad

def greedy_optimize(h, max_steps=2000):
    """Deterministic improvement using full‑epsilon transfers."""
    for _ in range(max_steps):
        s, _, cur_corr = worst_shift_info(h)
        grad = gradient_shift(h, s)
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-12:
            break
        # try full epsilon
        nh = h.copy()
        nh[i] -= max_eps
        nh[j] += max_eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        # try half epsilon
        eps = max_eps * 0.5
        nh[i] -= eps
        nh[j] += eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        break
    return h

# ---------- SLSQP refinement ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=600):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    rng = np.random.default_rng()
    x = enforce_sum(x, CURRENT_TARGET, rng)
    return x

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # initial vector
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = random_feasible_vector(180, rng)
    h0 = enforce_sum(np.clip(h0, 0.0, 1.0), h0.size / 2.0, rng)

    # maybe upscale to finer grid (2×) for more flexibility
    n0 = h0.size
    target_n = n0 * 2 if n0 * 2 <= 720 else n0
    if target_n != n0:
        xs_old = (np.arange(n0) + 0.5) * (2.0 / n0)
        xs_new = (np.arange(target_n) + 0.5) * (2.0 / target_n)
        h = np.interp(xs_new, xs_old, h0)
        h = enforce_sum(np.clip(h, 0.0, 1.0), target_n / 2.0, rng)
    else:
        h = h0.copy()

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ----- stochastic multi‑restart greedy search -----
    while time.time() - start_time < budget_s * 0.80:
        # generate start point
        if rng.random() < 0.5 and best_h is not None:
            start = perturb_random(best_h, rng, scale=0.5)
        else:
            start = random_feasible_vector(h.size, rng)
        # deterministic greedy refinement
        local = greedy_optimize(start, max_steps=3000)
        c_local = compute_c5(local)
        if c_local < best_c:
            best_h, best_c = local, c_local

        # occasional SLSQP polishing
        if rng.random() < 0.3:
            for beta in (200, 800, 2000, 5000):
                if time.time() - start_time > budget_s * 0.95:
                    break
                try:
                    refined = slsqp_optimize(best_h, beta, max_iter=800)
                except Exception:
                    continue
                c_ref = compute_c5(refined)
                if c_ref < best_c:
                    best_h, best_c = refined, c_ref

    # ----- final micro‑search (tiny moves) -----
    while time.time() - start_time < budget_s * 0.99:
        cand = perturb_random(best_h, rng, scale=0.02)
        c_cand = compute_c5(cand)
        if c_cand < best_c:
            best_h, best_c = cand, c_cand

    # clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
