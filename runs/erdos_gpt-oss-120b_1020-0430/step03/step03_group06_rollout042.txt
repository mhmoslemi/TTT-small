start_time = time.time()

    # propose move
    i, j = rng.integers(0,n, size=2)
    while i==j:
        j = rng.integers(0,n)
    max_pos = min(current_h[i], 1 - current_h[j])
    max_neg = min(current_h[j], 1 - current_h[i])
    if max_pos + max_neg == 0:
        continue
    eps = (rng.random() * (max_pos + max_neg) - max_neg) * step_scale
    new_h = current_h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    # compute new C5
    new_c = compute_c5(new_h)
    # acceptance
    if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c)/temperature):
        current_h, cur_c = new_h, new_c
        if new_c < best_c:
            best_h, best_c = new_h.copy(), new_c
    # update temperature and step scale schedule
    elapsed = time.time() - start_time
    # maybe linear cooling based on fraction of budget used
    fraction = elapsed / max_time
    temperature = T0 * (1 - fraction) + T_end * fraction

def sa_optimize(h0, rng, time_limit, start_temp=0.02, end_temp=0.0005,
                start_scale=1.0, end_scale=0.02):
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    start = time.time()
    while time.time() - start < time_limit:
        # compute fraction
        elapsed = time.time() - start
        fraction = elapsed / time_limit
        T = start_temp * (1 - fraction) + end_temp * fraction
        scale = start_scale * (1 - fraction) + end_scale * fraction
        # propose move
        i = rng.integers(0, h.size)
        j = rng.integers(0, h.size - 1)
        if j >= i:
            j += 1
        max_pos = min(h[i], 1.0 - h[j])
        max_neg = min(h[j], 1.0 - h[i])
        if max_pos + max_neg == 0:
            continue
        eps = (rng.random() * (max_pos + max_neg) - max_neg) * scale
        if eps == 0:
            continue
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        # acceptance
        accept = False
        if new_c < cur_c:
            accept = True
        else:
            # avoid overflow
            prob = np.exp(-(new_c - cur_c) / T) if T > 0 else 0.0
            if rng.random() < prob:
                accept = True
        if accept:
            h, cur_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
    return best_h, best_c

def logsumexp_corr_grad(x, beta, target, penalty):
    n = x.size
    corr = np.correlate(x, 1 - x, mode='full')
    bc = beta * corr
    m = bc.max()
    exp_bc = np.exp(bc - m)
    sumexp = exp_bc.sum()
    f = (m + np.log(sumexp)) / beta
    # penalty
    diff = x.sum() - target
    f += penalty * diff * diff
    # gradient
    w = exp_bc / sumexp   # weights
    # compute conv with w and x
    # w as array length 2n-1
    # compute conv_rev = convolution of w_rev and x
    w_rev = w[::-1]
    conv = np.convolve(w, x, mode='full')
    conv_rev = np.convolve(w_rev, x, mode='full')
    offset = n-1
    # gradient_i
    grad = 1.0 - conv_rev[offset:offset+n] - conv[offset:offset+n]
    # penalty gradient
    grad += 2.0 * penalty * diff
    return f, grad
```

import time
import numpy as np
from scipy.optimize import minimize

BETA_LIST = [10.0, 40.0, 120.0, 400.0, 800.0]


def compute_c5(h):
    """Exact discrete C₅ for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng):
    """Adjust h so that Σh = target while 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-12:
            continue
        h[i] += add
        diff = target - h.sum()
    # final tiny correction
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(1000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            if abs(add) < 1e-12:
                continue
            h[i] += add
            diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    v = rng.exponential(1.0, size=n)
    v = v / v.sum() * (n / 2.0)
    h = np.clip(v, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def sa_optimize(h0, rng, time_limit, start_temp=0.02, end_temp=0.0005,
                start_scale=1.0, end_scale=0.05):
    """Simulated annealing that preserves the sum exactly."""
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    n = h.size
    start = time.time()
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = start_temp * (1 - frac) + end_temp * frac
        scale = start_scale * (1 - frac) + end_scale * frac
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        max_pos = min(h[i], 1.0 - h[j])
        max_neg = min(h[j], 1.0 - h[i])
        if max_pos + max_neg == 0:
            continue
        eps = (rng.random() * (max_pos + max_neg) - max_neg) * scale
        if eps == 0.0:
            continue
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        accept = False
        if new_c < cur_c:
            accept = True
        else:
            if T > 0.0:
                prob = np.exp(-(new_c - cur_c) / T)
                if rng.random() < prob:
                    accept = True
        if accept:
            h, cur_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
    return best_h, best_c


def _logsumexp_corr_obj_grad(x, beta, target, penalty):
    """Smooth surrogate of max_k and its gradient (with sum penalty)."""
    n = x.size
    corr = np.correlate(x, 1 - x, mode="full")
    bc = beta * corr
    m = bc.max()
    exp_bc = np.exp(bc - m)
    sumexp = exp_bc.sum()
    f = (m + np.log(sumexp)) / beta
    diff = x.sum() - target
    f += penalty * diff * diff
    w = exp_bc / sumexp
    w_rev = w[::-1]
    conv = np.convolve(w, x, mode="full")
    conv_rev = np.convolve(w_rev, x, mode="full")
    off = n - 1
    grad = 1.0 - conv_rev[off : off + n] - conv[off : off + n]
    grad += 2.0 * penalty * diff
    return f, grad


def lbfgs_optimize(h0, beta, penalty=5000.0, max_iter=600):
    """L‑BFGS‑B on the smooth surrogate (penalised sum constraint)."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def fun(x):
        f, _ = _logsumexp_corr_obj_grad(x, beta, target, penalty)
        return f

    def jac(x):
        _, g = _logsumexp_corr_obj_grad(x, beta, target, penalty)
        return g

    res = minimize(
        fun,
        h0,
        method="L-BFGS-B",
        jac=jac,
        bounds=bounds,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return x
