def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    # We'll adjust a random index a small amount repeatedly
    # For up to 1000 iterations:
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])

            add = max(diff, -h[i])
        if abs(add) < 1e-12:

def improve_once(h, rng, step):
    n = len(h)
    corr_full = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr_full)
    k = max_idx - (n - 1)
    # compute overlapping region indices i (starting index for product)
    if k >= 0:
        region_len = n - k
        # product slice indexes
        h_a = h[:region_len]
        h_b = h[k:]
        # choose i where product minimal
        prod = h_a * h_b
        i = np.argmin(prod)  # index in h_a
        i_global = i  # corresponds to h[i]
        # partner index = i + k
        partner = i + k
    else:
        # negative k
        k_abs = -k
        region_len = n - k_abs
        h_a = h[k_abs:]
        h_b = h[:region_len]
        prod = h_a * h_b
        i = np.argmin(prod) # index in h_a, which corresponds to h[i + k_abs]
        i_global = i + k_abs  # position in h_a (original)
        partner = i # partner is i - k_abs
    # Choose j not equal to i_global nor partner
    candidates = np.arange(n)
    mask = (candidates != i_global) & (candidates != partner)
    if not np.any(mask):
        return h  # no alternative
    j = rng.choice(candidates[mask])
    # Determine direction: we want to move mass from j (likely larger) to i_global (likely smaller)
    # Determine epsilon bound
    eps = step * min(h[j], 1.0 - h[i_global])
    if eps <= 1e-12:
        return h
    h_new = h.copy()
    h_new[i_global] += eps
    h_new[j] -= eps
    # Clip but since eps respects bounds, should stay within.
    # However slight rounding may cause slightly out-of-bound; clip
    h_new = np.clip(h_new, 0.0, 1.0)
    # adjust sum if rounding error
    # I'll prefer not to call enforce_sum each time; we can rely on preserving sum up to floating errors.
    return h_new

def local_search(h_start, rng, time_limit):

            best_h = new_h.copy()
        # periodically decay step

def improve_worst(h, rng, step):
    n = h.size
    corr_full = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr_full)
    k = max_idx - (n - 1)
    if k >= 0:
        region_len = n - k
        prod = h[:region_len] * h[k:]
        i_local = np.argmin(prod)
        i = i_local
        partner = i + k
    else:
        k_abs = -k
        region_len = n - k_abs
        prod = h[k_abs:] * h[:region_len]
        i_local = np.argmin(prod)
        i = i_local + k_abs
        partner = i - k_abs
    # pick two donor indices distinct from i and partner
    # ensure there are at least two other indices
    if n <= 3:
        return h
    donors = rng.choice([idx for idx in range(n) if idx not in (i, partner)], size=2, replace=False)
    d1, d2 = donors[0], donors[1]
    eps = step * min(1 - h[i], 1 - h[partner], h[d1], h[d2])
    if eps <= 1e-12:
        return h
    h_new = h.copy()
    h_new[i] += eps / 2
    h_new[partner] += eps / 2
    h_new[d1] -= eps / 2
    h_new[d2] -= eps / 2
    # Clip to stay in [0,1]
    h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

def random_perturb(h, rng, step):
    n = h.size
    i, j = rng.choice(n, size=2, replace=False)
    eps = step * min(h[i], 1 - h[j])
    if eps <= 1e-12:
        return h
    h_new = h.copy()
    h_new[i] -= eps
    h_new[j] += eps
    return h_new

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def enforce_sum(h, target, rng, max_iter=2000):
    """Adjust h (in‑place) so that Σh = target while 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-12:
            continue
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with Σ = n/2."""
    v = rng.exponential(1.0, size=n)
    v = v / v.sum() * (n / 2.0)
    h = np.clip(v, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def compute_c5(h):
    """Exact discrete C₅ for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


# ----------------------------------------------------------------------
# Guided local moves
# ----------------------------------------------------------------------
def improve_worst(h, rng, step):
    """Transfer a little mass from two donors to the two positions
    that are worst for the lag with maximal overlap."""
    n = h.size
    corr_full = np.correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(corr_full)
    k = max_idx - (n - 1)               # signed lag

    if k >= 0:
        region_len = n - k
        prod = h[:region_len] * h[k:]
        i_local = np.argmin(prod)
        i = i_local
        partner = i + k
    else:
        k_abs = -k
        region_len = n - k_abs
        prod = h[k_abs:] * h[:region_len]
        i_local = np.argmin(prod)
        i = i_local + k_abs
        partner = i - k_abs

    # pick two donor indices different from i and partner
    if n <= 3:
        return h
    possible = [idx for idx in range(n) if idx not in (i, partner)]
    d1, d2 = rng.choice(possible, size=2, replace=False)

    eps = step * min(1 - h[i], 1 - h[partner], h[d1], h[d2])
    if eps <= 1e-12:
        return h

    h_new = h.copy()
    h_new[i] += eps / 2
    h_new[partner] += eps / 2
    h_new[d1] -= eps / 2
    h_new[d2] -= eps / 2
    return np.clip(h_new, 0.0, 1.0)


def random_perturb(h, rng, step):
    """Simple two‑index transfer preserving the sum."""
    n = h.size
    i, j = rng.choice(n, size=2, replace=False)
    eps = step * min(h[i], 1 - h[j])
    if eps <= 1e-12:
        return h
    h_new = h.copy()
    h_new[i] -= eps
    h_new[j] += eps
    return h_new


def local_search(h_start, rng, time_limit):
    """Hill‑climb with the guided “improve worst lag’’ move."""
    start = time.time()
    best_h = h_start.copy()
    best_c = compute_c5(best_h)
    cur_h = h_start.copy()
    cur_c = best_c
    step = 0.2
    it = 0
    while time.time() - start < time_limit:
        it += 1
        if rng.random() < 0.8:
            new_h = improve_worst(cur_h, rng, step)
        else:
            new_h = random_perturb(cur_h, rng, step)
        new_c = compute_c5(new_h)
        # accept if better, otherwise keep with tiny probability
        if new_c < cur_c or rng.random() < 0.001:
            cur_h, cur_c = new_h, new_c
        if new_c < best_c:
            best_h, best_c = new_h.copy(), new_c
        if it % 5000 == 0:
            step = max(step * 0.9, 0.005)
    return best_h, best_c


# ----------------------------------------------------------------------
# Smooth surrogate + SLSQP (optional final polishing)
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            if abs(add) < 1e-12:
                continue
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x
