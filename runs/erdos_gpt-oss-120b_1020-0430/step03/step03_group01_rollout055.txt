def derivative_wrt_lag(h, L):
    n = h.size
    d = np.empty(n)
    if L >= 0:
        # compute h_iL shift
        # part 1: indices where i+L < n
        d[:] = 1.0
        # subtract h[i+L]
        d[:n-L] -= h[L:]
        # subtract h[i-L]
        d[L:] -= h[:n-L]
    else:
        # L negative: treat similarly
        Lpos = -L
        d[:] = 1.0
        # subtract h[i-L] (i - (-Lpos) = i+Lpos)
        d[:n-Lpos] -= h[Lpos:]
        # subtract h[i+L] (i+(-Lpos) = i-Lpos)
        d[Lpos:] -= h[:n-Lpos]
    return d
```

def worst_lag_descent(h, max_iters, rng):
    best_h = h.copy()
    best_c, _ = compute_c5_and_lag(best_h)
    for it in range(max_iters):
        c, lag = compute_c5_and_lag(best_h)
        # compute derivative for lag
        d = derivative_wrt_lag(best_h, lag)
        i = np.argmax(d)
        j = np.argmin(d)
        if d[i] <= 0 and d[j] >= 0:
            # no improving direction
            break
        # compute epsilon
        eps = min(best_h[i], 1.0 - best_h[j])
        if eps < 1e-12:
            break
        eps *= 0.5  # reduce step size
        # update
        new_h = best_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c, _ = compute_c5_and_lag(new_h)
        if new_c < best_c - 1e-12:
            best_h = new_h
            best_c = new_c
        else:
            # if not improving, try smaller epsilon or random move
            # We'll try random perturb to escape local minima
            new_h = perturb_h(best_h, rng, scale=0.3)
            new_c = compute_c5(new_h)
            if new_c < best_c:
                best_h = new_h
                best_c = new_c
    return best_h, best_c
```

CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def compute_c5(h):
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)

def compute_c5_and_lag(h):
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr)
    return corr[idx] * (2.0 / h.size), idx - (h.size - 1)

def enforce_sum(h, target, rng, max_iter=5000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        # small correction distribute uniformly then clip
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
    return h

    return h

    nh[j] += eps

    return d

            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 240
        h0 = random_feasible_vector(n, rng)

    candidates = [h0]
    xs = (np.arange(n) + 0.5) * (2.0 / n)

    for f in (1, 2, 3, 4):
        h = 0.5 + 0.5 * np.sin(np.pi * f * xs)
        candidates.append(enforce_sum(np.clip(h, 0, 1), n / 2.0, rng))
        h = 0.5 + 0.5 * np.cos(np.pi * f * xs)
        candidates.append(enforce_sum(np.clip(h, 0, 1), n / 2.0, rng))

    for _ in range(3):
        mask = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        mask[idx] = 1.0
        candidates.append(mask)

    for _ in range(5):
        candidates.append(random_feasible_vector(n, rng))

    best_h = None
    best_c = float("inf")
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    beta_list = [5, 20, 80, 200, 500, 1000, 2000]

    while time.time() - start_time < budget_s * 0.85:
        deadline = start_time + budget_s * 0.85
        if rng.random() < 0.55:
            h_new, c_new = worst_lag_descent(best_h, max_iters=2500, rng=rng, deadline=deadline)
        else:
            h_new, c_new = hill_climb_multi(best_h, rng, max_iters=2000, init_step=0.5, n_trials=6)
        if c_new < best_c:
            best_h, best_c = h_new, c_new
        if rng.random() < 0.35:
            for beta in beta_list:
                if time.time() - start_time > budget_s * 0.85:
                    break
                try:
                    h_ref = slsqp_optimize(best_h, beta, max_iter=400)
                except Exception:
                    continue
                c_ref = compute_c5(h_ref)
                if c_ref < best_c:
                    best_h, best_c = h_ref, c_ref

    # optional up‑sampling if time permits
    if time.time() - start_time < budget_s * 0.95:
        target_n = min(720, max(480, n * 2))
        if target_n != n:
            old_x = (np.arange(n) + 0.5) * (2.0 / n)
            new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
            h_up = np.interp(new_x, old_x, best_h)
            h_up = enforce_sum(h_up, target_n / 2.0, rng)
            deadline = start_time + budget_s
            h_up, c_up = worst_lag_descent(h_up, max_iters=4000, rng=rng, deadline=deadline)
            for beta in beta_list:
                if time.time() - start_time > budget_s:
                    break
                try:
                    h_up = slsqp_optimize(h_up, beta, max_iter=400)
                except Exception:
                    pass
                c_up = compute_c5(h_up)
                if c_up < best_c:
                    best_h, best_c, n = h_up, c_up, target_n

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
