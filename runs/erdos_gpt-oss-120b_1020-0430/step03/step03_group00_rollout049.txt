def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # Initialize base vector from initial_h_values or generate random feasible at 360 points
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = random_feasible_vector(360, rng)
    # enforce sum (target 360/2 = 180)
    h0 = enforce_sum(np.clip(h0, 0, 1), h0.size/2, rng)
    # upsample to n=720 via interpolation
    n0 = h0.size
    n = 720  # maybe upsample factor =2 
    xs0 = (np.arange(n0)+0.5)*(2.0/n0)
    xs = (np.arange(n)+0.5)*(2.0/n)
    h = np.interp(xs, xs0, h0)
    h = enforce_sum(np.clip(h,0,1), n/2, rng)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # Phase 1: Intense hill climb targeted
    # We'll allocate ~0.6 of budget
    phase1_budget = budget_s * 0.6
    while time.time() - start < start + phase1_budget:
        # propose start: small perturb of best or random
        if rng.random() < 0.7:
            start_vec = perturb_random(best_h, rng, scale=0.3)
        else:
            start_vec = random_feasible_vector(n, rng)
        # hill climbing with many iterations
        h_local, c_local = hill_climb_targeted(start_vec, rng,
                                                max_iters=8000,
                                                start_step=0.5,
                                                target_sum=n/2)
        if c_local < best_c:
            best_h, best_c = h_local, c_local

    # Phase 2: SLSQP refinement with high beta schedule
    for beta in [5,20,80,200,800,2000,5000]:
        if time.time() - start > budget_s * 0.95:
            break
        try:
            h_ref = slsqp_optimize(best_h, beta, max_iter=800)
        except Exception:
            continue
        c_ref = compute_c5(h_ref)
        if c_ref < best_c:
            best_h, best_c = h_ref, c_ref

    # Phase 3: final random fine-tuning
    while time.time() - start < budget_s * 0.99:
        new_h = perturb_random(best_h, rng, scale=0.02)
        c_new = compute_c5(new_h)
        if c_new < best_c:
            best_h, best_c = new_h, c_new

    # Clean up
    best_h = np.clip(best_h, 0,1)
    best_h = enforce_sum(best_h, best_h.size/2, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)

import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def enforce_sum(h, target, rng):
    """Clip to [0,1] and adjust to make sum exactly target (simple water‑fill)."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # quick deterministic adjustment for tiny residuals
    if abs(diff) < 1e-12:
        return h
    # water‑fill style: add to those below 1, remove from those above 0
    while abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    # final catch‑all (should be very rare)
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    """Uniform random values projected onto the feasible box‑simplex."""
    v = rng.random(n)
    v = enforce_sum(v, n / 2.0, rng)
    return v

def compute_c5(h):
    """Discrete C5 for a vector h on [0,2] (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

# ---------- perturbations ----------
def perturb_random(h, rng, scale=0.2):
    """Transfer a small ε from i to j (preserving sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def perturb_targeted(h, grad, rng, scale=0.2):
    """Move mass from positive‑gradient to negative‑gradient coordinates."""
    pos = np.where((grad > 1e-9) & (h > 1e-12))[0]
    neg = np.where((grad < -1e-9) & (h < 1.0 - 1e-12))[0]
    if pos.size == 0 or neg.size == 0:
        return perturb_random(h, rng, scale)
    i = rng.choice(pos)
    j = rng.choice(neg)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return perturb_random(h, rng, scale)
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

# ---------- hill climbing ----------
def hill_climb_targeted(h_start, rng, max_iters=5000, start_step=0.5, step_decay=0.9):
    """Focused hill climb that always attacks the worst shift."""
    h = h_start.copy()
    n = h.size
    best_c = compute_c5(h)
    best_h = h.copy()
    step = start_step
    decay_interval = max(1, max_iters // 10)

    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr)
        max_corr = corr[idx]
        c_current = max_corr * (2.0 / n)
        k = idx - (n - 1)

        # gradient of the current worst shift
        grad = np.ones(n)
        if k >= 0:
            grad[:n - k] -= h[k:]
            grad[k:]   -= h[:n - k]
        else:
            kk = -k
            grad[:n - kk] -= h[kk:]
            grad[kk:]    -= h[:n - kk]

        new_h = perturb_targeted(h, grad, rng, scale=step)
        c_new = compute_c5(new_h)

        if c_new < best_c:
            best_c, best_h = c_new, new_h.copy()
        if c_new < c_current:
            h = new_h

        if (it + 1) % decay_interval == 0:
            step = max(step * step_decay, 1e-5)

    return best_h, best_c

# ---------- SLSQP smoothing ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def _softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp, scaled back
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=800):
    """Smooth‑max SLSQP refinement for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(_softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    x = enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x
