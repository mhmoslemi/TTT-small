def enforce_sum(h, target, rng=None):
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        caps = 1.0 - h
        total = caps.sum()
        if total < 1e-12:
            return h  # cannot adjust
        add = caps * (diff / total)
        h += add
    else:
        caps = h
        total = caps.sum()
        if total < 1e-12:
            return h
        sub = caps * (-diff / total)
        h -= sub
    # Clip again to ensure any numerical issues
    h = np.clip(h, 0.0, 1.0)
    # May still have small residual diff due to rounding; could repeat few times
    # For safety, iterate up to 3 times
    for _ in range(3):
        cur = h.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            caps = 1.0 - h
            total = caps.sum()
            if total < 1e-12:
                break
            add = caps * (diff / total)
            h += add
        else:
            caps = h
            total = caps.sum()
            if total < 1e-12:
                break
            sub = caps * (-diff / total)
            h -= sub
        h = np.clip(h, 0.0, 1.0)
    return h

def perturb_random(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n-1)
    if j >= i:
        j += 1 # ensure j != i (but keep in range)
    # Actually simpler: while loop ensure i!=j
    while j == i:
        j = rng.integers(0, n)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def perturb_targeted_move(h, grad, rng, scale=0.2):
    n = h.size
    pos_idx = np.where((grad > 1e-9) & (h > 1e-12))[0]
    neg_idx = np.where((grad < -1e-9) & (h < 1.0 - 1e-12))[0]
    if pos_idx.size == 0 or neg_idx.size == 0:
        return perturb_random(h, rng, scale)
    i = rng.choice(pos_idx)
    j = rng.choice(neg_idx)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_random(h, rng, scale)
    eps = max_eps * scale * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

Now compute_c5:

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def hill_climb_targeted(h_start, rng, max_iters, step_start):
    h = h_start.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)

    step = step_start
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        # compute worst shift k and gradient
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr)
        max_corr = corr[idx]
        c_current = max_corr * (2.0 / n)
        k = idx - (n - 1)
        # gradient
        grad = np.ones(n)
        if k >= 0:
            grad[:n - k] -= h[k:]
            grad[k:] -= h[:n - k]
        else:
            kk = -k
            grad[:n - kk] -= h[kk:]
            grad[kk:] -= h[:n - kk]

        new_h = perturb_targeted_move(h, grad, rng, scale=step)
        # enforce sum again? Not needed because perturb maintains sum; still clip due to numeric? Might be safe to call enforce_sum
        new_h = enforce_sum(np.clip(new_h, 0.0, 1.0), n/2.0, rng)
        c_new = compute_c5(new_h)

        if c_new < best_c:
            best_c = c_new
            best_h = new_h.copy()
        if c_new < c_current:
            h = new_h
        # else maybe accept if improvement on gradient?
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)

    return best_h, best_c

def slsqp_optimize(h0, beta, max_iter, rng):
    n = h0.size
    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0,1.0)] * n
    cons = {"type":"eq","fun": sum_eq_constraint}
    from scipy.optimize import minimize
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    x = enforce_sum(x, CURRENT_TARGET, rng)
    return x

import time
import numpy as np
from scipy.optimize import minimize

# ----- global parameters for SLSQP -----
CURRENT_TARGET = None
CURRENT_BETA = None

# ----- helper functions -----
def enforce_sum(h, target, rng=None):
    """Clip h to [0,1] and adjust linearly to achieve exact sum=target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # distribute proportionally to available capacity
    if diff > 0:
        caps = 1.0 - h
    else:
        caps = h
    total = caps.sum()
    if total <= 0:
        return h
    h += caps * (diff / total)
    # repeat a few times to clean rounding errors
    for _ in range(3):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            caps = 1.0 - h
        else:
            caps = h
        total = caps.sum()
        if total <= 0:
            break
        h += caps * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def perturb_random(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if j >= i:
        j += 1
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def perturb_targeted_move(h, grad, rng, scale=0.2):
    pos = np.where((grad > 1e-9) & (h > 1e-12))[0]
    neg = np.where((grad < -1e-9) & (h < 1.0 - 1e-12))[0]
    if pos.size == 0 or neg.size == 0:
        return perturb_random(h, rng, scale)
    i = rng.choice(pos)
    j = rng.choice(neg)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return perturb_random(h, rng, scale)
    eps = max_eps * scale * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb_targeted(h_start, rng, max_iters, step_start):
    h = h_start.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step_start
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr)
        k = idx - (n - 1)

        # gradient of the worst shift
        grad = np.ones(n)
        if k >= 0:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
        else:
            kk = -k
            grad[: n - kk] -= h[kk:]
            grad[kk:] -= h[: n - kk]

        new_h = perturb_targeted_move(h, grad, rng, scale=step)
        new_h = enforce_sum(np.clip(new_h, 0.0, 1.0), n / 2.0, rng)
        c_new = compute_c5(new_h)

        if c_new < best_c:
            best_c = c_new
            best_h = new_h.copy()
        if c_new < compute_c5(h):
            h = new_h

        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

# ----- SLSQP helpers -----
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter, rng):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return enforce_sum(x, CURRENT_TARGET, rng)

def optimize_at_resolution(h_init, n, time_allowed, rng, global_start, total_budget):
    local_start = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_c = compute_c5(h)

    # ---------- Simulated Annealing ----------
    T0 = 0.02
    Tmin = 0.0002
    alpha = 0.995
    T = T0
    c_cur = best_c
    while time.time() - local_start < time_allowed and T > Tmin:
        scale = max(T / T0 * 0.3, 0.005)
        cand = perturb_random(h, rng, scale=scale)
        c_new = compute_c5(cand)
        if c_new < best_c:
            best_c = c_new
            best_h = cand.copy()
        if c_new < c_cur or rng.random() < np.exp(-(c_new - c_cur) / T):
            h = cand
            c_cur = c_new
        T *= alpha

    # ---------- Targeted hill climbing ----------
    time_spent = time.time() - local_start

    if time_left > 4.0:
