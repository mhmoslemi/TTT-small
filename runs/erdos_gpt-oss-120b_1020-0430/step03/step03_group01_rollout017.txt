def perturb_h_sumpreserve(h, rng, scale):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)

import time
import numpy as np
from scipy.optimize import minimize

# globals for SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth log‑sum‑exp surrogate of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def enforce_sum(h, target, rng, max_iter=2000):
    """Adjust h to have Σh = target while staying in [0,1]."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add == 0:
            continue
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        for i in range(n):
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h


def compute_c5(h):
    """Exact discrete C5 using FFT (linear correlation)."""
    n = h.size
    a = h
    b_rev = (1.0 - h)[::-1]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.fft(a, size)
    B = np.fft.fft(b_rev, size)
    conv = np.fft.ifft(A * B).real
    c = conv[: 2 * n - 1]
    return np.max(c) * (2.0 / n)


def perturb_h_sumpreserve(h, rng, scale=0.2):
    """Transfer ε from i to j (preserves total)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def random_feasible_vector(n, rng):
    """Random h ∈[0,1]^n with Σh = n/2."""
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)


def crossover(parent1, parent2, rng):
    """Uniform crossover."""
    mask = rng.random(parent1.size) < 0.5
    child = np.where(mask, parent1, parent2)
    return child


def hill_climb_multi(h_start, rng, max_iters=2000, init_step=0.5, n_trials=5):
    """Simple hill‑climber using sum‑preserving perturbations."""
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h_sumpreserve(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
                improved = True
                break
        if not improved and rng.random() < 0.02:
            new_h = perturb_h_sumpreserve(h, rng, scale=step * 2.0)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return h, best_c


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum (tiny drift)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    end_time = start_time + budget_s * 0.96  # keep tiny margin for final clean‑up

    # -------------------------------------------------------------
    # 1. Determine resolution and initial vector
    # -------------------------------------------------------------
    if "initial_h_values" in globals():
        base_h = np.array(initial_h_values, dtype=float)
        base_n = base_h.size
    else:
        base_n = 360
        base_h = random_feasible_vector(base_n, rng)

    # we aim for a finer grid (720 points) if time allows
    target_n = 720
    if base_n != target_n:
        old_x = np.linspace(0.0, 2.0, base_n, endpoint=False) + (1.0 / base_n)
        new_x = np.linspace(0.0, 2.0, target_n, endpoint=False) + (1.0 / target_n)
        h0 = np.interp(new_x, old_x, base_h)
    else:
        h0 = base_h.copy()
    n = target_n
    h0 = enforce_sum(h0, n / 2.0, rng)

    # -------------------------------------------------------------
    # 2. Build initial population
    # -------------------------------------------------------------
    pop_size = 30
    population = [h0]
    pop_c = [compute_c5(h0)]
    for _ in range(pop_size - 1):
        h = random_feasible_vector(n, rng)
        population.append(h)
        pop_c.append(compute_c5(h))
    best_idx = int(np.argmin(pop_c))
    best_h = population[best_idx].copy()
    best_c = pop_c[best_idx]

    # -------------------------------------------------------------
    # 3. Evolutionary search
    # -------------------------------------------------------------
    mutation_scale = 0.5
    scale_decay = 0.9997
    last_report = time.time()
    while time.time() < end_time:
        # create child
        i, j = rng.choice(pop_size, size=2, replace=False)
        child = crossover(population[i], population[j], rng)
        # a few perturbations
        for _ in range(rng.integers(1, 4)):
            child = perturb_h_sumpreserve(child, rng, scale=mutation_scale)
        child = enforce_sum(child, n / 2.0, rng)
        c = compute_c5(child)

        # replace worst if better
        worst_idx = int(np.argmax(pop_c))
        if c < pop_c[worst_idx]:
            population[worst_idx] = child
            pop_c[worst_idx] = c
            if c < best_c:
                best_h, best_c = child.copy(), c

        # decay mutation scale
        mutation_scale = max(mutation_scale * scale_decay, 1e-4)

        # occasional early exit if we already beat target
        if best_c <= 0.3808:
            break

        # optional periodic reporting (no I/O, just keep time)
        if time.time() - last_report > 30.0:
            last_report = time.time()

    # -------------------------------------------------------------
    # 4. Local refinement (hill climbing)
    # -------------------------------------------------------------
    remaining = end_time - time.time()
    if remaining > 20.0:
        hc_iters = int(min(5000, remaining * 1000))
        best_h, best_c = hill_climb_multi(
            best_h, rng, max_iters=hc_iters, init_step=0.3, n_trials=6
        )

    # -------------------------------------------------------------
    # 5. Final SLSQP refinement with increasing β
    # -------------------------------------------------------------
    remaining = end_time - time.time()
    if remaining > 15.0:
        for beta in [5, 20, 80, 200, 500, 1000, 2000]:
            try:
                refined = slsqp_optimize(best_h, beta, max_iter=400)
                refined_c = compute_c5(refined)
                if refined_c < best_c:
                    best_h, best_c = refined, refined_c
                # stop early if already excellent
                if best_c <= 0.3808:
                    break
            except Exception:
                continue

    # -------------------------------------------------------------
    # 6. Final clean‑up
    # -------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
