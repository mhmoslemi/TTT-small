def binary_sa(n, rng, max_time):
    target = n // 2
    h = random_binary_vector(n, rng)
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    # lists of indices
    ones_idx = np.flatnonzero(h>0.5).tolist()
    zeros_idx = np.flatnonzero(h<0.5).tolist()
    start = time.time()
    T0 = 0.02
    Tmin = 1e-6
    T = T0
    decay = 0.9995
    it = 0
    while time.time() - start < max_time:
        # choose random 1 and 0
        if not ones_idx or not zeros_idx:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        # propose swap
        h[i] = 0.0
        h[j] = 1.0
        # update lists
        # swap positions in the lists
        # remove i from ones_idx, add to zeros_idx
        # remove j from zeros_idx, add to ones_idx
        # but need O(1) updates: we can locate positions in list and swap with last.
        # Simpler: recompute lists each few iterations
        # To keep code simple, we recompute lists each iteration: but that costs O(n). May be okay as n~300 and iterations maybe tens of thousands (< 1e5). So okay.
        c_new = compute_c5(h)
        accept = False
        if c_new < cur_c:
            accept = True
        else:
            if rng.random() < np.exp(-(c_new - cur_c) / max(T, 1e-12)):
                accept = True
        if accept:
            cur_c = c_new
            # update best
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
        # temperature decay
        T = max(T * decay, Tmin)
        it += 1
        # periodically recompute lists
        if it % 100 == 0:
            ones_idx = np.flatnonzero(h>0.5).tolist()
            zeros_idx = np.flatnonzero(h<0.5).tolist()

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Helper functions ----------
def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_to_feasible(v, target):
    """Project vector onto {0≤v≤1, sum(v)=target} by shifting then clipping."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Generate a random vector with 0≤h≤1 and sum = n/2."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_to_feasible(v, n / 2.0)


def upsample_initial(to_n):
    """Linear up‑sample the provided initial_h_values to length to_n."""
    init = np.array(initial_h_values, dtype=float)
    m = init.size
    xs_old = np.linspace(0.0, 2.0, m, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, to_n, endpoint=False)
    return np.interp(xs_new, xs_old, init)


# ---------- Soft‑max surrogate for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=500):
    """Run a single SLSQP optimisation using the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # exact sum correction (project onto feasible hyperplane)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ---------- Binary simulated annealing ----------
def random_binary_vector(n, rng):
    """Binary vector with exactly n/2 ones."""
    ones = n // 2
    v = np.zeros(n, dtype=float)
    v[rng.choice(n, size=ones, replace=False)] = 1.0
    return v


def binary_sa(n, rng, max_time, h0=None):
    """Simulated annealing on binary vectors (swap a 1 and a 0)."""
    target = n // 2
    if h0 is None:
        h = random_binary_vector(n, rng)
    else:
        h = h0.copy()
    # initialise index lists
    ones_idx = list(np.flatnonzero(h > 0.5))
    zeros_idx = list(np.where(h == 0)[0])

    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c

    start = time.time()
    T = 0.015          # initial temperature
    Tmin = 1e-6
    decay = 0.99985    # slow exponential cooling

    it = 0
    while time.time() - start < max_time:
        if not ones_idx or not zeros_idx:
            break
        i_pos = rng.integers(0, len(ones_idx))
        j_pos = rng.integers(0, len(zeros_idx))
        i = ones_idx[i_pos]
        j = zeros_idx[j_pos]

        # propose swap
        h[i] = 0.0
        h[j] = 1.0
        c_new = compute_c5(h)

        accept = False
        if c_new < cur_c:
            accept = True
        else:
            if rng.random() < np.exp(-(c_new - cur_c) / max(T, 1e-12)):
                accept = True

        if accept:
            cur_c = c_new
            # update index lists (swap positions)
            ones_idx[i_pos] = j
            zeros_idx[j_pos] = i
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0

        # temperature schedule
        T = max(T * decay, Tmin)
        it += 1
        # periodic sanity check (very cheap)
        if it % 2000 == 0:
            if cur_c > best_c * 1.2:  # reset if stuck badly
                h = best_h.copy()
                ones_idx = list(np.flatnonzero(h > 0.5))
                zeros_idx = list(np.where(h == 0)[0])

    return best_h, best_c


# ---------- Simple hill‑climb (deterministic) ----------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.2):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c
