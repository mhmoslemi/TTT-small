def compute_gradient_softmax(h, beta):
    n = len(h)
    conv = correlate_full(h)  # conv size 2n-1
    # compute weights via softmax
    bc = beta * conv
    # stable
    max_bc = bc.max()
    exp_bc = np.exp(bc - max_bc)
    w = exp_bc / exp_bc.sum()  # shape (2n-1,)
    # compute gradient
    grad = np.zeros(n)
    # offset range: -(n-1) .. (n-1); conv index k corresponds to offset = k - (n-1)
    # for each offset, weight w_k
    for k, wk in enumerate(w):
        offset = k - (n - 1)
        if wk == 0:
            continue
        # term: (1 - h[i+offset]) - h[i - offset]
        # For each i:
        # If i+offset in range, add wk*(1 - h[i+offset])
        # If i - offset in range, subtract wk*h[i - offset]
        if offset >= 0:
            # i+offset < n -> i < n - offset
            # for i in 0..n-offset-1
            i_end = n - offset
            # add wk * (1 - h[i+offset]) to grad[0:i_end]
            grad[:i_end] += wk * (1 - h[offset:])  # h[i+offset] = h[offset: i_end+offset]
            # subtract wk * h[i - offset] -> i >= offset
            # i from offset..n-1
            grad[offset:] -= wk * h[:n - offset]
        else:
            # offset negative: offset = -d where d>0
            d = -offset
            # term (1 - h[i - d]) - h[i + d]
            # i - d >=0 => i >= d: for i in d..n-1
            i_start = d
            # add wk * (1 - h[i - d]) to grad[d:]
            grad[d:] += wk * (1 - h[:n - d])
            # subtract wk * h[i + d] for i+d < n => i < n - d
            i_end = n - d
            grad[:i_end] -= wk * h[d:]
    # gradient of objective (softmax of corr) w.r.t h is computed as above (since objective = (1/β) * log_sum_exp(β* corr), gradient = sum w_k * grad(C_k)).
    return grad

def compute_gradient_softmax(h, beta):
    n = h.shape[0]
    conv = correlate_full(h)  # length 2n-1
    # compute softmax weights
    bc = beta * conv
    max_bc = bc.max()
    exp_bc = np.exp(bc - max_bc)
    w = exp_bc / exp_bc.sum()
    # forward convolution (w * h)
    # Using np.convolve
    forward_full = np.convolve(w, h, mode='full')
    start = n - 1
    forward = forward_full[start:start+n]
    # backward convolution using reversed w
    w_rev = w[::-1]
    backward_full = np.convolve(w_rev, h, mode='full')
    backward = backward_full[start:start+n]
    grad = 1.0 - forward - backward
    return grad

def enforce_sum(h, target, rng):
    """Adjust h so that Σh == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    enforce_sum(h, n / 2.0, rng)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def correlate_full(h):
    """Full linear correlation of h with (1‑h) via FFT."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Discrete C₅ for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)


# ----------------------------------------------------------------------
# Soft‑max surrogate and its gradient
# ----------------------------------------------------------------------
def compute_gradient_softmax(h, beta):
    """Gradient of (1/β)·log‑sum‑exp(β·Cₖ) w.r.t. h."""
    n = h.shape[0]
    conv = correlate_full(h)                     # Cₖ values, length 2n‑1
    bc = beta * conv
    m = bc.max()
    w = np.exp(bc - m)
    w /= w.sum()                                 # soft‑max weights, sum=1
    # forward term Σ wₖ·h[i+offset]
    forward_full = np.convolve(w, h, mode="full")
    start = n - 1
    forward = forward_full[start:start + n]
    # backward term Σ wₖ·h[i‑offset]
    backward_full = np.convolve(w[::-1], h, mode="full")
    backward = backward_full[start:start + n]
    return 1.0 - forward - backward


def projected_gradient_descent(h0, beta, steps, step_size, rng):
    """Simple projected gradient descent on the soft‑max surrogate."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    for it in range(steps):
        grad = compute_gradient_softmax(h, beta)
        alpha = step_size * (0.95 ** it)
        h -= alpha * grad
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h = enforce_sum(h, target, rng)
    return h


# ----------------------------------------------------------------------
# SLSQP on the same surrogate (kept from previous solution)
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    conv = correlate_full(x)
    bc = CURRENT_BETA * conv
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ----------------------------------------------------------------------
# Simulated annealing (unchanged)
# ----------------------------------------------------------------------
def simulated_annealing(start_h, rng, deadline, T0=0.02, Tf=1e-5, max_iters=1500):
    cur_h = start_h.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        T = T0 * ((Tf / T0) ** (it / max_iters))
        new_h = perturb_h(cur_h, rng, scale=0.35)
        new_c = compute_c5(new_h)
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / max(T, 1e-12)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
    return best_h, best_c


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.97

    # ---- initial vector ------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        enforce_sum(h0, h0.size / 2.0, rng)
    else:
        h0 = np.full(240, 0.5)
        enforce_sum(h0, h0.size / 2.0, rng)

    # ---- optional up‑sampling to 720 points ---------------------------
    target_n = 720
    if h0.size < target_n and target_n % h0.size == 0:
        factor = target_n // h0.size
        h = np.repeat(h0, factor)
        n = target_n
    else:
        h = h0.copy()
        n = h0.size
    enforce_sum(h, n / 2.0, rng)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ---- gradient‑descent phase ---------------------------------------
    time_mid = start_time + budget_s * 0.7
    betas = [5, 20, 80, 200, 800, 2000, 5000]
    for beta in betas:
        if time.time() > time_mid:
            break
        # refine current best
        h_opt = projected_gradient_descent(best_h, beta, steps=250, step_size=0.4, rng=rng)
        c_opt = compute_c5(h_opt)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
        # a couple of random restarts
        for _ in range(2):
            if time.time() > time_mid:
                break
            h_start = random_feasible_vector(n, rng)
            h_opt = projected_gradient_descent(h_start, beta, steps=250, step_size=0.4, rng=rng)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c:
                best_h, best_c = h_opt, c_opt
        # SLSQP fine‑tune
        if time.time() > time_mid:
            break
        try:
            h_slsqp = slsqp_optimize(best_h, beta, max_iter=250)
            c_slsqp = compute_c5(h_slsqp)
            if c_slsqp < best_c:
                best_h, best_c = h_slsqp, c_slsqp
        except Exception:
            pass

    # ---- simulated‑annealing sweep ------------------------------------
    time_sa_end = start_time + budget_s * 0.9
    while time.time() < time_sa_end:
        start = best_h if rng.random() < 0.7 else random_feasible_vector(n, rng)
        cand_h, cand_c = simulated_annealing(start, rng, deadline=time_sa_end,
                                            T0=0.1, Tf=1e-6, max_iters=1800)
        if cand_c < best_c:
            best_h, best_c = cand_h, cand_c

    # ---- final clean‑up ------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
