def targeted_move(h, rng, scale):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    lag = max_idx - (n - 1)
    # Determine overlapping region
    if lag >= 0:
        i_range = np.arange(0, n - lag)
        j_range = i_range + lag
    else:
        i_range = np.arange(-lag, n)
        j_range = i_range + lag  # will be < i_range
    # compute contributions
    contrib = h[i_range] * (1 - h[j_range])
    # select source i with max contrib
    src_idx = i_range[np.argmax(contrib)]
    # For targets, we want smallest (1 - h[j_range]), i.e., largest h[j_range]
    sorted_idx = np.argsort(1 - h[i_range + lag])  # ascending q (i.e., descending h)
    # iterate to find suitable target not same as src and not saturated
    target_idx = None
    for idx in sorted_idx:
        t = i_range[idx]
        if t != src_idx and h[t] < 1.0 - 1e-12:
            target_idx = t
            break
    if target_idx is None:
        # fallback random
        t = rng.integers(0, n)
        if t == src_idx:
            return h
        target_idx = t
    max_eps = min(h[src_idx], 1.0 - h[target_idx])
    if max_eps <= 1e-12:
        return h
    eps = scale * max_eps * rng.random()
    if eps <= 1e-12:
        return h
    new_h = h.copy()
    new_h[src_idx] -= eps
    new_h[target_idx] += eps
    # clip not needed as eps respects bounds
    return new_h

def random_move(h, rng, scale):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0.0:
            return h
        # swap roles
        i, j = j, i
    eps = scale * max_eps * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

class SoftmaxObj:
    def __init__(self, beta):
        self.beta = beta
    def __call__(self, x):
        corr = np.correlate(x, 1 - x, mode='full')
        bc = self.beta * corr
        max_bc = np.max(bc)
        # Compute softmax (log-sum-exp)
        # use stable trick
        return (max_bc + np.log(np.sum(np.exp(bc - max_bc)))) / self.beta

def slsqp_optimize(h0, beta, max_iter=250):
    n = h0.size
    target = n / 2.0
    obj = SoftmaxObj(beta)
    bounds = [(0.0,1.0)]*n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    # But lambda is a closure; we can define a function with target as default argument:
    def eq_con(x, t=target):
        return np.sum(x) - t
    cons = {"type": "eq", "fun": eq_con}
    res = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol':1e-12, 'disp':False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum as before
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # distribute diff
        n = x.size
        # simple approach: add diff/n to each, then clip
        x += diff / n
        x = np.clip(x, 0.0, 1.0)
        # after clipping sum may have changed; adjust again
        diff2 = target - x.sum()
        # adjust one element to fix
        if abs(diff2) > 1e-12:
            for i in range(n):
                add = min(diff2, 1.0 - x[i]) if diff2 > 0 else max(diff2, -x[i])
                x[i] += add
                diff2 = target - x.sum()
                if abs(diff2) < 1e-12:
                    break
    return x

# Utility functions

def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function h on [0,2] with uniform grid."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator, max_iter: int = 2000) -> np.ndarray:
    """Clip to [0,1] and adjust to satisfy Σh = target (within 1e-12)."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    n = h.size
    # Uniform correction first
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # Random redistribution of the remaining error
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add == 0.0:
            continue
        h[i] += add
        diff = target - h.sum()
    # Final clip (should be exact now)
    h = np.clip(h, 0.0, 1.0)
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector in [0,1]^n with Σ = n/2."""
    target = n / 2.0
    h = np.full(n, target / n)  # start at 0.5 everywhere

def targeted_move(h: np.ndarray, rng: np.random.Generator, scale: float) -> np.ndarray:
    """Transfer a small amount from a high‑contribution index to a low‑partner index."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(corr))
    lag = max_idx - (n - 1)  # could be negative

    if lag >= 0:
        i_range = np.arange(0, n - lag)
        j_range = i_range + lag
    else:
        i_range = np.arange(-lag, n)
        j_range = i_range + lag

    contrib = h[i_range] * (1.0 - h[j_range])
    src = i_range[np.argmax(contrib)]

    # target: want smallest (1 - h[j]) i.e. largest h[j]
    partner_vals = h[j_range]
    order = np.argsort(1.0 - partner_vals)  # ascending by (1 - h[j])
    tgt = None
    for idx in order:
        cand = i_range[idx]
        if cand != src and h[cand] < 1.0 - 1e-12:
            tgt = cand
            break
    if tgt is None:
        # fallback to a random different index
        cand = rng.integers(0, n)
        if cand == src:
            return h
        tgt = cand

    max_eps = min(h[src], 1.0 - h[tgt])
    if max_eps <= 1e-14:
        return h
    eps = scale * max_eps * rng.random()
    if eps <= 1e-14:
        return h
    new_h = h.copy()
    new_h[src] -= eps
    new_h[tgt] += eps
    return new_h


def random_move(h: np.ndarray, rng: np.random.Generator, scale: float) -> np.ndarray:
    """Transfer a small amount between two random coordinates (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        # try swapping the direction
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0.0:
            return h
        i, j = j, i
    eps = scale * max_eps * rng.random()
    if eps <= 1e-14:
        return h
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

# Smooth surrogate for SLSQP

class SoftmaxObj:
    """Smooth approximation of the max‑over‑k using log‑sum‑exp."""
    def __init__(self, beta: float):
        self.beta = beta

    def __call__(self, x: np.ndarray) -> float:
        corr = np.correlate(x, 1.0 - x, mode="full")
        bc = self.beta * corr
        m = np.max(bc)
        # stable log‑sum‑exp
        return (m + np.log(np.sum(np.exp(bc - m)))) / self.beta


class SumConstraint:
    """Equality constraint Σx = target."""
    def __init__(self, target: float):
        self.target = target

    def __call__(self, x: np.ndarray) -> float:
        return np.sum(x) - self.target


def slsqp_optimize(h0: np.ndarray, beta: float, max_iter: int = 300) -> np.ndarray:
    """Run SLSQP on the smooth surrogate for a given β."""
    n = h0.size
    target = n / 2.0
    obj = SoftmaxObj(beta)
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": SumConstraint(target)}
    res = minimize(
        obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # Repair tiny drift in the sum
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, target, np.random.default_rng())
    return x

            c_new = compute_c5(cand_up)
            if c_new < best_c:
                best_h, best_c = cand_up, c_new
            it_up += 1

    # ----- Final clean‑up -------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
