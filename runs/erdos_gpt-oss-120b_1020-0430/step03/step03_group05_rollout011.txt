def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb(h_start, rng, max_iters=2000, start_step=0.3):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global placeholders for the soft‑max surrogate used by SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def project_to_feasible(v, target):
    """Projection of v onto {0≤h≤1, Σh = target} via a bisection on the shift."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def random_feasible_binary(n, rng):
    """Fast generation of a random binary feasible vector (0/1, Σ = n/2)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def init_binary_from_initial(n, rng):
    """Create a binary start vector from the provided initial_h_values, if any."""
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)
        if init.size != n:
            xs_old = np.linspace(0.0, 2.0, init.size, endpoint=False)
            xs_new = np.linspace(0.0, 2.0, n, endpoint=False)
            init = np.interp(xs_new, xs_old, init)
        binary = (init > 0.5).astype(float)
        target = n // 2
        cur = int(binary.sum())
        if cur > target:
            ones = np.where(binary == 1)[0]
            idx = rng.choice(ones, size=cur - target, replace=False)
            binary[idx] = 0
        elif cur < target:
            zeros = np.where(binary == 0)[0]
            idx = rng.choice(zeros, size=target - cur, replace=False)
            binary[idx] = 1
        return binary
    return random_feasible_binary(n, rng)

def sa_optimize(n, rng, max_time):
    """Simulated‑annealing on binary vectors (swap moves)."""
    start = time.time()
    cur_h = init_binary_from_initial(n, rng).astype(float)
    best_h = cur_h.copy()
    best_c = compute_c5(cur_h)
    cur_c = best_c

    T0 = max(best_c * 0.2, 1e-6)
    T_min = 1e-8

    while time.time() - start < max_time:
        elapsed = time.time() - start
        frac = elapsed / max_time
        T = T0 * (1.0 - frac) ** 2 + T_min

        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_c = compute_c5(cur_h)
        delta = new_c - cur_c
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] = 1.0
            cur_h[j] = 0.0
    return best_h, best_c

def slsqp_optimize(h0, beta, rng, max_iter=300):
    """Run SLSQP with the soft‑max surrogate for a given beta."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # Exact sum correction (project onto feasible hyperplane)
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            if abs(add) < 1e-15:
                continue
            x[i] += add
            diff = target - x.sum()
    return x

def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb(h_start, rng, max_iters=1500, start_step=0.2):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c

# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    candidate_ns = [60, 84, 120, 168, 240, 300]
    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        if n0 not in candidate_ns:
            candidate_ns = [n0] + candidate_ns

    for n in candidate_ns:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining < 5.0:
            break

        # time budget for this discretisation
        time_for_n = min(remaining * 0.5, max(5.0, min(180.0, n * 0.6)))
        if time_for_n < 5.0:
            continue

        # ---- Simulated annealing (global) ----
        sa_time = time_for_n * 0.6
        h_sa, c_sa = sa_optimize(n, rng, sa_time)

        # ---- SLSQP refinement (smooth) ----
        h_cur = h_sa.astype(float)
        c_cur = c_sa
        for beta in (10.0, 50.0, 200.0, 800.0, 2000.0):
            if time.time() - start_time > budget_s * 0.96:
                break
            try:
                h_cur = slsqp_optimize(h_cur, beta, rng, max_iter=300)
            except Exception:
                continue
            c_cur = compute_c5(h_cur)
            if c_cur < best_c:
                best_c, best_h, best_n = c_cur, h_cur.copy(), n

        # ---- Final local hill‑climb ----
        if time.time() - start_time < budget_s * 0.98:
            h_cur, c_cur = hill_climb(h_cur, rng, max_iters=1500, start_step=0.2)
            if c_cur < best_c:
                best_c, best_h, best_n = c_cur, h_cur.copy(), n

    # ------------------------------------------------------------------
    # Fallback in the unlikely event nothing was found
    if best_h is None:
        best_n = 60
        best_h = np.full(best_n, 0.5)

    # Final feasibility enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, best_n)
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
