def project_to_feasible(y, target):
    lower, upper = 0.0, 1.0
    low = y.min() - upper
    high = y.max() - lower
    for _ in range(50):
        mid = (low + high) / 2.0
        proj = np.clip(y - mid, lower, upper)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    proj = np.clip(y - (low+high)/2.0, lower, upper)
    # ensure sum close; maybe adjust small diff by adding to interior values
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        # distribute diff uniformly among indices that are not at bounds.
        free = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if free.any():
            proj[free] += diff / free.sum()
    return proj

def compute_gradient(h, beta):
    n = h.shape[0]
    L = correlate_full(h)  # shape 2n-1
    # compute w (log-sum-exp)
    # For numerical stability, subtract max:
    maxL = np.max(L)
    expL = np.exp(beta * (L - maxL))
    w = expL / np.sum(expL)
    # convolution
    len_w = L.shape[0]  # 2n-1
    len_h = n
    len_conv = len_w + len_h - 1  # 3n-2
    N_fft = 1 << ((len_conv - 1).bit_length())
    H = np.fft.rfft(h, N_fft)
    W = np.fft.rfft(w, N_fft)
    conv1_full = np.fft.irfft(H * W, N_fft)[:len_conv]
    W_rev = np.fft.rfft(w[::-1], N_fft)
    conv2_full = np.fft.irfft(H * W_rev, N_fft)[:len_conv]
    start = n-1
    end = start + n
    conv1 = conv1_full[start:end]
    conv2 = conv2_full[start:end]
    grad = 1.0 - (conv1 + conv2)
    return grad

import time
import numpy as np

# ------------------------------------------------------------
# Basic feasibility helpers
# ------------------------------------------------------------
def enforce_sum(h, target, rng):
    """Adjust h to have sum == target while keeping 0 <= h <= 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    h = rng.random(n)
    target = n / 2.0
    s = h.sum()
    if s > 0:
        h = h * (target / s)
    else:
        h = np.full(n, target / n)
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, target, rng)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (preserves the sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def project_to_feasible(y, target):
    """Projection onto {0<=x<=1, sum(x)=target} using bisection on a shift."""
    lo = y.min() - 1.0      # ensures clip = 1 for all
    hi = y.max()            # ensures clip = 0 for all
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        if free.any():
            x[free] += diff / free.sum()
    return x


# ------------------------------------------------------------
# Correlation utilities
# ------------------------------------------------------------
def correlate_full(h):
    """Full linear correlation of h with (1-h) via FFT."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Discrete C5 = max_k ∫ h(x)(1‑h(x+k))dx (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)


# ------------------------------------------------------------
# Gradient of smooth max objective
# ------------------------------------------------------------
def compute_gradient(h, beta):
    """Gradient of (1/β)·log‑sum‑exp(β·L_k) where L_k = corr_k."""
    n = h.size
    L = correlate_full(h)                     # shape 2n‑1
    maxL = np.max(L)
    expL = np.exp(beta * (L - maxL))
    w = expL / expL.sum()                     # w_k ≥ 0, Σ w_k = 1

    # Convolution w * h and w_rev * h
    len_w = L.shape[0]                        # 2n‑1
    len_h = n
    conv_len = len_w + len_h - 1               # 3n‑2
    Nfft = 1 << ((conv_len - 1).bit_length())
    H = np.fft.rfft(h, Nfft)
    W = np.fft.rfft(w, Nfft)
    conv1_full = np.fft.irfft(H * W, Nfft)[:conv_len]
    W_rev = np.fft.rfft(w[::-1], Nfft)
    conv2_full = np.fft.irfft(H * W_rev, Nfft)[:conv_len]

    start = n - 1
    end = start + n
    conv1 = conv1_full[start:end]
    conv2 = conv2_full[start:end]

    grad = 1.0 - (conv1 + conv2)
    return grad

# Projected gradient descent on the smooth surrogate
# ------------------------------------------------------------
def pgd_optimize(start_h, beta_list, iters_per_beta, step0, step_decay, deadline):
    h = start_h.copy()
    target = h.size / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)

    for beta in beta_list:
        step = step0
        for _ in range(iters_per_beta):
            if time.time() >= deadline:
                return best_h, best_c
            grad = compute_gradient(h, beta)
            y = h - step * grad
            h = project_to_feasible(y, target)
            step *= step_decay
            c = compute_c5(h)
            if c < best_c:
                best_c, best_h = c, h.copy()
    return best_h, best_c
