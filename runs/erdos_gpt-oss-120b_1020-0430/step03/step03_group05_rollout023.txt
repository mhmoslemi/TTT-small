def binary_simulated_annealing(n, rng, time_limit):
    start = time.time()
    # init random binary vector with n/2 ones.
    h = np.zeros(n, dtype=int)
    h[:n//2] = 1
    rng.shuffle(h)
    best_h = h.copy()
    # compute C5 for binary
    best_c = compute_c5(h.astype(float))
    cur_c = best_c
    # Temperature schedule: start high then cool.
    # Setting T0 = best_c * 0.1 maybe
    T0 = best_c * 0.1
    T = T0
    decay = 0.995
    # maintain ones, zeros for fast sampling
    while time.time() - start < time_limit:
        # get indices of ones and zeros
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)  # or h == 0
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # propose swap
        h[i], h[j] = 0, 1
        new_c = compute_c5(h.astype(float))
        delta = new_c - cur_c
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            # accept
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # reject, revert
            h[i], h[j] = 1, 0
        # temperature cooling
        T *= decay
        # if T becomes too small, break
        if T < 1e-12:
            T = 1e-12
    return best_h, best_c

def de_optimize_real(n, rng, max_time, init=None):
    target = n / 2.0
    popsize = max(15, min(50, int(0.2 * n)))  # moderate
    pop = []
    # if init provided, add it
    if init is not None:
        # Ensure init length n
        if len(init) != n:
            # upsample to n
            # use interpolation
            xs = np.linspace(0.0, 2.0, len(init), endpoint=False)
            xs_new = np.linspace(0.0, 2.0, n, endpoint=False)
            init_interp = np.interp(xs_new, xs, init)
        else:
            init_interp = init
        pop.append(project_to_feasible(init_interp, target))
    # fill rest with random feasible vectors
    while len(pop) < popsize:
        pop.append(random_feasible_vector(n, rng))
    pop = np.array(pop)
    scores = np.array([compute_c5(ind) for ind in pop])
    best_idx = scores.argmin()
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]

    F = 0.6
    CR = 0.9
    start = time.time()
    while time.time() - start < max_time:
        for i in range(popsize):
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, 3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            # crossover
            cross = rng.random(n) < CR
            if not np.any(cross):

                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_h = trial.copy()
    return best_h, best_score

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global placeholders for the soft‑max surrogate used by SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None

# ----------------------------------------------------------------------
def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_to_feasible(v, target):
    """Projection of v onto {0≤h≤1, Σh = target} via bisection on a shift."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Fast generation of a random feasible vector (0≤h≤1, Σh=n/2)."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_to_feasible(v, n / 2.0)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.3, time_budget=None):
    start = time.time()
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        if time_budget is not None and time.time() - start > time_budget:
            break
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


def slsqp_optimize(h0, beta, rng, max_iter=500):
    """Run a single SLSQP optimisation with the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # exact sum correction (project onto feasible hyperplane)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def de_optimize_real(n, rng, max_time, init=None):
    """Simple DE for real‑valued vectors with projection onto the feasible set."""
    target = n / 2.0
    popsize = max(15, min(50, int(0.2 * n)))
    pop = []

    # seed with provided initial if any
    if init is not None:
        if len(init) != n:
            # upsample linearly
            xs_old = np.linspace(0.0, 2.0, len(init), endpoint=False)
            xs_new = np.linspace(0.0, 2.0, n, endpoint=False)
            init = np.interp(xs_new, xs_old, init)
        pop.append(project_to_feasible(init, target))

    while len(pop) < popsize:
        pop.append(random_feasible_vector(n, rng))

    pop = np.array(pop)
    scores = np.array([compute_c5(ind) for ind in pop])
    best_idx = scores.argmin()
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]

    F = 0.6
    CR = 0.9
    start = time.time()
    while time.time() - start < max_time:
        for i in range(popsize):
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, 3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(0, n)] = True
            trial = np.where(cross, mutant, pop[i])
            trial = project_to_feasible(trial, target)
            trial_score = compute_c5(trial)
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_h = trial.copy()
    return best_h, best_score


def binary_simulated_annealing(n, rng, time_limit):
    """Binary (0/1) simulated annealing respecting exactly n/2 ones."""
    start = time.time()
    h = np.zeros(n, dtype=int)
    h[: n // 2] = 1
    rng.shuffle(h)
    cur_c = compute_c5(h.astype(float))
    best_h = h.copy()
    best_c = cur_c

    T = cur_c * 0.1 + 1e-6
    decay = 0.995
    while time.time() - start < time_limit:
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # propose swap
        h[i], h[j] = 0, 1
        new_c = compute_c5(h.astype(float))
        delta = new_c - cur_c
        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / max(T, 1e-12)):
                accept = True
        if accept:
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = 1, 0
        T = max(T * decay, 1e-12)
    return best_h, best_c


def smooth_and_project(v, width, target):
    """Simple moving‑average smoothing of width `width` with projection."""
    if width <= 1:
        return project_to_feasible(v, target)
    kernel = np.ones(width) / width
    sm = np.convolve(v, kernel, mode="same")
    sm = np.clip(sm, 0.0, 1.0)
    return project_to_feasible(sm, target)


def upsample_initial(to_n):
    """Interpolate the global initial_h_values (if present) to length to_n."""
    init = np.array(initial_h_values, dtype=float)
    m = init.size
    xs_old = np.linspace(0.0, 2.0, m, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, to_n, endpoint=False)
    new = np.interp(xs_new, xs_old, init)
    return new


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # candidate discretisations – start medium, then larger if time permits
    candidate_ns = [120, 180, 240, 300]
    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        if n0 not in candidate_ns:
            candidate_ns = [n0] + candidate_ns

    for n in candidate_ns:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining < 10.0:
            break

        # allocate time for this n
        time_for_n = min(remaining * 0.55, max(20.0, n * 0.7))

        # ---- binary simulated annealing (hard bound) ----
        binary_time = time_for_n * 0.45
        bin_h, bin_c = binary_simulated_annealing(n, rng, binary_time)

        # Convert to float for further processing
        bin_h_float = bin_h.astype(float)

        # Generate a few smoothed candidates
        target_sum = n / 2.0
        candidates = [bin_h_float]
        for w in (3, 5, 7):
            candidates.append(smooth_and_project(bin_h_float, w, target_sum))

        # If a global initial exists and matches size, add it
        if "initial_h_values" in globals() and len(initial_h_values) == n:
            cand = np.array(initial_h_values, dtype=float)
            candidates.append(project_to_feasible(cand, target_sum))

        # Choose the best among candidates as seed for continuous refinement
        cand_scores = [compute_c5(c) for c in candidates]
        cur_h = candidates[int(np.argmin(cand_scores))].copy()
        cur_c = min(cand_scores)

        # Update global best if binary alone is better
        if bin_c < best_c:
            best_c, best_h, best_n = bin_c, bin_h_float.copy(), n

        # ---- differential evolution stage (real‑valued) ----
        de_time = time_for_n * 0.35
        h_de, c_de = de_optimize_real(n, rng, de_time, init=cur_h)
        if c_de < best_c:
            best_c, best_h, best_n = c_de, h_de.copy(), n
        cur_h, cur_c = h_de, c_de

        # ---- SLSQP refinement with increasing beta ----
        for beta in (10.0, 50.0, 200.0, 800.0, 2000.0, 8000.0):
            if time.time() - start_time > budget_s * 0.96:
                break
            cur_h = slsqp_optimize(cur_h, beta, rng, max_iter=500)
            cur_c = compute_c5(cur_h)
            if cur_c < best_c:
                best_c, best_h, best_n = cur_c, cur_h.copy(), n

        # ---- final short hill‑climb (real) ----
        if time.time() - start_time < budget_s * 0.98:
            h_final, c_final = hill_climb(cur_h, rng, max_iters=1500,
                                          start_step=0.2,
                                          time_budget=budget_s - (time.time() - start_time))
            if c_final < best_c:
                best_c, best_h, best_n = c_final, h_final.copy(), n

    # ------------------------------------------------------------------
    # Fallback if nothing was discovered
    if best_h is None:
        best_n = 60
        best_h = np.full(best_n, 0.5)

    # Ensure exact feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    best_h = project_to_feasible(best_h, target)

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
