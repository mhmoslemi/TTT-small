def targeted_move(h, k_abs, rng, scale=0.2):
    n = h.size
    L = n - k_abs
    if L <= 0:
        return h
    # Choose i uniformly from overlapping region start indices: i in [0, L-1] (if shift positive) else you might from [k_abs, n-1]? But for overlapping region symmetrical; it doesn't matter; we can choose i random from 0..L-1.
    i = rng.integers(0, L)
    # Determine second index: i2 = i + k_abs (could be bigger than n-1?), ensures within bounds.
    i2 = i + k_abs
    # Choose j outside overlapping region: indices [L, n-1]
    if L < n:
        j = rng.integers(L, n)  # random from L to n-1 inclusive
    else:
        # no complement region (k_abs=0). Use random j not equal to i or i2.
        j = rng.integers(0, n)
        while j == i or j == i2:
            j = rng.integers(0, n)
    # Compute max surplus from both i,i2 and j.
    max_eps = min( (1.0 - h[i]), (1.0 - h[i2]), h[j] )
    if max_eps <= 1e-12:
        return h
    eps = scale * max_eps * rng.random()
    # distribute eps from j to i and i2 equally.
    new_h = h.copy()
    new_h[i] += eps/2
    new_h[i2] += eps/2
    new_h[j] -= eps
    return new_h

def targeted_move(h, max_idx, rng, scale=0.3):
    n = h.size
    shift = max_idx - (n - 1)   # can be negative or positive
    k_abs = abs(shift)
    if k_abs == 0:
        # just use random perturb
        return perturb_h(h, rng, scale=scale)
    L = n - k_abs
    # Determine overlapping region based on sign
    if shift > 0:
        # i in [0, L-1]
        i = rng.integers(0, L)
        i2 = i + k_abs
        # pick j from complement region L..n-1
        if L < n:
            j = rng.integers(L, n)
        else:
            j = rng.integers(0, n)
            while j == i or j == i2:
                j = rng.integers(0, n)
    else:
        # shift negative, i in [k_abs, n-1]
        i = rng.integers(k_abs, n)
        i2 = i - k_abs
        if k_abs > 0:
            j = rng.integers(0, k_abs)
        else:
            j = rng.integers(0, n)
            while j == i or j == i2:
                j = rng.integers(0, n)
    # ensure unique and not same index.
    if i == i2:
        return h.copy()
    max_eps = min(1.0 - h[i], 1.0 - h[i2], h[j])
    if max_eps <= 1e-12:
        return h.copy()
    eps = scale * max_eps * rng.random()
    new_h = h.copy()
    new_h[i] += eps/2
    new_h[i2] += eps/2
    new_h[j] -= eps
    return new_h

import time, math

# ---------- utility functions ----------
def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h so that Σh = target while 0≤h≤1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        # tiny correction spread uniformly
        corr = diff / n
        h += corr
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # fallback deterministic sweep
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):

            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def swap_move(h, rng):
    """Swap two entries."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    while j == i:
        j = rng.integers(0, n)
    new_h = h.copy()
    new_h[i], new_h[j] = new_h[j], new_h[i]
    return new_h


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def compute_c5_info(h):
    """Return (value, k_abs, max_idx, full_corr)."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    max_idx = int(np.argmax(corr))
    max_val = corr[max_idx] * (2.0 / n)
    k_abs = abs(max_idx - (n - 1))
    return max_val, k_abs, max_idx, corr


def targeted_move(h, max_idx, rng, scale=0.4):
    """Move mass into the overlapping region of the worst lag."""
    n = h.size
    shift = max_idx - (n - 1)          # can be negative
    k_abs = abs(shift)
    if k_abs == 0:
        return perturb_h(h, rng, scale=scale)

    L = n - k_abs                      # length of overlap
    if shift > 0:
        # i from [0, L-1], i2 = i + k_abs, j from complement [L, n-1]
        i = rng.integers(0, L)
        i2 = i + k_abs
        if L < n:
            j = rng.integers(L, n)
        else:
            j = rng.integers(0, n)
            while j in (i, i2):
                j = rng.integers(0, n)
    else:
        # shift < 0: i from [k_abs, n-1], i2 = i - k_abs, j from [0, k_abs-1]
        i = rng.integers(k_abs, n)
        i2 = i - k_abs
        if k_abs > 0:
            j = rng.integers(0, k_abs)
        else:
            j = rng.integers(0, n)
            while j in (i, i2):
                j = rng.integers(0, n)

    if i == i2:  # shouldn't happen, but guard
        return h.copy()

    max_eps = min(1.0 - h[i], 1.0 - h[i2], h[j])
    if max_eps <= 1e-12:
        return h.copy()
    eps = scale * max_eps * rng.random()
    new_h = h.copy()
    new_h[i] += eps / 2.0
    new_h[i2] += eps / 2.0
    new_h[j] -= eps
    return new_h


# ---------- SLSQP surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny drift correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x
