def enforce_sum(h, target, rng, max_iter=5000):
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        # amount we can add to h[i]
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-12:
            continue
        h[i] += add
        diff = target - h.sum()
    # final correction: enforce sum exactly by adjusting one coordinate
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        i = rng.integers(0, n)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    # ensure within bounds
    np.clip(h, 0.0, 1.0, out=h)
    return h

import time
import numpy as np

# --------------------------------------------------------------
# Helper utilities
# --------------------------------------------------------------
def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h so that sum(h) == target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.shape[0]
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-12:
            continue
        h[i] += add
        diff = target - h.sum()
    # final tiny correction on a random index
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        i = rng.integers(0, n)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    np.clip(h, 0.0, 1.0, out=h)
    return h


def random_feasible_vector(n, rng):
    """Generate a random feasible vector (0≤h≤1, sum = n/2)."""
    target = n / 2.0
    v = rng.random(n)
    v = v / v.sum() * target
    v = np.clip(v, 0.0, 1.0)
    return enforce_sum(v, target, rng)


def find_opt_eps(C, b, lo, hi, tol=1e-12, max_iter=40):
    """Solve  min_{lo≤ε≤hi} max_k (C_k + b_k ε)  (convex 1‑D)."""
    if hi - lo < 1e-15:
        return lo, np.max(C + b * lo)

    left, right = lo, hi
    best_eps = left
    best_val = np.max(C + b * left)
    val_r = np.max(C + b * right)
    if val_r < best_val:
        best_val, best_eps = val_r, right

    for _ in range(max_iter):
        mid = (left + right) * 0.5
        arr = C + b * mid
        t = np.max(arr)
        # indices achieving the max (allow tiny tolerance)
        mask = arr >= t - 1e-12
        bmax = np.max(b[mask])
        bmin = np.min(b[mask])
        if bmin <= 0.0 <= bmax:
            best_eps, best_val = mid, t
            break
        if bmax < 0.0:
            left = mid
        else:
            right = mid

    # final check at current ends
    for eps in (left, right):
        val = np.max(C + b * eps)
        if val < best_val:
            best_val, best_eps = val, eps
    return best_eps, best_val

def optimise_h(h, rng, time_limit):
    """Local search using optimal epsilon transfers between two coordinates."""
    n = h.shape[0]
    target = n / 2.0
    C = np.correlate(h, 1.0 - h, mode='full')
    cur_max = np.max(C)
    best_h = h.copy()
    best_max = cur_max
    ks = np.arange(- (n - 1), n)                 # shift values
    stagnation = 0
    it = 0

    while time.time() < time_limit:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        hi, hj = h[i], h[j]
        eps_lo = max(-hj, hi - 1.0)
        eps_hi = min(hi, 1.0 - hj)
        if eps_hi - eps_lo < 1e-15:
            continue

        # ----- build linear coefficient b_k -----
        b = np.zeros_like(C, dtype=float)

        idx = i + ks
        mask = (idx >= 0) & (idx < n)
        if mask.any():
            b[mask] -= (1.0 - h[idx[mask]])

        idx = j + ks
        mask = (idx >= 0) & (idx < n)
        if mask.any():
            b[mask] += (1.0 - h[idx[mask]])

        idx = i - ks
        mask = (idx >= 0) & (idx < n)
        if mask.any():
            b[mask] += h[idx[mask]]

        idx = j - ks
        mask = (idx >= 0) & (idx < n)
        if mask.any():
            b[mask] -= h[idx[mask]]

        # ----- optimal epsilon for this pair -----
        eps_opt, new_max = find_opt_eps(C, b, eps_lo, eps_hi)

        if new_max < best_max - 1e-12:
            # accept move
            h[i] -= eps_opt
            h[j] += eps_opt
            # keep within bounds (tiny numerical drift)
            h[i] = np.clip(h[i], 0.0, 1.0)
            h[j] = np.clip(h[j], 0.0, 1.0)

            C = C + b * eps_opt
            best_max = new_max
            best_h = h.copy()
            stagnation = 0
        else:
            stagnation += 1

        it += 1

        # occasional full recompute to wipe floating error
        if it % 2000 == 0:
            C = np.correlate(h, 1.0 - h, mode='full')
            cur = np.max(C)
            if cur < best_max - 1e-12:
                best_max, best_h = cur, h.copy()
                stagnation = 0

        # restart if stuck for long
        if stagnation > 50000:
            # try a fresh random start
            h = random_feasible_vector(n, rng)
            C = np.correlate(h, 1.0 - h, mode='full')
            cur = np.max(C)
            best_max, best_h = cur, h.copy()
            stagnation = 0

    return best_h, best_max
