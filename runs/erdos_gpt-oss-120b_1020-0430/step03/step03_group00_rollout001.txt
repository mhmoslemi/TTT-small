def simulated_annealing(h, rng, time_limit):
    n = h.size
    target_sum = n/2.0
    best_h = h.copy()
    best_c = compute_c5(best_h)
    cur_h = h.copy()
    cur_c = best_c

    start = time.time()
    # parameters
    T = 0.5  # maybe 0.5
    Tmin = 1e-4
    alpha = 0.995
    # iteration count for schedule
    it = 0
    # scale for perturbation reduces with T maybe
    while time.time() - start < time_limit:
        # propose
        scale = 0.2
        new_h = perturb_random(cur_h, rng, scale=scale)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_h = new_h
            cur_c = new_c
            if new_c < best_c:
                best_h = new_h.copy()
                best_c = new_c
        # temperature decay
        it += 1
        if it % 200 == 0:
            T = max(T * alpha, Tmin)
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def enforce_sum(h, target, rng):
    """Clip to [0,1] and adjust to make sum exactly target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:  # need to increase
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i])
            add *= rng.random()
            h[i] += add
            diff -= add
        else:  # need to decrease
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i])
            sub *= rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum exactly n/2."""
    h = rng.random(n)
    h = enforce_sum(h, n / 2.0, rng)
    return h

def perturb_random(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (preserves sum, respects bounds)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def compute_c5(h):
    """Discrete C5 = max_k ∫ h(x)(1‑h(x+k))dx."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

# ---------- targeted hill climbing ----------
def hill_climb_targeted(h_start, rng, max_iters, start_step, target_sum):
    h = h_start.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(best_h)
    step = start_step
    decay_interval = max(1, max_iters // 10)

    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr)
        k = idx - (n - 1)
        # gradient of the worst shift term
        grad = np.ones(n)
        if k >= 0:
            grad[:n - k] -= h[k:]
            grad[k:]   -= h[:n - k]
        else:
            kk = -k
            grad[:n - kk] -= h[kk:]
            grad[kk:]     -= h[:n - kk]

        new_h = perturb_random(h, rng, scale=step)  # simple perturbation
        # try directed move if gradient suggests
        pos = np.where((grad > 1e-9) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-9) & (h < 1.0 - 1e-12))[0]
        if pos.size and neg.size:
            i = rng.choice(pos)
            j = rng.choice(neg)
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0:
                eps = max_eps * step * rng.random()
                new_h[i] -= eps
                new_h[j] += eps

        c_new = compute_c5(new_h)
        c_cur = compute_c5(h)

        if c_new < best_c:
            best_c, best_h = c_new, new_h.copy()
        if c_new < c_cur:
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)

    return best_h, best_c

# ---------- SLSQP surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    rng = np.random.default_rng()
    x = enforce_sum(x, CURRENT_TARGET, rng)
    return x

# ---------- simulated annealing ----------
def simulated_annealing(h, rng, time_limit):
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(best_h)
    cur_h = h.copy()
    cur_c = best_c

    start = time.time()
    T = 0.5
    Tmin = 1e-4
    alpha = 0.995
    it = 0

    while time.time() - start < time_limit:
        new_h = perturb_random(cur_h, rng, scale=0.2)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
        it += 1
        if it % 200 == 0:
            T = max(T * alpha, Tmin)
    return best_h, best_c

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # initial vector

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # --------- simulated annealing on current resolution ----------
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    sa_time = remaining * 0.45
    best_h, best_c = simulated_annealing(best_h, rng, sa_time)

    # --------- optional up‑sampling to finer grid ----------
    # try a single upscale if time permits
    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.6 and best_h.size * 2 <= 720:
        n = best_h.size * 2
        xs_old = (np.arange(best_h.size) + 0.5) * (2.0 / best_h.size)
        xs_new = (np.arange(n) + 0.5) * (2.0 / n)
        h_up = np.interp(xs_new, xs_old, best_h)
        h_up = enforce_sum(np.clip(h_up, 0.0, 1.0), n / 2.0, rng)

        # SA on the finer grid
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        sa_time = remaining * 0.35
        h_up, c_up = simulated_annealing(h_up, rng, sa_time)
        if c_up < best_c:
            best_h, best_c = h_up, c_up

    # --------- targeted hill‑climb refinement ----------
    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.9:
        h_climb, c_climb = hill_climb_targeted(best_h, rng,
                                               max_iters=2500,
                                               start_step=0.5,
                                               target_sum=best_h.size / 2.0)
        if c_climb < best_c:
            best_h, best_c = h_climb, c_climb

    # --------- SLSQP refinement schedule ----------
    beta_schedule = [5.0, 20.0, 80.0, 200.0, 800.0]
    for beta in beta_schedule:
        if time.time() - start_time > budget_s * 0.97:
            break
        try:
            h_ref = slsqp_optimize(best_h, beta, max_iter=400)
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_h, best_c = h_ref.copy(), c_ref
        except Exception:
            continue

    # --------- final tiny random tweaks ----------
    while time.time() - start_time < budget_s * 0.99:
        new_h = perturb_random(best_h, rng, scale=0.02)
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_h, best_c = new_h, new_c

    # clean‑up and final evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
