def project_onto_box_and_sum(v,total):
    # clip to [0,1] first
    x = np.clip(v, 0.0, 1.0)
    # adjust sum by water-filling
    # If sum too low, add delta equally to non-full entries, else subtract from non-zero entries.
    # Use while loop and compute count of free entries each time.
    # Implement efficiently with vectorized mask.
    for _ in range(100):
        diff = total - x.sum()
        if abs(diff) < 1e-12:
            break
        # mask entries not at bounds in direction of diff
        if diff > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        count = mask.sum()
        if count == 0:
            break
        delta = diff / count
        # apply but keep within bounds
        x[mask] += delta
        # clip again
        np.clip(x, 0.0, 1.0, out=x)
    return x
```

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

import time, numpy as np

# ----- projection onto [0,1] with fixed sum ---------------------------------
def proj_box_sum(v, total):
    """Project v onto the box [0,1] while enforcing sum(v)=total."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > total:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


# ----- C5 evaluation ---------------------------------------------------------
def compute_c5(h):
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


# ----- gradient descent on log-sum-exp smooth max ----------------------------
def gradient_descent_proj(h, beta_sched, iters_sched, step0, start, budget_s):
    """Projected gradient descent on the smooth max (log‑sum‑exp)."""
    n = h.size
    total = n / 2.0
    dx = 2.0 / n
    for beta, iters in zip(beta_sched, iters_sched):
        step = step0
        for it in range(iters):
            if time.time() - start > budget_s * 0.95:
                return h
            # overlaps for all shifts
            corr = np.correlate(h, 1 - h, mode="full")
            O = corr * dx
            max_O = O.max()
            exp_vals = np.exp(beta * (O - max_O))
            w_sum = exp_vals.sum()
            # convolution trick for the weighted h‑shifts
            conv1 = np.convolve(exp_vals, h, mode="full")[n - 1 : n - 1 + n]
            conv2 = np.convolve(exp_vals[::-1], h, mode="full")[n - 1 : n - 1 + n]
            grad = np.full_like(h, w_sum, dtype=float)
            grad -= conv1
            grad -= conv2
            grad_f = grad / w_sum
            h_new = h - step * grad_f
            h = proj_box_sum(h_new, total)
            if (it + 1) % 100 == 0:
                step *= 0.7
    return h


# ----- worst‑shift gradient (for hill‑climb) ---------------------------------
def worst_gradient(h):
    """Gradient of the overlap for the worst shift."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr)
    k = abs(idx - (n - 1))
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


# ----- small guided perturbation --------------------------------------------
def guided_perturb(h, grad, rng, scale=0.5):
    """Transfer a little mass from a high‑gradient index to a low‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


# ----- hill‑climb using the worst‑shift gradient -----------------------------
def guided_hill_climb(h_start, rng, max_iters=1500, step_scale=0.3):
    """Hill‑climb guided by the worst‑shift gradient."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


# ----- main entry point ------------------------------------------------------

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = proj_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)
