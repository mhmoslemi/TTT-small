def compute_c5(h):
    n = h.size
    rev = (1 - h)[::-1]
    # compute convolution using FFT
    size = 1 << ((2*n - 1).bit_length())
    fh = np.fft.rfft(h, n=size)
    f_rev = np.fft.rfft(rev, n=size)
    conv = np.fft.irfft(fh * f_rev, n=size)[:2*n-1]
    # conv corresponds to "full" convolution (like np.convolve)
    max_val = conv.max()
    return max_val * (2.0 / n)

def compute_c5_fast(h):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    size = 1 << ((2*n - 1).bit_length())

    conv = np.fft.irfft(H * R, n=size)[:2*n-1]

def propose_move(h, corr, rng, biased, scale):
    n = h.size
    if biased:
        # compute shift matrix for max correlation
        k = np.argmax(corr)
        shift = k - (n - 1)
        # overlapping indices:
        i0 = max(0, -shift)
        i1 = min(n, n - shift)
        idx = np.arange(i0, i1)
        if idx.size == 0:
            # fallback uniform
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)
        else:
            prod = h[idx] * (1 - h[idx + shift])
            thr_hi = np.quantile(prod, 0.7)
            thr_lo = np.quantile(prod, 0.3)
            high = idx[prod > thr_hi]
            low = idx[prod < thr_lo]
            if high.size == 0 or low.size == 0:
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                while j == i:
                    j = rng.integers(0, n)
            else:
                i = rng.choice(high)
                # choose j either from low or from any not equal i
                if low.size > 0:
                    j = rng.choice(low)
                else:
                    j = rng.integers(0, n)
                    while j == i:
                        j = rng.integers(0, n)

        j = rng.integers(0, n)

import time, math, numpy as np
from scipy.optimize import minimize

# ---------- helper functions ----------
def compute_c5_fast(h):
    """Discrete C5 using FFT (dx = 2 / n)."""
    n = h.size
    rev = (1 - h)[::-1]
    size = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=size)
    R = np.fft.rfft(rev, n=size)
    conv = np.fft.irfft(H * R, n=size)[: 2 * n - 1]
    return conv.max() * (2.0 / n)


def compute_diff(h, a):
    """Contribution of a single index a to the correlation gradient."""
    n = h.size
    size = 2 * n - 1
    diff = np.zeros(size, dtype=np.float64)
    idx1 = np.arange(n) + (n - 1) - a
    np.add.at(diff, idx1, 1 - h)
    idx2 = a + (n - 1) - np.arange(n)
    np.add.at(diff, idx2, -h)
    return diff


def project_to_sum_and_box(x, target, tol=1e-12):
    """Box [0,1] projection onto hyperplane sum(x)=target."""
    lo, hi = -1.0, np.max(x) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    resid = target - proj.sum()
    if abs(resid) > tol:
        for i in range(len(proj)):
            if resid > 0 and proj[i] < 1.0:
                inc = min(1.0 - proj[i], resid)
                proj[i] += inc
                resid -= inc
            elif resid < 0 and proj[i] > 0.0:
                dec = min(proj[i], -resid)
                proj[i] -= dec
                resid += dec
            if abs(resid) <= tol:
                break
    return proj


def enforce_sum(h, target, rng):
    """Iterative correction to meet the sum while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum=n/2."""
    v = rng.random(n)
    v = project_to_sum_and_box(v, n / 2.0)
    return v

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {"type": "eq", "fun": sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

        h0 = random_feasible_vector(n, rng)

    # SA phase
    sa_time = budget_s * 0.85
    end_sa = start + sa_time
    best_h, best_c = simulated_annealing_fast(h0, rng, end_sa, start)

    # final refinements with SLSQP if time permits
    remaining = budget_s - (time.time() - start)
    if remaining > 2.0:
        for beta in (200, 800, 2000, 8000, 20000):
            if time.time() - start > budget_s - 0.5:
                break
            try:
                refined = slsqp_optimize(best_h, beta, max_iter=300)
                rc = compute_c5_fast(refined)
                if rc < best_c:
                    best_h, best_c = refined, rc
            except Exception:
                continue

    # final clip & exact sum enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum_and_box(best_h, n / 2.0)
    final_c = compute_c5_fast(best_h)

    return best_h.tolist(), float(final_c), int(n)
