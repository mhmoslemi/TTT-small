Define correlate_full:

def correlate_full(h):
    n = h.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    # compute FFT of h and reversed (1-h)
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[:2*n-1]
    return conv

def slsqp_optimize(h0, beta, max_iter=350):
    n = h0.size
    target = n/2.0
    bounds = [(0.0,1.0)]*n
    cons = {'type':'eq','fun': lambda x: np.sum(x) - target}
    # Softmax objective closure capturing beta
    def obj(x):
        conv = correlate_full(x)
        # compute log-sum-exp of beta*conv
        bc = beta * conv
        m = bc.max()
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    res = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol':1e-12, 'disp':False})
    x = np.clip(res.x,0,1)
    # enforce sum exact
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        # fix diff by adjusting random elements
        for _ in range(1000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0,n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def enforce_sum(h, target, rng):
    """Adjust h so that sum(h) == target while keeping 0 <= h <= 1."""
    diff = target - h.sum()
    n = h.size
    if abs(diff) < 1e-12:
        return h
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            if h[i] >= 1.0:
                continue
            add = min(diff, 1.0 - h[i])
        else:
            if h[i] <= 0.0:
                continue
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # final tiny correction
    if abs(diff) > 1e-12:
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h


def random_feasible_vector(n, rng):
    """Uniform random vector in [0,1]^n with sum = n/2."""
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)


# ----------------------------------------------------------------------
# Correlation helpers (FFT based)
# ----------------------------------------------------------------------
def correlate_full(h):
    """Full linear correlation of h with (1‑h)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Discrete C₅ for step function h on [0,2] (dx = 2/n)."""
    n = h.size
    conv = correlate_full(h)
    return conv.max() * (2.0 / n)


# ----------------------------------------------------------------------
# Simple perturb / hill‑climb
# ----------------------------------------------------------------------
def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def hill_climb(h_start, rng, max_iters=1500, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        nh = perturb_h(h, rng, scale=step)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


def simulated_annealing(start_h, rng, deadline, T0=0.02, Tf=1e-5, max_iters=1500):
    cur_h = start_h.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        nh = perturb_h(cur_h, rng, scale=0.35)
        nc = compute_c5(nh)
        delta = nc - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_h, cur_c = nh, nc
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
    return best_h, best_c


# ----------------------------------------------------------------------
# Targeted hill‑climb (focus on the worst shift)
# ----------------------------------------------------------------------
def targeted_hill_climb(h_start, rng, max_iters=2000, step=0.05):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    for it in range(max_iters):
        conv = correlate_full(h)
        idx = int(np.argmax(conv))
        k = idx - (n - 1)                     # shift giving the maximum overlap
        i_min = max(0, -k)
        i_max = min(n - 1, n - 1 - k)
        if i_min > i_max:
            continue
        i = rng.integers(i_min, i_max + 1)
        j = i + k
        # donor that can lose some mass
        donors = np.where((h > step) & (np.arange(n) != i) & (np.arange(n) != j))[0]
        if donors.size == 0:
            donors = np.where(h > step)[0]
            if donors.size == 0:
                break
        d = rng.choice(donors)
        max_delta = min(step, h[d], 1.0 - h[i], 1.0 - h[j])
        if max_delta <= 1e-12:
            continue
        nh = h.copy()
        nh[i] += max_delta
        nh[j] += max_delta
        nh[d] -= max_delta
        np.clip(nh, 0.0, 1.0, out=nh)
        nc = compute_c5(nh)
        if nc < best_c - 1e-12:
            best_c, best_h, h = nc, nh.copy(), nh
        # occasional step decay
        if (it + 1) % (max_iters // 5) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


# ----------------------------------------------------------------------
# Smooth surrogate + SLSQP
# ----------------------------------------------------------------------
def slsqp_optimize(h0, beta, max_iter=350):
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}

    def obj(x):
        conv = correlate_full(x)
        bc = beta * conv
        m = bc.max()
        # stable log‑sum‑exp
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    res = minimize(
        obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny sum correction
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x

# Main entry point
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    main_deadline = start_time + budget_s * 0.90
    up_deadline = start_time + budget_s

    # ------------------------------------------------------------------
    # 1. Initial vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 240
        h0 = random_feasible_vector(n, rng)

    best_h, best_c = h0, compute_c5(h0)

    # ------------------------------------------------------------------
    # 2. Quick refinement with targeted hill‑climb
    # ------------------------------------------------------------------
    h_tmp, c_tmp = targeted_hill_climb(best_h, rng, max_iters=1800, step=0.07)
    if c_tmp < best_c:
        best_h, best_c = h_tmp, c_tmp

    # ------------------------------------------------------------------
    # 3. Main stochastic search
    # ------------------------------------------------------------------
    beta_list = [20, 80, 200, 800, 2000]
    while time.time() < main_deadline:
        # pick a strategy
        r = rng.random()
        if r < 0.35:
            cand_h, cand_c = simulated_annealing(best_h, rng, main_deadline,
                                                T0=0.03, Tf=1e-5, max_iters=1300)
        elif r < 0.65:
            cand_h, cand_c = targeted_hill_climb(best_h, rng,
                                                max_iters=2500, step=0.04)
        else:
            start = random_feasible_vector(n, rng)
            cand_h, cand_c = hill_climb(start, rng, max_iters=1500, start_step=0.5)

        if cand_c < best_c:
            best_h, best_c = cand_h, cand_c

        # SLSQP polishing
        for beta in beta_list:
            if time.time() >= main_deadline:
                break
            try:
                h_opt = slsqp_optimize(best_h, beta, max_iter=400)
                c_opt = compute_c5(h_opt)
                if c_opt < best_c:
                    best_h, best_c = h_opt, c_opt
            except Exception:
                continue

    # ------------------------------------------------------------------
    # 4. Optional up‑sampling (finer grid) if time permits
    # ------------------------------------------------------------------
    if time.time() < up_deadline and n * 2 <= 720:
        up_factor = 2
        h_up = np.repeat(best_h, up_factor)
        n_up = h_up.size
        h_up = enforce_sum(h_up, n_up / 2.0, rng)

        h_up, c_up = targeted_hill_climb(h_up, rng,
                                         max_iters=3500, step=0.03)
        if c_up < best_c:
            best_h, best_c, n = h_up, c_up, n_up

        for beta in beta_list:
            if time.time() >= up_deadline:
                break
            try:
                h_opt = slsqp_optimize(best_h, beta, max_iter=400)
                c_opt = compute_c5(h_opt)
                if c_opt < best_c:
                    best_h, best_c = h_opt, c_opt
            except Exception:
                continue

    # ------------------------------------------------------------------
    # 5. Final clean‑up
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
