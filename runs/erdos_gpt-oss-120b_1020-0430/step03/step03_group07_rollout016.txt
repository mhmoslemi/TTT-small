def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h in-place so that sum(h) == target while 0<=h<=1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):

            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    enforce_sum(h, n / 2.0, rng)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small ε from i to j (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def correlate_full(h):
    """Full linear correlation of h with (1-h) using FFT."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Discrete C5 for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)


# ------------------------------------------------------------
# Smooth surrogate for SLSQP
# ------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Log‑sum‑exp smoothing of max_k Σ_i x_i (1‑x_{i+k})."""
    conv = correlate_full(x)
    bc = CURRENT_BETA * conv
    m = bc.max()
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=400):
    """SLSQP on the smoothed surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        enforce_sum(x, CURRENT_TARGET, rng)
    return x


# ------------------------------------------------------------
# Local optimisation primitives
# ------------------------------------------------------------
def hill_climb(h_start, rng, deadline, max_iters=8000, start_step=0.3):
    """Greedy hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay = max(1, max_iters // 15)
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay == 0:
            step = max(step * 0.85, 1e-5)
    return best_h, best_c


def simulated_annealing(start_h, rng, deadline,
                        T0=0.04, Tf=1e-5, max_iters=2500):
    """Simple exponential‑cooling SA."""
    cur_h = start_h.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        new_h = perturb_h(cur_h, rng, scale=0.4)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
    return best_h, best_c


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Reserve a margin for final cleanup
    hard_deadline = start_time + budget_s
    main_deadline = start_time + 0.85 * budget_s
    upsample_deadline = start_time + 0.95 * budget_s

    # --------------------------------------------------------
    # Base dimension & initial vector
    # --------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        enforce_sum(h0, n / 2.0, rng)
    else:
        n = 180
        h0 = np.full(n, 0.5)
        enforce_sum(h0, n / 2.0, rng)

    # --------------------------------------------------------
    # Build diverse initial pool
    # --------------------------------------------------------
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    candidates = [h0]

    # sinusoidal families
    for f in range(1, 7):
        s = 0.5 + 0.5 * np.sin(np.pi * f * xs)
        candidates.append(enforce_sum(np.clip(s, 0.0, 1.0), n / 2.0, rng))
        c = 0.5 + 0.5 * np.cos(np.pi * f * xs)
        candidates.append(enforce_sum(np.clip(c, 0.0, 1.0), n / 2.0, rng))

    # logistic shapes
    for a in [0.8, 1.5, 3.0, 5.0]:
        for b in [0.4, 0.8, 1.2, 1.6]:
            l = 1.0 / (1.0 + np.exp(-a * (xs - b)))
            candidates.append(enforce_sum(np.clip(l, 0.0, 1.0), n / 2.0, rng))

    # linear ramps
    candidates.append(enforce_sum(np.clip(xs / 2.0, 0.0, 1.0), n / 2.0, rng))
    candidates.append(enforce_sum(np.clip(1.0 - xs / 2.0, 0.0, 1.0), n / 2.0, rng))

    # binary step
    step = np.zeros(n)
    step[: n // 2] = 1.0
    candidates.append(enforce_sum(step, n / 2.0, rng))

    # alternating
    alt = np.tile([0.0, 1.0], n // 2)
    candidates.append(enforce_sum(alt, n / 2.0, rng))

    # random draws
    for _ in range(5):
        candidates.append(random_feasible_vector(n, rng))

    # --------------------------------------------------------
    # Evaluate pool, keep top few
    # --------------------------------------------------------
    evaluated = [(compute_c5(h), h) for h in candidates]
    evaluated.sort(key=lambda x: x[0])
    top_k = 3
    best_h, best_c = evaluated[0][1].copy(), evaluated[0][0]

    # --------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------
    beta_schedule = [2, 5, 20, 80, 200, 800, 2000, 5000]

    while time.time() < main_deadline:
        # pick a start point
        if rng.random() < 0.7:
            start = perturb_h(best_h, rng, scale=0.35)
        else:
            start = random_feasible_vector(n, rng)

        # simulated annealing
        cand_h, cand_c = simulated_annealing(start, rng, main_deadline,
                                             T0=0.04, Tf=1e-5, max_iters=2500)
        if cand_c < best_c:
            best_h, best_c = cand_h.copy(), cand_c

        # hill‑climb refinement
        hc_deadline = min(main_deadline, time.time() + 5.0)
        h_opt, c_opt = hill_climb(best_h, rng, hc_deadline,
                                  max_iters=6000, start_step=0.3)
        if c_opt < best_c:
            best_h, best_c = h_opt.copy(), c_opt

        # SLSQP ladder
        for beta in beta_schedule:
            if time.time() >= main_deadline:
                break
            try:
                h_opt = slsqp_optimize(best_h, beta, max_iter=350)
                c_opt = compute_c5(h_opt)
                if c_opt < best_c:
                    best_h, best_c = h_opt.copy(), c_opt
            except Exception:
                continue

    # --------------------------------------------------------
    # Optional up‑sampling to finer grid
    # --------------------------------------------------------
    if time.time() < upsample_deadline and n * 2 <= 720:
        up_factor = 2
        h_up = np.repeat(best_h, up_factor)
        n_up = h_up.size
        enforce_sum(h_up, n_up / 2.0, rng)
        # quick hill‑climb on finer grid
        hc_deadline = min(upsample_deadline, time.time() + 6.0)
        h_up_opt, c_up_opt = hill_climb(h_up, rng, hc_deadline,
                                        max_iters=5000, start_step=0.25)
        if c_up_opt < best_c:
            best_h, best_c, n = h_up_opt.copy(), c_up_opt, n_up

        for beta in beta_schedule:
            if time.time() >= hard_deadline:
                break
            try:
                h_opt = slsqp_optimize(best_h, beta, max_iter=300)
                c_opt = compute_c5(h_opt)
                if c_opt < best_c:
                    best_h, best_c, n = h_opt.copy(), c_opt, h_opt.size
            except Exception:
                continue

    # --------------------------------------------------------
    # Final clean‑up
    # --------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
