def optimize_h(start_h, n, rng, budget):
    # returns improved h
    target = n/2.0
    mu = 1000.0
    h = start_h.copy()
    betas = [0.5, 2, 10, 50, 200, 800, 2000]
    for beta in betas:
        # L-BFGS-B
        bounds = [(0.0,1.0)]*n
        # Define closure functions that compute obj/grad given x:
        # to avoid recompute, we compute inside.
        def f_and_grad(x):
            # x vector
            # compute correlation
            corr = np.correlate(x, 1 - x, mode='full')
            # compute weights
            bc = beta * corr
            max_bc = np.max(bc)
            exp_bc = np.exp(bc - max_bc)
            sum_exp = np.exp(bc - max_bc).sum()
            w = exp_bc / sum_exp
            obj = np.dot(w, corr)
            # gradient of softmax part:
            nloc = n
            # compute gradient g_s
            g = np.zeros_like(x)
            shift_vals = np.arange(-(nloc-1), nloc)
            for wk, k in zip(w, shift_vals):
                if wk == 0:
                    continue
                i_start = max(0, -k)
                i_end = min(nloc, nloc - k)
                # slice
                # (1 - h[i+k]) for i in this range
                g[i_start:i_end] += wk * (1 - x[i_start + k:i_end + k])
                # -h[i-k]
                g[i_start:i_end] -= wk * x[i_start - k:i_end - k]
            # penalty:
            diff = np.sum(x) - target
            obj += mu * diff * diff
            g += 2 * mu * diff
            return obj, g
        # use SciPy minimize
        res = minimize(lambda x: f_and_grad(x)[0],
                       h,
                       method='L-BFGS-B',
                       jac=lambda x: f_and_grad(x)[1],
                       bounds=bounds,
                       options={'maxiter': 200, 'ftol':1e-12})
        h = np.clip(res.x, 0.0, 1.0)
        # after each beta, maybe do a small gradient descent step to correct sum exactly
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # adjust values linearly among interior points
            for _ in range(500):
                i = rng.integers(0, n)
                # ensure we don't exceed bounds
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h
```

import time, math
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h):
    """Exact discrete C5 for step function h on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def random_feasible_vector(n, rng):
    """Random vector in [0,1] with sum exactly n/2."""
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h

def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb(h_start, rng, max_iters=1500, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c

def refine_worst_shift(h, rng, steps=12, step0=0.02):
    """Gradient descent against the currently worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(steps):
        corr = np.correlate(h, 1 - h, mode="full")
        k = np.argmax(corr) - (n - 1)          # shift that attains max
        i0 = max(0, -k)
        i1 = min(n, n - k)
        g = np.zeros_like(h)
        g[i0:i1] += (1 - h[i0 + k:i1 + k]) - h[i0 - k:i1 - k]
        alpha = step0 * (0.9 ** it)
        h = h - alpha * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

# ---------- optimization core ----------
def optimize_h(h0, n, rng):
    """Continuation L‑BFGS‑B using soft‑max surrogate of the max shift."""
    target = n / 2.0
    mu = 1000.0                     # penalty weight for the sum constraint
    h = h0.copy()
    betas = [0.5, 2.0, 10.0, 50.0, 200.0, 800.0, 3000.0]

    shifts = np.arange(-(n - 1), n, dtype=int)   # length 2n‑1

    for beta in betas:
        bounds = [(0.0, 1.0)] * n

        def obj_and_grad(x):
            # correlation
            corr = np.correlate(x, 1 - x, mode="full")
            bc = beta * corr
            m = np.max(bc)
            exp_bc = np.exp(bc - m)
            sum_exp = exp_bc.sum()
            w = exp_bc / sum_exp                     # softmax weights
            obj = np.dot(w, corr)                    # smooth max

            # gradient of smooth max
            g = np.zeros_like(x)
            # loop over all possible shifts
            for wk, k in zip(w, shifts):
                if wk == 0.0:
                    continue
                i0 = max(0, -k)
                i1 = min(n, n - k)
                g[i0:i1] += wk * (1 - x[i0 + k:i1 + k])
                g[i0:i1] -= wk * x[i0 - k:i1 - k]

            # penalty for sum constraint
            diff = np.sum(x) - target
            obj += mu * diff * diff
            g += 2.0 * mu * diff
            return obj, g

        # L‑BFGS‑B with gradient supplied (jac=True expects (f,grad) return)
        res = minimize(obj_and_grad, h, method="L-BFGS-B", jac=True,
                       bounds=bounds, options={"maxiter": 200, "ftol": 1e-12})
        h = np.clip(res.x, 0.0, 1.0)

        # tiny exact sum correction
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h
