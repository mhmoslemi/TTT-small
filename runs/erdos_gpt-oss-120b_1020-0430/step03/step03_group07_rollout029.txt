import time, numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Basic utilities
# --------------------------------------------------------------
def enforce_sum(v, target, rng, max_iter=5000):
    diff = target - v.sum()
    n = v.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - v[i]) if diff > 0 else max(diff, -v[i])
        v[i] += add
        diff = target - v.sum()
    return v

def random_feasible_vector(n, rng):
    v = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(v[i], 1.0 - v[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        v[i] -= eps
        v[j] += eps
    v = np.clip(v, 0.0, 1.0)
    return enforce_sum(v, n / 2.0, rng)

def perturb(v, rng, scale=0.5):
    n = v.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return v.copy()
    max_eps = min(v[i], 1.0 - v[j])
    if max_eps <= 0:
        return v.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    out = v.copy()
    out[i] -= eps
    out[j] += eps
    return out

# --------------------------------------------------------------
# Correlation helpers (full linear correlation)
# --------------------------------------------------------------
def correlate_full(h):
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    fr = np.fft.rfft((1 - h)[::-1], L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv

def compute_c5_full(h):
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)

# --------------------------------------------------------------
# Symmetric representation (h(x)=h(2−x))
# --------------------------------------------------------------
def sym_h(s):
    return np.concatenate([s, s[::-1]])

def compute_c5(s):
    return compute_c5_full(sym_h(s))

# --------------------------------------------------------------
# Smooth surrogate (log-sum-exp) for SLSQP
# --------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    h = sym_h(x)
    conv = correlate_full(h)
    bc = CURRENT_BETA * conv
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(x0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = x0.size * 2
    m = x0.size
    CURRENT_TARGET = m / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * m
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, m)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def hill_climb(s0, rng, max_iters=2500, start_step=0.5):
    best = s0.copy()
    best_c = compute_c5(best)
    step = start_step
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        cand = perturb(best, rng, scale=step)
        c = compute_c5(cand)
        if c < best_c:
            best_c, best = c, cand
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-5)
    return best, best_c

def simulated_annealing(start, rng, deadline, T0=0.02, Tf=1e-5, max_iters=2000):
    cur = start.copy()
    cur_c = compute_c5(cur)
    best, best_c = cur.copy(), cur_c
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        cand = perturb(cur, rng, scale=0.35)
        c = compute_c5(cand)
        delta = c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur, cur_c = cand, c
            if cur_c < best_c:
                best, best_c = cur.copy(), cur_c
    return best, best_c

    return final_h.tolist(), float(final_c), int(final_h.size)
