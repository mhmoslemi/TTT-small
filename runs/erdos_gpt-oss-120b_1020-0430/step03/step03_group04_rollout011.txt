def simulated_annealing(start_h, rng, budget, start_time):
    # continue until time limit
    cur_h = start_h.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    it = 0
    while time.time() < budget:
        # schedule temperature
        t = max(0.001, cur_c * (0.99 ** (it // 1000)))
        # step scale decays over time
        scale = max(0.01, 0.5 * (0.999 ** it))
        # do a move
        new_h = perturb_transfer(cur_h, rng, scale=scale)
        new_c = compute_c5(new_h)
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / t):
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
        it += 1
    return best_h, best_c

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def enforce_sum_and_bounds(v, n, rng):
    h = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)

        if add == 0.0:
            continue
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_descent_proj(h, rng, max_iter=20, step_init=0.02):
    n = h.size
    target = n / 2.0
    step = step_init
    cur_c = compute_c5(h)
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        improved = False
        alpha = step
        while alpha > 1e-6:
            new_h = h - alpha * g
            new_h = np.clip(new_h, 0.0, 1.0)
            diff = target - new_h.sum()

    best_c = np.inf
    best_n = None

            x_tgt = np.linspace(0.0, 2.0, n, endpoint=False)

        starts.append(random_feasible_vector(n, rng))

        h_bin[idx] = 1.0

                    elite.append((h_sa.copy(), c_sa, n))

        if a_n != b_n:
