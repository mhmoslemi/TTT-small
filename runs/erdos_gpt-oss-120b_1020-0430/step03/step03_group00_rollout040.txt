Projection onto capped simplex:

Goal: find x' = proj_{[0,1], sum = S} (x). Equivalent to solving min ||x' - x||^2 s.t. 0<=x' <=1, sum x' = S.

We can use algorithm: sort x descending, then find threshold lambda such that after clipping to [0,1] (i.e., x_i - lambda clamped between 0 and 1) sum equals S.

import time
import numpy as np

# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C₅ for a step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / h.size))


def project_capped_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Projection of v onto {0 ≤ x ≤ 1, sum x = target_sum} via bisection."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(50):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random feasible vector (0≤h≤1, Σh=n/2)."""
    v = rng.random(n)
    return project_capped_simplex(v, n / 2.0)


# ---------- gradient descent on soft‑max approximation ----------
def gradient_descent_smooth(
    h_start: np.ndarray,
    betas,
    steps_per_beta: int,
    time_limit: float,
) -> tuple[np.ndarray, float]:
    h = h_start.copy()
    n = h.size
    target = n / 2.0
    best_h, best_c = h.copy(), compute_c5(h)

    for beta in betas:
        alpha = 0.5 / (beta + 1.0)            # initial step size
        for it in range(steps_per_beta):
            if time.time() > time_limit:
                return best_h, best_c

            corr = np.correlate(h, 1.0 - h, mode="full")
            bc = beta * corr
            m = np.max(bc)
            w = np.exp(bc - m)
            w /= w.sum()

            # gradient of the soft‑max objective
            term1 = np.convolve(w, 1.0 - h, mode="valid")
            term2 = np.convolve(w, h, mode="valid")[::-1]
            grad = term1 - term2

            h = h - alpha * grad
            h = project_capped_simplex(np.clip(h, 0.0, 1.0), target)

            if it % 100 == 0:
                c = compute_c5(h)
                if c < best_c:
                    best_c, best_h = c, h.copy()
        alpha *= 0.5          # tighten step size for next beta
    return best_h, best_c


# ---------- targeted hill climbing (worst‑shift gradient) ----------
def worst_shift_gradient(h: np.ndarray, k: int) -> np.ndarray:
    n = h.size
    grad = np.zeros_like(h)
    if k >= 0:
        grad[: n - k] = (1.0 - h[k:])
        grad[k:] -= h[: n - k]
    else:
        kk = -k
        grad[kk:] = (1.0 - h[: n - kk])
        grad[: n - kk] -= h[kk:]
    return grad


def targeted_hill_climb(
    h_start: np.ndarray,
    rng: np.random.Generator,
    max_iters: int,
    scale_start: float,
    time_limit: float,
) -> tuple[np.ndarray, float]:
    h = h_start.copy()
    n = h.size
    target = n / 2.0
    best_h, best_c = h.copy(), compute_c5(h)
    scale = scale_start
    decay_interval = max(1, max_iters // 10)

    for it in range(max_iters):
        if time.time() > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        k = idx - (n - 1)                           # shift giving the worst overlap
        cur_c = float(corr[idx] * (2.0 / n))

        grad = worst_shift_gradient(h, k)

        pos = np.where((grad > 1e-9) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-9) & (h < 1.0 - 1e-12))[0]

        if pos.size and neg.size:
            i = rng.choice(pos)
            j = rng.choice(neg)
            max_eps = min(h[i], 1.0 - h[j])
            eps = max_eps * scale * rng.random()
            new_h = h.copy()
            new_h[i] -= eps
            new_h[j] += eps
        else:
            # fallback random transfer
            i, j = rng.integers(0, n, size=2)
            while i == j:
                j = rng.integers(0, n)
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0.0:
                new_h = h.copy()
            else:
                eps = max_eps * scale * rng.random()
                new_h = h.copy()
                new_h[i] -= eps
                new_h[j] += eps

        new_h = project_capped_simplex(np.clip(new_h, 0.0, 1.0), target)
        new_c = compute_c5(new_h)

        if new_c < best_c:
            best_c, best_h = new_c, new_h.copy()
        if new_c < cur_c:
            h = new_h

        if (it + 1) % decay_interval == 0:

    return best_h, best_c
