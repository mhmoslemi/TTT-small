def project_to_feasible(v, target, low=0.0, high=1.0):
    # Clip
    v = np.clip(v, low, high)
    total = v.sum()
    diff = total - target
    if abs(diff) < 1e-12:
        return v
    # Use water-filling algorithm
    # Start with free indices
    free = np.ones_like(v, dtype=bool)
    while abs(diff) > 1e-12:
        free_idx = np.where(free)[0]
        if free_idx.size == 0:
            break
        # Amount to distribute per free variable
        delta = diff / free_idx.size
        # For each free index, attempt to adjust
        for i in free_idx:
            new_val = v[i] - delta
            if new_val < low:
                diff -= (v[i] - low)
                v[i] = low
                free[i] = False
            elif new_val > high:
                diff -= (high - v[i])
                v[i] = high
                free[i] = False
            else:
                v[i] = new_val
        diff = v.sum() - target
    return v
```

import time
import numpy as np

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def project_to_feasible(v, target, lo=0.0, hi=1.0):
    v = np.clip(v, lo, hi)
    diff = v.sum() - target
    if abs(diff) < 1e-12:
        return v
    free = np.ones_like(v, dtype=bool)
    # simple water-filling
    while abs(diff) > 1e-12:
        idx = np.where(free)[0]
        if idx.size == 0:
            break
        delta = diff / idx.size
        for i in idx:
            new = v[i] - delta
            if new < lo:
                diff -= (v[i] - lo)
                v[i] = lo
                free[i] = False
            elif new > hi:
                diff -= (hi - v[i])
                v[i] = hi
                free[i] = False
            else:
                v[i] = new
        diff = v.sum() - target
    return v

def worst_shift_and_grad(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr)                # 0 .. 2n-2
    k = idx - (n - 1)                    # shift (may be negative)
    g = np.zeros_like(h)
    if k >= 0:
        g[k:] += 1.0
        g[k:] -= h[: n - k]              # -h_{i-k}
        g[: n - k] -= h[k:]              # -h_{i+k}
    else:
        kk = -k
        g[: n - kk] += 1.0
        g[: n - kk] -= h[kk:]            # -h_{i+kk}
        g[kk:] -= h[: n - kk]            # -h_{i-kk}
    return k, g

def gradient_descent(h0, rng, iters=250, step0=0.5):
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h, best_c = h.copy(), compute_c5(h)
    for it in range(iters):
        _, g = worst_shift_and_grad(h)
        step = step0 * (0.95 ** it)
        h_new = h - step * g
        h_new = project_to_feasible(h_new, target)
        c = compute_c5(h_new)
        if c < best_c:
            best_c, best_h = c, h_new.copy()
        h = h_new
    return best_h, best_c

def local_search(h, rng, iters=200):
    n = h.size
    best_h, best_c = h.copy(), compute_c5(h)
    for _ in range(iters):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.1
        if abs(eps) < 1e-12:
            continue
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        c = compute_c5(cand)
        if c < best_c:
            best_c, best_h = c, cand
    return best_h, best_c

def random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

def random_continuous(n, rng):
    h = rng.random(n)
    target = n / 2.0
    scale = target / h.sum()
    h *= scale
    return project_to_feasible(h, target)

def upsample(h, n_new, rng):
    n_old = h.size
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_up = np.interp(x_new, x_old, h)
    return project_to_feasible(h_up, n_new / 2.0)
