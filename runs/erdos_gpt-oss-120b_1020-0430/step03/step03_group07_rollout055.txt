def correlate_full(a, b):
    """
    Full linear correlation of a with b: (a * b) [like np.correlate(a, b, mode='full')]
    returns array length len(a)+len(b)-1.
    """
    n = a.shape[0]
    m = b.shape[0]
    L = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b[::-1], L)
    conv = np.fft.irfft(fa * fb, L)[: n + m - 1]
    return conv

import time
import numpy as np

def correlation(a, b):
    """Full linear correlation a with b (like np.correlate(a,b,'full'))."""
    n = a.shape[0]
    m = b.shape[0]
    L = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b[::-1], L)
    conv = np.fft.irfft(fa * fb, L)[: n + m - 1]
    return conv

def compute_c5(h):
    n = h.size
    conv = correlation(h, 1 - h)
    return float(conv.max() * (2.0 / n))

def enforce_sum(v, target, rng, max_iter=10000):
    """Adjust v so that its sum equals target while staying in [0,1]."""
    diff = target - v.sum()
    n = v.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            i = rng.integers(0, n)
            cap = 1.0 - v[i]
            if cap <= 0.0:
                continue
            add = min(diff, cap)
            v[i] += add
            diff = target - v.sum()
        else:
            i = rng.integers(0, n)
            cap = v[i]
            if cap <= 0.0:
                continue
            sub = min(-diff, cap)
            v[i] -= sub
            diff = target - v.sum()
    return v

def random_feasible_vector(n, rng):
    v = rng.random(n)  # uniform in [0,1)
    target = n / 2.0
    v = enforce_sum(v, target, rng)
    return v

def gradient_descent(h0, target, beta_list, rng, deadline):
    """Projected gradient descent on the log‑sum‑exp surrogate."""
    h = h0.copy()
    cur_c = compute_c5(h)
    n = h.size
    for beta in beta_list:
        step = 0.5
        for _ in range(3000):
            if time.time() > deadline:
                return h, cur_c
            conv = correlation(h, 1 - h)
            maxc = conv.max()
            # softmax weights
            w = np.exp(beta * (conv - maxc))
            ws = w.sum()
            if ws == 0:
                w = np.ones_like(w) / w.size
            else:
                w /= ws
            # gradient of surrogate
            conv1 = correlation(h, w)
            conv2 = correlation(h, w[::-1])
            offset = w.size - 1
            grad = 1.0 - conv1[offset:offset + n] - conv2[offset:offset + n]
            # projected step
            new_h = h - step * grad
            new_h = np.clip(new_h, 0.0, 1.0)
            new_h = enforce_sum(new_h, target, rng)
            new_c = compute_c5(new_h)
            if new_c < cur_c:
                h, cur_c = new_h, new_c
                step = min(step * 1.05, 1.0)
            else:
                step *= 0.5
                if step < 1e-6:
                    break
    return h, cur_c

def hill_climb(h_start, rng, max_iters=2000, start_step=0.4):
    """Simple greedy hill‑climb using pairwise mass transfers."""
    h = h_start.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = enforce_sum(new_h, target, rng)
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c

import time
import numpy as np

def correlation(a, b):
    """Full linear correlation a with b (like np.correlate(a,b,'full'))."""
    n, m = a.shape[0], b.shape[0]
    L = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b[::-1], L)
    conv = np.fft.irfft(fa * fb, L)[: n + m - 1]
    return conv

def compute_c5(h):
    n = h.size
    conv = correlation(h, 1 - h)
    return float(conv.max() * (2.0 / n))

def enforce_sum(v, target, rng, max_iter=10000):
    """Adjust v so that Σv = target while keeping 0 ≤ v ≤ 1."""
    diff = target - v.sum()
    n = v.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            i = rng.integers(0, n)
            cap = 1.0 - v[i]
            if cap <= 0.0:
                continue
            add = min(diff, cap)
            v[i] += add
        else:
            i = rng.integers(0, n)
            cap = v[i]
            if cap <= 0.0:
                continue
            sub = min(-diff, cap)
            v[i] -= sub
        diff = target - v.sum()
    return v

def random_feasible_vector(n, rng):
    v = rng.random(n)               # uniform in [0,1)
    return enforce_sum(v, n / 2.0, rng)

def gradient_descent(h0, target, beta_list, rng, deadline):
    """Projected gradient descent on a smooth log‑sum‑exp surrogate."""
    h = h0.copy()

    n = h.size
    for beta in beta_list:
        step = 0.5
        for _ in range(3000):
            if time.time() > deadline:
                return h, cur_c
            conv = correlation(h, 1 - h)
            maxc = conv.max()
            w = np.exp(beta * (conv - maxc))
            ws = w.sum()
            if ws == 0:
                w = np.ones_like(w) / w.size
            else:
                w /= ws
            conv1 = correlation(h, w)
            conv2 = correlation(h, w[::-1])
            off = w.size - 1
            grad = 1.0 - conv1[off:off + n] - conv2[off:off + n]
            new_h = h - step * grad
            new_h = np.clip(new_h, 0.0, 1.0)
            new_h = enforce_sum(new_h, target, rng)
            new_c = compute_c5(new_h)
            if new_c < cur_c:
                h, cur_c = new_h, new_c
                step = min(step * 1.05, 1.0)
            else:
                step *= 0.5
                if step < 1e-6:
                    break
