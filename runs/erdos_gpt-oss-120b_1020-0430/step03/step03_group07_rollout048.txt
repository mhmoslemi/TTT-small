def gradient_for_shift(h, k):
    n = len(h)
    g = np.empty(n)
    # (1 - h[i+k])
    if k >= 0:
        g[:n-k] = 1 - h[k:]

        g[:n-kk] -= h[kk:]

    return g

import time, numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Basic utilities
# --------------------------------------------------------------
def enforce_sum(h, target, rng):
    """Adjust h so that Σh == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Uniform 0.5 start, then random walk preserving the sum."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def perturb_h(h, rng, scale=0.2):
    """Transfer ε from i to j (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def correlate_full(h):
    """Full linear correlation h * (1‑h) via FFT."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    fr = np.fft.rfft((1 - h)[::-1], L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Discrete C₅ = max lag of h·(1‑h) over [0,2]."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)


# --------------------------------------------------------------
# Gradient for a given lag
# --------------------------------------------------------------
def gradient_for_shift(h, k):
    """∂/∂h_i of V_k = Σ h_i (1‑h_{i+k}) for zero‑padded h."""
    n = h.size
    g = np.empty(n, dtype=h.dtype)
    if k >= 0:
        g[: n - k] = 1.0 - h[k:]
        g[n - k :] = 1.0
        g[k:] -= h[: n - k]
    else:
        kk = -k
        g[kk:] = 1.0 - h[: n - kk]
        g[: kk] = 1.0
        g[: n - kk] -= h[kk:]
    return g


def targeted_improvement(h, rng, max_moves=200):
    """Reduce the worst lag using a gradient‑guided mass transfer."""
    h = h.copy()
    c = compute_c5(h)
    n = h.size
    for _ in range(max_moves):
        conv = correlate_full(h)
        idx = np.argmax(conv)
        k = idx - (n - 1)
        g = gradient_for_shift(h, k)
        pos = np.where(g > 1e-8)[0]
        neg = np.where(g < -1e-8)[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(g[pos])]
        j = neg[np.argmin(g[neg])]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            break
        # try decreasing eps sizes
        improved = False
        for eps in (0.2, 0.1, 0.05, 0.02, 0.01):
            eps = min(max_eps, eps)
            if eps <= 0:
                continue
            new_h = h.copy()
            new_h[i] -= eps
            new_h[j] += eps
            new_c = compute_c5(new_h)
            if new_c < c - 1e-12:
                h, c = new_h, new_c
                improved = True
                break
        if not improved:
            # fallback to a small random perturb
            new_h = perturb_h(h, rng, scale=0.15)
            new_c = compute_c5(new_h)
            if new_c < c:
                h, c = new_h, new_c
    return h, c

def hill_climb(h_start, rng, max_iters=1500, start_step=0.4):
    """Greedy hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h.copy()
            h = new_h
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


def simulated_annealing(start_h, rng, deadline, T0=0.05, Tf=1e-5, max_iters=2000):
    """Exponential cooling SA."""
    cur_h = start_h.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        new_h = perturb_h(cur_h, rng, scale=0.35)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
    return best_h, best_c

# Smooth surrogate + SLSQP

CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq(x):
    return np.sum(x) - CURRENT_TARGET


def _smooth_obj(x):
    corr = correlate_full(x)
    bc = CURRENT_BETA * corr
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=500):
    """SLSQP on log‑sum‑exp smoothed max."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq}
    res = minimize(
        _smooth_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x
