def project_to_feasible(v, target):
    """Project onto [0,1] box and sum target using bisection shift."""
    # v may be any values, we need to shift by const c such that sum(clip(v + c,0,1)) = target.
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo+hi)/2
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo+hi)/2, 0.0, 1.0)

import time
import numpy as np
from scipy.optimize import minimize

# --------------------------- helpers ---------------------------------

def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_to_feasible(v, target):
    """Projection onto {0≤h≤1, Σh = target} via bisection."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Fast generation of a random feasible vector (0≤h≤1, Σh=n/2)."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_to_feasible(v, n / 2.0)


def upsample_initial(to_n):
    """Linear interpolation of the global initial_h_values to length to_n."""
    init = np.array(initial_h_values, dtype=float)
    m = init.size
    xs_old = np.linspace(0.0, 2.0, m, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, to_n, endpoint=False)
    new = np.interp(xs_new, xs_old, init)
    return new


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=3000, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


def slsqp_optimize(h0, beta, rng, max_iter=500):
    """Run a single SLSQP optimisation with the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # Exact sum correction (project onto feasible hyperplane)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def gradient_descent(h0, max_iters, step0, rng):
    """Projected gradient descent on the worst‑lag correlation."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step0
    for it in range(max_iters):
        # correlation for non‑negative lags
        corr_full = np.correlate(h, 1 - h, mode="full")
        corr = corr_full[n - 1 :]          # lags 0 … n‑1
        worst = int(np.argmax(corr))        # lag giving maximal overlap

        # gradient of that worst term
        grad = np.zeros_like(h)
        if worst < n:
            grad[: n - worst] += 1 - h[worst:]
            grad[worst:] -= h[: n - worst]

        # gradient step and projection
        h_new = h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_feasible(h_new, target)

        c = compute_c5(h_new)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        step *= 0.995
        h = h_new
    return best_h, best_c

                if time.time() - start_time > budget_s * 0.95:
                    break
                try:
                    h_cur = slsqp_optimize(h_cur, beta, rng, max_iter=600)
                except Exception:
                    continue
                c_cur = compute_c5(h_cur)
                if c_cur < best_c_n:
                    best_c_n = c_cur
                    best_h_n = h_cur.copy()
                if time.time() - slsqp_start > slsqp_time:
                    break

            # ----- final hill‑climb -----
            if time.time() - start_time < budget_s * 0.98:
                h_cur, c_cur = hill_climb(h_cur, rng, max_iters=2500, start_step=0.3)
                if c_cur < best_c_n:
                    best_c_n = c_cur
                    best_h_n = h_cur.copy()

        # ----- update global best -----
        if best_c_n is not None and best_c_n < best_c:
            best_c = best_c_n
            best_h = best_h_n.copy()
            best_n = n

    # fallback (unlikely)
    if best_h is None:
        best_n = 60
        best_h = np.full(best_n, 0.5)

    # final feasibility enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, best_n)
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
