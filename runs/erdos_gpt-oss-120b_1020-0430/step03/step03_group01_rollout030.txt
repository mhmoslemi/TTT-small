    n = h.size

    j = rng.integers(n)
    if i == j: return h.copy()
    max_eps = min(h[i], 1.0 - h[j])

    eps = np.clip(eps, -max_eps, max_eps)
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def gradient_softmax(h, beta):
    n = h.shape[0]
    corr_raw = np.correlate(h, 1 - h, mode='full')
    # avoid overflow
    maxc = np.max(beta * corr_raw)
    expb = np.exp(beta * corr_raw - maxc)
    w = expb / expb.sum()
    grad = np.zeros_like(h)
    # positive lags
    for i in range(n):
        # k from 0 to n-1-i
        kmax = n - 1 - i
        if kmax >= 0:
            # w_pos = w[n-1:n-1+kmax+1]
            w_pos = w[n-1:n-1 + kmax + 1]
            grad[i] += np.dot(w_pos, 1 - h[i:i + kmax + 1])
        # negative lags
        lmax = n - 1 - i
        if lmax >= 1:  # l from 1..lmax
            w_neg = w[n-1 - np.arange(1, lmax+1)]
            # w_neg shape lmax
            grad[i] -= np.dot(w_neg, h[i+1:i+1+lmax])
    return grad
```

def simulated_annealing(h, rng, max_iters, init_T, final_T):
    best = h.copy()
    best_c = compute_c5(best)
    cur = h.copy()
    cur_c = best_c
    T = init_T
    for it in range(max_iters):
        # linear cooling
        T = init_T * ((final_T / init_T) ** (it / max_iters))
        # produce a candidate
        scale = 0.5 * (1 - it / max_iters) + 0.01
        cand = random_transfer(cur, rng, scale=scale)
        cand_c = compute_c5(cand)
        delta = cand_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            # accept
            cur, cur_c = cand, cand_c
            if cand_c < best_c:
                best, best_c = cand.copy(), cand_c
    return best, best_c
```

def directed_transfer(h, lag_max, rng, scale):
    n = len(h)
    if lag_max >=0:
        contrib = np.empty(n)
        contrib[:n-lag_max] = h[:n-lag_max] * (1 - h[lag_max:])
        contrib[n-lag_max:] = h[n-lag_max:]
    else:
        lag_abs = -lag_max
        contrib = np.empty(n)
        contrib[:n-lag_abs] = h[lag_abs:] * (1 - h[:n-lag_abs])
        contrib[n-lag_abs:] = 0.0
    i_max = np.argmax(contrib)
    i_min = np.argmin(contrib)
    # ensure i_max != i_min
    if i_max == i_min:
        # fallback random
        return random_transfer(h, rng, scale)
    # compute max_eps transfer from i_max to i_min
    max_eps = min(h[i_max], 1.0 - h[i_min])

    i_min = np.argmin(contrib)
    if i_max == i_min:
        return random_transfer(h, rng, scale)
    max_eps = min(h[i_max], 1.0 - h[i_min])
    if max_eps <= 0.0:
        return random_transfer(h, rng, scale)
    eps = rng.random() * max_eps * scale
    new_h = h.copy()
    new_h[i_max] -= eps
    new_h[i_min] += eps
    return new_h

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # simple correction
        rng = np.random.default_rng()
        n = x.size
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x

from scipy.optimize import minimize

# globals used by the smooth surrogate
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng):

            break
        i = rng.integers(0, n)

            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])

    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def directed_transfer(h, lag, rng, scale):
    n = h.size
    if lag >= 0:
        end = n - lag
        contrib = np.empty(n)
        contrib[:end] = h[:end] * (1.0 - h[lag:])
        contrib[end:] = h[end:]
    else:
        lag_abs = -lag
        end = n - lag_abs
        contrib = np.empty(n)
        contrib[:end] = h[lag_abs:] * (1.0 - h[:end])

    i_min = np.argmin(contrib)
