def simulated_annealing(h, time_limit, rng):
   start_time = time.time()
   best_h = h.copy()
   best_c = compute_c5(best_h)
   cur_h = h.copy()
   cur_c = best_c
   T = 0.02   # maybe 0.01?
   while time.time() - start_time < time_limit:
       # propose move
       new_h = perturb_random(cur_h, rng, scale=0.3)
       new_c = compute_c5(new_h)
       if new_c < cur_c or rng.random() < math.exp(-(new_c - cur_c) / T):
           cur_h, cur_c = new_h, new_c
           if new_c < best_c:
               best_h, best_c = new_h.copy(), new_c
       # temperature schedule
       T *= 0.9995  # gradually cool
       if T < 1e-5: T = 1e-5
   return best_h, best_c
```

def enforce_sum(h, target, rng):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # simple correction: adjust a random index
    # Use while loop but limited iterations
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        idx = rng.integers(0, h.size)
        if diff > 0:
            # increase
            inc = min(diff, 1.0 - h[idx])
            if inc <= 0:
                continue
            h[idx] += inc
            diff -= inc

            dec = min(-diff, h[idx])
            if dec <= 0:
                continue
            h[idx] -= dec
            diff += dec

def sum_eq_constraint(x):

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # compute (log-sum-exp)/beta
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
def slsqp_optimize(h0, beta, max_iter=300):

    CURRENT_BETA = beta

    cons = {'type':'eq','fun':sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol':1e-12, 'disp':False})
    x = np.clip(res.x,0.0,1.0)
    x = enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

def enforce_sum(h, target, rng):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # adjust gradually
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        idx = rng.integers(0, h.size)
        if diff > 0:
            inc = min(diff, 1.0 - h[idx])
            if inc <= 0:
                continue
            h[idx] += inc
            diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec <= 0:
                continue
            h[idx] -= dec
            diff += dec
    # final correction if needed
    if abs(diff) > 1e-12:
        # distribute remaining diff to random index with capacity
        for _ in range(10):
            idx = rng.integers(0, h.size)
            if diff > 0 and h[idx] < 1.0:
                h[idx] = min(1.0, h[idx] + diff)
                break
            elif diff < 0 and h[idx] > 0.0:
                h[idx] = max(0.0, h[idx] + diff)
                break
    return h

import time, math
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def enforce_sum(h, target, rng):
    """Clip to [0,1] and adjust to make sum exactly target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, h.size)
        if diff > 0:
            inc = min(diff, 1.0 - h[i])
            if inc <= 0:
                continue
            h[i] += inc
            diff -= inc
        else:
            dec = min(-diff, h[i])
            if dec <= 0:
                continue
            h[i] -= dec
            diff += dec
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    """Uniform random vector, projected onto the hyperplane Σh = n/2."""
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)

def perturb_random(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    while i == j:
        j = rng.integers(0, n)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def compute_c5(h):
    """Discrete C5 = max_k ∫ h(x)(1-h(x+k))dx."""
    corr = np.correlate(h, 1 - h, mode="full")
    return corr.max() * (2.0 / h.size)

# ---------- hill climbing ----------
def hill_climb(h_start, rng, max_iters=1500, step_scale=0.3):
    h = h_start.copy()
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    cur_c = corr.max() * (2.0 / n)
    best_h, best_c = h.copy(), cur_c
    for _ in range(max_iters):
        idx = np.argmax(corr)
        k = idx - (n - 1)
        grad = np.ones(n)
        if k >= 0:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
        else:
            kk = -k
            grad[: n - kk] -= h[kk:]
            grad[kk:] -= h[: n - kk]
        pos = np.where((grad > 1e-9) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-9) & (h < 1.0 - 1e-12))[0]
        if pos.size and neg.size:
            i = rng.choice(pos)
            j = rng.choice(neg)
        else:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            while i == j:
                j = rng.integers(0, n)
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = max_eps * step_scale * rng.random()
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_corr = np.correlate(new_h, 1 - new_h, mode="full")
        new_c = new_corr.max() * (2.0 / n)
        if new_c < best_c:
            best_c, best_h = new_c, new_h.copy()
        if new_c < cur_c:
            h, cur_c, corr = new_h, new_c, new_corr
    return best_h, best_c

# ---------- SLSQP with smooth max ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def _softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return enforce_sum(x, CURRENT_TARGET, np.random.default_rng())

# ---------- simulated annealing ----------
def simulated_annealing(h0, time_limit, rng):
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    cur_h, cur_c = best_h.copy(), best_c
    T = 0.02
    start = time.time()
    while time.time() - start < time_limit:
        new_h = perturb_random(cur_h, rng, scale=0.3)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < math.exp(-delta / T):
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
        T = max(T * 0.9995, 1e-6)
    return best_h, best_c

# ---------- up‑sampling ----------
def upsample(h, n_new, rng):
    n_old = h.size
    xs_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    xs_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(xs_new, xs_old, h)
    return enforce_sum(np.clip(h_new, 0.0, 1.0), n_new / 2.0, rng)

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ---- initial vector (may be supplied) ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = random_feasible_vector(360, rng)
    h0 = enforce_sum(np.clip(h0, 0.0, 1.0), h0.size / 2.0, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    n0 = best_h.size
    upsampled = False
    n_up = n0 * 2

    # ---- main search loop ----
    while time.time() - start_time < budget_s * 0.85:
        elapsed = time.time() - start_time

        # up‑sample once after ~40% of allotted time
        if not upsampled and elapsed > budget_s * 0.4:
            best_h = upsample(best_h, n_up, rng)
            best_c = compute_c5(best_h)
            upsampled = True

        n = best_h.size

        # pick a start point
        if rng.random() < 0.7:
            start_h = perturb_random(best_h, rng, scale=0.4)
        else:
            start_h = random_feasible_vector(n, rng)

        # hill climbing
        max_iters = 2500 if n == n0 else 1200
        hc_h, hc_c = hill_climb(start_h, rng, max_iters=max_iters, step_scale=0.4)
        if hc_c < best_c:
            best_h, best_c = hc_h, hc_c

        # SLSQP refinement with increasing betas
        for beta in (5, 20, 80, 200, 800, 2000):
            if time.time() - start_time > budget_s * 0.85:
                break
            ref_h = slsqp_optimize(best_h, beta, max_iter=500)
            ref_c = compute_c5(ref_h)
            if ref_c < best_c:
                best_h, best_c = ref_h, ref_c

    # ---- simulated annealing on the best candidate ----
    remaining = budget_s - (time.time() - start_time)
    if remaining > 10:
        sa_h, sa_c = simulated_annealing(best_h, time_limit=remaining * 0.6, rng=rng)
        if sa_c < best_c:
            best_h, best_c = sa_h, sa_c

    # ---- final micro‑search (tiny random moves) ----
    while time.time() - start_time < budget_s * 0.99:

        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_h, best_c = new_h, new_c

    # ---- clean‑up and final evaluation ----
    best_h = enforce_sum(np.clip(best_h, 0.0, 1.0), best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
