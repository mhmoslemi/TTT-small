def hill_climb_binary(N, rng, end_time, target_max):
    # initialize random binary vector
    h = np.zeros(N, dtype=np.int8)
    # set N/2 ones at random positions
    ones = rng.choice(N, N//2, replace=False)
    h[ones] = 1
    mis = compute_mismatches(h)
    best_h = h.copy()
    best_val = mis.max()
    # store iteration count for random restart
    no_improve = 0
    # while time left and best_val > target_max
    while time.time() < end_time and best_val > target_max:
        # find worst shift
        k = int(mis.argmax())
        # if k ==0: set k = int(np.argsort(-mis)[1]) # second largest
        if k == 0:
            # find max over k>0
            k = int(mis[1:].argmax() + 1)
        # find mismatched indices I
        I = np.where((h[:N - k] == 1) & (h[k:] == 0))[0]   # i where h[i]==1, h[i+k]==0
        # find candidate zeros J meeting reduction condition
        # cond: h[j]==0, j>=k and h[j-k]==1, and j+k<N and h[j+k]==1
        # compute array condition
        # We'll compute boolean mask for all indices
        mask_J = np.zeros(N, dtype=bool)
        # j such that h[j]==0
        zeros = (h == 0)
        # for each j, check if j>=k and h[j - k]==1
        cond_a = np.zeros(N, dtype=bool)
        cond_b = np.zeros(N, dtype=bool)
        if k <= N-1:
            cond_a[k:] = h[:N - k] == 1   # j>=k and h[j - k]==1
            cond_b[:N - k] = h[k:] == 1   # j+k<N and h[j + k]==1
        mask_J = zeros & cond_a & cond_b
        J = np.where(mask_J)[0]
        performed = False
        if I.size > 0 and J.size > 0:
            i = rng.choice(I)
            j = rng.choice(J)
            mis_before = mis.copy()
            swap_update(h, mis, i, j)  # updates mis and h
            new_val = mis.max()
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
                no_improve = 0
            if new_val <= mis_before.max():
                performed = True
            else:
                # revert
                swap_update(h, mis, j, i)  # swap back
        if not performed:
            # random swap
            ones_idx = np.where(h == 1)[0]
            zeros_idx = np.where(h == 0)[0]
            i = rng.choice(ones_idx)
            j = rng.choice(zeros_idx)
            mis_before = mis.copy()
            swap_update(h, mis, i, j)
            new_val = mis.max()
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
                no_improve = 0
            if new_val <= mis_before.max():
                performed = True
            else:
                # revert
                swap_update(h, mis, j, i)
        if performed:
            # improvement made, reset no_improve counter
            no_improve = 0
        else:
            no_improve += 1
        # random restart if stagnant
        if no_improve >= 2000:
            # reinitialize
            h = np.zeros(N, dtype=np.int8)
            ones = rng.choice(N, N//2, replace=False)
            h[ones] = 1
            mis = compute_mismatches(h)
            no_improve = 0
    return best_h, best_val

def enforce_sum(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

import time, numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# basic utilities
# --------------------------------------------------------------
def compute_c5(h):
    """Discrete C5 for a step function on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return corr[n - 1 :].max() * (2.0 / n)


def compute_mismatches(h):
    """Array of mismatches for each non‑negative shift."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return corr[n - 1 :].astype(np.int32)


def enforce_sum(h, rng):
    """Enforce Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


# --------------------------------------------------------------
# incremental update for binary swaps
# --------------------------------------------------------------
def swap_update(h, mis, i, j):
    """
    Swap a one at i with a zero at j (i has 1, j has 0).
    Updates the mismatch array `mis` in place.
    """
    n = h.shape[0]

    # i : 1 → 0
    for k in range(1, n - i):
        mis[k] -= (1 - h[i + k])
    for p in range(i):
        if h[p]:
            mis[i - p] += 1

    # j : 0 → 1
    for k in range(1, n - j):
        mis[k] += (1 - h[j + k])
    for p in range(j):
        if h[p]:
            mis[j - p] -= 1

    h[i] = 0
    h[j] = 1


# --------------------------------------------------------------
# binary hill‑climb / simulated annealing
# --------------------------------------------------------------
def hill_climb_binary(N, rng, deadline, target_max):
    """
    Search for a binary vector of length N with weight N/2
    whose maximum mismatch does not exceed target_max.
    Returns best vector found and its max mismatch.
    """
    # initialise
    h = np.zeros(N, dtype=np.int8)
    ones = rng.choice(N, N // 2, replace=False)
    h[ones] = 1
    mis = compute_mismatches(h)
    best_h = h.copy()
    best_val = mis.max()
    stagnant = 0

    while time.time() < deadline and best_val > target_max:
        # worst shift (skip k=0 which is always 0)
        k = int(mis.argmax())
        if k == 0:
            k = int(mis[1:].argmax() + 1)

        # mismatched positions for this shift
        I = np.where((h[: N - k] == 1) & (h[k:] == 0))[0]
        # candidate zeros that would improve this shift
        zeros = h == 0
        cond_a = np.zeros(N, dtype=bool)
        cond_b = np.zeros(N, dtype=bool)
        if k < N:
            cond_a[k:] = h[: N - k] == 1          # j>=k and h[j-k]==1
            cond_b[: N - k] = h[k:] == 1          # j+k<N and h[j+k]==1
        J = np.where(zeros & cond_a & cond_b)[0]

        performed = False
        # try guided improvement
        if I.size and J.size:
            i = rng.choice(I)
            j = rng.choice(J)
            old_mis = mis.copy()
            swap_update(h, mis, i, j)
            new_val = mis.max()
            if new_val < best_val:
                best_val, best_h = new_val, h.copy()
                stagnant = 0
            if new_val <= old_mis.max():
                performed = True
            else:
                # revert
                swap_update(h, mis, j, i)

        # fallback random swap
        if not performed:
            ones_idx = np.where(h == 1)[0]
            zeros_idx = np.where(h == 0)[0]
            i = rng.choice(ones_idx)
            j = rng.choice(zeros_idx)
            old_mis = mis.copy()
            swap_update(h, mis, i, j)
            new_val = mis.max()
            if new_val < best_val:
                best_val, best_h = new_val, h.copy()
                stagnant = 0
            if new_val <= old_mis.max():
                performed = True
            else:
                swap_update(h, mis, j, i)

        if performed:
            stagnant = 0
        else:
            stagnant += 1

        # restart if no progress for a while
        if stagnant >= 2500:
            h[:] = 0
            ones = rng.choice(N, N // 2, replace=False)
            h[ones] = 1
            mis = compute_mismatches(h)
            stagnant = 0

    return best_h, best_val

# continuous refinement (soft‑max + SLSQP) – same as parent code
# --------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=400):
    """Soft‑max SLSQP refinement for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x
