def enforce_sum(h, target, rng):
    # Clip first
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Simple proportional adjustment: add diff/n to each element, then clip and repeat

        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n-1)
    if j >= i:
        j += 1
    # compute max epsilon for transfer from i to j
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        # try opposite direction
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0:
            return h.copy()
        # swap i,j and direction
        i, j = j, i
    eps = (rng.random()*2 - 1) * max_eps * scale
    # ensure within bounds
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    # Clip (should stay within bounds)
    new_h = np.clip(new_h, 0.0, 1.0)
    return new_h

import time, math
import numpy as np
from scipy.optimize import minimize

# ---------- Global helpers for SLSQP surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth log‑sum‑exp surrogate of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    mx = np.max(bc)
    # log‑sum‑exp trick
    return (mx + np.log(np.sum(np.exp(bc - mx)))) / CURRENT_BETA


# ---------- Core problem utilities ----------
def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng):
    """Adjust h so that Σh = target while respecting 0≤h≤1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Small proportional adjustments
    for _ in range(5):
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            return h
    # Greedy fix of remaining error
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if j >= i:
        j += 1
    # direction: from i to j
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        # try opposite direction
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0.0:
            return h.copy()
        i, j = j, i
    eps = (rng.random() * 2.0 - 1.0) * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return np.clip(new_h, 0.0, 1.0)


def hill_climb_multi(h_start, rng, max_iters=2000, init_step=0.5, n_trials=5):
    """Hill‑climb that evaluates several random perturbations per step."""
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c = c
                h = new_h
                improved = True
                break
        if not improved and rng.random() < 0.02:
            new_h = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(new_h)
            if c < best_c:
                best_c = c
                h = new_h
                improved = True
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return h, best_c


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    rng = np.random.default_rng()
    x = enforce_sum(x, CURRENT_TARGET, rng)
    return x


# ---------- Main optimisation routine ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ---- 0. Initial vector -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ---- 1. Small pool of heuristics ---------------------------------------
    candidates = [best_h]
    xs = (np.arange(n) + 0.5) * (2.0 / n)

    for f in range(1, 7):
        sin_h = 0.5 + 0.5 * np.sin(np.pi * f * xs)
        sin_h = np.clip(sin_h, 0.0, 1.0)
        candidates.append(enforce_sum(sin_h, n / 2.0, rng))
        cos_h = 0.5 + 0.5 * np.cos(np.pi * f * xs)
        cos_h = np.clip(cos_h, 0.0, 1.0)
        candidates.append(enforce_sum(cos_h, n / 2.0, rng))

    for _ in range(5):
        candidates.append(random_feasible_vector(n, rng))

    for cand in candidates:
        c = compute_c5(cand)
        if c < best_c:
            best_c = c
            best_h = cand.copy()

    # ---- 2. Main optimisation loop (hill‑climb + SLSQP) --------------------
    beta_list = [5, 20, 80, 200, 500, 1000, 2000]
    while time.time() - start_time < budget_s * 0.6:
        # choose start point
        if rng.random() < 0.6:
            start_vec = perturb_h(best_h, rng, scale=0.4)
        else:
            start_vec = random_feasible_vector(n, rng)

        h_hc, c_hc = hill_climb_multi(
            start_vec,
            rng,
            max_iters=3000,
            init_step=0.5,
            n_trials=10,
        )
        if c_hc < best_c:
            best_c, best_h = c_hc, h_hc.copy()

        # SLSQP refinements with increasing β
        for beta in beta_list:
            if time.time() - start_time > budget_s * 0.6:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, max_iter=500)
                c_ref = compute_c5(h_ref)
                if c_ref < best_c:
                    best_c, best_h = c_ref, h_ref.copy()
            except Exception:
                continue

    # ---- 3. Simulated annealing fine‑tuning -------------------------------
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 10.0:
        # estimate cheap cost per iteration (~1 ms)
        est_iter_time = 0.001
        max_iters_sa = int(min(200000, remaining / est_iter_time * 0.7))
        max_iters_sa = max(30000, max_iters_sa)

        T0 = best_c * 0.5                     # initial temperature
        T = T0
        decay = 0.99995                       # exponential decay per iteration
        max_step_initial = 0.3                # maximal ε at start

        h_cur = best_h.copy()
        c_cur = best_c

        for it in range(max_iters_sa):
            if time.time() - start_time > budget_s:
                break

            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1

            # choose transfer direction
            direction = 1 if rng.random() < 0.5 else -1
            if direction == 1:
                max_eps = min(h_cur[i], 1.0 - h_cur[j])
            else:
                max_eps = min(h_cur[j], 1.0 - h_cur[i])
            if max_eps <= 0.0:
                continue

            step = max_step_initial * (T / T0)
            eps = direction * min(max_eps, step * rng.random())

            h_new = h_cur.copy()
            if direction == 1:
                h_new[i] -= eps
                h_new[j] += eps
            else:
                h_new[j] -= eps
                h_new[i] += eps

            c_new = compute_c5(h_new)
            if c_new < c_cur or rng.random() < math.exp(-(c_new - c_cur) / (T + 1e-12)):
                h_cur, c_cur = h_new, c_new
                if c_new < best_c:
                    best_c, best_h = c_new, h_new.copy()

            T *= decay

    # ---- 4. Final local refinement ----------------------------------------
    h_fin, c_fin = hill_climb_multi(best_h, rng, max_iters=2000, init_step=0.2, n_trials=10)
    if c_fin < best_c:
        best_c, best_h = c_fin, h_fin.copy()

    # ---- 5. Clean‑up -------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
