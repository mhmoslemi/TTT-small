def hill_climb_binary(h, rng, max_iters=2000, accept_worse=False, temperature=0.01):
   h = h.copy()
   best_c = compute_c5(h)
   # precompute indices
   for it in range(max_iters):
       ones = np.where(h > 0.5)[0]  # indices of 1
       zeros = np.where(h < 0.5)[0]  # indices of 0
       if len(ones) == 0 or len(zeros) == 0:
           break
       i = rng.choice(ones)
       j = rng.choice(zeros)
       # swap
       h[i], h[j] = 0, 1
       c = compute_c5(h)
       if c < best_c or (accept_worse and rng.random() < np.exp(-(c - best_c) / temperature)):
           best_c = c
       else:
           # revert
           h[i], h[j] = 1, 0
   return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    best_h = None
    best_c = np.inf
    best_n = None

    # parent
    parent = globals().get('initial_h_values', None)
    if parent is not None:
        parent = np.array(parent, dtype=float)
    
    candidate_ns = [96, 120, 144, 180, 240]

    for n in candidate_ns:
        if time.time() - start_time > budget_s*0.95:
            break
        target_sum = n/2.0
        # generate start vectors
        starts = []
        # parent upsample/interp
        if parent is not None:
            x_src = np.linspace(0, 2, parent.size, endpoint=False)
            x_tgt = np.linspace(0, 2, n, endpoint=False)
            h_interp = np.interp(x_tgt, x_src, parent)
            # enforce bounds and sum
            h_interp = np.clip(h_interp, 0, 1)
            diff = target_sum - h_interp.sum()
            if abs(diff) > 1e-12:
                # simple correction
                for _ in range(2000):
                    i = rng.integers(0, n)
                    add = min(diff, 1.0 - h_interp[i]) if diff > 0 else max(diff, -h_interp[i])
                    h_interp[i] += add
                    diff = target_sum - h_interp.sum()
                    if abs(diff) < 1e-12:
                        break
            starts.append(h_interp)

        # random binary
        h_bin = np.zeros(n)
        idx = rng.choice(n, n//2, replace=False)
        h_bin[idx] = 1.0
        starts.append(h_bin)

        # random continuous feasible
        starts.append(random_feasible_vector(n, rng))

        # possibly small perturbed best so far with same n
        if best_n == n and best_h is not None:

        for base_h in starts:
            # ensure feasibility
            h = np.clip(base_h, 0.0, 1.0)
            diff = target_sum - h.sum()
            if abs(diff) > 1e-12:
                for _ in range(2000):

                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n
            else:

            # Phase 4: final local random perturb hill climb (continuous)
            h, c = hill_climb(h, rng, max_iters=2000, start_step=0.3)

            if time.time() - start_time > budget_s*0.99:

        if time.time() - start_time > budget_s*0.99:

    # Optional upsample refinement
    if best_h is not None and time.time() - start_time < budget_s*0.99:
        if best_n * 2 <= 300:

            x_src = np.linspace(0, 2, best_n, endpoint=False)
            x_tgt = np.linspace(0, 2, n_up, endpoint=False)
            h_up = np.interp(x_tgt, x_src, best_h)
            # enforce bounds and sum
            target_up = n_up / 2.0
            h_up = np.clip(h_up, 0.0, 1.0)

            if abs(diff) > 1e-12:

                    add = min(diff, 1 - h_up[i]) if diff > 0 else max(diff, -h_up[i])
                    h_up[i] += add

            # refine upsampled

            for beta in (5,20,80,200,500,1000):

            c_up = compute_c5(h_up)

    # final corrections
    if best_h is None:

    best_h = np.clip(best_h, 0.0, 1.0)

    target = best_n/2.0

            if abs(diff) < 1e-12:
                break

import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h):
    """Discrete C5 for uniform spacing on [0,2]."""
    n = h.size
    return float(np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / n))

def perturb_h(h, rng, scale=0.2):
    """Transfer ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def random_feasible_vector(n, rng):
    """Random vector in [0,1] with exact sum n/2."""
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Deterministic hill‑climb for continuous h."""
    h = h_start.copy()
    best_c = compute_c5(h)
    step = start_step
    decay = 0.98
    for _ in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, h = c, new_h
        step = max(step * decay, 1e-5)
    return h, best_c

def hill_climb_binary(h_start, rng, max_iters=3000):
    """Hill‑climb on binary vectors by swapping a 1 with a 0."""
    h = h_start.copy()
    best_c = compute_c5(h)
    for _ in range(max_iters):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0
        h[j] = 1
        c = compute_c5(h)
        if c < best_c:
            best_c = c
        else:
            h[i] = 1
            h[j] = 0
    return h, best_c

def gradient_refine(h, rng, step=0.02, max_iter=12):
    """Gradient step against the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        h = h - (step * (0.5 ** it)) * g
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

# ---------- SLSQP with soft‑max ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x
