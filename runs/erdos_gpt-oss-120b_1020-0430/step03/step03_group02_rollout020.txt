def run(seed=42,budget_s=1100,**kwargs):
    rng = np.random.default_rng(seed)
    start=time.time()
    best_h=None; best_c=inf; best_n=None
    candidate_ns = [84,96,108,120,144,168,192,240,256]
    # optionally sort by increasing n but we want to ensure we attempt larger n if time.
    for n in candidate_ns:
        # For each n, allow some restarts
        restarts = 3 if n<=192 else 2  # bigger n maybe less restarts.
        for r in range(restarts):
            if time.time() - start > 0.96 * budget_s: break
            if best_h is not None and best_n == n:
                h0 = best_h.copy()
            elif best_h is not None:
                # interpolate
                h0 = np.interp(np.linspace(0,2,n,endpoint=False),
                               np.linspace(0,2,best_n,endpoint=False),
                               best_h)
            else:
                # random feasible
                h0 = random_feasible_vector(n, rng)
            # hill climb
            h, c = guided_hill_climb(h0, rng, max_iters=3000, step_scale=0.5)
            # SLSQP refinement with increasing beta up to maybe 5000
            for beta in (10, 50, 200, 800, 3000, 8000):
                if time.time() - start > 0.98 * budget_s: break
                h = slsqp_optimize(h, beta, rng, max_iter=800)
                c = compute_c5(h)
                # if improvement record
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n
            # gradient refine
            if time.time() - start < 0.99*budget_s:
                h = gradient_refine(h, rng, step=0.015, max_iter=12)
                c=compute_c5(h)
                if c < best_c:
                    best_h,best_c,best_n = h.copy(),c,n
            # final short hill climb
            if time.time() - start < 0.995*budget_s:
                h, c = guided_hill_climb(h, rng, max_iters=2000, step_scale=0.3)
                if c < best_c:
                    best_h,best_c,best_n = h.copy(),c,n
    # final small random perturbation loop
    while time.time() - start < budget_s * 0.999 and best_h is not None:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc
    # enforce sum and clip
    best_h = np.clip(best_h,0.0,1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)

def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0,1.0)]*n
    constraints = {'type':'eq','fun': lambda x: np.sum(x) - CURRENT_TARGET}
    # define softmax objective inside using current_beta
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        bc = CURRENT_BETA * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=constraints,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum exactly
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

import time, numpy as np
from scipy.optimize import minimize

# globals for the SLSQP soft‑max objective
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, rng):
    """Adjust vector to satisfy Σh = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple random redistribution
    for _ in range(20000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Random feasible h (0≤h≤1, Σh=n/2)."""
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, rng)


def worst_gradient(h):
    """Gradient of the worst‑shift overlap."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]          # subtract h_{i+k}
        grad[k:]       -= h[: n - k]    # subtract h_{i‑k}
    return grad


def guided_perturb(h, grad, rng, scale=0.5):
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h_start, rng, max_iters=2500, step_scale=0.5):
    """Hill‑climb using gradient‑directed transfers."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    scale = step_scale
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        else:
            # keep current h (no acceptance of worse move)
            h = nh
        if (it + 1) % decay == 0:
            scale = max(scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h, rng, step=0.02, max_iter=8):
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        grad = worst_gradient(h)
        # move opposite to gradient (descent)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp, scaled back
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=400):
    """Soft‑max SLSQP optimisation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:

        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

    new_h = np.interp(x_new, x_old, old_h)
    new_h = np.clip(new_h, 0.0, 1.0)
    return new_h

            h_up, c_up = guided_hill_climb(h_up, rng, max_iters=2000, step_scale=0.3)
            for beta in (200, 800, 3000):
                if time.time() - start > budget_s:
                    break
                h_up = slsqp_optimize(h_up, beta, rng, max_iter=400)
                c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, best_n = h_up.copy(), c_up, n_up

    # ------------------------------------------------------------------
    # Final local search with tiny random transfers (remaining budget)
    # ------------------------------------------------------------------
    while time.time() - start < budget_s * 0.999 and best_h is not None:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # ------------------------------------------------------------------
    # Ensure feasibility and return
    # ------------------------------------------------------------------
    if best_h is None:
        best_n = 84
        best_h = np.full(best_n, 0.5)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
