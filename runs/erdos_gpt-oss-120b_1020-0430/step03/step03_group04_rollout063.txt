def delta_corr_swap(i, j, h, b):
    n = len(h)
    delta = np.zeros(n, dtype=int)
    # i-first
    s_i = np.arange(n - i)
    delta[s_i] -= b[i + s_i]
    # j-first
    s_j = np.arange(n - j)
    idx_j_s = j + s_j
    # new_j_first: if idx_j_s == i then 1 else b[idx_j_s]
    new_j_first = b[idx_j_s]
    mask = (idx_j_s == i)
    new_j_first[mask] = 1
    delta[s_j] += new_j_first
    # i-second
    s_i2 = np.arange(i + 1)
    p_i = i - s_i2
    # new_i_second = h[p_i] with adjustments at p_i == i (s=0) => 0, p_i == j => 1
    new_i_second = h[p_i].copy()
    new_i_second[0] = 0  # p_i == i
    if j <= i:
        s_j2 = i - j
        new_i_second[s_j2] = 1
    delta[s_i2] += new_i_second
    # j-second
    s_j2 = np.arange(j + 1)
    p_j = j - s_j2
    old_j_second = h[p_j]
    delta[s_j2] -= old_j_second
    return delta

def compute_c5(h):
    """Exact discrete C5 for step function h on [0,2] with uniform spacing."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def enforce_sum(arr, target, rng):
    """Adjust arr so that its sum equals target while respecting 0‑1 bounds."""
    diff = target - arr.sum()
    n = arr.size
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - arr[i])
            else:
                add = max(diff, -arr[i])
            arr[i] += add
            diff = target - arr.sum()
            if abs(diff) < 1e-12:
                break
    return arr

# Binary‑only hill‑climb (fast O(n) delta updates)

def delta_corr_swap(i, j, h, b):
    """
    Incremental change of the non‑negative correlation
    when swapping a 1 at i with a 0 at j.
    Returns an integer array delta of length n.
    """
    n = h.size
    delta = np.zeros(n, dtype=int)

    # i as first index (removed)
    delta[: n - i] -= b[i:]

    # j as first index (added)
    new_j_first = b[j:].copy()
    if i >= j:                     # the element at position i becomes 1
        new_j_first[i - j] = 1
    delta[: n - j] += new_j_first

    # i as second index (b[i] becomes 1)
    new_i_second = h[i::-1].copy()     # h[i], h[i‑1], … , h[0]
    new_i_second[0] = 0                # position i itself becomes 0 in h
    if j <= i:
        new_i_second[i - j] = 1        # position j becomes 1 in h
    delta[: i + 1] += new_i_second

    # j as second index (b[j] becomes 0)
    delta[: j + 1] -= h[j::-1]

    return delta


def binary_hill_climb(start_h, rng, deadline):
    """
    Greedy hill‑climb on binary vectors.
    Returns the best vector seen and its maximal overlap (integer).
    """
    n = start_h.size
    h = start_h.astype(np.int8).copy()
    b = 1 - h
    corr = np.correlate(h, b, mode="full")[n - 1 :].astype(int)
    max_corr = int(corr.max())

    best_h = h.copy()
    best_max = max_corr
    no_improve = 0

    while time.time() < deadline and no_improve < 5000:
        # propose a random 1↔0 swap
        i = rng.integers(n)
        while h[i] != 1:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] != 0:
            j = rng.integers(n)

        delta = delta_corr_swap(i, j, h, b)
        new_corr = corr + delta
        new_max = int(new_corr.max())

        if new_max < max_corr:
            # accept improvement
            corr = new_corr
            max_corr = new_max
            h[i] = 0
            h[j] = 1
            b[i] = 1
            b[j] = 0
            no_improve = 0
            if max_corr < best_max:
                best_max = max_corr
                best_h = h.copy()
                # early exit if already excellent
                if best_max <= 0:
                    break
        else:
            # occasional acceptance on plateau to escape local minima
            if new_max == max_corr and rng.random() < 0.001:
                corr = new_corr
                max_corr = new_max
                h[i] = 0
                h[j] = 1
                b[i] = 1
                b[j] = 0
            no_improve += 1

    return best_h, best_max

def gradient_refine(h, rng, step=0.02, max_iter=12):
    """Refine by moving against gradient of the worst shift."""
    n = h.size
    target = n / 2.0
    for _ in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)

        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]          # + (1 - h[i+k])
            g[k:] -= h[:upper]                # - h[i]

        h = h - step * g
        h = np.clip(h, 0.0, 1.0)

                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h
