def compute_gradient(h, s):
    n = len(h)
    b = 1 - h
    grad = np.zeros(n)
    if s >= 0:
        # b_shift = b[s:]  (size n-s)
        grad[:n-s] += b[s:]
        # a_shift = h[:n-s] (size n-s)
        grad[s:] -= h[:n-s]
    else:
        # s negative
        s_abs = -s
        # b_shift = b[:n-s_abs] (size n-s_abs)
        grad[s_abs:] += b[:n-s_abs]
        # a_shift = h[s_abs:] (size n-s_abs)
        grad[:n-s_abs] -= h[s_abs:]
    return grad

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def compute_c5_and_shift(h):
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr)
    return corr[idx] * (2.0 / h.size), idx

def binary_hill_climb(h_bin, rng, max_iters):
    # h_bin is binary array
    n = h_bin.size
    best = h_bin.copy()
    best_c = compute_c5(best.astype(float))
    for it in range(max_iters):
        # random swap
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if best[i] == best[j]:
            continue  # no swap
        # swap i and j
        new = best.copy()
        new[i], new[j] = new[j], new[i]
        new_c = compute_c5(new.astype(float))
        if new_c < best_c:
            best = new
            best_c = new_c
        # optional early break not needed
    return best.astype(float), best_c

import time
import numpy as np
from scipy.optimize import minimize

# ----------- Global placeholders for the soft‑max surrogate -------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=400):
    """Single SLSQP optimisation using the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # exact sum correction (project onto feasible hyperplane)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def compute_c5_and_shift(h):
    """Return (C5, index_of_max_shift) where index runs in 0..2n-2."""
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr)
    return corr[idx] * (2.0 / h.size), idx


def compute_gradient(h, shift_idx):
    """Sub‑gradient of the raw correlation at the maximal shift."""
    n = h.size
    s = shift_idx - (n - 1)          # signed shift
    b = 1 - h
    grad = np.zeros(n)
    if s >= 0:
        grad[: n - s] += b[s:]
        grad[s:] -= h[: n - s]
    else:
        s_abs = -s
        grad[s_abs:] += b[: n - s_abs]
        grad[: n - s_abs] -= h[s_abs:]
    return grad


def project_to_feasible(v, target):
    """Projection onto {0≤h≤1, Σh=target} via a simple bisection on a shift."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Fast generation of a random feasible vector (0≤h≤1, Σh=n/2)."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_to_feasible(v, n / 2.0)


# ---------------------- perturbation utilities -------------------------

def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.3):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


def binary_hill_climb(bin_vec, rng, max_iters=2000):
    """Swap‑based hill climbing on a binary (0/1) vector with exactly n/2 ones."""
    n = bin_vec.size
    best = bin_vec.copy()
    best_c = compute_c5(best.astype(float))
    for _ in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if best[i] == best[j]:
            continue
        cand = best.copy()
        cand[i], cand[j] = cand[j], cand[i]
        cand_c = compute_c5(cand.astype(float))
        if cand_c < best_c:
            best = cand
            best_c = cand_c
    return best.astype(float), best_c


def pgd_optimize(h0, rng, max_iters=800, init_step=0.5):
    """Projected gradient descent using the exact max‑shift subgradient."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    step = init_step
    for it in range(max_iters):
        c, idx = compute_c5_and_shift(h)
        grad = compute_gradient(h, idx)
        new_h = h - step * grad
        new_h = project_to_feasible(new_h, n / 2.0)
        new_c = compute_c5(new_h)
        if new_c < best_c - 1e-12:
            best_c = new_c
            best_h = new_h.copy()
            h = new_h
            step = min(step * 1.2, 5.0)
        else:
            step *= 0.6
        if step < 1e-5:
            break
    return best_h, best_c
