def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, scipy.optimize as opt
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    candidate_ns = [120, 200, 300, 400, 600]   # sorted ascending for early quick runs.

    # ensure includes initial's size
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 not in candidate_ns:
            candidate_ns = [n0] + candidate_ns

    for n in candidate_ns:
        elapsed = time.time() - start
        remaining = budget_s - elapsed
        if remaining < 5:
            break
        # allocate time for this n
        # allocate proportionally to n maybe or fixed portion; let's allocate a fraction of remaining.
        time_for_n = min(remaining * 0.5, max(5.0, min(200.0, n * 0.4)))
        # We'll keep track of start time for this n
        n_start = time.time()

        # generate initial population
        target = n / 2.0
        pop = []
        if 'initial_h_values' in globals():
            init_vec = upsample_initial(n)
            init_vec = project_to_feasible(init_vec, target)
            pop.append(init_vec)
        while len(pop) < max(10, int(0.05 * n)):  # popsize scale with n
            pop.append(random_feasible_vector(n, rng))
        pop = np.array(pop)
        scores = np.array([compute_c5(v) for v in pop])
        best_idx = scores.argmin()
        local_best_h = pop[best_idx].copy()
        local_best_c = scores[best_idx]

        # DE loop
        F = 0.8
        CR = 0.9
        popsize = pop.shape[0]
        # Run DE until time_for_n*0.6 used
        de_time_budget = time_for_n * 0.6
        while time.time() - n_start < de_time_budget:
            for i in range(popsize):
                # randomly choose three distinct indices
                idxs = list(range(popsize))
                idxs.remove(i)
                r1, r2, r3 = rng.choice(idxs, 3, replace=False)
                mutant = pop[r1] + F * (pop[r2] - pop[r3])
                # ensure within [0,1] via projection after crossover
                cross = rng.random(n) < CR
                if not np.any(cross):
                    cross[rng.integers(0, n)] = True
                trial = np.where(cross, mutant, pop[i])
                trial = project_to_feasible(trial, target)
                trial_c = compute_c5(trial)
                if trial_c < scores[i]:
                    pop[i] = trial
                    scores[i] = trial_c
                    if trial_c < local_best_c:
                        local_best_c = trial_c
                        local_best_h = trial.copy()
        # After DE, local refinement with SLSQP
        # We'll perform multiple restarts of SLSQP with increasing beta
        # Starting from local_best_h and few random restarts
        refine_start = time.time()
        # compute number of SLSQP runs possible within remaining time portion
        slsqp_time_budget = time_for_n * 0.3
        # We'll loop performing runs until budget exhausted
        while time.time() - refine_start < slsqp_time_budget:
            # seed with current best or random
            if rng.random() < 0.5:
                h0 = local_best_h.copy()
            else:
                h0 = random_feasible_vector(n, rng)
            # increase beta gradually
            for beta in (10., 50., 200., 800., 2000., 8000.):
                if time.time() - start > budget_s * 0.96:
                    break
                try:
                    h_opt = slsqp_optimize(h0, beta, rng, max_iter=800)
                except Exception:
                    continue
                c_opt = compute_c5(h_opt)
                if c_opt < local_best_c:
                    local_best_c = c_opt
                    local_best_h = h_opt.copy()
                # set h0 for next beta
                h0 = h_opt
                # break out if time low
                if time.time() - start > budget_s * 0.96:
                    break
        # final hill climb
        if time.time() - start < budget_s * 0.98:
            h_hc, c_hc = hill_climb(local_best_h, rng, max_iters=3000, start_step=0.2)
            if c_hc < local_best_c:
                local_best_c = c_hc
                local_best_h = h_hc

        # binary refinement: generate binary from local_best_h
        if time.time() - start < budget_s * 0.98:
            # threshold at median to get exactly n/2 ones
            sorted_idx = np.argsort(local_best_h)
            binary = np.zeros(n)
            binary[sorted_idx[-int(n/2):]] = 1.0
            # binary hill climbing via swaps
            binary_h, binary_c = binary_swap_hill_climb(binary, rng, max_iters=5000)
            if binary_c < local_best_c:
                local_best_c = binary_c
                local_best_h = binary_h

        # update global best
        if local_best_c < best_c:
            best_c = local_best_c
            best_h = local_best_h.copy()
            best_n = n

    # final check and projection
    if best_h is None:
        best_n = 120
        best_h = np.full(best_n, 0.5)
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        # adjust via projection
        best_h = project_to_feasible(best_h, target)

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)

def binary_swap_hill_climb(h, rng, max_iters=2000):
    # h is binary vector with sum = n/2 (float 0/1)
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(best_h)
    for it in range(max_iters):
        # choose i where h[i]==1, j where h[j]==0
        idx1 = rng.choice(np.where(best_h == 1)[0])
        idx0 = rng.choice(np.where(best_h == 0)[0])
        # swap
        trial = best_h.copy()
        trial[idx1] = 0
        trial[idx0] = 1
        c = compute_c5(trial)
        if c < best_c:
            best_c = c
            best_h = trial
    return best_h, best_c

def slsqp_optimize(h0, beta, rng, max_iter=800):
    target = len(h0) / 2.0
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    cons = {'type':'eq', 'fun': (lambda x: np.sum(x) - target)}
    bounds = [(0.0, 1.0)] * len(h0)
    res = opt.minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol':1e-12, 'disp':False})
    x = np.clip(res.x, 0.0, 1.0)
    # Project to meet sum
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # simple redistribution
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, len(x))
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x

import time
import numpy as np
import scipy.optimize as opt

# ---------- Utility functions ----------

def compute_c5(h):
    """Exact discrete C5 for a step function h on [0,2] (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_to_feasible(v, target):
    """Project onto {0 ≤ h ≤ 1, Σh = target} by shifting and clipping."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Fast random feasible vector (0 ≤ h ≤ 1, Σh = n/2)."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_to_feasible(v, n / 2.0)


def upsample_initial(to_n):
    """Linear interpolation of global initial_h_values to length to_n."""
    init = np.array(initial_h_values, dtype=float)
    m = init.size
    xs_old = np.linspace(0.0, 2.0, m, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, to_n, endpoint=False)
    return np.interp(xs_new, xs_old, init)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=3000, start_step=0.2):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-5)
    return best_h, best_c


def slsqp_optimize(h0, beta, rng, max_iter=800):
    """SLSQP refinement using a smooth log‑sum‑exp surrogate of the max."""
    n = h0.size
    target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    bounds = [(0.0, 1.0)] * n
    res = opt.minimize(
        obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # projection to fix any tiny sum error
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x


def binary_swap_hill_climb(h, rng, max_iters=5000):
    """Hill‑climb on a binary vector via swaps (preserves number of 1's)."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(best_h)
    ones = np.where(best_h == 1)[0]
    zeros = np.where(best_h == 0)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return best_h, best_c
    for _ in range(max_iters):
        i = rng.choice(ones)
        j = rng.choice(zeros)
        trial = best_h.copy()
        trial[i] = 0
        trial[j] = 1
        c = compute_c5(trial)
        if c < best_c:
            best_c = c
            best_h = trial
            # update index sets
            ones = np.where(best_h == 1)[0]
            zeros = np.where(best_h == 0)[0]
    return best_h, best_c


# ---------- Main optimisation routine ----------

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    best_h = None
    best_c = np.inf
    best_n = None

    # candidate discretisations (start small, end larger)
    candidate_ns = [120, 200, 300, 400, 600]

    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        if n0 not in candidate_ns:
            candidate_ns = [n0] + candidate_ns

    for n in candidate_ns:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining < 5.0:
            break

        # allocate a chunk of the remaining budget to this n
        time_for_n = min(remaining * 0.5, max(5.0, min(200.0, n * 0.4)))
        n_start = time.time()

        target = n / 2.0

        # ----- initialise population -----
        pop = []
        if "initial_h_values" in globals():
            pop.append(project_to_feasible(upsample_initial(n), target))
        while len(pop) < max(10, int(0.05 * n)):
            pop.append(random_feasible_vector(n, rng))
        pop = np.array(pop)
        scores = np.array([compute_c5(v) for v in pop])
        best_idx = scores.argmin()
        local_best_h = pop[best_idx].copy()
        local_best_c = scores[best_idx]

        # ----- Differential Evolution (global search) -----
        popsize = pop.shape[0]
        F = 0.8
        CR = 0.9
        de_time_budget = time_for_n * 0.6
        while time.time() - n_start < de_time_budget:
            for i in range(popsize):
                idxs = list(range(popsize))
                idxs.remove(i)
                r1, r2, r3 = rng.choice(idxs, 3, replace=False)
                mutant = pop[r1] + F * (pop[r2] - pop[r3])
                cross = rng.random(n) < CR
                if not np.any(cross):
                    cross[rng.integers(0, n)] = True
                trial = np.where(cross, mutant, pop[i])
                trial = project_to_feasible(trial, target)
                trial_c = compute_c5(trial)
                if trial_c < scores[i]:
                    pop[i] = trial
                    scores[i] = trial_c
                    if trial_c < local_best_c:
                        local_best_c = trial_c
                        local_best_h = trial.copy()

        # ----- SLSQP refinement (local search) -----
        slsqp_time_budget = time_for_n * 0.3
        slsqp_start = time.time()
        while time.time() - slsqp_start < slsqp_time_budget:
            # start from current best or a fresh random vector
            h0 = local_best_h.copy() if rng.random() < 0.6 else random_feasible_vector(n, rng)
            for beta in (10.0, 50.0, 200.0, 800.0, 2000.0, 8000.0):
                if time.time() - start_time > budget_s * 0.96:
                    break
                try:
                    h_opt = slsqp_optimize(h0, beta, rng, max_iter=800)
                except Exception:
                    continue
                c_opt = compute_c5(h_opt)
                if c_opt < local_best_c:
                    local_best_c = c_opt
                    local_best_h = h_opt.copy()
                h0 = h_opt
                if time.time() - start_time > budget_s * 0.96:
                    break

        # ----- Deterministic hill‑climb -----
        if time.time() - start_time < budget_s * 0.98:
            h_hc, c_hc = hill_climb(local_best_h, rng, max_iters=3000, start_step=0.2)
            if c_hc < local_best_c:
                local_best_c = c_hc
                local_best_h = h_hc

        # ----- Binary refinement (swap hill‑climb) -----
        if time.time() - start_time < budget_s * 0.98:
            # produce a binary vector with exactly n/2 ones
            order = np.argsort(local_best_h)
            binary = np.zeros(n)
            binary[order[-int(target):]] = 1.0
            bin_h, bin_c = binary_swap_hill_climb(binary, rng, max_iters=5000)
            if bin_c < local_best_c:
                local_best_c = bin_c
                local_best_h = bin_h

        # ----- Update global best -----
        if local_best_c < best_c:
            best_c = local_best_c
            best_h = local_best_h.copy()
            best_n = n

    # Fallback if nothing found
    if best_h is None:
        best_n = 120
        best_h = np.full(best_n, 0.5)

    # Final feasibility enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        best_h = project_to_feasible(best_h, target)

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
