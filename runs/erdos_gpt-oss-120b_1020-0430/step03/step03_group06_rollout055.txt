```python
def compute_corr(h):
    n = h.shape[0]
    N = 2*n
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(H * G, N)[:2*n-1]
    return corr
```

def hill_climb_targeted(h, rng, max_iters=20000, scale=0.5):
    best_h = h.copy()
    best_c = compute_c5(h)
    h_cur = h.copy()
    cur_c = best_c
    n = len(h)
    for it in range(max_iters):
        # compute correlation and max shift
        corr = compute_corr(h_cur)
        max_idx = np.argmax(corr)
        max_k = max_idx - (n - 1)
        # compute contributions for this shift
        if max_k >= 0:
            a = h_cur[:n-max_k] * (1.0 - h_cur[max_k:])
            offset = 0
        else:
            a = h_cur[-max_k:] * (1.0 - h_cur[:n+max_k])
            offset = -max_k
        i_local = np.argmax(a)
        i = i_local + offset
        # choose j:
        # pick random indices until find one with capacity to receive epsilon
        attempts = 0
        while True:
            j = rng.integers(0, n)
            if j != i and h_cur[j] < 1.0:
                break
            attempts += 1
            if attempts > 10:
                j = (i + rng.integers(1, n)) % n
                break
        max_eps = min(h_cur[i], 1.0 - h_cur[j])
        if max_eps <= 0:
            continue
        eps = max_eps * rng.random() * scale

            continue

            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()
        # optionally also accept equal or slight increase based on probability
        # but maybe not needed
    return best_h, best_c

def enforce_sum(h, target, rng, max_iter=10000):
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) <= 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-12:
            continue
        h[i] += add
        diff = target - h.sum()
    # final clip
    h = np.clip(h, 0.0, 1.0)
    # enforce again if sum off due to clipping
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # distribute remainder uniformly
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def hill_climb_targeted(h_start, rng, max_iters=20000, scale=0.5):
    h = h_start.copy()
    n = len(h)
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    no_improve = 0
    for it in range(max_iters):
        # occasionally random move
        if rng.random() < 0.3:
            i = rng.integers(0, n)
            j = rng.integers(0, n-1)
            if j >= i:
                j += 1
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0:
                continue
            eps = max_eps * rng.random() * scale
            new_h = h.copy()
            new_h[i] -= eps
            new_h[j] += eps
        else:
            # targeted move
            corr = compute_corr(h)
            max_idx = np.argmax(corr)
            k = max_idx - (n - 1)
            if k >= 0:
                a = h[:n-k] * (1.0 - h[k:])
                i_local = int(np.argmax(a))
                i = i_local
            else:
                a = h[-k:] * (1.0 - h[:n+k])
                i_local = int(np.argmax(a))
                i = i_local - k
            # choose j
            # try up to 5 attempts to find j with capacity
            for _ in range(5):
                j = rng.integers(0, n)

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global helpers for the smooth surrogate used by SLSQP
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    corr = compute_corr(x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------
def compute_corr(h):
    """Linear correlation h * (1‑h) using FFT (full mode)."""
    n = h.shape[0]
    N = 2 * n
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(H * np.conj(G), N)
    return corr[: 2 * n - 1]


def compute_c5(h):
    """Exact discrete C₅ for a step function."""
    n = h.shape[0]
    return np.max(compute_corr(h)) * (2.0 / n)


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h so that Σh = target while 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) <= 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-12:
            continue
        h[i] += add
        diff = target - h.sum()
    h = np.clip(h, 0.0, 1.0)
    # tiny uniform correction if needed
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    v = rng.exponential(1.0, size=n)
    v = v / v.sum() * (n / 2.0)
    h = np.clip(v, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


# ----------------------------------------------------------------------
# Hill climbing with a targeted move towards the worst shift
# ----------------------------------------------------------------------
def hill_climb_targeted(h_start, rng, max_iters=20000, scale=0.5):
    h = h_start.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    no_improve = 0
    for _ in range(max_iters):
        # 30 % random move, 70 % targeted
        if rng.random() < 0.3:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
        else:
            # compute the shift giving the current maximum overlap
            corr = compute_corr(h)
            max_idx = int(np.argmax(corr))
            k = max_idx - (n - 1)          # shift relative to centre
            # contribution of each i for this shift
            if k >= 0:
                a = h[: n - k] * (1.0 - h[k:])
                i_local = int(np.argmax(a))
                i = i_local
            else:
                a = h[-k:] * (1.0 - h[: n + k])
                i_local = int(np.argmax(a))
                i = i_local - k
            # pick a j with room to increase
            j = i
            for _ in range(5):
                cand = rng.integers(0, n)
                if cand != i and h[cand] < 1.0:
                    j = cand
                    break
        # transfer a small amount while preserving sum
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = max_eps * rng.random() * scale
        if eps <= 0:
            continue
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        if new_c < cur_c - 1e-12:
            h = new_h
            cur_c = new_c
            no_improve = 0
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
        else:
            no_improve += 1
            if no_improve >= 2000:
                scale = max(scale * 0.5, 0.01)
                no_improve = 0
    return best_h, best_c


# ----------------------------------------------------------------------
# SLSQP on smooth surrogate (continuation method)
# ----------------------------------------------------------------------
def slsqp_optimize(h0, beta, max_iter=800):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) <= 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            if abs(add) < 1e-12:
                continue
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----- initialise --------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.size

    else:
        n = 180
        target = n / 2.0
        h = np.concatenate([np.ones(int(target)), np.zeros(int(n - target))])
        rng.shuffle(h)
        h = h.astype(float)
        h = enforce_sum(h, target, rng)

    best_h, best_c = h.copy(), compute_c5(h)

    # ----- add a few analytic seeds (cosine shapes) -------------------
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    for f in range(1, 7):
        for a in np.linspace(0.1, 0.5, 9):
            cand = 0.5 + a * np.cos(np.pi * f * xs)
            cand = np.clip(cand, 0.0, 1.0)
            cand = enforce_sum(cand, n / 2.0, rng)
            c = compute_c5(cand)
            if c < best_c:
                best_h, best_c = cand.copy(), c

    # ----- main optimisation loop --------------------------------------
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break

        # local improvement on current grid size
        max_iters = int(20000 * (len(best_h) / 180))
        max_iters = min(max_iters, 80000)
        h_opt, c_opt = hill_climb_targeted(best_h, rng, max_iters=max_iters, scale=0.5)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

        # occasional SLSQP refinement
        if time.time() - start_time < budget_s * 0.90:
            for beta in (100, 400, 1200, 4000):
                if time.time() - start_time > budget_s * 0.90:
                    break
                try:
                    h_ref = slsqp_optimize(best_h, beta, max_iter=1000)
                except Exception:
                    continue
                c_ref = compute_c5(h_ref)
                if c_ref < best_c:
                    best_h, best_c = h_ref, c_ref

        # up‑sample if time permits and we are not at maximum resolution
        if len(best_h) >= 720:
            continue
        if time.time() - start_time > budget_s * 0.80:
            break
        best_h = np.repeat(best_h, 2)                     # factor‑2 up‑sample
        best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
        best_c = compute_c5(best_h)

    # ----- final polishing ---------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
