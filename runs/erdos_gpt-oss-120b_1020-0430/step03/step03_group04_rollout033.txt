def run(seed=42, budget_s=1100, **kwargs):
   rng = np.random.default_rng(seed)
   start_time = time.time()
   best_h = None
   best_c = np.inf
   best_n = None

   # candidate discretizations
   candidate_ns = [96, 120, 144, 180]

   for n in candidate_ns:
       if time.time() - start_time > budget_s * 0.95:
           break
       # generate start vectors
       start_vecs = []
       # parent interpolation if available
       if 'initial_h_values' in globals():
           parent = np.array(initial_h_values, dtype=float)
           # Interpolate parent to n points
           x_src = np.linspace(0.0, 2.0, parent.size, endpoint=False)
           x_tgt = np.linspace(0.0, 2.0, n, endpoint=False)
           h_interp = np.interp(x_tgt, x_src, parent)
           # enforce feasibility
           target = n / 2.0
           diff = target - h_interp.sum()
           # simple redistribution to correct sum
           if abs(diff) > 1e-12:
               for _ in range(2000):
                   i = rng.integers(0, n)
                   add = min(diff, 1.0 - h_interp[i]) if diff > 0 else max(diff, -h_interp[i])
                   h_interp[i] += add
                   diff = target - h_interp.sum()
                   if abs(diff) < 1e-12:
                       break
           start_vecs.append(h_interp)

       # random binary
       start_vecs.append(random_balanced_binary(n, rng))

       # random continuous feasible (existing function)
       start_vecs.append(random_feasible_vector(n, rng))

def compute_c5(h):
    """Discrete C5 for step function h on [0,2] with uniform spacing."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def random_feasible_vector(n, rng):
    """Uniform random vector in [0,1] with exact sum n/2."""
    h = rng.random(n)
    target = n / 2.0
    scale = target / h.sum()
    h *= scale
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h


def random_balanced_binary(n, rng):
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]          # + (1 - h[i+k])
            g[k:] -= h[:upper]                # - h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=400):

def binary_swap_improve(h, rng, max_iters, deadline):
    """Monotone improvement by swapping 1‑0 pairs that reduce the worst shift."""
    n = h.size
    best_h = h.copy()
    corr = np.correlate(h, 1 - h, mode="full")
    best_max = int(corr[n - 1 :].max())
    no_improve = 0

    for _ in range(max_iters):
        if time.time() > deadline:
            break

        corr = np.correlate(h, 1 - h, mode="full")
        counts = corr[n - 1 :]
        cur_max = int(counts.max())
        if cur_max < best_max:
            best_max = cur_max
            best_h = h.copy()
            no_improve = 0
        if cur_max == 0:
            break

        k = int(np.argmax(counts))
        if k == 0:
            break

        # search for a swappable pair at shift k
        prefix = h[: n - k]
        suffix = h[k:]
        A = np.where((prefix == 1) & (suffix == 0))[0]
        B = np.where((prefix == 0) & (suffix == 1))[0]

        if A.size == 0 or B.size == 0:
            # try other shifts
            found = False
            for kk in range(1, n):
                prefix = h[: n - kk]
                suffix = h[kk:]
                A = np.where((prefix == 1) & (suffix == 0))[0]
                B = np.where((prefix == 0) & (suffix == 1))[0]
                if A.size and B.size:
                    k = kk
                    found = True
                    break
            if not found:
                break
        # perform a random swap
        a = rng.choice(A)
        b = rng.choice(B)
        h[a], h[b] = 0.0, 1.0
        new_corr = np.correlate(h, 1 - h, mode="full")
        new_max = int(new_corr[n - 1 :].max())
        if new_max <= cur_max:
            no_improve = 0
            continue
        # revert
        h[a], h[b] = 1.0, 0.0
        no_improve += 1
        if no_improve >= 50:
            break

    return best_h, compute_c5(best_h)

                    best_h, best_c, best_n = h_bin.copy(), c_bin, n
                h = h_bin  # continue refinement from improved binary

            # ---- SLSQP refinement (soft‑max) ----
            if time.time() - start_time > budget_s * 0.95:
                break
            for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0):
                if time.time() - start_time > budget_s * 0.97:
                    break
                max_iter = 300 if beta < 200 else 600
                try:
                    h = slsqp_optimize(h, beta, rng, max_iter=max_iter)
                except Exception:
                    continue
                c = compute_c5(h)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n

            # ---- gradient refinement ----
            if time.time() - start_time < budget_s * 0.99:
                h = gradient_refine(h, rng, step=0.02, max_iter=16)
                c = compute_c5(h)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n

    # final polish on the overall best
    if best_h is not None and time.time() - start_time < budget_s * 0.998:
        best_h = gradient_refine(best_h, rng, step=0.02, max_iter=20)

    if best_h is None:
        best_n = 120
        best_h = np.full(best_n, 0.5)

    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            i = rng.integers(0, best_n)
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()
            if abs(diff) < 1e-12:
                break

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
