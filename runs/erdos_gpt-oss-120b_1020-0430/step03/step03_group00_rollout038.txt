def project_to_sum_box(x, target):
    # Clip to [0,1]
    x = np.clip(x, 0, 1)
    # Compute diff
    diff = target - x.sum()
    # while diff not near zero:
    # find indices not at bounds (0 or 1)
    free = np.where((x > 0) & (x < 1))[0]
    if free.size == 0:
        # cannot adjust, just add/subtract to some index
        # pick random index and clamp
        i = np.random.randint(len(x))
        x[i] = np.clip(x[i] + diff, 0, 1)
        return x
    # Distribute diff uniformly among free indices
    # Might overshoot, so iteratively clamp.
    while abs(diff) > 1e-12 and free.size > 0:
        delta = diff / free.size
        # attempt to add delta to free entries
        new_vals = x[free] + delta
        # compute overflow: new_vals >1 => overshoot; new_vals<0 => undershoot
        over = new_vals > 1
        under = new_vals < 0
        # for over entries, set to 1, update diff
        if np.any(over):
            overshoot = new_vals[over] - 1
            diff -= np.sum(overshoot)
            x[free[over]] = 1
        elif np.any(under):
            undershoot = -new_vals[under]
            diff += np.sum(undershoot)
            x[free[under]] = 0
        else:
            x[free] += delta
            diff = target - x.sum()
            break
        free = np.where((x > 0) & (x < 1))[0]
    # if still diff left due to rounding, adjust a random index
    if abs(diff) > 1e-12:
        i = np.random.randint(len(x))
        x[i] = np.clip(x[i] + diff, 0, 1)
    return x

def enforce_sum(h, target, rng):
    # h should be in [0,1] but may have slight sum error
    # Clip to [0,1] first
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    # Find indices that can be adjusted
    if diff > 0:
        # need to increase
        idx = np.where(h < 1.0)[0]
        if idx.size == 0:
            # nothing can increase; just add to random index and clip
            i = rng.integers(0, h.shape[0])
            h[i] = min(1.0, h[i] + diff)
        else:
            # distribute diff evenly
            # but ensure we don't exceed 1
            # Use while loop
            while diff > 1e-14 and idx.size > 0:
                add = diff / idx.size
                # limit add to capacity
                cap = 1.0 - h[idx]
                if np.any(cap < add):
                    # adjust those at bound individually
                    over = cap < add
                    # set those to 1
                    h[idx[over]] = 1.0
                    diff -= np.sum(cap[over])
                    idx = idx[~over]
                else:
                    h[idx] += add
                    diff = target - h.sum()
                    break
    else:
        # need to decrease
        idx = np.where(h > 0.0)[0]
        if idx.size == 0:
            i = rng.integers(0, h.shape[0])
            h[i] = max(0.0, h[i] + diff)
        else:
            while diff < -1e-14 and idx.size > 0:
                sub = (-diff) / idx.size
                cap = h[idx]
                if np.any(cap < sub):
                    over = cap < sub
                    h[idx[over]] = 0.0
                    diff += np.sum(cap[over])
                    idx = idx[~over]
                else:
                    h[idx] -= sub
                    diff = target - h.sum()
                    break
    # final correction
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        i = rng.integers(0, h.shape[0])
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def transfer_move(h, rng, max_frac=0.2):
    n = len(h)
    i = rng.integers(0, n)
    j = rng.integers(0, n-1)
    if j >= i:
        j += 1
    # donor i, receiver j
    if h[i] <= 0 or h[j] >= 1:
        # try again recursively limited attempts
        return h.copy()
    max_amount = min(h[i], 1.0 - h[j])
    if max_amount <= 0:
        return h.copy()
    delta = max_amount * max_frac * rng.random()
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h

def worst_shift_gradient(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    k = idx - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        # first part: (1 - h[i+k]) for i from 0..n-1-k
        grad[:n - k] += 1 - h[k:]   # shape (n-k)
        # second part: -h[i - k] for i from k..n-1
        grad[k:] -= h[:n - k]
    else:

        # i+k out of range? i+ (-kk) => i-kk
        grad[kk:] += 1 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad, k

    j = rng.choice(neg_idx)

    new_h[j] += delta

def simulated_annealing_time(h_start, rng, end_time, init_temp=0.02):

    best_h = h.copy()

def run(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    # Initialize h0
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n0 = 360  # default
        h0 = random_feasible_vector(n0, rng)
    # ensure valid
    n0 = h0.size
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = enforce_sum(h0, n0/2.0, rng)
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    # allocate time fractions
    total_budget = budget_s
    # Phase 1: SA restarts (coarse)
    phase1_end = start + total_budget * 0.55
    while time.time() < phase1_end:
        # pick start vector
        if rng.random() < 0.5:
            h_start = best_h.copy()
            # small perturb to avoid staying same
            h_start = transfer_move(h_start, rng, max_frac=0.3)
        else:
            h_start = random_feasible_vector(n0, rng)
        # run SA for about 2-3 seconds
        # compute remaining time for this iteration (but do not exceed phase1 end)
        sa_end = min(time.time() + 2.5, phase1_end)
        cand_h, cand_c = simulated_annealing_time(h_start, rng, sa_end)
        # refine with hill climb for small time (0.5 sec)
        hc_end = min(time.time() + 0.5, phase1_end)
        cand_h2, cand_c2 = hill_climb_targeted_time(cand_h, rng, hc_end)
        # choose best among cand_h2 (maybe improved) and cand_h
        if cand_c2 < cand_c:
            cand_h, cand_c = cand_h2, cand_c2
        if cand_c < best_c:
            best_h, best_c = cand_h.copy(), cand_c
    # Phase 2: Targeted hill climb on best coarse solution (more time)
    phase2_end = start + total_budget * 0.75
    if time.time() < phase2_end:
        hc_end = phase2_end
        best_h, best_c = hill_climb_targeted_time(best_h, rng, hc_end)
    # Phase 3: Upsample and refine fine solution
    phase3_end = start + total_budget * 0.95
    if time.time() < phase3_end:
        # upsample factor 2
        n_fine = n0 * 2
        xs0 = (np.arange(n0) + 0.5) * (2.0 / n0)

        # SA on fine

            fine_h, fine_c = fine_h2, fine_c2
        if fine_c < best_c:
            best_h, best_c = fine_h, fine_c
    # final cleanup
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)

def slsqp_refine(h, rng, betas=[5,20,80,200,800,2000], max_iter=500):
    from scipy.optimize import minimize
    n = h.size
    target = n/2.0
    # We'll define global variables like before or use closures; but we can avoid global.
    def sum_eq(x):
        return np.sum(x) - target
    # We'll use local variables for beta
    best_x = h.copy()
    best_c = compute_c5(best_x)
    for beta in betas:
        def soft_obj(x):
            corr = np.correlate(x, 1 - x, mode='full')
            bc = beta * corr
            m = np.max(bc)
            # log-sum-exp
            return (m + np.log(np.sum(np.exp(bc - m)))) / beta
        bounds = [(0.0, 1.0)] * n
        cons = {'type':'eq', 'fun': sum_eq}
        res = minimize(soft_obj, best_x, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol':1e-12, 'disp':False})
        x = np.clip(res.x, 0.0, 1.0)
        # enforce sum just in case
        x = enforce_sum(x, target, rng)
        c = compute_c5(x)
        if c < best_c:
            best_c = c
            best_x = x
    return best_x, best_c

For i >= n-k, out-of-range second factor considered zero since h[i+k] missing, so term = 0? Actually np.correlate with mode='full' uses zero padding: For shift k positive, correlation vector index offset includes zero contributions beyond boundaries. So for i > n-1-k, term = 0. So gradient for i beyond n-1-k does not have the first part (1 - h[i+k]) = 0, but also the second part maybe for i being the second factor of some other term? For i >= k, term where i = j+k -> j = i-k, term includes -h[j]; so gradient for i includes -h[i-k] when i >= k. For i < k, no second part.

Now we need to ensure gradient expression for k negative (i.e., shift left) correct. Using kk = -k. For k negative, the terms are h[i] * (1 - h[i+k]) = h[i] * (1 - h[i-kk]) where i-kk may be >=0 else out of range => treat as zero. This quantity defined for i>=kk, else zero.

Thus gradient for i:

- first part: (1 - h[i-kk]) for i>=kk else 0.

- second part: -h[i+kk] for i+kk < n (i <= n-1-kk) else 0.

import time, math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function h on [0,2] with uniform grid."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Clip to [0,1] and adjust so that sum(h) == target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
            if idx.size == 0:
                break
            add = diff / idx.size
            cap = 1.0 - h[idx]
            over = cap < add
            if np.any(over):
                h[idx[over]] = 1.0
                diff = target - h.sum()
                continue
            h[idx] += add
            diff = target - h.sum()
            break
        else:
            idx = np.where(h > 0.0)[0]
            if idx.size == 0:
                break
            sub = (-diff) / idx.size
            cap = h[idx]
            under = cap < sub
            if np.any(under):
                h[idx[under]] = 0.0
                diff = target - h.sum()
                continue
            h[idx] -= sub
            diff = target - h.sum()
            break
    # tiny correction
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.shape[0])
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector in [0,1]^n with exact sum n/2."""
    h = rng.random(n)
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, n / 2.0, rng)
    return h


def transfer_move(h: np.ndarray, rng: np.random.Generator, max_frac: float = 0.3) -> np.ndarray:
    """Move a small amount from one coordinate to another, preserving sum."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if j >= i:
        j += 1
    if h[i] <= 0.0 or h[j] >= 1.0:
        return h.copy()
    max_amount = min(h[i], 1.0 - h[j])
    if max_amount <= 0.0:
        return h.copy()
    delta = max_amount * max_frac * rng.random()
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h


def worst_shift_gradient(h: np.ndarray):
    """Gradient of the worst (max) correlation shift."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    k = idx - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        grad[: n - k] += 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[: n - kk]
        grad[: n - kk] -= h[kk:]
    return grad, k


def targeted_transfer(h: np.ndarray, grad: np.ndarray, rng: np.random.Generator,
                      max_frac: float = 0.3) -> np.ndarray:
    """Move mass from positive‑gradient (donor) to negative‑gradient (receiver) entries."""
    pos = np.where((grad > 1e-9) & (h > 1e-12))[0]
    neg = np.where((grad < -1e-9) & (h < 1.0 - 1e-12))[0]
    if pos.size == 0 or neg.size == 0:
        return transfer_move(h, rng, max_frac)
    i = rng.choice(pos)
    j = rng.choice(neg)
    max_amount = min(h[i], 1.0 - h[j])
    if max_amount <= 0.0:
        return transfer_move(h, rng, max_frac)
    delta = max_amount * max_frac * rng.random()
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h


def hill_climb_targeted_time(h_start: np.ndarray, rng: np.random.Generator,
                             end_time: float) -> (np.ndarray, float):
    """Greedy targeted hill‑climb until a wall‑clock deadline."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = 0.3
    it = 0
    while time.time() < end_time:
        grad, _ = worst_shift_gradient(h)
        new_h = targeted_transfer(h, grad, rng, max_frac=step)
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
        if new_c < compute_c5(h):
            h = new_h
        it += 1
        if it % 500 == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def simulated_annealing_time(h_start: np.ndarray, rng: np.random.Generator,
                             end_time: float, init_temp: float = 0.02) -> (np.ndarray, float):
    """Simple SA with continuous mass transfers, respecting a deadline."""
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    T = init_temp
    alpha = 0.9997
    while time.time() < end_time:
        new_h = transfer_move(h, rng, max_frac=0.3)
        new_c = compute_c5(new_h)
        d = new_c - cur_c
        if d < 0.0 or rng.random() < math.exp(-d / T):
            h = new_h
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()
        T = max(T * alpha, 1e-7)
    return best_h, best_c


def binary_annealing_time(h_start: np.ndarray, rng: np.random.Generator,
                          end_time: float, init_temp: float = 0.02) -> (np.ndarray, float):
    """SA on a 0‑1 vector (swap moves), with optional gradient‑guided swaps."""
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    T = init_temp
    alpha = 0.9995
    while time.time() < end_time:
        grad, _ = worst_shift_gradient(h)
        donors = np.where((h > 0.5) & (grad > 1e-9))[0]
        recips = np.where((h < 0.5) & (grad < -1e-9))[0]
        if donors.size and recips.size and rng.random() < 0.7:
            i = rng.choice(donors)
            j = rng.choice(recips)
        else:
            zeros = np.where(h < 0.5)[0]
            ones = np.where(h > 0.5)[0]
            i = rng.choice(zeros)
            j = rng.choice(ones)
        new_h = h.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]
        new_c = compute_c5(new_h)
        d = new_c - cur_c
        if d < 0.0 or rng.random() < math.exp(-d / T):
            h = new_h
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()
        T = max(T * alpha, 1e-7)
    return best_h, best_c


def slsqp_refine(h: np.ndarray, rng: np.random.Generator,
                 betas=(5, 20, 80, 200, 800, 2000), max_iter=400) -> (np.ndarray, float):
    """Final smooth‑max SLSQP refinement on the current h."""
    n = h.size
    target = n / 2.0

    def sum_eq(x):
        return np.sum(x) - target

    best_h = h.copy()
    best_c = compute_c5(best_h)

    for beta in betas:
        def soft_obj(x):
            corr = np.correlate(x, 1.0 - x, mode="full")
            bc = beta * corr
            m = np.max(bc)
            return (m + np.log(np.sum(np.exp(bc - m)))) / beta

        bounds = [(0.0, 1.0)] * n
        cons = {"type": "eq", "fun": sum_eq}
        try:
            res = minimize(soft_obj, best_h, method="SLSQP", bounds=bounds,
                           constraints=cons,
                           options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
        except Exception:
            continue
        x = np.clip(res.x, 0.0, 1.0)
        x = enforce_sum(x, target, rng)
        c = compute_c5(x)
        if c < best_c:
            best_c = c
            best_h = x
    return best_h, best_c

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    total_budget = budget_s

    # ---- initial vector -------------------------------------------------
    if "initial_h_values" in globals():

        hc_end = min(time.time() + 0.5, phase1_end)
        cand_h, cand_c = hill_climb_targeted_time(cand_h, rng, hc_end)

        if cand_c < best_c:
            best_h, best_c = cand_h.copy(), cand_c

    # ---- Phase 2: extended targeted hill‑climb -------------------------
    phase2_end = start_time + total_budget * 0.75
    if time.time() < phase2_end:
        best_h, best_c = hill_climb_targeted_time(best_h, rng, phase2_end)

    # ---- Phase 3: binary optimisation (if time permits) ---------------
    remaining = start_time + total_budget - time.time()
    if remaining > 30.0:
        bin_time = min(remaining - 10.0, 30.0)
        # create a binary version of the current best
        order = np.argsort(-best_h)
        bin_h = np.zeros(n0, dtype=float)
        bin_h[order[: n0 // 2]] = 1.0
        bin_h = enforce_sum(bin_h, n0 / 2.0, rng)

        bin_end = time.time() + bin_time
        bin_h, bin_c = binary_annealing_time(bin_h, rng, bin_end)
        if bin_c < best_c:
            best_h, best_c = bin_h, bin_c

    # ---- Phase 4: up‑sample to finer grid and refine --------------------
    remaining = start_time + total_budget - time.time()
    if remaining > 20.0:
        n_fine = n0 * 2
        xs0 = (np.arange(n0) + 0.5) * (2.0 / n0)
        xs = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
        h_fine = np.interp(xs, xs0, best_h)
        h_fine = enforce_sum(h_fine, n_fine / 2.0, rng)

        # SA on fine grid (about 60% of remaining time)
        sa_fine_end = time.time() + remaining * 0.55
        fine_h, fine_c = simulated_annealing_time(h_fine, rng, sa_fine_end)

        # targeted hill‑climb on fine grid (remaining time)
        hc_fine_end = start_time + total_budget
        fine_h, fine_c = hill_climb_targeted_time(fine_h, rng, hc_fine_end)

        # final SLSQP refinement if any time left
        if time.time() < start_time + total_budget:
            fine_h, fine_c = slsqp_refine(fine_h, rng)

        if fine_c < best_c:
            best_h, best_c = fine_h, fine_c

    # ---- final clean‑up -------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
