import time
import numpy as np
from scipy.optimize import minimize

# globals for the surrogate optimizer
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """Smooth max of the correlation using softmax with temperature 1/β."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def compute_c5(h):
    """Exact discrete C5 for step function h on [0,2] (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h, target, rng, max_iter=5000):
    """Project h onto the simplex Σh = target with bounds [0,1]."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        # small uniform correction
        corr = diff / n
        h += corr
        h = np.clip(h, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h

def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (keeps the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb_multi(h_start, rng, max_iters=2000, init_step=0.5, n_trials=5):
    """Greedy hill‑climb with several random trials per iteration."""
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            cand = perturb_h(h, rng, scale=step)
            c = compute_c5(cand)
            if c < best_c:
                best_c, h = c, cand
                improved = True
                break
        if not improved and rng.random() < 0.02:
            cand = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(cand)
            if c < best_c:
                best_c, h = c, cand
                improved = True
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)
        # early stop if no decent improvements for a while
        if not improved and it % decay == 0 and step < 1e-5:
            break
    return h, best_c

def slsqp_optimize(h0, beta, max_iter=400):
    """Run SLSQP on the smooth surrogate with given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x

def upsample(h, new_n, rng):
    old_n = h.size
    old_x = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    new_x = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    h_up = np.interp(new_x, old_x, h)
    h_up = enforce_sum(h_up, new_n / 2.0, rng)
    return h_up

            h_opt, c_opt = hill_climb_multi(start_vec, rng,
                                            max_iters=fine_hc_iters,
                                            init_step=0.4,
                                            n_trials=10)
            if c_opt < global_best_c:
                global_best_h, global_best_c = h_opt.copy(), c_opt
                h_fine = h_opt.copy()
            # occasional SLSQP refinement
            for beta in beta_list[-4:]:
                if time.time() >= end_time:
                    break
                try:
                    h_ref = slsqp_optimize(h_fine, beta, max_iter=600)
                except Exception:
                    continue
                c_ref = compute_c5(h_ref)
                if c_ref < global_best_c:
                    global_best_h, global_best_c = h_ref.copy(), c_ref
                    h_fine = h_ref.copy()

    # -----------------------------------------------------------------
    # 6. Final projection and exact sum enforcement
    # -----------------------------------------------------------------
    global_best_h = np.clip(global_best_h, 0.0, 1.0)
    global_best_h = enforce_sum(global_best_h, n / 2.0, rng)
    final_c = compute_c5(global_best_h)

    return global_best_h.tolist(), float(final_c), int(n)
