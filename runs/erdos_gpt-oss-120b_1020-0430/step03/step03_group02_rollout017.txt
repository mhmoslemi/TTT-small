    c = compute_c5(h)
    best_h, best_c = h.copy(), c
    T = init_temp
    for it in range(max_iter):

        # propose move:

def binary_local_search(h, rng, max_iters=5000):
    n=len(h)
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        if k == 0:
            break
        # find indices i where i < n - k and h[i]==1 and h[i+k]==0
        i_candidates = np.where((h[:n-k]==1) & (h[k:]==0))[0]
        j_candidates = np.where((h[:n-k]==0) & (h[k:]==1))[0]
        if i_candidates.size == 0 or j_candidates.size == 0:
            # no swap can reduce this shift; maybe try random swap
            i = rng.integers(0,n)
            j = rng.integers(0,n)
            if h[i]==h[j]:
                continue
        else:
            i = rng.choice(i_candidates)
            j = rng.choice(j_candidates)
        nh = h.copy()
        nh[i], nh[j] = nh[j], nh[i]
        nc = compute_c5(nh)
        if nc < best_c - 1e-12:
            best_h, best_c = nh.copy(), nc
        if nc <= best_c + 1e-12:
            h = nh
        # else maybe accept with small probability to escape

def to_binary(h):
    n = len(h)
    indices = np.argsort(-h)  # descending
    binary = np.zeros(n, dtype=float)
    binary[indices[:n//2]] = 1.0
    return binary

def random_perturb(h, rng, scale=0.1):
   nh = h.copy()
   for _ in range(int(h.size * scale)):
       i = rng.integers(0, h.size)
       max_eps = min(nh[i], 1 - nh[i])
       eps = rng.random() * max_eps * 0.1
       # randomly move to another index
       j = rng.integers(0, h.size)
       if i == j: continue
       # adjust both
       delta = min(eps, max_eps)
       nh[i] -= delta
       nh[j] += delta
   enforce_sum(nh, rng)
   return nh

import time, numpy as np
from scipy.optimize import minimize

# globals for the SLSQP soft‑max formulation
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    """Discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """Smooth approximation of max(corr) using β."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    """Soft‑max SLSQP optimisation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def enforce_sum(h, rng):
    """Enforce Σh = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    """Random feasible h (0≤h≤1, Σh=n/2)."""
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

def worst_gradient(h):
    """Gradient of the worst‑shift overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    """Transfer ε from a positive‑gradient entry to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h_start, rng, max_iters=2500, step_scale=0.5):
    """Hill‑climb using gradient‑directed transfers."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

def gradient_refine(h, rng, step=0.02, max_iter=8):
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        enforce_sum(h, rng)
    return h

def to_binary(h):
    """Round to a {0,1} vector with exact sum."""
    n = h.size
    idx = np.argsort(-h)[: n // 2]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    return b

def binary_local_search(h, rng, max_iters=2000):
    """Swap‑based search for binary vectors."""
    n = h.size
    best_h, best_c = h.copy(), compute_c5(h)
    h_cur = h.copy()
    for it in range(max_iters):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        if k == 0:
            break
        # candidate 1s that pair with a 0 at distance k
        A = np.where((h_cur[: n - k] == 1) & (h_cur[k:] == 0))[0]
        B = np.where((h_cur[: n - k] == 0) & (h_cur[k:] == 1))[0]
        if A.size == 0 or B.size == 0:
            break
        i = rng.choice(A)
        j = rng.choice(B)
        nh = h_cur.copy()
        nh[i], nh[j] = nh[j], nh[i]
        nc = compute_c5(nh)
        if nc < best_c - 1e-12:
            best_h, best_c = nh.copy(), nc
            h_cur = nh
        elif nc <= best_c + 1e-12:
            h_cur = nh
    return best_h, best_c
