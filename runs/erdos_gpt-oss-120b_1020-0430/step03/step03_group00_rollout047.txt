def simulated_annealing(h, time_budget, rng):
    n = len(h)
    # compute current C5
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c

    start = time.time()
    # SA schedule parameters
    T0 = 0.05   # set based on typical diff scale
    T_min = 1e-5
    max_iter = 10**9  # effectively unlimited
    # We'll use exponential cooling based on iteration count
    # Determine total iterations based on time: approximate target iter count = time_budget / est_time_per_iter
    # We'll just loop until time runs out.

    iter = 0
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        # temperature schedule based on elapsed ratio
        frac = elapsed / time_budget
        T = T0 * ((T_min / T0) ** frac)  # exponential cooling over full budget

        # determine move type probabilities
        if rng.random() < 0.7:
            # targeted move if possible
            new_h, new_c = perform_targeted_move(h, cur_c, rng, T)
        else:
            # random move
            new_h, new_c = perform_random_move(h, cur_c, rng, T)

        # accept
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-12)):
            h = new_h
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        iter += 1

    return best_h, best_c

def perform_targeted_move(h, cur_c, rng, T):
    n = len(h)
    # compute gradient of worst shift
    idx = np.argmax(np.correlate(h, 1 - h, mode='full'))
    k = idx - (n - 1)
    # compute grad arrays via vectorization
    i_arr = np.arange(n)
    grad = np.zeros(n)
    valid1 = (i_arr + k >= 0) & (i_arr + k < n)
    grad[valid1] += 1 - h[i_arr[valid1] + k]
    valid2 = (i_arr - k >= 0) & (i_arr - k < n)
    grad[valid2] -= h[i_arr[valid2] - k]

    pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
    neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
    if pos.size == 0 or neg.size == 0:
        return perform_random_move(h, cur_c, rng, T)
    i = rng.choice(pos)
    j = rng.choice(neg)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perform_random_move(h, cur_c, rng, T)
    step_factor = max(0.01, min(0.5, T * 10))
    eps = max_eps * step_factor * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    new_c = compute_c5(new_h)
    return new_h, new_c

def perform_random_move(h, cur_c, rng, T):

def hill_climb(h, max_iter, rng):
    n = len(h)
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_h = h.copy()
    cur_c = best_c
    for it in range(max_iter):
        # compute gradient for worst shift
        corr = np.correlate(cur_h, 1 - cur_h, mode='full')
        idx = np.argmax(corr)
        k = idx - (n - 1)
        # compute gradient as above
        i_arr = np.arange(n)
        grad = np.zeros(n)
        valid1 = (i_arr + k >= 0) & (i_arr + k < n)
        grad[valid1] += 1 - cur_h[i_arr[valid1] + k]
        valid2 = (i_arr - k >= 0) & (i_arr - k < n)
        grad[valid2] -= cur_h[i_arr[valid2] - k]
        # choose pos, neg
        pos = np.where((grad > 1e-12) & (cur_h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (cur_h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            # random small move
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            max_eps = min(cur_h[i], 1.0 - cur_h[j])
            if max_eps <= 0:
                continue
            eps = max_eps * 0.1 * rng.random()
        else:
            i = rng.choice(pos)
            j = rng.choice(neg)
            max_eps = min(cur_h[i], 1.0 - cur_h[j])
            if max_eps <= 0:
                continue
            eps = max_eps * 0.1 * rng.random()

        new_h[j] += eps
        new_c = compute_c5(new_h)
        if new_c < cur_c:
            cur_h = new_h
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()
        # optional: occasionally random restart? Not needed
    return best_h, best_c

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for vector h (sum(h)=len/2)."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / h.size)


def worst_shift_and_grad(h: np.ndarray):
    """Return worst shift index k and its gradient w.r.t. h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    k = idx - (n - 1)
    i = np.arange(n)
    grad = np.zeros(n)
    # (1 - h[i+k]) term where i+k in range
    v1 = (i + k >= 0) & (i + k < n)
    grad[v1] += 1.0 - h[i[v1] + k]
    # -h[i-k] term where i-k in range
    v2 = (i - k >= 0) & (i - k < n)
    grad[v2] -= h[i[v2] - k]
    return k, grad


def perform_random_move(h: np.ndarray, T: float, rng: np.random.Generator):
    n = h.size
    # try a few times to find a feasible pair
    for _ in range(5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        step = max(0.01, min(0.5, T * 10.0))

        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        return new_h, compute_c5(new_h)

    return h, compute_c5(h)

def simulated_annealing(start_h: np.ndarray, time_budget: float, rng: np.random.Generator):
    """Simple SA that preserves sum and box constraints."""
    start = time.time()
    cur_h = start_h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    T0 = 0.2
    T_min = 1e-5
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        frac = elapsed / time_budget
        T = T0 * ((T_min / T0) ** frac)  # exponential cooling
        # 70% targeted, otherwise random
        if rng.random() < 0.7:
            new_h, new_c = perform_targeted_move(cur_h, T, rng)
        else:
            new_h, new_c = perform_random_move(cur_h, T, rng)

        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-12)):
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
    return best_h, best_c

def enforce_sum_exact(h: np.ndarray, target: float, rng: np.random.Generator):

    h = np.clip(h, 0.0, 1.0)

            diff -= add

        remaining_stages = [sz for sz in stages if sz >= n]
        sz_sum = sum(remaining_stages)
        stage_budget = remaining * (n / sz_sum)

        # simulated annealing on this resolution
        h_sa, c_sa = simulated_annealing(best_h, stage_budget, rng)
        if c_sa < best_c:
            best_c, best_h = c_sa, h_sa.copy()

        # quick hill‑climb refinement (few thousand steps)
        if time.time() - start_time < total_search_budget:
            h_hc, c_hc = hill_climb(best_h, max_iter=5000, rng=rng)
            if c_hc < best_c:
                best_c, best_h = c_hc, h_hc.copy()

    # ---- final random fine‑tuning ---------------------------------------
    while time.time() - start_time < budget_s * 0.99:
        n = best_h.size
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0.0:
            continue
        eps = max_eps * 0.02 * rng.random()
        new_h = best_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c_new = compute_c5(new_h)
        if c_new < best_c:
            best_c, best_h = c_new, new_h.copy()

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum_exact(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
