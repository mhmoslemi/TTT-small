def project_onto_cap_simplex(v, target):
    L = np.min(v) - 1.0
    U = np.max(v)
    # binary loop 50 iters
    for _ in range(60):
        lam = (L + U) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            L = lam
        else:
            U = lam
    return np.clip(v - (L+U)/2.0, 0.0, 1.0)

def random_transfer(h, rng, scale=0.1):
    n = h.size
    # pick i (source) where h>0, j (dest) where h<1
    # We'll draw i from indices with h>1e-12, j from indices with h<1-1e-12
    # If none, just return copy.
    source_idxs = np.where(h > 1e-12)[0]
    dest_idxs = np.where(h < 1 - 1e-12)[0]
    if source_idxs.size == 0 or dest_idxs.size == 0:
        return h
    i = rng.choice(source_idxs)
    j = rng.choice(dest_idxs)
    max_eps = min(h[i], 1.0 - h[j])
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    # sum unchanged, bounds kept.
    return nh

def slsqp_refine(h, rng, max_iter=400):
    n = h.size
    target = n/2

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        # Use softmax with moderate beta to approximate max
        beta = 1000.0
        # compute softmax of corr*beta, but we need objective to approximate max of corr*dx? Actually we want to minimize max corr*dx, same as minimize max corr. So objective = softmax of corr.
        bc = corr * beta
        m = np.max(bc)
        # compute softmax value ~ (log sum exp)/beta
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    bounds = [(0.0, 1.0)] * n
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}

    res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraints,
                   options={'maxiter': max_iter, 'ftol': 1e-12})
    x = np.clip(res.x, 0.0, 1.0)
    # final projection adjust sum exactly
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = project_onto_cap_simplex(x, target)
    return x

def min_corr_gradient(h):
    n = h.size
    idx = np.argmin(np.correlate(h, h, mode='full'))
    k = idx - (n - 1)
    if k == 0:
        g = 2.0 * h
    else:
        k_abs = abs(k)
        g = np.zeros_like(h)
        # for shift positive (k>0)
        if k > 0:
            g[:n - k] += h[k:]
            g[k:] += h[:n - k]
        else:
            # k negative => shift left
            k_abs = -k
            g[:n - k_abs] += h[k_abs:]
            g[k_abs:] += h[:n - k_abs]
    return g

import time
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
def project_onto_cap_simplex(v, target):
    """Euclidean projection of v onto {0≤x≤1, sum x = target}."""
    L = np.min(v) - 1.0
    U = np.max(v)
    for _ in range(60):
        lam = (L + U) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            L = lam
        else:
            U = lam
    lam = (L + U) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = np.where((x > 0.0) & (x < 1.0))[0]
        if free.size:
            x[free] += diff / free.size
            x = np.clip(x, 0.0, 1.0)
    return x


# ------------------------------------------------------------
def compute_c5_from_h(h):
    """Exact discrete C5 for a step function on [0,2]."""
    n = h.size
    corr = np.correlate(h, h, mode="full")
    min_corr = np.min(corr)
    return 1.0 - (2.0 / n) * min_corr


# ------------------------------------------------------------
def min_corr_gradient(h):
    """Gradient of the minimum auto‑correlation Σ h_i h_{i+k}."""
    n = h.size
    idx = np.argmin(np.correlate(h, h, mode="full"))
    k = idx - (n - 1)
    if k == 0:
        return 2.0 * h
    g = np.zeros_like(h)
    if k > 0:
        g[: n - k] += h[k:]
        g[k:] += h[: n - k]
    else:  # k < 0
        k = -k
        g[: n - k] += h[k:]
        g[k:] += h[: n - k]
    return g


# ------------------------------------------------------------
def ascent_optimize(h0, rng, max_iter=1500, step_start=0.02, step_decay=0.997):
    """Projected gradient ascent on the minimum correlation."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    step = step_start
    best_h = h.copy()
    best_c = compute_c5_from_h(h)
    for _ in range(max_iter):
        g = min_corr_gradient(h)
        h_new = h + step * g
        h_new = project_onto_cap_simplex(h_new, target)
        c = compute_c5_from_h(h_new)
        if c < best_c:
            best_c, best_h, h = c, h_new.copy(), h_new
        step *= step_decay
    return best_h, best_c


# ------------------------------------------------------------
def random_feasible_vector(n, rng):
    """Uniform random vector projected onto the feasible set."""
    v = rng.random(n)
    return project_onto_cap_simplex(v, n / 2.0)


# ------------------------------------------------------------
def random_transfer(h, rng, scale=0.05):
    """Move a small amount of mass from a positive entry to a negative one."""
    pos = np.where(h > 1e-12)[0]
    neg = np.where(h < 1.0 - 1e-12)[0]
    if pos.size == 0 or neg.size == 0:
        return h.copy()
    i = rng.choice(pos)
    j = rng.choice(neg)
    max_eps = min(h[i], 1.0 - h[j])
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


# ------------------------------------------------------------
def penalized_objective(x, beta, target, penalty):
    """Soft‑max approximation of max_{k}∫h(1−h shifted) + penalty·(sum−target)²."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = corr * beta
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta + penalty * (np.sum(x) - target) ** 2


# ------------------------------------------------------------
def slsqp_refine(h, betas=(10, 50, 200, 1000, 5000), max_iter=150, penalty=1000.0):
    """SLSQP refinement using a soft‑max proxy for the max‑overlap."""
    n = h.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    for beta in betas:
        res = minimize(
            penalized_objective,
            h,
            args=(beta, target, penalty),
            method="SLSQP",
            bounds=bounds,
            options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
        )
        h = np.clip(res.x, 0.0, 1.0)
        h = project_onto_cap_simplex(h, target)
    return h
