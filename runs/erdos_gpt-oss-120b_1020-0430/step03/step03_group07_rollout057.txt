def enforce_sum(h, target, rng):
    """Adjust h so that Σh == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = diff if diff > 0 else diff
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h = enforce_sum(h, target, rng)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def compute_obj_and_grad(x):
    """Return smooth‑max objective and its gradient."""
    n = x.size
    # correlation of h with (1‑h)
    corr = correlate_full(x)                     # length 2n‑1

        g += 2.0 * CURRENT_LAMBDA * diff * np.ones_like(x)
    return f, g


def slsqp_optimize_grad(h0, beta, lam, max_iter=200):
    """Run SLSQP on the smooth‑max surrogate with analytical gradient."""
    global CURRENT_BETA, CURRENT_TARGET, CURRENT_LAMBDA
    n = h0.size
    CURRENT_BETA = beta
    CURRENT_TARGET = n / 2.0
    CURRENT_LAMBDA = lam

    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": eq_constraint, "jac": eq_jac}
    res = minimize(
        obj_fun,
        h0,
        method="SLSQP",
        jac=grad_fun,
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # final sum repair (rare)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        x = enforce_sum(x, CURRENT_TARGET, rng)
    return x


# ----------------------------------------------------------------------
# Local optimisation primitives
# ----------------------------------------------------------------------
def hill_climb(h_start, rng, max_iters=2000, start_step=0.4):
    """Greedy hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


def simulated_annealing(start_h, rng, deadline, T0=0.02, Tf=1e-5, max_iters=1500):
    """Simple exponential‑cooling SA."""
    cur_h = start_h.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        new_h = perturb_h(cur_h, rng, scale=0.35)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
    return best_h, best_c


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.96  # reserve a few seconds for cleanup

    # --------------------------------------------------------------
    # 1. Initial feasible vector
    # --------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 360
        h0 = np.full(n, 0.5)
        h0 = enforce_sum(h0, n / 2.0, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # --------------------------------------------------------------
    # 2. Parameters for continuation SLSQP
    # --------------------------------------------------------------
    beta_seq = [5, 20, 80, 200, 800, 2000]
    lam = 0.0                 # we rely on the equality constraint
    max_iter_per_beta = 200   # enough for convergence at each stage

    # --------------------------------------------------------------
    # 3. Main optimisation loop (random restarts + refinements)
    # --------------------------------------------------------------
    while time.time() < deadline:
        # starting point for this trial
        if rng.random() < 0.6:
            start = perturb_h(best_h, rng, scale=0.3)
        else:
            start = random_feasible_vector(n, rng)

        # continuation SLSQP
        cur = start.copy()
        for beta in beta_seq:
            if time.time() >= deadline - 5:
                break
            try:
                cur = slsqp_optimize_grad(cur, beta, lam, max_iter=max_iter_per_beta)
                # enforce exact sum after each stage
                cur = enforce_sum(cur, n / 2.0, rng)
            except Exception:
                pass

        # evaluate
        cur_c = compute_c5(cur)
        if cur_c < best_c:
            best_h, best_c = cur.copy(), cur_c

        # local hill‑climb refinement
        if time.time() < deadline:
            h_hc, c_hc = hill_climb(best_h, rng, max_iters=1500, start_step=0.35)
            if c_hc < best_c:
                best_h, best_c = h_hc.copy(), c_hc

        # occasional simulated annealing
        if rng.random() < 0.2 and time.time() < deadline:
            h_sa, c_sa = simulated_annealing(best_h, rng, deadline, T0=0.02, Tf=1e-5, max_iters=1200)
            if c_sa < best_c:
                best_h, best_c = h_sa.copy(), c_sa

        # up‑sample once we have a reasonably good candidate and still have time
        if n < 720 and time.time() < deadline - 30:
            h_up = np.repeat(best_h, 2)
            n_up = h_up.size
            h_up = enforce_sum(h_up, n_up / 2.0, rng)
            # repeat continuation on finer grid
            cur_up = h_up.copy()
            for beta in beta_seq:
                if time.time() >= deadline - 5:
                    break
                try:
                    cur_up = slsqp_optimize_grad(cur_up, beta, lam, max_iter=150)
                    cur_up = enforce_sum(cur_up, n_up / 2.0, rng)
                except Exception:
                    pass
            c_up = compute_c5(cur_up)
            if c_up < best_c:
                best_h, best_c, n = cur_up.copy(), c_up, n_up
                # single hill‑climb on the finer grid
                h_hc2, c_hc2 = hill_climb(best_h, rng, max_iters=1500, start_step=0.3)
                if c_hc2 < best_c:
                    best_h, best_c = h_hc2.copy(), c_hc2
            # after up‑sampling we keep the finer resolution for the rest of the run
            # (no further down‑sampling is performed)

    # --------------------------------------------------------------
    # 4. Final clean‑up and return
    # --------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
