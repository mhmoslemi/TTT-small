def binary_hill_climb(h0, rng, max_swaps):
    n = len(h0)
    ones = np.flatnonzero(h0 == 1)
    zeros = np.flatnonzero(h0 == 0)
    h = h0.copy()

    best_h = h.copy()
    no_improve = 0
    for s in range(max_swaps):
        # random 1 and 0
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0
        h[j] = 1
        # update lists
        # swap positions in arrays: we can just recompute ones and zeros each time? O(n). Acceptable for n=300 and 10k loops.
        # Actually we can update: ones = np.where(h==1)[0], zeros = np.where(h==0)[0] each iteration cost 2*n ~600; times 10k ~6M; fine.
        c = compute_c5(h)
        if c < best_c:

            best_h = h.copy()

            ones = np.flatnonzero(h == 1)

    for _ in range(60):
        mid = (lo+hi)/2
        s = np.sum(np.clip(v+mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo+hi)/2, 0.0, 1.0)

def binary_hill_climb(h0, rng, max_swaps):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    # Determine set of indices of ones and zeros for fast selection
    ones = np.flatnonzero(h == 1.0)
    zeros = np.flatnonzero(h == 0.0)
    no_improve = 0
    for _ in range(max_swaps):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        # update lists for next iteration: we can swap indices in the list
        # remove i from ones, add to zeros; similarly for j.
        # Efficient update:
        # replace i in ones with j (since after swap, j becomes 1)
        # and replace j in zeros with i (since after swap, i becomes 0)
        # But we need indices' positions in arrays; easier to recompute ones/zeros each iteration as small cost.
        # We'll recompute after acceptance or revert.
        c = compute_c5(h)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            # Update ones and zeros with new h
            ones = np.flatnonzero(h == 1.0)
            zeros = np.flatnonzero(h == 0.0)
            no_improve = 0
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
        if no_improve >= 2000:
            # reset to best
            h = best_h.copy()
            ones = np.flatnonzero(h == 1.0)
            zeros = np.flatnonzero(h == 0.0)
            no_improve = 0
    return best_h, best_c

def weighted_gradient(h, w):
    n = h.size
    w_rev = w[::-1]
    omhr = (1.0 - h)[::-1]
    conv1 = np.convolve(omhr, w_rev)
    conv2 = np.convolve(h, w_rev)
    idx = np.arange(n)
    term1 = conv1[2*n-2 - idx]
    term2 = conv2[n-1 + idx]
    return term1 - term2

def gd_optimize(h0, rng, phases):
    # phases is list of tuples (num_iter, beta, step)
    h = h0.copy()
    n = h.size
    target = n / 2.0
    for num_iter, beta, step in phases:
        for _ in range(num_iter):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            # compute softmax weights
            # using stable exp
            shifted = corr - max_corr
            w = np.exp(beta * shifted)
            w_sum = w.sum()
            if w_sum == 0:
                # all zero weight? should not happen
                w = np.ones_like(w) / w.size
            else:
                w = w / w_sum
            grad = weighted_gradient(h, w)
            h = h - step * grad
            # project onto feasible set
            h = np.clip(h, 0.0, 1.0)
            h = project_to_feasible(h, target)
    return h

import time
import numpy as np

# ---------- Helper utilities ----------
def compute_c5(h):
    """Exact C5 for a discrete step function."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_to_feasible(v, target):
    """Project onto {0≤h≤1, Σh = target} by shifting and clipping."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Random vector in [0,1] with sum = n/2."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_to_feasible(v, n / 2.0)


def random_binary_vector(n, rng):
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h


def upsample_initial(to_n):
    """Linear interpolation of the global initial_h_values to length to_n."""
    init = np.array(initial_h_values, dtype=float)
    m = init.size
    xs_old = np.linspace(0.0, 2.0, m, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, to_n, endpoint=False)
    new = np.interp(xs_new, xs_old, init)
    return new


def weighted_gradient(h, w):
    """
    Gradient of Σ w_s * corr_s where w corresponds to np.correlate output.
    """
    n = h.size
    w_rev = w[::-1]
    omhr = (1.0 - h)[::-1]
    conv1 = np.convolve(omhr, w_rev)          # length 3n‑2
    conv2 = np.convolve(h, w_rev)            # length 3n‑2
    idx = np.arange(n)
    term1 = conv1[2 * n - 2 - idx]           # w_{i+j} part
    term2 = conv2[n - 1 + idx]               # w_{i' + (n‑1‑i)} part
    return term1 - term2


def gd_optimize(h0, phases):
    """
    Gradient‑descent on the smooth soft‑max surrogate.
    phases = list of (iterations, beta, step) tuples.
    """
    h = h0.copy()
    n = h.size
    target = n / 2.0
    for iters, beta, step in phases:
        for _ in range(iters):
            corr = np.correlate(h, 1.0 - h, mode="full")
            max_corr = np.max(corr)
            shift = corr - max_corr                     # stability
            w = np.exp(beta * shift)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.ones_like(w) / w.size
            else:
                w /= w_sum
            grad = weighted_gradient(h, w)
            h = h - step * grad
            h = np.clip(h, 0.0, 1.0)
            h = project_to_feasible(h, target)
    return h


def binary_from_continuous(h):
    """Make a binary vector by keeping the top n/2 entries."""
    n = h.size
    k = n // 2
    idx = np.argpartition(h, -k)[-k:]
    bin_h = np.zeros(n, dtype=float)
    bin_h[idx] = 1.0
    return bin_h


def binary_hill_climb(h0, rng, max_swaps):
    """
    Simple deterministic hill‑climb on binary vectors.
    Swaps a random 1 with a random 0; keeps the change if C5 improves.
    """
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    no_improve = 0

    for _ in range(max_swaps):
        ones = np.flatnonzero(h == 1.0)
        zeros = np.flatnonzero(h == 0.0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        h[i] = 0.0
        h[j] = 1.0
        c = compute_c5(h)

        if c < best_c - 1e-12:
            best_c, best_h = c, h.copy()
            no_improve = 0
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1

        if no_improve >= 2000:
            h = best_h.copy()
            no_improve = 0

    return best_h, best_c
