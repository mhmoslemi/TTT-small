def perturb_random(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0,n)
    j = rng.integers(0,n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

import time
import numpy as np

def compute_c5(h):
    """Exact discrete C₅ for a vector h."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)

def project_onto_bounds_sum(v, target, iters=60):
    """Projection of v onto {0≤x≤1, sum x = target} using bisection."""
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(iters):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - ((lo + hi) * 0.5), 0.0, 1.0)

def perturb_random(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def _grad_softmax_max(h, beta):
    """Gradient of the β‑softmax approximation of maxₖ Iₖ."""
    n = h.size
    dx = 2.0 / n
    I = np.correlate(h, 1 - h, mode="full") * dx               # length 2n‑1
    I_max = I.max()
    w = np.exp(beta * (I - I_max))
    w /= w.sum()
    w_sym = w + w[::-1]                                         # symmetric weights
    conv = np.convolve(w_sym, h, mode="full")                  # length 3n‑2
    grad = dx - dx * conv[n - 1 : n - 1 + n]                    # size n
    return grad, I, I.max()

def run(seed=42, budget_s=1100, **kwargs):

    start = time.time()

    # ----- initialise h on a 720‑point grid -----
    base_n = 720
    if "initial_h_values" in globals():

        if h0.size != base_n:

            xs0 = (np.arange(h0.size) + 0.5) * (2.0 / h0.size)
            xs = (np.arange(base_n) + 0.5) * (2.0 / base_n)
            h = np.interp(xs, xs0, h0)
        else:
            h = h0.copy()
    else:
        h = rng.random(base_n)

    h = np.clip(h, 0.0, 1.0)
    target_sum = base_n / 2.0
    h = project_onto_bounds_sum(h, target_sum)

    # keep the best encountered vector
    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ----- optimisation schedule -----
    beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0, 800.0, 2000.0]
    # start with the current correlation and its max
    cur_grad, cur_I, cur_c = _grad_softmax_max(h, beta_schedule[0])
    cur_c = cur_I.max()
