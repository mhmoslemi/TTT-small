    cur = h.copy()
    cur_c = compute_c5(cur)

def init_binary(n, rng):
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        if init.size == n:
            # binarize by rounding to nearest integer of 0/1 while keeping sum half
            bin_vec = np.where(init >= 0.5, 1.0, 0.0)
            cur_sum = bin_vec.sum()
            target = n/2
            # adjust
            if cur_sum > target:
                # too many ones: set some of the smallest (closest to 0) ones to zero
                # find indices where init close to 0.5? but we can randomly flip excess ones
                excess = int(cur_sum - target)
                ones_idx = np.where(bin_vec == 1)[0]
                idx_to_flip = rng.choice(ones_idx, excess, replace=False)
                bin_vec[idx_to_flip] = 0
            elif cur_sum < target:
                deficit = int(target - cur_sum)
                zeros_idx = np.where(bin_vec == 0)[0]
                idx_to_flip = rng.choice(zeros_idx, deficit, replace=False)
                bin_vec[idx_to_flip] = 1
            return bin_vec
    # fallback random
    # Randomly pick n/2 indices to be 1
    idx = rng.choice(n, n // 2, replace=False)
    vec = np.zeros(n)
    vec[idx] = 1.0
    return vec

import time, math
import numpy as np
from scipy.optimize import minimize

# ---------- utility ----------
def compute_c5(h):
    """Exact discrete C5 for a step function h on [0,2] (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_to_feasible(v, target):
    """Project v onto {0≤x≤1, Σx=target} by a simple bisection on a shift."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


# ---------- binary annealing ----------
def init_binary(n, rng):
    """Create a feasible binary vector (sum = n/2). Use initial_h_values when possible."""
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)
        if init.size == n:
            bin_vec = (init >= 0.5).astype(float)
            cur = bin_vec.sum()
            target = n // 2
            if cur > target:
                excess = int(cur - target)
                ones = np.where(bin_vec == 1)[0]
                idx = rng.choice(ones, excess, replace=False)
                bin_vec[idx] = 0
            elif cur < target:
                deficit = int(target - cur)
                zeros = np.where(bin_vec == 0)[0]
                idx = rng.choice(zeros, deficit, replace=False)
                bin_vec[idx] = 1
            return bin_vec
    # random binary vector of correct weight
    idx = rng.choice(n, n // 2, replace=False)
    vec = np.zeros(n)
    vec[idx] = 1.0
    return vec


def binary_anneal_time_budget(n, rng, time_budget):
    """Simulated annealing on binary vectors, returns best found within time_budget."""
    start = time.time()
    cur = init_binary(n, rng)
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    # simple linear temperature schedule
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        t_frac = elapsed / time_budget
        temp = max(0.1, (1.0 - t_frac) * 8.0 + 0.1)   # 8→0.1
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        new = cur.copy()
        new[i] = 0.0
        new[j] = 1.0
        new_c = compute_c5(new)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < math.exp(-delta / temp):
            cur = new
            cur_c = new_c
            if new_c < best_c:
                best = new.copy()
                best_c = new_c
    return best, best_c


# ---------- continuous refinement ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) via log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=400):
    """Single SLSQP run with the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction (simple projection)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x


def hill_climb(h_start, rng, max_iters=2000, start_step=0.3):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        # transfer a small epsilon from i to j
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0:
                eps = (rng.random() - 0.5) * 2.0 * max_eps * step
                eps = max(-max_eps, min(eps, max_eps))
                new_h = h.copy()
                new_h[i] -= eps
                new_h[j] += eps
                c = compute_c5(new_h)
                if c < best_c:
                    best_c, best_h, h = c, new_h.copy(), new_h.copy()
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


# ---------- main routine ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_total = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # candidate discretisations (smaller first)
    ns = [60, 84, 120, 168, 240, 300]
    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        if n0 not in ns:
            ns = [n0] + ns

    for idx, n in enumerate(ns):
        elapsed = time.time() - start_total
        remaining = budget_s - elapsed
        if remaining < 5.0:
            break

        # allocate time for this resolution (roughly half remaining, capped)
        time_for_n = min(remaining * 0.5, max(10.0, n * 0.4))
        # -------- binary annealing --------
        t_bin = time_for_n * 0.6
        h_bin, c_bin = binary_anneal_time_budget(n, rng, t_bin)

        # -------- SLSQP refinement (optional) --------
        h_ref, c_ref = h_bin, c_bin
        if time.time() - start_total < budget_s * 0.95:
            # a short cascade of increasing β values
            for beta in (20.0, 100.0, 500.0):
                if time.time() - start_total > budget_s * 0.95:
                    break
                h_ref = slsqp_optimize(h_ref, beta, rng, max_iter=250)
                c_ref = compute_c5(h_ref)

        # -------- final hill‑climb (continuous) --------
        if time.time() - start_total < budget_s * 0.98:
            h_ref, c_ref = hill_climb(h_ref, rng, max_iters=1500, start_step=0.25)

        # keep best of the three candidates
        for cand_h, cand_c in ((h_bin, c_bin), (h_ref, c_ref)):
            if cand_c < best_c:
                best_c, best_h, best_n = cand_c, cand_h.copy(), n

    # fallback if nothing found
    if best_h is None:
        best_n = 60
        best_h = np.full(best_n, 0.5)
        best_c = compute_c5(best_h)

    # enforce feasibility (tiny numerical drift)
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, best_n)
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
