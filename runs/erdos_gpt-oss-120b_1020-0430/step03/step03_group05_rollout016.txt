    step = 0.2

        i_dec = int(np.argmin(g))

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Global placeholders for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth surrogate for max_k Σ_i x_i (1 - x_{i+k}) via log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε between two coordinates (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def project_to_feasible(v, target):
    """Projection onto {0≤h≤1, Σh = target} by a bisection on the shift."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Fast random feasible vector (0≤h≤1, Σh=n/2)."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_to_feasible(v, n / 2.0)


def hill_climb(h_start, rng, max_iters=3000, start_step=0.2):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-5)
    return best_h, best_c


def slsqp_optimize(h0, beta, rng, max_iter=500):
    """SLSQP optimisation on the smooth soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # exact sum correction (project onto hyperplane)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def gradient_optimize(h, rng, time_budget):
    """Increase the minimal auto‑correlation by pairwise mass transfers."""
    n = h.size
    target = n / 2.0
    h = project_to_feasible(h, target)
    best_h = h.copy()
    best_c = compute_c5(h)
    step = 0.2
    start = time.time()
    stagnation = 0
    while time.time() - start < time_budget:
        # worst (minimal) correlation shift
        corr = np.correlate(h, h, mode="full")
        S = corr[n - 1 :]
        kmin = int(np.argmin(S))

        # gradient for this shift
        g = np.roll(h, -kmin) + np.roll(h, kmin)
        i_inc = int(np.argmax(g))
        i_dec = int(np.argmin(g))

        max_inc = 1.0 - h[i_inc]
        max_dec = h[i_dec]
        if max_inc <= 0 or max_dec <= 0:
            h = perturb_h(h, rng, scale=step)
            continue

        eps = min(max_inc, max_dec, step * rng.random())
        if eps <= 0:
            step *= 0.9
            continue

        new_h = h.copy()
        new_h[i_inc] += eps
        new_h[i_dec] -= eps
        new_c = compute_c5(new_h)

        if new_c < best_c - 1e-12:
            best_c, best_h, h = new_c, new_h.copy(), new_h
            stagnation = 0
            step = min(step * 1.05, 0.5)
        else:
            stagnation += 1
            step *= 0.99
            if stagnation > 5000:
                stagnation = 0
                step = max(step, 1e-5)
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # base discretisation size
    base_n = len(initial_h_values) if "initial_h_values" in globals() else 300

    # ---------- Build a collection of starting vectors ----------
    seeds = []

    if "initial_h_values" in globals():
        seeds.append(np.array(initial_h_values, dtype=float))

    seeds.append(np.full(base_n, 0.5))

    xs = np.linspace(0.0, 2.0, base_n, endpoint=False)
    tri = np.maximum(0.0, 1.0 - np.abs(xs - 1.0))
    tri = project_to_feasible(tri, base_n / 2.0)
    seeds.append(tri)

    sine = (1.0 + np.cos(np.pi * xs)) / 2.0
    sine = project_to_feasible(sine, base_n / 2.0)
    seeds.append(sine)

    seeds.append(random_feasible_vector(base_n, rng))

    total_seeds = len(seeds)

    for idx, h0 in enumerate(seeds):
        elapsed = time.time() - start
        remaining = budget_s - elapsed
        if remaining < 8.0:
            break

        # share of remaining time for this seed (even split)
        share = remaining / (total_seeds - idx)

        # ---------- Gradient ascent phase ----------
        grad_time = min(share * 0.55, remaining * 0.55)
        if grad_time < 5.0:
            grad_time = 5.0
        h_opt, c_opt = gradient_optimize(h0.copy(), rng, grad_time)
        if c_opt < best_c:
            best_c, best_h, best_n = c_opt, h_opt.copy(), base_n

        # ---------- SLSQP refinement ----------
        for beta in (10.0, 50.0, 200.0, 800.0, 2000.0):
            if time.time() - start > budget_s * 0.94:
                break
            try:
                h_tmp = slsqp_optimize(h_opt, beta, rng, max_iter=500)
                c_tmp = compute_c5(h_tmp)
                if c_tmp < best_c:
                    best_c, best_h, best_n = c_tmp, h_tmp.copy(), base_n
                h_opt = h_tmp
            except Exception:
                continue

        # ---------- Final hill‑climb ----------
        if time.time() - start < budget_s * 0.97:
            h_opt, c_opt = hill_climb(h_opt, rng, max_iters=3000, start_step=0.1)
            if c_opt < best_c:
                best_c, best_h, best_n = c_opt, h_opt.copy(), base_n

    # ---------- Ensure exact feasibility ----------
    if best_h is None:
        best_n = base_n
        best_h = np.full(best_n, 0.5)

    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, best_n)
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
