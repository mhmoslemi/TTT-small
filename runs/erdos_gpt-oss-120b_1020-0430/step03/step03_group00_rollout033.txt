import time
import numpy as np
from scipy.optimize import differential_evolution

# ---------- projection onto capped simplex ----------
def _proj_capped(v, target):
    """Project v onto {0<=x<=1, sum(x)=target} via binary search on tau."""
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

# ---------- helper for feasible vectors ----------
def random_feasible(n, rng):
    v = rng.random(n)
    return _proj_capped(v, n / 2.0)

# ---------- C5 evaluation ----------
def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

# ---------- autocorrelation (FFT) ----------
def autocorr_fft(h):
    """Linear autocorrelation of h using FFT (no wrap‑around)."""
    n = h.size
    N = 1 << ((2 * n - 1).bit_length())
    a = np.zeros(N, dtype=h.dtype)
    a[:n] = h
    H = np.fft.rfft(a)
    conv = np.fft.irfft(np.abs(H), N)
    return conv[: 2 * n - 1]

# ---------- gradient refinement ----------
def refine_gradient(h, time_limit, start_time):
    n = h.size
    target = n / 2.0
    step = 0.05
    while time.time() - start_time < time_limit:
        corr = np.correlate(h, h, mode="full")
        min_idx = np.argmin(corr)
        k = min_idx - (n - 1)  # shift giving minimal autocorrelation
        grad = np.zeros_like(h)
        if k >= 0:
            grad[: n - k] += h[k:]
            grad[k:] += h[: n - k]
        else:
            kk = -k
            grad[: n - kk] += h[kk:]
            grad[kk:] += h[: n - kk]
        h = h + step * grad
        h = np.clip(h, 0.0, 1.0)
        h = _proj_capped(h, target)
        step *= 0.9995
    return h

# ---------- main optimisation ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- obtain initial h on 720 points ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = random_feasible(720, rng)
    n0 = h0.size
    # target number of points (use 720; if initial already 720 keep it)
    n = 720
    if n0 != n:
        xs0 = (np.arange(n0) + 0.5) * (2.0 / n0)
        xs = (np.arange(n) + 0.5) * (2.0 / n)
        h = np.interp(xs, xs0, h0)
    else:
        h = h0.copy()
    h = np.clip(h, 0.0, 1.0)
    h = _proj_capped(h, n / 2.0)

    # ---- block representation (M blocks) ----
    M = 72                     # 720 / 10
    L = n // M                 # block length = 10
    b0 = h.reshape(M, L).mean(axis=1)
    b0 = _proj_capped(b0, M / 2.0)

    # ---- differential evolution on block variables ----
    bounds = [(0.0, 1.0)] * M
    de_time = budget_s * 0.78
    de_start = time.time()
    target_sum_blocks = M / 2.0

    def obj_blocks(v):
        b = _proj_capped(v, target_sum_blocks)
        h_full = np.repeat(b, L)
        # exact C5
        return compute_c5(h_full)

    def de_cb(xk, convergence):
        return time.time() - de_start > de_time

    result = differential_evolution(
        obj_blocks,
        bounds,
        maxiter=400,
        seed=int(rng.integers(0, 1 << 32)),
        polish=False,
        callback=de_cb,
        updating="deferred",
        workers=1,
    )
    b_best = _proj_capped(result.x, target_sum_blocks)
    h_best = np.repeat(b_best, L)
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = _proj_capped(h_best, n / 2.0)

    # ---- local gradient refinement ----
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0:
        h_best = refine_gradient(
            h_best,
            time_limit=remaining * 0.9,
            start_time=time.time(),
        )
        h_best = np.clip(h_best, 0.0, 1.0)
        h_best = _proj_capped(h_best, n / 2.0)

    final_c5 = compute_c5(h_best)

    # ---- final tiny random search (optional) ----
    while time.time() - start < budget_s * 0.98:
        cand = h_best + rng.normal(scale=0.005, size=n)
        cand = np.clip(cand, 0.0, 1.0)
        cand = _proj_capped(cand, n / 2.0)
        c = compute_c5(cand)
        if c < final_c5:
            h_best, final_c5 = cand, c

    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = _proj_capped(h_best, n / 2.0)
    final_c5 = compute_c5(h_best)

    return h_best.tolist(), float(final_c5), int(n)
