def softmax_obj_and_grad(h, beta):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full') # length 2n-1
    # Compute stable softmax
    max_corr = corr.max()
    exp_vals = np.exp(beta * (corr - max_corr))
    sum_exp = exp_vals.sum()
    obj = (max_corr + np.log(sum_exp)) / beta

    # Compute weights
    w = exp_vals / sum_exp  # length 2n-1

    # Gradient
    # Initialize zero gradient
    g = np.zeros_like(h)
    # For each lag index m with shift = m - (n-1)
    for m, weight in enumerate(w):
        shift = m - (n - 1)
        if shift >= 0:
            # i from 0..n-1-shift
            i_range = slice(0, n - shift)
            # mask_fwd is 1 for these i
            g[i_range] += weight  # +1 term
            # Subtract h[i+shift] term
            g[i_range] -= weight * h[i_range + shift]
            # Subtract h[i-shift] term (where i-shift >=0) -> i from shift..n-1
            g[shift:] -= weight * h[:n - shift]
        else:
            s = -shift
            # shift negative, i from s..n-1 (i+shift >=0)
            i_start = s
            i_end = n
            g[i_start:] += weight
            g[i_start:] -= weight * h[i_start - s:]  # i+shift = i - s, h[i+shift]
            # subtract h[i - shift] term where i - shift < n
            # i - shift = i + s, must be < n, so i < n - s
            g[:n - s] -= weight * h[s:]
    return obj, g

But need to be careful with slicing: For shift>=0, g[i_range] += weight; but we need addition of weight for each i in that range. However weight is scalar, adding to each element yields same increment; that's correct. For subtraction of h[i+shift] term: we need to subtract weight * h[i+shift] for each i; using vectorized multiplication with slice h[i_range+shift] works.

Similarly subtract h[i-shift] term: For shift>=0, i - shift >=0 for i >= shift, i in shift..n-1. This corresponds to slice shift: n. h[:n-shift] corresponds to h[i - shift] where i from shift..n-1. Indeed because i - shift ranges from 0..n-1-shift.

Thus the code above for shift>=0 likely correct.

For shift<0, let s = -shift >0. Then i from s..n-1 (since i+shift >=0). So i_start = s. The mask_fwd is 1 for these i. So g[i_start:] += weight.

Now subtract h[i+shift] term (i+shift = i - s) for i in i_start..n-1, using h[i_start - s:] = h[:n-s]. That matches.

Subtract h[i - shift] term: i - shift = i + s, must be < n, i < n - s, so i in 0..n-s-1. So we subtract weight * h[i + s] for i in that range; that's h[s:].

def enforce_sum(h, target, rng, max_iter=1000):
    # same as earlier: clip and adjust sum.
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # final correction if needed
    if abs(diff) > 1e-12:
        # small distributed diff
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
    return h

def lbfgsb_optimize(h0, beta, max_iter=200, lam=1e3):

        # compute obj and gradient for current h x
        # compute correlation
        corr = np.correlate(x, 1 - x, mode='full')
        max_corr = corr.max()
        exp_vals = np.exp(beta * (corr - max_corr))
        sum_exp = exp_vals.sum()
        # softmax part
        obj = (max_corr + np.log(sum_exp)) / beta
        w = exp_vals / sum_exp
        # gradient softmax

        # loop over each lag (2n-1)
        n2 = n * 2 - 1
        for m in range(n2):
            weight = w[m]
            shift = m - (n - 1)
            if shift >= 0:
                # i from 0..n-1-shift
                i_end = n - shift
                g[:i_end] += weight
                g[:i_end] -= weight * x[shift:]
                g[shift:] -= weight * x[:i_end]
            else:
                s = -shift
                i_start = s
                i_end = n
                g[i_start:] += weight
                g[i_start:] -= weight * x[i_start - s:]  # i+shift = i - s
                g[:n - s] -= weight * x[s:]
        # penalty for sum constraint
        diff = x.sum() - target
        obj += lam * diff * diff
        g += 2 * lam * diff
        return obj, g
    bounds = [(0.0, 1.0)] * n
    res = minimize(lambda x: obj_grad(x), h0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum exact
    x = enforce_sum(x, target, np.random.default_rng())
    return x

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def enforce_sum(h, target, rng, max_iter=1000):
    """Clip to [0,1] and adjust entries so that sum equals target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
    return h


def random_binary(n, rng):
    """Return a length‑n binary array with exactly n/2 ones."""
    half = n // 2
    b = np.zeros(n, dtype=float)
    b[rng.choice(n, size=half, replace=False)] = 1.0
    return b


def greedy_swap_improve(b, rng, max_swaps=300, max_no_improve=30):
    """Local search on a binary vector: swap 1↔0 to lower max overlap."""
    n = b.size
    b = b.copy()
    no_improve = 0
    for _ in range(max_swaps):
        corr = np.correlate(b, 1.0 - b, mode="full")
        max_idx = np.argmax(corr)
        shift = max_idx - (n - 1)

        if shift >= 0:
            length = n - shift
            bad = np.where((b[:length] == 1) & (b[shift:] == 0))[0]
            good = np.where((b[:length] == 0) & (b[shift:] == 1))[0]
        else:
            s = -shift
            length = n - s
            bad = np.where((b[s:] == 1) & (b[:length] == 0))[0] + s
            good = np.where((b[s:] == 0) & (b[:length] == 1))[0] + s

        if bad.size and good.size:
            i = rng.choice(bad)
            j = rng.choice(good)
            b[i] = 0.0
            b[j] = 1.0
            no_improve = 0
        else:
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            b[i] = 0.0
            b[j] = 1.0
            no_improve += 1
            if no_improve >= max_no_improve:
                break
    return b


def lbfgsb_optimize(h0, beta, max_iter=300, lam=1e3):
    """L‑BFGS‑B on the smooth max objective + quadratic sum‑penalty."""
    n = h0.size
    target = n / 2.0

    def fun(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = corr.max()
        expv = np.exp(beta * (corr - m))
        s = expv.sum()
        obj = (m + np.log(s)) / beta
        diff = x.sum() - target
        obj += lam * diff * diff

        w = expv / s
        g = np.zeros_like(x)
        for m_idx, weight in enumerate(w):
            shift = m_idx - (n - 1)
            if shift >= 0:
                i_end = n - shift
                g[:i_end] += weight
                g[:i_end] -= weight * x[shift:]
                g[shift:] -= weight * x[:i_end]
            else:
                sft = -shift
                i_end = n - sft
                g[sft:] += weight
                g[sft:] -= weight * x[:i_end]
                g[:i_end] -= weight * x[sft:]
        g += 2.0 * lam * diff
        return obj, g

    bounds = [(0.0, 1.0)] * n
    res = minimize(fun, h0, method="L-BFGS-B", jac=True,
                   bounds=bounds, options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    x = enforce_sum(x, target, np.random.default_rng())
    return x


def fine_hill_climb(h, rng, max_iters=2000, init_step=0.5):
    """Small continuous hill‑climb (transfer ε between two coordinates)."""
    best = h.copy()
    best_c = compute_c5(best)
    step = init_step
    n = h.size
    for _ in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(best[i], 1.0 - best[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * step
        new = best.copy()
        new[i] -= eps
        new[j] += eps
        c = compute_c5(new)
        if c < best_c:
            best = new
            best_c = c
    return best, best_c

    while True:
        elapsed = time.time() - start
        if elapsed > budget_s * 0.90:
            break

        # random binary start → greedy swaps
        b = random_binary(n, rng)
        b = greedy_swap_improve(b, rng, max_swaps=250, max_no_improve=20)
        c = compute_c5(b)
        if c < best_c:
            best_c = c
            best_h = b.astype(float)

    # --- final continuous refinement --------------------------------
    if best_h is None:
        best_h = np.full(n, 0.5)

    h_ref = best_h.copy()
    for beta in (5, 20, 80, 200, 500, 1000):
        if time.time() - start > budget_s * 0.98:
            break
        h_ref = lbfgsb_optimize(h_ref, beta, max_iter=300)
        c = compute_c5(h_ref)
        if c < best_c:
            best_c = c
            best_h = h_ref.copy()

    # --- optional fine hill‑climb -----------------------------------
    if time.time() - start < budget_s * 0.99:
        best_h, best_c = fine_hill_climb(best_h, rng, max_iters=2000, init_step=0.5)

    # --- final cleanup ---------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
