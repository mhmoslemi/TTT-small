def project_onto_box_and_sum(v, target):
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(50):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = np.sum(x)
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    # Due to rounding, adjust sum maybe by small reallocation
    diff = target - np.sum(x)
    if abs(diff) > 1e-12:

        # distribute equally among entries with 0 < x_i < 1
        free = (x > 0) & (x < 1)
        nfree = np.count_nonzero(free)

            adjustment = diff / nfree
            x[free] += adjustment
            # clip again to ensure bounds

        free = (x > 0.0) & (x < 1.0)
        nfree = np.count_nonzero(free)
        if nfree > 0:
            adj = diff / nfree
            x[free] += adj
            x = np.clip(x, 0.0, 1.0)
    return x


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    v = rng.exponential(scale=1.0, size=n)
    v = v / v.sum() * (n / 2.0)
    return project_onto_box_and_sum(v, n / 2.0)


def subgradient_descent(h0, target, max_iters, start_alpha, rng):
    """Projected sub‑gradient descent on max_k overlap."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    N = h.size
    for it in range(1, max_iters + 1):
        corr = np.correlate(h, 1 - h, mode="full")
        k_idx = int(np.argmax(corr))
        shift = k_idx - (N - 1)            # offset in [-N+1, N-1]
        # gradient of the maximal shift
        grad = np.ones(N)
        if shift >= 0:
            s = shift
            grad[: N - s] -= h[s:]
            grad[s:] -= h[: N - s]
        else:
            s = -shift
            grad[s:] -= h[: N - s]
            grad[: N - s] -= h[s:]
        alpha = start_alpha / np.sqrt(it)
        v = h - alpha * grad
        h = project_onto_box_and_sum(v, target)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

# Main entry point
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # --------------------------------------------------------------
    # Determine dimension and initial vector
    # --------------------------------------------------------------
    if "initial_h_values" in globals():
        base = np.array(initial_h_values, dtype=float)
        n0 = base.size
        factor = 2                     # up‑sample factor
        n = n0 * factor
        h0 = np.repeat(base, factor)
        h0 = project_onto_box_and_sum(h0, n / 2.0)
    else:
        n = 360                         # a reasonably fine grid
        h0 = np.full(n, 0.5)
        h0 = project_onto_box_and_sum(h0, n / 2.0)

    target = n / 2.0

    # --------------------------------------------------------------
    # Build a diverse pool of starting points
    # --------------------------------------------------------------
    candidates = [h0]

    xs = (np.arange(n) + 0.5) * (2.0 / n)
    for f in (1, 2, 3, 4):
        for fun in (np.sin, np.cos):
            h = 0.5 + 0.5 * fun(np.pi * f * xs)
            h = np.clip(h, 0.0, 1.0)
            candidates.append(project_onto_box_and_sum(h, target))

    # linear ramps
    candidates.append(project_onto_box_and_sum(xs / 2.0, target))
    candidates.append(project_onto_box_and_sum(1.0 - xs / 2.0, target))

    # random feasible vectors
    for _ in range(8):
        candidates.append(random_feasible_vector(n, rng))

    # --------------------------------------------------------------
    # Pick the best initial candidate
    # --------------------------------------------------------------
    best_h = None
    best_c = float("inf")
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # --------------------------------------------------------------
    # Main optimisation loop: sub‑gradient + occasional SLSQP
    # --------------------------------------------------------------
    beta_list = [10.0, 40.0, 120.0, 400.0, 800.0]
    while time.time() - start_time < budget_s * 0.85:
        # choose a start point
        if rng.random() < 0.6:
            start = best_h.copy()
            noise = rng.normal(scale=0.07, size=n)
            start = project_onto_box_and_sum(start + noise, target)
        else:
            start = random_feasible_vector(n, rng)

        # remaining time decides iteration budget
        time_left = budget_s - (time.time() - start_time)
        max_iters = int(min(2500, time_left * 200))  # heuristic
        if max_iters < 200:
            break

        h_sgd, c_sgd = subgradient_descent(start, target, max_iters, start_alpha=0.8, rng=rng)

        if c_sgd < best_c:
            best_c = c_sgd
            best_h = h_sgd.copy()

            # refine with SLSQP on the smooth surrogate
            for beta in beta_list:
                if time.time() - start_time > budget_s * 0.95:
                    break
                try:
                    refined = slsqp_optimize(best_h, beta, rng, max_iter=500)
                except Exception:
                    continue
                c_ref = compute_c5(refined)
                if c_ref < best_c:
                    best_c = c_ref
                    best_h = refined.copy()

            # short extra sub‑gradient pass after refinement
            if time.time() - start_time < budget_s * 0.90:
                extra_iters = int(min(1200, (budget_s - (time.time() - start_time)) * 200))
                if extra_iters > 0:
                    h_extra, c_extra = subgradient_descent(best_h, target,
                                                           extra_iters, start_alpha=0.5, rng=rng)
                    if c_extra < best_c:
                        best_c = c_extra
                        best_h = h_extra.copy()

    # --------------------------------------------------------------
    # Final clean‑up (clip + exact sum enforcement)
    # --------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_and_sum(best_h, target)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
