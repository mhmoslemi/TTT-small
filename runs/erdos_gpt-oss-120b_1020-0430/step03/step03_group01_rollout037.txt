def compute_c5_fft(h):
    n = h.size
    a = h.astype(np.float64)
    b = (1.0 - h)[::-1]  # reversed
    m = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=m)
    fb = np.fft.rfft(b, n=m)
    conv = np.fft.irfft(fa * fb, n=m)
    # conv length m; first 2*n-1 entries correspond to correlation.
    # take first 2*n-1 values
    conv = conv[:2*n-1]
    return np.max(conv) * (2.0 / n)

def enforce_sum(h, target, rng, max_iter=2000):
    h = np.clip(h, 0., 1.)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        # amount we can adjust at i respecting bounds
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # final correction if small diff left

def hill_climb_sa(h0, rng, max_iters, init_scale, temp0):

   h = h0.copy()
   best_h = h.copy()
   best_c = compute_c5_fft(h)
   cur_c = best_c
   scale = init_scale
   for it in range(max_iters):
        T = temp0 * (1 - it / max_iters)
        # propose transfer
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        while j == i:
            j = rng.integers(0, n)
        # direction random
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_eps = min(h[src], 1.0 - h[dst])
        if max_eps <= 0:
            continue
        eps = (rng.random()*2-1) * max_eps * scale
        # ensure stay within bound
        eps = max(-max_eps, min(eps, max_eps))
        new_h = h.copy()
        new_h[src] -= eps
        new_h[dst] += eps
        # evaluate
        new_c = compute_c5_fft(new_h)
        # acceptance
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / (T + 1e-12)):
            h = new_h
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()
        # optionally adjust scale
        if (it+1) % (max_iters//10) == 0:
            # reduce scale
            scale = max(scale * 0.8, 1e-4)
   return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ---------- helper functions ----------
def enforce_sum(h, target, rng, max_iter=2000):
    """Clip to [0,1] and adjust to meet the exact sum."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def correlation_full(h):
    """Full zero‑padded correlation of h with (1‑h). Returns raw sums."""
    n = h.size
    a = h.astype(np.float64)
    b = (1.0 - h)[::-1]
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=m)
    fb = np.fft.rfft(b, n=m)
    conv = np.fft.irfft(fa * fb, n=m)
    return conv[: 2 * n - 1]


def compute_c5_fft(h):
    """Exact discrete C5 using FFT."""
    n = h.size
    return np.max(correlation_full(h)) * (2.0 / n)


# ---------- smooth surrogate for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def smooth_obj(x):
    corr = correlation_full(x) * (2.0 / x.size)
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(CURRENT_BETA * (corr - m))))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=400):
    """Refine h using SLSQP on the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        smooth_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum (tiny drift)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ---------- random constructions ----------
def random_binary_h(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h


def random_continuous_h(n, rng):
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)


# ---------- hill‑climb with simulated annealing ----------
def hill_climb_sa(h0, rng, max_iters, init_scale, temp0):
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    cur_c = compute_c5_fft(h)
    best_c = cur_c
    scale = init_scale
    for it in range(max_iters):
        T = temp0 * (1.0 - it / max_iters)
        # propose a transfer
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        while j == i:
            j = rng.integers(0, n)
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_eps = min(h[src], 1.0 - h[dst])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() * 2.0 - 1.0) * max_eps * scale
        eps = max(-max_eps, min(eps, max_eps))
        new_h = h.copy()
        new_h[src] -= eps
        new_h[dst] += eps
        new_c = compute_c5_fft(new_h)
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / (T + 1e-12)):
            h = new_h
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()
        if (it + 1) % (max_iters // 10 or 1) == 0:
            scale = max(scale * 0.8, 1e-4)
    return best_h, best_c


# ---------- main routine ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_limit = budget_s

    # ----- initial vector -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        target = n / 2.0
        h0 = enforce_sum(np.clip(h0, 0.0, 1.0), target, rng)
    else:
        n = 360
        h0 = random_binary_h(n, rng)
        target = n / 2.0

    best_h = h0.copy()
    best_c = compute_c5_fft(best_h)

    # ----- diverse seeds -----
    candidates = [best_h]
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    for f in (1, 2, 3, 4):
        h = 0.5 + 0.5 * np.sin(np.pi * f * xs)
        candidates.append(enforce_sum(np.clip(h, 0.0, 1.0), target, rng))
        h = 0.5 + 0.5 * np.cos(np.pi * f * xs)
        candidates.append(enforce_sum(np.clip(h, 0.0, 1.0), target, rng))
    for _ in range(5):
        candidates.append(random_binary_h(n, rng))

    for cand in candidates:
        c = compute_c5_fft(cand)
        if c < best_c:
            best_c, best_h = c, cand.copy()

    # ----- main optimisation loop -----
    max_sa_iters = 30000
    init_scale = 0.5
    temp0 = 0.005

    while time.time() - start < time_limit * 0.78:
        # start point: jittered best or fresh random binary
        if rng.random() < 0.7:
            h_start = best_h.copy()
            for _ in range(12):
                i, j = rng.integers(0, n, size=2)
                while j == i:
                    j = rng.integers(0, n)
                h_start[i], h_start[j] = h_start[j], h_start[i]
            h_start = enforce_sum(h_start, target, rng)
        else:
            h_start = random_binary_h(n, rng)

        remaining = time_limit - (time.time() - start)
        iters = min(max_sa_iters, max(2000, int(remaining * 2200)))
        h_sa, c_sa = hill_climb_sa(h_start, rng, iters, init_scale, temp0)
        if c_sa < best_c:
            best_c, best_h = c_sa, h_sa.copy()

        # SLSQP refinement ladder
        for beta in (20, 80, 200, 500, 1000, 2000):
            if time.time() - start > time_limit * 0.78:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, max_iter=800)
                c_ref = compute_c5_fft(h_ref)
                if c_ref < best_c:
                    best_c, best_h = c_ref, h_ref.copy()
            except Exception:
                pass

    # ----- upsample to finer grid if time permits -----
    if n < 720 and time.time() - start < time_limit * 0.94:
        target_n = 720
        old_x = (np.arange(n) + 0.5) * (2.0 / n)
        new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
        h_up = np.interp(new_x, old_x, best_h)
        target = target_n / 2.0
        h_up = enforce_sum(h_up, target, rng)
        n = target_n
        best_h = h_up
        best_c = compute_c5_fft(best_h)

        while time.time() - start < time_limit * 0.97:
            remaining = time_limit - (time.time() - start)
            iters = min(max_sa_iters, max(1500, int(remaining * 1500)))
            h_sa, c_sa = hill_climb_sa(best_h, rng, iters, init_scale * 0.5, temp0)
            if c_sa < best_c:
                best_c, best_h = c_sa, h_sa.copy()
            for beta in (100, 500, 1000, 2000, 5000):
                if time.time() - start > time_limit:
                    break
                try:
                    h_ref = slsqp_optimize(best_h, beta, max_iter=800)
                    c_ref = compute_c5_fft(h_ref)
                    if c_ref < best_c:
                        best_c, best_h = c_ref, h_ref.copy()
                except Exception:
                    pass
            if time.time() - start > time_limit * 0.99:
                break

    # ----- final clean‑up -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5_fft(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
