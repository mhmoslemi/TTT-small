def anneal_binary(h, rng, time_budget, start_temp, end_temp):
    best_h = h.copy()
    best_c = compute_c5_bin(h)
    cur_h = h.copy()
    cur_c = best_c
    t0 = start_temp
    t1 = end_temp
    start = time.time()
    while time.time() - start < time_budget:
        # compute temperature based on elapsed fraction
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = t0 * (t1/t0)**frac   # exponential cooling from t0 to t1
        # propose move: pick i with h_i=1 and j with h_j=0
        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur_h[i] = 0
        cur_h[j] = 1
        new_c = compute_c5_bin(cur_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] = 1
            cur_h[j] = 0
    return best_h, best_c
```

        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            if abs(add) < 1e-12:
                continue
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def binary_anneal(h_start, rng, time_budget, T_start=0.10, T_end=1e-5):
    """Simulated‑annealing on a binary (0/1) vector preserving the sum."""
    n = h_start.size
    cur_h = h_start.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    t0 = time.time()
    while True:
        elapsed = time.time() - t0
        if elapsed >= time_budget:
            break
        frac = elapsed / time_budget
        T = T_start * (1 - frac) + T_end * frac
        i = rng.integers(0, n)
        if cur_h[i] == 1:
            j = rng.integers(0, n)
            while cur_h[j] == 1:
                j = rng.integers(0, n)
        else:
            j = rng.integers(0, n)
            while cur_h[j] == 0:
                j = rng.integers(0, n)
        # swap
        cur_h[i] = 0
        cur_h[j] = 1
        new_c = compute_c5(cur_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            cur_h[i] = 1
            cur_h[j] = 0
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    base_n = 180

    # ---- initialise a binary vector of length base_n ----
    if "initial_h_values" in globals():
        h_init = np.array(initial_h_values, dtype=float)
        h_init = np.clip(h_init, 0.0, 1.0)
        if h_init.size != base_n:
            xs_old = np.linspace(0, 2, h_init.size, endpoint=False)
            xs_new = np.linspace(0, 2, base_n, endpoint=False)
            h_init = np.interp(xs_new, xs_old, h_init)
        h_bin = (h_init > 0.5).astype(int)
        diff = (base_n // 2) - h_bin.sum()
        if diff > 0:
            zeros = np.where(h_bin == 0)[0]
            h_bin[rng.choice(zeros, diff, replace=False)] = 1
        elif diff < 0:
            ones = np.where(h_bin == 1)[0]
            h_bin[rng.choice(ones, -diff, replace=False)] = 0
    else:
        h_bin = np.zeros(base_n, dtype=int)
        h_bin[rng.choice(base_n, base_n // 2, replace=False)] = 1

    # ---- Phase 1 : anneal on base_n ----
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    phase1_time = min(400.0, remaining * 0.5)  # up to 400 s or half the budget
    best_h_bin, best_c = binary_anneal(h_bin, rng, phase1_time)

    # ---- Phase 2 : upsample to 2× and anneal further ----
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 15.0:
        h_up = np.repeat(best_h_bin, 2)
        n_up = h_up.size
        target_up = n_up // 2
        if h_up.sum() != target_up:  # tiny repair just in case
            diff = target_up - h_up.sum()
            if diff > 0:
                zeros = np.where(h_up == 0)[0]
                h_up[rng.choice(zeros, diff, replace=False)] = 1
            else:
                ones = np.where(h_up == 1)[0]
                h_up[rng.choice(ones, -diff, replace=False)] = 0
        phase2_time = max(0.0, remaining - 8.0)  # leave ~8 s for final polishing
        best_h_bin, best_c = binary_anneal(h_up, rng, phase2_time)
        n_final = n_up
    else:
        n_final = base_n

    # ---- Phase 3 : continuous refinement with SLSQP ----
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    h_float = best_h_bin.astype(float)
    h_float = enforce_sum(h_float, n_final / 2.0, rng)

    if remaining > 5.0:
        beta_list = [10.0, 40.0, 120.0, 400.0, 800.0, 2000.0]
        for beta in beta_list:
            if time.time() - start_time > budget_s * 0.97:
                break
            try:
                h_float = slsqp_optimize(h_float, beta, max_iter=600)
                h_float = enforce_sum(h_float, n_final / 2.0, rng)
            except Exception:
                continue

    # final evaluation and clean‑up
    h_float = np.clip(h_float, 0.0, 1.0)
    h_float = enforce_sum(h_float, n_final / 2.0, rng)
    final_c = compute_c5(h_float)

    return h_float.tolist(), float(final_c), int(n_final)
