def softmax_grad(h, beta):
    n = h.shape[0]
    O = np.correlate(h, 1 - h, mode='full')
    max_O = O.max()
    exp_beta = np.exp(beta * (O - max_O))
    w = exp_beta / exp_beta.sum()
    grad = np.ones_like(h)  # from constant term sum of w = 1
    # iterate over idx, weight
    # shift = idx - (n - 1)
    for idx, weight in enumerate(w):
        shift = idx - (n - 1)
        if shift == 0:
            grad -= 2 * weight * h
        elif shift > 0:
            grad[: n - shift] -= weight * h[shift:]
            grad[shift:] -= weight * h[: n - shift]
        else: # shift <0
            s = -shift
            grad[s:] -= weight * h[: n - s]
            grad[: n - s] -= weight * h[s:]
    return grad

def softmax_obj_slsqp(x):
    O = np.correlate(x, 1 - x, mode='full')
    max_O = np.max(O)
    return (max_O + np.log(np.sum(np.exp(CURRENT_BETA * (O - max_O))))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj_slsqp, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum exactly
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, target=CURRENT_TARGET, rng=rng)
    return x

def enforce_sum(v, target=None, rng=None):
    n = v.shape[0]
    if target is None:
        target = n / 2.0
    diff = target - v.sum()
    # tolerance
    eps = 1e-12
    if abs(diff) < eps:
        return v
    # We'll use water-filling algorithm
    # We'll do iterative updates until diff satisfied
    # For speed, we can use loops but n small.
    # We'll compute indices each iteration
    while abs(diff) > eps:
        if diff > 0:
            idx = np.where(v < 1 - 1e-12)[0]
            if idx.size == 0:
                break
            # max possible increase
            max_inc = 1 - v[idx]
            total_inc = max_inc.sum()
            if total_inc <= diff:
                v[idx] = 1.0
                diff = target - v.sum()
                continue
            inc = diff / idx.size
            # allocate inc but not exceed per idx
            inc_vec = np.minimum(max_inc, inc)
            v[idx] += inc_vec
            diff = target - v.sum()
        else:
            idx = np.where(v > 1e-12)[0]
            if idx.size == 0:
                break
            max_dec = v[idx]
            total_dec = max_dec.sum()
            if total_dec <= -diff:
                v[idx] = 0.0
                diff = target - v.sum()
                continue
            dec = -diff / idx.size
            dec_vec = np.minimum(max_dec, dec)
            v[idx] -= dec_vec
            diff = target - v.sum()
    return v

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# basic utilities
# ----------------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def enforce_sum(v: np.ndarray, target: float = None, rng: np.random.Generator = None) -> np.ndarray:
    """Project v onto the box [0,1] and the hyperplane sum(v)=target (target = n/2)."""
    n = v.shape[0]
    if target is None:
        target = n / 2.0
    eps = 1e-12
    diff = target - v.sum()
    if abs(diff) < eps:
        return v
    # water‑filling / draining loop
    while abs(diff) > eps:
        if diff > 0.0:
            idx = np.where(v < 1.0 - eps)[0]
            if idx.size == 0:
                break
            max_inc = 1.0 - v[idx]
            total = max_inc.sum()
            if total <= diff:
                v[idx] = 1.0
                diff = target - v.sum()
                continue
            inc = diff / idx.size
            inc_vec = np.minimum(max_inc, inc)
            v[idx] += inc_vec
        else:
            idx = np.where(v > eps)[0]
            if idx.size == 0:
                break
            max_dec = v[idx]
            total = max_dec.sum()
            if total <= -diff:
                v[idx] = 0.0
                diff = target - v.sum()
                continue
            dec = -diff / idx.size
            dec_vec = np.minimum(max_dec, dec)
            v[idx] -= dec_vec
        diff = target - v.sum()
    return v


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector in [0,1]^n with sum n/2."""
    target = n / 2.0
    h = rng.random(n)
    s = h.sum()
    if s == 0.0:
        idx = rng.choice(n, int(target), replace=False)
        h[idx] = 1.0
    else:
        h *= target / s
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, target=target, rng=rng)
    return h


# ----------------------------------------------------------------------
# gradient of the worst overlap (used for guided hill climbing)
# ----------------------------------------------------------------------
def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray, rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h_start: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.5) -> tuple[np.ndarray, float]:
    """Hill‑climb driven by the gradient of the currently worst shift."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


# ----------------------------------------------------------------------
# smooth‑max objective and its gradient (used for fast descent)
# ----------------------------------------------------------------------
def softmax_obj_vec(h: np.ndarray, beta: float) -> float:
    O = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(O)
    return (m + np.log(np.sum(np.exp(beta * (O - m))))) / beta


def softmax_grad(h: np.ndarray, beta: float) -> np.ndarray:
    """Gradient of the soft‑max approximation of max_k O_k."""
    n = h.size
    O = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(O)
    exp_beta = np.exp(beta * (O - m))
    w = exp_beta / exp_beta.sum()                     # softmax weights, sum to 1
    grad = np.ones_like(h)                           # constant term (∂O/∂h_i contains +1)
    for idx, weight in enumerate(w):
        shift = idx - (n - 1)
        if shift == 0:
            grad -= 2.0 * weight * h
        elif shift > 0:
            grad[: n - shift] -= weight * h[shift:]
            grad[shift:]     -= weight * h[: n - shift]
        else:  # shift < 0
            s = -shift
            grad[s:]         -= weight * h[: n - s]
            grad[: n - s]    -= weight * h[s:]
    return grad


def gradient_descent_softmax(h0: np.ndarray, rng: np.random.Generator,
                            beta_start: float = 0.5, beta_factor: float = 1.5,
                            beta_max: float = 2000.0,
                            step_start: float = 0.12, step_decay: float = 0.7,
                            max_iter_per_beta: int = 150) -> tuple[np.ndarray, float]:
    """Simple projected gradient descent on the soft‑max objective."""
    h = h0.copy()
    n = h.shape[0]
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    beta = beta_start
    step = step_start
    while beta <= beta_max:
        for _ in range(max_iter_per_beta):
            g = softmax_grad(h, beta)
            nh = h - step * g
            nh = np.clip(nh, 0.0, 1.0)
            nh = enforce_sum(nh, target=target, rng=rng)
            c = compute_c5(nh)
            if c < best_c:
                best_c, best_h, h = c, nh.copy(), nh
        beta *= beta_factor
        step *= step_decay
    return best_h, best_c


# ----------------------------------------------------------------------
# SLSQP refinement (kept from the parent code)
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x: np.ndarray) -> float:
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj_slsqp(x: np.ndarray) -> float:
    O = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(O)
    return (m + np.log(np.sum(np.exp(CURRENT_BETA * (O - m))))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 400) -> np.ndarray:
    """SLSQP optimisation of the soft‑max objective for a fixed β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(_softmax_obj_slsqp, h0, method="SLSQP", bounds=bounds,
                   constraints=cons, options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    x = enforce_sum(x, target=CURRENT_TARGET, rng=rng)
    return x
