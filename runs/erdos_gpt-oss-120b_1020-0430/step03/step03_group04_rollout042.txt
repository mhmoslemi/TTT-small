    n = len(h)
    L = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1-h, n=L)

        if k == 0:
            # gradient of max mis (k=0) is (1-2h)
            g[:] = 1.0 - 2.0 * h
        else:
            # compute gradient of mis for shift k (positive direction)
            # For mis_k = sum_i h[i] * (1 - h[i+k]) for i where i+k < n
            # gradient w.r.t h[j] is:
            # For j in [0, n-k-1] (i part): derivative is (1 - h[j+k]) (from term where i=j)
            # For j in [k, n-1] (i+k part): derivative is -h[j-k] (from term where i=j-k)
            # So add contributions accordingly.
            # Equivalent to:
            # g[j] = (1 - h[j+k]) if j < n-k else 0
            # g[j] -= h[j-k] if j >= k else 0
            upper = n - k
            g[:upper] += 1.0 - h[k: k+upper]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        # Clip
        h = np.clip(h, 0.0, 1.0)
        # Adjust sum to target
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):

import time
import numpy as np

# ---------- correlation utilities ----------
def compute_corr_fft(h: np.ndarray) -> np.ndarray:
    """Linear correlation of h with (1‑h). Returns length 2·n‑1 array."""
    n = h.shape[0]
    rev = (1.0 - h)[::-1]
    L = 1 << ((2 * n - 1).bit_length())      # next power of two
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(rev, n=L)
    conv = np.fft.irfft(H * G, n=L)
    return conv[: 2 * n - 1]


def compute_c5_fast(h: np.ndarray) -> float:
    """C5 = (2/n) * max correlation, using the fast FFT routine."""
    n = h.shape[0]
    corr = compute_corr_fft(h)
    return float((2.0 / n) * np.max(corr))


# ---------- deterministic greedy improvement ----------
def improve_by_swaps(h: np.ndarray, max_iters: int, rng: np.random.Generator,
                    start_time: float, budget_s: float) -> np.ndarray:
    """Repeatedly reduce the worst shift by moving a 1 into the overlap region."""
    n = h.size
    for _ in range(max_iters):
        if time.time() - start_time > budget_s * 0.96:
            break
        corr = compute_corr_fft(h)
        max_idx = int(np.argmax(corr))
        max_mis = corr[max_idx]
        if max_mis <= 0:
            break
        k = max_idx - (n - 1)                     # signed shift
        off = abs(k)
        L = n - off
        if L <= 0:
            break
        if k >= 0:
            miss = (h[:L] == 1) & (h[off:off + L] == 0)
        else:
            miss = (h[:L] == 0) & (h[off:off + L] == 1)
        miss_idx = np.where(miss)[0]
        if miss_idx.size == 0:
            break
        i = miss_idx[0]
        src_candidates = np.where(h[L:] == 1)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[0] + L
        if k >= 0:
            h[i + off] = 1.0
            h[src] = 0.0
        else:
            h[i] = 1.0
            h[src] = 0.0
    return h


# ---------- simulated annealing with swaps ----------
def simulated_annealing_swap(h: np.ndarray, rng: np.random.Generator,
                            start_time: float, budget_s: float,
                            max_iters: int) -> tuple[np.ndarray, float]:
    """Swap‑based SA keeping weight exactly n/2."""
    n = h.size
    cur = h.copy()
    cur_c = compute_c5_fast(cur)
    best_h, best_c = cur.copy(), cur_c
    T0 = max(cur_c * 0.1, 1e-4)

    for it in range(max_iters):
        if time.time() - start_time > budget_s * 0.96:
            break
        T = T0 * (0.9995 ** it)
        # propose a swap 1↔0
        i = rng.integers(n)
        while cur[i] != 1.0:
            i = rng.integers(n)
        j = rng.integers(n)
        while cur[j] != 0.0 or j == i:
            j = rng.integers(n)
        cur[i], cur[j] = 0.0, 1.0
        new_c = compute_c5_fast(cur)
        accept = False
        if new_c < cur_c:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-(new_c - cur_c) / T):
                accept = True
        if accept:
            cur_c = new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur.copy()
        else:
            cur[i], cur[j] = 1.0, 0.0
    return best_h, best_c


# ---------- gradient refinement (continuous) ----------
def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                   step: float = 0.02, max_iter: int = 12) -> np.ndarray:
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = compute_corr_fft(h)
        idx = int(np.argmax(corr)) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:k + upper]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                ii = rng.integers(n)
                add = min(diff, 1.0 - h[ii]) if diff > 0 else max(diff, -h[ii])
                h[ii] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


# ---------- rotation‑based binary start ----------
def rotation_start(n: int, alpha: float, phase: float, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    frac = (i * alpha + phase) % 1.0
    h = (frac < 0.5).astype(float)
    target = n // 2
    cur = int(h.sum())
    diff = target - cur
    if diff > 0:
        zeros = np.where(h == 0)[0]
        choose = rng.choice(zeros, diff, replace=False)
        h[choose] = 1.0
    elif diff < 0:
        ones = np.where(h == 1)[0]
        choose = rng.choice(ones, -diff, replace=False)
        h[choose] = 0.0
    return h
