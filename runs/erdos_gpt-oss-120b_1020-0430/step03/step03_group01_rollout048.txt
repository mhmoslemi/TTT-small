def targeted_move(h, shift, rng, scale=0.5):

    n = len(h)
    # pick i uniformly where i+shift in bounds
    if shift >= 0:
        i = rng.integers(0, n - shift)
        j = rng.integers(0, n)
    else:
        i = rng.integers(-shift, n)
        j = rng.integers(0, n)
    # ensure donor j not equal to i or i+shift
    while j == i or j == i+shift:
        j = rng.integers(0, n)
    # compute max delta
    max_delta = min(h[j] / 2.0, 1.0 - h[i], 1.0 - h[i+shift])
    if max_delta <= 0:
        return h, False
    delta = scale * max_delta * rng.random()
    new_h = h.copy()
    new_h[j] -= 2*delta
    new_h[i] += delta
    new_h[i+shift] += delta
    # bounds are enforced already
    return new_h, True

import time, math
import numpy as np
from scipy.optimize import minimize

# globals for the smooth surrogate used by SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth LogSumExp surrogate of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=2000):
    """Adjust h so that Σh = target while respecting 0≤h≤1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    # simple redistribution loop
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # final correction if needed
    if abs(diff) > 1e-12:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
    return h


def random_feasible_vector(n, rng):
    """Generate a random feasible vector (0≤h_i≤1, Σh_i=n/2)."""
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)


def pairwise_move(h, rng, scale=0.5):
    """Transfer ε from i to j (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)

    eps = scale * max_eps * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h, True

def hill_climb_simple(h_start, rng, max_iters=500, step=0.3):
    """Straightforward hill‑climb using pairwise moves."""
    h = h_start.copy()
    best_c = compute_c5(h)

def slsqp_optimize(h0, beta, max_iter=400):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum exactly after numerical drift
    x = enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

            best_h, best_c = new_h, new_c

        # occasional short hill‑climb
        if rng.integers(0, 1500) == 0:
            hc_h, hc_c = hill_climb_simple(best_h, rng, max_iters=200, step=0.25)
            if hc_c < best_c:
                best_h, best_c = hc_h, hc_c

    # -------------------------------------------------------------
    # 5. SLSQP refinement with increasing smoothness
    # -------------------------------------------------------------
    for beta in beta_list:
        if time.time() - start_time > budget_s * 0.95:
            break
        try:
            refined = slsqp_optimize(best_h, beta, max_iter=400)
            rc = compute_c5(refined)
            if rc < best_c:
                best_h, best_c = refined, rc
        except Exception:
            continue

    # -------------------------------------------------------------
    # 6. Final clean‑up
    # -------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, N / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(N)
