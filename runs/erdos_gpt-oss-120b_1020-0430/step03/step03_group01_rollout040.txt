    best = b.copy()
    best_c = compute_c5_binary(b)
    cur = b.copy()
    cur_c = best_c
    temp = init_temp
    for it in range(max_iter):
        # choose random swap
        i = rng.choice(one_indices)
        j = rng.choice(zero_indices)
        # perform swap
        cur[i], cur[j] = 0, 1
        new_c = compute_c5_binary(cur)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            cur_c = new_c
            # update indices lists:
            # swap i from ones to zeros, and j from zeros to ones
            # ...
            if cur_c < best_c:
                best_c = cur_c
                best = cur.copy()
        else:
            # revert swap
            cur[i], cur[j] = 1, 0
        temp *= cooling
        # optionally keep best and restart
    return best, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utility functions ----------
def compute_c5(h):
    """Exact C5 for a discretised step function."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=2000):
    """Project h onto [0,1] and adjust to hit exact sum `target`."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # final uniform correction if needed
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def random_feasible_vector(n, rng):
    """Generate a random feasible vector (0≤h≤1, sum=n/2)."""
    h = rng.random(n)  # uniform in [0,1)
    h = enforce_sum(h, n / 2.0, rng)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_multi(h_start, rng, max_iters=2000, init_step=0.5, n_trials=5):
    """Multi‑trial hill‑climb using perturb_h."""
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c = c
                h = new_h
                improved = True
                break
        if not improved and rng.random() < 0.02:
            new_h = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(new_h)
            if c < best_c:
                best_c = c
                h = new_h
                improved = True
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return h, best_c


# ---------- SLSQP surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum after numerical drift
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ---------- binary utilities ----------
def binary_from_continuous(h):
    """Return the binary vector with top N/2 entries set to 1."""
    n = h.size
    idx = np.argsort(-h)
    b = np.zeros(n, dtype=float)
    b[idx[: n // 2]] = 1.0
    return b


def random_binary(n, rng):
    b = np.zeros(n, dtype=float)
    ones = rng.choice(n, n // 2, replace=False)
    b[ones] = 1.0
    return b


def binary_improve(b0, rng, time_budget):
    """Directed + random hill‑climb on binary vectors."""
    start = time.time()
    b = b0.copy()
    best_b = b.copy()
    best_c = compute_c5(b)
    n = b.size

    while time.time() - start < time_budget:
        # autocorrelation
        auto = np.correlate(b, b, mode="full")
        # ignore shift 0 (index n-1), consider positive shifts only
        shift = np.argmin(auto[n:]) + 1  # 1 … n-1

        # construct shifted version for this shift
        shifted = np.empty_like(b)
        shifted[: n - shift] = b[shift:]
        shifted[n - shift :] = 0.0

        # indices that would increase autocorrelation for this shift
        i_cand = np.where((b == 1) & (shifted == 0))[0]
        j_cand = np.where((b == 0) & (shifted == 1))[0]

        if i_cand.size and j_cand.size:
            i = rng.choice(i_cand)
            j = rng.choice(j_cand)
            b[i], b[j] = 0.0, 1.0
            c = compute_c5(b)
            if c < best_c:
                best_c = c
                best_b = b.copy()
            else:
                # revert
                b[i], b[j] = 1.0, 0.0
        else:
            # fallback random swap
            i = rng.choice(np.where(b == 1)[0])
            j = rng.choice(np.where(b == 0)[0])
            b[i], b[j] = 0.0, 1.0
            c = compute_c5(b)
            if c < best_c:
                best_c = c
                best_b = b.copy()
            else:
                b[i], b[j] = 1.0, 0.0

    return best_b, best_c


# ---------- main driver ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ---- base dimension and initial vector ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    h0 = np.clip(h0, 0.0, 1.0)
    h0 = enforce_sum(h0, n / 2.0, rng)

    # ---- diverse start pool ----
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    candidates = [h0]
    for f in range(1, 9):
        s = np.clip(0.5 + 0.5 * np.sin(np.pi * f * xs), 0.0, 1.0)
        candidates.append(enforce_sum(s, n / 2.0, rng))
        c = np.clip(0.5 + 0.5 * np.cos(np.pi * f * xs), 0.0, 1.0)
        candidates.append(enforce_sum(c, n / 2.0, rng))
    candidates.append(enforce_sum(xs / 2.0, n / 2.0, rng))
    candidates.append(enforce_sum(1.0 - xs / 2.0, n / 2.0, rng))
    mask = np.zeros(n)
    mask[: n // 2] = 1.0
    candidates.append(enforce_sum(mask, n / 2.0, rng))
    mask2 = np.zeros(n)
    start = n // 4
    mask2[start : start + n // 2] = 1.0
    candidates.append(enforce_sum(mask2, n / 2.0, rng))
    candidates.append(enforce_sum(np.arange(n) % 2, n / 2.0, rng))
    for _ in range(4):
        candidates.append(random_feasible_vector(n, rng))

    best_h = h0.copy()
    best_c = compute_c5(best_h)
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # ---- continuous optimisation (hill‑climb + SLSQP) ----
    while time.time() - start_time < budget_s * 0.55:
        if rng.random() < 0.7:
            start_vec = perturb_h(best_h, rng, scale=0.4)
        else:
            start_vec = random_feasible_vector(n, rng)

        h_opt, c_opt = hill_climb_multi(
            start_vec, rng, max_iters=2500, init_step=0.5, n_trials=8
        )
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()

        for beta in [5, 20, 80, 200, 500, 1000]:
            if time.time() - start_time >= budget_s * 0.55:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, max_iter=400)
            except Exception:
                continue
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_c = c_ref
                best_h = h_ref.copy()

    # ---- optional up‑sampling if time permits ----
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > budget_s * 0.15:
        target_n = min(720, max(360, n * 2))
        if target_n != n:
            old_x = (np.arange(n) + 0.5) * (2.0 / n)
            new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
            h_up = np.interp(new_x, old_x, best_h)
            h_up = enforce_sum(h_up, target_n / 2.0, rng)

            h_up, c_up = hill_climb_multi(
                h_up,
                rng,
                max_iters=3000,
                init_step=0.3,
                n_trials=6,
            )
            for beta in [5, 20, 80, 200, 500, 1000]:
                if time.time() - start_time >= budget_s:
                    break
                try:
                    h_up = slsqp_optimize(h_up, beta, max_iter=400)
                except Exception:
                    pass
                c_up = compute_c5(h_up)
                if c_up < best_c:
                    best_c = c_up
                    best_h = h_up.copy()
                    n = target_n

    # ---- binary refinement ----
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > budget_s * 0.10:
        b0 = binary_from_continuous(best_h)
        b_best, c_bin = binary_improve(b0, rng, remaining * 0.7)
        if c_bin < best_c:
            best_c = c_bin
            best_h = b_best.astype(float)

        # one final SLSQP pass on the binary optimum (may introduce small fractions)
        try:
            h_ref = slsqp_optimize(best_h, 1200, max_iter=200)
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_c = c_ref
                best_h = h_ref.copy()
        except Exception:
            pass

    # ---- final clean‑up ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
