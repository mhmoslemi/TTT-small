def local_search(h0, rng, time_limit):
    n = h0.size
    target = n // 2
    # ensure binary
    h = np.rint(h0).astype(int)
    # repair to exact sum
    cur_sum = h.sum()
    while cur_sum != target:
        if cur_sum > target:
            idx = rng.choice(np.where(h == 1)[0])
            h[idx] = 0
            cur_sum -= 1
        else:
            idx = rng.choice(np.where(h == 0)[0])
            h[idx] = 1
            cur_sum += 1
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c

    # SA parameters based on time
    # We'll compute number of iterations relative to time
    start = time.time()
    # Determine iteration budget: at most (time_limit - elapsed)/0.0002 sec per iter? Hard estimate.
    # We'll just set max_iters = 50000
    max_iters = 50000
    # Temperature schedule
    T0 = cur_c * 0.5 + 0.01
    Tf = 1e-4
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        T = T0 * ((Tf / T0) ** (it / max_iters))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i], h[j] = h[j], h[i]
        new_c = compute_c5(h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = h[j], h[i]

    # Targeted improvement loop
    while time.time() - start < time_limit:
        corr = compute_corr_full(h)
        max_val = int(np.max(corr))
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        k, _ = find_worst_shift(corr)
        pair = find_beneficial_swap(h, k)
        if pair is None:
            # no immediate beneficial swap; break to avoid infinite loop
            break
        i, j = pair
        h[i], h[j] = h[j], h[i]
        # compute new max
        corr = compute_corr_full(h)
        new_c = np.max(corr) * (2.0 / n)
        if new_c < cur_c:
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = h[j], h[i]
            # break possibly
            break
    return best_h, best_c

import time
import numpy as np

def compute_c5(h):
    """Exact discrete C5 for a step function (dx=2/len(h))."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def compute_corr_full(h):
    """Full correlation array np.correlate(h,1-h)."""
    return np.correlate(h, 1 - h, mode="full")

def find_worst_shift(corr):
    """Return shift k (relative to zero) giving max correlation and its value."""
    n = (len(corr) + 1) // 2
    idx = int(np.argmax(corr))
    k = idx - (n - 1)
    return k, float(corr[idx])

def find_beneficial_swap(h, k):
    """Find a pair (i,j) to swap 1↔0 that reduces the max correlation for shift k.
    Returns None if no such pair exists."""
    n = h.size
    if k >= 0:
        a = h[: n - k]
        b = h[k :]
        base = 0
    else:
        kp = -k
        a = h[kp :]
        b = h[: n - kp]
        base = kp
    mism_i = np.where((a == 1) & (b == 0))[0]
    mism_j = np.where((a == 0) & (b == 1))[0]
    if mism_i.size == 0 or mism_j.size == 0:
        return None
    i = base + int(mism_i[0])
    j = base + int(mism_j[0])
    return i, j

def local_search(h0, rng, time_limit):
    """Simulated‑annealing + targeted improvement on a binary vector."""
    n = h0.size
    target = n // 2
    h = np.rint(h0).astype(int)
    # repair exact sum
    cur = h.sum()
    while cur != target:
        if cur > target:
            i = rng.choice(np.where(h == 1)[0])
            h[i] = 0
            cur -= 1
        else:
            i = rng.choice(np.where(h == 0)[0])
            h[i] = 1
            cur += 1

    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    start = time.time()

    # ---------- simulated annealing ----------
    max_iters = 50000
    T0 = cur_c * 0.5 + 0.01
    Tf = 1e-4
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        T = T0 * ((Tf / T0) ** (it / max_iters))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = h[j], h[i]
        new_c = compute_c5(h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c = new_c
            if new_c < best_c:
                best_c, best_h = new_c, h.copy()
        else:
            h[i], h[j] = h[j], h[i]

    # ---------- targeted improvement ----------
    while time.time() - start < time_limit:
        corr = compute_corr_full(h)
        max_val = float(np.max(corr))
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c, best_h = cur_c, h.copy()
        k, _ = find_worst_shift(corr)
        pair = find_beneficial_swap(h, k)
        if pair is None:
            break
        i, j = pair
        h[i], h[j] = h[j], h[i]
        corr2 = compute_corr_full(h)
        new_c = np.max(corr2) * (2.0 / n)
        if new_c < cur_c:
            cur_c = new_c
            if new_c < best_c:
                best_c, best_h = new_c, h.copy()
        else:
            h[i], h[j] = h[j], h[i]
            break

    return best_h, best_c

def slsqp_refine(h, rng, max_iters=300):
    """Small SLSQP refinement using the smooth log‑sum‑exp surrogate."""
    try:
        from scipy.optimize import minimize
    except Exception:
        return h, compute_c5(h)

    n = h.size
    target = n / 2.0

    # start from a slightly perturbed continuous vector
    x0 = h.astype(float) + rng.normal(scale=0.01, size=n)
    x0 = np.clip(x0, 0.0, 1.0)
    diff = target - x0.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x0[i]) if diff > 0 else max(diff, -x0[i])
        if abs(add) < 1e-12:
            continue
        x0[i] += add
        diff = target - x0.sum()

    def sum_eq(x):
        return np.sum(x) - target

    def soft_obj(x, beta):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    cons = {"type": "eq", "fun": sum_eq}
    bounds = [(0.0, 1.0)] * n
    best_x, best_c = x0, compute_c5(x0)

    for beta in [10.0, 40.0, 120.0, 400.0]:
        try:
            res = minimize(lambda x: soft_obj(x, beta), best_x,
                           method="SLSQP", bounds=bounds,
                           constraints=cons,
                           options={"maxiter": max_iters,
                                    "ftol": 1e-12,
                                    "disp": False})
        except Exception:
            continue
        x = np.clip(res.x, 0.0, 1.0)

        diff = target - x.sum()
        for _ in range(500):

            i = rng.integers(0, n)

            if abs(add) < 1e-12:
                continue
            x[i] += add
            diff = target - x.sum()
        c_now = compute_c5(x)
        if c_now < best_c:
            best_c, best_x = c_now, x

    return best_x, best_c
