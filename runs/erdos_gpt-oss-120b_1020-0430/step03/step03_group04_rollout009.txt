def gradient_descent_refine(h, rng, max_iter=200, step=0.05):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        # compute correlation
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = idx
        g = np.zeros_like(h)
        if k >= 0:
            # left contribution
            g[:n - k] += 1.0 - h[k:]
            # right contribution
            g[k:] -= h[:n - k]
        else:
            kk = -k
            # left contribution (i+kk?) derived similarly
            g[kk:] += 1.0 - h[:n - kk]
            g[:n - kk] -= h[kk:]
        # projected gradient step
        h = h - step * g
        # clip to [0,1]
        np.clip(h, 0.0, 1.0, out=h)
        # adjust sum to target (simple redistribution)
        diff = target - h.sum()
        # use small iterative adjustment
        for _ in range(50):
            if abs(diff) < 1e-12:
                break
            # pick random index where we can add/subtract
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            if add != 0:
                h[i] += add
                diff = target - h.sum()
        # optionally reduce step
        step *= 0.995
    return h

import time, numpy as np

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def enforce_sum(h, target, rng, max_iter=2000):
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, h.size)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0:
            h[i] += add
            diff = target - h.sum()
    return h

def random_binary_vector(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n//2, replace=False)] = 1.0
    return h

def sinusoidal_vector(n):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    return 0.5 + 0.5 * np.cos(np.pi * x)

def random_continuous_vector(n, rng):
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    # clip and fix
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, target, rng)

def propose_move(h, rng, scale):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return None
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return None
    if rng.random() < 0.1:
        eps = max_eps  # full transfer (swap)
    else:
        eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
        eps = max(-max_eps, min(eps, max_eps))
    h[i] -= eps
    h[j] += eps
    return (i, j, eps)

def simulated_annealing(h0, rng, deadline):
    h = h0.copy()
    n = h.size
    target = n / 2.0
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    T0 = max(0.02 * cur_c, 1e-6)
    scale = 1.0
    it = 0
    while time.time() < deadline:
        # gradually shrink step and temperature
        if it % 5000 == 0 and it > 0:
            scale = max(scale * 0.9, 0.05)
        move = propose_move(h, rng, scale)
        if move is None:
            continue
        i, j, eps = move
        new_c = compute_c5(h)
        delta = new_c - cur_c
        accept = False
        if delta < 0:
            accept = True
        else:
            T = T0 * (0.9999 ** it) + 1e-12
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            # revert
            h[i] += eps
            h[j] -= eps
        it += 1
        # occasional sum correction (prevent drift)
        if it % 20000 == 0:
            diff = target - h.sum()
            if abs(diff) > 1e-12:
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
    return best_h, best_c

def gradient_descent_refine(h, rng, max_iter=200, step=0.05):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        k = np.argmax(corr) - (n - 1)   # shift
        g = np.zeros_like(h)
        if k >= 0:
            g[:n - k] += 1.0 - h[k:]
            g[k:] -= h[:n - k]
        else:
            kk = -k
            g[kk:] += 1.0 - h[:n - kk]
            g[:n - kk] -= h[kk:]
        h -= step * g
        np.clip(h, 0.0, 1.0, out=h)
        # fix sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # simple redistribution
            for _ in range(30):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                if add != 0:
                    h[i] += add
                    diff = target - h.sum()
                    if abs(diff) < 1e-12:
                        break
        step *= 0.995
    return h

import time

# -------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for step function h on [0,2] (uniform spacing)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator, max_iter: int = 2000) -> np.ndarray:
    """Adjust h in place so that sum(h) == target, staying in [0,1]."""
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, h.size)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
    return h


def random_binary_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h


def sinusoidal_vector(n: int) -> np.ndarray:
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    return 0.5 + 0.5 * np.cos(np.pi * x)


def random_continuous_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, target, rng)


def propose_move(h: np.ndarray, rng: np.random.Generator, scale: float):
    """Transfer ε from i to j (preserves sum). Returns (i,j,ε) or None."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)

    else:

        eps = max(-max_eps, min(eps, max_eps))
    h[i] -= eps
    h[j] += eps
    return (i, j, eps)

def gradient_descent_refine(h: np.ndarray, rng: np.random.Generator,
                           max_iter: int = 200, step: float = 0.05):
    """Project‑gradient steps against the worst shift."""
    n = h.size
