def gradient_softmax_max(h, beta):
    c = np.correlate(h, 1 - h, mode='full')
    # Ensure numeric stability
    bc = beta * c
    max_bc = np.max(bc)
    exp_bc = np.exp(bc - max_bc)  # length 2n-1
    w = exp_bc / exp_bc.sum()
    # Compute terms
    w_rev = w[::-1]
    termA = np.convolve(w_rev, 1 - h, mode='full')
    termB = np.convolve(w, h, mode='full')
    # slice index range
    n = h.size
    offset = n - 1
    grad = termA[offset:offset+n] - termB[offset:offset+n]
    return grad

def project_sum(h, target):
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # if diff > 0: need to increase sum
    if diff > 0:
        # capacity to increase:
        cap = np.sum(1.0 - h)
        if cap < 1e-12:
            return h
        # increase proportionally to capacity:
        inc = diff * (1.0 - h) / cap
        h = h + inc
        # after addition, some may exceed 1 due to rounding; clip and repeat recursively? But inc is computed such that sum exactly, but rounding could cause slight overshoot.
        # Clip and adjust recursively a few times.
    else:
        # diff < 0: need to decrease sum
        cap = np.sum(h)
        if cap < 1e-12:
            return h
        dec = diff * h / cap  # diff negative
        h = h + dec
    # Clip to [0,1] to avoid tiny errors
    np.clip(h, 0.0, 1.0, out=h)
    # After clipping, sum may be off by small epsilon; do a few corrective loops.
    for _ in range(5):
        err = target - h.sum()
        if abs(err) < 1e-12:
            break
        if err > 0:
            cap = np.sum(1.0 - h)
            if cap < 1e-12:
                break
            inc = err * (1.0 - h) / cap
            h += inc
        else:
            cap = np.sum(h)
            if cap < 1e-12:
                break
            dec = err * h / cap
            h += dec
        np.clip(h, 0.0, 1.0, out=h)
    return h

import time
import numpy as np

# ------------------------------------------------------------
# Helper functions
# ------------------------------------------------------------
def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2 / n)."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / h.size)


def project_sum(h, target, max_iter=10):
    """
    Adjust h in‑place so that sum(h) == target while keeping 0 ≤ h ≤ 1.
    Simple deterministic projection onto the affine hyperplane.
    """
    for _ in range(max_iter):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            cap = np.sum(1.0 - h)
            if cap < 1e-12:
                break
            inc = diff * (1.0 - h) / cap
            h = h + inc
        else:
            cap = np.sum(h)
            if cap < 1e-12:
                break
            dec = diff * h / cap
            h = h + dec
        np.clip(h, 0.0, 1.0, out=h)
    return h


def gradient_softmax_max(h, beta):
    """
    Gradient of the smooth max (log‑sum‑exp) of the overlap vector.
    Returns ∇_h  ( 1/β * log Σ_k e^{β c_k} ) where c_k = Σ_i h_i (1‑h_{i+k}).
    """
    n = h.size
    # correlation vector c_k
    c = np.correlate(h, 1.0 - h, mode="full")
    # softmax weights
    bc = beta * c
    m = np.max(bc)
    exp_bc = np.exp(bc - m)
    w = exp_bc / exp_bc.sum()            # length 2n‑1, w[idx] corresponds to shift s = idx-(n-1)

    # termA = Σ_s w_s (1‑h_{i+s})
    w_rev = w[::-1]
    termA_full = np.convolve(w_rev, 1.0 - h, mode="full")
    # termB = Σ_s w_s h_{i‑s}
    termB_full = np.convolve(w, h, mode="full")
    offset = n - 1
    termA = termA_full[offset : offset + n]
    termB = termB_full[offset : offset + n]
    return termA - termB
