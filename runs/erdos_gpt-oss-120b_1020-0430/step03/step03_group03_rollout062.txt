def greedy_optimize(h0, rng, time_limit):
    n = h0.size
    h = h0.copy()
    sum_target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    start = time.time()
    step_factor = 0.5
    stagnate = 0
    it = 0
    while time.time() - start < time_limit:
        # compute correlation and worst shifts
        sum_h = h.sum()
        self_corr = np.correlate(h, h, mode='full')
        corr = sum_h - self_corr
        max_corr_val = corr.max()
        # update current c if necessary
        cur_c = max_corr_val * (2.0 / n)
        if cur_c < best_c - 1e-12:
            best_c = cur_c
            best_h = h.copy()
            stagnate = 0
        else:
            stagnate += 1
        # compute worst shifts
        shift_idxs = np.where(corr == max_corr_val)[0]
        worst_shifts = shift_idxs - (n - 1)
        # compute gradient
        g = np.zeros_like(h)
        for k in worst_shifts:
            if k >= 0:
                m = n - k
                g[:m] += h[k:]
                g[k:] += h[:m]
            else:
                kk = -k
                m = n - kk
                g[kk:] += h[:m]
                g[:m] += h[kk:]
        u = np.argmin(g)
        v = np.argmax(g)
        if g[v] - g[u] < 1e-12:
            # random perturb
            # apply small random transfer
            i, j = rng.integers(0, n, size=2)
            if i != j:
                eps = min(h[i], 1 - h[j]) * rng.random() * 0.5
                if eps > 1e-12:
                    h[i] -= eps
                    h[j] += eps
            # occasional restart if stagnation large
            if stagnate > 2000:
                # restart from random feasible vector
                h = random_feasible_vector(n, rng)
                stagnate = 0
            continue
        # epsilon max
        eps_max = min(h[u], 1.0 - h[v])
        if eps_max <= 1e-12:
            # try opposite direction
            eps_max = min(h[v], 1.0 - h[u])
            if eps_max <= 1e-12:
                # random perturb to break
                i, j = rng.integers(0, n, size=2)
                eps = min(h[i], 1 - h[j]) * rng.random() * 0.2
                h[i] -= eps
                h[j] += eps
                continue
            else:
                u, v = v, u
        eps = eps_max * step_factor
        cand = h.copy()
        cand[u] -= eps
        cand[v] += eps
        cand_c = compute_c5(cand)
        if cand_c < cur_c - 1e-12:
            h = cand
            cur_c = cand_c
            # adjust step factor up slightly
            step_factor = min(step_factor * 1.05, 0.9)
            stagnate = 0
        else:
            # try smaller epsilon
            eps2 = eps * 0.5
            if eps2 > 1e-12:
                cand2 = h.copy()
                cand2[u] -= eps2
                cand2[v] += eps2
                cand2_c = compute_c5(cand2)
                if cand2_c < cur_c - 1e-12:
                    h = cand2
                    cur_c = cand2_c
                    step_factor = max(step_factor * 0.9, 0.05)
                    stagnate = 0
                else:
                    step_factor = max(step_factor * 0.8, 0.01)
            else:
                step_factor = max(step_factor * 0.8, 0.01)
        it += 1
        # occasional random restart if stuck
        if stagnate > 5000:
            # random restart preserving best solution's best components?
            h = random_feasible_vector(n, rng)
            stagnate = 0
    return best_h, best_c

import time, math, numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / h.size)

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    h = np.full(n, 0.5, dtype=float)
    for _ in range(n * 5):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    enforce_sum(h, n / 2.0, rng)
    return h

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # compute log-sum-exp in stable way
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {"type": "eq", "fun": sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

def compute_gradient(h, shifts):
    n = h.size
    g = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            m = n - k
            g[:m] += h[k:]
            g[k:] += h[:m]
        else:
            kk = -k
            m = n - kk
            g[kk:] += h[:m]
            g[:m] += h[kk:]
    return g

    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    step_factor = 0.5
    stagnate = 0
    start = time.time()
    while time.time() - start < time_limit:
        # correlation
        full_corr = np.correlate(h, 1 - h, mode='full')
        max_corr = full_corr.max()
        cur_c = max_corr * (2.0 / n)
        if cur_c < best_c - 1e-12:
            best_c = cur_c
            best_h = h.copy()
            stagnate = 0
        else:
            stagnate += 1
        # worst shifts
        idxs = np.where(full_corr == max_corr)[0]
        shifts = idxs - (n - 1)
        # gradient of sum of neighbor contributions
        g = compute_gradient(h, shifts)
        u = np.argmin(g)
        v = np.argmax(g)
        if g[v] - g[u] < 1e-12:
            # random small perturb
            i, j = rng.integers(0, n, size=2)
            if i != j:
                eps = min(h[i], 1 - h[j]) * rng.random() * 0.2
                if eps > 1e-12:
                    h[i] -= eps
                    h[j] += eps
            if stagnate > 2000:
                h = random_feasible_vector(n, rng)
                stagnate = 0
            continue
        eps_max = min(h[u], 1.0 - h[v])
        if eps_max <= 0:
            # try opposite
            eps_max = min(h[v], 1.0 - h[u])
            if eps_max <= 0:
                # random perturb
                i, j = rng.integers(0, n, size=2)
                eps = min(h[i], 1 - h[j]) * rng.random() * 0.2
                if eps > 1e-12:
                    h[i] -= eps
                    h[j] += eps

CURRENT_BETA = None

    """Adjust h in‑place so that sum(h)=target while 0≤h≤1."""
    diff = target - h.sum()

        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h
