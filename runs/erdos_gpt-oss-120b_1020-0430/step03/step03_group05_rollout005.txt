def compute_c5(h):
    n = h.size
    size = 1 << ((2*n-1).bit_length())
    # compute FFT with real-valued rfft.
    f_h = np.fft.rfft(h, n=size)
    f_y = np.fft.rfft((1-h)[::-1], n=size)
    conv = np.fft.irfft(f_h * f_y, n=size)[:2*n-1]
    max_corr = np.max(conv)
    # avoid small negative due to numeric, ensure >=0.
    return max_corr * (2.0 / n)

def hill_climb_time(h_start, rng, max_time, start_step=0.5, decay=0.99, min_step=1e-5):

    best = h.copy()

    start = time.time()
    it = 0
    while time.time() - start < max_time:

            h = new_h

def de_optimize(n, rng, max_time, init_vec=None, popsize=None, F=0.8, CR=0.5):

    target = n/2.0
    if popsize is None:
        popsize = min(200, max(30, n))  # maybe min(200, n)
    pop = []
    if init_vec is not None:
        pop.append(init_vec)
    while len(pop) < popsize:
        pop.append(random_feasible_vector(n, rng))
    pop = np.array(pop)
    scores = np.array([compute_c5(ind) for ind in pop])
    best_idx = scores.argmin()
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]

    start = time.time()
    
    # precompute index arrays for mutation quickly? We'll just random sample each iteration.

    while time.time() - start < max_time:
        for i in range(popsize):
            if time.time() - start >= max_time:
                break
            # mutation with three distinct indices not equal i
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, 3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            # Crossover
            cross = rng.random(n) < CR
            # ensure at least one true
            if not np.any(cross):

            trial = np.where(cross, mutant, pop[i])
            # project
            trial = project_to_feasible(trial, target)
            trial_score = compute_c5(trial)
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_h = trial.copy()
        # optional early stop if best haven't improved for many generations? Not needed.
    return best_h, best_score

import time
import numpy as np
from scipy.optimize import minimize

# globals for the SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth max via log-sum-exp on the overlap correlations."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def project_to_feasible(v, target, iters=60):
    """Project onto {0≤h≤1, Σh = target} by bisection on a shift."""
    lo, hi = -2.0, 2.0
    for _ in range(iters):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Generate a random vector satisfying 0≤h≤1 and Σh = n/2."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_to_feasible(v, n / 2.0)


def upsample_initial(to_n):
    """Interpolate global initial_h_values to length to_n (if present)."""
    init = np.array(initial_h_values, dtype=float)
    m = init.size
    xs_old = np.linspace(0.0, 2.0, m, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, to_n, endpoint=False)
    return np.interp(xs_new, xs_old, init)


def de_optimize(n, rng, max_time, init_vec=None, popsize=None, F=0.8, CR=0.5):
    """Simple DE with projection onto the feasible set."""
    target = n / 2.0
    if popsize is None:
        popsize = min(200, max(30, n))
    pop = []
    if init_vec is not None:
        pop.append(init_vec)
    while len(pop) < popsize:
        pop.append(random_feasible_vector(n, rng))
    pop = np.array(pop)
    scores = np.array([compute_c5(ind) for ind in pop])
    best_idx = scores.argmin()
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]

    start = time.time()
    while time.time() - start < max_time:
        for i in range(popsize):
            if time.time() - start >= max_time:
                break
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, 3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(0, n)] = True
            trial = np.where(cross, mutant, pop[i])
            trial = project_to_feasible(trial, target)
            trial_score = compute_c5(trial)
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_h = trial.copy()
    return best_h, best_score


def hill_climb_time(h_start, rng, max_time, start_step=0.5,
                    decay=0.97, min_step=1e-5):
    """Hill‑climb with a time budget."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    start = time.time()
    it = 0
    while time.time() - start < max_time:
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        it += 1
        if it % 500 == 0:
            step = max(step * decay, min_step)
    return best_h, best_c


def slsqp_optimize(h0, beta, rng, max_iter=100):
    """SLSQP refinement using the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # candidate discretisations (larger first)
    candidate_ns = [300, 270, 240, 210, 180, 150, 120, 90, 60]
    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        if n0 not in candidate_ns:
            candidate_ns = [n0] + candidate_ns

    for n in candidate_ns:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining < 10.0:
            break

        # allocate half of remaining time to this n (capped)
        time_for_n = min(remaining * 0.5, 400.0)
        target = n / 2.0

        # optional up‑sampled initial vector
        init_vec = None
        if "initial_h_values" in globals():
            vec = upsample_initial(n)
            init_vec = project_to_feasible(vec, target)

        # ----- Differential Evolution -----
        de_time = time_for_n * 0.75
        if de_time < 1.0:
            de_time = 1.0
        h_de, c_de = de_optimize(n, rng, de_time, init_vec=init_vec)

        if c_de < best_c:
            best_c, best_h, best_n = c_de, h_de.copy(), n

        # ----- Local hill‑climb -----
        elapsed_de = time.time() - start_time - elapsed
        hc_time = max(0.0, time_for_n - elapsed_de)
        if hc_time > 0.5:
            h_hc, c_hc = hill_climb_time(h_de, rng, max_time=hc_time,
                                         start_step=0.5)
            if c_hc < best_c:
                best_c, best_h, best_n = c_hc, h_hc.copy(), n

        # ----- Short SLSQP refinement if time permits -----
        elapsed_n = time.time() - start_time - elapsed
        left_n = time_for_n - elapsed_n
        if left_n > 5.0:
            try:
                h_ref = slsqp_optimize(best_h, beta=500, rng=rng, max_iter=100)
                c_ref = compute_c5(h_ref)
                if c_ref < best_c:
                    best_c, best_h, best_n = c_ref, h_ref.copy(), n
            except Exception:
                pass

    # Fallback if nothing found
    if best_h is None:
        best_n = 60
        best_h = np.full(best_n, 0.5)

    # Final feasibility enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, best_n)
            if diff > 0:
                add = min(diff, 1.0 - best_h[i])
                best_h[i] += add
                diff -= add
            else:
                sub = min(-diff, best_h[i])
                best_h[i] -= sub
                diff += sub

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
