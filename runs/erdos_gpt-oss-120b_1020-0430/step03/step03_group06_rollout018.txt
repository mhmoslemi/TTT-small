    corr = np.correlate(arr, 1 - arr, mode='full')
    return np.max(corr) * (2.0 / arr.size)

def repair_binary(vec, target_sum, rng):
    # vec is binary array (0/1) maybe not exactly sum
    cur = vec.sum()
    if cur > target_sum:
        # too many ones: need to turn off cur-target_sum ones
        indices = np.where(vec == 1)[0]
        to_flip = rng.choice(indices, size=int(cur - target_sum), replace=False)
        vec[to_flip] = 0
    elif cur < target_sum:
        indices = np.where(vec == 0)[0]
        to_flip = rng.choice(indices, size=int(target_sum - cur), replace=False)
        vec[to_flip] = 1
    return vec

def crossover(p1, p2, rng):
    mask = rng.integers(0, 2, size=p1.size, dtype=bool)
    child = np.where(mask, p1, p2)
    return child

        return vec

    j = rng.choice(zeros)

def hill_climb(h_start, rng, max_iters, start_step):
    h = h_start.copy()
    best = h.copy()
    best_c = compute_c5(best)
    step = start_step
    n = h.size
    for it in range(max_iters):
        i = rng.integers(0,n)
        j = rng.integers(0,n)
        while j == i:
            j = rng.integers(0,n)
        # compute max allowed epsilon to stay in [0,1]
        max_pos = min(h[i], 1.0 - h[j])  # moving from i to j (decrease i, increase j)
        max_neg = min(h[j], 1.0 - h[i])  # moving from j to i (increase i, decrease j)
        if max_pos + max_neg == 0:
            continue
        eps = (rng.random() * (max_pos + max_neg) - max_neg) * step
        # update
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        # clip maybe unnecessary because eps respects bounds
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best = new_h
            h = new_h
        # optional: accept if no improvement? Might use simulated annealing.
        # simple hill-climb: accept only if improves.
        # adjust step
        if (it+1) % (max_iters//10) == 0:
            step = max(step * 0.5, 1e-4)
    return best, best_c

def binary_hill_climb(vec, rng, max_iters):

    best_c = compute_c5(best.astype(float))
    n = vec.size
    for _ in range(max_iters):

            break

        cand[i]=0
        cand[j]=1

            best = cand
            best_c = c

import time, numpy as np
from scipy.optimize import minimize

# ---------- helpers ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def _softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def enforce_sum_float(v, target, rng):
    diff = target - v.sum()
    n = v.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - v[i]) if diff > 0 else max(diff, -v[i])
        if abs(add) < 1e-12:
            continue
        v[i] += add
        diff = target - v.sum()
    if abs(diff) > 1e-12:
        for _ in range(1000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - v[i]) if diff > 0 else max(diff, -v[i])
            if abs(add) < 1e-12:
                continue
            v[i] += add
            diff = target - v.sum()
    return v

def repair_binary(v, target, rng):
    cur = v.sum()
    if cur > target:
        idx = np.where(v == 1)[0]
        to_flip = rng.choice(idx, size=int(cur - target), replace=False)
        v[to_flip] = 0
    elif cur < target:
        idx = np.where(v == 0)[0]
        to_flip = rng.choice(idx, size=int(target - cur), replace=False)
        v[to_flip] = 1
    return v

def crossover(p1, p2, rng):
    mask = rng.integers(0, 2, size=p1.size, dtype=bool)
    return np.where(mask, p1, p2)

def mutate_swap(v, rng):
    ones = np.where(v == 1)[0]
    zeros = np.where(v == 0)[0]
    if ones.size == 0 or zeros.size == 0:
        return v
    i = rng.choice(ones)
    j = rng.choice(zeros)
    v[i], v[j] = 0, 1
    return v

def binary_hill_climb(vec, rng, max_iters=200000):
    best = vec.copy()
    best_c = compute_c5(best.astype(float))
    n = vec.size
    for _ in range(max_iters):
        ones = np.where(best == 1)[0]
        zeros = np.where(best == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best.copy()
        cand[i], cand[j] = 0, 1
        c = compute_c5(cand.astype(float))
        if c < best_c:
            best, best_c = cand, c
    return best, best_c

def continuous_hill_climb(h0, rng, max_iters=5000, start_step=0.5):
    h = h0.copy()
    best = h.copy()
    best_c = compute_c5(best)
    step = start_step
    n = h.size
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)
        max_pos = min(h[i], 1.0 - h[j])
        max_neg = min(h[j], 1.0 - h[i])
        if max_pos + max_neg == 0:
            continue
        eps = (rng.random() * (max_pos + max_neg) - max_neg) * step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c = compute_c5(new_h)
        if c < best_c:
            best, best_c = new_h, c
            h = new_h
        if (it + 1) % decay == 0:
            step = max(step * 0.5, 1e-4)
    return best, best_c

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(_softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            if abs(add) < 1e-12:
                continue
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # dimension
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 180
        h0 = np.full(n, 0.5)
    target = n // 2

    # initial population (binary)
    pop_sz = 120
    pop = np.empty((pop_sz, n), dtype=np.uint8)
    for i in range(pop_sz):
        inds = rng.choice(n, size=target, replace=False)
        v = np.zeros(n, dtype=np.uint8)
        v[inds] = 1
        pop[i] = v
    # seed with rounded initial vector
    bin0 = (h0 >= 0.5).astype(np.uint8)
    pop[0] = repair_binary(bin0, target, rng)

    # evaluate
    fitness = np.empty(pop_sz, dtype=float)
    for i in range(pop_sz):
        fitness[i] = compute_c5(pop[i].astype(float))
    best_idx = int(np.argmin(fitness))
    best_vec = pop[best_idx].copy()
    best_c = fitness[best_idx]

    # GA loop
    tournament_k = 3
    crossover_rate = 0.8
    mutation_rate = 0.2
    while time.time() - start < budget_s * 0.55:
        new_pop = np.empty_like(pop)
        new_pop[0] = best_vec  # elitism
        for i in range(1, pop_sz):
            # parent selection
            cand = rng.choice(pop_sz, size=tournament_k, replace=False)
            p1 = pop[cand[np.argmin(fitness[cand])]]
            cand = rng.choice(pop_sz, size=tournament_k, replace=False)
            p2 = pop[cand[np.argmin(fitness[cand])]]
            child = p1.copy()
            if rng.random() < crossover_rate:
                child = crossover(p1, p2, rng)
            child = repair_binary(child, target, rng)
            if rng.random() < mutation_rate:
                child = mutate_swap(child, rng)
            new_pop[i] = child
        # evaluate new generation
        for i in range(pop_sz):
            fitness[i] = compute_c5(new_pop[i].astype(float))
        pop = new_pop
        cur_idx = int(np.argmin(fitness))
        cur_c = fitness[cur_idx]
        if cur_c < best_c:
            best_c = cur_c
            best_vec = pop[cur_idx].copy()
            if best_c < 0.3808:
                break

    # binary hill climb
    if time.time() - start < budget_s * 0.70:
        bh_vec, bh_c = binary_hill_climb(best_vec, rng, max_iters=100000)
        if bh_c < best_c:
            best_c, best_vec = bh_c, bh_vec

    # continuous refinement
    if time.time() - start < budget_s * 0.80:
        h_float = best_vec.astype(float)
        ch_vec, ch_c = continuous_hill_climb(h_float, rng, max_iters=4000, start_step=0.5)
        if ch_c < best_c:
            best_c, best_vec = ch_c, ch_vec.astype(np.uint8)  # keep binary for next steps

    # SLSQP fine‑tuning (fractional)
    if time.time() - start < budget_s * 0.90:
        h_cont = best_vec.astype(float)
        beta_vals = [10.0, 40.0, 120.0, 400.0, 800.0]
        for beta in beta_vals:
            if time.time() - start > budget_s * 0.90:
                break
            try:
                h_cont = slsqp_optimize(h_cont, beta, max_iter=500)
            except Exception:
                continue
            c = compute_c5(h_cont)
            if c < best_c:
                best_c, best_vec = c, h_cont.copy()

    # optional up‑sampling
    if time.time() - start < budget_s * 0.97 and n * 2 <= 360:
        up = np.repeat(best_vec, 2)
        n_up = up.size
        target_up = n_up // 2
        up = enforce_sum_float(up.astype(float), target_up, rng)
        # a quick hill‑climb on upsampled grid
        up, up_c = continuous_hill_climb(up, rng, max_iters=3000, start_step=0.3)
        if up_c < best_c:
            best_c, best_vec, n = up_c, up, n_up

    # final clean‑up
    best_vec = np.clip(best_vec, 0.0, 1.0)
    if best_vec.dtype != float:
        best_vec = best_vec.astype(float)
    best_vec = enforce_sum_float(best_vec, n / 2.0, rng)
    final_c = compute_c5(best_vec)

    return best_vec.tolist(), float(final_c), int(n)
