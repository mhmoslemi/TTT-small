def binary_sa_optimize(n, rng, max_iter, initial_mask=None):
    if initial_mask is None:
        mask = np.zeros(n, dtype=np.float64)
        mask[:n//2] = 1.0
        rng.shuffle(mask)
    else:
        mask = initial_mask.astype(np.float64)
    best_mask = mask.copy()
    best_c = compute_c5_fft(mask)
    cur_mask = mask.copy()
    cur_c = best_c
    # maintain sets of indices
    ones = np.flatnonzero(cur_mask > 0.5)
    zeros = np.flatnonzero(cur_mask < 0.5)
    # schedule: start temp larger and decay.
    T0 = 0.01  # maybe bigger? We'll use 0.005? We'll adjust later.
    for it in range(max_iter):
        # choose random i from ones, j from zeros
        i = rng.integers(0, ones.size)
        j = rng.integers(0, zeros.size)
        idx_i = ones[i]
        idx_j = zeros[j]
        # swap in new mask
        cur_mask[idx_i] = 0.0
        cur_mask[idx_j] = 1.0
        new_c = compute_c5_fft(cur_mask)
        delta = new_c - cur_c
        # temperature
        T = T0 * (1 - it / max_iter)
        accept = delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12))
        if accept:
            cur_c = new_c
            # update best
            if new_c < best_c:
                best_c = new_c
                best_mask = cur_mask.copy()
            # update indices arrays
            # swap positions in ones and zeros arrays efficiently: 
            # replace idx_i in ones with idx_j? but idx_i becomes zero, idx_j becomes one
            # So we can assign: ones[i] = idx_j; zeros[j] = idx_i
            ones[i] = idx_j
            zeros[j] = idx_i
        else:
            # revert swap
            cur_mask[idx_i] = 1.0
            cur_mask[idx_j] = 0.0
        # optional early exit if time limit early
    return best_mask, best_c

def binary_sa_optimize(n, rng, max_iter=500000, time_limit=None):
    """SA on {0,1} vectors with exactly n/2 ones."""
    mask = np.zeros(n, dtype=np.float64)
    mask[: n // 2] = 1.0
    rng.shuffle(mask)
    best_mask = mask.copy()
    best_c = compute_c5_fft(mask)
    cur_c = best_c
    ones = np.flatnonzero(mask > 0.5)
    zeros = np.flatnonzero(mask < 0.5)
    T0 = 0.02
    for it in range(max_iter):
        if time_limit is not None and time.time() >= time_limit:
            break
        i_idx = rng.integers(0, len(ones))
        j_idx = rng.integers(0, len(zeros))
        idx_i = ones[i_idx]
        idx_j = zeros[j_idx]

        # swap
        mask[idx_i] = 0.0
        mask[idx_j] = 1.0
        new_c = compute_c5_fft(mask)
        delta = new_c - cur_c
        T = T0 * (1.0 - it / max_iter)
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_mask = mask.copy()
            # update index lists
            ones[i_idx] = idx_j
            zeros[j_idx] = idx_i
        else:
            # revert
            mask[idx_i] = 1.0
            mask[idx_j] = 0.0
    # final accurate evaluation
    best_c = compute_c5(best_mask)
    return best_mask, best_c

        h0 = random_feasible_vector(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ---- binary simulated annealing ----
    sa_time_limit = start_time + budget_s * 0.35
    mask, mask_c = binary_sa_optimize(n, rng, max_iter=500000, time_limit=sa_time_limit)
    if mask_c < best_c:
        best_h, best_c = mask.copy(), mask_c

        # optional smoothing of binary mask
        kernel = np.exp(-0.5 * ((np.arange(-3, 4)) / 1.5) ** 2)
        kernel /= kernel.sum()
        smooth_h = np.convolve(mask, kernel, mode="same")
        smooth_h = np.clip(smooth_h, 0.0, 1.0)
        smooth_h = enforce_sum(smooth_h, n / 2.0, rng)
        smooth_c = compute_c5(smooth_h)
        if smooth_c < best_c:
            best_h, best_c = smooth_h, smooth_c

    # ---- refinement by hill‑climbing ----
    remaining = deadline - time.time()
    if remaining > 30.0:
        max_hc_iters = int(min(8000, remaining * 5))
        h_opt, c_opt = hill_climb_multi(best_h, rng, max_iters=max_hc_iters,
                                         init_step=0.5, n_trials=8)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # ---- SLSQP ladder (smooth surrogate) ----
    remaining = deadline - time.time()
    if remaining > 15.0:
        beta_list = [20, 80, 200, 500, 1000, 2000]
        for beta in beta_list:
            if time.time() >= deadline:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, max_iter=400)
            except Exception:
                continue
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_h, best_c = h_ref, c_ref

    # ---- final short hill‑climb for polishing ----
    remaining = deadline - time.time()
    if remaining > 10.0:
        iters = int(min(3000, remaining * 5))
        h_final, c_final = hill_climb_multi(best_h, rng, max_iters=iters,
                                            init_step=0.2, n_trials=6)
        if c_final < best_c:
            best_h, best_c = h_final, c_final

    # ---- final clean‑up ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
