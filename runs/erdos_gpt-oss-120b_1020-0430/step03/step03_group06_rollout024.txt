def targeted_sa(h_start, rng, total_time, start_time):
    h = h_start.copy()
    n = h.size
    cur_c = np.correlate(h, 1 - h, mode='full')
    cur_max = int(cur_c.max())
    best_h = h.copy()
    best_score = cur_max

    # cooling schedule: linear from Tmax to Tmin
    Tmax = 5.0
    Tmin = 0.01
    while time.time() - start_time < total_time * 0.85:
        elapsed = time.time() - start_time
        frac = elapsed / (total_time * 0.85)
        T = Tmax * (1 - frac) + Tmin * frac

        # pick worst shift
        worst_idxs = np.flatnonzero(cur_c == cur_max)
        if worst_idxs.size == 0:
            break
        shift_idx = rng.choice(worst_idxs)
        shift = shift_idx - (n - 1)

        # get bad indices
        if shift >= 0:
            i_range = np.arange(n - shift)
            bad_i = i_range[(h[i_range] == 1) & (h[i_range + shift] == 0)]

import time
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Global variables for the smooth surrogate used by SLSQP
# --------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None

def _sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET

def _softmax_obj(x):
    """Smooth log‑sum‑exp surrogate of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def compute_c5(h):
    """Exact discrete C₅ for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def compute_max_overlap(h):
    """Maximum overlap count (before multiplying by dx)."""
    return int(np.max(np.correlate(h, 1 - h, mode="full")))

def enforce_sum_float(h, target, rng):
    """Tiny repair so that Σ h_i = target while 0≤h_i≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        if abs(add) < 1e-12:
            continue
        h[i] += add
        diff = target - h.sum()
    return h

def slsqp_optimize(h0, beta, rng, max_iter=300):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny exact‑sum repair
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            if abs(add) < 1e-12:
                continue
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def _bad_indices_for_shift(h, shift):
    """Indices i where h[i]=1 and h[i+shift]=0 (if within bounds)."""
    n = h.size
    if shift >= 0:
        i_range = np.arange(n - shift)
        bad = i_range[(h[i_range] == 1) & (h[i_range + shift] == 0)]
    else:
        s = -shift
        i_range = np.arange(s, n)
        bad = i_range[(h[i_range] == 1) & (h[i_range - s] == 0)]
    return bad

def targeted_sa(h_start, rng, budget_s, start_time):
    """Simulated annealing that preferentially fixes the worst shift."""
    h = h_start.copy()
    n = h.size
    cur_corr = np.correlate(h, 1 - h, mode="full")
    cur_max = int(cur_corr.max())
    best_h = h.copy()
    best_score = cur_max

    # annealing schedule parameters
    Tmax = 5.0
    Tmin = 0.01
    iters_without_improve = 0
    # use 85% of total budget for the SA phase
    limit = start_time + budget_s * 0.85

    while time.time() < limit:
        # linear cooling
        progress = (time.time() - start_time) / (budget_s * 0.85)
        T = Tmax * (1 - progress) + Tmin * progress

        # determine a worst shift
        worst_idxs = np.flatnonzero(cur_corr == cur_max)
        shift_idx = rng.choice(worst_idxs)
        shift = shift_idx - (n - 1)

        # pick a offending 1 if possible, else any 1
        bad = _bad_indices_for_shift(h, shift)
        if bad.size:
            i = rng.choice(bad)
        else:
            i = rng.choice(np.where(h == 1)[0])

        # pick a random 0
        j = rng.choice(np.where(h == 0)[0])
        if i == j:
            continue

        # perform the swap
        h[i] = 0
        h[j] = 1

        new_corr = np.correlate(h, 1 - h, mode="full")
        new_max = int(new_corr.max())
        delta = new_max - cur_max

        accept = False
        if delta <= 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True

        if accept:
            cur_corr = new_corr
            cur_max = new_max
            if cur_max < best_score:
                best_score = cur_max
                best_h = h.copy()
                iters_without_improve = 0
                if best_score <= 34:          # early success
                    break
            else:
                iters_without_improve += 1
        else:
            # revert swap
            h[i] = 1
            h[j] = 0
            iters_without_improve += 1

        # occasional restart to avoid stagnation
        if iters_without_improve > 8000:
            # restart from the current best (or a fresh random one)
            if rng.random() < 0.5:
                h = best_h.copy()
            else:
                h = np.zeros(n, dtype=int)
                ones = rng.choice(n, size=n // 2, replace=False)
                h[ones] = 1
            cur_corr = np.correlate(h, 1 - h, mode="full")
            cur_max = int(cur_corr.max())
            iters_without_improve = 0

    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # --------------------------------------------------------------
    # Choose dimension and an initial binary vector

    if "initial_h_values" in globals():

    else:

    # Phase 1 – targeted simulated annealing on the integer problem
    # --------------------------------------------------------------
    best_h_int, best_score = targeted_sa(h0, rng, budget_s, start_time)
    best_c5 = best_score * (2.0 / n)

    # early exit if already good enough
    if best_c5 < 0.3808 and (time.time() - start_time) > budget_s * 0.5:
        final_h = best_h_int.astype(float)
        final_h = enforce_sum_float(final_h, n / 2.0, rng)
        return final_h.tolist(), float(best_c5), int(n)

    # --------------------------------------------------------------
    # Phase 2 – smooth refinement with SLSQP (if time permits)
    # --------------------------------------------------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining > budget_s * 0.1:
        h_float = best_h_int.astype(float)
        beta_list = [10.0, 40.0, 120.0, 400.0, 800.0]
        for beta in beta_list:
            if time.time() - start_time > budget_s * 0.95:
                break

                h_float = slsqp_optimize(h_float, beta, rng, max_iter=500)
            except Exception:
                continue
            c5 = compute_c5(h_float)
            if c5 < best_c5:
                best_c5 = c5
                best_h_int = h_float.copy()

    # --------------------------------------------------------------
    # Final clean‑up: ensure exact sum and clip
    # --------------------------------------------------------------
    final_h = np.clip(best_h_int, 0.0, 1.0)
    final_h = enforce_sum_float(final_h, n / 2.0, rng)
    final_c5 = compute_c5(final_h)

    return final_h.tolist(), float(final_c5), int(n)
