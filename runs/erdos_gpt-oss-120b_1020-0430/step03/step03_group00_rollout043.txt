def gradient_shift(h, k):
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        # overlapping region i: 0..n-1-k
        i = np.arange(n - k)
        grad[i] += 1 - h[i + k]
        grad[i + k] -= h[i]
        # tail region i=n-k..n-1
        if k > 0:
            grad[n - k:] += 1
    else:
        kk = -k
        i = np.arange(kk, n)
        grad[i] += 1 - h[i - kk]
        grad[i - kk] -= h[i]
        # head region i=0..kk-1
        grad[:kk] += 1
    return grad

# ---------- helper utilities ----------
def enforce_sum(h, target, rng):
    """Adjust h in-place so that 0<=h<=1 and sum(h)=target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 20000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:  # need to increase somewhere
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i])
            add *= rng.random()
            h[i] += add
            diff -= add
        else:  # need to decrease somewhere
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i])
            sub *= rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    # final tiny fix if needed
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        if diff > 0:
            h[i] = min(1.0, h[i] + diff)
        else:
            h[i] = max(0.0, h[i] + diff)
    return h

def compute_c5_from_corr(corr, n):
    """C5 from correlation array."""
    return np.max(corr) * (2.0 / n)

def gradient_shift(h, k):
    """Gradient of C(k)=sum_i h_i(1-h_{i+k}) w.r.t. h (zero padding)."""
    n = h.shape[0]
    grad = np.zeros(n, dtype=float)
    if k >= 0:
        if k < n:
            i = np.arange(n - k)
            grad[i] += 1.0 - h[i + k]
            grad[i + k] -= h[i]
        if k > 0:
            grad[n - k :] += 1.0
    else:
        kk = -k
        if kk < n:
            i = np.arange(kk, n)
            grad[i] += 1.0 - h[i - kk]
            grad[i - kk] -= h[i]
        if kk > 0:
            grad[:kk] += 1.0
    return grad

    """
    Return the change to the full correlation array when moving eps mass
    from index i to j. Offsets = array of shifts corresponding to corr indices.
    """
    n = h.size
    pad = n - 1
    # zero‑padded version of h
    h_pad = np.concatenate([np.zeros(pad), h, np.zeros(pad)])
    base = pad
    # vectorised lookup
    term_i_plus = h_pad[i + offsets + base]
    term_j_plus = h_pad[j + offsets + base]
    term_i_minus = h_pad[i - offsets + base]
    term_j_minus = h_pad[j - offsets + base]
    delta = eps * ((term_i_plus - term_j_plus) + (term_i_minus - term_j_minus))
    return delta

# ---------- main optimisation ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    budget = budget_s

    # ---- initialise -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        m = h0.size
        # target size (>=720, <1000)
        n = max(720, m)
    else:
        n = 720
        h0 = rng.random(n)

    if h0.size != n:
        xs0 = (np.arange(h0.size) + 0.5) * (2.0 / h0.size)
        xs = (np.arange(n) + 0.5) * (2.0 / n)
        h = np.interp(xs, xs0, h0)
    else:
        h = h0.copy()

    h = np.clip(h, 0.0, 1.0)
    target_sum = n / 2.0
    h = enforce_sum(h, target_sum, rng)

    # correlation and offsets
    corr = np.correlate(h, 1.0 - h, mode="full")
    offsets = np.arange(n - 1, -n, -1)  # length 2n-1, maps idx -> shift

    # best solution trackers
    best_h = h.copy()
    best_corr = corr.copy()
    best_c = compute_c5_from_corr(best_corr, n)

    # current state (starts at best)
    cur_h = h.copy()
    cur_corr = corr.copy()
    cur_c = best_c

    # temperature schedule
    T0 = 0.002
    it = 0

    # ---- simulated annealing loop -----------------------------------
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget * 0.99:
            break
        # decreasing temperature
        frac = elapsed / budget
        T = T0 * (1.0 - frac) ** 2 + 1e-12

        # decide move type
        if rng.random() < 0.85:
            # targeted move based on worst shift
            idx_max = np.argmax(cur_corr)
            k = (n - 1) - idx_max  # shift associated with idx_max
            grad = gradient_shift(cur_h, k)

            pos = np.where((grad > 0.0) & (cur_h > 1e-12))[0]
            neg = np.where((grad < 0.0) & (cur_h < 1.0 - 1e-12))[0]

            if pos.size and neg.size:
                i = pos[np.argmax(grad[pos])]
                j = neg[np.argmin(grad[neg])]
                max_eps = min(cur_h[i], 1.0 - cur_h[j])
                eps = max_eps * (0.2 + 0.8 * rng.random())
            else:
                # fallback random pair
                i = rng.integers(n)
                while cur_h[i] < 1e-12:
                    i = rng.integers(n)
                j = rng.integers(n)
                while cur_h[j] > 1.0 - 1e-12:
                    j = rng.integers(n)
                max_eps = min(cur_h[i], 1.0 - cur_h[j])
                eps = max_eps * rng.random() * 0.2
        else:
            # small random move
            i = rng.integers(n)
            while cur_h[i] < 1e-12:
                i = rng.integers(n)
            j = rng.integers(n)
            while cur_h[j] > 1.0 - 1e-12:
                j = rng.integers(n)
            max_eps = min(cur_h[i], 1.0 - cur_h[j])
            eps = max_eps * rng.random() * 0.05

        if eps <= 0.0:
            continue

        # evaluate move
        delta = delta_corr_move(i, j, eps, cur_h, offsets)
        new_corr = cur_corr + delta
        new_c = compute_c5_from_corr(new_corr, n)
        delta_c = new_c - cur_c

        accept = False
        if delta_c <= 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-delta_c / T):
                accept = True

        if accept:
            # apply move
            cur_h[i] -= eps
            cur_h[j] += eps
            # enforce numeric bounds
            cur_h[i] = max(0.0, min(1.0, cur_h[i]))
            cur_h[j] = max(0.0, min(1.0, cur_h[j]))
            cur_corr = new_corr
            cur_c = new_c

            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
                best_corr = cur_corr.copy()

        it += 1

        # occasional gentle perturbation of the current solution
        if it % 20000 == 0:
            for _ in range(8):
                src = rng.integers(n)
                while cur_h[src] < 1e-12:
                    src = rng.integers(n)
                dst = rng.integers(n)
                while cur_h[dst] > 1.0 - 1e-12:
                    dst = rng.integers(n)
                max_eps = min(cur_h[src], 1.0 - cur_h[dst])
                eps = max_eps * rng.random() * 0.1
                if eps <= 0.0:
                    continue
                delta = delta_corr_move(src, dst, eps, cur_h, offsets)
                cur_corr += delta
                cur_h[src] -= eps
                cur_h[dst] += eps
                cur_c = compute_c5_from_corr(cur_corr, n)
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = cur_h.copy()
                    best_corr = cur_corr.copy()

    # ---- final fine‑grained search ----------------------------------
    while time.time() - start_time < budget * 0.998:
        i = rng.integers(n)
        if best_h[i] < 1e-12:
            continue
        j = rng.integers(n)
        if best_h[j] > 1.0 - 1e-12:
            continue
        max_eps = min(best_h[i], 1.0 - best_h[j])
        eps = max_eps * rng.random() * 0.02
        if eps <= 0.0:
            continue
        delta = delta_corr_move(i, j, eps, best_h, offsets)
        new_corr = best_corr + delta
        new_c = compute_c5_from_corr(new_corr, n)
        if new_c < best_c:
            best_c = new_c
            best_h[i] -= eps
            best_h[j] += eps
            best_corr = new_corr

    # ---- clean‑up ----------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target_sum, rng)
    final_corr = np.correlate(best_h, 1.0 - best_h, mode="full")
    final_c = compute_c5_from_corr(final_corr, n)

    return best_h.tolist(), float(final_c), int(n)
