import time

# ---------- utilities ----------
def compute_c5(h):
    """Discrete C5 for step function h (dx=2/n)."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)


def project_onto_capped_simplex(v, target):
    """Projection onto {0≤x≤1, sum x = target} via bisection on λ."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(50):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        if x.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Uniform random point on the capped simplex."""
    v = rng.random(n)
    return project_onto_capped_simplex(v, n / 2.0)
