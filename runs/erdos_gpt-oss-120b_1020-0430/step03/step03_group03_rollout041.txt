import time, math, numpy as np

def compute_c5(h):
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / h.size)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # Determine problem size
    if 'initial_h_values' in globals():
        init_arr = np.array(initial_h_values, dtype=float)
        n = init_arr.size
        bin_init = (init_arr > 0.5).astype(int)
    else:
        n = 180
        bin_init = None
    target_ones = n // 2

    # Helper to enforce exact number of ones
    def enforce_binary(v):
        cur = int(v.sum())
        if cur > target_ones:
            # turn excess ones to zero
            ones = np.where(v == 1)[0]
            to_zero = rng.choice(ones, size=cur - target_ones, replace=False)
            v[to_zero] = 0
        elif cur < target_ones:
            zeros = np.where(v == 0)[0]
            to_one = rng.choice(zeros, size=target_ones - cur, replace=False)
            v[to_one] = 1
        return v

    # Build a few starting binary vectors
    starters = []

    if bin_init is not None:
        starters.append(enforce_binary(bin_init.copy()))

    # random binary
    rand_bin = np.zeros(n, dtype=int)
    rand_bin[rng.choice(n, size=target_ones, replace=False)] = 1
    starters.append(rand_bin)

    # cosine‑threshold pattern
    xs = np.linspace(0, 2, n, endpoint=False)
    cos_vals = (1 + np.cos(np.pi * xs)) / 2.0
    cos_bin = (cos_vals > 0.5).astype(int)
    starters.append(enforce_binary(cos_bin))

    # select the best of the starters
    best_h = None
    best_c = float('inf')
    for h in starters:
        c = compute_c5(h.astype(float))
        if c < best_c:
            best_c = c
            best_h = h.copy()

    cur_h = best_h.copy()
    cur_c = best_c
    best_over_h = best_h.copy()
    best_over_c = best_c

    # time budget allocation
    elapsed = time.time() - start
    remaining_budget = max(0.0, budget_s - elapsed)
    # number of iterations – generous upper bound, will stop early by time
    max_iter = int(min(500000, remaining_budget * 4000))

    T_start = 0.5
    T_end = 1e-5

    for it in range(max_iter):
        if time.time() - start > budget_s * 0.99:
            break
        # temperature
        T = T_start * ((T_end / T_start) ** (it / max_iter))

        # correlation of current state
        corr = np.correlate(cur_h, 1 - cur_h, mode='full')
        max_idx = int(np.argmax(corr))
        shift = max_idx - (n - 1)

        # overlapping indices for this shift
        if shift >= 0:
            i_range = np.arange(0, n - shift)
        else:
            i_range = np.arange(-shift, n)
        i_plus = i_range + shift

        hi = cur_h[i_range]
        hj = cur_h[i_plus]

        bad_mask = (hi == 1) & (hj == 0)
        good_mask = (hi == 0) & (hj == 1)

        if bad_mask.any() and good_mask.any():
            # targeted swap between a bad and a good position
            i_bad = i_range[bad_mask][rng.integers(bad_mask.sum())]
            i_good = i_range[good_mask][rng.integers(good_mask.sum())]
            new_h = cur_h.copy()
            new_h[i_bad], new_h[i_good] = new_h[i_good], new_h[i_bad]
        else:
            # random 1–0 swap
            ones = np.where(cur_h == 1)[0]
            zeros = np.where(cur_h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i_bad = rng.choice(ones)
            i_good = rng.choice(zeros)
            new_h = cur_h.copy()
            new_h[i_bad], new_h[i_good] = new_h[i_good], new_h[i_bad]

        new_c = compute_c5(new_h.astype(float))
        delta = new_c - cur_c
        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            cur_h = new_h
            cur_c = new_c
            if cur_c < best_over_c:
                best_over_c = cur_c
                best_over_h = cur_h.copy()

    # final evaluation
    final_h = best_over_h.astype(float)
    final_c = compute_c5(final_h)

    return final_h.tolist(), float(final_c), int(n)
