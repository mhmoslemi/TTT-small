def simulated_annealing(h_start, rng, max_iters, init_temp, final_temp):
    h = h_start.copy()
    c = compute_c5(h)
    best_h, best_c = h.copy(), c
    n = h.size
    for it in range(max_iters):
        t = init_temp * (final_temp / init_temp) ** (it / max_iters)
        # propose move
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        # propose epsilon scaled with temperature and max_transfer
        eps = (rng.random() - 0.5) * 2 * max_transfer * t
        eps = max(-max_transfer, min(eps, max_transfer))
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        delta = new_c - c
        if delta < 0 or rng.random() < np.exp(-delta / (t + 1e-12)):
            h, c = new_h, new_c
            if c < best_c:
                best_c, best_h = c, new_h.copy()
    return best_h, best_c
```

def de_optimize(n, rng, max_time, init_vec=None):
    target = n / 2.0
    popsize = max(30, min(80, n//4)) # depends
    # initialise population
    pop = []
    if init_vec is not None:
        # downsample or upsample to size n
        init_vec_n = upsample(initial_vec, n)
        pop.append(project_to_feasible(init_vec_n, target))
    while len(pop) < popsize:
        pop.append(random_feasible_vector(n, rng))
    pop = np.array(pop)
    scores = np.array([compute_c5(ind) for ind in pop])
    best_idx = scores.argmin()
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]
    F = 0.5
    CR = 0.9
    start = time.time()
    while time.time() - start < max_time:
        for i in range(popsize):
            # mutation
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, 3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            # ensure within bounds may be done later after projection
            # crossover
            cross = rng.random(n) < CR
            if not cross.any():
                cross[rng.integers(0, n)] = True
            trial = np.where(cross, mutant, pop[i])
            # project
            trial = project_to_feasible(trial, target)
            trial_score = compute_c5(trial)
            if trial_score < scores[i]:
                pop[i], scores[i] = trial, trial_score
                if trial_score < best_score:
                    best_score, best_h = trial_score, trial.copy()
        # optional early exit if no improvement for many generations
        # skip for simplicity
    return best_h, best_score

def upsample(old_h, new_n):

    xs_new = np.linspace(0.0, 2.0, new_n, endpoint=False)

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utility functions ----------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for step function h on [0,2] (dx = 2/len)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def project_to_feasible(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0≤h≤1, Σh = target} by bisection on a shift."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Fast random point in {0≤h≤1, Σh=n/2}."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_to_feasible(v, n / 2.0)


def upsample(old_h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of old_h (size m) to size new_n on [0,2)."""
    m = old_h.size
    xs_old = np.linspace(0.0, 2.0, m, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(xs_new, xs_old, old_h)


# ---------- DE ----------
def de_optimize(n: int, rng: np.random.Generator, max_time: float,
                init_vec: np.ndarray | None = None):
    """Simple Differential Evolution with projection."""
    target = n / 2.0
    popsize = max(30, min(80, n // 4))
    pop = []

    if init_vec is not None:
        init = project_to_feasible(upsample(init_vec, n), target)
        pop.append(init)

    while len(pop) < popsize:
        pop.append(random_feasible_vector(n, rng))

    pop = np.array(pop)
    scores = np.array([compute_c5(ind) for ind in pop])
    best_idx = scores.argmin()
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]

    F, CR = 0.5, 0.9
    start = time.time()
    while time.time() - start < max_time:
        for i in range(popsize):
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, 3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            cross = rng.random(n) < CR
            if not cross.any():
                cross[rng.integers(0, n)] = True
            trial = np.where(cross, mutant, pop[i])
            trial = project_to_feasible(trial, target)
            trial_score = compute_c5(trial)
            if trial_score < scores[i]:
                pop[i], scores[i] = trial, trial_score
                if trial_score < best_score:
                    best_score, best_h = trial_score, trial.copy()
    return best_h, best_score


# ---------- SLSQP (soft‑max surrogate) ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ---------- hill climbing ----------
def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start: np.ndarray, rng: np.random.Generator,
               max_iters: int = 5000, start_step: float = 0.3):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h, h = c, new_h.copy(), new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
        if time.time() - _run_start > _budget:
            break
    return best_h, best_c


# ---------- Simulated annealing ----------
def simulated_annealing(h_start: np.ndarray, rng: np.random.Generator,
                        max_iters: int, init_temp: float, final_temp: float):
    h = h_start.copy()
    c = compute_c5(h)
    best_h, best_c = h.copy(), c
    n = h.size
    for it in range(max_iters):
        if time.time() - _run_start > _budget:
            break
        t = init_temp * (final_temp / init_temp) ** (it / max_iters)
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_transfer * t
        eps = max(-max_transfer, min(eps, max_transfer))
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        delta = new_c - c
        if delta < 0 or rng.random() < np.exp(-delta / (t + 1e-12)):
            h, c = new_h, new_c
            if c < best_c:
                best_c, best_h = c, new_h.copy()
    return best_h, best_c


# ---------- run ----------
def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    global _run_start, _budget
    _run_start = time.time()
    _budget = budget_s

    rng = np.random.default_rng(seed)
    best_h = None
    best_c = np.inf
    best_n = None

    # Candidate discretisations (small to larger)
    candidate_ns = [120, 180, 240, 300, 360, 420]
    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        if n0 not in candidate_ns:
            candidate_ns = [n0] + candidate_ns
    candidate_ns = sorted(set(candidate_ns))

    for idx, n in enumerate(candidate_ns):
        if time.time() - _run_start > _budget - 5.0:
            break
        remaining = _budget - (time.time() - _run_start)
        # allocate a slice of the remaining time for this n
        time_for_n = min(300.0, max(30.0, remaining * 0.6 / (len(candidate_ns) - idx)))
        # split the budget
        de_time = time_for_n * 0.5
        slsqp_time = time_for_n * 0.2
        hill_time = time_for_n * 0.15
        sa_time = time_for_n * 0.15

        # initialise DE with upsampled previous best if available
        init_vec = None
        if best_h is not None:
            init_vec = upsample(best_h, n)

        h_de, c_de = de_optimize(n, rng, de_time, init_vec=init_vec)
        if c_de < best_c:
            best_c, best_h, best_n = c_de, h_de.copy(), n

        # SLSQP refinements (only if time permits)
        if time.time() - _run_start < _budget - 5.0:
            h_cur = h_de
            for beta in (10.0, 30.0, 80.0, 200.0, 500.0, 1500.0):
                if time.time() - _run_start > _budget - 5.0:
                    break
                h_cur = slsqp_optimize(h_cur, beta, rng, max_iter=300)
                c_cur = compute_c5(h_cur)
                if c_cur < best_c:
                    best_c, best_h, best_n = c_cur, h_cur.copy(), n

        # hill climbing (time‑bounded)
        if time.time() - _run_start < _budget - 5.0:
            # use remaining part of hill_time but also respect global budget
            hc_iters = int(5000 * (hill_time / 30.0)) + 1000
            h_hc, c_hc = hill_climb(best_h, rng,
                                    max_iters=hc_iters, start_step=0.3)
            if c_hc < best_c:
                best_c, best_h, best_n = c_hc, h_hc.copy(), n

        # simulated annealing (time‑bounded)
        if time.time() - _run_start < _budget - 5.0:
            sa_iters = int(20000 * (sa_time / 30.0)) + 2000
            sa_iters = min(sa_iters, 60000)
            h_sa, c_sa = simulated_annealing(best_h, rng,
                                             max_iters=sa_iters,
                                             init_temp=0.2, final_temp=0.001)
            if c_sa < best_c:
                best_c, best_h, best_n = c_sa, h_sa.copy(), n

    # final feasibility enforcement
    if best_h is None:
        best_n = 60
        best_h = np.full(best_n, 0.5)
    else:
        best_h = np.clip(best_h, 0.0, 1.0)
        target = best_n / 2.0
        diff = target - best_h.sum()
        for _ in range(10000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, best_n)
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
