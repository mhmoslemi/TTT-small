def project_onto_feasible(v, target):
    # Solve min ||x - v||^2 s.t. 0<=x<=1, sum x = target
    # Equivalent to find tau such that x_i = clip(v_i - tau, 0, 1) satisfies sum = target.
    # This is monotone decreasing in tau.
    lo = -np.max(v) - 1.0  # ensures sum = N (all clipped to 1)
    hi = np.max(v) + 1.0   # ensures sum = 0
    for _ in range(50):
        tau = (lo + hi) / 2.0
        x = np.clip(v - tau, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    x = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    # correct residual if tiny diff due to numeric
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        # distribute diff among free variables (not at bounds)
        free = (x > 1e-12) & (x < 1-1e-12)
        if free.any():
            adj = diff / free.sum()
            x[free] += adj
    # Clip again to be safe
    x = np.clip(x, 0.0, 1.0)
    return x

import numpy as np

# -------------------------------------------------------------
# Core utilities
# -------------------------------------------------------------
def correlate_full(h):
    """Linear correlation of h with (1-h) using FFT, zero‑padding."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())  # next power of two
    fh = np.fft.rfft(h, L)
    fr = np.fft.rfft((1 - h)[::-1], L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    return correlate_full(h).max() * (2.0 / h.size)


# -------------------------------------------------------------
# Projection onto [0,1] with prescribed sum (capped simplex)
# -------------------------------------------------------------
def project_onto_feasible(v, target):
    """L2‑projection of v onto {0≤x≤1, Σx=target}."""
    lo = -np.max(v) - 1.0   # ensures sum = n after clipping
    hi = np.max(v) + 1.0    # ensures sum = 0 after clipping
    for _ in range(50):
        tau = (lo + hi) * 0.5
        x = np.clip(v - tau, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction if needed
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        if free.any():
            x[free] += diff / free.sum()
        x = np.clip(x, 0.0, 1.0)
    return x


def random_feasible_vector(n, rng):
    """Random vector in [0,1]^n with sum = n/2 (via projection)."""
    v = rng.random(n)
    return project_onto_feasible(v, n / 2.0)


# -------------------------------------------------------------
# Local perturbation / hill‑climb helpers
# -------------------------------------------------------------
def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j, preserving the sum."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.4):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h, h = c, new_h.copy(), new_h
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


# -------------------------------------------------------------
# Smooth objective (log‑sum‑exp) and its gradient
# -------------------------------------------------------------
def obj_and_grad(h, beta, target):
    """Return smooth surrogate of max corr and its gradient."""
    corr = correlate_full(h)
    m = corr.max()
    exp_terms = np.exp(beta * (corr - m))
    sum_exp = exp_terms.sum()
    prob = exp_terms / sum_exp               # softmax weights, sum to 1
    obj = m + np.log(sum_exp) / beta        # smooth max

    N = h.size
    grad = np.ones(N)                       # derivative of Σh_i part
    # subtract weighted shifted copies of h
    for offset in range(-N + 1, N):
        p = prob[offset + N - 1]
        if p == 0.0:
            continue
        if offset >= 0:
            grad[: N - offset] -= p * h[offset:]
            grad[offset :]    -= p * h[: N - offset]
        else:
            d = -offset
            grad[d:]          -= p * h[: N - d]
            grad[: N - d]     -= p * h[d:]
    return obj, grad


# -------------------------------------------------------------
# Projected gradient descent with backtracking line search
# -------------------------------------------------------------
def pgd_optimize(h0, beta_schedule, target, deadline):
    """Run PGD on the smooth surrogate, respecting the time limit."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)

    for beta in beta_schedule:
        for _ in range(200):                     # fixed inner iterations
            if time.time() >= deadline:
                return best_h, best_c
            obj, grad = obj_and_grad(h, beta, target)

            # backtracking line search
            step = 1.0
            while True:
                h_new = project_onto_feasible(h - step * grad, target)
                obj_new, _ = obj_and_grad(h_new, beta, target)
                if obj_new <= obj or step < 1e-8:
                    break
                step *= 0.5

            h = h_new
            c_now = compute_c5(h)
            if c_now < best_c:
                best_c, best_h = c_now, h.copy()
    return best_h, best_c


# -------------------------------------------------------------
# Main entry point
# -------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.96   # keep a little slack

    # ---- initial feasible vector -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        N = h0.size
        target = N / 2.0
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_onto_feasible(h0, target)
    else:
        N = 360
        target = N / 2.0
        h0 = project_onto_feasible(rng.random(N), target)

    # ---- build a few starting candidates -----------------------------------------
    starts = [h0]
    for _ in range(4):
        starts.append(random_feasible_vector(N, rng))

    # ---- optimisation parameters --------------------------------------------------
    beta_sched = [5.0, 15.0, 30.0, 80.0, 200.0, 500.0]

    best_h = None
    best_c = np.inf

    # ---- optimise each start ------------------------------------------------------
    for cand in starts:
        h_opt, c_opt = pgd_optimize(cand, beta_sched, target, deadline)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # ---- final local refinement ---------------------------------------------------
    if time.time() < deadline:
        best_h, best_c = hill_climb(best_h, rng, max_iters=3000, start_step=0.5)

    # ---- final clean‑up ------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_feasible(best_h, target)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(N)
