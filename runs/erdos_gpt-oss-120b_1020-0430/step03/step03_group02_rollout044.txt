def project_onto_feasible(y, target_sum):
    # y: array
    # need projection onto [0,1] box and sum target.
    # We'll use water-filling-like algorithm:
    x = np.clip(y, 0.0, 1.0)
    diff = target_sum - x.sum()
    # free indices where 0 < x < 1 (sum can be updated)
    # We'll iterate
    while abs(diff) > 1e-12:
        free = (x > 0) & (x < 1)
        n_free = free.sum()
        if n_free == 0:
            # cannot adjust further; break (shouldn't happen)
            break
        delta = diff / n_free
        # apply delta
        x[free] += delta
        # clamp again
        np.clip(x, 0.0, 1.0, out=x)
        diff = target_sum - x.sum()
    return x

def worst_gradient(h):
    n = h.size; corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad = 1 - 2*h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def slsqp_optimize(h0, beta, rng, max_iter=400):
    n = h0.size
    target = n / 2.0
    # define softmax objective
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        # compute softmax approx of max
        # use log-sum-exp stable
        m = np.max(beta * corr)
        # bc = beta*corr  actually m immediate: but we can compute as:
        # Using log-sum-exp: log(sum(exp(beta*corr))) = m/beta + log(sum(exp(beta*corr - m)))? Wait careful.
        # compute lse = log(sum(exp(beta*corr - m))) + m/beta? but we need objective = (1/beta) * logsumexp(beta*corr)
        # Equivalent: (1/beta)* (m + log(sum(exp(beta*corr - m)))).
        # So:
        lse = (m + np.log(np.sum(np.exp(beta * corr - m)))) / beta
        return lse

    cons = {'type':'eq', 'fun': lambda x: np.sum(x) - target}
    res = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons,

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return corr.max() * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    target = h.size / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst shift O_k = Σ h_i (1‑h_{i+k})."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad = np.ones_like(h)
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def subgradient_descent(
    h0: np.ndarray, rng: np.random.Generator, max_iter: int, step0: float
) -> np.ndarray:
    """Projected sub‑gradient descent on the worst‑shift max."""
    h = h0.copy()
    for it in range(max_iter):
        grad = worst_gradient(h)
        step = step0 / np.sqrt(it + 1.0)
        h_new = h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = enforce_sum(h_new, rng)
        if compute_c5(h_new) < compute_c5(h):
            h = h_new
    return h


def slsqp_optimize(
    h0: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int = 400
) -> np.ndarray:
    """Soft‑max SLSQP refinement for a given β."""
    n = h0.size
    target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        # (1/β) * log-sum-exp(β * corr)
        m = np.max(beta * corr)
        lse = (m + np.log(np.sum(np.exp(beta * corr - m)))) / beta
        return lse

    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    res = minimize(
        obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # quick random redistribution of the tiny leftover
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x


def guided_hill_climb(
    h_start: np.ndarray,
    rng: np.random.Generator,
    max_iters: int = 2000,
    step_scale: float = 0.5,
) -> tuple[np.ndarray, float]:
    """Hill‑climb using gradient‑directed transfers."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step_scale
    decay = max(1, max_iters // 5)
    n = h.size
    for it in range(max_iters):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0:
                eps = rng.random() * max_eps * step
                nh = h.copy()
                nh[i] -= eps
                nh[j] += eps
            else:
                nh = h.copy()
        else:
            nh = h.copy()
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c
