def targeted_perturb_h(h, rng, scale=0.2):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    k = np.argmax(corr)
    shift = k - (n - 1)
    # shift may be negative
    # compute contributions where i and i+shift within bounds
    if shift >= 0:
        idx = np.arange(0, n - shift)
        contrib = h[idx] * (1 - h[idx + shift])
    else:
        idx = np.arange(-shift, n)
        contrib = h[idx] * (1 - h[idx + shift])
    # if no contributions (should not happen)
    if contrib.size == 0:
        return perturb_h(h, rng, scale)
    # choose source high contribution, dest low contribution (different indices)
    hi_thr = np.quantile(contrib, 0.75)
    lo_thr = np.quantile(contrib, 0.25)
    high_idx = idx[contrib >= hi_thr]
    low_idx = idx[contrib <= lo_thr]
    # ensure they are not empty
    if high_idx.size == 0 or low_idx.size == 0:
        return perturb_h(h, rng, scale)
    i = rng.choice(high_idx)
    # choose j from all indices maybe not in high set; maybe from low set or random.
    # ensure i != j
    j = rng.choice(low_idx)
    if i == j:
        # choose random distinct
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)
    # transfer epsilon
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale)
    eps = rng.random() * max_eps * scale  # random in [0, max_eps*scale]
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

import time, math, numpy as np
from scipy.optimize import minimize

# ---------- helpers ----------
def compute_c5(h):
    """Discrete C5 for step function h on [0,2] with dx=2/len(h)."""
    n = h.size
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / n)

def enforce_sum(h, target, rng):
    """Adjust h (in‑place) so that sum(h)=target while 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    """Generate a random feasible h with 0≤h≤1 and sum=n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    enforce_sum(h, n / 2.0, rng)
    return h

def perturb_h(h, rng, scale=0.2):
    """Transfer ε between two coordinates (preserves sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

def targeted_perturb_h(h, rng, scale=0.2):
    """Move mass from a high‑contributing index (for worst shift) to a low one."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    k = np.argmax(corr)
    shift = k - (n - 1)
    if shift >= 0:
        idx = np.arange(0, n - shift)
        prod = h[idx] * (1 - h[idx + shift])
    else:
        idx = np.arange(-shift, n)
        prod = h[idx] * (1 - h[idx + shift])
    if prod.size == 0:
        return perturb_h(h, rng, scale)
    i = idx[np.argmax(prod)]
    j = idx[np.argmin(prod)]
    if i == j:
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale)
    eps = rng.random() * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

def simulated_annealing(h0, rng, max_iter, init_temp, final_temp, time_limit=None):
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    n = h.size
    if init_temp <= 0:

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {"type": "eq", "fun": sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

            start_h = targeted_perturb_h(best_h_n, rng, scale=0.3) if rng.random() < 0.7 else random_feasible_vector(n, rng)
            cand_h, cand_c = simulated_annealing(
                start_h,
                rng,
                max_iter=max_iter,
                init_temp=0.03,
                final_temp=1e-6,
                time_limit=sa_end,
            )
            if cand_c < best_c_n:
                best_c_n, best_h_n = cand_c, cand_h.copy()

        # ---------- SLSQP polishing ----------
        for beta in (200, 800, 2000, 8000):
            if time_left() < 2:
                break
            try:
                refined = slsqp_optimize(best_h_n, beta, max_iter=250)
                rc = compute_c5(refined)
                if rc < best_c_n:
                    best_c_n, best_h_n = rc, refined
            except Exception:
                continue

        best_h_n = np.clip(best_h_n, 0.0, 1.0)
        enforce_sum(best_h_n, n / 2.0, rng)
        best_c_n = compute_c5(best_h_n)

        if best_c_n < best_c_global:
            best_c_global = best_c_n
            best_h_global = best_h_n.copy()
            best_n = n

    # final polishing on the global best
    if best_h_global is not None and time_left() > 5:
        for beta in (2000, 8000, 20000):
            if time_left() < 2:
                break
            try:
                best_h_global = slsqp_optimize(best_h_global, beta, max_iter=200)
                best_c_global = compute_c5(best_h_global)
            except Exception:
                continue

    # safety fallback
    if best_h_global is None:
        n = n0
        best_h_global = random_feasible_vector(n, rng)
        best_c_global = compute_c5(best_h_global)
        best_n = n

    best_h_global = np.clip(best_h_global, 0.0, 1.0)
    enforce_sum(best_h_global, best_n / 2.0, rng)
    best_c_global = compute_c5(best_h_global)

    return best_h_global.tolist(), float(best_c_global), int(best_n)
