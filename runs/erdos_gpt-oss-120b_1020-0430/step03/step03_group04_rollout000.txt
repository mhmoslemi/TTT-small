    if init_h is None: init_h = random_binary_vector(n, rng)

    h = init_h.copy()

    cur_c = compute_c5(h)

    best_h = h.copy(); best_c = cur_c

    # maintain list of positions of ones and zeros for efficient random pick.

    ones = np.flatnonzero(h == 1)

    zeros = np.flatnonzero(h == 0)

    # Determine total number of iterations based on time left.

    i = 0

    while time.time() < time_limit:

        # pick random swap

        i_one = rng.integers(len(ones))

        i_zero = rng.integers(len(zeros))

        i_idx = ones[i_one]; j_idx = zeros[i_zero]

        # swap

        h[i_idx] = 0

        h[j_idx] = 1

        # compute new c

        new_c = compute_c5(h)

        # acceptance

        if new_c < cur_c:

            accept = True

        else:

            # SA probability

            # compute temperature

            # simple schedule: T = max(1e-5, init_T * (0.9999 ** i))

            zeros[i_zero] = i_idx

            if cur_c < best_c:

                best_c = cur_c

                best_h = h.copy()

        else:

            # revert swap

            h[i_idx] = 1

            h[j_idx] = 0

        i += 1

        # maybe occasional restart if no improvement for many iterations

        # can keep a counter

    return best_h, best_c

def compute_c5(h):

    n = h.shape[0]

    corr = np.correlate(h, 1 - h, mode="full")

    return float(corr.max() * (2.0 / n))

def binary_sa(n, rng, time_limit, init_h=None):

    if init_h is None:

        h = random_binary_vector(n, rng)

    else:

        h = init_h.copy()

    cur_c = compute_c5(h)

    best_h = h.copy(); best_c = cur_c

    ones = np.flatnonzero(h == 1)

    zeros = np.flatnonzero(h == 0)

    # temperature schedule parameters

    T0 = max(1e-5, cur_c * 0.02)  # starting temperature ~2% of current c

    T = T0

    decay = 0.9997

    iter_count = 0

    last_improve_iter = 0

    while time.time() < time_limit:

        # choose 1 and 0

        i_one = rng.integers(len(ones))

        i_zero = rng.integers(len(zeros))

        i_idx = ones[i_one]; j_idx = zeros[i_zero]

        # swap

        h[i_idx] = 0

        h[j_idx] = 1

        new_c = compute_c5(h)

        accept = False

        if new_c < cur_c:

            accept = True

        else:

            # SA acceptance

            if T > 0:

                if rng.random() < np.exp(-(new_c - cur_c) / T):

                    accept = True

        if accept:

            cur_c = new_c

            # update ones and zeros arrays

            ones[i_one] = j_idx

            zeros[i_zero] = i_idx

            if cur_c < best_c:

                best_c = cur_c

                best_h = h.copy()

                last_improve_iter = iter_count

        else:

            # revert swap

            h[i_idx] = 1

            h[j_idx] = 0

        iter_count += 1

        # update temperature

        T = max(1e-8, T * decay)

        # optional restart if stagnation long

        if iter_count - last_improve_iter > 200000:

            h = best_h.copy()

            zeros = np.flatnonzero(h == 0)

            last_improve_iter = iter_count

    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h):
    """Discrete C5 for step function h on [0,2] with uniform spacing."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return float(corr.max() * (2.0 / n))

def random_binary_vector(n, rng):
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=int)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1
    return h

# ---------- binary simulated annealing ----------
def binary_sa(n, rng, deadline, init_h=None):
    """SA on binary vectors (swap a 1 with a 0)."""
    if init_h is None:
        h = random_binary_vector(n, rng)
    else:
        h = init_h.copy()
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    ones = np.flatnonzero(h == 1)
    zeros = np.flatnonzero(h == 0)

    T0 = max(1e-6, cur_c * 0.02)
    T = T0
    decay = 0.9997
    it = 0
    last_improve = 0

    while time.time() < deadline:
        i_one = rng.integers(len(ones))
        i_zero = rng.integers(len(zeros))
        i = ones[i_one]
        j = zeros[i_zero]

        # perform swap
        h[i] = 0
        h[j] = 1
        new_c = compute_c5(h)

        accept = False
        if new_c < cur_c:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-(new_c - cur_c) / T):
                accept = True

        if accept:
            cur_c = new_c
            ones[i_one] = j
            zeros[i_zero] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
                last_improve = it
        else:
            # revert swap
            h[i] = 1
            h[j] = 0

        it += 1
        T = max(1e-9, T * decay)

        # stagnation restart
        if it - last_improve > 200_000:
            h = best_h.copy()
            cur_c = best_c
            ones = np.flatnonzero(h == 1)
            zeros = np.flatnonzero(h == 0)
            last_improve = it
            T = T0

    return best_h, best_c

# ---------- smooth‑max SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=300):
    """Single SLSQP run with soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x

# ---------- gradient refinement ----------
def gradient_refine(h, rng, step=0.02, max_iter=12):
    """Local gradient descent on the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]          # + (1 - h[i+k])
            g[k:] -= h[:upper]                # - h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h
