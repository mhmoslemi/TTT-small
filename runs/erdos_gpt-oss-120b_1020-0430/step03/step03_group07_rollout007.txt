import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (all top‑level)
# ----------------------------------------------------------------------
def enforce_sum(h, target, rng):
    """Adjust vector h in-place so that its sum equals target, staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j, preserving the sum."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def correlate_full(h):
    """Full linear correlation of h with (1-h) using FFT."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)


# ----------------------------------------------------------------------
# SLSQP surrogate optimisation (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1-x_{i+k}) using log‑sum‑exp."""
    conv = correlate_full(x)
    bc = CURRENT_BETA * conv
    m = bc.max()
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=350):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ----------------------------------------------------------------------
# Local greedy hill‑climb
# ----------------------------------------------------------------------
def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Greedy hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Reserve a little margin for final clean‑up
    deadline = start_time + budget_s * 0.97

    # --------------------------------------------------------------
    # 1. Determine dimension and starting point
    # --------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 360
        h0 = np.full(n, 0.5)
        h0 = enforce_sum(h0, n / 2.0, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # --------------------------------------------------------------
    # 2. Small deterministic pool
    # --------------------------------------------------------------
    xs = np.arange(n) * (2.0 / n) + (1.0 / n)  # mid‑point sample
    for f in (1, 2, 3):
        cand = 0.5 + 0.5 * np.sin(np.pi * f * xs)
        cand = np.clip(cand, 0.0, 1.0)
        cand = enforce_sum(cand, n / 2.0, rng)
        c = compute_c5(cand)
        if c < best_c:
            best_c, best_h = c, cand.copy()
        cand = 0.5 + 0.5 * np.cos(np.pi * f * xs)
        cand = np.clip(cand, 0.0, 1.0)
        cand = enforce_sum(cand, n / 2.0, rng)
        c = compute_c5(cand)
        if c < best_c:
            best_c, best_h = c, cand.copy()

    # binary half‑step
    half = np.zeros(n)
    half[: n // 2] = 1.0
    half = enforce_sum(half, n / 2.0, rng)
    c = compute_c5(half)
    if c < best_c:
        best_c, best_h = c, half.copy()

    # alternating 0/1
    alt = np.tile([0.0, 1.0], n // 2)[:n]
    alt = enforce_sum(alt, n / 2.0, rng)
    c = compute_c5(alt)
    if c < best_c:
        best_c, best_h = c, alt.copy()

    # --------------------------------------------------------------
    # 3. Main stochastic optimisation loop (SLSQP continuation + hill‑climb)
    # --------------------------------------------------------------
    beta_schedule = [1.0, 5.0, 15.0, 40.0, 120.0, 350.0, 800.0]
    # Rough estimate: one full restart takes ~3‑4 s, so we stop early if time low.
    while time.time() < deadline * 0.8:
        start_vec = random_feasible_vector(n, rng)
        h_opt = start_vec
        # continuation over β
        for beta in beta_schedule:
            if time.time() >= deadline:
                break
            h_opt = slsqp_optimize(h_opt, beta, max_iter=300)
        # local greedy refinement
        if time.time() < deadline:
            h_opt, c_opt = hill_climb(h_opt, rng, max_iters=2500, start_step=0.35)
        else:
            c_opt = compute_c5(h_opt)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()
        # early exit if we already beat the target comfortably
        if best_c < 0.38075:
            break

    # --------------------------------------------------------------
    # 4. Final refinement on the current best
    # --------------------------------------------------------------
    if time.time() < deadline:
        h_ref, c_ref = hill_climb(best_h, rng, max_iters=5000, start_step=0.25)
        if c_ref < best_c:
            best_c, best_h = c_ref, h_ref.copy()

    # --------------------------------------------------------------
    # 5. Optional up‑sampling to finer grid (if time permits)
    # --------------------------------------------------------------
    if time.time() < deadline and n * 2 <= 720:
        up_factor = 2
        h_up = np.repeat(best_h, up_factor)
        n_up = h_up.size
        h_up = enforce_sum(h_up, n_up / 2.0, rng)

        # continuation on finer grid (fewer SLSQP iterations)
        for beta in beta_schedule:
            if time.time() >= deadline:
                break
            h_up = slsqp_optimize(h_up, beta, max_iter=250)

        if time.time() < deadline:
            h_up, c_up = hill_climb(h_up, rng, max_iters=4000, start_step=0.2)
            if c_up < best_c:
                best_c, best_h = c_up, h_up.copy()
                n = n_up

    # --------------------------------------------------------------
    # 6. Final clean‑up
    # --------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
