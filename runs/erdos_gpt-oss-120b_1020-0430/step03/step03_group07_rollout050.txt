def reduce_max_shift(h, rng, attempts=20):
    n = h.size
    conv = correlate_full(h)
    max_idx = int(np.argmax(conv))

    h_new = h.copy()
    h_new[i] -= eps
    h_new[j] += eps
    # enforce sum not needed (sum preserved)

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global variables used by the SLSQP surrogate
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def enforce_sum(v, target, rng):
    """Adjust v in-place so that sum(v) == target while keeping 0<=v<=1."""
    diff = target - v.sum()
    n = v.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - v[i])
        else:
            add = max(diff, -v[i])
        v[i] += add
        diff = target - v.sum()
    return v


def random_feasible(n, rng):
    """Return a random vector in [0,1]^n with sum = n/2."""
    v = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(v[i], 1.0 - v[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        v[i] -= eps
        v[j] += eps
    v = np.clip(v, 0.0, 1.0)
    return enforce_sum(v, n / 2.0, rng)


def perturb(v, rng, scale=0.2):
    """Transfer a small amount from one index to another (preserves sum)."""
    n = v.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return v.copy()
    max_eps = min(v[i], 1.0 - v[j])
    if max_eps <= 0:
        return v.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new = v.copy()
    new[i] -= eps
    new[j] += eps
    return new


def correlate_full(v):
    """Full linear correlation of v with (1-v) using FFT."""
    n = v.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(v, L)
    rev = (1 - v)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fv * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(v):
    """Discrete C5 for step vector v (dx = 2/n)."""
    return correlate_full(v).max() * (2.0 / v.size)


# ----------------------------------------------------------------------
# SLSQP surrogate (smooth approximation of max via log‑sum‑exp)
# ----------------------------------------------------------------------
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    conv = correlate_full(x)
    bc = CURRENT_BETA * conv
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(x0, beta, max_iter=600):
    """Run SLSQP on the smooth surrogate for a given beta."""
    global CURRENT_TARGET, CURRENT_BETA
    n = x0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny correction of the sum
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def hill_climb(v0, rng, deadline, max_iters=8000, start_step=0.5):
    """Greedy hill‑climb with time check."""
    v = v0.copy()
    best_v = v.copy()
    best_c = compute_c5(v)
    step = start_step
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        cand = perturb(v, rng, scale=step)
        c = compute_c5(cand)
        if c < best_c:
            best_c = c
            best_v = cand.copy()
            v = cand
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-6)
    return best_v, best_c


def reduce_max_shift(v, rng, attempts=12):
    """Targeted move that tries to lower the currently maximal shift."""
    n = v.size
    conv = correlate_full(v)
    idx = int(np.argmax(conv))
    s = idx - (n - 1)          # shift value
    a = np.zeros_like(v)
    if s >= 0:
        a[: n - s] = 1 - v[s:]
    else:
        a[-s:] = 1 - v[: n + s]
    prod = v * a
    i = int(np.argmax(prod))
    j = int(np.argmin(prod))
    if i == j:
        return v, compute_c5(v)
    max_eps = min(v[i], 1 - v[j])
    if max_eps <= 0:
        return v, compute_c5(v)
    eps = max_eps * 0.5
    cand = v.copy()
    cand[i] -= eps
    cand[j] += eps
    return cand, compute_c5(cand)
