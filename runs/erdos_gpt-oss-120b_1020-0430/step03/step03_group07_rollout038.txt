def compute_overlap_array(h):
    n = h.shape[0]
    L = 1 << ((2*n-1).bit_length())
    fh = np.fft.rfft(h, L)
    fr = np.fft.rfft(h[::-1], L)
    conv = np.fft.irfft(fh * fr, L)[:2*n-1]
    O = conv[n-1:]  # length n, O_k for k=0..n-1
    return O
```

def correlate_full(h):
    n = h.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[:2*n-1]
    return conv

def targeted_hill_climb(h_start, rng, deadline, max_iters=10000, step_initial=0.5, step_decay=0.99):
    h = h_start.copy()
    cur_c = compute_c5(h)
    step = step_initial
    n = h.size
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # compute correlation full
        conv = correlate_full(h)
        # get worst shift index k (max product)
        k = np.argmax(conv[n-1:])  # index in 0..n-1
        limit = n - k
        if limit <= 1:
            # cannot improve further at this shift; maybe random move
            new_h = perturb_h(h, rng, scale=step)
            new_c = compute_c5(new_h)
            if new_c < cur_c:
                h, cur_c = new_h, new_c
            # decay step occasionally
            if (it+1) % 1000 == 0:
                step *= step_decay
            continue
        # try random attempts
        improved = False
        for _ in range(5):
            i = rng.integers(0, limit)
            j = rng.integers(0, limit)
            if i == j:
                continue
            hi = h[i]
            hj = h[j]
            if hi <= 0.0 or hj >= 1.0:
                continue
            delta = h[j + k] - h[i + k]
            if delta <= 0:
                continue
            max_eps = min(hi, 1.0 - hj)
            eps = max_eps * step
            if eps <= 0:
                continue
            new_h = h.copy()
            new_h[i] -= eps
            new_h[j] += eps
            new_c = compute_c5(new_h)
            if new_c < cur_c:
                h, cur_c = new_h, new_c
                improved = True
                break
        if not improved:
            # fallback generic perturb
            new_h = perturb_h(h, rng, scale=step)
            new_c = compute_c5(new_h)
            if new_c < cur_c:
                h, cur_c = new_h, new_c
        # decay step regularly
        if (it+1) % 2000 == 0:
            step = max(step * step_decay, 1e-4)
    return h, cur_c

CURRENT_TARGET = None
CURRENT_BETA = None

# ----------------------------------------------------------------------
# Basic feasibility utilities
# ----------------------------------------------------------------------
def enforce_sum(h, target, rng):
    """Adjust h so that Σh == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-14:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        h = enforce_sum(h, target, rng)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

# ----------------------------------------------------------------------
# Correlation utilities (FFT based)
# ----------------------------------------------------------------------
def correlate_full(h):
    """Full linear correlation of h with (1-h) using FFT (zero‑padding)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())  # next power of two
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    n = h.size
    max_corr = conv[n - 1 :].max()
    return max_corr * (2.0 / n)


# ----------------------------------------------------------------------
# Smooth surrogate for max (log‑sum‑exp) and SLSQP optimizer
# ----------------------------------------------------------------------
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of the max over k of Σ h_i (1‑h_{i+k})."""
    corr = correlate_full(x)
    bc = CURRENT_BETA * corr
    m = bc.max()
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=350):
    """Run SLSQP on the smoothed objective for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny correction of the sum (numerical drift)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ----------------------------------------------------------------------
# Targeted hill‑climbing using the worst shift
# ----------------------------------------------------------------------
def targeted_hill_climb(h_start, rng, deadline, max_iters=20000,
                       step_initial=0.5, step_decay=0.99):
    """Improve h by moving mass to increase the minimum overlap."""
    h = h_start.copy()
    cur_c = compute_c5(h)
    n = h.size
    step = step_initial
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # worst shift (max product)
        conv = correlate_full(h)
        k = int(np.argmax(conv[n - 1 :]))  # shift with max product
        limit = n - k
        improved = False
        if limit > 1:
            for _ in range(5):
                i = rng.integers(0, limit)
                j = rng.integers(0, limit)
                if i == j:
                    continue
                hi = h[i]
                hj = h[j]
                if hi <= 0.0 or hj >= 1.0:
                    continue
                delta = h[j + k] - h[i + k]
                if delta <= 0.0:
                    continue
                max_eps = min(hi, 1.0 - hj)
                eps = max_eps * step
                if eps <= 0.0:
                    continue
                new_h = h.copy()
                new_h[i] -= eps
                new_h[j] += eps
                new_c = compute_c5(new_h)
                if new_c < cur_c:
                    h, cur_c = new_h, new_c
                    improved = True
                    break
        if not improved:
            # generic perturb as a fallback
            new_h = perturb_h(h, rng, scale=step)
            new_c = compute_c5(new_h)
            if new_c < cur_c:
                h, cur_c = new_h, new_c
        if (it + 1) % 2000 == 0:
            step = max(step * step_decay, 1e-4)
    return h, cur_c


# ----------------------------------------------------------------------
# Upsampling utilities
# ----------------------------------------------------------------------
def upsample_h(h, factor, rng):
    """Linear interpolation up‑sampling by an integer factor."""
    n = h.size
    n_new = n * factor
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    target = n_new / 2.0
    h_new = enforce_sum(h_new, target, rng)
    return h_new


# ----------------------------------------------------------------------
# Local optimisation driver (SLSQP + targeted hill‑climb)
# ----------------------------------------------------------------------
def optimize_on_grid(start_h, rng, deadline):
    """Apply a ladder of SLSQP refinements followed by a directed hill‑climb."""
    h = start_h.copy()
    best_h = h
    best_c = compute_c5(h)
    beta_list = [5.0, 20.0, 80.0, 200.0, 800.0, 2000.0, 5000.0]
    for beta in beta_list:
        if time.time() >= deadline:
            break
        try:
            h_s = slsqp_optimize(h, beta, max_iter=350)
            c_s = compute_c5(h_s)
            if c_s < best_c:
                best_c, best_h = c_s, h_s.copy()
            h = h_s
        except Exception:
            continue
    # final directed hill‑climb
    if time.time() < deadline:
        hc_h, hc_c = targeted_hill_climb(best_h, rng, deadline,
                                         max_iters=20000, step_initial=0.5)
        if hc_c < best_c:
            best_c, best_h = hc_c, hc_h.copy()
    return best_h, best_c


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # time allocation
    phase1_deadline = start_time + budget_s * 0.75
    phase2_deadline = start_time + budget_s * 0.95
    final_deadline = start_time + budget_s - 0.05

    # ------------------------------------------------------------------
    # 1. Initial vector (use parent if provided)
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = random_feasible_vector(240, rng)
    n0 = h0.size
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = enforce_sum(h0, n0 / 2.0, rng)

    best_h = h0
    best_c = compute_c5(best_h)

    # ------------------------------------------------------------------
    # 2. Repeated random‑restart optimisation on the base grid
    # ------------------------------------------------------------------
    while time.time() < phase1_deadline:
        if rng.random() < 0.6:
            start = perturb_h(best_h, rng, scale=0.3)
        else:
            start = random_feasible_vector(n0, rng)
        # allocate a modest sub‑deadline so we never overrun phase1
        sub_deadline = min(time.time() + 30.0, phase1_deadline)
        h_opt, c_opt = optimize_on_grid(start, rng, sub_deadline)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # ------------------------------------------------------------------
    # 3. Upsample and refine if time permits (once or twice)
    # ------------------------------------------------------------------
    while time.time() < phase2_deadline:
        # stop growing the grid after it exceeds 1440 points
        if best_h.size * 2 > 1440:
            break
        h_up = upsample_h(best_h, 2, rng)
        n_up = h_up.size
        sub_deadline = min(time.time() + 40.0, phase2_deadline)
        h_opt, c_opt = optimize_on_grid(h_up, rng, sub_deadline)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
        else:
            # no improvement – break to save time
            break

    # ------------------------------------------------------------------
    # 4. Final clean‑up (exact feasibility)
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
