import time
import numpy as np

# -------------------------------------------------
# Helper functions (top‑level, no closures or lambdas)
# -------------------------------------------------
def compute_c5(h):
    """Discrete C5 for a step function on [0,2] with n points."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def enforce_sum(h, rng):
    """Adjust h to satisfy Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple random redistribution
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Random feasible h (0≤h≤1, Σh=n/2)."""
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h = enforce_sum(h, rng)
    return h


def worst_gradient(h):
    """Gradient of the worst‑shift overlap O_k = Σ h_i(1‑h_{i+k})."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h, grad, rng, scale=0.5):
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h_start, rng, max_iters=4000, step_scale=0.6):
    """Hill‑climb directed by the worst‑shift gradient."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def simulated_annealing(h_start, rng, start_time, budget_s,
                        target_c=0.3808,
                        init_temp=0.002, final_temp=1e-8, scale=0.5):
    """SA that preserves the sum by moving mass between two indices."""
    n = h_start.size
    h = h_start.copy()
    c = compute_c5(h)
    best_h, best_c = h.copy(), c
    while True:
        now = time.time()
        elapsed = now - start_time
        if elapsed > 0.96 * budget_s or best_c < target_c:
            break
        # temperature decreases with elapsed fraction
        frac = elapsed / budget_s
        T = init_temp * (final_temp / init_temp) ** frac
        # propose a move
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * scale
        h[i] -= eps
        h[j] += eps
        c_new = compute_c5(h)
        if c_new < c:
            accept = True
        else:
            accept = (T > 0.0) and (rng.random() < np.exp(-(c_new - c) / T))
        if accept:
            c = c_new
            if c < best_c:
                best_c, best_h = c, h.copy()
        else:
            # revert
            h[i] += eps
            h[j] -= eps
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # -------------------------------------------------
    # Determine candidate discretisations
    # -------------------------------------------------
    base_n = 192
    if "initial_h_values" in globals():
        base_n = len(initial_h_values)
    candidate_ns = [base_n] + [84, 96, 108, 120, 132, 144, 156, 168, 180, 192]
    candidate_ns = [n for n in candidate_ns if n > 0]
    # -------------------------------------------------
    # Explore each n while time permits
    # -------------------------------------------------
    for n in candidate_ns:
        if time.time() - start > 0.94 * budget_s:
            break

        # ---- initialise h for this n ---------------------------------
        if "initial_h_values" in globals() and len(initial_h_values) == n:
            h = np.array(initial_h_values, dtype=float)
        elif best_h is not None and best_n == n:
            h = best_h.copy()
        elif best_h is not None:
            # interpolate best known solution to new resolution
            x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h = np.interp(x_new, x_old, best_h)
        else:
            h = random_feasible_vector(n, rng)

        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)

        # ---- local hill‑climb -----------------------------------------
        h, c = guided_hill_climb(h, rng, max_iters=5000, step_scale=0.6)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

        # ---- simulated annealing ---------------------------------------
        h_sa, c_sa = simulated_annealing(h, rng, start, budget_s, target_c=0.3808)
        if c_sa < best_c:
            best_h, best_c, best_n = h_sa.copy(), c_sa, n

        # ---- final polish hill‑climb -----------------------------------
        h, c = guided_hill_climb(best_h, rng, max_iters=3000, step_scale=0.4)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

    # -------------------------------------------------
    # Final feasibility check & return
    # -------------------------------------------------
    if best_h is None:
        best_n = 84
        best_h = np.full(best_n, 0.5)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
