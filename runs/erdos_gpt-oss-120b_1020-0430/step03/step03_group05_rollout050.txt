def project_to_feasible(v, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo+hi)/2.0
        s = np.sum(np.clip(v+mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo+hi)/2.0, 0.0, 1.0)

def neighbor_transfer(h, rng, scale):
    n = h.size
    i = rng.integers(0,n)
    j = rng.integers(0,n)
    while i==j:
        j = rng.integers(0,n)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    eps = (rng.random()*2 - 1) * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h
```

    cur = h0.copy()
    cur_c = compute_c5_fft(cur)
    best = cur.copy()
    best_c = cur_c
    n = cur.size
    start = time.time()
    # Determine number of temperature steps
    # We'll adapt based on time budget: allocate loops via while loop.
    T = start_temp
    cooling_rate = 0.99
    # scale for epsilon as proportional to T
    while time.time() - start < time_budget and T > end_temp:
        # attempts per T
        attempts = max(200, n*5)
        for _ in range(attempts):

def hill_climb(h0, rng, max_iters=5000, init_step=0.2):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5_fft(h)
    step = init_step
    for it in range(max_iters):
        new_h = neighbor_transfer(h, rng, scale=step)
        new_c = compute_c5_fft(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
            h = new_h
        # schedule step decay: every decays
        if (it+1) % (max_iters//10) == 0:
            step *= 0.7
            if step < 1e-4:
                step = 1e-4
    return best_h, best_c

import time
import numpy as np

# ---------- feasibility utilities ----------
def project_to_feasible(v: np.ndarray, target: float) -> np.ndarray:
    """Shift+clip projection onto {0≤x≤1, sum x = target}."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Fast generation of a random vector with 0≤h≤1 and sum = n/2."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_to_feasible(v, n / 2.0)


# ---------- objective ----------
def compute_c5_fft(h: np.ndarray) -> float:
    """Exact discrete C5 using FFT (dx = 2/n)."""
    n = h.shape[0]
    m = n * 2
    H = np.fft.rfft(h, m)
    G = np.fft.rfft(np.flip(1.0 - h), m)
    conv = np.fft.irfft(H * G, m)
    max_corr = np.max(conv[: 2 * n - 1])
    return max_corr * (2.0 / n)


# ---------- local perturbations ----------
def neighbor_transfer(h: np.ndarray, rng: np.random.Generator, scale: float) -> np.ndarray:
    """Transfer ε from i to j (preserves sum, respects bounds)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    while i == j:
        j = rng.integers(0, n)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() * 2.0 - 1.0) * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def neighbor_swap(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Swap two entries (sum unchanged)."""
    n = h.size
    i, j = rng.choice(n, size=2, replace=False)
    new_h = h.copy()
    new_h[i], new_h[j] = new_h[j], new_h[i]
    return new_h


def neighbor(h: np.ndarray, rng: np.random.Generator, scale: float, p_swap: float = 0.2) -> np.ndarray:
    """Choose between a swap and a transfer."""
    if rng.random() < p_swap:
        return neighbor_swap(h, rng)
    return neighbor_transfer(h, rng, scale)


# ---------- simulated annealing ----------
def simulated_annealing(
    h0: np.ndarray,
    rng: np.random.Generator,
    time_budget: float,
    start_temp: float = 0.2,
    end_temp: float = 1e-5,
) -> tuple[np.ndarray, float]:
    """Simple SA keeping the sum constraint exactly."""
    cur = h0.copy()
    cur_c = compute_c5_fft(cur)
    best = cur.copy()
    best_c = cur_c
    T = start_temp
    cooling = 0.99
    n = cur.size
    start = time.time()

    while time.time() - start < time_budget and T > end_temp:
        # attempts per temperature level
        attempts = max(200, n * 5)
        scale = min(1.0, T * 20.0)  # larger moves early
        for _ in range(attempts):
            new_h = neighbor(cur, rng, scale)
            new_c = compute_c5_fft(new_h)
            if new_c < cur_c:
                cur, cur_c = new_h, new_c
                if new_c < best_c:
                    best, best_c = new_h.copy(), new_c
            else:
                # Metropolis acceptance
                if rng.random() < np.exp(-(new_c - cur_c) / (T + 1e-12)):
                    cur, cur_c = new_h, new_c
        T *= cooling
    return best, best_c


# ---------- hill climbing ----------
def hill_climb(
    h_start: np.ndarray,
    rng: np.random.Generator,
    max_iters: int = 5000,
    init_step: float = 0.3,
) -> tuple[np.ndarray, float]:
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5_fft(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)

    for it in range(max_iters):
        new_h = neighbor_transfer(h, rng, step)
        new_c = compute_c5_fft(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


# ---------- up‑sampling ----------
def upsample(old_h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a step function to a finer grid."""
    old_n = old_h.size
    xs_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(xs_new, xs_old, old_h)


# ---------- main driver ----------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----- determine target discretisation -----
    if "initial_h_values" in globals():
        init_h = np.array(initial_h_values, dtype=float)
        n = init_h.size
        init_h = project_to_feasible(init_h, n / 2.0)
    else:
        n = 300
        init_h = random_feasible_vector(n, rng)

    best_h = init_h.copy()
    best_c = compute_c5_fft(best_h)

    # ----- optional low‑resolution exploration -----
    low_res_n = 60
    if budget_s > 200:
        low_res_time = min(0.25 * budget_s, 60.0)
        # generate a random start for low resolution
        low_start = random_feasible_vector(low_res_n, rng)
        low_best_h, low_best_c = simulated_annealing(
            low_start, rng, low_res_time, start_temp=0.2, end_temp=1e-5
        )
        # up‑sample and project to target resolution
        up_h = upsample(low_best_h, n)
        up_h = project_to_feasible(up_h, n / 2.0)
        up_c = compute_c5_fft(up_h)
        if up_c < best_c:
            best_h, best_c = up_h, up_c

    # ----- main simulated annealing loops (multiple restarts) -----
    while True:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining < 20.0:
            break

        # allocate a slice of time for this SA run
        time_for_sa = min(remaining * 0.4, 50.0)

        # choose a start point (random or perturbed best)
        if rng.random() < 0.5:
            start_h = random_feasible_vector(n, rng)
        else:
            start_h = neighbor_transfer(best_h, rng, scale=0.5)

        cand_h, cand_c = simulated_annealing(
            start_h, rng, time_for_sa, start_temp=0.2, end_temp=1e-5
        )
        if cand_c < best_c:
            best_h, best_c = cand_h, cand_c

        # short polishing hill‑climb
        if remaining > 10.0:
            cand_h2, cand_c2 = hill_climb(cand_h, rng, max_iters=2000, init_step=0.2)
            if cand_c2 < best_c:
                best_h, best_c = cand_h2, cand_c2

    # ----- final intensive hill‑climb -----
    elapsed = time.time() - start_time
    if budget_s - elapsed > 5.0:
        best_h, best_c = hill_climb(best_h, rng, max_iters=6000, init_step=0.3)

    # ----- enforce exact feasibility (tiny numeric drift) -----
    best_h = np.clip(best_h, 0.0, 1.0)
    target = n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()

    final_c = compute_c5_fft(best_h)

    return best_h.tolist(), float(final_c), int(n)
