def gradient_step(h, target, beta, step):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_c = corr.max()
    w = np.exp(beta * (corr - max_c))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0/(len(w)))
    else:
        w /= w_sum
    # compute forward and backward via convolution
    forward = np.convolve(w, 1 - h, mode='full')[n-1:n-1+n]
    backward = np.convolve(w[::-1], h, mode='full')[n-1:n-1+n]
    g = forward - backward
    h_new = h - step * g
    h_new = np.clip(h_new, 0.0, 1.0)
    # project onto sum
    h_new = project_onto_sum_and_bounds(h_new, target)
    return h_new

import time, math, numpy as np
from scipy.optimize import minimize

# ------------------- basic utilities -------------------
def compute_c5(h):
    """Discrete C5 for step function h (dx = 2 / len(h))."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / h.size)


def project_onto_sum_and_bounds(x, target, max_iter=60):
    """Euclidean projection of x onto {0≤x≤1, sum(x)=target}."""
    low, high = -1.0, 1.0
    for _ in range(max_iter):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    return np.clip(x - (low + high) * 0.5, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Generate a random h with 0≤h≤1 and sum = n/2."""
    h = np.full(n, 0.5, dtype=float)
    for _ in range(n * 5):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.5
        h[i] -= eps
        h[j] += eps
    return project_onto_sum_and_bounds(h, n / 2.0)


def perturb_h(h, rng, scale=0.2):
    """Transfer ε between two coordinates (preserves sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new


def biased_perturb_h(h, rng, scale=0.2):
    """Bias transfer away from positions contributing most to the max overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    k = np.argmax(corr)
    shift = k - (n - 1)
    i0 = max(0, -shift)
    i1 = min(n, n - shift)
    idx = np.arange(i0, i1)
    prod = h[idx] * (1.0 - h[idx + shift])
    if prod.size == 0:
        return perturb_h(h, rng, scale)
    thr_hi = np.quantile(prod, 0.70)
    thr_lo = np.quantile(prod, 0.30)
    high = idx[prod > thr_hi]
    low = idx[prod < thr_lo]
    if high.size == 0 or low.size == 0:
        return perturb_h(h, rng, scale)
    i = rng.choice(high)
    j = rng.choice(low)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new


def simulated_annealing(h0, rng, max_iter, init_temp, final_temp):
    """Simple SA on C5 (lower is better)."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    n = h.size
    log_ratio = math.log(final_temp / init_temp) if init_temp > 0 else 0.0
    for it in range(1, max_iter + 1):
        T = init_temp * math.exp(log_ratio * it / max_iter)
        cand = biased_perturb_h(h, rng, scale=0.3) if rng.random() < 0.7 else perturb_h(h, rng, scale=0.3)
        cand_c = compute_c5(cand)
        diff = cand_c - cur_c
        if diff < 0 or rng.random() < math.exp(-diff / max(T, 1e-12)):
            h, cur_c = cand, cand_c
            if cand_c < best_c:
                best_c, best_h = cand_c, cand.copy()
    return best_h, best_c


# ------------------- SLSQP surrogate -------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on a smooth approximation of max."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {"type": "eq", "fun": sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_onto_sum_and_bounds(x, CURRENT_TARGET)


# ------------------- gradient descent on smooth max -------------------
def gradient_step(h, target, beta, step):
    """One gradient update for the soft‑max approximation."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    w = np.exp(beta * (corr - m))
    w_sum = w.sum()
    if w_sum == 0.0:
        w.fill(1.0 / w.size)
    else:
        w /= w_sum

    # forward term: sum_s w_s * (1 - h[i+s])
    forward = np.convolve(w, 1.0 - h, mode="full")[n - 1 : n - 1 + n]
    # backward term: sum_s w_s * h[i-s] = convolution with reversed weights
    backward = np.convolve(w[::-1], h, mode="full")[n - 1 : n - 1 + n]
    g = forward - backward

    h_new = h - step * g
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_onto_sum_and_bounds(h_new, target)

        step = base_step / math.sqrt(beta)
        for _ in range(iters):
            h = gradient_step(h, target, beta, step)
    return h

    beta_sched = [5, 20, 80, 300, 1200, 5000]
    iters_sched = [500, 400, 300, 250, 150, 100]  # total ≈1700 steps per restart
    base_step = 0.5

    # allocate ~70 % of the budget for this phase
    search_deadline = start + budget_s * 0.70
    while time.time() < search_deadline:
        # start from a perturbed best or a fresh random point
        if rng.random() < 0.55:
            start_h = perturb_h(best_h, rng, scale=0.4)
        else:
            start_h = random_feasible_vector(n, rng)

        # smooth‑max gradient descent
        h_opt = grad_descent_h(start_h, target, beta_sched, iters_sched, base_step)

        # brief local SA polishing
        h_opt, c_opt = simulated_annealing(
            h_opt, rng, max_iter=1200, init_temp=0.02, final_temp=1e-6
        )
        if c_opt < best_c:
            best_h, best_c = h_opt.copy(), c_opt

    # ----- final SLSQP polishing (remaining time) -----
    for beta in (8000, 20000, 50000):
        if time.time() > start + budget_s:
            break
        try:
            best_h = slsqp_optimize(best_h, beta, max_iter=250)
            best_c = compute_c5(best_h)
        except Exception:
            continue

    # ----- final projection and return -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_sum_and_bounds(best_h, target)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
