def gradient_refine_long(h, rng, max_iter=200, step=0.05):
    for it in range(max_iter):
        # compute correlation and worst shift (k)
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr) - (h.size -1)
        k = abs(idx)
        # gradient g
        n = h.size
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]  # derivative w.r.t h_i for i<upper
            g[k:] -= h[:upper]       # derivative w.r.t h_{i+k}
        # update step size: decaying
        step_sz = step * (0.95 ** it)  # slower decay?
        h = h - step_sz * g
        # clip to [0, 1]
        h = np.clip(h, 0.0, 1.0)
        # correct sum:
        diff = (n / 2.0) - h.sum()
        if abs(diff) > 1e-12:
            # we can adjust at random positions not at bounds.
            for _ in range(200):
                i = rng.integers(0, n)

                if diff > 0:
                    add = min(diff, 1.0 - h[i])

                    add = max(diff, -h[i])
                if add != 0:
                    h[i] += add
                    diff = (n / 2.0) - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

import time, numpy as np

def compute_c5(h):
    """Maximum overlap for step function h on [0,2] with uniform spacing."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def random_feasible_vector(n, rng):
    """Uniform random vector in [0,1] with exact sum n/2."""
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    # fix rounding errors
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)

def perturb_h(h, rng, scale=0.2):
    """Transfer ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if j >= i:
        j += 1
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb(h_start, rng, max_iters=2000, start_step=0.4):
    """Deterministic hill‑climb – accept only improvements."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        cand = perturb_h(h, rng, scale=step)
        c = compute_c5(cand)
        if c < best_c:
            best_c, best_h, h = c, cand.copy(), cand
        if (it + 1) % decay == 0:
            step = max(step * 0.9, 1e-5)
    return best_h, best_c

def gradient_refine_long(h, rng, max_iter=200, step=0.05):
    """Gradient descent on the worst shift (more iterations than original)."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            up = n - k
            g[:up] += 1.0 - h[k:]          # ∂/∂h_i for i<up
            g[k:]  -= h[:up]               # ∂/∂h_{i+k}
        step_sz = step * (0.97 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # cheap redistribution
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                if add != 0.0:
                    h[i] += add
                    diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

        parent = np.clip(parent, 0.0, 1.0)
        target = n_parent / 2.0
        diff = target - parent.sum()
        if abs(diff) > 1e-12:
            for _ in range(2000):
                i = rng.integers(0, n_parent)
                add = min(diff, 1.0 - parent[i]) if diff > 0 else max(diff, -parent[i])
                parent[i] += add
                diff = target - parent.sum()
                if abs(diff) < 1e-12:
                    break
        best_h, best_c, best_n = parent.copy(), compute_c5(parent), n_parent

    # -----------------------------------------------------------
    # 2.  Grid sizes to explore
    candidate_ns = [96, 120, 144, 180, 240, 300]
    if best_n is not None:
        candidate_ns.append(best_n)
    candidate_ns = sorted(set(candidate_ns))

    # -----------------------------------------------------------
    # 3.  Main search over each n
    for n in candidate_ns:
        if time.time() - start_time > budget_s * 0.96:
            break

        # base interpolation from current best (if any)
        base_interp = None
        if best_h is not None:
            x_src = np.linspace(0.0, 2.0, best_n, endpoint=False)
            x_tgt = np.linspace(0.0, 2.0, n, endpoint=False)
            base_interp = np.interp(x_tgt, x_src, best_h)

        # number of independent starts (keep modest)
        starts = 6
        for s in range(starts):
            if time.time() - start_time > budget_s * 0.96:
                break

            # ----- initialise start vector -----
            if s == 0 and base_interp is not None:
                h = base_interp.copy()
            elif s == 1:
                h = random_feasible_vector(n, rng)
            elif s == 2:
                h = np.zeros(n)
                idx = rng.choice(n, n // 2, replace=False)
                h[idx] = 1.0
            else:
                # perturb current global best if same size, else random
                if best_n == n and best_h is not None:
                    h = best_h + 0.05 * rng.standard_normal(n)
                else:
                    h = random_feasible_vector(n, rng)

            # enforce bounds & exact sum
            h = np.clip(h, 0.0, 1.0)
            target = n / 2.0
            diff = target - h.sum()
            if abs(diff) > 1e-12:
                for _ in range(2000):
                    i = rng.integers(0, n)
                    add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                    h[i] += add
                    diff = target - h.sum()
                    if abs(diff) < 1e-12:
                        break

            cur_h, cur_c = h, compute_c5(h)

            # ----- Simulated annealing (time‑capped) -----
            time_left = budget_s - (time.time() - start_time)
            max_iters = int(min(25000, max(1000, time_left * 4000)))
            T0 = max(cur_c * 0.02, 1e-5)
            step0 = 0.5
            for it in range(max_iters):
                if time.time() - start_time > budget_s * 0.96:
                    break
                # exponential decay of temperature & step
                tau = max_iters / 5.0
                T = T0 * np.exp(-it / tau)
                step = step0 * np.exp(-it / tau)

                i = rng.integers(0, n)
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                max_eps = min(cur_h[i], 1.0 - cur_h[j])
                if max_eps <= 0.0:
                    continue
                eps = (rng.random() - 0.5) * 2.0 * max_eps * step
                cand = cur_h.copy()
                cand[i] -= eps
                cand[j] += eps
                cand_c = compute_c5(cand)

                if cand_c < cur_c:
                    cur_h, cur_c = cand, cand_c
                else:
                    if rng.random() < np.exp(-(cand_c - cur_c) / (T + 1e-12)):
                        cur_h, cur_c = cand, cand_c
                if cur_c < best_c:
                    best_h, best_c, best_n = cur_h.copy(), cur_c, n

            # ----- Gradient refinement (many iterations) -----
            cur_h = gradient_refine_long(cur_h, rng, max_iter=120, step=0.05)
            cur_c = compute_c5(cur_h)
            if cur_c < best_c:
                best_h, best_c, best_n = cur_h.copy(), cur_c, n

            # ----- Deterministic hill‑climb (short) -----
            cur_h, cur_c = hill_climb(cur_h, rng, max_iters=1500, start_step=0.3)
            if cur_c < best_c:
                best_h, best_c, best_n = cur_h.copy(), cur_c, n

    # -----------------------------------------------------------
    # 4.  Final intensive refinement on the global best
    if best_h is not None:
        best_h = gradient_refine_long(best_h, rng, max_iter=300, step=0.04)
        best_h, _ = hill_climb(best_h, rng, max_iters=4000, start_step=0.2)

    # -----------------------------------------------------------
    # 5.  Safety checks & final objective
    if best_h is None:
        best_n = 120
        best_h = np.full(best_n, 0.5)
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            i = rng.integers(0, best_n)
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()
            if abs(diff) < 1e-12:
                break
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
