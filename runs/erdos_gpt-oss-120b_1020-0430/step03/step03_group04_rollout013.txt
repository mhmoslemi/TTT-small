def _enforce_sum_and_bounds(h, target, rng):
    """Clip to [0,1] and adjust to hit the exact target sum."""
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add == 0.0:
            continue
        h[i] += add
        diff = target - h.sum()
    return h

def _compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def _gradient_softmax(h, beta):
    """Gradient of the soft‑max surrogate of max_k c_k."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    maxc = np.max(corr)
    exp = np.exp(beta * (corr - maxc))
    w = exp / exp.sum()
    g = np.zeros_like(h)

    # shifts ≥ 0
    for m in range(0, n):
        wpos = w[m + n - 1]
        if wpos == 0.0:
            continue
        L = n - m
        if L <= 0:
            continue
        g[:L] += wpos * dx * (1.0 - h[m:])
        g[m:] -= wpos * dx * h[:L]

    # shifts < 0
    for m in range(1, n):
        wneg = w[n - 1 - m]
        if wneg == 0.0:
            continue
        L = n - m
        g[m:] += wneg * dx * (1.0 - h[:L])
        g[:L] -= wneg * dx * h[m:]

    return g, corr

            h_rand = rng.random(n)
            h_rand *= target / h_rand.sum()
            init_vecs.append(_enforce_sum_and_bounds(h_rand, target, rng))

        # perturb the current best if it has same n
        if best_n == n and best_h is not None:
            pert = np.array(best_h, copy=True) + 0.02 * rng.standard_normal(n)
            init_vecs.append(_enforce_sum_and_bounds(pert, target, rng))

        for h0 in init_vecs:
            if time.time() > end_time * 0.99:
                break

            h_cur = _enforce_sum_and_bounds(h0.copy(), target, rng)

            # soft‑max PGD with increasing beta
            for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0):
                if time.time() > end_time * 0.99:
                    break
                iters = 800 if beta < 200 else 1200
                h_cur = _pgd_optimize(h_cur, beta, step0=0.2,
                                      max_iter=iters, rng=rng)

            # sub‑gradient refinement on the worst shift
            if time.time() < end_time * 0.98:
                h_cur = _subgradient_refine(h_cur, rng,
                                            init_step=0.4, max_iter=800)

            # evaluate exact C5
            cur_c = _compute_c5(h_cur)
            if cur_c < best_c:
                best_h, best_c, best_n = h_cur.copy(), cur_c, n

            # final hill‑climbing on the true objective
            if time.time() < end_time * 0.97:
                h_hc, c_hc = _hill_climb_exact(h_cur, rng,
                                               max_iters=3000, step_scale=0.4)
                if c_hc < best_c:
                    best_h, best_c, best_n = h_hc.copy(), c_hc, n

        # extra random attempts while time permits (lightweight)
        while time.time() < end_time * 0.95:
            h_rand = rng.random(n)
            h_rand *= target / h_rand.sum()
            h_rand = _enforce_sum_and_bounds(h_rand, target, rng)
            # short PGD just to get a quick improvement
            for beta in (5.0, 20.0, 80.0):
                h_rand = _pgd_optimize(h_rand, beta, step0=0.2,
                                       max_iter=600, rng=rng)
            c_rand = _compute_c5(h_rand)
            if c_rand < best_c:
                best_h, best_c, best_n = h_rand.copy(), c_rand, n
            break  # keep this loop cheap

    # fallback in the unlikely case nothing was found
    if best_h is None:
        best_n = 180
        best_h = np.full(best_n, 0.5)

    # final safety projection and evaluation
    best_h = _enforce_sum_and_bounds(np.clip(best_h, 0.0, 1.0),
                                     best_n / 2.0, rng)
    final_c = _compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
