def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s
    # Determine n
    if "initial_h_values" in globals():
        h_initial = np.array(initial_h_values, dtype=float)
        n = h_initial.size
        h_initial = np.clip(h_initial, 0.0, 1.0)
        enforce_sum(h_initial, n/2.0, rng)
        best_h = h_initial.copy()
        best_c = compute_c5(best_h)
    else:
        n = 180
        best_h = None
        best_c = float('inf')
    # Helper to run optimization with given beta and max_iter
    def opt_with_beta(x0, beta, max_iter=200):
        # Use L-BFGS-B with gradient and penalty
        penalty = 1e4
        bounds = [(0.0, 1.0)] * x0.size
        def fun(x):
            f, _ = softmax_obj_grad(x, beta, penalty)
            return f
        def jac(x):
            _, g = softmax_obj_grad(x, beta, penalty)
            return g
        res = minimize(fun, x0, method='L-BFGS-B', jac=jac,
                       bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-12})
        x = np.clip(res.x, 0.0, 1.0)
        enforce_sum(x, n/2.0, rng)
        return x

    # Phase 1: many multistart low-beta optimization
    phase1_end = start + budget_s * 0.70
    while time.time() < phase1_end:
        # generate start
        x0 = random_feasible_vector(n, rng)
        # low beta start
        x_opt = opt_with_beta(x0, beta=50, max_iter=200)
        c = compute_c5(x_opt)
        if c < best_c:
            best_c = c
            best_h = x_opt.copy()
    # Phase 2: refinement with higher beta
    for beta in (200, 800, 2000, 8000, 20000):
        if time.time() > deadline:
            break
        if best_h is None:
            # if no best yet, generate random
            best_h = random_feasible_vector(n, rng)
        best_h = opt_with_beta(best_h, beta=beta, max_iter=200)

    # Final cleanup
    best_h = np.clip(best_h, 0.0, 1.0)
    enforce_sum(best_h, n/2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)

import time, numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h):
    """Discrete C5 for step function h on [0,2] with dx=2/len(h)."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)

def enforce_sum(h, target, rng):
    """Adjust h in‑place so that sum(h)=target while respecting 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    """Generate a random h∈[0,1]^n with sum=n/2."""
    h = np.full(n, 0.5)
    # random pairwise transfers
    for _ in range(n * 5):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    enforce_sum(h, n / 2.0, rng)
    return h

# ---------- smooth max (log‑sum‑exp) and its gradient ----------
def softmax_obj_grad(x, beta, penalty):
    """
    Return f(x) = softmax approximation of max correlation + penalty,
    and its gradient w.r.t. x.
    """
    n = x.size
    target = n * 0.5
    # correlation of x with (1-x)
    corr = np.correlate(x, 1 - x, mode='full')
    s = beta * corr
    m = np.max(s)
    exp_s = np.exp(s - m)
    sum_exp = np.sum(exp_s)
    f = (m + np.log(sum_exp)) / beta
    w = exp_s / sum_exp                      # weights, sum to 1

    # gradient of weighted correlation
    conv1 = np.convolve(w, x, mode='full')
    conv2 = np.convolve(w, x[::-1], mode='full')
    term1 = conv1[n - 1 : n - 1 + n]                     # sum_i w_i * x[l - shift_i]
    term2 = conv2[2 * n - 2 - np.arange(n)]              # sum_i w_i * x[l + shift_i]
    grad_corr = 1.0 - (term1 + term2)

    # quadratic penalty for the sum constraint
    p = target - np.sum(x)
    f_total = f + penalty * p * p
    grad = grad_corr - 2.0 * penalty * p
    return f_total, grad

# ---------- L‑BFGS‑B wrapper ----------
def lbfgsb_optimize(x0, beta, max_iter, penalty, rng):
    """Run L‑BFGS‑B on the smooth max objective."""
    bounds = [(0.0, 1.0)] * x0.size

    def fun(x):
        f, _ = softmax_obj_grad(x, beta, penalty)
        return f

    def jac(x):
        _, g = softmax_obj_grad(x, beta, penalty)
        return g

    res = minimize(fun, x0, method='L-BFGS-B', jac=jac,
                   bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-12})
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, x0.size * 0.5, rng)
    return x

# ---------- main run ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s

    # initialise problem size
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        enforce_sum(h0, n * 0.5, rng)
        best_h = h0.copy()
        best_c = compute_c5(best_h)
    else:
        n = 180
        best_h = None
        best_c = float('inf')

    # parameters
    low_beta = 50
    high_betas = (200, 800, 2000, 8000, 20000)
    penalty = 1e4
    max_iter_low = 200
    max_iter_high = 200

    # -----------------------------------------------------------------
    # Phase 1 – many multistart low‑beta L‑BFGS‑B runs
    # -----------------------------------------------------------------
    phase1_end = start_time + budget_s * 0.70
    while time.time() < phase1_end:
        x0 = random_feasible_vector(n, rng)
        x_opt = lbfgsb_optimize(x0, low_beta, max_iter_low, penalty, rng)
        c = compute_c5(x_opt)
        if c < best_c:
            best_c, best_h = c, x_opt.copy()

    # -----------------------------------------------------------------
    # Phase 2 – incremental refinement with larger betas
    # -----------------------------------------------------------------
    for beta in high_betas:
        if time.time() > deadline:
            break
        if best_h is None:
            best_h = random_feasible_vector(n, rng)
        best_h = lbfgsb_optimize(best_h, beta, max_iter_high, penalty, rng)
        best_c = compute_c5(best_h)

    # -----------------------------------------------------------------
    # final clean‑up
    # -----------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    enforce_sum(best_h, n * 0.5, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
