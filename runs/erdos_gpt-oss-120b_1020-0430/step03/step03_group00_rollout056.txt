def project_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=100):
    # find lambda such that sum_i clip(v_i - lam, lower, upper) = target
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):

        x = np.clip(v - lam, lower, upper)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
        if hi - lo < tol:
            break
    return np.clip(v - (lo+hi)/2.0, lower, upper)

def smooth_max_and_grad(h, beta):
    n = h.size
    corr = np.correlate(h, 1-h, mode='full')
    max_corr = corr.max()
    # compute weights
    exp_term = np.exp(beta * (corr - max_corr))
    w = exp_term / exp_term.sum()
    # compute conv terms
    w_rev = w[::-1]
    conv_plus = np.convolve(w_rev, h, mode='full')
    conv_minus = np.convolve(w, h, mode='full')
    start = n-1
    A = conv_plus[start:start+n]
    B = conv_minus[start:start+n]
    grad = 1.0 - A - B
    smooth_val = max_corr + np.log(exp_term.sum()) / beta
    return smooth_val, grad, corr
```

def project_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-10, max_iter=100):
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, lower, upper)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
        if hi - lo < tol:
            break
    x = np.clip(v - (lo+hi)*0.5, lower, upper)
    # final correction: distribute residual diff among non-full entries
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # find indices not at bounds
        free = np.where((x > lower + 1e-12) & (x < upper - 1e-12))[0]
        if free.size > 0:
            # distribute equally
            x[free] += diff / free.size
            # clip again
            x = np.clip(x, lower, upper)
        else:
            # fallback: adjust one entry
            i = np.argmin(np.abs(v - (lo+hi)*0.5))
            x[i] = max(lower, min(upper, x[i] + diff))
    return x
```

import time
import numpy as np

# ---------- projection onto box & sum ----------
def project_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-10, max_iter=100):
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, lower, upper)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
        if hi - lo < tol:
            break
    x = np.clip(v - (lo + hi) * 0.5, lower, upper)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = np.where((x > lower + 1e-12) & (x < upper - 1e-12))[0]
        if free.size:
            x[free] += diff / free.size
            x = np.clip(x, lower, upper)
        else:
            i = np.argmin(np.abs(v - (lo + hi) * 0.5))
            x[i] = min(upper, max(lower, x[i] + diff))
    return x

# ---------- basic evaluation ----------
def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

# ---------- smooth max and its gradient ----------
def smooth_max_and_grad(h, beta):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = corr.max()
    exp_term = np.exp(beta * (corr - max_corr))
    w = exp_term / exp_term.sum()
    w_rev = w[::-1]

    conv_plus = np.convolve(w_rev, h, mode="full")
    conv_minus = np.convolve(w, h, mode="full")
    start = n - 1
    A = conv_plus[start:start + n]
    B = conv_minus[start:start + n]
    grad = 1.0 - A - B
    smooth_val = max_corr + np.log(exp_term.sum()) / beta
    return smooth_val, grad, corr

                 beta_start=5.0, beta_end=800.0, lr=0.5):

    j = rng.integers(0, n)

def perturb_targeted(h, grad, rng, scale=0.2):
    pos = np.where((grad > 1e-9) & (h > 1e-12))[0]
    neg = np.where((grad < -1e-9) & (h < 1.0 - 1e-12))[0]
    if pos.size == 0 or neg.size == 0:
        return perturb_random(h, rng, scale)

    j = rng.choice(neg)

def hill_climb_targeted(h_start, rng, max_iters, start_step, target_sum):
    h = h_start.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr)
        k = idx - (n - 1)

        grad = np.ones(n)
        if k >= 0:
            grad[:n - k] -= h[k:]
            grad[k:] -= h[:n - k]
        else:
            kk = -k
            grad[:n - kk] -= h[kk:]
            grad[kk:] -= h[:n - kk]

            best_c = c_new
            best_h = new_h.copy()

            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta

    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_box_sum(x, CURRENT_TARGET, lower=0.0, upper=1.0)
    return x

    h0 = project_box_sum(np.clip(h0, 0.0, 1.0), h0.size / 2.0, lower=0.0, upper=1.0)

    # ---- upsample to final resolution (≤720) ----
    n0 = h0.size
    final_n = min(720, n0 * 2)
    if final_n != n0:
        xs0 = (np.arange(n0) + 0.5) * (2.0 / n0)
        xs = (np.arange(final_n) + 0.5) * (2.0 / final_n)
        h = np.interp(xs, xs0, h0)
    else:
        h = h0.copy()
    h = project_box_sum(np.clip(h, 0.0, 1.0), final_n / 2.0, lower=0.0, upper=1.0)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    total_budget = float(budget_s)

    # ---- Phase 1: PGD with random restarts ----
    phase1_end = start_time + total_budget * 0.5
    while time.time() < phase1_end:
        if rng.random() < 0.7:
            start = perturb_random(best_h, rng, scale=0.2)
        else:
            start = random_feasible_vector(final_n, rng)
        remaining = phase1_end - time.time()
        slice_time = min(5.0, remaining)
        if slice_time < 1.0:
            break
        h_loc, c_loc = pgd_optimize(
            start,
            final_n / 2.0,
            rng,
            time_limit=slice_time,
            beta_start=5.0,
            beta_end=800.0,
            lr=0.5,
        )
        if c_loc < best_c:
            best_h, best_c = h_loc.copy(), c_loc

    # ---- Phase 2: targeted hill climbing ----
    phase2_end = start_time + total_budget * 0.75
    while time.time() < phase2_end:
        start = perturb_random(best_h, rng, scale=0.3)
        h_loc, c_loc = hill_climb_targeted(
            start, rng, max_iters=1200, start_step=0.5, target_sum=final_n / 2.0
        )
        if c_loc < best_c:
            best_h, best_c = h_loc.copy(), c_loc

    # ---- Phase 3: SLSQP refinements ----
    beta_schedule = [5.0, 20.0, 80.0, 200.0, 800.0, 2000.0]
    for beta in beta_schedule:
        if time.time() - start_time > total_budget * 0.95:
            break
        try:
            h_ref = slsqp_optimize(best_h, beta, max_iter=500)
        except Exception:
            continue
        c_ref = compute_c5(h_ref)
        if c_ref < best_c:
            best_h, best_c = h_ref.copy(), c_ref

    # ---- Phase 4: final micro search ----
    while time.time() - start_time < total_budget * 0.99:
        new_h = perturb_random(best_h, rng, scale=0.02)
        c_new = compute_c5(new_h)
        if c_new < best_c:
            best_h, best_c = new_h.copy(), c_new

    # ---- final clean‑up ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0, lower=0.0, upper=1.0)

    # tiny adjustment for any residual sum error
    total = best_h.sum()
    target = best_h.size / 2.0
    if abs(total - target) > 1e-12:
        diff = target - total
        for i in range(best_h.size):
            if (diff > 0 and best_h[i] < 1.0) or (diff < 0 and best_h[i] > 0.0):
                best_h[i] = min(1.0, max(0.0, best_h[i] + diff))
                break

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
