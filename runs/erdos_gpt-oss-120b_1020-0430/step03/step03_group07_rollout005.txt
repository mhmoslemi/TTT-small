def conv_full(a,b):
    L = 1 << ((len(a) + len(b) - 1).bit_length())
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    conv = np.fft.irfft(fa*fb, L)[:len(a)+len(b)-1]
    return conv

    # Clip first
    v = np.clip(v, 0.0, 1.0)
    # Compute current sum
    cur_sum = v.sum()
    diff = target - cur_sum
    if abs(diff) < 1e-12:
        return v
    # Determine free indices where 0 < v < 1
    free = np.where((v > 0) & (v < 1))[0]
    # If no free indices (rare), just skip adjusting
    if free.size == 0:
        # If diff > 0 need to raise some zeros, if diff < 0 lower some ones.
        # Choose random indices among zeros or ones that can be moved.
        # Use enforce_sum fallback
        return enforce_sum(v, target, np.random.default_rng())
    # Distribute diff equally
    # But ensure not exceed bounds
    # Use iterative method:
    while abs(diff) > 1e-12 and free.size > 0:
        delta = diff / free.size
        # Adjust
        v_new = v[free] + delta
        # Clip new values
        over = v_new > 1.0
        under = v_new < 0.0
        # For values that exceed bounds, set to bound and remove from free
        # Compute actual change applied
        actual_delta = np.where(over, 1.0 - v[free], np.where(under, -v[free], delta))
        v[free] += actual_delta
        # Update diff

        # Update free set: those still strictly interior
        free = np.where((v > 0) & (v < 1))[0]
        # If free unchanged and diff not zero, break to avoid infinite loop
        if free.size == 0:
            break
    # If still diff not zero due to rounding, use enforce_sum fallback
    if abs(diff) > 1e-12:
        v = enforce_sum(v, target, np.random.default_rng())

import time
import numpy as np

# ------------------------------------------------------------
# Basic linear algebra helpers
# ------------------------------------------------------------
def project_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project vector onto [0,1] box and enforce Σv = target (simple iterative)."""
    for _ in range(10):
        v = np.clip(v, 0.0, 1.0)
        diff = target - v.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        idx = np.where(mask)[0]
        if idx.size == 0:
            break
        v[idx] += diff / idx.size
    return np.clip(v, 0.0, 1.0)


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector in [0,1]^n with Σ = n/2."""
    v = rng.random(n)
    target = n / 2.0
    return project_sum(v, target)


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small amount ε from i to j (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def correlate_full(h: np.ndarray) -> np.ndarray:
    """Full linear correlation of h with (1-h) (equivalent to np.correlate)."""
    n = h.size
    if n <= 512:
        return np.correlate(h, 1 - h, mode="full")
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def conv_full(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a * b (full). Uses direct method for small sizes."""
    if a.size * b.size <= 200_000:
        return np.convolve(a, b)
    L = 1 << ((a.size + b.size - 1).bit_length())
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    conv = np.fft.irfft(fa * fb, L)[: a.size + b.size - 1]
    return conv


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)


def gradient_subgrad(h: np.ndarray, beta: float) -> np.ndarray:
    """
    Subgradient of the smoothed max_{k} Σ_i h_i (1 - h_{i+k}).
    Uses softmax weights with temperature 1/beta.
    """
    n = h.size
    c = correlate_full(h)
    max_c = c.max()
    if beta <= 0.0:
        w = np.full_like(c, 1.0 / c.size)
    else:
        # stable softmax
        exp_arg = beta * (c - max_c)
        w = np.exp(exp_arg)
        s = w.sum()
        if s == 0.0:
            w = np.full_like(c, 1.0 / c.size)
        else:
            w = w / s
    # grad = conv(w, 1-h) - conv(w, h) (slice n-1 … n-1+n-1)
    conv1 = conv_full(w, 1 - h)
    conv2 = conv_full(w, h)
    start = n - 1
    return conv1[start : start + n] - conv2[start : start + n]


def hill_climb_refine(h0: np.ndarray, rng: np.random.Generator,
                     max_iters: int = 2000, step: float = 0.04):
    """Simple greedy hill‑climb (accept only improvements)."""
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    h = best_h.copy()
    target = best_h.size / 2.0
    for _ in range(max_iters):
        h_new = perturb_h(h, rng, scale=step)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
    return best_h, best_c
