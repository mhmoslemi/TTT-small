def project_box_simplex(v, target, max_iter=50, tol=1e-12):
    lo = -1.0
    hi = 1.0
    for _ in range(max_iter):
        tau = (lo + hi) / 2.0
        proj = np.clip(v - tau, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) < tol:
            return proj
        if s > target:
            lo = tau
        else:
            hi = tau
    return np.clip(v - (lo+hi)/2.0, 0.0, 1.0)

def perturb_transfer(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    while i == j:
        j = rng.integers(0, n)
    # compute allowed epsilon
    max_eps = min(h[i], 1.0 - h[j])

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # return (m + log sum exp(bc - m)) / beta
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False}
    )
    x = np.clip(res.x, 0.0, 1.0)
    # due to possible numerical sum deviation, project
    x = project_box_simplex(x, CURRENT_TARGET)
    return x

def simulated_annealing(h0, time_limit, rng):
    n = h0.size
    target = n / 2.0
    current = h0.copy()
    current_c = compute_c5(current)
    best_h = current.copy()
    best_c = current_c
    start = time.time()
    # parameters
    T = 0.05  # initial temperature
    T_min = 1e-5
    alpha = 0.998  # cooling per iteration?
    # We'll update T gradually each iteration.
    # We can compute total possible steps based on time but we just loop.
    iter_count = 0
    while time.time() - start < time_limit:
        new_h = perturb_transfer(current, rng, scale=0.2)
        new_c = compute_c5(new_h)
        # acceptance
        if new_c < current_c:
            accept = True
        else:
            # compute acceptance probability
            prob = np.exp(-(new_c - current_c) / max(T, 1e-12))
            accept = rng.random() < prob
        if accept:
            current = new_h
            current_c = new_c
        if new_c < best_c:
            best_h = new_h.copy()
            best_c = new_c
        # cooling
        T = max(T * alpha, T_min)
        iter_count += 1
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ---------- projection onto [0,1] simplex ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def project_box_simplex(v, target, max_iter=50, tol=1e-12):
    """Project vector v onto the intersection of box [0,1] and sum(v)=target."""
    lo, hi = -1.0, 1.0
    for _ in range(max_iter):
        tau = (lo + hi) * 0.5
        proj = np.clip(v - tau, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = tau
        else:
            hi = tau
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

# ---------- objective ----------
def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

# ---------- perturbations ----------
def perturb_transfer(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    while i == j:
        j = rng.integers(0, n)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        # try opposite direction
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0.0:
            return h.copy()
        eps = max_eps * scale * rng.random()
        nh = h.copy()
        nh[j] -= eps
        nh[i] += eps
        return nh
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

# ---------- SLSQP with softmax ----------
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_simplex(x, CURRENT_TARGET)
