def anneal_binary(h0, rng, deadline, T0=1.0, Tf=1e-4, max_iters=200000):
    n = h0.size
    # Precompute indices of ones and zeros for picking randomly.
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    # For fast random selection of a 1 and a 0, we can maintain arrays of indices.
    ones = np.flatnonzero(h > 0.5)
    zeros = np.flatnonzero(h <= 0.5)
    for it in range(max_iters):
        if time.time() >= deadline: break
        frac = it / max_iters
        T = T0 * ((Tf/T0)**frac)
        # choose random swap
        if len(ones)==0 or len(zeros)==0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i], h[j] = 0.0, 1.0
        # update lists
        ones = ones[ones != i]
        zeros = zeros[zeros != j]
        # after swap, i becomes 0, j becomes 1
        # we could use python list conversion for speed but n small; can recompute after each swap.
        # We'll just recompute via flatnonzero each iteration; it's cheap for n=512.
        # Actually we can recompute later after acceptance decision.
        # Let's recompute after acceptance.

def sa_binary(h_start, rng, time_deadline, T0=1.0, Tf=1e-5, max_iters=500000):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    n = h.size
    for it in range(max_iters):
        if time.time() >= time_deadline:
            break
        # cooling
        frac = it / max_iters
        T = T0 * ((Tf/T0) ** frac)
        # choose random 1 and 0
        # we can get indices of ones and zeros via rng.choice with probability?
        i = rng.integers(n)
        # ensure h[i]==1
        while h[i] < 0.5:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] > 0.5:
            j = rng.integers(n)
        # swap
        h[i], h[j] = 0.0, 1.0
        new_c = compute_c5(h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):

            if cur_c < best_c:

                best_h = h.copy()

def enforce_sum(h, target, rng):
    """Adjust h so that sum(h) == target, staying in [0,1]."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        # final deterministic correction on first element
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h


def random_feasible_vector(n, rng):
    """Create a random vector in [0,1]^n with exact sum n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def correlate_full(h):
    """Full linear correlation of h with (1-h) via FFT."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Exact discrete C5 (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)

    """Log‑sum‑exp smoothing of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = correlate_full(x)
    bc = CURRENT_BETA * corr
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

    """Run SLSQP on smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n

    res = minimize(
        softmax_obj,

        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()

    return x

def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Greedy hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-6)
    return best_h, best_c

def sa_binary(h_start, rng, deadline, T0=1.0, Tf=1e-5, max_iters=200000):
    """Simulated annealing on binary vectors (swap moves)."""
    h = h_start.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # exponential cooling
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)

        # pick a random 1 and a random 0
        i = rng.integers(n)
        while h[i] < 0.5:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] > 0.5:
            j = rng.integers(n)

        # swap
        h[i], h[j] = 0.0, 1.0
        new_c = compute_c5(h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = 1.0, 0.0
    return best_h, best_c

def thue_morse(n):
    """Balanced Thue‑Morse binary sequence of length n (n must be a power of 2)."""
    seq = np.fromiter((bin(i).count("1") & 1 for i in range(n)), dtype=np.float64)
    # ensure exact half ones
    ones = int(seq.sum())
    target = n // 2
    if ones > target:
        # turn some ones to zero
        idx = np.flatnonzero(seq == 1.0)[: ones - target]
        seq[idx] = 0.0
    elif ones < target:
        idx = np.flatnonzero(seq == 0.0)[: target - ones]
        seq[idx] = 1.0
    return seq

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    total_budget = budget_s

    # --------------------------------------------------------------
    # Choose discretisation size
    # --------------------------------------------------------------
    n = 1024  # power of two gives fast FFTs

    # --------------------------------------------------------------
    # Build initial candidate pool
    # --------------------------------------------------------------
    candidates = []

    # Thue‑Morse balanced binary
    candidates.append(thue_morse(n))

    # Random balanced binaries
    for _ in range(3):
        perm = rng.permutation(n)
        h = np.zeros(n, dtype=np.float64)
        h[perm[: n // 2]] = 1.0
        candidates.append(h)

    # If a parent construction is supplied, interpolate it to current n
    if "initial_h_values" in globals():
        old_h = np.array(initial_h_values, dtype=np.float64)
        old_n = old_h.size
        xs_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
        xs_new = np.linspace(0.0, 2.0, n, endpoint=False)
        interp = np.interp(xs_new, xs_old, old_h)
        interp = np.clip(interp, 0.0, 1.0)
        interp = enforce_sum(interp, n / 2.0, rng)
        candidates.append(interp)

    # --------------------------------------------------------------
    # Select best starter
    # --------------------------------------------------------------
    best_h = None
    best_c = float("inf")
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # --------------------------------------------------------------
    # Binary simulated annealing phase
    # --------------------------------------------------------------
    deadline_sa = start_time + total_budget * 0.55
    while time.time() < deadline_sa:
        # diversify start points
        if rng.random() < 0.7:
            start = best_h.copy()
            # sprinkle a few random swaps
            for _ in range(8):
                i = rng.integers(n)
                j = rng.integers(n)
                if (start[i] == 1.0) != (start[j] == 1.0):
                    start[i], start[j] = start[j], start[i]
        else:
            perm = rng.permutation(n)
            start = np.zeros(n, dtype=np.float64)
            start[perm[: n // 2]] = 1.0

        cand_h, cand_c = sa_binary(start, rng, deadline_sa,
                                   T0=1.0, Tf=1e-5, max_iters=25000)
        if cand_c < best_c:
            best_c = cand_c
            best_h = cand_h.copy()
        if best_c < 0.3808:
            break  # target already beaten

    # --------------------------------------------------------------
    # Continuous refinement (hill‑climb + SLSQP ladder)
    # --------------------------------------------------------------
    deadline_hc = start_time + total_budget * 0.75
    if time.time() < deadline_hc:
        h_opt, c_opt = hill_climb(best_h, rng, max_iters=8000, start_step=0.3)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()

    beta_list = [10.0, 30.0, 100.0, 300.0, 1000.0, 3000.0]
    deadline_slsqp = start_time + total_budget * 0.94
    for beta in beta_list:
        if time.time() >= deadline_slsqp:
            break
        try:
            h_opt = slsqp_optimize(best_h, beta, max_iter=500)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt.copy()
        except Exception:
            continue

    # --------------------------------------------------------------
    # Optional up‑sampling refinement if time permits
    # --------------------------------------------------------------
    if time.time() < start_time + total_budget * 0.99 and best_h.size * 2 <= 2048:
        up_factor = 2
        h_up = np.repeat(best_h, up_factor)
        n_up = h_up.size
        h_up = enforce_sum(h_up, n_up / 2.0, rng)

        # hill‑climb on finer grid
        h_up_opt, c_up_opt = hill_climb(h_up, rng,
                                        max_iters=12000, start_step=0.2)
        if c_up_opt < best_c:
            best_c, best_h = c_up_opt, h_up_opt.copy()

        # final SLSQP passes on refined grid
        for beta in beta_list:
            if time.time() >= start_time + total_budget:
                break
            try:
                h_opt = slsqp_optimize(best_h, beta, max_iter=500)
                c_opt = compute_c5(h_opt)
                if c_opt < best_c:
                    best_c, best_h = c_opt, h_opt.copy()
            except Exception:
                continue

    # --------------------------------------------------------------
    # Clean‑up and final return
    # --------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
