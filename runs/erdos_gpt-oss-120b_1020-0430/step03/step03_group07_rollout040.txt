def enforce_sum(v, target):
    v = np.clip(v, 0, 1)
    diff = target - v.sum()
    if abs(diff) < 1e-12:
        return v
    # vector of free indices where adjustment possible
    # if diff >0, we can increase entries <1; else decrease entries >0
    # We'll iterate adjusting uniformly among free entries.
    # Due to bounds, may need multiple passes.
    max_iter = 1000
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = (v < 1 - 1e-12)
        else:
            free = (v > 1e-12)
        count = free.sum()
        if count == 0:
            break
        delta = diff / count
        # apply delta clipped to bounds
        v_new = v.copy()
        # select free indices
        v_new[free] += delta
        # clip to [0,1]
        v_new = np.clip(v_new, 0, 1)
        diff = target - v_new.sum()
        v = v_new
    return v

import time
import numpy as np

# Utility helpers

def next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()


def enforce_sum(v: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Project v onto [0,1]^n and enforce Σv = target via random adjustments."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    n = v.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0.0:
            add = min(diff, 1.0 - v[i])
        else:
            add = max(diff, -v[i])
        v[i] += add
        diff = target - v.sum()
    return v


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small amount ε from i to j (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def correlate_full(h: np.ndarray) -> np.ndarray:
    """Linear correlation of h with (1‑h) via FFT."""
    n = h.shape[0]
    L = next_pow2(2 * n - 1)
    fh = np.fft.rfft(h, L)
    rev = (1.0 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)


def autocorr(given: np.ndarray) -> np.ndarray:
    """Autocorrelation g(k)=∑_i h_i h_{i+k} (zero‑padded)."""
    n = given.size
    L = next_pow2(2 * n - 1)
    fft_h = np.fft.rfft(given, L)
    g = np.fft.irfft(fft_h * np.conj(fft_h), L)[: 2 * n - 1]
    return g


def obj_grad_min_g(h: np.ndarray, beta: float, target: float) -> tuple:
    """
    Returns (softmax_obj, gradient, min_g, max_corr) for the current h.
    Softmax objective approximates max_k Σ_i h_i(1‑h_{i+k}).
    Gradient is ∇_h (softmax objective) respecting the derived formula.
    """
    n = h.size
    # FFT lengths
    Lc = next_pow2(3 * n - 2)

    # FFT of h (once, for convolutions)
    fft_h = np.fft.rfft(h, Lc)

    # ----- autocorrelation g(k) -----
    Lg = next_pow2(2 * n - 1)
    fft_h_small = np.fft.rfft(h, Lg)
    g = np.fft.irfft(fft_h_small * np.conj(fft_h_small), Lg)[: 2 * n - 1]

    const = h.sum()                     # should be very close to target
    corr = const - g                    # Σ_i h_i(1‑h_{i+k})
    max_corr = corr.max()

    # ----- softmax weights -----
    # w_k ∝ exp(β * corr_k)  (stable)
    bc = beta * (corr - max_corr)
    w = np.exp(bc)
    w_sum = w.sum()
    w /= w_sum                          # normalized weights, Σ w = 1

    # ----- gradient computation -----
    # conv1 = w * h (linear convolution), extract centre n entries
    fft_w = np.fft.rfft(w, Lc)
    conv1_full = np.fft.irfft(fft_w * fft_h, Lc)
    start = n - 1
    conv1 = conv1_full[start:start + n]

    w_rev = w[::-1]
    fft_w_rev = np.fft.rfft(w_rev, Lc)
    conv2_full = np.fft.irfft(fft_w_rev * fft_h, Lc)
    conv2 = conv2_full[start:start + n]

    conv_sum = conv1 + conv2
    grad = 1.0 - conv_sum               # ∇ softmax objective

    # ----- softmax objective value -----
    obj = max_corr + np.log(w_sum) / beta

    min_g = g.min()
    return obj, grad, min_g, max_corr


def pgd_optimize(h_start: np.ndarray, beta: float, deadline: float,
                 rng: np.random.Generator) -> tuple:
    """
    Simple projected gradient descent on the smoothed objective.
    Returns the best (h, C5) encountered before the deadline.
    """
    n = h_start.size
    target = n / 2.0
    h = h_start.copy()
    # initial quality
    g0 = autocorr(h)
    best_c = 1.0 - (2.0 / n) * g0.min()
    best_h = h.copy()
    step = 0.5

    while time.time() < deadline:
        obj, grad, min_g, _ = obj_grad_min_g(h, beta, target)
        # trial update
        new_h = h - step * grad
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = enforce_sum(new_h, target, rng)

        # evaluate C5 for the trial vector
        g_new = autocorr(new_h)
        c_new = 1.0 - (2.0 / n) * g_new.min()

        if c_new < best_c:
            best_c = c_new
            best_h = new_h.copy()
            step = min(step * 1.1, 1.0)          # enlarge step after improvement
        else:
            step *= 0.5                           # shrink step when no improvement
            if step < 1e-6:
                break

        h = new_h
    return best_h, best_c


def hill_climb(h_start: np.ndarray, rng: np.random.Generator,
               max_iters: int = 2000, start_step: float = 0.5) -> tuple:
    """
    Greedy hill‑climb: keep a move only if it improves C₅.
    """
    n = h_start.size
    target = n / 2.0
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)

    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c

    rng = np.random.default_rng(seed)
    start_time = time.time()

    # 2. Main optimisation loop (PGD + hill‑climb + restarts)
    # ------------------------------------------------------------------
    betas = [5.0, 20.0, 80.0, 200.0, 800.0, 2000.0, 5000.0]

    deadline_pgd = start_time + budget_s * 0.80

    for beta in betas:

        # several restarts for each beta
        for _ in range(3):

            start_vec = perturb_h(best_h, rng, scale=0.35)
            cand_h, cand_c = pgd_optimize(start_vec, beta, deadline_pgd, rng)
            if cand_c < best_c:
                best_c = cand_c

    # ------------------------------------------------------------------
    # 3. Refine with deterministic hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline_main:
        refined_h, refined_c = hill_climb(best_h, rng,
                                          max_iters=3000,
                                          start_step=0.3)
        if refined_c < best_c:
            best_c = refined_c
            best_h = refined_h.copy()

    # ------------------------------------------------------------------
    # 4. Final clean‑up (clip & exact sum)
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
