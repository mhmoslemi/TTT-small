def project_to_feasible(v, target):
    lo, hi = -2, 2
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) / 2, 0.0, 1.0)

def gradient_descent_refine(h, max_iter=2000, init_step=0.2):

    step = init_step

            v = project_to_feasible(v, target)
            c = compute_c5(v)

                best_h = v
                best_c = c
                step = alpha  # keep the successful step size
                improved = True
                break
        if not improved:

            step *= 0.5
            if step < 1e-5:
                break
    return best_h, best_c

            break
    return best_h, best_score

def compute_gradient(h, s):
    n = h.shape[0]
    g = np.zeros_like(h)
    if s >= 0:
        # (1 - h[i + s]) term
        g[:n - s] += (1 - h[s:])
        # -h[i - s] term
        g[s:] -= h[:n - s]

        g[k:] += (1 - h[:n - k])
        # -h[i + k] term for i < n - k
        g[:n - k] -= h[k:]
    return g

CURRENT_TARGET = None
CURRENT_BETA = None

def project_to_feasible(v, target):
    """Projection onto {0<=h<=1, sum(h)=target} via bisection on shift."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def worst_shift_and_value(h):
    """Return raw shift s (integer) that maximizes correlation and its raw value."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr)
    s = idx - (n - 1)
    return s, corr[idx]

def compute_gradient(h, s):
    """Gradient of raw correlation for shift s."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if s >= 0:
        g[:n - s] += (1 - h[s:])
        g[s:] -= h[:n - s]
    else:
        k = -s
        g[k:] += (1 - h[:n - k])
        g[:n - k] -= h[k:]
    return g

CURRENT_TARGET = None
CURRENT_BETA = None


def project_to_feasible(v, target):
    """Projection onto {0≤h≤1, Σh=target} via a simple bisection on the shift."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def worst_shift_and_value(h):
    """Return the raw shift s (integer) maximising correlation and its raw value."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr)
    s = idx - (n - 1)
    return s, corr[idx]


def compute_gradient(h, s):
    """Gradient of raw correlation for a given shift s."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if s >= 0:
        g[:n - s] += (1 - h[s:])
        g[s:] -= h[:n - s]
    else:
        k = -s
        g[k:] += (1 - h[:n - k])
        g[:n - k] -= h[k:]
    return g


def random_feasible_vector(n, rng):
    """Fast generation of a random feasible vector (0≤h≤1, Σh=n/2)."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)

def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""

def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
