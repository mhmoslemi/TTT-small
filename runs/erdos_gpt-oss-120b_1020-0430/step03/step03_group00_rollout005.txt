def simulated_annealing(h0, rng, n_steps, max_temp=1.0, min_temp=0.001):
    h = h0.copy()
    current_c = compute_c5(h)
    best_h = h.copy()
    best_c = current_c
    n = h.size
    for it in range(n_steps):
        T = max_temp * (min_temp / max_temp) ** (it / n_steps)
        # propose move
        i = rng.integers(0,n)
        j = rng.integers(0,n)
        if i == j:
            continue
        max_eps = min(h[i], 1 - h[j])
        if max_eps <= 0:
            continue
        eps = max_eps * 0.2 * rng.random()  # choose scale 0.2
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c_new = compute_c5(new_h)
        delta = c_new - current_c
        if delta < 0 or rng.random() < np.exp(-delta / T):
            h = new_h
            current_c = c_new
            if c_new < best_c:
                best_h = new_h.copy()
                best_c = c_new
    return best_h, best_c
```

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utility helpers ----------
def enforce_sum(h, target, rng):
    """Clip to [0,1] and adjust to make sum exactly target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i])
            add *= rng.random()
            h[i] += add
            diff -= add
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i])
            sub *= rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    """Uniform random vector repaired to satisfy Σh = n/2."""
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)

def perturb_random(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    eps_max = min(h[i], 1.0 - h[j])
    if eps_max <= 0.0:
        return h.copy()
    eps = eps_max * scale * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def compute_c5(h):
    """Discrete C5 = max_k ∫ h(x)(1-h(x+k)) dx."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

# ---------- targeted hill climbing ----------
def hill_climb_targeted(h_start, rng, max_iters, start_step, target_sum):
    h = h_start.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr)
        max_corr = corr[idx]
        c_cur = max_corr * (2.0 / n)
        k = idx - (n - 1)

        # gradient of the worst shift
        grad = np.ones(n)
        if k >= 0:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
        else:
            kk = -k
            grad[: n - kk] -= h[kk:]
            grad[kk:] -= h[: n - kk]

        new_h = perturb_random(h, rng, scale=step)  # simple random move
        # occasionally use gradient‑directed move
        if rng.random() < 0.6:
            pos = np.where((grad > 1e-9) & (h > 1e-12))[0]
            neg = np.where((grad < -1e-9) & (h < 1.0 - 1e-12))[0]
            if pos.size and neg.size:
                i = rng.choice(pos)
                j = rng.choice(neg)
                eps_max = min(h[i], 1.0 - h[j])
                if eps_max > 0:
                    eps = eps_max * step * rng.random()
                    new_h = h.copy()
                    new_h[i] -= eps
                    new_h[j] += eps

        c_new = compute_c5(new_h)

        if c_new < best_c:
            best_c = c_new
            best_h = new_h.copy()
        if c_new < c_cur:
            h = new_h

        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

# ---------- SLSQP surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp, stable
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # small repair to guarantee exact sum
    x = enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

# ---------- simulated annealing ----------
def simulated_annealing(h0, rng, n_steps, max_temp=0.5, min_temp=0.001):
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    n = h.size
    for it in range(n_steps):
        T = max_temp * (min_temp / max_temp) ** (it / n_steps)
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        eps_max = min(h[i], 1.0 - h[j])
        if eps_max <= 0.0:
            continue
        eps = eps_max * 0.2 * rng.random()
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c_new = compute_c5(new_h)
        d = c_new - cur_c
        if d < 0 or rng.random() < np.exp(-d / max(T, 1e-12)):
            h = new_h
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = new_h.copy()
    return best_h, best_c
