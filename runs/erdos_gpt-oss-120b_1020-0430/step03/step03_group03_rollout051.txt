def gradient_softmax(h, beta):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    # compute softmax probabilities
    max_corr = np.max(beta * corr)
    exp_beta_corr = np.exp(beta * corr - max_corr)
    p = exp_beta_corr / exp_beta_corr.sum()
    # gradient init
    grad = np.ones_like(h)  # term A
    # term B: - sum_s p_s * h_{i+s}
    # term C: - sum_s p_s * h_{i-s}
    # for each shift s
    # we can compute convolution via np.dot of p and shifted h.
    # Use for loops; n up to maybe 360, 2n-1 ~ 719, so loops okay.
    for s in range(-(n-1), n):
        w = p[s + n - 1]
        if w == 0:
            continue
        # contributions to grad[i] from h[i+s] and h[i-s]
        if s >= 0:
            # i + s in range for i in 0..n-1-s
            grad[:n-s] -= w * h[s:]
            grad[s:] -= w * h[:n-s]  # i-s for i>=s -> i-s = i-s indexing? Actually for i>=s, i-s >=0, i-s < n, h[i-s] = h[0:n-s]
        else:
            # s negative: let t = -s >0
            t = -s
            # i + s = i - t, valid for i>=t
            grad[t:] -= w * h[:n-t]
            # i - s = i + t, valid for i<=n-1-t
            grad[:n-t] -= w * h[t:]
    return grad

def project_onto_capped_simplex(v, target):
    # clip to [0,1] after projection onto hyperplane
    # We'll solve using bisection for lambda such that sum(min(max(v - lambda, 0), 1)) = target
    # v is numpy array.
    lo = -np.max(v)  # lower bound for lambda
    hi = np.max(v)   # upper bound for lambda

    # Let's set lo = -1e5, hi = 1e5.
    lo = -1e5
    hi = 1e5
    for _ in range(60):

        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam

    n = h.size
    target = n / 2.0

        h_new = project_onto_capped_simplex(h_new, target)

            if c < best_c:
                best_c = c
                best_h = h.copy()
    return best_h, best_c

def gradient_descent(h, beta, lr, max_iter, rng):
    n = h.size
    target = n/2.0
    best_h = h.copy()
    best_c = compute_c5(h)

def gradient_softmax(h, beta):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    # compute p (softmax weights)
    # compute max to avoid overflow
    max_val = np.max(beta * corr)
    exp_beta_corr = np.exp(beta * corr - max_val)
    p = exp_beta_corr / exp_beta_corr.sum()
    # p is length 2n-1
    # compute term1 and term2 via correlation
    term1_full = np.correlate(p, h, mode='full')
    term1 = term1_full[n-1 : n-1 + n]
    p_rev = p[::-1]
    term2_full = np.correlate(p_rev, h, mode='full')
    term2 = term2_full[n-1 : n-1 + n]
    grad = 1.0 - term1 - term2
    return grad

Check shape: term1 and term2 are length n.

def project_onto_capped_simplex(v, target):
    # bisection for lambda
    lo = -1e5
    hi = 1e5
    for _ in range(60):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

import time, math, numpy as np

# -------------------- basic utilities --------------------
def compute_c5(h):
    """Discrete C5 for step function h (dx = 2 / len(h))."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)


def project_onto_capped_simplex(v, target):
    """Euclidean projection of v onto {0<=x<=1, sum(x)=target}."""
    lo, hi = -1e5, 1e5               # wide enough bounds for the Lagrange multiplier
    for _ in range(60):               # binary search, enough for double precision
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)


def enforce_sum(h, target, rng):
    """Force sum(h)=target while keeping 0<=h<=1."""
    h[:] = project_onto_capped_simplex(h, target)
    return h


def random_feasible_vector(n, rng):
    """Generate a random feasible h with sum=n/2."""
    target = n / 2.0
    h = rng.random(n)
    h = h * target / h.sum()
    enforce_sum(h, target, rng)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount between two coordinates (preserves sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new


# -------------------- gradient of smooth max --------------------
def gradient_softmax(h, beta):
    """Gradient of (1/β)·log‑sum‑exp(β·corr) w.r.t. h."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    # softmax weights
    max_val = np.max(beta * corr)
    exp_beta = np.exp(beta * corr - max_val)
    p = exp_beta / exp_beta.sum()                # shape (2n‑1,)

    # term1 = sum_s p_s * h_{i+s}
    term1_full = np.correlate(p, h, mode="full")
    term1 = term1_full[n - 1 : n - 1 + n]

    # term2 = sum_s p_s * h_{i‑s}  (use reversed p)
    p_rev = p[::-1]
    term2_full = np.correlate(p_rev, h, mode="full")
    term2 = term2_full[n - 1 : n - 1 + n]

    grad = 1.0 - term1 - term2
    return grad


def gradient_descent(h0, beta, lr, max_iter, rng):
    """Projected gradient descent on the smooth max objective."""
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)

    for it in range(max_iter):
        grad = gradient_softmax(h, beta)
        h = h - lr * grad
        h = project_onto_capped_simplex(h, target)

        if (it % 100) == 0 or it == max_iter - 1:
            c = compute_c5(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    return best_h, best_c


# -------------------- local deterministic improvement --------------------
def local_improve(h, max_iter, rng):
    """Move mass from high‑product to low‑product locations for the worst lag."""
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)

    for _ in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        k_star = int(np.argmax(corr) - (n - 1))          # lag giving the max overlap
        i0 = max(0, -k_star)
        i1 = min(n, n - k_star)
        if i0 >= i1:
            break
        idx = np.arange(i0, i1)
        prod = h[idx] * (1 - h[idx + k_star])
        if prod.size == 0:
            break

        high_thr = np.quantile(prod, 0.80)
        low_thr = np.quantile(prod, 0.20)
        high = idx[prod > high_thr]
        low = idx[prod < low_thr]

        if high.size == 0 or low.size == 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
        else:
            i = rng.choice(high)
            j = rng.choice(low)

        h[j] += eps
        # (sum unchanged, bounds respected)

        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    # final projection for safety
    best_h = project_onto_capped_simplex(best_h, target)
