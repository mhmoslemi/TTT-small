def fft_convolve(a, b):
    n = a.shape[0] + b.shape[0] - 1
    L = 1 << ((n-1).bit_length())
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    conv = np.fft.irfft(fa*fb, L)
    return conv[:n]
```

def target_hill_climb(h, rng, max_iters=1000, step=0.1):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iters):
        corr = correlate_full(best_h)
        k_max = np.argmax(corr)
        shift = k_max - (n-1)
        # choose a random index i for which i+shift in range
        if shift >= 0:
            i = rng.integers(0, n - shift)
            j = i + shift
        else:
            i = rng.integers(-shift, n)
            j = i + shift
        # compute potential improvement by moving epsilon from high to low
        # if best_h[i] > best_h[j], we can move from i to j to reduce cross-corr
        if best_h[i] > best_h[j]:
            eps = min(step, best_h[i] - best_h[j])
            new_h = best_h.copy()
            new_h[i] -= eps
            new_h[j] += eps
        else:
            # maybe move opposite direction
            eps = min(step, best_h[j] - best_h[i])
            new_h = best_h.copy()
            new_h[j] -= eps
            new_h[i] += eps
        # enforce bounds and sum (should be preserved)
        new_h = np.clip(new_h, 0, 1)
        # sum unchanged because we moved same epsilon, but due to clipping may need adjust
        # adjust by redistributing any diff
        diff = best_h.sum() - new_h.sum()
        if abs(diff) > 1e-12:
            new_h = enforce_sum_exact(new_h, best_h.sum(), rng)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h
    return best_h, best_c

            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        # allocate equally
        delta = diff / np.count_nonzero(mask)
        # update with clipping
        h[mask] += delta
        # enforce bounds again
        h = np.clip(h, 0.0, 1.0)

def project_to_feasible(h, target, max_iter=200):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        # indices that can move in direction of diff
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        delta = diff / np.count_nonzero(mask)
        # add delta but clip
        add = np.minimum(delta, 1.0 - h[mask]) if diff > 0 else np.maximum(delta, -h[mask])
        # adjust diff for possible saturation
        actual = add.sum()
        h[mask] += add
        # recompute diff after actual adjustment
        diff = target - h.sum()
        # if actual small, break
        if abs(actual) < 1e-12:
            break
    return h
```

    best_c = compute_c5(best_h)
    for it in range(max_iters):

        # find shift with maximum cross-corr
        k_max = np.argmax(corr)
        shift = k_max - (n - 1)
        # choose random i that has partner j = i + shift (within bounds)
        if shift >= 0:
            i = rng.integers(0, n - shift)

        else:
            i = rng.integers(-shift, n)
            j = i + shift

            eps = min(step, best_h[i] - best_h[j])

            new_h[i] -= eps
            new_h[j] += eps

def simulated_annealing_binary(h, rng, max_iters, T0, Tf):
    cur_h = h
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    for it in range(max_iters):
        T = T0 * (Tf/T0) ** (it / max_iters)
        # propose: swap a 1 and a 0
        ones = np.where(cur_h > 0.5)[0]
        zeros = np.where(cur_h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        new_h = cur_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
    return best_h, best_c
```

def hill_climb_shift(h, rng, max_iters=20000, step_scale=0.2):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(best_h)
    for it in range(max_iters):
        # compute correlation
        corr = correlate_full(best_h)
        k_max = np.argmax(corr)
        shift = k_max - (n - 1)
        # select random i valid for shift
        if shift >= 0:
            i = rng.integers(0, n - shift)
            j = i + shift
        else:
            i = rng.integers(-shift, n)
            j = i + shift
        # Determine direction: want to move mass from higher to lower
        if best_h[i] > best_h[j]:
            epsilon = min(step_scale * (best_h[i] - best_h[j]), best_h[i])
            # also ensure not exceed upper bound at j
            epsilon = min(epsilon, 1 - best_h[j])
            if epsilon <= 0:
                continue
            new_h = best_h.copy()
            new_h[i] -= epsilon
            new_h[j] += epsilon
        else:
            epsilon = min(step_scale * (best_h[j] - best_h[i]), best_h[j])
            epsilon = min(epsilon, 1 - best_h[i])
            if epsilon <= 0:
                continue
            new_h = best_h.copy()
            new_h[j] -= epsilon
            new_h[i] += epsilon

        new_c = compute_c5(new_h)
        if new_c < best_c - 1e-12:

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic feasibility utilities
# ----------------------------------------------------------------------
def project_to_feasible(h, target, max_iter=200):
    """Clip to [0,1] and adjust to meet sum == target (simple water‑filling)."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        delta = diff / np.count_nonzero(mask)
        if diff > 0:
            add = np.minimum(delta, 1.0 - h[mask])
        else:
            add = np.maximum(delta, -h[mask])
        h[mask] += add
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    return project_to_feasible(h, n / 2.0)


# ----------------------------------------------------------------------
# Correlation utilities (full linear correlation via FFT)
# ----------------------------------------------------------------------
def correlate_full(h):
    """Full linear correlation of h with (1‑h) (zero‑padded FFT)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Discrete C₅ for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)


# ----------------------------------------------------------------------
# Convolution helper (full linear convolution via FFT)
# ----------------------------------------------------------------------
def fft_convolve(a, b):
    """Full linear convolution using FFT."""
    n = a.shape[0] + b.shape[0] - 1
    L = 1 << ((n - 1).bit_length())
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    conv = np.fft.irfft(fa * fb, L)[:n]
    return conv


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) – useful for gradient steps
# ----------------------------------------------------------------------
def smooth_objective(h, beta):
    corr = correlate_full(h)
    m = corr.max()
    return (m + np.log(np.sum(np.exp(beta * (corr - m))))) / beta


# ----------------------------------------------------------------------
# Projected gradient descent on the smooth surrogate
# ----------------------------------------------------------------------
def projected_gradient_descent(h0, beta, target, rng, max_iter=80, step0=0.5):
    n = h0.size
    h = h0.copy()
    best_val = smooth_objective(h, beta)
    step = step0
    for it in range(max_iter):
        corr = correlate_full(h)
        m = corr.max()
        w = np.exp(beta * (corr - m))
        w_sum = w.sum()
        if w_sum == 0:
            break
        w /= w_sum
        # two convolutions
        conv1 = fft_convolve(w, h)
        conv2 = fft_convolve(w[::-1], h)
        off = n - 1
        g = -(conv1[off:off + n] + conv2[off:off + n])
        step = step0 / (1 + it)
        h_new = h - step * g
        h_new = project_to_feasible(h_new, target)
        val = smooth_objective(h_new, beta)
        if val < best_val:
            best_val = val
            h = h_new
            step0 = step * 1.1
        else:
            step0 *= 0.5
    return h, best_val


# ----------------------------------------------------------------------
# Targeted hill‑climb (focus on the worst shift)
# ----------------------------------------------------------------------
def hill_climb_shift(h, rng, max_iters=20000, step_scale=0.2):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(best_h)
    for it in range(max_iters):
        corr = correlate_full(best_h)
        k_max = int(np.argmax(corr))
        shift = k_max - (n - 1)
        if shift >= 0:
            i = rng.integers(0, n - shift)
            j = i + shift
        else:
            i = rng.integers(-shift, n)
            j = i + shift
        hi, hj = best_h[i], best_h[j]
        if hi == hj:
            continue
        if hi > hj:
            eps = min(step_scale * (hi - hj), hi, 1.0 - hj)
            if eps <= 0:
                continue
            new_h = best_h.copy()
            new_h[i] -= eps
            new_h[j] += eps
        else:
            eps = min(step_scale * (hj - hi), hj, 1.0 - hi)
            if eps <= 0:
                continue
            new_h = best_h.copy()
            new_h[j] -= eps
            new_h[i] += eps
        new_h = project_to_feasible(new_h, best_h.sum())
        new_c = compute_c5(new_h)
        if new_c + 1e-12 < best_c:
            best_c = new_c
            best_h = new_h
        if (it + 1) % 5000 == 0 and rng.random() < 0.2:
            # occasional random perturbation to escape plateaus
            ii = rng.integers(0, n)
            jj = rng.integers(0, n)
            if ii != jj:
                max_eps = min(best_h[ii], 1.0 - best_h[jj])
                if max_eps > 0:
                    eps = rng.random() * max_eps * step_scale
                    tmp = best_h.copy()
                    tmp[ii] -= eps
                    tmp[jj] += eps
                    tmp = project_to_feasible(tmp, best_h.sum())
                    c_tmp = compute_c5(tmp)
                    if c_tmp < best_c:
                        best_c = c_tmp
                        best_h = tmp
    return best_h, best_c


def multi_stage_hill_climb(h, rng):
    stages = [(0.5, 5000), (0.2, 10000), (0.05, 15000)]
    cur_h, cur_c = h, compute_c5(h)
    for scale, iters in stages:
        cur_h, cur_c = hill_climb_shift(cur_h, rng, max_iters=iters, step_scale=scale)
    return cur_h, cur_c


# ----------------------------------------------------------------------
# Binary simulated annealing (swap a 1 and a 0)
# ----------------------------------------------------------------------
def simulated_annealing_binary(h, rng, max_iters=60000, T0=0.02, Tf=1e-5):
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    n = h.size
    for it in range(max_iters):
        T = T0 * ((Tf / T0) ** (it / max_iters))
        ones = np.where(cur_h > 0.5)[0]
        zeros = np.where(cur_h <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        new_h = cur_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
    return best_h, best_c

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    n = n0
