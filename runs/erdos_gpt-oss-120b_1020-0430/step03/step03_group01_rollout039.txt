def project_onto_box_and_sum(x, target):
    # find lambda such that sum(clamp(x+lambda,0,1)) = target.
    # Use binary search between low = -1, high = 1.
    lo, hi = -1.0, 1.0

        s = np.clip(x+mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x+hi, 0.0, 1.0)

def grad_descent(h0, target, beta_list, max_iter_per_beta, step0, rng, time_budget, start_time):
    h = h0.copy()
    bestlocal_h = h.copy()
    bestlocal_c = compute_c5(h)
    for beta in beta_list:
        # compute possible iterations based on remaining time
        remaining = time_budget - (time.time() - start_time)
        if remaining < 0.5:
            break
        max_iter = min(max_iter_per_beta, int(remaining * 20))  # approx 20 iters per second
        # if max_iter <=0: break
        for it in range(max_iter):
            corr = np.correlate(h, 1 - h, mode='full')
            bc = beta * corr
            max_bc = np.max(bc)
            w = np.exp(bc - max_bc)
            w /= w.sum()
            b = 1 - h
            w_rev = w[::-1]
            conv1 = np.convolve(w_rev, b, mode='full')
            conv2 = np.convolve(w, h, mode='full')
            offset = len(h) - 1
            grad = conv1[offset:offset+len(h)] - conv2[offset:offset+len(h)]

                cur_c = compute_c5(h)
                if cur_c < bestlocal_c:
                    bestlocal_c = cur_c
                    bestlocal_h = h.copy()
    return bestlocal_h, bestlocal_c

import time
import numpy as np

def _proj_sum_box(x, target):
    """Project x onto the box [0,1] while enforcing sum(x)=target."""
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + hi, 0.0, 1.0)


def _grad_descent(h0, target, beta_list, iters_per_beta, step0, rng,
                  start_time, budget):
    """Smooth‑max gradient descent with projection."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(best_h)
    v = np.zeros_like(h)            # momentum
    for beta in beta_list:
        remaining = budget - (time.time() - start_time)
        if remaining < 0.5:
            break
        max_iter = min(iters_per_beta, int(remaining * 20))
        if max_iter <= 0:
            break
        for it in range(max_iter):
            corr = np.correlate(h, 1 - h, mode="full")
            bc = beta * corr
            max_bc = np.max(bc)
            w = np.exp(bc - max_bc)
            w /= w.sum()
            b = 1 - h
            w_rev = w[::-1]
            conv1 = np.convolve(w_rev, b, mode="full")
            conv2 = np.convolve(w, h, mode="full")
            off = len(h) - 1
            grad = conv1[off : off + len(h)] - conv2[off : off + len(h)]
            v = 0.9 * v + grad
            alpha = step0 / (1.0 + it / 100.0)
            h = _proj_sum_box(h - alpha * v, target)
            if (it + 1) % 30 == 0:
                c = compute_c5(h)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
        if rng.random() < 0.3:            # tiny jitter after each β
            h = _proj_sum_box(h + rng.normal(scale=0.02, size=len(h)),
                              target)
    return best_h, best_c
