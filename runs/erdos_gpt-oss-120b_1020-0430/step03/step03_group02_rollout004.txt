import time, numpy as np

def compute_c5(h):
    n = h.shape[0]
    nfft = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, n=nfft)
    G = np.fft.rfft(1.0-h, n=nfft)
    corr = np.fft.irfft(H * np.conj(G), n=nfft)[:2*n-1]
    return np.max(corr) * (2.0 / n)

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.shape[0]
    # While diff not zero and attempts left
    # Use small iterations to adjust gradually.
    while abs(diff) > 1e-12:

def guided_perturb(h, grad, rng, scale=0.5):
    i = np.argmax(grad)
    j = np.argmin(grad)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    # if diff very small, done
    if abs(diff) < 1e-12:
        return h
    # We'll adjust using random selection but up to max attempts
    attempts = 0
    max_attempts = n * 10
    while abs(diff) > 1e-12 and attempts < max_attempts:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-12:
            attempts += 1
            continue
        h[i] += add
        diff = target - h.sum()
        attempts = 0  # reset attempts after successful adjustment
    # if not converged (should not happen), scale linearly
    if abs(diff) > 1e-12:
        # distribute uniformly among all entries within bounds
        # simple approach:
        # compute residual capacity per index
        if diff > 0:
            caps = 1.0 - h
        else:
            caps = -h
        total_cap = caps.sum()
        if total_cap > 0:
            h += caps / total_cap * diff
    # clip final
    np.clip(h, 0.0, 1.0, out=h)
    # final correct using scaling
    # ensure mass matches (small numeric tolerance)
    return h

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------- global helpers -----------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 using FFT for speed."""
    n = h.shape[0]
    nfft = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=nfft)
    G = np.fft.rfft(1.0 - h, n=nfft)
    corr = np.fft.irfft(H * np.conj(G), n=nfft)[: 2 * n - 1]
    return float(np.max(corr) * (2.0 / n))


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h)==target while staying in [0,1]."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    n = h.size
    attempts = 0
    max_attempts = n * 10
    while abs(diff) > 1e-12 and attempts < max_attempts:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-12:
            attempts += 1
            continue
        h[i] += add
        diff = target - h.sum()
        attempts = 0
    if abs(diff) > 1e-12:
        # fallback linear redistribution
        if diff > 0:
            caps = 1.0 - h
        else:
            caps = -h
        cap_sum = caps.sum()
        if cap_sum > 0:
            h += caps / cap_sum * diff
    np.clip(h, 0.0, 1.0, out=h)
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random vector in [0,1] with sum == n/2."""
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, target, rng)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst shift overlap."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray, rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer epsilon from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                     max_iters: int = 2000, step_scale: float = 0.5) -> tuple:
    """Local refinement using gradient‑directed transfers."""
    best = h.copy()
    best_c = compute_c5(best)
    scale = step_scale
    for it in range(max_iters):
        grad = worst_gradient(best)
        nh = guided_perturb(best, grad, rng, scale=scale)
        c = compute_c5(nh)
        if c < best_c:
            best, best_c = nh, c
        if (it + 1) % max(1, max_iters // 5) == 0:
            scale = max(scale * 0.7, 1e-4)
    return best, best_c


# -------------- Soft‑max SLSQP refinement (global vars) --------------
def sum_eq_constraint(x: np.ndarray) -> float:
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x: np.ndarray) -> float:
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 500) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj, h0, method="SLSQP", bounds=bounds,
        constraints=cons, options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, CURRENT_TARGET, rng)
    return x


# --------------------------- GA core ---------------------------
def genetic_optimize(n: int, rng: np.random.Generator, time_limit: float) -> tuple:
    """Simple GA with elitism and occasional local refinement."""
    target = n / 2.0
    popsize = 60
    # initialise population
    pop = []
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size != n:
            x_old = np.linspace(0.0, 2.0, h0.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(x_new, x_old, h0)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, target, rng)
        pop.append(h0)
    while len(pop) < popsize:
        pop.append(random_feasible_vector(n, rng))
    pop = np.array(pop)
    fitness = np.array([compute_c5(ind) for ind in pop])
    best_idx = np.argmin(fitness)
    best = pop[best_idx].copy()
    best_f = fitness[best_idx]
    start = time.time()
    generation = 0
    max_generations = 2000
    tournament_k = 3
    cx_prob = 0.7
    mut_rate = 0.12
    mut_delta = 0.25
    while generation < max_generations and (time.time() - start) < time_limit:
        generation += 1
        new_pop = [best.copy()]  # elitism
        while len(new_pop) < popsize:
            # tournament selection
            idx = rng.integers(0, popsize, size=tournament_k)
            p1 = pop[idx[np.argmin(fitness[idx])]]
            idx = rng.integers(0, popsize, size=tournament_k)
            p2 = pop[idx[np.argmin(fitness[idx])]]
            # crossover
            if rng.random() < cx_prob:
                lam = rng.random()
                child = lam * p1 + (1.0 - lam) * p2
            else:
                child = p1.copy()
            # mutation
            mask = rng.random(child.shape) < mut_rate
            if mask.any():
                pert = (rng.random(child.shape) * 2.0 - 1.0) * mut_delta
                child = child + mask * pert
            child = np.clip(child, 0.0, 1.0)
            child = enforce_sum(child, target, rng)
            new_pop.append(child)
        pop = np.array(new_pop)
        fitness = np.array([compute_c5(ind) for ind in pop])
        cur_best_idx = np.argmin(fitness)
        cur_best_f = fitness[cur_best_idx]
        if cur_best_f < best_f - 1e-9:
            best = pop[cur_best_idx].copy()
            best_f = cur_best_f
            # occasional local refinement on new best
            if (time.time() - start) < time_limit * 0.8:
                refined, refined_f = guided_hill_climb(best, rng, max_iters=1200, step_scale=0.4)
                if refined_f < best_f:
                    best, best_f = refined, refined_f
    return best, best_f

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # Primary search on n = 192 (or the size of the parent if provided)
    n = 192
    if "initial_h_values" in globals():
        n = int(len(initial_h_values))
    time_for_ga = (budget_s - (time.time() - start)) * 0.70
    best_h, best_f = genetic_optimize(n, rng, time_limit=time_for_ga)

    # Final intensive local refinement
    if time.time() - start < budget_s * 0.95:
        best_h, best_f = guided_hill_climb(best_h, rng, max_iters=3000, step_scale=0.3)

    # One round of soft‑max SLSQP to polish
    if time.time() - start < budget_s * 0.97:
        for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0):
            if time.time() - start > budget_s * 0.98:
                break
            best_h = slsqp_optimize(best_h, beta, rng, max_iter=500)
            best_f = compute_c5(best_h)

    # Optional up‑sampling if time permits
    if time.time() - start < budget_s * 0.99:
        if best_h.size * 2 <= 300:
            up = np.repeat(best_h, 2)
            up, up_f = guided_hill_climb(up, rng, max_iters=2000, step_scale=0.3)
            if up_f < best_f:
                best_h, best_f = up, up_f

    # Ensure feasibility before returning
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_h.size / 2.0
    best_h = enforce_sum(best_h, target, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
