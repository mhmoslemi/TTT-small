def gradient_softmax(h, beta):
    n = h.size
    conv = correlate_full(h)   # size 2n-1
    # compute weights
    m = conv.max()
    exp_vals = np.exp(beta * (conv - m))
    w = exp_vals / exp_vals.sum()
    # compute gradient vector
    g = np.zeros_like(h)
    # lag from -(n-1) to n-1
    for lag in range(-n+1, n):
        wk = w[lag + n - 1]
        if wk == 0:
            continue
        # compute term2, term3
        if lag >= 0:
            term2 = np.concatenate((h[lag:], np.zeros(lag)))
            term3 = np.concatenate((np.zeros(lag), h[:n - lag]))
        else:
            L = -lag
            term2 = np.concatenate((np.zeros(L), h[:n - L]))
            term3 = np.concatenate((h[L:], np.zeros(L)))
        g += wk * (1.0 - term2 - term3)
    return g
```

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    n = h.size
    # use water-filling
    if diff > 0:
        cap = 1.0 - h
        total_cap = cap.sum()
        if total_cap > 0:
            add = diff * cap / total_cap
            h = h + add
        else:
            # fallback random
            pass
    else:
        cap = h
        total_cap = cap.sum()
        if total_cap > 0:
            sub = (-diff) * cap / total_cap
            h = h - sub
        else:
            pass
    # after adjustment may exceed bounds due to numerical error; clip and repeat small random fix
    np.clip(h, 0.0, 1.0, out=h)
    # ensure sum exact within tolerance:
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # final small correction via random index adjustments
        for _ in range(1000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
    return h

    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # we need to add diff, distribute proportionally to remaining capacity
        cap = 1.0 - h
        total_cap = cap.sum()
        if total_cap > 0:
            h = h + diff * cap / total_cap
        else:

            h = h + diff / h.size

import time
import numpy as np
from scipy.optimize import minimize

# -------------------------------------------------------------
# Basic feasibility utilities
# -------------------------------------------------------------
def enforce_sum(h, target, rng):
    """Random walk correction: adjust h so that sum(h)==target while 0<=h<=1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def enforce_sum_fast(h, target):
    """Deterministic projection onto the hyperplane sum=h_target with box [0,1]."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        cap = 1.0 - h
        total = cap.sum()
        if total > 0:
            h = h + diff * cap / total
    else:
        cap = h
        total = cap.sum()
        if total > 0:
            h = h + diff * cap / total
    # final clean‑up
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
    return h


def random_feasible_vector(n, rng):
    """Roughly uniform random point in [0,1]^n with Σh = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.5
        h[i] -= eps
        h[j] += eps
    np.clip(h, 0.0, 1.0, out=h)
    return enforce_sum_fast(h, n / 2.0)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


# -------------------------------------------------------------
# Correlation and objective
# -------------------------------------------------------------
def correlate_full(h):
    """Linear correlation of h with (1‑h) (full mode)."""
    rev = (1.0 - h)[::-1]
    return np.convolve(h, rev, mode="full")


def compute_c5(h):
    """Exact C₅ for a discrete step function."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)


# -------------------------------------------------------------
# Gradient of the smooth (log‑sum‑exp) surrogate
# -------------------------------------------------------------
def gradient_softmax(h, beta):
    """∇_h  (1/β) log Σ_k exp(β·corr_k) ."""
    n = h.shape[0]
    conv = correlate_full(h)               # length 2n‑1
    m = conv.max()
    exp_vals = np.exp(beta * (conv - m))
    w = exp_vals / exp_vals.sum()          # softmax weights, sum=1

    # term_minus = Σ_k w_k * h_{i‑k}
    conv_minus = np.convolve(w, h, mode="full")
    # term_plus = Σ_k w_k * h_{i+k}
    conv_plus = np.convolve(w[::-1], h, mode="full")

    offset = n - 1
    term_minus = conv_minus[offset: offset + n]
    term_plus = conv_plus[offset: offset + n]

    return 1.0 - term_plus - term_minus


def gradient_descent_smooth(h0, rng, deadline):
    """Projected gradient descent on the smoothed surrogate."""
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)

    beta = 5.0
    base_lr = 0.2
    it = 0
    while time.time() < deadline:
        # increase β gradually (focus on the true max)
        if it % 300 == 0 and beta < 800:
            beta = min(beta * 1.3, 800)

        g = gradient_softmax(h, beta)
        lr = base_lr / (1 + it * 0.001)
        h_new = h - lr * g
        np.clip(h_new, 0.0, 1.0, out=h_new)
        h_new = enforce_sum_fast(h_new, target)

        c = compute_c5(h_new)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        h = h_new
        it += 1

        # occasional tiny random jitter to avoid stagnation
        if rng.random() < 0.005:
            noise = rng.normal(scale=0.01, size=n)
            h_tmp = h + noise
            np.clip(h_tmp, 0.0, 1.0, out=h_tmp)
            h_tmp = enforce_sum_fast(h_tmp, target)
            h = h_tmp
    return best_h, best_c

def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay = max(1, max_iters // 10)
    for it in range(max_iters):

        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c

# -------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    conv = correlate_full(x)
    bc = CURRENT_BETA * conv
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # final sum correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 0) & (x < 1)
        if mask.any():
            x[mask] += diff / mask.sum()
        np.clip(x, 0.0, 1.0, out=x)
        x = enforce_sum_fast(x, CURRENT_TARGET)
    return x
