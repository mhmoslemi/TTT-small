            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i])
            add *= rng.random()
            h[i] += add
            diff -= add
        else:  # need to decrease
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i])
            sub *= rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    """Random vector in [0,1] with exact sum n/2."""
    h = rng.random(n)
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, n / 2.0, rng)
    return h

def compute_c5(h):
    """Discrete C5 for step vector h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def perturb_random(h, rng, scale=0.2):
    """Transfer a small amount from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def perturb_targeted(h, grad, rng, scale=0.2):
    """Move mass from positive‑gradient indices to negative‑gradient ones."""
    pos = np.where((grad > 1e-9) & (h > 1e-12))[0]
    neg = np.where((grad < -1e-9) & (h < 1.0 - 1e-12))[0]
    if pos.size == 0 or neg.size == 0:
        return perturb_random(h, rng, scale)
    i = rng.choice(pos)
    j = rng.choice(neg)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return perturb_random(h, rng, scale)
    eps = max_eps * scale * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb_targeted(h_start, rng, max_iters, start_step, target_sum):
    """Greedy descent focusing on the worst shift."""
    h = h_start.copy()
    n = h.size
    best_h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode="full")
    best_c = np.max(corr) * (2.0 / n)
    step = start_step
    decay_interval = max(1, max_iters // 8)

    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr)
        max_corr = corr[idx]
        c_current = max_corr * (2.0 / n)
        k = idx - (n - 1)

        # gradient of the worst shift
        grad = np.ones(n)
        if k >= 0:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
        else:
            kk = -k
            grad[: n - kk] -= h[kk:]
            grad[kk:] -= h[: n - kk]

        new_h = perturb_targeted(h, grad, rng, scale=step)
        corr_new = np.correlate(new_h, 1.0 - new_h, mode="full")
        c_new = np.max(corr_new) * (2.0 / n)

        if c_new < best_c:
            best_c = c_new
            best_h = new_h.copy()
        if c_new < c_current:
            h = new_h

        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)

    return best_h, best_c

# ----- SLSQP surrogate (softmax of correlations) -----
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # ensure exact sum
    rng = np.random.default_rng()
    x = enforce_sum(x, CURRENT_TARGET, rng)
    return x

# ----- main entry point -----
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ---- initial vector ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = random_feasible_vector(180, rng)
    h0 = enforce_sum(np.clip(h0, 0.0, 1.0), h0.size / 2.0, rng)

    # ---- decide base resolution ----
    n0 = h0.size
    # ensure we have at least 360 points for reasonable shape
    if n0 < 360:
        n_small = n0
        n = 2 * n_small
        xs0 = (np.arange(n_small) + 0.5) * (2.0 / n_small)
        xs = (np.arange(n) + 0.5) * (2.0 / n)
        h = np.interp(xs, xs0, h0)
    else:
        n = n0
        h = h0.copy()
    h = enforce_sum(np.clip(h, 0.0, 1.0), n / 2.0, rng)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ----- parameters -----
    high_n = max(720, n)          # final target resolution
    upsample_done = False
    # schedule for SLSQP betas (more aggressive for high n)
    beta_schedule_low = [5.0, 20.0, 80.0, 200.0, 800.0]
    beta_schedule_high = [5.0, 20.0, 80.0, 200.0, 800.0, 2000.0, 5000.0]

    # ----- optimization loop -----
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.99:
            break

        # up‑sample when time permits and we haven't yet
        if (not upsample_done) and elapsed > budget_s * 0.30 and n < high_n:
            # interpolate best_h onto finer grid
            xs_old = (np.arange(n) + 0.5) * (2.0 / n)
            n = high_n
            xs_new = (np.arange(n) + 0.5) * (2.0 / n)
            h_fine = np.interp(xs_new, xs_old, best_h)
            best_h = enforce_sum(np.clip(h_fine, 0.0, 1.0), n / 2.0, rng)
            best_c = compute_c5(best_h)
            upsample_done = True
            continue  # restart loop with new n

        # pick starting point
        if rng.random() < 0.7:
            start = perturb_random(best_h, rng, scale=0.3)
        else:
            start = random_feasible_vector(n, rng)

        # hill climbing
        max_iters_hc = 3000 if n >= 720 else 1800
        h_local, c_local = hill_climb_targeted(
            start, rng, max_iters=max_iters_hc, start_step=0.6, target_sum=n / 2.0
        )
        if c_local < best_c:
            best_h, best_c = h_local.copy(), c_local

        # SLSQP refinement
        betas = beta_schedule_high if n >= 720 else beta_schedule_low
        for beta in betas:
            if time.time() - start_time >= budget_s * 0.95:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, max_iter=600 if n >= 720 else 400)
            except Exception:
                continue
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_h, best_c = h_ref.copy(), c_ref

        # tiny random tweaks (micro‑search)
        for _ in range(5):
            if time.time() - start_time >= budget_s * 0.99:
                break
            new_h = perturb_random(best_h, rng, scale=0.02)
            c_new = compute_c5(new_h)
            if c_new < best_c:
                best_h, best_c = new_h.copy(), c_new

    # ----- final cleanup -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
