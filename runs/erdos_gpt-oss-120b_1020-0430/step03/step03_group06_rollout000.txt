def enforce_sum_bin(h, target, rng):
    # h binary array (0/1) dtype int, target integer (n/2)
    diff = int(target - h.sum())
    if diff > 0:
        # need to turn diff zeros to ones
        zero_idxs = np.flatnonzero(h == 0)
        if zero_idxs.size >= diff:
            chosen = rng.choice(zero_idxs, size=diff, replace=False)
            h[chosen] = 1
        else:
            # fallback: set everything to 1 then reduce; but diff should be small.
            h[:] = 1
            # then set extra ones to zero
            surplus = h.sum() - target
            one_idxs = np.flatnonzero(h == 1)
            chosen = rng.choice(one_idxs, size=surplus, replace=False)
            h[chosen] = 0
    elif diff < 0:
        # need to turn -diff ones to zero
        one_idxs = np.flatnonzero(h == 1)
        chosen = rng.choice(one_idxs, size=-diff, replace=False)
        h[chosen] = 0
    # ensure binary
    h = np.clip(h, 0, 1).astype(int)
    return h
```

def simulated_annealing_binary(h0, rng, time_limit):
    start = time.time()
    n = h0.size
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    target = n // 2
    # Determine schedule: we can compute total steps as max_steps = 200000 maybe
    max_steps = int(1e6)  # large
    # precompute list of indices with ones and zeros to quickly pick swap
    # keep sets updated after each swap
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        # Temperature schedule based on fraction of allocated steps done?
        # Use exponential cooling: T = start_temp * (end_temp/start_temp)^(elapsed / time_limit)
        start_temp = 5e-3
        end_temp = 1e-6
        T = start_temp * (end_temp / start_temp) ** (elapsed / time_limit)
        # Choose random swap
        # get list of ones and zeros each time? maintain arrays?
        one_idxs = np.flatnonzero(h == 1)
        zero_idxs = np.flatnonzero(h == 0)
        if one_idxs.size == 0 or zero_idxs.size == 0:
            break
        i = rng.choice(one_idxs)
        j = rng.choice(zero_idxs)
        # swap
        h[i] = 0
        h[j] = 1
        new_c = compute_c5(h)
        delta = new_c - cur_c
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0:
                accept = rng.random() < np.exp(-delta / T)
        if accept:
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1
            h[j] = 0
    return best_h, best_c
```

def worst_shift_descent(h, rng, time_limit, start_step=0.5, end_step=1e-4):
    start = time.time()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    n = h.size
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        step = start_step * (end_step/start_step) ** frac  # exponential decay
        # compute correlation and shift
        corr = np.correlate(h, 1 - h, mode='full')
        k = np.argmax(corr)
        shift = k - (n - 1)
        # find source and dest
        if shift >=0:
            valid_idx = np.arange(n - shift)
            src_mask = h[valid_idx] > 1e-12
            dst_mask = h[valid_idx] < 1 - 1e-12
            src_idx = valid_idx[src_mask]
            dst_idx = valid_idx[dst_mask]
            if src_idx.size == 0 or dst_idx.size == 0:
                # maybe skip
                continue
            src_scores = (1 - h[src_idx + shift]) * h[src_idx]
            i_src = src_idx[np.argmax(src_scores)]
            # dest choose minimal (1 - h + maybe also h low)
            dest_vals = 1 - h[dst_idx + shift]
            j_dest = dst_idx[np.argmin(dest_vals)]
        else:
            shift_abs = -shift
            valid_idx = np.arange(shift_abs, n)
            src_mask = h[valid_idx] > 1e-12
            dst_mask = h[valid_idx] < 1 - 1e-12
            src_idx = valid_idx[src_mask]
            dst_idx = valid_idx[dst_mask]
            if src_idx.size == 0 or dst_idx.size == 0:
                continue
            src_scores = (1 - h[src_idx - shift_abs]) * h[src_idx]
            i_src = src_idx[np.argmax(src_scores)]
            dest_vals = 1 - h[dst_idx - shift_abs]
            j_dest = dst_idx[np.argmin(dest_vals)]
        max_eps = min(h[i_src], 1 - h[j_dest])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps * step
        if eps <= 0:
            continue
        # apply
        h[i_src] -= eps
        h[j_dest] += eps
        new_c = compute_c5(h)
        if new_c < cur_c:
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            # revert
            h[i_src] += eps
            h[j_dest] -= eps
    return best_h, best_c

def simulated_annealing_binary(h0, rng, time_limit):
    start = time.time()
    n = h0.size
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    # keep track of indices with 1 and 0 for fast sampling
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        # temperature
        start_temp = 5e-3
        end_temp = 1e-6
        T = start_temp * (end_temp / start_temp) ** (elapsed / time_limit)
        # choose random source and dest
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0
        h[j] = 1
        new_c = compute_c5(h)
        delta = new_c - cur_c
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0:
                accept = rng.random() < np.exp(-delta / T)
        if accept:
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            # revert
            h[i] = 1
            h[j] = 0
    return best_h, best_c

import time
import numpy as np

# --------------------------------------------------------------
# Utility: exact discrete C5 for a step function on [0,2]
# --------------------------------------------------------------
def compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx  (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


# --------------------------------------------------------------
# Sum/box enforcement helpers
# --------------------------------------------------------------
def enforce_sum_continuous(h, target, rng, max_iter=5000):
    """Adjust h (float) so that sum(h) == target while 0<=h<=1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-12:
            continue
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(1000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            if abs(add) < 1e-12:
                continue
            h[i] += add
            diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def enforce_sum_bin(h, target, rng):
    """Adjust binary vector h (0/1) so that sum(h) == target (int)."""
    cur = int(h.sum())
    diff = int(target - cur)
    n = h.size
    if diff > 0:
        zeros = np.flatnonzero(h == 0)
        if zeros.size >= diff:
            idx = rng.choice(zeros, size=diff, replace=False)
            h[idx] = 1
        else:
            h[:] = 1
            surplus = h.sum() - target
            ones = np.flatnonzero(h == 1)
            idx = rng.choice(ones, size=int(surplus), replace=False)
            h[idx] = 0
    elif diff < 0:
        ones = np.flatnonzero(h == 1)
        idx = rng.choice(ones, size=-diff, replace=False)
        h[idx] = 0
    return np.clip(h, 0, 1).astype(int)


def random_binary_feasible(n, rng):
    """Random binary vector with exactly n/2 ones."""
    target = n // 2
    h = np.zeros(n, dtype=int)
    ones = rng.choice(n, size=target, replace=False)
    h[ones] = 1
    return h


# --------------------------------------------------------------
# Simulated annealing on binary vectors
# --------------------------------------------------------------
def simulated_annealing_binary(h0, rng, time_limit):
    start = time.time()
    n = h0.size
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    while True:
        if time.time() - start >= time_limit:
            break
        # temperature schedule (exponential)
        start_temp = 5e-3
        end_temp = 1e-6
        elapsed = time.time() - start
        T = start_temp * (end_temp / start_temp) ** (elapsed / time_limit)

        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # propose swap
        h[i] = 0
        h[j] = 1
        new_c = compute_c5(h)
        delta = new_c - cur_c
        accept = False
        if delta < 0:
            accept = True
        else:
            accept = rng.random() < np.exp(-delta / T) if T > 0 else False
        if accept:
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            # revert
            h[i] = 1
            h[j] = 0
    return best_h, best_c


# --------------------------------------------------------------
# Continuous "worst-shift" descent
# --------------------------------------------------------------
def worst_shift_descent(h0, rng, time_limit, start_step=0.5, end_step=1e-4):
    """Move mass from high-contributing indices to low‑contributing ones."""
    start = time.time()
    h = h0.copy()
    n = h.size
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    while True:
        if time.time() - start >= time_limit:
            break
        # decaying step size
        elapsed = time.time() - start
        frac = elapsed / time_limit
        step = start_step * (end_step / start_step) ** frac

        # locate shift with maximal overlap
        corr = np.correlate(h, 1 - h, mode="full")
        k = np.argmax(corr)
        shift = k - (n - 1)

        # valid index range for this shift
        if shift >= 0:
            idx = np.arange(n - shift)
            src_mask = h[idx] > 1e-12
            dst_mask = h[idx] < 1 - 1e-12
            src_idx = idx[src_mask]
            dst_idx = idx[dst_mask]
            if src_idx.size == 0 or dst_idx.size == 0:
                continue
            src_scores = (1 - h[src_idx + shift]) * h[src_idx]
            i_src = src_idx[np.argmax(src_scores)]
            dest_vals = 1 - h[dst_idx + shift]
            j_dst = dst_idx[np.argmin(dest_vals)]
        else:
            s = -shift
            idx = np.arange(s, n)
            src_mask = h[idx] > 1e-12
            dst_mask = h[idx] < 1 - 1e-12
            src_idx = idx[src_mask]
            dst_idx = idx[dst_mask]
            if src_idx.size == 0 or dst_idx.size == 0:
                continue
            src_scores = (1 - h[src_idx - s]) * h[src_idx]
            i_src = src_idx[np.argmax(src_scores)]
            dest_vals = 1 - h[dst_idx - s]
            j_dst = dst_idx[np.argmin(dest_vals)]

        max_eps = min(h[i_src], 1.0 - h[j_dst])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps * step
        if eps <= 0:
            continue

        # apply move
        h[i_src] -= eps
        h[j_dst] += eps
        new_c = compute_c5(h)
        if new_c < cur_c:
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            # revert
            h[i_src] += eps
            h[j_dst] -= eps
    return best_h, best_c


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----- Determine problem size and initial vector -----
    if "initial_h_values" in globals():
        init_h = np.array(initial_h_values, dtype=float)
        n = init_h.size
        init_h = np.clip(init_h, 0.0, 1.0)
        target = n / 2.0
        init_h = enforce_sum_continuous(init_h, target, rng)
    else:
        n = 180
        target = n / 2.0
        init_h = np.full(n, 0.5, dtype=float)
        init_h = enforce_sum_continuous(init_h, target, rng)

    best_h = init_h.copy()
    best_c = compute_c5(best_h)

    # ----- Small initial refinement (continuous) -----
    time_used = time.time() - start_time
    init_refine_time = min(budget_s * 0.08, max(0.0, budget_s - time_used - 0.2 * budget_s))
    if init_refine_time > 1.0:
        h_ref, c_ref = worst_shift_descent(best_h.copy(), rng,
                                           time_limit=init_refine_time,
                                           start_step=0.5,
                                           end_step=1e-4)
        if c_ref < best_c:
            best_h, best_c = h_ref, c_ref

    # ----- Main loop: binary SA + continuous refinement -----
    while True:
        elapsed = time.time() - start_time
        time_left = budget_s - elapsed
        if time_left < 12.0 or best_c < 0.3808:
            break

        # pick a starting binary vector
        if rng.random() < 0.3 and "initial_h_values" in globals():
            bin_start = (best_h > 0.5).astype(int)
            bin_start = enforce_sum_bin(bin_start, n // 2, rng)
        else:
            bin_start = random_binary_feasible(n, rng)

        # allocate time for SA (≈30% of remaining, capped)
        sa_time = min(5.0, time_left * 0.3)
        bin_best, bin_best_c = simulated_annealing_binary(bin_start, rng, sa_time)

        # consider binary solution directly
        if bin_best_c < best_c:
            best_h = bin_best.astype(float)
            best_c = bin_best_c

        # continuous refinement of the binary result
        time_left = budget_s - (time.time() - start_time)
        if time_left < 5.0:
            continue
        refine_time = min(5.0, time_left * 0.3)
        h_cont, c_cont = worst_shift_descent(bin_best.astype(float), rng,
                                             time_limit=refine_time,
                                             start_step=0.3,
                                             end_step=1e-4)
        if c_cont < best_c:
            best_h, best_c = h_cont, c_cont

    # ----- Optional up‑sampling and final refinement -----
    elapsed = time.time() - start_time
    time_left = budget_s - elapsed
    if time_left > 8.0 and n * 2 <= 360:
        h_up = np.repeat(best_h, 2)
        n_up = h_up.size
        target_up = n_up / 2.0
        h_up = enforce_sum_continuous(h_up, target_up, rng)
        # use remaining time for a final descent on up‑sampled grid
        h_up_ref, c_up_ref = worst_shift_descent(h_up, rng,
                                                time_limit=time_left,
                                                start_step=0.3,
                                                end_step=1e-4)
        if c_up_ref < best_c:
            best_h, best_c, n = h_up_ref, c_up_ref, n_up

    # ----- Final clean‑up -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum_continuous(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
