def gradient_descent_step(h, target, rng):
    # compute overlap S
    S = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(S)
    max_k = np.where(S == max_val)[0]  # list of indices
    # compute gradient g: shape n
    n = h.size
    g = np.zeros(n)
    for k in max_k:
        shift = k - (n - 1)  # convert to lag: correlation indices from -(n-1) to (n-1)
        # compute contributions:
        if shift >= 0:
            # For i where i+shift < n: (1 - h[i+shift]) term
            # loop over i = 0..n-1-shift
            i_range = slice(0, n-shift)
            g[i_range] += (1 - h[i_range + shift])
            # For i where i-shift >=0: -h[i-shift] term
            i_range2 = slice(shift, n)
            g[i_range2] -= h[i_range2 - shift]
        else:
            # shift negative: similar
            s = -shift
            # i+shift = i - s
            i_range = slice(s, n)
            g[i_range] += (1 - h[i_range - s])
            # i - shift = i + s
            i_range2 = slice(0, n-s)
            g[i_range2] -= h[i_range2 + s]
    # Now g_i >0 means increasing h_i increases S(max), so to reduce, we want to decrease h_i where g_i>0 and increase where g_i<0.
    pos_indices = np.where(g > 1e-12)[0]
    neg_indices = np.where(g < -1e-12)[0]
    if pos_indices.size == 0 or neg_indices.size == 0:
        # cannot improve
        return h, False
    i_pos = pos_indices[np.argmax(g[pos_indices])]  # most positive gradient
    i_neg = neg_indices[np.argmin(g[neg_indices])]  # most negative gradient (most negative)
    # compute max epsilon: limited by h[i_pos] and (1 - h[i_neg])
    max_eps = min(h[i_pos], 1.0 - h[i_neg])
    if max_eps <= 0:
        return h, False
    # adapt epsilon proportionally to gradient magnitude
    # Choose epsilon = min(max_eps, 0.5*max_eps) maybe.
    eps = max_eps * 0.5
    new_h = h.copy()
    new_h[i_pos] -= eps
    new_h[i_neg] += eps
    # enforce sum: current sum unchanged because we moved, but clipping may cause diff if eps forced to boundaries: we ensured within limits, so sum unchanged.
    return new_h, True

def gradient_step_with_line_search(h, target, rng):
    n = len(h)
    S = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(S)
    max_idx = np.where(S == max_val)[0]  # array
    # compute gradient g
    g = np.zeros(n)
    for idx in max_idx:
        k = idx - (n - 1)  # shift
        if k >= 0:
            # contributions
            g[:n-k] += (1 - h[k:])
            g[k:] -= h[:n-k]
        else:
            s = -k
            g[s:] += (1 - h[:n-s])
            g[:n-s] -= h[s:]
    pos_mask = g > 1e-12
    neg_mask = g < -1e-12
    if not pos_mask.any() or not neg_mask.any():
        return h, False
    # choose i_pos with max g, i_neg with min g
    i_pos = np.argmax(g * pos_mask)  # note zeros for false; but may pick zero if all small; use pos_mask indices
    # Better: use where
    pos_inds = np.where(pos_mask)[0]
    i_pos = pos_inds[np.argmax(g[pos_inds])]
    neg_inds = np.where(neg_mask)[0]
    i_neg = neg_inds[np.argmin(g[neg_inds])]
    max_eps = min(h[i_pos], 1 - h[i_neg])
    if max_eps <= 0:
        return h, False
    # try factors
    factors = [1.0, 0.5, 0.25, 0.125]
    cur_c = max_val
    for f in factors:
        eps = max_eps * f
        h_new = h.copy()
        h_new[i_pos] -= eps
        h_new[i_neg] += eps
        # compute new max after move
        S_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c = np.max(S_new)
        if new_c < cur_c - 1e-12:
            return h_new, True
    # no improvement
    return h, False

def _softmax_obj(x):
    """Smooth log‑sum‑exp surrogate of max_{k} Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)

        if improved:
            h = h_new
            cur_c = compute_c5(h)
            if cur_c < best_c:
                best_c, best_h = cur_c, h.copy()
                fails = 0
            else:
                fails += 1
        else:
            h = perturb_h(h, rng, scale=0.2)
            fails += 1
        if fails > 200:
            break
    return best_h, best_c


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            if abs(add) < 1e-12:
                continue
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----- choose a working dimension -----
    n = 240                     # even, gives finer discretisation than 180
    target = n / 2.0
    dx = 2.0 / n

    # ----- generate a diverse pool of starters -----
    candidates = []

    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, target, rng)
        candidates.append(h0)

    xs = (np.arange(n) + 0.5) * dx
    for f in (1, 2, 3, 4):
        for fun in (np.sin, np.cos):
            h = 0.5 + 0.5 * fun(np.pi * f * xs)
            h = np.clip(h, 0.0, 1.0)
            candidates.append(enforce_sum(h, target, rng))
    # linear ramps
    candidates.append(enforce_sum(xs / 2.0, target, rng))
    candidates.append(enforce_sum(1.0 - xs / 2.0, target, rng))
    # random feasible vectors
    for _ in range(12):
        candidates.append(random_feasible_vector(n, rng))

    # ----- evaluate pool and keep the most promising -----
    vals = [compute_c5(h) for h in candidates]
    order = np.argsort(vals)[:3]                     # top‑3 starters
    best_h = None
    best_c = float("inf")

    # ----- dedicated optimisation for each top starter -----
    remaining_budget = lambda: budget_s - (time.time() - start_time)
    for idx in order:
        if remaining_budget() <= 0:
            break
        h_start = candidates[idx]
        # generous sub‑gradient descent
        time_allow = min(remaining_budget() * 0.4, 250.0)
        h_opt, c_opt = gradient_descent_optimize(
            h_start, target, rng, max_iters=8000, time_limit=time_allow
        )
        if c_opt < best_c:
            best_h, best_c = h_opt.copy(), c_opt

        # quick SLSQP polishing
        for beta in (20, 80, 200, 800):
            if remaining_budget() < 5:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, max_iter=400)
            except Exception:
                continue
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_h, best_c = h_ref.copy(), c_ref

    # ----- up‑sampling to a finer grid if time permits -----
    if remaining_budget() > 60 and n * 2 <= 480:
        h_up = np.repeat(best_h, 2)
        n_up = h_up.size
        target_up = n_up / 2.0
        h_up = enforce_sum(h_up, target_up, rng)

        time_allow = min(remaining_budget() * 0.45, 200.0)
        h_up_opt, c_up_opt = gradient_descent_optimize(
            h_up, target_up, rng, max_iters=8000, time_limit=time_allow
        )
        if c_up_opt < best_c:
            best_h, best_c, n = h_up_opt.copy(), c_up_opt, n_up

        for beta in (40, 160, 640, 2560):
            if remaining_budget() <= 0:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, max_iter=400)
            except Exception:
                continue
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_h, best_c = h_ref.copy(), c_ref

    # ----- final clean‑up -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
