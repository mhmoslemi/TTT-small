def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad = 1 - 2*h
    else:
        grad[:n - k] -= h[k:]
        grad[k:] -= h[:n - k]
    return grad

def guided_hill_climb_cont(h, rng, max_iters=2000, step_scale=0.5):
    cur_h = h.copy()
    best_h = cur_h.copy()
    best_c = compute_c5(cur_h)
    scale = step_scale
    decay_interval = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(cur_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            continue
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * scale
        cur_h[i] -= eps
        cur_h[j] += eps
        # No sum change (sub + add)
        c = compute_c5(cur_h)
        if c < best_c:
            best_c = c
            best_h = cur_h.copy()
        else:
            # revert the step
            cur_h[i] += eps
            cur_h[j] -= eps
        if (it + 1) % decay_interval == 0:
            scale = max(scale * 0.7, 1e-6)
    return best_h, best_c

def enforce_sum(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # Adjust using random index moves
    # while diff not small:
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add == 0:
            continue
        h[i] += add
        diff = target - h.sum()
    return h

import time, numpy as np

def compute_c5(h):
    """Discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def worst_gradient(h):
    """Gradient of the worst‑shift overlap O_k = Σ h_i(1‑h_{i+k})."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        return 1.0 - 2.0 * h
    grad = np.ones_like(h)
    grad[: n - k] -= h[k:]
    grad[k:] -= h[: n - k]
    return grad


def enforce_sum(h, rng):
    """Adjust vector so that Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        if add == 0:
            continue
        h[i] += add
        diff = target - h.sum()
    return h


def binary_from_continuous(h):
    """Convert a continuous profile to a binary one with exactly n/2 ones."""
    n = h.size
    target = n // 2
    idx = np.argsort(-h)  # descending
    bin_h = np.zeros(n, dtype=float)
    bin_h[idx[:target]] = 1.0
    return bin_h


def random_binary(n, rng):
    """Random feasible 0‑1 vector with n/2 ones."""
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h


def binary_hill_climb(h, rng, max_iters=3000):
    """Swap‑based hill‑climb on binary vectors."""
    cur = h.copy()
    best = cur.copy()
    best_c = compute_c5(cur)
    n = cur.size
    for it in range(max_iters):
        grad = worst_gradient(cur)
        pos_one = np.where((cur == 1) & (grad > 0))[0]
        neg_zero = np.where((cur == 0) & (grad < 0))[0]
        if pos_one.size and neg_zero.size:
            i = int(rng.choice(pos_one))
            j = int(rng.choice(neg_zero))
        else:
            ones = np.where(cur == 1)[0]
            zeros = np.where(cur == 0)[0]
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
        cur[i], cur[j] = cur[j], cur[i]
        c = compute_c5(cur)
        if c < best_c - 1e-12:
            best_c, best = c, cur.copy()
        else:
            cur[i], cur[j] = cur[j], cur[i]  # revert
    return best, best_c


def guided_hill_climb_cont(h, rng, max_iters=2000, step_scale=0.5):
    """Continuous gradient‑directed hill‑climb."""
    cur = h.copy()
    best = cur.copy()
    best_c = compute_c5(cur)
    scale = step_scale
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(cur)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            continue
        max_eps = min(cur[i], 1.0 - cur[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * scale
        cur[i] -= eps
        cur[j] += eps
        c = compute_c5(cur)
        if c < best_c:
            best_c, best = c, cur.copy()
        else:
            cur[i] += eps
            cur[j] -= eps
        if (it + 1) % decay == 0:
            scale = max(scale * 0.7, 1e-6)
    return best, best_c
