def gradient_softmax(h, beta):
    c = np.correlate(h, 1 - h, mode='full')
    w = np.exp(beta * c - np.max(beta*c))  # stabilize
    w /= w.sum()
    n = h.size
    g = np.zeros(n)
    for d in range(-n+1, n):
        wd = w[d + n -1]
        if wd == 0:
            continue
        if d >= 0:
            # slice overlapping region length n-d
            # add (1 - h[i+d]) part:
            g[:n-d] += wd * (1 - h[d:])
            # subtract h[i-d] part: i-d = i - d? Actually term - h_{i-d}
            # When d>0, i-d is index i-d >=0 for i in d..n-1 => slice
            g[d:] -= wd * h[:n-d]
        else:
            # d negative, let t = -d >0
            t = -d
            # i from t..n-1
            # (1 - h[i+d]) = (1 - h[i - t])
            g[t:] += wd * (1 - h[:n-t])   # since i+d = i - t
            # - h[i-d] = -h[i + t]
            g[:n-t] -= wd * h[t:]
    return g

    c = np.correlate(h, 1 - h, mode='full')
    bc = beta * c
    bmax = np.max(bc)
    exp_bc = np.exp(bc - bmax)  # shape (2n-1)

def project_onto_box_and_sum(x, target):
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # simple water-filling using iterative scaling
    # Use while loop to handle case capacity insufficient (should not happen because we start within bounds).
    # We'll adjust step by step maybe 2 iterations.
    for _ in range(10):
        if diff > 0:
            capacity = 1.0 - x
            total = capacity.sum()
            if total < 1e-12:
                break
            delta = diff * capacity / total
            x += delta
        else:
            capacity = x
            total = capacity.sum()
            if total < 1e-12:
                break
            delta = diff * capacity / total
            x += delta  # since diff negative, subtract
        x = np.clip(x, 0.0, 1.0)
        diff = target - x.sum()
        if abs(diff) < 1e-12:
            break
    # final fallback using random adjustments if not fixed.
    if abs(diff) > 1e-8:
        rng = np.random.default_rng()
        for _ in range(2000):
            i = rng.integers(0, len(x))
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
            if abs(diff) < 1e-12:
                break
    return x

def pgd_softmax(h0, beta, max_iter=200, step_size=1.0):
    h = h0.copy()
    target = h.size / 2.0
    for it in range(max_iter):
        grad = softmax_grad(h, beta)
        # compute step size adaptively via backtracking
        step = step_size
        f_curr = softmax_obj_val(h, beta)
        # backtrack
        for _ in range(20):
            h_new = project_onto_box_and_sum(h - step * grad, target)
            f_new = softmax_obj_val(h_new, beta)
            if f_new <= f_curr - 1e-4 * step * np.dot(grad, grad):
                h = h_new
                break
            step *= 0.5
        else:
            # no improvement, break
            break
    return h

def project_onto_box_and_sum(x, target):
    """Project x onto [0,1]^n while preserving sum = target (assumes feasible)."""
    # clip to bounds
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # iterative water-filling
    for _ in range(10):
        if diff > 0:
            capacity = 1.0 - x
            total = capacity.sum()
            if total < 1e-12:
                break
            delta = diff * capacity / total
            x += delta
        else:
            capacity = x
            total = capacity.sum()
            if total < 1e-12:
                break
            delta = diff * capacity / total
            x += delta
        x = np.clip(x, 0.0, 1.0)
        diff = target - x.sum()
        if abs(diff) < 1e-12:
            break
    # fallback random adjustments if still not exact
    if abs(diff) > 1e-8:
        rng = np.random.default_rng()
        for _ in range(2000):
            i = rng.integers(0, len(x))

            x[i] += add
            diff = target - x.sum()
            if abs(diff) < 1e-12:
                break
    return x

def softmax_obj_val(h, beta):
    c = np.correlate(h, 1.0 - h, mode='full')
    bc = beta * c
    bmax = np.max(bc)
    return (bmax + np.log(np.sum(np.exp(bc - bmax)))) / beta

### softmax_grad

def softmax_grad(h, beta):
    n = h.size
    c = np.correlate(h, 1.0 - h, mode='full')
    bc = beta * c
    bmax = np.max(bc)
    exp_bc = np.exp(bc - bmax)
    w = exp_bc / np.sum(exp_bc)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    # loop over shifts
    for d in range(-n+1, n):
        wd = w[d + n - 1]
        if wd == 0.0:
            continue
        if d >= 0:
            # (1 - h_{i+d}) term
            grad[:n-d] += wd * one_minus_h[d:]
            # - h_{i-d}
            grad[d:] -= wd * h[:n-d]
        else:
            t = -d
            grad[t:] += wd * one_minus_h[:n-t]
            grad[:n-t] -= wd * h[t:]
    return grad

def pgd_softmax(h0, beta, max_iter=200, step_size=1.0):
    h = h0.copy()
    target = h.size / 2.0
    for it in range(max_iter):
        grad = softmax_grad(h, beta)
        f_curr = softmax_obj_val(h, beta)
        step = step_size
        # backtracking loop
        while True:
            h_new = project_onto_box_and_sum(h - step * grad, target)
            f_new = softmax_obj_val(h_new, beta)
            if f_new <= f_curr - 1e-5 * step * np.dot(grad, grad):
                h = h_new
                break
            step *= 0.5
            if step < 1e-8:
                # cannot improve
                break
    return h

def compute_c5(h):
    """Exact discrete C5 for step function h on [0,2] (uniform spacing)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def project_onto_box_and_sum(x, target):

        return x

            x += delta

        if accept:
            cur_h, cur_c = cand_h, cand_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
        it += 1
    return best_h, best_c


# ------------------------------------------------------------
# Smooth max (soft‑max) surrogate and its gradient
# ------------------------------------------------------------
def softmax_obj_val(h, beta):
    c = np.correlate(h, 1.0 - h, mode="full")
    bc = beta * c
    bmax = np.max(bc)
    return (bmax + np.log(np.sum(np.exp(bc - bmax)))) / beta


def softmax_grad(h, beta):
    n = h.size
    c = np.correlate(h, 1.0 - h, mode="full")
    bc = beta * c
    bmax = np.max(bc)
    exp_bc = np.exp(bc - bmax)
    w = exp_bc / np.sum(exp_bc)          # normalized weights over shifts
    grad = np.zeros(n)
    one_minus = 1.0 - h
    # contribution of each shift d
    for d in range(-n + 1, n):
        wd = w[d + n - 1]
        if wd == 0.0:
            continue
        if d >= 0:
            grad[: n - d] += wd * one_minus[d:]      # (1 - h_{i+d})
            grad[d:]       -= wd * h[: n - d]        # - h_{i-d}
        else:
            t = -d
            grad[t:]       += wd * one_minus[: n - t]   # (1 - h_{i-d})
            grad[: n - t]  -= wd * h[t:]                # - h_{i+d}
    return grad


def pgd_softmax(h0, beta, max_iter=200, step_size=0.5):
    """Projected gradient descent on the soft‑max surrogate."""
    h = h0.copy()
    target = h.size / 2.0
    for _ in range(max_iter):
        grad = softmax_grad(h, beta)
        f_curr = softmax_obj_val(h, beta)
        step = step_size
        # Armijo back‑tracking
        while True:
            h_new = project_onto_box_and_sum(h - step * grad, target)
            f_new = softmax_obj_val(h_new, beta)
            if f_new <= f_curr - 1e-5 * step * np.dot(grad, grad):
                h = h_new
                break
            step *= 0.5
            if step < 1e-8:
                break
    return h


# ------------------------------------------------------------
# Main driver
# ------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    def update_best(h, c, n):
        nonlocal best_h, best_c, best_n
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

    # ----------------------------------------------------------------
    # Determine candidate discretisations
    # ----------------------------------------------------------------
    candidate_ns = [72, 96, 108, 120, 144, 180, 240]
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        if parent.size not in candidate_ns:
            candidate_ns.insert(0, parent.size)
    else:
        parent = None

    # ----------------------------------------------------------------
    # Process each n (small to large)
    # ----------------------------------------------------------------
    for n in candidate_ns:
        if time.time() - start_time > budget_s * 0.92:
            break
        target = n / 2.0
        starts = []

        # 1) parent (interpolated if needed)
        if parent is not None:
            if parent.size == n:
                starts.append(parent)
            else:
                xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
                xt = np.linspace(0.0, 2.0, n, endpoint=False)
                starts.append(np.interp(xt, xs, parent))
        # 2) random continuous feasible
        starts.append(random_feasible_vector(n, rng))
        # 3) random binary feasible
        h_bin = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h_bin[idx] = 1.0
        starts.append(h_bin)
        # 4) a few extra random starts
        for _ in range(2):
            starts.append(random_feasible_vector(n, rng))

        for h0 in starts:
            if time.time() - start_time > budget_s * 0.95:
                break
            h = project_onto_box_and_sum(np.clip(h0, 0.0, 1.0), target)

            # ---- smooth‑max descent with increasing β ----
            for beta in (1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1000.0):
                if time.time() - start_time > budget_s * 0.96:
                    break
                max_iter = 80 if beta < 20 else 180 if beta < 200 else 260
                h = pgd_softmax(h, beta, max_iter=max_iter, step_size=0.5)

            # ---- exact C5 evaluation and bookkeeping ----
            c = compute_c5(h)
            update_best(h, c, n)

            # ---- deterministic hill‑climb ----
            if time.time() - start_time < budget_s * 0.97:
                h_hc, c_hc = hill_climb(h, rng, max_iters=1500, start_step=0.4)
                update_best(h_hc, c_hc, n)
                h = h_hc

            # ---- simulated annealing refinement ----
            if time.time() - start_time < budget_s * 0.99:
                deadline = start_time + budget_s * 0.99
                h_sa, c_sa = simulated_annealing(h, rng, deadline)
                update_best(h_sa, c_sa, n)

    # ----------------------------------------------------------------
    # Optional up‑sampling refinement (double resolution)
    # ----------------------------------------------------------------
    if best_h is not None and time.time() - start_time < budget_s * 0.85:
        n_up = best_n * 2
        if n_up <= 300:
            xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
            xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
            h_up = np.interp(xt, xs, best_h)
            h_up = project_onto_box_and_sum(h_up, n_up / 2.0)

            for beta in (1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1000.0):
                if time.time() - start_time > budget_s * 0.98:
                    break
                h_up = pgd_softmax(h_up, beta, max_iter=250, step_size=0.5)

            c_up = compute_c5(h_up)
            update_best(h_up, c_up, n_up)

            # final local hill‑climb on the up‑sampled solution
            if time.time() - start_time < budget_s * 0.999:
                h_hc, c_hc = hill_climb(h_up, rng, max_iters=1800, start_step=0.3)
                update_best(h_hc, c_hc, n_up)

    # ----------------------------------------------------------------
    # Safety fallback
    # ----------------------------------------------------------------
    if best_h is None:
        best_n = 120
        best_h = np.full(best_n, 0.5)

    best_h = project_onto_box_and_sum(best_h, best_n / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
