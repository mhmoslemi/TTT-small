def guided_perturb_top_bottom(h, grad, rng, scale):
    # Determine threshold for top and bottom
    n = len(h)
    k = max(1, n//20)  # top 5%
    # Get indices sorted by grad descending
    order = np.argsort(grad)
    donors = order[-k:]  # largest
    receivers = order[:k]  # smallest
    i = rng.choice(donors)
    j = rng.choice(receivers)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh
```

    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(corr)
    # indices of shifts attaining max (within tolerance)
    idxs = np.where(corr >= max_val - eps)[0]
    grad = np.zeros_like(h)
    for idx in idxs:

        shift = abs(k)
        if shift == 0:
            grad += 1.0 - 2.0 * h
        else:
            # compute contributions

def slsqp_refine(x0, beta, target, rng, max_iter=200):
    n = x0.size
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full')
        maxc = np.max(c)
        lse = maxc + np.log(np.sum(np.exp(beta * (c - maxc))))
        return lse / beta
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, x0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol':1e-12, 'disp':False})
    x = np.clip(res.x, 0.0, 1.0)
    # repair sum
    diff = target - x.sum()
    # same enforcement as before
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = target - x.sum()
    return x
```

import time
import math
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# basic utilities
# --------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] (vector h)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator, max_iter: int = 2000) -> np.ndarray:
    """Adjust h so that sum(h) == len(h)/2 while keeping 0≤h≤1."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
        it += 1
    return h


# --------------------------------------------------------------
# gradient of the worst shift (subgradient)
# --------------------------------------------------------------
def worst_gradient_sum(h: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    """Subgradient of max_k Σ h_i (1‑h_{i+k}) (sum over all worst shifts)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    idxs = np.where(corr >= max_val - eps)[0]
    grad = np.zeros_like(h)
    for idx in idxs:
        k = idx - (n - 1)               # shift (can be negative)
        if k == 0:
            grad += 1.0 - 2.0 * h
        elif k > 0:                     # positive shift
            shift = k
            grad[: n - shift] += 1.0 - h[shift:]
            grad[shift:] -= h[: n - shift]
        else:                           # negative shift
            shift = -k
            grad[shift:] += 1.0 - h[: n - shift]
            grad[: n - shift] -= h[shift:]
    if len(idxs) > 1:
        grad /= len(idxs)
    return grad


# --------------------------------------------------------------
# guided perturbation (mass transfer)
# --------------------------------------------------------------
def guided_perturb_top_bottom(h: np.ndarray,
                              grad: np.ndarray,
                              rng: np.random.Generator,
                              scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a high‑gradient index to a low‑gradient one."""
    n = h.size
    k = max(1, n // 20)                     # 5 % of indices
    order = np.argsort(grad)
    donors = order[-k:]                     # largest gradients
    receivers = order[:k]                   # smallest gradients
    i = int(rng.choice(donors))
    j = int(rng.choice(receivers))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

# local search using guided perturbations
# --------------------------------------------------------------
def local_search_with_guided(h_start: np.ndarray,
                             rng: np.random.Generator,
                             deadline: float) -> tuple:
    """Hill‑climb with guided transfers until deadline."""
    h = h_start.copy()
    curr_c = compute_c5(h)
    best_h = h.copy()
    best_c = curr_c
    scale = 0.5
    while time.time() < deadline:
        grad = worst_gradient_sum(h)
        nh = guided_perturb_top_bottom(h, grad, rng, scale)
        nc = compute_c5(nh)
        if nc < best_c:
            best_c, best_h = nc, nh.copy()
        if nc < curr_c:
            h, curr_c = nh, nc
            scale = min(scale * 1.05, 0.8)
        else:
            scale = max(scale * 0.95, 0.01)
        # occasional random scale reduction
        if rng.random() < 0.001:
            scale = max(scale * 0.5, 0.01)
    return best_h, best_c


# --------------------------------------------------------------
# SLSQP refinement via smooth max (log‑sum‑exp)
# --------------------------------------------------------------
def _softmax_obj(x: np.ndarray, beta: float) -> float:
    c = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(c)
    lse = m + np.log(np.sum(np.exp(beta * (c - m))))
    return lse / beta


def slsqp_refine(x0: np.ndarray,
                 beta: float,
                 target: float,
                 rng: np.random.Generator,
                 max_iter: int = 300) -> np.ndarray:
    n = x0.size

    def obj(x):
        return _softmax_obj(x, beta)

    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, x0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    x = enforce_sum(x, rng)
    return x


def slsqp_refine_full(h_start: np.ndarray,
                      rng: np.random.Generator,
                      deadline: float) -> tuple:
    """Run SLSQP with a short β schedule until deadline."""
    n = h_start.size
    target = n / 2.0
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    betas = (10.0, 100.0, 500.0, 2000.0)
    for beta in betas:
        if time.time() >= deadline:
            break
        h = slsqp_refine(h, beta, target, rng, max_iter=250)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
    return best_h, best_c

        if nc < curr_c:
