def project_to_simplex_box(x, target, max_iter=50):
    lo = -np.max(x)  # sum = 0
    hi = 1 - np.min(x)  # sum = n
    for _ in range(max_iter):
        mid = (lo + hi) / 2
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x + (lo+hi)/2, 0.0, 1.0)
    return y

def get_grad_cache(n):
    cache = _gradient_cache.get(n)
    if cache is None:
        I = np.arange(n)[:,None]
        J = np.arange(n)[None,:]
        shift = J - I
        w_idx = shift + (n - 1)
        mask_A = (J >= I).astype(np.float64)
        mask_B = (J <= I).astype(np.float64)
        cache = (w_idx, mask_A, mask_B)
        _gradient_cache[n] = cache

def smooth_max_and_grad(h, beta):
    c = np.correlate(h, 1 - h, mode='full')
    bc = beta * c
    max_bc = np.max(bc)
    exp_bc = np.exp(bc - max_bc)
    sum_exp = np.sum(exp_bc)
    smooth = (max_bc + np.log(sum_exp)) / beta
    w = exp_bc / sum_exp
    n = h.size
    w_idx, mask_A, mask_B = get_grad_cache(n)
    w_mat = w[w_idx]  # shape (n,n)
    gA = (w_mat * mask_A).dot(1 - h)
    gB = (w_mat * mask_B).dot(h)
    grad = gA - gB
    return smooth, grad

def gradient_descent(h0, target, betas, steps_list, init_step, rng, start_time, time_budget):
    h = h0.copy()
    n = h.size

                return h

    bc = CURRENT_BETA * c
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(_softmax_obj, h0, method="SLSQP", bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)

    diff = CURRENT_TARGET - x.sum()

        # adjust by simple projection
        x = project_to_simplex_box(x, CURRENT_TARGET)

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global cache for gradient computation
# ----------------------------------------------------------------------
_gradient_cache = {}

def _grad_cache(n):
    """Return index matrix and masks for given dimension."""
    if n not in _gradient_cache:
        I = np.arange(n)[:, None]
        J = np.arange(n)[None, :]
        shift = J - I
        w_idx = shift + (n - 1)
        mask_A = (J >= I).astype(np.float64)
        mask_B = (J <= I).astype(np.float64)
        _gradient_cache[n] = (w_idx, mask_A, mask_B)
    return _gradient_cache[n]

# ----------------------------------------------------------------------
# Projection onto {0≤x_i≤1, Σx_i = target}
# ----------------------------------------------------------------------
def _project_box_simplex(x, target, iters=50):
    lo = -np.max(x)          # sum=0 after clipping
    hi = 1.0 - np.min(x)     # sum=n after clipping
    for _ in range(iters):
        mid = (lo + hi) * 0.5
        y = np.clip(x + mid, 0.0, 1.0)
        if y.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)

# ----------------------------------------------------------------------
# Core problem utilities
# ----------------------------------------------------------------------
def compute_c5(h):
    """Exact discrete C₅ for a step vector h."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def random_feasible_vector(n, target, rng):
    """Uniform random point, projected onto the feasible set."""
    x = rng.random(n)
    return _project_box_simplex(x, target)

def perturb_h(h, target, rng, scale=0.5):
    """Transfer a small amount between two coordinates (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if j >= i:
        j += 1
    max_pos = min(h[i], 1.0 - h[j])
    max_neg = min(h[j], 1.0 - h[i])
    if max_pos + max_neg <= 0:
        return h.copy()
    eps = rng.uniform(-max_neg, max_pos) * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    nh = np.clip(nh, 0.0, 1.0)
    # tiny sum correction if needed
    diff = target - nh.sum()
    if abs(diff) > 1e-12:
        nh = _project_box_simplex(nh, target)
    return nh

def hill_climb(h_start, rng, max_iters=2500, step_start=0.5):
    """Simple hill‑climb: accept only improving moves."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step_start
    n = h.size
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        max_pos = min(h[i], 1.0 - h[j])
        max_neg = min(h[j], 1.0 - h[i])
        if max_pos + max_neg == 0:
            continue
        eps = rng.uniform(-max_neg, max_pos) * step
        nh = h.copy()
        nh[i] -= eps
        nh[j] += eps
        nh = np.clip(nh, 0.0, 1.0)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-6)
    return best_h, best_c

# Smooth surrogate and its gradient
# ----------------------------------------------------------------------
def smooth_max_and_grad(h, beta):
    """Log‑sum‑exp of correlations and its gradient w.r.t. h."""
    c = np.correlate(h, 1 - h, mode="full")
    bc = beta * c
    m = np.max(bc)
    exp_bc = np.exp(bc - m)
    sum_exp = np.sum(exp_bc)
    smooth = (m + np.log(sum_exp)) / beta
    w = exp_bc / sum_exp                       # weights, sum to 1
    n = h.size
    w_idx, mask_A, mask_B = _grad_cache(n)
    w_mat = w[w_idx]                           # (n,n) matrix of weights
    gA = (w_mat * mask_A).dot(1 - h)
    gB = (w_mat * mask_B).dot(h)
    grad = gA - gB
    return smooth, grad

def gradient_descent(h0, target, betas, steps, init_step, rng,
                     start_time, time_budget):
    """Continuation on β with simple projected gradient steps."""
    h = h0.copy()
    n = h.size
    for beta, nsteps in zip(betas, steps):
        for it in range(nsteps):
            if time.time() - start_time > time_budget:
                return h
            _, g = smooth_max_and_grad(h, beta)
            step = init_step * (1.0 - it / nsteps)
            h = h - step * g / n
            h = _project_box_simplex(h, target)
    return h

# SLSQP refinement on the smooth surrogate
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None

def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def _softmax_obj(x):
    c = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * c
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    """Run a few SLSQP iterations on the log‑sum‑exp surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(_softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # final exact‑sum repair via projection
    if abs(x.sum() - CURRENT_TARGET) > 1e-12:
        x = _project_box_simplex(x, CURRENT_TARGET)
    return x
