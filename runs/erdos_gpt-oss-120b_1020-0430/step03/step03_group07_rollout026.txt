import numpy as np
from scipy.optimize import minimize

def make_full_h(half):
    return np.concatenate([half, 1 - half[::-1]])

def correlate_full(h):
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv

def compute_c5_raw(h):
    conv = correlate_full(h)
    return conv.max()

    conv = correlate_full(h)

def make_full_h(half):
    """Construct full vector of length 2*len(half) with complement symmetry."""
    return np.concatenate([half, 1 - half[::-1]])


def correlate_full(h):
    """Full linear correlation of h with (1‑h) using FFT (zero‑padding)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5_raw(h):
    """Raw maximum overlap (without the dx scaling)."""
    return correlate_full(h).max()


def smooth_obj_and_grad(half, beta):
    """
    Smoothed objective (log‑sum‑exp of the full correlation) and its gradient.
    Returns (obj, grad_half) where grad_half has the same shape as half.
    """
    h = make_full_h(half)
    conv = correlate_full(h)          # length 2n‑1
    maxc = conv.max()

    # stable softmax weights
    exp_shift = np.exp(beta * (conv - maxc))
    sum_exp = exp_shift.sum()
    w = exp_shift / sum_exp

    n = h.size
    offset = w.size - 1               # = 2n‑2

    conv1_full = np.convolve(w[::-1], 1 - h, mode='full')
    conv1 = conv1_full[offset : offset + n]
