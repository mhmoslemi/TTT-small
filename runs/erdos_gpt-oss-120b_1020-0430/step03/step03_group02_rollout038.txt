def simulated_annealing(h, rng, time_limit):
    start = time.time()
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    # temperature schedule parameters
    T0 = 0.01 * cur_c # maybe small
    T_end = 1e-6
    # total iterations limited by time
    while time.time() - start < time_limit:
        # progress fraction
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = T0 * (1 - frac) + T_end * frac  # linear decay or exponential
        # propose move
        n = cur_h.size
        # pick two indices i!=j at random
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # choose epsilon from normal distribution scaled by T or constant
        # epsilon max limited by maintain bounds
        max_eps = min(cur_h[i], 1 - cur_h[j]) if cur_h[i] > 0 else 0
        max_eps = max_eps if max_eps > 0 else min(1 - cur_h[i], cur_h[j]) # we could also move opposite direction; consider both directions.
        # Actually we can allow epsilon both positive (move from i to j) and negative (move from j to i)
        # Choose direction randomly: sign = +/-1
        if rng.random() < 0.5:
            # move from i to j
            max_move = min(cur_h[i], 1 - cur_h[j])
            if max_move <= 0:
                continue
            eps = rng.random() * max_move * 0.5  # scale
            new_h = cur_h.copy()
            new_h[i] -= eps
            new_h[j] += eps
        else:
            # move from j to i
            max_move = min(cur_h[j], 1 - cur_h[i])
            if max_move <= 0:
                continue
            eps = rng.random() * max_move * 0.5
            new_h = cur_h.copy()
            new_h[j] -= eps
            new_h[i] += eps
        # maybe also random small uniform perturbation to many indices.
        new_c = compute_c5(new_h)
        # acceptance
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / (T + 1e-12)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
    return best_h, best_c

def simulated_annealing(h, rng, time_limit):
    start = time.time()
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    n = h.size
    # set parameters
    max_eps_factor = 0.5
    # schedule - linear decay of temperature from initial T0 to T_end
    T0 = cur_c * 0.02  # maybe small
    T_end = cur_c * 1e-6
    # could also use exponential: T = T0 * (T_end/T0) ** (elapsed/limit)
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = T0 * (1 - frac) + T_end * frac
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        # decide direction; compute max move
        if rng.random() < 0.5:
            max_move = min(cur_h[i], 1 - cur_h[j])
            if max_move <= 0:
                continue
            eps = rng.random() * max_move * max_eps_factor
            new_h = cur_h.copy()
            new_h[i] -= eps
            new_h[j] += eps
        else:
            max_move = min(cur_h[j], 1 - cur_h[i])
            if max_move <= 0:
                continue
            eps = rng.random() * max_move * max_eps_factor
            new_h = cur_h.copy()
            new_h[j] -= eps
            new_h[i] += eps
        new_c = compute_c5(new_h)
        # acceptance
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / (T + 1e-12)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
    return best_h, best_c

def optimal_perturb(h, grad, rng, max_iters=None):
    n = h.size
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    # direction: move from i to j (i has larger gradient, decreases)
    max_move = min(h[i], 1 - h[j])
    if max_move <= 0:
        return h
    # compute O (overlap) values:
    O = np.correlate(h, 1 - h, mode='full')
    # compute a vector for this pair:
    shifts = np.arange(-(n-1), n)
    a = np.zeros_like(O, dtype=float)
    # contributions from i: -h[i+shift] - h[i-shift]
    idx_plus = i + shifts
    mask = (idx_plus >= 0) & (idx_plus < n)
    a[mask] -= h[idx_plus[mask]]
    idx_minus = i - shifts
    mask = (idx_minus >= 0) & (idx_minus < n)
    a[mask] -= h[idx_minus[mask]]
    # contributions from j: +h[j+shift] + h[j-shift]
    idx_plus = j + shifts
    mask = (idx_plus >= 0) & (idx_plus < n)
    a[mask] += h[idx_plus[mask]]
    idx_minus = j - shifts
    mask = (idx_minus >= 0) & (idx_minus < n)
    a[mask] += h[idx_minus[mask]]
    # define function max_val(eps):
    #   return np.max(O + eps*a)
    # ternary search in [0, max_move] for minimal.
    lo, hi = 0.0, max_move
    for _ in range(30):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(O + m1 * a)
        f2 = np.max(O + m2 * a)
        if f1 > f2:
            lo = m1
        else:
            hi = m2
    eps_opt = (lo + hi) / 2.0
    # generate new h
    new_h = h.copy()
    new_h[i] -= eps_opt
    new_h[j] += eps_opt
    return new_h
```

import time
import numpy as np

# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add == 0.0:
            continue
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random feasible h (0≤h≤1, Σh=n/2)."""
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, rng)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‐shift overlap O_k = Σ h_i(1‑h_{i+k})."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]      # subtract h_{i+k}
        grad[k:]       -= h[: n - k]  # subtract h_{i‑k}
    return grad


def optimal_perturb_pair(h: np.ndarray, i: int, j: int) -> np.ndarray:
    """
    Transfer an optimal ε from index i to index j (h[i] decreases,
    h[j] increases) to minimise max_k O_k.
    """
    n = h.size
    max_move = min(h[i], 1.0 - h[j])
    if max_move <= 0.0:
        return h
    # baseline overlaps (unscaled)
    O = np.correlate(h, 1.0 - h, mode="full")
    shifts = np.arange(-(n - 1), n)  # length 2*n-1
    a = np.zeros_like(O, dtype=float)

    # contributions from i (positive)
    idx = i + shifts
    mask = (idx >= 0) & (idx < n)
    a[mask] += h[idx[mask]]
    idx = i - shifts
    mask = (idx >= 0) & (idx < n)
    a[mask] += h[idx[mask]]

    # contributions from j (negative)
    idx = j + shifts
    mask = (idx >= 0) & (idx < n)
    a[mask] -= h[idx[mask]]
    idx = j - shifts
    mask = (idx >= 0) & (idx < n)
    a[mask] -= h[idx[mask]]

    if np.allclose(a, 0.0):
        # no effect on any O_k; take a small move
        eps_opt = max_move
    else:
        # ternary search on [0, max_move] for min of max(O + eps*a)
        lo, hi = 0.0, max_move
        for _ in range(30):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            f1 = np.max(O + m1 * a)
            f2 = np.max(O + m2 * a)
            if f1 > f2:
                lo = m1
            else:
                hi = m2
        eps_opt = (lo + hi) / 2.0

    nh = h.copy()
    nh[i] -= eps_opt
    nh[j] += eps_opt
    return nh


def optimal_hill_climb(h0: np.ndarray,
                       rng: np.random.Generator,
                       max_iters: int = 2500) -> tuple[np.ndarray, float]:
    """Fine‑grained hill‑climb using optimal line‑search transfers."""
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    n = h.size
    no_improve = 0
    patience = max_iters // 10

    for it in range(max_iters):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            break
        nh = optimal_perturb_pair(h, i, j)
        nc = compute_c5(nh)
        if nc < cur_c - 1e-12:
            h, cur_c = nh, nc
            no_improve = 0
            if cur_c < best_c:
                best_h, best_c = h.copy(), cur_c
        else:
            no_improve += 1
            if no_improve >= patience:
                # try a random pair to escape possible stall
                i = rng.integers(0, n)
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                nh = optimal_perturb_pair(h, i, j)
                nc = compute_c5(nh)
                if nc < cur_c:
                    h, cur_c = nh, nc
                no_improve = 0
    return best_h, best_c


def simulated_annealing(h0: np.ndarray,
                        rng: np.random.Generator,
                        time_limit: float) -> tuple[np.ndarray, float]:
    """Very cheap SA using pairwise ε‑moves."""
    start = time.time()
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    n = cur_h.size
    T0 = cur_c * 0.02
    T_end = cur_c * 1e-6

    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = T0 * (1.0 - frac) + T_end * frac

        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        if rng.random() < 0.5:
            max_move = min(cur_h[i], 1.0 - cur_h[j])
            if max_move <= 0.0:
                continue
            eps = rng.random() * max_move * 0.5
            nh = cur_h.copy()
            nh[i] -= eps
            nh[j] += eps
        else:
            max_move = min(cur_h[j], 1.0 - cur_h[i])
            if max_move <= 0.0:
                continue
            eps = rng.random() * max_move * 0.5
            nh = cur_h.copy()
            nh[j] -= eps
            nh[i] += eps

        nc = compute_c5(nh)
        if nc < cur_c or rng.random() < np.exp(-(nc - cur_c) / (T + 1e-12)):
            cur_h, cur_c = nh, nc
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
    return best_h, best_c

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
