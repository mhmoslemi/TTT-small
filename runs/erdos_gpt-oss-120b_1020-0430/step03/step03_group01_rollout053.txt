def compute_c5_binary(h):
    n = h.size
    # compute autocorrelation full
    H = np.fft.rfft(h, 2*n)
    corr = np.fft.irfft(H * np.conj(H), n=2*n)[:2*n-1]
    # compute overlap via prefix sums using np.cumsum
    pref = np.concatenate(([0.0], np.cumsum(h)))
    # k from -(n-1) to (n-1)
    # For positive k (>0): overlapping i indices from 0 to n-1-k => sum = pref[n - k] - pref[0]
    # For negative k (<0): overlapping i indices from -k to n-1 => sum = pref[n] - pref[-k]
    # for k=0: overlap = pref[n] = n/2
    # compute vector overlap similarly.
    # Compute vector of length 2n-1:
    ks = np.arange(-(n-1), n)  # inclusive
    overlap = np.empty(2*n-1)
    # positive ks
    pos = ks > 0
    kpos = ks[pos]
    overlap[pos] = pref[n - kpos]  # sum upto n-k
    # negative ks
    neg = ks < 0
    kneg = -ks[neg]
    # overlapping sum of h_i where i >= kneg (since shift negative)
    overlap[neg] = pref[n] - pref[kneg]
    # zero
    overlap[ks == 0] = pref[n]
    diff = overlap - corr
    max_diff = np.max(diff)
    return max_diff * (2.0 / n)

But careful: For k positive, overlapping i up to n-1-k, sum = pref[n - k] - pref[0] = pref[n - k]; correct.
For k negative, shift negative means moving (1-h) left relative to h. Overlap indices i such that i+k >=0 => i >= -k. So sum over i from -k to n-1 => total sum - sum_{i=0}^{-k-1} = pref[n] - pref[-k].

```
    n = h_init.size
    h = h_init.copy()
    best_h = h.copy()
    best_c = compute_c5_binary(h)
    cur_c = best_c
    T0 = 0.1  # maybe set based on initial difference
    # Determine schedule: T = T0 * (1 - (elapsed/ time_budget)) maybe.
    while time.time() - start < time_budget:
        # propose swap
        ones_idx = np.where(h==1)[0]
        zeros_idx = np.where(h==0)[0]
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        # swap
        h[i], h[j] = 0, 1
        new_c = compute_c5_binary(h)
        delta = new_c - cur_c
        # accept if improvement or with probability
        elapsed = time.time() - start
        T = max(1e-6, T0 * (1 - elapsed / time_budget))
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # reject, swap back
            h[i], h[j] = 1, 0
    return best_h, best_c
```

```
def fast_c5(h):
    n = h.size
    H = np.fft.rfft(h, n=2*n)
    corr = np.fft.irfft(H * np.conj(H), n=2*n)[:2*n-1]
    # prefix sums of h as float64
    pref = np.concatenate(([0.0], np.cumsum(h)))
    # ks from -(n-1) to (n-1)
    ks = np.arange(-(n-1), n)
    # initialize overlap array
    overlap = np.empty(2*n-1, dtype=np.float64)
    # positive ks
    pos = ks > 0
    kp = ks[pos]
    overlap[pos] = pref[n - kp]
    # negative
    neg = ks < 0
    kn = -ks[neg]
    overlap[neg] = pref[n] - pref[kn]
    # zero
    overlap[ks == 0] = pref[n]
    diff = overlap - corr
    max_diff = np.max(diff)
    return max_diff * (2.0 / n)
```

def simulated_annealing(h_init, rng, time_limit):
    n = h_init.size
    # compute ones and zeros indices once at start
    h = h_init.copy()
    cur_c = fast_c5(h)
    best_h = h.copy()
    best_c = cur_c
    start = time.time()
    # schedule: linear cooling from T0 to T_end
    T0 = 0.5  # initial temperature
    T_end = 1e-4
    while time.time() - start < time_limit:
        # proportion of elapsed
        elapsed = time.time() - start
        t_frac = elapsed / time_limit
        T = T0 * (1 - t_frac) + T_end * t_frac
        # pick random swap
        # choose i from ones, j from zeros
        ones_idx = np.where(h == 1.0)[0]
        zeros_idx = np.where(h == 0.0)[0]
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        # swap
        h[i], h[j] = 0.0, 1.0
        new_c = fast_c5(h)
        delta = new_c - cur_c
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = 1.0, 0.0
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ---------- global cache for fast C5 ----------
_fast_c5_cache = {}

# ---------- helpers ----------
def enforce_sum(arr, target, rng, max_iter=5000):
    arr = np.clip(arr, 0.0, 1.0)
    diff = target - arr.sum()
    n = arr.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - arr[i])
        else:
            add = max(diff, -arr[i])
        arr[i] += add
        diff = target - arr.sum()
    if abs(diff) > 1e-12:
        corr = diff / n
        arr += corr
        arr = np.clip(arr, 0.0, 1.0)
    return arr


def compute_c5(h):
    """Exact discrete C5 using np.correlate."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def fast_c5(h):
    """Fast C5 for binary (or any) h using FFT and prefix sums."""
    n = h.size
    cache = _fast_c5_cache.get(n)
    if cache is None:
        ks = np.arange(-(n - 1), n)
        pos = ks > 0
        neg = ks < 0
        _fast_c5_cache[n] = (ks, pos, neg)
        cache = (ks, pos, neg)
    ks, pos, neg = cache

    H = np.fft.rfft(h, n=2 * n)
    corr = np.fft.irfft(H * np.conj(H), n=2 * n)[: 2 * n - 1]

    pref = np.concatenate(([0.0], np.cumsum(h)))
    overlap = np.empty(2 * n - 1, dtype=np.float64)
    kp = ks[pos]
    overlap[pos] = pref[n - kp]
    kn = -ks[neg]
    overlap[neg] = pref[n] - pref[kn]
    overlap[ks == 0] = pref[n]

    diff = overlap - corr
    return np.max(diff) * (2.0 / n)


def binary_from_continuous(h):
    n = h.size
    idx = np.argsort(-h)
    mask = np.zeros(n, dtype=np.float64)
    mask[idx[: n // 2]] = 1.0
    return mask


# ---------- simulated annealing ----------
def simulated_annealing(h_start, rng, time_limit):
    h = h_start.copy()
    cur_c = fast_c5(h)
    best_h, best_c = h.copy(), cur_c
    t0 = time.time()
    T0, T_end = 0.05, 1e-4

    n = h.size
    while time.time() - t0 < time_limit:
        elapsed = time.time() - t0
        T = T0 * (1.0 - elapsed / time_limit) + T_end * (elapsed / time_limit)

        i = rng.integers(n)
        hi = h[i]
        # find opposite type index
        while True:
            j = rng.integers(n)
            if h[j] != hi:
                break

        h[i], h[j] = h[j], h[i]
        new_c = fast_c5(h)
        delta = new_c - cur_c
        if delta <= 0.0 or rng.random() < np.exp(-delta / T):
            cur_c = new_c
            if new_c < best_c:
                best_c, best_h = new_c, h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    return best_h, best_c


def hill_climb(h_start, rng, max_iter=2000):
    h = h_start.copy()
    cur_c = fast_c5(h)
    n = h.size
    for _ in range(max_iter):

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny drift correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x += diff / n
        x = np.clip(x, 0.0, 1.0)
    return x
