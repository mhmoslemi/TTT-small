    start = time.time()

def enforce_sum(h, target, rng):
    # Clip then adjust by random redistribution.
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # Use iterative approach.
    max_iter = 10000
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # add to random index with room
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i])
            # random fraction of max
            add = add * rng.random()
            h[i] += add
            diff -= add
        else:
            # subtract
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i])
            sub = sub * rng.random()
            h[i] -= sub
            diff += sub
    if abs(diff) > 1e-12:
        # final adjust a random index
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    # final adjustment maybe to correct rounding errors.
    return h

def enforce_sum(h, target, rng):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 10000
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i])
            add *= rng.random()
            h[i] += add
            diff -= add
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i])
            sub *= rng.random()
            h[i] -= sub
            diff += sub
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

    return h

def perturb_random(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

            new_h = perturb_random(best_h, rng, scale=0.02)
            c_new = compute_c5(new_h)
            if c_new < best_c:
                best_h, best_c = new_h.copy(), c_new

    # final fine‑tune
    while time.time() - start_time < budget_s * 0.99:
        new_h = perturb_random(best_h, rng, scale=0.01)
        c_new = compute_c5(new_h)
        if c_new < best_c:
            best_h, best_c = new_h.copy(), c_new

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
