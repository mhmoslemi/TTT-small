def simulated_annealing(h0, rng, time_left):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    c = best_c
    T0 = (best_c * 0.05) maybe 0.01
    iter =0
    max_iters = large (maybe 200000)
    while time.time() < deadline and iter < max_iters:
        # propose move
        i = rng.integers(0,n)
        j = rng.integers(0,n-1)
        if j >= i: j+=1
        # compute epsilon bounds
        hi = h[i]; hj = h[j]
        max_eps_pos = min(hi, 1 - hj)
        max_eps_neg = min(hj, 1 - hi)
        if max_eps_pos+max_eps_neg ==0:
            continue
        eps = rng.random() * (max_eps_pos+max_eps_neg) - max_eps_neg
        # optionally scale eps: by 0.2? we can sample from interval directly, no scaling.
        # If eps very small may be negligible: but okay.
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps

        # compute C5
        new_c = compute_c5(new_h)
        delta = new_c - c
        # schedule T
        # linear decay: T = T0 * (1 - iter/max_iters)
        T = T0 * (1 - iter / max_iters)
        if delta < 0 or rng.random() < np.exp(-delta / max(T,1e-12)):
            h = new_h
            c = new_c
            if c < best_c:
                best_c = c
                best_h = h.copy()
        iter+=1

import time
import numpy as np

# ------------------------------------------------------------
# Core utilities
# ------------------------------------------------------------
def compute_c5(h):
    """Exact C5 for step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng):
    """Adjust h to have Σh = target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-14:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-14:
            continue
        h[i] += add
        diff = target - h.sum()
    # final tiny correction if needed
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        for _ in range(1000):
            if abs(diff) < 1e-14:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            if abs(add) < 1e-14:
                continue
            h[i] += add
            diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Generate a random feasible vector (0≤h≤1, Σh=n/2)."""
    v = rng.exponential(1.0, size=n)
    v = v / v.sum() * (n / 2.0)
    v = np.clip(v, 0.0, 1.0)
    return enforce_sum(v, n / 2.0, rng)

def hill_climb(h_start, rng, max_iters=2500, start_step=0.5):
    """Simple hill‑climb: keep only improving moves."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        hi, hj = h[i], h[j]
        max_pos = min(hi, 1.0 - hj)
        max_neg = min(hj, 1.0 - hi)
        if max_pos + max_neg <= 0:
            continue
        eps = (rng.random() * (max_pos + max_neg)) - max_neg
        eps *= step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-6)
    return best_h, best_c


def simulated_annealing(h_start, rng, max_iters=20000, start_temp=0.01):
    """SA with pair‑wise mass transfers, preserving Σh."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    c = best_c
    n = h.size
    for it in range(max_iters):
        T = start_temp * (1.0 - it / max_iters)
        if T < 1e-12:
            T = 1e-12
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----- initial vector -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 180
        h0 = np.full(n, 0.5)
        h0 = enforce_sum(h0, n / 2.0, rng)

    target = n / 2.0

    # ----- generate diverse starting points --------------------------------
    candidates = [h0]
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx

    for f in range(1, 4):
        for fun in (np.sin, np.cos):
            h = 0.5 + 0.5 * fun(np.pi * f * xs)
            h = np.clip(h, 0.0, 1.0)
            candidates.append(enforce_sum(h, target, rng))

    candidates.append(enforce_sum(xs / 2.0, target, rng))
    candidates.append(enforce_sum(1.0 - xs / 2.0, target, rng))

    for _ in range(8):
        candidates.append(random_feasible_vector(n, rng))

    # ----- keep the best of the pool --------------------------------------
    best_h = None
    best_c = float("inf")
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    global_best_h, global_best_c = best_h, best_c

    # ----- hill‑climb on a few promising starts ---------------------------
    # sort by their C5 and pick the top few
    cand_sorted = sorted(candidates, key=lambda hh: compute_c5(hh))
    for h_start in cand_sorted[:5]:
        h_opt, c_opt = hill_climb(h_start, rng, max_iters=6000, start_step=0.5)
        if c_opt < global_best_c:
            global_best_h, global_best_c = h_opt, c_opt

    # ----- simulated annealing (time‑aware) -------------------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 0.2:
        # rough estimate: 2000 iterations per second
        max_sa_iters = int(min(200000, remaining * 2000))
        h_sa, c_sa = simulated_annealing(global_best_h, rng,
                                         max_iters=max_sa_iters,
                                         start_temp=0.01)
        if c_sa < global_best_c:
            global_best_h, global_best_c = h_sa, c_sa

    # ----- final fine hill‑climb -----------------------------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 0.1:
        fine_iters = int(min(40000, remaining * 2500))
        h_fine, c_fine = hill_climb(global_best_h, rng,
                                    max_iters=fine_iters,
                                    start_step=0.1)
        if c_fine < global_best_c:
            global_best_h, global_best_c = h_fine, c_fine

    # ----- optional up‑sampling -------------------------------------------
    if (time.time() - start_time) < budget_s * 0.98 and n * 2 <= 360:
        n2 = n * 2
        h_up = np.repeat(global_best_h, 2)
        target2 = n2 / 2.0
        h_up = enforce_sum(h_up, target2, rng)

        remaining = budget_s - (time.time() - start_time)
        if remaining > 0.1:
            it_up = int(min(25000, remaining * 2500))
            h_up_opt, c_up_opt = hill_climb(h_up, rng,
                                            max_iters=it_up,
                                            start_step=0.3)
            if c_up_opt < global_best_c:
                global_best_h, global_best_c, n = h_up_opt, c_up_opt, n2

    # ----- final clean‑up -------------------------------------------------
    global_best_h = np.clip(global_best_h, 0.0, 1.0)
    global_best_h = enforce_sum(global_best_h, n / 2.0, rng)
    final_c = compute_c5(global_best_h)

    return global_best_h.tolist(), float(final_c), int(n)
