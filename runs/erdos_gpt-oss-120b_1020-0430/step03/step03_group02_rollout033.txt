    corr = np.correlate(h, 1.0 - h, mode="full")

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=400):
    """Soft‑max SLSQP refinement for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # ------------------------------------------------------------------
    # initialise from parent if provided
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h_parent = np.array(initial_h_values, dtype=float)
        h_parent = np.clip(h_parent, 0.0, 1.0)
        h_parent = enforce_sum(h_parent, rng)
        c_parent = compute_c5(h_parent)
        best_h, best_c, best_n = h_parent, c_parent, h_parent.size

    # candidate discretisations (prefer finer grids)
    cand_ns = [96, 120, 144, 168, 192, 216, 240, 256]
    if best_n is not None:
        cand_ns = [best_n] + [n for n in cand_ns if n != best_n]

    # ------------------------------------------------------------------
    # main multi‑restart optimisation loop
    # ------------------------------------------------------------------
    for n in cand_ns:
        if time.time() - start > 0.94 * budget_s:
            break
        # number of independent restarts for this n
        restarts = 6 if n != best_n else 8
        for r in range(restarts):
            if time.time() - start > 0.94 * budget_s:
                break
            # initialise h for this restart
            if best_h is not None and best_n == n and r == 0:
                h = best_h.copy()
            elif "initial_h_values" in globals() and best_h is not None:
                # interpolate parent to current resolution
                x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
                x_new = np.linspace(0.0, 2.0, n, endpoint=False)
                h = np.interp(x_new, x_old, best_h)
                h = np.clip(h, 0.0, 1.0)
                h = enforce_sum(h, rng)
            else:
                h = random_feasible_vector(n, rng)

            # 1) guided hill‑climb
            h, c = guided_hill_climb(h, rng, max_iters=3000, step_scale=0.6)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

            # 2) SLSQP refinement with increasing β
            for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0):
                if time.time() - start > 0.96 * budget_s:
                    break
                maxit = 800 if beta >= 500 else 500
                h = slsqp_optimize(h, beta, rng, max_iter=maxit)
                c = compute_c5(h)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n

            # 3) fine gradient refinement
            if time.time() - start < 0.97 * budget_s:
                h = gradient_refine(h, rng, step=0.02, max_iter=12)
                c = compute_c5(h)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n

            # 4) second guided hill‑climb (shorter)
            if time.time() - start < 0.98 * budget_s:
                h, c = guided_hill_climb(h, rng, max_iters=2000, step_scale=0.4)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n

            # 5) small guided perturbations around the current global best
            for _ in range(10):
                if time.time() - start > 0.999 * budget_s:
                    break
                grad = worst_gradient(best_h)
                nh = guided_perturb(best_h, grad, rng, scale=0.02)
                nc = compute_c5(nh)
                if nc < best_c:
                    best_h, best_c = nh.copy(), nc

    # ------------------------------------------------------------------
    # optional up‑sampling (if time permits)
    # ------------------------------------------------------------------
    if best_h is not None and time.time() - start < 0.99 * budget_s:
        if best_n * 2 <= 300:
            h_up = np.repeat(best_h, 2)
            n_up = h_up.size
            h_up, c_up = guided_hill_climb(h_up, rng, max_iters=2500, step_scale=0.5)
            for beta in (5.0, 20.0, 80.0, 200.0):
                if time.time() - start > budget_s:
                    break
                h_up = slsqp_optimize(h_up, beta, rng, max_iter=400)
                c_up = compute_c5(h_up)
                if c_up < best_c:
                    best_h, best_c, best_n = h_up.copy(), c_up, n_up

    # ------------------------------------------------------------------
    # final simulated‑annealing‑style random transfers (remaining budget)
    # ------------------------------------------------------------------
    while time.time() - start < budget_s * 0.999 and best_h is not None:
        n = best_h.size
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * 0.05
        nh = best_h.copy()
        nh[i] -= eps
        nh[j] += eps
        c_new = compute_c5(nh)
        if c_new < best_c:
            best_h, best_c = nh, c_new

    # ------------------------------------------------------------------
    # final feasibility enforcement and return
    # ------------------------------------------------------------------
    if best_h is None:
        best_n = 96
        best_h = np.full(best_n, 0.5)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
