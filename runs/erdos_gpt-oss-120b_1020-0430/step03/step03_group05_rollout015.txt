import time
import numpy as np
from scipy.optimize import minimize

# ---------- Global placeholders for the soft‑max surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) via log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_to_feasible(v, target):
    """Projection of v onto {0≤h≤1, Σh = target} via a shift bisection."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Fast random feasible vector (0≤h≤1, Σh=n/2)."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_to_feasible(v, n / 2.0)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=3000, start_step=0.2):
    """Simple deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


def slsqp_optimize(h0, beta, rng, max_iter=500):
    """Run a single SLSQP optimisation with the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def annealing_optimize(h0, rng, max_iter, start_temp, end_temp):
    """Simulated‑annealing style local search using perturb_h."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iter):
        # exponential temperature schedule
        t = start_temp * ((end_temp / start_temp) ** (it / max_iter))
        scale = max(0.05, t)  # step size linked to temperature
        new_h = perturb_h(h, rng, scale=scale)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h.copy()
        # Metropolis acceptance
        if c < compute_c5(h) or rng.random() < np.exp(-(c - best_c) / (t + 1e-12)):
            h = new_h
    return best_h, best_c


def upsample_initial(to_n):
    """Linear interpolation of the global initial_h_values to length to_n."""
    init = np.array(globals()["initial_h_values"], dtype=float)
    m = init.size
    xs_old = np.linspace(0.0, 2.0, m, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, to_n, endpoint=False)
    return np.interp(xs_new, xs_old, init)


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # candidate discretisations (coarse to fine)
    candidate_ns = [120, 180, 240, 300]
    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        if n0 not in candidate_ns:
            candidate_ns = [n0] + candidate_ns

    best_h = None
    best_c = np.inf
    best_n = None

    for n in candidate_ns:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.85:
            break
        remaining = budget_s - elapsed
        if remaining < 5.0:
            break

        # ----- build starting pool -----
        starts = []
        if "initial_h_values" in globals() and n == len(initial_h_values):
            starts.append(np.array(initial_h_values, dtype=float))
        elif "initial_h_values" in globals():
            starts.append(project_to_feasible(upsample_initial(n), n / 2.0))

        # add a few random feasible vectors
        for _ in range(4):
            starts.append(random_feasible_vector(n, rng))

        # allocate a rough time slice for this n
        time_for_n = min(remaining * 0.6, max(30.0, n * 0.4))

        # ---- optimise each start ----
        for h0 in starts:
            if time.time() - start_time > budget_s * 0.92:
                break
            h = h0.copy()
            # progressive SLSQP with increasing beta
            for beta in (10.0, 50.0, 200.0, 800.0, 2000.0, 8000.0, 20000.0):
                if time.time() - start_time > budget_s * 0.90:
                    break
                try:
                    h = slsqp_optimize(h, beta, rng, max_iter=400)
                except Exception:
                    continue
            c = compute_c5(h)
            if c < best_c:
                best_c, best_h, best_n = c, h.copy(), n

            # ---- annealing fine‑tune ----
            if time.time() - start_time > budget_s * 0.90:
                break
            # allocate a few seconds based on remaining budget
            rem = budget_s - (time.time() - start_time)
            iters = int(min(8000, max(2000, rem * 20)))
            h_ann, c_ann = annealing_optimize(
                h, rng, max_iter=iters, start_temp=0.1, end_temp=1e-5
            )
            if c_ann < best_c:
                best_c, best_h, best_n = c_ann, h_ann.copy(), n

        # optional brief hill‑climb on the current best of this n
        if best_h is not None and best_h.size == n:
            rem = budget_s - (time.time() - start_time)
            if rem > 5.0:
                iters = int(min(5000, rem * 250))
                h_hc, c_hc = hill_climb(best_h, rng, max_iters=iters, start_step=0.15)
                if c_hc < best_c:
                    best_c, best_h, best_n = c_hc, h_hc.copy(), n

    # ----- final global polishing -----
    if best_h is not None:
        rem = budget_s - (time.time() - start_time)
        if rem > 3.0:
            iters = int(min(10000, rem * 300))
            best_h, best_c = hill_climb(best_h, rng, max_iters=iters, start_step=0.1)

    # enforce feasibility one last time
    if best_h is None:
        n = 120
        best_h = np.full(n, 0.5)
        best_n = n
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_h.size / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, best_h.size)
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
