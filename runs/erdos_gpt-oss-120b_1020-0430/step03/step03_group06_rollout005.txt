def gradient_step(h, target, step_scale):
    # compute worst shift
    n = len(h)
    auto_corr = np.correlate(h, h, mode='full')
    min_idx = np.argmin(auto_corr)
    shift = min_idx - (n - 1)
    # compute neighbor sums
    neighbor = np.zeros_like(h)
    if shift >= 0:
        # i+shift term
        neighbor[:n-shift] += h[shift:]
        # i-shift term
        neighbor[shift:] += h[:n-shift]
    else:
        s = -shift
        # i+shift term (shift negative)
        neighbor[s:] += h[:n-s]
        neighbor[:n-s] += h[s:]
    # direction is neighbor (positive)
    max_n = neighbor.max()
    if max_n == 0:
        return h
    d = neighbor / max_n  # normalized to [0,1]
    step = step_scale * target / n  # maybe 0.01*0.5=0.005
    h_new = h + step * d
    # clip if needed
    h_new = np.clip(h_new, 0.0, 1.0)
    # enforce sum (possibly using enforce_sum)
    h_new = enforce_sum(h_new, target, rng)
    return h_new

import time
import numpy as np
from scipy.optimize import minimize

# Basic utilities

def enforce_sum(h, target, rng):
    """Adjust h so that sum = target while 0<=h<=1 (random walk)."""
    diff = target - h.sum()
    n = h.size
    # simple random walk to fix the sum
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-12:
            continue
        h[i] += add
        diff = target - h.sum()
    # final tiny correction if needed
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(1000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            if abs(add) < 1e-12:
                continue
            h[i] += add
            diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    v = rng.exponential(1.0, size=n)
    v = v / v.sum() * (n / 2.0)
    h = np.clip(v, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def perturb_h(h, rng, scale=0.5):
    """Transfer a small amount between two coordinates (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    while j == i:
        j = rng.integers(0, n)
    max_pos = min(h[i], 1.0 - h[j])
    max_neg = min(h[j], 1.0 - h[i])
    if max_pos + max_neg == 0:
        return h.copy()
    eps = (rng.random() * (max_pos + max_neg)) - max_neg
    eps *= scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return np.clip(new_h, 0.0, 1.0)


def compute_c5(h):
    """Exact discrete C₅ for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


# ----------------------------------------------------------------------
# Smooth surrogate for SLSQP (log‑sum‑exp of overlaps)
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny exact‑sum repair
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            if abs(add) < 1e-12:
                continue
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

# Gradient‑descent style improvement focusing on the worst shift
# ----------------------------------------------------------------------
def worst_shift_neighbor(h):
    """Return (shift, neighbor_sum) for the shift with minimal auto‑correlation."""
    n = h.size
    auto = np.correlate(h, h, mode="full")
    min_idx = int(np.argmin(auto))
    shift = min_idx - (n - 1)          # integer offset, can be negative
    neighbor = np.zeros_like(h)

    if shift == 0:
        neighbor = 2.0 * h
    elif shift > 0:
        neighbor[: n - shift] += h[shift:]
        neighbor[shift:] += h[: n - shift]
    else:  # shift < 0
        s = -shift
        neighbor[s:] += h[: n - s]
        neighbor[: n - s] += h[s:]

    return shift, neighbor


def gradient_step(h, target, step_scale, rng):
    """One gradient‑like step aimed at improving the worst shift."""
    shift, neighbor = worst_shift_neighbor(h)
    max_n = neighbor.max()
    if max_n == 0.0:
        return h
    # normalized direction (0‑1)
    d = neighbor / max_n
    # step size proportional to average mass
    step = step_scale * target / h.size
    h_new = h + step * d
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = enforce_sum(h_new, target, rng)
    return h_new

    start_time = time.time()
    total = budget_s

    # ------------------------------------------------------------------

    if "initial_h_values" in globals():

    best_h = h0.copy()

        c_new = compute_c5(h_new)

        if c_new < best_c:
            best_h, best_c = h_new.copy(), c_new

            # after a genuine improvement run a quick SLSQP sweep

            c_rand = compute_c5(h_rand)
            if c_rand < best_c:

    # Upsample to a finer grid if time permits
    # ------------------------------------------------------------------

        n_up = h_up.size
        target_up = n_up / 2.0
        h_up = enforce_sum(h_up, target_up, rng)

        # refine on finer grid
        best_h_up = h_up.copy()
        best_c_up = compute_c5(best_h_up)
        iter_up = 0
        while time.time() - start_time < total * 0.97:
            step_scale = 0.03 * max(1.0 - iter_up / 4000.0, 0.1)
            h_new = gradient_step(best_h_up, target_up, step_scale, rng)
            c_new = compute_c5(h_new)
            if c_new < best_c_up:
                best_h_up, best_c_up = h_new.copy(), c_new
                # quick SLSQP sweep
                for beta in beta_list:
                    if time.time() - start_time > total * 0.97:
                        break
                    h_ref = slsqp_optimize(best_h_up, beta, max_iter=300)
                    c_ref = compute_c5(h_ref)
                    if c_ref < best_c_up:
                        best_h_up, best_c_up = h_ref.copy(), c_ref
            iter_up += 1

        # keep the better of coarse vs fine
        if best_c_up < best_c:
            best_h, best_c, n = best_h_up.copy(), best_c_up, n_up

    # ------------------------------------------------------------------
    # Final clean‑up
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
