def gradient_of_shift(h, k):
    n = h.size
    grad = np.ones_like(h)
    if k>=0:
        # forward shift term: h_{j+k}
        grad[:n-k] -= h[k:]
        # backward shift term: h_{j-k}
        grad[k:] -= h[:n-k]
    else:
        k_abs = -k
        # forward term h_{j+k}=h_{j - k_abs}
        grad[k_abs:] -= h[:n-k_abs]
        # backward term h_{j - k}=h_{j + k_abs}
        grad[:n - k_abs] -= h[k_abs:]
    return grad

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def enforce_sum(h, target, rng):
    """Adjust h in-place so that sum(h) == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0.0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Return a random vector in [0,1]^n with sum = n/2."""
    h = rng.random(n)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    return enforce_sum(h, target, rng)


def correlate_full(h):
    """Full linear correlation of h with (1‑h) using FFT."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)


def gradient_of_shift(h, k):
    """Gradient of the worst‑shift term for a given shift k (int)."""
    n = h.size
    grad = np.ones_like(h)
    if k >= 0:
        if k < n:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
    else:  # k < 0
        k = -k
        if k < n:
            grad[k:] -= h[: n - k]
            grad[: n - k] -= h[k:]
    return grad


def worst_shift_and_gradient(h):
    """Return the shift giving max overlap and its gradient."""
    conv = correlate_full(h)
    idx = int(np.argmax(conv))
    n = h.size
    k = idx - (n - 1)
    grad = gradient_of_shift(h, k)
    return k, grad


def gradient_descent(
    h,
    target,
    rng,
    max_iters,
    init_step=0.5,
    enforce_every=10,
    time_limit=None,
):
    """Projected sub‑gradient descent on maxₖ ∑ hᵢ(1‑hᵢ₊ₖ)."""
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    start = time.time()
    for it in range(max_iters):
        if time_limit is not None and time.time() > time_limit:
            break
        k, grad = worst_shift_and_gradient(h)
        step = init_step / np.sqrt(it + 1.0)
        h_new = h - step * grad
        np.clip(h_new, 0.0, 1.0, out=h_new)

        if it % enforce_every == 0:
            diff = target - h_new.sum()
            if abs(diff) > 1e-9:
                h_new = enforce_sum(h_new, target, rng)

        h = h_new
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c


def random_perturb(h, rng, scale=0.1):
    """Transfer a small amount between two random positions (keeps sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new


def upsample_vector(h, factor, rng):
    """Linear interpolation up‑sampling by integer factor, then enforce sum."""
    n = h.size
    n_up = n * factor
    xs_old = (np.arange(n) + 0.5) * (2.0 / n)
    xs_new = (np.arange(n_up) + 0.5) * (2.0 / n_up)
    h_up = np.interp(xs_new, xs_old, h)
    target = n_up / 2.0
    h_up = np.clip(h_up, 0.0, 1.0)
    return enforce_sum(h_up, target, rng)
