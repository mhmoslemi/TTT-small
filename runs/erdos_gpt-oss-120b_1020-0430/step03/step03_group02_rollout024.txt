def anneal(h_init, rng, max_iters, start_temp, end_temp):
    h = h_init.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    N = len(h)
    for it in range(max_iters):
        T = start_temp * (end_temp/start_temp) ** (it / max_iters)
        # propose a swap
        # choose i where h[i]=1, j where h[j]=0
        ones = np.where(h>0.5)[0]
        zeros = np.where(h<0.5)[0]
        if len(ones)==0 or len(zeros)==0: break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i], h[j] = h[j], h[i]
        c = compute_c5(h)
        delta = c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / (T+1e-12)):

            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = h[j], h[i]
    return best_h, best_c

def guided_hill_climb(h_start, rng, max_iters=2000, step_scale=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    N = h.size
    decay_interval = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        # find indices with max/min gradient
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps * step_scale
        h[i] -= eps
        h[j] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        if (it+1) % decay_interval == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)

    n = h.size

CURRENT_TARGET = None
CURRENT_BETA = None
def sum_eq_constraint(x): return np.sum(x) - CURRENT_TARGET
def softmax_obj(x):
    corr = np.correlate(x, 1-x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # smooth max: log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
def slsqp_optimize(h0,beta,rng,max_iter=300):
    global CURRENT_TARGET,CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n/2
    CURRENT_BETA = beta
    bounds = [(0.,1.)]*n
    cons = {'type':'eq','fun':sum_eq_constraint}
    from scipy.optimize import minimize
    res = minimize(softmax_obj,h0,method='SLSQP',bounds=bounds,constraints=cons,
                   options={'maxiter':max_iter,'ftol':1e-12,'disp':False})
    x = np.clip(res.x,0,1)
    # enforce sum exactly
    diff = CURRENT_TARGET - x.sum()
    if abs(diff)>1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0,n)
            add = min(diff, 1-x[i]) if diff>0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

import time, numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust vector so that Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k = Σ h_i(1‑h_{i+k})."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_hill_climb(h_start: np.ndarray, rng: np.random.Generator,
                     max_iters: int = 2000, step_scale: float = 0.4):
    """Hill‑climb using gradient‑directed transfers."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0.0:
                eps = rng.random() * max_eps * step_scale
                h[i] -= eps
                h[j] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def random_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h


def anneal_binary(h: np.ndarray, rng: np.random.Generator,
                  steps: int, start_temp: float, end_temp: float):
    """Simulated annealing on binary vectors (swap 1↔0)."""
    N = h.size
    cur = h.copy()
    cur_c = compute_c5(cur)
    best_h, best_c = cur.copy(), cur_c
    for it in range(steps):
        T = start_temp * (end_temp / start_temp) ** (it / steps)
        i, j = rng.integers(N, size=2)
        if cur[i] == cur[j]:
            continue
        cur[i], cur[j] = cur[j], cur[i]          # swap
        c = compute_c5(cur)
        delta = c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            cur_c = c
            if c < best_c:
                best_c, best_h = c, cur.copy()
        else:
            cur[i], cur[j] = cur[j], cur[i]      # revert
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                   step: float = 0.02, max_iter: int = 8) -> np.ndarray:
    """Fine‑tune using worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        k = abs(np.argmax(corr) - (n - 1))
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h


def binarize(h: np.ndarray) -> np.ndarray:
    """Return a binary vector (n/2 ones) taking the largest entries of h."""
    n = h.size
    idx = np.argsort(-h)[: n // 2]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    return b

CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq(x):
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 300) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq}
    res = minimize(_softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()

            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x
