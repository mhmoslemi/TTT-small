def simulated_annealing(h0, rng, max_iters, init_temp, final_temp):

    h = h0.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c

    for it in range(max_iters):
        t = init_temp * ((final_temp/init_temp) ** (it / max_iters))
        # compute gradient
        grad = worst_gradient(h)
        # pick move
        # probability to select gradient-based or random move
        if rng.random() < 0.8:
            i = int(np.argmax(grad))
            j = int(np.argmin(grad))
        else:
            i = rng.integers(0, h.size)
            j = rng.integers(0, h.size)
            while j == i:
                j = rng.integers(0, h.size)
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        # epsilon scale based on temperature: some fraction of max_eps
        eps = max_eps * rng.random() * (0.5 + 0.5 * t / init_temp) # higher at high temp.

        # maybe sometimes also negative move (swap direction)
        # Actually we always move from i to j.

        nh = h.copy()
        nh[i] -= eps
        nh[j] += eps

        c = compute_c5(nh)

        if c < cur_c:
            h, cur_c = nh, c
            if c < best_c:
                best_h, best_c = nh.copy(), c
        else:
            # accept with prob exp(-(c-cur)/t)
            if t > 0 and rng.random() < np.exp(-(c - cur_c) / t):
                h, cur_c = nh, c

    return best_h, best_c

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def simulated_annealing(h0, rng, max_iters, temp0, temp_end):
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    n = h.size
    for it in range(max_iters):
        # temperature schedule exponential decay
        t = temp0 * ((temp_end / temp0) ** (it / max_iters))
        # compute gradient for direction
        grad = worst_gradient(h)
        # with prob 0.9 choose gradient-based move
        if rng.random() < 0.9:
            i = int(np.argmax(grad))
            j = int(np.argmin(grad))
        else:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        # step size proportional to temperature
        # At high temp allow large moves, but limited by max_eps
        eps = max_eps * rng.random() * (0.3 + 0.7 * t / temp0)  # between 0.3 and 1 factor area
        if eps <= 0:
            continue
        nh = h.copy()
        nh[i] -= eps
        nh[j] += eps
        c_new = compute_c5(nh)
        # accept if better or with probability
        if c_new < cur_c or (t > 0 and rng.random() < np.exp(-(c_new - cur_c) / t)):
            h, cur_c = nh, c_new
            if c_new < best_c:
                best_c, best_h = c_new, nh.copy()
    return best_h, best_c

import time, numpy as np

def compute_c5(h):
    """Discrete C5 for a step function on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def worst_gradient(h):
    """Sub‑gradient of the worst‑shift overlap."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        return 1.0 - 2.0 * h
    grad = np.ones_like(h)
    grad[: n - k] -= h[k:]
    grad[k:] -= h[: n - k]
    return grad


def guided_hill_climb(h, rng, max_iters=2000, step_scale=0.5):
    """Simple gradient‑directed hill climbing."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    scale = step_scale
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            break
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = max_eps * rng.random() * scale
        if eps <= 0:
            continue
        h[i] -= eps
        h[j] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        else:
            # revert non‑improving step
            h[i] += eps
            h[j] -= eps
        if (it + 1) % decay == 0:
            scale = max(scale * 0.7, 1e-4)
    return best_h, best_c


def anneal_with_budget(h, rng, start, budget_s):
    """Simulated‑annealing loop that respects the global time budget."""
    n = h.size
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    temp0, temp_end = 5e-4, 5e-8
    max_iters = 2_000_000
    it = 0
    while True:
        remaining = budget_s - (time.time() - start)
        if remaining < 0.1:
            break
        batch = min(4000, int(remaining * 4000))
        if batch <= 0:
            break
        for _ in range(batch):
            # exponential temperature schedule
            t = temp0 * ((temp_end / temp0) ** (it / max_iters))
            grad = worst_gradient(h)
            if rng.random() < 0.9:
                i = int(np.argmax(grad))
                j = int(np.argmin(grad))
            else:
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                while j == i:
                    j = rng.integers(0, n)
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0:
                eps = max_eps * rng.random() * (0.2 + 0.8 * t / temp0)
                if eps > 0:
                    nh = h.copy()
                    nh[i] -= eps
                    nh[j] += eps
                    c_new = compute_c5(nh)
                    accept = c_new < cur_c or (t > 0 and rng.random() < np.exp(-(c_new - cur_c) / t))
                    if accept:
                        h, cur_c = nh, c_new
                        if c_new < best_c:
                            best_c, best_h = c_new, nh.copy()
            it += 1
    return best_h, best_c
