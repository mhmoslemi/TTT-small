```
def transfer_move(h, k, rng, scale):
    n = len(h)
    if k >= 0:
        i = rng.integers(0, n - k)
        j = i + k
    else:
        i = rng.integers(-k, n)
        j = i + k  # j = i + k < i
    # ensure i != j else no effect
    # compute max epsilon we can move
    max_eps = min(h[i], 1 - h[j])
    if max_eps <= 0:
        return h
    eps = rng.random() * max_eps * scale
    # could also random sign? we always move from i to j (h_i high, h_j low).
    if eps == 0:
        return h
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h
```

```
def select_pair(h, k, rng):
    n = len(h)
    if k >= 0:
        valid = slice(0, n - k)
        i_vals = np.arange(0, n - k)
        contrib = h[valid] * (1 - h[valid + k])
    else:
        valid = slice(-k, n)
        i_vals = np.arange(-k, n)
        contrib = h[valid] * (1 - h[valid + k])
    total = contrib.sum()
    if total == 0:
        # fallback random
        i = rng.integers(len(i_vals))
        return i_vals[i], i_vals[i] + k
    # choose according to contribution
    probs = contrib / total
    # generate random sample via cumulative
    r = rng.random()
    cum = np.cumsum(probs)
    idx = np.searchsorted(cum, r)
    i = i_vals[idx]
    j = i + k
    return i, j
```

```
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)
        return int(i), int(j)

    total = contrib.sum()
    if total > 0 and rng.random() < 0.9:
        probs = contrib / total
        r = rng.random()
        cum = np.cumsum(probs)
        idx = np.searchsorted(cum, r, side="right")
        i = i_vals[idx]
        j = i + k
    else:
        # fallback uniform choice inside the valid range
        if k >= 0:
            i = rng.integers(0, n - k)
        else:
            i = rng.integers(-k, n)
        j = i + k
    return int(i), int(j)


def local_search(h0, rng, time_limit):
    """Targeted hill‑climb focusing on the currently worst shift."""
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    scale = 0.5
    it = 0
    while time.time() - start < time_limit:
        it += 1
        corr = np.correlate(h, 1 - h, mode="full")
        max_idx = int(np.argmax(corr))
        k = max_idx - (n - 1)
        max_c = corr[max_idx] * (2.0 / n)

        if max_c < best_c:
            best_c = max_c
            best_h = h.copy()

        i, j = select_pair(h, k, rng)
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            h_new = perturb_h(h, rng, scale=scale)
        else:
            eps = rng.random() * max_eps * scale
            if eps <= 1e-12:
                h_new = h.copy()
            else:
                h_new = h.copy()
                h_new[i] -= eps
                h_new[j] += eps

        c_new = compute_c5(h_new)

        # accept improvements
        if c_new < max_c - 1e-12:
            h = h_new
            scale = max(scale * 0.995, 1e-4)
        else:
            # occasional uphill move to escape plateaus
            if rng.random() < 0.001:
                h = h_new
            else:
                if rng.random() < 0.02:
                    h = perturb_h(h, rng, scale=scale)

        if it % 200 == 0:
            scale = max(scale * 0.9, 1e-4)

    # final feasibility enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    target = n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()
    return best_h, compute_c5(best_h)


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # candidate discretisations (start coarse, finish fine)
    candidate_ns = [60, 84, 120, 168, 240, 300, 360, 420, 480]
    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        if n0 not in candidate_ns:
            candidate_ns = [n0] + candidate_ns
    candidate_ns = sorted(set(candidate_ns))

    for n in candidate_ns:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.98:
            break
        remaining = budget_s - elapsed
        # allocate a modest chunk of the remaining budget to this n
        time_for_n = min(remaining * 0.5, max(5.0, n * 0.4))

        # build a small pool of seeds
        seeds = []
        if best_h is not None:
            up = upsample(best_h, n)
            up = project_to_feasible(up, n / 2.0)
            seeds.append(up)
        if "initial_h_values" in globals():
            init = np.array(initial_h_values, dtype=float)
            up = upsample(init, n)
            up = project_to_feasible(up, n / 2.0)
            seeds.append(up)
        while len(seeds) < 3:
            seeds.append(random_feasible_vector(n, rng))

        time_per_seed = time_for_n / len(seeds)
        for h0 in seeds:
            h_opt, c_opt = local_search(h0, rng, time_per_seed)
            if c_opt < best_c:
                best_c, best_h, best_n = c_opt, h_opt.copy(), n

        # optional SLSQP polishing if time permits
        if time.time() - start_time < budget_s * 0.96:
            for beta in (50.0, 200.0, 800.0, 2000.0):
                if time.time() - start_time > budget_s * 0.96:
                    break
                try:
                    h_ref = slsqp_optimize(best_h, beta, rng, max_iter=400)
                except Exception:
                    continue
                c_ref = compute_c5(h_ref)
                if c_ref < best_c:
                    best_c, best_h = c_ref, h_ref.copy()

    # final feasibility enforcement
    if best_h is None:
        best_n = 60
        best_h = np.full(best_n, 0.5)

    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(10000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, best_n)
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
```
