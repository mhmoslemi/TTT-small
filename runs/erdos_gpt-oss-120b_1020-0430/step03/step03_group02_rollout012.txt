def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.empty_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[:] = 0.0
        # for i < n-k: term from (1 - h[i+k])
        grad[: n - k] += 1.0 - h[k:]
        # for i >= k: term from -h[i-k]
        grad[k:] -= h[: n - k]
    return grad
```

def guided_perturb(h, grad, rng, scale=0.5):
    # choose i with max grad and j with min grad
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        # pick random distinct indices
        i, j = rng.choice(h.size, size=2, replace=False)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh
```

def guided_hill_climb(h, rng, max_iters=5000, step_scale=0.5, decay_every=500, decay_factor=0.7):
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_h = h.copy()
    cur_c = best_c
    n = h.size
    step = step_scale
    for it in range(max_iters):
        grad = worst_gradient(cur_h)

import time, numpy as np
from scipy.optimize import minimize

# ---------- helpers ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] with n points."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple greedy redistribution
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # tiny fallback (should be almost zero)
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random h satisfying 0≤h≤1 and Σh=n/2."""
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, rng)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the currently worst overlap shift."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        return 1.0 - 2.0 * h
    grad = np.zeros_like(h)
    grad[: n - k] += 1.0 - h[k:]         # (1 - h_{i+k})
    grad[k:]     -= h[: n - k]          # -h_{i-k}
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a high‑gradient index to a low‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        i, j = rng.choice(h.size, size=2, replace=False)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 5000, step_scale: float = 0.5,
                      decay_every: int = 600, decay_factor: float = 0.7):
    """Hill‑climb using gradient‑directed transfers."""
    best_h, best_c = h.copy(), compute_c5(h)
    cur_h, cur_c = best_h.copy(), best_c
    step = step_scale
    for it in range(max_iters):
        grad = worst_gradient(cur_h)
        nh = guided_perturb(cur_h, grad, rng, scale=step)
        nc = compute_c5(nh)
        if nc < cur_c:
            cur_h, cur_c = nh, nc
            if nc < best_c:
                best_h, best_c = nh.copy(), nc
        if (it + 1) % decay_every == 0:
            step = max(step * decay_factor, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.02, max_iter: int = 8):
    """Fine‑tune using explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:]     -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h


# ----- soft‑max SLSQP optimizer (same as parent) -----
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 500):
    """Soft‑max SLSQP refinement for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj, h0, method="SLSQP",
        bounds=bounds, constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x
