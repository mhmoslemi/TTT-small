```
import cvxpy as cp

def solve_milp(n, time_limit):
    h = cp.Variable(n, boolean=True)  # binary
    # compute M
    M = n * (n + 1) // 2
    W = cp.Variable(M, boolean=True)
    # compute start indices
    start = np.empty(n, dtype=int)
    s = 0
    for k in range(n):
        start[k] = s
        s += n - k

    # constraints list
    constraints = []
    # sum constraint
    constraints.append(cp.sum(h) == n/2)
    # product constraints
    for k in range(n):
        L = n - k
        idx_start = start[k]
        # for each i in range(L):
        # h[i] + h[i+k] - 1 <= w <= min(h[i], h[i+k]
        for i in range(L):
            w_idx = idx_start + i
            constraints.append(W[w_idx] <= h[i])
            constraints.append(W[w_idx] <= h[i+k])
            constraints.append(W[w_idx] >= h[i] + h[i+k] - 1)
    # t variable
    t = cp.Variable(integer=True)
    constraints.append(t >= 0)
    # shift constraints
    for k in range(1, n):
        L = n - k
        idx_start = start[k]
        overlap = cp.sum(W[idx_start: idx_start + L])
        a_k = cp.sum(h[:L])
        b_k = cp.sum(h[k:])
        constraints.append(t >= a_k - overlap)
        constraints.append(t >= b_k - overlap)

    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.CBC, timeLimit=time_limit, msg=False)
    # check status
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        return None, None
    h_val = np.rint(h.value).astype(float) # convert to 0/1 float
    t_val = int(round(t.value))
    return h_val, t_val
```

import time
import numpy as np
import cvxpy as cp

# -------------------------------------------------------------
# Basic utilities
# -------------------------------------------------------------
def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.3):
    """Simple hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c - 1e-12:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-5)
    return best_h, best_c


def make_feasible(h):
    """Project a floating vector onto {0≤h≤1, Σh = n/2} by simple redistribution."""
    n = h.size
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    rng = np.random.default_rng(0)
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


# -------------------------------------------------------------
# MILP optimizer (binary formulation)
# -------------------------------------------------------------
def milp_optimize(n, time_limit=30):
    """
    Solve the exact binary min‑max overlap problem for a given n.
    Returns (h_vector, c5_value) on success, (None, None) otherwise.
    """
    if n % 2 != 0:
        return None, None
    target = n // 2

    # variables
    h = cp.Variable(n, boolean=True)
    M = n * (n + 1) // 2
    w = cp.Variable(M, boolean=True)
    t = cp.Variable(integer=True)

    # start indices for each shift k
    start = np.empty(n, dtype=int)
    cur = 0
    for k in range(n):
        start[k] = cur
        cur += n - k

    constraints = [cp.sum(h) == target,
                   t >= 0]

    # product linearisation constraints
    for k in range(n):
        L = n - k
        base = start[k]
        for i in range(L):
            wi = w[base + i]
            hi = h[i]
            hj = h[i + k]
            constraints.append(wi <= hi)
            constraints.append(wi <= hj)
            constraints.append(wi >= hi + hj - 1)

    # max‑overlap constraints for each non‑zero shift (both signs)
    for k in range(1, n):
        L = n - k
        base = start[k]
        overlap = cp.sum(w[base:base + L])
        a_k = cp.sum(h[:L])          # Σ_{i=0}^{L-1} h_i
        b_k = cp.sum(h[k:])          # Σ_{i=k}^{n-1} h_i
        constraints.append(t >= a_k - overlap)
        constraints.append(t >= b_k - overlap)

    # objective
    prob = cp.Problem(cp.Minimize(t), constraints)

    # solve (CBC, with optional time limit)
    try:
        prob.solve(solver=cp.CBC, max_seconds=time_limit, verbose=False)
    except Exception:
        return None, None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None, None

    h_val = np.rint(h.value).astype(float)  # binary 0/1
    t_val = int(round(t.value))
    c5_val = (2.0 / n) * t_val
    return h_val, c5_val

# Main entry point
# -------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # candidate discretisations (small → moderate)
    candidate_ns = [60, 70, 80, 90, 100]

    # if a parent solution exists, also try its size (usually 300)
    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        if n0 not in candidate_ns:
            candidate_ns = [n0] + candidate_ns

    TARGET_C5 = 0.3808

    for n in candidate_ns:
        # time check
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining < 5.0:
            break

        # -------------------------------------------------
        # 1) MILP stage (binary exact)
        # -------------------------------------------------
        # allocate at most half of remaining budget, but cap for safety
        milp_time = min(remaining * 0.5, 30.0)
        h_bin, c5_bin = milp_optimize(n, time_limit=milp_time)
        if h_bin is not None and c5_bin is not None:
            if c5_bin < best_c:
                best_c, best_h, best_n = c5_bin, h_bin.copy(), n
            # -------------------------------------------------
            # 2) Continuous refinement (hill climb)
            # -------------------------------------------------
            after = time.time()
            remaining2 = budget_s - after
            if remaining2 > 2.0:
                # small allocation for refinement
                iters = min(4000, int(remaining2 * 1500))
                h_ref, c5_ref = hill_climb(h_bin, rng, max_iters=iters, start_step=0.3)
                if c5_ref < best_c:
                    best_c, best_h, best_n = c5_ref, h_ref.copy(), n

            if best_c <= TARGET_C5:
                break  # good enough

        # -------------------------------------------------
        # 3) If MILP failed or not sufficient, fall back to simple random search
        # -------------------------------------------------
        # quick random sampling for the same n
        if time.time() - start_time > budget_s * 0.95:
            break
        trials = 2000
        target_sum = n / 2.0
        for _ in range(trials):
            h_rand = rng.random(n)
            h_rand = h_rand / h_rand.sum() * target_sum
            h_rand = np.clip(h_rand, 0.0, 1.0)
            # simple projection to satisfy box & sum
            total = h_rand.sum()
            diff = target_sum - total
            if abs(diff) > 1e-12:
                # adjust a random coordinate
                i = rng.integers(0, n)
                h_rand[i] = np.clip(h_rand[i] + diff, 0.0, 1.0)
            c_val = compute_c5(h_rand)
            if c_val < best_c:
                best_c, best_h, best_n = c_val, h_rand.copy(), n
            if best_c <= TARGET_C5:
                break
        if best_c <= TARGET_C5:
            break

    # -------------------------------------------------
    # 4) If still nothing, use the provided initial_h_values as last resort
    # -------------------------------------------------
    if best_h is None:
        if "initial_h_values" in globals():
            h0 = np.array(initial_h_values, dtype=float)
            best_h = h0.copy()
            best_n = h0.size
            best_c = compute_c5(best_h)
        else:

            best_n = 60
            best_h = np.full(best_n, 0.5)
            best_c = compute_c5(best_h)

    # Final feasibility enforcement (clip & exact sum)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = make_feasible(best_h)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
