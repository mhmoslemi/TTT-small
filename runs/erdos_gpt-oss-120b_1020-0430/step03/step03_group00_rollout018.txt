def compute_c5_fft(h):
    n = h.size
    g = 1 - h
    N = 1 << ((2*n - 1).bit_length())
    H = np.fft.fft(h, N)
    G = np.fft.fft(g[::-1], N)
    conv = np.fft.ifft(H * G).real
    # conv length N, but only first 2n-1 indices are valid
    c = conv[:2*n-1]
    return np.max(c) * (2.0 / n)
```

def hill_climb(h_start, rng, max_iters, step_scale):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    for it in range(max_iters):
        c_vals = compute_corr_fft(h)
        max_idx = np.argmax(c_vals)
        max_c = c_vals[max_idx] * (2.0 / n)
        k = max_idx - (n - 1)
        # compute gradient: vector of shape n
        grad = np.ones(n)
        if k >= 0:
            grad[:n - k] -= h[k:]
            grad[k:] -= h[:n - k]
        else:
            kk = -k
            grad[:n - kk] -= h[kk:]
            grad[kk:] -= h[:n - kk]

        # identify candidate indices
        pos_idx = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg_idx = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos_idx.size > 0 and neg_idx.size > 0:
            i = pos_idx[np.argmax(grad[pos_idx])]  # largest grad
            j = neg_idx[np.argmin(grad[neg_idx])]  # most negative (smallest)
            max_eps = min(h[i], 1.0 - h[j])
            eps = max_eps * step_scale * rng.random()
            new_h = h.copy()
            new_h[i] -= eps
            new_h[j] += eps
        else:
            # fallback random transfer
            new_h = perturb_random(h, rng, scale=step_scale)

        new_c = compute_c5(new_h)
        if new_c < best_c - 1e-12:
            best_c, best_h = new_c, new_h.copy()
        if new_c < max_c - 1e-12:
            h = new_h
        # step decay
        if (it+1) % (max_iters // 5 + 1) == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c
```

def softmax_obj(x):
    # x is vector of length n
    # compute correlation via FFT same as compute_c5(but not multiply by 2/n)
    n = x.size
    g = 1 - x
    N = 1 << ((2*n - 1).bit_length())
    H = np.fft.fft(x, N)
    G = np.fft.fft(g[::-1], N)
    conv = np.fft.ifft(H * G).real[:2*n - 1]
    # compute scaled by 2/n
    vals = conv * (2.0 / n)
    # softmax: (1/beta) * log sum exp(beta*vals)
    m = np.max(beta * vals)
    return (m + np.log(np.sum(np.exp(beta * vals - m)))) / beta
```

def enforce_sum(h, target, rng):
    """Clip to [0,1] and adjust to make the sum exactly target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # simple water‑filling style repair
    it = 0
    while abs(diff) > 1e-12 and it < 2000:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
            if idx.size == 0:
                break
            add = diff / idx.size
            # avoid overflow beyond 1
            overflow = np.maximum(h[idx] + add - 1.0, 0.0)
            if np.any(overflow):
                h[idx[overflow > 0]] = 1.0
                diff = target - h.sum()
                it += 1
                continue
            h[idx] += add
        else:
            idx = np.where(h > 0.0)[0]
            if idx.size == 0:
                break
            sub = diff / idx.size   # diff is negative
            under = np.maximum(-h[idx] - sub, 0.0)
            if np.any(under):
                h[idx[under > 0]] = 0.0
                diff = target - h.sum()
                it += 1

    return h

def _fft_length(n):
    """Smallest power‑of‑two ≥ 2*n‑1."""
    return 1 << ((2 * n - 1).bit_length())


def compute_corr_fft(h):
    """Full cross‑correlation of h and (1‑h) using FFT, unscaled."""
    n = h.size
    g = 1.0 - h
    N = _fft_length(n)
    H = np.fft.fft(h, N)
    G = np.fft.fft(g[::-1], N)      # reverse g for correlation
    conv = np.fft.ifft(H * G).real[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Discrete C5 = (2/n) * max correlation."""
    n = h.size
    c = compute_corr_fft(h)
    return np.max(c) * (2.0 / n)

def hill_climb(h_start, rng, max_iters=3000, step_scale=0.5):
    """Deterministic gradient‑based local search."""
    h = h_start.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)

    decay_interval = max(1, max_iters // 5)

    for it in range(max_iters):
        # correlation and worst shift
        corr = compute_corr_fft(h)
        idx = np.argmax(corr)
        max_c = corr[idx] * (2.0 / n)
        k = idx - (n - 1)

        # gradient of the worst shift (change of max_c w.r.t. h_i)
        grad = np.ones(n)
        if k >= 0:
            grad[: n - k] -= h[k:]
            grad[k :] -= h[: n - k]
        else:
            kk = -k
            grad[: n - kk] -= h[kk:]
            grad[kk :] -= h[: n - kk]

        # choose best transfer direction
        pos = np.where((grad > 1e-9) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-9) & (h < 1.0 - 1e-12))[0]

        if pos.size and neg.size:
            i = pos[np.argmax(grad[pos])]      # high positive gradient – decrease
            j = neg[np.argmin(grad[neg])]      # most negative gradient – increase
            max_eps = min(h[i], 1.0 - h[j])
            eps = max_eps * step_scale * rng.random()
            new_h = h.copy()
            new_h[i] -= eps
            new_h[j] += eps
        else:
            new_h = perturb_random(h, rng, scale=step_scale)

        if new_c < best_c - 1e-12:
            best_c, best_h = new_c, new_h.copy()
        if new_c < max_c - 1e-12:
            h = new_h

        if (it + 1) % decay_interval == 0:
            step_scale = max(step_scale * 0.7, 1e-4)

    return best_h, best_c

    # Final fine‑grained random tweaks
    # ---------------------------------------------------------
    while time.time() - start_time < budget_s * 0.99:
        new_h = perturb_random(best_h, rng, scale=0.02)
        c_new = compute_c5(new_h)
        if c_new < best_c:
            best_h, best_c = new_h.copy(), c_new

    # ---------------------------------------------------------
    # Clean‑up and return
    # ---------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
