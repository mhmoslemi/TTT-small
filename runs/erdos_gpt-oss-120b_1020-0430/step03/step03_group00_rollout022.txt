def simulated_annealing(h0, rng, time_budget):
    n = h0.size
    best = h0.copy()
    best_c = compute_c5(best)
    cur = h0.copy()
    cur_c = best_c
    start = time.time()
    T0 = 0.005
    Tf = 1e-6
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = T0 * (1 - frac) + Tf * frac
        # choose random pair i,j
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(cur[i], 1.0 - cur[j])
        if max_eps <= 0:
            continue
        # scale epsilon based on temperature ratio
        scale = T / T0
        eps = max_eps * scale * rng.random()
        if eps < 1e-12:
            continue
        new = cur.copy()
        new[i] -= eps
        new[j] += eps
        new_c = compute_c5(new)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = new
            cur_c = new_c
            if new_c < best_c:
                best = new.copy()
                best_c = new_c
    return best, best_c
```

import time
import numpy as np
from scipy.optimize import minimize

# ----- basic helpers -------------------------------------------------
def enforce_sum(h, target, rng):
    """Clip to [0,1] and adjust by transferring mass until sum == target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 2000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:  # need to add mass
            cand = np.where(h < 1.0 - 1e-12)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            add = min(diff, 1.0 - h[i])
            add *= rng.random()
            h[i] += add
            diff -= add
        else:          # need to remove mass
            cand = np.where(h > 1e-12)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            sub = min(-diff, h[i])
            sub *= rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:  # final correction if needed
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    """Uniform random in [0,1] then repair to sum = n/2."""
    h = rng.random(n)
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)

def compute_c5(h):
    """Discrete C5 = max_k ∫ h(x)(1‑h(x+k)) dx."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def perturb_random(h, rng, scale=0.2):

    if i == j:

    if max_eps <= 0:

        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0:
            return h.copy()

        nh = h.copy()
        nh[j] -= eps
        nh[i] += eps
        return nh
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps

    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # repair possible tiny sum error
    rng = np.random.default_rng()
    return enforce_sum(x, CURRENT_TARGET, rng)

# ----- simulated annealing -------------------------------------------
def simulated_annealing(h0, rng, time_budget):
    n = h0.size
    best = h0.copy()
    best_c = compute_c5(best)
    cur = best.copy()
    cur_c = best_c
    start = time.time()
    T0, Tf = 5e-3, 1e-6
    while time.time() - start < time_budget:
        t = time.time() - start
        frac = t / time_budget
        T = T0 * (1 - frac) + Tf * frac
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(cur[i], 1.0 - cur[j])
        if max_eps <= 0:
            max_eps = min(cur[j], 1.0 - cur[i])
            if max_eps <= 0:
                continue
            # swap direction
            i, j = j, i
        scale = T / T0
        eps = max_eps * scale * rng.random()
        if eps < 1e-12:
            continue
        nh = cur.copy()
        nh[i] -= eps
        nh[j] += eps
        new_c = compute_c5(nh)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur, cur_c = nh, new_c
            if new_c < best_c:
                best, best_c = nh.copy(), new_c
    return best, best_c

# ----- main entry ----------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----- initial vector ------------------------------------------------
    if "initial_h_values" in globals():
        h0_low = np.array(initial_h_values, dtype=float)
    else:
        h0_low = random_feasible_vector(180, rng)  # fallback if not provided
    n_low = h0_low.size
    n = n_low * 2  # work on doubled grid (≈360→720)

    # two up‑sampling strategies
    xs_low = (np.arange(n_low) + 0.5) * (2.0 / n_low)
    xs = (np.arange(n) + 0.5) * (2.0 / n)

    h_interp = np.interp(xs, xs_low, h0_low)
    h_repeat = np.repeat(h0_low, 2)

    h_interp = enforce_sum(np.clip(h_interp, 0.0, 1.0), n / 2.0, rng)
    h_repeat = enforce_sum(np.clip(h_repeat, 0.0, 1.0), n / 2.0, rng)

    cand_h = h_interp
    cand_c = compute_c5(cand_h)
    if compute_c5(h_repeat) < cand_c:
        cand_h, cand_c = h_repeat, compute_c5(h_repeat)

    best_h, best_c = cand_h.copy(), cand_c

    # ----- main optimisation ------------------------------------------------
    beta_schedule = [5, 20, 80, 200, 800, 3200, 12800]

    while time.time() - start_time < budget_s * 0.70:
        # random start (biased towards current best)
        if rng.random() < 0.6:
            start_vec = perturb_random(best_h, rng, scale=0.3)
        else:
            start_vec = random_feasible_vector(n, rng)

        # targeted hill‑climb
        h_loc, c_loc = hill_climb_targeted(start_vec, rng, max_iters=1800, start_step=0.5)
        if c_loc < best_c:
            best_h, best_c = h_loc, c_loc

        # SLSQP refinements
        for beta in beta_schedule:
            if time.time() - start_time > budget_s * 0.70:
                break
            h_ref = slsqp_optimize(best_h, beta, max_iter=500)
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_h, best_c = h_ref, c_ref

    # ----- fine‑grained hill climb -----------------------------------------
    h_fine, c_fine = hill_climb_targeted(best_h, rng, max_iters=2500, start_step=0.15)
    if c_fine < best_c:
        best_h, best_c = h_fine, c_fine

    # ----- simulated annealing --------------------------------------------
    remaining = budget_s - (time.time() - start_time) - budget_s * 0.05
    if remaining > 5:
        h_sa, c_sa = simulated_annealing(best_h, rng, time_budget=remaining)
        if c_sa < best_c:
            best_h, best_c = h_sa, c_sa

    # ----- final tiny random tweaks ----------------------------------------
    while time.time() - start_time < budget_s * 0.99:
        new_h = perturb_random(best_h, rng, scale=0.005)
        c_new = compute_c5(new_h)
        if c_new < best_c:
            best_h, best_c = new_h, c_new

    # ----- clean up --------------------------------------------------------
    best_h = enforce_sum(np.clip(best_h, 0.0, 1.0), best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
