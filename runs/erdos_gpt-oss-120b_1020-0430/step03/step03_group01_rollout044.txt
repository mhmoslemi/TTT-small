def slsqp_optimize(h0, beta, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n/2.0
    CURRENT_BETA = beta
    bounds = [(0.0,1.0)]*n
    cons = {'type':'eq','fun':sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter,
                                              'ftol':1e-12,'disp':False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum if drift
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:

            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])

import time
import numpy as np
from scipy.optimize import minimize

# globals for the smooth surrogate used by SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth surrogate of max correlation via log‑sum‑exp."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # safe log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng):
    """Project h onto the simplex Σh = target with 0≤h≤1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        # small equal distribute correction
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
    if abs(diff) > 1e-9:
        for i in range(n):
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    target = n / 2.0
    h = rng.random(n)
    s = h.sum()
    if s > 0:
        h *= target / s
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, target, rng)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2500, step=0.5, trials=5):
    """Simple hill‑climb using pairwise transfers."""
    h = h_start.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(trials):
            new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
                improved = True
                break
        if not improved and rng.random() < 0.02:
            new_h = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
                improved = True
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)
    return h, best_c


def slsqp_optimize(h0, beta, max_iter=400):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # -------------------------------------------------------------
    # 1. Base dimension and initial vector
    # -------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 180
        h0 = random_feasible_vector(n, rng)

    # -------------------------------------------------------------
    # 2. Build a diverse pool of candidates
    # -------------------------------------------------------------
    candidates = [h0]
    xs = (np.arange(n) + 0.5) * (2.0 / n)

    for f in (1, 2, 3, 5, 7):
        s = 0.5 + 0.5 * np.sin(np.pi * f * xs)
        candidates.append(enforce_sum(np.clip(s, 0.0, 1.0), n / 2.0, rng))
        c = 0.5 + 0.5 * np.cos(np.pi * f * xs)
        candidates.append(enforce_sum(np.clip(c, 0.0, 1.0), n / 2.0, rng))

    candidates.append(enforce_sum(xs / 2.0, n / 2.0, rng))
    candidates.append(enforce_sum(1.0 - xs / 2.0, n / 2.0, rng))

    mask = np.zeros(n)
    mask[: n // 2] = 1.0
    candidates.append(enforce_sum(mask, n / 2.0, rng))

    mask2 = np.zeros(n)
    start_idx = n // 4
    mask2[start_idx : start_idx + n // 2] = 1.0
    candidates.append(enforce_sum(mask2, n / 2.0, rng))

    mask3 = (np.arange(n) % 2).astype(float)
    candidates.append(enforce_sum(mask3, n / 2.0, rng))

    for _ in range(4):
        candidates.append(random_feasible_vector(n, rng))

    # -------------------------------------------------------------
    # 3. Pick best initial candidate
    # -------------------------------------------------------------
    best_h = None
    best_c = np.inf
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # -------------------------------------------------------------
    # 4. Main optimisation loop (hill‑climb + SLSQP continuation)
    # -------------------------------------------------------------
    beta_seq = [5, 20, 80, 200, 500, 1000, 2000]
    time_limit_main = start_time + 0.70 * budget_s

    while time.time() < time_limit_main:
        # occasional random restart
        if rng.random() < 0.6:
            start_vec = perturb_h(best_h, rng, scale=0.4)
        else:
            start_vec = random_feasible_vector(n, rng)

        # hill‑climb refinement
        h_hc, c_hc = hill_climb(start_vec, rng, max_iters=2500, step=0.5, trials=6)
        if c_hc < best_c:
            best_h, best_c = h_hc, c_hc

        # SLSQP continuation with increasing β
        for beta in beta_seq:
            if time.time() >= time_limit_main:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, max_iter=400)
            except Exception:
                continue
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_h, best_c = h_ref, c_ref

    # -------------------------------------------------------------
    # 5. Upsampling & fine optimisation (if time permits)
    # -------------------------------------------------------------
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 0.20 * budget_s:
        target_n = min(720, max(360, n * 2))
        if target_n != n:
            old_x = (np.arange(n) + 0.5) * (2.0 / n)
            new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
            h_up = np.interp(new_x, old_x, best_h)
            n = target_n
            h_up = enforce_sum(h_up, n / 2.0, rng)

            # hill‑climb on the finer grid
            h_up, c_up = hill_climb(h_up, rng, max_iters=4000, step=0.3, trials=5)
            if c_up < best_c:
                best_h, best_c = h_up, c_up

            # SLSQP refinement on fine grid
            for beta in beta_seq:
                if time.time() - start_time >= budget_s:
                    break
                try:
                    h_up = slsqp_optimize(best_h, beta, max_iter=400)
                except Exception:
                    continue
                c_up = compute_c5(h_up)
                if c_up < best_c:
                    best_h, best_c = h_up, c_up

    # -------------------------------------------------------------
    # 6. Final clean‑up
    # -------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
