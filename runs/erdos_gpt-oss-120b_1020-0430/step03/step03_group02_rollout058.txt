def smoothed_obj_and_grad(h, beta):
    c = np.correlate(h, 1 - h, mode='full')
    # length m = 2n-1
    # compute weights = softmax(beta*c)
    maxc = np.max(beta*c)
    exp = np.exp(beta*c - maxc)
    sumexp = np.sum(exp)
    w = exp / sumexp
    # compute gradient
    n = h.size
    grad = np.zeros_like(h, dtype=float)
    # contributions from each shift i (index shift = i-(n-1))
    for idx, wi in enumerate(w):
        shift = idx - (n - 1)
        if shift >= 0:
            # term (1 - h[p+shift]) for p where p+shift<n => p in [0, n-1-shift]
            # and -h[p-shift] for p where p-shift>=0 => p in [shift, n-1]
            # We'll vectorize
            # part1: for p in 0..n-1-shift: add wi*(1 - h[p+shift])
            # part2: for p in shift..n-1: add wi*(-h[p-shift])
            # Instead compute using slices
            # part1
            grad[:n-shift] += wi * (1 - h[shift:])
            # part2
            grad[shift:] += wi * (-h[:n-shift])
        else:
            # shift negative: shift = -s where s>0.
            s = -shift
            # For shift negative, we have p+shift = p-s (if p-s>=0)
            # Use same formula: (1 - h[p+shift]) where p+shift>=0: p >= s
            # and -h[p-shift] where p-shift<n: p-s < n => p < n+s
            # So treat symmetrically:
            # part1: for p in s..n-1: add wi*(1 - h[p-s])
            # part2: for p in 0..n-1-s: add wi*(-h[p+s])
            grad[s:] += wi * (1 - h[:n-s])
            grad[:n-s] += wi * (-h[s:])
    # grad computed
    obj = (np.log(sumexp) + maxc) / beta  # since maxc = max(beta*c) = beta*max(c) but we used subtract shift; Actually we did exp(beta*c - maxc), so maxc = max(beta*c). So logsumexp = maxc + log(sumexp). Then divide by beta.
    return obj, grad

def project_feasible(h, target, rng):
    # clip
    h = np.clip(h, 0., 1.)
    diff = target - h.sum()
    # adjust with random indices until diff approx 0
    # simple while loop up to some max tries
    tries = 0
    while abs(diff) > 1e-12 and tries < 10000:
        i = rng.integers(0, h.size)
        if diff > 0:
            inc = min(diff, 1.0 - h[i])
            h[i] += inc
            diff -= inc
        else:
            dec = min(-diff, h[i])
            h[i] -= dec
            diff += dec
        tries += 1
    return h

import numpy as np

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def project_feasible(h, target, rng):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # simple water-filling with random indices
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        i = rng.integers(0, h.size)
        if diff > 0:
            inc = min(diff, 1.0 - h[i])
            h[i] += inc
            diff -= inc
        else:
            dec = min(-diff, h[i])
            h[i] -= dec
            diff += dec
        it += 1
    # If still not exact due to rounding, adjust using deterministic fill
    if abs(diff) > 1e-12:
        # add diff uniformly across feasible entries
        # compute indices that can be increased/decreased
        if diff > 0:
            idx = np.where(h < 1.0)[0]
            if idx.size > 0:
                inc = diff / idx.size
                inc = np.minimum(inc, 1.0 - h[idx])
                h[idx] += inc
        else:
            idx = np.where(h > 0.0)[0]
            if idx.size > 0:
                dec = -diff / idx.size
                dec = np.minimum(dec, h[idx])
                h[idx] -= dec
    return h

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)

        grad[k:]       -= h[: n - k]    # subtract h_{i‑k}

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()

        return h.copy()

    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h_start, rng, max_iters=2000, step_scale=0.5):
    h = h_start.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)

def gradient_refine(h, rng, step=0.02, max_iter=8):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            # gradient of O_k
            g[: n - k] += 1.0 - h[k:]
            g[k:]       -= h[: n - k]
        # update

    return h

def smoothed_obj_and_grad(h, beta):
    # compute correlation
    c = np.correlate(h, 1 - h, mode='full')
    beta_c = beta * c
    max_bc = np.max(beta_c)
    exp_shifted = np.exp(beta_c - max_bc)
    sumexp = np.exp(max_bc) * np.sum(exp_shifted)  # but we can keep as sumexp = np.sum(np.exp(beta_c - max_bc))
    # Actually simpler: compute weighted average directly
    w = exp_shifted / np.sum(exp_shifted)
    n = h.size
    grad = np.zeros_like(h, dtype=float)
    # Compute gradient contributions
    # loop over each shift
    # For performance we can vectorize with slicing inside loop
    for idx, wi in enumerate(w):
        shift = idx - (n - 1)
        if shift >= 0:
            s = shift
            # part1: grad[:n-s] += wi * (1 - h[s:])
            grad[:n-s] += wi * (1 - h[s:])

        else:

            grad[s:] += wi * (1 - h[:n-s])
            # part2: grad[:n-s] += wi * (-h[s:])
            grad[:n-s] += wi * (-h[s:])
    obj = (max_bc + np.log(np.sum(exp_shifted))) / beta
    return obj, grad

def smoothed_pg_optimize(h0, beta, rng, max_iter=200):

    return h

            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target, rng)
    return h


# ----------------------------------------------------------------------
# smooth (log‑sum‑exp) objective and its gradient
# ----------------------------------------------------------------------
def smoothed_obj_and_grad(h: np.ndarray, beta: float) -> tuple:
    """Return soft‑max C5 and its gradient."""
    c = np.correlate(h, 1.0 - h, mode="full")
    beta_c = beta * c
    max_bc = np.max(beta_c)
    exp_shifted = np.exp(beta_c - max_bc)
    w = exp_shifted / np.sum(exp_shifted)                     # soft‑max weights
    n = h.size
    grad = np.zeros_like(h, dtype=float)

    for idx, wi in enumerate(w):
        shift = idx - (n - 1)
        if shift >= 0:
            s = shift
            grad[: n - s] += wi * (1.0 - h[s:])                # (1‑h_{i+shift})
            grad[s:]      += wi * (-h[: n - s])                # -h_{i‑shift}
        else:
            s = -shift
            grad[s:]      += wi * (1.0 - h[: n - s])           # (1‑h_{i‑s})
            grad[: n - s] += wi * (-h[s:])                     # -h_{i+s}
    obj = (max_bc + np.log(np.sum(exp_shifted))) / beta
    return obj, grad


def smoothed_pg_optimize(h0: np.ndarray, beta: float,
                         rng: np.random.Generator, max_iter: int = 200) -> np.ndarray:
    """Projected gradient descent on the smooth objective."""
    h = h0.copy()
    target = h.size / 2.0
    step = 0.2
    for _ in range(max_iter):
        obj, grad = smoothed_obj_and_grad(h, beta)
        h_new = h - step * grad
        h_new = project_feasible(h_new, target, rng)
        new_obj, _ = smoothed_obj_and_grad(h_new, beta)
        if new_obj < obj:
            h = h_new
            step = min(step * 1.1, 1.0)
        else:
            step *= 0.5
            if step < 1e-6:
                break
    return h


# ----------------------------------------------------------------------
# main driver
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs) -> tuple:
    rng = np.random.default_rng(seed)
    start = time.time()

    best_h = None
    best_c = np.inf
    best_n = None

    # ------------------------------------------------------------------
    # start from parent construction if supplied
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0.copy(), c0, h0.size

    # ------------------------------------------------------------------
    # candidate discretisations
    # ------------------------------------------------------------------
    candidates = [84, 96, 108, 120, 144, 168, 192, 216, 240]
    if best_n is not None:
        if best_n not in candidates:
            candidates = [best_n] + candidates
        else:
            candidates = [best_n] + [n for n in candidates if n != best_n]

    # ------------------------------------------------------------------
    # main optimisation loop over different n
    # ------------------------------------------------------------------
    for n in candidates:
        if time.time() - start > budget_s * 0.94:
            break

        restarts = 6 if n <= 120 else 3
        for r in range(restarts):
            if time.time() - start > budget_s * 0.94:
                break

            if r == 0 and best_h is not None and best_n == n:
                h = best_h.copy()
            elif r == 1 and best_h is not None:
                x_old = np.linspace(0, 2, best_h.size, endpoint=False)
                x_new = np.linspace(0, 2, n, endpoint=False)
                h = np.interp(x_new, x_old, best_h)
                h = np.clip(h, 0.0, 1.0)
                h = enforce_sum(h, rng)
            else:
                h = random_feasible_vector(n, rng)

            # quick hill‑climb to get a decent starting point
            h, _ = guided_hill_climb(h, rng, max_iters=1200, step_scale=0.5)

            # smooth optimisation with increasing β
            for beta in (1, 5, 20, 80, 200, 500, 1000, 2000, 5000):
                if time.time() - start > budget_s * 0.96:
                    break
                iters = 150 if beta < 200 else 80
                h = smoothed_pg_optimize(h, beta, rng, max_iter=iters)
                c = compute_c5(h)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n

            # final fine‑grained refinement
            h = gradient_refine(h, rng, step=0.01, max_iter=12)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

    # ------------------------------------------------------------------
    # random tiny guided perturbations while time permits
    # ------------------------------------------------------------------
    while time.time() - start < budget_s * 0.999 and best_h is not None:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # ------------------------------------------------------------------
    # optional up‑sampling refinement (if we still have time)
    # ------------------------------------------------------------------
    if best_h is not None and time.time() - start < budget_s * 0.95:
        up_n = best_n * 2
        if up_n <= 320:
            up_h = np.repeat(best_h, 2)
            up_h, _ = guided_hill_climb(up_h, rng, max_iters=1500, step_scale=0.5)
            for beta in (5, 20, 80, 200, 500, 1000, 2000, 5000):
                if time.time() - start > budget_s * 0.99:
                    break
                iters = 120 if beta < 200 else 60
                up_h = smoothed_pg_optimize(up_h, beta, rng, max_iter=iters)
            up_h = gradient_refine(up_h, rng, step=0.01, max_iter=12)
            up_c = compute_c5(up_h)
            if up_c < best_c:
                best_h, best_c, best_n = up_h.copy(), up_c, up_n

    # ------------------------------------------------------------------
    # final feasibility check and return
    # ------------------------------------------------------------------
    if best_h is None:
        best_n = 96
        best_h = np.full(best_n, 0.5)

    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
