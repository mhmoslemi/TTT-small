import time
import numpy as np

# --------------------------------------------------------------
# Helper utilities
# --------------------------------------------------------------
def enforce_sum(h, target, rng):
    """Adjust h in-place so that sum(h) == target while keeping 0 <= h <= 1."""
    diff = target - h.sum()
    n = h.size
    # simple random adjustments
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            candidates = np.where(h < 1.0)[0]
            if candidates.size == 0:
                break
            i = rng.choice(candidates)
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            candidates = np.where(h > 0.0)[0]
            if candidates.size == 0:
                break
            i = rng.choice(candidates)
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    h = rng.random(n)
    # project onto hyperplane with simple iterative clipping
    target = n / 2.0
    # initial scaling
    scale = target / h.sum()
    h *= scale
    # clip and fix residual
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h = enforce_sum(h, target, rng)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def correlate_full(h):
    """Full linear correlation of h with (1‑h) using FFT (zero‑padding)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())  # next power of two
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Discrete C₅ for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)


def hill_climb(h_start, rng, max_iters=2000, step_start=0.4, deadline=None):
    """Greedy hill‑climb: only accept improvements."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step_start
    decay_interval = max(1, max_iters // 5)
    for it in range(max_iters):
        if deadline is not None and time.time() >= deadline:
            break
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-6)
    return best_h, best_c


def simulated_annealing(start_h, rng, deadline, T0=0.02, Tf=1e-6, max_iters=2000):
    """Simple exponential‑cooling SA."""
    cur_h = start_h.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        new_h = perturb_h(cur_h, rng, scale=0.35)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
    return best_h, best_c


def smooth_h(h):
    """Apply a small three‑point smoothing kernel."""
    kernel = np.array([0.25, 0.5, 0.25])
    sm = np.convolve(h, kernel, mode='same')
    sm = np.clip(sm, 0.0, 1.0)
    return sm


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    overall_deadline = start_time + budget_s * 0.98  # reserve a bit for final cleanup

    # ---- 1. Base vector -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 180
        h0 = enforce_sum(np.full(n, 0.5), n / 2.0, rng)

    # ---- 2. Small pool of diverse seeds --------------------------------
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    pool = [h0]
    for f in (0.5, 1.0, 1.5, 2.0):
        pool.append(enforce_sum(np.clip(0.5 + 0.5 * np.sin(np.pi * f * xs), 0, 1), n / 2.0, rng))
        pool.append(enforce_sum(np.clip(0.5 + 0.5 * np.cos(np.pi * f * xs), 0, 1), n / 2.0, rng))
    # a few random vectors
    for _ in range(800):
        pool.append(random_feasible_vector(n, rng))

    # ---- 3. Pick the best initial candidate -----------------------------
    best_h = None
    best_c = float("inf")
    for h in pool:
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # ---- 4. Main refinement loop ----------------------------------------
    while time.time() < overall_deadline:
        # start from a perturbed best or a fresh random vector
        if rng.random() < 0.6:
            start = perturb_h(best_h, rng, scale=0.3)
        else:
            start = random_feasible_vector(n, rng)

        # allocate a budgeted number of iterations based on remaining time
        remaining = overall_deadline - time.time()
        if remaining <= 0:
            break
        max_iters = int(min(6000, max(500, remaining * 3000)))  # rough 3000 iters per sec

        # greedy hill‑climb
        h_opt, c_opt = hill_climb(start, rng, max_iters=max_iters,
                                  step_start=0.4,
                                  deadline=overall_deadline)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()

        # occasional simulated annealing
        if rng.random() < 0.2:
            cand_h, cand_c = simulated_annealing(start, rng,
                                                 deadline=overall_deadline,
                                                 T0=0.03, Tf=1e-6,
                                                 max_iters=1500)
            if cand_c < best_c:
                best_c, best_h = cand_c, cand_h.copy()

        # occasional smoothing step
        if rng.random() < 0.1:
            sm = smooth_h(best_h)
            sm = enforce_sum(sm, n / 2.0, rng)
            c_sm = compute_c5(sm)
            if c_sm < best_c:
                best_c, best_h = c_sm, sm.copy()

    # ---- 5. Upsample to a finer grid if time permits --------------------
    up_deadline = start_time + budget_s * 0.99
    if time.time() < up_deadline and n * 2 <= 720:
        n_up = n * 2
        xs_old = (np.arange(n) + 0.5) * (2.0 / n)
        xs_new = (np.arange(n_up) + 0.5) * (2.0 / n_up)
        h_up = np.interp(xs_new, xs_old, best_h)
        h_up = np.clip(h_up, 0.0, 1.0)
        h_up = enforce_sum(h_up, n_up / 2.0, rng)

        # refine on the finer grid
        while time.time() < up_deadline:
            # compute a modest iteration budget
            rem = up_deadline - time.time()
            if rem <= 0:
                break
            max_iters = int(min(5000, max(400, rem * 2500)))
            h_opt, c_opt = hill_climb(h_up, rng, max_iters=max_iters,
                                      step_start=0.2,
                                      deadline=up_deadline)
            if c_opt < best_c:
                best_c, best_h, n = c_opt, h_opt.copy(), n_up
                h_up = h_opt

            # smoothing on the fine grid
            if rng.random() < 0.15:
                sm = smooth_h(best_h)
                sm = enforce_sum(sm, n / 2.0, rng)
                c_sm = compute_c5(sm)
                if c_sm < best_c:
                    best_c, best_h = c_sm, sm.copy()
                    h_up = best_h

    # ---- 6. Final clean‑up ------------------------------------------------
    final_h = np.clip(best_h, 0.0, 1.0)
    final_h = enforce_sum(final_h, final_h.size / 2.0, rng)
    final_c = compute_c5(final_h)

    return final_h.tolist(), float(final_c), int(final_h.size)
