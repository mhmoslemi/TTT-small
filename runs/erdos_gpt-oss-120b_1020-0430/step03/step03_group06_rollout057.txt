def targeted_move(h, k, rng, scale=0.5):
    n = h.size
    # compute overlapping range
    if k >= 0:
        i_start = 0
        i_end = n - k  # exclusive
        offset = k
    else:
        i_start = -k
        i_end = n
        offset = k  # negative

    # compute contributions
    # Use vectorized:
    i_range = np.arange(i_start, i_end)
    contributions = h[i_range] * (1.0 - h[i_range + offset])

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def enforce_sum(h, target, rng):
    """Adjust h so that Σh = target while 0≤h≤1 (simple projection)."""
    diff = target - h.sum()
    n = h.size
    # simple iterative redistribution
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-12:
            continue
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Random binary‑fractional feasible vector (sum = n/2)."""
    h = rng.random(n)
    h = enforce_sum(h, n / 2.0, rng)
    return h


def compute_c5(h):
    """Exact discrete C₅ for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def max_corr_and_shift(h):
    """Return (max_corr, shift) where shift = idx-(n-1)."""
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr)
    maxv = corr[idx]
    shift = idx - (h.size - 1)
    return float(maxv), int(shift)


def perturb_h(h, rng, scale=0.5):
    """Transfer a small amount between a random 1‑slot and a 0‑slot."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    while j == i:
        j = rng.integers(0, n)
    max_pos = min(h[i], 1.0 - h[j])
    max_neg = min(h[j], 1.0 - h[i])
    if max_pos + max_neg == 0:
        return h.copy()
    eps = (rng.random() * (max_pos + max_neg)) - max_neg
    eps *= scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    np.clip(nh, 0.0, 1.0, out=nh)
    return nh


def targeted_move(h, k, rng, scale=0.5):
    """Move a little mass from a position that hurts the worst shift to one that helps."""
    n = h.size
    if k >= 0:
        i_start, i_end, off = 0, n - k, k
    else:
        i_start, i_end, off = -k, n, k
    rng_inds = np.arange(i_start, i_end)

    # contributions for the worst shift
    contrib = h[rng_inds] * (1.0 - h[rng_inds + off])
    i_max = rng_inds[np.argmax(contrib)]

    # candidate destinations: where we can increase h and contribution is smallest
    j_start, j_end = i_start, i_end
    j_inds = np.arange(j_start, j_end)
    possible = h[j_inds] < 1.0
    if not np.any(possible):
        return h
    loss = h[j_inds] * (1.0 - h[j_inds + off])
    loss[~possible] = np.inf
    # avoid moving back to the same index
    if j_start <= i_max < j_end:
        loss[i_max - j_start] = np.inf
    j_min = j_inds[np.argmin(loss)]

    max_transfer = min(h[i_max], 1.0 - h[j_min])
    if max_transfer <= 0:
        return h
    eps = rng.random() * max_transfer * scale
    nh = h.copy()
    nh[i_max] -= eps
    nh[j_min] += eps
    np.clip(nh, 0.0, 1.0, out=nh)
    return nh
