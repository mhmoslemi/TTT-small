def random_feasible_vector(n, rng):
    """Random vector in [0,1]^n with sum = n/2."""
    v = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(v[i], 1.0 - v[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        v[i] -= eps
        v[j] += eps
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    # simple projection onto bounded simplex
    return project_to_feasible(v, target)


def project_to_feasible(v, target, lower=0.0, upper=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lower<=x<=upper, sum(x)=target}."""
    lo, hi = -2.0, 2.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, lower, upper)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return proj


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def correlate_full(h):
    """Linear correlation of h with (1-h) using FFT."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)

# Softmax‑gradient utilities
# ----------------------------------------------------------------------
def gradient_softmax(h, beta):
    """Gradient of (1/β)·log‑sum‑exp(β·corr) w.r.t. h."""
    n = h.size
    offset = n - 1
    conv = correlate_full(h)                # length 2n‑1
    m = conv.max()
    w = np.exp(beta * (conv - m))
    w /= w.sum()

    # term1 via prefix sums
    w_pos = w[offset:]                     # ℓ ≥ 0
    w_neg = w[:offset]                     # ℓ < 0
    cum_pos = np.cumsum(w_pos)             # length n
    w_neg_rev = w_neg[::-1]                # reverse ℓ < 0 order
    cum_neg_rev = np.cumsum(w_neg_rev)     # length n‑1
    idx = np.arange(n)
    sum2 = np.empty(n)
    sum2[:-1] = cum_neg_rev[n - 2 - idx[:-1]]
    sum2[-1] = 0.0
    term1 = cum_pos + sum2

    # term2 via convolution w*h and w_rev*h
    L = 1 << ((len(w) + len(h) - 1).bit_length())
    w_fft = np.fft.rfft(w, L)
    h_fft = np.fft.rfft(h, L)
    conv_wh = np.fft.irfft(w_fft * h_fft, L)[: len(w) + len(h) - 1]
    part1 = conv_wh[offset: offset + n]

    w_rev = w[::-1]
    w_rev_fft = np.fft.rfft(w_rev, L)
    conv_rev = np.fft.irfft(w_rev_fft * h_fft, L)[: len(w) + len(h) - 1]
    part2 = conv_rev[offset: offset + n]

    term2 = part1 + part2
    grad = term1 - term2
    return grad, conv


def projected_gradient_descent(
    h0, target, beta_seq, steps_per_beta, step_size, rng
):
    """Projected gradient descent on softmax‑smoothed max‑overlap."""
    h = np.clip(h0, 0.0, 1.0)
    h = project_to_feasible(h, target)
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c

    for beta in beta_seq:
        for _ in range(steps_per_beta):
            grad, _ = gradient_softmax(h, beta)
            h_new = project_to_feasible(h - step_size * grad, target)
            new_c = compute_c5(h_new)
            if new_c < cur_c:
                h, cur_c = h_new, new_c
            if new_c < best_c:
                best_h, best_c = h_new.copy(), new_c
    return best_h, best_c

    h = h_start.copy()

    step = start_step
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


# ----------------------------------------------------------------------
# SLSQP refinement (smooth surrogate)
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    conv = correlate_full(x)
    bc = CURRENT_BETA * conv
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """SLSQP on the smooth max‑overlap surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny post‑processing to restore exact sum
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ----------------------------------------------------------------------
# Resampling utilities
# ----------------------------------------------------------------------
def resample_h(h, new_n):
    """Linear interpolation resampling of h onto new_n points."""
    old_n = h.size
    xs_old = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    xs_new = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    return np.interp(xs_new, xs_old, h)


# ----------------------------------------------------------------------
# Main optimisation driver
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.94  # keep a safety margin

    # --------------------------------------------------------------
    # Determine base dimension and initial vector
    # --------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        target = n / 2.0
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_to_feasible(h0, target)
    else:
        n = 360
        target = n / 2.0
        h0 = random_feasible_vector(n, rng)

    best_h, best_c = h0.copy(), compute_c5(h0)

    # --------------------------------------------------------------
    # Coarse optimisation on a reduced grid
    # --------------------------------------------------------------
    n_small = max(60, n // 6)            # 60‑360 → 60‑60
    target_small = n_small / 2.0
    # build a few small‑scale starting points
    small_starts = [
        resample_h(best_h, n_small),
        random_feasible_vector(n_small, rng),
        random_feasible_vector(n_small, rng),
    ]
    if "initial_h_values" in globals():
        small_starts.append(resample_h(np.array(initial_h_values, dtype=float), n_small))

    beta_seq_small = [1.0, 5.0, 20.0, 80.0]
    steps_small = 200

    small_best = None
    small_best_c = float("inf")
    for h_start in small_starts:
        h_start = project_to_feasible(h_start, target_small)
        h_opt, c_opt = projected_gradient_descent(
            h_start, target_small, beta_seq_small, steps_small, step_size=0.05, rng=rng
        )
        if c_opt < small_best_c:
            small_best_c, small_best = c_opt, h_opt.copy()
        # one greedy hill‑climb refinement
        h_hc, c_hc = hill_climb(h_opt, rng, max_iters=1500, start_step=0.3)
        if c_hc < small_best_c:
            small_best_c, small_best = c_hc, h_hc.copy()
        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # Upsample to full size and refine
    # --------------------------------------------------------------
    if small_best is not None:
        h_up = resample_h(small_best, n)
        h_up = project_to_feasible(h_up, target)
        up_c = compute_c5(h_up)
        if up_c < best_c:
            best_h, best_c = h_up.copy(), up_c

        # further gradient descent on full grid
        beta_seq_big = [5.0, 20.0, 80.0, 200.0, 800.0]
        h_up_opt, c_up_opt = projected_gradient_descent(
            best_h, target, beta_seq_big, steps_per_beta=300, step_size=0.02, rng=rng
        )
        if c_up_opt < best_c:
            best_h, best_c = h_up_opt.copy(), c_up_opt

    # --------------------------------------------------------------
    # Intensive local refinement loop (hill‑climb + SLSQP)
    # --------------------------------------------------------------
    while time.time() < deadline:
        # start from a perturbed best or a fresh random point
        if rng.random() < 0.65:
            start = perturb_h(best_h, rng, scale=0.25)
        else:
            start = random_feasible_vector(n, rng)

        # hill‑climb
        h_hc, c_hc = hill_climb(start, rng, max_iters=2500, start_step=0.35)
        if c_hc < best_c:
            best_h, best_c = h_hc.copy(), c_hc

        # short SLSQP refinements with increasing β
        for beta in [200.0, 800.0, 2000.0]:
            if time.time() >= deadline:
                break
            try:
                h_s = slsqp_optimize(best_h, beta, max_iter=250)
                c_s = compute_c5(h_s)
                if c_s < best_c:
                    best_h, best_c = h_s.copy(), c_s
            except Exception:
                continue

    # --------------------------------------------------------------
    # Final clean‑up
    # --------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_feasible(best_h, target)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
