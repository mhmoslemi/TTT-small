def compute_c5_fast(h):
    n = h.shape[0]
    g = 1 - h
    size = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, size)
    G = np.fft.rfft(g, size)
    corr = np.fft.irfft(H * np.conj(G), size)  # length size
    # Take first 2n-1 elements (linear correlation)
    corr = corr[: 2*n - 1]
    # max correlation
    max_corr = corr.max()
    return float(max_corr * (2.0 / n))

def gradient_refine_fast(h, rng, step=0.02, max_iter=30):
    n = h.size
    target = n/2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1-h, mode='full')
        idx = np.argmax(corr) - (n-1)
        k = idx
        g = np.zeros_like(h)
        if k >= 0:
            upper = n - k
            if upper > 0:
                g[:upper] += 1 - h[k:]
                g[k:] -= h[:upper]
        else:
            kp = -k
            upper = n - kp
            if upper > 0:
                g[kp:] += 1 - h[:upper]
                g[:upper] -= h[kp:]
        # take step

def simulated_annealing_fast(h0, rng, deadline):
    h = h0.copy()
    cur_c = compute_c5_fast(h)
    best_h = h.copy()
    best_c = cur_c
    # initial temperature as fraction of cur_c
    T0 = max(0.01 * cur_c, 1e-4)
    it = 0
    while time.time() < deadline:
        # adapt step decreasing
        step = max(0.5 * (0.999 ** it), 1e-5)
        new_h = perturb_h(h, rng, scale=step)
        new_c = compute_c5_fast(new_h)
        if new_c < cur_c:
            accept = True
        else:
            T = T0 * (0.999 ** it) + 1e-12
            accept = rng.random() < np.exp(-(new_c - cur_c) / T)
        if accept:
            h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, h.copy()
        it += 1
        if it % 1000 == 0:

# ---------- basic utilities ----------
def compute_c5_fast(h: np.ndarray) -> float:
    """Fast C5 using FFT (linear correlation)."""
    n = h.shape[0]
    g = 1.0 - h
    # next power of two >= 2*n-1
    size = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, size)
    G = np.fft.rfft(g, size)
    corr = np.fft.irfft(H * np.conj(G), size)
    corr = corr[: 2 * n - 1]          # linear correlation
    return float(corr.max() * (2.0 / n))


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector in [0,1] with exact sum n/2."""
    h = rng.random(n)
    target = n / 2.0
    scale = target / h.sum()
    h *= scale
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h


def make_feasible(h: np.ndarray, n: int, rng: np.random.Generator) -> np.ndarray:
    """Clip to [0,1] and adjust sum to n/2."""
    np.clip(h, 0.0, 1.0, out=h)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):

            h[i] += add
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_fast(h_start: np.ndarray, rng: np.random.Generator,
                    max_iters: int = 2000, start_step: float = 0.5):
    """Deterministic hill‑climb using fast C5 evaluation."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5_fast(h)
    step = start_step
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5_fast(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-5)
    return best_h, best_c


def gradient_refine_fast(h: np.ndarray, rng: np.random.Generator,
                         step: float = 0.02, max_iter: int = 30):
    """Project gradient of the worst shift, with sum correction."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = idx
        g = np.zeros_like(h)
        if k >= 0:
            up = n - k
            if up > 0:
                g[:up] += 1.0 - h[k:]
                g[k:] -= h[:up]
        else:
            kp = -k
            up = n - kp
            if up > 0:
                g[kp:] += 1.0 - h[:up]
                g[:up] -= h[kp:]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


def simulated_annealing_fast(h0: np.ndarray, rng: np.random.Generator,
                             deadline: float):
    """Simple annealing respecting the sum constraint."""
    h = h0.copy()
    cur_c = compute_c5_fast(h)
    best_h = h.copy()
    best_c = cur_c
    T0 = max(0.01 * cur_c, 1e-4)
    it = 0
    while time.time() < deadline:
        step = max(0.5 * (0.999 ** it), 1e-5)
        cand_h = perturb_h(h, rng, scale=step)
        cand_c = compute_c5_fast(cand_h)
        if cand_c < cur_c:
            accept = True
        else:
            T = T0 * (0.999 ** it) + 1e-12
            accept = rng.random() < np.exp(-(cand_c - cur_c) / T)
        if accept:
            h, cur_c = cand_h, cand_c
            if cur_c < best_c:
                best_c, best_h = cur_c, h.copy()
        it += 1
        if it % 2000 == 0 and time.time() >= deadline:
            break
    return best_h, best_c
