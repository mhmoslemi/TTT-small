def compute_c5_fast(h):
    n = h.size
    L = 2*n - 1
    # next power of 2
    m = 1 << (L-1).bit_length()
    # pad
    h_pad = np.zeros(m)
    y_rev_pad = np.zeros(m)
    h_pad[:n] = h
    y_rev_pad[:n] = (1 - h)[::-1]
    H = np.fft.rfft(h_pad)
    Y = np.fft.rfft(y_rev_pad)
    conv = np.fft.irfft(H * Y, n=m)[:L]
    # conv is correlation as np.correlate returns.
    max_corr = np.max(conv)
    return max_corr * (2.0 / n)
```

This is O(m log m). For n up to 2000, still fast.

def simulated_annealing(h0, rng, timelimit):
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    # params
    T0 = 0.01
    Tf = 1e-5
    # number of iterations approx
    # compute max iterations: while time < limit
    while time.time() - start < timelimit:
        elapsed = time.time() - start
        frac = elapsed / timelimit
        T = T0 * (Tf/T0)**frac  # exponential cooling
        # propose move
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j: continue
        max_eps = min(h[i], 1 - h[j])
        if max_eps <= 0: continue
        eps = rng.random() * max_eps * 0.2  # 20% of max
        # also allow negative transfer??? Actually we always move from i to j
        # but we could also move opposite direction via different i/j
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c_new = compute_c5(new_h)
        delta = c_new - best_c
        if c_new < best_c:
            best_c = c_new
            best_h = new_h.copy()
        # acceptance condition: compare to current c?
        cur_c = compute_c5(h)
        dcur = c_new - cur_c
        if dcur < 0 or rng.random() < np.exp(-dcur / max(T, 1e-12)):
            h = new_h
        # optionally random restart if stagnation
    return best_h, best_c

def correlate_fft(a, b):
    n = a.size
    m = b.size
    L = n + m - 1
    size = 1 << (L - 1).bit_length()
    A = np.zeros(size)
    B = np.zeros(size)
    A[:n] = a
    B[:m] = b[::-1]   # reverse b for correlation
    FA = np.fft.rfft(A)
    FB = np.fft.rfft(B)
    conv = np.fft.irfft(FA * FB, n=size)[:L]
    return conv
```

        while time.time() - start < time_limit:

            # compute softmax weights

            w = exp_term / exp_term.sum()

import time
import numpy as np
from scipy.optimize import minimize

# ----- Global placeholders for the soft‑max surrogate used by SLSQP -----
CURRENT_TARGET = None
CURRENT_BETA = None


# ----- Correlation via FFT (full linear correlation) -----
def correlate_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear correlation of a and b using FFT (no padding beyond needed)."""
    n, m = a.size, b.size
    L = n + m - 1
    size = 1 << (L - 1).bit_length()  # next power of two
    A = np.zeros(size, dtype=np.float64)
    B = np.zeros(size, dtype=np.float64)
    A[:n] = a
    B[:m] = b[::-1]          # reverse for correlation
    FA = np.fft.rfft(A)
    FB = np.fft.rfft(B)
    conv = np.fft.irfft(FA * FB, n=size)[:L]
    return conv


# ----- Exact C5 for a given step function (using the fast correlation) -----
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function h on [0,2]."""
    n = h.size
    corr = correlate_fft(h, 1.0 - h)
    return np.max(corr) * (2.0 / n)


# ----- Projection onto {0 ≤ h ≤ 1, Σ h = target} via bisection on a shift -----
def project_to_feasible(v: np.ndarray, target: float) -> np.ndarray:
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


# ----- Random feasible vector (0 ≤ h ≤ 1, Σ h = target) -----
def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_to_feasible(v, n / 2.0)


# ----- Upsample the global initial_h_values (if present) -----
def upsample_initial(to_n: int) -> np.ndarray:
    init = np.array(initial_h_values, dtype=float)
    m = init.size
    xs_old = np.linspace(0.0, 2.0, m, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, to_n, endpoint=False)
    return np.interp(xs_new, xs_old, init)


# ----- Equality constraint for SLSQP -----
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


# ----- Smoothed max (log‑sum‑exp) objective for SLSQP -----
def softmax_obj(x):
    corr = correlate_fft(x, 1.0 - x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


# ----- SLSQP refinement with a given β -----
def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int = 400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # exact sum correction (simple greedy fix)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ----- Simple hill‑climb (deterministic improvements) -----
def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start: np.ndarray, rng: np.random.Generator, max_iters: int = 1500,
               start_step: float = 0.3):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


# ----- Smooth descent using a soft‑max surrogate (projected gradient) -----
def smooth_descent(h0: np.ndarray, rng: np.random.Generator,
                  time_limit: float, betas: list) -> tuple:
    """Projected gradient descent on the log‑sum‑exp surrogate."""
    h = np.clip(h0, 0.0, 1.0)
    target = h.size / 2.0
    # enforce exact sum once at start
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(500):
            i = rng.integers(0, h.size)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break

    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    iter_cnt = 0
    per_beta = time_limit / len(betas)

    for beta in betas:
        beta_start = time.time()
        while time.time() - beta_start < per_beta:
            if time.time() - start >= time_limit:
                return best_h, best_c

            # correlation h with 1‑h
            C_full = correlate_fft(h, 1.0 - h)
            maxC = np.max(C_full)
            # stable soft‑max weights
            exp_term = np.exp(beta * (C_full - maxC))
            w = exp_term / exp_term.sum()

            # weighted sum of h shifted forward and backward
            n = h.size
            idx0 = n - 1
            s_full = correlate_fft(w, h)
            s_plus = s_full[idx0:idx0 + n]
            w_rev = w[::-1]
            s_rev_full = correlate_fft(w_rev, h)
            s_minus = s_rev_full[idx0:idx0 + n]

            grad = 1.0 - s_plus - s_minus

            iter_cnt += 1
            eta = 0.5 / np.sqrt(iter_cnt)

            h_new = h - eta * grad
            np.clip(h_new, 0.0, 1.0, out=h_new)

            # enforce sum constraint
            diff = target - h_new.sum()
            if abs(diff) > 1e-12:
                for _ in range(200):
                    i = rng.integers(0, n)
                    add = min(diff, 1.0 - h_new[i]) if diff > 0 else max(diff, -h_new[i])
                    h_new[i] += add
                    diff = target - h_new.sum()
                    if abs(diff) < 1e-12:
                        break

            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()

            # simple Metropolis acceptance to avoid premature stagnation
            c_cur = compute_c5(h)
            if c_new <= c_cur or rng.random() < np.exp(-(c_new - c_cur) / max(eta, 1e-12)):
                h = h_new
                # c_cur will be recomputed in next loop
    return best_h, best_c
