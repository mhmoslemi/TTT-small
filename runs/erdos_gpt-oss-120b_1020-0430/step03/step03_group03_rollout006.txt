def compute_c5(h):
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)

def enforce_sum(h, target, rng, max_iter=10000):
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            # add to i
            add = min(diff, 1.0 - h[i])
            if add <= 0:
                continue
            h[i] += add
        else:
            # subtract from i
            sub = min(-diff, h[i])
            if sub <= 0:
                continue
            h[i] -= sub
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    h = np.full(n, 0.5)
    for _ in range(n * 3):
        i, j = rng.integers(0, n, size=2)
        if i == j:

        max_eps = min(h[i], 1.0 - h[j])

        h[i] -= eps
        h[j] += eps

def crossover(parent1, parent2, rng):
    mask = rng.random(parent1.size) < 0.5
    child = np.where(mask, parent1, parent2)
    return child

def mutate(child, rng, mut_rate=0.1, mut_scale=0.05):
    # random offset for each gene with prob mut_rate
    rand = rng.random(child.size)
    idx = np.where(rand < mut_rate)[0]
    if idx.size > 0:
        offsets = (rng.random(idx.size) - 0.5) * 2.0 * mut_scale
        child[idx] += offsets
        child[idx] = np.clip(child[idx], 0.0, 1.0)
    return child

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # compute smooth max: (m + log(sum(exp(bc - m))))/beta
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
```

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {"type": "eq", "fun": sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x
```

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

def biased_perturb_h(h, rng, scale=0.2):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    k = np.argmax(corr)
    shift = k - (n - 1)
    i0 = max(0, -shift)
    i1 = min(n, n - shift)
    idx = np.arange(i0, i1)
    if idx.size == 0:
        return perturb_h(h, rng, scale)
    prod = h[idx] * (1 - h[idx + shift])
    thr_hi = np.quantile(prod, 0.7)
    thr_lo = np.quantile(prod, 0.3)
    high = idx[prod > thr_hi]
    low = idx[prod < thr_lo]
    if high.size == 0 or low.size == 0:
        return perturb_h(h, rng, scale)
    i = rng.choice(high)
    j = rng.choice(low)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

def simulated_annealing(h0, rng, max_iter, init_temp, final_temp):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    n = h.size
    log_ratio = math.log(final_temp / init_temp) if init_temp > 0 else 0.0
    for it in range(1, max_iter + 1):
        T = init_temp * math.exp(log_ratio * it / max_iter)
        cand = biased_perturb_h(h, rng, scale=0.3) if rng.random() < 0.7 else perturb_h(h, rng, scale=0.3)
        cand_c = compute_c5(cand)
        diff = cand_c - cur_c
        if diff < 0 or rng.random() < math.exp(-diff / max(T, 1e-12)):
            h, cur_c = cand, cand_c
            if cand_c < best_c:
                best_c, best_h = cand_c, cand.copy()
    return best_h, best_c

import time, math, numpy as np
from scipy.optimize import minimize

# ---------- basic helpers ----------
def compute_c5(h):
    """Discrete C5 for step function h (dx = 2/n)."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / h.size)

def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h in‑place so that sum(h)=target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            if add <= 0:
                continue
            h[i] += add
        else:
            sub = min(-diff, h[i])
            if sub <= 0:
                continue
            h[i] -= sub
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    """Random h with 0≤h≤1 and sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 3):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps
        h[i] -= eps
        h[j] += eps
    enforce_sum(h, n / 2.0, rng)
    return h

# ---------- genetic operators ----------
def crossover(p1, p2, rng):
    mask = rng.random(p1.size) < 0.5
    return np.where(mask, p1, p2)

def mutate(child, rng, mut_rate=0.1, mut_scale=0.05):
    mask = rng.random(child.size) < mut_rate
    cnt = mask.sum()
    if cnt:
        offs = (rng.random(cnt) - 0.5) * 2.0 * mut_scale
        child[mask] += offs
        np.clip(child, 0.0, 1.0, out=child)
    return child

# ---------- simulated annealing helpers ----------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

def biased_perturb_h(h, rng, scale=0.2):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    k = np.argmax(corr)
    shift = k - (n - 1)
    i0 = max(0, -shift)
    i1 = min(n, n - shift)
    idx = np.arange(i0, i1)
    if idx.size == 0:
        return perturb_h(h, rng, scale)
    prod = h[idx] * (1.0 - h[idx + shift])
    thr_hi = np.quantile(prod, 0.7)
    thr_lo = np.quantile(prod, 0.3)
    high = idx[prod > thr_hi]
    low = idx[prod < thr_lo]
    if high.size == 0 or low.size == 0:
        return perturb_h(h, rng, scale)
    i = rng.choice(high)
    j = rng.choice(low)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

def simulated_annealing(h0, rng, max_iter, init_temp, final_temp):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    log_ratio = math.log(final_temp / init_temp) if init_temp > 0 else 0.0
    for it in range(1, max_iter + 1):
        T = init_temp * math.exp(log_ratio * it / max_iter)
        cand = biased_perturb_h(h, rng, scale=0.3) if rng.random() < 0.7 else perturb_h(h, rng, scale=0.3)
        cand_c = compute_c5(cand)
        diff = cand_c - cur_c
        if diff < 0 or rng.random() < math.exp(-diff / max(T, 1e-12)):
            h, cur_c = cand, cand_c
            if cand_c < best_c:
                best_c, best_h = cand_c, cand.copy()
    return best_h, best_c

# ---------- SLSQP smooth‑max refinement ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {"type": "eq", "fun": sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

        new_pop = [population[i].copy() for i in sorted_idx[:elite_num]]

        # fill rest of population
        while len(new_pop) < popsize:
            # tournament select two parents
            cand = rng.choice(popsize, size=tournament_k, replace=False)
