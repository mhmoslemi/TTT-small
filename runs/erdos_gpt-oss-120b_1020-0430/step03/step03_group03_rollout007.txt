def project_sum_bounds(h, target):
    # ensure 0<=h_i<=1, sum target
    # We'll use iterative water-filling
    # Start with clipping
    h = np.clip(h, 0.0, 1.0)
    # Compute diff = target - sum(h)
    diff = target - h.sum()
    tol = 1e-12
    # Iteratively adjust
    free = np.ones_like(h, dtype=bool)  # initially all free
    while abs(diff) > tol and np.any(free):
        # number of free entries
        cnt = free.sum()
        # adjust amount per free
        delta = diff / cnt
        # apply
        h[free] += delta
        # Clip again
        over = h > 1.0
        under = h < 0.0
        # fix overs/unders; set to bound and mark as not free
        if over.any():
            h[over] = 1.0
            free[over] = False
        if under.any():
            h[under] = 0.0
            free[under] = False
        diff = target - h.sum()
    # If diff remains (due to numerical issues) adjust smallest.
    if abs(diff) > 1e-10:
        # adjust the free entries arbitrarily
        # simply add diff to first free or to any entry within bounds
        idx = np.argmax(free) if free.any() else np.argmax(h)
        h[idx] += diff
        h = np.clip(h,0.0,1.0)
        # final enforce sum by scaling
        h = h * (target / h.sum())
    return h

def softmax_weights(C, beta):
    maxC = C.max()
    exp_vals = np.exp(beta * (C - maxC))
    w = exp_vals / exp_vals.sum()
    return w

def gradient_softmax(h, beta):
    C = np.correlate(h, 1 - h, mode='full')
    w = softmax_weights(C, beta)
    # w length 2n-1
    # compute conv of h with w and w_rev
    conv1 = np.convolve(h, w, mode='full')
    conv2 = np.convolve(h, w[::-1], mode='full')
    center = len(h) - 1
    grad = 1.0 - conv1[center:center+len(h)] - conv2[center:center+len(h)]
    return grad

import time, math, numpy as np

# ------------------------------------------------------------
# basic evaluation
# ------------------------------------------------------------
def compute_c5(h):
    """Discrete C5 for step function h on [0,2] with dx = 2/len(h)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


# ------------------------------------------------------------
# projection onto box [0,1] and hyperplane sum = target
# ------------------------------------------------------------
def project_sum_bounds(h, target, tol=1e-12):
    """Clip to [0,1] and adjust so that sum(h)=target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    free = np.ones_like(h, dtype=bool)

    while abs(diff) > tol and np.any(free):
        cnt = free.sum()
        delta = diff / cnt
        h[free] += delta
        over = h > 1.0
        under = h < 0.0
        if over.any():
            h[over] = 1.0
            free[over] = False
        if under.any():
            h[under] = 0.0
            free[under] = False
        diff = target - h.sum()

    if abs(diff) > 1e-9:
        # very small remaining error – distribute to any free entry
        idx = np.where(free)[0]
        if idx.size:
            i = idx[0]
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
        else:
            # all entries are at a bound; rescale proportionally (unlikely)
            h = np.clip(h * (target / h.sum()), 0.0, 1.0)
    return h


# ------------------------------------------------------------
# gradient of soft‑max (log‑sum‑exp) of the correlation vector
# ------------------------------------------------------------
def gradient_softmax(h, beta):
    """∇_h  (1/β) log Σ_k exp(β C_k)  where C_k = corr of h with 1‑h."""
    # correlation values C_k
    C = np.correlate(h, 1.0 - h, mode="full")
    m = C.max()
    w = np.exp(beta * (C - m))
    w /= w.sum()                     # soft‑max weights, length 2n‑1
    # convolution to obtain Σ_k w_k * h_{j‑k} and Σ_k w_k * h_{j+k}
    conv1 = np.convolve(h, w, mode="full")          # length 3n‑2
    conv2 = np.convolve(h, w[::-1], mode="full")   # length 3n‑2
    n = h.size
    off = n - 1
    sum_j_minus_k = conv1[off : off + n]
    sum_j_plus_k = conv2[off : off + n]
    grad = 1.0 - sum_j_plus_k - sum_j_minus_k
    return grad

def upsample_h(old_h, new_n):
    old_n = len(old_h)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(new_x, old_x, old_h)

    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- resolution -------------------------------------------------
    n = 200                               # 200 points → dx = 0.01
    target_sum = n * 0.5                  # ∫h = 1  ⇒  Σh = n/2

    # ---- initialise -------------------------------------------------
    if "initial_h_values" in globals():
        h = upsample_h(np.array(initial_h_values, dtype=float), n)
    else:
        h = np.full(n, 0.5)
    h = project_sum_bounds(h, target_sum)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ---- gradient‑based smoothing phases ----------------------------
    beta_schedule = [30, 100, 300, 1000, 3000, 10000]
    grad_deadline = start + budget_s * 0.85

    for beta in beta_schedule:
        if time.time() >= grad_deadline:
            break
        # step size tuned to beta
        eta = 0.25 / math.sqrt(beta)
        # allocate iterations based on remaining time (≈1800 it/s)
        remaining = max(0.0, grad_deadline - time.time())
        max_iter = int(min(20000, remaining * 1800))
        max_iter = max(1200, max_iter)

        for it in range(max_iter):
            # compute gradient of smooth max
            g = gradient_softmax(h, beta)
            # gradient step + projection
            h_new = project_sum_bounds(np.clip(h - eta * g, 0.0, 1.0), target_sum)

            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_c, best_h = c_new, h_new.copy()
            h = h_new

            # occasional random jitter to escape plateaus
            if it % 500 == 0 and rng.random() < 0.3:
                noise = rng.normal(scale=0.015, size=n)
                h_noise = project_sum_bounds(np.clip(h + noise, 0.0, 1.0), target_sum)
                c_noise = compute_c5(h_noise)
                if c_noise < best_c:
                    best_c, best_h = c_noise, h_noise.copy()
                h = h_noise

            if time.time() >= grad_deadline:
                break

    # ---- high‑beta polishing (subgradient‐like) --------------------
    final_deadline = start + budget_s
    beta = 20000
    eta = 0.05 / math.sqrt(beta)

    while time.time() < final_deadline:
        g = gradient_softmax(best_h, beta)
        h_new = project_sum_bounds(np.clip(best_h - eta * g, 0.0, 1.0), target_sum)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()
        else:
            eta *= 0.9            # reduce step if no improvement
        if eta < 1e-7:
            break

    # ---- final clean‑up --------------------------------------------
    best_h = project_sum_bounds(np.clip(best_h, 0.0, 1.0), target_sum)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
