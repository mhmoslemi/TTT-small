def gradient_descent_step(h, alpha, rng):
    # compute correlation
    corr = np.correlate(h, 1 - h, mode='full')
    # find lag index of max correlation
    # correlation length 2n-1, index offset = n-1 for lag=0
    max_idx = np.argmax(corr)
    k = max_idx - (h.size - 1)  # lag (can be negative)
    # compute gradient
    n = h.size
    g = np.ones(n)
    # contributions
    if k >= 0:
        # overlapping region i in [0, n-k-1]
        g[:n-k] -= h[k:]  # h[i+k]
        # for second term, i in [k, n-1] for i - k >= 0
        g[k:] -= h[:n-k]  # h[i - k]
    else:
        # k negative; let k = -m >0, lag negative equivalent to positive shift opposite direction
        m = -k
        # For k negative, overlap i in [m, n-1]
        # h_i (1 - h_{i - m})? Actually formula: sum_i h_i (1 - h_{i+k}), but for k negative, i+k < i; overlapping region i from -k to n-1? We'll just compute using generic method: 
        # Let's compute product contributions: we need h_i+h_{i+k} for i where both valid.
        # Implementation: for all i that have i + k in [0, n-1] and i - k in [0, n-1].
        # Equivalent to using same formula as positive but with sign.
        # Simpler: compute two arrays shift_plus and shift_minus:
        # For any i, if i + k in range => subtract h[i+k]; if i - k in range => subtract h[i-k].
        # Since k negative, i + k means i - m.
        # So:
        g[m:] -= h[:n-m]  # h[i + k] => h[i - m]
        g[:n-m] -= h[m:]  # h[i - k] => h[i + m]
    # Now step
    h_new = h - alpha * g
    # Clip and enforce sum
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = enforce_sum(h_new, n/2.0, rng)
    return h_new

def gradient_improve(h, rng, max_iters=500, init_alpha=0.5, decay=0.97):
    best_h = h.copy()
    best_c = compute_c5(best_h)
    alpha = init_alpha
    for it in range(max_iters):
        h_new = gradient_descent_step(best_h, alpha, rng)
        c_new = compute_c5(h_new)
        if c_new < best_c - 1e-8:
            best_h, best_c = h_new, c_new
            alpha = min(alpha/decay, 1.0)  # maybe increase? Actually if improvement, we can keep step size constant.
        else:
            alpha *= decay
        if alpha < 1e-5:
            break
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# Global variables for SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None

def enforce_sum(h, target, rng, max_iter=5000):
    """Clip to [0,1] and adjust to achieve the exact sum `target`."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def compute_c5(h):
    """Exact discrete C5 for step function h on [0,2] (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb_multi(h_start, rng, max_iters=2000, init_step=0.5, n_trials=5):
    """Simple hill‑climb with multiple random perturbations per iteration."""
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c = c
                h = new_h
                improved = True
                break
        if not improved and rng.random() < 0.02:
            new_h = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(new_h)
            if c < best_c:
                best_c = c
                h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return h, best_c

def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """Smooth log‑sum‑exp surrogate of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum (tiny numerical drift)
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x

def gradient_descent_step(h, alpha, rng):
    """One projected gradient step targeting the worst lag."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(corr)
    k = max_idx - (n - 1)  # lag (can be negative)
    g = np.ones(n)
    if k >= 0:
        if k < n:
            g[:n - k] -= h[k:]          # subtract h[i+k]
            g[k:] -= h[:n - k]          # subtract h[i-k]
    else:
        m = -k
        if m < n:
            g[m:] -= h[:n - m]          # subtract h[i+k] = h[i-m]
            g[:n - m] -= h[m:]          # subtract h[i-k] = h[i+m]
    h_new = h - alpha * g
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = enforce_sum(h_new, n / 2.0, rng)
    return h_new

def gradient_improve(h, rng, max_iters=300, init_alpha=0.5, decay=0.95):
    """Repeated projected gradient steps with simple line‑search."""
    best_h = h.copy()
    best_c = compute_c5(best_h)
    alpha = init_alpha
    for _ in range(max_iters):
        h_candidate = gradient_descent_step(best_h, alpha, rng)
        c_candidate = compute_c5(h_candidate)
        if c_candidate < best_c - 1e-9:
            best_h, best_c = h_candidate, c_candidate
            # keep step size for next iteration
        else:
            alpha *= decay
        if alpha < 1e-5:
            break
    return best_h, best_c
