def pgd_optimize(h0, beta, step_size, max_iter, rng):
    h = h0.copy()
    target = h.size / 2
    for i in range(max_iter):
        # compute corr_k = target - np.correlate(h,h,'full')
        T = np.correlate(h, h, mode='full')
        corr = target - T
        # compute weights w
        expb = np.exp(beta * corr - np.max(beta*corr)) # for stability
        w = expb / expb.sum()
        # compute w_sym
        w_sym = w + w[::-1]
        # gradient
        c = np.correlate(w_sym, h, mode='full')
        grad = -c[(h.size-1):(h.size-1)+h.size]
        # step
        h = h - step_size * grad
        # project
        h = project_to_feasible(h, target)
        # optional early stop if improvement small
    return h

def grad_softmax(h, beta, target):
    # compute correlation of h with itself
    T = np.correlate(h, h, mode='full')
    corr = target - T  # length 2n-1
    # softmax weights
    maxval = np.max(beta * corr)
    # subtract max for numeric stability: compute w = exp(beta*corr-maxval)/sum
    expb = np.exp(beta * corr - maxval)
    w = expb / np.sum(expb)
    # symmetrize
    w_sym = w + w[::-1]
    # compute gradient: - correlate(w_sym, h)
    c = np.correlate(w_sym, h, mode='full')
    # want slice (n-1): (n-1)+n
    n = h.size
    grad = -c[(n-1):(n-1)+n]
    return grad

import time
import numpy as np

# ------------------------------------------------------------
# Basic utilities
# ------------------------------------------------------------
def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_to_feasible(v, target):
    """Project v onto {0≤h≤1, Σh = target}."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Fast random feasible vector (0≤h≤1, Σh=n/2)."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_to_feasible(v, n / 2.0)


def upsample_vector(v, new_n):
    """Linear interpolation of a vector defined on [0,2)."""
    old_n = v.size
    xs_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(xs_new, xs_old, v)


# ------------------------------------------------------------
# Gradient of the smooth max (log‑sum‑exp) objective
# ------------------------------------------------------------
def grad_softmax(h, beta, target):
    """Gradient of (1/β)log Σ exp(β·corr) w.r.t. h."""
    n = h.size
    # corr_k = target - Σ h_i h_{i+k}
    T = np.correlate(h, h, mode="full")          # length 2n‑1
    corr = target - T
    # softmax weights
    bc = beta * corr
    m = np.max(bc)
    exp_bc = np.exp(bc - m)
    w = exp_bc / np.sum(exp_bc)                 # length 2n‑1
    # symmetrised weights for the gradient
    w_sym = w + w[::-1]
    # gradient = - correlate(w_sym, h)
    c = np.correlate(w_sym, h, mode="full")
    grad = -c[(n - 1):(n - 1) + n]
    return grad


# ------------------------------------------------------------
# Projected gradient descent on the smooth objective
# ------------------------------------------------------------
def pgd_optimize(h0, target, rng, max_time, start_time):
    """Run a short PGD schedule, aborting when max_time is spent."""
    h = h0.copy()
    # (beta, step size, iterations) schedule
    schedules = [
        (0.5, 1.0, 300),
        (2.0, 0.8, 300),
        (8.0, 0.6, 250),
        (32.0, 0.4, 250),
        (128.0, 0.3, 200),
        (512.0, 0.2, 200),
        (2048.0, 0.1, 200),
    ]
    for beta, step, iters in schedules:
        for _ in range(iters):
            if time.time() - start_time >= max_time:
                return h
            grad = grad_softmax(h, beta, target)
            norm = np.linalg.norm(grad)
            if norm > 0:
                lr = step / norm
                h = h - lr * grad
            h = np.clip(h, 0.0, 1.0)
            h = project_to_feasible(h, target)
    return h


# ------------------------------------------------------------
# Simple hill‑climbing (pairwise epsilon transfer)
# ------------------------------------------------------------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new


def hill_climb(h_start, rng, max_iters=2000, start_step=0.3):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


# ------------------------------------------------------------
# Local optimisation wrapper (PGD + optional hill‑climb)
# ------------------------------------------------------------
def local_optimize_h(h0, rng, max_time):
    start = time.time()
    target = h0.size / 2.0
    # PGD phase
    h = pgd_optimize(h0, target, rng, max_time, start)
    # small hill‑climb if time permits
    elapsed = time.time() - start
    remaining = max_time - elapsed
    if remaining > 5.0:
        # scale iterations to remaining time (≈200 iters per sec)
        hc_iters = int(min(3000, remaining * 200))
        h, _ = hill_climb(h, rng, max_iters=hc_iters, start_step=0.3)
    return h, compute_c5(h)


# ------------------------------------------------------------

# ------------------------------------------------------------
