def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # Determine n
    base_n = 180
    # Use upsample to 360
    n = base_n * 2
    target = n / 2

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size != base_n:
            # if initial size not 180, still use as base
            base_n = h0.size
            n = base_n * 2
            target = n/2
        # upsample by repeat
        h0 = np.repeat(h0, 2)
    else:
        # random feasible vector
        # generate weight half half
        v = rng.exponential(1.0, size=n)
        v = v / v.sum() * target
        h0 = np.clip(v, 0.0, 1.0)
        # project to sum exactly
        h0 = project_onto_box_simplex(h0, target)

    # Ensure feasible
    h = np.clip(h0, 0.0, 1.0)
    h = project_onto_box_simplex(h, target)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    def projection(v):
        return project_onto_box_simplex(v, target)

    # gradient descent schedule
    betas = [5, 20, 80, 200, 800, 2000]
    # We'll set max iter per beta based on time
    for beta in betas:
        # step size schedule start
        step = 0.5
        # compute max iter allowed: e.g., 2000 or less based on remaining time
        max_iter = 2000
        for it in range(max_iter):
            if time.time() - start > budget_s * 0.90:
                break
            # compute corr
            corr = np.correlate(h, 1.0 - h, mode='full')
            # compute softmax weights
            bc = beta * corr
            max_bc = np.max(bc)
            exp_bc = np.exp(bc - max_bc)
            w = exp_bc / np.sum(exp_bc)
            # compute gradients
            term1 = np.correlate(w, 1.0 - h, mode='full')[n-1:n-1+n]
            term2 = np.convolve(w, h, mode='full')[n-1:n-1+n]
            grad = term1 - term2
            # update
            new_h = projection(h - step * grad)
            new_c = compute_c5(new_h)
            if new_c < best_c - 1e-12:
                best_c, best_h = new_c, new_h.copy()
                h = new_h
                # possibly increase step
                step = min(step * 1.1, 1.0)
            else:
                # reject, reduce step
                step = max(step * 0.5, 1e-5)
                # optionally keep current h unchanged
                # we might also try to move anyway sometimes; but keep simple.
        if time.time() - start > budget_s * 0.90:

    # final local hill climbing from best_h
    # Hill climb similar to earlier but more iterations
    def hill_climb_local(h_start, max_iters=5000, start_step=0.5):
        h = h_start.copy()
        best_h_local = h.copy()
        best_c_local = compute_c5(h)
        step = start_step
        n_local = h.size
        for it in range(max_iters):
            if time.time() - start > budget_s * 0.99:
                break
            i = rng.integers(0, n_local)
            j = rng.integers(0, n_local)
            while j == i:
                j = rng.integers(0, n_local)
            max_pos = min(h[i], 1.0 - h[j])
            max_neg = min(h[j], 1.0 - h[i])
            if max_pos + max_neg == 0:
                continue
            eps = (rng.random() * (max_pos + max_neg)) - max_neg
            eps *= step
            new_h = h.copy()
            new_h[i] -= eps
            new_h[j] += eps
            new_h = np.clip(new_h, 0.0, 1.0)  # sum preserved (due to eps)

            if c < best_c_local:

                best_h_local = new_h.copy()
                h = new_h
                step = min(step * 1.05, 1.0)
            else:

        return best_h_local, best_c_local

        if c_fine < best_c:

    # final projection enforce
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_simplex(best_h, target)

    final_c = compute_c5(best_h)

import time
import numpy as np
from scipy.optimize import minimize

# -------------------------------------------------------------
# Helper utilities
# -------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C₅ for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / h.size))


def project_onto_box_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Projection of v onto the set {x | 0≤x_i≤1, sum x_i = target}.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo = float(np.min(v - 1.0))   # ensures all x_i ≤ 1 after shift
    hi = float(np.max(v))         # ensures all x_i ≥ 0 after shift
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction for numerical drift
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        idx = np.where(mask)[0]
        if idx.size:
            add = diff / idx.size
            x[idx] = np.clip(x[idx] + add, 0.0, 1.0)
    return x


def softmax_weights(corr: np.ndarray, beta: float) -> np.ndarray:
    """Log‑sum‑exp based softmax on the correlation vector."""
    bc = beta * corr
    m = np.max(bc)
    exp_bc = np.exp(bc - m)
    return exp_bc / exp_bc.sum()


def gradient_surrogate(h: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth surrogate (soft‑max of the correlations)
    with respect to h, for a fixed beta.
    """
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    w = softmax_weights(corr, beta)

    # term1 = Σ_k w_k * (1 - h_{i+k})
    term1 = np.correlate(w, 1.0 - h, mode="full")[n - 1 : n - 1 + n]
    # term2 = Σ_k w_k * h_{i-k}
    term2 = np.convolve(w, h, mode="full")[n - 1 : n - 1 + n]
    return term1 - term2


def slsqp_optimize(h0: np.ndarray, beta: float, max_iter: int = 500) -> np.ndarray:
    """One‑shot SLSQP refinement on the smooth surrogate."""
    n = h0.size
    target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        exp_bc = np.exp(bc - m)
        return (m + np.log(np.sum(exp_bc))) / beta

    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # final exact‑sum projection
    return project_onto_box_simplex(x, target)


def hill_climb_local(h_start: np.ndarray, rng: np.random.Generator,
                    max_iters: int = 6000, start_step: float = 0.4):
    """Simple pairwise hill‑climb (keeps the sum exactly)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    for it in range(max_iters):
        if (it & 127) == 0 and time.time() - _START_TIME > _BUDGET * 0.99:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        max_pos = min(h[i], 1.0 - h[j])   # amount we can move from i → j
        max_neg = min(h[j], 1.0 - h[i])   # amount we can move from j → i
        if max_pos + max_neg == 0:
            continue
        eps = (rng.random() * (max_pos + max_neg)) - max_neg
        eps *= step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        # sum preserved; values stay in [0,1] because of max_pos/max_neg
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h.copy()
            h = new_h
            step = min(step * 1.05, 1.0)
        else:
            step = max(step * 0.95, 1e-5)
    return best_h, best_c

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    global _START_TIME, _BUDGET
    _START_TIME = time.time()
    _BUDGET = budget_s

    rng = np.random.default_rng(seed)

    # ---------- initial construction ----------
    base_n = 180
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size != base_n:
            base_n = h0.size
        # up‑sample by simple repetition (keeps total integral)
        h0 = np.repeat(h0, 2)
    else:
        n_tmp = base_n * 2
        target_tmp = n_tmp / 2.0
        v = rng.exponential(1.0, size=n_tmp)
        v = v / v.sum() * target_tmp
        h0 = np.clip(v, 0.0, 1.0)
        h0 = project_onto_box_simplex(h0, target_tmp)

    n = h0.size
    target = n / 2.0
    h = project_onto_box_simplex(h0, target)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ---------- smooth‑gradient descent ----------
    betas = [5.0, 20.0, 80.0, 200.0, 800.0, 2000.0, 5000.0]
    for beta in betas:
        step = 0.5
        max_iter = 2000
        for it in range(max_iter):
            if time.time() - _START_TIME > _BUDGET * 0.88:
                break
            grad = gradient_surrogate(h, beta)
            new_h = project_onto_box_simplex(h - step * grad, target)
            new_c = compute_c5(new_h)
            if new_c < best_c - 1e-12:
                best_c, best_h = new_c, new_h.copy()
                h = new_h
                step = min(step * 1.10, 1.0)
            else:
                step = max(step * 0.55, 1e-5)
        if time.time() - _START_TIME > _BUDGET * 0.88:
            break

    # ---------- optional SLSQP refinement ----------
    if time.time() - _START_TIME < _BUDGET * 0.80:
        for beta in (200.0, 800.0, 2000.0):
            if time.time() - _START_TIME > _BUDGET * 0.80:
                break
            try:
                refined = slsqp_optimize(best_h, beta, max_iter=600)
                c_ref = compute_c5(refined)
                if c_ref < best_c:
                    best_c, best_h = c_ref, refined.copy()
            except Exception:
                continue

    # ---------- final local hill climb ----------
    if time.time() - _START_TIME < _BUDGET * 0.95:
        h_fine, c_fine = hill_climb_local(best_h, rng, max_iters=8000, start_step=0.3)
        if c_fine < best_c:
            best_c, best_h = c_fine, h_fine.copy()

    # final clean‑up
    best_h = project_onto_box_simplex(np.clip(best_h, 0.0, 1.0), target)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
