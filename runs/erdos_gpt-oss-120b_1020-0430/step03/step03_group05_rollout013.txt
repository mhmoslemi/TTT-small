def slsqp_optimize_analytic(h0, beta, rng, max_iter=2000, ftol=1e-12):
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        m = np.max(beta * corr)
        # compute log-sum-exp in stable form
        bc = beta * corr
        max_bc = np.max(bc)
        val = (max_bc + np.log(np.sum(np.exp(bc - max_bc)))) / beta
        return val
    def grad(x):
        corr = np.correlate(x, 1 - x, mode='full')
        bc = beta * corr
        max_bc = np.max(bc)
        exp_bc = np.exp(bc - max_bc)
        w = exp_bc / exp_bc.sum()
        # compute gradient
        n = x.size
        grad = np.zeros(n)
        # w indexed by offset index offset + n -1
        for offset in range(-n+1, n):
            wi = w[offset + n - 1]
            if wi == 0.0:
                continue
            if offset >= 0:
                grad[:n - offset] += wi * (1 - x[offset:])
                grad[offset:] -= wi * x[:n - offset]
            else:
                d = -offset
                grad[d:] += wi * (1 - x[:n - d])
                grad[:n - d] -= wi * x[d:]
        return grad
    # equality constraint sum(x) = target
    def cons_eq(x):
        return np.sum(x) - target
    cons = {'type': 'eq', 'fun': cons_eq, 'jac': lambda x: np.ones_like(x)}
    res = minimize(obj, h0, method='SLSQP', jac=grad,
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': ftol, 'disp': False})
    # clip and project to ensure feasibility
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # simple projection: adjust with uniform distribution across non-saturated? 
        # Use project_to_feasible defined earlier.
        x = project_to_feasible(x, target)
    return x

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utility functions ----------
def compute_c5(h):
    """Exact C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_to_feasible(v, target):
    """Project v onto {0≤h≤1, Σh = target} via a simple bisection on the shift."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Fast generation of a random feasible vector (0≤h≤1, Σh=n/2)."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_to_feasible(v, n / 2.0)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.3):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 15)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h, h = c, new_h.copy(), new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.85, 1e-5)
    return best_h, best_c


# ---------- smooth‑max objective with analytic gradient ----------
def smooth_obj_and_grad(x, beta):
    """Return (value, gradient) of the log‑sum‑exp approximation of max_k Σ h_i(1‑h_{i+k})."""
    n = x.size
    corr = np.correlate(x, 1 - x, mode="full")
    bc = beta * corr
    max_bc = np.max(bc)
    exp_bc = np.exp(bc - max_bc)
    w = exp_bc / exp_bc.sum()          # softmax weights, length 2n‑1
    # value (stable log‑sum‑exp)
    val = (max_bc + np.log(exp_bc.sum())) / beta

    # gradient
    grad = np.zeros(n)
    # offset runs from -(n-1) … (n-1)
    for offset in range(-n + 1, n):
        wi = w[offset + n - 1]
        if wi == 0.0:
            continue
        if offset >= 0:
            # first term: (1‑h_{j+offset})
            grad[: n - offset] += wi * (1 - x[offset:])
            # second term: –h_{j‑offset}
            grad[offset:] -= wi * x[: n - offset]
        else:
            d = -offset
            grad[d:] += wi * (1 - x[: n - d])
            grad[: n - d] -= wi * x[d:]
    return val, grad


def slsqp_optimize_analytic(h0, beta, max_iter=500, ftol=1e-12):
    """Run SLSQP on the smooth‑max surrogate, using the analytic gradient."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def fun(x):
        val, _ = smooth_obj_and_grad(x, beta)
        return val

    def jac(x):
        _, g = smooth_obj_and_grad(x, beta)
        return g

    def eq_cons(x):
        return np.sum(x) - target

    cons = {"type": "eq", "fun": eq_cons, "jac": lambda x: np.ones_like(x)}

    res = minimize(
        fun,
        h0,
        method="SLSQP",
        jac=jac,
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": ftol, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # final projection in case of tiny feasibility drift
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = project_to_feasible(x, target)
    return x


# ---------- main optimisation driver ----------
