    for i in range(n):
        # parity of number of ones in binary representation of i

def binary_sa(h_bin, time_budget, rng):

    best_h = h_bin.copy()
    best_c = compute_c5(best_h)
    current_h = h_bin.copy()
    current_c = best_c
    t0 = time.time()
    # parameters
    T0 = 0.5 * best_c  # initial temperature? set maybe based on initial_c
    Tf = 0.001
    N = 200000  # approx number of iterations
    for it in range(N):
        elapsed = time.time() - t0
        if elapsed > time_budget:
            break
        # compute temperature schedule linear
        tau = it / N
        T = T0 * (1 - tau) + Tf * tau
        # propose swap
        ones = np.where(current_h > 0.5)[0]
        zeros = np.where(current_h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # create trial
        trial = current_h.copy()
        trial[i] = 0.0
        trial[j] = 1.0
        trial_c = compute_c5(trial)
        delta = trial_c - current_c
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            current_h = trial
            current_c = trial_c
            if trial_c < best_c:
                best_c = trial_c
                best_h = trial.copy()
    return best_h, best_c

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def slsqp_optimize(h0, beta, rng, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta

        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

def hill_climb(h_start, rng, max_iters=2000, start_step=0.3):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step

import time
import math
import numpy as np
from scipy.optimize import minimize

# globals for the soft‑max surrogate used in SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_to_feasible(v, target):
    """Projection onto {0≤h≤1, Σh = target} by bisection."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.3):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


def slsqp_optimize(h0, beta, rng, max_iter=500):
    """SLSQP optimisation of the smooth surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # enforce exact sum by a simple projection onto the hyperplane
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def thue_morse(n):
    """Binary Thue‑Morse sequence of length n (n must be a power of two)."""
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = bin(i).count("1") % 2
    return seq


def random_binary(n, rng):
    """Random binary vector with exactly n/2 ones (n even)."""
    arr = np.zeros(n, dtype=float)
    arr[: n // 2] = 1.0
    rng.shuffle(arr)
    return arr


def simulated_annealing(h_start, time_budget, rng):
    """Simple SA on the exact C5 objective (preserves sum)."""
    h_cur = h_start.copy()
    cur_c = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_c = cur_c
    t0 = time.time()
    T0 = cur_c
    Tf = cur_c * 1e-3 if cur_c > 0 else 1e-6
    while True:
        elapsed = time.time() - t0
        if elapsed > time_budget:
            break
        tau = elapsed / time_budget
        T = T0 * (1 - tau) + Tf * tau
        h_new = perturb_h(h_cur, rng, scale=0.5)
        c_new = compute_c5(h_new)
        delta = c_new - cur_c
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            h_cur = h_new
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # candidate discretisations (larger ones first for more impact)
    candidate_ns = [256, 240]
    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        candidate_ns = [n0] + candidate_ns

    for n in candidate_ns:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining < 10.0:
            break

        # allocate a chunk of the remaining budget for this n
        # give a larger share to larger n
        alloc = min(remaining * 0.5, max(30.0, n * 0.25))

        # ----- initialise starting point -----
        if "initial_h_values" in globals() and len(initial_h_values) == n:
            h0 = np.array(initial_h_values, dtype=float)
        else:
            if (n & (n - 1)) == 0:                     # power of two → Thue‑Morse
                h0 = thue_morse(n)
            else:
                h0 = random_binary(n, rng)

        h0 = np.clip(h0, 0.0, 1.0)
        target_sum = n / 2.0
        if abs(h0.sum() - target_sum) > 1e-12:
            h0 = project_to_feasible(h0, target_sum)

        # ----- simulated annealing -----
        t_sa = alloc * 0.6
        h_sa, c_sa = simulated_annealing(h0, t_sa, rng)
        cur_h, cur_c = h_sa, c_sa
        if cur_c < best_c:
            best_c, best_h, best_n = cur_c, cur_h.copy(), n

        # ----- SLSQP refinement with increasing beta -----
        for beta in (10.0, 50.0, 200.0, 800.0, 2000.0):

            cur_c = compute_c5(cur_h)
            if cur_c < best_c:

        # ----- deterministic hill climbing -----
        if time.time() - start_time < budget_s * 0.98:
            cur_h, cur_c = hill_climb(cur_h, rng, max_iters=2000, start_step=0.3)
            if cur_c < best_c:
                best_c, best_h, best_n = cur_c, cur_h.copy(), n

        # ----- short final SA for fine‑tuning -----
        if time.time() - start_time < budget_s * 0.99:
            t_sa2 = max(5.0, (budget_s - (time.time() - start_time)) * 0.1)
            h_sa2, c_sa2 = simulated_annealing(cur_h, t_sa2, rng)
            if c_sa2 < best_c:
                best_c, best_h, best_n = c_sa2, h_sa2.copy(), n

    # ----- ensure feasibility of the final answer -----
    if best_h is None:
        best_n = 256
        best_h = np.full(best_n, 0.5)

    best_h = np.clip(best_h, 0.0, 1.0)

    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(best_n)
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
