def slsqp_optimize(h0, beta, max_iter=200, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    n = h0.size
    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {"type": "eq", "fun": sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons, options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, rng)
    return x
```

Define `sum_eq_constraint` and `softmax_obj`:

```
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # compute log-sum-exp safely
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
```

import time, math, numpy as np

# ---------- basic utilities ----------
def compute_c5(h):
    """Discrete C5 for step function on [0,2] with step size 2/n."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / h.size)


def enforce_sum(h, target, rng):
    """Adjust h in‑place so that sum(h)=target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # tiny residual correction
    residual = target - h.sum()
    if abs(residual) > 1e-10:
        h += residual / n
        np.clip(h, 0.0, 1.0, out=h)
        residual = target - h.sum()
        if abs(residual) > 1e-10:
            idx = np.argmin(h) if residual < 0 else np.argmax(h)
            h[idx] = np.clip(h[idx] + residual, 0.0, 1.0)
    return h

def local_improve(h, rng, max_no_improve=80):
    """Greedy descent using reduce_max_move."""
    cur = h.copy()
    cur_c = compute_c5(cur)
    no_improve = 0
    while no_improve < max_no_improve:
        cand = reduce_max_move(cur, rng)
        cand_c = compute_c5(cand)
        if cand_c < cur_c - 1e-12:
            cur, cur_c = cand, cand_c
            no_improve = 0
        else:
            no_improve += 1
    return cur, cur_c


# ---------- smooth max (softmax) for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=120, rng=None):
    """Smooth‑max SLSQP refinement."""
    if rng is None:
        rng = np.random.default_rng()
    n = h0.size
    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {"type": "eq", "fun": sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-9, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, rng)
    return x
