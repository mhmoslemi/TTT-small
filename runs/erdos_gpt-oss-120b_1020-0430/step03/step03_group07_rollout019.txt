def correlate_full(h):
    n = len(h)
    L = 1 << ((2*n - 1).bit_length())
    f = np.fft.rfft(h, L)
    g = np.fft.rfft((1 - h)[::-1], L)
    conv = np.fft.irfft(f * g, L)[:2*n-1]
    return conv

def compute_c5(h):
    return correlate_full(h).max() * (2.0 / len(h))

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c

def simulated_annealing_binary(start_a, rng, deadline, max_iters=200000, T0=0.01, Tf=1e-5):
    a = start_a.copy()
    cur_c = compute_c5(a)
    best_a = a.copy()
    best_c = cur_c
    n = a.size
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # temperature schedule
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        # propose swap
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if a[i] == a[j]:
            continue
        # swap
        a[i], a[j] = a[j], a[i]
        new_c = compute_c5(a)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_a = a.copy()
        else:
            # revert swap
            a[i], a[j] = a[j], a[i]
    return best_a, best_c

import time
import math
import numpy as np
from scipy.optimize import minimize

# ---------- utility ----------
def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_binary_vector(n, rng):
    a = np.zeros(n, dtype=float)
    a[rng.choice(n, n // 2, replace=False)] = 1.0
    return a

def adjust_binary_sum(a, target_one, rng):
    cur = int(a.sum())
    diff = target_one - cur
    if diff == 0:
        return a
    if diff > 0:
        zeros = np.where(a == 0)[0]
        flip = rng.choice(zeros, diff, replace=False)
        a[flip] = 1.0
    else:
        ones = np.where(a == 1)[0]
        flip = rng.choice(ones, -diff, replace=False)
        a[flip] = 0.0
    return a

def correlate_full(h):
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(h, L)
    g = np.fft.rfft((1 - h)[::-1], L)
    conv = np.fft.irfft(f * g, L)[: 2 * n - 1]
    return conv

def compute_c5(h):
    return correlate_full(h).max() * (2.0 / h.size)

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        nh = perturb_h(h, rng, scale=step)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c

CURRENT_BETA = None

def softmax_obj(x):

    m = bc.max()

    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta

    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    return x

# ---------- binary simulated annealing ----------
def simulated_annealing_binary(start_a, rng, deadline, max_iters=200000,
                              T0=0.01, Tf=1e-5):
    a = start_a.copy()
    cur_c = compute_c5(a)
    best_a, best_c = a.copy(), cur_c
    n = a.size
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if a[i] == a[j]:
            continue
        a[i], a[j] = a[j], a[i]
        new_c = compute_c5(a)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            cur_c = new_c
            if cur_c < best_c:
                best_c, best_a = cur_c, a.copy()
        else:
            a[i], a[j] = a[j], a[i]
    return best_a, best_c

                best_h, best_c, n = h_up_hc, c_up_hc, n_up

        # SLSQP on finer grid
        for beta in beta_list:
            if time.time() >= deadline:
                break
            try:
                h_opt = slsqp_optimize(best_h, beta, max_iter=300)
                c_opt = compute_c5(h_opt)
                if c_opt < best_c:
                    best_h, best_c = h_opt, c_opt
            except Exception:
                continue

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
