    best_h = h.copy()

    lr = init_lr
    for it in range(max_iter):

        k_idx = np.argmax(corr)
        c_val = corr[k_idx] * (2.0 / len(best_h))
        if c_val < best_c - 1e-9:
            best_c = c_val
            # optional store
        # compute gradient g for this shift
        shift = k_idx - (len(best_h)-1)
        n = len(best_h)

            if shift < n:
                g[:n-shift] += (1 - best_h[shift:])
                g[shift:] -= best_h[:n-shift]
        else:
            s = -shift
            # negative shift

                break

                lr *= 0.5

            # if cannot improve, break

    return best_h, best_c

def compute_c5(h):

def project_to_feasible(v, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

    h = h0.copy()
    best_c = compute_c5(h)
    lr = init_lr
    n = h.size
    start = time.time()
    for it in range(max_iter):
        if time_limit is not None and (time.time() - start) > time_limit:
            break

        k_idx = np.argmax(corr)
        current_c = corr[k_idx] * (2.0 / n)
        if current_c < best_c - 1e-12:

def hill_climb_simple(h0, target, rng, max_iters=2000, time_limit=None):
    h = h0.copy()
    best_c = compute_c5(h)
    n = h.size
    step_scale = 0.3
    decay_every = max(1, max_iters // 10)
    start = time.time()
    for it in range(max_iters):
        if time_limit is not None and (time.time() - start) > time_limit:
            break
        # propose transfer
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * step_scale
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        # ensure feasibility (sum unchanged)
        c = compute_c5(new_h)
        if c < best_c:
            h = new_h
            best_c = c
        if (it + 1) % decay_every == 0:
            step_scale = max(step_scale * 0.9, 1e-5)
    return h, best_c

    h = h0.copy()
    best_c = compute_c5(h)
    lr = init_lr
    n = h.size
    start = time.time()
    for it in range(max_iter):
        if time_limit is not None and (time.time() - start) > time_limit:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = corr.max()
        c_curr = max_val * (2.0 / n)
        if c_curr < best_c - 1e-12:
            best_c = c_curr
        # find all shifts achieving max
        max_idxs = np.where(corr == max_val)[0]
        # compute average gradient
        g = np.zeros_like(h)
        for idx in max_idxs:
            shift = idx - (n - 1)
            g += gradient_for_shift(h, shift)
        g /= len(max_idxs)
        grad_norm = np.linalg.norm(g)
        if grad_norm == 0:
            break
        # step direction normalized

        # try line search

                break

                lr_trial *= 0.5
        if not improved:
            # reduce learning rate for next iteration
            lr *= 0.5
            if lr < 1e-5:
                break

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Global placeholders for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) via log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp (bc - m ≤ 0)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=500):
    """One SLSQP refinement using the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # exact sum correction (project onto the hyperplane Σh = target)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

# ---------- Feasibility utilities ----------
def project_to_feasible(v, target):
    """Project onto {0≤h≤1, Σh = target} via a bisection on a uniform shift."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def random_feasible_vector(n, rng):
    """Fast random feasible vector (0≤h≤1, Σh = n/2)."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_to_feasible(v, n / 2.0)

def upsample_initial(to_n):
    """Linear interpolation of the global initial_h_values to length to_n."""
    init = np.array(initial_h_values, dtype=float)
    n0 = init.size
    xs_old = np.linspace(0.0, 2.0, n0, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, to_n, endpoint=False)
    new = np.interp(xs_new, xs_old, init)
    return project_to_feasible(new, to_n / 2.0)

# ---------- Objective ----------
def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)

# ---------- Gradient for a given shift ----------
def gradient_for_shift(h, shift):
    """Sub‑gradient of the overlap for a particular shift."""
    n = h.size
    g = np.zeros_like(h)
    if shift >= 0:
        if shift < n:
            g[:n - shift] += (1 - h[shift:])
            g[shift:] -= h[:n - shift]
    else:
        s = -shift
        if s < n:
            g[s:] += (1 - h[:n - s])
            g[:n - s] -= h[s:]
    return g

# ---------- Gradient descent on the max‑overlap ----------
def gradient_descent(h0, target, max_iter=20000, init_lr=0.5, time_limit=None):
    """Projected gradient descent on the maximal overlap."""
    h = h0.copy()
    best_c = compute_c5(h)
    lr = init_lr
    n = h.size
    start = time.time()
    for _ in range(max_iter):
        if time_limit is not None and (time.time() - start) > time_limit:
            break
        corr = np.correlate(h, 1 - h, mode="full")
        max_val = corr.max()
        c_curr = max_val * (2.0 / n)
        if c_curr < best_c - 1e-12:
            best_c = c_curr
        # all shifts attaining the maximal value
        max_idxs = np.where(corr == max_val)[0]
        g = np.zeros_like(h)
        for idx in max_idxs:
            shift = idx - (n - 1)
            g += gradient_for_shift(h, shift)
        g /= len(max_idxs)
        norm = np.linalg.norm(g)
        if norm == 0.0:
            break
        direction = g / norm
        # line search
        improved = False
        lr_trial = lr
        for _ in range(10):
            step = lr_trial * direction
            h_new = project_to_feasible(h - step, target)
            c_new = compute_c5(h_new)
            if c_new < best_c - 1e-12:
                h = h_new
                best_c = c_new
                lr = min(lr * 1.2, 2.0)
                improved = True
                break
            lr_trial *= 0.5
        if not improved:
            lr *= 0.5
            if lr < 1e-5:
                break
    return h, best_c

# ---------- Simple hill‑climb (transfer ε) ----------
def hill_climb_simple(h0, target, rng, max_iters=2000, time_limit=None):
    """Transfer small ε between coordinates, preserving the sum."""
    h = h0.copy()
    best_c = compute_c5(h)
    n = h.size
    step_scale = 0.3
    decay_every = max(1, max_iters // 10)
    start = time.time()
    for it in range(max_iters):
        if time_limit is not None and (time.time() - start) > time_limit:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * step_scale
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c = compute_c5(new_h)
        if c < best_c:
            h = new_h
            best_c = c
        if (it + 1) % decay_every == 0:
            step_scale = max(step_scale * 0.9, 1e-5)
    return h, best_c

# ---------- Main optimisation for a fixed n ----------
def optimise_fixed_n(n, rng, total_time):
    target = n / 2.0
    start = time.time()
    # initial guess
    if "initial_h_values" in globals():
        h0 = upsample_initial(n)
    else:
        h0 = random_feasible_vector(n, rng)

    # -------------------------------------------------
    # Primary refinement: long gradient descent + hill‑climb
    primary_time = total_time * 0.7
    gd_time = primary_time * 0.5
    hc_time = primary_time - gd_time
    h_cur, c_cur = gradient_descent(h0, target, max_iter=200000,
                                    init_lr=0.5, time_limit=gd_time)
    h_cur, c_cur = hill_climb_simple(h_cur, target, rng,
                                     max_iters=10000, time_limit=hc_time)

    best_h, best_c = h_cur, c_cur

    # -------------------------------------------------
    # Random restarts while time permits
    while time.time() - start < total_time - 5.0:
        remaining = total_time - (time.time() - start)
        # short trial
        h_rand = random_feasible_vector(n, rng)
        gd_t = min(2.0, remaining * 0.4)
        h_try, c_try = gradient_descent(h_rand, target, max_iter=8000,
                                        init_lr=0.5, time_limit=gd_t)
        hc_t = min(1.0, remaining - (time.time() - start) - 0.5)
        h_try, c_try = hill_climb_simple(h_try, target, rng,
                                         max_iters=3000, time_limit=hc_t)
        if c_try < best_c:
            best_h, best_c = h_try, c_try

    # -------------------------------------------------
    # Final SLSQP polishing if time left
    if time.time() - start < total_time - 2.0:
        for beta in (10.0, 50.0, 200.0, 800.0, 2000.0, 8000.0):
            if time.time() - start > total_time - 0.5:
                break
            best_h = slsqp_optimize(best_h, beta, rng, max_iter=500)
            best_c = compute_c5(best_h)

    return best_h, best_c

# ---------- Entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
