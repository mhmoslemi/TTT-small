def gradient_descent_smooth_max(h, beta, steps, lr):
    for step in range(steps):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = corr.max()
        w = np.exp(beta * (corr - max_corr))
        w_sum = w.sum()
        w = w / w_sum
        w_flip = w[::-1]
        term1 = np.convolve(w_flip, 1 - h, mode='full')
        term2 = np.convolve(w, h, mode='full')
        grad = term1[(len(h)-1):(2*len(h)-1)] - term2[(len(h)-1):(2*len(h)-1)]
        # gradient descent step
        h = h - lr * grad
        # projected onto [0,1], enforce sum
        h = np.clip(h, 0.0, 1.0)
        enforce_sum(h, len(h)/2.0, rng)
    return h
```

def subgradient_descent(h0, lr0, beta_decay, steps):
    h = h0.copy()
    lr = lr0
    for t in range(steps):
        corr = np.correlate(h, 1 - h, mode='full')
        k = np.argmax(corr) - (len(h)-1)  # shift
        # compute subgradient
        grad = np.zeros_like(h)
        n = len(h)
        # For i where i + k < n:
        if k >= 0:
            grad[:n-k] += 1 - h[k:]
            grad[k:] -= h[:n-k]
        else:
            # k negative
            kk = -k
            grad[kk:] += 1 - h[:-kk]
            grad[:n-kk] -= h[kk:]
        # step
        h = h - lr * grad
        # clip & sum enforcement
        h = np.clip(h, 0, 1)
        enforce_sum(h, len(h)/2, rng)
        lr *= beta_decay
    return h
```

    # Use iterative method: subtract diff/n, then clip.
    for _ in range(10):
        diff = h.sum() - target
        if abs(diff) < 1e-9:
            break
        h -= diff / len(h)
        # clip
        np.clip(h, 0.0, 1.0, out=h)
    # final correction: if still slight diff, adjust randomly
    diff = h.sum() - target
    if abs(diff) > 1e-8:
        # distribute diff over non-boundary elements
        free = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
        if free.size > 0:
            adjustment = diff / free.size
            h[free] -= adjustment
            np.clip(h, 0.0, 1.0, out=h)
        else:
            # fallback to random pairwise

def compute_c5(h):
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()

import time
import math
import numpy as np

# ---------- basic helpers ----------
def compute_c5(h):
    """Discrete C5 for step function h (dx=2/n)."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)


def enforce_sum(h, target, rng):
    """Adjust h in‑place so that sum(h)=target while respecting 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def project_sum_box(h, target):
    """Project onto box [0,1] and the hyperplane sum=h_target."""
    np.clip(h, 0.0, 1.0, out=h)
    diff = h.sum() - target
    if abs(diff) > 1e-9:
        free = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
        if free.size:
            h[free] -= diff / free.size
            np.clip(h, 0.0, 1.0, out=h)
        else:  # fallback – rare
            enforce_sum(h, target, np.random.default_rng())
    return h


def random_feasible_vector(n, rng):
    """Generate a random h with 0≤h≤1 and sum=n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    enforce_sum(h, n / 2.0, rng)
    return h


# ---------- subgradient descent (max‑corr) ----------
def subgradient_descent(h0, rng, steps, lr, decay):
    h = h0.copy()
    n = h.size
    target = n / 2.0
    for _ in range(steps):
        corr = np.correlate(h, 1 - h, mode="full")
        k = np.argmax(corr) - (n - 1)           # shift attaining the max
        grad = np.zeros_like(h)

        if k >= 0:
            nk = n - k
            grad[:nk] += 1 - h[k:]
            grad[k:] -= h[:nk]
        else:
            kk = -k
            nk = n - kk
            grad[kk:] += 1 - h[:nk]
            grad[:nk] -= h[kk:]

        h -= lr * grad
        project_sum_box(h, target)
        lr *= decay
    return h


# ---------- smooth‑max descent (SLSQP‑like) ----------
def smooth_max_descent(h0, rng, beta, steps, lr):
    h = h0.copy()
    n = h.size
    target = n / 2.0
    for _ in range(steps):
        corr = np.correlate(h, 1 - h, mode="full")
        m = corr.max()
        w = np.exp(beta * (corr - m))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        w_flip = w[::-1]
        term1 = np.convolve(w_flip, 1 - h, mode="full")
        term2 = np.convolve(w, h, mode="full")
        start = n - 1
        end = start + n
        grad = term1[start:end] - term2[start:end]

        h -= lr * grad
        project_sum_box(h, target)
    return h


# ---------- up‑sampling ----------
def upsample_h(old, n_new):
    n_old = old.size
    if n_old == n_new:
        return old.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, old)
    enforce_sum(h_new, n_new / 2.0, np.random.default_rng())
    return h_new


# ---------- main routine ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ---- initialise -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        # use a finer discretisation for more flexibility
        n_target = max(360, n0)
        h = upsample_h(h0, n_target)
    else:
        n_target = 360
        h = random_feasible_vector(n_target, rng)

    project_sum_box(h, n_target / 2.0)
    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ---- stochastic subgradient exploration -------------------------
    main_time = budget_s * 0.80
    while time.time() - start_time < main_time:
        # choose a start point
        if rng.random() < 0.7:
            start = best_h + rng.normal(scale=0.02, size=n_target)
        else:
            start = random_feasible_vector(n_target, rng)
        project_sum_box(start, n_target / 2.0)

        # hyper‑parameters for this run
        steps = int(rng.integers(1200, 2600))
        lr = rng.uniform(0.008, 0.025)
        decay = rng.uniform(0.995, 0.999)

        h_new = subgradient_descent(start, rng, steps, lr, decay)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_h, best_c = h_new, c_new
            # a quick smooth‑max polish after a successful improvement
            for beta in (20, 80):
                h_pol = smooth_max_descent(best_h, rng, beta, steps=600,
                                           lr=0.1 / math.sqrt(beta))
                c_pol = compute_c5(h_pol)
                if c_pol < best_c:
                    best_h, best_c = h_pol, c_pol

    # ---- final polishing (high‑beta) --------------------------------
    for beta in (200, 800, 3200, 12800):
        if time.time() - start_time > budget_s:
            break
        best_h = smooth_max_descent(best_h, rng, beta,
                                    steps=1800,
                                    lr=0.1 / math.sqrt(beta))
        best_c = compute_c5(best_h)

    # ---- final projection & return ----------------------------------
    project_sum_box(best_h, n_target / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n_target)
