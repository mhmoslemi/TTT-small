def improve_binary(h, rng, now, start_time, time_limit):
    n = len(h)
    best_h = h.copy()
    best_c = compute_c5(best_h)
    # While time left and improvements possible:
    while time.time() - start_time < time_limit:
        # compute correlation, find worst shift:
        corr = np.correlate(best_h, 1 - best_h, mode="full")
        idx = np.argmax(corr)
        k = idx - (n - 1)
        # compute candidate i and j
        start_i = max(0, -k)
        end_i = min(n, n - k)
        i_range = np.arange(start_i, end_i)
        i_k = i_range + k
        # find candidates
        cond_i = (best_h[i_range] == 1) & (best_h[i_k] == 0)
        cond_j = (best_h[i_range] == 0) & (best_h[i_k] == 1)
        cand_i = i_range[cond_i]
        cand_j = i_range[cond_j]
        if cand_i.size == 0 or cand_j.size == 0:
            # cannot improve current worst shift; break
            break
        # choose random i and j from candidates
        i = rng.choice(cand_i)
        j = rng.choice(cand_j)
        # perform swap
        best_h[i] = 0
        best_h[j] = 1
        # compute new C5
        c_new = compute_c5(best_h)
        if c_new < best_c:
            best_c = c_new
        # continue (loop will recompute worst shift)
    return best_h, best_c

def improve_binary_targeted(h, rng, start_time, time_limit):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(best_h)
    while True:
        elapsed = time.time() - start_time
        if elapsed >= time_limit:
            break
        # compute worst shift
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        idx = np.argmax(corr)
        k = idx - (n - 1)
        # compute candidates
        start_i = max(0, -k)
        end_i = min(n, n - k)
        i_range = np.arange(start_i, end_i)
        i_k = i_range + k
        cond_i = (best_h[i_range] == 1) & (best_h[i_k] == 0)
        cond_j = (best_h[i_range] == 0) & (best_h[i_k] == 1)
        cand_i = i_range[cond_i]
        cand_j = i_range[cond_j]

        if cand_i.size > 0 and cand_j.size > 0:
            i = rng.choice(cand_i)
            j = rng.choice(cand_j)
            best_h[i] = 0
            best_h[j] = 1
            # update c
            c_new = compute_c5(best_h)
            if c_new < best_c:
                best_c = c_new
            # continue loop
        else:
            # no direct improvement for this worst shift; break out of loop
            break
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utilities ----------
def enforce_sum(h, target, rng):
    """Clip to [0,1] and adjust to make sum exactly target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:  # need to increase some entries
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i])
            add *= rng.random()
            h[i] += add
            diff -= add
        else:  # need to decrease
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i])
            sub *= rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    """Uniform random vector in [0,1] with exact sum n/2."""
    h = rng.random(n)
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, n / 2.0, rng)
    return h

def perturb_random(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def compute_c5(h):
    """Exact discrete C₅."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

# ---------- hill climbing ----------
def hill_climb_targeted(h_start, rng, max_iters, start_step, target_sum):
    """Targeted gradient hill‑climb on the worst shift."""
    h = h_start.copy()
    n = h.size
    best_h = h.copy()
    corr = np.correlate(h, 1 - h, mode="full")
    best_c = np.max(corr) * (2.0 / n)

    step = start_step
    decay_interval = max(1, max_iters // 10)

    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr)
        k = idx - (n - 1)

        # gradient of the worst shift (approximation used by perturb_targeted)
        grad = np.ones(n)
        if k >= 0:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
        else:
            kk = -k
            grad[: n - kk] -= h[kk:]
            grad[kk:] -= h[: n - kk]

        # targeted perturbation
        pos = np.where((grad > 1e-9) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-9) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            new_h = perturb_random(h, rng, scale=step)
        else:
            i = rng.choice(pos)
            j = rng.choice(neg)
            max_eps = min(h[i], 1.0 - h[j])

                new_h = h.copy()
                new_h[i] -= eps

        corr_new = np.correlate(new_h, 1 - new_h, mode="full")

        if c_new < (np.max(corr) * (2.0 / n)):
            h = new_h

        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)

    return best_h, best_c

# ---------- SLSQP with soft‑max barrier ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x
