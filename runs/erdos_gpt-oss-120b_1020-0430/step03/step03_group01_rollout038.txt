Given n, lags idx = np.arange(-(n-1), n) (length L = 2*n-1). For each lag l, compute w[l] as exp(beta * corr[l] - max_bc) / sum_exp (these are weights). For each lag, compute pos and neg arrays:

def soft_obj_and_grad(h, beta):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    bc = beta * corr
    max_bc = np.max(bc)
    exp_bc = np.exp(bc - max_bc)
    sum_exp = np.exp(bc - max_bc).sum()
    obj = (max_bc + np.log(sum_exp)) / beta
    w = exp_bc / sum_exp
    grad = np.zeros_like(h, dtype=float)
    # loop over lags
    # idx=0 corresponds to lag = -(n-1)
    for offset, weight in enumerate(w):
        lag = offset - (n-1)
        if lag >= 0:
            # pos: i from 0..n-1-lag
            # pos[i] = 1 - h[i+lag]
            grad[:n-lag] += weight * (1 - h[lag:])
            # neg: i from lag..n-1
            # neg[i] = h[i-lag]
            grad[lag:] -= weight * h[:n-lag]
        else:
            pos_len = n + lag  # since lag negative
            # pos i from -lag..n-1
            # For i from -lag..n-1: i+lag is >=0.
            # Equivalent to i from -lag to n-1; index i+lag = i+lag.
            # Actually implement similar using shifted slices.
            # When lag negative, lag = -k, k>0
            k = -lag
            # pos: i from k..n-1, i+lag = i-k
            grad[k:] += weight * (1 - h[:n-k])
            # neg: i from 0..n-k-1, i-lag = i + k
            grad[:n-k] -= weight * h[k:]
    return obj, grad

def compute_soft_obj_and_grad(h, beta):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    bc = beta * corr
    max_bc = np.max(bc)
    exp_bc = np.exp(bc - max_bc)
    sum_exp = np.sum(exp_bc)
    obj = (max_bc + np.log(sum_exp)) / beta
    w = exp_bc / sum_exp
    grad = np.zeros_like(h)
    # loop as before using n
    for offset, weight in enumerate(w):
        lag = offset - (n - 1)
        if lag >= 0:
            grad[:n - lag] += weight * (1 - h[lag:])
            grad[lag:] -= weight * h[:n - lag]
        else:
            k = -lag
            grad[k:] += weight * (1 - h[:n - k])
            grad[:n - k] -= weight * h[k:]
    return obj, grad

def project_sum_bounds(h, target):
    # Clip already done.
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    # Determine indices that are not at bounds
    free = (h > 0) & (h < 1)
    if np.all(~free):
        # all at bounds; adjust uniformly (may break bounds)
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        return h
    # Solve water-filling
    # allocate diff equally across free indices, but may hit bounds; iterate.
    while abs(diff) > 1e-12:
        free_idx = np.where(free)[0]
        if len(free_idx) == 0:
            # can't adjust, break
            break
        add = diff / len(free_idx)
        # Apply add
        h[free_idx] += add
        # Clip any that exceed bounds and update diff
        over = h > 1
        under = h < 0
        if np.any(over):
            excess = h[over] - 1
            diff -= excess.sum()
            h[over] = 1
        if np.any(under):
            deficit = h[under]
            diff += deficit.sum()
            h[under] = 0
        # Update free mask
        free = (h > 0) & (h < 1)
        total = h.sum()
        diff = target - total
        if np.abs(diff) < 1e-12:
            break
    return h

import time
import numpy as np
from scipy.optimize import minimize, differential_evolution

# ---------- helpers ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality: sum(x) == CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k corr_k."""
    n = x.size
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    s = np.exp(bc - m).sum()
    return (m + np.log(s)) / CURRENT_BETA


def softmax_grad(x):
    """Gradient of softmax_obj."""
    n = x.size
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    exp_bc = np.exp(bc - m)
    w = exp_bc / exp_bc.sum()
    grad = np.zeros_like(x)
    for k, wk in enumerate(w):
        lag = k - (n - 1)
        if lag >= 0:
            grad[: n - lag] += wk * (1 - x[lag:])
            grad[lag:] -= wk * x[: n - lag]
        else:
            q = -lag
            grad[q:] += wk * (1 - x[: n - q])
            grad[: n - q] -= wk * x[q:]
    return grad


def compute_c5(h):
    """Exact discrete C₅ (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=2000):
    """Adjust h to have Σh = target while staying in [0,1]."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        # final uniform correction
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Random h ∈ [0,1]^n with Σh = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def compute_soft_obj_and_grad(h, beta):
    """Softmax objective + gradient for given beta."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    bc = beta * corr
    m = np.max(bc)
    exp_bc = np.exp(bc - m)
    sum_exp = exp_bc.sum()
    obj = (m + np.log(sum_exp)) / beta
    w = exp_bc / sum_exp
    grad = np.zeros_like(h)
    for k, wk in enumerate(w):
        lag = k - (n - 1)
        if lag >= 0:
            grad[: n - lag] += wk * (1 - h[lag:])
            grad[lag:] -= wk * h[: n - lag]
        else:
            q = -lag
            grad[q:] += wk * (1 - h[: n - q])
            grad[: n - q] -= wk * h[q:]
    return obj, grad


def slsqp_optimize(h0, beta, max_iter=400):
    """SLSQP on smooth surrogate (with analytical gradient)."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        jac=softmax_grad,
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x += diff / n
        x = np.clip(x, 0.0, 1.0)
    return x


# ---------- main ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # ----- base size and initial candidates -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 360
        xs = (np.arange(n) + 0.5) * (2.0 / n)
        h0 = 0.5 + 0.5 * np.cos(np.pi * xs)  # cosine start
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = enforce_sum(h0, n / 2.0, rng)

    xs = (np.arange(n) + 0.5) * (2.0 / n)
    # triangle shape (peak 1 at centre)
    tri = 1.0 - np.abs(xs - 1.0)
    tri = np.clip(tri, 0.0, 1.0)
    tri = enforce_sum(tri, n / 2.0, rng)
    # cosine shape (same as h0 if not already)
    cos = 0.5 + 0.5 * np.cos(np.pi * xs)
    cos = np.clip(cos, 0.0, 1.0)
    cos = enforce_sum(cos, n / 2.0, rng)

    candidates = [h0, tri, cos]
    for _ in range(3):
        candidates.append(random_feasible_vector(n, rng))

    best_h = None
    best_c = float("inf")
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    global_best_h, global_best_c = best_h, best_c

    target = n / 2.0
    beta_schedule = [50, 200, 800, 2000, 5000]

    # ----- main optimisation loop -----
    while time.time() - start < budget_s * 0.8:
        # start from a perturbed best
        h = perturb_h(global_best_h, rng, scale=0.3)
        # simple projected gradient descent on surrogate
        beta = 200
        for it in range(250):
            obj, grad = compute_soft_obj_and_grad(h, beta)
            grad -= grad.mean()               # keep sum invariant in linear step
            step = 0.6 / np.sqrt(it + 1)      # tuned step size
            h_new = h - step * grad
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = enforce_sum(h_new, target, rng)
            c_new = compute_c5(h_new)
            if c_new < global_best_c:
                global_best_c = c_new
                global_best_h = h_new.copy()
            # accept if better, else occasional acceptance
            if c_new < compute_c5(h) or rng.random() < 0.01:
                h = h_new

        # SLSQP refinement with increasing sharpness
        for beta in beta_schedule:
            if time.time() - start > budget_s * 0.8:
                break
            try:
                h_ref = slsqp_optimize(global_best_h, beta, max_iter=400)
                c_ref = compute_c5(h_ref)
                if c_ref < global_best_c:
                    global_best_c = c_ref
                    global_best_h = h_ref.copy()
            except Exception:
                pass

    # ----- optional up‑sampling and fine‑tuning -----
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    if remaining > budget_s * 0.15:
        target_n = min(720, n * 2)
        if target_n != n:
            old_x = (np.arange(n) + 0.5) * (2.0 / n)
            new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
            up_h = np.interp(new_x, old_x, global_best_h)
            up_h = enforce_sum(up_h, target_n / 2.0, rng)
            # brief hill‑climb on finer grid
            for _ in range(1200):
                up_h = perturb_h(up_h, rng, scale=0.15)
                up_h = enforce_sum(up_h, target_n / 2.0, rng)
                c_up = compute_c5(up_h)
                if c_up < global_best_c:
                    global_best_c = c_up
                    global_best_h = up_h.copy()
                    n = target_n
            # final SLSQP on fine grid
            for beta in [2000, 5000]:
                try:
                    h_ref = slsqp_optimize(global_best_h, beta, max_iter=300)
                    c_ref = compute_c5(h_ref)
                    if c_ref < global_best_c:
                        global_best_c = c_ref
                        global_best_h = h_ref.copy()
                except Exception:
                    pass

    # ----- Fourier‑basis post‑processing (small DE) -----
    remaining = budget_s - (time.time() - start)
    if remaining > budget_s * 0.1:
        F = 3                                 # use three frequencies
        dim = 1 + 2 * F
        bounds = [(-2.0, 2.0)] * dim

        def coeff_to_h(coeff):
            h = coeff[0] + np.zeros_like(xs)
            for f in range(1, F + 1):
                c_cos = coeff[2 * f - 1]
                c_sin = coeff[2 * f]
                h += c_cos * np.cos(np.pi * f * xs)
                h += c_sin * np.sin(np.pi * f * xs)
            h = np.clip(h, 0.0, 1.0)
            h = enforce_sum(h, target, rng)
            return h

        def de_obj(coeff):
            return compute_c5(coeff_to_h(coeff))

        try:
            de_res = differential_evolution(
                de_obj,
                bounds,
                maxiter=30,
                popsize=12,
                seed=int(rng.integers(2**31)),
                polish=False,
                updating="deferred",
                workers=1,
            )
            h_de = coeff_to_h(de_res.x)
            c_de = compute_c5(h_de)
            if c_de < global_best_c:
                global_best_c = c_de
                global_best_h = h_de.copy()
        except Exception:
            pass

    # ----- final clean‑up -----
    global_best_h = np.clip(global_best_h, 0.0, 1.0)
    global_best_h = enforce_sum(global_best_h, n / 2.0, rng)
    final_c = compute_c5(global_best_h)

    return global_best_h.tolist(), float(final_c), int(n)
