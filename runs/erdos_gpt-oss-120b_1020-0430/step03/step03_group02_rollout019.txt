        for it in range(max_iters_per_beta):

            g = beta * corr
            maxg = np.max(g)

            w = exp_g / exp_g.sum()

import time, numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / h.size)

def enforce_sum(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1 - 2 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]    -= h[: n - k]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h, rng, max_iters=2000, step_scale=0.5):
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
def compute_c5(h):
    """Discrete C5 for step function on [0,2] (h size = n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, rng):
    """Project h onto {0≤h≤1, Σh = n/2} by random redistribution."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h):
    """Sub‑gradient of the worst shift overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

    """Transfer a small amount from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h, rng, max_iters=2000, step_scale=0.5):

    best_h = h.copy()

    """Projected gradient descent on the soft‑max approximation of max_k."""
    n = h.size
    best_h = h.copy()

            w_rev = w[::-1]

            h_up, c_up = guided_hill_climb(h_up, rng, max_iters=2000, step_scale=0.3)
            if c_up < best_c:
                best_h, best_c, best_n = h_up.copy(), c_up, h_up.size

    # ------------------------------------------------------------------
    # final intensive local search on the current best solution
    # ------------------------------------------------------------------
    while time.time() - start < budget_s * 0.999 and best_h is not None:
        # occasional full random restart
        if rng.random() < 0.02:
            h_rand = rng.random(best_n)
            h_rand = np.clip(h_rand, 0.0, 1.0)
            h_rand = enforce_sum(h_rand, rng)
            h_rand, c_rand = guided_hill_climb(h_rand, rng, max_iters=800, step_scale=0.5)
            if c_rand < best_c:
                best_h, best_c = h_rand.copy(), c_rand
            continue

        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

        # occasional soft‑max descent sweep
        if rng.random() < 0.1:
            h_tmp, c_tmp = softmax_gradient_descent(
                best_h.copy(),
                rng,
                beta_seq=[5, 15, 30, 60, 120],
                iters_per_beta=200,
                step0=0.1,
            )
            if c_tmp < best_c:
                best_h, best_c = h_tmp.copy(), c_tmp

    # ------------------------------------------------------------------
    # final feasibility enforcement and return
    # ------------------------------------------------------------------
    if best_h is None:
        best_n = 84
        best_h = np.full(best_n, 0.5)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
