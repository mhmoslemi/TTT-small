def correlate_fft(x, y):
    n = x.size + y.size - 1
    m = 1 << (n - 1).bit_length()
    X = np.fft.rfft(x, m)
    Y = np.fft.rfft(y[::-1], m)  # reversed y
    c = np.fft.irfft(X * Y, m)
    return c[:n]

import time
import numpy as np

# ---------- projection onto [0,1] and sum ----------
def enforce_sum_box(u, target):
    """Project vector u onto the intersection of the box [0,1]^n and the hyperplane sum = target."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        v = np.clip(u - mid, 0.0, 1.0)
        s = v.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    v = np.clip(u - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - v.sum()
    if abs(diff) > 1e-12:
        free = np.where((v > 1e-12) & (v < 1 - 1e-12))[0]
        if free.size:
            v[free[0]] += diff
            v = np.clip(v, 0.0, 1.0)
    return v

def random_feasible(n, rng):
    return enforce_sum_box(rng.random(n), n / 2.0)

# ---------- FFT‑based convolution / correlation ----------
def conv_fft(a, b):
    n = a.size + b.size - 1
    m = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, m)
    B = np.fft.rfft(b, m)
    c = np.fft.irfft(A * B, m)
    return c[:n]

def correlate_fft(a, b):
    return conv_fft(a, b[::-1])

# ---------- exact C5 evaluation ----------
def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / h.size)

# ---------- main optimisation ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----- target resolution -----
    target_n = 720

    # ----- initialise h -----

            xs_new = np.arange(target_n) / target_n * 2.0
            h0 = np.interp(xs_new, xs_old, h0)
        h = enforce_sum_box(h0, target_n / 2.0)
    else:
        h = random_feasible(target_n, rng)

    n = target_n
    best_h = h.copy()
    best_c = compute_c5(best_h)

    beta = 200.0          # smoothing parameter
    step = 0.5            # gradient step scale
    it = 0
    last_improve = 0

    # ----- main loop -----
    while time.time() - start_time < budget_s * 0.94:
        # correlation and soft‑max weights
        corr = correlate_fft(h, 1.0 - h)
        bc = beta * corr
        max_bc = bc.max()
        w_unnorm = np.exp(bc - max_bc)
        w_sum = w_unnorm.sum()
        w = w_unnorm / w_sum

        # gradient of the smoothed objective
        A = conv_fft(w, 1.0 - h)[n - 1 : n - 1 + n]

        grad = (2.0 / n) * (A - B)
