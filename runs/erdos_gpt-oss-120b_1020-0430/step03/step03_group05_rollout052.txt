def gradient_max_step(h, eta):
    # compute correlation and argmax
    corr = np.correlate(h, 1 - h, mode='full')
    k = np.argmax(corr) # index in [0, 2n-2]; shift = k - (n-1)
    shift = k - (len(h) - 1)
    n = len(h)
    # compute g
    g = np.ones_like(h)
    # add contributions of h[i+shift] and h[i-shift] where in bounds
    if shift >= 0:
        g[:n-shift] -= h[shift:]
        g[shift:] -= h[:n-shift]
    else:
        s = -shift
        g[s:] -= h[:n-s]
        g[:n-s] -= h[s:]
    # gradient descent step
    new_h = h - eta * g
    # clip, project onto sum
    new_h = np.clip(new_h, 0.0, 1.0)
    new_h = project_to_feasible(new_h, target=len(h)/2.0)
    return new_h

def de_optimize_general(n, rng, time_limit):
    target = n/2
    popsize = max(30, int(0.2*n))
    # initial pop:
    pop = []
    if 'initial_h_values' in globals():
        init = upsample_initial(n)
        init = project_to_feasible(init, target)
        pop.append(init)
    # add sinusoidal:
    xs = np.linspace(0, 2, n, endpoint=False)
    sin = 0.5 + 0.5*np.sin(2*np.pi*xs/2)  # period 2? Actually x from 0 to2, sin(π*x) maybe.
    # let's use sin = 0.5 + 0.5*np.sin(np.pi*xs) => ranges [0,1].
    sin = 0.5 + 0.5*np.sin(np.pi*xs)
    sin = project_to_feasible(sin, target)
    pop.append(sin)
    # fill remaining with random feasible.
    while len(pop) < popsize:
        pop.append(random_feasible_vector(n, rng))
    pop = np.array(pop)
    scores = np.array([compute_c5(ind) for ind in pop])
    best_idx = scores.argmin()
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]
    # DE loop
    while time.time() < time_limit:
        for i in range(popsize):
            # mutation: selects three distinct random individuals
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, 3, replace=False)
            # adaptive parameters
            F = rng.uniform(0.4, 0.9)  # mutation factor
            CR = rng.uniform(0.0, 1.0)  # crossover probability
            mutant = pop[r1] + F*(pop[r2] - pop[r3])
            # binomial crossover
            cross = rng.random(n) < CR
            # ensure at least one dimension from mutant
            if not np.any(cross):
                cross[rng.integers(0,n)] = True
            trial = np.where(cross, mutant, pop[i])
            # project to feasible
            trial = project_to_feasible(trial, target)
            trial_score = compute_c5(trial)
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_h = trial.copy()
    return best_h, best_score

def gradient_refine(h, max_iter, rng, time_limit):
    n = h.size
    target = n/2.0
    best_h = h.copy()
    best_score = compute_c5(best_h)
    eta = 0.01
    # simple decreasing schedule: after certain no improvements, reduce eta.
    no_improve = 0
    start = time.time()
    while time.time() < time_limit and no_improve < 200 and max_iter>0:
        # compute max shift
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k = np.argmax(corr)
        # shift:
        shift = k - (n-1)
        # compute gradient g
        g = np.ones_like(best_h)
        # apply contributions
        if shift >= 0:
            # overlapping indices:
            g[:n-shift] -= best_h[shift:]
            g[shift:] -= best_h[:n-shift]
        else:
            s = -shift
            g[s:] -= best_h[:n-s]
            g[:n-s] -= best_h[s:]
        # step
        new_h = best_h - eta * g
        # clip and project
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = project_to_feasible(new_h, target)
        new_score = compute_c5(new_h)
        if new_score < best_score - 1e-6:
            best_h = new_h
            best_score = new_score
            no_improve = 0
            # potentially increase eta slightly
            eta = min(0.05, eta*1.05)
        else:
            no_improve += 1
            eta = max(0.001, eta*0.7)
        max_iter -= 1
    return best_h, best_score

def de_optimize_time_limited(n, rng, end_time):
    target = n/2.0
    popsize = max(30, min(50, int(0.2*n)))  # maybe
    pop = []
    # include upsampled initial if present
    if 'initial_h_values' in globals():
        init = upsample_initial(n)
        init = project_to_feasible(init, target)
        pop.append(init)
    # include sinusoidal guess
    xs = np.linspace(0, 2, n, endpoint=False)
    sin_guess = 0.5 + 0.5*np.sin(np.pi*xs)  # range [0,1]
    sin_guess = project_to_feasible(sin_guess, target)
    pop.append(sin_guess)
    # fill random
    while len(pop) < popsize:
        pop.append(random_feasible_vector(n, rng))
    pop = np.array(pop)
    scores = np.array([compute_c5(ind) for ind in pop])
    best_idx = scores.argmin()
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]
    while time.time() < end_time:
        for i in range(popsize):
            if time.time() >= end_time:
                break
            # Mutation
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, 3, replace=False)
            F = rng.uniform(0.4, 0.9)
            CR = rng.random()
            mutant = pop[r1] + F*(pop[r2] - pop[r3])
            # Crossover
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(0, n)] = True
            trial = np.where(cross, mutant, pop[i])
            trial = project_to_feasible(trial, target)
            trial_score = compute_c5(trial)
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_h = trial.copy()
    return best_h, best_score

def gradient_refine_time_limited(h_start, rng, end_time):
    h = h_start.copy()
    n = h.size
    target = n/2.0
    best_h = h.copy()
    best_score = compute_c5(best_h)
    eta = 0.01
    no_improve = 0
    while time.time() < end_time:
        # compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        k = np.argmax(corr)
        shift = k - (n - 1)
        # compute gradient
        g = np.ones_like(h)
        if shift >= 0:
            # shift forward
            end = n - shift
            if end > 0:
                g[:end] -= h[shift:]
                g[shift:] -= h[:end]
        else:
            s = -shift
            end = n - s
            if end > 0:
                g[s:] -= h[:end]
                g[:end] -= h[s:]
        # step
        new_h = h - eta * g
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = project_to_feasible(new_h, target)
        new_score = compute_c5(new_h)
        if new_score < best_score - 1e-8:
            h = new_h
            best_h = new_h.copy()
            best_score = new_score
            no_improve = 0
            eta = min(0.05, eta * 1.05)
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                eta = max(0.001, eta * 0.7)
            # maybe occasional random perturb
            if rng.random() < 0.01:
                # small random jitter
                new_h = h + rng.normal(scale=0.01, size=n)
                new_h = np.clip(new_h, 0.0, 1.0)
                new_h = project_to_feasible(new_h, target)
                new_score = compute_c5(new_h)
                if new_score < best_score:
                    h = new_h
                    best_h = new_h.copy()
                    best_score = new_score
                    no_improve = 0
    return best_h, best_score

import time
import numpy as np

# ---------- Basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for step function h (dx = 2/len(h))."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_to_feasible(v: np.ndarray, target: float) -> np.ndarray:
    """Projection onto {0≤h≤1, Σh = target} via bisection on a shift."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector satisfying 0≤h≤1 and Σh = n/2."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_to_feasible(v, n / 2.0)


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def upsample_initial(to_n: int) -> np.ndarray:
    """Linear interpolation of the global initial_h_values (if present)."""
    init = np.array(initial_h_values, dtype=float)
    m = init.size
    xs_old = np.linspace(0.0, 2.0, m, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, to_n, endpoint=False)
    return np.interp(xs_new, xs_old, init)

        if vec.size != n:
            vec = np.interp(
                np.linspace(0.0, 2.0, n, endpoint=False),
                np.linspace(0.0, 2.0, vec.size, endpoint=False),
                vec,
            )
        vec = np.clip(vec, 0.0, 1.0)
        vec = project_to_feasible(vec, target)
        pop.append(vec)

    while len(pop) < popsize:
        pop.append(random_feasible_vector(n, rng))

    pop = np.array(pop)
    scores = np.array([compute_c5(ind) for ind in pop])
    best_idx = scores.argmin()
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]

    # DE main loop
    while time.time() < end_time:
        for i in range(popsize):
            if time.time() >= end_time:
                break
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, 3, replace=False)
            F = rng.uniform(0.4, 0.9)
            CR = rng.random()
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(0, n)] = True
            trial = np.where(cross, mutant, pop[i])
            trial = project_to_feasible(trial, target)
            trial_score = compute_c5(trial)
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_h = trial.copy()
    return best_h, best_score


# ---------- Gradient refinement on the current maximal shift ----------
def gradient_refine_time_limited(
    h_start: np.ndarray,
    rng: np.random.Generator,
    end_time: float,
) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the currently maximal overlap shift."""
    h = h_start.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(best_h)
    eta = 0.01
    no_improve = 0

    while time.time() < end_time:
        corr = np.correlate(h, 1 - h, mode="full")
        k = np.argmax(corr)
        shift = k - (n - 1)

        # gradient of the max‑shift overlap
        g = np.ones_like(h)
        if shift >= 0:
            end = n - shift
            if end > 0:
                g[:end] -= h[shift:]
                g[shift:] -= h[:end]
        else:
            s = -shift
            end = n - s
            if end > 0:
                g[s:] -= h[:end]
                g[:end] -= h[s:]

        new_h = h - eta * g
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = project_to_feasible(new_h, target)
        new_score = compute_c5(new_h)

        if new_score < best_score - 1e-8:
            h = new_h
            best_h = new_h.copy()
            best_score = new_score
            no_improve = 0
            eta = min(0.05, eta * 1.05)
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                eta = max(0.001, eta * 0.7)
            # occasional random jitter to escape flat regions
            if rng.random() < 0.01:
                jitter = h + rng.normal(scale=0.01, size=n)
                jitter = np.clip(jitter, 0.0, 1.0)
                jitter = project_to_feasible(jitter, target)
                jitter_score = compute_c5(jitter)
                if jitter_score < best_score:
                    h = jitter
                    best_h = jitter.copy()
                    best_score = jitter_score
                    no_improve = 0
                    eta = min(0.05, eta * 1.05)
    return best_h, best_score


# ---------- Final deterministic hill‑climb ----------
def hill_climb_time_limited(
    h_start: np.ndarray,
    rng: np.random.Generator,
    end_time: float,
    start_step: float = 0.3,
) -> tuple[np.ndarray, float]:
    """Greedy hill‑climb with decreasing step size."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(best_h)
    step = start_step
    n = h.size
    it = 0
    while time.time() < end_time:
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
            step = max(step * 0.9, 1e-4)
        else:
            # tiny chance to wander
            if rng.random() < 0.01:
                h = new_h
        it += 1
        if it % 500 == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


# ---------- Main driver ----------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    best_h = None
    best_c = np.inf
    best_n = None

    # ---------- Stage 1 : coarse (n=120) ----------
    n1 = 120
    remaining = budget_s - (time.time() - start_time)
    if remaining > 80:
        end_de = start_time + min(remaining * 0.35, 130.0)
        h1, c1 = de_optimize_time_limited(n1, rng, end_de)
        # gradient refinement
        end_grad = start_time + min(budget_s - (time.time() - start_time), 90.0)
        h1, c1 = gradient_refine_time_limited(h1, rng, end_grad)
        # hill‑climb
        end_hc = start_time + min(budget_s - (time.time() - start_time), 70.0)
        h1, c1 = hill_climb_time_limited(h1, rng, end_hc)
        if c1 < best_c:
            best_c, best_h, best_n = c1, h1.copy(), n1

    # ---------- Stage 2 : medium (n=240) ----------
    n2 = 240
    remaining = budget_s - (time.time() - start_time)
    if remaining > 150:
        # upsample best from previous stage if any
        extra = None
        if best_h is not None and best_n != n2:
            extra = np.interp(
                np.linspace(0.0, 2.0, n2, endpoint=False),
                np.linspace(0.0, 2.0, best_n, endpoint=False),
                best_h,
            )
        end_de = start_time + min(remaining * 0.40, 200.0)
        h2, c2 = de_optimize_time_limited(n2, rng, end_de, extra_vec=extra)
        # gradient refinement
        end_grad = start_time + min(budget_s - (time.time() - start_time), 120.0)
        h2, c2 = gradient_refine_time_limited(h2, rng, end_grad)
        # hill‑climb
        end_hc = start_time + min(budget_s - (time.time() - start_time), 80.0)
        h2, c2 = hill_climb_time_limited(h2, rng, end_hc)
        if c2 < best_c:
            best_c, best_h, best_n = c2, h2.copy(), n2

    # ---------- Stage 3 : fine (n=300) ----------
    n3 = 300
    remaining = budget_s - (time.time() - start_time)
    if remaining > 130:
        extra = None
        if best_h is not None and best_n != n3:
            extra = np.interp(
                np.linspace(0.0, 2.0, n3, endpoint=False),
                np.linspace(0.0, 2.0, best_n, endpoint=False),
                best_h,
            )
        end_de = start_time + min(remaining * 0.45, 250.0)
        h3, c3 = de_optimize_time_limited(n3, rng, end_de, extra_vec=extra)
        # gradient refinement
        end_grad = start_time + min(budget_s - (time.time() - start_time), 150.0)
        h3, c3 = gradient_refine_time_limited(h3, rng, end_grad)
        # hill‑climb
        end_hc = start_time + min(budget_s - (time.time() - start_time), 80.0)
        h3, c3 = hill_climb_time_limited(h3, rng, end_hc)
        if c3 < best_c:
            best_c, best_h, best_n = c3, h3.copy(), n3

    # ---------- Final sanity projection ----------
    if best_h is None:
        best_n = 120
        best_h = np.full(best_n, 0.5)

    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    if abs(best_h.sum() - target) > 1e-12:
        best_h = project_to_feasible(best_h, target)

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
