def subgradient_descent(h_start, rng, max_iters, step0):
    h = h_start.copy()
    n = h.size
    step = step0
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = 0.9995 # per iteration, maybe compute after some steps
    for it in range(max_iters):
        # compute worst shift
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr)
        k = idx - (n - 1)
        # compute gradient same as earlier
        grad = np.ones(n)
        if k >= 0:
            grad[:n - k] -= h[k:]
            grad[k:] -= h[:n - k]
        else:
            kk = -k
            grad[:n - kk] -= h[kk:]
            grad[kk:] -= h[:n - kk]

import time
import numpy as np
from scipy.optimize import minimize

# ---------- global vars for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

# ---------- utilities ----------
def enforce_sum(h, target, rng=None):
    """Clip to [0,1] and adjust to hit the exact target sum."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        caps = 1.0 - h
        cap_sum = caps.sum()
        if cap_sum < 1e-12:
            i = rng.integers(len(h))
            h[i] = min(1.0, h[i] + diff)
        else:
            add = diff * caps / cap_sum
            h += add
    else:
        caps = h
        cap_sum = caps.sum()
        if cap_sum < 1e-12:
            i = rng.integers(len(h))
            h[i] = max(0.0, h[i] + diff)
        else:
            sub = (-diff) * caps / cap_sum
            h -= sub
    h = np.clip(h, 0.0, 1.0)
    # tiny residual fix
    if abs(h.sum() - target) > 1e-10 and rng is not None:
        i = rng.integers(len(h))
        h[i] = np.clip(h[i] + (target - h.sum()), 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    h = rng.random(n)
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)

def perturb_random(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def perturb_targeted(h, grad, rng, scale=0.2):
    pos = np.where((grad > 1e-9) & (h > 1e-12))[0]
    neg = np.where((grad < -1e-9) & (h < 1.0 - 1e-12))[0]
    if pos.size == 0 or neg.size == 0:
        return perturb_random(h, rng, scale)
    i = rng.choice(pos)
    j = rng.choice(neg)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return perturb_random(h, rng, scale)
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def hill_climb_targeted(h_start, rng, max_iters=1500, start_step=0.5, target_sum=None):
    h = h_start.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr)
        k = idx - (n - 1)
        # gradient of the worst shift
        grad = np.ones(n)
        if k >= 0:
            grad[:n - k] -= h[k:]
            grad[k:] -= h[:n - k]
        else:
            kk = -k
            grad[:n - kk] -= h[kk:]
            grad[kk:] -= h[:n - kk]
        new_h = perturb_targeted(h, grad, rng, scale=step)
        c_new = compute_c5(new_h)
        if c_new < best_c:
            best_c = c_new
            best_h = new_h.copy()
        if c_new < compute_c5(h):
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

def subgradient_descent(h_start, rng, max_iters=2000, step0=0.4):
    h = h_start.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step0
    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr)
        k = idx - (n - 1)
        grad = np.ones(n)
        if k >= 0:
            grad[:n - k] -= h[k:]
            grad[k:] -= h[:n - k]
        else:
            kk = -k
            grad[:n - kk] -= h[kk:]
            grad[kk:] -= h[:n - kk]
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size and neg.size:
            i = rng.choice(pos)
            j = rng.choice(neg)
            max_eps = min(h[i], 1.0 - h[j])
            eps = max_eps * step * rng.random()
            h[i] -= eps
            h[j] += eps
        # occasional random jitter
        if rng.random() < 0.03:
            i2 = rng.integers(0, n)
            j2 = rng.integers(0, n)
            if i2 != j2:
                max_eps2 = min(h[i2], 1.0 - h[j2])
                eps2 = max_eps2 * step * rng.random()
                h[i2] -= eps2
                h[j2] += eps2
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        step *= 0.9995
    return best_h, best_c

def slsqp_optimize(h0, beta, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    total_budget = budget_s

    # ----- initialise at base resolution (360 if supplied else 180) -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        h0 = enforce_sum(np.clip(h0, 0.0, 1.0), n0 / 2.0, rng)
    else:
        n0 = 180
        h0 = random_feasible_vector(n0, rng)

    # ----- upscale to 720 points -----
    n = n0 * 2
    xs0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.interp(xs, xs0, h0)
    h = enforce_sum(np.clip(h, 0.0, 1.0), n / 2.0, rng)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ----- primary stochastic search (up to 70% of budget) -----
    while time.time() - start_time < total_budget * 0.70:
        # time‑aware iteration count
        remaining = total_budget - (time.time() - start_time)
        # adapt number of inner steps: more when we have plenty of time
        max_iters = 2500 if remaining > total_budget * 0.20 else 800
        if rng.random() < 0.65:
            start = perturb_random(best_h, rng, scale=0.35)
        else:
            start = random_feasible_vector(n, rng)
        h_local, c_local = hill_climb_targeted(
            start, rng, max_iters=max_iters, start_step=0.6, target_sum=n / 2.0
        )
        if c_local < best_c:
            best_h, best_c = h_local, c_local
        # occasional sub‑gradient refinement
        if rng.random() < 0.25:
            h_sub, c_sub = subgradient_descent(best_h, rng, max_iters=1500, step0=0.5)
            if c_sub < best_c:
                best_h, best_c = h_sub, c_sub

    # ----- SLSQP refinement with extended beta schedule (up to 95% budget) -----
    beta_schedule = [5, 20, 80, 200, 800, 3000, 8000]
    for beta in beta_schedule:
        if time.time() - start_time > total_budget * 0.95:
            break
        try:
            h_ref = slsqp_optimize(best_h, beta, max_iter=800)
        except Exception:
            continue
        c_ref = compute_c5(h_ref)
        if c_ref < best_c:
            best_h, best_c = h_ref, c_ref

    # ----- final microscopic random search (up to 99% budget) -----
    while time.time() - start_time < total_budget * 0.99:
        new_h = perturb_random(best_h, rng, scale=0.01)
        c_new = compute_c5(new_h)
        if c_new < best_c:
            best_h, best_c = new_h, c_new

    # ----- cleanup and final evaluation -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
