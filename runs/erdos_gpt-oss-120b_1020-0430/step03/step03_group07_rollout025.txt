def swap_search(h_bin, rng, max_iters, best_c):
    n = len(h_bin)
    while max_iters > 0:
        conv = correlate_full(h_bin)
        cmax = conv.max() * (2.0 / n)
        if cmax < best_c - 1e-12:
            best_c = cmax
            # continue exploring improvements
        # find worst shift
        worst_idx = np.argmax(conv)
        k = worst_idx - (n - 1)
        # Determine mismatched and matched indices
        if k >= 0:
            left = np.arange(n - k)
            right = left + k
        else:
            k_abs = -k
            left = np.arange(k_abs, n)
            right = left - k_abs
        # mismatched: h[left]==1, h[right]==0
        mismatched = left[(h_bin[left] > 0.5) & (h_bin[right] < 0.5)]
        matched = left[(h_bin[left] < 0.5) & (h_bin[right] > 0.5)]
        if len(mismatched) == 0 or len(matched) == 0:
            # cannot improve, break
            break
        # pick random pair
        i = rng.choice(mismatched)
        j = rng.choice(matched)
        # swap
        h_new = h_bin.copy()
        h_new[i] = 0
        h_new[j] = 1
        # compute new C5
        c_new = compute_c5(h_new)
        max_iters -= 1
        if c_new < best_c - 1e-12:
            h_bin = h_new
            best_c = c_new
            # optionally restart with new conv
        # optionally accept with prob for SA?
    return h_bin, best_c

    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h

# ----------------------------------------------------------------------
# Correlation utilities (linear correlation of h with (1-h))
# ----------------------------------------------------------------------
def correlate_full(h):
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv

def compute_c5(h):
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)

# ----------------------------------------------------------------------
# Smooth max surrogate for SLSQP
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    conv = correlate_full(x)
    bc = CURRENT_BETA * conv
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

# ----------------------------------------------------------------------
# Binary local improvement: swap a mismatched 1 with a matched 0
# ----------------------------------------------------------------------
def swap_search(h_bin, rng, max_iters, current_best):
    """Descend on max overlap by swapping 1↔0 at the worst shift."""
    n = h_bin.size
    best_h = h_bin
    best_c = current_best
    it = 0
    while it < max_iters:
        conv = correlate_full(best_h)
        c_val = conv.max() * (2.0 / n)
        if c_val < best_c - 1e-12:
            best_c = c_val
        # worst shift
        worst_idx = int(conv.argmax())
        k = worst_idx - (n - 1)          # shift = i - (n-1)
        if k >= 0:
            left = np.arange(n - k)
            right = left + k
        else:
            left = np.arange(-k, n)
            right = left + k
        # mismatched: h[left]==1, h[right]==0
        # matched   : h[left]==0, h[right]==1
        mism = left[(best_h[left] > 0.5) & (best_h[right] < 0.5)]
        matc = left[(best_h[left] < 0.5) & (best_h[right] > 0.5)]
        if mism.size == 0 or matc.size == 0:
            break
        i = rng.choice(mism)
        j = rng.choice(matc)
        new_h = best_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_c = compute_c5(new_h)
        it += 1
        if new_c < best_c - 1e-12:
            best_h = new_h
            best_c = new_c
            # restart from fresh conv in next loop iteration
        # else reject
    return best_h, best_c

# ----------------------------------------------------------------------
# Continuous hill‑climb (small epsilon moves) – optional refinement
# ----------------------------------------------------------------------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.96  # keep a safety margin

    # ----- initial vector ------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 180
        h0 = np.full(n, 0.5)
        h0 = enforce_sum(h0, n / 2.0, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ----- smooth SLSQP refinement (continuation) -----------------------
    beta_list = [20.0, 80.0, 200.0, 800.0]
    for beta in beta_list:
        if time.time() > deadline:
            break
        try:
            h_opt = slsqp_optimize(best_h, beta, max_iter=400)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c:
                best_h, best_c = h_opt, c_opt
        except Exception:
            pass

    # ----- create several binary starting points ------------------------
    binary_inits = []
    # binary rounding of current best
    h_bin = (best_h > 0.5).astype(float)
    h_bin = enforce_sum(h_bin, n / 2.0, rng)
    binary_inits.append(h_bin)

    # a few random binary vectors
    for _ in range(4):
        binary_inits.append(random_binary_vector(n, rng))

    # ----- binary swap‑search on each init -----------------------------
    for init in binary_inits:
        if time.time() > deadline:
            break
        # allocate a modest budget per init, shrinking as time passes
        remaining = deadline - time.time()
        iters = int(min(12000, remaining * 5))  # heuristic
        h_res, c_res = swap_search(init, rng, max_iters=iters, current_best=best_c)
        if c_res < best_c:
            best_h, best_c = h_res, c_res

    # ----- occasional extra random swaps to escape local minima ---------
    while time.time() < deadline:
        # perform a quick random batch of swaps
        batch = 2000
        h_tmp, c_tmp = swap_search(best_h, rng, max_iters=batch, current_best=best_c)
        if c_tmp < best_c:
            best_h, best_c = h_tmp, c_tmp
        else:
            # small perturbation (swap a few random bits) and retry
            flips = max(1, int(0.005 * n))
            for _ in range(flips):
                ones = np.where(best_h > 0.5)[0]
                zeros = np.where(best_h < 0.5)[0]
                if ones.size == 0 or zeros.size == 0:
                    break
                i = rng.choice(ones)
                j = rng.choice(zeros)
                best_h[i] = 0.0
                best_h[j] = 1.0
            best_h = enforce_sum(best_h, n / 2.0, rng)
            best_c = compute_c5(best_h)

    # ----- final continuous refinement ----------------------------------
    for beta in beta_list:
        if time.time() > deadline:
            break
        try:
            h_opt = slsqp_optimize(best_h, beta, max_iter=350)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c:
                best_h, best_c = h_opt, c_opt
        except Exception:
            pass

    # ----- optional hill‑climb for fine tuning -------------------------
    if time.time() < deadline:
        h_opt, c_opt = hill_climb(best_h, rng, max_iters=2500, start_step=0.3)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # ----- final clean‑up ------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
