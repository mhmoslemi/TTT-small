import time
import numpy as np
from scipy.optimize import minimize

# Global placeholders for surrogate optimization
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=2000):
    """Adjust h so that Σh = target while respecting bounds [0,1]."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        # final uniform correction, then clip again
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if j >= i:
        j += 1
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_multi(h_start, rng, max_iters=2000, init_step=0.5, n_trials=6):
    """Hill‑climb with several random perturbations per step."""
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay = max(1, max_iters // 12)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
                improved = True
                break
        if not improved and rng.random() < 0.03:
            new_h = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)
    return h, best_c


def slsqp_optimize(h0, beta, max_iter=400):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    h = np.full(n, 0.5)
    for _ in range(n * 4):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.35
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def random_binary_feasible(n, rng):
    """Generate a binary vector with exactly n/2 ones."""
    mask = np.zeros(n, dtype=float)
    mask[: n // 2] = 1.0
    rng.shuffle(mask)
    return mask


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # -----------------------------------------------------------------
    # 1. Base dimension and initial vector
    # -----------------------------------------------------------------
    n = 360
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        h0 = random_feasible_vector(n, rng)

    # -----------------------------------------------------------------
    # 2. Assemble initial candidates
    # -----------------------------------------------------------------
    candidates = [h0]
    xs = (np.arange(n) + 0.5) * (2.0 / n)

    for f in (1, 2, 3, 4, 5):
        h = 0.5 + 0.5 * np.sin(np.pi * f * xs)
        candidates.append(enforce_sum(np.clip(h, 0, 1), n / 2.0, rng))
        h = 0.5 + 0.5 * np.cos(np.pi * f * xs)
        candidates.append(enforce_sum(np.clip(h, 0, 1), n / 2.0, rng))

    candidates.append(enforce_sum(xs / 2.0, n / 2.0, rng))
    candidates.append(enforce_sum(1.0 - xs / 2.0, n / 2.0, rng))
    candidates.append(random_binary_feasible(n, rng))
    for _ in range(4):
        candidates.append(random_feasible_vector(n, rng))

    # -----------------------------------------------------------------
    # 3. Initial best
    # -----------------------------------------------------------------
    best_h = None
    best_c = float("inf")
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # -----------------------------------------------------------------
    # 4. Main optimisation loop (hill‑climb + SLSQP ladder)
    # -----------------------------------------------------------------
    beta_list = [5, 20, 80, 200, 500, 1000, 2000]
    hc_iters = 3000
    while time.time() - start_time < budget_s * 0.78:
        # Choose start point
        if rng.random() < 0.6:
            start_vec = perturb_h(best_h, rng, scale=0.5)
        elif rng.random() < 0.4:
            start_vec = random_binary_feasible(n, rng).astype(float)
            start_vec = enforce_sum(start_vec, n / 2.0, rng)
        else:
            start_vec = random_feasible_vector(n, rng)

        # Hill‑climb
        h_opt, c_opt = hill_climb_multi(start_vec, rng,
                                        max_iters=hc_iters,
                                        init_step=0.6,
                                        n_trials=10)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()

        # SLSQP refinement
        for beta in beta_list:
            if time.time() - start_time > budget_s * 0.78:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, max_iter=800)
            except Exception:
                continue
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_c, best_h = c_ref, h_ref.copy()

    # -----------------------------------------------------------------
    # 5. Upsample to finer grid if time permits
    # -----------------------------------------------------------------
    elapsed = time.time() - start_time
    if budget_s - elapsed > budget_s * 0.20:
        target_n = n * 2
        old_x = (np.arange(n) + 0.5) * (2.0 / n)
        new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
        h_up = np.interp(new_x, old_x, best_h)
        h_up = enforce_sum(h_up, target_n / 2.0, rng)

        h_up_opt, c_up_opt = hill_climb_multi(h_up, rng,
                                              max_iters=5000,
                                              init_step=0.4,
                                              n_trials=12)
        best_h, best_c, n = h_up_opt.copy(), c_up_opt, target_n

        for beta in beta_list:
            if time.time() - start_time > budget_s:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, max_iter=800)
            except Exception:
                continue
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_h, best_c = h_ref.copy(), c_ref

    # -----------------------------------------------------------------
    # 6. Final clean‑up
    # -----------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
