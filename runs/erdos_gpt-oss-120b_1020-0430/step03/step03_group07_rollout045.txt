import time
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Basic feasibility utilities
# --------------------------------------------------------------
def enforce_sum(h, target, rng):
    """Project h onto {0<=h<=1, sum(h)=target} by local adjustments."""
    diff = target - h.sum()
    n = h.size
    for _ in range(10000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small ε from i to j (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


# --------------------------------------------------------------
# Correlation / objective
# --------------------------------------------------------------
def correlate_full(h):
    """Linear correlation of h with (1‑h) using FFT (zero‑padded)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Discrete C₅ for step function h (dx = 2/n)."""
    return correlate_full(h).max() * (2.0 / h.size)


# --------------------------------------------------------------
# Smooth surrogate for SLSQP
# --------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = correlate_full(x)
    bc = CURRENT_BETA * corr
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=400):
    """SLSQP on smoothed max‑overlap."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def hill_climb(h_start, rng, max_iters=3000, start_step=0.4):
    """Greedy hill‑climb with decaying step size."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_int = max(1, max_iters // 12)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h, h = c, new_h.copy(), new_h
        if (it + 1) % decay_int == 0:
            step = max(step * 0.75, 1e-5)
    return best_h, best_c


def spectral_smooth(h, rng, factor=0.6, cutoff=0.3):
    """Dampen high‑frequency components of h."""
    N = h.size
    H = np.fft.rfft(h)
    cut = int(cutoff * len(H))
    H[cut:] *= factor
    h_new = np.fft.irfft(H, n=N)
    h_new = np.clip(h_new, 0.0, 1.0)
    return enforce_sum(h_new, N / 2.0, rng)

            global_best_h, global_best_c = h_cur.copy(), c_cur

        # intensive hill‑climb
        h_hc, c_hc = hill_climb(global_best_h, rng, max_iters=4000, start_step=0.35)
        if c_hc < global_best_c:
            global_best_h, global_best_c = h_hc.copy(), c_hc

        # spectral smoothing attempts
        for cutoff in [0.2, 0.3, 0.45]:
            if time.time() >= deadline_main:
                break
            h_sp = spectral_smooth(global_best_h, rng, factor=0.5, cutoff=cutoff)
            h_sp_hc, c_sp_hc = hill_climb(h_sp, rng, max_iters=2500, start_step=0.25)
            if c_sp_hc < global_best_c:
                global_best_h, global_best_c = h_sp_hc.copy(), c_sp_hc

    # ---- main stochastic search loop -------------------------------------------------
    while time.time() < deadline_main:
        # choose start point
        if rng.random() < 0.6:
            start = perturb_h(global_best_h, rng, scale=0.3)
        else:
            start = random_feasible_vector(n, rng)
        improve_from(start)

    # ---- optional up‑sampling --------------------------------------------------------
    if time.time() < deadline_up and n * 2 <= 720:
        up_factor = 2
        h_up = np.repeat(global_best_h, up_factor)
        n_up = h_up.size
        h_up = enforce_sum(h_up, n_up / 2.0, rng)
        # refine on finer grid
        improve_from(h_up)
        n = n_up
        global_best_h = global_best_h[:n]  # ensure correct length

    # ---- final cleanup ----------------------------------------------------------------
    global_best_h = np.clip(global_best_h, 0.0, 1.0)
    global_best_h = enforce_sum(global_best_h, global_best_h.size / 2.0, rng)
    final_c = compute_c5(global_best_h)

    return global_best_h.tolist(), float(final_c), int(global_best_h.size)
