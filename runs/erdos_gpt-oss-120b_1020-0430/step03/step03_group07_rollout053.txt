CURRENT_TARGET = None
CURRENT_BETA = None


def enforce_sum(h, target, rng):
    """Adjust h to satisfy Σh = target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 20000:
        if diff > 0:
            idxs = np.where(h < 1 - 1e-12)[0]
            if idxs.size == 0:
                break
            i = rng.choice(idxs)
            add = min(diff, 1 - h[i])
            h[i] += add
        else:
            idxs = np.where(h > 1e-12)[0]
            if idxs.size == 0:
                break
            i = rng.choice(idxs)
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
        attempts += 1
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with Σ = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.5
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h = enforce_sum(h, target, rng)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def correlate_full(h):
    """Full linear correlation of h with (1‑h) using FFT."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())  # next power of two
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Discrete C₅ for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)

# SLSQP smoothing (log‑sum‑exp surrogate of max)
# ----------------------------------------------------------------------
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    conv = correlate_full(x)
    bc = CURRENT_BETA * conv
    m = bc.max()
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        x = enforce_sum(x, CURRENT_TARGET, rng)
    return x

def hill_climb_strong(h_start, rng, max_iters=2000, start_step=0.5):

def simulated_annealing(start_h, rng, deadline, T0=0.02, Tf=1e-6, max_iters=1500):
    """Exponential‑cooling simulated annealing."""
    cur_h = start_h.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        new_h = perturb_h(cur_h, rng, scale=0.35)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
    return best_h, best_c

    # --------------------------------------------------------------
    fine_deadline = end_time - 2.0  # small safety margin
    best_h_fine = h_fine.copy()
    best_c_fine = compute_c5(best_h_fine)

    while time.time() < fine_deadline:
        # randomised start
        if rng.random() < 0.5:
            start = perturb_h(best_h_fine, rng, scale=0.3)
        else:
            start = random_feasible_vector(n_fine, rng)

        # simulated annealing (more iterations)
        cand_h, cand_c = simulated_annealing(
            start, rng, fine_deadline, T0=0.05, Tf=1e-6, max_iters=2500
        )
        if cand_c < best_c_fine:
            best_h_fine, best_c_fine = cand_h, cand_c

        # strong hill climbing (moderate length)
        hc_h, hc_c = hill_climb_strong(best_h_fine, rng, max_iters=4000, start_step=0.3)
        if hc_c < best_c_fine:
            best_h_fine, best_c_fine = hc_h, hc_c

        # occasional SLSQP refinement (light)
        if rng.random() < 0.2:
            for beta in [5.0, 20.0, 80.0, 200.0]:
                if time.time() >= fine_deadline:
                    break
                try:
                    opt_h = slsQP_optimize(best_h_fine, beta, max_iter=250)
                    opt_c = compute_c5(opt_h)
                    if opt_c < best_c_fine:
                        best_h_fine, best_c_fine = opt_h, opt_c
                except Exception:
                    continue

    # --------------------------------------------------------------
    # 5. Final clean‑up
    # --------------------------------------------------------------
    best_h_fine = np.clip(best_h_fine, 0.0, 1.0)
    best_h_fine = enforce_sum(best_h_fine, n_fine / 2.0, rng)
    final_c = compute_c5(best_h_fine)

    return best_h_fine.tolist(), float(final_c), int(best_h_fine.size)
