    for _ in range(max_iter):

            break

import time
import numpy as np

def compute_c5(h):
    """Exact C5 for discrete step function h (dx = 2 / n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h, target, rng, max_iter=2000):
    """Clamp h to [0,1] and adjust to hit exact sum=target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        # distribute the remaining error uniformly
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def neighbor_sum_array(idx, k, h):
    """h[idx+k] + h[idx-k] with zero outside [0,n)."""
    n = h.size
    idx = np.asarray(idx)
    high = np.where(idx + k < n, h[idx + k], 0.0)
    low = np.where(idx - k >= 0, h[idx - k], 0.0)
    return high + low

def delta_corr(i, j, h, L):
    """Full correlation delta for moving ε from i to j (length 2n‑1)."""
    n = h.size
    iL = i + L
    mask_iL = (iL >= 0) & (iL < n)
    term_i_plus = np.zeros_like(L, dtype=float)
    term_i_plus[mask_iL] = h[iL[mask_iL]]

    i_minusL = i - L
    mask_i_minusL = (i_minusL >= 0) & (i_minusL < n)
    term_i_minus = np.zeros_like(L, dtype=float)
    term_i_minus[mask_i_minusL] = h[i_minusL[mask_i_minusL]]

    jL = j + L
    mask_jL = (jL >= 0) & (jL < n)
    term_j_plus = np.zeros_like(L, dtype=float)
    term_j_plus[mask_jL] = h[jL[mask_jL]]

    j_minusL = j - L
    mask_j_minusL = (j_minusL >= 0) & (j_minusL < n)
    term_j_minus = np.zeros_like(L, dtype=float)
    term_j_minus[mask_j_minusL] = h[j_minusL[mask_j_minusL]]

    return (term_i_plus + term_i_minus) - (term_j_plus + term_j_minus)

def run(seed=42, budget_s=1100, **kwargs):

    start_time = time.time()

    # ------------------------------------------------------------------
    # 1. Initialise h
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.size
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        h = enforce_sum(h, target, rng)
    else:
        n = 180
        target = n / 2.0
        h = np.full(n, 0.5)
        h = enforce_sum(h, target, rng)

    # pre‑compute helpers
    L = np.arange(-(n - 1), n, dtype=int)   # lags for full correlation

    # current correlation and best solution
    S = np.correlate(h, 1.0 - h, mode="full")
    cur_max = S.max()
    best_h = h.copy()
    best_c = cur_max * (2.0 / n)

    # ------------------------------------------------------------------
    # 2. Main optimisation loop – guided moves + occasional random tries
    # ------------------------------------------------------------------
    step = 0.3                     # fraction of transferable mass
    decay = 0.9995                 # exponential decay of step
    min_step = 1e-4
    max_no_improve = 2000
    no_improve = 0
    accepted = 0

    while time.time() - start_time < budget_s * 0.95:
        # periodic exact recompute to cure drift
        if accepted and accepted % 5000 == 0:
            S = np.correlate(h, 1.0 - h, mode="full")
            cur_max = S.max()

        # worst lag (absolute value, for neighbour sums)
        lag_idx = int(np.argmax(S))
        lag = abs(lag_idx - (n - 1))

        src = np.where(h > 1e-12)[0]
        sink = np.where(h < 1.0 - 1e-12)[0]

                    prob = np.exp((cur_max - new_max) / temperature)
                    accept = rng.random() < prob
                if accept:
                    h[i] -= epsilon
                    h[j] += epsilon
                    S = new_S
                    cur_max = new_max
                    accepted += 1
                    no_improve = 0
                    cur_c = cur_max * (2.0 / n)
                    if cur_c < best_c:
                        best_c = cur_c
                        best_h = h.copy()

        # step decay
        step = max(step * decay, min_step)

        # --- long stall – large random perturbation ----------------------------
        if no_improve > max_no_improve:
            for _ in range(100):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                max_eps = min(h[i], 1.0 - h[j])
                if max_eps < 1e-12:
                    continue
                eps = max_eps * rng.random()
                h[i] -= eps
                h[j] += eps
            S = np.correlate(h, 1.0 - h, mode="full")
            cur_max = S.max()
            no_improve = 0
            step = max(step, 0.05)   # give the search a little boost

    # ------------------------------------------------------------------
    # 3. Final clean‑up
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    best_c = compute_c5(best_h)

    return best_h.tolist(), float(best_c), int(n)
