def project_to_feasible(v, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo+hi)*0.5, 0.0, 1.0)

def subgradient_descent(h0, target, rng, max_iters=3000, init_step=0.5, decay=0.98):
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = init_step
    for it in range(max_iters):
        # compute correlation vector
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = np.max(corr)
        # get lag(s) of max
        lag = np.argmax(corr) - (n - 1)
        # gradient
        g = np.ones(n)
        k = lag
        if k >= 0:
            g[:n-k] -= h[k:]
            if k > 0:
                g[k:] -= h[:n-k]
        else:
            kp = -k
            g[kp:] -= h[:n-kp]
            g[:n-kp] -= h[kp:]
        # update
        h_new = h - step * g
        # symmetrize
        h_new = (h_new + h_new[::-1]) * 0.5
        # project
        h_new = project_to_feasible(h_new, target)
        # compute C
        c_new = compute_c5(h_new)
        # accept if improves best
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        # update h for next iteration
        h = h_new
        # decay step
        step *= decay
        if step < 1e-5:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_c = np.inf
    best_h = None
    best_n = None

    candidate_ns = [60,84,120,168,240,300]

    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 not in candidate_ns:
            candidate_ns = [n0] + candidate_ns

    for n in candidate_ns:
        target = n / 2.0
        # compute number of restarts based on remaining time
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining < 5:
            break
        # Determine a reasonable number of restarts, e.g., min(30, remaining // 5)
        # At least 5 restarts.
        max_restarts = max(5, int(min(40, remaining // 8)))  # each ~ <2 secs
        # optionally allocate rest of remaining to this n.
        for r in range(max_restarts):
            # check time
            if time.time() - start_time > budget_s * 0.95:
                break
            # init vector
            if 'initial_h_values' in globals() and n == len(initial_h_values):
                h0 = np.array(initial_h_values, dtype=float)
                # ensure feasible (maybe small rounding)
                h0 = np.clip(h0, 0., 1.)
                h0 = project_to_feasible(h0, target)
            elif 'initial_h_values' in globals() and n != len(initial_h_values):
                init_vec = upsample_initial(n)
                init_vec = np.clip(init_vec, 0., 1.)
                h0 = project_to_feasible(init_vec, target)
            else:
                h0 = random_feasible_vector(n, rng)
            # run subgradient descent
            # choose parameters random
            init_step = rng.uniform(0.2, 1.0)
            decay = rng.uniform(0.94, 0.99)
            max_iters = int(3000 * (0.8 + 0.4 * rng.random())) # 2400-4200
            h_opt, c_opt = subgradient_descent(h0, target, rng,
                                               max_iters=max_iters,
                                               init_step=init_step,
                                               decay=decay)
            # hill climb for fine tuning
            if time.time() - start_time < budget_s * 0.97:
                h_opt2, c_opt2 = hill_climb(h_opt, rng, max_iters=2000, start_step=0.2)
                if c_opt2 < c_opt:
                    h_opt, c_opt = h_opt2, c_opt2
            # update best
            if c_opt < best_c:
                best_c, best_h, best_n = c_opt, h_opt.copy(), n
        # optional early termination if best_c already low enough (<0.3808)
        if best_c < 0.3808:
            break

    # fallback if nothing found
    if best_h is None:
        best_n = 60
        best_h = np.full(best_n, 0.5)

    # final feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = best_n/2.0 - best_h.sum()
    if abs(diff) > 1e-12:
        # simple correction loop
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, best_n)
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = best_n/2.0 - best_h.sum()

    best_h = h.copy()

    step = init_step

import time
import numpy as np

# ---------- helper functions ----------
def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def project_to_feasible(v, target):
    """Projection onto {0≤h≤1, Σh = target} via bisection on a uniform shift."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def random_feasible_vector(n, rng):
    """Generate a random vector with 0≤h≤1 and Σh = n/2."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_to_feasible(v, n / 2.0)

def upsample_initial(to_n):
    """Linear interpolation of the global initial_h_values to a new length."""
    init = np.array(initial_h_values, dtype=float)
    m = init.size
    xs_old = np.linspace(0.0, 2.0, m, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, to_n, endpoint=False)
    return np.interp(xs_new, xs_old, init)

def interpolate_to_n(src_vec, src_n, dst_n):
    """Interpolate src_vec (size src_n) to size dst_n on [0,2)."""
    xs_src = np.linspace(0.0, 2.0, src_n, endpoint=False)
    xs_dst = np.linspace(0.0, 2.0, dst_n, endpoint=False)
    return np.interp(xs_dst, xs_src, src_vec)

def perturb_h(h, rng, scale=0.2):
    """Transfer a small ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb(h_start, rng, max_iters=2000, start_step=0.2):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-5)
    return best_h, best_c

def subgradient_descent(h0, target, rng, max_iters=3000,
                        init_step=0.5, decay=0.97):
    """Sub‑gradient descent on the max‑overlap objective."""
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = init_step
    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode="full")
        lag = np.argmax(corr) - (n - 1)          # integer shift
        # gradient of C(lag) w.r.t. h
        g = np.ones(n)
        k = lag
        if k >= 0:
            g[:n - k] -= h[k:]
            if k > 0:
                g[k:] -= h[:n - k]
        else:
            kp = -k
            g[kp:] -= h[:n - kp]
            g[:n - kp] -= h[kp:]
        # descent step and projection
        h_new = h - step * g
        h_new = project_to_feasible(h_new, target)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        step *= decay
        if step < 1e-5:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_c = np.inf
    best_h = None
    best_n = None

    candidate_ns = [60, 84, 120, 168, 240, 300]
    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        if n0 not in candidate_ns:
            candidate_ns = [n0] + candidate_ns

    prev_best_h = None
    prev_n = None

    for n in candidate_ns:
        elapsed = time.time() - start_time
        if budget_s - elapsed < 5.0:
            break
        target = n / 2.0

        # decide how many restarts we can afford for this n
        remaining = budget_s - (time.time() - start_time)
        max_restarts = max(5, int(min(40, remaining // 8)))
        for _ in range(max_restarts):
            if time.time() - start_time > budget_s * 0.95:
                break

            # ----- initialise a feasible vector -----
            use_prev = prev_best_h is not None and rng.random() < 0.6
            if use_prev:
                h0 = interpolate_to_n(prev_best_h, prev_n, n)
                h0 = project_to_feasible(h0, target)
            elif "initial_h_values" in globals():
                if n == len(initial_h_values):
                    h0 = np.array(initial_h_values, dtype=float)
                else:
                    h0 = upsample_initial(n)
                h0 = np.clip(h0, 0.0, 1.0)
                h0 = project_to_feasible(h0, target)
            else:
                h0 = random_feasible_vector(n, rng)

            # ----- sub‑gradient descent -----
            init_step = rng.uniform(0.2, 1.0)
            decay = rng.uniform(0.94, 0.99)
            max_iters = int(3000 * (0.8 + 0.4 * rng.random()))
            h_opt, c_opt = subgradient_descent(
                h0, target, rng,
                max_iters=max_iters,
                init_step=init_step,
                decay=decay,
            )

            # ----- fine hill‑climb -----
            if time.time() - start_time < budget_s * 0.97:
                h_opt, c_opt = hill_climb(h_opt, rng, max_iters=2000, start_step=0.1)

            if c_opt < best_c:
                best_c, best_h, best_n = c_opt, h_opt.copy(), n

        # keep best of this resolution for possible up‑sampling
        if best_h is not None and best_n == n:
            prev_best_h = best_h.copy()
            prev_n = n

        # early exit if we already beat the target
        if best_c < 0.3808:
            break

    # ----- final intensive refinement on the overall best -----
    if best_h is not None:
        target = best_n / 2.0
        best_h, best_c = subgradient_descent(
            best_h, target, rng,
            max_iters=8000,
            init_step=0.2,
            decay=0.99,
        )
        best_h, best_c = hill_climb(best_h, rng, max_iters=3000, start_step=0.05)

    if best_h is None:
        best_n = 60
        best_h = np.full(best_n, 0.5)
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = best_n / 2.0 - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):

            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = best_n / 2.0 - best_h.sum()
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
