def run(seed=42, budget_s=1100, **kwargs):
    rng= np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # Determine candidate ns (maybe [96,120,144,168,192,240,384])
    candidate_ns = [96,120,144,168,192,240,384]
    # If initial_h_values present, get it and use its n as candidate first.
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        best_h = np.clip(h0, 0.0, 1.0)
        # enforce sum
        best_h = enforce_sum(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = best_h.size

    # For each n:
    for n in candidate_ns:
        if time.time() - start > 0.90 * budget_s:
            break
        # maintain local best for this n
        local_best_h = None
        local_best_c = np.inf

        # If best_n matches n and best_h exists, start from that.
        if best_n == n and best_h is not None:
            seeds = [best_h.copy()]
        else:
            seeds = []

        # Generate some random seeds (maybe up to 5 or until time)
        while len(seeds) < 5 and time.time() - start < 0.93 * budget_s:
            h = random_feasible_vector(n, rng)
            seeds.append(h)

        for h in seeds:
            # Apply hill climb
            h, c = hill_climb(h, rng, max_iters=5000, step_scale=0.5)
            if c < local_best_c:
                local_best_c, local_best_h = c, h.copy()
            # Get SLSQP refinement with a schedule of betas
            for beta in [1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1000.0]:
                if time.time() - start > 0.95 * budget_s:
                    break
                h = slsqp_optimize(h, beta, rng, max_iter=600)
                c = compute_c5(h)
                if c < local_best_c:
                    local_best_c, local_best_h = c, h.copy()
            # gradient refine
            if time.time() - start < 0.96 * budget_s:
                h = gradient_refine(h, rng, step=0.02, max_iter=12)
                c = compute_c5(h)
                if c < local_best_c:
                    local_best_c, local_best_h = c, h.copy()
            # final short hill climb
            if time.time() - start < 0.97 * budget_s:
                h, c = hill_climb(h, rng, max_iters=2000, step_scale=0.3)
                if c < local_best_c:
                    local_best_c, local_best_h = c, h.copy()

        # Update global best
        if local_best_c < best_c:
            best_c = local_best_c
            best_h = local_best_h.copy()
            best_n = n

    # Additional upsampling to larger n if time left
    if best_h is not None and best_n < 384 and time.time() - start < 0.99 * budget_s:
        # upsample by factor 2 repeatedly until n reaches 384 or time is low
        while best_n < 384 and time.time() - start < 0.99 * budget_s:
            # upsample by factor 2
            best_h = np.repeat(best_h, 2)
            best_n = best_h.size
            # refine
            best_h, c = hill_climb(best_h, rng, max_iters=3000, step_scale=0.4)
            best_c = c
            # SLSQP a few betas
            for beta in [5.0, 20.0, 80.0]:
                if time.time() - start > budget_s:
                    break
                best_h = slsqp_optimize(best_h, beta, rng, max_iter=400)

            # gradient refine
            best_h = gradient_refine(best_h, rng, step=0.01, max_iter=8)
            best_c = compute_c5(best_h)

    # final random perturbations for remaining time
    while time.time() - start < budget_s * 0.999 and best_h is not None:

    best_h = enforce_sum(best_h, rng)

import time, numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    """Discrete C5 for step function on [0,2]."""
    c = np.correlate(h, 1 - h, mode="full")
    return np.max(c) * (2.0 / h.size)

def enforce_sum(h, rng):
    """Adjust h so that sum(h)=len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple random redistribution
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum=n/2."""
    h = rng.random(n)
    h *= (n / 2.0) / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, rng)

def worst_gradient(h):
    """Gradient of the worst (largest) overlap Corr_k."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]     -= h[: n - k]
    return grad

def guided_transfer(h, grad, rng, scale=0.5):
    """Move mass from a high‑gradient index to a low‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def hill_climb(h0, rng, max_iters=3000, step_scale=0.5):
    """Simple hill‑climb using guided transfers."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    scale = step_scale
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_transfer(h, grad, rng, scale=scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            scale = max(scale * 0.7, 1e-4)
    return best_h, best_c

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    """Smooth max‑overlap optimisation via SLSQP."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return enforce_sum(x, rng)

def gradient_refine(h, rng, step=0.02, max_iter=8):
    """Fine‑grained descent on the worst shift."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        k = abs(np.argmax(corr) - (n - 1))
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] = 1.0 - h[k:]
            g[k:]    -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h
