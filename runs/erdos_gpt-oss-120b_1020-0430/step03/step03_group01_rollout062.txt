def weighted_shift_sum(h, w):
    n = len(h)
    offset = n - 1
    g = np.zeros(n)
    for lag in range(- (n-1), n):
        w_l = w[lag + offset]
        if w_l == 0:
            continue
        if lag >= 0:
            g[:n-lag] += w_l * h[lag:]
        else:
            g[-lag:] += w_l * h[:n+lag]
    return g

import time
import numpy as np
from scipy.optimize import minimize

# ---------- helper utilities ----------

def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h so that sum(h) == target while keeping 0 <= h <= 1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        # fallback uniform correction
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def compute_c5(h):
    """Exact discrete C5 for step function h on [0,2] with uniform grid."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def random_feasible_vector(n, rng):
    """Generate a random feasible vector (0<=h_i<=1, sum = n/2)."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_multi(h_start, rng, max_iters=2000, init_step=0.5, n_trials=5):
    """Simple multi‑trial hill‑climb."""
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
                improved = True
                break
        if not improved and rng.random() < 0.02:
            new_h = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
                improved = True
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return h, best_c


# ---------- smooth surrogate + projected gradient ----------

CURRENT_TARGET = None
CURRENT_BETA = None


def _softmax_weights(corr, beta):
    """Return normalized soft‑max weights for a correlation vector."""
    b = beta * corr
    b_max = np.max(b)
    exp_b = np.exp(b - b_max)
    return exp_b / np.sum(exp_b)


def gradient_descent(
    h_start,
    target,
    beta_schedule,
    step0,
    step_decay,
    iters_per_beta,
    rng,
    end_time,
    start_time,
):
    """Projected gradient descent on the log‑sum‑exp surrogate."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)

    n = h.size
    offset = n - 1

    for beta in beta_schedule:
        # initialise step size proportional to beta (larger beta → smaller steps)
        step = step0 * 5.0 / max(beta, 1.0)
        for itr in range(iters_per_beta):
            if time.time() > end_time:
                return best_h, best_c
            corr = np.correlate(h, 1.0 - h, mode="full")
            w = _softmax_weights(corr, beta)
            w_sum = w + w[::-1]                         # symmetric kernel
            conv_full = np.convolve(h, w_sum, mode="full")
            shift_sum = conv_full[offset : offset + n]   # Σ_k w_k h_{i+k} + Σ_k w_k h_{i‑k}
            grad = 1.0 - shift_sum
            h = h - step * grad
            h = np.clip(h, 0.0, 1.0)
            h = enforce_sum(h, target, rng)
            c = compute_c5(h)
            if c < best_c:
                best_c, best_h = c, h.copy()
            # step decay schedule
            if (itr + 1) % max(1, iters_per_beta // 5) == 0:
                step *= step_decay
        # small random shake before next beta
        if time.time() > end_time:
            break
        h = perturb_h(h, rng, scale=step0 * 0.5)
        h = enforce_sum(h, target, rng)
    return best_h, best_c


# ---------- SLSQP refinement (same as previous implementation) ----------

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on the smooth surrogate for a fixed β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum (tiny numerical drift)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x
