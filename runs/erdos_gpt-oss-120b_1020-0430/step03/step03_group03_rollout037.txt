def biased_step(h, rng, step_frac=0.5):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    ks = np.argmax(corr)
    offset = ks - (n - 1)  # shift
    if offset >= 0:
        idx = np.arange(0, n - offset)
    else:
        idx = np.arange(-offset, n)  # because i+offset >=0 -> i >= -offset
    if idx.size == 0:
        # shift too large? Should not happen as correlation at extreme shift is just sum h (1) maybe.
        return h
    prod = h[idx] * (1 - h[idx + offset])
    # pick high and low
    src = idx[np.argmax(prod)]
    dst_candidates = idx[np.argmin(prod)]
    # Actually there may be multiple low; we can just pick that
    dst = dst_candidates
    # Ensure src != dst; if equal, choose other minimal
    if src == dst:
        # try alternative
        sorted_idx = np.argsort(prod)
        src = idx[sorted_idx[-1]]
        dst = idx[sorted_idx[0]]
        if src == dst:
            # randomly pick different indices
            src, dst = rng.choice(idx, size=2, replace=False)
    max_transfer = min(h[src], 1 - h[dst])
    if max_transfer <= 0:
        # fallback random perturb
        return h
    eps = step_frac * max_transfer * (rng.random())
    # update
    h_new = h.copy()
    h_new[src] -= eps
    h_new[dst] += eps
    return h_new

def transfer_step(h, rng, step_frac):

    n = len(h)

    corr = np.correlate(h, 1 - h, mode="full")

    ks = np.argmax(corr)

    offset = ks - (n - 1)

    # compute valid indices
    if offset >= 0:
        idx = np.arange(0, n - offset)
    else:
        idx = np.arange(-offset, n)
    # compute product
    prod = h[idx] * (1 - h[idx + offset])
    # find src, dst indices
    src = idx[np.argmax(prod)]
    # find dst = idx with min prod
    dst = idx[np.argmin(prod)]
    if src == dst:
        # fallback random picks
        if len(idx) >= 2:
            src, dst = rng.choice(idx, size=2, replace=False)
        else:
            # nothing to do
            return h
    # compute transfer magnitude
    max_transfer = min(h[src], 1 - h[dst])
    if max_transfer <= 0:
        # random fallback: choose random pair
        i, j = rng.choice(n, size=2, replace=False)
        max_transfer = min(h[i], 1 - h[j])
        if max_transfer <= 0:
            return h
        src, dst = i, j
    # epsilon as fraction step_frac of max_transfer * random
    eps = max_transfer * step_frac * rng.random()
    # Update
    h_new = h.copy()
    h_new[src] -= eps
    h_new[dst] += eps
    return h_new

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)

    start_time = time.time()
    # Determine n (samples)
    # If initial_h_values exists, use its length; else default 200
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        # clip to [0,1], enforce sum
        np.clip(h0, 0.0, 1.0, out=h0)
        enforce_sum(h0, n / 2.0, rng)
    else:
        n = 200
        h0 = random_feasible_vector(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # current solution for hill climbing
    curr_h = best_h.copy()
    curr_c = best_c

    # Reserve time for final polishing:
    polish_reserve = 15.0  # seconds (enough for a few SLSQP runs)
    # pre-calc total budget.
    total_budget = budget_s

    # Main hill climbing loop
    while time.time() - start_time < total_budget - polish_reserve:
        elapsed = time.time() - start_time
        progress = elapsed / (total_budget - polish_reserve)
        # schedule parameters:
        # step fraction decreasing
        step_frac = 0.4 * (1 - progress) + 0.02  # between ~0.42 to 0.02
        # temperature for acceptance
        init_temp = 5e-3
        final_temp = 5e-7
        temp = init_temp * ((final_temp / init_temp) ** progress)

        # decide move
        if rng.random() < 0.02:  # occasional random restart
            cand = random_feasible_vector(n, rng)
        else:
            cand = transfer_step(curr_h, rng, step_frac)

        cand_c = compute_c5(cand)

        # update best
        if cand_c < best_c:
            best_c = cand_c
            best_h = cand.copy()

        # acceptance
        if cand_c < curr_c or rng.random() < math.exp(-(cand_c - curr_c) / (temp + 1e-12)):
            curr_h = cand
            curr_c = cand_c

    # Final polishing with SLSQP and high beta
    # We'll try a few betas
    for beta in (200, 800, 2000, 8000):
        if time.time() - start_time > total_budget:
            break
        try:
            refined = slsqp_optimize(best_h, beta, max_iter=300)
            rc = compute_c5(refined)
            if rc < best_c:
                best_c = rc
                best_h = refined.copy()
        except Exception:
            pass

    # final clip and enforce sum
    enforce_sum(best_h, n / 2.0, rng)
    best_h = np.clip(best_h, 0.0, 1.0)

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)

    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta

def transfer_step(h, rng, step_frac):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode="full")
    ks = np.argmax(corr)
    offset = ks - (n - 1)
    if offset >= 0:
        idx = np.arange(0, n - offset, dtype=int)
    else:
        idx = np.arange(-offset, n, dtype=int)
    if idx.size < 2:
        # can't transfer; random perturb
        i, j = rng.choice(n, size=2, replace=False)
        max_eps = min(h[i], 1 - h[j])
        if max_eps <= 0:
            return h
        eps = max_eps * step_frac * rng.random()
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        return h_new

    prod = h[idx] * (1 - h[idx + offset])
    src = idx[np.argmax(prod)]
    dst = idx[np.argmin(prod)]
    if src == dst:
        # alternative selection
        if idx.size >= 2:
            src, dst = rng.choice(idx, size=2, replace=False)
        else:
            # fallback random pair
            i, j = rng.choice(n, size=2, replace=False)
            src, dst = i, j

    max_transfer = min(h[src], 1 - h[dst])
    if max_transfer <= 0:
        # try random pair
        i, j = rng.choice(n, size=2, replace=False)
        max_transfer = min(h[i], 1 - h[j])
        src, dst = i, j
        if max_transfer <= 0:
            return h
    eps = max_transfer * step_frac * rng.random()
    h_new = h.copy()
    h_new[src] -= eps
    h_new[dst] += eps
    return h_new

def project_sum_box(h, target, tol=1e-12, max_iter=100):
    lo = -1.0
    hi = 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        proj = np.clip(h + mid, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) < tol:
            return proj
        if s > target:
            hi = mid
        else:
            lo = mid
    return proj

def gradient_softmax(h, beta):
    n = h.size
    a = np.correlate(h, 1 - h, mode="full")  # length 2n-1
    max_a = a.max()
    # compute exponentials scaled
    exp_vals = np.exp(beta * (a - max_a))
    sum_exp = exp_vals.sum()
    p = exp_vals / sum_exp   # shape (2n-1)
    # compute gradient

import time, math, numpy as np
from scipy.optimize import minimize

# global for SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    """Discrete C5 for step function h on [0,2] with uniform grid."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)

def enforce_sum(h, target, rng):
    """Adjust h in‑place so that sum(h)=target while respecting bounds."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # final clip
    np.clip(h, 0.0, 1.0, out=h)
    return h

def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1] with sum n/2."""
    h = np.full(n, 0.5, dtype=float)
    for _ in range(n * 5):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.5
        h[i] -= eps
        h[j] += eps
    enforce_sum(h, n / 2.0, rng)
    return h

def transfer_step(h, rng, step_frac):
    """Move a fraction of mass from a high‑product index to a low‑product one."""
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode="full")
    ks = int(np.argmax(corr))
    offset = ks - (n - 1)
    # valid indices where both h[i] and h[i+offset] exist
    if offset >= 0:
        idx = np.arange(0, n - offset, dtype=int)
    else:
        idx = np.arange(-offset, n, dtype=int)
    if idx.size < 2:
        i, j = rng.choice(n, size=2, replace=False)
        max_transfer = min(h[i], 1 - h[j])
        if max_transfer <= 0:
            return h
        eps = max_transfer * step_frac * rng.random()
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        return h_new
    prod = h[idx] * (1 - h[idx + offset])
    src = idx[int(np.argmax(prod))]
    dst = idx[int(np.argmin(prod))]
    if src == dst:
        src, dst = rng.choice(idx, size=2, replace=False)
    max_transfer = min(h[src], 1 - h[dst])
    if max_transfer <= 0:
        i, j = rng.choice(n, size=2, replace=False)
        max_transfer = min(h[i], 1 - h[j])
        src, dst = i, j
        if max_transfer <= 0:
            return h
    eps = max_transfer * step_frac * rng.random()
    h_new = h.copy()
    h_new[src] -= eps
    h_new[dst] += eps
    return h_new

def subgradient(h):
    """Subgradient of the max‑overlap functional at h."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    ks = int(np.argmax(corr))
    offset = ks - (n - 1)
    grad = np.ones_like(h)  # term1 = 1 everywhere
    if offset >= 0:
        # i+offset in range for i < n-offset
        grad[: n - offset] = 1 - h[offset:]
        # subtract h[i - offset] where i >= offset
        grad[offset:] -= h[: n - offset]
        # for i >= n-offset, term1 stays 1 (out‑of‑range)
    else:
        d = -offset
        # i+offset in range for i >= d
        grad[d:] = 1 - h[: n - d]
        # subtract h[i - offset] = h[i + d] for i < n-d
        grad[: n - d] -= h[d:]
        # for i < d, term1 stays 1 (out‑of‑range)
    return grad

def project_box_sum(v, target, tol=1e-12, max_iter=60):
    """Project v onto [0,1]^n while enforcing sum(v)=target."""
    lo, hi = -1.0, 1.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) < tol:
            return proj
        if s > target:
            hi = mid
        else:
            lo = mid
    # final return
    proj = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    return proj

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    a = np.correlate(x, 1 - x, mode="full")
    b = CURRENT_BETA * a
    m = np.max(b)
    lse = m + np.log(np.sum(np.exp(b - m)))
    return lse / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=250):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

def compute_c5(h):
    """Discrete C5 for a step function on [0,2] using uniform grid."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)


def enforce_sum(h, target, rng):
    """Adjust h in‑place so that Σh = target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()

def transfer_step(h, rng, step_frac):
    """Move a fraction of mass from a high‑product index to a low‑product one."""
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode="full")
    ks = int(np.argmax(corr))
    offset = ks - (n - 1)

    if offset >= 0:
        idx = np.arange(0, n - offset, dtype=int)
    else:
        idx = np.arange(-offset, n, dtype=int)

    if idx.size < 2:
        i, j = rng.choice(n, size=2, replace=False)
        max_transfer = min(h[i], 1 - h[j])
        if max_transfer <= 0:
            return h
        eps = max_transfer * step_frac * rng.random()
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        return h_new

    prod = h[idx] * (1 - h[idx + offset])
    src = idx[int(np.argmax(prod))]
    dst = idx[int(np.argmin(prod))]
    if src == dst:
        src, dst = rng.choice(idx, size=2, replace=False)

        max_transfer = min(h[i], 1 - h[j])
        src, dst = i, j
        if max_transfer <= 0:
            return h

    eps = max_transfer * step_frac * rng.random()
    h_new = h.copy()
    h_new[src] -= eps
    h_new[dst] += eps
    return h_new


def subgradient(h):
    """Subgradient of max⁡_{k}∫h(x)(1−h(x+k))dx at the current h."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    ks = int(np.argmax(corr))
    offset = ks - (n - 1)
    grad = np.ones_like(h)            # term 1 for every index

    if offset >= 0:
        # i+offset valid for i < n‑offset
        grad[: n - offset] = 1 - h[offset:]
        # subtract h[i‑offset] for i ≥ offset
        grad[offset:] -= h[: n - offset]
    else:
        d = -offset
        # i+offset valid for i ≥ d
        grad[d:] = 1 - h[: n - d]
        # subtract h[i‑offset] = h[i+d] for i < n‑d
        grad[: n - d] -= h[d:]
    return grad


def project_box_sum(v, target, tol=1e-12, max_iter=60):
    """
    Project v onto the intersection of the box [0,1]^n
    and the hyperplane Σx = target.
    """
    lo, hi = -1.0, 1.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) < tol:
            return proj
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    a = np.correlate(x, 1 - x, mode="full")
    b = CURRENT_BETA * a
    m = np.max(b)
    lse = m + np.log(np.sum(np.exp(b - m)))
    return lse / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=250):
    """Smooth‑max SLSQP refinement."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x
