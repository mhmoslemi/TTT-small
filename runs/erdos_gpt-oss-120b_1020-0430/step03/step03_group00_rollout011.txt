def project_to_simplex_box(v, target, tol=1e-12, max_iter=100):
    lo = -np.max(v)
    hi = 1 - np.min(v)
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        proj = np.clip(v + mid, 0, 1)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo+hi)/2.0, 0, 1)

def convolve_center(w, h):
    n = len(h)
    L = 1 << ((len(w) + n - 1).bit_length())
    W = np.fft.rfft(w, n=L)
    H = np.fft.rfft(h, n=L)
    conv_full = np.fft.irfft(W * H, n=L)[:len(w) + n - 1]
    # extract center part length n

import time
import numpy as np

# ---------- projection onto [0,1] with sum constraint ----------
def _project_sum_box(v, target, tol=1e-12, max_iter=100):
    """Project v onto {x | 0<=x<=1, sum x = target}."""
    lo = -np.max(v)
    hi = 1.0 - np.min(v)
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

# ---------- correlation & C5 ----------
def _corr_full(h):
    """Linear correlation of h with (1-h). Returns length 2n-1 array."""
    n = h.size
    L = 1 << ((2 * n).bit_length())          # power of two >= 2n
    fh = np.fft.rfft(h, n=L)
    f1mh = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(fh * np.conj(f1mh), n=L)[: 2 * n - 1]
    return corr

def compute_c5_fft(h):
    """Exact discrete C5 for step vector h."""
    n = h.size
    max_corr = np.max(_corr_full(h))
    return max_corr * (2.0 / n)

# ---------- convolution helper ----------
def _conv_center(w, h):
    """Linear convolution w*h, returning the n central entries (j from 0..n-1)."""
    n = h.size
    L = 1 << ((len(w) + n - 1).bit_length())
    W = np.fft.rfft(w, n=L)
    H = np.fft.rfft(h, n=L)
    full = np.fft.irfft(W * H, n=L)[: len(w) + n - 1]
    start = n - 1
    return full[start:start + n]

# ---------- gradient of soft‑maxed overlap ----------
def gradient_softmax(h, beta):
    """Gradient of (1/beta) log sum exp(beta * corr) w.r.t. h."""
    n = h.size
    corr = _corr_full(h)                       # length 2n-1
    m = np.max(corr)
    w = np.exp(beta * (corr - m))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum                               # softmax weights, sum 1

    conv1 = _conv_center(w, h)                  # sum_k w_k * h[j-k]
    conv2 = _conv_center(w[::-1], h)            # sum_k w_k * h[j+k]
    grad = 1.0 - conv2 - conv1
    return grad

# ---------- random feasible vectors ----------
def random_feasible_vector(n, rng):
    """Uniform random in [0,1] then project to sum=n/2."""
    v = rng.random(n)
    return _project_sum_box(v, n / 2.0)

def perturb_random_move(h, rng, scale=0.1):
    """Transfer a small amount ε from i to j (preserves sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh
