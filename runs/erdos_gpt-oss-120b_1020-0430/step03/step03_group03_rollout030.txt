def de_optimize(n, target, rng, max_evals):
    # parameters
    pop_size = min(30, max(5, int(0.3 * n))) ??? maybe 30.
    # Initialize population
    pop = np.empty((pop_size, n))
    for i in range(pop_size):
        pop[i] = random_feasible_vector(n, rng)  # maybe different each.
    fitness = np.array([compute_c5(ind) for ind in pop])
    best_idx = np.argmin(fitness)
    best = pop[best_idx].copy()
    best_fit = fitness[best_idx]
    evals = pop_size
    
    # DE parameters
    F = 0.8
    CR = 0.9
    
    while evals < max_evals:
        for i in range(pop_size):
            # choose three distinct indices a,b,c
            idxs = list(range(pop_size))
            idxs.remove(i)
            a,b,c = rng.choice(idxs, size=3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            # repair: clip
            mutant = np.clip(mutant, 0.0, 1.0)
            enforce_sum(mutant, target, rng)  # modifies in place
            # crossover
            cross = np.empty(n)
            j_rand = rng.integers(n)
            for j in range(n):
                if rng.random() < CR or j == j_rand:
                    cross[j] = mutant[j]
                else:
                    cross[j] = pop[i][j]
            # repair again (to ensure sum)
            enforce_sum(cross, target, rng)
            # evaluate
            f = compute_c5(cross)
            evals += 1
            if f < fitness[i]:
                pop[i] = cross
                fitness[i] = f
                if f < best_fit:
                    best_fit = f
                    best = cross.copy()
            if evals >= max_evals:
                break
        # optional adaptation of F,CR based on progress
    return best, best_fit
```

        pop[i] = vec
        fitness[i] = compute_c5(vec)
    evals = popsize
    # best tracking
    best_idx = np.argmin(fitness)
    best_vec = pop[best_idx].copy()
    best_fit = fitness[best_idx]
    # DE parameters
    F = 0.8
    CR = 0.9
    # Heuristics: adapt F/CR maybe gradually
    # Main loop

    while abs(diff) > 1e-12:

import time, numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Discrete C5 for step function h (dx=2/n)."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)


def project_sum(x, target, rng):
    """Adjust x in‑place so that sum(x)=target while 0≤x≤1."""
    diff = target - x.sum()
    while abs(diff) > 1e-12:
        if diff > 0:
            slack = 1.0 - x
            i = np.argmax(slack)
            add = min(diff, slack[i])
            if add <= 0:
                break
            x[i] += add
            diff -= add
        else:
            i = np.argmax(x)
            sub = min(-diff, x[i])
            if sub <= 0:
                break
            x[i] -= sub
            diff += sub
    return x


def random_feasible_vector(n, rng):
    """Generate a random h with 0≤h≤1 and sum=n/2."""
    target = n / 2.0
    x = rng.random(n)
    x = x * (target / x.sum())
    np.clip(x, 0.0, 1.0, out=x)
    project_sum(x, target, rng)
    return x

def de_optimize(n, target, rng, init_h, time_limit, max_evals):
    start = time.time()
    popsize = max(5, int(0.12 * n))
    popsize = min(popsize, 30)
    pop = np.empty((popsize, n))
    fitness = np.empty(popsize)

    pop[0] = np.clip(init_h, 0.0, 1.0)
    project_sum(pop[0], target, rng)
    fitness[0] = compute_c5(pop[0])

    for i in range(1, popsize):
        vec = random_feasible_vector(n, rng)
        pop[i] = vec
        fitness[i] = compute_c5(vec)

    evals = popsize
    best_idx = np.argmin(fitness)
    best_vec = pop[best_idx].copy()
    best_fit = fitness[best_idx]

    F = 0.8
    CR = 0.9

    while evals < max_evals and (time.time() - start) < time_limit:
        for i in range(popsize):
            idxs = list(range(popsize))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)

            mutant = pop[a] + F * (pop[b] - pop[c])
            np.clip(mutant, 0.0, 1.0, out=mutant)
            project_sum(mutant, target, rng)

            trial = pop[i].copy()
            j_rand = rng.integers(n)
            rand_vals = rng.random(n)
            mask = (rand_vals < CR) | (np.arange(n) == j_rand)
            trial[mask] = mutant[mask]
            project_sum(trial, target, rng)

            f = compute_c5(trial)
            evals += 1
            if f < fitness[i]:
                pop[i] = trial
                fitness[i] = f
                if f < best_fit:
                    best_fit = f
                    best_vec = trial.copy()
            if evals >= max_evals or (time.time() - start) >= time_limit:
                break
    return best_vec, best_fit


# -------------------------------------------------
# SLSQP refinement using smooth max approximation
# -------------------------------------------------
CURRENT_TARGET = 0.0
CURRENT_BETA = 0.0


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=200):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    project_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x
