    # The shift that maximizes correlation is the one that minimizes product sum.
    prod_corr = np.correlate(h, h, mode='full')
    # compute sum_i h[i] * h[i+shift]; shift from -(n-1) to (n-1)
    # correlation_k = sum_i h[i] - prod_corr_shift; shift considered only non-negative? We'll just consider shift>=0.
    # Better to compute shift maximizing correlation for positive shift. We'll compute correlation directly for all shifts using compute_c5? That returns max of sum_i h[i]*(1-h[i+shift]).
    # compute_c5 uses correlation with 1-h. So we need shift of that max.
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)  # index in full array
    shift = max_idx - (n - 1)  # shift integer (can be negative)
    if shift < 0:
        # invert orientation for symmetry; we can treat positive shift
        shift = -shift
        # also we need to reflect indices; but product sum symmetric, so fine.
    # Now find i with high contribution: product = h[i] * h[i+shift] (if shift + i < n)
    # compute contributions array:
    if shift >= n:
        # shift out of range -> correlation zero; pick random move
        return perturb_h(h, rng, scale=0.2)
    contrib = h[:n-shift] * h[shift:]  # length n-shift
    # choose i weighted by contrib (higher products have higher chance)
    total_contrib = contrib.sum()
    if total_contrib <= 0:
        # no overlapping mass; random move
        return perturb_h(h, rng, scale=0.2)
    # sample i proportional to contrib
    probs = contrib / total_contrib
    i = rng.choice(np.arange(n-shift), p=probs)
    # choose j random from all indices where we can move mass
    # we need to ensure we can reduce product, moving mass from i (which contributes to overlap)
    # we can transfer epsilon from i to j where either j+shift or j-shift is low (i.e., where product small)
    # Simpler: pick j random.
    j = rng.integers(0, n)
    if i == j:
        # pick another
        j = (j+1) % n
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale=0.2)
    epsilon = rng.random() * max_eps * 0.5  # move small portion
    new_h = h.copy()
    new_h[i] -= epsilon
    new_h[j] += epsilon
    return new_h

def hill_climb_targeted(start_h, rng, max_iters, init_step=0.4):

    h = start_h.copy()
    best_c = compute_c5(h)
    step = init_step
    n = len(h)
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        # compute shift causing max correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        shift = max_idx - (n - 1)
        # generate a candidate
        if rng.random() < 0.7:
            new_h = targeted_perturb(h, rng, shift, scale=step)
        else:
            new_h = perturb_h(h, rng, scale=step)
        # Evaluate
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            h = new_h
            # maybe reduce step less aggressively?
        # else maybe accept some worsening occasionally
        elif rng.random() < 0.01:
            # accept bad move
            h = new_h
        # step decay
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.8, 1e-5)
    return h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# Global variables for the SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth log‑sum‑exp surrogate of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp shifted by m for numerical stability
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h so that Σh = target while respecting 0≤h≤1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        # tiny correction spread uniformly
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def targeted_perturb(h, rng, shift, scale=0.5):
    """Move mass away from a pair (i,i+shift) that contributes heavily."""
    n = h.size
    if shift == 0 or abs(shift) >= n:
        return perturb_h(h, rng, scale=scale)
    if shift < 0:
        shift = -shift
    # contribution of each i to overlap product h[i]*h[i+shift]
    contrib = h[: n - shift] * h[shift:]
    total = contrib.sum()
    if total <= 0.0:
        return perturb_h(h, rng, scale=scale)
    probs = contrib / total
    i = rng.choice(np.arange(n - shift), p=probs)
    j = rng.integers(0, n)
    if i == j:
        j = (j + 1) % n
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale=scale)
    eps = rng.random() * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_targeted(h_start, rng, max_iters=3000, init_step=0.5, sabotage_prob=0.01):
    """Hill‑climb using both random and targeted moves."""
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    n = h.size
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        # most offending shift
        corr = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        shift = max_idx - (n - 1)
        # generate candidate
        if rng.random() < 0.65:
            new_h = targeted_perturb(h, rng, shift, scale=step)
        else:
            new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            h = new_h
        elif rng.random() < sabotage_prob:
            # occasional acceptance of a worse move
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.8, 1e-5)
    return h, best_c


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=[cons],
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum (tiny numerical drift)
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x


def run(seed=42, budget_s=1100, **kwargs):
    """Optimise a step function h on [0,2] to minimise the Erdős C₅ bound."""
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # -------------------------------------------------------------
    # 1. Determine dimension and initial vector(s)
    # -------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 240
        h0 = random_feasible_vector(n, rng)

    # -------------------------------------------------------------
    # 2. Build a diverse candidate pool
    # -------------------------------------------------------------
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    candidates = [h0]

    # sinusoidal families
    for f in range(1, 9):
        s = 0.5 + 0.5 * np.sin(np.pi * f * xs)
        candidates.append(enforce_sum(np.clip(s, 0.0, 1.0), n / 2.0, rng))
        c = 0.5 + 0.5 * np.cos(np.pi * f * xs)
        candidates.append(enforce_sum(np.clip(c, 0.0, 1.0), n / 2.0, rng))

    # linear ramps
    candidates.append(enforce_sum(xs / 2.0, n / 2.0, rng))
    candidates.append(enforce_sum(1.0 - xs / 2.0, n / 2.0, rng))

    # binary blocks
    mask = np.zeros(n)
    mask[: n // 2] = 1.0
    candidates.append(enforce_sum(mask, n / 2.0, rng))

    mask = np.zeros(n)
    start = n // 4
    mask[start : start + n // 2] = 1.0
    candidates.append(enforce_sum(mask, n / 2.0, rng))

    mask = np.arange(n) % 2
    candidates.append(enforce_sum(mask, n / 2.0, rng))

    # random feasible vectors
    for _ in range(6):
        candidates.append(random_feasible_vector(n, rng))

    # -------------------------------------------------------------
    # 3. Pick the best among the pool
    # -------------------------------------------------------------
    global_best_h = None
    global_best_c = float("inf")
    for h in candidates:
        c = compute_c5(h)
        if c < global_best_c:
            global_best_c = c
            global_best_h = h.copy()

    # -------------------------------------------------------------
    # 4. Main optimisation loop (targeted hill‑climb + SLSQP ladder)
    # -------------------------------------------------------------
    beta_list = [10, 50, 200, 800, 2000]
    hc_iters = 3000

    while time.time() - start_time < budget_s * 0.75:
        # choose a start point
        if rng.random() < 0.6:
            start_vec = perturb_h(global_best_h, rng, scale=0.4)
        else:
            start_vec = random_feasible_vector(n, rng)

        # targeted hill climbing
        h_opt, c_opt = hill_climb_targeted(start_vec, rng, max_iters=hc_iters, init_step=0.5)

        if c_opt < global_best_c:
            global_best_h, global_best_c = h_opt.copy(), c_opt

        # SLSQP refinements with increasing β
        for beta in beta_list:
            if time.time() - start_time > budget_s * 0.75:
                break
            try:
                h_ref = slsqp_optimize(global_best_h, beta, max_iter=400)
            except Exception:
                continue
            c_ref = compute_c5(h_ref)
            if c_ref < global_best_c:
                global_best_h, global_best_c = h_ref.copy(), c_ref

    # -------------------------------------------------------------
    # 5. Optional up‑sampling and fine‑tuning if time permits
    # -------------------------------------------------------------
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > budget_s * 0.15:
        target_n = min(720, max(360, n * 2))
        if target_n != n:
            old_x = (np.arange(n) + 0.5) * (2.0 / n)
            new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
            h_up = np.interp(new_x, old_x, global_best_h)
            h_up = enforce_sum(h_up, target_n / 2.0, rng)

            # short hill‑climb on the finer grid
            h_up, c_up = hill_climb_targeted(h_up, rng, max_iters=4000, init_step=0.3)

            # SLSQP refinements on fine grid
            for beta in beta_list:
                if time.time() - start_time > budget_s:
                    break
                try:
                    h_up = slsqp_optimize(h_up, beta, max_iter=400)
                except Exception:
                    pass
                c_up = compute_c5(h_up)
                if c_up < global_best_c:
                    global_best_h, global_best_c, n = h_up.copy(), c_up, target_n

    # -------------------------------------------------------------
    # 6. Final clean‑up (clip + exact sum)
    # -------------------------------------------------------------
    global_best_h = np.clip(global_best_h, 0.0, 1.0)
    global_best_h = enforce_sum(global_best_h, n / 2.0, rng)
    final_c = compute_c5(global_best_h)

    return global_best_h.tolist(), float(final_c), int(n)
