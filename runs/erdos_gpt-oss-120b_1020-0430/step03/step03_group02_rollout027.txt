def softmax_grad(h, beta):
    n = len(h)
    corr = np.correlate(h, 1-h, mode='full')
    O = corr
    # softmax weights
    max_O = np.max(O)
    exp_beta_O = np.exp(beta * (O - max_O))
    w = exp_beta_O / np.sum(exp_beta_O)  # sum to 1, length L = 2n-1
    # gradient
    grad = np.ones_like(h)
    for shift in range(-n+1, n):
        w_shift = w[shift + n - 1]
        if w_shift == 0:
            continue
        # indices where i+shift is in range
        if shift >= 0:
            # i from 0..n-1-shift: h_{i+shift} exists
            grad[:n-shift] -= w_shift * h[shift:]
            # i >= shift: h_{i-shift} exists
            grad[shift:] -= w_shift * h[:n-shift]
        else:
            s = -shift
            # shift negative: i+shift = i - s
            # i from s..n-1: h_{i+shift} exists
            grad[s:] -= w_shift * h[:n-s]
            # i - shift = i + s exists for i from 0..n-1-s
            grad[:n-s] -= w_shift * h[s:]
    return grad, np.max(O)
```

def guided_move(h, grad, rng, max_transfer):
    pos_idx = np.argmax(grad)
    neg_idx = np.argmin(grad)
    if pos_idx == neg_idx:
        return h
    max_eps = min(h[pos_idx], 1.0 - h[neg_idx])
    eps = min(max_eps, max_transfer)
    if eps <= 0:
        return h
    h_new = h.copy()
    h_new[pos_idx] -= eps
    h_new[neg_idx] += eps
    return h_new
```

def softmax_grad_and_obj(h, beta):
    """Gradient of (1/β)·log Σ exp(β·O_k) where O_k = correlation."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    # soft‑max weights (sum to 1)
    exp_beta = np.exp(beta * (corr - max_corr))
    w = exp_beta / exp_beta.sum()
    # convolution w * h gives Σ_k w_k·h_{i+k}
    conv = np.convolve(w, h, mode="full")
    s = conv[n - 1 : n - 1 + n]          # length n
    grad = 1.0 - 2.0 * s
    # soft‑max objective value
    obj = max_corr + np.log(exp_beta.sum()) / beta
    return grad, obj

    best_c = np.inf
    best_n = None

    # start from parent if supplied
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0, c0, h0.size

    # candidate discretisations (small → medium)
    n_candidates = [84, 96, 108, 120, 144, 168, 192, 216, 240, 256]
    if best_n is not None:
        # try the current size first
        n_candidates = [best_n] + [n for n in n_candidates if n != best_n]

    # ----------------------------------------------------------------
    # repeated stochastic optimisation cycles
    # ----------------------------------------------------------------
    while time.time() - start < budget_s * 0.97:
        # rotate through n values (simple round‑robin)
        elapsed = time.time() - start
        n = n_candidates[int(elapsed // 5) % len(n_candidates)]

        # initialise h for this n
        if best_h is not None and best_n == n:
            h = best_h.copy()
        elif best_h is not None:
            # linear interpolation from the best known solution
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h = np.interp(x_new, x_old, best_h)
            h = np.clip(h, 0.0, 1.0)
            h = enforce_sum(h, rng)
        else:
            h = random_feasible_vector(n, rng)

        # ----- soft‑max descent with increasing β -----
        beta_schedule = [(1.0, 80), (5.0, 80), (20.0, 80), (80.0, 120), (200.0, 150)]
        for beta, iters in beta_schedule:
            # step size tuned to β and problem size
            step = 0.03 / (np.sqrt(beta) * (n / 192.0))
            h = softmax_descent(h, beta, step, rng, max_iter=iters)

        # ----- greedy hill‑climb -----
        max_hc = 2500 if n <= 192 else 1500
        h, c = guided_hill_climb(h, rng, max_iters=max_hc, step_scale=0.5)

        # a short fine‑grained climb
        h, c = guided_hill_climb(h, rng, max_iters=800, step_scale=0.2)

        c = compute_c5(h)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

        # optional up‑sampling if time permits (increase resolution)
        if time.time() - start > budget_s * 0.90:
            break

    # ----------------------------------------------------------------
    # final refinement with tiny random transfers (exploitation)
    # ----------------------------------------------------------------
    while time.time() - start < budget_s * 0.998 and best_h is not None:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # final feasibility enforcement
    if best_h is None:
        best_n = 84
        best_h = np.full(best_n, 0.5)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
