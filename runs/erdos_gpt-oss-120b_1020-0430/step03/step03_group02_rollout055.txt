def multi_shift_hill_climb(h, rng, max_iters=5000, m_shifts=5, step_start=0.1, step_decay=0.999):
    n = len(h)
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step_start
    for it in range(max_iters):
        O = np.correlate(h, 1 - h, mode='full')
        max_O = O.max()
        c = max_O * 2.0 / n   # dx factor
        if c < best_c:
            best_c = c; best_h = h.copy()
        # select top shifts
        # you can ignore shift indices at edges where O may be small
        # Compute top m
        top_idx = np.argpartition(O, -m_shifts)[-m_shifts:]
        shifts = top_idx - (n - 1)  # convert index to shift
        # compute gradient
        grad = np.full_like(h, float(m_shifts))
        for s in shifts:

import time, numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global placeholders for SLSQP constraints
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    """Discrete C5 for a step function on [0,2] with n points."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)

# ----------------------------------------------------------------------
# Equality constraint for SLSQP
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

# Soft‑max objective (smooth approximation of max_k O_k)
def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    """SLSQP refinement of h with softened max‑overlap objective."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # exact sum correction (pairwise redistribution)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def enforce_sum(h, rng):
    """Adjust h so that Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 10000:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        if add != 0:
            h[i] += add
            diff = target - h.sum()
        attempts += 1
    if abs(diff) > 1e-12:  # final tiny fix
        h[-1] += diff
        h[-1] = max(0.0, min(1.0, h[-1]))
    return h

def random_binary_feasible(n, rng):
    """Binary feasible vector with exactly n/2 ones."""
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

def worst_gradient(h):
    """Gradient of worst shift overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    """Transfer ε from a high‑gradient index to a low‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h_start, rng, max_iters=2500, step_scale=0.5):
    """Hill‑climb using the worst‑shift gradient."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

def gradient_refine(h, rng, step=0.02, max_iter=8):
    """Fine‑tune using explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h

            j = rng.integers(0, n)
            while i == j:
                j = rng.integers(0, n)
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        eps = step * max_transfer * rng.random()
        if eps <= 0:
            continue
        h[i] -= eps
        h[j] += eps
        step *= step_decay
    return best_h, best_c

# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # ------------------------------------------------------------------
    # Use parent construction if available
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0, c0, h0.size

    # ------------------------------------------------------------------
    # Candidate discretisations (keep them modest to stay in time)
    # ------------------------------------------------------------------
    MAX_N = 720
    base_ns = list(range(96, MAX_N + 1, 24))  # 96,120,…,720
    if best_n is not None and best_n not in base_ns:
        base_ns = [best_n] + base_ns
    candidate_ns = sorted(set(base_ns))

    # ------------------------------------------------------------------
    # Main optimisation loop over discretisations
    # ------------------------------------------------------------------
    for n in candidate_ns:
        if time.time() - start > 0.92 * budget_s:
            break

        # number of restarts for this n (more for small n)
        restarts = 2 if n <= 240 else 1

        for r in range(restarts):
            if time.time() - start > 0.92 * budget_s:
                break

            # initialise h for this n
            if best_h is not None and best_n == n and r == 0:
                h = best_h.copy()
            elif best_h is not None:
                # linear interpolation from best known shape
                x_old = np.linspace(0, 2, best_n, endpoint=False)
                x_new = np.linspace(0, 2, n, endpoint=False)
                h = np.interp(x_new, x_old, best_h)
                h = np.clip(h, 0.0, 1.0)
            else:
                h = random_binary_feasible(n, rng)

            h = enforce_sum(h, rng)

            # ----------------------------------------------------------------
            # Multi‑shift hill‑climb (coarse)
            # ----------------------------------------------------------------
            max_it = 2500 if n <= 240 else 1800 if n <= 480 else 1200
            h, c = multi_shift_hill_climb(h, rng,
                                          max_iters=max_it,
                                          m_shifts=7,
                                          step_start=0.12,
                                          step_decay=0.998)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

            # ----------------------------------------------------------------
            # Guided hill‑climb (worst‑shift)
            # ----------------------------------------------------------------
            h, c = guided_hill_climb(h, rng,
                                     max_iters=2000,
                                     step_scale=0.5)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

            # ----------------------------------------------------------------
            # SLSQP refinements with increasing β
            # ----------------------------------------------------------------
            for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0, 10000.0):
                if time.time() - start > 0.96 * budget_s:
                    break
                maxit = 500 if beta >= 200 else 400
                h = slsqp_optimize(h, beta, rng, max_iter=maxit)
                c = compute_c5(h)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n

            # ----------------------------------------------------------------
            # Fine gradient refinement
            # ----------------------------------------------------------------
            if time.time() - start < 0.97 * budget_s:
                h = gradient_refine(h, rng, step=0.02, max_iter=6)
                c = compute_c5(h)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n

            # ----------------------------------------------------------------
            # Short final guided hill‑climb
            # ----------------------------------------------------------------
            if time.time() - start < 0.98 * budget_s:
                h, c = guided_hill_climb(h, rng, max_iters=1200, step_scale=0.3)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n

    # ----------------------------------------------------------------------
    # Optional up‑sampling cascade (while time permits)
    # ----------------------------------------------------------------------
    while best_h is not None and time.time() - start < 0.99 * budget_s:
        if best_n * 2 > MAX_N:
            break
        h_up = np.repeat(best_h, 2)
        n_up = h_up.size

        # moderate local search on up‑sampled vector
        h_up, c_up = multi_shift_hill_climb(h_up, rng,
                                            max_iters=1500,
                                            m_shifts=9,
                                            step_start=0.1,
                                            step_decay=0.997)
        if c_up < best_c:
            best_h, best_c, best_n = h_up.copy(), c_up, n_up

        for beta in (5.0, 20.0, 80.0, 200.0, 500.0):
            if time.time() - start > budget_s:
                break
            h_up = slsqp_optimize(h_up, beta, rng, max_iter=300)
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, best_n = h_up.copy(), c_up, n_up

        h_up = gradient_refine(h_up, rng, step=0.02, max_iter=4)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up.copy(), c_up, n_up

        # prepare for possible next up‑sampling
        best_h, best_n = h_up, n_up

    # ----------------------------------------------------------------------
    # Final tiny random perturbations (use remaining budget)
    # ----------------------------------------------------------------------
    while time.time() - start < budget_s * 0.999 and best_h is not None:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # ----------------------------------------------------------------------
    # Final feasibility enforcement and return
    # ----------------------------------------------------------------------
    if best_h is None:
        best_n = 96
        best_h = np.full(best_n, 0.5)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
