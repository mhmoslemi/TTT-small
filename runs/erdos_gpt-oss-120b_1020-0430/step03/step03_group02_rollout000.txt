def enforce_sum(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # small tolerance; we use while loop with limit of maybe 1000 adjustments.
    # If diff > 0, need to add diff by increasing random indices not at 1.
    # If diff < 0, need to subtract.
    # We'll adjust per step max amount = min(diff, free capacity)
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            capacity = 1.0 - h[i]
            if capacity <= 0:
                continue
            add = min(diff, capacity)
        else:
            capacity = h[i]  # how much we can subtract
            if capacity <= 0:
                continue
            add = max(diff, -capacity)  # diff negative
        h[i] += add
        diff = target - h.sum()
    # If still not exactly target floating small diff, we can adjust first element
    if abs(diff) > 1e-9:
        h[0] += diff
    return h

def compute_c5(h):
    """Discrete C5 for step function on [0,2] with n samples."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, rng):
    """Adjust h in-place so that sum(h) = n/2 while keeping 0≤h≤1."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # Simple random redistribution
    for _ in range(2000):
        if abs(diff) < 1e-12:

            if cap <= 0.0:

            if cap <= 0.0:
                continue
            add = max(diff, -cap)
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-9:  # final tiny tweak
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    h = rng.random(n)
    target = n / 2.0
    h = h * (target / h.sum())
    h = np.clip(h, 0.0, 1.0)
    enforce_sum(h, rng)
    return h


def worst_gradient(h):
    """Gradient of the worst shift overlap O_k = Σ h_i(1‑h_{i+k})."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h, grad, rng, scale=0.5):
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h_start, rng, max_iters=3000, step_scale=0.5):
    """Hill‑climb using gradient‑directed transfers."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


# --------------------------------------------------------------
# Main optimisation routine
# --------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # Determine discretisation size
    if "initial_h_values" in globals():
        n = int(len(initial_h_values))
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        enforce_sum(h0, rng)
    else:
        n = 192
        h0 = None

    popsize = 180
    tournament = 3
    pc = 0.7          # crossover prob
    pm = 0.2          # mutation prob per child
    max_no_improve = 30

    # ----- initialise population -----
    population = []
    fitness = []
    if h0 is not None:
        population.append(h0.copy())
        fitness.append(compute_c5(h0))
    while len(population) < popsize:
        h = random_feasible_vector(n, rng)
        population.append(h)
        fitness.append(compute_c5(h))

    best_idx = int(np.argmin(fitness))
    best_h = population[best_idx].copy()
    best_c = fitness[best_idx]
    no_improve_gen = 0
    generation = 0

    # ----- evolutionary loop -----
    while time.time() - start < budget_s * 0.94:
        generation += 1
        new_pop = [best_h.copy()]          # elitism: keep current best
        new_fit = [best_c]

        while len(new_pop) < popsize:
            # tournament selection for two parents
            idx1 = rng.choice(popsize, tournament, replace=False)
            p1 = idx1[np.argmin(np.take(fitness, idx1))]
            idx2 = rng.choice(popsize, tournament, replace=False)
            p2 = idx2[np.argmin(np.take(fitness, idx2))]
            parent1 = population[p1]
            parent2 = population[p2]

            # crossover
            if rng.random() < pc:
                alpha = rng.random()
                child = alpha * parent1 + (1.0 - alpha) * parent2
            else:
                child = parent1.copy()

            # mutation
            if rng.random() < pm:
                if rng.random() < 0.5:
                    sigma = 0.05
                    child = child + rng.normal(scale=sigma, size=n)
                    child = np.clip(child, 0.0, 1.0)
                else:
                    grad = worst_gradient(child)
                    child = guided_perturb(child, grad, rng, scale=0.1)

            # repair sum constraint
            enforce_sum(child, rng)

            # evaluate
            c = compute_c5(child)

            # occasional local refinement when a new personal best appears
            if c < best_c and rng.random() < 0.1:
                child, c = guided_hill_climb(child, rng, max_iters=500, step_scale=0.3)

            new_pop.append(child)
            new_fit.append(c)

            if c < best_c:
                best_c = c
                best_h = child.copy()
                no_improve_gen = 0

        # replace old generation
        population = new_pop
        fitness = new_fit

        # stagnation handling
        no_improve_gen += 1
        if no_improve_gen >= max_no_improve:
            pm = min(0.5, pm * 1.3)          # boost mutation
            # inject fresh random individuals
            for i in range(popsize // 2, popsize):
                h = random_feasible_vector(n, rng)
                population[i] = h
                fitness[i] = compute_c5(h)
            no_improve_gen = 0

    # ----- final local optimisation -----
    remaining = budget_s - (time.time() - start)
    if remaining > 0.5:
        # modest hill‑climb; limit iteration count by remaining time estimate
        max_iters = min(5000, int(remaining * 2000))  # ~2000 iters per second
        best_h, best_c = guided_hill_climb(best_h, rng, max_iters=max_iters, step_scale=0.3)

    # final feasibility enforcement and evaluation
    enforce_sum(best_h, rng)
    best_c = compute_c5(best_h)

    return best_h.tolist(), float(best_c), int(n)
