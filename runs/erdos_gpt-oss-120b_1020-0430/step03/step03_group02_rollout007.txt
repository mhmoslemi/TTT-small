def pairwise_exchange(h, rng, max_iters, max_step_scale):
   best_h = h.copy()
   best_c = compute_c5(best_h)
   step_scale = max_step_scale
   for it in range(max_iters):
        # choose i (source) and j (dest)
        i = rng.integers(len(h))
        j = rng.integers(len(h))
        if i == j: continue
        # direction: if best_h[i] > best_h[j] try to move from i to j else swap direction
        if best_h[i] > best_h[j]:
            max_eps = min(best_h[i], 1 - best_h[j])
            direction = -1
        else:
            max_eps = min(best_h[j], 1 - best_h[i])
            direction = 1
        if max_eps <= 0: continue
        eps = rng.random() * max_eps * step_scale
        nh = best_h.copy()
        if direction == -1:
            nh[i] -= eps
            nh[j] += eps
        else:
            nh[i] += eps
            nh[j] -= eps
        c = compute_c5(nh)
        if c < best_c:
            best_h = nh
            best_c = c
        # schedule
        if (it+1) % (max_iters//10) == 0:
            step_scale *= 0.7
   return best_h,best_c

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utility functions ----------
def compute_c5(h):
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, rng):
    """Project h onto Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random feasible h."""
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h


def worst_gradient(h):
    """Gradient of the worst‑shift overlap."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h, grad, rng, scale=0.5):
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h_start, rng, max_iters=3000, step_scale=0.5):
    """Local hill‑climb using the worst‑shift gradient."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_h, best_c, h = nh.copy(), c, nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-5)
    return best_h, best_c

            maxit = 600 if beta >= 200 else 400
            h = slsqp_optimize(h, beta, rng, max_iter=maxit)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, cur_n

        # fine gradient refinement
        if time.time() - start < 0.90 * budget_s:
            h = gradient_refine(h, rng, step=0.01, max_iter=12)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, cur_n

        # random pairwise exchanges
        if time.time() - start < 0.95 * budget_s:
            h, c = pairwise_exchange(h, rng, max_iters=4000, init_step=0.5)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, cur_n

        # final guided hill‑climb
        if time.time() - start < 0.99 * budget_s:
            h, c = guided_hill_climb(h, rng, max_iters=2000, step_scale=0.3)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, cur_n

    # -------- final sanity and return ----------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
