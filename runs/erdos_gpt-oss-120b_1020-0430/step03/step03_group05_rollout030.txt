def de_optimize_sym(n, rng, max_time, init_x=None):
    half_n = n // 2
    target = n / 4.0
    popsize = max(20, min(80, int(0.2 * half_n)))  # e.g., 60
    pop = []
    if init_x is not None:
        pop.append(init_x.copy())
    while len(pop) < popsize:
        pop.append(random_feasible_vector_sym(half_n, rng, target))
    pop = np.array(pop)
    # compute scores
    scores = np.array([compute_c5(build_h_from_x(x, n)) for x in pop])
    best_idx = scores.argmin()
    best_x = pop[best_idx].copy()
    best_score = scores[best_idx]

    F = 0.5
    CR = 0.9

    start = time.time()
    while time.time() - start < max_time:
        for i in range(popsize):
            # mutation indices: choose r1,r2,r3 distinct from i
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, 3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            # crossover
            cross = rng.random(half_n) < CR
            if not cross.any():
                cross[rng.integers(0, half_n)] = True
            trial = np.where(cross, mutant, pop[i])
            # project to feasible
            trial = project_to_feasible(trial, target)
            trial_score = compute_c5(build_h_from_x(trial, n))
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_x = trial.copy()
    return best_x, best_score

def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # choose n
    n = 600
    if n % 2 != 0:
        n += 1
    target = n // 2

    # Build initial x
    init_x = None
    if 'initial_h_values' in globals():
        # upsample to n
        h_init = upsample_initial(n)
        # symmetrize
        h_sym = (h_init + h_init[::-1]) / 2.0
        # clip to [0,1]
        h_sym = np.clip(h_sym, 0., 1.)
        # take half
        half_n = n // 2
        x0 = h_sym[:half_n]
        # project to meet half target
        x0 = project_to_feasible(x0, n/4.0)
        init_x = x0

    # allocate DE time
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    de_time = min(remaining * 0.5, 400.0)  # cap
    # Ensure some time left for local refinements
    if de_time < 30:
        de_time = max(0, remaining - 30)  # fallback

    best_x, best_score = de_optimize_sym(n, rng, de_time, init_x)
    best_h = build_h_from_x(best_x, n)
    best_c = best_score

    # Hill climbing
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    # allocate hill climbing up to remaining*0.4 or at most 800 iterations
    if remaining > 5:
        max_iters = int(min(8000, remaining * 200))  # heuristic: 200 it per second
        # compute time limit for hill climb:
        hc_deadline = start_time + remaining * 0.45
        best_h, best_c = hill_climb(best_h, rng, max_iters=max_iters, start_step=0.3, time_limit=hc_deadline)

    # SLSQP refinement if time remains
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 5:
        # Run a few beta steps
        cur_h = best_h.copy()
        cur_c = best_c
        for beta in (10.0, 50.0, 200.0, 800.0, 2000.0):
            if time.time() - start_time > budget_s * 0.98:
                break
            cur_h = slsqp_optimize(cur_h, beta, rng, max_iter=300)
            cur_c = compute_c5(cur_h)
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
    # Final feasibility enforcement
    best_h = np.clip(best_h, 0., 1.)
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        # adjust by distributing diff
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)

def compute_c5(h):

    return np.max(corr) * (2.0 / n)

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb(h_start, rng, max_iters=2000, start_step=0.5, time_limit=None):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        if time_limit is not None and time.time() >= time_limit:
            break
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Helper functions ----------
def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_to_feasible(v, target):
    """Project v onto {0≤x≤1, sum(x)=target} by bisection on a shift."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def upsample_initial(to_n):
    """Linear interpolation of the global initial_h_values to length to_n."""
    init = np.array(initial_h_values, dtype=float)
    m = init.size
    xs_old = np.linspace(0.0, 2.0, m, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, to_n, endpoint=False)
    return np.interp(xs_new, xs_old, init)


def build_h_from_x(x, n):
    """Create a symmetric full h from half-vector x (n even)."""
    half = n // 2
    h = np.empty(n, dtype=float)
    h[:half] = x
    h[half:] = x[::-1]
    return h


def random_feasible_vector_sym(dim, rng, target):
    """Generate a random feasible half‑vector (0≤x≤1, sum=target)."""
    v = rng.random(dim)
    v = v / v.sum() * target
    return project_to_feasible(v, target)


# ---------- Differential Evolution (symmetric) ----------
def de_optimize_sym(n, rng, max_time, init_x=None):
    """DE on the symmetric representation (half the variables)."""
    half = n // 2
    target = n / 4.0
    popsize = max(20, min(80, int(0.2 * half)))
    pop = []
    if init_x is not None:
        pop.append(init_x.copy())
    while len(pop) < popsize:
        pop.append(random_feasible_vector_sym(half, rng, target))
    pop = np.array(pop)
    scores = np.array([compute_c5(build_h_from_x(x, n)) for x in pop])
    best_idx = scores.argmin()
    best_x = pop[best_idx].copy()
    best_score = scores[best_idx]

    F = 0.5
    CR = 0.9
    start = time.time()
    while time.time() - start < max_time:
        for i in range(popsize):
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, 3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            cross = rng.random(half) < CR
            if not cross.any():
                cross[rng.integers(0, half)] = True
            trial = np.where(cross, mutant, pop[i])
            trial = project_to_feasible(trial, target)
            trial_score = compute_c5(build_h_from_x(trial, n))
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_x = trial.copy()
    return best_x, best_score


# ---------- Local perturbation ----------
def perturb_h(h, rng, scale=0.2):
    """Transfer ε between two coordinates while preserving the total sum."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5, time_limit=None):
    """Hill‑climb accepting only improvements; optional time limit."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        if time_limit is not None and time.time() >= time_limit:
            break
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


# ---------- SLSQP with smooth max ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=300):
    """Run SLSQP on the exact objective using a smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Choose an even discretisation size.
    n = 600
    if n % 2 != 0:
        n += 1
    target_sum = n // 2

    # Prepare an initial symmetric half‑vector if initial data exist.
    init_x = None
    if "initial_h_values" in globals():
        h_up = upsample_initial(n)
        h_sym = (h_up + h_up[::-1]) / 2.0
        h_sym = np.clip(h_sym, 0.0, 1.0)
        init_x = project_to_feasible(h_sym[: n // 2], n / 4.0)

    # Allocate time for DE, ensuring some budget remains for refinement.
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 250:
        de_time = min(500.0, remaining - 200.0)
    else:
        de_time = max(5.0, remaining * 0.5)
    de_time = max(0.0, min(de_time, remaining))

    best_x, best_score = de_optimize_sym(n, rng, de_time, init_x)
    best_h = build_h_from_x(best_x, n)
    best_c = best_score

    # Hill‑climbing refinement (time‑bounded).
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 10:
        max_iters = int(min(8000, remaining * 200))
        hc_deadline = start_time + remaining * 0.45
        best_h, best_c = hill_climb(best_h, rng, max_iters=max_iters,
                                    start_step=0.3, time_limit=hc_deadline)

    # SLSQP refinement with increasing β, if time permits.
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 5:
        cur_h = best_h.copy()
        cur_c = best_c
        for beta in (10.0, 50.0, 200.0, 800.0, 2000.0):
            if time.time() - start_time > budget_s * 0.98:
                break
            cur_h = slsqp_optimize(cur_h, beta, rng, max_iter=300)
            cur_c = compute_c5(cur_h)
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()

    # Final feasibility enforcement.
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target_sum - best_h.sum()

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
