import time, math
import numpy as np
from scipy.optimize import minimize

# ---------- correlation and objective ---------- #
def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

# ---------- perturbation utilities ---------- #
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def hill_climb(h, rng, max_iters=5000, start_step=0.5):
    best_h = h.copy()
    best_c = compute_c5(best_h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        cand = perturb_h(best_h, rng, scale=step)
        c = compute_c5(cand)
        if c < best_c:
            best_c, best_h = c, cand.copy()
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-5)
    return best_h, best_c

# ---------- gradient refinement ---------- #
def gradient_refine(h, rng, step=0.02, max_iter=20):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        h -= step * (0.5 ** it) * g
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

# ---------- simulated annealing ---------- #
def simulated_annealing(h0, rng, deadline):
    n = h0.size
    target = n / 2.0
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    it = 0
    T0 = max(0.01 * cur_c, 1e-4)
    while time.time() < deadline and it < 20000:
        step = max(0.5 * (0.999 ** it), 5e-4)
        cand = perturb_h(cur_h, rng, scale=step)
        cand_c = compute_c5(cand)
        accept = False
        if cand_c < cur_c:
            accept = True
        else:
            T = T0 * (0.999 ** it) + 1e-12
            if rng.random() < math.exp(-(cand_c - cur_c) / T):
                accept = True
        if accept:
            cur_h, cur_c = cand, cand_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
        it += 1
    return best_h, best_c

# ---------- soft‑max surrogate for SLSQP ---------- #
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=800):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x

# ---------- up‑sampling helper ---------- #
def upsample_and_fix(h, factor, rng):
    n = h.size
    n_up = n * factor
    x_src = np.linspace(0.0, 2.0, n, endpoint=False)
    x_tgt = np.linspace(0.0, 2.0, n_up, endpoint=False)
    h_up = np.interp(x_tgt, x_src, h)
    target = n_up / 2.0
    diff = target - h_up.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n_up)
            add = min(diff, 1.0 - h_up[i]) if diff > 0 else max(diff, -h_up[i])
            h_up[i] += add
            diff = target - h_up.sum()
            if abs(diff) < 1e-12:
                break
    return h_up

# ---------- main driver ---------- #
