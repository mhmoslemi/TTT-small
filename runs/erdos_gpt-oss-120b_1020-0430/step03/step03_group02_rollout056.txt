import time, math, numpy as np
from scipy.optimize import minimize

# globals for the SLSQP soft‑max trick
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    """Discrete C5 for a step function on [0,2] with uniform spacing."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, rng):
    """Adjust vector h to satisfy Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Random feasible h (0≤h≤1, Σh=n/2)."""
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=400):
    """Soft‑max SLSQP optimisation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def worst_gradient(h):
    """Gradient of the worst‑shift overlap O_k = Σ h_i(1‑h_{i+k})."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h, grad, rng, scale=0.5):
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h_start, rng, max_iters=3000, step_scale=0.5):
    """Hill‑climb using gradient‑directed transfers."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h, rng, step=0.02, max_iter=8):
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0

                h[i] += add
                diff = target - h.sum()
    return h

    h = h0.copy()
    c = compute_c5(h)
    best_h, best_c = h.copy(), c
    n = h.size
    it = 0
    T0 = 0.002
    T_end = 1e-7
    while time.time() < end_time:
        it += 1
        T = max(T_end, T0 * (0.9995 ** it))
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_ij = min(h[i], 1.0 - h[j])
        max_ji = min(h[j], 1.0 - h[i])
        if max_ij <= 0 and max_ji <= 0:
            continue
        if max_ij > 0 and max_ji > 0:
            if rng.random() < 0.5:
                src, dst, max_eps = i, j, max_ij
            else:
                src, dst, max_eps = j, i, max_ji
        elif max_ij > 0:
            src, dst, max_eps = i, j, max_ij
        else:
            src, dst, max_eps = j, i, max_ji
        eps = max_eps * rng.random() * 0.6
        nh = h.copy()
        nh[src] -= eps
        nh[dst] += eps
        nc = compute_c5(nh)
        delta = nc - c
        if delta < 0 or rng.random() < math.exp(-delta / T):
            h, c = nh, nc
            if nc < best_c:
                best_h, best_c = nh.copy(), nc
    return best_h, best_c


# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # ------------------------------------------------------------------
    # start from parent construction if it exists
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0, c0, h0.size

    candidate_ns = [84, 96, 108, 120, 144, 168, 192, 240]
    if best_n is not None:
        candidate_ns = [best_n] + [n for n in candidate_ns if n != best_n]

    # ------------------------------------------------------------------
    # optimisation over several discretisations
    # ------------------------------------------------------------------
    for n in candidate_ns:
        if time.time() - start > 0.94 * budget_s:
            break

        # initialise h for this n
        if best_n == n and best_h is not None:
            h = best_h.copy()
        elif best_h is not None:
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h = np.interp(x_new, x_old, best_h)
            h = np.clip(h, 0.0, 1.0)
            h = enforce_sum(h, rng)
        else:
            if rng.random() < 0.5:
                h = np.zeros(n)
                h[rng.choice(n, n // 2, replace=False)] = 1.0
            else:
                h = random_feasible_vector(n, rng)

        # ------------------------------------------------------------------
        # guided hill‑climb
        # ------------------------------------------------------------------
        h, c = guided_hill_climb(h, rng, max_iters=2500, step_scale=0.5)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

        # ------------------------------------------------------------------
        # increasing‑β SLSQP refinement
        # ------------------------------------------------------------------
        for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0):
            if time.time() - start > 0.96 * budget_s:
                break
            maxit = 600 if beta >= 200 else 400
            h = slsqp_optimize(h, beta, rng, max_iter=maxit)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

        # ------------------------------------------------------------------
        # fine gradient refinement
        # ------------------------------------------------------------------
        if time.time() - start < 0.97 * budget_s:
            h = gradient_refine(h, rng, step=0.02, max_iter=8)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

        # ------------------------------------------------------------------
        # short final guided hill‑climb
        # ------------------------------------------------------------------
        if time.time() - start < 0.98 * budget_s:
            h, c = guided_hill_climb(h, rng, max_iters=1500, step_scale=0.3)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

    # ------------------------------------------------------------------
    # optional up‑sampling & refinement if time allows
    # ------------------------------------------------------------------
    if best_h is not None and time.time() - start < 0.99 * budget_s:
        if best_n * 2 <= 384:
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, best_n * 2, endpoint=False)
            up = np.interp(x_new, x_old, best_h)
            up = np.clip(up, 0.0, 1.0)
            up = enforce_sum(up, rng)
            up, c_up = guided_hill_climb(up, rng, max_iters=2000, step_scale=0.3)
            for beta in (5.0, 20.0, 80.0, 200.0, 500.0):
                if time.time() - start > budget_s:
                    break
                up = slsqp_optimize(up, beta, rng, max_iter=300)
            c_up = compute_c5(up)
            if c_up < best_c:
                best_h, best_c, best_n = up.copy(), c_up, best_n * 2

    # ------------------------------------------------------------------
    # simulated‑annealing refinement with remaining budget
    # ------------------------------------------------------------------
    remaining = budget_s - (time.time() - start)
    if best_h is not None and remaining > 5.0:
        best_h, best_c = simulated_annealing(best_h, rng, remaining, start)

    # ------------------------------------------------------------------
    # final feasibility enforcement and return
    # ------------------------------------------------------------------
    if best_h is None:
        best_n = 84
        best_h = np.full(best_n, 0.5)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
