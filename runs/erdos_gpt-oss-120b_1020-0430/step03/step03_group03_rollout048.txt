def biased_perturb_multi(h, rng, scale=0.2, top_ratio=0.1):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    # select ks where corr >= max_corr * (1 - top_ratio)
    threshold = max_corr * (1 - top_ratio)
    ks = np.where(corr >= threshold)[0] - (n - 1)  # shift indices
    # compute scores
    scores = np.zeros(n)
    for k in ks:
        # i from 0 to n-1 such that i+k in [0,n-1]
        i0 = max(0, -k)
        i1 = min(n, n - k)
        idx = np.arange(i0, i1)
        scores[idx] += (1 - h[idx + k])
    # also weight by h_i maybe
    scores = scores * h
    # Now select high score and low score indices
    if scores.max() == scores.min():
        return perturb_h(h, rng, scale)
    # choose high
    hi_thr = np.quantile(scores, 0.8)
    lo_thr = np.quantile(scores, 0.2)
    high_idx = np.where(scores >= hi_thr)[0]
    low_idx = np.where(scores <= lo_thr)[0]
    if high_idx.size == 0 or low_idx.size == 0:
        return perturb_h(h, rng, scale)
    i = rng.choice(high_idx)
    j = rng.choice(low_idx)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new
```

def simulated_annealing_multi(start, rng, max_iter, init_temp, final_temp):
    h = start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    n = len(h)
    log_ratio = math.log(final_temp / init_temp) if init_temp>0 else 0.0
    for it in range(1, max_iter+1):
        T = init_temp * math.exp(log_ratio * it / max_iter)
        # choose perturb
        if rng.random() < 0.7:
            cand = biased_perturb_multi(h, rng, scale=0.2)
        else:
            cand = perturb_h(h, rng, scale=0.2)
        cand_c = compute_c5(cand)
        diff = cand_c - cur_c
        if diff < 0 or rng.random() < math.exp(-diff / max(T, 1e-12)):
            h, cur_c = cand, cand_c
            if cand_c < best_c:
                best_c, best_h = cand_c, cand.copy()
    return best_h, best_c

    n = h.size
    i, j = rng.integers(0, n, size=2)

def reduce_max_shift_move(h, rng, step=0.2):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    k_star = np.argmax(corr) - (n - 1)
    # compute product contributions for this shift
    i0 = max(0, -k_star)
    i1 = min(n, n - k_star)
    idx = np.arange(i0, i1)
    prod = h[idx] * (1 - h[idx + k_star])
    if prod.size == 0:
        return perturb_h(h, rng, scale=step)
    i_local = np.argmax(prod)
    j_local = np.argmin(prod)
    i = idx[i_local]

    n = h.size
    for _ in range(10000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])

            continue
        h[i] += add
        diff = target - h.sum()
    return h

        if i == j:

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

def biased_perturb_multi(h, rng, scale=0.2, top_ratio=0.1):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    threshold = max_corr * (1 - top_ratio)
    ks = np.where(corr >= threshold)[0] - (n - 1)
    scores = np.zeros(n)
    for k in ks:
        i0 = max(0, -k)
        i1 = min(n, n - k)
        idx = np.arange(i0, i1)
        scores[idx] += (1 - h[idx + k])
    scores *= h
    if scores.max() == scores.min():
        return perturb_h(h, rng, scale)
    hi_thr = np.quantile(scores, 0.8)
    lo_thr = np.quantile(scores, 0.2)
    high_idx = np.where(scores >= hi_thr)[0]
    low_idx = np.where(scores <= lo_thr)[0]
    if high_idx.size == 0 or low_idx.size == 0:
        return perturb_h(h, rng, scale)
    i = rng.choice(high_idx)
    j = rng.choice(low_idx)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale)
    eps = max_eps * scale * rng.random()
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

def reduce_max_shift_move(h, rng, step=0.2):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    k_star = np.argmax(corr) - (n - 1)
    i0 = max(0, -k_star)
    i1 = min(n, n - k_star)
    idx = np.arange(i0, i1)
    prod = h[idx] * (1 - h[idx + k_star])
    if prod.size == 0:
        return perturb_h(h, rng, scale=step)
    i_local = np.argmax(prod)
    j_local = np.argmin(prod)
    i = idx[i_local]
    j = idx[j_local]
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale=step)
    eps = max_eps * step * rng.random()
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

    best_h = h.copy()

    cur_c = best_c

# SLSQP smoothing helpers
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=200):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta

    res = minimize(softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,

    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

import time, math, numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.size
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / n)

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    for _ in range(10000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add == 0:
            continue
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    enforce_sum(h, n / 2.0, rng)
    return h

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

def biased_perturb_multi(h, rng, scale=0.2, top_ratio=0.1):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    threshold = max_corr * (1 - top_ratio)
    ks = np.where(corr >= threshold)[0] - (n - 1)
    scores = np.zeros(n)
    for k in ks:
        i0 = max(0, -k)
        i1 = min(n, n - k)
        idx = np.arange(i0, i1)
        scores[idx] += (1 - h[idx + k])
    scores *= h
    if scores.max() == scores.min():
        return perturb_h(h, rng, scale)
    hi_thr = np.quantile(scores, 0.8)
    lo_thr = np.quantile(scores, 0.2)
    high_idx = np.where(scores >= hi_thr)[0]
    low_idx = np.where(scores <= lo_thr)[0]
    if high_idx.size == 0 or low_idx.size == 0:
        return perturb_h(h, rng, scale)
    i = rng.choice(high_idx)
    j = rng.choice(low_idx)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale)
    eps = max_eps * scale * rng.random()
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

def reduce_max_shift_move(h, rng, step=0.2):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    k_star = np.argmax(corr) - (n - 1)
    i0 = max(0, -k_star)
    i1 = min(n, n - k_star)
    idx = np.arange(i0, i1)
    prod = h[idx] * (1 - h[idx + k_star])
    if prod.size == 0:
        return perturb_h(h, rng, scale=step)
    i_local = np.argmax(prod)
    j_local = np.argmin(prod)
    i = idx[i_local]
    j = idx[j_local]
    if i == j:
        return perturb_h(h, rng, scale=step)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale=step)
    eps = max_eps * step * rng.random()
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

def simulated_annealing_multi(start, rng, max_iter, init_temp, final_temp):
    h = start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    n = h.size
    log_ratio = math.log(final_temp / init_temp) if init_temp > 0 else 0.0
    for it in range(1, max_iter + 1):
        T = init_temp * math.exp(log_ratio * it / max_iter) if init_temp > 0 else 0.0
        r = rng.random()
        if r < 0.5:
            cand = reduce_max_shift_move(h, rng, step=0.2)
        elif r < 0.8:
            cand = biased_perturb_multi(h, rng, scale=0.2)
        else:
            cand = perturb_h(h, rng, scale=0.2)
        cand_c = compute_c5(cand)
        diff = cand_c - cur_c
        if diff < 0 or (T > 0 and rng.random() < math.exp(-diff / T)):
            h, cur_c = cand, cand_c
            if cand_c < best_c:
                best_c, best_h = cand_c, cand.copy()
    return best_h, best_c

# --- SLSQP smoothing helpers ---
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=200):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {"type": "eq", "fun": sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_h_overall = None
    best_c_overall = np.inf
    best_n = None

    candidate_ns = [180, 240, 300]
    per_n_budget = budget_s / len(candidate_ns)

    for n in candidate_ns:
        if time.time() - start_time >= budget_s:
            break
        n_start = time.time()

        if "initial_h_values" in globals() and len(initial_h_values) == n:
            h0 = np.array(initial_h_values, dtype=float)
        else:
            h0 = random_feasible_vector(n, rng)
        h0 = np.clip(h0, 0.0, 1.0)
        enforce_sum(h0, n / 2.0, rng)
        best_h = h0.copy()
        best_c = compute_c5(best_h)

        # ---- Simulated annealing phase (≈80% of per_n_budget) ----
        while time.time() - n_start < per_n_budget * 0.8:
            if rng.random() < 0.6:
                start = perturb_h(best_h, rng, scale=0.3)
            else:
                start = random_feasible_vector(n, rng)

            remaining = per_n_budget * 0.8 - (time.time() - n_start)
            if remaining < 0.5:
                break
            max_iter = int(min(25000, remaining * 2000))
            max_iter = max(2000, max_iter)

            cand_h, cand_c = simulated_annealing_multi(
                start, rng, max_iter, init_temp=0.05, final_temp=1e-6
            )
            if cand_c < best_c:
                best_h, best_c = cand_h.copy(), cand_c

            for beta in (100, 500, 2000):
                if time.time() - n_start >= per_n_budget * 0.8:
                    break
                try:
                    refined = slsqp_optimize(best_h, beta, max_iter=200)
                    rc = compute_c5(refined)
                    if rc < best_c:
                        best_h, best_c = refined, rc
                except Exception:
                    continue

        # ---- Final polishing within remaining per_n_budget ----
        while time.time() - n_start < per_n_budget:
            for beta in (8000, 20000):
                if time.time() - n_start >= per_n_budget:
                    break
                try:
                    best_h = slsqp_optimize(best_h, beta, max_iter=200)
                    best_c = compute_c5(best_h)
                except Exception:
                    continue

        best_h = np.clip(best_h, 0.0, 1.0)
        enforce_sum(best_h, n / 2.0, rng)
        final_c = compute_c5(best_h)
        if final_c < best_c_overall:
            best_c_overall = final_c
            best_h_overall = best_h.copy()
            best_n = n

    # Use any remaining time for a final high‑beta polish of the best solution
    while time.time() - start_time < budget_s:
        if best_h_overall is None:
            break
        try:
            best_h_overall = slsqp_optimize(best_h_overall, 20000, max_iter=200)
            best_c_overall = compute_c5(best_h_overall)
        except Exception:
            break
        if time.time() - start_time > budget_s:
            break

    best_h_overall = np.clip(best_h_overall, 0.0, 1.0)
    enforce_sum(best_h_overall, best_n / 2.0, rng)
    final_c = compute_c5(best_h_overall)
    return best_h_overall.tolist(), float(final_c), int(best_n)
