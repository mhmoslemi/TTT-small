import time
import numpy as np
from scipy.optimize import minimize

# ---------- Global variables for the SLSQP surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None

# ---------- Helper utilities ----------
def enforce_sum(h, target, rng):
    """Adjust h so that sum(h) == target while keeping 0 <= h <= 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def correlate_full(h):
    """Full linear correlation of h with (1-h) using FFT."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Discrete C5 for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)


# ---------- Surrogate for SLSQP (smooth max) ----------
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Log-sum-exp smoothing of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = correlate_full(x)
    bc = CURRENT_BETA * corr
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ---------- Local optimisation primitives ----------
def hill_climb(h_start, rng, deadline, max_iters=2000, start_step=0.5):
    """Greedy hill‑climb (accept only improvements), stops at deadline."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


def simulated_annealing(start_h, rng, deadline, T0=0.02, Tf=1e-5, max_iters=1500):
    """Simple SA with exponential cooling."""
    cur_h = start_h.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        new_h = perturb_h(cur_h, rng, scale=0.35)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
    return best_h, best_c

    rng = np.random.default_rng(seed)
    start_time = time.time()
    # time slices
    main_deadline = start_time + 0.78 * budget_s

        h0 = np.full(n, 0.5)
        h0 = enforce_sum(h0, n / 2.0, rng)

    # ----- 2. build diverse starting candidates -----
    candidates = [h0]
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx

    # sinusoidal and cosine families
    for f in range(1, 7):
        h = 0.5 + 0.5 * np.sin(np.pi * f * xs)
        candidates.append(enforce_sum(np.clip(h, 0.0, 1.0), n / 2.0, rng))
        h = 0.5 + 0.5 * np.cos(np.pi * f * xs)
        candidates.append(enforce_sum(np.clip(h, 0.0, 1.0), n / 2.0, rng))
        # triangular wave
        h = 0.5 + 0.5 * np.abs(np.sin(np.pi * f * xs))
        candidates.append(enforce_sum(np.clip(h, 0.0, 1.0), n / 2.0, rng))

    # linear ramps
    candidates.append(enforce_sum(np.clip(xs / 2.0, 0.0, 1.0), n / 2.0, rng))
    candidates.append(enforce_sum(np.clip(1.0 - xs / 2.0, 0.0, 1.0), n / 2.0, rng))

    # binary step and alternating patterns
    step = np.zeros(n)
    step[: n // 2] = 1.0
    candidates.append(enforce_sum(step, n / 2.0, rng))

    alt = np.tile([0.0, 1.0], n // 2)
    candidates.append(enforce_sum(alt, n / 2.0, rng))

    # some random feasible vectors
    for _ in range(5):
        candidates.append(random_feasible_vector(n, rng))

    # ----- 3. pick best among pool -----
    best_h = None
    best_c = float("inf")
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    global_best_h = best_h
    global_best_c = best_c

    # ----- 4. Main optimisation loop (low resolution) -----
    beta_list = [5.0, 20.0, 80.0, 200.0, 800.0, 2000.0]

        # choose a start point
        if rng.random() < 0.65:
            start = perturb_h(global_best_h, rng, scale=0.3)
        else:
            start = random_feasible_vector(n, rng)

        # simulated annealing
        cand_h, cand_c = simulated_annealing(
            start, rng, main_deadline, T0=0.03, Tf=1e-5, max_iters=1800
        )
        if cand_c < global_best_c:
            global_best_h, global_best_c = cand_h.copy(), cand_c

        # hill‑climb refinement
        h_opt, c_opt = hill_climb(
            global_best_h, rng, main_deadline, max_iters=2500, start_step=0.4
        )
        if c_opt < global_best_c:
            global_best_h, global_best_c = h_opt.copy(), c_opt

        # SLSQP continuation with increasing β
        for beta in beta_list:
            if time.time() >= main_deadline:
                break
            try:
                h_opt = slsqp_optimize(global_best_h, beta, max_iter=400)
                c_opt = compute_c5(h_opt)
                if c_opt < global_best_c:
                    global_best_h, global_best_c = h_opt.copy(), c_opt
            except Exception:
                continue

    # ----- 5. Upsample to finer grid (if time permits) -----
    if time.time() < final_deadline and n * 2 <= 960:
        up_factor = 2
        h_up = np.repeat(global_best_h, up_factor)
        n_up = h_up.size
        target_up = n_up / 2.0
        h_up = enforce_sum(h_up, target_up, rng)

        # new deadlines for fine phase
        up_deadline = final_deadline

        # fine‑grained SA
        cand_h, cand_c = simulated_annealing(
            h_up, rng, up_deadline, T0=0.02, Tf=1e-5, max_iters=1500
        )
        if cand_c < global_best_c:
            global_best_h, global_best_c = cand_h.copy(), cand_c

        # fine‑grained hill‑climb
        h_opt, c_opt = hill_climb(
            global_best_h, rng, up_deadline, max_iters=5000, start_step=0.25

        if c_opt < global_best_c:

        # SLSQP on fine grid with higher β
        beta_fine = [5.0, 20.0, 80.0, 200.0, 800.0, 2000.0, 5000.0, 10000.0]
        for beta in beta_fine:
            if time.time() >= up_deadline:

                if c_opt < global_best_c:
                    global_best_h, global_best_c = h_opt.copy(), c_opt
            except Exception:
                continue

    # ----- 6. Final clean‑up -----
    global_best_h = np.clip(global_best_h, 0.0, 1.0)
    global_best_h = enforce_sum(global_best_h, global_best_h.size / 2.0, rng)
    final_c = compute_c5(global_best_h)

    return global_best_h.tolist(), float(final_c), int(global_best_h.size)
