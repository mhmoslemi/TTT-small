    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample_interpolate(old_h, new_n):
    old_n = old_h.size
    xs_old = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    xs_new = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    new_h = np.interp(xs_new, xs_old, old_h)
    # clip to [0,1]
    new_h = np.clip(new_h, 0.0, 1.0)
    return new_h

def simulated_annealing(h0, target_sum, rng, time_limit):
    n = h0.size
    h = h0.copy()
    # ensure sum
    h = enforce_sum(h, target_sum, rng)
    c_curr = compute_c5(h)
    best_h = h.copy()
    best_c = c_curr

    start = time.time()
    # initial temperature: perhaps set based on typical delta (maybe 0.01)
    T = 0.5
    T_min = 1e-5
    alpha = 0.995
    # schedule: after each n moves, reduce T
    moves_per_temp = max(1000, n)
    move_count = 0

    while time.time() - start < time_limit:
        # propose a move
        i = rng.integers(0, n)
        if h[i] <= 0:
            continue
        j = rng.integers(0, n)
        if h[j] >= 1:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        # scale eps with temperature
        # epsilon can be up to max_eps; we sample eps = max_eps * rng.random() * min(1, T*5)
        scale = min(1.0, T * 5.0)
        eps = max_eps * rng.random() * scale
        if eps <= 0:
            continue
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps

        # compute new C5
        c_new = compute_c5(new_h)
        delta = c_new - c_curr
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = new_h
            c_curr = c_new
            # update best
            if c_new < best_c:
                best_c = c_new
                best_h = new_h.copy()
        move_count += 1
        if move_count >= moves_per_temp:
            # reduce temperature
            T = max(T * alpha, T_min)
            move_count = 0
            # occasional sum enforcement due to floating errors
            h = enforce_sum(h, target_sum, rng)
            c_curr = compute_c5(h)  # recompute after enforcement
    # final enforcement
    best_h = enforce_sum(best_h, target_sum, rng)
    best_c = compute_c5(best_h)
    return best_h, best_c

    rng = np.random.default_rng(seed)
    start_time = time.time()
    
    # initial vector

        h_current = np.array(initial_h_values, dtype=float)

        h_current = np.full(n0, 0.5)
    # enforce proper sum

    # allocate proportional to n (more time at larger n)

    # If no further resolution, fallback to SA on current.
    for idx, n in enumerate(n_list):
        # compute remaining time
        now = time.time()

        time_left = budget_s - elapsed
        if time_left <= 0:
            break
        if idx == 0:

import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def enforce_sum(h, target, rng):
    """Clip to [0,1] and adjust to make sum exactly target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i])
            add *= rng.random()
            h[i] += add
            diff -= add
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i])
            sub *= rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        if diff > 0:
            h[i] = min(1.0, h[i] + diff)
        else:
            h[i] = max(0.0, h[i] + diff)
    return h

def compute_c5(h):
    """Exact discrete C₅."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def upsample_interpolate(old_h, new_n):
    """Linear interpolation to a finer grid."""
    old_n = old_h.size
    xs_old = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    xs_new = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    new_h = np.interp(xs_new, xs_old, old_h)
    return np.clip(new_h, 0.0, 1.0)

# ---------- simulated annealing ----------
def simulated_annealing(h0, target_sum, rng, time_limit):
    n = h0.size
    h = enforce_sum(h0.copy(), target_sum, rng)
    c_curr = compute_c5(h)
    best_h, best_c = h.copy(), c_curr

    start = time.time()
    T = 1.0
    T_min = 1e-5
    alpha = 0.995
    moves_per_temp = max(500, n)
    moves = 0

    while time.time() - start < time_limit:
        # propose transfer
        i = rng.integers(0, n)
        if h[i] <= 0.0:
            continue
        j = rng.integers(0, n)
        if h[j] >= 1.0:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        scale = min(1.0, T * 5.0)
        eps = max_eps * rng.random() * scale
        if eps <= 0.0:
            continue
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps

        c_new = compute_c5(new_h)
        delta = c_new - c_curr
        if delta < 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = new_h
            c_curr = c_new
            if c_new < best_c:
                best_c, best_h = c_new, new_h.copy()

        moves += 1
        if moves >= moves_per_temp:
            T = max(T * alpha, T_min)
            moves = 0
            h = enforce_sum(h, target_sum, rng)
            c_curr = compute_c5(h)

    best_h = enforce_sum(best_h, target_sum, rng)
    best_c = compute_c5(best_h)
    return best_h, best_c

# ---------- perturbations for hill climbing ----------
def perturb_random(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def perturb_targeted(h, grad, k, rng, scale=0.2):
    pos = np.where((grad > 1e-9) & (h > 1e-12))[0]
    neg = np.where((grad < -1e-9) & (h < 1.0 - 1e-12))[0]
    if pos.size == 0 or neg.size == 0:
        return perturb_random(h, rng, scale)
    i = rng.choice(pos)
    j = rng.choice(neg)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return perturb_random(h, rng, scale)
    eps = max_eps * scale * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

# ---------- targeted hill climbing ----------
def hill_climb_targeted(h_start, rng, max_iters, start_step, target_sum, time_limit=None):
    h = h_start.copy()
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    best_c = np.max(corr) * (2.0 / n)
    best_h = h.copy()
    step = start_step
    decay_interval = max(1, max_iters // 10)
    start = time.time()
    for it in range(max_iters):
        if time_limit is not None and time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr)
        max_corr = corr[idx]
        c_curr = max_corr * (2.0 / n)
        k = idx - (n - 1)

        # gradient of the worst shift
        grad = np.ones(n)
        if k >= 0:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
        else:
            kk = -k
            grad[: n - kk] -= h[kk:]
            grad[kk:] -= h[: n - kk]

        new_h = perturb_targeted(h, grad, k, rng, scale=step)
        corr_new = np.correlate(new_h, 1 - new_h, mode="full")
        c_new = np.max(corr_new) * (2.0 / n)

        if c_new < best_c:
            best_c, best_h = c_new, new_h.copy()
        if c_new < c_curr:
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    best_h = enforce_sum(best_h, target_sum, rng)
    best_c = compute_c5(best_h)
    return best_h, best_c

# ---------- smooth max + SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x
