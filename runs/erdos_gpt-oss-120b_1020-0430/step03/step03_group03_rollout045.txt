import time, math, numpy as np
from scipy.optimize import minimize

# ---------- core utilities ----------
def compute_c5(h):
    """Discrete C5 for step function h on [0,2] with dx = 2/len(h)."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)


def enforce_sum(h, target, rng):
    """Adjust h in place to satisfy sum(h)=target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    # simple proportional correction with random picks
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # tiny remainder fix
    if abs(diff) > 1e-12:
        # distribute remaining diff evenly (might violate bounds by epsilon)
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
        enforce_sum(h, target, rng)
    return h


# ---------- SLSQP surrogate for smooth max ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smoothed max using log-sum-exp with temperature 1/CURRENT_BETA."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log-sum-exp stable
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=200):
    """Run SLSQP on smooth max approximation."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {"type": "eq", "fun": sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x


# ---------- Main optimisation ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # ---- initialise ----
    n = 240  # points (dx = 2/n)
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        old_n = h0.size
        # linear interpolation to new grid
        x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x_new, x_old, h0)
    else:
        h = np.full(n, 0.5)
        # add small random jitter while preserving sum
        for _ in range(500):
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0:
                continue
            eps = max_eps * rng.random() * 0.4
            h[i] -= eps
            h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    enforce_sum(h, n / 2.0, rng)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ---- iterative improvement ----
    while time.time() - start < budget_s - 5.0:  # keep a few seconds for polishing
        # compute current worst shift
        corr = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        s = max_idx - (n - 1)   # shift (positive means h shifted right)
        cur_c = corr[max_idx] * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()

        # inner directed moves focusing on this shift
        inner_iters = 2000
        for _ in range(inner_iters):
            if time.time() - start > budget_s - 5.0:
                break
            # overlapping index range for shift s
            if s >= 0:
                i_start = 0
                i_end = n - s
            else:
                i_start = -s
                i_end = n
            idx = np.arange(i_start, i_end)
            prod = h[idx] * (1.0 - h[idx + s])
            # high‑product and low‑product positions
            hi_loc = np.argmax(prod)
            lo_loc = np.argmin(prod)
            if hi_loc == lo_loc:
                # pick a different low if possible
                cand = np.where(prod != prod[hi_loc])[0]
                if cand.size:
                    lo_loc = rng.choice(cand)
                else:
                    # fallback random transfer
                    i, j = rng.integers(0, n, size=2)
                    if i != j:
                        max_eps = min(h[i], 1.0 - h[j])
                        if max_eps > 0:
                            eps = max_eps * rng.random() * 0.03
                            h[i] -= eps
                            h[j] += eps
                    continue
            i_high = idx[hi_loc]
            i_low = idx[lo_loc]
            # transfer a modest amount from high to low
            max_eps = min(h[i_high], 1.0 - h[i_low])
            if max_eps <= 0:
                continue
            eps = max_eps * rng.random() * 0.04
            h[i_high] -= eps
            h[i_low] += eps

            # occasional random jitter to avoid stagnation
            if rng.random() < 0.008:
                i, j = rng.integers(0, n, size=2)
                if i != j:
                    max_eps = min(h[i], 1.0 - h[j])
                    if max_eps > 0:
                        eps = max_eps * rng.random() * 0.06
                        h[i] -= eps
                        h[j] += eps

        # occasional SLSQP refinement if time permits
        if time.time() - start < budget_s - 8.0:
            try:
                refined = slsqp_optimize(h, beta=8000, max_iter=150)
                rc = compute_c5(refined)
                if rc < best_c:
                    best_c = rc
                    best_h = refined.copy()
                    h = refined.copy()
                    continue
            except Exception:
                pass

        # small chance to re‑anchor around the best solution
        if rng.random() < 0.07:
            h = best_h.copy()
            for _ in range(20):
                i, j = rng.integers(0, n, size=2)
                if i == j:
                    continue
                max_eps = min(h[i], 1.0 - h[j])
                if max_eps <= 0:
                    continue
                eps = max_eps * rng.random() * 0.02
                h[i] -= eps
                h[j] += eps

        enforce_sum(h, n / 2.0, rng)

    # ---- final polishing with high‑beta SLSQP ----
    for beta in (15000, 30000, 60000):
        if time.time() - start > budget_s:
            break
        try:
            best_h = slsqp_optimize(best_h, beta=beta, max_iter=200)
        except Exception:
            continue

    best_h = np.clip(best_h, 0.0, 1.0)
    enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
