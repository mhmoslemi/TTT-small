import time
import numpy as np

# ----------------------- utilities -----------------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function h on [0,2] with uniform spacing."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> None:
    """Exact sum correction by moving mass between feasible coordinates."""
    diff = target - h.sum()
    n = h.size
    # simple random redistribution; will terminate because capacity exists
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector in [0,1] with exact sum n/2."""
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    adjust_sum(h, target, rng)
    return h


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_local(h_start: np.ndarray, rng: np.random.Generator,
                     max_iters: int = 2000, step: float = 0.2):
    """Deterministic hill‑climb (accept only improvements)."""
    h = np.clip(h_start, 0.0, 1.0)
    n = h.size
    target = n / 2.0
    adjust_sum(h, target, rng)
    best_h = h.copy()
    best_c = compute_c5(h)
    for _ in range(max_iters):
        cand = perturb_h(h, rng, scale=step)
        c = compute_c5(cand)
        if c < best_c:
            best_c = c
            best_h = cand.copy()
            h = cand
    return best_h, best_c


def gradient_sa_optimize(h0: np.ndarray, rng: np.random.Generator, deadline: float):
    """Simulated‑annealing style descent using the exact worst‑shift gradient."""
    n = h0.size
    target = n / 2.0
    h = np.clip(h0, 0.0, 1.0)
    adjust_sum(h, target, rng)
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c

    start = time.time()
    T0 = cur_c * 0.1 + 1e-12  # initial temperature

    while time.time() < deadline:
        # progress fraction for schedules
        frac = (time.time() - start) / max(1e-12, (deadline - start))
        step = 0.10 * (1.0 - frac) + 1e-5 * frac
        # worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)

        # gradient of C_k
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            g = np.zeros_like(h)
            g[: n - k] += 1.0 - h[k:]
            g[k :] -= h[: n - k]

        # propose new point
        new_h = h - step * g
        new_h = np.clip(new_h, 0.0, 1.0)
        adjust_sum(new_h, target, rng)
        new_c = compute_c5(new_h)

        # acceptance rule
        if new_c < cur_c:
            accept = True
        else:
            T = T0 * max(0.0, 1.0 - frac)
            accept = rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-12))

        if accept:
            h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, h.copy()

        # occasional random jump to avoid stagnation
        if rng.random() < 0.001:
            jump = perturb_h(h, rng, scale=step)
            jc = compute_c5(jump)
            if jc < cur_c:
                h, cur_c = jump, jc
                if cur_c < best_c:
                    best_c, best_h = cur_c, h.copy()
    return best_h, best_c


def upsample_h(h_src: np.ndarray, n_target: int, rng: np.random.Generator) -> np.ndarray:
    """Linear interpolation of a step function to a finer grid, with exact sum."""
    x_src = np.linspace(0.0, 2.0, h_src.size, endpoint=False)
    x_tgt = np.linspace(0.0, 2.0, n_target, endpoint=False)
    h_tgt = np.interp(x_tgt, x_src, h_src)
    target = n_target / 2.0
    adjust_sum(h_tgt, target, rng)
    return h_tgt


# ----------------------- main driver -----------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    final_deadline = start_time + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    # parent solution (if provided)
    parent = globals().get("initial_h_values")
    if parent is not None:
        parent = np.array(parent, dtype=float)

    # candidate discretisations (moderate size)
    candidate_ns = [144, 180, 240, 300]

    for n in candidate_ns:
        # stop early if we are out of time
        if time.time() >= final_deadline:
            break

        # allocate a fair share of the remaining budget for this n
        remaining_time = final_deadline - time.time()
        if remaining_time < 20.0:
            break
        # simple equal sharing amongst the remaining n values
        ns_left = len(candidate_ns) - candidate_ns.index(n)
        per_n_deadline = time.time() + remaining_time / ns_left

        # assemble a few diverse starts
        starts = []

        if parent is not None:
            if parent.size == n:
                starts.append(parent.copy())
            else:
                starts.append(upsample_h(parent, n, rng))

        # random binary feasible
        h_bin = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h_bin[idx] = 1.0
        starts.append(h_bin)

        # random continuous feasible
        starts.append(random_feasible_vector(n, rng))

        # run optimisation from each start
        for start_idx, h0 in enumerate(starts):
            if time.time() >= per_n_deadline:
                break
            # give each start a slice of the remaining time
            starts_left = len(starts) - start_idx
            local_deadline = time.time() + max(5.0, (per_n_deadline - time.time()) / starts_left)

            # gradient‑SA phase
            h_opt, c_opt = gradient_sa_optimize(h0, rng, local_deadline)

            # hill‑climb refinement
            h_ref, c_ref = hill_climb_local(h_opt, rng, max_iters=2500, step=0.2)

            if c_ref < best_c:
                best_c, best_h, best_n = c_ref, h_ref.copy(), n

    # final safety checks
    if best_h is None:
        best_n = 180
        best_h = np.full(best_n, 0.5)

    best_h = np.clip(best_h, 0.0, 1.0)
    target_sum = best_n / 2.0
    adjust_sum(best_h, target_sum, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
