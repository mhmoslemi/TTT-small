def compute_c5_fast(h):
    n = h.size
    dx = 2.0 / n
    # compute correlation via FFT
    # conv of h with reversed (1-h)
    rev = (1.0 - h)[::-1]
    m = n + n - 1
    L = 1 << (m - 1).bit_length()  # next power of two
    H = np.fft.rfft(h, L)
    R = np.fft.rfft(rev, L)
    conv = np.fft.irfft(H * R, L)[:m]
    return np.max(conv) * dx
```

import time, math
import numpy as np
from scipy.optimize import minimize

# ---------- helpers ----------
def sum_eq_constraint(x):
    return np.sum(x) - CUR_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CUR_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CUR_BETA

def compute_c5_fast(h):
    """max_k ∫ h(x)(1-h(x+k))dx approximated by FFT."""
    n = h.size
    dx = 2.0 / n
    rev = (1.0 - h)[::-1]
    m = 2 * n - 1
    L = 1 << (m - 1).bit_length()          # next power of two
    H = np.fft.rfft(h, L)
    R = np.fft.rfft(rev, L)
    conv = np.fft.irfft(H * R, L)[:m]
    return np.max(conv) * dx

def project_to_feasible(v, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def random_feasible_vector(n, rng):
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_to_feasible(v, n / 2.0)

def upsample_initial(to_n):
    init = np.array(initial_h_values, dtype=float)
    m = init.size
    xs_old = np.linspace(0.0, 2.0, m, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, to_n, endpoint=False)
    new = np.interp(xs_new, xs_old, init)
    return new

def binary_from_continuous(arr):
    n = arr.size
    m = n // 2
    idx = np.argpartition(-arr, m)[:m]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    return b

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(n)
    j = rng.integers(n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    h2 = h.copy()
    h2[i] -= eps
    h2[j] += eps
    return h2

def hill_climb(h_start, rng, max_iters=1500, start_step=0.3):
    h = h_start.copy()
    best = h.copy()
    best_c = compute_c5_fast(h)
    step = start_step
    decay = max(1, max_iters // 8)
    for it in range(max_iters):
        cand = perturb_h(h, rng, scale=step)
        c = compute_c5_fast(cand)
        if c < best_c:
            best_c, best = c, cand.copy()
            h = cand
        if (it + 1) % decay == 0:
            step = max(step * 0.85, 1e-4)
    return best, best_c

def slsqp_optimize(h0, beta, rng, max_iter=300):
    global CUR_TARGET, CUR_BETA
    n = h0.size
    CUR_TARGET = n / 2.0
    CUR_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj, h0, method="SLSQP", bounds=bounds,
        constraints=cons, options={"maxiter": max_iter, "ftol": 1e-12, "disp": False}
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CUR_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CUR_TARGET - x.sum()
    return x

def anneal_binary(h, rng, time_budget):
    n = h.size
    cur = h.copy()
    cur_c = compute_c5_fast(cur)
    best = cur.copy()
    best_c = cur_c
    start = time.time()
    # linear cooling
    T0, T1 = 0.4, 1e-4
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = T0 * (1 - frac) + T1 * frac
        # propose swap 1->0,0->1
        i = rng.integers(n)
        j = rng.integers(n)
        if cur[i] == cur[j]:
            continue
        # ensure i is 1 and j is 0
        if cur[i] == 0:
            i, j = j, i
        # perform swap
        cur[i] = 0.0
        cur[j] = 1.0
        cand_c = compute_c5_fast(cur)
        delta = cand_c - cur_c
        accept = delta < 0 or rng.random() < math.exp(-delta / T)
        if accept:
            cur_c = cand_c
            if cand_c < best_c:
                best_c, best = cand_c, cur.copy()
        else:
            # revert
            cur[i] = 1.0
            cur[j] = 0.0
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_total = time.time()
    best_h, best_c, best_n = None, np.inf, None

    # Decide working resolution
    if "initial_h_values" in globals():

    # binary start

    remaining = budget_s - (time.time() - start_total)
    t_ann = remaining * 0.68
    t_sl = remaining * 0.22
    t_hc = remaining * 0.08

    # ---- simulated annealing (binary) ----
    h_ann, c_ann = anneal_binary(h_bin, rng, t_ann)

    # ---- continuous refinement (SLSQP) ----
    cur = h_ann.copy()
    for beta in (10.0, 50.0, 200.0, 800.0, 2000.0):

    if time.time() - start_total < budget_s * 0.98:

    best_h, best_c, best_n = cur, c_cur, n
