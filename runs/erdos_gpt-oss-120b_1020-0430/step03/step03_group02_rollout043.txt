    n = len(h)
    size = 2*n-1
    delta = np.zeros(size, dtype=int)
    # positive shifts where i <= n-1-s, i.e., s in 1..n-1-i
    s_pos = np.arange(1, n - i)  # length n-i-1?

def reduce_worst_shift_swap(h, O_current):
    # find worst shift index
    idx = np.argmax(O_current)
    if idx == (len(O_current) - 1) // 2:
        # O_0 maybe zero
        return False
    shift = idx - (len(O_current) // 2)
    n = len(h)
    # Determine sign
    if shift > 0:
        # positive shift
        mismatched_1 = np.where((h[:n-shift] == 1) & (h[shift:] == 0))[0]  # i positions
        # candidate 0 where opposite mismatched
        mismatched_0 = np.where((h[:n-shift] == 0) & (h[shift:] == 1))[0]
    else:
        shift_abs = -shift
        mismatched_1 = np.where((h[shift_abs:] == 1) & (h[:n-shift_abs] == 0))[0] + shift_abs  # positions i (where i>=shift_abs)
        # Actually for negative shift, O_{-s} = sum_{i=s}^{n-1} h_i * (1 - h_{i-s});

        # Use same logic but swapped indices.
        mismatched_0 = np.where((h[shift_abs:] == 0) & (h[:n-shift_abs] == 1))[0] + shift_abs

    i = mismatched_1[0]
    j = mismatched_0[0]

    # removal i
    delta_i = -compute_delta(h, i)

def compute_delta(h, i):
    n = h.shape[0]
    size = 2*n - 1
    delta = np.zeros(size, dtype=int)
    mid = n - 1
    # positive shifts where i <= n-1-s -> s in 1 .. n-1-i
    if i < n - 1:
        s_pos = np.arange(1, n - i)
        delta[mid + s_pos] += 1 - h[i + s_pos]
    # positive shifts where i >= s -> s in 1 .. i
    if i > 0:
        s_sub = np.arange(1, i + 1)
        delta[mid + s_sub] -= h[i - s_sub]
    # negative shifts where i >= t -> t in 1 .. i
    if i > 0:
        t_pos = np.arange(1, i + 1)
        delta[mid - t_pos] += 1 - h[i - t_pos]
    # negative shifts where i + t <= n-1 -> t in 1 .. n-1-i
    if i < n - 1:
        t_sub = np.arange(1, n - i)
        delta[mid - t_sub] -= h[i + t_sub]
    return delta

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]  # subtract h_{i+k}
        grad[k:] -= h[: n - k]  # subtract h_{i-k}
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

                h[i] += add
                diff = target - h.sum()
    return h


# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    best_h = None
    best_c = np.inf
    best_n = None

    # ------------------------------------------------------------------
    # if a parent solution is supplied, evaluate it
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        # enforce exact sum
        target0 = h0.size / 2.0
        diff0 = target0 - h0.sum()
        if abs(diff0) > 1e-12:
            for _ in range(2000):
                i = rng.integers(0, h0.size)
                add = min(diff0, 1.0 - h0[i]) if diff0 > 0 else max(diff0, -h0[i])
                h0[i] += add
                diff0 = target0 - h0.sum()
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0.copy(), c0, h0.size

    # ------------------------------------------------------------------
    # discretisation sizes to try (larger sizes give finer resolution)
    # ------------------------------------------------------------------
    candidate_ns = [256, 224, 192]
    if best_n is not None and best_n not in candidate_ns:
        candidate_ns = [best_n] + candidate_ns

    for n in candidate_ns:
        if time.time() - start > 0.94 * budget_s:
            break

        # ------------------------------------------------------------------
        # initialise h for this n
        # ------------------------------------------------------------------
        if best_n == n and best_h is not None:
            h = best_h.copy()
        elif "initial_h_values" in globals() and n != len(initial_h_values):
            # linear interpolation of the supplied parent
            x_old = np.linspace(0, 2, len(initial_h_values), endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h = np.interp(x_new, x_old, initial_h_values)
            h = np.clip(h, 0.0, 1.0)
            # enforce exact sum
            target = n / 2.0
            diff = target - h.sum()
            if abs(diff) > 1e-12:
                for _ in range(2000):
                    i = rng.integers(0, n)
                    add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                    h[i] += add
                    diff = target - h.sum()
        else:
            # start from a greedy binary construction (often a good basin)
            h, _ = greedy_construct(n)

        # ------------------------------------------------------------------
        # continuous local search – guided hill‑climb
        # ------------------------------------------------------------------
        h, c = guided_hill_climb(h, rng, max_iters=2000, step_scale=0.6)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

        # ------------------------------------------------------------------
        # SLSQP refinement with an increasing β‑schedule
        # ------------------------------------------------------------------
        for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0, 10000.0):
            if time.time() - start > 0.96 * budget_s:
                break
            maxit = 600 if beta >= 200 else 400
            h = slsqp_optimize(h, beta, rng, max_iter=maxit)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

        # ------------------------------------------------------------------
        # fine‑grained gradient refinement
        # ------------------------------------------------------------------
        if time.time() - start < 0.97 * budget_s:
            h = gradient_refine(h, rng, step=0.02, max_iter=8)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

        # ------------------------------------------------------------------
        # final short guided hill‑climb
        # ------------------------------------------------------------------
        if time.time() - start < 0.98 * budget_s:
            h, c = guided_hill_climb(h, rng, max_iters=1500, step_scale=0.3)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

        # ------------------------------------------------------------------
        # optional up‑sampling + refinement (if time permits)
        # ------------------------------------------------------------------
        if (
            best_h is not None
            and time.time() - start < 0.99 * budget_s
            and best_n * 2 <= 256
        ):
            h_up = np.repeat(best_h, 2)
            n_up = h_up.size
            h_up, _ = guided_hill_climb(h_up, rng, max_iters=2000, step_scale=0.3)
            for beta in (5.0, 20.0, 80.0, 200.0):
                if time.time() - start > budget_s:
                    break
                h_up = slsqp_optimize(h_up, beta, rng, max_iter=300)
                c_up = compute_c5(h_up)
                if c_up < best_c:
                    best_h, best_c, best_n = h_up.copy(), c_up, n_up

    # ------------------------------------------------------------------
    # final random tiny perturbations until time runs out
    # ------------------------------------------------------------------
    while time.time() - start < budget_s * 0.999 and best_h is not None:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # ------------------------------------------------------------------
    # final feasibility enforcement and return
    # ------------------------------------------------------------------
    if best_h is None:
        best_n = 256
        best_h = np.full(best_n, 0.5)

    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, best_n)
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
