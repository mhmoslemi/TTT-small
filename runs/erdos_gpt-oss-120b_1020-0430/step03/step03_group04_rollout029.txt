        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        return new_h

    def hill_climb(h, max_iters=2000, start_step=0.5):
        n = h.shape[0]
        best_h = h.copy()
        best_c = compute_c5(best_h)
        step = start_step
        decay_interval = max(1, max_iters // 10)
        for it in range(max_iters):

            c = compute_c5(cand)
            if c < best_c:
                best_c = c
                best_h = cand.copy()
                h = cand
            if (it+1) % decay_interval == 0:
                step = max(step*0.9, 1e-5)
        return best_h, best_c

    # define function to adjust sum to target (repair)
    def repair_sum(h, target):
        diff = target - h.sum()
        n = len(h)
        if abs(diff) < 1e-12:
            return h
        # use greedy process
        for _ in range(2000):
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
        return h

    candidate_ns = [72, 96, 120]
    for n in candidate_ns:
        if time.time() - start_time > budget_s * 0.95:
            break
        target = n/2
        # try to solve MILP binary for this n
        try:
            # Build indices
            i_list = []

            y_idx_per_k = [[] for _ in range(n)]
            for k in range(1, n):
                for i in range(n - k):

                    i_list.append(i)
                    j_list.append(i + k)
                    k_list.append(k)
                    y_idx_per_k[k].append(idx)
            num_pairs = len(i_list)
            # Define variables
            x = cp.Variable(n, boolean=True)
            y = cp.Variable(num_pairs, boolean=True)
            z = cp.Variable()
            constraints = []
            constraints.append(cp.sum(x) == target)
            # product constraints
            for p in range(num_pairs):
                i = i_list[p]; j = j_list[p]
                constraints.append(y[p] <= x[i])
                constraints.append(y[p] <= x[j])
                constraints.append(y[p] >= x[i] + x[j] - 1)
            # shift constraints
            for k in range(1, n):
                S = cp.sum(x[:n - k])
                if y_idx_per_k[k]:
                    sum_y = cp.sum(y[y_idx_per_k[k]])
                else:
                    sum_y = 0
                L = S - sum_y
                constraints.append(L <= z)
            constraints.append(z >= 0)
            prob = cp.Problem(cp.Minimize(z), constraints)
            # Solve with GLPK_MI
            prob.solve(solver=cp.GLPK_MI, glpk={'msg_lev': 'GLP_MSG_OFF'})
        except Exception as e:
            # if MILP fails, skip
            continue
        if prob.status not in ["optimal", "optimal_inaccurate"]:
            # didn't find feasible, skip
            continue
        x_sol = np.round(x.value).astype(float)  # binary
        # correct any rounding errors
        x_sol = repair_sum(x_sol, target)
        c_val = compute_c5(x_sol)
        if c_val < best_c:
            best_h, best_c, best_n = x_sol.copy(), c_val, n
        # local refinement
        if time.time() - start_time < budget_s * 0.9:
            # optionally hill climb
            h_ref, c_ref = hill_climb(best_h, max_iters=3000, start_step=0.5)
            if c_ref < best_c:
                best_h, best_c, best_n = h_ref.copy(), c_ref, n

    # if best not found, fallback to random hill climbing
    if best_h is None:
        n = 180
        target = n/2
        h = np.full(n, 0.5)
        h = repair_sum(h, target)
        best_h, best_c, best_n = hill_climb(h, max_iters=5000, start_step=0.6)

    # final repair and compute final c
    best_h = np.clip(best_h, 0., 1.)
    best_h = repair_sum(best_h, best_n/2)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)

import time
import numpy as np

# try to import cvxpy; if unavailable we will fall back to pure heuristics
try:
    import cvxpy as cp
    CVX_AVAILABLE = True
except Exception:  # pragma: no cover
    CVX_AVAILABLE = False


def compute_c5(h):
    """Discrete C5 for step function h on [0,2] with uniform spacing."""
    n = h.size
    return float(np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / n))


def repair_sum(h, target):
    """Adjust h in-place (using small moves) so that sum(h) == target."""
    diff = target - h.sum()
    n = h.size
    if abs(diff) < 1e-12:
        return h
    rng = np.random.default_rng()
    for _ in range(5000):
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small ε from one coordinate to another, preserving the sum."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Simple deterministic hill‑climber (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        cand = perturb_h(h, rng, scale=step)
        c = compute_c5(cand)
        if c < best_c:
            best_c = c
            best_h = cand.copy()
            h = cand
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-5)
    return best_h, best_c


def solve_binary_milp(n, rng):
    """Return a binary vector of length n (sum=n/2) minimizing max overlap."""
    target = n // 2
    # build pair lists
    i_list, j_list, k_list = [], [], []
    y_idx_per_k = [[] for _ in range(n)]
    for k in range(1, n):
        for i in range(n - k):
            idx = len(i_list)
            i_list.append(i)
            j_list.append(i + k)
            k_list.append(k)
            y_idx_per_k[k].append(idx)
    num_pairs = len(i_list)

    # cvxpy model
    x = cp.Variable(n, boolean=True)
    y = cp.Variable(num_pairs, boolean=True)
    z = cp.Variable()
    cons = [cp.sum(x) == target, z >= 0]

    # product constraints
    for p in range(num_pairs):
        i, j = i_list[p], j_list[p]
        cons.append(y[p] <= x[i])
        cons.append(y[p] <= x[j])
        cons.append(y[p] >= x[i] + x[j] - 1)

    # shift constraints
    for k in range(1, n):
        S = cp.sum(x[: n - k])
        if y_idx_per_k[k]:
            sum_y = cp.sum(y[np.array(y_idx_per_k[k], dtype=int)])
        else:
            sum_y = 0
        cons.append(S - sum_y <= z)

    prob = cp.Problem(cp.Minimize(z), cons)
    # GLPK_MI is widely available; fall back to CBC if needed
    solvers = []
    if CVX_AVAILABLE:
        try:
            prob.solve(solver=cp.GLPK_MI, glpk={"msg_lev": "GLP_MSG_OFF"})
            solvers.append("GLPK_MI")
        except Exception:
            pass
        if prob.status not in ["optimal", "optimal_inaccurate"]:

                prob.solve(solver=cp.CBC, cbc={"msg_lev": "GLP_MSG_OFF"})
                solvers.append("CBC")
            except Exception:
                pass
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        return None, None
    x_val = np.rint(x.value).astype(float)  # ensure binary
    x_val = repair_sum(x_val, target)
    c_val = compute_c5(x_val)
    return x_val, c_val
