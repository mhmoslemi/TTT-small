def binary_hill_climb(h_bin, rng, max_iters, patience):
    best_h = h_bin.copy()
    best_c = compute_c5(best_h.astype(float))
    n = len(h_bin)
    ones = np.flatnonzero(best_h)
    zeros = np.flatnonzero(1 - best_h)
    it_without_improve = 0
    for it in range(max_iters):
        # random swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # perform swap
        cand = best_h.copy()
        cand[i] = 0
        cand[j] = 1
        c = compute_c5(cand.astype(float))
        if c < best_c - 1e-12:
            best_h = cand
            best_c = c
            # update indices arrays
            ones[ones == i] = j  # actually i removed, j added. 
            zeros[zeros == j] = i
            it_without_improve = 0
        else:
            it_without_improve += 1
        if it_without_improve >= patience:
            # random restart or perturb
            break
    return best_h, best_c, it_without_improve

import time
import numpy as np
from scipy.optimize import minimize

# globals for the SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function h (dx = 2/n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator, max_iter: int = 5000) -> np.ndarray:
    """Adjust h to satisfy Σh = target while staying in [0,1]."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0.0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # final uniform correction if necessary
    if abs(diff) > 1e-12:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                if diff > 0.0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def sum_eq_constraint(x: np.ndarray) -> float:
    """Equality constraint Σx_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x: np.ndarray) -> float:
    """Smooth approximation of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp stable
    val = (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
    return val


def slsqp_optimize(h0: np.ndarray, beta: float, max_iter: int = 400) -> np.ndarray:
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum (tiny drift)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        n = x.size
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def random_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=int)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1
    return h


def target_swap(h_bin: np.ndarray, rng: np.random.Generator) -> np.ndarray | None:
    """Swap a 1/0 pair that directly reduces the current worst shift."""
    n = h_bin.size
    corr = np.correlate(h_bin, 1 - h_bin, mode="full")
    k = int(np.argmax(corr))
    shift = k - (n - 1)          # signed shift
    if shift == 0:
        return None
    idx = np.arange(n)
    if shift > 0:
        ok = idx + shift < n
        A = idx[(h_bin == 1) & ok & (h_bin[idx + shift] == 0)]
        B = idx[(h_bin == 0) & ok & (h_bin[idx + shift] == 1)]
    else:
        s = -shift
        ok = idx >= s
        A = idx[(h_bin == 1) & ok & (h_bin[idx - s] == 0)]
        B = idx[(h_bin == 0) & ok & (h_bin[idx - s] == 1)]
    if A.size == 0 or B.size == 0:
        return None
    i = rng.choice(A)
    j = rng.choice(B)
    new_h = h_bin.copy()
    new_h[i] = 0
    new_h[j] = 1
    return new_h


def binary_hill_climb(h_start: np.ndarray,
                      rng: np.random.Generator,
                      max_iters: int = 4000,
                      patience: int = 1500) -> tuple[np.ndarray, float, int]:
    """Hill‑climb on binary vectors using targeted swaps."""
    best_h = h_start.copy()
    best_c = compute_c5(best_h.astype(float))
    it_no_improve = 0
    n = best_h.size
    for it in range(max_iters):
        cand = target_swap(best_h, rng)
        if cand is None:
            # fallback: random 1‑0 swap
            ones = np.flatnonzero(best_h)
            zeros = np.flatnonzero(1 - best_h)
            i = rng.choice(ones)
            j = rng.choice(zeros)
            cand = best_h.copy()
            cand[i] = 0
            cand[j] = 1
        c = compute_c5(cand.astype(float))
        if c < best_c - 1e-12:
            best_h, best_c = cand, c
            it_no_improve = 0
        else:
            it_no_improve += 1
        if it_no_improve >= patience:
            break
    return best_h, best_c, it_no_improve


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # 1. Determine grid size and initial continuous vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        base_h = np.array(initial_h_values, dtype=float)
        base_n = base_h.size
    else:
        base_h = None
        base_n = 360

    target_n = max(720, base_n * 2)          # finer grid for better exploration
    if base_h is not None and base_n != target_n:
        old_x = (np.arange(base_n) + 0.5) * (2.0 / base_n)
        new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
        h_cont = np.interp(new_x, old_x, base_h)
    else:
        # generic start – flat 0.5 (will be projected later)
        h_cont = np.full(target_n, 0.5, dtype=float)

    target_sum = target_n / 2.0
    h_cont = enforce_sum(h_cont, target_sum, rng)

    # ------------------------------------------------------------------
    # 2. Initialise binary search
    # ------------------------------------------------------------------
    best_bin = random_binary(target_n, rng)
    best_c = compute_c5(best_bin.astype(float))
    best_cont = best_bin.astype(float).copy()
    # also keep the interpolated continuous start as a contender
    c0 = compute_c5(h_cont)
    if c0 < best_c:
        best_cont = h_cont.copy()
        best_c = c0
        best_bin = (h_cont > 0.5).astype(int)          # a quick binarisation for future swaps

    # ------------------------------------------------------------------
    # 3. Binary hill‑climbing with restarts (major part of budget)
    # ------------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.60:
        # choose start point
        if rng.random() < 0.7:
            start_bin = best_bin.copy()
        else:
            start_bin = random_binary(target_n, rng)

        bin_h, bin_c, _ = binary_hill_climb(start_bin, rng,
                                            max_iters=5000,
                                            patience=2000)
        if bin_c < best_c:
            best_bin = bin_h
            best_c = bin_c
            best_cont = bin_h.astype(float)

    # ------------------------------------------------------------------
    # 4. Continuous refinement with SLSQP ladder (remaining budget)
    # ------------------------------------------------------------------
    beta_list = [10, 30, 100, 300, 800, 1500, 3000]
    h_cur = best_cont.copy()
    for beta in beta_list:
        if time.time() - start_time > budget_s * 0.95:
            break
        try:
            h_cur = slsqp_optimize(h_cur, beta, max_iter=800)
        except Exception:
            continue
        c_cur = compute_c5(h_cur)
        if c_cur < best_c:
            best_c = c_cur
            best_cont = h_cur.copy()

    # ------------------------------------------------------------------
    # 5. Final clean‑up
    # ------------------------------------------------------------------
    best_cont = np.clip(best_cont, 0.0, 1.0)
    best_cont = enforce_sum(best_cont, target_sum, rng)
    final_c = compute_c5(best_cont)

    # return as Python list for the harness
    return best_cont.tolist(), float(final_c), int(target_n)
