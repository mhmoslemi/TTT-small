def perturb_random(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    # compute max epsilon to keep within [0,1]
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

import time
import numpy as np

# ---------- helpers ----------
def project_sum_bounds(x, target, rng=None):
    """Project x onto the box [0,1] while enforcing sum == target."""
    low, high = -1.0, 1.0
    for _ in range(50):
        mid = (low + high) * 0.5
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + low, 0.0, 1.0)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        free = (y > 1e-12) & (y < 1.0 - 1e-12)
        if free.any():
            y[free] += diff / free.sum()
        elif rng is not None:
            i = rng.integers(0, y.size)
            y[i] = np.clip(y[i] + diff, 0.0, 1.0)
    return np.clip(y, 0.0, 1.0)


def random_feasible_vector(n, rng):
    v = rng.random(n)
    return project_sum_bounds(v, n / 2.0, rng)


def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def perturb_random(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def hill_climb_targeted(h_start, rng, max_iters, start_step, target_sum):
    h = h_start.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay = max(1, max_iters // 10)

    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr)
        k = idx - (n - 1)

        grad = np.ones(n)
        if k >= 0:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
        else:
            kk = -k
            grad[: n - kk] -= h[kk:]
            grad[kk:] -= h[: n - kk]

        pos = np.where((grad > 1e-9) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-9) & (h < 1.0 - 1e-12))[0]

        if pos.size and neg.size:
            i = rng.choice(pos)
            j = rng.choice(neg)
            max_eps = min(h[i], 1.0 - h[j])
            eps = max_eps * step * rng.random()
            new_h = h.copy()
            new_h[i] -= eps
            new_h[j] += eps
        else:
            new_h = perturb_random(h, rng, scale=step)

        new_h = project_sum_bounds(new_h, target_sum, rng)
        c_new = compute_c5(new_h)

        if c_new < best_c:
            best_c, best_h = c_new, new_h.copy()
        if c_new < compute_c5(h):
            h = new_h

        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)

    return best_h, best_c


def smooth_step(h, beta, alpha, rng):
    """One projected gradient step on the log‑sum‑exp (soft‑max) surrogate."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    m = corr.max()
    w = np.exp(beta * (corr - m))
    w /= w.sum()

    # weighted neighbor sum S_i = Σ_k w_k (h_{i+k}+h_{i-k})
    S = np.zeros(n)
    for shift in range(-(n - 1), n):
        wk = w[shift + n - 1]
        if wk == 0.0:
            continue
        if shift >= 0:
            end = n - shift
            S[:end] += wk * h[shift:]
            S[shift:] += wk * h[:end]
        else:
            kk = -shift
            end = n - kk
            S[:end] += wk * h[kk:]
            S[kk:] += wk * h[:end]

    grad = dx * (1.0 - S)          # gradient of the surrogate
    nh = h - alpha * grad
    nh = np.clip(nh, 0.0, 1.0)
    return project_sum_bounds(nh, n / 2.0, rng)


# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # ---- initialise h ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = random_feasible_vector(360, rng)
    n0 = h0.size
    up_factor = 2                         # double resolution
    n = n0 * up_factor
    xs0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.interp(xs, xs0, h0)
    h = project_sum_bounds(h, n / 2.0, rng)

    best_h = h.copy()
    best_c = compute_c5(best_h)
    cur_h = h.copy()
    cur_c = best_c

    # ---- parameters ----
    beta_sched = [5, 15, 50, 150, 500, 1500, 4000]
    max_beta = beta_sched[-1]

    # main stochastic search
    while time.time() - start < budget_s * 0.78:
        # pick a start point
        if rng.random() < 0.7:
            cur_h = perturb_random(best_h, rng, scale=0.3)
        else:
            cur_h = random_feasible_vector(n, rng)
        cur_h = project_sum_bounds(cur_h, n / 2.0, rng)
        cur_c = compute_c5(cur_h)

        # gradient descent on the smooth surrogate
        for beta in beta_sched:
            base_alpha = 200.0 * beta / max_beta   # larger steps for larger beta
            alpha = base_alpha
            for _ in range(500):
                if time.time() - start > budget_s * 0.78:
                    break
                nh = smooth_step(cur_h, beta, alpha, rng)
                nc = compute_c5(nh)

                if nc < best_c:
                    best_c, best_h = nc, nh.copy()
                if nc < cur_c:
                    cur_h, cur_c = nh, nc
                    alpha = min(alpha * 1.05, base_alpha * 1.5)
                else:
                    alpha = max(alpha * 0.7, base_alpha * 0.1)

        # local refinement using targeted hill‑climb
        h_loc, c_loc = hill_climb_targeted(cur_h, rng, max_iters=800,
                                           start_step=0.4, target_sum=n / 2.0)
        if c_loc < best_c:
            best_c, best_h = c_loc, h_loc.copy()

    # final tiny random tweaks
    while time.time() - start < budget_s * 0.99:
        nh = perturb_random(best_h, rng, scale=0.01)
        nc = compute_c5(nh)
        if nc < best_c:
            best_c, best_h = nc, nh.copy()

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_bounds(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
