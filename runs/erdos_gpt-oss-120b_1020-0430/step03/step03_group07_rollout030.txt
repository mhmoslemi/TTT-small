def binary_to_target(b, target, rng):
    current = int(b.sum())
    diff = int(target - current)
    if diff > 0:
        zero_indices = np.where(b == 0)[0]
        select = rng.choice(zero_indices, diff, replace=False)
        b[select] = 1.0
    elif diff < 0:
        one_indices = np.where(b == 1)[0]
        select = rng.choice(one_indices, -diff, replace=False)
        b[select] = 0.0
    return b

        if eps <= 0:
            break
        h[i] -= eps
        h[j] += eps

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i h_i(1-h_{i+k})."""
    conv = correlate_full(x)
    bc = CURRENT_BETA * conv
    m = bc.max()
    return (m + np.log(np.exp(bc - m).sum())) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """Refine h0 by minimizing the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA

def enforce_sum(h, target, rng):
    """Adjust h so that Σh == target while preserving 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Random vector in [0,1]^n with Σ = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def correlate_full(h):
    """Linear correlation of h with (1‑h) using FFT."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Discrete C₅ for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)


def candidate_block_pattern(n, L, rng):
    """Repeating blocks of L ones then L zeros, corrected to exact sum."""
    if L <= 0:
        return np.full(n, 0.5)
    block = np.concatenate([np.ones(L), np.zeros(L)])
    reps = int(np.ceil(n / (2 * L)))
    pat = np.tile(block, reps)[:n].astype(float)
    target = n // 2
    cur = int(pat.sum())
    diff = target - cur
    if diff > 0:
        zeros = np.where(pat == 0)[0]
        if zeros.size >= diff:
            sel = rng.choice(zeros, diff, replace=False)
            pat[sel] = 1.0
    elif diff < 0:
        ones = np.where(pat == 1)[0]
        if ones.size >= -diff:
            sel = rng.choice(ones, -diff, replace=False)
            pat[sel] = 0.0
    return pat


def irrational_binary(n, alpha, rng):
    """Binary sequence based on irrational rotation."""
    t = (np.arange(n) * alpha) % 1.0
    b = (t < 0.5).astype(float)
    target = n // 2
    cur = int(b.sum())
    diff = target - cur
    if diff > 0:
        zeros = np.where(b == 0)[0]
        if zeros.size >= diff:
            b[rng.choice(zeros, diff, replace=False)] = 1.0
    elif diff < 0:
        ones = np.where(b == 1)[0]
        if ones.size >= -diff:
            b[rng.choice(ones, -diff, replace=False)] = 0.0
    return b


def gradient_descent_local(h, rng, steps=1500, init_step=0.04):
    """Greedy reduction of the worst overlap using exact gradient."""
    n = h.size
    step = init_step
    best_c = compute_c5(h)
    for _ in range(steps):
        conv = correlate_full(h)
        idx_max = int(conv.argmax())
        d = (n - 1) - idx_max  # shift (positive = right)
        grad = np.zeros(n)
        if d >= 0:
            i1 = np.arange(0, n - d)
            grad[i1] = 1.0 - h[i1 + d]
            i2 = np.arange(d, n)
            grad[i2] -= h[i2 - d]
        else:
            dp = -d
            i1 = np.arange(dp, n)
            grad[i1] = 1.0 - h[i1 - dp]
            i2 = np.arange(0, n - dp)
            grad[i2] -= h[i2 + dp]
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[grad[pos].argmax()]
        j = neg[grad[neg].argmin()]
        max_transfer = min(h[i], 1.0 - h[j])
        eps = min(step, max_transfer)
        if eps <= 0:
            break
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        c_new = compute_c5(h_new)
        if c_new < best_c - 1e-12:
            h = h_new
            best_c = c_new
            # keep step size
        else:
            step *= 0.7
            if step < 1e-5:
                break
    return h


def generate_candidates(n, rng):
    """Construct a diverse pool of feasible step functions."""
    cand = []
    # initial parent if available
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
        cand.append(h0)
    # random fractional vectors
    for _ in range(5):
        cand.append(random_feasible_vector(n, rng))
    # random binary vectors
    for _ in range(5):
        b = np.zeros(n)
        b[rng.choice(n, n // 2, replace=False)] = 1.0
        cand.append(b)
    # block patterns
    for L in range(1, min(21, n // 2 + 1)):
        cand.append(candidate_block_pattern(n, L, rng))
    # irrational binary patterns
    alphas = [np.sqrt(2), (np.sqrt(5) - 1) / 2, np.e, np.pi]
    for a in alphas:
        cand.append(irrational_binary(n, a, rng))
    # sinusoidal families
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    for f in range(1, 4):
        h = 0.5 + 0.5 * np.sin(np.pi * f * xs)
        cand.append(enforce_sum(np.clip(h, 0, 1), n / 2.0, rng))
        h = 0.5 + 0.5 * np.cos(np.pi * f * xs)
        cand.append(enforce_sum(np.clip(h, 0, 1), n / 2.0, rng))
    # linear ramps
    h = xs / 2.0
    cand.append(enforce_sum(np.clip(h, 0, 1), n / 2.0, rng))
    h = 1.0 - xs / 2.0
    cand.append(enforce_sum(np.clip(h, 0, 1), n / 2.0, rng))
    return cand


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline_main = start_time + budget_s * 0.9  # reserve time for final steps
    # ----------------------------------------------------------
    # Base dimension
    # ----------------------------------------------------------
    if "initial_h_values" in globals():
        n = len(initial_h_values)
    else:
        n = 360
    # ----------------------------------------------------------
    # Initial candidate pool
    # ----------------------------------------------------------
    candidates = generate_candidates(n, rng)
    best_h = None
    best_c = float("inf")
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    # ----------------------------------------------------------
    # Main optimisation loop
    # ----------------------------------------------------------
    beta_list = [5.0, 20.0, 80.0, 200.0, 800.0, 2000.0]
    while time.time() < deadline_main:
        # new start point
        if rng.random() < 0.7:
            start = perturb_h(best_h, rng, scale=0.3)
        else:
            start = random_feasible_vector(n, rng)
        # gradient descent refinement
        start = gradient_descent_local(start, rng, steps=1500, init_step=0.04)
        c_start = compute_c5(start)
        if c_start < best_c:
            best_c, best_h = c_start, start.copy()
        # SLSQP continuation over β
        for beta in beta_list:
            if time.time() >= deadline_main:
                break
            try:
                opt_h = slsqp_optimize(best_h, beta, max_iter=300)
            except Exception:
                continue
            c_opt = compute_c5(opt_h)
            if c_opt < best_c:
                best_c, best_h = c_opt, opt_h.copy()
        # final local cleanup
        best_h = gradient_descent_local(best_h, rng, steps=800, init_step=0.02)
        best_c = compute_c5(best_h)
        # early stop if already very good
        if best_c <= 0.38075:
            break
    # ----------------------------------------------------------
    # Optional up‑sampling to a finer grid
    # ----------------------------------------------------------
    if time.time() < start_time + budget_s * 0.99 and best_h.size * 2 <= 800:
        up_h = np.repeat(best_h, 2)
        up_h = enforce_sum(up_h, up_h.size / 2.0, rng)
        # refine on finer grid
        up_h = gradient_descent_local(up_h, rng, steps=2500, init_step=0.02)
        for beta in beta_list:
            if time.time() >= start_time + budget_s:
                break
            try:
                up_h = slsqp_optimize(up_h, beta, max_iter=300)
            except Exception:
                pass
        up_h = gradient_descent_local(up_h, rng, steps=1000, init_step=0.01)
        up_c = compute_c5(up_h)
        if up_c < best_c:
            best_c, best_h = up_c, up_h.copy()
    # ----------------------------------------------------------
    # Final clean‑up
    # ----------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
