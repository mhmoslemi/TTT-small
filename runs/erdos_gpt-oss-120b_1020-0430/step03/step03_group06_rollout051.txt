def perturb_move(h, rng, step):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n-1)
    if j >= i: j += 1  # ensure i != j

    if max_pos + max_neg == 0:

    # ensure eps within limits (should be)
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps

def compute_c5_fft(h):
    n = h.size
    dx = 2.0 / n
    a = h
    b = 1.0 - h
    L = 1 << ((2*n - 1).bit_length())
    FA = np.fft.rfft(a, L)
    FB = np.fft.rfft(b, L)
    conv = np.fft.irfft(FA * FB, L)[:2*n-1]
    return np.max(conv) * dx
```

def simulated_annealing(h_start, rng, time_limit):
    start = time.time()
    h_cur = h_start.copy()
    cur_c = compute_c5_fft(h_cur)
    best_h = h_cur.copy()
    best_c = cur_c
    n = h_cur.size
    max_step = 0.5
    T0 = 0.001
    T_end = 1e-7
    # we can compute a schedule based on elapsed fraction
    while True:
        elapsed = time.time() - start
        if elapsed > time_limit:
            break
        t = elapsed / time_limit
        # exponential cooling: T = T0 * (T_end/T0) ** t
        T = T0 * ((T_end / T0) ** t)
        step = max_step * (T / T0) ** 0.5
        # propose move
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        max_pos = min(h_cur[i], 1.0 - h_cur[j])
        max_neg = min(h_cur[j], 1.0 - h_cur[i])
        if max_pos + max_neg == 0:
            continue
        eps = (rng.random() * (max_pos + max_neg) - max_neg) * step
        if eps == 0.0:
            continue
        h_new = h_cur.copy()
        h_new[i] -= eps
        h_new[j] += eps
        new_c = compute_c5_fft(h_new)
        # acceptance
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-12)):
            h_cur = h_new
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h_new.copy()
    return best_h, best_c

def compute_c5_fft(h):
    n = h.size
    dx = 2.0 / n
    a = h.astype(np.float64)
    b = 1.0 - a
    L = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    conv = np.fft.irfft(fa * fb, L)[:2*n-1]
    return np.max(conv) * dx
```

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def compute_c5(h):
    """Exact C5 using linear correlation (as in the evaluator)."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)


def compute_c5_fft(h):
    """Fast C5 using FFT (for internal optimisation)."""
    n = h.size
    dx = 2.0 / n
    a = h.astype(np.float64)
    b = 1.0 - a
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    conv = np.fft.irfft(fa * fb, L)[: 2 * n - 1]
    return np.max(conv) * dx


def enforce_sum(h, target, rng):
    """Adjust h so that Σh = target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-12:
            continue
        h[i] += add
        diff = target - h.sum()
    # tiny final sweep
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(1000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            if abs(add) < 1e-12:
                continue
            h[i] += add
            diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h≤1, Σh=n/2)."""
    v = rng.exponential(1.0, size=n)
    v = v / v.sum() * (n / 2.0)
    h = np.clip(v, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def perturb_h(h, rng, scale=0.3):
    """Transfer a small amount between two coordinates (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if j >= i:
        j += 1
    max_pos = min(h[i], 1.0 - h[j])
    max_neg = min(h[j], 1.0 - h[i])
    if max_pos + max_neg == 0:
        return h.copy()
    eps = (rng.random() * (max_pos + max_neg) - max_neg) * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.2):
    """Simple greedy hill‑climb using the exact C5."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        max_pos = min(h[i], 1.0 - h[j])
        max_neg = min(h[j], 1.0 - h[i])
        if max_pos + max_neg == 0:
            continue
        eps = (rng.random() * (max_pos + max_neg) - max_neg) * step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def simulated_annealing(h_start, rng, time_limit):
    """Simulated annealing using the fast FFT C5."""
    start = time.time()
    h_cur = h_start.copy()
    cur_c = compute_c5_fft(h_cur)
    best_h = h_cur.copy()
    best_c = cur_c
    n = h_cur.size
    max_step = 0.5
    T0 = 0.001
    T_end = 1e-7
    while True:
        elapsed = time.time() - start
        if elapsed > time_limit:
            break
        t = elapsed / time_limit
        T = T0 * ((T_end / T0) ** t)  # exponential cooling
        step = max_step * (T / T0) ** 0.5
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        max_pos = min(h_cur[i], 1.0 - h_cur[j])
        max_neg = min(h_cur[j], 1.0 - h_cur[i])
        if max_pos + max_neg == 0:
            continue
        eps = (rng.random() * (max_pos + max_neg) - max_neg) * step
        if eps == 0.0:
            continue
        h_new = h_cur.copy()
        h_new[i] -= eps
        h_new[j] += eps
        new_c = compute_c5_fft(h_new)
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-12)):
            h_cur = h_new
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h_new.copy()
    return best_h, best_c
