def anneal_optimize(h0, rng, init_temp, final_temp, max_iters):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    temp = init_temp
    decay = (final_temp / init_temp) ** (1.0 / max_iters)
    for i in range(max_iters):
        if temp < final_temp:
            break
        # propose move
        i1 = rng.integers(0, n)
        i2 = rng.integers(0, n)
        if i2 == i1:
            continue
        max_pos = min(h[i1], 1 - h[i2])
        max_neg = min(h[i2], 1 - h[i1])
        if max_pos + max_neg == 0:
            continue
        eps = (rng.random() * (max_pos + max_neg)) - max_neg
        # maybe scale by step size similarly
        eps *= 0.5 # moderate
        new_h = h.copy()
        new_h[i1] -= eps
        new_h[i2] += eps
        new_h = np.clip(new_h, 0.0, 1.0)
        c = compute_c5(new_h)
        delta = c - best_c
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            h = new_h
            if c < best_c:
                best_c = c
                best_h = new_h.copy()
        temp *= decay
    return best_h, best_c
```

```
def compute_c5_fft(h):
    n = h.size
    dx = 2.0 / n
    m = 2*n - 1
    size = 1 << (m-1).bit_length()
    H = np.fft.rfft(h, n=size)
    G = np.fft.rfft(1 - h, n=size)
    # compute linear correlation via zero padding? We need correlation (non-circular) with zero padding.
    # The linear correlation can be obtained as irfft(H * np.conj(G), n=size). But that gives circular correlation mixing wrap-around.
    # To avoid wrap-around, we can pad h and 1-h with zeros to length size, but we need the reverse of one of them.
    # correlation of a and b (non-circular) is convolution of a and reverse(b).
    # So define b_rev = (1 - h)[::-1]
    b_rev = (1 - h)[::-1]
    B = np.fft.rfft(b_rev, n=size)
    conv = np.fft.irfft(H * B, n=size)[:m]
    # conv index shift: conv[k] corresponds to sum_i h[i] * (1-h)[i + (k - (n-1))]
    # Because convolution index zero corresponds to i=0 with b_rev index 0 -> (1-h)[n-1]
    # Actually we need max of correlation for k from -(n-1) to (n-1) (excluding zero?). np.correlate returns length 2n-1 where index offset map: lag = i - (n-1)
    # We'll just compute max of conv because max over all k same as original.
    return np.max(conv) * dx
```

import time
import numpy as np
from scipy.optimize import minimize

# globals for SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None

def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def _softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def compute_c5_fft(h):
    n = h.size
    dx = 2.0 / n
    m = 2 * n - 1
    size = 1 << (m - 1).bit_length()
    H = np.fft.rfft(h, n=size)
    B = np.fft.rfft((1 - h)[::-1], n=size)
    conv = np.fft.irfft(H * B, n=size)[:m]
    return np.max(conv) * dx

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        if abs(add) < 1e-12:
            continue
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(1000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            if abs(add) < 1e-12:
                continue
            h[i] += add
            diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)

def random_feasible_vector(n, rng):
    v = rng.exponential(1.0, size=n)
    v = v / v.sum() * (n / 2.0)
    h = np.clip(v, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)

def binary_random_vector(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return enforce_sum(h, n / 2.0, rng)

def smooth_vector(v):
    kern = np.array([0.25, 0.5, 0.25])
    s = np.convolve(v, kern, mode="same")
    return s

def hill_climb(h_start, rng, max_iters=4000, step=0.5, time_limit=None, start_time=None, budget=None):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5_fft(h)
    n = h.size
    decay_int = max(1, max_iters // 10)
    for it in range(max_iters):
        if time_limit is not None and time.time() - start_time > time_limit * budget:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)
        max_pos = min(h[i], 1.0 - h[j])
        max_neg = min(h[j], 1.0 - h[i])
        if max_pos + max_neg == 0:
            continue
        eps = (rng.random() * (max_pos + max_neg)) - max_neg
        eps *= step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_h = np.clip(new_h, 0.0, 1.0)
        c = compute_c5_fft(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_int == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(_softmax_obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons, options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        n = x.size
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            if abs(add) < 1e-12:
                continue
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def anneal_optimize(h0, rng, init_temp=0.1, final_temp=1e-5, max_iters=20000, time_limit=None, start_time=None, budget=None):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5_fft(h)
    n = h.size
    temp = init_temp
    decay = (final_temp / init_temp) ** (1.0 / max_iters)
    for i in range(max_iters):
        if time_limit is not None and time.time() - start_time > time_limit * budget:
            break
        i1 = rng.integers(0, n)
        i2 = rng.integers(0, n)
        if i2 == i1:
            continue
        max_pos = min(h[i1], 1.0 - h[i2])
        max_neg = min(h[i2], 1.0 - h[i1])
        if max_pos + max_neg == 0:
            continue
        eps = (rng.random() * (max_pos + max_neg)) - max_neg
        eps *= 0.5
        new_h = h.copy()
        new_h[i1] -= eps
        new_h[i2] += eps
        new_h = np.clip(new_h, 0.0, 1.0)
        c = compute_c5_fft(new_h)
        delta = c - best_c
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            h = new_h
            if c < best_c:
                best_c = c
                best_h = new_h.copy()
        temp *= decay
        if temp < final_temp:
            break
    return best_h, best_c

def thue_morse(N):
    seq = np.zeros(N, dtype=float)
    for i in range(N):
        seq[i] = bin(i).count('1') % 2
    return seq

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # initial vector
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 180
        h0 = np.full(n, 0.5)
        h0 = enforce_sum(h0, n / 2.0, rng)

    target = n / 2.0
    candidates = [h0]

    # binary random & smoothed
    for _ in range(3):
        h_bin = binary_random_vector(n, rng)
        candidates.append(h_bin)
        candidates.append(enforce_sum(smooth_vector(h_bin), target, rng))

    # Thue–Morse
    tm = thue_morse(n)
    candidates.append(enforce_sum(tm, target, rng))
    candidates.append(enforce_sum(smooth_vector(tm), target, rng))

    # few continuous random
    for _ in range(4):
        candidates.append(random_feasible_vector(n, rng))

    # pick best start
    best_h = None
    best_c = float('inf')
    for h in candidates:
        c = compute_c5_fft(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    global_h, global_c = best_h, best_c

    beta_list = [5.0, 15.0, 40.0, 120.0, 400.0, 800.0]

    while time.time() - start_time < budget_s * 0.78:
        if rng.random() < 0.7:
            start = global_h.copy()
            # small perturbation
            i, j = rng.integers(0, n, size=2)

            max_neg = min(start[j], 1.0 - start[i])

                eps *= 0.3
                start[i] -= eps
                start[j] += eps
                start = np.clip(start, 0.0, 1.0)
        else:

        h_opt, c_opt = hill_climb(start, rng, max_iters=3500, step=0.5,
                                  time_limit=budget_s, start_time=start_time, budget=0.78)
        if c_opt < global_c:
            for beta in beta_list:
                if time.time() - start_time > budget_s * 0.78:
                    break
                try:
                    h_opt = slsqp_optimize(h_opt, beta, max_iter=400)
                except Exception:
                    continue
                c_opt = compute_c5_fft(h_opt)
                if c_opt < global_c:
                    global_h, global_c = h_opt.copy(), c_opt
            if c_opt < global_c:
                global_h, global_c = h_opt.copy(), c_opt

            h_anneal, c_anneal = anneal_optimize(global_h, rng,
                                                init_temp=0.05, final_temp=1e-6,
                                                max_iters=12000,

            if c_anneal < global_c:

        n_up = h_up.size

        h_up = enforce_sum(h_up, target_up, rng)
        h_up_opt, c_up_opt = hill_climb(h_up, rng, max_iters=5000, step=0.4,

        for beta in beta_list:
            if time.time() - start_time > budget_s:
                break

                h_up_opt = slsqp_optimize(h_up_opt, beta, max_iter=300)
            except Exception:
                pass
            c_up_opt = compute_c5_fft(h_up_opt)

    final_c = compute_c5_fft(global_h)

    return global_h.tolist(), float(final_c), int(n)
