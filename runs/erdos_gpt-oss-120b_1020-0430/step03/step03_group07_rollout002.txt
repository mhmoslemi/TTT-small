def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Determine n and initial h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.size
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        # adjust sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h = enforce_sum(h, target, rng)
    else:
        n = 360  # or any
        h = np.full(n, 0.5)
        h = enforce_sum(h, n/2.0, rng)

    # Ensure sum correct
    target = n/2.0
    # Phase 1: pairwise targeted improvement
    deadline_phase1 = start_time + budget_s * 0.75
    step = 0.5  # initial step size
    best_h = h.copy()
    best_c = compute_c5(h)
    while time.time() < deadline_phase1:
        O = compute_O_array(h)
        c = O.max()
        if c < best_c:
            best_c = c; best_h = h.copy()
        # find argmax
        k_star = int(np.argmax(O) - (n - 1))
        # choose indices and transfer
        if k_star == 0:
            # random pair transfer
            i = rng.integers(0, n)
            j = rng.integers(0, n-1)
            if j >= i:
                j += 1
            hi = h[i]; hj = h[j]
            max_eps = min(hi, 1 - hj)
            if max_eps > 0:
                eps = step * max_eps * rng.random()
                h[i] = hi - eps
                h[j] = hj + eps
        else:
            if k_star > 0:
                max_i = n - k_star
                if max_i <= 0:
                    continue
                i = rng.integers(0, max_i)
                j = i + k_star
            else:
                # k_star negative
                min_i = -k_star
                i = rng.integers(min_i, n)
                j = i + k_star
            hi = h[i]; hj = h[j]
            if hi < hj:
                delta = min(step * (hj - hi), 1 - hi, hj)
                if delta > 0:
                    h[i] = hi + delta
                    h[j] = hj - delta
            elif hi > hj:
                delta = min(step * (hi - hj), hi, 1 - hj)
                if delta > 0:
                    h[i] = hi - delta
                    h[j] = hj + delta
            else:
                # equal, random small transfer
                i2 = rng.integers(0, n)
                j2 = rng.integers(0, n-1)
                if j2 >= i2:
                    j2 += 1
                hi2 = h[i2]; hj2 = h[j2]
                max_eps = min(hi2, 1 - hj2)
                if max_eps > 0:
                    eps = step * max_eps * rng.random()
                    h[i2] = hi2 - eps
                    h[j2] = hj2 + eps
        # occasional random perturb to avoid stuck
        if rng.random() < 0.01:
            # random transfer between any two (maybe bigger step)
            i = rng.integers(0, n)
            j = rng.integers(0, n-1)
            if j >= i:
                j += 1
            hi = h[i]; hj = h[j]
            max_eps = min(hi, 1 - hj)
            if max_eps > 0:
                eps = step * max_eps * rng.random()
                h[i] = hi - eps
                h[j] = hj + eps
        # decay step
        step = max(step * 0.9995, 1e-6)

    # Phase 2: SLSQP refinement
    beta_list = [5, 20, 80, 200, 800, 2000]
    # allocate remaining time
    deadline_total = start_time + budget_s - 0.1  # leave small margin
    for beta in beta_list:
        if time.time() >= deadline_total:
            break
        try:
            h_opt = slsqp_optimize(best_h, beta, max_iter=500)

                best_h = h_opt
        except Exception:
            continue

    # Final cleanup
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size/2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def enforce_sum(h, target, rng):
    """Adjust h so that sum(h) == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def correlate_full(h):
    """Full linear correlation of h with (1‑h) using FFT."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())  # next power of two
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_O_array(h):
    """Array of overlap integrals O_k for all integer shifts."""
    conv = correlate_full(h)               # length 2n‑1
    dx = 2.0 / h.size
    return conv * dx


def compute_c5(h):
    """Maximum overlap (the C₅ value) for a discrete step function."""
    return compute_O_array(h).max()


# ----------------------------------------------------------------------
# Smooth surrogate + SLSQP (same as previous code)
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Log‑sum‑exp smoothing of the max overlap."""
    corr = correlate_full(x)
    bc = CURRENT_BETA * corr
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """SLSQP minimisation of the smoothed max overlap."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x
