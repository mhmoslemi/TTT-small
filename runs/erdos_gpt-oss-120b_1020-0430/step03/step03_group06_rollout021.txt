def feasible_move(h,i,j):

CURRENT_TARGET = None

def _softmax_obj(x):
    # smooth approximation of max_k Σ_i x_i (1-x_{i+k})

    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


# --------------------------------------------------------------
# Core utilities
# --------------------------------------------------------------
def compute_c5_fft(h):
    """Exact C5 for a step function h using circular correlation (FFT)."""
    n = h.size
    # cross‑correlation h ⋆ (1-h)
    conv = np.real(np.fft.ifft(np.fft.fft(h) * np.conj(np.fft.fft(1.0 - h))))
    return np.max(conv) * (2.0 / n)


def enforce_sum(h, target, rng):
    """Adjust h so Σh = target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-12:
            continue
        h[i] += add
        diff = target - h.sum()
    # tiny final fix
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(1000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            if abs(add) < 1e-12:
                continue
            h[i] += add
            diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    v = rng.exponential(1.0, size=n)
    v = v / v.sum() * (n / 2.0)
    h = np.clip(v, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def perturb_h(h, rng, scale=0.5):
    """Transfer a small amount between two coordinates (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    while j == i:
        j = rng.integers(0, n)
    max_pos = min(h[i], 1.0 - h[j])
    max_neg = min(h[j], 1.0 - h[i])
    if max_pos + max_neg == 0:
        return h.copy()
    eps = (rng.random() * (max_pos + max_neg)) - max_neg
    eps *= scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return np.clip(new_h, 0.0, 1.0)


def slsqp_optimize(h0, beta, max_iter=800):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            if abs(add) < 1e-12:
                continue
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# --------------------------------------------------------------
# Local search primitives
# --------------------------------------------------------------
def hill_climb_local(h0, rng, max_iters=2500, start_step=0.5):
    """Simple hill‑climb: keep only improving moves, step size decays."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5_fft(h)
    n = h.size
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        max_pos = min(h[i], 1.0 - h[j])
        max_neg = min(h[j], 1.0 - h[i])
        if max_pos + max_neg == 0:
            continue
        eps = (rng.random() * (max_pos + max_neg)) - max_neg
        eps *= step
        if eps == 0:
            continue
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c = compute_c5_fft(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


def simulated_annealing(h_start, rng, time_budget, start_temp=0.02, decay=0.9995):
    """SA on the exact C5 objective (circular correlation)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5_fft(h)
    cur_c = best_c
    n = h.size
    it = 0
    start = time.time()
    while time.time() - start < time_budget:
        T = max(1e-6, start_temp * (decay ** it))
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        max_pos = min(h[i], 1.0 - h[j])
        max_neg = min(h[j], 1.0 - h[i])
        if max_pos + max_neg == 0:
            it += 1
            continue
        eps = (rng.random() * (max_pos + max_neg)) - max_neg
        # optional scaling of move size by temperature
        eps *= T / start_temp
        if eps == 0:
            it += 1
            continue
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c_new = compute_c5_fft(new_h)
        delta = c_new - cur_c
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            h = new_h
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = new_h.copy()
        it += 1
    return best_h, best_c


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ---- initial vector -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 180
        h0 = random_feasible_vector(n, rng)

    target = n / 2.0
    best_h = h0.copy()
    best_c = compute_c5_fft(best_h)

    # ---- generate a modest pool of seeds -------------------------------
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx
    pool = [best_h]
    for f in (1, 2, 3):
        for fun in (np.sin, np.cos):
            h = 0.5 + 0.4 * fun(np.pi * f * xs)
            h = np.clip(h, 0.0, 1.0)
            pool.append(enforce_sum(h, target, rng))
    for _ in range(8):
        pool.append(random_feasible_vector(n, rng))

    for h in pool:
        c = compute_c5_fft(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # ---- main stochastic search -----------------------------------------
    # parameters
    beta_list = [200, 800, 2000]
    while time.time() - start_time < budget_s * 0.85:
        # pick start point
        if rng.random() < 0.75:
            start = perturb_h(best_h, rng, scale=0.3)
        else:
            start = random_feasible_vector(n, rng)

        # hill‑climb
        h_hc, c_hc = hill_climb_local(start, rng, max_iters=3000, start_step=0.6)
        if c_hc < best_c:
            best_h, best_c = h_hc, c_hc

        # if a worthwhile improvement, try SLSQP refinement
        if c_hc < best_c - 1e-5:
            for beta in beta_list:
                if time.time() - start_time >= budget_s * 0.85:
                    break
                try:
                    h_opt = slsqp_optimize(best_h, beta, max_iter=600)
                except Exception:
                    continue
                c_opt = compute_c5_fft(h_opt)
                if c_opt < best_c:
                    best_h, best_c = h_opt.copy(), c_opt

        # occasional simulated annealing
        remaining = budget_s - (time.time() - start_time)
        if remaining > 15:
            sa_time = min(remaining * 0.15, 30.0)
            h_sa, c_sa = simulated_annealing(best_h, rng, sa_time)
            if c_sa < best_c:
                best_h, best_c = h_sa, c_sa

    # ---- final polishing phase -----------------------------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 5:
        # final hill‑climb with finer steps
        h_fine, c_fine = hill_climb_local(best_h, rng, max_iters=8000, start_step=0.2)
        if c_fine < best_c:
            best_h, best_c = h_fine, c_fine
        # final SLSQP with high β
        for beta in (2000, 4000):
            if time.time() - start_time >= budget_s:
                break
            try:
                h_opt = slsqp_optimize(best_h, beta, max_iter=1000)
            except Exception:
                continue
            c_opt = compute_c5_fft(h_opt)
            if c_opt < best_c:
                best_h, best_c = h_opt.copy(), c_opt

    # ---- clean‑up --------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_c = compute_c5_fft(best_h)

    return best_h.tolist(), float(final_c), int(n)
