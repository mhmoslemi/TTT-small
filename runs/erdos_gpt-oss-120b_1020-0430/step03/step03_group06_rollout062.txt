import time
import numpy as np

def compute_c5(h):
    """Exact C5 value for a step function h."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def _initial_binary(N, rng):
    """Create a random binary vector with exactly N/2 ones."""
    a = np.zeros(N, dtype=np.int8)
    ones = rng.choice(N, size=N // 2, replace=False)
    a[ones] = 1
    return a


def _binary_from_initial(N, rng):
    """If a suitable initial_h_values exists, turn it into a binary vector."""
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size == N:
            a = (h0 > 0.5).astype(np.int8)
            target = N // 2
            cur = int(a.sum())
            if cur > target:
                idx = np.where(a == 1)[0]
                rm = rng.choice(idx, size=cur - target, replace=False)
                a[rm] = 0
            elif cur < target:
                idx = np.where(a == 0)[0]
                add = rng.choice(idx, size=target - cur, replace=False)
                a[add] = 1
            return a
    return None


def simulated_annealing(N, rng, time_limit):
    """
    Simulated annealing on a binary vector of length N.
    Returns the best binary vector found and its max correlation value.
    """
    # initialise binary vector
    a = _binary_from_initial(N, rng)
    if a is None:
        a = _initial_binary(N, rng)

    b = 1 - a
    c = np.correlate(a, b, mode="full").astype(np.int32)
    cur_max = int(c.max())
    best_max = cur_max
    best_a = a.copy()

    # pre‑compute index tables for fast updates
    arr = np.arange(N, dtype=np.int32)
    left_idx = (arr[:, None] - arr[None, :] + N - 1).astype(np.int32)   # shape (N,N)
    right_idx = (arr[None, :] - arr[:, None] + N - 1).astype(np.int32)  # shape (N,N)
    center = N - 1

    # annealing schedule
    T0 = 4.0
    Tmin = 0.001
    restart_limit = max(5000, N * 30)
    steps_since_best = 0
    start = time.time()

    while time.time() - start < time_limit:
        # temperature (linear cooling)
        frac = (time.time() - start) / time_limit
        T = max(Tmin, T0 * (1.0 - frac))

        # pick a random 1 and a random 0
        p = rng.integers(N)
        while a[p] == 0:
            p = rng.integers(N)
        q = rng.integers(N)
        while a[q] == 1:
            q = rng.integers(N)

        # backup current state
        c_old = c.copy()
        a_old = a.copy()

        # update correlation using fancy‑index add/subtract
        np.subtract.at(c, left_idx[p], b)
        np.add.at(c, left_idx[q], b)
        np.add.at(c, right_idx[p], a)
        np.subtract.at(c, right_idx[q], a)

        # contributions of the swap itself
        c[center] -= 2
        d = q - p
        c[center + d] += 1
        c[center - d] += 1

        # perform the swap
        a[p], a[q] = a[q], a[p]
        b[:] = 1 - a

        new_max = int(c.max())
        delta = new_max - cur_max

        # acceptance test
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                steps_since_best = 0
            else:
                steps_since_best += 1
        else:
            # revert
            c[:] = c_old
            a[:] = a_old
            b[:] = 1 - a
            steps_since_best += 1

        # random restart if we have been stuck too long
        if steps_since_best >= restart_limit:
            a = _initial_binary(N, rng)
            b = 1 - a
            c = np.correlate(a, b, mode="full").astype(np.int32)
            cur_max = int(c.max())
            steps_since_best = 0

    return best_a, best_max


def run(seed=42, budget_s=1100, **kwargs):

    start_time = time.time()

    # primary dimension (larger N gives a finer bound for the same max correlation)
    N_primary = 200
