        Q[rs_i:rs_i+n] -= eps * h_rev
        # forward j
        fs_j = f_start[j]
        Q[fs_j:fs_j+n] += eps * h
        # reverse j
        rs_j = r_start[j]
        Q[rs_j:rs_j+n] += eps * h_rev
        # Compute new minQ
        new_minQ = Q.min()
        # Score = new_minQ
        delta = new_minQ - minQ
        accept = False
        if delta > 0:
            accept = True
        else:
            # probability
            prob = np.exp(delta / T) if T > 0 else 0.0
            if rng.random() < prob:
                accept = True
        if accept:
            # Update h, minQ, score
            h[i] = hi - eps
            h[j] = hj + eps
            minQ = new_minQ
            score = new_minQ
        else:

            # revert forward i
            Q[fs_i:fs_i+n] += eps * h

            Q[rs_i:rs_i+n] += eps * h_rev
            # revert forward j
            Q[fs_j:fs_j+n] -= eps * h
            # revert reverse j
            Q[rs_j:rs_j+n] -= eps * h_rev
            # h unchanged
        # Temperature decay
        T = max(T * decay, T_min)
        it += 1
    return h, Q, minQ, score

# ---------- utilities ----------
def enforce_sum(h, target, rng):
    """Clip to [0,1] and adjust to make sum exactly target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i])
            add *= rng.random()
            h[i] += add
            diff -= add
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i])
            sub *= rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        if diff > 0:
            h[i] = min(1.0, h[i] + diff)
        else:
            h[i] = max(0.0, h[i] + diff)
    return h

def compute_c5(h):
    """Exact discrete C₅."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

    start = time.time()

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:

        accept = False
        if delta > 0:
            accept = True
        else:
            prob = np.exp(delta / T) if T > 0 else 0.0
            if rng.random() < prob:
                accept = True
        if accept:
            h[i] = hi - eps
            h[j] = hj + eps
            minQ = new_minQ
            if minQ > best_minQ:
                best_minQ = minQ
                best_h = h.copy()
        else:
            # revert Q using original values
            Q[fs_i:fs_i + n] += eps * h
            Q[rs_i:rs_i + n] += eps * h_rev
            Q[fs_j:fs_j + n] -= eps * h
            Q[rs_j:rs_j + n] -= eps * h_rev
        T = max(T * decay, T_min)
        it += 1
    return best_h, best_minQ

# ---------- SLSQP refinement ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    """Smooth approximation followed by SLSQP."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------ initial vector ------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n0 = 180
        h0 = rng.random(n0)
        h0 = enforce_sum(np.clip(h0, 0.0, 1.0), n0 / 2.0, rng)

    # ------ target resolution ------
    target_n = 720
    if h0.size < target_n:
        # up‑sample by linear interpolation
        xs_old = (np.arange(h0.size) + 0.5) * (2.0 / h0.size)
        xs_new = (np.arange(target_n) + 0.5) * (2.0 / target_n)
        h = np.interp(xs_new, xs_old, h0)
    else:
        h = h0[:target_n].copy()
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, target_n / 2.0, rng)

    # ------ initial correlation ------
    Q = np.correlate(h, h, mode="full")
    minQ = Q.min()
    best_h = h.copy()
    best_minQ = minQ

    # ------ simulated annealing phase ------
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    sa_time = remaining * 0.7  # majority of the time
    h_sa, minQ_sa = sa_optimize(best_h, Q.copy(), minQ, rng, sa_time)
    if minQ_sa > best_minQ:
        best_h, best_minQ = h_sa, minQ_sa

    # ------ SLSQP refinement (quick) ------
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    beta_sched = [5.0, 20.0, 80.0, 200.0, 800.0]
    for beta in beta_sched:
        if time.time() - start_time > budget_s * 0.95:
            break
        try:
            h_ref = slsqp_optimize(best_h, beta, max_iter=300)
        except Exception:
            continue
        c_ref = compute_c5(h_ref)
        if c_ref < compute_c5(best_h):
            best_h = h_ref
            best_minQ = (target_n / 2.0) - c_ref * (target_n / 2.0)  # not crucial

    # ------ final tiny random search ------
    while time.time() - start_time < budget_s * 0.99:
        # small transfer move
        n = best_h.size
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        hi, hj = best_h[i], best_h[j]
        max_transfer = min(hi, 1.0 - hj)
        if max_transfer <= 1e-12:
            continue
        eps = max_transfer * rng.random() * 0.02
        new_h = best_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c_new = compute_c5(new_h)
        if c_new < compute_c5(best_h):
            best_h = new_h

    # ------ clean‑up ------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
