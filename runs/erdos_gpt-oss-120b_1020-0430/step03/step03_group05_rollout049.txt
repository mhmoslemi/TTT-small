    best_c = np.inf
    best_n = None

    # Determine n candidates
    candidates = [120, 180, 240, 300]  # sorted increasing
    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        if n0 not in candidates:
            candidates = [n0] + candidates

    remaining_budget = budget_s

    for n in candidates:
        t0 = time.time()
        elapsed = t0 - start_time
        remaining = budget_s - elapsed
        if remaining < 10:
            break

        # allocate time to this n proportionally: maybe 0.25 for low, 0.35 for high.
        # We'll do simple: allocate min(remaining*0.4, max(5, n*0.5))
        time_for_n = min(remaining * 0.4, max(5.0, n * 0.5))
        # split: de_time 0.5, slsqp_time 0.3, greedy_time 0.2
        de_time = time_for_n * 0.5
        slsqp_time = time_for_n * 0.3
        greedy_time = time_for_n * 0.2

        # Initial vector: upsample from initial_h_values if present else random.
        if "initial_h_values" in globals() and len(initial_h_values) == n:
            h0 = np.array(initial_h_values, dtype=float)
        else:
            # if we have a higher resolution initial, upsample; else random.
            if "initial_h_values" in globals():
                # upsample

                h0 = np.interp(xs_new, xs_old, init)
                # ensure feasible
                h0 = project_to_feasible(h0, n/2)
            else:
                h0 = random_feasible_vector(n, rng)

        # DE optimization
        h_de, c_de = de_optimize(n, rng, de_time, init=h0)

def compute_c5(h):
    """Exact discrete C5 for step function h on [0,2] (dx = 2/len(h))."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)


def project_to_feasible(v, target):
    """Shift‑and‑clip projection onto {0≤h≤1, Σh = target}."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
