def project_onto_box_sum(v, target):
    """
    Project vector v onto feasible set {x: 0 <= x <= 1, sum(x) = target}.
    Returns projected vector.
    """
    # Use bisection on λ:
    lo = -1.0   # lambda lower bound
    hi = 1.0    # lambda upper bound
    for _ in range(60):
        lam = (lo+hi)/2
        x = np.clip(v - lam, 0, 1)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo+hi)/2, 0, 1)

def gradient_descent_optimize(h0, rng, time_budget):
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    # field for no improve detection
    last_improve_iter = 0
    iteration = 0
    while time.time() - start < time_budget:
        # schedule
        elapsed = time.time() - start
        frac = elapsed / time_budget
        beta = 5.0 * (2000.0 / 5.0) ** frac  # geometric from 5 to 2000
        step = 0.6 / (beta + 0.1)
        s = 1.0 - h
        g = np.correlate(h, s, mode='full')
        maxg = g.max()
        w = np.exp(beta * (g - maxg))
        w_sum = w.sum()
        w /= w_sum
        # gradients
        A = np.correlate(w, s, mode='full')[:n]
        B = np.convolve(w, h, mode='full')[n-1:n-1+n]
        grad = A - B
        # update
        h = project_onto_feasible(h - step * grad, target)
        iteration += 1
        # evaluate every 500 iters
        if iteration % 500 == 0:
            cur_c = compute_c5(h)
            if cur_c < best_c:
                best_c, best_h = cur_c, h.copy()
                last_improve_iter = iteration
                # early exit if good enough
                if best_c <= 0.38079:
                    break
            else:
                # if no improvement for many iters, add random noise
                if iteration - last_improve_iter > 5000:
                    # jitter best_h
                    noise = rng.normal(scale=0.05, size=n)
                    h = project_onto_feasible(best_h + noise, target)
                    last_improve_iter = iteration
    # final best
    final_c = compute_c5(best_h)
    if final_c < best_c:
        best_c, best_h = final_c, best_h
    return best_h, best_c
```

import time
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# basic utilities
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for step function h on [0,2] with dx=2/len(h)."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)


def project_onto_feasible(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto {x : 0 <= x_i <= 1, sum(x) = target}.
    Uses bisection on the Lagrange multiplier λ.
    """
    lo, hi = -1e6, 1e6
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random feasible vector (0≤h≤1, sum = n/2)."""
    v = rng.random(n)
    return project_onto_feasible(v, n / 2.0)


# ------------------------------------------------------------
# gradient‑descent optimizer (smooth max)
# ------------------------------------------------------------
def gradient_descent_optimize(h0: np.ndarray,
                              rng: np.random.Generator,
                              time_budget: float) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the soft‑max approximation of max overlap."""
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)

    start = time.time()
    last_improve = 0
    it = 0
    beta_start, beta_end = 5.0, 2000.0

    while time.time() - start < time_budget:
        # schedule β geometrically from start to end
        frac = (time.time() - start) / time_budget
        beta = beta_start * (beta_end / beta_start) ** frac
        step = 0.6 / (beta + 0.1)

        s = 1.0 - h
        g = np.correlate(h, s, mode="full")
        maxg = g.max()
        w = np.exp(beta * (g - maxg))
        w /= w.sum()

        # gradient: A = corr(w, s)[:n], B = conv(w, h)[n-1:n-1+n]
        A = np.correlate(w, s, mode="full")[:n]
        B = np.convolve(w, h, mode="full")[n - 1:n - 1 + n]
        grad = A - B

        # projected step (gradient sum is zero, but we keep the projection for bounds)
        h = project_onto_feasible(h - step * grad, target)

        it += 1
        if it % 500 == 0:
            cur_c = compute_c5(h)
            if cur_c < best_c:
                best_c, best_h = cur_c, h.copy()
                last_improve = it
                if best_c <= 0.38079:          # early success guard
                    break
            elif it - last_improve > 5000:
                # stagnation – jitter around the current best
                noise = rng.normal(scale=0.05, size=n)
                h = project_onto_feasible(best_h + noise, target)
                last_improve = it

    # final check
    final_c = compute_c5(best_h)
    if final_c < best_c:
        best_c, best_h = final_c, best_h.copy()
    return best_h, best_c


# ------------------------------------------------------------
# SLSQP refinement (log‑sum‑exp objective)
# ------------------------------------------------------------
CURRENT_BETA = None
CURRENT_TARGET = None


def _sum_eq_constraint(x: np.ndarray) -> float:
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x: np.ndarray) -> float:
    s = 1.0 - x
    g = np.correlate(x, s, mode="full")
    maxg = g.max()
    # stabilized log‑sum‑exp
    z = np.exp(CURRENT_BETA * (g - maxg))
    return (maxg + np.log(z.sum())) / CURRENT_BETA


def slsqp_refine(h: np.ndarray, beta: float, max_iter: int) -> np.ndarray:
    """Run a short SLSQP optimisation of the smoothed max‑overlap."""
    global CURRENT_BETA, CURRENT_TARGET
    n = h.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(_softmax_obj, h, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    return project_onto_feasible(x, CURRENT_TARGET)

# ------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_total = time.time()

    # ----------------------------------------------------------------
    # initialise h
    # ----------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_onto_feasible(h0, n / 2.0)
    else:
        n = 180
        h0 = random_feasible(n, rng)

    # ----------------------------------------------------------------
    # gradient‑descent phase (≈85 % of the budget)
    # ----------------------------------------------------------------
    grad_budget = budget_s * 0.85

    # ----------------------------------------------------------------
    # SLSQP polishing with the remaining time (≈15 % of the budget)
    # ----------------------------------------------------------------
    time_used = time.time() - start_total
    remain = max(0.0, budget_s - time_used)
    if remain > 5.0:                               # leave a small safety margin
        for beta in (2000.0, 5000.0, 20000.0):
            if time.time() - start_total > budget_s:
                break
            try:
                h_opt = slsqp_refine(h_opt, beta, max_iter=250)
                c_opt = compute_c5(h_opt)
            except Exception:
                continue

    # final safety projection & evaluation
    # ----------------------------------------------------------------
    h_opt = np.clip(h_opt, 0.0, 1.0)
    h_opt = project_onto_feasible(h_opt, n / 2.0)
    final_c = compute_c5(h_opt)

    return h_opt.tolist(), float(final_c), int(n)
