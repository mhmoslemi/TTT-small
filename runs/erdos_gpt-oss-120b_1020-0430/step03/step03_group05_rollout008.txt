def compute_c5_fast(h):
    n = h.size
    # correlation: conv of h and reversed (1-h)
    rev = (1 - h)[::-1]
    nfft = 1 << ((2*n - 1).bit_length())
    # Compute using rfft for speed
    # converted to complex
    H = np.fft.rfft(h, n=nfft)
    R = np.fft.rfft(rev, n=nfft)
    conv = np.fft.irfft(H * R, n=nfft)
    # conv length nfft, but relevant part is first 2*n -1
    corr = conv[:2*n-1]
    return np.max(corr) * (2.0 / n)
```

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Global placeholders for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint_global(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def full_correlation(h):
    """Cross‑correlation of h with 1‑h using FFT (full length)."""
    n = h.shape[0]
    rev = (1.0 - h)[::-1]
    nfft = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=nfft)
    R = np.fft.rfft(rev, n=nfft)
    conv = np.fft.irfft(H * R, n=nfft)
    return conv[: 2 * n - 1]


def softmax_obj_global(x):
    """Smooth approximation of max using log‑sum‑exp with CURRENT_BETA."""
    beta = CURRENT_BETA
    corr = full_correlation(x)
    bc = beta * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta


def compute_c5_fast(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    n = h.shape[0]
    corr = full_correlation(h)
    return np.max(corr) * (2.0 / n)


def project_to_feasible(v, target):
    """Projection of v onto {0≤h≤1, Σh = target} via bisection on the shift."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def enforce_sum_and_bounds(v, target, rng):
    """Clip to [0,1] and adjust a single coordinate to hit the required sum."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    if abs(diff) < 1e-12:
        return v
    # try a few random coordinates
    for _ in range(10):
        i = rng.integers(0, v.size)
        if diff > 0:
            add = min(diff, 1.0 - v[i])
        else:
            add = max(diff, -v[i])
        if abs(add) > 0:
            v[i] += add
            diff = target - v.sum()
            if abs(diff) < 1e-12:
                break
    if abs(diff) > 1e-12:
        v = project_to_feasible(v, target)
    return v

    """Fast random feasible vector (0≤h≤1, Σh=n/2)."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_to_feasible(v, n / 2.0)

def hill_climb(h_start, rng, max_iters=10000, step_start=0.2, step_decay=0.99):
    """Simple monotone hill‑climb (transfer ε between two coordinates)."""
    h = h_start.copy()
    best_c = compute_c5_fast(h)
    step = step_start
    n = h.size
    target = n / 2.0

            h = new_h
