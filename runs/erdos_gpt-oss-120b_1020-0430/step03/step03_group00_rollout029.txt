def targeted_move(h, grad, sigma, rng):
    pos = np.where((grad > 1e-9) & (h > 1e-12))[0]
    neg = np.where((grad < -1e-9) & (h < 1.0 - 1e-12))[0]
    if pos.size == 0 or neg.size == 0:
        # fallback random move
        i = rng.integers(0, h.size)
        j = rng.integers(0, h.size)
        while i == j:
            j = rng.integers(0, h.size)
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            return h
        eps = max_eps * sigma * rng.random()
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        return h_new
    i = rng.choice(pos)
    j = rng.choice(neg)
    max_eps = min(h[i], 1.0 - h[j])
    eps = max_eps * sigma * rng.random()
    h_new = h.copy()
    h_new[i] -= eps
    h_new[j] += eps
    return h_new

def gradient_descent_targeted(h0, budget_s, start_time, rng):
    h = h0.copy()
    n = h.size
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    # pseudo-simulated annealing schedule:
    # start sigma = 0.5, end sigma = 0.001
    # start T = maybe 0.01
    # timeframe total = budget_s - (time.time() - start_time)
    # We'll compute allowed_time = budget_s - (time.time() - start_time)
    while True:
        now = time.time()
        elapsed = now - start_time
        if elapsed >= budget_s:
            break
        # compute gradient and worst shift
        k, _, grad = worst_shift_and_grad(h)
        # schedule sigma: linear decay based on elapsed fraction
        fraction = (elapsed) / budget_s
        sigma = 0.5 * (1 - fraction) + 0.001
        T = 0.01 * (1 - fraction) + 0.0001
        # propose move
        h_new = targeted_move(h, grad, sigma, rng)
        # ensure still within bounds and sum (transferred)
        # evaluate
        c_new = compute_c5(h_new)
        # update best
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        # acceptance
        if c_new < cur_c:
            h = h_new
            cur_c = c_new
        else:
            if rng.random() < np.exp((cur_c - c_new) / T):
                h = h_new
                cur_c = c_new
        # occasional random restart: with small prob, replace with random feasible vector near best
        if rng.random() < 0.001:
            # random perturb around best
            h = best_h + rng.normal(scale=0.05, size=n)
            h = enforce_sum(np.clip(h,0,1), n/2.0, rng)
            cur_c = compute_c5(h)
    return best_h, best_c

def enforce_sum(h, target, rng):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # iterate small number up to 1000 times
    max_iter = 1000
    it = 0
    # use while loop
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:
            # need to add mass to indices with h < 1
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i])
            # add some random proportion
            add = add * rng.random()
            h[i] += add
            diff -= add
        else:
            # need to remove mass
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i])
            sub = sub * rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    # final correction if still diff
    if abs(diff) > 1e-12:
        # adjust one random index
        i = rng.integers(0, h.size)
        if diff > 0:
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
        else:
            h[i] = np.clip(h[i] + diff, 0.0, 1.0) # diff negative
    return h

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False}
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum using a new rng for reproducibility
    x = enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

Define worst_shift_and_grad:

def worst_shift_and_grad(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    max_corr = corr[idx]
    k = idx - (n - 1)
    grad = np.ones(n)
    if k >= 0:
        grad[:n - k] -= h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[:n - kk] -= h[kk:]
        grad[kk:] -= h[:n - kk]
    return k, max_corr, grad

def perturb_random(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    h_new = h.copy()
    h_new[i] -= eps
    h_new[j] += eps
    return h_new

import time
import numpy as np
from scipy.optimize import minimize

# ----- global for SLSQP surrogate -----
CURRENT_TARGET = None
CURRENT_BETA = None

def enforce_sum(h, target, rng):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 1000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i])
            add *= rng.random()
            h[i] += add
            diff -= add
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i])
            sub *= rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def worst_shift_and_grad(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr)
    k = idx - (n - 1)
    grad = np.ones(n)
    if k >= 0:
        grad[:n - k] -= h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[:n - kk] -= h[kk:]
        grad[kk:] -= h[:n - kk]
    return k, grad

def targeted_move(h, grad, sigma, rng):
    pos = np.where((grad > 1e-9) & (h > 1e-12))[0]
    neg = np.where((grad < -1e-9) & (h < 1.0 - 1e-12))[0]
    if pos.size == 0 or neg.size == 0:
        # fallback random transfer
        i = rng.integers(0, h.size)
        j = rng.integers(0, h.size)
        while i == j:
            j = rng.integers(0, h.size)
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            return h.copy()
        eps = max_eps * sigma * rng.random()
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        return h_new
    i = rng.choice(pos)
    j = rng.choice(neg)
    max_eps = min(h[i], 1.0 - h[j])
    eps = max_eps * sigma * rng.random()
    h_new = h.copy()
    h_new[i] -= eps
    h_new[j] += eps
    return h_new

def perturb_random(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    h_new = h.copy()
    h_new[i] -= eps
    h_new[j] += eps
    return h_new

def binary_projection(h):
    n = h.size
    k = n // 2
    idx = np.argpartition(h, -k)[-k:]
    b = np.zeros_like(h)
    b[idx] = 1.0
    return b

def gradient_descent_targeted(h_start, total_budget, start_time, rng):
    h = h_start.copy()
    n = h.shape[0]
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    while True:
        elapsed = time.time() - start_time
        if elapsed >= total_budget:
            break
        _, grad = worst_shift_and_grad(h)
        frac = elapsed / total_budget
        sigma = 0.5 * (1 - frac) + 0.001
        T = 0.01 * (1 - frac) + 1e-6
        h_new = targeted_move(h, grad, sigma, rng)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        if c_new < cur_c:
            h = h_new
            cur_c = c_new
        else:
            if rng.random() < np.exp(-(c_new - cur_c) / T):
                h = h_new
                cur_c = c_new
        if rng.random() < 0.001:
            h = best_h + rng.normal(scale=0.02, size=n)
            h = enforce_sum(np.clip(h, 0.0, 1.0), n / 2.0, rng)
            cur_c = compute_c5(h)
    return best_h, best_c

# ----- SLSQP surrogate utilities -----
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x
