def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_limit = budget_s
    # Determine initial vector and n0
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        # enforce sum exactly
        h0 = enforce_sum(h0, n0/2.0, rng)
    else:
        n0 = 240
        h0 = random_feasible_vector(n0, rng)
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # Phase 1: Optimize at n0
    phase1_end = start_time + 0.55 * time_limit
    # Use multiple restarts, each with SLSQP ladder and hill climbing
    # Determine number of restarts adaptively based on time.

    while time.time() < phase1_end:
        # Choose start vector: either perturb current best or random
        if rng.random() < 0.7:
            start_h = perturb_h(best_h, rng, scale=0.25)
        else:
            start_h = random_feasible_vector(n0, rng)

        # Ladder of betas
        for beta in [5, 20, 80, 200, 800, 2000]:
            if time.time() > phase1_end:
                break
            try:
                h_opt = slsqp_optimize(start_h, beta, max_iter=400)
                c_opt = compute_c5(h_opt)
                if c_opt < best_c:
                    best_h, best_c = h_opt.copy(), c_opt
                start_h = h_opt  # continue with improved start
            except Exception:
                pass

        # hill climb refine
        if time.time() < phase1_end:
            h_hc, c_hc = hill_climb(best_h, rng, max_iters=3000, start_step=0.3)
            if c_hc < best_c:
                best_h, best_c = h_hc.copy(), c_hc

    # Phase 2: Upsample if time left
    # Determine upsample factor up to ~720 or 1080
    remaining = time.time() - start_time
    if time.time() < start_time + time_limit:
        # decide factor
        n_up = min(720, best_h.size * 2)
        if n_up > best_h.size:
            # upsample by repeat
            factor = n_up // best_h.size
            h_up = np.repeat(best_h, factor)
            # enforce sum
            h_up = enforce_sum(h_up, n_up/2.0, rng)
            best_h = h_up
            best_c = compute_c5(best_h)

            phase2_end = start_time + time_limit * 0.95
            # Simple hill climbing many iterations
            while time.time() < phase2_end:
                h_hc, c_hc = hill_climb(best_h, rng, max_iters=5000, start_step=0.2)
                if c_hc < best_c:
                    best_h, best_c = h_hc.copy(), c_hc
                # occasional SLSQP with higher beta
                for beta in [800, 2000, 5000]:
                    try:
                        h_opt = slsqp_optimize(best_h, beta, max_iter=250)
                        c_opt = compute_c5(h_opt)
                        if c_opt < best_c:
                            best_h, best_c = h_opt.copy(), c_opt
                    except Exception:
                        pass
    # Final cleaning
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size/2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)

def binary_sa(start, rng, deadline):
    h = start.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)

    T0 = best_c*0.02
    Tf = 1e-6
    max_iters = 20000
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if h[i] == h[j]:
            continue
        # ensure i is 1 and j is 0 for swap
        if h[i] < h[j]:
            i, j = j, i
        # now h[i]==1, h[j]==0
        # swap
        h[i] = 0.0
        h[j] = 1.0
        c = compute_c5(h)
        delta = c - best_c
        # Temperature schedule
        frac = it / max_iters
        T = T0 * ((Tf/T0) ** frac)
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            if c < best_c:
                best_c, best_h = c, h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c

def enforce_sum(h, target, rng):
    """Adjust h so that Σh == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h = enforce_sum(h, target, rng)
    return h

        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


# --------------------------------------------------------------
# Correlation utilities (FFT based)
# --------------------------------------------------------------
def correlate_full(h):
    """Full linear correlation of h with (1-h) using FFT (zero‑padding)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)


# --------------------------------------------------------------
# Smooth surrogate for SLSQP
# --------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Log‑sum‑exp smoothing of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = correlate_full(x)
    bc = CURRENT_BETA * corr
    m = bc.max()
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=350):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# --------------------------------------------------------------
# Purely combinatorial SA on binary vectors
# --------------------------------------------------------------
def binary_sa(start_bin, rng, deadline):
    """Swap‑based simulated annealing on a binary vector (sum = n/2)."""
    h = start_bin.astype(float).copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)

    # temperature schedule (exponential)
    T0 = best_c * 0.02 + 1e-6
    Tf = 1e-6
    max_iters = 20000

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if h[i] == h[j]:
            continue          # not a swap
        # enforce i is 1, j is 0
        if h[i] < h[j]:
            i, j = j, i
        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        c = compute_c5(h)
        delta = c - best_c
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            if c < best_c:
                best_c, best_h = c, h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c


# --------------------------------------------------------------
# Simple greedy hill‑climb (continuous)
# --------------------------------------------------------------
def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Greedy hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    budget = budget_s

    # ------------------------------------------------------------------
    # 1. Base resolution and initial candidate set
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 240
        h0 = random_feasible_vector(n, rng)

    # a few hand‑crafted patterns (including a triangular shape)
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx
    candidates = [h0]

    # triangular (tent) shape
    tri = 1.0 - np.abs(xs - 1.0)
    candidates.append(enforce_sum(np.clip(tri, 0.0, 1.0), n / 2.0, rng))

    # sin / cos families
    for f in range(1, 5):
        s = 0.5 + 0.5 * np.sin(np.pi * f * xs)
        candidates.append(enforce_sum(np.clip(s, 0.0, 1.0), n / 2.0, rng))
        c = 0.5 + 0.5 * np.cos(np.pi * f * xs)
        candidates.append(enforce_sum(np.clip(c, 0.0, 1.0), n / 2.0, rng))

    # binary block
    block = np.zeros(n)
    block[: n // 2] = 1.0
    candidates.append(enforce_sum(block, n / 2.0, rng))

    # alternating 0/1
    alt = np.tile([0.0, 1.0], n // 2)
    candidates.append(enforce_sum(alt, n / 2.0, rng))

    # three random feasible vectors
    for _ in range(3):
        candidates.append(random_feasible_vector(n, rng))

    # ------------------------------------------------------------------
    # 2. Pick best among the pool
    # ------------------------------------------------------------------
    best_h = None
    best_c = float("inf")
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # ------------------------------------------------------------------
    # 3. Continuous optimisation at base resolution
    # ------------------------------------------------------------------
    beta_list = [5.0, 20.0, 80.0, 200.0, 800.0, 2000.0]
    phase1_end = start_time + 0.55 * budget

    while time.time() < phase1_end:
        # start either from perturbation of current best or a fresh random point
        if rng.random() < 0.65:
            start = perturb_h(best_h, rng, scale=0.3)
        else:
            start = random_feasible_vector(n, rng)

        # SLSQP ladder
        for beta in beta_list:
            if time.time() >= phase1_end:
                break
            try:
                h_opt = slsqp_optimize(start, beta, max_iter=400)
                c_opt = compute_c5(h_opt)
                if c_opt < best_c:
                    best_h, best_c = h_opt.copy(), c_opt
                start = h_opt
            except Exception:
                continue

        # greedy hill‑climb refinement
        if time.time() < phase1_end:
            h_hc, c_hc = hill_climb(best_h, rng, max_iters=2500, start_step=0.3)
            if c_hc < best_c:
                best_h, best_c = h_hc.copy(), c_hc

    # ------------------------------------------------------------------
    # 4. Binary simulated annealing (discrete exploration)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.85 * budget
    # create binary start from the current best (top n/2 entries)
    idx_sorted = np.argsort(-best_h)[: n // 2]
    bin_start = np.zeros(n)
    bin_start[idx_sorted] = 1.0

    bin_h, bin_c = binary_sa(bin_start, rng, phase2_end)
    if bin_c < best_c:
        best_h, best_c = bin_h.copy(), bin_c

    # a short continuous polishing after binary SA
    if time.time() < phase2_end:
        h_pol, c_pol = hill_climb(best_h, rng, max_iters=3000, start_step=0.2)
        if c_pol < best_c:
            best_h, best_c = h_pol.copy(), c_pol

    # ------------------------------------------------------------------
    # 5. Optional up‑sampling to finer grid
    # ------------------------------------------------------------------
    time_now = time.time()
    if time_now < start_time + budget:
        # aim for at most 720 points (factor 2) if not already there
        if best_h.size * 2 <= 720:
            factor = 2
            h_fine = np.repeat(best_h, factor)
            n_fine = h_fine.size
            # binary up‑sampling keeps the exact sum, but enforce for safety
            h_fine = enforce_sum(h_fine, n_fine / 2.0, rng)

            # brief hill‑climb on fine grid
            fine_deadline = start_time + budget
            h_fine, c_fine = hill_climb(h_fine, rng, max_iters=5000, start_step=0.2)
            if c_fine < best_c:
                best_h, best_c = h_fine.copy(), c_fine

            # final SLSQP pass at higher β on fine grid
            for beta in [800.0, 2000.0, 5000.0]:
                if time.time() >= fine_deadline:
                    break
                try:
                    h_opt = slsqp_optimize(best_h, beta, max_iter=250)
                    c_opt = compute_c5(h_opt)
                    if c_opt < best_c:
                        best_h, best_c = h_opt.copy(), c_opt
                except Exception:
                    continue

    # ------------------------------------------------------------------
    # 6. Final cleaning and return
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
