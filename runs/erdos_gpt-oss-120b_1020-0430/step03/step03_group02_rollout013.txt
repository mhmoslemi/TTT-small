def hill_climb(h, rng, max_iters=5000, step_scale=0.5):
    n = len(h)
    best_c = compute_c5(h)
    best_h = h.copy()
    for it in range(max_iters):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            # Random swap maybe
            i, j = rng.choice(n, 2, replace=False)
        max_eps = min(h[i], 1.0 - h[j]) if h[i] > 0 else 0
        if max_eps <= 0:
            # If cannot move, try reverse direction (j->i)
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0:
                # pick random small epsilon
                max_eps = 1e-6
                i, j = j, i
        eps = step_scale * max_eps * rng.random()
        nh = h.copy()
        nh[i] -= eps
        nh[j] += eps
        # compute c
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h = c, nh.copy()
        # decide acceptance: greedy or SA?
        # We'll accept if c < current or with small prob.
        if c < best_c or rng.random() < 0.001:
            h = nh
        # optionally reduce step scale
    return best_h, best_c

    beta_O = beta * O
    m = np.max(beta_O)
    exp_shift = np.exp(beta_O - m)
    sum_exp = np.exp(-m) * np.sum(exp_shift) * np.exp(m) ??? Actually we already have exp_shift = np.exp(beta_O - m); sum_exp = np.sum(exp_shift)
    weights = exp_shift / sum_exp

def smooth_max_obj_grad(h, beta, lam):
    n = h.shape[0]
    O = np.correlate(h, 1.0 - h, mode='full')
    beta_O = beta * O
    m = np.max(beta_O)
    exp_shift = np.exp(beta_O - m)
    sum_exp = np.sum(exp_shift)
    weights = exp_shift / sum_exp
    # gradient
    g = np.zeros_like(h)
    # loops
    for idx, w in enumerate(weights):
        if w == 0:
            continue
        k = idx - (n - 1)
        if k >= 0:
            # i < n-k
            g[:n - k] += w * (1 - h[k:])
            # i >= k
            g[k:] += w * (-h[:n - k])
        else:
            k2 = -k
            # i >= k2
            g[k2:] += w * (1 - h[:n - k2])
            # i < n - k2
            g[:n - k2] += w * (-h[k2:])
    # penalty for sum
    sum_h = h.sum()
    diff = sum_h - (n / 2.0)
    f_smooth = (m + np.log(sum_exp)) / beta
    f = f_smooth + lam * diff * diff
    g += 2 * lam * diff
    return f, g
```

But note that gradient from penalty is constant 2*lam*diff for each component, which is vector added to g, fine.

def enforce_sum(h, rng, max_iter=2000):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # while diff not zero
    i = 0
    while abs(diff) > 1e-12 and i < max_iter:
        idx = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[idx])
        else:
            add = max(diff, -h[idx])
        h[idx] += add
        diff = target - h.sum()
        i += 1
    # if still diff (unlikely), scale whole vector
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        # final adjust
        diff2 = target - h.sum()
        if abs(diff2) > 1e-12:
            # distribute small correction equally
            h += diff2 / n
            h = np.clip(h, 0.0, 1.0)

import time, numpy as np
from scipy.optimize import minimize

# -------------------------------------------------------------
def compute_c5(h):
    """Discrete C5 = max_k Σ h_i (1-h_{i+k}) * (2/n)."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h, rng, max_iter=2000):
    """Adjust h so that Σh = n/2 while keeping 0≤h≤1."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
        it += 1
    # fallback tiny global fix
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
    return h

def random_binary_vector(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

    """Gradient of the worst shift O_k = Σ h_i (1‑h_{i+k})."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')

    n = h.size
