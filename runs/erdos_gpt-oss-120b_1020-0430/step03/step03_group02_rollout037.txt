import time, numpy as np

# ------------------------------------------------------------
# Utility functions
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def proj_box_simplex(v: np.ndarray, target: float, iters: int = 30) -> np.ndarray:
    """
    Project v onto the set {x : 0 <= x <= 1, sum(x) = target}.
    Simple bisection on the Lagrange multiplier.
    """
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(iters):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        if x.sum() > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


# ------------------------------------------------------------
# Gradient descent on the smooth max objective
# ------------------------------------------------------------
def gradient_descent(
    h: np.ndarray, rng: np.random.Generator, time_limit: float, max_iter: int = 8000
):
    """
    Minimise the smooth maximum of the overlap integral.
    Returns the best vector encountered and its C5 value.
    """
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(best_h)

    beta = 5.0
    beta_max = 2000.0
    step = 0.5
    start = time.time()
    it = 0

    while it < max_iter and (time.time() - start) < time_limit:
        # correlation and current C5
        O = np.correlate(h, 1 - h, mode="full")
        c_val = np.max(O) * (2.0 / n)
        if c_val < best_c:
            best_c = c_val
            best_h = h.copy()

        # smooth‑max weights
        maxO = np.max(O)
        expO = np.exp(beta * (O - maxO))
        w = expO / np.sum(expO)

        # gradient of the smooth max
        grad = np.ones_like(h)
        offset = n - 1
        for idx, weight in enumerate(w):
            shift = idx - offset
            if shift >= 0:
                grad[: n - shift] -= weight * h[shift:]
            else:
                s = -shift
                grad[s:] -= weight * h[: n - s]

        # gradient step + projection
        h = h - step * grad
        h = proj_box_simplex(h, target)

        # enforce symmetry (helps the problem) and re‑project
        h = (h + h[::-1]) * 0.5
        h = proj_box_simplex(h, target)

        # schedule
        if it % 200 == 0 and it > 0:
            step = max(step * 0.8, 1e-4)
        if it % 300 == 0 and beta < beta_max:
            beta = min(beta * 1.5, beta_max)

        it += 1

    return best_h, best_c
