def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def perturb_h(h, rng, scale=0.2):
    n = h.shape[0]
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    # Determine bound on epsilon: transfer from i to j
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    # ensure eps within bounds
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb(h_start, rng, max_iters=2000, start_step=0.6):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.shape[0]
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-5)
    return best_h, best_c

def simulated_annealing(h0, rng, max_iters=20000):
    n = h0.shape[0]
    target = n / 2.0
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    T0 = max(0.01 * cur_c, 1e-4)
    for it in range(max_iters):
        step = max(0.5 * (0.999 ** it), 5e-4)
        cand_h = perturb_h(cur_h, rng, scale=step)
        cand_c = compute_c5(cand_h)

            accept = True

        if accept:
            cur_h, cur_c = cand_h, cand_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
    return best_h, best_c

def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.shape[0]
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            up = n - k
            g[:up] += 1.0 - h[k:] 
            g[k:] -= h[:up]       
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def sum_eq_constraint(x):

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.shape[0]
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )

            x[i] += add
            diff = CURRENT_TARGET - x.sum()

                break
    return x

import time, math
import numpy as np
from scipy.optimize import minimize

# ---------------------- utilities ---------------------- #
def compute_c5(h):
    """Exact discrete C5 for step function h on [0,2] with uniform spacing."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def random_feasible_vector(n, rng):
    """Uniform random vector in [0,1] with exact sum n/2."""
    h = rng.random(n)
    target = n / 2.0
    scale = target / h.sum()
    h *= scale
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.6):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-5)
    return best_h, best_c


def simulated_annealing(h0, rng, max_iters=20000):
    """Simple SA that respects the sum constraint."""
    n = h0.size
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    T0 = max(0.01 * cur_c, 1e-4)
    for it in range(max_iters):
        step = max(0.5 * (0.999 ** it), 5e-4)
        cand_h = perturb_h(cur_h, rng, scale=step)
        cand_c = compute_c5(cand_h)
        if cand_c < cur_c:
            accept = True
        else:
            T = T0 * (0.999 ** it) + 1e-12
            accept = rng.random() < math.exp(-(cand_c - cur_c) / T)
        if accept:
            cur_h, cur_c = cand_h, cand_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
    return best_h, best_c


def gradient_refine(h, rng, step=0.02, max_iter=12):
    """Refine by moving against gradient of the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            up = n - k
            g[:up] += 1.0 - h[k:]
            g[k:] -= h[:up]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


# ----------- SLSQP surrogate for max‑overlap ----------- #
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x

    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c = float("inf")
    best_n = None
    target_c = 0.3808  # stop early if we beat this

    parent = None

    # candidate discretisations (keep modest, all <=300)
    n_candidates = [180, 240, 300]

    for n in n_candidates:
        if time.time() - start > budget_s * 0.94:
            break

        # build a collection of diverse starting vectors
        start_vecs = []

        # 1) parent (exact or interpolated)
        if parent is not None:
            if parent.size == n:
                start_vecs.append(parent.copy())
            else:
                xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
                xt = np.linspace(0.0, 2.0, n, endpoint=False)
                start_vecs.append(np.interp(xt, xs, parent))

        # 2) best solution up‑sampled (if any)
        if best_h is not None and best_n != n:
            xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            start_vecs.append(np.interp(xt, xs, best_h))

        # 3) random continuous feasible vectors
        for _ in range(3):
            start_vecs.append(random_feasible_vector(n, rng))

        # 4) random binary feasible vector
        h_bin = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h_bin[idx] = 1.0
        start_vecs.append(h_bin)

        for h0 in start_vecs:
            if time.time() - start > budget_s * 0.98:
                break

            # --- enforce feasibility -----------------
            h = np.clip(h0, 0.0, 1.0)
            target = n / 2.0
            diff = target - h.sum()
            if abs(diff) > 1e-12:
                for _ in range(2000):
                    i = rng.integers(n)
                    add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                    h[i] += add
                    diff = target - h.sum()
                    if abs(diff) < 1e-12:
                        break

            # stage 1: hill‑climb (quick local improvement)
            h, c = hill_climb(h, rng, max_iters=3000, start_step=0.6)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n
                if best_c <= target_c:
                    return best_h.tolist(), float(best_c), int(best_n)

            # stage 2: simulated annealing (larger exploration)
            remaining = budget_s - (time.time() - start)
            max_sa_iters = int(min(40000, max(5000, remaining * 30)))
            h_sa, c_sa = simulated_annealing(h, rng, max_iters=max_sa_iters)
            if c_sa < best_c:
                best_h, best_c, best_n = h_sa.copy(), c_sa, n
                if best_c <= target_c:
                    return best_h.tolist(), float(best_c), int(best_n)

            # stage 3: gradient refinement (tiny tweak)
            h_ref = gradient_refine(h_sa, rng, step=0.02, max_iter=12)
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_h, best_c, best_n = h_ref.copy(), c_ref, n
                if best_c <= target_c:
                    return best_h.tolist(), float(best_c), int(best_n)

    # ---------------- final refinement ---------------- #
    if best_h is not None and time.time() - start < budget_s * 0.99:
        # optional up‑sampling (if still small)
        if best_n * 2 <= 300:
            n_up = best_n * 2
            xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
            xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
            h_up = np.interp(xt, xs, best_h)
            target_up = n_up / 2.0
            diff = target_up - h_up.sum()
            if abs(diff) > 1e-12:
                for _ in range(2000):
                    i = rng.integers(n_up)
                    add = min(diff, 1.0 - h_up[i]) if diff > 0 else max(diff, -h_up[i])
                    h_up[i] += add
                    diff = target_up - h_up.sum()
                    if abs(diff) < 1e-12:
                        break
            best_h, best_n = h_up, n_up

        # a short cascade of SLSQP passes with growing β
        for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0):
            if time.time() - start > budget_s * 0.99:
                break
            try:
                best_h = slsqp_optimize(best_h, beta, rng, max_iter=500)
            except Exception:
                continue
            best_c = compute_c5(best_h)

        # final gradient polish
        best_h = gradient_refine(best_h, rng, step=0.02, max_iter=12)
        best_c = compute_c5(best_h)

    # final safety checks
    if best_h is None:
        best_n = 180
        best_h = np.full(best_n, 0.5)

    best_h = np.clip(best_h, 0.0, 1.0)
