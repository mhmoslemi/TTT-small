def projected_gradient_ascent(h, rng, deadline):
    h = h.copy()
    n = h.size
    step_factor = 0.5
    decay = 0.9995 # per iteration
    while time.time() < deadline:
        # compute product sums via correlation
        corr = np.correlate(h, h, mode='full')
        P = corr[n-1:]   # length n
        k_min = int(np.argmin(P))
        if k_min == 0:
            # gradient for k=0: dP = 2*h
            g = 2 * h
        else:
            g = np.zeros_like(h)
            upper = n - k_min
            g[:upper] += h[k_min:]
            g[k_min:] += h[:upper]
        # project to zero sum:
        g -= np.mean(g)
        # compute max feasible step
        pos = g > 1e-12
        neg = g < -1e-12
        # compute bounds
        if pos.any():
            max_pos = np.min((1 - h[pos]) / g[pos])
        else:
            max_pos = np.inf
        if neg.any():
            max_neg = np.min((-h[neg]) / g[neg])  # note g negative so -h/g positive
        else:
            max_neg = np.inf
        alpha_max = min(max_pos, max_neg)
        if not np.isfinite(alpha_max) or alpha_max <= 0:
            break
        alpha = step_factor * alpha_max
        # update
        h += alpha * g
        # ensure sum preserved (some rounding errors)
        # maybe renormalize by subtracting mean diff?
        diff = (n/2.0) - h.sum()
        if abs(diff) > 1e-12:
            # distribute diff uniformly (or via random)
            h += diff / n
        # ensure clipping just in case small numeric overshoot
        h = np.clip(h, 0.0, 1.0)
        # decay step factor
        step_factor = max(step_factor * decay, 0.01)
    return h

def simulated_annealing(h0, rng, deadline):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    n = h.size
    # parameters
    T0 = max(0.01 * cur_c, 1e-4)
    it = 0
    while time.time() < deadline and it < 15000:
        # generate perturb via moving epsilon from i to j
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2 * max_eps * 0.1  # small step
        # ensure within bounds
        eps = max(-max_eps, min(eps, max_eps))
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / (T0 * (0.999 ** it) + 1e-12)):
            h = new_h
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        it += 1
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ---------- global variables for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Sum of h must equal target."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ h_i (1 - h_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 = max_k ∫ h(x)(1-h(x+k))dx."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def slsqp_optimize(h0, beta, rng, max_iter=400):
    """SLSQP with soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # tiny sum correction
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

    """Ascent on the minimal product sum (equivalently descent on max overlap)."""
    h = h.copy()

    step_factor = 0.9

    while time.time() < deadline:
        # product sums for all shifts (non‑negative lags)
        corr = np.correlate(h, h, mode="full")
        prod = corr[n - 1 :]                     # length n, shift k = 0..n-1
        k_min = int(np.argmin(prod))

        # gradient of the worst product sum
        if k_min == 0:
            g = 2.0 * h
        else:
            g = np.zeros_like(h)
            upper = n - k_min
            g[:upper] += h[k_min:]
            g[k_min:] += h[:upper]

        # project to zero‑sum subspace
        g -= np.mean(g)

        # feasible step size respecting bounds
        pos = g > 1e-12
        neg = g < -1e-12
        if pos.any():
            max_pos = np.min((1.0 - h[pos]) / g[pos])
        else:
            max_pos = np.inf
        if neg.any():
            max_neg = np.min((-h[neg]) / g[neg])
        else:
            max_neg = np.inf
        alpha_max = min(max_pos, max_neg)
        if not np.isfinite(alpha_max) or alpha_max <= 0:
            break
        alpha = step_factor * alpha_max
        h += alpha * g
        # keep numerical sum exact
        diff = (n / 2.0) - h.sum()
        if abs(diff) > 1e-12:
            h += diff / n
        # clip (should be unnecessary but safe)
        h = np.clip(h, 0.0, 1.0)

        step_factor = max(step_factor * decay, 0.01)
    return h


def simulated_annealing(h0, rng, deadline):
    """Simple SA that respects the sum constraint."""
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    n = h.size
    T0 = max(0.01 * cur_c, 1e-4)
    it = 0
    while time.time() < deadline and it < 15000:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            it += 1
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.1
        eps = max(-max_eps, min(eps, max_eps))
        cand = h.copy()
        cand[i] -= eps
        cand[j] += eps
        cand_c = compute_c5(cand)
        if cand_c < cur_c:
            h, cur_c = cand, cand_c
            if cur_c < best_c:
                best_h, best_c = h.copy(), cur_c
        else:
            T = T0 * (0.999 ** it) + 1e-12
            if rng.random() < np.exp(-(cand_c - cur_c) / T):
                h, cur_c = cand, cand_c
                if cur_c < best_c:
                    best_h, best_c = h.copy(), cur_c
        it += 1
    return best_h, best_c


def enforce_sum(h, target, rng):
    """Redistribute small mass to hit the exact target."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # ---------- choose resolution ----------
    n = 300                     # highest sensible size (<1k)
    tgt = n / 2.0

    # ---------- initial vector ----------
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
        xt = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(xt, xs, parent)
    else:
        # smooth random start
        h = rng.random(n)
        kernel = np.array([0.25, 0.5, 0.25])
        h = np.convolve(h, kernel, mode="same")
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, tgt, rng)

    # ---------- projected gradient ascent ----------
    pg_deadline = start + budget_s * 0.55
    h = projected_gradient_ascent(h, rng, pg_deadline)

    # ---------- soft‑max SLSQP refinement ----------
    best_h, best_c = h.copy(), compute_c5(h)
    beta_seq = [5, 20, 80, 200, 500, 1000, 2000, 5000]
    for beta in beta_seq:
        if time.time() > start + budget_s * 0.85:
            break
        h_opt = slsqp_optimize(best_h, beta, rng, max_iter=500)
        c = compute_c5(h_opt)
        if c < best_c:
            best_h, best_c = h_opt.copy(), c

    # ---------- final annealing ----------
    anneal_deadline = start + budget_s * 0.97
    if time.time() < anneal_deadline:
        h_sa, c_sa = simulated_annealing(best_h, rng, anneal_deadline)
        if c_sa < best_c:
            best_h, best_c = h_sa.copy(), c_sa

    # ---------- safety pass ----------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, tgt, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
