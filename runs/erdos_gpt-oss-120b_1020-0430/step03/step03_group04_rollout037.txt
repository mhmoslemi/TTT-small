def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    # define final half size
    final_n = 300   # choose even
    final_m = final_n // 2

    best_c = np.inf
    best_h_half = None

    # if parent exists
    if 'initial_h_values' in globals():
        parent_full = np.array(initial_h_values, dtype=float)
        n_parent = parent_full.size
        if n_parent % 2 != 0:
            # if odd, ignore last point for symmetry
            parent_full = parent_full[:n_parent-1]
            n_parent = parent_full.size

        h_half_parent = np.empty(m_parent)
        for i in range(m_parent):

            x_tgt = np.linspace(0.0, 1.0, final_m, endpoint=False)

def symmetrize_half(h_half):
    """Construct full step function from its half (mirrored complement)."""
    m = h_half.size
    n = 2 * m
    full = np.empty(n, dtype=float)
    full[:m] = h_half
    full[m:] = 1.0 - h_half[::-1]
    return full


def compute_c5_half(h_half):
    """Exact C5 for a half–vector (full function is symmetric)."""
    full = symmetrize_half(h_half)
    n = full.size
    corr = np.correlate(full, 1.0 - full, mode="full")
    return float(np.max(corr) * (2.0 / n))


def random_feasible_half(m, rng):
    """Uniform random values in [0,1] for the half‑vector."""
    return rng.random(m)


def random_binary_half(m, rng):
    """Random 0/1 half‑vector."""
    return rng.integers(0, 2, size=m).astype(float)


def hill_climb_half(h_start, rng, max_iters=2000, start_step=0.6, decay=0.9):
    """Simple hill‑climb on the half‑vector (preserves constraints automatically)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5_half(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        i = rng.integers(0, h.size)
        max_eps = min(h[i], 1.0 - h[i])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * step
        new_h = h.copy()
        new_h[i] = np.clip(new_h[i] + eps, 0.0, 1.0)
        c = compute_c5_half(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * decay, 1e-4)
    return best_h, best_c

    """Smooth approximation of max_k overlap for half‑vector x."""
    global CURRENT_BETA
    full = symmetrize_half(x)
    corr = np.correlate(full, 1.0 - full, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def simulated_annealing_half(h0, rng, deadline):
    """Simple SA on half‑vector respecting bounds (sum is automatic)."""
    h = h0.copy()
    cur_c = compute_c5_half(h)
    best_h = h.copy()

    T0 = max(0.01 * cur_c, 1e-4)
