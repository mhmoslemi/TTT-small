def smooth_gradient(h, beta):

    a = -beta * corr

def project_feasible(y, target):
    x = np.clip(y, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # If diff > 0: need to add diff. Compute capacity = sum(1 - x)
    cap = (1 - x).sum()
    if diff > 0 and cap > 0:
        # distribute proportionally
        x += diff * (1 - x) / cap
    elif diff < 0:
        # need to subtract
        cap = x.sum()
        if cap > 0:
            x += diff * x / cap
    # clip again
    x = np.clip(x, 0.0, 1.0)
    # Recheck diff
    diff = target - x.sum()
    # If still off, fallback to random adjustment loop
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, len(x))
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
            if abs(diff) < 1e-12:
                break
    return x
```

def adam_optimize(h0, rng, beta_schedule, iters_per_beta, lr=0.1):
    h = h0.copy()
    n = h.size
    target = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_h = h.copy()
    best_c = compute_c5(h)
    
    for beta in beta_schedule:
        for _ in range(iters_per_beta):
            t += 1
            # compute gradient
            loss, grad = smooth_loss_and_grad(h, beta)
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            step = lr / (np.sqrt(v_hat) + eps)
            h = h - step * m_hat
            # projection
            h = project_feasible(h, target, rng)
            # compute c5 occasionally
            if t % 10 == 0:
                c = compute_c5(h)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
    # final best
    return best_h, best_c
```

def smooth_loss_and_grad(h, beta):
    # correlation h with itself
    corr = np.convolve(h, h, mode='full')
    a = -beta * corr
    max_a = np.max(a)
    exp_shift = np.exp(a - max_a)
    sum_exp = np.sum(exp_shift)
    # weights
    w = exp_shift / sum_exp
    # gradient via convolution
    start = len(h) - 1
    conv = np.convolve(w, h, mode='full')
    conv_rev = np.convolve(w[::-1], h, mode='full')
    grad = - (conv[start:start+len(h)] + conv_rev[start:start+len(h)])
    loss = (max_a + np.log(sum_exp)) / beta
    return loss, grad
```

def adam_optimize(h0, rng, beta_schedule, iters_per_beta, lr):
    h = h0.copy()
    n = h.shape[0]
    target = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_h = h.copy()
    best_c = compute_c5(h)
    for beta in beta_schedule:
        for _ in range(iters_per_beta):
            t += 1
            loss, grad = smooth_loss_and_grad(h, beta)
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            step = lr / (np.sqrt(v_hat) + eps)
            h = h - step * m_hat
            h = project_feasible(h, target, rng)
            # occasional check
            if t % 10 == 0:
                c = compute_c5(h)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
    # final check
    c = compute_c5(h)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    return best_h, best_c

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb_local(h0, rng, max_iters=1500, step=0.4):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
            h = new_h
        # optionally reduce step
        if it % (max_iters // 5 + 1) == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c

import time
import numpy as np

# ------------------- basic utilities ------------------- #
def compute_c5(h):
    """Exact discrete C5 for step function h on [0,2] with uniform spacing."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def project_feasible(x, target, rng):
    """Clip to [0,1] and adjust to satisfy sum == target."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    if diff > 0:
        cap = (1.0 - x).sum()
        if cap > 1e-12:
            x = x + diff * (1.0 - x) / cap
    else:
        total = x.sum()
        if total > 1e-12:
            x = x + diff * x / total
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, len(x))
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
            if abs(diff) < 1e-12:
                break
    return x


def random_feasible_vector(n, rng):
    """Uniform random vector in [0,1] with exact sum n/2."""
    h = rng.random(n)
    target = n / 2.0
    h = h / h.sum() * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h


def upsample_vector(v, new_n):
    """Linear interpolation of vector v defined on [0,2] to length new_n."""
    src_x = np.linspace(0.0, 2.0, v.size, endpoint=False)
    tgt_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(tgt_x, src_x, v)


# ------------------- smooth surrogate ------------------- #
def smooth_loss_and_grad(h, beta):
    """
    Softmax surrogate for max_k (n/2 - Σ h_i h_{i+k}) .
    Returns (loss, gradient) wrt h.
    """
    # correlation of h with itself (full length 2n-1)
    corr = np.convolve(h, h, mode="full")
    a = -beta * corr
    max_a = np.max(a)
    exp_shift = np.exp(a - max_a)
    sum_exp = np.sum(exp_shift)
    w = exp_shift / sum_exp                     # softmax weights
    start = len(h) - 1                         # index where zero lag aligns
    conv = np.convolve(w, h, mode="full")
    conv_rev = np.convolve(w[::-1], h, mode="full")
    grad = - (conv[start:start + len(h)] + conv_rev[start:start + len(h)])
    loss = (max_a + np.log(sum_exp)) / beta
    return loss, grad


# ------------------- Adam optimiser ------------------- #
def adam_optimize(h0, rng, beta_schedule, iters_per_beta, lr=0.05):
    """
    Adam optimisation of the smooth surrogate.
    Returns best vector found and its exact C5.
    """
    h = h0.copy()
    n = h.shape[0]
    target = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_h = h.copy()
    best_c = compute_c5(h)
    best_loss = np.inf
    no_improve = 0
    max_no_improve = 5 * len(beta_schedule)

    for beta in beta_schedule:
        for _ in range(iters_per_beta):
            t += 1
            loss, grad = smooth_loss_and_grad(h, beta)
            # Adam updates
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            step = lr / (np.sqrt(v_hat) + eps)
            h = h - step * m_hat
            h = project_feasible(h, target, rng)

            # progress tracking
            if loss < best_loss - 1e-12:
                best_loss = loss
                no_improve = 0
            else:
                no_improve += 1
            if no_improve >= max_no_improve:
                break

            if t % 10 == 0:
                c = compute_c5(h)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
        if no_improve >= max_no_improve:
            break

    # final check
    c = compute_c5(h)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    return best_h, best_c


# ------------------- local perturbation ------------------- #
def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_local(h0, rng, max_iters=2000, step=0.3):
    """Simple hill‑climb that only accepts improvements."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


# ------------------- main driver ------------------- #
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # optional parent construction
    parent = None
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)

    candidate_ns = [96, 120, 144, 180, 240]

    beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1000.0]
    iters_per_beta = 40  # same for all n; cheap enough

    for n in candidate_ns:
        if time.time() - start_time > budget_s * 0.95:
            break
        target = n / 2.0
        start_vectors = []

        # upsample / use parent
        if parent is not None:
            h_up = upsample_vector(parent, n) if parent.size != n else parent.copy()
            start_vectors.append(h_up)

        # random continuous feasible
        start_vectors.append(random_feasible_vector(n, rng))

        # random binary feasible
        h_bin = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h_bin[idx] = 1.0
        start_vectors.append(h_bin)

        # low‑frequency sinusoid
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        freq = rng.integers(1, 4)
        h_sin = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / 2.0)
        start_vectors.append(project_feasible(h_sin, target, rng))

        # smoothed random noise
        noise = rng.random(n)
        kernel = np.array([0.25, 0.5, 0.25])
        smooth = np.convolve(noise, kernel, mode="same")
        start_vectors.append(project_feasible(smooth, target, rng))

        # slight perturbation of current global best (if same size)
        if best_n == n and best_h is not None:
            start_vectors.append(best_h + 0.02 * rng.standard_normal(n))

        for base_h in start_vectors:
            if time.time() - start_time > budget_s * 0.98:
                break
            h = project_feasible(base_h, target, rng)

            # quick evaluation
            c0 = compute_c5(h)
            if c0 < best_c:
                best_c = c0
                best_h = h.copy()
                best_n = n

            # Adam optimisation on smooth surrogate
            h_opt, c_opt = adam_optimize(h, rng, beta_schedule, iters_per_beta, lr=0.05)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt.copy()
                best_n = n

    # final local refinement on the best vector found
    if best_h is not None:
        h_refined, c_refined = hill_climb_local(best_h, rng, max_iters=2000, step=0.3)
        if c_refined < best_c:
            best_c = c_refined
            best_h = h_refined
            best_n = best_h.shape[0]

    # optional up‑sampling refinement (if space permits)
    if (
        best_h is not None
        and best_n * 2 <= 300
        and time.time() - start_time < budget_s * 0.97
    ):
        n_up = best_n * 2
        h_up = upsample_vector(best_h, n_up)
        h_up = project_feasible(h_up, n_up / 2.0, rng)
        h_up_opt, c_up_opt = adam_optimize(
            h_up, rng, beta_schedule[:5], iters_per_beta=30, lr=0.05
        )
        h_up_opt, c_up_opt = hill_climb_local(h_up_opt, rng, max_iters=1200, step=0.3)
        if c_up_opt < best_c:
            best_c = c_up_opt
            best_h = h_up_opt
            best_n = n_up

    # safety net
    if best_h is None:
        best_n = 100
        best_h = np.full(best_n, 0.5)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, best_n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
