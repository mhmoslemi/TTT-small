def subgrad_descent(h, max_iters, init_alpha):
    best_h = h.copy()
    best_c = compute_c5(h)
    alpha = init_alpha
    for it in range(1, max_iters+1):
        # compute conv
        conv = correlate_full(h)
        # choose k with max (argmax)
        idx = np.argmax(conv)
        shift = idx - (h.size - 1)  # because conv length = 2n-1, center at n-1 corresponds to shift 0
        # gradient
        n = h.size
        if shift == 0:
            g = 1 - 2*h
        elif shift > 0:
            g = np.ones(n)
            g[:n - shift] -= h[shift:]
            g[shift:] -= h[:n - shift]
        else: # shift < 0
            k = -shift
            g = np.ones(n)
            g[k:] -= h[:n - k]
            g[:n - k] -= h[k:]
        # step size schedule
        alpha = init_alpha / np.sqrt(it)  # decreasing
        # gradient step (descend)
        y = h - alpha * g
        # project onto [0,1] intersect sum = target:
        h = project_onto_capped_simplex(y, target=n/2)
        # evaluate
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
            # optionally restart step size? maybe reset alpha.
    return best_h, best_c

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    # Determine direction: move from i to j
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        # try opposite direction
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0.0:
            return h.copy()
        # swap
        i, j = j, i
    eps = rng.random() * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def subgrad_descent(h_start, max_iters, init_alpha):
    h = h_start.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(1, max_iters+1):
        conv = correlate_full(h)
        idx = int(np.argmax(conv))
        shift = idx - (n - 1)
        # compute gradient
        if shift == 0:
            g = 1.0 - 2.0 * h
        elif shift > 0:
            k = shift
            g = np.ones(n)
            # subtract h[i+k] for i in [0, n-k-1]
            g[:n - k] -= h[k:]
            # subtract h[i-k] for i in [k, n-1]
            g[k:] -= h[:n - k]
        else:
            k = -shift
            g = np.ones(n)
            # shift negative: treat as shift = -k
            # subtract h[i+k] = h[i - k] for i >= k (i-k >=0)
            g[k:] -= h[:n - k]
            # subtract h[i-k] = h[i + k] for i < n - k
            g[:n - k] -= h[k:]
        # step size
        alpha = init_alpha / np.sqrt(it)
        y = h - alpha * g
        # project onto [0,1] and sum = target
        h = project_onto_capped_simplex(y, target)
        # evaluate
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def correlate_full(h):
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv

def compute_c5(h):
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)

import time
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Core utilities
# --------------------------------------------------------------
def correlate_full(h: np.ndarray) -> np.ndarray:
    """Linear correlation of h with (1‑h) using FFT (zero‑pad)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())  # next power of two
    fh = np.fft.rfft(h, L)
    rev = (1.0 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h: np.ndarray) -> float:
    """Discrete C₅ for step function h (dx = 2 / n)."""
    return correlate_full(h).max() * (2.0 / h.size)


# --------------------------------------------------------------
# Projection onto the capped simplex (0 ≤ x ≤ 1, sum = target)
# --------------------------------------------------------------
def project_onto_capped_simplex(y: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of y onto {0≤x≤1, Σx=target}."""
    lo, hi = -np.max(y) - 1.0, np.max(y) + 1.0
    for _ in range(35):
        lam = (lo + hi) * 0.5
        x = np.clip(y - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(y - lam, 0.0, 1.0)


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random point in [0,1]^n with sum = n/2."""
    y = rng.random(n)
    return project_onto_capped_simplex(y, n / 2.0)


# --------------------------------------------------------------
# Simple perturbation (transfer ε from i to j)
# --------------------------------------------------------------
def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    # try moving from i to j
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        # opposite direction
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0.0:
            return h.copy()
        i, j = j, i
    eps = rng.random() * max_eps * scale
    out = h.copy()
    out[i] -= eps
    out[j] += eps
    return out


# --------------------------------------------------------------
# Hill‑climbing refinement
# --------------------------------------------------------------
def hill_climb(h_start: np.ndarray, rng: np.random.Generator, deadline: float, start_step: float = 0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, int((deadline - time.time()) // 0.05))
    it = 0
    while time.time() < deadline:
        it += 1
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h, h = c, new_h.copy(), new_h
        if (it % decay_interval) == 0:
            step = max(step * 0.7, 1e-5)
            decay_interval = max(1, int((deadline - time.time()) // 0.05))
    return best_h, best_c


# --------------------------------------------------------------
# Subgradient descent on the max‑overlap objective
# --------------------------------------------------------------
def subgrad_descent(h_start: np.ndarray, init_alpha: float, deadline: float):
    h = h_start.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    it = 0
    while time.time() < deadline:
        it += 1
        conv = correlate_full(h)
        idx = int(np.argmax(conv))
        shift = idx - (n - 1)
        k = abs(shift)
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            g = np.ones(n)
            g[: n - k] -= h[k:]
            g[k:] -= h[: n - k]
        alpha = init_alpha / np.sqrt(it)
        y = h - alpha * g
        h = project_onto_capped_simplex(y, target)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
    return best_h, best_c


# --------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and SLSQP refinement
# --------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    conv = correlate_full(x)
    bc = CURRENT_BETA * conv
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, max_iter: int = 300) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # final projection onto the capped simplex to repair any drift
    return project_onto_capped_simplex(x, CURRENT_TARGET)

    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------

    h = 1.0 - xs / 2.0

        if time.time() < stage_up_deadline:

            if c_up < global_best_c:

                    h_up = opt_h
            except Exception:

    # 4. Final clean‑up
    # ----------------------------------------------------------
    global_best_h = np.clip(global_best_h, 0.0, 1.0)
    global_best_h = project_onto_capped_simplex(global_best_h, global_best_h.size / 2.0)
    final_c = compute_c5(global_best_h)
