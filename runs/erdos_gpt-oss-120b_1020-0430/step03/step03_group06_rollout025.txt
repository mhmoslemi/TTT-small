def compute_c5(h):
    """Exact discrete C5 for step function h on [0,2] (dx=2/len(h))."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)


def enforce_sum(h, target, rng):
    """Adjust h so that Σh = target while 0≤h≤1 (simple iterative repair)."""
    diff = target - h.sum()
    n = h.size
    # basic redistribution
    for _ in range(3000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-12:
            continue
        h[i] += add
        diff = target - h.sum()
    # final tiny correction if needed
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(1000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            if abs(add) < 1e-12:
                continue
            h[i] += add
            diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Return a random vector in [0,1]^n with sum=n/2."""
    v = rng.exponential(1.0, size=n)
    v = v / v.sum() * (n / 2.0)
    h = np.clip(v, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount between two coordinates (preserves sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    while j == i:
        j = rng.integers(0, n)
    max_pos = min(h[i], 1.0 - h[j])
    max_neg = min(h[j], 1.0 - h[i])
    if max_pos + max_neg == 0:
        return h.copy()
    eps = (rng.random() * (max_pos + max_neg)) - max_neg
    eps *= scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return np.clip(nh, 0.0, 1.0)


def binary_swap_move(h, rng):
    """Swap a 1 and a 0 to keep the sum unchanged (h binary)."""
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return h.copy()
    i = rng.choice(ones)
    j = rng.choice(zeros)
    nh = h.copy()
    nh[i] = 0.0
    nh[j] = 1.0
    return nh


def binary_hill_climb(h, rng, max_iters=2000):
    """Simple hill‑climb on binary vectors (swap moves only)."""
    best = h.copy()
    best_c = compute_c5(best)
    for _ in range(max_iters):
        nh = binary_swap_move(best, rng)
        c = compute_c5(nh)
        if c < best_c:
            best, best_c = nh, c
    return best, best_c


# --------------------------------------------------------------
# SLSQP on smooth log‑sum‑exp surrogate
# --------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(_softmax_obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons, options={"maxiter": max_iter,
                                             "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # tiny exact‑sum repair
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            if abs(add) < 1e-12:
                continue
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

# main entry point
# --------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    # ----- initialise ----- #
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        target = n / 2.0
        h0 = enforce_sum(np.clip(h0, 0.0, 1.0), target, rng)
    else:
        n = 180
        target = n / 2.0
        h0 = np.full(n, 0.5)
        h0 = enforce_sum(h0, target, rng)

    # ----- population ----- #
    pop_sz = 30
    population = [(h0, compute_c5(h0))]
    while len(population) < pop_sz:
        if rng.random() < 0.5:
            # binary random
            idx = rng.choice(n, size=int(target), replace=False)
            h = np.zeros(n)
            h[idx] = 1.0
        else:
            h = random_feasible_vector(n, rng)
        population.append((h, compute_c5(h)))
    population.sort(key=lambda x: x[1])
    best_h, best_c = population[0][0].copy(), population[0][1]

    # ----- evolutionary loop ----- #
    while time.time() - t0 < budget_s * 0.78:
        # select two parents from top half
        p1 = population[rng.integers(0, pop_sz // 2)][0]
        p2 = population[rng.integers(0, pop_sz // 2)][0]
        child = (p1 + p2) * 0.5
        child = np.clip(child, 0.0, 1.0)
        child = enforce_sum(child, target, rng)

        if rng.random() < 0.3:
            child = perturb_h(child, rng, scale=0.1)
            child = enforce_sum(child, target, rng)

        c_child = compute_c5(child)
        if c_child < population[-1][1]:
            population[-1] = (child, c_child)
            population.sort(key=lambda x: x[1])
            if c_child < best_c:
                best_h, best_c = child.copy(), c_child

        # occasional binary refinement of current best
        if rng.random() < 0.04:
            # binarise best_h by taking largest entries
            bin_h = np.zeros_like(best_h)
            order = np.argsort(-best_h)
            bin_h[order[:int(target)]] = 1.0
            bin_h, bin_c = binary_hill_climb(bin_h, rng, max_iters=2500)
            if bin_c < best_c:
                best_h, best_c = bin_h.copy(), bin_c
                population[0] = (best_h, best_c)
                population.sort(key=lambda x: x[1])

    # ----- intensive local search on best ----- #
    # fine hill‑climb (swap + small transfers)
    for _ in range(3):
        # swap‑only hill‑climb
        best_h, best_c = binary_hill_climb(best_h, rng, max_iters=3000)
        # fractional perturbation hill‑climb
        h_tmp, c_tmp = best_h.copy(), best_c
        for _ in range(2000):
            nh = perturb_h(h_tmp, rng, scale=0.15)
            nh = enforce_sum(nh, target, rng)
            c = compute_c5(nh)
            if c < c_tmp:
                h_tmp, c_tmp = nh, c
        if c_tmp < best_c:
            best_h, best_c = h_tmp, c_tmp

    # ----- SLSQP refinement ----- #
    betas = [20.0, 80.0, 320.0, 1280.0, 5120.0]
    for beta in betas:
        if time.time() - t0 > budget_s * 0.94:
            break
        try:
            cand = slsqp_optimize(best_h, beta, max_iter=500)
        except Exception:
            continue
        c = compute_c5(cand)
        if c < best_c:
            best_h, best_c = cand.copy(), c

    # ----- up‑sampling & final polish ----- #
    if time.time() - t0 < budget_s * 0.98 and n * 2 <= 360:
        up_h = np.repeat(best_h, 2)
        up_n = up_h.size
        up_target = up_n / 2.0
        up_h = enforce_sum(up_h, up_target, rng)

        up_h, up_c = binary_hill_climb(up_h, rng, max_iters=4000)
        for beta in betas:
            if time.time() - t0 > budget_s:
                break
            try:
                up_h = slsqp_optimize(up_h, beta, max_iter=400)
            except Exception:
                pass
            up_c = compute_c5(up_h)
            if up_c < best_c:
                best_h, best_c, n = up_h.copy(), up_c, up_n

    # ----- final clean‑up ----- #
    best_h = enforce_sum(np.clip(best_h, 0.0, 1.0), n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
