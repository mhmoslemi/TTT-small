def binary_hill_climb(b, rng, max_iters=2000):
    n = len(b)
    best = b.copy()
    # compute pair counts
    pair = np.zeros(2*n-1,dtype=int)
    for s in range(-(n-1), n):
        if s >= 0:
            pair[s + n -1] = np.dot(b[:n-s], b[s:])
        else:
            t = -s
            pair[s + n -1] = np.dot(b[t:], b[:n-t])
    best_min = pair.min()
    for it in range(max_iters):
        # sample random indices
        i = rng.choice(np.where(b==1)[0])
        j = rng.choice(np.where(b==0)[0])
        # swap
        b[i]=0; b[j]=1
        # recompute min pairs
        # can compute directly
        pair_new = np.zeros_like(pair)
        for s in range(-(n-1), n):
            if s >= 0:
                pair_new[s + n -1] = np.dot(b[:n-s], b[s:])
            else:
                t = -s
                pair_new[s + n -1] = np.dot(b[t:], b[:n-t])
        cur_min = pair_new.min()
        if cur_min > best_min:
            best_min = cur_min
            best = b.copy()
        else:
            # revert
            b[i]=1; b[j]=0
    # return best binary vector
    return best

def worst_shift_gradient(h):
    n = len(h)
    corr = np.correlate(h, 1-h, mode='full')
    idx = np.argmax(corr) - (n-1)
    k = idx
    g = np.zeros_like(h)
    if k >= 0:
        # i ranges 0..n-1-k
        i0 = np.arange(0, n - k)
        g[i0] += (1 - h[i0 + k])
        if k > 0:
            i1 = np.arange(k, n)
            g[i1] -= h[i1 - k]
    else:
        t = -k
        i0 = np.arange(t, n)
        g[i0] += (1 - h[i0 - t])
        i1 = np.arange(0, n - t)
        g[i1] -= h[i1 + t]
    return g

import time
import math
import numpy as np
from scipy.optimize import minimize

# ---------- global helpers for soft‑max surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h):
    """Exact discrete C5 for step function h on [0,2] with uniform spacing."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=1500, start_step=0.5):
    """Simple hill‑climb that accepts only improvements."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay = max(1, max_iters // 8)
    for it in range(max_iters):
        cand = perturb_h(h, rng, scale=step)
        c = compute_c5(cand)
        if c < best_c:
            best_c, best_h, h = c, cand.copy(), cand
        if (it + 1) % decay == 0:
            step = max(step * 0.85, 1e-4)
    return best_h, best_c


def worst_shift_gradient(h):
    """Gradient of the worst shift (the one that attains the max overlap)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)          # shift with maximal overlap
    k = idx
    g = np.zeros_like(h)
    if k >= 0:
        if n - k > 0:
            i0 = slice(0, n - k)
            g[i0] += 1.0 - h[i0.start + k : i0.stop + k]
        if k > 0:
            i1 = slice(k, n)
            g[i1] -= h[i1.start - k : i1.stop - k]
    else:
        t = -k
        i0 = slice(t, n)
        g[i0] += 1.0 - h[i0.start - t : i0.stop - t]
        i1 = slice(0, n - t)
        g[i1] -= h[i1.start + t : i1.stop + t]
    return g


def binary_hill_climb(b, rng, max_iters=800):
    """Hill‑climb on binary vectors using the min‑pair (max‑overlap) objective."""
    n = b.size
    target = n // 2
    best = b.copy()
    best_corr = np.correlate(b, b, mode="full")
    best_min = best_corr.min()
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]
    for _ in range(max_iters):
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        b[i], b[j] = 0, 1
        corr = np.correlate(b, b, mode="full")
        cur_min = corr.min()
        if cur_min > best_min:
            best_min, best = cur_min, b.copy()
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
        else:
            b[i], b[j] = 1, 0          # revert
    return best


# ---------- soft‑max surrogate with gradient (global vars) ----------
def softmax_obj_and_grad(x):
    """Return f, grad of (1/β)log‑sum‑exp(β·corr) with quadratic penalty for sum."""
    n = x.size
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    beta = CURRENT_BETA
    bc = beta * corr
    m = np.max(bc)
    exp_bc = np.exp(bc - m)
    sum_exp = np.sum(exp_bc)
    f = (m + np.log(sum_exp)) / beta
    w = exp_bc / sum_exp                     # soft‑max weights
    grad = np.zeros_like(x)

    # contributions from each shift
    for s in range(-(n - 1), n):
        ws = w[s + n - 1]
        if ws == 0.0:
            continue
        if s >= 0:
            if n - s > 0:
                i0 = slice(0, n - s)
                grad[i0] += ws * (1.0 - x[i0.start + s : i0.stop + s])
            if s > 0:
                i1 = slice(s, n)
                grad[i1] -= ws * x[i1.start - s : i1.stop - s]
        else:
            t = -s
            i0 = slice(t, n)
            grad[i0] += ws * (1.0 - x[i0.start - t : i0.stop - t])
            i1 = slice(0, n - t)
            grad[i1] -= ws * x[i1.start + t : i1.stop + t]

    # quadratic penalty for sum constraint
    diff = x.sum() - CURRENT_TARGET
    f += 1e6 * diff * diff
    grad += 2e6 * diff * np.ones_like(x)
    return f, grad


def obj_lbfgsb(x):
    f, _ = softmax_obj_and_grad(x)
    return f


def grad_lbfgsb(x):
    _, g = softmax_obj_and_grad(x)
    return g


def lbfgsb_optimize(h0, beta, max_iter=80):
    """L‑BFGS‑B on the soft‑max surrogate with penalty for the linear constraint."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        obj_lbfgsb,
        h0,
        method="L-BFGS-B",
        jac=grad_lbfgsb,
        bounds=bounds,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # final exact sum correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x
