def gradient_step(h, rng, step):
    n = h.shape[0]
    # find worst shift
    corr = np.correlate(h, 1-h, mode='full')
    # compute f_k = corr*dx; want max. Use argmax index.
    idx = np.argmax(corr) - (n-1)
    k = abs(idx)  # shift magnitude (non-negative)
    # compute gradient
    g = np.zeros_like(h)
    if k == 0:
        # gradient is 1 - 2h as derived (since corr = sum h_i - sum h_i^2)
        g[:] = 1 - 2 * h
    else:
        # compute using loops vectorized
        # g[i] = 1 - (h[i+k] + h[i-k]) with zero out-of-range
        # using slicing
        upper = n - k
        # For i in 0..upper-1: h[i+k] exists
        g[:upper] += 1 - h[k:]
        # For i in k..n-1: h[i-k] exists
        g[k:] -= h[:upper]
    # gradient descent: h_new = h - step * g
    # after step, enforce bounds [0,1] and sum constraint
    h_new = h - step * g
    # clip
    h_new = np.clip(h_new, 0., 1.)
    # adjust sum
    diff = (n/2) - h_new.sum()
    if abs(diff) > 1e-12:
        # redistribute diff to random indices that are not at bounds
        for _ in range(1000):
            i = rng.integers(0,n)
            if diff > 0:
                add = min(diff, 1.-h_new[i])
            else:
                add = max(diff, -h_new[i])
            if add == 0: continue
            h_new[i] += add
            diff = (n/2)-h_new.sum()
            if abs(diff) < 1e-12:
                break
    return h_new

def subgradient_descent(h, rng, max_iters=200, step0=0.3):
    n = h.shape[0]
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    for it in range(max_iters):
        # find worst shift
        corr = np.correlate(cur_h, 1-cur_h, mode='full')
        idx = np.argmax(corr) - (n-1)
        k = abs(idx)
        # compute subgradient (for active shift)
        g = np.zeros_like(cur_h)
        if k == 0:
            g[:] = 1 - 2*cur_h
        else:
            upper = n - k
            g[:upper] += 1 - cur_h[k:]
            g[k:] -= cur_h[:upper]
        # step size schedule: 
        step = step0 / np.sqrt(it+1)
        # take step
        new_h = cur_h - step * g
        # project onto [0,1] and sum constraint
        new_h = np.clip(new_h, 0., 1.)
        diff = (n/2) - new_h.sum()
        if abs(diff) > 1e-12:
            # redistribute diff to random entries
            for _ in range(2000):
                i = rng.integers(0,n)
                if diff > 0:
                    add = min(diff, 1.-new_h[i])
                else:
                    add = max(diff, -new_h[i])
                if add == 0: continue
                new_h[i] += add
                diff = (n/2)-new_h.sum()
                if abs(diff) < 1e-12:
                    break
        # check improvement:
        new_c = compute_c5(new_h)
        if new_c < cur_c:
            cur_h, cur_c = new_h, new_c
        # optional: always accept new_h? maybe.
    return cur_h

import time
import math
import numpy as np
from scipy.optimize import minimize

# ---------- core utilities ---------- #
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for step function h on [0,2] with uniform spacing."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small amount from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(
    h_start: np.ndarray,
    rng: np.random.Generator,
    max_iters: int = 1500,
    start_step: float = 0.6,
) -> tuple[np.ndarray, float]:
    """Deterministic hill‑climb – accept only improvements."""
    h = h_start.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        cand = perturb_h(h, rng, scale=step)
        c = compute_c5(cand)
        if c < best_c:
            best_c = c
            h = cand
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return h, best_c


def gradient_refine(
    h: np.ndarray,
    rng: np.random.Generator,
    step: float = 0.02,
    max_iter: int = 12,
) -> np.ndarray:
    """Refine by moving against gradient of the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            up = n - k
            g[:up] += 1.0 - h[k:]
            g[k:] -= h[:up]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector in [0,1] with exact sum n/2."""
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h


def subgradient_descent(
    h: np.ndarray,
    rng: np.random.Generator,
    max_iters: int = 80,
    step0: float = 0.3,
) -> np.ndarray:
    """Sub‑gradient descent on the max‑overlap objective."""
    n = h.size
    target = n / 2.0
    cur = h.copy()
    cur_c = compute_c5(cur)
    for it in range(max_iters):
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(cur)
        if k == 0:
            g[:] = 1.0 - 2.0 * cur
        else:
            up = n - k
            g[:up] += 1.0 - cur[k:]
            g[k:] -= cur[:up]
        step = step0 / math.sqrt(it + 1.0)
        nxt = cur - step * g
        nxt = np.clip(nxt, 0.0, 1.0)
        diff = target - nxt.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - nxt[i]) if diff > 0 else max(diff, -nxt[i])
                nxt[i] += add
                diff = target - nxt.sum()
                if abs(diff) < 1e-12:
                    break
        nxt_c = compute_c5(nxt)
        if nxt_c < cur_c:
            cur, cur_c = nxt, nxt_c
    return cur


# ---------- SLSQP with soft‑max surrogate ---------- #
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x: np.ndarray) -> float:
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x: np.ndarray) -> float:
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(
    h0: np.ndarray,
    beta: float,
    rng: np.random.Generator,
    max_iter: int = 400,
) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x
