    n = h0.size
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c

    for it in range(max_iter):
        # temperature schedule
        t = init_temp * ((final_temp / init_temp) ** (it / max_iter))
        # propose swap
        # choose i where h[i] == 1, j where h[j] == 0
        ones = np.flatnonzero(h > 0.5)
        zeros = np.flatnonzero(h < 0.5)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = h[j], h[i]
        new_c = compute_c5(h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < math.exp(-delta / t):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = h[j], h[i]
        # Optionally early exit if no improvement for many steps
    return best_h, best_c

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n/2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type":"eq","fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol':1e-12, 'disp':False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum exactly
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        # adjust by small amounts on random entries
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x

import time, math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper functions (same as previous implementation where useful)
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth log‑sum‑exp surrogate of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h so that Σh = target while respecting 0≤h≤1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # final tidy‑up if needed
    if abs(diff) > 1e-12:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:

                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def simulated_annealing(h0, rng, time_budget):
    """SA that swaps a 1 and a 0 while keeping the sum fixed."""
    n = h0.size
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c

    start = time.time()
    it = 0
    while time.time() - start < time_budget:
        it += 1
        # temperature schedule (geometric)
        t = T0 * ((T_end / T0) ** (it / 500000.0))

        i = rng.integers(n)
        j = rng.integers(n)
        if cur_h[i] == cur_h[j]:
            continue

        new_c = compute_c5(cur_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < math.exp(-delta / (t + 1e-12)):
            cur_c = new_c
            if new_c < best_c:

                best_h = cur_h.copy()
        else:

    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
