import time, math, numpy as np
from scipy.optimize import minimize

# ----------------------- basic utilities -----------------------
def compute_c5(h):
    """Discrete C5 for step function h (dx = 2/n)."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)

def enforce_sum(h, target, rng):
    """Adjust h (in place) so that sum(h) == target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    """Generate a random feasible h with 0≤h≤1 and sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    enforce_sum(h, n / 2.0, rng)
    return h

def perturb_h(h, rng, scale=0.2):
    """Transfer ε between two coordinates (preserves total sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

def biased_perturb_h(h, rng, scale=0.2):
    """Bias perturb away from positions contributing most to the current max overlap."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    k = np.argmax(corr)
    shift = k - (n - 1)
    i0 = max(0, -shift)
    i1 = min(n, n - shift)
    idx = np.arange(i0, i1)
    prod = h[idx] * (1 - h[idx + shift])
    if prod.size == 0:
        return perturb_h(h, rng, scale)
    thr_hi = np.quantile(prod, 0.70)
    thr_lo = np.quantile(prod, 0.30)
    high = idx[prod > thr_hi]
    low = idx[prod < thr_lo]
    if high.size == 0 or low.size == 0:
        return perturb_h(h, rng, scale)
    i = rng.choice(high)
    j = rng.choice(low)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

def targeted_move(h, rng, scale=0.2):
    """Move ε from a high‑product index to its shifted partner to directly cut the current max overlap."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    k = np.argmax(corr)
    shift = k - (n - 1)
    if shift == 0:
        return perturb_h(h, rng, scale)
    i0 = max(0, -shift)
    i1 = min(n, n - shift)
    idx = np.arange(i0, i1)
    prod = h[idx] * (1 - h[idx + shift])
    if prod.size == 0:
        return perturb_h(h, rng, scale)
    thr_hi = np.quantile(prod, 0.7)
    high = idx[prod > thr_hi]
    if high.size == 0:
        return perturb_h(h, rng, scale)
    # pick a source from high‑product set
    src = rng.choice(high)
    dst = src + shift
    # ensure destination bound and try to pick a src whose dst is not saturated
    max_eps = min(h[src], 1.0 - h[dst])
    if max_eps <= 0:
        # fallback to generic perturb
        return perturb_h(h, rng, scale)
    eps = rng.random() * max_eps * scale
    new = h.copy()
    new[src] -= eps
    new[dst] += eps
    return new

# ----------------------- SLSQP smoothing -----------------------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """Smooth approximation to the max overlap."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on the smooth max‑approximation."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {"type": "eq", "fun": sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

# ----------------------- Simulated Annealing -----------------------
def simulated_annealing(h0, rng, max_iter, init_temp, final_temp):
    """SA with a mix of targeted, biased and random moves."""
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    n = h.size
    log_ratio = math.log(final_temp / init_temp) if init_temp > 0 else 0.0
    for it in range(1, max_iter + 1):
        T = init_temp * math.exp(log_ratio * it / max_iter)
        r = rng.random()
        if r < 0.5:
            cand = targeted_move(h, rng, scale=0.3)
        elif r < 0.8:
            cand = biased_perturb_h(h, rng, scale=0.3)
        else:
            cand = perturb_h(h, rng, scale=0.3)
        cand_c = compute_c5(cand)
        diff = cand_c - cur_c
        if diff < 0 or rng.random() < math.exp(-diff / max(T, 1e-12)):
            h, cur_c = cand, cand_c
            if cand_c < best_c:
                best_h, best_c = cand.copy(), cand_c
    return best_h, best_c

def hill_climb(h, rng, max_iters=2000):
    """Deterministic descent using targeted moves."""
    cur = h.copy()
    cur_c = compute_c5(cur)
    for _ in range(max_iters):
        cand = targeted_move(cur, rng, scale=0.1)
        cand_c = compute_c5(cand)
        if cand_c + 1e-12 < cur_c:
            cur, cur_c = cand, cand_c
    return cur, cur_c

# ----------------------- main entry point -----------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # ---- initial vector ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        enforce_sum(h0, n / 2.0, rng)
    else:
        n = 200
        h0 = random_feasible_vector(n, rng)

    best_h, best_c = h0.copy(), compute_c5(h0)

    # ---- simulated annealing phase (≈85% of budget) ----
    sa_budget = budget_s * 0.85
    while time.time() - start < sa_budget:
        remaining = sa_budget - (time.time() - start)
        if remaining < 1.0:
            break
        # decide start point
        if rng.random() < 0.7:
            start_h = perturb_h(best_h, rng, scale=0.25)
        else:
            start_h = random_feasible_vector(n, rng)
        # allocate iterations roughly proportional to remaining time
        iters = int(min(80000, max(2000, remaining * 800)))
        cand_h, cand_c = simulated_annealing(start_h, rng,
                                             max_iter=iters,
                                             init_temp=0.05,
                                             final_temp=1e-8)
        if cand_c < best_c:
            best_h, best_c = cand_h, cand_c

        # occasional SLSQP refinement (light)
        for beta in (100, 400, 1600):
            if time.time() - start > sa_budget:
                break
            try:
                refined = slsqp_optimize(best_h, beta, max_iter=250)
                rc = compute_c5(refined)
                if rc < best_c:
                    best_h, best_c = refined, rc
            except Exception:
                continue

    # ---- final polishing phase (remaining budget) ----
    while time.time() - start < budget_s:
        # hill climbing
        best_h, best_c = hill_climb(best_h, rng, max_iters=1500)
        # higher‑beta SLSQP refinements
        for beta in (5000, 20000, 50000):
            if time.time() - start > budget_s:
                break
            try:
                best_h = slsqp_optimize(best_h, beta, max_iter=500)
                best_c = compute_c5(best_h)
            except Exception:
                continue
        # final tiny random shake (to escape possible plateau)
        if time.time() - start > budget_s:
            break
        trial = perturb_h(best_h, rng, scale=0.05)
        tc = compute_c5(trial)
        if tc + 1e-12 < best_c:
            best_h, best_c = trial, tc

    # ---- enforce constraints and return ----
    best_h = np.clip(best_h, 0.0, 1.0)
    enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
