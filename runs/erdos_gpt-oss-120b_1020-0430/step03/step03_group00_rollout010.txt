import time
import numpy as np
from scipy.optimize import minimize

# ---------- helper utilities ----------
def enforce_sum(h, target, rng):
    """Clip to [0,1] and adjust by random transfers until sum == target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 20000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i])
            add *= rng.random()
            h[i] += add
            diff -= add
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i])
            sub *= rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)

def perturb_random(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def perturb_targeted(h, grad, shift, rng, scale=0.2):
    pos = np.where((grad > 1e-9) & (h > 1e-12))[0]
    neg = np.where((grad < -1e-9) & (h < 1.0 - 1e-12))[0]
    if pos.size == 0 or neg.size == 0:
        return perturb_random(h, rng, scale)
    i = rng.choice(pos)
    j = rng.choice(neg)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return perturb_random(h, rng, scale)
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def compute_c5(h):
    """Exact discrete C5 for vector h."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def hill_climb_targeted(h_start, rng, max_iters=1500, start_step=0.5):
    """Greedy descent on the worst shift using the sub‑gradient."""
    h = h_start.copy()
    n = h.size
    best_h = h.copy()
    corr = np.correlate(h, 1 - h, mode="full")
    best_c = np.max(corr) * (2.0 / n)

    step = start_step
    decay_interval = max(1, max_iters // 10)

    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr)
        max_corr = corr[idx]
        c_cur = max_corr * (2.0 / n)
        k = idx - (n - 1)

        # gradient of the worst shift
        grad = np.ones(n)
        if k >= 0:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
        else:
            kk = -k
            grad[: n - kk] -= h[kk:]
            grad[kk:] -= h[: n - kk]

        new_h = perturb_targeted(h, grad, k, rng, scale=step)
        corr_new = np.correlate(new_h, 1 - new_h, mode="full")
        c_new = np.max(corr_new) * (2.0 / n)

        if c_new < best_c:
            best_c = c_new
            best_h = new_h.copy()
        if c_new < c_cur:
            h = new_h

        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)

    return best_h, best_c

# ---------- SLSQP surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp trick
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # final repair to satisfy the sum exactly
    rng = np.random.default_rng()
    x = enforce_sum(x, CURRENT_TARGET, rng)
    return x

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ---------- initial vector ----------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size                     # keep the original size (usually 360)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    h0 = enforce_sum(np.clip(h0, 0.0, 1.0), n / 2.0, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    beta_schedule = [5.0, 20.0, 80.0, 200.0, 800.0, 2000.0]

    # ---------- main optimisation loop ----------
    # We keep most of the time for low‑dim search; high‑res final polish uses the same size.
    while time.time() - start_time < budget_s * 0.86:
        # pick a start point
        if rng.random() < 0.55:
            start = perturb_random(best_h, rng, scale=0.35)
        else:
            start = random_feasible_vector(n, rng)

        # targeted hill‑climb (more iterations because cheap)
        h_loc, c_loc = hill_climb_targeted(start, rng, max_iters=2500, start_step=0.5)
        if c_loc < best_c:
            best_h, best_c = h_loc.copy(), c_loc

        # SLSQP refinement with increasing beta
        for beta in beta_schedule:
            if time.time() - start_time > budget_s * 0.95:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, max_iter=800)
            except Exception:
                continue
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_h, best_c = h_ref.copy(), c_ref

    # ---------- final micro‑search ----------
    while time.time() - start_time < budget_s * 0.99:
        new_h = perturb_random(best_h, rng, scale=0.02)
        c_new = compute_c5(new_h)
        if c_new < best_c:
            best_h, best_c = new_h.copy(), c_new

    # ---------- clean‑up ----------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
