def compute_c5(h):
    n = h.size
    m = 1 << ((2*n-1).bit_length())
    fh = np.fft.rfft(h, m)
    f1mh = np.fft.rfft(1 - h, m)
    conv = np.fft.irfft(fh * f1mh, m)[: 2*n-1]  # The linear convolution result.
    max_val = np.max(conv)
    return max_val * (2.0 / n)

def simulated_annealing(h0, rng, max_iter, init_temp, final_temp, init_scale=0.5, final_scale=0.05):

    best_h = h0.copy(); best_c = compute_c5(best_h)

    curr_h = h0.copy(); curr_c = best_c

    for i in range(max_iter):
        t = i / max_iter
        temp = init_temp * (final_temp / init_temp) ** t
        scale = init_scale * (final_scale / init_scale) ** t
        new_h = perturb_h(curr_h, rng, scale=scale)
        new_c = compute_c5(new_h)
        delta = new_c - curr_c
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            curr_h = new_h; curr_c = new_c
            if curr_c < best_c:
                best_c = curr_c; best_h = curr_h.copy()
    return best_h, best_c

def enforce_sum(h, target, rng, max_iter=5000):

        if abs(diff) < 1e-12:

        if diff > 0:
            add = min(diff, 1.0 - h[i])

        delta = new_c - cur_c
        if delta < 0 or (temp > 0 and rng.random() < np.exp(-delta / temp)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
    return best_h, best_c


def slsqp_optimize(h0, beta, max_iter=300):
    """SLSQP refinement of the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny drift correction
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x


def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] with a low Erdős C₅ bound."""
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Base dimension and initial vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    # ------------------------------------------------------------------
    # Assemble a modest pool of candidates
    # ------------------------------------------------------------------
    candidates = [h0]
    xs = (np.arange(n) + 0.5) * (2.0 / n)

    # triangular bump (integrates to 1)
    tri = np.where(xs <= 1, xs, 2 - xs)
    candidates.append(enforce_sum(tri, n / 2.0, rng))

    # cosine bump
    cosine = (1 + np.cos(np.pi * (xs - 1))) / 2.0
    candidates.append(enforce_sum(cosine, n / 2.0, rng))

    # linear ramps
    candidates.append(enforce_sum(xs / 2.0, n / 2.0, rng))
    candidates.append(enforce_sum(1 - xs / 2.0, n / 2.0, rng))

    # sinusoidal phases
    for f in (1, 2, 3):
        s = 0.5 + 0.5 * np.sin(np.pi * f * xs)
        c = 0.5 + 0.5 * np.cos(np.pi * f * xs)
        candidates.append(enforce_sum(s, n / 2.0, rng))
        candidates.append(enforce_sum(c, n / 2.0, rng))

    # random feasible vectors
    for _ in range(5):
        candidates.append(random_feasible_vector(n, rng))

    # ------------------------------------------------------------------
    # Pick the best of the pool
    # ------------------------------------------------------------------
    best_h = None
    best_c = np.inf
    for cand in candidates:
        c = compute_c5(cand)
        if c < best_c:
            best_c = c
            best_h = cand.copy()

    # ------------------------------------------------------------------
    # Main optimisation loop (hill‑climb → SA → SLSQP)
    # ------------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.6:
        # starting point for this round
        if rng.random() < 0.7:
            start_vec = perturb_h(best_h, rng, scale=0.5)
        else:
            start_vec = random_feasible_vector(n, rng)

        # hill‑climb
        h_hc, c_hc = hill_climb_multi(
            start_vec, rng, max_iters=3000, init_step=0.5, n_trials=8
        )
        if c_hc < best_c:
            best_c, best_h = c_hc, h_hc.copy()

        # simulated annealing
        h_sa, c_sa = simulated_annealing(
            best_h,
            rng,
            max_iter=8000,
            init_temp=1e-2,
            final_temp=1e-5,
            init_scale=0.5,
            final_scale=0.02,
        )
        if c_sa < best_c:
            best_c, best_h = c_sa, h_sa.copy()

        # SLSQP ladder
        for beta in (50, 200, 800, 2000, 5000):
            if time.time() - start_time > budget_s * 0.6:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, max_iter=500)
            except Exception:
                continue
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_c, best_h = c_ref, h_ref.copy()

    # ------------------------------------------------------------------
    # Optional up‑sampling if time permits
    # ------------------------------------------------------------------
    elapsed = time.time() - start_time
    if budget_s - elapsed > budget_s * 0.1:
        target_n = min(720, max(n * 2, n + 120))
        if target_n != n:
            old_x = (np.arange(n) + 0.5) * (2.0 / n)
            new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
            up_h = np.interp(new_x, old_x, best_h)
            up_h = enforce_sum(up_h, target_n / 2.0, rng)

            # modest hill‑climb + SA on the finer grid
            up_h, _ = hill_climb_multi(
                up_h, rng, max_iters=2000, init_step=0.4, n_trials=6
            )
            up_h, _ = simulated_annealing(
                up_h,
                rng,
                max_iter=5000,
                init_temp=5e-3,
                final_temp=1e-5,
                init_scale=0.4,
                final_scale=0.02,
            )
            for beta in (200, 800, 2000, 5000):
                if time.time() - start_time > budget_s:
                    break
                up_h = slsqp_optimize(up_h, beta, max_iter=400)
                c_up = compute_c5(up_h)
                if c_up < best_c:
                    best_c, best_h, n = c_up, up_h.copy(), target_n

    # ------------------------------------------------------------------
    # Final clean‑up
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
