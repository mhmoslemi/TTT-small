def annealing_binary(h0, rng, n_iters, init_T, final_T):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    n = h.size
    log_ratio = math.log(final_T / init_T) if init_T>0 else 0.0
    for it in range(n_iters):
        T = init_T * math.exp(log_ratio * it / n_iters)
        # propose swap
        i = rng.integers(0, n)
        while h[i]==0:
            i = rng.integers(0,n)
        j = rng.integers(0, n)
        while h[j]==1:
            j = rng.integers(0,n)
        new_h = h.copy()
        new_h[i]=0
        new_h[j]=1
        new_c = compute_c5(new_h)
        diff = new_c - cur_c
        if diff < 0 or rng.random() < math.exp(-diff/T):
            h, cur_c = new_h, new_c
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
    return best_h, best_c

def find_swap_pair(h, shift, rng):
    n = h.size
    start = max(0, -shift)
    end = min(n, n-shift)
    # compute product at each i within overlapping region
    idx = np.arange(start, end)
    prod = h[idx] * (1 - h[idx+shift])
    # candidates for i: h=1 and product=1 (i.e., product == 1)
    cand_i = idx[(h[idx]==1) & (h[idx+shift]==0)]
    # candidates for j: h=0 and h[j+shift]==1
    cand_j = idx[(h[idx]==0) & (h[idx+shift]==1)]
    if cand_i.size==0 or cand_j.size==0:
        return None
    i = rng.choice(cand_i)
    j = rng.choice(cand_j)
    return i, j

def annealing_binary(h0, rng, max_iter, init_T=0.05, final_T=1e-5):
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    # compute initial
    corr = np.correlate(h, 1-h, mode='full')
    max_c = corr.max()
    best_c = max_c * (2.0/n)
    cur_c = best_c
    log_ratio = math.log(final_T / init_T) if init_T>0 else 0.0
    for it in range(max_iter):
        T = init_T * math.exp(log_ratio * it / max_iter)
        use_biased = rng.random() < 0.7
        if use_biased:
            # compute shift_max
            idx = np.argmax(corr)
            shift = idx - (n - 1)
            pair = find_swap_pair(h, shift, rng)
            if pair is None:
                # fallback to random swap
                i = rng.integers(0, n)
                while h[i]==0:
                    i = rng.integers(0,n)
                j = rng.integers(0,n)
                while h[j]==1:
                    j = rng.integers(0,n)
            else:
                i,j = pair
        else:
            i = rng.integers(0, n)
            while h[i]==0:
                i = rng.integers(0,n)
            j = rng.integers(0,n)
            while h[j]==1:
                j = rng.integers(0,n)
        new_h = h.copy()
        new_h[i]=0
        new_h[j]=1
        new_corr = np.correlate(new_h, 1-new_h, mode='full')
        new_max_c = new_corr.max()

            corr = new_corr
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()

def find_swap_pair(h, shift, rng):
    n = h.size
    start = max(0, -shift)
    end = min(n, n - shift)
    idx = np.arange(start, end)
    # conditions
    src_mask = (h[idx] == 1) & (h[idx + shift] == 0)
    dst_mask = (h[idx] == 0) & (h[idx + shift] == 1)
    src = idx[src_mask]
    dst = idx[dst_mask]
    if src.size == 0 or dst.size == 0:
        return None
    i = rng.choice(src)
    j = rng.choice(dst)
    return i, j
```

import time, math, numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h):
    """Discrete C5 for step function h (dx = 2 / n)."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)

def enforce_sum(h, target, rng):
    """Adjust h in‑place so that sum(h)=target while 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_binary(n, rng):
    """Balanced 0/1 vector of length n (exactly n/2 ones)."""
    h = np.zeros(n, dtype=int)
    h[rng.choice(n, n // 2, replace=False)] = 1
    return h

# ---------- biased swap for binary SA ----------
def find_swap_pair(h, shift, rng):
    """Return indices i (1->0) and j (0->1) that improve overlap for given shift."""
    n = h.size
    start = max(0, -shift)
    end = min(n, n - shift)
    idx = np.arange(start, end)
    src_mask = (h[idx] == 1) & (h[idx + shift] == 0)
    dst_mask = (h[idx] == 0) & (h[idx + shift] == 1)
    src = idx[src_mask]
    dst = idx[dst_mask]
    if src.size == 0 or dst.size == 0:
        return None
    return int(rng.choice(src)), int(rng.choice(dst))

# ---------- simulated annealing on binary vectors ----------
def annealing_binary(start_h, rng, max_iter, init_T=0.08, final_T=1e-6):
    """SA that swaps a 1 and a 0 while preserving the sum."""
    h = start_h.copy()
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    cur_c = np.max(corr) * (2.0 / n)
    best_h, best_c = h.copy(), cur_c
    log_ratio = math.log(final_T / init_T) if init_T > 0 else 0.0
    for it in range(max_iter):
        T = init_T * math.exp(log_ratio * it / max_iter)
        use_biased = rng.random() < 0.7
        if use_biased:
            idx = np.argmax(corr)
            shift = idx - (n - 1)
            pair = find_swap_pair(h, shift, rng)
            if pair is None:
                # fall back to uniform swap
                i = rng.integers(0, n)
                while h[i] == 0:
                    i = rng.integers(0, n)
                j = rng.integers(0, n)
                while h[j] == 1:
                    j = rng.integers(0, n)
            else:
                i, j = pair
        else:
            i = rng.integers(0, n)
            while h[i] == 0:
                i = rng.integers(0, n)
            j = rng.integers(0, n)
            while h[j] == 1:
                j = rng.integers(0, n)
        new_h = h.copy()
        new_h[i] = 0
        new_h[j] = 1
        new_corr = np.correlate(new_h, 1 - new_h, mode="full")
        new_c = np.max(new_corr) * (2.0 / n)
        diff = new_c - cur_c
        if diff < 0 or rng.random() < math.exp(-diff / max(T, 1e-12)):
            h, corr, cur_c = new_h, new_corr, new_c
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
    return best_h, best_c

# ---------- deterministic local improvement ----------
def deterministic_improve(h, rng):
    """Repeatedly apply a beneficial swap for the current maximal shift."""
    n = h.size
    while True:
        corr = np.correlate(h, 1 - h, mode="full")
        max_c = np.max(corr)
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        pair = find_swap_pair(h, shift, rng)
        if pair is None:
            break
        i, j = pair
        h[i] = 0
        h[j] = 1
        new_max = np.max(np.correlate(h, 1 - h, mode="full"))
        if new_max < max_c:
            continue
        # revert if no improvement
        h[i] = 1
        h[j] = 0
        break
    return h

# ---------- SLSQP refinement (smooth max) ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=200):
    """SLSQP on smooth max approximation."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {"type": "eq", "fun": sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x
