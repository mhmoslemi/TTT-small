    cur_c = compute_c5(h)
    while time.time() - start < time_budget:

        max_idx = np.argmax(corr_full)
        # compute shift k = max_idx - (n-1)  (since mode full includes negative shifts)
        k = max_idx - (n-1)
        if k < 0:
            # negative shift: treat similarly by swapping roles
            # we can just consider absolute value because correlation symmetric? Actually not exactly due to zero padding.
            # But we can handle both.
        shift = k
        # compute arrays for shift
        if shift >= 0:
            h_i = h[:n-shift]
            h_j = h[shift:]
        else:

            shift_abs = -shift
            h_i = h[shift_abs:]
            h_j = h[:n-shift_abs]
        # donors: where h_i>0 and h_j<1

            i_a = a_idx

        new_c = compute_c5(h)

def targeted_local_search(h, rng, time_budget):
    start = time.time()
    n = h.size
    cur_c = compute_c5(h)
    while time.time() - start < time_budget:
        corr_full = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr_full)
        k = max_idx - (n - 1)  # shift
        if k >= 0:
            hi = h[:n - k]
            hj = h[k:]
            # mapping: donor index in original hi index = idx_i, acceptor index = idx_j (same)
            # For donor d in hi, actual index in h is d
            # For acceptor a in hi, index = a
            # For shift negative case, we will map accordingly.
        else:
            k_abs = -k
            hi = h[k_abs:]
            hj = h[:n - k_abs]
            # donor index in hi corresponds to original index = d + k_abs
            # acceptor index in hi corresponds to a + k_abs
        # Compute donor and acceptor weights
        donor_weights = hi * (1 - hj)  # large when hi high, hj low
        acceptor_weights = (1 - hi) * hj  # large when hi low, hj high
        # Avoid zero weights
        if donor_weights.sum() == 0 or acceptor_weights.sum() == 0:
            # fallback random perturb and continue
            h = perturb_h(h, rng, scale=0.1)
            cur_c = compute_c5(h)
            continue
        # sample donors and acceptors (maybe via weighted random)
        # To reduce overhead, we can sample uniformly among indices with weight > 0
        donor_candidates = np.where(donor_weights > 0)[0]
        acceptor_candidates = np.where(acceptor_weights > 0)[0]
        if donor_candidates.size == 0 or acceptor_candidates.size == 0:
            h = perturb_h(h, rng, scale=0.1)
            cur_c = compute_c5(h)
            continue
        d_local = rng.choice(donor_candidates)
        a_local = rng.choice(acceptor_candidates)
        # map to original indices
        if k >= 0:
            d_idx = d_local
            a_idx = a_local
        else:
            d_idx = d_local + k_abs
            a_idx = a_local + k_abs
        # amount epsilon
        max_eps = min(h[d_idx], 1.0 - h[a_idx])
        if max_eps <= 0:
            continue
        # pick epsilon as fraction of max_eps (e.g., up to 0.5)
        eps = max_eps * rng.random() * 0.5
        # apply
        h[d_idx] -= eps
        h[a_idx] += eps
        new_c = compute_c5(h)
        if new_c < cur_c:
            cur_c = new_c
        else:
            # revert
            h[d_idx] += eps
            h[a_idx] -= eps
            # occasional acceptance to escape
            if rng.random() < 0.001:
                # accept anyway
                cur_c = new_c
            else:
                # maybe try random small perturbation
                pass
    return h, cur_c

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Helper functions ----------

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth log‑sum‑exp approximation of max_k Σ_i x_i (1 - x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for a step function h on [0,2] (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=3000, start_step=0.3):
    """Simple hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


def slsqp_optimize(h0, beta, rng, max_iter=500):
    """Run a single SLSQP optimisation with the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # final exact sum correction (project onto hyperplane)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def project_to_feasible(v, target):
    """Projection onto {0≤h≤1, Σh = target} via bisection on the shift."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Fast random feasible vector (0≤h≤1, Σh=n/2)."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_to_feasible(v, n / 2.0)


def upsample_initial(to_n):
    """Linear interpolation of the global initial_h_values to length to_n."""
    init = np.array(initial_h_values, dtype=float)
    m = init.size
    xs_old = np.linspace(0.0, 2.0, m, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, to_n, endpoint=False)
    return np.interp(xs_new, xs_old, init)


def de_optimize(n, rng, max_time):
    """Simple DE with projection onto the feasible set."""
    target = n / 2.0
    popsize = max(20, min(50, int(0.2 * n)))
    pop = []

    # seed with upsampled initial if available
    if "initial_h_values" in globals():
        init_vec = upsample_initial(n)
        init_vec = project_to_feasible(init_vec, target)
        pop.append(init_vec)

    while len(pop) < popsize:
        pop.append(random_feasible_vector(n, rng))

    pop = np.array(pop)
    scores = np.array([compute_c5(ind) for ind in pop])
    best_idx = scores.argmin()
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]

    F = 0.5
    CR = 0.9
    start = time.time()
    while time.time() - start < max_time:
        for i in range(popsize):
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, 3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(0, n)] = True
            trial = np.where(cross, mutant, pop[i])
            trial = project_to_feasible(trial, target)
            trial_score = compute_c5(trial)
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_h = trial.copy()
    return best_h, best_score


def targeted_local_search(h, rng, time_budget):
    """Directed local search that reduces the worst shift."""
    start = time.time()
    n = h.size
    cur_c = compute_c5(h)

    while time.time() - start < time_budget:
        corr_full = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr_full)
        k = max_idx - (n - 1)  # shift (can be negative)

        if k >= 0:
            hi = h[: n - k]
            hj = h[k:]
            offset = 0
        else:
            k_abs = -k
            hi = h[k_abs:]
            hj = h[: n - k_abs]
            offset = k_abs

        # donor: high hi, low hj  --> hi * (1-hj)
        donor_weights = hi * (1 - hj)
        # acceptor: low hi, high hj --> (1-hi) * hj
        acceptor_weights = (1 - hi) * hj

        if donor_weights.sum() == 0 or acceptor_weights.sum() == 0:
            # fallback to a random perturbation
            h = perturb_h(h, rng, scale=0.1)
            cur_c = compute_c5(h)
            continue

        donor_idxs = np.where(donor_weights > 0)[0]
        acceptor_idxs = np.where(acceptor_weights > 0)[0]

        if donor_idxs.size == 0 or acceptor_idxs.size == 0:
            h = perturb_h(h, rng, scale=0.1)
            cur_c = compute_c5(h)
            continue

        d_local = rng.choice(donor_idxs)
        a_local = rng.choice(acceptor_idxs)

        d_idx = d_local + offset
        a_idx = a_local + offset

        max_eps = min(h[d_idx], 1.0 - h[a_idx])
        if max_eps <= 0:
            continue
        eps = max_eps * rng.random() * 0.5   # up to half of what is possible

        h[d_idx] -= eps
        h[a_idx] += eps
        new_c = compute_c5(h)

        if new_c < cur_c:
            cur_c = new_c
        else:
            # revert
            h[d_idx] += eps
            h[a_idx] -= eps
            # occasional uphill acceptance to escape plateaus
            if rng.random() < 0.001:
                cur_c = new_c
    return h, cur_c
