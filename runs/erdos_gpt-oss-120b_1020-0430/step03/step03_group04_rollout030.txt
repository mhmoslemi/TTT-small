def solve_lp(n, rng):
    # Build cp variables
    h = cp.Variable(n)
    s = cp.Variable((n, n))
    t = cp.Variable()
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n/2,
    ]
    # product constraints
    # s <= h_i
    for i in range(n):
        constraints.append(s[i, :] <= h[i])  # broadcast row
        constraints.append(s[:, i] <= h[i])  # broadcast column
        # lower bound: s >= h_i + h_j - 1
        # Equivalent: s[i,j] + 1 >= h[i] + h[j]
        constraints.append(s[i, :] + 1 >= h[i] + h) # this is shape n
        # s >= 0 also covered implicitly maybe not needed
    constraints.append(s >= 0)

    # shift constraints
    for k in range(n):
        # sum_i s[i, i+k] for i=0..n-k-1
        idx_i = cp.arange(n-k)  # can't index with variable? Use sum of diag
        # Use matrix: s[:n-k, k:] should correspond to pairs (i,i+k)
        constraints.append(cp.sum(s[:n-k, k:]) >= t)

    # objective: maximize t
    prob = cp.Problem(cp.Maximize(t), constraints)
    prob.solve(solver=cp.ECOS, max_iters=5000, abstol=1e-8, reltol=1e-8, feastol=1e-8)
    # get h values
    h_val = np.clip(h.value, 0, 1)
    # adjust sum to n/2 by scaling (since small numeric).
    diff = n/2 - h_val.sum()
    if abs(diff) > 1e-8:
        # try redistribution similarly
        for _ in range(1000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h_val[i]) if diff > 0 else max(diff, -h_val[i])
            h_val[i] += add
            diff = n/2 - h_val.sum()
            if abs(diff) < 1e-9:
                break
    return h_val, prob.value

def solve_lp_maxmin(n, rng):
    import cvxpy as cp
    h = cp.Variable(n)
    s = cp.Variable((n, n))
    t = cp.Variable()
    constraints = [h >= 0, h <= 1, cp.sum(h) == n/2, s >= 0]

    # Upper bound constraints: s <= h_i and s <= h_j
    for i in range(n):
        constraints.append(s[i, :] <= h[i])
        constraints.append(s[:, i] <= h[i])
        constraints.append(s[i, :] + 1 >= h[i] + h)
    # shift constraints
    for k in range(n):
        constraints.append(cp.sum(s[:n-k, k:]) >= t)
    # objective
    prob = cp.Problem(cp.Maximize(t), constraints)
    # solve
    prob.solve(solver=cp.ECOS, max_iters=5000, abstol=1e-9, reltol=1e-9, feastol=1e-9)
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        return None, -np.inf
    h_val = np.clip(h.value, 0, 1)
    # adjust sum
    target = n/2
    diff = target - h_val.sum()
    # small redistribution similar to earlier
    for _ in range(1000):
        if abs(diff) < 1e-10:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h_val[i]) if diff > 0 else max(diff, -h_val[i])
        h_val[i] += add
        diff = target - h_val.sum()
    return h_val, float(t.value)
```

import time

def compute_c5(h):
    """Exact discrete C5 for step function h on [0,2] with uniform spacing."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c - 1e-12:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-5)
    return best_h, best_c


def gradient_refine(h, rng, step=0.02, max_iter=30):
    """Refine by moving against gradient of the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]    # + (1 - h[i+k])
            g[k:] -= h[:upper]          # - h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)

                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


def simulated_annealing(h0, rng, deadline):
    """Simple SA that respects the sum constraint."""
    n = h0.size
    target = n / 2.0
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    T0 = max(0.01 * cur_c, 1e-4)
    it = 0
    while time.time() < deadline and it < 12000:
        step = max(0.5 * (0.999 ** it), 5e-4)
        cand_h = perturb_h(cur_h, rng, scale=step)
        cand_c = compute_c5(cand_h)
        accept = False
        if cand_c < cur_c:

def solve_lp_maxmin(n, rng):
    """Solve linear relaxation:
       maximize t  s.t. for every shift k,
       sum_i s[i,i+k] >= t,  and s obey McCormick bounds with h.
    """
    h = cp.Variable(n)
    s = cp.Variable((n, n))
    t = cp.Variable()
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n / 2,
        s >= 0,
    ]
    # McCormick envelope: s_ij <= h_i, s_ij <= h_j, s_ij >= h_i + h_j - 1
    for i in range(n):
        constraints.append(s[i, :] <= h[i])
        constraints.append(s[:, i] <= h[i])
        constraints.append(s[i, :] + 1 >= h[i] + h)
    # shift constraints
    for k in range(n):
        constraints.append(cp.sum(s[: n - k, k:]) >= t)
    prob = cp.Problem(cp.Maximize(t), constraints)
    prob.solve(solver=cp.ECOS, max_iters=5000, abstol=1e-9,
               reltol=1e-9, feastol=1e-9, verbose=False)
    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None, -np.inf
    h_val = np.clip(h.value, 0.0, 1.0)
    # exact feasibility cleanup
    target = n / 2.0
    diff = target - h_val.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h_val[i]) if diff > 0 else max(diff, -h_val[i])
        h_val[i] += add
        diff = target - h_val.sum()
    return h_val, float(t.value)

            h_hc, c_hc = hill_climb(h, rng, max_iters=1200, start_step=0.6)
            if c_hc < best_c:
                best_h, best_c, best_n = h_hc.copy(), c_hc, n

            # ----------------------------------------------------
            # gradient refinement
            # ----------------------------------------------------
            h_gr = gradient_refine(h_hc, rng, step=0.025, max_iter=20)
            c_gr = compute_c5(h_gr)
            if c_gr < best_c:
                best_h, best_c, best_n = h_gr.copy(), c_gr, n

            # ----------------------------------------------------
            # simulated annealing (time‑capped)
            # ----------------------------------------------------
            deadline_sa = start_time + budget_s * 0.96
            if time.time() < deadline_sa:
                h_sa, c_sa = simulated_annealing(h_gr, rng, deadline_sa)
                if c_sa < best_c:
                    best_h, best_c, best_n = h_sa.copy(), c_sa, n

            # ----------------------------------------------------
            # final hill‑climb
            # ----------------------------------------------------
            if time.time() - start_time < budget_s * 0.98:
                h_fc, c_fc = hill_climb(h_sa, rng, max_iters=1500, start_step=0.4)
                if c_fc < best_c:
                    best_h, best_c, best_n = h_fc.copy(), c_fc, n

            # early exit if time almost exhausted
            if time.time() - start_time > budget_s * 0.99:
                break

    # ------------------------------------------------------------
    # optional up‑sampling refinement
    # ------------------------------------------------------------
    if best_h is not None and time.time() - start_time < budget_s * 0.97:
        if best_n * 2 <= 300:
            n_up = best_n * 2
            x_src = np.linspace(0.0, 2.0, best_n, endpoint=False)
            x_up = np.linspace(0.0, 2.0, n_up, endpoint=False)
            h_up = np.interp(x_up, x_src, best_h)
            target_up = n_up / 2.0
            diff = target_up - h_up.sum()
            for _ in range(2000):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n_up)
                add = min(diff, 1.0 - h_up[i]) if diff > 0 else max(diff, -h_up[i])
                h_up[i] += add
                diff = target_up - h_up.sum()
            h_up, c_up = hill_climb(h_up, rng, max_iters=2200, start_step=0.5)
            if c_up < best_c:
                best_h, best_c, best_n = h_up, c_up, n_up

    # ------------------------------------------------------------
    # final safety checks
    # ------------------------------------------------------------
    if best_h is None:
        best_n = 120
        best_h = np.full(best_n, 0.5)

    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    diff = target - best_h.sum()
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, best_n)
        add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
        best_h[i] += add
        diff = target - best_h.sum()

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
