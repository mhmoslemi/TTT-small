import time, numpy as np
from scipy.optimize import minimize

# global variables for SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None

def _sum_eq_constraint(x): return np.sum(x) - CURRENT_TARGET
def _softmax_obj(x): 
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h, target, rng):
    # adjust to meet sum target (like previous function)
    diff = target - h.sum()
    n = h.size
    # simple loop
    for _ in range(10000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-12:
            continue
        h[i] += add
        diff = target - h.sum()
    # final clip
    return np.clip(h, 0.0, 1.0)

def random_binary_vector(n, rng):
    # n should be even
    half = n // 2
    h = np.zeros(n, dtype=float)
    ones_idx = rng.choice(n, size=half, replace=False)
    h[ones_idx] = 1.0
    return h

def simulated_annealing_binary(n, rng, time_limit):
    # initial
    h = random_binary_vector(n, rng)
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    start = time.time()
    # estimate initial temperature
    deltas = []
    for _ in range(200):
        # pick opposite bits
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        while h[i] == h[j]:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
        h[i], h[j] = h[j], h[i]  # swap
        new_c = compute_c5(h)
        delta = new_c - cur_c
        deltas.append(delta)
        # revert
        h[i], h[j] = h[j], h[i]
    pos_deltas = [d for d in deltas if d > 0]
    T0 = max(np.mean(pos_deltas) if pos_deltas else 0.01, 0.01)
    T_end = 1e-4

    # main loop
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit - 0.1:  # keep small margin
            break
        ratio = elapsed / time_limit
        T = T0 * (1 - ratio) + T_end * ratio
        # random opposite bits
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        while h[i] == h[j]:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
        # swap
        h[i], h[j] = h[j], h[i]
        new_c = compute_c5(h)
        delta = new_c - cur_c
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            # accept
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            # reject, revert
            h[i], h[j] = h[j], h[i]
    return best_h, best_c

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(_softmax_obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        # simple repair using random adjustments
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            if abs(add) < 1e-12:
                continue
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global helpers for the smooth surrogate used by SLSQP
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    """Smooth log‑sum‑exp surrogate of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


# ----------------------------------------------------------------------
# Problem‑specific utilities
# ----------------------------------------------------------------------
def compute_c5(h):
    """Exact discrete C₅ for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng):
    """Adjust h so that Σh = target while 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-12:
            continue
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_binary_vector(n, rng):
    """Binary vector with exactly n/2 ones (n must be even)."""
    half = n // 2
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, size=half, replace=False)
    h[ones] = 1.0
    return h


def simulated_annealing_binary(n, rng, time_limit):
    """Simple SA on binary vectors (preserves sum)."""
    h = random_binary_vector(n, rng)
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    start = time.time()

    # estimate a temperature scale
    deltas = []
    for _ in range(200):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        while h[i] == h[j]:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
        h[i], h[j] = h[j], h[i]
        new_c = compute_c5(h)
        deltas.append(new_c - cur_c)
        h[i], h[j] = h[j], h[i]
    pos = [d for d in deltas if d > 0]
    T0 = max(np.mean(pos) if pos else 0.01, 0.01)
    T_end = 1e-4

    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit - 0.1:
            break
        ratio = elapsed / time_limit
        T = T0 * (1 - ratio) + T_end * ratio

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        while h[i] == h[j]:
            i = rng.integers(0, n)
            j = rng.integers(0, n)

        h[i], h[j] = h[j], h[i]                # swap
        new_c = compute_c5(h)
        delta = new_c - cur_c
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]            # revert
    return best_h, best_c


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            if abs(add) < 1e-12:
                continue
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def hill_climb(h_start, rng, max_iters=2500, start_step=0.5):
    """Local hill‑climb: keep only improving epsilon‑transfers."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)
        max_pos = min(h[i], 1.0 - h[j])
        max_neg = min(h[j], 1.0 - h[i])
        if max_pos + max_neg == 0:
            continue
        eps = (rng.random() * (max_pos + max_neg)) - max_neg
        eps *= step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

        start_candidates = [best_h_bin, init_h]
    else:
        start_candidates = [best_h_bin]

    # ------------------------------------------------------------------

    refine_budget = total_budget - (time.time() - start_time)
    best_h = best_h_bin.copy()
    best_c = best_c_bin

    beta_list = [10.0, 50.0, 200.0, 800.0, 2000.0]
    for beta in beta_list:
        if time.time() - start_time >= total_budget - 5.0:

        except Exception:

        # short hill‑climb after each SLSQP stage

            best_h = h_hc.copy()

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
