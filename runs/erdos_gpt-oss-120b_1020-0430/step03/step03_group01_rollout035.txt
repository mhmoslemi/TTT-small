def compute_c5(h):
    n = h.size
    # Using FFT:
    N = 1 << ((2*n-1).bit_length())
    # Flip 1-h
    b = np.flip(1.0 - h)
    H = np.fft.rfft(h, n=N)
    B = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(H * B, n=N)
    # conv array size N, but relevant portion first 2n-1
    max_val = np.max(conv[:2*n-1])
    return max_val * (2.0/n)

        for _ in range(n_trials):
            new_h = perturb_h_transfer(h, rng, scale=step)
            # maybe random swap with prob?

            c = compute_c5(new_h)

def random_feasible_vector(n, rng):
    # generate random numbers in [0, 1] and scale to target
    h = rng.random(n)
    target = n/2.0
    # scale to target
    h = h * (target / h.sum())
    # clip and adjust
    h = np.clip(h, 0.0, 1.0)
    # compute diff
    diff = target - h.sum()
    # adjust by distributing diff uniformly to non-saturated entries
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        # choose random index where we can add/subtract
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])

        diff = target - h.sum()

def perturb_h_transfer(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    # after epsilon may slightly violate bounds due to floating errors; clip
    new_h = np.clip(new_h, 0.0, 1.0)
    return new_h

        x = np.clip(x, 0.0, 1.0)

def perturb_h_transfer(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return np.clip(new_h, 0.0, 1.0)

def perturb_h_swap(h, rng):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    new_h = h.copy()
    new_h[i], new_h[j] = new_h[j], new_h[i]
    return new_h

def full_corr(x):
    n = x.size
    N = 1 << ((2 * n - 1).bit_length())
    b = np.flip(1.0 - x)
    H = np.fft.rfft(x, n=N)
    B = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(H * B, n=N)
    return conv[: 2 * n - 1]

def compute_c5(h):
    n = h.size
    max_val = np.max(full_corr(h))
    return max_val * (2.0 / n)

# ---------- SLSQP surrogate ----------
CURRENT_BETA = 1.0

def sum_eq_constraint(x, target):
    return np.sum(x) - target

def softmax_obj(x):
    corr = full_corr(x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=400):
    global CURRENT_BETA
    CURRENT_BETA = beta
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint, "args": (target,)}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x += diff / n
        x = np.clip(x, 0.0, 1.0)
    return x

# ---------- hill climbing ----------
def hill_climb(start_h, rng, max_iters=3000, step_init=0.5,
               step_decay=0.7, decay_interval=300, n_trials=8):
    h = start_h.copy()
    best = compute_c5(h)
    step = step_init
    n = h.size
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h_transfer(h, rng, scale=step)
            if rng.random() < 0.2:
                new_h = perturb_h_swap(new_h, rng)
            c = compute_c5(new_h)
            if c < best:
                best, h = c, new_h
                improved = True
                break
        if not improved and rng.random() < 0.05:
            new_h = perturb_h_transfer(h, rng, scale=step * 2.0)
            c = compute_c5(new_h)
            if c < best:
                best, h = c, new_h
                improved = True
        if (it + 1) % decay_interval == 0:
            step = max(step * step_decay, 1e-4)
    return h, best

# ---------- main ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 180
        h0 = random_feasible_vector(n, rng)

    best_h, best_c = h0, compute_c5(h0)

    # diverse starter pool
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    pool = [best_h]
    for f in range(1, 7):
        for func in (np.sin, np.cos):
            h = 0.5 + 0.5 * func(np.pi * f * xs)
            h = np.clip(h, 0.0, 1.0)
            h = enforce_sum(h, n / 2.0, rng)
            pool.append(h)
    for _ in range(6):
        pool.append(random_feasible_vector(n, rng))

    for h in pool:
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # ------- main search -------
    main_deadline = start_time + budget_s * 0.8
    beta_seq = [5, 20, 80, 200, 500, 1000, 2000, 5000]
    while time.time() < main_deadline:
        start_vec = (
            perturb_h_transfer(best_h, rng, scale=0.3)
            if rng.random() < 0.6
            else random_feasible_vector(n, rng)
        )
        h_opt, c_opt = hill_climb(
            start_vec,
            rng,
            max_iters=3000,
            step_init=0.5,
            n_trials=8,
            decay_interval=300,
            step_decay=0.7,
        )
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()

        for beta in beta_seq:
            if time.time() >= main_deadline:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, max_iter=500)
            except Exception:
                continue
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_c, best_h = c_ref, h_ref.copy()

    # ------- finer grid if time left -------
    if time.time() < start_time + budget_s:
        target_n = min(720, max(360, n * 2))
        if target_n != n:
            old_x = (np.arange(n) + 0.5) * (2.0 / n)
            new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
            h_up = np.interp(new_x, old_x, best_h)
            best_h = enforce_sum(h_up, target_n / 2.0, rng)
            n = target_n

        h_opt2, c_opt2 = hill_climb(
            best_h,
            rng,
            max_iters=2000,
            step_init=0.3,
            n_trials=6,
            decay_interval=200,
            step_decay=0.7,
        )
        if c_opt2 < best_c:
            best_c, best_h = c_opt2, h_opt2.copy()

        for beta in (2000, 5000, 10000):
            if time.time() >= start_time + budget_s:
                break
            try:
                h_ref2 = slsqp_optimize(best_h, beta, max_iter=500)
            except Exception:
                continue
            c_ref2 = compute_c5(h_ref2)
            if c_ref2 < best_c:
                best_c, best_h = c_ref2, h_ref2.copy()

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
