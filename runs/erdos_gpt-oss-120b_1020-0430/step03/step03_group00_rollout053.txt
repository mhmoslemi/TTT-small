def simulated_annealing(h0, n, time_limit, rng, init_temp=10.0, cooling_rate=0.95, min_temp=0.001, steps_per_temp=200):
    h = h0.copy()
    c_cur = compute_c5(h)
    best_h = h.copy()
    best_c = c_cur
    T = init_temp
    start = time.time()
    while T > min_temp and time.time() - start < time_limit:
        for _ in range(steps_per_temp):
            # propose
            i = rng.integers(0, n)
            j = rng.integers(0, n-1)
            while j == i:
                j = rng.integers(0, n)
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0:
                continue
            eps = max_eps * rng.random() * 0.5  # 0-0.5 fraction of max
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
            c_new = compute_c5(h_new)
            delta = c_new - c_cur
            if delta < 0 or rng.random() < np.exp(-delta / T):
                h = h_new
                c_cur = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
        T *= cooling_rate
    return best_h, best_c

# ---------- helpers ----------
def enforce_sum(h, target, rng):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    it = 0
    while abs(diff) > 1e-12 and it < 10000:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i])
            add *= rng.random()
            h[i] += add
            diff -= add
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i])
            sub *= rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        if diff > 0:
            h[i] = min(1.0, h[i] + diff)
        else:
            h[i] = max(0.0, h[i] + diff)
    return np.clip(h, 0.0, 1.0)

def random_feasible_vector(n, rng):
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)

def perturb_random(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    while j == i:
        j = rng.integers(0, n)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * rng.random() * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

def perturb_targeted(h, grad, shift, rng, scale=0.2):
    pos = np.where((grad > 1e-9) & (h > 1e-12))[0]
    neg = np.where((grad < -1e-9) & (h < 1.0 - 1e-12))[0]
    if pos.size == 0 or neg.size == 0:
        return perturb_random(h, rng, scale)
    i = rng.choice(pos)
    j = rng.choice(neg)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return perturb_random(h, rng, scale)
    eps = max_eps * rng.random() * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def hill_climb_targeted(h_start, rng, max_iters=2000, start_step=0.5, target_sum=None):
    h = h_start.copy()
    n = h.size
    best_h, best_c = h.copy(), compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr)
        max_corr = corr[idx]
        c_cur = max_corr * (2.0 / n)
        k = idx - (n - 1)

        # gradient of the worst shift
        grad = np.ones(n)
        if k >= 0:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
        else:
            kk = -k
            grad[: n - kk] -= h[kk:]
            grad[kk:] -= h[: n - kk]

        new_h = perturb_targeted(h, grad, k, rng, scale=step)
        c_new = compute_c5(new_h)

        if c_new < best_c:
            best_c, best_h = c_new, new_h.copy()
        if c_new < c_cur:
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

def simulated_annealing(h0, n, time_limit, rng,
                        init_temp=10.0, cooling_rate=0.95,
                        min_temp=0.001, steps_per_temp=200):
    h = h0.copy()
    c_cur = compute_c5(h)
    best_h, best_c = h.copy(), c_cur
    T = init_temp
    start = time.time()
    while T > min_temp and time.time() - start < time_limit:
        for _ in range(steps_per_temp):
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            while j == i:
                j = rng.integers(0, n)
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0.0:
                continue
            eps = max_eps * rng.random() * 0.5
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
            c_new = compute_c5(h_new)
            delta = c_new - c_cur
            if delta < 0 or rng.random() < np.exp(-delta / T):
                h, c_cur = h_new, c_new
                if c_new < best_c:
                    best_c, best_h = c_new, h_new.copy()
        T *= cooling_rate
    return best_h, best_c
