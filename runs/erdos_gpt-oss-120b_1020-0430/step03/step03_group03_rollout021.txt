    n = len(h)
    grad = np.zeros(n)
    if shift >= 0:
        s = shift
        # i <= n-1-s: sum part 1
        mask_sum = np.arange(n) <= n - 1 - s

def subgradient_optimize(h0, rng, max_iter, eta0=0.01, decay=0.9995):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iter):
        shift = get_worst_shift(h)
        grad = subgradient_for_shift(h, shift)
        eta = eta0 * (decay ** it)
        h_new = h - eta * grad
        # Clip and enforce sum
        np.clip(h_new, 0.0, 1.0, out=h_new)
        enforce_sum(h_new, len(h) / 2.0, rng)
        c = compute_c5(h_new)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        # accept if improvement else maybe still accept? Keep h = h_new to continue progress.
        h = h_new
    return best_h, best_c

import time, math, numpy as np
from scipy.optimize import minimize

# ------------------- basic utilities -------------------
def compute_c5(h):
    """Discrete C5 for step function h on [0,2] with dx = 2/len(h)."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)

def enforce_sum(h, target, rng):
    """Adjust h in‑place so that sum(h)=target while respecting 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    """Generate random h with 0≤h≤1 and sum=n/2."""
    h = np.full(n, 0.5)
    # random swaps preserving sum and bounds
    for _ in range(n * 5):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    enforce_sum(h, n / 2.0, rng)
    return h

def perturb_h(h, rng, scale=0.2):
    """Transfer ε between two coordinates (preserves sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

def biased_perturb_h(h, rng, scale=0.2):
    """Bias transfer away from positions contributing most to the max overlap."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    k = np.argmax(corr)
    shift = k - (n - 1)
    i0 = max(0, -shift)
    i1 = min(n, n - shift)
    idx = np.arange(i0, i1)
    prod = h[idx] * (1 - h[idx + shift])
    if prod.size == 0:
        return perturb_h(h, rng, scale)
    thr_hi = np.quantile(prod, 0.70)
    thr_lo = np.quantile(prod, 0.30)
    high = idx[prod > thr_hi]
    low = idx[prod < thr_lo]
    if high.size == 0 or low.size == 0:
        return perturb_h(h, rng, scale)
    i = rng.choice(high)
    j = rng.choice(low)
    max_eps = min(h[i], 1.0 - h[j])

    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

# ------------------- SLSQP smoothing -------------------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=200):
    """Run SLSQP on smooth max‑approximation."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {"type": "eq", "fun": sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

def hill_climb_one_step(h, rng, scale=0.05):
    """Perform a single biased move aimed at reducing the current max overlap."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    k = np.argmax(corr)
    shift = k - (n - 1)

    if shift >= 0:
        i_start, i_end = 0, n - shift
    else:
        i_start, i_end = -shift, n
    idx = np.arange(i_start, i_end)
    prod = h[idx] * (1 - h[idx + shift])

    if prod.size == 0:
        return perturb_h(h, rng, scale)

    thr_hi = np.quantile(prod, 0.75)
    thr_lo = np.quantile(prod, 0.25)
    high = idx[prod > thr_hi]
    low = idx[prod < thr_lo]
    if high.size == 0 or low.size == 0:
        return perturb_h(h, rng, scale)

    i = rng.choice(high)
    j = rng.choice(low)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale)

    eps = rng.random() * max_eps * scale
    cand = h.copy()
    cand[i] -= eps
    cand[j] += eps
    return cand

            break
        try:
            best_h = slsqp_optimize(best_h, beta, max_iter=200)
            best_c = compute_c5(best_h)
        except Exception:
            continue

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
