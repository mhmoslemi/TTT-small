def subgradient_refine(h, rng, max_iter=20, init_step=0.02, eps_rel=1e-4):
    n = len(h); target = n/2.0; step = init_step
    for it in range(max_iter):
        corr = np.correlate(h, 1-h, mode="full")
        max_corr = corr.max()
        # Choose active shifts
        # tolerance: 1e-6 relative or absolute? Use max_corr*1e-6
        tol = max(1e-9, max_corr * 1e-6)
        active = np.where(corr >= max_corr - tol)[0] - (n-1)  # shift values
        g = np.zeros_like(h)
        for shift in active:
            k = abs(shift)
            if k == 0:
                g += 1 - 2*h
            else:
                upper = n - k
                # i from 0..upper-1: forward part
                g[:upper] += 1 - h[k:]
                # i from k..n-1: backward part subtract
                g[k:] -= h[:upper]
        # Average
        if active.size > 1:
            g /= active.size
        # Take a step
        new_h = h - step * g
        # Clip and project to sum
        new_h = np.clip(new_h, 0.0, 1.0)
        # Fix sum
        diff = target - new_h.sum()
        if abs(diff) > 1e-12:
            # add diff by scaling? Use simple redistribution
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - new_h[i]) if diff > 0 else max(diff, -new_h[i])
                new_h[i] += add
                diff = target - new_h.sum()
                if abs(diff) < 1e-12:
                    break
        # Acceptance: if new C5 < old or with line search reducing step if not)
        old_c = compute_c5(h)
        new_c = compute_c5(new_h)
        if new_c <= old_c:
            h = new_h
            # optionally increase step
            step *= 1.05
        else:
            # reduce step
            step *= 0.5
        # ensure step not too small
        if step < 1e-6:
            step = 1e-6
    return h

def binary_sa(n, rng, time_limit):
    h = np.zeros(n, dtype=np.float64)
    idx_ones = rng.choice(n, n//2, replace=False)
    h[idx_ones] = 1.0
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_h = h
    cur_c = best_c
    T0 = best_c * 0.1 + 1e-3
    it = 0
    while time.time() < time_limit:
        # update temperature
        T = T0 * (0.9995 ** it)
        # find positions
        ones = np.flatnonzero(cur_h > 0.5)
        zeros = np.flatnonzero(cur_h < 0.5)
        if len(ones)==0 or len(zeros)==0:
            break
        i = rng.integers(0, len(ones))
        j = rng.integers(0, len(zeros))
        i_idx = ones[i]
        j_idx = zeros[j]
        # swap
        cur_h[i_idx] = 0.0
        cur_h[j_idx] = 1.0
        cand_c = compute_c5(cur_h)
        delta = cand_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c = cand_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i_idx] = 1.0
            cur_h[j_idx] = 0.0
        it += 1
        # if time left low, break
    return best_h, best_c

def softmax_obj(x, beta):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = beta * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta

def slsqp_optimize(h0, beta, rng, max_iter=400):
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}
    res = minimize(lambda x: softmax_obj(x, beta), h0, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # fix sum exactly
    diff = target - x.sum()
    # simple loop for adjustment
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = target - x.sum()
    return x

import time
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# basic utilities
# --------------------------------------------------------------

def compute_c5(h):
    """Exact discrete C5 for step function h on [0,2] with uniform spacing."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=500, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay = 0.99
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % max(1, max_iters // 5) == 0:
            step = max(step * decay, 1e-4)
    return best_h, best_c

def subgradient_refine(h, rng, max_iter=20, init_step=0.02, tol_rel=1e-6):
    """Subgradient descent on the max‑correlation objective."""
    n = h.size
    target = n / 2.0
    step = init_step
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = corr.max()
        tol = max(1e-12, max_corr * tol_rel)
        active_shifts = np.where(corr >= max_corr - tol)[0] - (n - 1)

        if active_shifts.size == 0:
            break

        g = np.zeros_like(h)
        for shift in active_shifts:
            k = abs(shift)
            if k == 0:
                g += 1.0 - 2.0 * h
            else:
                upper = n - k
                g[:upper] += 1.0 - h[k:]
                g[k:] -= h[:upper]
        g /= active_shifts.size

        new_h = h - step * g
        new_h = np.clip(new_h, 0.0, 1.0)

        # project to the exact sum
        diff = target - new_h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - new_h[i]) if diff > 0 else max(diff, -new_h[i])
                new_h[i] += add
                diff = target - new_h.sum()
                if abs(diff) < 1e-12:
                    break

        old_c = compute_c5(h)
        new_c = compute_c5(new_h)
        if new_c <= old_c:
            h = new_h
            step = min(step * 1.05, 1.0)
        else:
            step = max(step * 0.5, 1e-6)
    return h

# soft‑max SLSQP optimizer (continuous)
# --------------------------------------------------------------

def softmax_obj(x, beta):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = beta * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta


def slsqp_optimize(h0, beta, rng, max_iter=400):
    """SLSQP minimisation of the smooth max‑correlation surrogate."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    res = minimize(
        lambda x: softmax_obj(x, beta),
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # exact sum correction
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
            if abs(diff) < 1e-12:
                break
    return x

def binary_sa(n, rng, deadline, init_h=None):
    """Simulated annealing on binary vectors (exact half‑ones)."""
    if init_h is not None:
        cur_h = init_h.copy()
    else:
        cur_h = np.zeros(n, dtype=np.float64)
        cur_h[rng.choice(n, n // 2, replace=False)] = 1.0

    best_h = cur_h.copy()
    best_c = compute_c5(cur_h)
    cur_c = best_c
    T0 = best_c * 0.1 + 1e-3
    it = 0

    while time.time() < deadline:
        T = T0 * (0.9995 ** it)

        ones = np.flatnonzero(cur_h > 0.5)
        zeros = np.flatnonzero(cur_h < 0.5)
        if ones.size == 0 or zeros.size == 0:
            break

        i_idx = ones[rng.integers(0, ones.size)]
        j_idx = zeros[rng.integers(0, zeros.size)]

        # propose swap

        cur_h[j_idx] = 1.0

            cur_c = cand_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i_idx] = 1.0
            cur_h[j_idx] = 0.0

        it += 1
    return best_h, best_c
