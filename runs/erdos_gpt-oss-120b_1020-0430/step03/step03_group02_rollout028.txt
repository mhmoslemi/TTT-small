def softmax_obj_grad(h, beta):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    maxc = np.max(corr)
    exp_terms = np.exp(beta * (corr - maxc))
    Z = exp_terms.sum()
    obj = (maxc + np.log(Z)) / beta

    w = exp_terms / Z  # length 2n-1
    # compute conv of w with (1 - h) and h
    # use np.convolve which is O(n^2) but okay for n ~ 200.
    conv1 = np.convolve(w, 1 - h, mode='full')
    conv2 = np.convolve(w, h, mode='full')
    # extract gradient at indices n-1 to 2n-2 (size n)
    grad = conv1[n-1: n-1 + n] - conv2[n-1: n-1 + n]
    # gradient of obj w.r.t h.
    # Also need to multiply by something? Let's verify derivative of obj: obj = (1/β)log Σ_i exp(β*corr_i). d(obj)/dh = Σ_i (exp(β*corr_i)/Σ_j exp(β*corr_j)) * d(corr_i)/dh. That's w_i * d(corr_i)/dh. Our w is normalized weights accordingly. So gradient = sum_i w_i * grad_corr_i; we computed that via convolution, indeed.

def slsqp_optimize_softmax(h0, beta, rng, max_iter=400):
    n = h0.size
    target = n/2.0
    bounds = [(0,1)]*n
    constraints = {"type": "eq", "fun": lambda x: np.sum(x)-target}
    # define objective with gradient:
    def fun(x):
        obj, _ = softmax_obj_grad(x, beta)
        return obj
    def jac(x):
        _, grad = softmax_obj_grad(x, beta)
        return grad
    res = minimize(fun, h0, method='SLSQP', jac=jac, bounds=bounds,
                   constraints=constraints,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x,0,1)
    # enforce sum exactly via redistribution (like before)
    diff = target - x.sum()
    # small correction loop
    for _ in range(2000):
        if abs(diff) < 1e-12: break
        i = rng.integers(0,n)
        add = min(diff,1-x[i]) if diff>0 else max(diff,-x[i])
        x[i] += add
        diff = target - x.sum()
    return x

def softmax_obj_grad(h, beta):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    maxc = np.max(corr)
    exp_terms = np.exp(beta * (corr - maxc))
    Z = np.sum(exp_terms)
    obj = (maxc + np.log(Z)) / beta
    w = exp_terms / Z
    w_rev = w[::-1]
    conv1 = np.convolve(w_rev, 1 - h, mode='full')
    conv2 = np.convolve(w_rev, h, mode='full')
    grad = conv1[n-1:n-1+n] - conv2[n-1:n-1+n]
    return obj, grad
```

def slsqp_optimize_softmax(h0, beta, rng, max_iter=400):
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type":"eq", "fun": lambda x: np.sum(x) - target}
    def fun(x):
        obj, _ = softmax_obj_grad(x, beta)
        return obj
    def jac(x):
        _, g = softmax_obj_grad(x, beta)
        return g
    res = minimize(fun, h0, method='SLSQP', jac=jac,
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol':1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum
    diff = target - x.sum()

        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = target - x.sum()
    return x

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    s = idx  # signed shift
    grad = np.ones_like(h)
    if s == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        if s > 0:
            grad[: n - s] -= h[s:]
            grad[s:] -= h[: n - s]
        else:
            # s < 0, shift left
            s_abs = -s
            grad[s_abs:] -= h[: n - s_abs]
            grad[: n - s_abs] -= h[s_abs:]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh
```

    best_c = compute_c5(h)

        if (it + 1) % decay == 0:

    return best_h, best_c

def compute_c5(h):
    """Exact discrete C5 for a step function on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def enforce_sum(h, rng):
    """Adjust h so that Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(20000):

        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()

def random_feasible_vector(n, rng):
    """Return a random vector in [0,1]^n with sum = n/2."""
    h = rng.random(n)
    target = n / 2.0
    h = h * target / h.sum()
    h = np.clip(h, 0.0, 1.0)

def softmax_obj_grad(h, beta):
    """
    Smoothed objective  (1/β)·log Σ exp(β·corr_i) and its gradient.
    corr = np.correlate(h, 1 - h, mode='full')
    """
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    maxc = np.max(corr)
    # stable log‑sum‑exp
    exp_terms = np.exp(beta * (corr - maxc))
    Z = np.sum(exp_terms)
    obj = (maxc + np.log(Z)) / beta

    w = exp_terms / Z               # normalized weights, length 2n‑1
    w_rev = w[::-1]                # reverse for correct convolution direction
    conv1 = np.convolve(w_rev, 1 - h, mode="full")
    conv2 = np.convolve(w_rev, h, mode="full")
    grad = conv1[n - 1 : n - 1 + n] - conv2[n - 1 : n - 1 + n]
    return obj, grad

        return g

        fun,

        method="SLSQP",
        jac=jac,
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

# ----------------------------------------------------------------------
def worst_gradient(h):
    """Gradient of the worst shift O_k = Σ h_i (1‑h_{i+k})."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h, grad, rng, scale=0.5):
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h_start, rng, max_iters=2000, step_scale=0.5):
    """Hill‑climb using gradient‑directed transfers."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    best_h = None
    best_c = np.inf
    best_n = None

    # ------------------------------------------------------------------
    # start from parent if present
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        best_h, best_c, best_n = h0.copy(), compute_c5(h0), h0.size

    beta_schedule = [5, 20, 80, 200, 500, 1000, 2000, 5000]
    candidate_ns = [96, 120, 144, 168, 192, 216, 240]

    # ------------------------------------------------------------------
    # optimisation on the (possibly) inherited solution
    # ------------------------------------------------------------------
    if best_h is not None:
        h = best_h.copy()
        for beta in beta_schedule:
            if time.time() - start > 0.94 * budget_s:
                break
            h = slsqp_optimize_softmax(h, beta, rng, max_iter=500)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, h.size
        if time.time() - start < 0.96 * budget_s:
            h, c = guided_hill_climb(h, rng, max_iters=1500, step_scale=0.3)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, h.size

    # ------------------------------------------------------------------
    # search over other discretisations
    # ------------------------------------------------------------------
    for n in candidate_ns:
        if time.time() - start > 0.94 * budget_s:
            break

        # initialise a base vector for this n
        if best_h is not None:
            if best_n == n:
                base = best_h.copy()
            else:
                x_old = np.linspace(0, 2, best_n, endpoint=False)
                x_new = np.linspace(0, 2, n, endpoint=False)
                base = np.interp(x_new, x_old, best_h)
                base = np.clip(base, 0.0, 1.0)
                base = enforce_sum(base, rng)
        else:
            base = random_feasible_vector(n, rng)

        # a few initial candidates (base + two random)
        inits = [base]
        for _ in range(2):
            inits.append(random_feasible_vector(n, rng))

        for h0 in inits:
            if time.time() - start > 0.96 * budget_s:
                break
            h = h0.copy()
            # soft‑max SLSQP schedule
            for beta in beta_schedule:
                if time.time() - start > 0.98 * budget_s:
                    break
                h = slsqp_optimize_softmax(h, beta, rng, max_iter=400)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

            # short hill‑climb refinement
            if time.time() - start < 0.99 * budget_s:
                h2, c2 = guided_hill_climb(h, rng, max_iters=1200, step_scale=0.3)
                if c2 < best_c:
                    best_h, best_c, best_n = h2.copy(), c2, n

    # ------------------------------------------------------------------
    # optional up‑sampling refinement (if time remains)
    # ------------------------------------------------------------------
    if best_h is not None and time.time() - start < 0.95 * budget_s:
        n_up = best_n * 2
        if n_up <= 480:  # keep the problem moderate
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n_up, endpoint=False)
            h_up = np.interp(x_new, x_old, best_h)
            h_up = np.clip(h_up, 0.0, 1.0)
            h_up = enforce_sum(h_up, rng)
            for beta in beta_schedule:
                if time.time() - start > budget_s * 0.99:
                    break
                h_up = slsqp_optimize_softmax(h_up, beta, rng, max_iter=400)
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, best_n = h_up.copy(), c_up, n_up

    # ------------------------------------------------------------------
    # final tiny random guided perturbations while time permits
    # ------------------------------------------------------------------
    while time.time() - start < budget_s * 0.999 and best_h is not None:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # ------------------------------------------------------------------
    # final feasibility check and return
    # ------------------------------------------------------------------
    if best_h is None:
        best_n = 96
        best_h = np.full(best_n, 0.5)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
