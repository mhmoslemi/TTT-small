def subgrad_step(h, step, target, rng):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    k = np.argmax(corr) - (n - 1)
    grad = np.ones(n)
    if k >= 0:
        ks = k
        grad[:n - ks] -= h[ks:]
        grad[ks:] -= h[:n - ks]
    else:
        ks = -k
        grad[ks:] -= h[:n - ks]
        grad[:n - ks] -= h[ks:]
    h_new = h - step * grad
    np.clip(h_new,0.0,1.0,out=h_new)
    enforce_sum(h_new, target, rng)
    return h_new, corr

import time, math, numpy as np

def compute_c5(h):
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / h.size)

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i, j = rng.integers(0, n, size=2)
        if i == j:

        h[i] -= eps
        h[j] += eps
    enforce_sum(h, n / 2.0, rng)
    return h

import time, math, numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / h.size)

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    enforce_sum(h, n / 2.0, rng)
    return h

def subgrad_step(h, step, target, rng):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    k = np.argmax(corr) - (n - 1)
    grad = np.ones(n)
    if k >= 0:
        ks = k
        if ks < n:
            grad[:n - ks] -= h[ks:]
            grad[ks:]     -= h[:n - ks]
    else:
        ks = -k
        if ks < n:
            grad[ks:]     -= h[:n - ks]
            grad[:n - ks] -= h[ks:]
    h_new = h - step * grad
    np.clip(h_new, 0.0, 1.0, out=h_new)
    enforce_sum(h_new, target, rng)
    return h_new, corr

def gradient_descent_optimize(h_start, time_limit, rng):
    n = h_start.size
    target = n / 2.0
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    it = 0
    step0 = 0.5
    start = time.time()
    while time.time() - start < time_limit:
        it += 1
        step = step0 / np.sqrt(it)
        h, corr = subgrad_step(h, step, target, rng)
        c = np.max(corr) * (2.0 / n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if rng.random() < 0.01:
            noise = rng.normal(scale=0.02, size=n)
            h_noise = h + noise
            np.clip(h_noise, 0.0, 1.0, out=h_noise)
            enforce_sum(h_noise, target, rng)
            c_noise = compute_c5(h_noise)
            if c_noise < best_c:
                best_c = c_noise
                best_h = h_noise.copy()
                h = h_noise
    return best_h, best_c
