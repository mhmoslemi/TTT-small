def project_onto_capped_simplex(v, target, tol=1e-9):
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(100):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
        if hi - lo < tol:
            break
    x = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    # ensure sum close to target; optionally adjust tiny diff
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # distribute diff among non-saturated interior entries
        # find indices with 0 < x_i < 1; if none, adjust any
        interior = np.where((x > 0) & (x < 1))[0]
        if interior.size > 0:
            # adjust uniformly
            x[interior] += diff / interior.size
            # clip again
            x = np.clip(x, 0.0, 1.0)
        else:
            # fallback to simple correction on random entries
            for i in np.where(x < 1)[0]:
                add = min(diff, 1 - x[i])
                x[i] += add
                diff -= add
                if abs(diff) < 1e-12:
                    break
    return x

def project_onto_capped_simplex(v, target):
    # Equivalent to solving min ||x-v||^2 s.t. 0<=x<=1, sum x = target.
    # Implementation using algorithm from "Projection onto the capped simplex".
    # Steps:
    # 1. Clip v to [0,1] (since bounds)
    # 2. If sum(clipped) == target: return clipped.
    # 3. Else, we need to find lambda such that sum(clip(v - lambda, 0, 1)) = target.
    # Use bisection with vectorized clipping.
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2.0, 0.0, 1.0)

def project_onto_capped_simplex(v, target, tol=1e-12, max_iter=60):
    # returns projection onto [0,1]^n with sum = target
    # Using bisection on lambda.
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Ensure sum exactly target within tolerance
    diff = target - x.sum()
    if abs(diff) > tol:
        # distribute diff among interior points
        interior = np.where((x > 0) & (x < 1))[0]
        if interior.size > 0:
            x[interior] += diff / interior.size
            x = np.clip(x, 0.0, 1.0)
        else:
            # add to any point not at upper bound
            for i in range(x.size):
                add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
                x[i] += add
                diff = target - x.sum()
                if abs(diff) < tol:
                    break
    return x

def correlate_rev(a, b):
    """Full linear correlation of a with reversed b (zero padding)."""
    n = a.shape[0]

        if hi - lo < tol:
            break
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > tol:
        interior = np.where((x > 0) & (x < 1))[0]
        if interior.size:
            x[interior] += diff / interior.size
            x = np.clip(x, 0.0, 1.0)
        else:
            for i in range(x.size):
                add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
                x[i] += add
                diff = target - x.sum()
                if abs(diff) < tol:
                    break
    return x


# ----------------------------------------------------------------------
# Simple random feasible vectors
# ----------------------------------------------------------------------
def random_feasible_vector(n, rng):
    v = rng.random(n)
    return project_onto_capped_simplex(v, n / 2.0)


# ----------------------------------------------------------------------
# Small hill‑climb for final polishing
# ----------------------------------------------------------------------
def hill_climb(h_start, rng, max_iters=1000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        i = rng.integers(0, h.size)
        j = rng.integers(0, h.size)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


# ----------------------------------------------------------------------
# Gradient descent on the smooth log‑sum‑exp surrogate
# ----------------------------------------------------------------------
def gradient_optimize_start(h0, beta_list, max_iter_per_beta, target):
    h = h0.copy()
    for beta in beta_list:
        lr = 0.15 / np.sqrt(beta)  # modest step size that shrinks with β
        for _ in range(max_iter_per_beta):
            S = h.sum()
            corr_hh = correlate_rev(h, h)          # Σ_i h_i h_{i+k}
            f = S - corr_hh                         # array length 2n‑1
            max_f = f.max()
            exp_vals = np.exp(beta * (f - max_f))
            w = exp_vals / exp_vals.sum()          # softmax weights, sum to 1
            w_comb = w + w[::-1]                   # w_s + w_{‑s}
            conv = conv_full(w_comb[::-1], h)      # length 3n‑2
            start = w_comb.size - 1
            grad = 1.0 - conv[start:start + h.size]  # ∇F
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_onto_capped_simplex(h_new, target)
            if np.max(np.abs(h_new - h)) < 1e-12:
                break
            h = h_new
    return h


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.97  # small safety margin

    # ---- initialise base dimension and vector --------------------------------
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n = 360
        h0 = np.full(n, 0.5)
    target = n / 2.0
    h0 = project_onto_capped_simplex(np.clip(h0, 0.0, 1.0), target)

    # ---- build a diversified pool of start vectors ---------------------------
    starts = [h0]
    xs = (np.arange(n) + 0.5) * (2.0 / n)

    for f in (1, 2, 3):
        s = 0.5 + 0.5 * np.sin(np.pi * f * xs)
        starts.append(project_onto_capped_simplex(np.clip(s, 0.0, 1.0), target))
        c = 0.5 + 0.5 * np.cos(np.pi * f * xs)
        starts.append(project_onto_capped_simplex(np.clip(c, 0.0, 1.0), target))

    # linear ramps
    ramp = xs / 2.0
    starts.append(project_onto_capped_simplex(np.clip(ramp, 0.0, 1.0), target))
    ramp2 = 1.0 - xs / 2.0
    starts.append(project_onto_capped_simplex(np.clip(ramp2, 0.0, 1.0), target))

    # a few random feasible vectors
    for _ in range(5):
        starts.append(random_feasible_vector(n, rng))

    # ---- quick exploration of the pool ---------------------------------------
    best_h = None
    best_c = float('inf')
    beta_small = [5, 20, 80, 200]

    for h_start in starts:
        if time.time() >= deadline:
            break
        h_opt = gradient_optimize_start(h_start, beta_small, max_iter_per_beta=60, target=target)
        h_opt, c_val = hill_climb(h_opt, rng, max_iters=1500, start_step=0.4)
        if c_val < best_c:
            best_c = c_val
            best_h = h_opt.copy()

    # ---- intensive refinement of the current best -----------------------------
    if best_h is not None and time.time() < deadline:
        beta_big = [5, 20, 80, 200, 800, 2000]
        # allocate generous iteration budget per β (time‑aware)
        remaining = deadline - time.time()
        iters_per_beta = 120  # fine‑grained enough; still well within budget
        h_refined = gradient_optimize_start(best_h, beta_big, max_iter_per_beta=iters_per_beta, target=target)
        h_refined, c_val = hill_climb(h_refined, rng, max_iters=2500, start_step=0.3)
        if c_val < best_c:
            best_c = c_val
            best_h = h_refined.copy()

    # ---- optional up‑sampling to a finer grid --------------------------------
    if best_h is not None and time.time() < deadline and n * 2 <= 720:
        n_up = n * 2
        target_up = n_up / 2.0
        # linear interpolation (piecewise‑linear refinement)
        h_up = np.empty(n_up)
        h_up[0::2] = best_h
        avg = (best_h[:-1] + best_h[1:]) / 2.0
        h_up[1:-1:2] = avg
        h_up[-1] = best_h[-1]
        h_up = project_onto_capped_simplex(np.clip(h_up, 0.0, 1.0), target_up)

        beta_big = [5, 20, 80, 200, 800, 2000]
        h_up = gradient_optimize_start(h_up, beta_big, max_iter_per_beta=80, target=target_up)
        h_up, c_up = hill_climb(h_up, rng, max_iters=2000, start_step=0.2)
        if c_up < best_c:
            best_c = c_up
            best_h = h_up.copy()
            n = n_up

    # ---- final clean‑up -------------------------------------------------------
    if best_h is None:
        best_h = np.full(n, 0.5)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_capped_simplex(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
