def swap_move(h, rng):
    # h binary np.int8
    ones_idx = np.where(h == 1)[0]
    zeros_idx = np.where(h == 0)[0]
    if len(ones_idx) == 0 or len(zeros_idx) == 0:

    i = rng.choice(ones_idx)
    j = rng.choice(zeros_idx)
    h_new = h.copy()
    h_new[i] = 0
    h_new[j] = 1
    return h_new

def binary_simulated_annealing(n, rng, time_limit, start_h=None):
    # returns (best_h, best_c)
    # start_h optional; else random binary.
    duration = time_limit
    deadline = time.time() + duration
    if start_h is None:
        cur_h = random_binary(n, rng)
    else:
        cur_h = start_h.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    # schedule
    T0 = 0.02
    Tf = 1e-6
    total = duration
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - (deadline - duration)
        frac = (elapsed) / total
        T = T0 * ((Tf/T0) ** frac)
        # propose swap
        new_h = swap_move_binary(cur_h, rng)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / max(T,1e-12)):
                accept = True
        if accept:
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
        it += 1
    return best_h, best_c
```

def continuous_refine(h0, rng, time_limit):
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    deadline = time.time() + time_limit
    # first apply hill_climb
    while time.time() < deadline:
        # hill climb for maybe 2000 iters
        # we can call hill_climb with max_iters limited by time left
        remaining = deadline - time.time()
        max_iters = int(min(2000, remaining * 1000))  # assuming 1000 iters per second? approximate
        h_opt, c_opt = hill_climb(best_h, rng, max_iters=max_iters, start_step=0.4)

        run_time = min(12.0, remaining)  # seconds per chunk
        # SA within chunk
        chunk_deadline = time.time() + run_time
        # Reset temperature schedule per chunk
        start_chunk = time.time()
        while time.time() < chunk_deadline:
            elapsed = time.time() - start_chunk
            frac = elapsed / run_time
            T = T0 * ((Tf/T0) ** frac)

            new_c = compute_c5(new_h)

                accept = True
            else:

                    accept = True
            if accept:

def perturb_h(h, rng, scale=0.2):

    i = rng.integers(0, n)
    j = rng.integers(0, n)

        return h.copy()

def enforce_sum(h, target, rng):
    """Adjust h so that sum(h) == target while keeping 0 <= h <= 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add

    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    total_budget = budget_s

    # ------------------------------------------------------------------
    # Initial vector (use provided one if present)
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h_init = np.array(initial_h_values, dtype=float)
        n0 = h_init.size
    else:
        h_init = None
        n0 = 360

    target_n = max(480, n0)
    if target_n % 2:
        target_n += 1

    if h_init is not None and h_init.size != target_n:
        h0 = upscale(h_init, target_n)
    else:
        h0 = h_init if h_init is not None else np.full(target_n, 0.5)

    h0 = np.clip(h0, 0.0, 1.0)
    h0 = enforce_sum(h0, target_n / 2.0, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ------------------------------------------------------------------
    # 1. Binary simulated annealing (majority of time)
    # ------------------------------------------------------------------
    binary_time = total_budget * 0.65
    binary_deadline = start_time + binary_time

    cur_binary = random_binary(target_n, rng)
    cur_c = compute_c5(cur_binary)
    if cur_c < best_c:
        best_h, best_c = cur_binary.copy(), cur_c

    T0, Tf = 0.03, 1e-6
    while time.time() < binary_deadline:
        remaining = binary_deadline - time.time()
        chunk = min(12.0, remaining)
        chunk_deadline = time.time() + chunk
        start_chunk = time.time()
        while time.time() < chunk_deadline:
            elapsed = time.time() - start_chunk
            frac = elapsed / chunk
            T = T0 * ((Tf / T0) ** frac)
            new_h = swap_move_binary(cur_binary, rng)
            new_c = compute_c5(new_h)
            delta = new_c - cur_c
            accept = False
            if delta < 0:
                accept = True
            else:
                if rng.random() < np.exp(-delta / max(T, 1e-12)):
                    accept = True
            if accept:
                cur_binary, cur_c = new_h, new_c
                if cur_c < best_c:
                    best_h, best_c = cur_binary.copy(), cur_c
        if rng.random() < 0.15 and time.time() < binary_deadline:
            cur_binary = random_binary(target_n, rng)
            cur_c = compute_c5(cur_binary)
            if cur_c < best_c:
                best_h, best_c = cur_binary.copy(), cur_c

    # ------------------------------------------------------------------
    # 2. Continuous refinement (remaining time)
    # ------------------------------------------------------------------
    cont_deadline = start_time + total_budget
    while time.time() < cont_deadline:
        remaining = cont_deadline - time.time()
        if remaining <= 0.5:
            break
        max_iters = int(min(3000, remaining * 500))
        h_opt, c_opt = hill_climb(best_h, rng, max_iters=max_iters, start_step=0.4)
        if c_opt < best_c:
            best_h, best_c = h_opt.copy(), c_opt
        for beta in [5, 20, 80, 200, 800]:
            if time.time() >= cont_deadline:
                break
            try:
                h_opt = slsqp_optimize(best_h, beta, max_iter=300)
                c_opt = compute_c5(h_opt)
                if c_opt < best_c:
                    best_h, best_c = h_opt.copy(), c_opt
            except Exception:
                continue
        if rng.random() < 0.3:
            best_h = perturb_h(best_h, rng, scale=0.3)
            best_h = enforce_sum(best_h, best_h.size / 2.0, rng)

    # ------------------------------------------------------------------
    # 3. Optional up‑sampling if time permits
    # ------------------------------------------------------------------
    leftover = total_budget - (time.time() - start_time)
    if leftover > 5.0 and best_h.size * 2 <= 800:
        up_h = np.repeat(best_h, 2)
        up_n = up_h.size
        up_h = enforce_sum(up_h, up_n / 2.0, rng)
        max_iters = int(min(4000, leftover * 500))
        up_opt, up_c = hill_climb(up_h, rng, max_iters=max_iters, start_step=0.3)
        if up_c < best_c:
            best_h, best_c = up_opt, up_c

    # ------------------------------------------------------------------
    # Final clean‑up
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
