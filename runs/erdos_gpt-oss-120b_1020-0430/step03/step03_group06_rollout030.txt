    lo = np.min(v) - 1.0
    hi = np.max(v)

        mid = (lo+hi)/2.0
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid

    # ensure sum exactly target within tolerance:
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        # distribute diff across indices not at boundaries
        # find indices where 0 < w_i < 1
        free = np.where((w > 1e-12) & (w < 1-1e-12))[0]

                grad[:n-k] -= dx * h[k:]
                # h[i-k] term
                grad[k:]   -= dx * h[:n-k]
            else:
                k = -s
                grad[k:]   -= dx * h[:n-k]
                grad[:n-k] -= dx * h[k:]
        # step size
        step = step0 / np.sqrt(it + 1)
        h_new = h - step * grad
        # project onto feasible region
        h_new = proj_box_sum(h_new, target)
        # compute objective
        c = compute_c5(h_new)
        # update best
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        # replace h for next iteration
        h = h_new
    return best_h, best_c

def hill_climb(h0, rng, max_iters=5000, step=0.5):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    for it in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        # compute max possible transfer
        max_pos = min(h[i], 1.0 - h[j])
        max_neg = min(h[j], 1.0 - h[i])
        if max_pos + max_neg == 0:
            continue
        eps = (rng.random() * (max_pos + max_neg)) - max_neg
        eps *= step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        # no need to project because we ensure within bounds
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
    return best_h, best_c

def simple_project(h, target):
    w = np.clip(h, 0.0, 1.0)
    s = w.sum()
    diff = target - s
    if abs(diff) > 1e-12:
        # distribute diff proportionally to free space (1 - w_i) for diff>0 else w_i
        if diff > 0:
            # free capacities
            caps = 1.0 - w
            total_cap = caps.sum()
            if total_cap > 0:
                w += diff * caps / total_cap
                # clip again
                w = np.clip(w, 0.0, 1.0)
        else:
            # reduction capacities
            caps = w
            total_cap = caps.sum()
            if total_cap > 0:
                w += diff * caps / total_cap
                w = np.clip(w, 0.0, 1.0)
        # after this, diff may be small residual; correct iteratively
        # iterate a few times
        for _ in range(5):
            s = w.sum()
            diff = target - s
            if abs(diff) < 1e-12:
                break
            # simple adjust on a random free index
            free = np.where((w > 1e-12) & (w < 1-1e-12))[0]
            if free.size == 0:
                break
            i = free[0]
            delta = np.clip(diff, -w[i], 1 - w[i])
            w[i] += delta
    return w

def anneal(h0, rng, time_limit):
    h = h0.copy()
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    start = time.time()
    # schedule: initial T = 0.01 maybe relative to c? Or maybe 0.005.
    # We'll gradually lower T to near zero.

    # Determine number of steps maybe 100k or time-limited.
    # Use exponential cooling: T = T0 * (0.995 ** step). Step increments each iteration.

    T0 = 0.01
    step = 0
    while time.time() - start < time_limit:
        # propose
        i = rng.integers(0, h.size)
        j = rng.integers(0, h.size-1)
        if j >= i:
            j += 1
        # compute possible transfer bounds as before
        max_pos = min(h[i], 1.0 - h[j])
        max_neg = min(h[j], 1.0 - h[i])
        if max_pos + max_neg == 0:
            continue
        eps = (rng.random() * (max_pos + max_neg)) - max_neg
        eps *= 0.3  # scale
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        # no out-of-bound thanks to eps bounds
        c_new = compute_c5(new_h)
        delta = c_new - c
        T = T0 * (0.9995 ** step)
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = new_h
            c = c_new
            if c < best_c:
                best_c = c
                best_h = h.copy()
        step += 1
    return best_h, best_c

def subgradient_optimize(h0, target, dx, time_limit, rng, step0=5.0):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    # compute initial c
    corr = np.correlate(h, 1 - h, mode='full')
    best_c = np.max(corr) * dx
    it = 0
    start = time.time()
    while time.time() - start < time_limit:
        # compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        g = corr * dx
        max_g = np.max(g)
        # find worst shifts
        # tolerance can be relative e.g., 1e-12
        mask = (g >= max_g - 1e-12)
        shifts = np.where(mask)[0] - (n - 1)
        # compute gradient
        grad = np.zeros_like(h)
        # loop over shifts
        for s in shifts:
            if s >= 0:
                k = s
                grad[:n - k] -= dx * h[k:]
                grad[k:] -= dx * h[:n - k]
            else:
                k = -s
                grad[k:] -= dx * h[:n - k]
                grad[:n - k] -= dx * h[k:]
        # step size schedule
        step = step0 / np.sqrt(it + 1)
        h_new = h - step * grad
        # project: clip and adjust sum
        w = np.clip(h_new, 0.0, 1.0)
        s = w.sum()
        diff = target - s
        # simple uniform adjustment loop
        for _ in range(5):
            if abs(diff) < 1e-12:
                break
            w += diff / n
            w = np.clip(w, 0.0, 1.0)
            diff = target - w.sum()
        h = w
        # update best if improved
        c = max_g
        if c < best_c:
            best_c = c
            best_h = h.copy()
        it += 1
    return best_h, best_c

import time
import numpy as np

# --------------------------------------------------------------
# Core utilities
# --------------------------------------------------------------
def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def proj_box_sum(v, target):
    """Exact projection of v onto {0≤x≤1, Σx=target} using water‑filling."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(50):
        mid = (lo + hi) / 2.0
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        free = np.where((w > 1e-12) & (w < 1 - 1e-12))[0]
        if free.size:
            i = free[0]
            w[i] = np.clip(w[i] + diff, 0.0, 1.0)
    return w


def simple_project(v, target):
    """Fast approximate projection: clip then uniform correction."""
    w = np.clip(v, 0.0, 1.0)
    n = w.size
    diff = target - w.sum()
    for _ in range(5):
        if abs(diff) < 1e-12:
            break
        w += diff / n
        w = np.clip(w, 0.0, 1.0)
        diff = target - w.sum()
    return w


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    v = rng.exponential(1.0, size=n)
    v = v / v.sum() * (n / 2.0)
    return proj_box_sum(v, n / 2.0)


def hill_climb(h0, rng, max_iters=5000, step=0.5):
    """Simple hill‑climb that preserves the sum."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    for _ in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        max_pos = min(h[i], 1.0 - h[j])
        max_neg = min(h[j], 1.0 - h[i])
        if max_pos + max_neg == 0:
            continue
        eps = (rng.random() * (max_pos + max_neg)) - max_neg
        eps *= step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h, h = c, new_h.copy(), new_h
    return best_h, best_c


def subgradient_optimize(h0, target, dx, time_limit, rng, step0=5.0):
    """Subgradient descent on max_{k} ∫h(1‑h(x+k)) dx."""
    h = h0.copy()
    n = h.size
    # initial value
    corr = np.correlate(h, 1 - h, mode="full")
    best_c = np.max(corr) * dx
    best_h = h.copy()
    it = 0
    start = time.time()
    while time.time() - start < time_limit:
        corr = np.correlate(h, 1 - h, mode="full")
        g = corr * dx
        max_g = np.max(g)
        mask = (g >= max_g - 1e-12)
        shifts = np.where(mask)[0] - (n - 1)
        grad = np.zeros_like(h)
        for s in shifts:
            if s >= 0:
                k = s
                grad[:n - k] -= dx * h[k:]
                grad[k:] -= dx * h[:n - k]
            else:
                k = -s
                grad[k:] -= dx * h[:n - k]
                grad[:n - k] -= dx * h[k:]
        step = step0 / np.sqrt(it + 1)
        h_new = h - step * grad
        h = simple_project(h_new, target)
        if max_g < best_c:
            best_c, best_h = max_g, h.copy()
        it += 1
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # ---- determine dimension and initial vector
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = proj_box_sum(h0, n / 2.0)
    else:
        n = 180
        h0 = np.full(n, 0.5)
        h0 = proj_box_sum(h0, n / 2.0)

    target = n / 2.0
    dx = 2.0 / n

    # ---- build a modest pool of diverse starts
    candidates = [h0]
    xs = (np.arange(n) + 0.5) * dx
    for f in (1, 2, 3, 4):
        for func in (np.sin, np.cos):
            h = 0.5 + 0.5 * func(np.pi * f * xs)
            h = np.clip(h, 0.0, 1.0)
            h = proj_box_sum(h, target)
            candidates.append(h)
    candidates.append(proj_box_sum(xs / 2.0, target))
    candidates.append(proj_box_sum(1.0 - xs / 2.0, target))
    for _ in range(6):
        candidates.append(random_feasible_vector(n, rng))

    best_h = None
    best_c = float("inf")
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # ---- main optimisation phases --------------------------------
    total_budget = budget_s
    # 1) Subgradient descent (about 65% of time)
    sub_budget = total_budget * 0.65
    while time.time() - start_time < sub_budget:
        # choose start point
        if rng.random() < 0.7:
            start = simple_project(best_h + rng.normal(scale=0.1, size=n), target)
        else:
            start = random_feasible_vector(n, rng)
        # allocate a short time slice
        remaining = sub_budget - (time.time() - start_time)
        slice_time = min(6.0, remaining)
        h_opt, c_opt = subgradient_optimize(
            start, target, dx, slice_time, rng, step0=rng.uniform(3.0, 6.0)
        )
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()
        # occasional hill‑climb after a good subgradient run
        if c_opt < best_c or rng.random() < 0.15:
            h_hc, c_hc = hill_climb(best_h, rng, max_iters=3000, step=0.4)
            if c_hc < best_c:
                best_c, best_h = c_hc, h_hc.copy()

    # 2) Fine‑grained hill‑climb (remaining time)
    while time.time() - start_time < total_budget * 0.95:
        h_hc, c_hc = hill_climb(best_h, rng, max_iters=5000, step=0.3)
        if c_hc < best_c:
            best_c, best_h = c_hc, h_hc.copy()
        # try a perturbed restart occasionally
        pert = simple_project(best_h + rng.normal(scale=0.05, size=n), target)
        h_hc2, c_hc2 = hill_climb(pert, rng, max_iters=2000, step=0.3)
        if c_hc2 < best_c:
            best_c, best_h = c_hc2, h_hc2.copy()

    # ---- final clean‑up ------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = proj_box_sum(best_h, target)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
