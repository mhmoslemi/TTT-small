def hill_climb(h_start, rng, max_iters=2500, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)
        max_pos = min(h[i], 1.0 - h[j])
        max_neg = min(h[j], 1.0 - h[i])
        if max_pos + max_neg == 0:
            continue
        eps = (rng.random() * (max_pos + max_neg)) - max_neg
        eps *= step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global helpers for the smooth surrogate used by SLSQP
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    """Smooth log‑sum‑exp surrogate of max_{k} Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

# Problem‑specific utilities
# ----------------------------------------------------------------------
def compute_c5(h):
    """Exact discrete C₅ for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

        if diff > 0:
            inc = np.minimum(delta, 1 - v[free])
            v[free] += inc
        else:
            dec = np.maximum(delta, -v[free])
            v[free] += dec
        diff = target - v.sum()
    return np.clip(v, 0.0, 1.0)


def random_binary_vector(n, target, rng):
    """Binary vector of length n with exactly target ones (target integer)."""
    target = int(round(target))
    idx = rng.permutation(n)[:target]
    h = np.zeros(n, dtype=np.uint8)
    h[idx] = 1
    return h


def hill_climb(h_start, rng, max_iters=2500, start_step=0.5):
    """Projected hill‑climb on continuous variables (preserves sum)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)
        max_pos = min(h[i], 1.0 - h[j])
        max_neg = min(h[j], 1.0 - h[i])
        if max_pos + max_neg == 0:
            continue
        eps = (rng.random() * (max_pos + max_neg)) - max_neg
        eps *= step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    if abs(CURRENT_TARGET - x.sum()) > 1e-12:
        x = project_to_feasible(x, CURRENT_TARGET)
    return x

# Main entry point
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # --------------------------------------------------------------
    # Determine dimension and initial feasible vector
    # --------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        target = n / 2.0
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_to_feasible(h0, target)
        global_best_h = h0.copy()
        global_best_c = compute_c5(global_best_h)
    else:
        n = 180
        target = n / 2.0
        # start from a random binary feasible vector
        global_best_h = random_binary_vector(n, target, rng).astype(float)
        global_best_c = compute_c5(global_best_h)

    # --------------------------------------------------------------
    # Simulated Annealing on binary vectors
    # --------------------------------------------------------------
    max_time_sa = budget_s * 0.55
    # initialise SA with a fresh random binary vector
    cur_h = random_binary_vector(n, target, rng).copy()
    cur_c = compute_c5(cur_h.astype(float))
    best_bin_h = cur_h.copy()
    best_bin_c = cur_c

    temp = 0.02
    temp_decay = 0.9997

    while time.time() - start_time < max_time_sa:
        ones = np.flatnonzero(cur_h)
        zeros = np.flatnonzero(1 - cur_h)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # propose swap
        cur_h[i] = 0
        cur_h[j] = 1
        new_c = compute_c5(cur_h.astype(float))
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            cur_c = new_c
            if cur_c < best_bin_c:
                best_bin_c = cur_c
                best_bin_h = cur_h.copy()
        else:
            # revert
            cur_h[i] = 1
            cur_h[j] = 0

        temp *= temp_decay

        # occasional shake to escape stagnation
        if rng.random() < 0.001:
            for _ in range(5):
                a, b = rng.integers(0, n, size=2)
                if cur_h[a] != cur_h[b]:
                    cur_h[a], cur_h[b] = cur_h[b], cur_h[a]
            cur_c = compute_c5(cur_h.astype(float))

        if time.time() - start_time >= max_time_sa:
            break

    # adopt best binary solution if it improves the global best
    if best_bin_c < global_best_c:
        global_best_c = best_bin_c
        global_best_h = best_bin_h.astype(float)

    # --------------------------------------------------------------
    # Local continuous refinement (hill‑climb)
    # --------------------------------------------------------------
    max_time_hc = budget_s * 0.85
    while time.time() - start_time < max_time_hc:
        h_opt, c_opt = hill_climb(global_best_h, rng, max_iters=2000, start_step=0.5)
        if c_opt < global_best_c:
            global_best_h, global_best_c = h_opt.copy(), c_opt
        # early exit if we have already reached the target bound
        if global_best_c <= 0.3808:
            break

    # --------------------------------------------------------------
    # Optional SLSQP polishing
    # --------------------------------------------------------------
    if time.time() - start_time < budget_s * 0.97:
        for beta in (40.0, 200.0, 800.0):
            if time.time() - start_time > budget_s * 0.97:
                break
            try:
                h_opt = slsqp_optimize(global_best_h, beta, max_iter=400)
                c_opt = compute_c5(h_opt)
                if c_opt < global_best_c:
                    global_best_h, global_best_c = h_opt.copy(), c_opt
            except Exception:
                pass

    # --------------------------------------------------------------
    # Final clean‑up
    # --------------------------------------------------------------
    global_best_h = np.clip(global_best_h, 0.0, 1.0)
    global_best_h = project_to_feasible(global_best_h, n / 2.0)
    final_c = compute_c5(global_best_h)

    return global_best_h.tolist(), float(final_c), int(n)
