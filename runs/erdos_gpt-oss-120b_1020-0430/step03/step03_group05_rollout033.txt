def sa_binary_optimize(n, rng, time_limit):
    # Returns best binary vector and its C5.
    # Use initial vector: alternating pattern.
    best = np.arange(n) % 2  # 0,1 pattern start with 0? Actually this yields 0,1,0,1,... which has half ones.
    # but if n is even, sum = n/2 (since pattern has as many 0 as 1).
    # Optionally random start also.
    best = best.astype(float) # We'll treat as float for compute.
    best_c = compute_c5_fft(best)
    # random restarts: keep best overall
    # We'll maintain current vector cur and cur_c.
    cur = best.copy()
    cur_c = best_c
    # schedule: start T = (some factor) * (initial max - min?), we can set initial T = 0.01 maybe.
    T0 = 0.01
    T = T0
    iteration = 0
    start = time.time()
    while time.time() - start < time_limit:
        # propose swap
        # pick i from ones, j from zeros
        ones_idx = np.flatnonzero(cur == 1)
        zeros_idx = np.flatnonzero(cur == 0)
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        # propose new vector
        new = cur.copy()
        new[i] = 0
        new[j] = 1
        new_c = compute_c5_fft(new)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_c = new_c
            if cur_c < best_c:
                best = cur.copy()
                best_c = cur_c
        # update temperature gradually
        iteration += 1
        # simplistic cooling: every 1000 iter reduce T
        if iteration % 1000 == 0:
            T *= 0.9
            if T < 1e-6:
                T = 1e-6
    return best, best_c

    return compute_c5_fft(h)

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

    corr = np.fft.irfft(H * np.conj(O_fft - H), n=L)
    max_corr = np.max(corr[: 2 * n - 1])
    return max_corr * (2.0 / n)


def compute_c5(h: np.ndarray) -> float:
    """Thin wrapper for the optimizer."""
    return compute_c5_fft(h)


def sum_eq_constraint(x: np.ndarray) -> float:
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x: np.ndarray) -> float:
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 800) -> np.ndarray:
    """SLSQP refinement of the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_BETA = beta
    n = h0.size
    CURRENT_TARGET = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        # simple redistribution to restore the exact sum
        for _ in range(2 * n):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Move a random ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() * 2.0 - 1.0) * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_fine(h: np.ndarray, rng: np.random.Generator,
                    max_iters: int = 5000) -> tuple[np.ndarray, float]:
    """Longer deterministic hill‑climb with gradual step decay."""
    best = h.copy()
    best_c = compute_c5(best)
    scale = 0.4
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new = perturb_h(best, rng, scale=scale)
        new_c = compute_c5(new)
        if new_c < best_c:
            best = new
            best_c = new_c
        if (it + 1) % decay_interval == 0:
            scale = max(scale * 0.85, 1e-6)
    return best, best_c


def random_binary_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a binary vector with exactly n/2 ones."""
    vec = np.zeros(n)
    vec[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return vec


def sa_binary_optimize(n: int, rng: np.random.Generator,
                       time_limit: float) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing on binary vectors (exact sum = n/2)."""
    # start from the alternating 0101… pattern
    cur = (np.arange(n) % 2).astype(float)
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    T = 0.01
    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        ones = np.flatnonzero(cur == 1)
        zeros = np.flatnonzero(cur == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        new = cur.copy()
        new[i] = 0
        new[j] = 1
        new_c = compute_c5(new)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < math.exp(-delta / T):
            cur = new
            cur_c = new_c
            if cur_c < best_c:
                best = cur.copy()
                best_c = cur_c
        it += 1
        if it % 2000 == 0:
            T = max(T * 0.7, 1e-7)
    return best, best_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    best_h = None
    best_c = np.inf
    best_n = None

    # candidate discretisations (all even)
    candidate_ns = [60, 84, 120, 168, 240, 300]
    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        if n0 not in candidate_ns:
            candidate_ns = [n0] + candidate_ns

    for n in candidate_ns:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining < 8.0:
            break

        # allocate a modest slice of the remaining budget to this size
        time_for_n = min(remaining * 0.6, max(15.0, min(300.0, n * 1.0)))

        # ---- binary simulated annealing (≈40% of the slice) ----
        time_sa = time_for_n * 0.4
        # start from the alternating pattern
        best_bin, best_bin_c = sa_binary_optimize(n, rng, time_sa * 0.5)

        # a second SA run starting from a random binary vector
        rand_start = random_binary_vector(n, rng)
        cur = rand_start.copy()
        cur_c = compute_c5(cur)
        T = 0.01
        start_sa = time.time()
        while time.time() - start_sa < time_sa * 0.5:
            ones = np.flatnonzero(cur == 1)
            zeros = np.flatnonzero(cur == 0)
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            new = cur.copy()
            new[i] = 0
            new[j] = 1
            new_c = compute_c5(new)
            delta = new_c - cur_c
            if delta < 0 or rng.random() < math.exp(-delta / T):
                cur = new
                cur_c = new_c
                if cur_c < best_bin_c:
                    best_bin = cur.copy()
                    best_bin_c = cur_c
            # very slow cooling – we only need a rough search here
            T = max(T * 0.9999, 1e-8)

        # ---- prepare starting continuous vectors ----
        starts = [(best_bin.astype(float), best_bin_c)]

        # if a parent construction of the same size exists, also try it
        if "initial_h_values" in globals() and n == len(initial_h_values):
            init_vec = np.array(initial_h_values, dtype=float)
            init_vec = np.clip(init_vec, 0.0, 1.0)
            init_c = compute_c5(init_vec)
            starts.append((init_vec, init_c))

        # ---- refinement for each start ----
        for h_start, _ in starts:
            h_cur = h_start.copy()

            # a sequence of SLSQP refinements with increasing beta
            for beta in (10.0, 50.0, 200.0, 800.0, 2000.0, 5000.0):
                if time.time() - start_time > budget_s * 0.96:
                    break
                h_cur = slsqp_optimize(h_cur, beta, rng, max_iter=800)
                c_cur = compute_c5(h_cur)
                if c_cur < best_c:
                    best_c, best_h, best_n = c_cur, h_cur.copy(), n

            # final deterministic hill‑climb
            if time.time() - start_time < budget_s * 0.98:
                h_cur, c_cur = hill_climb_fine(h_cur, rng, max_iters=4000)
                if c_cur < best_c:
                    best_c, best_h, best_n = c_cur, h_cur.copy(), n

    # fallback in the unlikely case that nothing was found
    if best_h is None:
        best_n = 60
        best_h = np.full(best_n, 0.5)
        best_c = compute_c5(best_h)

    # make sure the final vector satisfies the constraints exactly
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, best_n)
            if diff > 0:
                add = min(diff, 1.0 - best_h[i])
            else:
                add = max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
