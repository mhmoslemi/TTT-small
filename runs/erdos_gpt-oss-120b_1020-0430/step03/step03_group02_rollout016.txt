def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def worst_shift_info(h):
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr) - (h.size - 1)
    k = abs(idx)
    # produce gradient as earlier if needed
    return k, corr, idx

def gradient_move(h, rng, step_scale=0.5):
    n = h.size
    k, _, _ = worst_shift_info(h)
    # compute gradient g
    grad = np.ones_like(h)
    if k == 0:
        grad = 1 - 2*h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    i_max = np.argmax(grad)  # positive biggest
    i_min = np.argmin(grad)  # negative biggest
    # max reduce at i_max and increase at i_min
    if i_max == i_min:
        return h
    max_eps = min(h[i_max], 1 - h[i_min])
    if max_eps <= 0:
        return h
    eps = rng.random() * max_eps * step_scale
    nh = h.copy()
    nh[i_max] -= eps
    nh[i_min] += eps
    return nh

def pair_move(h, rng, step_scale=0.5):
    n = h.size
    k, _, _ = worst_shift_info(h)
    if k == 0:
        # fallback to simple gradient_transfer
        return gradient_move(h, rng, step_scale)
    # compute product for each i where both indices defined
    p = h[: n - k] * h[k:]
    # indices for i: 0..n-k-1
    i_max = np.argmax(p)
    i_min = np.argmin(p)
    # ensure capacity
    eps_cap = min(h[i_max], h[i_max + k], 1 - h[i_min], 1 - h[i_min + k])
    if eps_cap <= 0:
        # maybe try alternative indices (second best) - can skip
        return h
    eps = eps_cap * step_scale * rng.random()
    nh = h.copy()
    nh[i_max] -= eps
    nh[i_max + k] -= eps
    nh[i_min] += eps
    nh[i_min + k] += eps
    return nh

def hill_climb(h, rng, n_iter=5000, step_scale=0.5, use_pair=True):
    best = h.copy()
    best_c = compute_c5(best)
    for it in range(n_iter):
        if use_pair:
            nh = pair_move(h, rng, step_scale)
        else:
            nh = gradient_move(h, rng, step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best, best_c = nh.copy(), c
            h = nh  # continue from improved
        else:
            # maybe accept with small probability to escape
            # simple: occasional random walk
            if rng.random() < 0.001:
                h = nh
        # decaying step_scale
        if (it+1) % (n_iter // 5) == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best, best_c

import time, numpy as np

# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) == len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple random redistribution, limited iterations
    for _ in range(10000):
        if abs(diff) <= 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector in [0,1]^n with sum = n/2."""
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, rng)


def worst_shift_k(h: np.ndarray) -> int:
    """Return the shift k (>=0) attaining the maximal overlap."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    return abs(idx)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k = Σ h_i(1‑h_{i+k})."""
    n = h.size
    k = worst_shift_k(h)
    grad = np.ones_like(h)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Move ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


# ---------- optimization moves ----------
def gradient_move(h: np.ndarray, rng: np.random.Generator,
                  step_scale: float = 0.5) -> np.ndarray:
    """Single gradient‑directed transfer preserving the sum."""
    grad = worst_gradient(h)
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * step_scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def pair_move(h: np.ndarray, rng: np.random.Generator,
              step_scale: float = 0.5) -> np.ndarray:
    """Transfer ε from the worst‑product pair to the best‑product pair."""
    n = h.size
    k = worst_shift_k(h)
    if k == 0:
        return gradient_move(h, rng, step_scale)
    # product for each overlapping pair
    prod = h[: n - k] * h[k:]
    i_max = int(np.argmax(prod))
    i_min = int(np.argmin(prod))
    # capacity checks
    cap = min(h[i_max], h[i_max + k], 1.0 - h[i_min], 1.0 - h[i_min + k])
    if cap <= 0.0:
        return h.copy()
    eps = rng.random() * cap * step_scale
    nh = h.copy()
    nh[i_max] -= eps
    nh[i_max + k] -= eps
    nh[i_min] += eps
    nh[i_min + k] += eps
    return nh


def hill_climb(h: np.ndarray, rng: np.random.Generator,
               n_iter: int = 1000, step_scale: float = 0.5,
               use_pair: bool = True) -> tuple[np.ndarray, float]:
    """Local improvement using repeated transfers."""
    best_h = h.copy()
    best_c = compute_c5(best_h)
    cur_h = h.copy()
    decay = max(1, n_iter // 5)
    for it in range(n_iter):
        nh = pair_move(cur_h, rng, step_scale) if use_pair else gradient_move(cur_h, rng, step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_h, best_c = nh.copy(), c
            cur_h = nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c
