def random_feasible_vector(n, rng):
    h = rng.random(n)
    target = n/2
    # First scale to target sum: scaling factor = target / sum
    h = h * (target / h.sum())
    # Clip may affect sum, so fix with loop
    h = np.clip(h, 0, 1)
    diff = target - h.sum()
    # while diff not zero etc.
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def hill_climb(h, rng, max_iters=2000, step_init=0.5):
    best_h = h.copy()
    best_c = compute_c5(best_h)  # use fast corr?
    step = step_init

            step *= 0.9
            step = max(step, 1e-5)
    return best_h, best_c

def simulated_annealing(h, rng, deadline):
    n = len(h)
    target = n/2
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    it = 0
    # temperature schedule: start with T0 = cur_c * 0.1? Use cur_c intensity.
    T0 = max(cur_c * 0.1, 1e-4)  # maybe 0.05.
    while time.time() < deadline:
        # adapt step based on it
        step = max(0.5 * (0.9995 ** it), 1e-5)
        cand_h = perturb_h(cur_h, rng, scale=step)
        cand_c = compute_c5(cand_h)
        if cand_c < cur_c:
            cur_h, cur_c = cand_h, cand_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
        else:
            # acceptance probability
            T = T0 * (0.999 ** it) + 1e-12
            if rng.random() < np.exp(-(cand_c - cur_c) / T):
                cur_h, cur_c = cand_h, cand_c
        it += 1
    return best_h, best_c

def gradient_refine(h, rng, max_iter=20, step_start=0.02):
    n = h.size
    target = n/2.0
    step = step_start
    for it in range(max_iter):
        corr = correlation_array(h)
        idx = np.argmax(corr)  # shift index in [0,2n-2]
        shift = idx - (n - 1)  # can be negative; but correlation is symmetric? Not necessary.
        k = abs(shift)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            # For i in [0, n-k-1]:
            g[:n-k] += 1.0 - h[k:]
            # For i in [k, n-1]:
            g[k:] -= h[:n-k]
        # update
        h = h - step * g
        h = np.clip(h, 0.0, 1.0)
        # fix sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # simple proportional scaling? Use loop.
            for _ in range(200):
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
        step *= 0.8  # reduce step
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # Get parent if exists
    parent = None
    if 'initial_h_values' in globals():
        parent_array = np.array(initial_h_values, dtype=float)
        if parent_array.ndim == 1:
            parent = parent_array

    # Candidate discretizations
    candidate_ns = [96, 120, 144, 180, 240]  # balanced range, exclude tiny.

    # Iterate over n
    for n in candidate_ns:
        if time.time() - start_time > budget_s * 0.95:
            break

        # Create list of start vectors
        start_vectors = []

        # 1) parent upsample/downsample to n
        if parent is not None:
            # if parent size matches n: use directly
            if parent.size == n:
                start_vectors.append(parent.copy())
            else:
                # Interpolate
                x_src = np.linspace(0.0, 2.0, parent.size, endpoint=False)
                x_tgt = np.linspace(0.0, 2.0, n, endpoint=False)
                h_interp = np.interp(x_tgt, x_src, parent)
                start_vectors.append(h_interp)

        # 2) random feasible vectors (few)
        for _ in range(3):
            start_vectors.append(random_feasible_vector(n, rng))

        # Optionally include best so far at same n

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Correlation utilities ----------
def correlation_array(h: np.ndarray) -> np.ndarray:
    n = h.size
    N = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    corr_full = np.fft.irfft(H * np.conj(G), N)
    return corr_full[: 2 * n - 1]


def compute_c5(h: np.ndarray) -> float:
    n = h.size
    return float(np.max(correlation_array(h)) * (2.0 / n))


# ---------- Feasibility helpers ----------
def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> None:
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, h.size)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    np.clip(h, 0.0, 1.0, out=h)
    enforce_sum(h, target, rng)
    return h


# ---------- Local moves ----------
def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h0: np.ndarray, rng: np.random.Generator,
               max_iters: int = 1500, step_init: float = 0.6) -> tuple:
    h = h0.copy()
    best_h, best_c = h.copy(), compute_c5(h)
    step = step_init
    decay = max(1, max_iters // 8)
    for it in range(max_iters):
        cand = perturb_h(h, rng, scale=step)
        c = compute_c5(cand)
        if c < best_c:
            best_c, best_h, h = c, cand.copy(), cand
        if (it + 1) % decay == 0:
            step = max(step * 0.85, 1e-5)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    max_iter: int = 30, step_start: float = 0.02) -> np.ndarray:
    n = h.size
    target = n / 2.0
    step = step_start
    for _ in range(max_iter):
        corr = correlation_array(h)
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        k = abs(shift)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - step * g
        np.clip(h, 0.0, 1.0, out=h)
        enforce_sum(h, target, rng)
        step *= 0.8
    return h


def simulated_annealing(h0: np.ndarray, rng: np.random.Generator,
                        deadline: float) -> tuple:
    n = h0.size
    target = n / 2.0
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    it = 0
    T0 = max(cur_c * 0.08, 1e-5)
    while time.time() < deadline:
        step = max(0.5 * (0.9995 ** it), 5e-5)
        cand = perturb_h(cur_h, rng, scale=step)
        cand_c = compute_c5(cand)
        if cand_c < cur_c:
            cur_h, cur_c = cand, cand_c
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
        else:
            T = T0 * (0.999 ** it) + 1e-12
            if rng.random() < np.exp(-(cand_c - cur_c) / T):
                cur_h, cur_c = cand, cand_c
        it += 1
    return best_h, best_c


# ---------- SLSQP with soft‑max ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = correlation_array(x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 300) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, rng)
    return x

    candidate_ns = [96, 120, 144, 180, 240]

    for n in candidate_ns:
        if time.time() - start_time > budget_s * 0.95:
            break

        starts = []

        # parent interpolation / direct
        if parent is not None:
            if parent.size == n:
                starts.append(parent.copy())
            else:
                xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
                xt = np.linspace(0.0, 2.0, n, endpoint=False)
                starts.append(np.interp(xt, xs, parent))

        # random feasible vectors
        for _ in range(2):
            starts.append(random_feasible_vector(n, rng))

        # binary feasible
        bin_vec = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        bin_vec[idx] = 1.0
        starts.append(bin_vec)

        if best_n == n and best_h is not None:
            starts.append(best_h + 0.02 * rng.standard_normal(n))

        for h0 in starts:
            if time.time() - start_time > budget_s * 0.96:
                break

            h = np.clip(h0, 0.0, 1.0)
            enforce_sum(h, n / 2.0, rng)

            # quick hill‑climb
            h, c = hill_climb(h, rng, max_iters=1200, step_init=0.6)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

            # simulated annealing (time‑capped)
            deadline_sa = start_time + budget_s * 0.95
            if time.time() < deadline_sa:
                h_sa, c_sa = simulated_annealing(h, rng, deadline_sa)
                if c_sa < best_c:
                    best_h, best_c, best_n = h_sa.copy(), c_sa, n
                h = h_sa

            # gradient refinement
            h = gradient_refine(h, rng, max_iter=35, step_start=0.02)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

            # SLSQP smoothing sequence
            for beta in (5, 20, 80, 200, 500):
                if time.time() - start_time > budget_s * 0.97:
                    break
                h = slsqp_optimize(h, beta, rng, max_iter=250)
                c = compute_c5(h)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n

            # final long hill‑climb
            if time.time() - start_time < budget_s * 0.98:
                h, c = hill_climb(h, rng, max_iters=2500, step_init=0.4)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n

    # ----- optional up‑sampling refinement -----
    if best_h is not None and time.time() - start_time < budget_s * 0.94:
        if best_n * 2 <= 300:
            n_up = best_n * 2
            xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
            xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
            h_up = np.interp(xt, xs, best_h)
            enforce_sum(h_up, n_up / 2.0, rng)
            h_up, _ = hill_climb(h_up, rng, max_iters=1800, step_init=0.5)
            for beta in (5, 20, 80, 200):
                if time.time() - start_time > budget_s * 0.98:
                    break
                h_up = slsqp_optimize(h_up, beta, rng, max_iter=250)
            h_up = gradient_refine(h_up, rng, max_iter=30, step_start=0.02)
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, best_n = h_up, c_up, n_up

    # ----- final safety fix -----
    if best_h is None:
        best_n = 120
        best_h = np.full(best_n, 0.5)
    best_h = np.clip(best_h, 0.0, 1.0)
    enforce_sum(best_h, best_n / 2.0, rng)

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
