        total_cap = capacities.sum()
        if total_cap < diff - 1e-12:

            diff = target - h.sum()

def gradient_worst_shift(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr)
    k = idx - (n - 1)
    grad = np.ones(n)
    if k >= 0:
        grad[:n - k] -= h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[:n - kk] -= h[kk:]
        grad[kk:] -= h[:n - kk]
    # Also gradient for other shifts may be positive; but we only compute for worst shift.
    return grad, k

def enforce_sum(h, target):
    # Ensure sum equals target while staying within [0,1]
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Clip modifications iteratively
    # Use while loop limited.
    max_iter = 20
    for _ in range(max_iter):
        # Determine where we can adjust: if diff > 0 => increase those not at 1; else decrease those > 0.
        if diff > 0:
            slack = 1.0 - h
            mask = slack > 1e-12
        else:
            slack = h
            mask = slack > 1e-12
        if not np.any(mask):
            break
        # amount to add/subtract per variable
        delta = diff / mask.sum()
        # clamp delta to not exceed slack
        delta = np.clip(delta, -slack[mask], slack[mask])
        # For simplicity, uniformly apply min magnitude:
        # Actually we want to apply same delta for all mask entries but clamp after.
        # To keep simple: apply delta to all mask entries (with clipping)
        # This might not exactly use full diff, but we repeat iter.
        adjust = np.zeros_like(h)
        adjust[mask] = np.clip(delta, -slack[mask], slack[mask])
        h = h + adjust
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
    # final clip
    h = np.clip(h, 0.0, 1.0)
    # might still have small diff, adjust with random small change
    if abs(target - h.sum()) > 1e-8:
        # distribute remaining diff randomly among non-boundary entries
        diff = target - h.sum()
        mask = (h > 0) & (h < 1)
        if np.any(mask):
            idx = np.where(mask)[0]
            # add diff equally approx
            h[idx] += diff / idx.size
            h = np.clip(h, 0.0, 1.0)
    # final scaling if still off (just to be safe)
    h = np.clip(h, 0.0, 1.0)
    # If sum still off due to rounding, adjust first element
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h

def slsqp_optimize(h0, beta, max_iter=300):
    n = h0.size
    target = n / 2.0
    def sum_eq(x):
        return np.sum(x) - target
    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        # stable log-sum-exp
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    bounds = [(0.0, 1.0)] * n
    constraints = {'type': 'eq', 'fun': sum_eq}
    res = minimize(objective, h0, method='SLSQP', bounds=bounds, constraints=constraints,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    # Clip and enforce sum as safety
    x = np.clip(res.x, 0.0, 1.0)
    x = enforce_sum(x, target)
    return x

import time

from scipy.optimize import minimize

# -------------------------------------------------------------
#  Constraint handling
# -------------------------------------------------------------
def enforce_sum(h, target):
    """Adjust h (in [0,1]) so that sum(h) == target (preserving bounds)."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Increase if needed
    if diff > 0:
        caps = 1.0 - h
        total = caps.sum()
        if total < diff - 1e-12:          # cannot satisfy, fill all
            h = np.ones_like(h)
            diff = target - h.sum()
            caps = 1.0 - h
            total = caps.sum()
        inc = diff * caps / total
        inc = np.minimum(inc, caps)
        h = h + inc
    else:                                 # diff < 0, need to decrease
        caps = h
        total = caps.sum()
        if total < -diff - 1e-12:          # cannot satisfy, empty all
            h = np.zeros_like(h)
            diff = target - h.sum()
            caps = h
            total = caps.sum()
        dec = (-diff) * caps / total
        dec = np.minimum(dec, caps)
        h = h - dec
    # Residual correction (tiny)
    residual = target - h.sum()
    if abs(residual) > 1e-10:
        # adjust first non‑boundary entry if possible
        idx = 0
        if residual > 0:
            add = min(residual, 1.0 - h[idx])
            h[idx] += add
        else:
            sub = min(-residual, h[idx])
            h[idx] -= sub
    return np.clip(h, 0.0, 1.0)

def random_feasible_vector(n, rng):
    """Uniform random vector in [0,1] with exact sum n/2."""
    h = rng.random(n)
    return enforce_sum(h, n / 2.0)

def compute_c5(h):
    """Discrete C5 = max correlation * (2/N)."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)

# -------------------------------------------------------------
#  Gradient of worst shift
# -------------------------------------------------------------
def gradient_worst_shift(h):
    """Return gradient of the correlation term attaining the maximum."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr)
    k = idx - (n - 1)                      # shift index
    grad = np.ones(n)
    if k >= 0:
        grad[:n - k] -= h[k:]
        grad[k:]    -= h[:n - k]
    else:
        kk = -k
        grad[:n - kk] -= h[kk:]
        grad[kk:]     -= h[:n - kk]
    return grad, k

    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def gradient_descent_local(h, deadline, rng):
    """Drive h down the max‑correlation direction until time runs out."""
    n = h.size
    target = n / 2.0

#  SLSQP refinement using smooth max (soft‑max) surrogate

def slsqp_optimize(h0, beta, max_iter=300):
    n = h0.size
    target = n / 2.0

    def sum_eq(x):
        return np.sum(x) - target

    def softmax_obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        # stable log‑sum‑exp
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    cons = {"type": "eq", "fun": sum_eq}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )

    start = time.time()

    # ---- starting vector ------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
    else:
        n0 = 360
        h0 = random_feasible_vector(n0, rng)

    # up‑sample to a finer grid (2×)
    target_n = n0 * 2
    xs0 = np.arange(n0) * (2.0 / n0)
    xs  = np.arange(target_n) * (2.0 / target_n)
    h = np.interp(xs, xs0, h0)
    h = enforce_sum(np.clip(h, 0.0, 1.0), target_n / 2.0)

    best_h, best_c = h.copy(), compute_c5(h)

    # ---- primary gradient‑descent phase ---------------------------------
    primary_deadline = start + budget_s * 0.55
    best_h, best_c = gradient_descent_local(best_h, primary_deadline, rng)

    # ---- short random‑restart searches -----------------------------------
    while time.time() - start < budget_s * 0.78:
        # allocate a small slice (≈2 % of total budget) for each restart
        remaining = budget_s * 0.78 - (time.time() - start)
        slice_time = min(budget_s * 0.02, remaining)
        if slice_time <= 0.0:
            break
        rnd = random_feasible_vector(target_n, rng)
        deadline = time.time() + slice_time
        h_try, c_try = gradient_descent_local(rnd, deadline, rng)
        if c_try < best_c:
            best_h, best_c = h_try.copy(), c_try

    # ---- SLSQP refinement ------------------------------------------------
    for beta in [10, 30, 100, 300, 1000, 3000]:
        if time.time() - start > budget_s * 0.92:
            break
        h_ref = slsqp_optimize(best_h, beta, max_iter=400)
        c_ref = compute_c5(h_ref)
        if c_ref < best_c:
            best_h, best_c = h_ref.copy(), c_ref

    # ---- final tiny random tweaks ----------------------------------------
    while time.time() - start < budget_s * 0.99:
        new_h = perturb_random(best_h, rng, scale=0.005)
        new_h = enforce_sum(np.clip(new_h, 0.0, 1.0), target_n / 2.0)
        c_new = compute_c5(new_h)
        if c_new < best_c:
            best_h, best_c = new_h.copy(), c_new

    # ---- clean‑up ---------------------------------------------------------
    best_h = enforce_sum(np.clip(best_h, 0.0, 1.0), target_n / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
