def compute_c5_fft(h):
    n = h.size
    N = 2*n
    ha = np.fft.rfft(h, N)
    hb = np.fft.rfft(1 - h, N)
    corr = np.fft.irfft(ha * np.conj(hb), N)
    # corr length N, but we need linear correlation of length 2n-1: the first 2n-1 elements.
    corr = corr[:2*n-1]
    return np.max(corr) * (2.0 / n)

def enforce_sum(h, target, rng):
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    diff = target - total
    # if diff is negligible, just return
    if abs(diff) < 1e-12:
        return h
    # adjust vector to satisfy sum exactly
    # We can add/subtract from randomly selected non-saturating indices
    max_iter = 10000
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i])
            # scale with random factor to avoid overshoot
            add *= rng.random()
            h[i] += add
            diff -= add
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i])
            sub *= rng.random()
            h[i] -= sub
            diff += sub
    # after loop, if still diff, adjust a random element
    if abs(diff) >= 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def perturb_random(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def perturb_targeted(h, grad, rng, scale=0.2):
    pos = np.where((grad > 1e-9) & (h > 1e-12))[0]
    neg = np.where((grad < -1e-9) & (h < 1.0 - 1e-12))[0]
    if pos.size == 0 or neg.size == 0:
        return perturb_random(h, rng, scale)
    i = rng.choice(pos)
    j = rng.choice(neg)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return perturb_random(h, rng, scale)
    eps = max_eps * scale * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def compute_corr_fft(h):
    n = h.shape[0]
    N = 2 * n
    ha = np.fft.rfft(h, N)
    hb = np.fft.rfft(1.0 - h, N)
    corr = np.fft.ifft(ha * np.conj(hb), N).real
    # Take the first 2*n-1 elements for linear correlation
    return corr[: 2 * n - 1]

def compute_c5_approx(h):
    corr = compute_corr_fft(h)
    return np.max(corr) * (2.0 / h.size)

def compute_c5_exact(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def hill_climb_targeted(h_start, rng, max_iters, start_step, target_sum):
    h = h_start.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5_approx(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        corr = compute_corr_fft(h)
        idx = np.argmax(corr)
        max_corr = corr[idx]
        c_current = max_corr * (2.0 / n)
        k = idx - (n - 1)
        # gradient of worst shift
        grad = np.ones(n)
        if k >= 0:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
        else:
            kk = -k
            grad[: n - kk] -= h[kk:]
            grad[kk:] -= h[: n - kk]
        new_h = perturb_targeted(h, grad, rng, scale=step)
        # enforce constraints
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = enforce_sum(new_h, target_sum, rng)
        c_new = compute_c5_approx(new_h)
        if c_new < best_c:
            best_c = c_new
            best_h = new_h.copy()
        if c_new < c_current:
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

def sum_eq_constraint(x):

def softmax_obj(x):
    corr = compute_corr_fft(x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # use log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic feasibility utilities
# ----------------------------------------------------------------------
def enforce_sum(h, target, rng):
    """Clip entries to [0,1] and adjust to make the sum exactly target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    max_iter = 10000
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i])
            add *= rng.random()
            h[i] += add
            diff -= add
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i])
            sub *= rng.random()
            h[i] -= sub
            diff += sub
    if abs(diff) >= 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h


def random_feasible_vector(n, rng):
    """Uniform random vector in [0,1] repaired to satisfy Σh = n/2."""
    h = rng.random(n)
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, n / 2.0, rng)
    return h


# ----------------------------------------------------------------------
# Perturbations
# ----------------------------------------------------------------------
def perturb_random(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def perturb_targeted(h, grad, rng, scale=0.2):
    """Move mass from indices with positive gradient to those with negative gradient."""
    pos = np.where((grad > 1e-9) & (h > 1e-12))[0]
    neg = np.where((grad < -1e-9) & (h < 1.0 - 1e-12))[0]
    if pos.size == 0 or neg.size == 0:
        return perturb_random(h, rng, scale)
    i = rng.choice(pos)
    j = rng.choice(neg)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return perturb_random(h, rng, scale)
    eps = max_eps * scale * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


# ----------------------------------------------------------------------
# Correlation / C5 computation (fast via FFT)
# ----------------------------------------------------------------------
def compute_corr_fft(h):
    """Linear correlation of h with (1‑h) using zero‑padded FFT."""
    n = h.shape[0]
    N = 2 * n
    ha = np.fft.rfft(h, N)
    hb = np.fft.rfft(1.0 - h, N)
    corr = np.fft.ifft(ha * np.conj(hb), N).real
    return corr[: 2 * n - 1]


def compute_c5_approx(h):
    """Fast approximate C₅ (uses FFT, sufficient for internal decisions)."""
    corr = compute_corr_fft(h)
    return np.max(corr) * (2.0 / h.size)


def compute_c5_exact(h):
    """Exact C₅ as required for final reporting (uses np.correlate)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


# ----------------------------------------------------------------------
# Targeted hill climbing (fast version)
# ----------------------------------------------------------------------
def hill_climb_targeted(h_start, rng, max_iters, start_step, target_sum):
    h = h_start.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5_approx(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        corr = compute_corr_fft(h)
        idx = np.argmax(corr)
        max_corr = corr[idx]
        c_current = max_corr * (2.0 / n)
        k = idx - (n - 1)

        # gradient of the worst shift
        grad = np.ones(n)
        if k >= 0:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
        else:
            kk = -k
            grad[: n - kk] -= h[kk:]
            grad[kk:] -= h[: n - kk]

        new_h = perturb_targeted(h, grad, rng, scale=step)
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = enforce_sum(new_h, target_sum, rng)
        c_new = compute_c5_approx(new_h)

        if c_new < best_c:
            best_c = c_new
            best_h = new_h.copy()
        if c_new < c_current:
            h = new_h

        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


# ----------------------------------------------------------------------
# SLSQP refinement on a smooth approximation of max
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = compute_corr_fft(x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----- initial vector (may be supplied) -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = random_feasible_vector(180, rng)
    h0 = enforce_sum(np.clip(h0, 0.0, 1.0), h0.size / 2.0, rng)

    # ----- up‑sample to 720 points -----
    target_n = 720
    xs0 = (np.arange(h0.size) + 0.5) * (2.0 / h0.size)
    xs = (np.arange(target_n) + 0.5) * (2.0 / target_n)
    h = np.interp(xs, xs0, h0)
    h = enforce_sum(np.clip(h, 0.0, 1.0), target_n / 2.0, rng)

    best_h = h.copy()
    best_c = compute_c5_exact(best_h)

    # ----- main optimisation loop -----
    main_budget = budget_s * 0.85
    beta_schedule = [5.0, 20.0, 80.0, 200.0, 800.0, 2000.0, 5000.0]
    while time.time() - start_time < main_budget:
        # enough time for at least one more major step?
        if budget_s - (time.time() - start_time) < 30.0:
            break

        # start point: perturb current best or random fresh
        if rng.random() < 0.6:
            start = perturb_random(best_h, rng, scale=0.3)
        else:
            start = random_feasible_vector(target_n, rng)

        # targeted hill‑climb
        h_local, _ = hill_climb_targeted(
            start,
            rng,
            max_iters=5000,
            start_step=0.5,
            target_sum=target_n / 2.0,
        )
        c_local = compute_c5_exact(h_local)
        if c_local < best_c:
            best_h, best_c = h_local.copy(), c_local

        # SLSQP refinements with increasingly large β
        for beta in beta_schedule:
            if time.time() - start_time > budget_s * 0.95:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, max_iter=800)
            except Exception:
                continue
            c_ref = compute_c5_exact(h_ref)
            if c_ref < best_c:
                best_h, best_c = h_ref.copy(), c_ref

    # ----- final micro‑search (tiny random moves) -----
    while time.time() - start_time < budget_s * 0.99:
        new_h = perturb_random(best_h, rng, scale=0.02)
        c_new = compute_c5_exact(new_h)
        if c_new < best_c:
            best_h, best_c = new_h.copy(), c_new

    # ----- clean‑up -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5_exact(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
