    decay = (end_temp/start_temp)**(1/max_iters)

    best_c = compute_c5(best_h)
    cur_h = h0.copy()
    cur_c = best_c
    for it in range(max_iters):
        # propose move: similar donor/receiver but random
        # maybe move epsilon from random i to random j while staying feasible
        i = rng.integers(n)
        j = rng.integers(n)
        if i==j:
            continue
        max_eps = min(cur_h[i], 1-cur_h[j])
        if max_eps<=0:
            continue
        eps = max_eps * rng.random() * 0.5
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            cur_h = new_h
            cur_c = new_c

                best_c = cur_c
                best_h = cur_h.copy()
        temp *= decay
        # optionally break if time
    return best_h, best_c

def corr_full(h):
    n = h.size
    L = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1-h), L)
    corr = np.fft.irfft(H * G, L)[:2*n-1]
    return corr

def compute_c5(h):
    n = h.size
    corr = corr_full(h)
    return np.max(corr) * (2.0 / n)

        O_nonneg = corr[n-1:]  # length n

def improve_targeted(h0, rng, max_iters=4000, patience=500):
    n = h0.size
    h = h0.copy()
    best_c = compute_c5(h)
    no_improve = 0
    eps_min = 1e-8
    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode='full')
        O_nonneg = corr[n-1:]  # length n
        k = int(np.argmax(O_nonneg))
        # compute neighbor sum s
        if k == 0:
            s = 2 * h
        else:
            right = np.concatenate([h[k:], np.zeros(k)])
            left = np.concatenate([np.zeros(k), h[:k]])
            s = right + left
        donors = np.where((s < 1 - 1e-9) & (h > eps_min))[0]
        receivers = np.where((s > 1 + 1e-9) & (h < 1 - eps_min))[0]
        if donors.size and receivers.size:
            i = rng.choice(donors)
            j = rng.choice(receivers)
            max_eps = min(h[i], 1 - h[j])
            if max_eps > eps_min:
                eps = max_eps * rng.random() * 0.4
            else:
                eps = 0.0
        else:
            # random pair
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            max_eps = min(h[i], 1 - h[j])
            if max_eps <= eps_min:
                continue
            eps = max_eps * rng.random() * 0.3
        if eps <= eps_min:
            continue
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        # ensure within bounds
        new_h = np.clip(new_h, 0.0, 1.0)
        new_c = compute_c5(new_h)
        if new_c + 1e-12 < best_c:
            best_c = new_c
            h = new_h
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= patience:
            # random perturb to escape
            for _ in range(3):
                a, b = rng.integers(n, size=2)
                if a==b: continue
                max_eps2 = min(h[a], 1 - h[b])

                    h[a] -= eps2
                    h[b] += eps2
            h = np.clip(h, 0.0, 1.0)
            best_c = compute_c5(h)
            no_improve = 0
    return h, best_c

            j = rng.choice(receivers)
            max_eps = min(h[i], 1.0 - h[j])
            eps = max_eps * rng.random() * 0.4
        else:
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= eps_min:
                continue
            eps = max_eps * rng.random() * 0.3

        if eps <= eps_min:
            continue

        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_h = np.clip(new_h, 0.0, 1.0)

        new_c = compute_c5(new_h)
        if new_c + 1e-12 < best_c:
            best_c = new_c
            h = new_h
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= patience:
            # small random shake
            for _ in range(3):
                a, b = rng.integers(n, size=2)
                if a == b:
                    continue
                max_eps2 = min(h[a], 1.0 - h[b])
                if max_eps2 > eps_min:
                    eps2 = max_eps2 * rng.random() * 0.5
                    h[a] -= eps2
                    h[b] += eps2
            h = np.clip(h, 0.0, 1.0)
            best_c = compute_c5(h)
            no_improve = 0

    return h, best_c


# ---------- main driver ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    best_h = None
    best_c = np.inf
    best_n = None

    # work with 300 points (good compromise between accuracy and speed)
    n = 300
    target_sum = n / 2.0

    # ---- try the provided parent solution first ----
    if "initial_h_values" in globals():
        try:
            h0 = upsample_initial(n)
            h0 = project_to_feasible(h0, target_sum)
            h_loc, c_loc = improve_targeted(h0, rng, max_iters=5000, patience=400)
            # SLSQP refinement
            h_cur = h_loc
            c_cur = c_loc
            for beta in (10.0, 50.0, 200.0, 800.0, 2000.0, 8000.0):
                if time.time() - start_time > budget_s * 0.94:
                    break
                h_cur = slsqp_optimize(h_cur, beta, rng, max_iter=500)
                c_cur = compute_c5(h_cur)
                if c_cur < best_c:
                    best_c, best_h, best_n = c_cur, h_cur.copy(), n
            if c_cur < best_c:
                best_c, best_h, best_n = c_cur, h_cur.copy(), n
        except Exception:
            pass

    # ---- random restarts with quick local search ----
    while time.time() - start_time < budget_s * 0.90:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 6.0:
            break

        h0 = random_feasible_vector(n, rng)
        # allocate iterations proportional to remaining time (max 2500)
        iters = int(min(2500, remaining * 20))
        h_loc, c_loc = improve_targeted(h0, rng, max_iters=iters, patience=300)

        if c_loc < best_c:
            best_c, best_h, best_n = c_loc, h_loc.copy(), n

        # light SLSQP polishing if time permits
        if time.time() - start_time < budget_s * 0.95:
            for beta in (10.0, 100.0, 500.0):
                if time.time() - start_time > budget_s * 0.95:
                    break
                h_ref = slsqp_optimize(h_loc, beta, rng, max_iter=300)
                c_ref = compute_c5(h_ref)
                if c_ref < best_c:
                    best_c, best_h, best_n = c_ref, h_ref.copy(), n
                h_loc = h_ref

    # ---- fallback if nothing found ----
    if best_h is None:
        best_n = n
        best_h = np.full(best_n, 0.5)

    # enforce feasibility one last time
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, best_n)
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target_sum - best_h.sum()

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
