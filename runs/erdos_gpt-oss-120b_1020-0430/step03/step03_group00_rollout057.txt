def autocorr_full(h):
    n = len(h)
    n_fft = 1 << (2*n - 1).bit_length()  # next power of two
    H = np.fft.rfft(h, n=n_fft)
    Hr = np.fft.rfft(h[::-1], n=n_fft)

    # The first 2n-1 entries correspond to correlation for k = -(n-1) .. (n-1)
    return conv[:2*n-1]

def compute_c5_fast(h):
    n = h.shape[0]
    # Use real FFT to compute correlation (convolution with flipped version)
    # Zero pad to length L = next power of two >= 2n
    L = 1 << ((2*n - 1).bit_length())
    # Compute FFT of h and reversed h
    H = np.fft.rfft(h, n=L)
    Hr = np.fft.rfft(h[::-1], n=L)
    conv = np.fft.irfft(H * Hr, n=L)
    # Use first 2n-1 values
    min_S = conv[:2*n - 1].min()
    # C5 = 1 - 2*min_S/n
    return 1.0 - 2.0 * min_S / n

def perturb_random(h, rng, scale=0.2):
    n = h.size
    # select i where h[i] > 0
    i_candidates = np.where(h > 1e-12)[0]
    if i_candidates.size == 0:
        return h.copy()
    i = rng.choice(i_candidates)
    j_candidates = np.where(h < 1 - 1e-12)[0]
    if j_candidates.size == 0:
        return h.copy()
    j = rng.choice(j_candidates)
    max_eps = min(h[i], 1 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

import time
import numpy as np

# ---------- basic utilities ----------
def enforce_sum(h, target, rng):
    """Clip to [0,1] and adjust to make the sum exactly target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # simple proportional correction with random redistribution
    max_iter = 1000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:  # need to add mass
            idx = np.where(h < 1.0)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i])
            add *= rng.random()
            h[i] += add
            diff -= add
        else:      # need to remove mass
            idx = np.where(h > 0.0)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i])
            sub *= rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    # final tiny fix if needed
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    """Random vector in [0,1] with sum exactly n/2."""
    h = rng.random(n)
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, n / 2.0, rng)
    return h

def perturb_random(h, rng, scale=0.2):
    """Transfer a small amount from a random positive entry to a random free entry."""
    n = h.size
    pos = np.where(h > 1e-12)[0]
    if pos.size == 0:
        return h.copy()
    i = rng.choice(pos)
    free = np.where(h < 1 - 1e-12)[0]
    if free.size == 0:
        return h.copy()
    j = rng.choice(free)
    max_eps = min(h[i], 1 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def perturb_targeted_move(h, grad, shift, rng, scale=0.2):
    """Move mass from gradient‑positive to gradient‑negative positions."""
    pos = np.where((grad > 1e-9) & (h > 1e-12))[0]
    neg = np.where((grad < -1e-9) & (h < 1 - 1e-12))[0]
    if pos.size == 0 or neg.size == 0:
        return perturb_random(h, rng, scale)
    i = rng.choice(pos)
    j = rng.choice(neg)
    max_eps = min(h[i], 1 - h[j])
    if max_eps <= 0:
        return perturb_random(h, rng, scale)
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def upsample(h, target_n):
    """Linear interpolation of h (defined on [0,2]) to target_n samples."""
    n = h.size
    xs_old = (np.arange(n) + 0.5) * (2.0 / n)
    xs_new = (np.arange(target_n) + 0.5) * (2.0 / target_n)
    h_new = np.interp(xs_new, xs_old, h)
    return h_new

# ---------- evaluation ----------
def evaluate_state(h):
    """Return (C5, worst shift k, full auto‑correlation array S)."""
    n = h.shape[0]
    n_fft = 2 * n                     # length for FFT (zero‑padded)
    H = np.fft.rfft(h, n=n_fft)
    Hr = np.fft.rfft(h[::-1], n=n_fft)
    conv = np.fft.irfft(H * Hr, n=n_fft)
    S = conv[:2 * n - 1]              # auto‑correlation for shifts -(n-1)..(n-1)
    min_idx = int(np.argmin(S))
    min_S = float(S[min_idx])
    if min_S < 0.0:
        min_S = 0.0
    C5 = 1.0 - 2.0 * min_S / n
    shift = min_idx - (n - 1)
    return C5, shift, S

# ---------- simulated annealing ----------
def simulated_annealing(h0, end_time, rng):
    """SA that preserves the sum; returns best vector and its C5."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    cur_c, cur_shift, cur_S = evaluate_state(h)
    best_h = h.copy()
    best_c = cur_c
    sa_start = time.time()
    sa_budget = max(end_time - sa_start, 1e-12)
    T0, T_min = 0.001, 1e-6
    sc0, sc_min = 0.5, 0.001

    while True:
        now = time.time()
        if now >= end_time:
            break
        elapsed = now - sa_start
        frac = elapsed / sa_budget
        T = T0 * (1 - frac) + T_min * frac      # linear cooling
        scale = sc0 * (1 - frac) + sc_min * frac

        # gradient for the current worst shift
        k = cur_shift
        grad = np.ones(n)
        if k >= 0:
            grad[:n - k] -= h[k:]
            grad[k:]     -= h[:n - k]
        else:
            kk = -k
            grad[:n - kk] -= h[kk:]
            grad[kk:]    -= h[:n - kk]

        # propose a move
        if rng.random() < 0.7:
            new_h = perturb_targeted_move(h, grad, k, rng, scale)
        else:
            new_h = perturb_random(h, rng, scale)

        new_c, new_shift, new_S = evaluate_state(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = new_h
            cur_c = new_c
            cur_shift = new_shift
            cur_S = new_S
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    total_budget = budget_s

    # initial vector
    if 'initial_h_values' in globals():

    else:

    best_c, _, _ = evaluate_state(best_h)

    # -------- Stage 1: SA at current resolution ----------
    stage1_end = start_time + total_budget * 0.60
    if time.time() < stage1_end:
        best_h, best_c = simulated_annealing(best_h, stage1_end, rng)

    # -------- Stage 2: upscale and SA ----------
    # target resolution (at most 720)
    target_n = min(720, best_h.size * 2)
    if target_n != best_h.size and time.time() < start_time + total_budget * 0.95:
        # interpolate to finer grid
        h_fine = upsample(best_h, target_n)
        h_fine = enforce_sum(np.clip(h_fine, 0.0, 1.0), target_n / 2.0, rng)
        fine_end = start_time + total_budget * 0.95
        h_fine, c_fine = simulated_annealing(h_fine, fine_end, rng)
        if c_fine < best_c:

    # -------- Stage 3: final tiny random tweaks ----------
    final_end = start_time + total_budget * 0.99
    while time.time() < final_end:
        trial = perturb_random(best_h, rng, scale=0.02)
        c_trial, _, _ = evaluate_state(trial)
        if c_trial < best_c:

    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c, _, _ = evaluate_state(best_h)
