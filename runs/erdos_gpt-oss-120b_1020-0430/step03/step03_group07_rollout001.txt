def directed_hill_climb(h, rng, max_iters=2000, scale=0.3):
    best_h = h.copy()
    best_c = compute_c5(best_h)
    n = len(h)
    for it in range(max_iters):
        conv = correlate_full(best_h)
        idx_max = np.argmax(conv)
        offset = idx_max - (n - 1)
        # Determine possible i_max (high contribution)
        if offset >= 0:
            valid_i = np.arange(n - offset)
            contrib = best_h[valid_i] * (1 - best_h[valid_i + offset])
        else:
            offset_abs = -offset
            valid_i = np.arange(offset_abs, n)
            contrib = best_h[valid_i] * (1 - best_h[valid_i - offset_abs])
        i_max = valid_i[np.argmax(contrib)]
        # Determine j candidates where (1 - h[j+offset]) small
        # we'll try up to 5 attempts random among low contribution indices
        # compute low_contrib mask
        if offset >= 0:
            low_contrib = best_h[valid_i] * (1 - best_h[valid_i + offset])
        else:
            low_contrib = best_h[valid_i] * (1 - best_h[valid_i - offset_abs])
        # we want low product, i.e., maybe near zero.
        # pick indices where low_contrib is minimal
        # get indices sorted by low_contrib ascending
        sorted_idx = np.argsort(low_contrib)
        # try a few candidates
        found = False
        for rank in range(min(5, len(sorted_idx))):
            j = valid_i[sorted_idx[rank]]
            if j == i_max:
                continue
            # compute max transfer
            max_transfer = min(best_h[i_max], 1.0 - best_h[j])
            if max_transfer <= 1e-12:
                continue
            eps = scale * max_transfer * rng.random()
            new_h = best_h.copy()
            new_h[i_max] -= eps
            new_h[j] += eps
            c = compute_c5(new_h)
            if c < best_c:
                best_h, best_c = new_h, c
                found = True
                break
        if not found:
            # fallback to random perturb
            new_h = perturb_h(best_h, rng, scale=scale)
            c = compute_c5(new_h)
            if c < best_c:
                best_h, best_c = new_h, c
        # optional step size decay
        # could gradually reduce scale but ignore.
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# -------------------- feasibility utilities --------------------
def enforce_sum(h, target, rng):
    """Adjust h so that sum(h) == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    # simple random walk to fix the sum
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # final clean‑up (unlikely needed)
    if abs(diff) > 1e-12:
        # distribute uniformly over free capacity
        free_up = 1.0 - h
        free_down = h
        if diff > 0:
            cap = free_up.sum()
            if cap > 0:
                h += diff * free_up / cap
        else:
            cap = free_down.sum()
            if cap > 0:
                h += diff * free_down / cap
    return h

def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    h = rng.random(n)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # scale to meet the sum roughly
    cur = h.sum()
    if cur > 0:
        h *= target / cur
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, target, rng)

def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = np.clip(eps, -max_eps, max_eps)
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

# -------------------- correlation helpers --------------------
def correlate_full(h):
    """Full linear correlation of h with (1‑h) using FFT."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv

def compute_c5(h):
    """Discrete C₅ for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)

# -------------------- smooth surrogate for SLSQP --------------------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """Log‑sum‑exp smoothing of maxₖ Σᵢ hᵢ(1‑hᵢ₊ₖ)."""
    conv = correlate_full(x)
    bc = CURRENT_BETA * conv
    m = bc.max()
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=600):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-9, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # fix any tiny sum drift
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

# -------------------- directed hill‑climb --------------------
def directed_hill_climb(h, rng, max_iters=3000, scale=0.25):
    """Greedy local search focusing on the worst lag."""
    best_h = h.copy()
    best_c = compute_c5(best_h)
    n = best_h.size
    for it in range(max_iters):
        conv = correlate_full(best_h)
        idx_max = int(conv.argmax())
        offset = idx_max - (n - 1)  # shift index (positive → right)
        # valid indices for this offset
        if offset >= 0:
            vi = np.arange(n - offset)
            contrib = best_h[vi] * (1 - best_h[vi + offset])
        else:
            off = -offset
            vi = np.arange(off, n)
            contrib = best_h[vi] * (1 - best_h[vi - off])
        i_max = vi[int(contrib.argmax())]

        # pick a low‑contribution partner j
        low_order = np.argsort(contrib)  # ascending
        moved = False
        for rank in low_order[:5]:
            j = vi[rank]
            if j == i_max:
                continue
            max_transfer = min(best_h[i_max], 1.0 - best_h[j])
            if max_transfer <= 1e-12:
                continue
            eps = scale * max_transfer * rng.random()
            new_h = best_h.copy()
            new_h[i_max] -= eps
            new_h[j] += eps
            new_c = compute_c5(new_h)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
                moved = True
                break
        if not moved:
            # fallback random perturb
            new_h = perturb_h(best_h, rng, scale=scale)
            new_c = compute_c5(new_h)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
    return best_h, best_c

# -------------------- simulated annealing --------------------
def simulated_annealing(start_h, rng, deadline, T0=0.1, Tf=1e-5, max_iters=2500):
    cur_h = start_h.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        new_h = perturb_h(cur_h, rng, scale=0.4)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
    return best_h, best_c
