import time
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Global placeholders for the smooth surrogate used by SLSQP
# --------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    """Smooth log‑sum‑exp surrogate of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


# --------------------------------------------------------------
# Helper utilities (identical to the previous solution)
# --------------------------------------------------------------
def compute_c5(h):
    """Exact discrete C₅ for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng):
    """Adjust h so that Σh = target while 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-12:
            continue
        h[i] += add
        diff = target - h.sum()
    # final tiny correction if needed
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(1000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            if abs(add) < 1e-12:
                continue
            h[i] += add
            diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    v = rng.exponential(1.0, size=n)
    v = v / v.sum() * (n / 2.0)
    h = np.clip(v, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def perturb_h(h, rng, scale=0.5):
    """Transfer a small amount between two coordinates (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    while j == i:
        j = rng.integers(0, n)
    max_pos = min(h[i], 1.0 - h[j])
    max_neg = min(h[j], 1.0 - h[i])
    if max_pos + max_neg == 0:
        return h.copy()
    eps = (rng.random() * (max_pos + max_neg)) - max_neg
    eps *= scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return np.clip(new_h, 0.0, 1.0)


def hill_climb(h_start, rng, max_iters=2500, start_step=0.5):
    """Simple hill‑climb: keep only improving moves."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)
        max_pos = min(h[i], 1.0 - h[j])
        max_neg = min(h[j], 1.0 - h[i])
        if max_pos + max_neg == 0:
            continue
        eps = (rng.random() * (max_pos + max_neg)) - max_neg
        eps *= step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            if abs(add) < 1e-12:
                continue
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_limit = budget_s

    # --------------------------------------------------------------
    # Determine dimension and initial vector
    # --------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 180
        h0 = np.full(n, 0.5)
        h0 = enforce_sum(h0, n / 2.0, rng)

    target = n / 2.0
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx

    # --------------------------------------------------------------
    # Build a diverse pool of starting points
    # --------------------------------------------------------------
    candidates = [h0]

    # smooth sin/cos patterns
    for f in range(1, 4):
        for fun in (np.sin, np.cos):
            h = 0.5 + 0.5 * fun(np.pi * f * xs)
            h = np.clip(h, 0.0, 1.0)
            candidates.append(enforce_sum(h, target, rng))

    # a few random feasible vectors
    for _ in range(8):
        candidates.append(random_feasible_vector(n, rng))

    # --------------------------------------------------------------
    # Add Sturmian (Beatty) sequences parametrised by phi
    # --------------------------------------------------------------
    for _ in range(3000):
        phi = rng.random()
        b = ((np.arange(n) * phi) % 1) < 0.5
        b = b.astype(float)
        diff = int(target - b.sum())
        if diff > 0:
            zero_idx = np.where(b == 0)[0]
            if zero_idx.size >= diff:
                chosen = rng.choice(zero_idx, size=diff, replace=False)
                b[chosen] = 1.0
        elif diff < 0:
            one_idx = np.where(b == 1)[0]
            if one_idx.size >= -diff:
                chosen = rng.choice(one_idx, size=-diff, replace=False)
                b[chosen] = 0.0
        candidates.append(b)

    # --------------------------------------------------------------
    # Pick the best among the pool
    # --------------------------------------------------------------
    best_h = None
    best_c = float("inf")
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    global_best_h = best_h
    global_best_c = best_c

    # --------------------------------------------------------------
    # Main optimisation loop (hill‑climb + SLSQP)
    # --------------------------------------------------------------
    beta_list = [10.0, 40.0, 120.0, 400.0, 800.0]
    while time.time() - start_time < time_limit * 0.86:
        # Choose a start point
        if rng.random() < 0.7:
            start = perturb_h(global_best_h, rng, scale=0.3)
        else:
            start = random_feasible_vector(n, rng)

        # Local hill‑climb
        h_opt, c_opt = hill_climb(start, rng, max_iters=3000, start_step=0.5)

        # If we improved, try SLSQP with increasing beta
        if c_opt < global_best_c:
            for beta in beta_list:
                if time.time() - start_time > time_limit * 0.86:
                    break
                try:
                    h_opt = slsqp_optimize(h_opt, beta, max_iter=500)
                except Exception:
                    continue
                c_opt = compute_c5(h_opt)
                if c_opt < global_best_c:
                    global_best_h, global_best_c = h_opt.copy(), c_opt

        if c_opt < global_best_c:
            global_best_h, global_best_c = h_opt.copy(), c_opt

        # Occasionally perform a finer hill‑climb
        if time.time() - start_time < time_limit * 0.70 and c_opt < global_best_c:
            h_fine, c_fine = hill_climb(global_best_h, rng,
                                        max_iters=5000, start_step=0.2)
            if c_fine < global_best_c:
                global_best_h, global_best_c = h_fine.copy(), c_fine

    # Optional up‑sampling and refinement (if time permits)
    # --------------------------------------------------------------
    if time.time() - start_time < time_limit * 0.94 and n * 2 <= 360:

        n_up = h_up.size
        target_up = n_up / 2.0
        h_up = enforce_sum(h_up, target_up, rng)

        # refine on the finer grid
        h_up_opt, c_up_opt = hill_climb(h_up, rng,
                                        max_iters=5000, start_step=0.3)
        for beta in beta_list:
            if time.time() - start_time > time_limit:
                break
            try:
                h_up_opt = slsqp_optimize(h_up_opt, beta, max_iter=500)
            except Exception:
                pass
            c_up_opt = compute_c5(h_up_opt)
            if c_up_opt < global_best_c:
                global_best_h, global_best_c, n = h_up_opt.copy(), c_up_opt, n_up

        if time.time() - start_time < time_limit * 0.99:
            h_fine_up, c_fine_up = hill_climb(global_best_h, rng,
                                              max_iters=5000, start_step=0.2)
            if c_fine_up < global_best_c:
                global_best_h, global_best_c = h_fine_up.copy(), c_fine_up

    # --------------------------------------------------------------
    # Final clean‑up (clip + exact sum enforcement) and return
    # --------------------------------------------------------------
    global_best_h = np.clip(global_best_h, 0.0, 1.0)
    global_best_h = enforce_sum(global_best_h, n / 2.0, rng)
    final_c = compute_c5(global_best_h)

    return global_best_h.tolist(), float(final_c), int(n)
