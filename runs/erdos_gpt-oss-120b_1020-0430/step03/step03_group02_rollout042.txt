def proj_capped(v, target, lower=0.0, upper=1.0, eps=1e-12):
    # solve min ||x - v||^2 s.t. sum x = target, lower <= x <= upper
    # can be done via bisection on λ as above.
    # Define function f(λ) = sum_i clip(v_i - λ, lower, upper) - target.
    # f is decreasing in λ.
    # Find λ where f(λ)=0.
    # Initialise λ_low = min(v) - upper, λ_high = max(v) - lower.
    low = np.min(v) - upper
    high = np.max(v) - lower
    # ensure f(low) >= 0, f(high) <= 0 (since decreasing).
    # compute f at bounds
    def f(lam):
        return np.clip(v - lam, lower, upper).sum() - target
    # adjust if necessary
    # sometimes target may be out of feasible range: ensure feasible.
    # Since target = n/2 <= n*upper and >= n*lower, feasible.
    # Might need enlarge interval.
    while f(low) < 0:
        low -= 1.0
    while f(high) > 0:
        high += 1.0
    for _ in range(50):
        mid = (low + high) / 2.0
        fm = f(mid)
        if fm > 0:
            low = mid
        else:
            high = mid
    x = np.clip(v - (low+high)/2.0, lower, upper)
    # final correction to ensure sum close to target (tiny rounding)
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        # distribute diff to entries not at bounds
        free = (x > lower+1e-12) & (x < upper-1e-12)
        if free.any():
            x[free] += diff / free.sum()
        else:
            # just adjust all (should not happen)
            x += diff / len(x)
    return x

def pgd_optimize(h0, beta_schedule, target, max_iter_per_beta, rng, alpha0=0.01):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    for beta_idx, beta in enumerate(beta_schedule):
        alpha = alpha0
        for it in range(max_iter_per_beta):
            # compute O
            corr = np.correlate(h, 1 - h, mode='full')
            O = corr[h.size - 1: h.size - 1 + h.size]  # length n
            O_max = O.max()
            w = np.exp(beta * (O - O_max))
            W = w.sum()
            # FFT conv
            # compute w_rev
            w_rev = np.concatenate([w[:1], w[:0:-1]])  # w[0], then w[n-1...1]
            # FFT
            h_fft = np.fft.rfft(h)
            w_fft = np.fft.rfft(w)
            w_rev_fft = np.fft.rfft(w_rev)
            conv = np.fft.irfft(w_fft * h_fft, n=h.size)
            conv_rev = np.fft.irfft(w_rev_fft * h_fft, n=h.size)
            grad = W - conv - conv_rev

            # Adam update

def softmax_obj_and_grad(h, beta):
    n = h.size
    # correlation of h with (1 - h)
    corr = np.correlate(h, 1 - h, mode='full')
    O = corr[n-1: n-1 + n]
    Omax = O.max()
    # compute softmax weights
    w = np.exp(beta * (O - Omax))
    Z = w.sum()
    # gradient
    pre = np.cumsum(w)
    const_term = pre[::-1]  # length n
    # forward correlation w with h
    fwd = np.correlate(w, h, mode='full')[n-1: n-1 + n]
    back = np.convolve(w, h, mode='full')[:n]
    grad = const_term - fwd - back
    # compute objective (softmax):
    obj = (np.log(Z) + beta * Omax) / beta  # since log(sum exp(beta O)) = log(Z) + beta Omax
    return obj, grad

def proj_capped(v, target, lower=0.0, upper=1.0):
    # want to solve min ||x - v||^2 s.t. sum x = target, lower <= x <= upper
    # Equivalent to find lambda such that sum clip(v - lambda) = target
    # Use bisection.
    lo = np.min(v) - upper
    hi = np.max(v) - lower
    # define function f(lambda)
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, lower, upper)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, lower, upper)
    # final adjustment tiny diff
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        # distribute among non-boundary elements
        free = (x > lower + 1e-12) & (x < upper - 1e-12)
        if free.any():
            x[free] += diff / free.sum()
        else:
            # fallback: adjust all uniformly within bounds
            x += diff / len(x)
            x = np.clip(x, lower, upper)
    return x

    pre = np.cumsum(w)
    const = pre[::-1]  # length n
    # forward correlation w * h
    fwd_full = np.correlate(w, h, mode='full')
    fwd = fwd_full[n-1: n-1 + n]
    # backward convolution w * h
    back_full = np.convolve(w, h, mode='full')
    back = back_full[:n]
    grad = const - fwd - back
    # objective (softmax)
    obj = (np.log(Z) + beta * Omax) / beta
    return obj, grad

Double-check dims: w size n, h size n. np.correlate(w,h,mode='full') returns length 2n-1. For positive shift j, we get Σ_i w_i * h_{i + j} (i from 0 to n-1-j). That's what we need for forward contributions. We're slicing from n-1 to n-1+n = n-1 to 2n-2 inclusive, which yields index j = 0..n-1. Good.

The backward contributions using convolution: np.convolve(w, h, mode='full') yields Σ_i w_i * h_{j - i} for each j. For j = 0..n-1, this indeed yields Σ_{i=0}^j w_i * h_{j - i}. That's our backward contributions.

Thus gradient correct.

import time
import numpy as np

# -------------------------------------------------------------
# basic utilities
# -------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2] (linear correlation)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def proj_capped(v: np.ndarray, target: float, lo: float = 0.0, hi: float = 1.0) -> np.ndarray:
    """Projection onto the box [lo,hi] with an exact sum constraint."""
    # find λ such that sum clip(v-λ) = target; the function is monotone decreasing.
    left = np.min(v) - hi
    right = np.max(v) - lo
    for _ in range(30):
        mid = (left + right) * 0.5
        x = np.clip(v - mid, lo, hi)
        s = x.sum()
        if s > target:
            left = mid
        else:
            right = mid
    x = np.clip(v - (left + right) * 0.5, lo, hi)
    # tiny correction for floating point drift
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = (x > lo + 1e-12) & (x < hi - 1e-12)
        if free.any():
            x[free] += diff / free.sum()
        else:
            x += diff / len(x)
        x = np.clip(x, lo, hi)
    return x


# -------------------------------------------------------------
# gradient of the soft‑max (smoothed max) objective
# -------------------------------------------------------------
def softmax_obj_and_grad(h: np.ndarray, beta: float):
    """Return smooth max value and its gradient w.r.t. h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    O = corr[n - 1 : n - 1 + n]                 # O_k for k = 0..n‑1
    Omax = O.max()
    w = np.exp(beta * (O - Omax))               # soft‑max weights, max weight = 1
    Z = w.sum()

    # constant part: sum_{k ≤ n‑1‑j} w_k
    pref = np.cumsum(w)
    const = pref[::-1]                          # length n

    # forward contributions: Σ_{k≤n‑1‑j} w_k * h_{j+k}
    fwd_full = np.correlate(w, h, mode="full")
    fwd = fwd_full[n - 1 : n - 1 + n]

    # backward contributions: Σ_{k≤j} w_k * h_{j‑k}
    back_full = np.convolve(w, h, mode="full")
    back = back_full[:n]

    grad = const - fwd - back
    obj = (np.log(Z) + beta * Omax) / beta
    return obj, grad


# -------------------------------------------------------------
# Adam‑style projected gradient optimizer
# -------------------------------------------------------------
def optimize_h(h0: np.ndarray, target_sum: float, rng, start_time: float, budget: float):
    """Adam with projection on the capped simplex."""
    h = h0.copy()
    n = h.size
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0
    betas = [5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0]

    for beta in betas:
        # learning rate scaled roughly as 1/√beta
        lr = 0.08 / (np.sqrt(beta) + 1e-12)
        max_it = 350
        for _ in range(max_it):
            if time.time() - start_time > budget * 0.96:
                return h
            _, grad = softmax_obj_and_grad(h, beta)
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            h = proj_capped(h - step, target_sum)

        # occasional jitter to escape tiny plateaus
        if beta != betas[-1]:
            h = proj_capped(h + rng.normal(scale=0.01, size=n), target_sum)

    return h


# -------------------------------------------------------------
# simple hill‑climb used for final fine‑tuning
# -------------------------------------------------------------
def worst_gradient(h: np.ndarray):
    """Gradient of the worst (max) overlap O_k = Σ h_i(1‑h_{i+k}) (both directions)."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]          # subtract h_{i+k}
        grad[k:]       -= h[: n - k]    # subtract h_{i‑k}
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray, rng, scale: float = 0.5):
    """Move a little mass from a positive‑gradient entry to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h_start: np.ndarray, rng, max_iters: int = 2000, step_scale: float = 0.3):
    """Hill‑climb driven by the worst‑shift gradient."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c = float("inf")
    best_n = None

    # -----------------------------------------------------------------
    # start from the parent construction if it exists
    # -----------------------------------------------------------------
    if "initial_h_values" in globals():

        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0, c0, h0.size

    # -----------------------------------------------------------------
    # candidate discretisations (prefer the one from the parent)
    # -----------------------------------------------------------------
    cand = [96, 108, 120, 144, 168, 192, 216, 240]
    if best_n is not None:
        cand = [best_n] + [n for n in cand if n != best_n]

    # -----------------------------------------------------------------
    # optimisation loop over discretisations
    # -----------------------------------------------------------------
    for n in cand:
        if time.time() - start > budget_s * 0.94:
            break
        target = n / 2.0

        # initialise h for this n
        if best_n == n and best_h is not None:
            h = best_h.copy()
        elif best_h is not None:
            # linear interpolation of the best known solution
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h = np.interp(x_new, x_old, best_h)
            h = proj_capped(h, target)
        else:
            h = rng.random(n)
            h = proj_capped(h, target)

        # -------------------------------------------------------------
        # a few random restarts + Adam optimisation
        # -------------------------------------------------------------
        restarts = 3
        for r in range(restarts):
            if time.time() - start > budget_s * 0.92:
                break
            if r > 0:
                # perturbed start around the best known vector
                h0 = np.clip(rng.normal(loc=best_h, scale=0.07, size=n), 0.0, 1.0)
                h0 = proj_capped(h0, target)
                h = h0

            h_opt = optimize_h(h, target, rng, start, budget_s)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c:
                best_h, best_c, best_n = h_opt.copy(), c_opt, n

        # -------------------------------------------------------------
        # short guided hill‑climb for extra polishing
        # -------------------------------------------------------------
        if best_h is not None and time.time() - start < budget_s * 0.90:
            h, c = guided_hill_climb(best_h, rng, max_iters=1500, step_scale=0.25)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

    # -----------------------------------------------------------------
    # final random fine‑tuning until time runs out
    # -----------------------------------------------------------------
    while best_h is not None and time.time() - start < budget_s * 0.998:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # -----------------------------------------------------------------
    # sanity check and return
    # -----------------------------------------------------------------
    if best_h is None:
        best_n = 120
        best_h = np.full(best_n, 0.5)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = proj_capped(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
