def worst_shift_improvement(h, rng, max_iter, step=0.5, eps_max=0.02):
   # step factor for epsilon relative to allowed max
   n = h.size
   best = h.copy()
   best_c5 = compute_c5(best)
   for it in range(max_iter):
       # compute autocorr
       c = np.correlate(h, h, mode='full')
       s = int(np.argmin(c) - (n-1))
       # compute L = forward + backward
       if s >= 0:
           forward = np.concatenate([h[s:], np.zeros(s)])
           backward = np.concatenate([np.zeros(s), h[:n-s]])
       else:
           ss = -s
           forward = np.concatenate([np.zeros(ss), h[:n-ss]])
           backward = np.concatenate([h[ss:], np.zeros(ss)])
       L = forward + backward
       # identify low and high L indices
       lo_thr = np.quantile(L, 0.2)
       hi_thr = np.quantile(L, 0.8)
       low = np.where(L <= lo_thr)[0]
       high = np.where(L >= hi_thr)[0]
       if low.size == 0 or high.size == 0:
           # fallback random perturb
           i, j = rng.integers(0, n, size=2)
       else:
           i = rng.choice(low)
           j = rng.choice(high)
       if i == j:
           continue
       max_eps = min(h[i], 1.0 - h[j])
       if max_eps <= 0:
           continue
       eps = max_eps * step * rng.random()  # small step relative to max
       # ensure eps not too tiny
       if eps < 1e-12:
           continue
       new = h.copy()
       new[i] -= eps
       new[j] += eps
       c5 = compute_c5(new)
       if c5 < best_c5 - 1e-12:
           best_c5 = c5
           best = new
           h = new
           # early exit if below target
           if best_c5 < 0.3808:
               break
       else:
           # occasionally accept worse with small probability for diversification
           if rng.random() < 0.001:
               h = new
   return best, best_c5

import time, math, numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h):
    """Discrete C5 for step function h (dx = 2/n)."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / h.size)


def enforce_sum(h, target, rng):
    """Adjust h in‑place so that sum(h)=target while 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    # simple random fill‑in
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # final tiny correction if needed
    if abs(diff) > 1e-9:
        slack = np.where((diff > 0) & (h < 1.0))[0] if diff > 0 else np.where((diff < 0) & (h > 0.0))[0]
        if slack.size:
            inc = diff / slack.size
            inc = np.clip(inc, -np.min(h[slack]), 1.0 - np.max(h[slack]))
            h[slack] += inc
    return h


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h≤1, sum = n/2)."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    enforce_sum(h, n / 2.0, rng)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer ε between two coordinates (preserves sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new


# ---------- SLSQP surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=200):
    """SLSQP on smooth approximation of max correlation."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {"type": "eq", "fun": sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x
