import time
import numpy as np

# -------------------------------------------------------------
# Helper: exact C5 evaluation for a discrete step function
# -------------------------------------------------------------
def compute_c5(h):
    """Exact discrete C₅ for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


# -------------------------------------------------------------
# Projection onto [0,1] with a prescribed sum (linear equality)
# -------------------------------------------------------------
def project_onto_sum_bounds(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """
    Find lambda such that sum_i clip(v_i + lambda, lower, upper) = target.
    The sum is monotone in lambda, we solve by bisection.
    """
    lo = -np.max(v - lower) - 1.0   # guarantees all clipped to lower -> sum = lower*n
    hi = np.max(upper - v) + 1.0    # guarantees all clipped to upper -> sum = upper*n
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        proj = np.clip(v + mid, lower, upper)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            hi = mid
        else:
            lo = mid
    # final return
    return np.clip(v + (lo + hi) * 0.5, lower, upper)

# Random feasible vector (continuous, respects bounds and sum)
# -------------------------------------------------------------
def random_feasible_vector(n, rng):
    v = rng.random(n)
    return project_onto_sum_bounds(v, n / 2.0)

# Gradient of the smooth max (log-sum-exp) surrogate
# -------------------------------------------------------------
def gradient_softmax(h, beta):
    """
    Gradient of (1/beta) * log sum exp(beta * corr) where
    corr = np.correlate(h, 1-h, mode='full').
    """
    n = h.size
    b = 1.0 - h
    corr = np.correlate(h, b, mode="full")
    # stable softmax weights
    max_c = np.max(corr)
    exp_vals = np.exp(beta * (corr - max_c))
    w = exp_vals / exp_vals.sum()          # length 2n-1
    grad = np.zeros_like(h)

    # shifts from -(n-1) .. (n-1)
    for d in range(-(n - 1), n):
        w_d = w[d + (n - 1)]
        if w_d == 0.0:
            continue
        if d >= 0:
            # add b[i+d] to grad[i] for i=0..n-d-1
            grad[: n - d] += w_d * b[d:]
            # subtract h[i] from grad[i+d] for i=0..n-d-1
            grad[d:] -= w_d * h[: n - d]
        else:
            t = -d
            # add b[i] to grad[i+t] for i=0..n-t-1
            grad[t:] += w_d * b[: n - t]
            # subtract h[i+t] from grad[i] for i=0..n-t-1
            grad[: n - t] -= w_d * h[t:]
    return grad

                if it % 80 == 0:
                    c_val = compute_c5(h_new)
                    if c_val < best_c:
                        best_c = c_val
                        best_h = h_new.copy()
                h = h_new

        # a short hill‑climb style fine‑tuning (pairwise transfers)
        for _ in range(250):
            if time.time() - start_time > budget_s * 0.96:
                break
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # max feasible epsilon to keep bounds
            max_pos = min(h[i], 1.0 - h[j])
            max_neg = min(h[j], 1.0 - h[i])
            if max_pos + max_neg == 0.0:
                continue
            eps = (rng.random() * (max_pos + max_neg)) - max_neg
            eps *= 0.3
            cand = h.copy()
            cand[i] -= eps
            cand[j] += eps
            cand = np.clip(cand, 0.0, 1.0)
            cand = project_onto_sum_bounds(cand, target)
            c_val = compute_c5(cand)
            if c_val < best_c:
                best_c = c_val
                best_h = cand.copy()
                h = cand

    # ---------------------------------------------------------
    # Optional up‑sampling to a finer grid (if time permits)
    # ---------------------------------------------------------
    if time.time() - start_time < budget_s * 0.94 and n * 2 <= 360:
        h_up = np.repeat(best_h, 2)
        n_up = h_up.size
        target_up = n_up / 2.0
        h_up = project_onto_sum_bounds(h_up, target_up)

        # refine on the finer grid with the same schedule
        for beta in beta_schedule:
            step = 0.5 / (1.0 + np.sqrt(beta))
            iters = max(1, int(800 / len(beta_schedule)))
            for it in range(iters):
                if time.time() - start_time > budget_s * 0.99:
                    break
                grad = gradient_softmax(h_up, beta)
                h_up = project_onto_sum_bounds(h_up - step * grad, target_up)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_c = c_up
            best_h = h_up
            n = n_up
            target = target_up

    # ---------------------------------------------------------
    # Final clean‑up
    # ---------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_sum_bounds(best_h, target)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
