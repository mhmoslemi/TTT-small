def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    # try up to 10 * n iterations
    for _ in range(10 * n):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-12:
            continue
        h[i] += add
        diff = target - h.sum()
    # clip final
    return np.clip(h, 0.0, 1.0)

def perturb_swap(h, rng, scale=1.0):
    # Choose i,j
    n=h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n-1)
    if j >= i:
        j+=1
    # Determine max amount we can transfer from i to j while staying within [0,1]
    max_transfer = min(h[i], 1.0 - h[j])
    # also can transfer opposite direction
    max_transfer_neg = min(h[j], 1.0 - h[i])
    total_range = max_transfer + max_transfer_neg
    if total_range <= 0:
        return h.copy()
    eps = (rng.random() * total_range) - max_transfer_neg
    eps *= scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    # Clip small deviations
    new_h = np.clip(new_h, 0.0, 1.0)
    return new_h

def anneal(h_initial, rng, budget, time_start, n):
    best_h = h_initial.copy()
    best_c = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_c = best_c
    T_initial = 0.5  # magnitude for C5 reductions maybe
    # Let's schedule exponential cooling.
    # Determine number of steps based on budget and time consumption.
    # We'll loop while time < budget * 0.8
    while time.time() - time_start < budget * 0.8:
        # Temperature schedule: T = T_initial * (0.99 ** steps)
        # We'll track steps count.
        # approximate step count
        # We'll compute a simple schedule using elapsed fraction.
        frac = (time.time() - time_start) / (budget * 0.8)
        T = T_initial * (0.01 ** frac)  # exponential decrease from T_initial -> T_initial*0.01
        # propose new
        cand_h = perturb_swap(cur_h, rng, scale=0.5)
        cand_c = compute_c5(cand_h)
        if cand_c < cur_c or rng.random() < np.exp(-(cand_c - cur_c) / max(T, 1e-6)):
            cur_h = cand_h
            cur_c = cand_c
            if cur_c < best_c:
                best_h = cur_h.copy()
                best_c = cur_c
    return best_h, best_c

def l_bfgs_optimize(h0, beta, target, rng, max_iter=500):
    # define objective
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        softmax = (m + np.log(np.sum(np.exp(bc - m)))) / beta
        # penalty
        penalty = 1e3 * (np.sum(x) - target) ** 2
        return softmax + penalty
    bounds = [(0.0, 1.0)] * len(h0)
    res = minimize(obj, h0, method='L-BFGS-B', bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-12})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum exactly
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # adjust using random indices as before
        x = enforce_sum(x, target, rng)
    return x

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global placeholders for the smooth surrogate used by SLSQP
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    """Smooth log‑sum‑exp surrogate of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


# ----------------------------------------------------------------------
# Problem‑specific utilities
# ----------------------------------------------------------------------
def compute_c5(h):
    """Exact discrete C₅ for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng):
    """Adjust h so that Σh = target while 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(10 * n):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-12:
            continue
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    v = rng.exponential(1.0, size=n)
    v = v / v.sum() * (n / 2.0)
    h = np.clip(v, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def perturb_swap(h, rng, scale=0.5):
    """Swap a small amount between two coordinates (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if j >= i:
        j += 1
    max_pos = min(h[i], 1.0 - h[j])
    max_neg = min(h[j], 1.0 - h[i])
    total = max_pos + max_neg
    if total <= 0.0:
        return h.copy()
    eps = (rng.random() * total) - max_neg
    eps *= scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return np.clip(new_h, 0.0, 1.0)


def hill_climb(h_start, rng, max_iters=2500, start_step=0.5):
    """Simple hill‑climb: keep only improving moves."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        max_pos = min(h[i], 1.0 - h[j])
        max_neg = min(h[j], 1.0 - h[i])
        total = max_pos + max_neg
        if total == 0.0:
            continue
        eps = (rng.random() * total) - max_neg
        eps *= step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x
