def compute_c5_fft(h):
    n = h.size
    # pad to length N = 2*n for linear convolution
    N = 2*n
    H = np.fft.rfft(np.concatenate([h, np.zeros(n)]), n=N)

import time, numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)

def enforce_sum(h, target, rng, max_iter=2000):
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        # adjust on index i only, clamp
        add = np.clip(diff, -h[i], 1.0 - h[i])
        if abs(add) < 1e-12:
            continue
        h[i] += add
        diff = target - h.sum()
    # if still not exact, do small corrections
    if abs(diff) > 1e-12:
        for _ in range(500):
            i = rng.integers(0, n)
            add = np.clip(diff, -h[i], 1.0 - h[i])
            if abs(add) < 1e-12:
                continue
            h[i] += add
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

def random_feasible_vector(n, rng):
    v = rng.exponential(1.0, size=n)
    v = v / v.sum() * (n / 2.0)
    h = np.clip(v, 0.0, 1.0)
    return enforce_sum(h, n/2.0, rng)

def perturb_h(h, rng, scale=0.5):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n-1)
    if j >= i:
        j += 1
    # maximum epsilon that can be transferred from i to j
    max_pos = min(h[i], 1.0 - h[j])
    max_neg = min(h[j], 1.0 - h[i])
    if max_pos + max_neg == 0:
        return h.copy()
    eps = (rng.random() * (max_pos + max_neg)) - max_neg
    eps *= scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    # clip just in case numeric errors
    new_h = np.clip(new_h, 0.0, 1.0)
    return new_h

def hill_climb(h_start, rng, max_iters=5000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        max_pos = min(h[i], 1.0 - h[j])
        max_neg = min(h[j], 1.0 - h[i])
        if max_pos + max_neg == 0:
            continue
        eps = (rng.random() * (max_pos + max_neg)) - max_neg
        eps *= step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_h = np.clip(new_h, 0.0, 1.0)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def _softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=800):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(_softmax_obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons, options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            if abs(add) < 1e-12:
                continue
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def upsample(h, new_n):
    n = h.size
    # linear interpolation using midpoints positions
    old_x = (np.arange(n) + 0.5) * (2.0 / n)
    new_x = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    return np.interp(new_x, old_x, h)

    start = time.time()

def compute_c5(h):
    """Exact discrete C₅ for a step function h (dx = 2/len(h))."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=2000):
    """Adjust h so that Σh = target while 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = np.clip(diff, -h[i], 1.0 - h[i])
        if abs(add) < 1e-12:
            continue
        h[i] += add
        diff = target - h.sum()
    # tiny cleanup if still off
    if abs(diff) > 1e-12:

            i = rng.integers(0, n)
            add = np.clip(diff, -h[i], 1.0 - h[i])

                continue
            h[i] += add
            diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    v = rng.exponential(1.0, size=n)
    v = v / v.sum() * (n / 2.0)
    h = np.clip(v, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def perturb_h(h, rng, scale=0.5):
    """Transfer a small amount between two coordinates (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if j >= i:
        j += 1
    max_pos = min(h[i], 1.0 - h[j])
    max_neg = min(h[j], 1.0 - h[i])
    if max_pos + max_neg == 0:
        return h.copy()
    eps = (rng.random() * (max_pos + max_neg)) - max_neg
    eps *= scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return np.clip(new_h, 0.0, 1.0)


def hill_climb(h_start, rng, max_iters=5000, start_step=0.5):
    """Greedy hill‑climb: keep only improving moves, with step decay."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        max_pos = min(h[i], 1.0 - h[j])
        max_neg = min(h[j], 1.0 - h[i])
        if max_pos + max_neg == 0:
            continue
        eps = (rng.random() * (max_pos + max_neg)) - max_neg
        eps *= step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_h = np.clip(new_h, 0.0, 1.0)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=800):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):

            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            if abs(add) < 1e-12:
                continue
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
