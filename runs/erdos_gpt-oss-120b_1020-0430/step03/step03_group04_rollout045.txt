```python
def adjust_sum(h, target, rng, max_iter=5000):
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) <= 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h
```

```python
def compute_c5(h):

    return float(np.max(corr) * (2.0 / n))
```

def gradient_descent(h, rng, step=0.1, max_iter=30):
    n = h.size
    target = n/2.0
    for it in range(max_iter):
        # compute correlation, find worst shift
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)  # shift index (negative to positive)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]  # (1 - h_{i+k})
            g[k:] -= h[:upper]        # -h_{i-k}
        # step with decaying factor
        step_sz = step * (0.9 ** it)
        h = h - step_sz * g
        # clip and adjust sum
        h = np.clip(h, 0.0, 1.0)
        # adjust sum if needed
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h = adjust_sum(h, target, rng)
    return h

    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n/2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)]*n
    cons = {'type':'eq', 'fun': sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': max_iter, 'ftol':1e-12, 'disp':False})
    x = np.clip(res.x,0,1)
    # exact sum correction:
    diff = CURRENT_TARGET - x.sum()
    if abs(diff)>1e-12:
        x = adjust_sum(x, CURRENT_TARGET, rng)
    return x

import time
import numpy as np
from scipy.optimize import minimize

# ------------------- utilities ------------------- #
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h):
    """Discrete C5 for step function h on [0,2] with uniform spacing."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum(h, target, rng, max_iter=5000):
    """Move mass between coordinates until sum(h) == target (within tolerance)."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) <= 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Uniform random vector in [0,1] with exact sum n/2."""
    h = rng.random(n)
    target = n / 2.0
    scale = target / h.sum()
    h *= scale
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h = adjust_sum(h, target, rng)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=3000, start_step=0.6):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


def gradient_descent(h, rng, step=0.1, max_iter=30):
    """Gradient descent on the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.9 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h = adjust_sum(h, target, rng)
    return h


def simulated_annealing(h0, rng, deadline):
    """Simple SA (preserves sum) with exponential cooling."""
    n = h0.size
    target = n / 2.0
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    T0 = max(0.01 * cur_c, 1e-4)
    it = 0
    while time.time() < deadline and it < 20000:
        step = max(0.5 * (0.999 ** it), 5e-4)
        cand_h = perturb_h(cur_h, rng, scale=step)
        cand_c = compute_c5(cand_h)
        accept = False
        if cand_c < cur_c:
            accept = True
        else:
            T = T0 * (0.999 ** it) + 1e-12
            if rng.random() < np.exp(-(cand_c - cur_c) / T):
                accept = True
        if accept:
            cur_h, cur_c = cand_h, cand_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
        it += 1
    return best_h, best_c


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=400):
    """SLSQP on smooth max‑approximation."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = adjust_sum(x, CURRENT_TARGET, rng)
    return x

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # Parent solution if provided
    parent = None
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)

    # Candidate discretisations (moderate sizes)
    candidate_ns = [120, 150, 180, 210, 240, 300]
    if parent is not None:
        candidate_ns = [parent.size] + [n for n in candidate_ns if n != parent.size]

    for n in candidate_ns:
        if time.time() - start_time > budget_s * 0.97:
            break
        target = n / 2.0
        starts = []

        # 1) parent interpolated
        if parent is not None:
            xs_src = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xs_tgt = np.linspace(0.0, 2.0, n, endpoint=False)
            starts.append(np.interp(xs_tgt, xs_src, parent))

        # 2) random feasible
        starts.append(random_feasible_vector(n, rng))

        # 3) binary feasible
        h_bin = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h_bin[idx] = 1.0
        starts.append(h_bin)

        # 4) perturbation of current best (if same n)
        if best_n == n and best_h is not None:
            starts.append(best_h + 0.02 * rng.standard_normal(n))

        for h0 in starts:
            # enforce feasibility
            h = np.clip(h0, 0.0, 1.0)
            diff = target - h.sum()
            if abs(diff) > 1e-12:
                h = adjust_sum(h, target, rng)

            # Phase A: long hill‑climb
            h, c = hill_climb(h, rng, max_iters=5000, start_step=0.8)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

            # Phase B: aggressive gradient descent
            h = gradient_descent(h, rng, step=0.2, max_iter=30)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

            # Phase C: SLSQP continuation
            for beta in (10.0, 50.0, 200.0, 500.0, 1000.0, 2000.0):
                if time.time() - start_time > budget_s * 0.96:
                    break
                h = slsqp_optimize(h, beta, rng, max_iter=800)
                c = compute_c5(h)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n

            # Phase D: simulated annealing (time‑capped)
            deadline = start_time + budget_s * 0.98
            if time.time() < deadline:
                h_sa, c_sa = simulated_annealing(h, rng, deadline)
                if c_sa < best_c:
                    best_h, best_c, best_n = h_sa.copy(), c_sa, n
                h = h_sa

            # Phase E: final refinement hill‑climb
            if time.time() - start_time < budget_s * 0.99:
                h, c = hill_climb(h, rng, max_iters=3000, start_step=0.5)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n

            if time.time() - start_time > budget_s * 0.99:
                break

    # ------------------- up‑sampling refinement ------------------- #
    if best_h is not None and time.time() - start_time < budget_s * 0.99:
        if best_n * 2 <= 400:
            n_up = best_n * 2
            xs_src = np.linspace(0.0, 2.0, best_n, endpoint=False)
            xs_up = np.linspace(0.0, 2.0, n_up, endpoint=False)
            h_up = np.interp(xs_up, xs_src, best_h)
            h_up = adjust_sum(np.clip(h_up, 0.0, 1.0), n_up / 2.0, rng)

            h_up, c_up = hill_climb(h_up, rng, max_iters=4000, start_step=0.6)
            for beta in (10.0, 50.0, 200.0, 500.0, 1000.0):
                if time.time() - start_time > budget_s * 0.995:
                    break
                h_up = slsqp_optimize(h_up, beta, rng, max_iter=600)
            h_up = gradient_descent(h_up, rng, step=0.1, max_iter=20)
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, best_n = h_up, c_up, n_up

    # ------------------- final safety checks ------------------- #
    if best_h is None:
        best_n = 100
        best_h = np.full(best_n, 0.5)

    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        best_h = adjust_sum(best_h, target, rng)

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
