def enforce_sum(h, target, rng, max_iter=10000):
    # Compute diff, adjust by random indices
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            # we need to increase sum, add to h[i] if possible
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # if still not exact, distribute evenly
    if abs(diff) > 1e-12:
        # adjust smallest values maybe
        # fallback uniform distribution
        correction = diff / n
        h += correction
        # clip again & recompute diff
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # final pass
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def compute_c5(h, dx=None):

def simulated_annealing(h0, rng, time_budget):
    start = time.time()
    n = h0.size
    dx = 2.0 / n
    target = n / 2.0
    h_curr = enforce_sum(h0, target, rng)
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, dx)
    curr_c = best_c
    T = 0.02  # initial temperature
    T_end = 1e-8
    cooling = 0.99995
    iteration = 0
    offset = None
    while True:
        elapsed = time.time() - start
        if elapsed > time_budget:
            break
        # compute scale from temperature
        scale = max(0.001, min(1.0, T * 1.5))  # adjust factor
        # decide move type
        if iteration % 250 == 0:
            # compute offset for directed move
            corr = np.correlate(h_curr, 1.0 - h_curr, mode='full')
            lag_idx = np.argmax(corr)
            offset = lag_idx - (n - 1)
            h_new = directed_move_from_offset(h_curr, rng, offset, scale)
        else:
            h_new = transfer_move(h_curr, rng, scale)
        # compute new cost
        c_new = compute_c5(h_new, dx)
        delta = c_new - curr_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h_curr = h_new
            curr_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        # cooling
        T = max(T_end, T * cooling)
        iteration += 1
        # occasional sum correction
        if iteration % 5000 == 0:
            diff = target - h_curr.sum()
            if abs(diff) > 1e-9:
                # adjust random index
                i = rng.integers(0, n)
                # adjust within bounds
                if diff > 0:
                    add = min(diff, 1 - h_curr[i])
                else:
                    add = max(diff, -h_curr[i])
                h_curr[i] += add
                curr_c = compute_c5(h_curr, dx)  # recompute after correction
    return best_h, best_c

import time
import numpy as np

# ------------------------------------------------------------
# Core utilities
# ------------------------------------------------------------
def compute_c5(h, dx=None):
    """Exact discrete C5 for step function h (dx = 2 / len(h))."""
    if dx is None:
        dx = 2.0 / h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h so that sum(h) == target while keeping 0 ≤ h ≤ 1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # tiny correction if needed
    if abs(diff) > 1e-12:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
    return h


def random_feasible_continuous(n, rng):
    """Generate a random vector in [0,1] with sum = n/2."""
    h = rng.random(n)
    target = n / 2.0
    h = enforce_sum(h, target, rng)
    return h


def transfer_move(h, rng, scale):
    """Move a random amount from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    while j == i:
        j = rng.integers(0, n)

    # pick direction uniformly
    if rng.random() < 0.5:
        max_move = min(h[i], 1.0 - h[j])
        if max_move > 0:
            eps = rng.random() * max_move * scale
            new_h = h.copy()
            new_h[i] -= eps
            new_h[j] += eps
            return new_h
        # fallback opposite direction
        max_move = min(h[j], 1.0 - h[i])
        if max_move > 0:
            eps = rng.random() * max_move * scale
            new_h = h.copy()
            new_h[j] -= eps
            new_h[i] += eps
            return new_h
        return h.copy()
    else:
        max_move = min(h[j], 1.0 - h[i])
        if max_move > 0:
            eps = rng.random() * max_move * scale
            new_h = h.copy()
            new_h[j] -= eps
            new_h[i] += eps
            return new_h
        max_move = min(h[i], 1.0 - h[j])
        if max_move > 0:
            eps = rng.random() * max_move * scale
            new_h = h.copy()
            new_h[i] -= eps
            new_h[j] += eps
            return new_h
        return h.copy()


def directed_move_from_offset(h, rng, offset, scale):
    """Transfer mass along the lag that currently attains the maximum."""
    n = h.size
    if offset >= 0:
        i = rng.integers(0, n - offset)
        j = i + offset
    else:
        i = rng.integers(-offset, n)
        j = i + offset

    # try to reduce the term h[i]*(1-h[j])
    if h[i] > 0 and h[j] < 1:
        max_move = min(h[i], 1.0 - h[j])
        if max_move > 0:
            eps = rng.random() * max_move * scale
            new_h = h.copy()
            new_h[i] -= eps
            new_h[j] += eps
            return new_h

    # otherwise move the opposite way
    if h[j] > 0 and h[i] < 1:
        max_move = min(h[j], 1.0 - h[i])
        if max_move > 0:
            eps = rng.random() * max_move * scale
            new_h = h.copy()
            new_h[j] -= eps
            new_h[i] += eps
            return new_h

    # fallback
    return transfer_move(h, rng, scale)

def simulated_annealing(h0, rng, time_budget):
    """Simple SA using transfer moves and occasional directed moves."""
    start = time.time()
    n = h0.size
    dx = 2.0 / n
    target = n / 2.0
    h_curr = enforce_sum(h0, target, rng)
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, dx)
    curr_c = best_c

    T = 0.02                     # initial temperature
    T_end = 1e-8
    cooling = 0.99995
    it = 0
    offset = None

    while True:
        if time.time() - start > time_budget:
            break

        # temperature‑dependent step size
        scale = max(0.001, min(1.0, T * 1.5))

        # every few iterations try a directed move
        if it % 250 == 0:
            corr = np.correlate(h_curr, 1.0 - h_curr, mode="full")
            lag_idx = np.argmax(corr)
            offset = lag_idx - (n - 1)
            h_new = directed_move_from_offset(h_curr, rng, offset, scale)
        else:
            h_new = transfer_move(h_curr, rng, scale)

        c_new = compute_c5(h_new, dx)
        delta = c_new - curr_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h_curr = h_new
            curr_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()

        T = max(T_end, T * cooling)
        it += 1

        # occasional tiny sum‑drift correction
        if it % 5000 == 0:
            diff = target - h_curr.sum()
            if abs(diff) > 1e-9:
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h_curr[i]) if diff > 0 else max(diff, -h_curr[i])
                h_curr[i] += add
                curr_c = compute_c5(h_curr, dx)

    return best_h, best_c


def hill_climb_time_limited(h0, rng, time_budget):
    """Greedy hill‑climb with decreasing step size, limited by time."""
    start = time.time()
    n = h0.size
    dx = 2.0 / n
    h = h0.copy()
    best_c = compute_c5(h, dx)
    step = 0.5
    it = 0
    while True:
        if time.time() - start > time_budget:
            break
        h_new = transfer_move(h, rng, step)
        c_new = compute_c5(h_new, dx)
        if c_new < best_c:
            h = h_new
            best_c = c_new
        it += 1
        if it % 800 == 0:
            step = max(step * 0.7, 0.001)
    return h, best_c


def upsample_h(h, target_n, rng):
    """Linear interpolation to a finer grid, followed by sum‑repair."""
    n = len(h)
    old_x = (np.arange(n) + 0.5) * (2.0 / n)
    new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
    h_new = np.interp(new_x, old_x, h)
    h_new = enforce_sum(h_new, target_n / 2.0, rng)
    return h_new


def initial_pool(n, rng):
    """Create a modest pool of diverse feasible vectors."""
    pool = []

    # 1. possibly provided construction
    try:
        init = np.array(initial_h_values, dtype=float)
        if init.size == n:
            init = np.clip(init, 0.0, 1.0)
            init = enforce_sum(init, n / 2.0, rng)
            pool.append(init)
    except NameError:
        pass

    # 2. random continuous
    pool.append(random_feasible_continuous(n, rng))

    # 3. sinusoidal patterns
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    for f in (1, 2):
        h = 0.5 + 0.5 * np.sin(np.pi * f * xs)
        h = np.clip(h, 0.0, 1.0)
        pool.append(enforce_sum(h, n / 2.0, rng))
        h = 0.5 + 0.5 * np.cos(np.pi * f * xs)
        h = np.clip(h, 0.0, 1.0)
        pool.append(enforce_sum(h, n / 2.0, rng))

    # 4. binary masks
    mask = np.zeros(n)
    mask[: n // 2] = 1.0
    pool.append(enforce_sum(mask, n / 2.0, rng))
    mask = np.zeros(n)
    start = n // 4
    mask[start : start + n // 2] = 1.0
    pool.append(enforce_sum(mask, n / 2.0, rng))
    mask = np.arange(n) % 2
    pool.append(enforce_sum(mask, n / 2.0, rng))

    return pool

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # --------------------------------------------------------
    # 1. Choose base resolution and initial candidate
    # --------------------------------------------------------
    base_n = 360
    pool = initial_pool(base_n, rng)
    best_h = None
    best_c = float("inf")
    for h in pool:
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    n = base_n

    # --------------------------------------------------------
    # 2. Main simulated‑annealing phase (≈70 % of budget)
    # --------------------------------------------------------
    elapsed = time.time() - start_time
    sa_budget = max(0.0, budget_s * 0.70 - elapsed)
    best_h, best_c = simulated_annealing(best_h, rng, sa_budget)

    # --------------------------------------------------------
    # 3. Optional up‑sampling and refinement if time permits
    # --------------------------------------------------------
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed

    # Up‑sample to at most 720 points when we have enough time
    if remaining > budget_s * 0.15:
        target_n = min(720, n * 2)
        if target_n != n:
            best_h = upsample_h(best_h, target_n, rng)
            n = target_n
            # a second SA on the finer grid
            elapsed = time.time() - start_time
            sa2_budget = max(0.0, budget_s * 0.20 - elapsed)
            if sa2_budget > 1.0:
                best_h, best_c = simulated_annealing(best_h, rng, sa2_budget)

    # --------------------------------------------------------
    # 4. Final greedy hill‑climb using remaining time
    # --------------------------------------------------------
    elapsed = time.time() - start_time
    final_budget = max(0.0, budget_s - elapsed)
    if final_budget > 1.0:
        best_h, best_c = hill_climb_time_limited(best_h, rng, final_budget)

    # --------------------------------------------------------
    # 5. Clean‑up (clip + exact sum)
    # --------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
