def project_onto_box_sum(v, target):
    # clip to [0,1]
    v = np.clip(v, 0.0, 1.0)
    # adjust sum via shift λ: we need to find λ such that sum(clip(v + λ,0,1)) = target.
    # Because after clipping, contribution saturates at boundaries.
    # This requires solving piecewise linear; we can use bisection on λ.
    lo = -1.0
    hi = 1.0
    # broaden bounds: shift may be up or down up to 1
    for _ in range(50):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

Note: The monotonic property: As λ increases, sum of clipped increases. Indeed because adding λ raises values (subject to cap). So bisection works.

def gradient_maxlag(h):
    n = len(h)
    # compute convolution using FFT for speed
    # correlation array same as compute_c5 but without scaling
    # we need the index of max.
    # we can use np.correlate (O(n^2)), but for n up to 400 OK. Use that for simplicity.
    corr = np.correlate(h, 1 - h, mode='full')

    g = np.ones_like(h)
    if k >= 0:

def perturb_h(h, rng, scale=0.5):
    n = len(h)
    i, j = rng.integers(0, n, size=2)
    while j == i:
        j = rng.integers(0, n)
    max_pos = min(h[i], 1 - h[j])  # max we can move from i to j
    max_neg = min(h[j], 1 - h[i])  # max we can move from j to i (i gets from j)
    total = max_pos + max_neg
    if total <= 0:
        return h.copy()
    eps = (rng.random() * total) - max_neg
    eps *= scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    new_h = np.clip(new_h, 0.0, 1.0)
    # after clipping sum may drift slightly; project onto sum again
    new_h = project_onto_box_sum(new_h, n/2.0)  # but moving small, sum stays basically.
    return new_h

    a = h

def simple_hill_climb(h, rng, iters=1000, step=0.2):
    best_h = h.copy()
    best_c = compute_c5_fast(best_h)
    n = len(h)
    for _ in range(iters):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_pos = min(best_h[i], 1 - best_h[j])
        max_neg = min(best_h[j], 1 - best_h[i])
        total = max_pos + max_neg
        if total == 0:
            continue
        eps = (rng.random() * total) - max_neg
        eps *= step
        new_h = best_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_h = np.clip(new_h, 0.0, 1.0)
        # enforce sum
        new_h = project_onto_box_sum(new_h, n/2.0)
        c = compute_c5_fast(new_h)
        if c < best_c:
            best_h, best_c = new_h, c
    return best_h, best_c

def enforce_sum(v, target):
    """Project v onto [0,1]^n and adjust to have exact sum = target."""
    w = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        w_mid = np.clip(w + mid, 0.0, 1.0)
        s = w_mid.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    w = np.clip(w + (lo + hi) * 0.5, 0.0, 1.0)

    diff = target - w.sum()

        for i in range(n):

                w[i] += add

            elif diff < 0 and w[i] > 0.0:
                sub = min(-diff, w[i])
                w[i] -= sub
                diff += sub
    return w


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    v = rng.exponential(1.0, size=n)
    v = v / v.sum() * (n / 2.0)
    return enforce_sum(v, n / 2.0)


def perturb_h(h, rng, scale=0.5):
    """Transfer a small amount between two coordinates (preserves sum)."""
    n = len(h)
    i, j = rng.integers(0, n, size=2)
    while j == i:
        j = rng.integers(0, n)
    max_pos = min(h[i], 1.0 - h[j])
    max_neg = min(h[j], 1.0 - h[i])
    tot = max_pos + max_neg
    if tot == 0.0:
        return h.copy()
    eps = (rng.random() * tot) - max_neg
    eps *= scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return enforce_sum(new_h, n / 2.0)


def compute_c5(h):
    """Exact discrete C5 using linear correlation (np.correlate)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def compute_c5_fast(h):
    """Fast C5 evaluation using FFT (same result up to floating error)."""
    n = h.shape[0]
    L = 2 * n
    A = np.rfft(np.concatenate([h, np.zeros(n)]), n=L)
    B = np.rfft(np.concatenate([1 - h, np.zeros(n)]), n=L)
    conv = np.irfft(A * np.conj(B), n=L)
    max_corr = np.max(conv[: 2 * n - 1])
    return max_corr * (2.0 / n)


def corr_fft(h):
    """Linear convolution h * (1-h) via FFT."""
    n = h.shape[0]
    L = 2 * n
    A = np.rfft(np.concatenate([h, np.zeros(n)]), n=L)
    B = np.rfft(np.concatenate([1 - h, np.zeros(n)]), n=L)
    conv = np.irfft(A * np.conj(B), n=L)
    return conv[: 2 * n - 1]

    stall = 0

def simple_hill_climb(h0, rng, iters=2000, step=0.2):
    """Greedy hill climb using pairwise mass transfers."""
    n = len(h0)
    best_h = h0.copy()
    best_c = compute_c5_fast(best_h)

    for _ in range(iters):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_pos = min(best_h[i], 1.0 - best_h[j])
        max_neg = min(best_h[j], 1.0 - best_h[i])
        tot = max_pos + max_neg
        if tot == 0.0:
            continue
        eps = (rng.random() * tot) - max_neg
        eps *= step
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        cand = enforce_sum(cand, n / 2.0)
        c = compute_c5_fast(cand)
        if c < best_c:
            best_h, best_c = cand, c
    return best_h, best_c


def binary_round(h):
    """Force h to a 0‑1 vector (with possible fractional remainder)."""
    n = len(h)
    target = n / 2.0
    whole = int(np.floor(target))
    frac = target - whole
    idx = np.argsort(-h)  # descending
    hb = np.zeros_like(h)
    hb[idx[:whole]] = 1.0
    if frac > 1e-12:
        hb[idx[whole]] = frac
    return hb


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Base dimension and initial vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        base_n = len(initial_h_values)
        base_h = np.array(initial_h_values, dtype=float)
        base_h = enforce_sum(base_h, base_n / 2.0)
    else:
        base_n = 180
        base_h = np.full(base_n, 0.5)
        base_h = enforce_sum(base_h, base_n / 2.0)

    # ------------------------------------------------------------------
    # Candidate dimensions (up to about 400 points)
    # ------------------------------------------------------------------
    n_list = [base_n]
    for f in (1.2, 1.5, 2.0):
        n = int(round(base_n * f))
        if n <= 400:
            n_list.append(n)
    n_list = sorted(set(n_list))

    global_best_h = None
    global_best_c = float("inf")
    global_best_n = None

    # ------------------------------------------------------------------
    # Optimisation per dimension
    # ------------------------------------------------------------------
    for n in n_list:
        if time.time() - start > budget_s * 0.95:
            break

        # build starting vectors for this n
        if n == base_n:
            h_base = base_h.copy()
        else:
            # simple linear interpolation from the base function
            xs_old = np.linspace(0, 2, base_n, endpoint=False) + 1.0 / base_n
            xs_new = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
            h_base = np.interp(xs_new, xs_old, base_h)
            h_base += rng.normal(scale=0.02, size=n)  # small noise
            h_base = enforce_sum(h_base, n / 2.0)

        starts = [h_base]
        for _ in range(2):
            starts.append(random_feasible_vector(n, rng))

        for h0 in starts:
            if time.time() - start > budget_s * 0.94:
                break

            # gradient descent
            h_opt, c_opt = gradient_descend_optimize(
                h0, rng, n, budget_s, start
            )

            # optional fine‑grained hill climb
            if time.time() - start < budget_s * 0.90:
                h_opt, c_opt = simple_hill_climb(h_opt, rng, iters=3000, step=0.15)

            # try binary rounding (often improves C5)
            hb = binary_round(h_opt)
            c_bin = compute_c5_fast(hb)
            if c_bin < c_opt:
                h_opt, c_opt = hb, c_bin

            if c_opt < global_best_c:
                global_best_h, global_best_c, global_best_n = (
                    h_opt.copy(),
                    c_opt,
                    n,
                )
                # early exit if we beat the target bound
                if global_best_c < 0.3808:
                    break
        if global_best_c < 0.3808:
            break

    # ------------------------------------------------------------------
    # Final clean‑up and exact evaluation
    # ------------------------------------------------------------------
    if global_best_h is None:
        global_best_h = base_h.copy()
        global_best_n = base_n
        global_best_c = compute_c5(global_best_h)

    # ensure exact sum and box constraints
    global_best_h = enforce_sum(global_best_h, global_best_n / 2.0)
    final_c = compute_c5(global_best_h)

    return global_best_h.tolist(), float(final_c), int(global_best_n)
