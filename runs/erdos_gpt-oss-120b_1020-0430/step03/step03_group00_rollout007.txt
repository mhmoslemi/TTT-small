def enforce_sum(h, target, rng):
    h = np.clip(h, 0., 1.)
    diff = target - h.sum()
    # if diff near zero, return h
    # else distribute diff uniformly among entries that are not at bound
    # use loop to avoid infinite.
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-12) & (h < 1-1e-12)
        nfree = np.count_nonzero(free)
        if nfree == 0:
            # adjust any random index
            idx = rng.integers(0, h.size)
            h[idx] = np.clip(h[idx] + diff, 0., 1.)
            break
        # distribute diff equally:
        delta = diff / nfree
        # but limit delta by bounds
        add = np.minimum(delta, 1 - h[free])
        sub = np.maximum(delta, -h[free])
        # Since delta could be positive or negative, we need to adjust accordingly.
        # Actually we need to add delta but limited by bounds. Compute allowed.
        allowed = np.where(delta > 0,
                           np.minimum(delta, 1 - h[free]),
                           np.maximum(delta, -h[free]))
        h[free] += allowed
        diff = target - h.sum()
    # final fix if still diff
    if abs(diff) > 1e-9:
        idx = rng.integers(0, h.size)
        h[idx] = np.clip(h[idx] + diff, 0., 1.)
    return h

def hill_climb_targeted(h_start, rng, max_iters=1500, start_step=0.5, target_sum=None):
    h = h_start.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr)
        max_corr = corr[idx]
        c_current = max_corr * (2.0 / n)
        k = idx - (n - 1)
        # gradient of worst shift (similar as previous)
        grad = np.ones(n)
        if k >= 0:
            grad[:n - k] -= h[k:]
            grad[k:] -= h[:n - k]
        else:
            kk = -k
            grad[:n - kk] -= h[kk:]
            grad[kk:] -= h[:n - kk]
        # propose perturb_targeted: transfer from positive grad to negative grad
        pos = np.where((grad > 1e-9) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-9) & (h < 1 - 1e-12))[0]
        if pos.size > 0 and neg.size > 0:
            i = rng.choice(pos)
            j = rng.choice(neg)
            max_eps = min(h[i], 1 - h[j])
            eps = max_eps * step * rng.random()
            new_h = h.copy()
            new_h[i] -= eps
            new_h[j] += eps
        else:
            # fallback random transfer in direction of gradient sign random
            new_h = perturb_random_transfer(h, rng, scale=step)
        # evaluate
        corr_new = np.correlate(new_h, 1 - new_h, mode='full')
        c_new = np.max(corr_new) * (2.0 / n)
        if c_new < best_c:
            best_c = c_new
            best_h = new_h.copy()
        if c_new < c_current:
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    # after loop, ensure sum
    best_h = enforce_sum(best_h, target_sum if target_sum is not None else n/2., rng)
    return best_h, best_c

def sum_eq_constraint(x):

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # compute log-sum-exp in stable way
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0., 1.)
    x = enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utility functions ----------
def enforce_sum(h, target, rng):
    """Clip to [0,1] and adjust to make sum exactly target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # distribute diff among non‑boundary entries
    for _ in range(1000):
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nfree = np.count_nonzero(free)
        if nfree == 0:
            break
        delta = diff / nfree
        # limit delta by bounds
        add = np.where(delta > 0,
                       np.minimum(delta, 1.0 - h[free]),
                       np.maximum(delta, -h[free]))
        h[free] += add
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
    if abs(diff) >= 1e-9:  # final tiny correction
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    """Uniform random vector repaired to satisfy Σh = n/2."""
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)

def upsample(h, n_target):
    """Linear interpolation from current length to n_target."""
    n0 = h.size
    xs0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    xs = (np.arange(n_target) + 0.5) * (2.0 / n_target)
    return np.interp(xs, xs0, h)

def compute_c5(h):
    """Discrete C₅ for a vector h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

# ---------- perturbations ----------
def perturb_random_transfer(h, rng, scale=0.2):
    """Transfer ε from i to j (preserves sum)."""
    pos = np.where(h > 1e-12)[0]
    neg = np.where(h < 1.0 - 1e-12)[0]
    if pos.size == 0 or neg.size == 0:
        return h.copy()
    i = rng.choice(pos)
    j = rng.choice(neg)
    max_eps = min(h[i], 1.0 - h[j])
    eps = max_eps * scale * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def perturb_direction(h, rng, scale=0.1):
    """Small move along a zero‑mean random direction."""
    n = h.size
    d = rng.standard_normal(n)
    d -= d.mean()
    pos = d > 0
    neg = d < 0
    step_pos = np.min((1.0 - h[pos]) / d[pos]) if np.any(pos) else np.inf
    step_neg = np.max((-h[neg]) / d[neg]) if np.any(neg) else -np.inf
    # feasible interval [step_neg, step_pos]
    if step_pos == np.inf and step_neg == -np.inf:
        return h.copy()
    # symmetric bound
    s_max = min(step_pos, -step_neg) if step_pos != np.inf and step_neg != -np.inf else (step_pos if step_neg == -np.inf else -step_neg)
    if s_max <= 0:
        return h.copy()
    s = (rng.random() * 2.0 - 1.0) * s_max * scale
    new_h = h + s * d
    new_h = np.clip(new_h, 0.0, 1.0)
    # Preserve sum exactly (original sum = target)
    target = np.sum(h)
    new_h = enforce_sum(new_h, target, rng)
    return new_h

# ---------- local optimisation ----------
def hill_climb_targeted(h_start, rng, max_iters=1500, start_step=0.5, target_sum=None):
    """Hill‑climb using gradient of worst shift."""
    h = h_start.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(best_h)
    step = start_step
    decay_interval = max(1, max_iters // 10)

    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr)
        max_corr = corr[idx]
        c_current = max_corr * (2.0 / n)
        k = idx - (n - 1)

        # gradient of the worst shift
        grad = np.ones(n)
        if k >= 0:
            grad[: n - k] -= h[k:]
            grad[k :] -= h[: n - k]
        else:
            kk = -k
            grad[: n - kk] -= h[kk:]
            grad[kk :] -= h[: n - kk]

        # targeted transfer from +grad to -grad positions
        pos = np.where((grad > 1e-9) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-9) & (h < 1.0 - 1e-12))[0]
        if pos.size and neg.size:
            i = rng.choice(pos)
            j = rng.choice(neg)
            max_eps = min(h[i], 1.0 - h[j])
            eps = max_eps * step * rng.random()
            new_h = h.copy()
            new_h[i] -= eps
            new_h[j] += eps
        else:
            new_h = perturb_random_transfer(h, rng, scale=step)

        c_new = compute_c5(new_h)

        if c_new < best_c:
            best_c = c_new
            best_h = new_h.copy()
        if c_new < c_current:
            h = new_h

        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)

    if target_sum is not None:
        best_h = enforce_sum(best_h, target_sum, rng)
    return best_h, best_c

# ---------- SLSQP surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    total_budget = budget_s

    # ----- initialise best vector -----
    desired_n = 720
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size != desired_n:
            h0 = upsample(h0, desired_n)
    else:
        h0 = random_feasible_vector(desired_n, rng)

    h0 = enforce_sum(np.clip(h0, 0.0, 1.0), desired_n / 2.0, rng)
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ----- phase 1: intensive hill‑climbing -----
    phase1_end = start_time + total_budget * 0.92
    while time.time() < phase1_end:
        # choose start point
        if rng.random() < 0.55:
            start = perturb_direction(best_h, rng, scale=0.25)
        else:
            start = random_feasible_vector(desired_n, rng)

        # hill‑climb
        h_loc, c_loc = hill_climb_targeted(
            start,
            rng,
            max_iters=2500,
            start_step=0.6,
            target_sum=desired_n / 2.0,
        )
        if c_loc < best_c:
            best_h, best_c = h_loc, c_loc

    # ----- phase 2: SLSQP refinement -----
    beta_schedule = [5.0, 20.0, 80.0, 200.0, 800.0, 2000.0, 5000.0]
    phase2_end = start_time + total_budget * 0.98
    for beta in beta_schedule:
        if time.time() >= phase2_end:
            break
        try:
            h_ref = slsqp_optimize(best_h, beta, max_iter=800)
        except Exception:
            continue
        c_ref = compute_c5(h_ref)
        if c_ref < best_c:
            best_h, best_c = h_ref, c_ref

    # ----- phase 3: simulated‑annealing style micro search -----
    phase3_end = start_time + total_budget * 0.999
    temperature = 0.15
    while time.time() < phase3_end:
        cand = perturb_direction(best_h, rng, scale=0.05)
        c_cand = compute_c5(cand)
        delta = c_cand - best_c
        if delta < 0 or rng.random() < np.exp(-delta / (temperature + 1e-12)):
            best_h, best_c = cand, c_cand
        # very slow cooling
        if rng.random() < 0.01:
            temperature = max(temperature * 0.99, 1e-4)

    # ----- final clean‑up -----
    best_h = enforce_sum(np.clip(best_h, 0.0, 1.0), desired_n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
