def slsqp_optimize_grad(h0, beta, max_iter=200):
    n = h0.size
    target_sum = n/2.0
    bounds = [(0.0, 1.0)] * n
    eq_cons = {"type": "eq", "fun": lambda h: np.sum(h) - target_sum,
               "jac": lambda h: np.ones_like(h)}
    def fun(h):
        corr = np.correlate(h, 1 - h, mode='full') * (2.0/n)  # dx factor
        max_val = np.max(beta * corr)
        # compute logsumexp
        lse = max_val + np.log(np.sum(np.exp(beta * corr - max_val)))
        return lse / beta
    def jac(h):
        # compute corr, w, conv as above
        dx = 2.0 / n
        corr = np.correlate(h, 1 - h, mode='full') * dx
        # compute normalized weights
        max_beta_corr = np.max(beta * corr)
        exp_vals = np.exp(beta * corr - max_beta_corr)
        w = exp_vals / np.sum(exp_vals)
        conv1 = np.convolve(w, h, mode='full')
        conv2 = np.convolve(w[::-1], h, mode='full')
        offset = n - 1
        grad = dx * (1.0 - conv1[offset:offset+n] - conv2[offset:offset+n])
        return grad
    res = minimize(fun, h0, method='SLSQP', jac=jac,
                   bounds=bounds, constraints=[eq_cons],
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    # Clip and enforce sum exactly
    x = np.clip(res.x, 0.0, 1.0)
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:

class SmoothMax:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n
        self.dx = 2.0 / n
    def fun(self, h):
        corr = np.correlate(h, 1 - h, mode='full') * self.dx
        beta_corr = self.beta * corr
        max_bc = np.max(beta_corr)
        lse = max_bc + np.log(np.sum(np.exp(beta_corr - max_bc)))
        return lse / self.beta
    def jac(self, h):
        corr = np.correlate(h, 1 - h, mode='full') * self.dx
        beta_corr = self.beta * corr
        max_bc = np.max(beta_corr)
        exp_vals = np.exp(beta_corr - max_bc)
        w = exp_vals / np.sum(exp_vals)
        # convolve
        conv1 = np.convolve(w, h, mode='full')
        conv2 = np.convolve(w[::-1], h, mode='full')
        offset = self.n - 1
        grad = self.dx * (1.0 - conv1[offset:offset+self.n] - conv2[offset:offset+self.n])
        return grad

import numpy as np
from scipy.optimize import minimize

# -------------------- basic utilities -------------------- #
def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n)
    return np.max(corr)


def enforce_sum(h, target, rng, max_iter=5000):
    """Clamp to [0,1] and adjust to hit exact target sum."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        # small uniform correction then clip again
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Start from 0.5 everywhere and randomly shuffle mass."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    return enforce_sum(h, target, rng)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (preserves total)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

# -------------------- smooth surrogate for SLSQP -------------------- #
class EqualityConstraint:
    def __init__(self, target):
        self.target = target

    def fun(self, h):
        return np.sum(h) - self.target

    def jac(self, h):
        return np.ones_like(h)


class SmoothMax:
    """Log‑sum‑exp surrogate of max_k corr_k with its gradient."""
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n
        self.dx = 2.0 / n

    def fun(self, h):
        corr = np.correlate(h, 1 - h, mode="full") * self.dx
        bc = self.beta * corr
        m = np.max(bc)
        lse = m + np.log(np.sum(np.exp(bc - m)))
        return lse / self.beta

    def jac(self, h):
        corr = np.correlate(h, 1 - h, mode="full") * self.dx
        bc = self.beta * corr
        m = np.max(bc)
        exp_vals = np.exp(bc - m)
        w = exp_vals / np.sum(exp_vals)            # softmax weights, length 2n‑1
        # convolution to obtain Σ_k w_k h_{i±k}
        conv1 = np.convolve(w, h, mode="full")
        conv2 = np.convolve(w[::-1], h, mode="full")
        off = self.n - 1
        grad = self.dx * (1.0 - conv1[off:off + self.n] - conv2[off:off + self.n])
        return grad


def slsqp_optimize_grad(h0, beta, max_iter=200):
    """Run SLSQP on the smooth surrogate, respecting bounds and sum."""
    n = h0.size
    target = n / 2.0
    obj = SmoothMax(beta, n)
    eq = EqualityConstraint(target)
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": eq.fun, "jac": eq.jac}
    res = minimize(
        obj.fun,
        h0,
        method="SLSQP",
        jac=obj.jac,
        bounds=bounds,
        constraints=[cons],
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # final repair to exact sum
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        x = enforce_sum(x, target, rng)
    return x
