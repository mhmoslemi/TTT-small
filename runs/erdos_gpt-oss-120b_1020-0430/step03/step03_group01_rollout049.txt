def enforce_sum(h, target, rng):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    if abs(diff) < 1e-12:
        return h
    # Distribute diff proportionally to slack
    # While diff not zero, adjust random indices.
    for _ in range(10000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0,n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-12:
            continue
        h[i] += add
        diff = target - h.sum()
    # final adjustment
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        # Ensure sum extremely close
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    target_n = 360
    # initial h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.size
    else:
        n = target_n
        h = np.full(n, 0.5)  # sum = n/2
    h = enforce_sum(h, n/2, rng)
    best_h = h.copy()
    best_c = compute_c5(best_h)

    # function to attempt refinement
    def attempt_refine(h_current, beta, max_iter=200):
        nonlocal best_h, best_c, rng
        # add small random perturbation initially to avoid same solution
        h0 = perturb_h(h_current, rng, scale=0.1) # define perturb_h similar to earlier

        # call L-BFGS-B
        target = h0.size / 2
        lam = 0.1 * beta  # scale lambda
        # objective function returns f, grad
        def fun(x):
            f, g = objective_and_grad(x, beta, target, lam)
            return f, g
        bounds = [(0.0, 1.0)] * h0.size
        res = scipy.optimize.minimize(fun, h0, method='L-BFGS-B', jac=True, bounds=bounds,

            best_h = x.copy()
        return x, c

    beta_schedule = [10, 50, 200, 500, 1500, 5000]

import time
import numpy as np
from scipy.optimize import minimize

# -------------------------------------------------
# Utility functions (top level, no closures)
# -------------------------------------------------
def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=5000):
    """Clip h to [0,1] and adjust so that sum(h) == target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-12:
            continue
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        # small uniform correction (rare)
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def random_feasible_vector(n, rng):
    """Generate a random feasible vector (0≤h≤1, sum = n/2)."""
    h = rng.random(n)  # uniform in [0,1)
    target = n / 2.0
    # scale to target sum, then fix any violations
    scale = target / h.sum()
    h *= scale
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, target, rng)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def smooth_max_corr_and_grad(h, beta):
    """Smooth approximation of max_k Corr_k and its gradient."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    bc = beta * corr
    max_bc = np.max(bc)
    exp_bc = np.exp(bc - max_bc)
    sum_exp = exp_bc.sum()
    smooth = (max_bc + np.log(sum_exp)) / beta
    w = exp_bc / sum_exp                      # weights, sum to 1

    grad = np.zeros_like(h)
    # loop over all integer shifts r = -(n-1)..(n-1)
    for r in range(-n + 1, n):
        w_r = w[r + n - 1]
        if r >= 0:
            i_end = n - r
            grad[:i_end] += w_r * (1 - h[r:n])
            grad[r:n] -= w_r * h[:i_end]
        else:
            s = -r
            grad[s:n] += w_r * (1 - h[0:n - s])
            grad[0:n - s] -= w_r * h[s:n]
    return smooth, grad


def objective_and_grad(x, beta, target, lam):
    """Smooth max objective + quadratic penalty for sum constraint, with gradient."""
    smooth, grad_s = smooth_max_corr_and_grad(x, beta)
    diff = x.sum() - target
    penalty = lam * diff * diff
    grad = grad_s + 2 * lam * diff * np.ones_like(x)
    return smooth + penalty, grad


def lbfgsb_refine(h0, beta, max_iter, rng):
    """One L-BFGS-B refinement pass on the smooth surrogate."""
    n = h0.size
    target = n / 2.0
    lam = 1e3                     # penalty weight (fixed)
    bounds = [(0.0, 1.0)] * n

    # tiny random jitter to escape flat regions
    h_start = perturb_h(h0, rng, scale=0.05)

    def fun(x):
        f, g = objective_and_grad(x, beta, target, lam)
        return f, g

    opts = {"maxfun": max_iter, "ftol": 1e-12, "disp": False}
    try:
        res = minimize(fun, h_start, method="L-BFGS-B", jac=True,
                       bounds=bounds, options=opts)
        x = np.clip(res.x, 0.0, 1.0)
        x = enforce_sum(x, target, rng)
    except Exception:
        x = h_start
    return x

    """Optimise a step function h on [0,2] to minimise the Erdős C₅ bound."""

    start = n // 4
    mask2[start : start + n // 2] = 1.0
    candidates.append(enforce_sum(mask2, n / 2.0, rng))

    candidates.append(enforce_sum(np.arange(n) % 2, n / 2.0, rng))

    # random vectors
    for _ in range(3):
        candidates.append(random_feasible_vector(n, rng))

        else:
            h_start = best_h.copy()

        # small hill‑climb style perturbations before each ladder
        for _ in range(4):

        for beta in beta_schedule:
