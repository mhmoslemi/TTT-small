def smooth_obj_and_grad(h, beta):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    # compute softmax weights
    # Use stable log-sum-exp:
    max_beta_corr = np.max(beta * corr)
    exp_terms = np.exp(beta * corr - max_beta_corr)
    sum_exp = np.sum(exp_terms)
    w = exp_terms / sum_exp
    # compute term2 = correlation of w with h
    term2_full = np.correlate(w, h, mode='full')
    start = n - 1
    term2 = term2_full[start:start + n]
    # term3 via reversed w
    w_rev = w[::-1]
    term3_full = np.correlate(w_rev, h, mode='full')
    term3 = term3_full[start:start + n]
    grad = 1.0 - term2 - term3

def enforce_sum(h, target):
    # adjust h to have sum target while staying in [0,1]
    # use simple water-filling: shift mass among elements.
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Determine indices that can be increased/decreased
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    # If no free indices, cannot adjust; just return
    if not np.any(mask):
        return h
    # compute total capacity
    cap = np.where(diff > 0, 1.0 - h, h)
    total_cap = cap[mask].sum()
    # avoid division by zero
    if total_cap == 0:
        return h
    # distribute proportionally
    delta = diff * cap / total_cap
    h += delta
    # Clip
    np.clip(h, 0.0, 1.0, out=h)
    # Since rounding may cause slight error, fix recursively if needed
    # (call itself once more)
    diff2 = target - h.sum()
    if abs(diff2) > 1e-9:
        # adjust a single element
        if diff2 > 0:
            # need increase, choose index with h<1
            idx = np.argmax(1.0 - h)
            inc = min(diff2, 1.0 - h[idx])
            h[idx] += inc
        else:
            idx = np.argmax(h)
            dec = min(-diff2, h[idx])
            h[idx] -= dec
    return h

def smooth_obj_and_grad(h, beta):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    # stable log-sum-exp
    beta_corr = beta * corr
    max_bc = np.max(beta_corr)
    exp_shifted = np.exp(beta_corr - max_bc)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    # term2: correlation of w with h
    term2_full = np.correlate(w, h, mode='full')
    start = n - 1
    term2 = term2_full[start:start + n]
    # term3: correlation of reversed w with h
    w_rev = w[::-1]
    term3_full = np.correlate(w_rev, h, mode='full')
    term3 = term3_full[start:start + n]
    grad = 1.0 - term2 - term3
    phi = (max_bc + np.log(sum_exp)) / beta
    return phi, grad

import time, math, numpy as np
from scipy.optimize import minimize

# ------------------- core helpers -------------------
def compute_c5(h):
    """Exact discrete C5 for step function h (dx=2/n)."""
    n = h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)


def enforce_sum(h, target, max_iter=500):
    """Adjust h in‑place so that sum(h)=target while keeping 0≤h≤1."""
    for _ in range(max_iter):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        cap = np.where(diff > 0, 1.0 - h, h)          # how much each entry can move
        total = cap[mask].sum()
        if total <= 0:
            break
        delta = diff * cap / total
        h += delta
        np.clip(h, 0.0, 1.0, out=h)
    # tiny post‑correction
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argmax(1.0 - h)
            h[idx] += min(diff, 1.0 - h[idx])
        else:
            idx = np.argmax(h)
            h[idx] -= min(-diff, h[idx])
    return h


def random_feasible_vector(n, rng):
    """Generate a random h with 0≤h≤1 and sum=n/2."""
    h = rng.random(n)
    enforce_sum(h, n / 2.0)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer ε between two coordinates (preserves sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new


def simulated_annealing(h0, rng, max_iter=5000, init_temp=0.02, final_temp=1e-6):
    """Simple SA on the true C5 (lower is better)."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    n = h.size
    log_ratio = math.log(final_temp / init_temp) if init_temp > 0 else 0.0
    for it in range(1, max_iter + 1):
        T = init_temp * math.exp(log_ratio * it / max_iter)
        cand = perturb_h(h, rng, scale=0.3)
        cand_c = compute_c5(cand)
        diff = cand_c - cur_c
        if diff < 0 or rng.random() < math.exp(-diff / max(T, 1e-12)):
            h, cur_c = cand, cand_c
            if cand_c < best_c:
                best_c, best_h = cand_c, cand.copy()
    return best_h, best_c


# ------------------- smooth max proxy -------------------
def smooth_obj_and_grad(h, beta):
    """Log‑sum‑exp surrogate of max correlation and its gradient."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    beta_corr = beta * corr
    max_bc = np.max(beta_corr)
    exp_shifted = np.exp(beta_corr - max_bc)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp                     # softmax weights, sum to 1

    # term2 = Σ_k w_k * h_{i+k}
    term2_full = np.correlate(w, h, mode="full")
    start = n - 1
    term2 = term2_full[start:start + n]

    # term3 = Σ_k w_k * h_{i-k}
    w_rev = w[::-1]
    term3_full = np.correlate(w_rev, h, mode="full")
    term3 = term3_full[start:start + n]

    grad = 1.0 - term2 - term3
    phi = (max_bc + np.log(sum_exp)) / beta
    return phi, grad


def lbfgs_optimize(h0, beta, max_iter=200):
    """L‑BFGS‑B on the smooth surrogate with a quadratic sum penalty."""
    n = h0.size
    target = n / 2.0
    lam = 1000.0

    def fun(x):
        phi, grad = smooth_obj_and_grad(x, beta)
        diff = x.sum() - target
        val = phi + lam * diff * diff
        grad_total = grad + 2.0 * lam * diff * np.ones_like(x)
        return val, grad_total

    bounds = [(0.0, 1.0)] * n
    res = minimize(fun, h0, method="L-BFGS-B", jac=True, bounds=bounds,
                   options={"maxiter": max_iter, "ftol": 1e-12})
    x = res.x
    enforce_sum(x, target)
    return x
