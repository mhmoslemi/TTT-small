def propose_move(h, rng, scale):
    n = h.size
    i, j = rng.integers(0,n, size=2)
    while i==j: j = rng.integers(0,n)
    # direction: moving from i to j (i loses, j gains)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0: return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    # ensure within bounds:
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

import time
import numpy as np
from scipy.optimize import minimize

# Global placeholders for the SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp
    val = (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
    return val


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=2000):
    """Adjust h to have Σh = target while keeping 0≤h≤1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # small uniform correction
    h += diff / h.size
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, h.size)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    h = rng.random(n)
    h = enforce_sum(h, n / 2.0, rng)
    return h


def propose_move(h, rng, scale):
    """Generate a neighboring vector: either epsilon transfer or a swap."""
    n = h.size
    if rng.random() < 0.5:
        # epsilon transfer
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        while i == j:
            j = rng.integers(0, n)
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            # try opposite direction
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0:
                return h
            i, j = j, i
        eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
        eps = max(-max_eps, min(eps, max_eps))
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        return new_h
    else:
        # swap two entries
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        while i == j:
            j = rng.integers(0, n)
        new_h = h.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]
        return new_h


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on the smooth surrogate with given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum (tiny drift)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def run(seed=42, budget_s=1100, **kwargs):
    """Optimise a step function h on [0,2] to minimise the Erdős C₅ bound."""
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # -------------------------------------------------------------
    # 1. Initialise h
    # -------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # -------------------------------------------------------------
    # 2. Simulated annealing
    # -------------------------------------------------------------
    # total iterations (will be cut by time)
    max_iters = 500_000
    # temperature schedule
    T0 = 0.005
    T_final = 1e-7
    alpha = -np.log(T_final / T0) / max_iters

    cur_h = best_h.copy()
    cur_c = best_c
    scale_init = 0.5

    for it in range(max_iters):
        # time guard
        if time.time() - start_time > budget_s * 0.85:
            break
        # occasional random restart
        if rng.random() < 0.001:
            cur_h = random_feasible_vector(n, rng)
            cur_c = compute_c5(cur_h)

        # propose move
        scale = scale_init * (T0 * np.exp(-alpha * it)) / T0  # decay with temperature
        new_h = propose_move(cur_h, rng, scale)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0:

        else:
            T = T0 * np.exp(-alpha * it)

        if accept:
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:

    beta_list = [20, 80, 200, 500, 1000, 2000]

        except Exception:

    # -------------------------------------------------------------
    # 4. Final clean‑up
    # -------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
