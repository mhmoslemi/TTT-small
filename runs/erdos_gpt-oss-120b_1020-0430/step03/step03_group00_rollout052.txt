def gradient_of_shift(h, k):
    n = h.size
    grad = np.zeros(n)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        l = -k
        grad[l:] += 1.0 - h[:n - l]
        grad[:n - l] -= h[l:]
    return grad
```

def hill_climb_targeted(h_start, rng, max_iters, start_step, target_sum):
    h = h_start.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr)
        max_corr = corr[idx]
        c_current = max_corr * (2.0 / n)
        k = idx - (n - 1)
        # gradient of correlation at shift k
        grad = gradient_of_shift(h, k)
        # choose move
        pos = np.where((grad > 1e-9) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-9) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            # fallback random
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0:
                continue
            eps = max_eps * step * rng.random()
            new_h = h.copy()
            new_h[i] -= eps
            new_h[j] += eps
        else:
            i = rng.choice(pos)
            j = rng.choice(neg)
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0:
                continue
            eps = max_eps * step * rng.random()
            new_h = h.copy()
            new_h[i] -= eps
            new_h[j] += eps
        # compute new C5
        new_c = compute_c5(new_h)
        # Accept if improves
        if new_c < c_current:
            h = new_h
            c_current = new_c
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c
```

import time, math, numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def enforce_sum(h, target, rng):
    """Clip to [0,1] and adjust to make sum exactly target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:  # need to increase some entries
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i])
            add *= rng.random()
            h[i] += add
            diff -= add
        else:        # need to decrease some entries
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i])
            sub *= rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        if diff > 0:
            h[i] = min(1.0, h[i] + diff)
        else:
            h[i] = max(0.0, h[i] + diff)
    return h

def random_feasible_vector(n, rng):
    """Uniform random vector then repaired to meet sum=n/2."""
    h = rng.random(n)
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, n / 2.0, rng)
    return h

def perturb_random(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j, preserving the sum."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def compute_c5(h):
    """Discrete C5 = max correlation * (2 / n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def gradient_of_shift(h, k):
    """Gradient of correlation at shift k w.r.t. each entry of h."""
    n = h.shape[0]
    grad = np.zeros(n)
    if k >= 0:
        grad[: n - k] += 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    else:
        l = -k
        grad[l:] += 1.0 - h[: n - l]
        grad[: n - l] -= h[l:]
    return grad

# ---------- hill climbing ----------
def hill_climb_targeted(h_start, rng, max_iters, start_step, target_sum):
    h = h_start.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(best_h)
    step = start_step
    decay_interval = max(1, max_iters // 10)

    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr)
        max_corr = corr[idx]
        c_current = max_corr * (2.0 / n)
        k = idx - (n - 1)

        grad = gradient_of_shift(h, k)

        pos = np.where((grad > 1e-9) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-9) & (h < 1.0 - 1e-12))[0]

        if pos.size and neg.size:
            i = rng.choice(pos)
            j = rng.choice(neg)
            max_eps = min(h[i], 1.0 - h[j])
            eps = max_eps * step * rng.random()
            new_h = h.copy()
            new_h[i] -= eps
            new_h[j] += eps
        else:
            new_h = perturb_random(h, rng, scale=step)

        c_new = compute_c5(new_h)

        if c_new < c_current:
            h = new_h
            c_current = c_new

        if c_new < best_c:
            best_c = c_new
            best_h = new_h.copy()

        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)

    return best_h, best_c

# ---------- SLSQP with smooth max ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

# ---------- Simulated annealing ----------
def simulated_annealing(h0, rng, time_limit):
    n = h0.size
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c

    T = 0.005          # initial temperature
    Tmin = 1e-6
    decay = 0.9995

    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0.0:
            continue
        eps = max_eps * rng.random() * 0.5
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < math.exp(-delta / max(T, Tmin)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        T = max(T * decay, Tmin)
        it += 1
    return best_h, best_c
