def lp_refine(h, max_iter=5, rng=None):
    n = h.size
    dx = 2.0 / n
    target_sum = n / 2.0
    for it in range(max_iter):
        # constants based on old h
        coeffs = []  # list of np.ndarray length n
        consts = np.empty(n)
        for k in range(n):
            # shift h by k
            h_shift = np.roll(h, -k)  # h_{i+k}
            const_k = np.dot(h, h_shift)  # sum_i h_i h_{i+k}
            consts[k] = const_k
            # coeff = h_{i-k} + h_{i+k}
            coeff_k = np.roll(h, k) + h_shift  # (i-k) plus (i+k)
            coeffs.append(coeff_k)

        h_var = cp.Variable(n)
        t_var = cp.Variable()

import time, numpy as np, cvxpy as cp
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# global placeholders used by SLSQP soft‑max refinement
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    """Discrete C5 for a step function on [0,2] using the given samples."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

# ----------------------------------------------------------------------
# constraints / objective for the soft‑max SLSQP stage
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp (smooth max)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    """Soft‑max SLSQP optimisation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction (quick redistribution)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def random_feasible_vector(n, rng):
    """Random feasible h (0≤h≤1, Σh=n/2)."""
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

def enforce_sum(h, rng):
    """Adjust vector to satisfy Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

# ----------------------------------------------------------------------
def worst_gradient(h):
    """Gradient of the worst‑shift overlap O_k = Σ h_i(1‑h_{i+k})."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]          # subtract h_{i+k}
        grad[k:]       -= h[: n - k]    # subtract h_{i‑k}
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h_start, rng, max_iters=3000, step_scale=0.5):
    """Hill‑climb using gradient‑directed transfers."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

def gradient_refine(h, rng, step=0.02, max_iter=8):
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def lp_refine(h, max_iter=5, best_c=np.inf, rng=None):
    """
    Convex‑concave (linearised) LP refinement.
    Each iteration solves:
        minimise   t
        s.t.       coeff_k·h + (n/2)·t ≥ (n/2) + const_k   for all k
                  0 ≤ h ≤ 1, Σh = n/2, t ≥ 0, t ≤ best_c (if finite)
    """
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        # linearisation data
        consts = np.empty(n)
        coeffs = []
        for k in range(n):
            h_shift = np.roll(h, -k)
            consts[k] = np.dot(h, h_shift)
            coeffs.append(np.roll(h, k) + h_shift)   # length‑n vector
        # CVXPY variables
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = [h_var >= 0, h_var <= 1,
                cp.sum(h_var) == target,
                t_var >= 0]
        if best_c < np.inf:
            cons.append(t_var <= best_c)
        for k in range(n):
            cons.append(coeffs[k] @ h_var + target * t_var >= target + consts[k])
        prob = cp.Problem(cp.Minimize(t_var), cons)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=500, verbose=False)
        except Exception:
            prob.solve(solver=cp.SCS, warm_start=True, max_iters=2000, verbose=False)
        if h_var.value is None:
            break
        h_new = np.clip(h_var.value, 0.0, 1.0)
        if rng is not None:
            h_new = enforce_sum(h_new, rng)
        else:
            # quick deterministic fix
            diff = target - h_new.sum()
            h_new += diff / n
            h_new = np.clip(h_new, 0.0, 1.0)
        c_new = compute_c5(h_new)
        # accept only if actual C5 improves
        if c_new + 1e-9 < best_c:
            h = h_new
            best_c = c_new
        else:
            break
    return h, best_c
