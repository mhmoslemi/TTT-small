import time
import numpy as np
from scipy.optimize import minimize

# ------- helper functions -------------------------------------------------

def compute_c5(h):
    """Exact discrete C5 for step function h on [0,2] with dx = 2/len(h)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=5000):
    """Project h onto [0,1]^n with exact sum = target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # fallback correction if needed
    if abs(diff) > 1e-12:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    # start from uniform 0.5 then random transfers
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount epsilon between two coordinates (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_multi(h_start, rng, max_iters=2000, init_step=0.5, n_trials=5):
    """Simple hill‑climb with multiple candidate perturbations per step."""
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c = c
                h = new_h
                improved = True
                break
        if not improved and rng.random() < 0.02:
            new_h = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(new_h)
            if c < best_c:
                best_c = c
                h = new_h
                improved = True
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return h, best_c


# ------- smooth surrogate + SLSQP -----------------------------------------

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth log‑sum‑exp surrogate of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp stabilized
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum (tiny drift)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ------- main optimisation ------------------------------------------------

def run(seed=42, budget_s=1100, **kwargs):
    """Optimise a step function h on [0,2] to minimise the Erdős C5 bound."""
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # -----------------------------------------------------------------
    # Determine discretisation size
    # -----------------------------------------------------------------
    base_h = None
    if "initial_h_values" in globals():
        base_h = np.array(initial_h_values, dtype=float)
    base_n = base_h.size if base_h is not None else 360
    # aim for a finer grid but cap at 720 for speed
    n = min(720, max(360, base_n * 2))

    target_sum = n / 2.0
    # -----------------------------------------------------------------
    # Build initial population
    # -----------------------------------------------------------------
    pop_size = 40
    pop = []

    # include upsampled initial (if any)
    if base_h is not None:
        if base_n != n:
            old_x = (np.arange(base_n) + 0.5) * (2.0 / base_n)
            new_x = (np.arange(n) + 0.5) * (2.0 / n)
            up = np.interp(new_x, old_x, base_h)
            up = enforce_sum(up, target_sum, rng)
        else:
            up = enforce_sum(base_h, target_sum, rng)
        pop.append(up)

    # deterministic masks
    mask = np.zeros(n)
    mask[: n // 2] = 1.0
    pop.append(enforce_sum(mask, target_sum, rng))

    mask2 = np.zeros(n)
    start = n // 4
    mask2[start : start + n // 2] = 1.0
    pop.append(enforce_sum(mask2, target_sum, rng))

    alt = np.arange(n) % 2
    pop.append(enforce_sum(alt.astype(float), target_sum, rng))

    # linear ramps
    pop.append(enforce_sum(np.linspace(0.0, 1.0, n), target_sum, rng))
    pop.append(enforce_sum(np.linspace(1.0, 0.0, n), target_sum, rng))

    # sinusoidal family
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    for f in (1, 2, 3, 4):
        h = 0.5 + 0.5 * np.sin(np.pi * f * xs)
        pop.append(enforce_sum(np.clip(h, 0.0, 1.0), target_sum, rng))
        h = 0.5 + 0.5 * np.cos(np.pi * f * xs)
        pop.append(enforce_sum(np.clip(h, 0.0, 1.0), target_sum, rng))

    # random feasible vectors
    while len(pop) < pop_size:
        pop.append(random_feasible_vector(n, rng))

    pop = pop[:pop_size]

    # -----------------------------------------------------------------
    # Evaluate initial fitness
    # -----------------------------------------------------------------
    fitness = [compute_c5(h) for h in pop]
    best_idx = int(np.argmin(fitness))
    best_h = pop[best_idx].copy()
    best_c = fitness[best_idx]

    # -----------------------------------------------------------------
    # Evolutionary optimisation
    # -----------------------------------------------------------------
    elite_frac = 0.2
    mutation_rate = 0.35
    hillclimb_prob = 0.25
    max_ga_iters = 1000000  # effectively unlimited; loop exits by time

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.70:
            break

        elite_cnt = max(2, int(pop_size * elite_frac))
        # select elites (best fitness)
        sorted_idx = sorted(range(pop_size), key=lambda i: fitness[i])
        elites = [pop[i] for i in sorted_idx[:elite_cnt]]
        elite_fit = [fitness[i] for i in sorted_idx[:elite_cnt]]

        # build weighted selection distribution (better fitness => higher weight)
        weights = np.array([1.0 / (f + 1e-12) for f in elite_fit])
        weights /= weights.sum()

        new_pop = elites.copy()
        while len(new_pop) < pop_size:
            # parent choice
            p_idx = rng.choice(len(elites), size=2, replace=False, p=weights)
            p1, p2 = elites[p_idx[0]], elites[p_idx[1]]
            alpha = rng.random()
            child = alpha * p1 + (1.0 - alpha) * p2
            child = np.clip(child, 0.0, 1.0)
            child = enforce_sum(child, target_sum, rng)

            if rng.random() < mutation_rate:
                child = perturb_h(child, rng, scale=0.2)

            if rng.random() < hillclimb_prob:
                child, _ = hill_climb_multi(
                    child, rng, max_iters=250, init_step=0.3, n_trials=4
                )

            new_pop.append(child)

        # evaluate new generation
        new_fitness = [compute_c5(h) for h in new_pop]
        # update global best
        min_idx = int(np.argmin(new_fitness))
        if new_fitness[min_idx] < best_c:
            best_c = new_fitness[min_idx]
            best_h = new_pop[min_idx].copy()

        pop = new_pop
        fitness = new_fitness

        # occasional SLSQP refinement of the global best
        if rng.random() < 0.04:
            try:
                refined = slsqp_optimize(best_h, beta=200, max_iter=250)
                c_ref = compute_c5(refined)
                if c_ref < best_c:
                    best_c = c_ref
                    best_h = refined.copy()
            except Exception:
                pass

    # -----------------------------------------------------------------
    # Final intensive local search
    # -----------------------------------------------------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 120.0:  # allocate ~2 minutes if possible
        best_h, best_c = hill_climb_multi(
            best_h, rng, max_iters=5000, init_step=0.15, n_trials=8
        )

    # -----------------------------------------------------------------
    # SLSQP ladder for final polishing
    # -----------------------------------------------------------------
    beta_schedule = [5, 20, 80, 200, 500, 1000, 2000, 5000]
    for beta in beta_schedule:
        if time.time() - start_time >= budget_s:
            break
        try:
            best_h = slsqp_optimize(best_h, beta=beta, max_iter=600)
            best_c = compute_c5(best_h)
        except Exception:
            pass

    # -----------------------------------------------------------------
    # Clean‑up and return
    # -----------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target_sum, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
