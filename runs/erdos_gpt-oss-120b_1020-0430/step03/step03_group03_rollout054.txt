def gradient_logsumexp(h, beta):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    # compute weights
    maxc = corr.max()
    exp_term = np.exp(beta * (corr - maxc))
    w = exp_term / exp_term.sum()
    # convs
    L = 2 * n - 1
    # conv2: sum w[k] * h[i - shift]
    conv2 = np.convolve(w, h, mode='full')
    conv2 = conv2[n-1:n-1+n]  # length n
    # conv1: sum w[k] * h[i + shift]
    w_rev = w[::-1]
    conv1_full = np.convolve(w_rev, h, mode='full')
    conv1 = conv1_full[L-1:L-1+n]  # start at 2n-2
    grad = 1.0 - (conv1 + conv2)
    return grad, corr
```

def gradient_descent(h, beta, n_iter, step_size, rng):
    n = h.size
    target_sum = n / 2.0
    for _ in range(n_iter):
        grad, _ = gradient_logsumexp(h, beta)
        # move opposite gradient
        h = h - step_size * grad

        # enforce sum using deterministic correction
        # compute diff

def random_feasible_vector(n, rng):
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    enforce_sum(h, n / 2.0, rng)
    return h

import time
import math
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# basic utilities
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function h on [0,2] with dx = 2/len(h)."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h in‑place so that sum(h)=target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    # simple random walk to fix the sum
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random h∈[0,1]^n with Σh=n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    enforce_sum(h, n / 2.0, rng)
    return h


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.3) -> np.ndarray:
    """Transfer ε between two coordinates (preserves sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

# smooth‑max gradient utilities
# ------------------------------------------------------------
def gradient_logsumexp(h: np.ndarray, beta: float):
    """Gradient of the log‑sum‑exp (soft‑max) of the overlaps."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")          # length 2n‑1
    maxc = corr.max()
    w = np.exp(beta * (corr - maxc))
    w /= w.sum()                                          # weights sum to 1

    # convolution that yields Σ_k w_k h_{i±k}
    L = 2 * n - 1
    conv2 = np.convolve(w, h, mode="full")[n - 1 : n - 1 + n]          # Σ w_k h_{i‑k}
    conv1 = np.convolve(w[::-1], h, mode="full")[L - 1 : L - 1 + n]   # Σ w_k h_{i+k}
    grad = 1.0 - (conv1 + conv2)                                     # ∂L/∂h_i
    return grad, corr


def gradient_descent(
    h0: np.ndarray,
    beta: float,
    n_iter: int,
    step: float,
    rng: np.random.Generator,
) -> np.ndarray:
    """Projected gradient descent on the smooth max objective."""
    h = h0.copy()
    target = h.size / 2.0
    for _ in range(n_iter):
        grad, _ = gradient_logsumexp(h, beta)
        h -= step * grad
        np.clip(h, 0.0, 1.0, out=h)
        # enforce the exact sum constraint
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
            if free.size == 0:
                enforce_sum(h, target, rng)
            else:
                h[free] += diff / free.size
                np.clip(h, 0.0, 1.0, out=h)
                enforce_sum(h, target, rng)
    return h


# ------------------------------------------------------------
# SLSQP refinement (soft‑max surrogate)
# ------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp, divided by beta
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, max_iter: int = 250) -> np.ndarray:
    """Run SLSQP on the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {"type": "eq", "fun": sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x
