def pair_transfer_optimize(h0, rng, max_iters, time_limit, initial_factor=0.5, factor_decay=0.99):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    factor = initial_factor
    n = h.size
    while time.time() - start < time_limit and max_iters>0:
        max_val, shift = max_overlap_and_shift(h)
        if max_val * (2.0 / n) < best_c: # actual c5 with scaling
            best_c = max_val * (2.0 / n)
            best_h = h.copy()
        # compute product array for this shift
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:

            j_range = i_range + shift
        # compute products
        prod = h[i_range] * (1 - h[j_range])
        # find max product index
        idx = np.argmax(prod)
        i = i_range[idx]
        j = j_range[idx]
        max_eps = min(h[i], 1 - h[j])
        if max_eps <= 0:
            # try random pair
            i = rng.integers(0, n)
            j = (i + shift) % n  # modulo? not using wrap; but shift may be negative, use i+shift if within bounds else skip.
            # Too messy; fallback: random transfer
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            max_eps = min(h[i], 1 - h[j])

                factor *= factor_decay

def slsqp_optimize(h0, beta, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n/2
    CURRENT_BETA = beta
    bounds = [(0.,1.)]*n
    cons = {'type':'eq','fun':sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter':max_iter,'ftol':1e-12,'disp':False})
    x = np.clip(res.x,0.,1.)
    # enforce sum
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # correct by adding diff/n equally but clip
        correction = diff / n
        x += correction
        x = np.clip(x,0.,1.)
        diff = target - x.sum()
        if abs(diff) > 1e-12:
            # small greedy fix
            for i in range(n):
                if abs(diff) < 1e-12: break
                add = min(diff, 1.-x[i]) if diff>0 else max(diff, -x[i])
                x[i] += add
                diff = target - x.sum()
    return x

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
#  Global variables used by the smooth surrogate (softmax) for SLSQP
# ----------------------------------------------------------------------
CURRENT_TARGET = 0.0
CURRENT_BETA = 0.0


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1-x_{i+k}) using log-sum-exp."""
    # full convolution h * (1-h)
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h so that Σh = target (0≤h≤1)."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-14:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # final correction if needed
    if abs(diff) > 1e-14:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-14:
            for i in range(n):
                if abs(diff) < 1e-14:
                    break
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random feasible h with Σh = n/2 and 0≤h≤1."""
    h = np.full(n, 0.5)
    # random pairwise swaps
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        for _ in range(2000):
            if abs(diff) < 1e-14:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_multi(h_start, rng, max_iters=2000, init_step=0.5, n_trials=5):
    """Hill‑climb with multiple random perturbations per step."""
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c = c
                h = new_h
                improved = True
                break
        if not improved and rng.random() < 0.02:
            new_h = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(new_h)
            if c < best_c:
                best_c = c
                h = new_h
                improved = True
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return h, best_c


def slsqp_optimize(h0, beta, max_iter=400):
    """Refine h using SLSQP on the smooth surrogate with given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        n = x.size
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x


def max_overlap_and_shift(h):
    """Return (max_corr, shift) where shift = idx-(n-1)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    max_val = float(corr[idx])
    shift = idx - (h.size - 1)
    return max_val, shift


def pair_transfer_optimize(h0, rng, max_iters=5000, time_limit=30.0,
                          init_factor=0.5, decay=0.999):
    """Specialised local search moving mass from i to i+shift to lower max overlap."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    factor = init_factor

    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        max_val, shift = max_overlap_and_shift(h)
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        # determine index ranges for this shift
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            i_range = np.arange(-shift, n)
            j_range = i_range + shift
        if i_range.size == 0:
            factor = init_factor
            continue
        prod = h[i_range] * (1.0 - h[j_range])
        idx_max = int(np.argmax(prod))
        i = int(i_range[idx_max])
        j = int(j_range[idx_max])
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            factor = init_factor
            continue
        eps = factor * max_eps
        if eps <= 0.0:
            factor = init_factor
            continue
        # transfer eps from i to j
        h[i] -= eps
        h[j] += eps
        # numerical safety
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
        factor = max(1e-4, factor * decay)
    # final check
    final_c = compute_c5(best_h)
    return best_h, final_c


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # -------------------------------------------------
    # 1. Initialise h
    # -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    # -------------------------------------------------
    # 2. Build a modest pool of diverse starting vectors
    # -------------------------------------------------
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    candidates = [h0]

    for f in (1, 2, 3, 4, 5):
        for func in (np.sin, np.cos):
            h = 0.5 + 0.5 * func(np.pi * f * xs)
            h = np.clip(h, 0.0, 1.0)
            candidates.append(enforce_sum(h, n / 2.0, rng))

    # linear ramps
    candidates.append(enforce_sum(xs / 2.0, n / 2.0, rng))
    candidates.append(enforce_sum(1.0 - xs / 2.0, n / 2.0, rng))

    # binary masks
    mask = np.zeros(n)
    mask[: n // 2] = 1.0
    candidates.append(enforce_sum(mask, n / 2.0, rng))

    mask = np.zeros(n)
    start = n // 4
    mask[start : start + n // 2] = 1.0
    candidates.append(enforce_sum(mask, n / 2.0, rng))

    mask = np.arange(n) % 2
    candidates.append(enforce_sum(mask, n / 2.0, rng))

    # random vectors
    for _ in range(3):
        candidates.append(random_feasible_vector(n, rng))

    # -------------------------------------------------
    # 3. Pick the best among the pool
    # -------------------------------------------------
    best_h = None
    best_c = float("inf")
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    global_best_h = best_h
    global_best_c = best_c

    # -------------------------------------------------
    # 4. Main optimisation: hill‑climb + SLSQP ladder
    # -------------------------------------------------
    beta_list = [5, 20, 80, 200, 500, 1000, 2000, 5000]
    main_time_limit = budget_s * 0.70
    while time.time() - start_time < main_time_limit:
        # start point selection
        if rng.random() < 0.65:
            start_vec = perturb_h(global_best_h, rng, scale=0.35)
        else:
            start_vec = random_feasible_vector(n, rng)

        # hill climb
        h_hc, c_hc = hill_climb_multi(
            start_vec, rng, max_iters=1800, init_step=0.4, n_trials=6
        )
        if c_hc < global_best_c:
            global_best_h, global_best_c = h_hc.copy(), c_hc

        # SLSQP refinement with increasing β
        for beta in beta_list:
            if time.time() - start_time >= main_time_limit:
                break
            try:
                h_ref = slsqp_optimize(global_best_h, beta, max_iter=500)
                c_ref = compute_c5(h_ref)
                if c_ref < global_best_c:
                    global_best_h, global_best_c = h_ref.copy(), c_ref
            except Exception:
                continue

    # -------------------------------------------------
    # 5. Targeted pair‑transfer optimisation (remaining time)
    # -------------------------------------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 15.0:
        # allocate attempts proportionally
        attempts = max(1, int(remaining // 30))
        per_attempt = remaining / (attempts + 1)
        best_h = global_best_h
        best_c = global_best_c
        for _ in range(attempts):
            # small random perturb before pair‑transfer
            h_start = perturb_h(best_h, rng, scale=0.2) if rng.random() < 0.5 else best_h.copy()
            h_opt, c_opt = pair_transfer_optimize(
                h_start, rng, max_iters=8000, time_limit=per_attempt, init_factor=0.5
            )
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt.copy()
                global_best_h, global_best_c = best_h.copy(), best_c
        # final fine‑tune on the current best
        h_opt, c_opt = pair_transfer_optimize(
            global_best_h, rng, max_iters=12000, time_limit=remaining * 0.9, init_factor=0.5
        )
        if c_opt < global_best_c:
            global_best_h, global_best_c = h_opt.copy(), c_opt

    # -------------------------------------------------
    # 6. Final clean‑up
    # -------------------------------------------------
    global_best_h = np.clip(global_best_h, 0.0, 1.0)
    global_best_h = enforce_sum(global_best_h, n / 2.0, rng)
    final_c = compute_c5(global_best_h)

    return global_best_h.tolist(), float(final_c), int(n)
