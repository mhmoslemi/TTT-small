import time
import numpy as np
from scipy.optimize import minimize

# ------------------- utilities ------------------- #
def compute_c5(h):
    """Exact discrete C5 for step function h on [0,2] with uniform spacing."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def random_feasible_vector(n, rng):
    """Uniform random vector in [0,1] with exact sum n/2."""
    h = rng.random(n)
    target = n / 2.0
    scale = target / h.sum()
    h *= scale
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h


def random_binary_vector(n, rng):
    """Binary vector with exactly n/2 ones (n must be even)."""
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = np.clip(eps, -max_eps, max_eps)
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def worst_shift_move(h, step_factor):
    """Targeted mass transfer along the current worst shift."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        return h, False
    a = h[: n - k]
    b = h[k:]
    diff = a - b
    i = int(np.argmax(diff))
    max_eps = min(a[i], 1.0 - b[i])
    if max_eps <= 0.0:
        return h, False
    eps = max_eps * step_factor
    new_h = h.copy()
    new_h[i] -= eps
    new_h[i + k] += eps
    return new_h, True


def hill_climb(h_start, rng, max_iters=2000, start_step=0.6):
    """Improvement loop mixing random perturbations and worst‑shift moves."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        # try a directed move along the worst shift
        new_h, moved = worst_shift_move(h, step_factor=step)
        if moved:
            c = compute_c5(new_h)
            if c < best_c:
                best_c, best_h, h = c, new_h.copy(), new_h
                if (it + 1) % decay_every == 0:
                    step = max(step * 0.9, 1e-6)
                continue  # success, skip random perturbation this iteration
        # fallback to random perturbation
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h, h = c, new_h.copy(), new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-6)
    return best_h, best_c


def gradient_refine(h, rng, step=0.02, max_iter=12):
    """Refine by moving against gradient of the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]       # + (1 - h[i+k])
            g[k:] -= h[:upper]             # - h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        # enforce exact sum after each iteration
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


# ----------- soft‑max surrogate for SLSQP ----------- #
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=400):
    """Run a single SLSQP optimisation with the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x


def simulated_annealing(h0, rng, deadline):
    """Simple SA that respects the sum constraint."""
    n = h0.size
    target = n / 2.0
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    T0 = max(0.01 * cur_c, 1e-4)
    it = 0
    while time.time() < deadline and it < 8000:
        step = max(0.5 * (0.999 ** it), 5e-4)
        cand_h = perturb_h(cur_h, rng, scale=step)
        cand_c = compute_c5(cand_h)
        accept = False
        if cand_c < cur_c:
            accept = True
        else:
            T = T0 * (0.999 ** it) + 1e-12
            if rng.random() < np.exp(-(cand_c - cur_c) / T):
                accept = True
        if accept:
            cur_h, cur_c = cand_h, cand_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
        it += 1
    return best_h, best_c


# ------------------- main driver ------------------- #
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # parent solution (if supplied)
    parent = None
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)

    # candidate discretisations (moderate sizes)
    candidate_ns = [72, 96, 120, 144, 180, 240, 300]

    # Helper to enforce feasibility after any manual change
    def enforce_sum(h):
        n = h.size
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(2000):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
        np.clip(h, 0.0, 1.0, out=h)
        return h

    # Core improvement routine for a given starting vector
    def improve_one(h):
        nonlocal best_h, best_c, best_n
        # Stage 1: aggressive hill‑climb
        h, c = hill_climb(h, rng, max_iters=2500, start_step=0.7)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, h.size

        # Stage 2: targeted smoothing of worst shift (few passes)
        for _ in range(8):
            h, moved = worst_shift_move(h, step_factor=0.5)
            if not moved:
                break
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, h.size

        # Stage 3: gradient refinement
        h = gradient_refine(h, rng, step=0.02, max_iter=15)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, h.size

        # Stage 4: SLSQP with increasing beta
        for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 3000.0):
            if time.time() - start_time > budget_s * 0.96:
                break
            max_iter = 300 if beta < 200 else 600
            try:
                h = slsqp_optimize(h, beta, rng, max_iter=max_iter)
            except Exception:
                continue
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, h.size

        # Stage 5: final hill‑climb
        if time.time() - start_time < budget_s * 0.99:
            h, c = hill_climb(h, rng, max_iters=2000, start_step=0.4)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, h.size

        return h

    # Main loop over resolutions
    for n in candidate_ns:
        if time.time() - start_time > budget_s * 0.95:
            break

        # Prepare some diverse starting vectors
        starts = []

        # 0) Interpolated parent (if available)
        if parent is not None:
            x_src = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            x_tgt = np.linspace(0.0, 2.0, n, endpoint=False)
            starts.append(np.interp(x_tgt, x_src, parent))

        # 1) Random continuous feasible
        starts.append(random_feasible_vector(n, rng))

        # 2) Random binary feasible
        if n % 2 == 0:
            starts.append(random_binary_vector(n, rng))

        # 3) Slightly perturbed best of same size (if we already have)
        if best_n == n and best_h is not None:
            starts.append(best_h + 0.02 * rng.standard_normal(n))

        # Limit attempts per resolution
        attempts = 0
        while attempts < 8 and time.time() - start_time < budget_s * 0.94:
            h0 = starts[attempts % len(starts)]
            h0 = enforce_sum(h0.copy())
            improve_one(h0)
            attempts += 1

    # Optional up‑sampling refinement (if budget permits)
    if best_h is not None and time.time() - start_time < budget_s * 0.99:
        if best_n * 2 <= 300:
            n_up = best_n * 2
            x_src = np.linspace(0.0, 2.0, best_n, endpoint=False)
            x_tgt = np.linspace(0.0, 2.0, n_up, endpoint=False)
            h_up = np.interp(x_tgt, x_src, best_h)
            h_up = enforce_sum(h_up)
            h_up = hill_climb(h_up, rng, max_iters=1800, start_step=0.5)
            for beta in (5.0, 20.0, 80.0, 200.0, 500.0):
                if time.time() - start_time > budget_s * 0.995:
                    break
                h_up = slsqp_optimize(h_up, beta, rng, max_iter=400)
            h_up = gradient_refine(h_up, rng, step=0.02, max_iter=10)
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, best_n = h_up, c_up, n_up

    # Final safety check
    if best_h is None:
        best_n = 120
        best_h = np.full(best_n, 0.5)

    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            i = rng.integers(0, best_n)
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()
            if abs(diff) < 1e-12:
                break

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
