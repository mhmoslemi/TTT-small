def compute_c5_fft(h):
    n = h.size
    L = 2 * n
    # compute convolution via FFT
    f = np.fft.rfft(h, L)
    g = np.fft.rfft((1-h)[::-1], L)
    conv = np.fft.irfft(f * g, L)

    corr = conv[:2*n-1]
    # compute max
    return float(corr.max() * (2.0 / n))

import time, numpy as np
from scipy.optimize import minimize

# ---------- fast C5 evaluation ----------
def compute_c5(h):
    n = h.shape[0]
    L = 2 * n
    h_pad = np.concatenate((h, np.zeros(n)))
    rev_pad = np.concatenate(((1.0 - h)[::-1], np.zeros(n)))
    conv = np.fft.irfft(np.fft.rfft(h_pad) * np.fft.rfft(rev_pad), n=L)
    corr = conv[:2 * n - 1]
    return float(corr.max() * (2.0 / n))

# ---------- utilities ----------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def random_feasible_vector(n, rng):
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

# ---------- hill climbing ----------
def hill_climb(h, rng, max_iters=2000, step=0.5):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(best_h)
    cur_step = step
    decay = max_iters // 10 or 1
    for it in range(max_iters):
        cand = perturb_h(best_h, rng, scale=cur_step)
        c = compute_c5(cand)
        if c < best_c:
            best_c, best_h = c, cand.copy()
        if (it + 1) % decay == 0:
            cur_step = max(cur_step * 0.9, 1e-4)
    return best_h, best_c

def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        h = h - step * (0.5 ** it) * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        for _ in range(200):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

# ---------- SLSQP surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

# ---------- simulated annealing ----------
def simulated_annealing(h0, rng, deadline):
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    T0 = max(0.01 * cur_c, 1e-4)
    it = 0
    n = cur_h.size
    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0:
            continue
        # slowly decaying step
        scale = max(0.1, 0.5 ** (it / 20000.0))
        eps = (rng.random() * 2 - 1) * max_eps * scale
        cand = cur_h.copy()
        cand[i] -= eps
        cand[j] += eps
        c = compute_c5(cand)
        if c < cur_c:
            cur_h, cur_c = cand, c
            if c < best_c:
                best_c, best_h = c, cand.copy()
        else:
            T = T0 * np.exp(-0.0005 * it) + 1e-12
            if rng.random() < np.exp(-(c - cur_c) / T):
                cur_h, cur_c = cand, c
        it += 1
    return best_h, best_c
