def gradient_descent_transfer(h, rng, max_iters=10000, step=0.05):
    n = len(h)
    target = n/2
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iters):
        # compute gradient for worst shift
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1 - 2*h
        else:
            upper = n - k
            g[:upper] += 1 - h[k:]   # for i from 0 to n-k-1
            g[k:] -= h[:upper]       # for i from k to n-1
        # find indices
        pos = np.argmax(g)  # max gradient
        neg = np.argmin(g)  # min gradient
        g_pos = g[pos]
        g_neg = g[neg]
        if g_pos <= 0 and g_neg >= 0:
            # no improvement direction found (maybe local optimum)
            break
        eps = min(h[pos], 1 - h[neg], step)
        if eps <= 1e-12:
            break
        # do transfer
        h[pos] -= eps
        h[neg] += eps
        # ensure feasibility (clip + sum adjust)
        # after transfer sum remains same, and each stays in [0,1] by eps bound.
        # compute new C5
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        # optional: adjust step adaptively
        # maybe reduce step gradually
        if it % 500 == 0 and step > 1e-4:
            step *= 0.5
    return best_h, best_c
```

def slsqp_optimize(h0, beta, rng, max_iter, deadline):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    # Use minimize with maxiter limited, but if time is low break early.
    # We'll check time before calling.
    if time.time() > deadline:
        return h0
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x

    bc = CURRENT_BETA * corr
    m = np.max(bc)

import time
import numpy as np
from scipy.optimize import minimize

# Global variables used by the soft‑max surrogate
CURRENT_TARGET = None
CURRENT_BETA = None

# -------------------------- utilities -------------------------- #
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function h on [0,2] with uniform spacing."""
    n = h.size
    return float(np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / n))


def make_feasible(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Clip to [0,1] and adjust linearly so that sum(h)=len(h)/2."""
    n = h.size
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector in [0,1] with exact sum n/2."""
    h = rng.random(n)
    target = n / 2.0
    scale = target / h.sum()
    h *= scale
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


# ---------------------- gradient transfer ---------------------- #
def gradient_transfer(h: np.ndarray, max_iters: int, init_step: float,
                     deadline: float) -> tuple[np.ndarray, float]:
    """
    Deterministic descent using the gradient of the worst shift.
    Moves mass from the index with largest positive gradient to the one
    with most negative gradient.
    """
    n = h.size
    step = init_step
    best_h = h.copy()
    best_c = compute_c5(h)
    no_improve = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g = 1 - 2 * h
        else:
            upper = n - k
            g[:upper] += 1 - h[k:]
            g[k:] -= h[:upper]
        pos = int(np.argmax(g))
        neg = int(np.argmin(g))
        g_pos = g[pos]
        g_neg = g[neg]
        if g_pos <= 0 and g_neg >= 0:
            break
        eps = min(h[pos], 1 - h[neg], step)
        if eps <= 1e-12:
            break
        h[pos] -= eps
        h[neg] += eps
        c = compute_c5(h)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if (it + 1) % (max_iters // 10 + 1) == 0:
            step = max(step * 0.5, 1e-6)
        if no_improve > max_iters // 10:
            break
    return best_h, best_c


# ------------------------ simulated annealing ------------------------ #
def simulated_annealing(h0: np.ndarray, rng: np.random.Generator,
                        deadline: float) -> tuple[np.ndarray, float]:
    n = h0.size
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    T0 = max(0.01 * cur_c, 1e-4)
    it = 0
    while time.time() < deadline:
        step = max(0.5 * (0.999 ** it), 5e-4)
        cand_h = perturb_h(cur_h, rng, scale=step)
        cand_c = compute_c5(cand_h)
        if cand_c < cur_c:
            accept = True
        else:
            T = T0 * (0.999 ** it) + 1e-12
            accept = rng.random() < np.exp(-(cand_c - cur_c) / T)
        if accept:
            cur_h, cur_c = cand_h, cand_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
        it += 1
    return best_h, best_c


# ---------------------- soft‑max surrogate for SLSQP ---------------------- #
def sum_eq_constraint(x: np.ndarray) -> float:
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x: np.ndarray) -> float:
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int, deadline: float) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    if time.time() > deadline:
        return h0
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x
