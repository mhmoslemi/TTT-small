def worst_shift_and_gradient(h):
    n = h.size
    # compute convolution for (h, 1 - h)
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    k = idx - (n - 1)   # shift index (i.e., j = i + k)
    max_val = corr[idx]

    # compute gradient
    grad = np.ones(n)
    if k >= 0:
        # i only where both terms defined:
        # For i in [0..n - k -1], term -h[i+k] appears; for i in [k..n-1], term -h[i-k] appears.
        grad[:n - k] -= h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[:n - kk] -= h[kk:]
        grad[kk:] -= h[:n - kk]
    # gradient of f_k (i.e., max) with respect to h.

    return k, max_val, grad

    max_corr, k, grad = corr_and_worst_grad(h)

        elapsed = time.time() - start

    dx = 2.0 / n
    h = h_start.copy()
    max_corr, k, grad = corr_and_worst_grad(h)
    best_h = h.copy()
    best_c = max_corr * dx
    c_curr = best_c

    start = time.time()
    it = 0
    while it < max_iters and time.time() - start < time_limit:
        # compute temperature
        frac = it / max_iters
        T = 0.05 * (1 - frac)  # temperature for acceptance
        # compute scale for perturb
        scale = max_scale * (1 - frac) + min_scale
        # choose to move using gradient or random?

import time
import numpy as np
from scipy.optimize import minimize

# ---------- feasibility utilities ----------
def enforce_sum(h, target, rng):
    """Clip to [0,1] and adjust to make sum exactly target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-14 and it < max_iter:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-14)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i])
            add *= rng.random()
            h[i] += add
            diff -= add
        else:
            idx = np.where(h > 1e-14)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i])
            sub *= rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-14:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)

# ---------- objective ----------
def corr_and_worst_grad(h):
    """Return (max_corr, shift, gradient) for the worst overlap shift."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr)
    max_corr = corr[idx]
    k = idx - (n - 1)
    grad = np.ones(n)
    if k >= 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    else:
        kk = -k
        grad[: n - kk] -= h[kk:]
        grad[kk:] -= h[: n - kk]
    return max_corr, k, grad

def compute_c5(h):
    max_corr, _, _ = corr_and_worst_grad(h)
    return max_corr * (2.0 / h.size)

# ---------- annealed hill climbing ----------
def annealed_hill_climb(h_start, rng, max_iters, max_scale, min_scale, target_sum, time_limit):
    """Simulated‑annealing style hill climbing using guided transfers."""
    n = h_start.size
    dx = 2.0 / n
    h = h_start.copy()
    max_corr, _, grad = corr_and_worst_grad(h)
    c_curr = max_corr * dx
    best_h, best_c = h.copy(), c_curr
    start = time.time()
    it = 0
    while it < max_iters and (time.time() - start) < time_limit:
        frac = it / max_iters
        T = 0.07 * (1.0 - frac)          # temperature for acceptance
        scale = max_scale * (1.0 - frac) + min_scale  # move magnitude

        # guided or random transfer
        if rng.random() < 0.7:
            pos = np.where((grad > 1e-9) & (h > 1e-12))[0]
            neg = np.where((grad < -1e-9) & (h < 1.0 - 1e-12))[0]
            if pos.size and neg.size:
                i = rng.choice(pos)
                j = rng.choice(neg)
                max_eps = min(h[i], 1.0 - h[j])
                eps = max_eps * scale * rng.random()
                new_h = h.copy()
                new_h[i] -= eps
                new_h[j] += eps
            else:
                # fallback random transfer
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                while i == j:
                    j = rng.integers(0, n)
                max_eps = min(h[i], 1.0 - h[j])
                eps = max_eps * scale * rng.random()
                new_h = h.copy()
                new_h[i] -= eps
                new_h[j] += eps
        else:
            # pure random transfer
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            while i == j:
                j = rng.integers(0, n)
            max_eps = min(h[i], 1.0 - h[j])
            eps = max_eps * scale * rng.random()
            new_h = h.copy()
            new_h[i] -= eps
            new_h[j] += eps

        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = enforce_sum(new_h, target_sum, rng)

        max_corr_new, _, _ = corr_and_worst_grad(new_h)
        c_new = max_corr_new * dx

        if c_new < c_curr or (T > 0 and np.log(rng.random()) < -(c_new - c_curr) / T):
            h = new_h
            c_curr = c_new
            max_corr, _, grad = corr_and_worst_grad(h)
            if c_new < best_c:
                best_c, best_h = c_new, h.copy()
        it += 1
    return best_h, best_c

# ---------- up‑sampling ----------
def upsample_h(h, factor, rng):
    n = h.size
    new_n = n * factor
    xs_old = (np.arange(n) + 0.5) * (2.0 / n)
    xs_new = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    h_new = np.interp(xs_new, xs_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = enforce_sum(h_new, new_n / 2.0, rng)
    return h_new

# ---------- SLSQP refinement ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

    n0 = h0.size

            c_ref = compute_c5(refined)

    # ---- final tiny random tweaks ----
    time_left = budget_s - (time.time() - start_time)

        new_h = best_h.copy()
