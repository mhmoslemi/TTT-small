import time
import numpy as np
from scipy.optimize import minimize  # only for potential fallback, not heavily used


# ---------- basic utilities ----------
def compute_c5(h):
    """Exact discrete C5 for step function h on [0,2] with uniform spacing."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Simple hill‑climb that only accepts improvements."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


def gradient_refine(h, rng, step=0.02, max_iter=12):
    """Gradient step against the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            up = n - k
            g[:up] += 1.0 - h[k:]
            g[k:] -= h[:up]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


# ---------- binary simulated annealing ----------
def binary_sa_impl(start_bin, rng, deadline, max_iters=10_000_000):
    """
    SA on a binary vector that keeps exactly n/2 ones.
    Returns best binary vector (int) and the best max correlation (int).
    """
    n = start_bin.size
    cur_h = start_bin.copy()
    hd = 2 * cur_h - 1                 # +1 for ones, -1 for zeros
    cur_c = np.correlate(cur_h, 1 - cur_h, mode="full")
    cur_cmax = cur_c.max()
    best_h = cur_h.copy()
    best_cmax = cur_cmax

    # maintain lists of positions
    ones = [i for i in range(n) if cur_h[i] == 1]
    zeros = [i for i in range(n) if cur_h[i] == 0]

    delta = np.empty(2 * n - 1, dtype=int)

    T = 1.0
    cool = 0.9995
    it = 0
    check_step = 2000

    while it < max_iters:
        # time check every few thousand steps
        if (it & (check_step - 1)) == 0:
            if time.time() > deadline:
                break

        # choose random 1 and 0
        pos_one = rng.integers(len(ones))
        i_one = ones[pos_one]
        pos_zero = rng.integers(len(zeros))
        i_zero = zeros[pos_zero]

        # build delta
        delta.fill(0)
        s1 = n - 1 - i_one
        s2 = n - 1 - i_zero
        delta[s1 : s1 + n] += hd
        delta[s2 : s2 + n] -= hd
        delta[n - 1] -= 2
        d = i_zero - i_one
        delta[n - 1 + d] += 1
        delta[n - 1 - d] += 1

        new_cmax = (cur_c + delta).max()

        accept = False
        if new_cmax <= cur_cmax:
            accept = True
        else:
            if rng.random() < np.exp(-(new_cmax - cur_cmax) / max(T, 1e-12)):
                accept = True

        if accept:
            # apply swap
            cur_h[i_one] = 0
            cur_h[i_zero] = 1
            hd[i_one] = -1
            hd[i_zero] = 1
            cur_c = cur_c + delta
            cur_cmax = new_cmax

            # update best
            if cur_cmax < best_cmax:
                best_cmax = cur_cmax
                best_h = cur_h.copy()

            # update lists
            ones[pos_one] = ones[-1]
            ones.pop()
            zeros.append(i_one)

            zeros[pos_zero] = zeros[-1]
            zeros.pop()
            ones.append(i_zero)

        T *= cool
        it += 1

    return best_h, best_cmax


def binary_sa(start_bin, rng, deadline, max_total_iters=2_000_000):
    """
    Run several independent SA restarts within the allotted time.
    """
    best_h = None
    best_cmax = np.iinfo(np.int64).max
    while time.time() < deadline and max_total_iters > 0:
        # give each restart a modest slice of time
        sub_deadline = min(deadline, time.time() + 5.0)
        h, cmax = binary_sa_impl(start_bin, rng, sub_deadline, max_iters= max_total_iters)
        max_total_iters -= (sub_deadline - time.time()) * 2000  # rough estimate
        if cmax < best_cmax:
            best_cmax = cmax
            best_h = h.copy()
        # small perturbation to seed next restart
        start_bin = start_bin.copy()
        # flip a few random bits to keep diversity
        for _ in range(3):
            i = rng.integers(0, start_bin.size)
            j = rng.integers(0, start_bin.size)
            if start_bin[i] != start_bin[j]:
                start_bin[i], start_bin[j] = start_bin[j], start_bin[i]
    return best_h, best_cmax


# ---------- high‑level driver ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.96  # reserve a little for final cleanup

    parent = None
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)

    best_h = None
    best_c5 = np.inf
    best_n = None

    candidate_ns = [180, 144, 120, 96, 72]

    for n in candidate_ns:
        if time.time() > deadline:
            break

        # ----- prepare initial binary vector -----
        if parent is not None and parent.size == n:
            # round parent to binary then correct exact sum
            h_bin = (parent >= 0.5).astype(int)
            diff = (n // 2) - h_bin.sum()
            if diff != 0:
                idx0 = np.where(h_bin == 0)[0]
                idx1 = np.where(h_bin == 1)[0]
                if diff > 0:
                    chosen = rng.choice(idx0, diff, replace=False)
                    h_bin[chosen] = 1
                else:
                    chosen = rng.choice(idx1, -diff, replace=False)
                    h_bin[chosen] = 0
        else:
            h_bin = np.zeros(n, dtype=int)
            ones = rng.choice(n, n // 2, replace=False)
            h_bin[ones] = 1

        # ----- binary simulated annealing -----
        sub_deadline = min(deadline, time.time() + budget_s * 0.2)
        bin_best, bin_cmax = binary_sa(h_bin, rng, sub_deadline, max_total_iters=1_500_000)
        if bin_best is None:
            continue
        c5_bin = (2.0 / n) * bin_cmax
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = bin_best.astype(float)
            best_n = n

        # ----- local continuous refinement -----
        if time.time() < deadline:
            # start from the binary solution
            h_float = best_h.copy()
            # quick hill‑climb
            h_float, c5_float = hill_climb(h_float, rng, max_iters=2500, start_step=0.4)
            if c5_float < best_c5:
                best_c5 = c5_float
                best_h = h_float.copy()
                best_n = n

            # gradient refinement
            h_float = gradient_refine(h_float, rng, step=0.02, max_iter=12)
            c5_float = compute_c5(h_float)
            if c5_float < best_c5:
                best_c5 = c5_float
                best_h = h_float.copy()
                best_n = n

        # ----- optional use of the original parent (if size matches) -----
        if parent is not None and parent.size == n and time.time() < deadline:
            h_parent = np.clip(parent, 0.0, 1.0)
            # enforce exact sum
            target = n / 2.0
            diff = target - h_parent.sum()
            if abs(diff) > 1e-12:
                for _ in range(2000):
                    i = rng.integers(0, n)
                    add = min(diff, 1.0 - h_parent[i]) if diff > 0 else max(diff, -h_parent[i])
                    h_parent[i] += add
                    diff = target - h_parent.sum()
                    if abs(diff) < 1e-12:
                        break
            c5_parent = compute_c5(h_parent)
            if c5_parent < best_c5:
                best_c5 = c5_parent
                best_h = h_parent.copy()
                best_n = n

    # ---------- final safety adjustments ----------
    if best_h is None:
        best_n = 100
        best_h = np.full(best_n, 0.5)

    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            i = rng.integers(0, best_n)
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()
            if abs(diff) < 1e-12:
                break

    final_c5 = compute_c5(best_h)

    return best_h.tolist(), float(final_c5), int(best_n)
