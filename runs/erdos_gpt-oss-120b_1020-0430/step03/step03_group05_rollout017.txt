```
def smooth_obj_and_grad(x, beta):
    n = x.shape[0]
    r = 1.0 - x
    corr = np.correlate(x, r, mode='full')
    # compute stable log-sum-exp
    z = beta * corr
    maxz = np.max(z)
    exp_z = np.exp(z - maxz)
    sum_exp = np.sum(exp_z)
    obj = (maxz + np.log(sum_exp)) / beta
    p = exp_z / sum_exp  # shape 2n-1
    p_rev = p[::-1]
    offset = 2 * n - 2
    conv1 = np.convolve(r, p_rev, mode='full')
    conv2 = np.convolve(x, p_rev, mode='full')
    grad = conv1[offset:offset + n] - conv2[offset:offset + n]
    return obj, grad
```

    sumexp = np.sum(exp_z)
    obj = (maxz + np.log(sumexp)) / beta
    p = exp_z / sumexp          # length 2n-1
    p_rev = p[::-1]
    offset = 2 * n - 2
    conv1 = np.convolve(r, p_rev, mode='full')
    conv2 = np.convolve(x, p_rev, mode='full')
    grad = conv1[offset:offset + n] - conv2[offset:offset + n]
    return obj, grad

def continuation_optimize(x0, target, rng, max_iters_per_beta=250):
    x = project_to_feasible(x0, target)
    beta_seq = [5.0, 20.0, 80.0, 320.0, 1280.0]
    for beta in beta_seq:
        obj, grad = smooth_obj_and_grad(x, beta)
        step = 0.2
        for _ in range(max_iters_per_beta):
            norm = np.linalg.norm(grad) + 1e-12
            direction = -grad / norm
            x_new = project_to_feasible(x + step * direction, target)
            obj_new, grad_new = smooth_obj_and_grad(x_new, beta)
            if obj_new < obj:
                x, obj, grad = x_new, obj_new, grad_new
                step = min(step * 1.1, 0.5)
            else:
                step *= 0.5
                if step < 1e-8:
                    break
    return x

def perturb_h(h, rng, scale=0.3):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = scale * max_eps * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb_exact(h, rng, iters=200):
    best = h.copy()
    best_c = compute_c5(best)
    for _ in range(iters):
        cand = perturb_h(best, rng, scale=0.3)
        c = compute_c5(cand)
        if c < best_c:
            best = cand
            best_c = c
    return best, best_c

    xs_new = np.linspace(0.0, 2.0, to_n, endpoint=False)

def project_to_feasible(v, target):
    lo, hi = -2.0, 2.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
