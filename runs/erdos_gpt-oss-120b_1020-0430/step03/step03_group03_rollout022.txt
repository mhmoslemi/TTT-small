# ---------- basic utilities ----------
def compute_c5(h):
    """Discrete C5 for step function h on [0,2] (dx = 2/n)."""
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / h.size)

def enforce_sum(h, target, rng):
    """Adjust h in‑place so that sum(h)=target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    # increase
    while diff > 1e-12:
        idx = rng.choice(np.where(h < 1)[0])
        add = min(diff, 1.0 - h[idx])
        h[idx] += add
        diff = target - h.sum()
    # decrease
    while diff < -1e-12:
        idx = rng.choice(np.where(h > 0)[0])
        sub = min(-diff, h[idx])
        h[idx] -= sub
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    """Start from 0.5 everywhere and perform random transfers."""
    h = np.full(n, 0.5)
    for _ in range(5 * n):

        if i == j:

        h[i] -= eps
        h[j] += eps

    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=200):
    """Smooth‑max (softmax) SLSQP refinement."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {"type": "eq", "fun": sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        enforce_sum(h0, n / 2.0, rng)
    else:
        n = 180
        h0 = random_feasible_vector(n, rng)

    best_h, best_c = local_descent(h0, rng, max_iters=300, init_step=0.2)

    # ----- many random restarts with local descent -----
    while time.time() - start < budget_s * 0.80:
        if best_c < 0.38078:  # good enough, break early
            break
        # start from a fresh random point or a perturbation of the best
        if rng.random() < 0.5:
            start_h = random_feasible_vector(n, rng)
        else:
            start_h = perturb_h(best_h, rng, scale=0.25)
        enforce_sum(start_h, n / 2.0, rng)
        cand_h, cand_c = local_descent(start_h, rng, max_iters=250, init_step=0.15)
        if cand_c < best_c:
            best_h, best_c = cand_h, cand_c

    # ----- occasional SLSQP polishing during the remaining budget -----
    beta_list = (200, 800, 2000, 8000, 20000)
    for beta in beta_list:
        if time.time() - start > budget_s * 0.95:
            break
        try:
            refined = slsqp_optimize(best_h, beta, max_iter=250)
            rc = compute_c5(refined)
            if rc < best_c:
                best_h, best_c = refined, rc
        except Exception:
            continue

    # final ensure feasibility and compute exact C5
    best_h = np.clip(best_h, 0.0, 1.0)
    enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
