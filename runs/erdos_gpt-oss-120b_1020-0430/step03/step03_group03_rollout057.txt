Procedure improve_local(h, iters, step_frac):
    for iter:
        compute overlap array corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (len(h) - 1)
        # compute grad array (size n)

        # For forward direction:
        idx_f = np.arange(0, len(h)-k)  # i where i+k in bounds
        grad[idx_f] -= h[idx_f + k]
        # For backward direction:
        idx_b = np.arange(k, len(h))  # i where i-k in bounds -> i-k >=0 -> start from k
        grad[idx_b] -= h[idx_b - k]
        # Actually we added contributions twice for those i that have both forward and backward. that's correct (two terms)
        # Now find i_decr = argmax(grad) (largest positive). j_incr = argmin(grad) (most negative).
        i_decr = np.argmax(grad)
        j_incr = np.argmin(grad)
        if grad[i_decr] <= 0 and grad[j_incr] >= 0:
            # cannot improve further on this shift
            break
        # Transfer epsilon
        max_transfer = min(h[i_decr], 1 - h[j_incr])
        if max_transfer <= 0:
            break
        eps = step_frac * max_transfer
        h[i_decr] -= eps
        h[j_incr] += eps
        # optionally normalize sum? Already sum unchanged
    return h

def gradient_descent_max_shift(h, steps=200, step_frac=0.3, rng=None):
    n = h.shape[0]
    # Make a copy
    h = h.copy()
    for s in range(steps):
        # compute corr and find max shift
        corr = np.correlate(h, 1 - h, mode='full')
        k = np.argmax(corr) - (n - 1)
        # compute gradient for that shift
        grad = np.full(n, 1.0)
        if k >= 0:
            # forward: i in [0, n-1-k] subtract h[i+k]
            grad[:n - k] -= h[k:]
            # backward: i in [k, n-1] subtract h[i - k]
            grad[k:] -= h[:n - k]
        else:
            sneg = -k
            # forward: i in [sneg, n-1] subtract h[i - sneg]
            grad[sneg:] -= h[:n - sneg]
            # backward: i in [0, n-1-sneg] subtract h[i + sneg]
            grad[:n - sneg] -= h[sneg:]
        # i_decr: largest positive grad (increase of h_i would increase overlap, so we want to decrease it)
        i_decr = int(np.argmax(grad))
        # i_incr: most negative grad (increase reduces overlap)
        i_incr = int(np.argmin(grad))
        if grad[i_decr] <= 0 and grad[i_incr] >= 0:
            break
        max_transfer = min(h[i_decr], 1.0 - h[i_incr])
        if max_transfer <= 0:
            break
        eps = max_transfer * step_frac
        h[i_decr] -= eps
        h[i_incr] += eps
    return h

from scipy.optimize import minimize

# ------------------------------------------------------------
# helpers
# ------------------------------------------------------------
def compute_c5(h):
    """Discrete C5 for step function h (dx = 2 / len(h))."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)


def enforce_sum(h, target, rng):
    """Adjust h in‑place so that sum(h) == target while 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Random vector in [0,1]^n with sum n/2 (continuous)."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    enforce_sum(h, n / 2.0, rng)
    return h


def random_binary_feasible(n, rng):
    """Binary vector with exactly n/2 ones (n must be even)."""
    perm = rng.permutation(n)
    h = np.zeros(n, dtype=float)
    h[perm[: n // 2]] = 1.0
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer ε between two coordinates (preserves sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new


def gradient_descent_max_shift(h, steps=250, max_step=0.5, min_step=0.02, rng=None):
    """Iteratively reduce the worst‑shift overlap using directed mass transfers."""
    n = h.size
    h = h.copy()
    for s in range(steps):
        # current worst shift
        corr = np.correlate(h, 1 - h, mode="full")
        k = int(np.argmax(corr)) - (n - 1)

        # gradient of the overlap for this shift (preserves sum)
        grad = np.full(n, 1.0)
        if k >= 0:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
        else:
            sneg = -k
            grad[sneg:] -= h[: n - sneg]
            grad[: n - sneg] -= h[sneg:]

        i_decr = int(np.argmax(grad))      # decreasing this entry helps
        i_incr = int(np.argmin(grad))      # increasing this entry helps

        if grad[i_decr] <= 0 and grad[i_incr] >= 0:
            break  # cannot improve this shift

        max_transfer = min(h[i_decr], 1.0 - h[i_incr])
        if max_transfer <= 0:
            break

        step_frac = max_step - (max_step - min_step) * (s / steps)
        eps = max_transfer * step_frac
        h[i_decr] -= eps
        h[i_incr] += eps
    return h

# smooth‑max SLSQP refinement
# ------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=200):
    """SLSQP on a smooth approximation of max_k overlap."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {"type": "eq", "fun": sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # --------------------------------------------------------
    # initialise
    # --------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        enforce_sum(h0, n / 2.0, rng)
    else:
        n = 180
        h0 = random_feasible_vector(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # --------------------------------------------------------
    # stochastic exploration + directed gradient descent
    # --------------------------------------------------------
    grad_phase = budget_s * 0.85
    while time.time() - start_time < grad_phase:
        # choose a start point
        if rng.random() < 0.6:
            start = perturb_h(best_h, rng, scale=0.25)
        else:
            start = random_binary_feasible(n, rng) if rng.random() < 0.5 else random_feasible_vector(n, rng)

        # directed refinement
        h_opt = gradient_descent_max_shift(start, steps=250, max_step=0.5, min_step=0.02, rng=rng)
        enforce_sum(h_opt, n / 2.0, rng)

        c = compute_c5(h_opt)
        if c < best_c:
            best_h, best_c = h_opt.copy(), c

    # --------------------------------------------------------
    # final polishing with increasing beta (SLSQP)
    # --------------------------------------------------------
    for beta in (50, 200, 800, 2000, 8000, 20000):
        if time.time() - start_time > budget_s:
            break
        try:
            cand = slsqp_optimize(best_h, beta, max_iter=250)
            cand_c = compute_c5(cand)
            if cand_c < best_c:
                best_h, best_c = cand, cand_c
        except Exception:
            continue

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
