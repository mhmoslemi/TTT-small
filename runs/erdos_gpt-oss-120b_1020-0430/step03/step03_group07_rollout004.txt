def compute_gradient_for_lag(h, k):
    n = h.size
    grad = np.ones(n)
    if k >= 0:
        # h[i+k] exists for i in 0..n-1-k
        grad[: n - k] -= h[k: ]  # h[i+k] with i up to n-1-k

        grad[kk: ] -= h[: n - kk]
        # (i-k) = i+kk -> exist for i in 0..n-1-kk
        grad[: n - kk] -= h[kk: ]
    return grad

def binary_sa(bin_h, rng, deadline):
    # bin_h is boolean or 0/1 array.
    n = bin_h.size
    cur = bin_h.copy()
    cur_c = compute_c5(cur.astype(float))
    best = cur.copy()
    best_c = cur_c
    # schedule
    start = time.time()
    while time.time() < deadline:
        # temperature schedule (progress as fraction of remaining)
        fraction = (time.time() - start) / (deadline - start)
        T = max(1e-5, (1 - fraction))
        # propose swap
        # select i where cur[i]==1, j where cur[j]==0
        ones = np.flatnonzero(cur)
        zeros = np.flatnonzero(~cur)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur[i] = 0
        cur[j] = 1
        new_c = compute_c5(cur.astype(float))
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            # accept
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best = cur.copy()
        else:
            # revert swap
            cur[i] = 1
            cur[j] = 0
    return best, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ------------------- basic utilities -------------------
def enforce_sum(h, target, rng):
    """Adjust h so that sum(h) == target, keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    """Random vector in [0,1]^n with sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps_max = min(h[i], 1.0 - h[j])
        if eps_max <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * eps_max * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)

def perturb_h(h, rng, scale=0.2):
    """Transfer a random ε from i to j (preserves sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def correlate_full(h):
    """Full linear correlation of h with (1‑h) using FFT."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv

def compute_c5(h):
    """Discrete C₅ for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)

# ------------------- SLSQP surrogate -------------------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """Log‑sum‑exp smoothing of max_k Σ_i x_i (1‑x_{i+k})."""
    conv = correlate_full(x)
    bc = CURRENT_BETA * conv
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

# ------------------- hill climbing -------------------
def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        nh = perturb_h(h, rng, scale=step)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c

# ------------------- worst‑lag gradient descent -------------------
def gradient_worst_lag(h, k):
    n = h.size
    grad = np.ones(n)
    if k >= 0:
        grad[: n - k] -= h[k: ]
        grad[k: ] -= h[: n - k]
    else:
        kk = -k
        grad[kk: ] -= h[: n - kk]
        grad[: n - kk] -= h[kk: ]
    return grad

def improve_worst_lag(h, rng, max_iters=500, eps_scale=0.5):
    n = h.size
    for _ in range(max_iters):
        conv = correlate_full(h)
        idx = int(conv.argmax())
        k = (n - 1) - idx
        grad = gradient_worst_lag(h, k)
        pos = int(np.argmax(grad))
        neg = int(np.argmin(grad))
        if grad[pos] <= 0 or grad[neg] >= 0:
            break
        max_eps = min(h[pos], 1.0 - h[neg])
        if max_eps <= 0:
            break
        eps = max_eps * eps_scale * rng.random()
        h[pos] -= eps
        h[neg] += eps
    return h

# ------------------- binary simulated annealing -------------------
def binary_sa(bin_h, rng, deadline):
    """Swap‑based SA on a binary vector (fixed number of ones)."""
    cur = bin_h.copy()
    cur_c = compute_c5(cur.astype(float))
    best = cur.copy()
    best_c = cur_c
    start = time.time()
    n = cur.size
    while time.time() < deadline:
        frac = (time.time() - start) / max(deadline - start, 1e-12)
        T = max(1e-5, 1.0 - frac)  # linear cooling
        ones = np.flatnonzero(cur)
        zeros = np.flatnonzero(~cur)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur[i] = 0
        cur[j] = 1
        new_c = compute_c5(cur.astype(float))
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            cur_c = new_c
            if cur_c < best_c:
                best_c, best = cur_c, cur.copy()
        else:
            cur[i] = 1
            cur[j] = 0
    return best, best_c

# ------------------- main entry point -------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s
    # phases
    deadline_main = start_time + budget_s * 0.78
    deadline_up   = start_time + budget_s * 0.95

    # initial vector
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(360, 0.5)
    n = h0.size
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = enforce_sum(h0, n / 2.0, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ---------- main stochastic search ----------
    while time.time() < deadline_main:
        # start point
        if rng.random() < 0.7:
            start = perturb_h(best_h, rng, scale=0.35)
        else:
            start = random_feasible_vector(n, rng)

        # focused gradient descent on worst lag
        iters = int(400 + 200 * rng.random())
        cand = improve_worst_lag(start, rng, max_iters=iters, eps_scale=0.6)
        cand_c = compute_c5(cand)
        if cand_c < best_c:
            best_h, best_c = cand.copy(), cand_c

        # occasional generic hill climb
        if rng.random() < 0.25:
            hc, cc = hill_climb(best_h, rng, max_iters=1500, start_step=0.3)
            if cc < best_c:
                best_h, best_c = hc.copy(), cc

        # SLSQP refinement ladder
        for beta in (5.0, 20.0, 80.0, 200.0, 800.0):
            if time.time() >= deadline_main:
                break
            try:
                cand = slsqp_optimize(best_h, beta, max_iter=250)
                cand_c = compute_c5(cand)
                if cand_c < best_c:
                    best_h, best_c = cand.copy(), cand_c
            except Exception:
                continue

    # ---------- optional up‑sampling ----------
    if time.time() < deadline_up and n * 2 <= 800:
        n2 = n * 2
        up = np.repeat(best_h, 2)
        up = enforce_sum(up, n2 / 2.0, rng)
        # a few refinement cycles on finer grid
        while time.time() < deadline_up - 20:
            start = perturb_h(up, rng, scale=0.25) if rng.random() < 0.5 else random_feasible_vector(n2, rng)
            up = improve_worst_lag(start, rng, max_iters=800, eps_scale=0.5)
            c_up = compute_c5(up)
            if c_up < best_c:
                best_h, best_c, n = up.copy(), c_up, n2
        # keep upsampled version if better
        if best_h.size == n2:
            pass

    # ---------- binary rounding & SA ----------
    n = best_h.size
    idx = np.argsort(-best_h)
    bin_h = np.zeros(n, dtype=float)
    bin_h[idx[: n // 2]] = 1.0
    bin_c = compute_c5(bin_h)
    if bin_c < best_c:
        best_h, best_c = bin_h.copy(), bin_c

    deadline_bin = start_time + budget_s * 0.99
    if time.time() < deadline_bin:
        bin_bool = best_h >= 0.5
        # enforce exact count
        target = n // 2
        cur_cnt = int(bin_bool.sum())
        if cur_cnt != target:
            if cur_cnt > target:
                drop = rng.choice(np.flatnonzero(bin_bool), size=cur_cnt - target, replace=False)
                bin_bool[drop] = False
            else:
                add = rng.choice(np.flatnonzero(~bin_bool), size=target - cur_cnt, replace=False)
                bin_bool[add] = True
        best_bin, best_bin_c = binary_sa(bin_bool.astype(np.int8), rng, deadline_bin)
        if best_bin_c < best_c:
            best_h, best_c = best_bin.astype(float), best_bin_c

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
