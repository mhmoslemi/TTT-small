import time
import numpy as np

# ---------- Helper utilities ----------
def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_to_feasible(v, target):
    """Projection of v onto {0≤h≤1, Σh = target} via bisection on a shift."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Fast generation of a random feasible vector (0≤h≤1, Σh=n/2)."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_to_feasible(v, n / 2.0)


def upsample_initial(to_n):
    """Linear interpolation of global initial_h_values to length to_n."""
    init = np.array(initial_h_values, dtype=float)
    m = init.size
    xs_old = np.linspace(0.0, 2.0, m, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, to_n, endpoint=False)
    return np.interp(xs_new, xs_old, init)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5, patience=200):
    """Greedy hill‑climb with early stop when no improvement for `patience` steps."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    no_improve = 0
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c - 1e-12:
            best_c = c
            best_h = new_h.copy()
            h = new_h
            no_improve = 0
        else:
            no_improve += 1
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
        if no_improve >= patience:
            break
    return best_h, best_c


def subgradient_step(h, target, eta):
    """One projected sub‑gradient step for the max‑correlation objective."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    k_idx = int(np.argmax(corr))
    k = k_idx - (n - 1)            # shift (may be negative)

    grad = np.zeros_like(h)
    if k >= 0:
        grad[: n - k] = 1.0 - h[k:]          # (1 - h[i+k])
        grad[k:]   -= h[: n - k]            # -h[i-k]
    else:
        s = -k
        grad[s:]   = 1.0 - h[: n - s]        # (1 - h[i-k]) with k negative
        grad[: n - s] -= h[s:]              # -h[i+k]

    new_h = h - eta * grad
    new_h = np.clip(new_h, 0.0, 1.0)
    if abs(new_h.sum() - target) > 1e-12:
        new_h = project_to_feasible(new_h, target)
    return new_h


# ---------- Main optimisation routine ----------
def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Choose a moderately sized discretisation; larger n gives more flexibility.
    n = 240
    target = n / 2.0

    # Prepare a list of starting vectors (random + up‑sampled initial if available)
    starts = []
    # a few random starts
    for _ in range(5):
        starts.append(random_feasible_vector(n, rng))
    # include up‑sampled initial if present
    if "initial_h_values" in globals():
        init_vec = upsample_initial(n)
        init_vec = project_to_feasible(init_vec, target)
        starts.append(init_vec)

    best_h = None
    best_c = np.inf

    # Main optimisation loop – iterate over starting points
    for h0 in starts:
        # ----- Greedy hill‑climb -----
        h_cur, c_cur = hill_climb(h0, rng, max_iters=4000, start_step=0.5, patience=300)
        if c_cur < best_c:
            best_c, best_h = c_cur, h_cur.copy()

        # ----- Projected sub‑gradient descent -----
        eta0 = 0.5
        decay = 0.9995
        eta = eta0
        it = 0
        while True:
            elapsed = time.time() - start_time
            if elapsed > budget_s * 0.96:
                break
            h_next = subgradient_step(h_cur, target, eta)
            c_next = compute_c5(h_next)
            if c_next < best_c - 1e-12:
                best_c, best_h = c_next, h_next.copy()
            # accept if improvement, otherwise keep current
            if c_next < c_cur:
                h_cur, c_cur = h_next, c_next
            eta *= decay
            it += 1
            if it % 500 == 0 and elapsed > budget_s * 0.9:
                break

        # ----- Light random perturbation phase (to escape plateaus) -----
        while True:
            elapsed = time.time() - start_time
            if elapsed > budget_s * 0.98:
                break
            h_trial = perturb_h(h_cur, rng, scale=0.05)
            c_trial = compute_c5(h_trial)
            if c_trial < best_c:
                best_c, best_h = c_trial, h_trial.copy()
                h_cur, c_cur = h_trial, c_trial
            else:
                # occasionally accept small worsening to explore
                if rng.random() < 0.01:
                    h_cur, c_cur = h_trial, c_trial
            if time.time() - start_time > budget_s * 0.99:
                break

    # Fallback – should never happen but ensures a valid return.
    if best_h is None:
        best_h = np.full(n, 0.5)
        best_c = compute_c5(best_h)

    # Final feasibility enforcement (numerical drift correction).
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        best_h = project_to_feasible(best_h, target)

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
