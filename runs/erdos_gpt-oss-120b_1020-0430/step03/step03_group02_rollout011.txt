import time, numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    """Discrete C5 value for step function defined by array h."""
    # Correlation of h with (1-h) over full range, scaled by dx = 2/n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h, rng):
    """Adjust h so that sum(h) = len(h)/2 while keeping 0<=h<=1."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # Simple random redistribution
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    """Generate a random feasible vector with given size."""
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, rng)

def worst_gradient(h):
    """Gradient of the overlap O_k attaining the current max."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    """Transfer epsilon from a large‑gradient index to a small‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp for smooth max
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=800):
    """Smooth‑max SLSQP refinement for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum repair (tiny)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def anneal(h0, rng, budget_s, start_time):
    """Simulated‑annealing style local search."""
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    n = h.size
    max_iters = 300000
    for it in range(max_iters):
        if time.time() - start_time > budget_s * 0.99:
            break
        # temperature linearly decreasing from 1e-3 to 1e-6
        T = 1e-3 * (1.0 - it / max_iters) + 1e-6 * (it / max_iters)
        # choose move type
        if rng.random() < 0.6:
            grad = worst_gradient(h)
            i = int(np.argmax(grad))
            j = int(np.argmin(grad))
        else:
            # random feasible pair
            pos = np.where(h > 1e-12)[0]
            neg = np.where(h < 1.0 - 1e-12)[0]
            if pos.size == 0 or neg.size == 0:
                continue
            i = rng.choice(pos)
            j = rng.choice(neg)
            if i == j:
                continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        # eps scaled by temperature (larger T → larger steps)
        eps = max_eps * min(1.0, T * 500.0)
        if eps <= 0.0:
            eps = max_eps * 0.01
        nh = h.copy()
        nh[i] -= eps
        nh[j] += eps
        c_nh = compute_c5(nh)
        delta = c_nh - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = nh
            cur_c = c_nh
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        # occasional reset to best found to avoid drift
        if (it + 1) % 5000 == 0:
            h = best_h.copy()
            cur_c = best_c
    return best_h, best_c

CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h):
    """Discrete C5 value for step function given by array h."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, rng):
    """Scale/adjust h so that sum(h) = len(h)/2 while keeping 0≤h≤1."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    h = rng.random(n)

def worst_gradient(h):
    """Gradient of the overlap O_k attaining the current maximum."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def anneal(h0, rng, budget_s, start_time):
    """Simulated‑annealing style search."""
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    max_iters = 300_000
    for it in range(max_iters):
        if time.time() - start_time > budget_s * 0.99:
            break
        # temperature drops linearly from 1e-3 to 1e-6
        T = 1e-3 * (1.0 - it / max_iters) + 1e-6 * (it / max_iters)

        if rng.random() < 0.6:
            grad = worst_gradient(h)
            i = int(np.argmax(grad))
            j = int(np.argmin(grad))
        else:

            neg = np.where(h < 1.0 - 1e-12)[0]
