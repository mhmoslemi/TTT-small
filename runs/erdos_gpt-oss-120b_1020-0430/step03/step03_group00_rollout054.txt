def delta_corr_move(h, g, i_src, i_dst, eps):
    n = h.shape[0]
    delta = np.zeros(2*n-1, dtype=np.float64)
    # -eps * g[i_src + k] part
    start = n-1 - i_src
    delta[start:start+n] -= eps * g
    # +eps * g[i_dst + k] part
    start = n-1 - i_dst
    delta[start:start+n] += eps * g
    # +eps * h[i_src - k] part -> idx = (n-1 + i_src) - np.arange(n)
    idx = (n-1 + i_src) - np.arange(n)
    delta[idx] += eps * h
    # -eps * h[i_dst - k] part
    idx = (n-1 + i_dst) - np.arange(n)
    delta[idx] -= eps * h
    return delta

    n = h.shape[0]

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utilities ----------
def enforce_sum(h, target, rng):
    """Clip to [0,1] and adjust to make sum exactly target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    # simple proportional redistribution to non‑saturated entries
    idx_free = np.where((h > 0.0) & (h < 1.0))[0]
    if idx_free.size == 0:
        # fallback: adjust a random entry
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
        return h
    # distribute diff equally (may overshoot, but loop will correct)
    add = diff / idx_free.size
    h[idx_free] += add
    # recurse if still out of bounds
    return enforce_sum(h, target, rng)

def compute_c5(h):
    """Discrete C5 = max correlation * (2 / n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

# ---------- delta for a two‑point transfer ----------
def delta_corr_move(h, g, i_src, i_dst, eps):
    """Incremental update of correlation for moving eps from i_src to i_dst."""
    n = h.shape[0]
    delta = np.zeros(2 * n - 1, dtype=np.float64)

    # -eps * g[i_src + k]
    start = n - 1 - i_src
    delta[start:start + n] -= eps * g
    # +eps * g[i_dst + k]
    start = n - 1 - i_dst
    delta[start:start + n] += eps * g

    # +eps * h[i_src - k]  (indices = (n-1+i_src) - np.arange(n))
    idx = (n - 1 + i_src) - np.arange(n)
    delta[idx] += eps * h
    # -eps * h[i_dst - k]
    idx = (n - 1 + i_dst) - np.arange(n)
    delta[idx] -= eps * h
    return delta

# ---------- Simulated annealing ----------
def simulated_annealing(h0, time_limit, rng):
    """Anneal h0 for at most time_limit seconds, returns best (h, max_corr)."""
    h = h0.copy()
    n = h.shape[0]
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    cur_max = corr.max()
    best_h = h.copy()
    best_max = cur_max

    T0 = 5.0
    alpha = 0.9995
    T = T0
    start = time.time()
    while time.time() - start < time_limit:
        # pick source and destination with slack
        src_candidates = np.where(h > 1e-12)[0]
        if src_candidates.size == 0:
            break
        dst_candidates = np.where(h < 1.0 - 1e-12)[0]
        if dst_candidates.size == 0:
            break
        i_src = rng.choice(src_candidates)
        i_dst = rng.choice(dst_candidates)
        if i_src == i_dst:
            continue
        max_eps = min(h[i_src], 1.0 - h[i_dst])
        if max_eps <= 0.0:
            continue
        scale = min(0.5, T / T0)  # shrink step with temperature
        eps = max_eps * scale * rng.random()
        if eps <= 0.0:
            continue

        delta = delta_corr_move(h, g, i_src, i_dst, eps)
        new_corr = corr + delta
        new_max = new_corr.max()

        # Metropolis acceptance on the unscaled max correlation
        if new_max <= cur_max or rng.random() < np.exp(-(new_max - cur_max) / (T + 1e-12)):
            # accept move
            h[i_src] -= eps
            h[i_dst] += eps
            g[i_src] = 1.0 - h[i_src]
            g[i_dst] = 1.0 - h[i_dst]

                best_h = h.copy()
        T *= alpha
        if T < 1e-6:
            T = 0.0
    return best_h, best_max

# ---------- SLSQP (log‑sum‑exp smoothing) ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

    # ----- Phase 1 : low‑resolution SA -----
    elapsed = time.time() - start_time
    remaining = max(0.0, budget_s - elapsed)
    phase1_budget = remaining * 0.20
    if phase1_budget > 5.0:
        h1, max_corr1 = simulated_annealing(h0, phase1_budget, rng)
    else:
        h1, max_corr1 = h0, np.correlate(h0, 1 - h0, mode="full").max()
    best_h = h1
    best_max_corr = max_corr1

    # ----- Phase 2 : upscale and SA -----
    factor = 2
    n2 = n0 * factor
    xs0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    xs2 = (np.arange(n2) + 0.5) * (2.0 / n2)
    h2 = np.interp(xs2, xs0, best_h)
    h2 = np.clip(h2, 0.0, 1.0)
    h2 = enforce_sum(h2, n2 / 2.0, rng)

    elapsed = time.time() - start_time
    remaining = max(0.0, budget_s - elapsed)
    phase2_budget = remaining * 0.70
    if phase2_budget > 5.0:
        h2_opt, max_corr2 = simulated_annealing(h2, phase2_budget, rng)
        best_h = h2_opt
        best_max_corr = max_corr2
    else:
        best_h = h2
        best_max_corr = np.correlate(h2, 1 - h2, mode="full").max()

    # ----- Phase 3 : SLSQP refinement (log‑sum‑exp) -----
    elapsed = time.time() - start_time
    remaining = max(0.0, budget_s - elapsed)
    # give about half of the rest to a quick SLSQP sweep
    slsqp_budget = remaining * 0.5
    if slsqp_budget > 2.0:
        beta_schedule = [5.0, 20.0, 80.0, 200.0, 800.0, 2000.0]
        for beta in beta_schedule:
            if time.time() - start_time > budget_s * 0.95:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, max_iter=300)
            except Exception:
                continue
            corr = np.correlate(h_ref, 1 - h_ref, mode="full")
            max_corr = corr.max()
            if max_corr < best_max_corr - 1e-12:
                best_h = h_ref
                best_max_corr = max_corr

    # ----- Phase 4 : tiny random tweaks (micro‑search) -----
    while time.time() - start_time < budget_s * 0.99:
        n = best_h.size
        src_idx = rng.choice(np.where(best_h > 1e-12)[0])
        dst_idx = rng.choice(np.where(best_h < 1.0 - 1e-12)[0])
        if src_idx == dst_idx:
            continue
        eps_max = min(best_h[src_idx], 1.0 - best_h[dst_idx])
        eps = eps_max * 0.05 * rng.random()
        if eps <= 0.0:
            continue
        cand = best_h.copy()
        cand[src_idx] -= eps
        cand[dst_idx] += eps
        cand = enforce_sum(np.clip(cand, 0.0, 1.0), n / 2.0, rng)
        c_cand = compute_c5(cand)
        c_best = compute_c5(best_h)
        if c_cand < c_best:
            best_h = cand
            best_max_corr = np.correlate(best_h, 1 - best_h, mode="full").max()

    # ----- final clean‑up -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
