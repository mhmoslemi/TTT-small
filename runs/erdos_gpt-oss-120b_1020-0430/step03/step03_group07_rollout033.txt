        step[block_len*2:block_len*3] = 1  # two blocks maybe
        candidates.append(enforce_sum(step, n/2, rng))

    # compute best initial
    best_h = None; best_c = float('inf')
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c = c; best_h = h.copy()

    # optimization phases
    # phase 1: SLSQP with increasing beta repeated several times
    beta_list = [5, 20, 80, 200, 800, 2000]
    for beta in beta_list:
        if time.time() >= deadline: break

            h_opt = slsqp_optimize(best_h, beta, max_iter=400)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt.copy()

            pass

    # optional upsampling if time left and n*2 <= 800

        h_opt, c_opt = hill_climb(best_h, rng, max_iters=6000, start_step=0.2)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()
        # SLSQP final
        for beta in beta_list:
            if time.time() >= deadline: break

                h_opt = slsqp_optimize(best_h, beta, max_iter=400)

                if c_opt < best_c:
                    best_c, best_h = c_opt, h_opt.copy()
            except Exception:
                pass

    best_h = np.clip(best_h, 0, 1)
    best_h = enforce_sum(best_h, best_h.size/2, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)

def enforce_sum(h, target, rng):
    """Project h onto [0,1]^n with sum = target using a simple iterative scheme."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    # random walk that preserves the sum
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        max_eps = min(h[i], 1.0 - h[i])  # how much we could move from i
        if diff > 0:  # need to add mass
            add = min(diff, max_eps)
        else:        # need to remove mass
            add = -min(-diff, max_eps)
        h[i] -= add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = np.clip(eps, -max_eps, max_eps)
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def correlate_full(h):
    """Full linear correlation of h with (1-h) using FFT."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Discrete C5 for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)

CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Log‑sum‑exp smoothing of max_k Σ_i x_i (1‑x_{i+k})."""
    conv = correlate_full(x)
    bc = CURRENT_BETA * conv
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=350):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

            h_opt = slsqp_optimize(best_h, beta, max_iter=400)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt.copy()
        except Exception:
            pass

    # b) Gradient descent on smooth max (high beta)
    for beta in (200.0, 800.0, 2000.0):
        if time.time() >= deadline:
            break
        try:
            h_opt, c_opt = grad_descent_optimize(
                best_h, beta, max_iters=2000, step0=0.3, rng=rng, deadline=deadline
            )
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt.copy()
        except Exception:
            pass

    # c) Intensive hill‑climbing / SA phase
    while time.time() < deadline:
        # occasional random restart
        if rng.random() < 0.2:
            start = random_feasible_vector(n, rng)
        else:
            start = perturb_h(best_h, rng, scale=0.35)

        # hill climb
        h_opt, c_opt = hill_climb(start, rng, max_iters=3000, start_step=0.4)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()

        # simulated annealing (short bursts)
        cand_h, cand_c = simulated_annealing(best_h, rng, deadline,
                                             T0=0.07, Tf=1e-5, max_iters=2500)
        if cand_c < best_c:
            best_c, best_h = cand_c, cand_h.copy()

    # ------------------------------------------------------------------
    # 5. Optional up‑sampling and refinement (if time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline * 0.99 and n * 2 <= 800:
        up_factor = 2
        best_h = np.repeat(best_h, up_factor)
        n = best_h.size
        best_h = enforce_sum(best_h, n / 2.0, rng)

        # Hill‑climb on finer grid
        h_opt, c_opt = hill_climb(best_h, rng, max_iters=5000, start_step=0.25)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()

        # Gradient descent on fine grid
        for beta in (800.0, 2000.0):
            if time.time() >= deadline:
                break
            try:
                h_opt, c_opt = grad_descent_optimize(
                    best_h, beta, max_iters=1500, step0=0.2, rng=rng, deadline=deadline
                )
                if c_opt < best_c:
                    best_c, best_h = c_opt, h_opt.copy()
            except Exception:
                pass

        # Final SLSQP passes on refined grid
        for beta in beta_list:
            if time.time() >= deadline:
                break
            try:
                h_opt = slsqp_optimize(best_h, beta, max_iter=400)
                c_opt = compute_c5(h_opt)
                if c_opt < best_c:
                    best_c, best_h = c_opt, h_opt.copy()
            except Exception:
                pass

    # ------------------------------------------------------------------
    # 6. Final clean‑up
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
