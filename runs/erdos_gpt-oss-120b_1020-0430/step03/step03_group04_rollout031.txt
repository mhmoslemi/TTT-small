def binary_sa(h_init, rng, max_iters, start_temp):
    h = h_init.copy()
    n = h.shape[0]
    cur_max = max_overlap(h)
    best_h = h.copy()
    best_max = cur_max
    T = start_temp
    # compute schedule: T *= cooling factor each iteration (or linear)
    for it in range(max_iters):
        # choose random swap: pick i w/ h[i]=1, j w/ h[j]=0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i], h[j] = 0, 1
        new_max = max_overlap(h)
        delta = new_max - cur_max
        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = 1, 0
        # update temperature
        # cooling schedule: geometric
        T *= 0.9995
        # optionally break early if T low, but will continue.
    return best_h, best_max

def project_onto_simplex_box(y, T):
    # y is vector, need to find lambda such that sum_i min(max(y_i + lambda, 0), 1) = T
    # Solution can be done by bisection on lambda in range [-max(y), 1 - min(y)].
    low = -np.max(y) - 1  # some lower bound
    high = 1 - np.min(y) + 1  # upper bound
    for _ in range(50):
        mid = (low + high) / 2.0
        x = np.clip(y + mid, 0.0, 1.0)
        s = x.sum()
        if s > T:
            high = mid
        else:
            low = mid
    return np.clip(y + (low + high) / 2.0, 0.0, 1.0)

import time
import numpy as np

# ---------- utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for step function h on [0,2] with uniform spacing."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr[n - 1 :])  # shifts k = 0..n-1
    return float(max_corr) * (2.0 / n)


def enforce_feasibility(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Clip to [0,1] and adjust to hit the exact sum `target` by moving mass."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, h.size)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h


def project_onto_box_sum(y: np.ndarray, target: float) -> np.ndarray:
    """Projection of y onto {x | 0≤x≤1, sum x = target} using bisection on the shift λ."""
    low = -np.max(y)  # makes all entries ≤0 after shift
    high = 1.0 - np.min(y)  # makes all entries ≥1 after shift
    for _ in range(25):
        mid = (low + high) / 2.0
        x = np.clip(y + mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            high = mid
        else:
            low = mid
    return np.clip(y + (low + high) / 2.0, 0.0, 1.0)


def subgradient_for_shift(h: np.ndarray, k: int) -> np.ndarray:
    """Subgradient of the linear piece corresponding to shift k."""
    n = h.size
    g = np.zeros_like(h)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        up = n - k
        g[:up] += 1.0 - h[k:]
        g[k:] -= h[:up]
    return g


def subgradient_descent(
    h0: np.ndarray,
    target: float,
    rng: np.random.Generator,
    max_iters: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the convex max‑overlap objective."""
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    # initial value
    corr = np.correlate(h, 1 - h, mode="full")
    maxv = np.max(corr[n - 1 :])
    best_c = float(maxv) * (2.0 / n)

    step0 = 5.0  # base step length
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode="full")
        O = corr[n - 1 :]
        maxv = O.max()
        active = np.where(np.abs(O - maxv) < 1e-12)[0]

        # average subgradient of all active shifts
        g = np.zeros_like(h)
        for k in active:
            g += subgradient_for_shift(h, int(k))
        g /= len(active)

        norm = np.linalg.norm(g)
        if norm > 0.0:
            direction = g / norm
        else:
            direction = g

        alpha = step0 / np.sqrt(it + 1.0)
        y = h - alpha * direction
        h = project_onto_box_sum(y, target)

        # evaluate
        corr = np.correlate(h, 1 - h, mode="full")
        maxv = np.max(corr[n - 1 :])
        c = float(maxv) * (2.0 / n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c


def perturb_h_sum_preserve(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_simple(h0: np.ndarray, rng: np.random.Generator, max_iters: int = 2000, start_step: float = 0.4):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_every = max(1, max_iters // 5)
    for it in range(max_iters):
        cand = perturb_h_sum_preserve(h, rng, scale=step)
        c = compute_c5(cand)
        if c < best_c:
            best_c = c
            best_h = cand.copy()
            h = cand
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector in [0,1] with exact sum n/2."""
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99  # safety margin

    # determine problem size
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        n = parent.size
    else:
        parent = None
        n = 180

    target = n / 2.0
    best_h = None
    best_c = np.inf
    best_n = n

    # ---------- generate starting vectors ----------
    starts = []

    if parent is not None:
        starts.append(parent.copy())

    # binary rounding of parent (if available)
    if parent is not None:
        h_bin = np.rint(parent).astype(float)
        h_bin = enforce_feasibility(h_bin, target, rng)
        starts.append(h_bin)

    # a few random continuous feasible vectors
    for _ in range(3):
        starts.append(random_feasible_vector(n, rng))

    # a few random binary vectors
    for _ in range(2):
        h_bin = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h_bin[idx] = 1.0
        starts.append(h_bin)

    # ---------- optimise each start ----------
    for h0 in starts:
        if time.time() > deadline:
            break
        h = enforce_feasibility(h0.copy(), target, rng)

        # quick hill‑climb to get a decent local point
        h, c = hill_climb_simple(h, rng, max_iters=1500, start_step=0.6)
        if c < best_c:
            best_c, best_h, best_n = c, h.copy(), n

        # sub‑gradient descent (major work)
        remain = deadline - time.time()
        if remain > 2.0:
            max_iters = int(min(40000, remain * 2500))  # approx 2500 it/s
            h, c = subgradient_descent(h, target, rng, max_iters, deadline)
            if c < best_c:
                best_c, best_h, best_n = c, h.copy(), n

        # final short hill‑climb polish
        if time.time() < deadline:
            h, c = hill_climb_simple(h, rng, max_iters=1500, start_step=0.4)
            if c < best_c:
                best_c, best_h, best_n = c, h.copy(), n

    # ---------- final safety polish ----------
    if best_h is None:
        best_n = n
        best_h = np.full(best_n, 0.5)

    best_h = enforce_feasibility(best_h, best_n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
