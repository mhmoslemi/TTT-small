def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # compute softmax: log sum exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # precise sum correction by redistribution
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

import time, numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global variables for SLSQP soft‑max objective
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    """discrete C5 for step function on [0,2] (dx = 2 / n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, rng):
    """Adjust h to satisfy Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Random continuous feasible h (0≤h≤1, Σh=n/2)."""
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def worst_gradient(h):
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]          # subtract h_{i+k}
        grad[k:] -= h[: n - k]          # subtract h_{i-k}
    return grad


def guided_perturb(h, grad, rng, scale=0.5):
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=400):
    """Soft‑max SLSQP optimisation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def hill_climb(start_h, rng, max_iters, step_scale_initial):
    """Simulated‑annealing‑style hill‑climb on the worst‑shift objective."""
    n = start_h.size
    cur_h = start_h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    step_scale = step_scale_initial
    temp0 = max((cur_c - 0.25) * 0.1, 1e-4)
    temp_final = temp0 * 0.01
    for it in range(max_iters):
        t = temp0 * ((temp_final / temp0) ** (it / max_iters)) if temp0 > 0 else 0.0
        grad = worst_gradient(cur_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if rng.random() < 0.02:          # occasional random indices
            i = rng.integers(0, n)
            j = rng.integers(0, n)
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps * step_scale
        nh = cur_h.copy()
        nh[i] -= eps
        nh[j] += eps
        nc = compute_c5(nh)
        accept = False
        if nc < cur_c:
            accept = True
        elif t > 0 and rng.random() < np.exp(-(nc - cur_c) / t):
            accept = True
        if accept:
            cur_h, cur_c = nh, nc
            if nc < best_c:
                best_h, best_c = nh.copy(), nc
        if (it + 1) % max(1, max_iters // 10) == 0:
            step_scale = max(step_scale * 0.8, 0.01)
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # ------------------------------------------------------------------
    # start from parent construction if supplied
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0, c0, h0.size

    candidate_n = [84, 96, 108, 120, 144, 168, 192, 216, 240]
    if best_n is not None:
        order_n = [best_n] + [n for n in candidate_n if n != best_n]
    else:
        order_n = candidate_n[:]

    # ------------------------------------------------------------------
    # main search: many random restarts with hill‑climb + SLSQP
    # ------------------------------------------------------------------
    for n in order_n:
        while time.time() - start < 0.94 * budget_s:
            # ---------- initial vector ----------
            if rng.random() < 0.5:
                h = np.zeros(n)
                h[rng.choice(n, n // 2, replace=False)] = 1.0
            else:
                h = random_feasible_vector(n, rng)

            # occasional interpolation from current best
            if best_h is not None and best_n != n and rng.random() < 0.3:
                x_old = np.linspace(0, 2, best_n, endpoint=False)
                x_new = np.linspace(0, 2, n, endpoint=False)
                h = np.interp(x_new, x_old, best_h)
                h = np.clip(h, 0.0, 1.0)
                h = enforce_sum(h, rng)

            # ---------- hill‑climb ----------
            max_it = 2500 if n <= 144 else 2000
            h_loc, c_loc = hill_climb(h, rng, max_iters=max_it, step_scale_initial=0.5)

            # ---------- SLSQP refinement ----------
            for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0):
                if time.time() - start > 0.96 * budget_s:
                    break
                h_loc = slsqp_optimize(h_loc, beta, rng, max_iter=400)
                c_loc = compute_c5(h_loc)
                if c_loc < best_c:
                    best_h, best_c, best_n = h_loc.copy(), c_loc, n

            if c_loc < best_c:
                best_h, best_c, best_n = h_loc.copy(), c_loc, n

            if best_c < 0.3808:
                break
        if best_c < 0.3808:
            break

    # ------------------------------------------------------------------
    # optional up‑sampling of the best solution
    # ------------------------------------------------------------------
    if best_h is not None and time.time() - start < 0.99 * budget_s:
        if best_n * 2 <= 300:
            h_up = np.repeat(best_h, 2)
            n_up = h_up.size
            h_up, c_up = hill_climb(h_up, rng, max_iters=2000, step_scale_initial=0.3)
            for beta in (5.0, 20.0, 80.0, 200.0):
                if time.time() - start > budget_s:
                    break
                h_up = slsqp_optimize(h_up, beta, rng, max_iter=300)
                c_up = compute_c5(h_up)
                if c_up < best_c:
                    best_h, best_c, best_n = h_up.copy(), c_up, n_up

    # ------------------------------------------------------------------
    # final tiny guided perturbations (remaining budget)
    # ------------------------------------------------------------------
    while time.time() - start < budget_s * 0.999 and best_h is not None:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # ------------------------------------------------------------------
    # final refinement on the best solution
    # ------------------------------------------------------------------
    if best_h is not None and time.time() - start < budget_s * 0.998:
        best_h, best_c = hill_climb(best_h, rng, max_iters=5000, step_scale_initial=0.2)
        for beta in (500.0, 1000.0, 2000.0):
            if time.time() - start > budget_s:
                break
            best_h = slsqp_optimize(best_h, beta, rng, max_iter=400)
            best_c = compute_c5(best_h)

    # ------------------------------------------------------------------
    # ensure feasibility and return
    # ------------------------------------------------------------------
    if best_h is None:
        best_n = 84
        best_h = np.full(best_n, 0.5)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
