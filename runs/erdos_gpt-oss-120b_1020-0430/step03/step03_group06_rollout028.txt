def solve_milp_binary(n, time_limit):
    import cvxpy as cp
    m = n // 2
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(h) == m, t >= 0]
    # create dict for y variables: maybe store as list per shift.
    for k in range(-(n-1), n):
        start_i = max(0, -k)
        end_i = min(n-1, n-1 - k)
        length = end_i - start_i + 1
        if length <= 0: continue
        region_h = h[start_i:end_i+1]  # cp slice
        sum_h_region = cp.sum(region_h)
        y_vars = []
        for i in range(start_i, end_i+1):
            j = i + k
            y = cp.Variable(boolean=True)
            # linking constraints
            constraints.append(y <= h[i])
            constraints.append(y <= h[j])
            constraints.append(y >= h[i] + h[j] - 1)
            y_vars.append(y)
        sum_y = cp.sum(y_vars) if y_vars else 0
        constraints.append(sum_h_region - sum_y <= t)
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CBC, warm_start=True, max_iters=100000, max_seconds=time_limit)
    except Exception:
        # fallback to GLPK_MI
        prob.solve(solver=cp.GLPK_MI, max_seconds=time_limit)
    if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE, cp.FEASIBLE]:
        # maybe not solved
        # Return None to indicate failure.
        return None, None
    h_val = np.round(h.value).astype(float)  # may already be integer.
    t_val = float(t.value)
    return h_val, t_val

def hill_climb_binary(h, rng, max_iters=1000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    for _ in range(max_iters):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        new_h = h.copy()
        new_h[i] = 0
        new_h[j] = 1
        c = compute_c5(new_h)
        if c < best_c - 1e-12:
            best_c = c
            best_h = new_h.copy()
            h = new_h

            zeros = np.where(h <= 0.5)[0]

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

import time
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Core evaluation
# --------------------------------------------------------------
def compute_c5(h):
    """Exact C₅ for a discrete step function."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


# --------------------------------------------------------------
# Simple hill‑climb for continuous vectors
# --------------------------------------------------------------
def hill_climb(h_start, rng, max_iters=3000, start_step=0.5):
    """Transfer a small amount between two coordinates (preserves sum)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        i, j = rng.integers(0, n, size=2)
        while j == i:
            j = rng.integers(0, n)
        max_pos = min(h[i], 1.0 - h[j])
        max_neg = min(h[j], 1.0 - h[i])
        if max_pos + max_neg == 0:
            continue
        eps = (rng.random() * (max_pos + max_neg)) - max_neg
        eps *= step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


# --------------------------------------------------------------
# Greedy improvement for binary vectors
# --------------------------------------------------------------
def greedy_improve_binary(h, rng, max_iters=5000):
    """Repeatedly fix the worst shift by swapping mismatched bits."""
    n = h.size
    scale = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h)
    no_improve = 0
    for _ in range(max_iters):
        corr = np.correlate(h, 1 - h, mode="full")
        max_idx = int(np.argmax(corr))
        cur_c = corr[max_idx] * scale
        if cur_c < best_c - 1e-12:
            best_c = cur_c
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 300:
                break
        shift = max_idx - (n - 1)
        start = max(0, -shift)
        end = min(n - 1, n - 1 - shift)
        if start > end:
            continue
        i_range = np.arange(start, end + 1)
        j_range = i_range + shift
        bad = i_range[(h[i_range] == 1) & (h[j_range] == 0)]
        good = i_range[(h[i_range] == 0) & (h[j_range] == 1)]
        if bad.size and good.size:
            i = int(rng.choice(bad))
            j = int(rng.choice(good))
            h[i] = 0.0
            h[j] = 1.0
        else:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
            h[i] = 0.0
            h[j] = 1.0
    return best_h, best_c


# --------------------------------------------------------------
# Smooth surrogate for SLSQP (log‑sum‑exp of the correlation)
# --------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=400):
    """Run SLSQP on the smooth surrogate for a fixed β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny exact‑sum repair
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = np.random.randint(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            if abs(add) < 1e-12:
                continue
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

# --------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # ----------------------------------------------------------
    # Determine problem size
    # ----------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        # enforce exact integral
        target = n / 2.0
        diff = target - h0.sum()
        if abs(diff) > 1e-12:
            i = rng.integers(0, n)
            h0[i] = np.clip(h0[i] + diff, 0.0, 1.0)
        h0 = np.clip(h0, 0.0, 1.0)
    else:
        n = 180
        h0 = np.full(n, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ----------------------------------------------------------
    # Random binary restarts with greedy refinement
    # ----------------------------------------------------------
    while time.time() - start < budget_s * 0.65:
        # random binary seed
        h_bin = np.zeros(n, dtype=float)
        ones = rng.choice(n, size=n // 2, replace=False)
        h_bin[ones] = 1.0
        h_opt, c_opt = greedy_improve_binary(h_bin, rng, max_iters=5000)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()
        # occasional plain hill‑climb from the current best
        if rng.random() < 0.3:
            h_hc, c_hc = hill_climb(best_h, rng, max_iters=2000, start_step=0.4)
            if c_hc < best_c:
                best_c, best_h = c_hc, h_hc.copy()

    # ----------------------------------------------------------
    # SLSQP refinement with increasing β
    # ----------------------------------------------------------
    beta_seq = [20.0, 80.0, 200.0, 600.0, 1500.0]
    for beta in beta_seq:
        if time.time() - start > budget_s * 0.95:
            break
        refined = slsqp_optimize(best_h, beta, max_iter=500)
        c_ref = compute_c5(refined)
        if c_ref < best_c:
            best_c, best_h = c_ref, refined.copy()

    # ----------------------------------------------------------
    # Final cleaning – enforce exact sum
    # ----------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    target = n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        # adjust a single coordinate
        i = rng.integers(0, n)
        best_h[i] = np.clip(best_h[i] + diff, 0.0, 1.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
