def perturb_move_biased(h, rng, scale):
    n = len(h)
    # choose donor i with probability proportional to h[i] (prefer non-zero)
    probs_i = h / h.sum() if h.sum() > 0 else np.ones(n)/n
    i = rng.choice(n, p=probs_i)
    # choose recipient j with probability proportional to (1 - h[j])
    probs_j = (1.0 - h) / ((1.0 - h).sum())
    j = rng.choice(n, p=probs_j)
    if i == j:
        j = (j+1) % n
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0: return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def random_feasible_vector(n, rng):

def hill_climb(h_start, rng, max_iters=2000, start_step=0.2):

    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = len(h)
    step = start_step
    decay_iter = max(1, max_iters // 10)
    for it in range(max_iters):
        # propose perturb_move_biased with scale=step
        new_h = perturb_move_biased(h, rng, step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it+1) % decay_iter == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Global placeholders for the soft‑max surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_to_feasible(v, target):
    """Project v onto {0≤h≤1, Σh = target} via a bisection on a shift."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Fast generation of a random feasible vector (0≤h≤1, Σh=n/2)."""
    v = rng.random(n)
    return project_to_feasible(v, n / 2.0)


def perturb_move_biased(h, rng, scale):
    """Transfer a small amount ε from donor i to recipient j (preserves sum)."""
    n = h.size
    # donor probabilities ~ h (prefer non‑zero)
    dh = h.copy()
    dh_sum = dh.sum()
    if dh_sum == 0:
        probs_i = np.full(n, 1.0 / n)
    else:
        probs_i = dh / dh_sum
    i = rng.choice(n, p=probs_i)
    # recipient probabilities ~ (1‑h) (prefer non‑full)
    rec = 1.0 - h
    rec_sum = rec.sum()
    if rec_sum == 0:
        probs_j = np.full(n, 1.0 / n)
    else:
        probs_j = rec / rec_sum
    j = rng.choice(n, p=probs_j)
    if i == j:
        j = (j + 1) % n
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def anneal_optimize(start_h, rng, time_budget,
                    start_temp=0.05, end_temp=1e-7,
                    scale_start=1.0, scale_end=0.01):
    """Simulated annealing that respects the sum constraint."""
    start = time.time()
    cur_h = start_h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        frac = elapsed / time_budget
        temp = start_temp * (1.0 - frac) + end_temp * frac
        scale = scale_start * (1.0 - frac) + scale_end * frac
        cand_h = perturb_move_biased(cur_h, rng, scale)
        cand_c = compute_c5(cand_h)
        accept = False
        if cand_c < cur_c:
            accept = True
        else:
            # Metropolis acceptance
            if temp > 0:
                prob = np.exp(-(cand_c - cur_c) / temp)
                if rng.random() < prob:
                    accept = True
        if accept:
            cur_h = cand_h
            cur_c = cand_c
            if cand_c < best_c:
                best_c = cand_c
                best_h = cand_h.copy()
    return best_h, best_c


def slsqp_optimize(h0, beta, rng, max_iter=800):
    """Run SLSQP on the smooth surrogate with given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-13, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact projection to the hyperplane
    x = project_to_feasible(x, CURRENT_TARGET)
    return x


def hill_climb(h_start, rng, max_iters=3000, start_step=0.3):
    """Deterministic hill‑climb (accept only improvements) with decaying step."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_move_biased(h, rng, step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-5)
    return best_h, best_c

    if remain > 5.0:
        # allocate remaining time to hill‑climb (iterations proportional to time)
        iters = int(min(8000, remain * 1500))
        best_h, best_c = hill_climb(best_h, rng, max_iters=iters, start_step=0.15)

    # ---------- Final feasibility enforcement ----------
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        best_h = project_to_feasible(best_h, target_sum)

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
