            global_best_h = h_up
            n = n_up

    global_best_h = np.clip(global_best_h, 0.0, 1.0)
    global_best_h = enforce_sum(global_best_h, global_best_h.size / 2.0, rng)
    final_c = compute_c5(global_best_h)

    return global_best_h.tolist(), float(final_c), int(global_best_h.size)

def adam_fine_tune(h0, rng, start_time, deadline, beta=200.0, steps=2000, lr=0.01):
    h = h0.copy()
    n = h.size
    target = n/2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    for t in range(1, steps+1):
        if time.time() >= deadline:
            break
        # compute gradient of softmax_obj
        conv = correlate_full(h)
        # softmax weights
        maxc = conv.max()
        e = np.exp(beta * (conv - maxc))
        p = e / e.sum()
        # gradient: for each i, sum_k p_k * ((1 - h_{i+k}) - h_{i-k})
        # compute term1 = conv of p with (1 - h) shifted forward
        # term2 = conv of p with h shifted backward ?
        # We'll compute gradient via explicit loops for simplicity given n moderate.
        grad = np.zeros(n)
        # compute for each i the sum over k where i+k < n:
        # term pos = Σ_{k: i+k<n} p_{k} * (1 - h_{i+k})
        # term neg = Σ_{k: i-k>=0} p_{k} * h_{i-k}
        # But p indexed by shift k from -(n-1)..(n-1); we need mapping.
        # Instead compute using convolution approach: expand p to array of length 2n-1, treat as zero-padded and compute shifted contributions.
        # We'll build arrays for (1 - h) and h padded appropriately.
        L = 2*n - 1
        # Pad (1-h) with zeros on both sides to length L
        one_minus_h_pad = np.concatenate([np.zeros(n-1), 1 - h, np.zeros(n-1)])
        h_pad = np.concatenate([np.zeros(n-1), h, np.zeros(n-1)])
        # p array maps to shifts: index offset = n-1
        # pos term: for each i, sum_{k} p_{k} * (1 - h)_{i+k}
        # which corresponds to (p convolve one_minus_h_pad)[i + (n-1) + (n-1)]? Let's compute via direct loops maybe simpler.
        for i in range(n):
            # positive contributions
            # k range such that i+k < n -> k from -(i) to n-1-i
            k_min = -i
            k_max = n - 1 - i
            # convert k to index in p: idx = k + (n-1)
            idxs = np.arange(k_min, k_max+1) + (n-1)
            pos = np.sum(p[idxs] * (1 - h[i + np.arange(k_min, k_max+1)])
)
            # negative contributions: k such that i - k >=0 => k from i - (n-1) to i
            k_min2 = i - (n-1)
            k_max2 = i
            idxs2 = np.arange(k_min2, k_max2+1) + (n-1)
            neg = np.sum(p[idxs2] * h[i - np.arange(k_min2, k_max2+1)])
            grad[i] = pos - neg
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h = h - step
        # project onto [0,1] and sum
        h = np.clip(h, 0, 1)
        # adjust sum to target
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # distribute diff over free elements
            free = (h > 0) & (h < 1)
            if free.any():
                add = diff / free.sum()
                h[free] += add
                # clip again
                h = np.clip(h, 0, 1)
            else:
                # fallback random correction
                h = enforce_sum(h, target, rng)
    # after loop, enforce sum exactly
    h = enforce_sum(h, target, rng)
    c = compute_c5(h)
    return h, c

import time
import numpy as np
from scipy.optimize import minimize

# -------------------------------------------------
# Helper utilities (feasibility, perturbation, etc.)
# -------------------------------------------------
def enforce_sum(h, target, rng):
    """Adjust h in-place so that sum(h) == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = diff if diff > 0 else diff
        if diff > 0:
            add = min(add, 1.0 - h[i])
        else:
            add = max(add, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.4
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount between two coordinates (preserves the sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


# -------------------------------------------------
# Correlation / objective
# -------------------------------------------------
def correlate_full(h):
    """Linear correlation of h with (1‑h) reversed, using FFT."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    return correlate_full(h).max() * (2.0 / h.size)


# -------------------------------------------------
# Smooth surrogate for max (log‑sum‑exp)
# -------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """β‑scaled log‑sum‑exp of the correlation values."""
    conv = correlate_full(x)
    bc = CURRENT_BETA * conv
    m = bc.max()
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=400):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # small sum correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = diff if diff > 0 else diff
            if diff > 0:
                add = min(add, 1.0 - x[i])
            else:
                add = max(add, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# -------------------------------------------------
# Simple hill‑climb (greedy)
# -------------------------------------------------
def hill_climb(h_start, rng, max_iters=2000, start_step=0.4):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay = max(1, max_iters // 8)
    for it in range(max_iters):
        nh = perturb_h(h, rng, scale=step)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.94  # keep a safety margin

    # ----- 1. grid size and initial vector -----------------
    if "initial_h_values" in globals():

        h0 = np.array(initial_h_values, dtype=float)

    # ----- 3. pick best initial candidate ------------------
    best_h = None

    for h in candidates:

        if c < best_c:

    global_best_h, global_best_c = best_h, best_c

    # ----- 4. optimisation parameters ----------------------
    beta_list = [5.0, 20.0, 80.0, 200.0, 800.0, 2000.0]
