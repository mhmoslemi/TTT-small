def enforce_sum(h, target, rng, max_iter=2000):
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        # Choose random index
        i = rng.integers(0, n)
        # If diff > 0, we need to increase h[i]; max increase is 1 - h[i]
        # If diff < 0, we need to decrease; max decrease is h[i]
        max_delta = (1 - h[i]) if diff > 0 else -h[i]
        # delta = clamp(diff, -max_delta, max_delta)
        # Actually if max_delta > 0, we can adjust
        if diff > 0:
            delta = min(diff, max_delta)
        else:
            delta = max(diff, max_delta)  # note max_delta negative
        if abs(delta) < 1e-15:
            continue
        h[i] += delta
        diff = target - h.sum()
    return h

def differential_evolution_optimize(n, rng, h_initial, budget_s, start_time):
    target = n/2.0
    NP = max(20, n // 6)  # ~30
    pop = np.empty((NP, n), dtype=float)
    # first individual
    if h_initial is not None:
        pop[0] = h_initial.copy()
    else:
        pop[0] = random_feasible_vector(n, rng)
    for i in range(1, NP):
        pop[i] = random_feasible_vector(n, rng)
    fitness = np.array([compute_c5(ind) for ind in pop])
    best_idx = np.argmin(fitness)
    best_h = pop[best_idx].copy()
    best_f = fitness[best_idx]

    F = 0.8
    CR = 0.9
    # success counter for adaptation
    successes = 0
    trials = 0

    while time.time() - start_time < budget_s:
        # optional adaptation after each generation
        for i in range(NP):
            # ensure still within time for at least candidate generation
            if time.time() - start_time >= budget_s:
                break
            # select a,b,c distinct from i
            idxs = list(range(NP))
            idxs.remove(i)
            a,b,c = rng.choice(idxs, size=3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            enforce_sum(mutant, target, rng)
            # crossover produce trial vector
            jrand = rng.integers(0, n)
            trial = np.empty(n, dtype=float)
            # vectorized:
            mask = rng.random(n) < CR
            mask[jrand] = True
            trial[mask] = mutant[mask]
            trial[~mask] = pop[i][~mask]
            enforce_sum(trial, target, rng)
            f = compute_c5(trial)
            trials += 1
            if f <= fitness[i]:
                pop[i] = trial
                fitness[i] = f
                successes += 1
                if f < best_f:
                    best_f = f
                    best_h = trial.copy()
        # optional adaptation every generation
        # adjust F & CR based on success rate
        if trials > 0:
            success_rate = successes / trials
            # simple adaptation: if success_rate > 0.2 increase F? else decrease
            if success_rate > 0.2:
                F = min(F + 0.02, 0.9)
                CR = min(CR + 0.02, 0.95)
            else:
                F = max(F - 0.02, 0.5)
                CR = max(CR - 0.02, 0.6)
            successes = 0
            trials = 0
    return best_h, best_f

def local_optim(h, rng, iters):
    best = h.copy()
    best_c = compute_c5(best)
    n = h.size
    target = n/2.0
    for _ in range(iters):
        i, j = rng.integers(0, n, size=2)
        if i == j: continue
        max_eps = min(best[i], 1 - best[j])
        if max_eps <= 0: continue
        # random epsilon small relative to 0.1

        cand = best.copy()
        cand[i] -= eps

        # sum remains same

        if cand[i] < 0 or cand[i] > 1 or cand[j] < 0 or cand[j] > 1:

        c = compute_c5(cand)
        if c < best_c:
            best, best_c = cand, c
    return best, best_c

def compute_c5(h):
    """Discrete C5 for step function h on [0,2] with n points."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=2000):
    """Adjust h in‑place so that sum(h)=target while 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            delta = min(diff, 1.0 - h[i])
        else:
            delta = max(diff, -h[i])
        if abs(delta) < 1e-15:
            continue
        h[i] += delta
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    h = np.full(n, 0.5)
    # random swaps preserving bounds
    for _ in range(n * 5):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    enforce_sum(h, n / 2.0, rng)
    return h


# ---------- SLSQP surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp stable
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """Smooth‑max minimisation with SLSQP, preserving sum and bounds."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {"type": "eq", "fun": sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x


# ---------- Differential Evolution ----------
def differential_evolution_optimize(n, rng, h_initial, budget_s, start_time):
    """Population‑based DE on C5 with sum constraint."""
    target = n / 2.0
    NP = max(20, n // 6)                     # ~30 for n=180
    pop = np.empty((NP, n), dtype=float)
    pop[0] = h_initial.copy()
    for i in range(1, NP):
        pop[i] = random_feasible_vector(n, rng)
    fitness = np.array([compute_c5(ind) for ind in pop])
    best_idx = np.argmin(fitness)
    best_h = pop[best_idx].copy()
    best_f = fitness[best_idx]

    F, CR = 0.8, 0.9
    succ, trials = 0, 0

    while time.time() - start_time < budget_s:
        for i in range(NP):
            if time.time() - start_time >= budget_s:
                break
            idxs = list(range(NP))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            enforce_sum(mutant, target, rng)

            jrand = rng.integers(0, n)
            mask = rng.random(n) < CR
            mask[jrand] = True
            trial = np.where(mask, mutant, pop[i])
            enforce_sum(trial, target, rng)

            f = compute_c5(trial)
            trials += 1
            if f <= fitness[i]:
                pop[i] = trial
                fitness[i] = f
                succ += 1
                if f < best_f:
                    best_f, best_h = f, trial.copy()

        # simple adaptive control
        if trials:
            rate = succ / trials
            if rate > 0.2:
                F = min(F + 0.02, 0.9)
                CR = min(CR + 0.02, 0.95)
            else:
                F = max(F - 0.02, 0.5)
                CR = max(CR - 0.02, 0.6)
            succ, trials = 0, 0

    return best_h, best_f


# ---------- Refinement ----------
def refine_h(best_h, budget_s, start_time):
    """Apply a cascade of SLSQP refinements (increasing beta)."""
    best_c = compute_c5(best_h)
    for beta in (50, 200, 800, 2000, 8000, 20000):
        if time.time() - start_time >= budget_s:
            break
        try:
            refined = slsqp_optimize(best_h, beta, max_iter=400)
            rc = compute_c5(refined)
            if rc < best_c:
                best_c, best_h = rc, refined
        except Exception:
            continue
    return best_h, best_c


# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # initialise
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        np.clip(h0, 0.0, 1.0, out=h0)
        enforce_sum(h0, n / 2.0, rng)
    else:
        n = 180
        h0 = random_feasible_vector(n, rng)

    # allocate ~70 % of time for DE, rest for refinement
    de_budget = budget_s * 0.70
    best_h, _ = differential_evolution_optimize(
        n, rng, h0, budget_s=de_budget, start_time=start
    )

    # final polishing with SLSQP
    if time.time() - start < budget_s:
        best_h, _ = refine_h(best_h, budget_s, start)

    # final safety checks
    enforce_sum(best_h, n / 2.0, rng)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
