def local_improve(h, max_iters=1000):
    n = h.size
    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode='full')
        k_idx = np.argmax(corr)
        shift = k_idx - (n - 1)
        # compute product
        i_start = max(0, -shift)
        i_end = min(n, n - shift)
        idx = np.arange(i_start, i_end)
        prod = h[idx] * (1 - h[idx + shift])
        # find high product and low product indices
        if prod.size < 2:
            break
        hi_idx = idx[np.argmax(prod)]
        lo_idx = idx[np.argmin(prod)]
        # we will try to move epsilon from hi_idx to lo_idx (preserve sum)
        max_eps = min(h[hi_idx], 1 - h[lo_idx])
        if max_eps <= 1e-12:
            # try swapped direction (increase hi, decrease lo)
            max_eps2 = min(h[lo_idx], 1 - h[hi_idx])
            if max_eps2 <= 1e-12:
                break
            eps = max_eps2 * 0.5
            new = h.copy()
            new[hi_idx] += eps
            new[lo_idx] -= eps
        else:
            eps = max_eps * 0.5
            new = h.copy()
            new[hi_idx] -= eps
            new[lo_idx] += eps
        new_c = compute_c5(new)
        if new_c < compute_c5(h) - 1e-9:
            h = new
        else:
            # try smaller epsilon?
            break
    return h

import time, math, numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / h.size)

def enforce_sum(h, target, rng, max_iter=5000):
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-15:
            continue
        h[i] += add
        diff = target - h.sum()
    return h

def random_binary_vector(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n//2, replace=False)
    h[idx] = 1.0
    return h

def random_continuous_vector(n, rng):
    h = rng.random(n)
    enforce_sum(h, n/2., rng)
    return h

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    # smooth approximation of max over corr
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

import time, numpy as np
from scipy.optimize import minimize

# globals for SLSQP objective
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    """Discrete C5 for step function h (dx = 2 / n)."""
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / h.size)

def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h in‑place so that sum(h)=target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-15:
            continue
        h[i] += add
        diff = target - h.sum()
    return h

def random_binary_vector(n, rng):
    """Feasible binary h with exactly n/2 ones."""
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def random_continuous_vector(n, rng):
    """Uniform random h, then project onto the bounded simplex."""
    h = rng.random(n)
    enforce_sum(h, n / 2.0, rng)
    return h

def perturb_h(h, rng, scale=0.2):
    """Transfer ε between two coordinates (preserves sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

# ----- SLSQP with smooth max approximation -----
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=80):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {"type": "eq", "fun": sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

    """Run SLSQP sequentially over a schedule of betas."""
    h = h0.copy()
    for beta in beta_sched:
        h = slsqp_optimize(h, beta, max_iter=max_iter)
    return h

def local_improve(h, max_iters=2000, rng=None, time_limit=None, start_time=None):

            eps = max_eps * 0.5

            new[hi] += eps

            new[hi] -= eps
