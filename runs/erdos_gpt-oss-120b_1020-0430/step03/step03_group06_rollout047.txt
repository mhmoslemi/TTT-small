import time, math, numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global helpers for smooth surrogate used by SLSQP
# ----------------------------------------------------------------------
_CURRENT_TARGET = None
_CURRENT_BETA = None


def _sum_eq_constraint(x):
    return np.sum(x) - _CURRENT_TARGET


def _softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = _CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / _CURRENT_BETA


# ----------------------------------------------------------------------
# Problem‑specific utilities
# ----------------------------------------------------------------------
def compute_c5(h):
    """Exact discrete C₅ for a step function h (dx = 2/len(h))."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)


def enforce_sum(h, target, rng):
    """Adjust h so that Σh = target while 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-12:
            continue
        h[i] += add
        diff = target - h.sum()
    # tiny final fix
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(1000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            if abs(add) < 1e-12:
                continue
            h[i] += add
            diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_binary_vector(n, rng):
    """Binary vector with exactly n/2 ones."""
    idx = rng.choice(n, n // 2, replace=False)
    h = np.zeros(n, dtype=float)
    h[idx] = 1.0
    return h


def random_feasible_vector(n, rng):
    """Continuous feasible vector (0≤h≤1, Σh=n/2)."""
    v = rng.exponential(1.0, size=n)
    v = v / v.sum() * (n / 2.0)
    h = np.clip(v, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def perturb_swap(h, rng, swaps=3):
    """Swap a few 1↔0 entries (preserves exact sum)."""
    h = h.copy()
    n = h.size
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return h
    for _ in range(swaps):
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
    return h


def hill_climb_random(h0, rng, max_iters=2500, start_step=0.5):
    """Simple hill‑climb using continuous transfers."""
    h = h0.copy()
    best_h, best_c = h.copy(), compute_c5(h)
    step = start_step
    n = h.size
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        i, j = rng.integers(0, n, size=2)
        while j == i:
            j = rng.integers(0, n)
        max_pos = min(h[i], 1.0 - h[j])
        max_neg = min(h[j], 1.0 - h[i])
        if max_pos + max_neg == 0:
            continue
        eps = (rng.random() * (max_pos + max_neg)) - max_neg
        eps *= step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h.copy()
            h = new_h
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


def hill_climb_targeted(h0, rng, max_iters=2000, start_step=0.3):
    """Greedy climb focusing on the currently worst shift."""
    h = h0.copy()
    best_h, best_c = h.copy(), compute_c5(h)
    curr_c = best_c
    step = start_step
    n = h.size
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        # worst shift
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        # build w_i = 1 - h[i+shift] for overlapping indices, else 0
        w = np.zeros(n, dtype=float)
        i_start = max(0, -shift)
        i_end = min(n, n - shift)
        w[i_start:i_end] = 1.0 - h[i_start + shift : i_end + shift]
        # choose donor (large w*h) and receiver (large (1-w)*(1-h))
        i_scores = h * w
        j_scores = (1.0 - h) * (1.0 - w)
        i_idx = np.argmax(i_scores)
        j_idx = np.argmax(j_scores)
        if i_idx == j_idx:
            i_idx = rng.integers(0, n)
            j_idx = rng.integers(0, n)
            while j_idx == i_idx:
                j_idx = rng.integers(0, n)
        eps = step * min(h[i_idx], 1.0 - h[j_idx])
        if eps < 1e-12:
            step = max(step * 0.7, 1e-6)
            continue
        new_h = h.copy()
        new_h[i_idx] -= eps
        new_h[j_idx] += eps
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h.copy()
        if c <= curr_c:
            h, curr_c = new_h, c
        else:
            # occasional uphill acceptance
            if rng.random() < 0.001:
                h, curr_c = new_h, c
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-6)
    return best_h, best_c


def slsqp_optimize(h0, beta, max_iter=800):
    """Run SLSQP on the smooth surrogate for a given β."""
    global _CURRENT_TARGET, _CURRENT_BETA
    n = h0.size
    _CURRENT_TARGET = n / 2.0
    _CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = _CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            if abs(add) < 1e-12:
                continue
            x[i] += add
            diff = _CURRENT_TARGET - x.sum()
    return x

def run(seed=42, budget_s=1100, **kwargs):

    start_time = time.time()

    # Determine dimension and starting point

        n = h0.size

    else:
        n = 180
        h0 = random_binary_vector(n, rng)

    global_best_h = h0.copy()
    global_best_c = compute_c5(global_best_h)

    # ------------------------------------------------------------------
    # Parameters
    # ------------------------------------------------------------------
    beta_list = [20.0, 80.0, 200.0, 600.0, 1200.0, 2400.0]
    time_limit = budget_s * 0.90

    # ------------------------------------------------------------------
    while time.time() - start_time < time_limit:
        # start point
        if rng.random() < 0.6:
            start = perturb_swap(global_best_h, rng, swaps=2)
        else:
            start = random_binary_vector(n, rng)

        # random hill‑climb
        h1, c1 = hill_climb_random(start, rng, max_iters=1800, start_step=0.5)

        # targeted hill‑climb
        h2, c2 = hill_climb_targeted(h1, rng, max_iters=1800, start_step=0.3)

        # accept improvement
        if c2 < global_best_c:
            global_best_h, global_best_c = h2.copy(), c2

            # smooth SLSQP refinements
            for beta in beta_list:
                if time.time() - start_time > budget_s * 0.95:
                    break
                try:
                    refined = slsqp_optimize(global_best_h, beta, max_iter=500)
                except Exception:
                    continue
                c_ref = compute_c5(refined)
                if c_ref < global_best_c:
                    global_best_h, global_best_c = refined.copy(), c_ref

        # occasional up‑sampling if time permits
        if time.time() - start_time > budget_s * 0.70 and n * 2 <= 360:
            up = np.repeat(global_best_h, 2)
            n_up = up.size
            up = enforce_sum(up, n_up / 2.0, rng)
            h_up, c_up = hill_climb_random(up, rng, max_iters=1200, start_step=0.3)
            h_up, c_up = hill_climb_targeted(h_up, rng, max_iters=1200, start_step=0.2)
            if c_up < global_best_c:

    # ------------------------------------------------------------------
    # Final clean‑up
    # ------------------------------------------------------------------
    global_best_h = np.clip(global_best_h, 0.0, 1.0)
    global_best_h = enforce_sum(global_best_h, n / 2.0, rng)
    final_c = compute_c5(global_best_h)

    return global_best_h.tolist(), float(final_c), int(n)
