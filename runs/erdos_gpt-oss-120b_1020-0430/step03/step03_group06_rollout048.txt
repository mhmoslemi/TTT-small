def apply_delta(i, delta, h, rev, corr):
    n = h.shape[0]
    # compute indices arrays
    j = np.arange(n)
    idx1 = n-1 + i - j  # shift = i - j

def simulated_annealing(h0, max_iters, time_budget, rng):
  n = len(h0)
  h = h0.copy()
  rev = 1.0 - h
  # compute initial corr
  corr = np.correlate(h, rev, mode='full')
  cur_c = np.max(corr) * (2.0 / n)
  best_h = h.copy()
  best_c = cur_c
  idx1_cache, idx2_cache = precompute_idx(n)

  # schedule values
  total_iters = max_iters
  start_time = time.time()
  end_time = start_time + time_budget

  for it in range(total_iters):
      if time.time() > end_time:
          break
      # select i,j
      i = rng.integers(0, n)
      j = rng.integers(0, n-1)
      if j >= i:
          j += 1  # ensure j != i
      # compute max transfer both directions
      max_pos = min(h[i], 1.0 - h[j])  # move from i to j
      max_neg = min(h[j], 1.0 - h[i])  # move from j to i (negative eps)
      if max_pos + max_neg == 0:
          continue
      # pick direction
      if max_pos > 0 and (max_neg == 0 or rng.random() < 0.5):
          # positive eps
          eps = rng.random() * max_pos
          delta_i = -eps
          delta_j = eps
      else:
          eps = rng.random() * max_neg
          delta_i = eps
          delta_j = -eps
      # apply changes
      # We'll keep copies of current max_corr before? We'll compute after.
      # apply delta to i
      apply_delta(i, delta_i, h, rev, corr, idx1_cache, idx2_cache)
      # apply delta to j
      apply_delta(j, delta_j, h, rev, corr, idx1_cache, idx2_cache)

      new_c = np.max(corr) * (2.0 / n)

      # temperature
      T = 0.01 * ((1e-5/0.01) ** (it / total_iters))  # exponential from 0.01 to 1e-5
      # accept condition
      if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / T):
          cur_c = new_c
          # optionally update best if improved
          if new_c < best_c:
              best_c = new_c
              best_h = h.copy()
      else:
          # reject: revert changes
          # revert delta on j then i (reverse)
          # we can apply delta of opposite sign.
          apply_delta(j, -delta_j, h, rev, corr, idx1_cache, idx2_cache)
          apply_delta(i, -delta_i, h, rev, corr, idx1_cache, idx2_cache)
          # cur_c unchanged

  return best_h, best_c

def enforce_sum(h, target, rng):

    total = h.sum()
    diff = target - total
    max_iter = 10000
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        # choose index where we can adjust in desired direction
        if diff > 0:
            # need increase

import time

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def compute_c5(h):
    """Exact discrete C5 for a step function on [0,2] with n = len(h)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng):
    """Adjust h so that sum(h)=target while keeping 0<=h<=1."""
    n = h.size
    diff = target - h.sum()
    max_iter = 10000
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to increase some entries
            idxs = np.where(h < 1.0)[0]
            if idxs.size == 0:
                break
            i = rng.choice(idxs)
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            # need to decrease some entries
            idxs = np.where(h > 0.0)[0]
            if idxs.size == 0:
                break
            i = rng.choice(idxs)
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
    # tiny final correction if anything remains
    if abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            h[i] = min(1.0, h[i] + diff)
        else:
            h[i] = max(0.0, h[i] + diff)
    np.clip(h, 0.0, 1.0, out=h)
    return h

def _precompute_idx(n):
    """Cache of index arrays needed for the O(n) update."""
    j = np.arange(n)
    idx1 = [(n - 1) + i - j for i in range(n)]
    idx2 = [(n - 1) + j - i for i in range(n)]
    return idx1, idx2


def _apply_delta(i, delta, h, rev, corr, idx1_cache, idx2_cache):
    """Apply a change delta to h[i] (and rev[i]) and update corr in O(n)."""
    # use current rev and h (before modification)
    np.add.at(corr, idx1_cache[i], delta * rev)
    np.add.at(corr, idx2_cache[i], -delta * h)
    h[i] += delta
    rev[i] = 1.0 - h[i]

# ----------------------------------------------------------------------
def _simulated_annealing(h0, max_time, rng, max_iters=500_000):
    n = h0.size
    idx1_cache, idx2_cache = _precompute_idx(n)
    h = h0.copy()
    rev = 1.0 - h
    corr = np.correlate(h, rev, mode="full")
    cur_c = np.max(corr) * (2.0 / n)
    best_h = h.copy()
    best_c = cur_c

    # temperature schedule
    T0, T_end = 0.02, 1e-5
    start = time.time()
    it = 0
    while it < max_iters and (time.time() - start) < max_time:
        # pick two distinct indices
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        max_pos = min(h[i], 1.0 - h[j])   # move i -> j
        max_neg = min(h[j], 1.0 - h[i])   # move j -> i
        if max_pos + max_neg == 0.0:
            it += 1
            continue

        # decide direction and magnitude
        if max_pos > 0.0 and (max_neg == 0.0 or rng.random() < 0.5):
            eps = rng.random() * max_pos
            di, dj = -eps, eps
        else:
            eps = rng.random() * max_neg
            di, dj = eps, -eps

        # apply tentative move
        _apply_delta(i, di, h, rev, corr, idx1_cache, idx2_cache)
        _apply_delta(j, dj, h, rev, corr, idx1_cache, idx2_cache)

        new_c = np.max(corr) * (2.0 / n)

        # temperature for this iteration (linear in log‑scale)
        T = T0 * ((T_end / T0) ** (it / max_iters))
        accept = False
        if new_c < cur_c:
            accept = True
        else:
            if rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-12)):
                accept = True

        if accept:
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            _apply_delta(j, -dj, h, rev, corr, idx1_cache, idx2_cache)
            _apply_delta(i, -di, h, rev, corr, idx1_cache, idx2_cache)

        it += 1
    return best_h, best_c
