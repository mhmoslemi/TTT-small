def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta

        return np.sum(x) - CURRENT_TARGET
    def soft_obj(x):
        corr = np.correlate(x, 1 - x, mode="full")

    res = minimize(soft_obj, h0, method="SLSQP", bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})

            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])

import time
import numpy as np
from scipy.optimize import minimize

# Core utilities

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    it = 0
    while abs(diff) > 1e-12 and it < 5000:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
        it += 1
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random feasible h (0≤h≤1, Σh=n/2)."""
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    it = 0
    while abs(diff) > 1e-12 and it < 5000:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
        it += 1
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k = Σ h_i(1‑h_{i+k})."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h_start: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 3000, step_scale: float = 0.5):
    """Greedy hill‑climb using the worst‑shift gradient."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

CURRENT_BETA = None


def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 400) -> np.ndarray:
    """Smooth‑max SLSQP optimisation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta

    def sum_eq(x):
        return np.sum(x) - CURRENT_TARGET

    def soft_obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = CURRENT_BETA * corr
        m = np.max(bc)
        # log‑sum‑exp for smooth maximum
        return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq}
    res = minimize(
        soft_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    it = 0
    while abs(diff) > 1e-12 and it < 5000:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
        it += 1
    return x

        h0 = enforce_sum(h0, rng)
        h0, c0 = guided_hill_climb(h0, rng, max_iters=2000, step_scale=0.5)
        best_h, best_c = h0, c0
    else:
        h0 = random_feasible_vector(n, rng)
        h0, c0 = guided_hill_climb(h0, rng, max_iters=1500, step_scale=0.5)
        best_h, best_c = h0, c0

    # ------------------------------------------------------------------
    # Random‑restart exploration with local refinement
    # ------------------------------------------------------------------
    while time.time() - start < budget_s * 0.90:
        h = random_feasible_vector(n, rng)

        # Occasionally blend with the current best to explore neighbourhoods
        if rng.random() < 0.3:
            alpha = rng.random()
            h = alpha * h + (1.0 - alpha) * best_h
            h = np.clip(h, 0.0, 1.0)
            h = enforce_sum(h, rng)

        # Local greedy refinement
        h, c = guided_hill_climb(h, rng, max_iters=1500, step_scale=0.5)

        if c < best_c:
            best_h, best_c = h.copy(), c

            # Light SLSQP polishing (few β values)
            for beta in (5.0, 20.0, 80.0):
                if time.time() - start > budget_s * 0.85:
                    break
                h_s = slsqp_optimize(best_h, beta, rng, max_iter=300)
                c_s = compute_c5(h_s)
                if c_s < best_c:
                    best_h, best_c = h_s.copy(), c_s

    # ------------------------------------------------------------------
    # Final intensive refinement on the incumbent
    # ------------------------------------------------------------------
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0:
        max_iters = int(min(5000, remaining * 10))
        best_h, best_c = guided_hill_climb(best_h, rng,
                                           max_iters=max_iters,
                                           step_scale=0.3)

    # ------------------------------------------------------------------
    # Ensure feasibility and return
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
