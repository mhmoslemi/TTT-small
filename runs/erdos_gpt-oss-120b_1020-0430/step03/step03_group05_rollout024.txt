    start = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # candidate n's
    candidate_ns = [300]  # start with 300

    # maybe also include 240, 360 if time.
    if time_remaining > 20:
        candidate_ns = [240, 300, 360, 420, 480, 600]

    for n in candidate_ns:
        if time.time() - start >= budget_s * 0.97:
            break
        target = n/2.0

        # generate starting vectors
        # If initial exists, upsample
        init_vec = None
        if 'initial_h_values' in globals():
            init_vec = upsample_initial(n)
            init_vec = project_to_feasible(init_vec, target)

        # number of restarts governed by time
        restarts = 0
        while time.time() - start < budget_s*0.95:
            # generate new start
            if restarts == 0 and init_vec is not None:
                h0 = init_vec.copy()
            else:
                h0 = random_feasible_vector(n, rng)
            # apply slsqp series
            h_cur = h0
            for beta in (10.0, 100.0, 500.0, 2000.0, 8000.0):
                if time.time() - start >= budget_s*0.95:
                    break
                h_cur = slsqp_optimize(h_cur, beta, rng, max_iter=300)
            c = compute_c5(h_cur)
            if c < best_c:
                best_c, best_h, best_n = c, h_cur.copy(), n
            # optional hill climb
            if time.time() - start < budget_s*0.92:
                h_hc, c_hc = hill_climb(h_cur, rng, max_iters=500, start_step=0.2)
                if c_hc < best_c:
                    best_c, best_h, best_n = c_hc, h_hc.copy(), n
            restarts += 1
            # Break after reasonable number of restarts to avoid infinite loop
            if restarts > 30:
                break
    # fallback to initial if best not found
    if best_h is None:
        if 'initial_h_values' in globals():
            h_init = np.array(initial_h_values, dtype=float)
            n0 = h_init.size
            best_h = project_to_feasible(h_init, n0/2)
            best_n = n0
            best_c = compute_c5(best_h)
        else:
            n0 = 60
            best_h = np.full(n0, 0.5)
            best_n = n0
            best_c = compute_c5(best_h)

    # Ensure feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n/2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        # adjust greedily
        for _ in range(5000):
            if abs(diff) <= 1e-12:
                break
            i = rng.integers(0, best_n)
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)

import numpy as np
from scipy.optimize import minimize

# Global placeholders for the smooth max surrogate used by SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for step function h on [0,2] (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_to_feasible(v, target):
    """Project vector v onto {0≤h≤1, Σh = target} via a simple bisection shift."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Fast generation of a random feasible vector (0≤h≤1, Σh=n/2)."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_to_feasible(v, n / 2.0)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.3):
    """Greedy hill‑climb that only accepts improving moves."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


def slsqp_optimize(h0, beta, rng, max_iter=500):
    """Run SLSQP optimisation with the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # exact sum correction by greedy redistribution
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def upsample_initial(to_n):
    """Linear interpolation of global initial_h_values to length to_n."""
    init = np.array(initial_h_values, dtype=float)
    m = init.size
    xs_old = np.linspace(0.0, 2.0, m, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, to_n, endpoint=False)
    return np.interp(xs_new, xs_old, init)


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    best_h = None
    best_c = np.inf
    best_n = None

    # Primary discretisation (the record was obtained at n=300)
    n = 300
    target = n / 2.0

    # Prepare an up‑sampled initial if available
    init_vec = None
    if "initial_h_values" in globals():
        if len(initial_h_values) != n:
            init_vec = upsample_initial(n)
        else:
            init_vec = np.array(initial_h_values, dtype=float)
        init_vec = project_to_feasible(init_vec, target)

    # Main optimisation loop: multiple random restarts + SLSQP + hill‑climb
    betas = (10.0, 50.0, 200.0, 800.0, 2000.0, 8000.0)
    max_restarts = 30
    restart = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.96:
            break
        if restart >= max_restarts:
            break

        # Starting vector
        if restart == 0 and init_vec is not None:
            h0 = init_vec.copy()
        else:
            h0 = random_feasible_vector(n, rng)

        # SLSQP refinement with increasing β
        h_cur = h0
        for beta in betas:
            if time.time() - start_time >= budget_s * 0.96:
                break
            h_cur = slsqp_optimize(h_cur, beta, rng, max_iter=300)

        c_cur = compute_c5(h_cur)
        if c_cur < best_c:
            best_c, best_h, best_n = c_cur, h_cur.copy(), n

        # Optional hill‑climb for fine‑tuning
        if time.time() - start_time < budget_s * 0.94:
            h_hc, c_hc = hill_climb(h_cur, rng, max_iters=3000, start_step=0.25)
            if c_hc < best_c:
                best_c, best_h, best_n = c_hc, h_hc.copy(), n

        restart += 1

    # If nothing was obtained, fall back to a simple feasible vector
    if best_h is None:
        n = 60
        best_h = np.full(n, 0.5)
        best_n = n
        best_c = compute_c5(best_h)

    # Final feasibility enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, best_n)
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
