def improve_h(h, rng, total_iters, step_initial):
    h = h.copy()
    step = step_initial
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(total_iters):
        A = np.correlate(h, h, mode='full')[len(h)-1:]  # length n
        k_w = np.argmin(A)  # worst shift
        # compute gradient for worst shift
        grad = np.zeros_like(h)
        if k_w > 0:
            grad[:len(h)-k_w] += h[k_w:]
            grad[k_w:] += h[:len(h)-k_w]
        # choose max and min indices
        i_max = np.argmax(grad)
        i_min = np.argmin(grad)
        # compute max transfer possible
        max_transfer = min(step, 1 - h[i_max], h[i_min])
        if max_transfer <= 1e-12:
            # reduce step maybe
            step *= 0.5
            continue
        # transfer
        h[i_max] += max_transfer
        h[i_min] -= max_transfer
        # enforce sum? sum remains same due to transfer (no rounding errors)
        # compute C5
        c = compute_c5(h)
        if c < best_c - 1e-9:
            best_c = c
            best_h = h.copy()
        # adapt step: maybe decrease gradually
        if (it+1) % 500 == 0:
            step *= 0.9
        # early termination if step tiny
        if step < 1e-6:
            break
    return best_h, best_c

import time, numpy as np
from scipy.optimize import minimize

# ---------- utilities ----------
def enforce_sum(h, target, rng, max_iter=5000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

# ---------- smooth surrogate for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET
def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # tiny sum correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

# ---------- targeted transfer local search ----------
def transfer_search(h0, rng, max_iters=2000, step_init=0.15,
                    decay_interval=300):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step_init
    for it in range(max_iters):
        # autocorrelation A_k = Σ h_i h_{i+k}
        A = np.correlate(h, h, mode="full")[n-1:]
        k = int(np.argmin(A))
        # gradient of A_k w.r.t. h (for this k)
        grad = np.zeros_like(h)
        if k > 0:
            grad[:n-k] += h[k:]
            grad[k:]   += h[:n-k]
        i_max = int(np.argmax(grad))
        i_min = int(np.argmin(grad))
        transfer = min(step, 1.0 - h[i_max], h[i_min])
        if transfer <= 0:
            step *= 0.5
            continue
        h[i_max] += transfer
        h[i_min] -= transfer
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
            step = min(step_init, step * 1.1)
        if (it + 1) % decay_interval == 0:
            step *= 0.9
            if step < 1e-6:
                break
        # occasional random jitter
        if rng.random() < 0.02:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                tr = min(step, h[i], 1.0 - h[j])
                if tr > 0:
                    h[i] -= tr
                    h[j] += tr
    return best_h, best_c

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # ---- initialise ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 240
        h0 = random_feasible_vector(n, rng)
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # some quick diverse seeds
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    candidates = [best_h]
    for f in (1, 2, 3, 4):
        s = np.clip(0.5 + 0.5 * np.sin(np.pi * f * xs), 0, 1)
        candidates.append(enforce_sum(s, n / 2.0, rng))
        c = np.clip(0.5 + 0.5 * np.cos(np.pi * f * xs), 0, 1)
        candidates.append(enforce_sum(c, n / 2.0, rng))
    mask = np.zeros(n)
    mask[:n//2] = 1.0
    candidates.append(enforce_sum(mask, n / 2.0, rng))
    mask = np.arange(n) % 2
    candidates.append(enforce_sum(mask, n / 2.0, rng))
    for _ in range(5):
        candidates.append(random_feasible_vector(n, rng))

    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # ---- main optimisation loop ----
    beta_list = [10, 50, 200, 500, 1000]
    while time.time() - start < budget_s * 0.85:
        # pick a start point
        if rng.random() < 0.6:
            start_vec = perturb_h(best_h, rng, scale=0.3)
        else:
            start_vec = random_feasible_vector(n, rng)
        # local targeted transfer search
        h_opt, c_opt = transfer_search(start_vec, rng,
                                       max_iters=1800,
                                       step_init=0.2,
                                       decay_interval=250)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()
        # occasional SLSQP refinement
        if rng.random() < 0.35:
            for beta in beta_list:
                if time.time() - start > budget_s * 0.85:
                    break
                try:
                    h_ref = slsqp_optimize(best_h, beta, max_iter=300)
                except Exception:
                    continue
                c_ref = compute_c5(h_ref)
                if c_ref < best_c:
                    best_c, best_h = c_ref, h_ref.copy()

    # final polishing with highest beta
    try:
        h_final = slsqp_optimize(best_h, beta=1500, max_iter=400)
        cf = compute_c5(h_final)
        if cf < best_c:
            best_c, best_h = cf, h_final.copy()
    except Exception:
        pass

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
