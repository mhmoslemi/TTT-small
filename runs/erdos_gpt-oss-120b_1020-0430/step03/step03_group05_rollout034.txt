def hill_climb(h0, rng, max_iters, initial_step=0.4):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = initial_step
    for it in range(max_iters):
        i, j = rng.integers(0, h.size, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2 * max_eps * step
        # ensure eps within [-max_eps, max_eps]
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        # else maybe accept if random? Not needed
        # reduce step after some iterations
        if (it+1) % (max_iters // 10 + 1) == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def pattern_search(n: int, max_time: float):
    """Exhaustive search over three‑block patterns (1, c, 0) with all offsets."""
    target = n / 2.0
    best_c = np.inf
    best_vec = None
    start = time.time()
    h = np.empty(n, dtype=np.float64)

    # L1 = number of points with value 1
    for L1 in range(int(target)):
        if time.time() - start > max_time:
            break
        # minimal L2 to keep c ≤ 1
        L2_min = int(np.ceil(target - L1))
        if L2_min < 1:
            L2_min = 1
        L2_max = n - L1
        for L2 in range(L2_min, L2_max + 1):
            if time.time() - start > max_time:
                break
            c = (target - L1) / L2
            if c < 0.0 or c > 1.0:
                continue
            Z_total = n - L1 - L2
            # try every possible placement of the leading zeros
            for offset in range(Z_total + 1):
                if time.time() - start > max_time:
                    break
                h.fill(0.0)
                if L1:
                    h[offset : offset + L1] = 1.0
                if L2:
                    h[offset + L1 : offset + L1 + L2] = c
                c_val = compute_c5(h)
                if c_val < best_c:
                    best_c = c_val
                    best_vec = h.copy()
    return best_vec, best_c


def hill_climb(h0: np.ndarray, rng: np.random.Generator, max_iters: int = 2000, initial_step: float = 0.5):
    """Simple deterministic hill‑climb using mass transfer between coordinates."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = initial_step
    n = h.size
    decay_interval = max(1, max_iters // 10)

    for it in range(max_iters):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * step
        eps = max(-max_eps, min(eps, max_eps))
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c
