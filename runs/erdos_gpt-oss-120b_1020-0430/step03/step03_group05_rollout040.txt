def compute_c5(h):
    n = h.shape[0]
    a = h
    b = 1 - h
    # full correlation via convolution
    # pad to length 2n-1
    # use np.fft with size = 2**ceil_pow2(2*n)
    size = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b[::-1], size)  # flip b for correlation
    conv = np.fft.irfft(fa * fb, size)[:2*n-1]
    return np.max(conv) * (2.0 / n)

def compute_c5(h):
    n = h.shape[0]
    # convolution of h and reversed (1-h)
    corr = np.convolve(h, (1-h)[::-1], mode='full')
    # max
    return np.max(corr) * (2.0 / n)

def project_to_feasible(v, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    # ensure within bounds
    if eps > max_eps:
        eps = max_eps
    if eps < -max_eps:
        eps = -max_eps
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb(h_start, rng, max_iters=5000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c

def slsqp_optimize(h0, beta, rng, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = opt.minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # correct sum precisely
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def hill_climb_binary(h_start, rng, max_iters=10000):

    best_h = h.copy()

    n = len(h)
    for it in range(max_iters):
        # pick random positions: one with 1 and one with 0

        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i], h[j] = h[j], h[i]
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        else:
            # revert
            h[i], h[j] = h[j], h[i]
    return best_h, best_c

import time
import numpy as np
import scipy.optimize as opt

# globals for soft‑max surrogate
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) via log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    n = h.shape[0]
    # correlation = convolution with reversed (1‑h)
    corr = np.convolve(h, (1 - h)[::-1], mode="full")
    return np.max(corr) * (2.0 / n)


def project_to_feasible(v, target):
    """Projection onto {0≤h≤1, Σh = target} using a bisection on the shift."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Fast random feasible vector (0≤h≤1, Σh = n/2)."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_to_feasible(v, n / 2.0)


def upsample_initial(to_n):
    """Linear interpolation of the provided initial_h_values to a new length."""
    init = np.array(initial_h_values, dtype=float)
    m = init.size
    xs_old = np.linspace(0.0, 2.0, m, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, to_n, endpoint=False)
    return np.interp(xs_new, xs_old, init)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=5000, start_step=0.5):
    """Greedy hill‑climb using small transfers; only accepts improvements."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


def slsqp_optimize(h0, beta, rng, max_iter=800):
    """SLSQP optimisation of the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = opt.minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # exact sum correction (project onto the hyperplane)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def hill_climb_binary(h_start, rng, max_iters=8000):
    """Hill‑climb on binary vectors (exact sum, values in {0,1})."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    for _ in range(max_iters):
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap the pair
        h[i], h[j] = h[j], h[i]
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        else:
            # revert
            h[i], h[j] = h[j], h[i]
    return best_h, best_c


def optimize_n(n, rng, time_budget):
    """Optimise a given discretisation size n within the allotted time."""
    start = time.time()
    target = n / 2.0
    best_h = None
    best_c = np.inf

    # seed with (upsampled) parent if available
    if "initial_h_values" in globals():
        h_up = upsample_initial(n)
        h_up = np.clip(h_up, 0.0, 1.0)
        h_up = project_to_feasible(h_up, target)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_c, best_h = c_up, h_up.copy()

    # main random‑restart loop
    while True:
        elapsed = time.time() - start
        remaining = time_budget - elapsed
        if remaining < 5.0:
            break

        # ----- binary hill‑climb -----
        half = n // 2
        h_bin = np.zeros(n)
        h_bin[:half] = 1.0
        rng.shuffle(h_bin)
        max_iters_bin = int(min(8000, remaining * 1500))
        max_iters_bin = max(2000, max_iters_bin)
        h_bin_opt, c_bin_opt = hill_climb_binary(h_bin, rng, max_iters_bin)
        if c_bin_opt < best_c:
            best_c, best_h = c_bin_opt, h_bin_opt.copy()

        # ----- continuous SLSQP refinement -----
        h_cur = best_h.copy()
        for beta in (1.0, 10.0, 100.0, 500.0, 2000.0, 8000.0):
            if time.time() - start > time_budget * 0.92:
                break
            h_cur = slsqp_optimize(h_cur, beta, rng, max_iter=800)
            c_cur = compute_c5(h_cur)
            if c_cur < best_c:
                best_c, best_h = c_cur, h_cur.copy()

        # ----- final greedy hill‑climb -----
        if time.time() - start < time_budget * 0.95:
            rem = time_budget - (time.time() - start)
            max_iters_cont = int(min(12000, rem * 2000))
            max_iters_cont = max(3000, max_iters_cont)
            h_cur, c_cur = hill_climb(h_cur, rng, max_iters=max_iters_cont, start_step=0.5)
            if c_cur < best_c:
                best_c, best_h = c_cur, h_cur.copy()

    return best_h, best_c
