def targeted_move(h, rng, scale=0.1):
    n = len(h)
    corr = compute_correlation_array(h)  # full.
    idx_max = np.argmax(corr)
    shift = idx_max - (n - 1)
    # compute contributions
    if shift >= 0:
        contributions = np.zeros_like(h)
        contributions[:n-shift] = h[:n-shift] * (1 - h[shift:])
    else:
        s = -shift
        contributions = np.zeros_like(h)
        contributions[s:] = h[s:] * (1 - h[:n-s])
    # indices
    i = np.argmax(contributions)
    # find candidate j:
    # pick random j not equal i with allowed capacity.
    attempts = 0
    while attempts < 10:
        j = rng.integers(0, n)
        if j != i:
            # candidate must have capacity
            if h[j] < 1.0 and h[i] > 0.0:
                break
        attempts += 1
    # compute max epsilon that respects bounds
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = max_eps * rng.random() * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def enforce_sum(h, target, rng, max_iter=2000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        # choose random index
        i = rng.integers(0, n)
        # adjust as much as possible
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # final correction if needed
    if abs(diff) > 1e-12:
        # distribute uniformly
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h
```

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Global variables for SLSQP surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth surrogate for max_k Σ_i x_i (1‑x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp trick
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_corr_fft(h):
    """Linear correlation of h with (1‑h) using zero‑padded FFT."""
    n = h.shape[0]
    b = 1.0 - h
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    A = np.fft.rfft(h, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * np.conj(B), n=size)[: 2 * n - 1]
    return np.real(conv)


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    n = h.shape[0]
    corr = compute_corr_fft(h)
    return np.max(corr) * (2.0 / n)


def enforce_sum(h, target, rng, max_iter=2000):
    """Clip to [0,1] and adjust to achieve Σh = target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        # uniform correction, may need a second pass
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1] with Σ = n/2."""
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def targeted_move(h, rng, scale=0.1):
    """Move mass from a point contributing most to the worst shift."""
    n = h.size
    corr = compute_corr_fft(h)
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)  # negative shifts allowed
    contrib = np.zeros_like(h)
    if shift >= 0:
        contrib[: n - shift] = h[: n - shift] * (1.0 - h[shift:])
    else:
        s = -shift
        contrib[s:] = h[s:] * (1.0 - h[: n - s])
    i = int(np.argmax(contrib))
    # pick a different index with capacity
    attempts = 0
    while attempts < 12:
        j = rng.integers(0, n)
        if j != i and h[j] < 1.0 and h[i] > 0.0:
            break
        attempts += 1
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * rng.random() * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_targeted(h0, rng, max_iters=2500, init_scale=0.5):
    """Hybrid hill‑climb using random and targeted moves."""
    h = h0.copy()
    best_c = compute_c5(h)
    scale = init_scale
    for it in range(max_iters):
        # 70 % targeted, 30 % random
        if rng.random() < 0.7:
            new_h = targeted_move(h, rng, scale=scale)
        else:
            new_h = perturb_h(h, rng, scale=scale)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            h = new_h
        # decay scale periodically
        if (it + 1) % 200 == 0:
            scale = max(scale * 0.7, 1e-4)
    return h, best_c


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum (tiny drift)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        n = x.size
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x
