def greedy_targeted_move(h, k, grad, max_transfer=0.9):
    pos_idx = np.where((grad > 1e-9) & (h > 1e-12))[0]
    neg_idx = np.where((grad < -1e-9) & (h < 1 - 1e-12))[0]
    if pos_idx.size ==0 or neg_idx.size==0:
        return h, False
    i = pos_idx[np.argmax(grad[pos_idx])]  # largest positive gradient
    j = neg_idx[np.argmin(grad[neg_idx])]  # most negative gradient
    max_eps = min(h[i], 1 - h[j])
    if max_eps <= 0:
        return h, False
    eps = max_transfer * max_eps
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h, True

def compute_gradient(h,k):
    n = len(h)
    grad = np.ones(n)
    if k >= 0:
        grad[:n - k] -= h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[:n - kk] -= h[kk:]
        grad[kk:] -= h[:n - kk]
    return grad

import time
import numpy as np
from scipy.optimize import minimize

# ----- utilities -----
def enforce_sum(h, target, rng):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i]) * rng.random()
            h[i] += add
            diff -= add
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i]) * rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    h = rng.random(n)
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)

def perturb_random(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def compute_c5(h):
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)

def compute_gradient(h, k):
    n = h.size
    grad = np.ones(n)
    if k >= 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    else:
        kk = -k
        grad[: n - kk] -= h[kk:]
        grad[kk:] -= h[: n - kk]
    return grad

# ----- SLSQP surrogate -----
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
