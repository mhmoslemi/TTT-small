def propose_move(h, rng, step_scale, swap_prob=0.2):
    n = h.size
    i = rng.integers(0,n)
    j = rng.integers(0,n-1)
    if j >= i: j+=1
    if rng.random() < swap_prob:
        new_h = h.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]
        return new_h
    else:
        # transfer epsilon
        max_pos = min(h[i], 1.0 - h[j])
        max_neg = min(h[j], 1.0 - h[i])
        if max_pos + max_neg <= 0:
            return h.copy()
        eps = (rng.random() * (max_pos + max_neg)) - max_neg
        eps *= step_scale
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        return new_h

def compute_c5_fft(h):
    n = h.size
    # compute correlation using FFT of length L = 2*n
    L = 2*n
    # optional: we can use np.fft.rfft for real efficiency but okay.
    H = np.fft.fft(h, L)
    G = np.fft.fft((1 - h)[::-1], L)
    corr = np.fft.ifft(H * G).real
    # take first 2*n-1 entries (full correlation)
    c = corr[:2*n-1]
    return np.max(c) * (2.0 / n)

def compute_c5_fft(h):
    n = h.size
    g_rev = (1 - h)[::-1]
    L = 2 * n
    H = np.fft.fft(h, L)
    G = np.fft.fft(g_rev, L)

    return np.max(conv) * (2.0 / n)

def hill_climb_sa(h_start, rng, remaining_time, max_iters=20000, start_temp=0.005, end_temp=5e-6):
    best_h = h_start.copy()
    best_c = compute_c5_fft(best_h)
    curr_h = h_start.copy()
    curr_c = best_c
    n = h_start.size
    step_scale = 0.5
    start = time.time()
    it = 0
    while it < max_iters:
        # time check
        if time.time() - start > remaining_time:
            break
        # linear temperature schedule based on progress relative to max_iters
        t = it / max_iters
        temp = start_temp * (1 - t) + end_temp * t
        # propose move
        new_h = propose_move(curr_h, rng, step_scale)
        # ensure constraints if needed (should be kept)
        new_c = compute_c5_fft(new_h)
        delta = new_c - curr_c
        if delta <= 0 or rng.random() < math.exp(-delta / temp):
            # accept
            curr_h = new_h
            curr_c = new_c
            if new_c < best_c:

def compute_c5_fft(h):
    n = h.size
    g_rev = (1.0 - h)[::-1]
    L = 2 * n
    H = np.fft.fft(h, L)
    G = np.fft.fft(g_rev, L)
    conv = np.fft.ifft(H * G).real[:2*n-1]
    return np.max(conv) * (2.0 / n)

def random_binary_vector(n, rng):
    arr = np.zeros(n, dtype=float)
    idx = rng.choice(n, n//2, replace=False)
    arr[idx] = 1.0
    return arr

def propose_move(h, rng, step_scale, swap_prob=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n-1)
    if j >= i:
        j += 1
    if rng.random() < swap_prob:
        new_h = h.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]
        return new_h
    # transfer epsilon preserving sum
    max_pos = min(h[i], 1.0 - h[j])
    max_neg = min(h[j], 1.0 - h[i])
    if max_pos + max_neg <= 0:
        return h.copy()
    eps = (rng.random() * (max_pos + max_neg)) - max_neg
    eps *= step_scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    # clip may not be necessary because eps limited, but for safety:
    new_h[i] = min(max(new_h[i], 0.0), 1.0)
    new_h[j] = min(max(new_h[j], 0.0), 1.0)
    return new_h

def hill_climb_sa(h_start, rng, remaining_time, max_iters=200000, start_temp=0.005, end_temp=5e-6):
    best_h = h_start.copy()
    best_c = compute_c5_fft(best_h)
    curr_h = h_start.copy()
    curr_c = best_c
    step_scale = 0.5
    start = time.time()
    for it in range(max_iters):
        if time.time() - start >= remaining_time:
            break
        t = it / max_iters
        temp = start_temp * (1 - t) + end_temp * t
        new_h = propose_move(curr_h, rng, step_scale)
        new_c = compute_c5_fft(new_h)
        delta = new_c - curr_c
        if delta <= 0 or rng.random() < math.exp(-delta / max(temp, 1e-12)):
            curr_h = new_h
            curr_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()
        # decay step scale
        if it % (max_iters // 10 + 1) == 0:
            step_scale = max(step_scale * 0.9, 1e-4)
    return best_h, best_c

def _softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=800):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(_softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})

    H = np.fft.fft(h, L)
    G = np.fft.fft(g_rev, L)
    conv = np.fft.ifft(H * G).real[: 2 * n - 1]
    return np.max(conv) * (2.0 / n)


def random_binary_vector(n, rng):
    """Binary vector with exactly n/2 ones."""
    a = np.zeros(n, dtype=float)
    a[rng.choice(n, n // 2, replace=False)] = 1.0
    return a


def propose_move(h, rng, step_scale, swap_prob=0.2):
    """Generate a feasible neighbour of h."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if j >= i:
        j += 1
    # occasional full swap
    if rng.random() < swap_prob:
        new_h = h.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]
        return new_h
    # transfer a small amount ε, preserving sum
    max_pos = min(h[i], 1.0 - h[j])
    max_neg = min(h[j], 1.0 - h[i])
    if max_pos + max_neg <= 0:
        return h.copy()
    eps = (rng.random() * (max_pos + max_neg)) - max_neg
    eps *= step_scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    # safety clipping
    new_h[i] = min(max(new_h[i], 0.0), 1.0)
    new_h[j] = min(max(new_h[j], 0.0), 1.0)
    return new_h


def hill_climb_sa(h_start, rng, remaining_time, max_iters=200000,
                  start_temp=0.005, end_temp=5e-6):
    """Simulated‑annealing hill climb, returns best vector seen."""
    best_h = h_start.copy()
    best_c = compute_c5_fft(best_h)
    cur_h = h_start.copy()
    cur_c = best_c
    step = 0.5
    start = time.time()
    for it in range(max_iters):
        if time.time() - start >= remaining_time:
            break
        t = it / max_iters
        temp = start_temp * (1 - t) + end_temp * t
        new_h = propose_move(cur_h, rng, step)
        new_c = compute_c5_fft(new_h)
        delta = new_c - cur_c
        if delta <= 0 or rng.random() < math.exp(-delta / max(temp, 1e-12)):
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
        if it % (max_iters // 10 + 1) == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


# ----------------------------------------------------------------------
# Smooth surrogate for SLSQP (same as the reference code)
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=800):
    """Run SLSQP on the log‑sum‑exp surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            if abs(add) < 1e-12:
                continue
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

# Main entry point
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ---- initial vector ------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n0 / 2.0, rng)
    else:
        n0 = 180
        h0 = random_binary_vector(n0, rng)

    best_h = h0
    best_c = compute_c5_fft(best_h)

    # ---- phase 0: SA on the base grid ---------------------------------
    phase0_budget = budget_s * 0.35
    while time.time() - start_time < phase0_budget:
        # choose a start point
        if rng.random() < 0.5:
            h_start = best_h.copy()
            for _ in range(2):
                i, j = rng.choice(h_start.size, 2, replace=False)
                h_start[i], h_start[j] = h_start[j], h_start[i]
        else:
            h_start = random_binary_vector(n0, rng)

        rem = max(0.0, phase0_budget - (time.time() - start_time))
        if rem <= 0:
            break
        h_loc, c_loc = hill_climb_sa(
            h_start, rng, remaining_time=rem, max_iters=150000
        )
        if c_loc < best_c:
            best_c, best_h = c_loc, h_loc

    # ---- iterative up‑sampling & refinement ----------------------------
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break

        # up‑sample by factor 2
        best_h = np.repeat(best_h, 2)
        best_h = enforce_sum(best_h, best_h.size / 2.0, rng)

        remaining = max(0.0, budget_s - (time.time() - start_time))
        if remaining <= 0:
            break

        # 60 % SA, 30 % SLSQP, rest for final tidy‑up
        sa_budget = remaining * 0.6
        slsqp_budget = remaining * 0.3

        # SA refinement
        h_loc, c_loc = hill_climb_sa(
            best_h, rng, remaining_time=sa_budget, max_iters=300000
        )
        if c_loc < best_c:
            best_c, best_h = c_loc, h_loc

        # SLSQP refinement (if time permits)
        if time.time() - start_time < budget_s * 0.99:
            for beta in [10, 40, 120, 400, 800]:
                if time.time() - start_time > budget_s * 0.99:
                    break
                try:
                    h_sls = slsqp_optimize(best_h, beta, max_iter=800)
                except Exception:
                    continue
                c_sls = compute_c5_fft(h_sls)
                if c_sls < best_c:
                    best_c, best_h = c_sls, h_sls

        # stop after reaching 720 points or when no time left
        if best_h.size >= 720:
            if time.time() - start_time < budget_s * 0.95 and best_h.size < 1440:
                continue
            else:
                break

    # ---- final clean‑up ------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5_fft(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
