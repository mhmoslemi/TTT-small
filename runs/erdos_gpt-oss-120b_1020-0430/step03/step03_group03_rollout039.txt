def random_transfer(h, rng, scale):
    n = len(h)
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h
    delta_max = min(h[i], 1 - h[j])
    if delta_max <= 0:
        return h
    delta = rng.random() * delta_max * scale
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h

def compute_c5(h):
    """Discrete C5 for step function on [0,2] using correlation."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)


def enforce_sum(h, target, rng):
    """Adjust h in‑place so that sum(h)=target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size

            continue
        h[i] += add
        diff = target - h.sum()
    return h

def random_transfer(h, rng, scale=0.3):
    """Transfer a random amount between two positions (preserves sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new


def targeted_transfer(h, rng, shift, scale=0.3, top_frac=0.3, low_frac=0.3):
    """Move mass from positions whose partner (i+shift) is large to ones where it is small."""
    n = h.size
    if shift >= 0:
        idx = np.arange(0, n - shift)
    else:
        idx = np.arange(-shift, n)
    if idx.size == 0:
        return random_transfer(h, rng, scale)

    partner = h[idx + shift]
    th_high = np.quantile(partner, 1 - top_frac)
    th_low = np.quantile(partner, low_frac)
    src = idx[partner >= th_high]
    dst = idx[partner <= th_low]
    if src.size == 0 or dst.size == 0:
        return random_transfer(h, rng, scale)
    i = rng.choice(src)
    j = rng.choice(dst)
    if i == j:
        return random_transfer(h, rng, scale)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return random_transfer(h, rng, scale)
    eps = rng.random() * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

    # ----- first SA phase (70 % of budget) ---------------------------
    sa_budget = budget_s * 0.70
    while time.time() - start < sa_budget:
        # start point: modest perturbation of current best or fresh random
        if rng.random() < 0.5:
            start_h = random_transfer(best_h, rng, scale=0.5)
        else:
            start_h = random_feasible_vector(n, rng)

        # allocate iterations based on remaining time
        remaining = max(0.0, sa_budget - (time.time() - start))
        iters = int(min(50_000, max(2_000, remaining * 2_000)))
        cand_h, cand_c = anneal(
            start_h,
            rng,
            n_iter=iters,
            init_temp=0.5,
            final_temp=1e-4,
            scale=0.25,
        )
        if cand_c < best_c:
            best_h, best_c = cand_h.copy(), cand_c

        # occasional SLSQP refinement with moderate beta
        if time.time() - start < sa_budget:
            try:
                refined = slsqp_optimize(best_h, beta=800, max_iter=250)
                rc = compute_c5(refined)
                if rc < best_c:
                    best_h, best_c = refined, rc
            except Exception:
                pass

    # ----- optional up‑sampling and second SA phase --------------------
    remaining_time = budget_s - (time.time() - start)
    if remaining_time > 30 and n < 500:
        n2 = min(500, n * 2)
        best_h = upscale(best_h, n2, rng)
        n = n2
        best_c = compute_c5(best_h)

        sa2_budget = remaining_time * 0.85
        start2 = time.time()
        while time.time() - start2 < sa2_budget:
            start_h = random_transfer(best_h, rng, scale=0.4)
            remaining2 = max(0.0, sa2_budget - (time.time() - start2))
            iters2 = int(min(50_000, max(2_000, remaining2 * 2_000)))
            cand_h, cand_c = anneal(
                start_h,
                rng,
                n_iter=iters2,
                init_temp=0.5,
                final_temp=1e-4,
                scale=0.2,
            )
            if cand_c < best_c:
                best_h, best_c = cand_h.copy(), cand_c

            if time.time() - start2 < sa2_budget:
                try:
                    refined = slsqp_optimize(best_h, beta=2000, max_iter=200)
                    rc = compute_c5(refined)
                    if rc < best_c:
                        best_h, best_c = refined, rc
                except Exception:
                    pass

    # ----- final polishing (high‑beta SLSQP) --------------------------
    beta = 2000
    while time.time() - start < budget_s - 2.0:
        try:
            best_h = slsqp_optimize(best_h, beta=beta, max_iter=150)
            best_c = compute_c5(best_h)
            beta = int(beta * 1.5)  # gradually strengthen smoothing
        except Exception:
            break

    # ----- final clean‑up ---------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
