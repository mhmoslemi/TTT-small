def compute_c5_fast(h):
    n = len(h)
    g = 1.0 - h
    g_rev = g[::-1]
    m = 1
    while m < 2*n:
        m <<= 1

    H = np.fft.rfft(h, m)

import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # -------------------------------------------------------------
    # 1. Problem size and initial data
    # -------------------------------------------------------------
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        init_real = np.clip(np.array(initial_h_values, dtype=np.float64), 0.0, 1.0)
    else:
        n = 360
        init_real = None

    if n % 2 == 1:               # enforce even length for exact half‑sum
        n += 1
        if init_real is not None:
            init_real = np.pad(init_real, (0, 1), constant_values=0.5)

    target_sum = n // 2           # exact number of ones for binary vectors

    # -------------------------------------------------------------
    # 2. Core helpers
    # -------------------------------------------------------------
    def compute_c5(h):
        """exact discrete C5 for current length n"""
        return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / n)

    def enforce_sum(h, tgt):
        """clip to [0,1] and adjust to meet exact sum"""
        h = np.clip(h, 0.0, 1.0)
        diff = tgt - h.sum()
        size = h.size
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, size)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = tgt - h.sum()
        if abs(diff) > 1e-12:            # final uniform correction
            h += diff / size
            h = np.clip(h, 0.0, 1.0)
        return h

    def random_binary():
        h = np.zeros(n, dtype=np.float64)
        h[:target_sum] = 1.0
        rng.shuffle(h)
        return h

    # -------------------------------------------------------------
    # 3. Targeted hill‑climb on binary vectors
    # -------------------------------------------------------------
    arr = np.arange(n)

    def hill_climb(start_h, steps):
        cur_h = start_h.copy()
        cur_c = compute_c5(cur_h)
        best_h, best_c = cur_h.copy(), cur_c

        for _ in range(steps):
            corr = np.correlate(cur_h, 1 - cur_h, mode="full")
            worst = corr.max()
            cur_c = worst

            # indices where the worst value occurs
            worst_idxs = np.where(corr == worst)[0]
            idx = rng.choice(worst_idxs)
            shift = idx - (n - 1)

            # positions i (a 1) that actually contribute to this worst lag
            cond_i = (cur_h == 1) & (arr + shift < n) & (cur_h[arr + shift] == 0)
            i_cands = np.where(cond_i)[0]

            if i_cands.size:
                i = rng.choice(i_cands)
                # choose a zero j that *does not* contribute at this lag
                cond_j = (cur_h == 0) & ((arr + shift >= n) | (cur_h[arr + shift] == 1))
                j_cands = np.where(cond_j)[0]
                if j_cands.size:
                    j = rng.choice(j_cands)
                else:
                    j = rng.choice(np.where(cur_h == 0)[0])
                new_h = cur_h.copy()
                new_h[i] = 0.0
                new_h[j] = 1.0
            else:
                # fallback – simple random swap
                ones = np.where(cur_h == 1)[0]
                zeros = np.where(cur_h == 0)[0]
                i, j = rng.choice(ones), rng.choice(zeros)
                new_h = cur_h.copy()
                new_h[i] = 0.0
                new_h[j] = 1.0

            new_c = compute_c5(new_h)

            # accept if not worsening (tiny uphill moves are occasionally allowed)
            if new_c <= cur_c or rng.random() < 0.001:
                cur_h, cur_c = new_h, new_c
                if cur_c < best_c:
                    best_h, best_c = cur_h.copy(), cur_c
        return best_h, best_c

    # -------------------------------------------------------------
    # 4. Initialise global best
    # -------------------------------------------------------------
    global_best_h = random_binary()
    global_best_c = compute_c5(global_best_h)

    if init_real is not None:
        # ensure exact sum for the real start
        init_real = enforce_sum(init_real, target_sum)
        c_init = compute_c5(init_real)
        if c_init < global_best_c:
            global_best_h, global_best_c = init_real.copy(), c_init

    # -------------------------------------------------------------
    # 5. Main binary hill‑climb loop (≈60 % of budget)
    # -------------------------------------------------------------
    while time.time() - start_time < 0.60 * budget_s:
        # start from a perturbed best or a fresh random vector
        if rng.random() < 0.7:
            start = global_best_h.copy()
            for _ in range(rng.integers(1, 4)):
                ones = np.where(start == 1)[0]
                zeros = np.where(start == 0)[0]
                i, j = rng.choice(ones), rng.choice(zeros)
                start[i], start[j] = 0.0, 1.0
        else:
            start = random_binary()
        h_opt, c_opt = hill_climb(start, steps=rng.integers(2000, 3000))
        if c_opt < global_best_c:
            global_best_h, global_best_c = h_opt.copy(), c_opt

    # -------------------------------------------------------------
    # 6. SLSQP refinement on the (possibly fractional) best vector
    # -------------------------------------------------------------
    CURRENT_TARGET = None
    CURRENT_BETA = None

    def sum_eq_constraint(x):
        return np.sum(x) - CURRENT_TARGET

    def softmax_obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = CURRENT_BETA * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

    def slsqp_optimize(h0, beta, max_iter=300):
        nonlocal CURRENT_TARGET, CURRENT_BETA
        nloc = h0.size
        CURRENT_TARGET = nloc / 2.0
        CURRENT_BETA = beta
        bounds = [(0.0, 1.0)] * nloc
        cons = {"type": "eq", "fun": sum_eq_constraint}
        res = minimize(
            softmax_obj,
            h0,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
        )
        x = np.clip(res.x, 0.0, 1.0)
        # exact sum repair
        diff = CURRENT_TARGET - x.sum()
        if abs(diff) > 1e-12:
            for _ in range(2000):
                i = rng.integers(0, nloc)
                add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
                x[i] += add
                diff = CURRENT_TARGET - x.sum()
                if abs(diff) < 1e-12:
                    break
        return x

    beta_schedule = [10, 30, 100, 300, 800, 1500]
    for beta in beta_schedule:
        if time.time() - start_time > 0.90 * budget_s:
            break
        try:
            refined = slsqp_optimize(global_best_h, beta, max_iter=500)
        except Exception:
            continue
        refined = enforce_sum(refined, target_sum)
        c_ref = compute_c5(refined)
        if c_ref < global_best_c:
            global_best_h, global_best_c = refined.copy(), c_ref

    # -------------------------------------------------------------
    # 7. Optional up‑sampling to a finer grid (if time permits)
    # -------------------------------------------------------------
    elapsed = time.time() - start_time
    if budget_s - elapsed > 0.15 * budget_s:
        n2 = n * 2
        # linear interpolation of the current best
        old_x = (np.arange(n) + 0.5) * (2.0 / n)
        new_x = (np.arange(n2) + 0.5) * (2.0 / n2)
        h_interp = np.interp(new_x, old_x, global_best_h)
        # binarise and repair to exact half‑sum
        h_bin = (h_interp >= 0.5).astype(np.float64)
        target_sum2 = n2 // 2
        h_bin = enforce_sum(h_bin, target_sum2)

        arr2 = np.arange(n2)

        def compute_c5_n2(h):
            return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / n2)

        def hill_climb_n2(start_h, steps):
            cur_h = start_h.copy()
            cur_c = compute_c5_n2(cur_h)
            best_h, best_c = cur_h.copy(), cur_c
            for _ in range(steps):
                corr = np.correlate(cur_h, 1 - cur_h, mode="full")
                worst = corr.max()
                cur_c = worst
                worst_idxs = np.where(corr == worst)[0]
                idx = rng.choice(worst_idxs)
                shift = idx - (n2 - 1)

                cond_i = (cur_h == 1) & (arr2 + shift < n2) & (cur_h[arr2 + shift] == 0)
                i_cands = np.where(cond_i)[0]
                if i_cands.size:
                    i = rng.choice(i_cands)
                    cond_j = (cur_h == 0) & ((arr2 + shift >= n2) | (cur_h[arr2 + shift] == 1))
                    j_cands = np.where(cond_j)[0]
                    if j_cands.size:
                        j = rng.choice(j_cands)
                    else:
                        j = rng.choice(np.where(cur_h == 0)[0])
                    new_h = cur_h.copy()
                    new_h[i] = 0.0
                    new_h[j] = 1.0
                else:
                    ones = np.where(cur_h == 1)[0]
                    zeros = np.where(cur_h == 0)[0]
                    i, j = rng.choice(ones), rng.choice(zeros)
                    new_h = cur_h.copy()
                    new_h[i] = 0.0
                    new_h[j] = 1.0
                new_c = compute_c5_n2(new_h)
                if new_c <= cur_c or rng.random() < 0.001:
                    cur_h, cur_c = new_h, new_c
                    if cur_c < best_c:
                        best_h, best_c = cur_h.copy(), cur_c
            return best_h, best_c

        best_h2, best_c2 = h_bin, compute_c5_n2(h_bin)
        while time.time() - start_time < budget_s:
            # small random perturbation of current best
            pert = best_h2.copy()
            for _ in range(rng.integers(1, 4)):
                ones = np.where(pert == 1)[0]
                zeros = np.where(pert == 0)[0]
                i, j = rng.choice(ones), rng.choice(zeros)
                pert[i], pert[j] = 0.0, 1.0
            h_opt2, c_opt2 = hill_climb_n2(pert, steps=2000)
            if c_opt2 < best_c2:
                best_h2, best_c2 = h_opt2.copy(), c_opt2
        if best_c2 < global_best_c:
            global_best_h, global_best_c, n = best_h2.copy(), best_c2, n2
            target_sum = n // 2

    # -------------------------------------------------------------
    # 8. Final clean‑up and return
    # -------------------------------------------------------------
    global_best_h = np.clip(global_best_h, 0.0, 1.0)
    global_best_h = enforce_sum(global_best_h, target_sum)
    final_c = compute_c5(global_best_h)

    return global_best_h.tolist(), float(final_c), int(n)
