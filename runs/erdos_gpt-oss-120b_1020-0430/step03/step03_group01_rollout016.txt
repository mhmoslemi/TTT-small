```
def hill_climb_multi(h_start, rng, max_iters=5000, init_step=0.5, n_trials=10, decay=0.999):
    h = h_start.copy()
    best_c = compute_c5(h)
    curr_c = best_c
    step = init_step
    for it in range(max_iters):
        # generate n_trials candidates
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h_combined(h, rng, scale=step)
            new_c = compute_c5(new_h)
            # compute acceptance probability
            if new_c < curr_c - 1e-12:
                # strictly better
                h = new_h
                curr_c = new_c
                if new_c < best_c:
                    best_c = new_c
                improved = True
                break
            else:
                # accept worsening with small probability
                # temperature = step ... maybe
                T = step * 0.01
                if rng.random() < np.exp(-(new_c - curr_c) / (T + 1e-12)):
                    h = new_h
                    curr_c = new_c
                    # not update best
                    break
        # occasional larger jump
        if not improved and rng.random() < 0.01:
            new_h = perturb_h_combined(h, rng, scale=step*2.0)
            new_c = compute_c5(new_h)
            if new_c < curr_c:
                h = new_h
                curr_c = new_c
                if new_c < best_c:
                    best_c = new_c
        # decay step size
        step = max(step * decay, 1e-5)
    return h, best_c
```

import time
import numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    # smooth approximation of max_k sum_i x_i (1 - x_{i+k})
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def compute_c5(h):
    """Exact discretized C5."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h, target, rng):
    """Clip to [0,1] and adjust to hit the exact linear sum."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    # Simple redistribution loop
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add == 0:
            continue
        h[i] += add
        diff = target - h.sum()
    # final tiny correction
    if abs(diff) > 1e-12:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12 and n > 0:
            # distribute remaining error to first element
            h[0] += diff
            h[0] = np.clip(h[0], 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    """Random vector in [0,1]^n with sum = n/2."""
    # start from uniform and scale
    h = rng.random(n)
    # bring to target sum
    target = n / 2.0
    h = h / h.sum() * target
    # now clip and rebalance
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, target, rng)
    return h

def perturb_h(h, rng, scale=0.2):
    """Transfer ε between two entries preserving sum."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def swap_h(h, rng):
    """Swap two entries (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    new_h = h.copy()
    new_h[i], new_h[j] = new_h[j], new_h[i]
    return new_h

def perturb_h_combined(h, rng, scale):
    """Either a transfer or a swap, chosen randomly."""
    if rng.random() < 0.7:
        return perturb_h(h, rng, scale)
    else:
        return swap_h(h, rng)

def slsqp_optimize(h0, beta, max_iter=800):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum (tiny drift)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        x = enforce_sum(x, CURRENT_TARGET, rng)
    return x

def hill_climb_multi(h_start, rng, max_iters=8000, init_step=0.5, n_trials=12, decay=0.9995):
    """Local search that accepts improving moves and occasional uphill moves."""
    h = h_start.copy()
    best_h = h.copy()
    curr_c = compute_c5(h)
    best_c = curr_c
    step = init_step
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h_combined(h, rng, step)
            new_c = compute_c5(new_h)
            if new_c < curr_c - 1e-14:
                h = new_h
                curr_c = new_c
                if new_c < best_c:
                    best_c = new_c
                    best_h = new_h.copy()
                improved = True
                break
            else:
                # occasional uphill acceptance
                T = step * 0.01
                if rng.random() < np.exp(-(new_c - curr_c) / (T + 1e-14)):
                    h = new_h
                    curr_c = new_c
                    break
        if not improved and rng.random() < 0.01:
            # larger jump
            new_h = perturb_h_combined(h, rng, step * 2.0)
            new_c = compute_c5(new_h)
            if new_c < curr_c:
                h = new_h
                curr_c = new_c
                if new_c < best_c:
                    best_c = new_c
                    best_h = new_h.copy()
        step = max(step * decay, 1e-5)
    return best_h, best_c

    beta_list = [5, 20, 80, 200, 500, 1000, 2000, 5000]

    # ---------- main stochastic search ----------
    while time.time() - start_time < budget_s * 0.80:
        start_vec = perturb_h(best_h, rng, scale=0.5) if rng.random() < 0.7 else random_feasible_vector(n, rng)
        h_opt, c_opt = hill_climb_multi(
            start_vec, rng, max_iters=8000, init_step=0.5, n_trials=12, decay=0.9995
        )
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
        for beta in beta_list:
            if time.time() - start_time > budget_s * 0.85:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, max_iter=1200)
            except Exception:
                continue
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_h, best_c = h_ref, c_ref

    # ---------- final fine‑tuning ----------
    if budget_s - (time.time() - start_time) > budget_s * 0.10:
        h_fin, c_fin = hill_climb_multi(
            best_h, rng, max_iters=15000, init_step=0.2, n_trials=20, decay=0.9997
        )
        if c_fin < best_c:
            best_h, best_c = h_fin, c_fin

    # ---------- optional up‑sampling ----------
    if time.time() - start_time < budget_s * 0.95 and n < 720:
        new_n = min(720, n * 2)
        old_x = (np.arange(n) + 0.5) * (2.0 / n)
        new_x = (np.arange(new_n) + 0.5) * (2.0 / new_n)
        h_up = np.interp(new_x, old_x, best_h)
        h_up = enforce_sum(h_up, new_n / 2.0, rng)
        h_up, _ = hill_climb_multi(
            h_up, rng, max_iters=6000, init_step=0.3, n_trials=12, decay=0.9994
        )
        for beta in beta_list:
            if time.time() - start_time > budget_s:
                break
            try:
                h_up = slsqp_optimize(h_up, beta, max_iter=1200)
            except Exception:
                pass
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, n = h_up.copy(), c_up, new_n

    # ---------- clean‑up ----------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
