def compute_c5(h):

def anneal_binary(h, n, start_time, budget_s, rng):
    """Perform simulated annealing on binary vector h to minimize C5."""
    # h assumed binary integer array (float values but 0/1)
    # Manage sets
    h = h.astype(int).copy()
    ones = set(np.where(h==1)[0])
    zeros = set(np.where(h==0)[0])
    # compute current objective
    O_curr = int(np.max(np.correlate(h, 1 - h, mode='full')))
    best_h = h.copy()
    best_O = O_curr
    # parameters
    max_iters = 50000  # will be limited by time
    # Choose temperature schedule
    T0 = O_curr * 0.5 + 1.0  # start temperature relative to possible delta; ensure >0
    for it in range(max_iters):
        # check time
        if time.time() - start_time > budget_s * 0.99:
            break
        # propose swap
        i = rng.choice(list(ones))
        j = rng.choice(list(zeros))
        # perform swap
        h[i] = 0
        h[j] = 1
        # evaluate
        O_new = int(np.max(np.correlate(h, 1 - h, mode='full')))
        delta = O_new - O_curr
        # temperature
        T = T0 * (1 - it / max_iters)
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-8)):
            # accept swap
            O_curr = O_new
            # update sets
            ones.remove(i)
            zeros.remove(i)
            ones.add(j)
            zeros.add(j)
            if O_curr < best_O:
                best_O = O_curr
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1
            h[j] = 0
        # optional occasional random restart after stagnation
        if (it+1) % 5000 == 0:
            # if no improvement in last segment: maybe random perturb
            if best_O < O_curr:
                # we have improvement
                pass
            else:
                # random perturb: swap a few random pairs to diversify
                for _ in range(3):
                    i2 = rng.choice(list(ones))
                    j2 = rng.choice(list(zeros))
                    h[i2] = 0; h[j2] = 1
                    ones.remove(i2); zeros.remove(i2)
                    ones.add(j2); zeros.add(j2)
                O_curr = int(np.max(np.correlate(h, 1 - h, mode='full')))
    # output best solution scaled to float
    return best_h.astype(float), (best_O * (2.0 / n))

import time
import numpy as np

def compute_c5(h):
    """Exact discrete C5 for a step function on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return float(corr.max() * (2.0 / n))

def max_overlap(h):
    """Maximum overlap O_k = Σ h_i (1‑h_{i+k}) (integer, without dx)."""
    return int(np.correlate(h, 1 - h, mode="full").max())

def greedy_balanced_set(n, rng):
    """Greedy construction of a binary vector (n/2 ones) trying to equalise pairwise distances."""
    m = n // 2
    S = set()
    dist = np.zeros(n, dtype=int)          # distance counts, index 0 unused
    for _ in range(m):
        best_idx = -1
        best_min = -1
        # current minimal count (ignoring distance 0)
        cur_min = dist[1:].min()
        for i in range(n):
            if i in S:
                continue
            # simulate adding i
            new = dist.copy()
            for j in S:
                d = abs(i - j)
                new[d] += 1
            min_new = new[1:].min()
            if min_new > best_min:
                best_min = min_new
                best_idx = i
            elif min_new == best_min and rng.random() < 0.5:
                best_idx = i
        # commit best_idx
        for j in S:
            dist[abs(best_idx - j)] += 1
        S.add(best_idx)
    h = np.zeros(n, dtype=int)
    h[list(S)] = 1
    return h.astype(float)

def anneal_binary(start_h, n, start_time, budget_s, rng):
    """Simulated‑annealing on a binary vector to minimise max overlap."""
    # internal integer representation
    h = start_h.astype(int).copy()
    # initialise index lists for fast random selection
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    O_curr = max_overlap(h)
    best_h = h.copy()
    best_O = O_curr

    max_iters = 150_000
    T0 = max(1.0, O_curr * 0.5)                     # initial temperature
    for it in range(max_iters):
        if time.time() - start_time > budget_s * 0.99:
            break

        # pick random 1 and 0
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]

        # tentative swap
        h[i] = 0
        h[j] = 1
        O_new = max_overlap(h)
        delta = O_new - O_curr
        T = T0 * (1.0 - it / max_iters)
        accept = delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12))

        if accept:
            O_curr = O_new
            # update the index lists in‑place
            idx_i = ones.index(i)
            idx_j = zeros.index(j)
            ones[idx_i] = j
            zeros[idx_j] = i
            if O_curr < best_O:
                best_O = O_curr
                best_h = h.copy()
        else:
            # revert
            h[i] = 1
            h[j] = 0

        # occasional cheap refresh of lists (prevent hidden bias)
        if (it + 1) % 5000 == 0:
            ones = list(np.where(h == 1)[0])
            zeros = list(np.where(h == 0)[0])

    return best_h.astype(float), best_O * (2.0 / n)

    start = time.time()
    best_h = None
    best_c = float("inf")

    # ------------------------------------------------------------------

        elif best_h is not None:

            for _ in range(5):

            if best_c < 0.3805:

        # keep the current best for possible up‑sampling later
        if best_n == n and best_h is not None:

    # ------------------------------------------------------------------
    # final feasibility check and return
    # ------------------------------------------------------------------
    if best_h is None:
        # fallback: simple 0‑1 alternation

        h = np.zeros(n, dtype=float)
        h[::2] = 1.0
        best_h, best_n = h, n
        best_c = compute_c5(best_h)

    # enforce exact half‑weight (tiny correction if needed)
    best_h = np.clip(best_h, 0.0, 1.0)
    cur_sum = int(best_h.sum())
    target = best_n // 2
    if cur_sum != target:
        zeros = np.where(best_h == 0)[0]
        ones = np.where(best_h == 1)[0]
        if cur_sum < target:
            flip = rng.choice(zeros, target - cur_sum, replace=False)
            best_h[flip] = 1.0
        else:
            flip = rng.choice(ones, cur_sum - target, replace=False)
            best_h[flip] = 0.0

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
