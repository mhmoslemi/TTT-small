def make_binary(h):
    bin_h = (h > 0.5).astype(float)
    target = h.size // 2
    cur_sum = int(bin_h.sum())
    diff = cur_sum - target
    if diff > 0:

        idx = np.where(bin_h == 1)[0][:diff]
        bin_h[idx] = 0
    elif diff < 0:
        idx = np.where(bin_h == 0)[0][: -diff]
        bin_h[idx] = 1
    return bin_h

import time
import math
import numpy as np

# ------------------------------------------------------------
# Utility functions for binary step functions
# ------------------------------------------------------------
def make_binary(v):
    """Threshold to {0,1} and enforce exactly half ones."""
    b = (v > 0.5).astype(float)
    n = b.size
    target = n // 2
    cur = int(b.sum())
    diff = cur - target
    if diff > 0:
        idx = np.where(b == 1)[0][:diff]
        b[idx] = 0.0
    elif diff < 0:
        idx = np.where(b == 0)[0][: -diff]
        b[idx] = 1.0
    return b


def apply_swap_incrementally(c, h, i, new_val, idx, off):
    """Change h[i] to new_val, update full correlation c (vectorised)."""
    delta = new_val - h[i]
    if delta == 0.0:
        return
    # term where i is the first factor
    c[idx - i + off] += delta * (1.0 - h[idx])
    # term where i is the second factor
    c[i - idx + off] -= delta * h[idx]
    h[i] = new_val


def swap_pair(c, h, i, j, idx, off):
    """Swap values h[i] and h[j] (binary) and update correlation."""
    vi, vj = h[i], h[j]
    if vi == vj:
        return
    apply_swap_incrementally(c, h, i, vj, idx, off)
    apply_swap_incrementally(c, h, j, vi, idx, off)


def perturb_binary(h, rng, swaps=3):
    """Perform a few random swaps, keeping the sum unchanged."""
    n = h.size
    out = h.copy()
    for _ in range(swaps):
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if out[i] == out[j]:
            continue
        out[i], out[j] = out[j], out[i]
    return out


def compute_c5(h):
    """Exact C5 for a binary step function."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max()) * (2.0 / n)


# ------------------------------------------------------------
# Core simulated‑annealing routine
# ------------------------------------------------------------
def anneal_from(init_h, global_start, global_end, stop_time, rng):
    """Anneal a binary vector until stop_time, using the global schedule."""
    n = init_h.size
    idx = np.arange(n)
    off = n - 1

    h = init_h.copy()
    c = np.correlate(h, 1.0 - h, mode="full").astype(np.float64)
    cur_max = c.max()
    best_h = h.copy()
    best_c = cur_max * (2.0 / n)

    while True:
        now = time.time()
        if now >= stop_time:
            break

        # linear cooling based on remaining global time
        T = max(1e-6, 0.2 * (global_end - now) / (global_end - global_start))

        # propose a swap between a 1 and a 0
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if h[i] == h[j]:
            continue

        old_max = cur_max
        swap_pair(c, h, i, j, idx, off)
        new_max = c.max()

        if new_max <= old_max or rng.random() < math.exp(-(new_max - old_max) / T):
            cur_max = new_max
            cur_c = new_max * (2.0 / n)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            # revert the move
            swap_pair(c, h, i, j, idx, off)

    return best_h, best_c


def hill_climb(h, c, idx, off, steps, rng, time_limit):
    """Greedy improvement (accept only if the max overlap decreases)."""
    n = h.size
    cur_max = c.max()
    end = time.time() + steps / 1e5 + time_limit  # crude time guard
    for _ in range(steps):
        if time.time() > end:
            break
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if h[i] == h[j]:
            continue
        old_max = cur_max
        swap_pair(c, h, i, j, idx, off)
        new_max = c.max()
        if new_max < old_max:
            cur_max = new_max
        else:
            swap_pair(c, h, i, j, idx, off)
    return h, c

# ------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    global_end = start + budget_s

    # ---- initial dimension and vector ---------------------------------
    if "initial_h_values" in globals():
        h0_raw = np.array(initial_h_values, dtype=float)
        n0 = h0_raw.size
        # keep the original size for phase‑1; later we may upsample
        h0 = make_binary(h0_raw)
    else:
        n0 = 180
        h0 = np.zeros(n0, dtype=float)
        h0[: n0 // 2] = 1.0
        rng.shuffle(h0)

    # ---- phase‑1: explore many restarts on the original grid ----------
    phase1_end = start + budget_s * 0.70
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # a few basic starters (binary initial, random, perturbed)
    starters = [best_h]
    for _ in range(4):
        rnd = np.zeros(n0, dtype=float)
        rnd[: n0 // 2] = 1.0
        rng.shuffle(rnd)
        starters.append(rnd)

    # main restart loop
    while time.time() < phase1_end:
        # choose a starting vector: sometimes the current best, else random
        if rng.random() < 0.5 and best_h is not None:
            start_vec = perturb_binary(best_h, rng, swaps=5)
        else:
            start_vec = starters[rng.integers(len(starters))]
        # give each restart at most 12 seconds (or the remaining phase‑1 time)
        now = time.time()
        restart_stop = min(phase1_end, now + 12.0)
        local_h, local_c = anneal_from(start_vec, start, global_end, restart_stop, rng)
        if local_c < best_c:
            best_c = local_c
            best_h = local_h

    # ---- optional up‑sampling and refinement -------------------------
    remaining = global_end - time.time()
    if remaining > 30.0 and best_h.size * 2 <= 720:
        # double the resolution by repetition
        up_h = np.repeat(best_h, 2)
        n_up = up_h.size
        idx_up = np.arange(n_up)
        off_up = n_up - 1
        # start from the up‑sampled version, run annealing until the very end
        up_h, up_c = anneal_from(up_h, start, global_end, global_end, rng)
        if up_c < best_c:
            best_c, best_h = up_c, up_h
        # final quick greedy sweep
        c_full = np.correlate(best_h, 1.0 - best_h, mode="full").astype(np.float64)
        best_h, c_full = hill_climb(best_h, c_full, idx_up, off_up, 200000, rng, 5.0)
        best_c = compute_c5(best_h)

    # ---- final clean‑up ------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    # enforce exact half‑ones (tiny correction if needed)
    diff = int(best_h.sum() - best_h.size // 2)
    if diff != 0:
        if diff > 0:
            idx = np.where(best_h == 1.0)[0][:diff]
            best_h[idx] = 0.0
        else:
            idx = np.where(best_h == 0.0)[0][: -diff]
            best_h[idx] = 1.0
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
