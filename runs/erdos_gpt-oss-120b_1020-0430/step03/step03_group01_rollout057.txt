def targeted_perturb(h, rng, scale=0.2):
    n = h.size
    # compute full correlation
    corr = np.correlate(h, 1 - h, mode='full')
    lag = np.argmax(corr) - (n - 1)
    k = lag
    # compute product for this lag
    if k >= 0:
        i_range = np.arange(n - k)
        prod = h[i_range] * h[i_range + k]
    else:
        kp = -k
        i_range = np.arange(n - kp)
        prod = h[i_range + kp] * h[i_range]  # same
    # find index i where prod high, and both h components high
    high_mask = prod > (0.5)  # tune
    # maybe pick from top 10% by product
    if np.any(high_mask):
        candidates = np.where(high_mask)[0]
        i = rng.choice(candidates)
        # decide which of the two indices to decrease: choose the higher h value
        if k >= 0:
            i1 = i
            i2 = i + k
        else:
            i1 = i - k
            i2 = i
        # pick the larger h
        if h[i1] >= h[i2]:
            dec_idx = i1
        else:
            dec_idx = i2
        # find a low index to increase
        low_mask = h < 0.3
        if np.any(low_mask):
            inc_idx = rng.choice(np.where(low_mask)[0])
            # compute max epsilon
            max_eps = min(h[dec_idx], 1.0 - h[inc_idx])
            if max_eps > 0:
                eps = scale * max_eps * rng.random()
                new_h = h.copy()
                new_h[dec_idx] -= eps
                new_h[inc_idx] += eps
                return new_h
    # fallback random perturb
    return perturb_h(h, rng, scale=scale)

def hill_climb_targeted(h_start, rng, max_iters=5000, init_scale=0.5, decay=0.99):
    h = h_start.copy()
    best_c = compute_c5(h)
    scale = init_scale
    for it in range(max_iters):
        new_h = targeted_perturb(h, rng, scale=scale)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            h = new_h
            # maybe keep scale higher after improvement?
        else:
            # maybe accept with probability decreasing (simulated annealing)
            # temperature schedule
            T = 0.01 * (1 - it / max_iters)  # example
            if rng.random() < np.exp((best_c - c) / (T + 1e-12)):
                h = new_h
        scale = max(scale * decay, 1e-4)
    return h, best_c

    if abs(diff) > 1e-10:
        # distribute diff to random indices respecting bounds

import time
import numpy as np
from scipy.optimize import minimize

# globals for SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1-x_{i+k}) via log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h so that Σh = target while respecting 0≤h≤1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        # small global correction
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def targeted_perturb(h, rng, scale=0.2):
    """Move mass from a high‑product pair at the worst lag to a low‑value site."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    lag = np.argmax(corr) - (n - 1)          # lag giving maximal overlap
    k = lag
    if k >= 0:
        i_range = np.arange(n - k)
        prod = h[i_range] * h[i_range + k]
        i = i_range[np.argmax(prod)]
        i1, i2 = i, i + k
    else:
        kp = -k
        i_range = np.arange(n - kp)
        prod = h[i_range + kp] * h[i_range]
        i = i_range[np.argmax(prod)]
        i1, i2 = i + kp, i
    dec_idx = i1 if h[i1] >= h[i2] else i2
    low = np.where(h < 0.3)[0]
    if low.size == 0:
        return perturb_h(h, rng, scale)
    inc_idx = rng.choice(low)
    max_eps = min(h[dec_idx], 1.0 - h[inc_idx])
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale)
    eps = scale * max_eps * rng.random()
    new_h = h.copy()
    new_h[dec_idx] -= eps
    new_h[inc_idx] += eps
    return new_h


def hill_climb_targeted(h_start, rng, max_iters=5000, init_scale=0.5, decay=0.998):
    """Hill‑climb using targeted perturbations."""
    h = h_start.copy()
    best_c = compute_c5(h)
    scale = init_scale
    for it in range(max_iters):
        new_h = targeted_perturb(h, rng, scale=scale)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            h = new_h
        else:
            # occasional Metropolis acceptance
            T = max(0.01 * (1.0 - it / max_iters), 1e-4)
            if rng.random() < np.exp((best_c - c) / T):
                h = new_h
        scale = max(scale * decay, 1e-5)
    return h, best_c


def slsqp_optimize(h0, beta, max_iter=400):
    """Refine h using SLSQP on the smooth surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum (tiny drift)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        x = enforce_sum(x, CURRENT_TARGET, rng)
    return x

        if rng.random() < 0.15:
            for beta in (50, 200, 500, 1000):
                if time.time() - start_time > budget_s * 0.85:
                    break
                try:
                    h_ref = slsqp_optimize(best_h, beta, max_iter=300)
                except Exception:
                    continue
                c_ref = compute_c5(h_ref)
                if c_ref < best_c:
                    best_c, best_h = c_ref, h_ref.copy()

    # -------------------------------------------------------------
    # optional up‑sampling if time permits
    # -------------------------------------------------------------
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > budget_s * 0.1:
        target_n = min(720, max(480, n * 2))
        if target_n != n:
            old_x = (np.arange(n) + 0.5) * (2.0 / n)
            new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
            h_up = np.interp(new_x, old_x, best_h)
            h_up = enforce_sum(h_up, target_n / 2.0, rng)
            h_up, c_up = hill_climb_targeted(h_up, rng, max_iters=6000, init_scale=0.4, decay=0.998)
            # SLSQP fine‑tuning on up‑sampled version
            for beta in (200, 500, 1000, 2000):
                if time.time() - start_time > budget_s:
                    break
                try:
                    h_up = slsqp_optimize(h_up, beta, max_iter=300)
                except Exception:
                    pass
                c_up = compute_c5(h_up)
                if c_up < best_c:
                    best_c, best_h, n = c_up, h_up.copy(), target_n

    # final cleanup
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
