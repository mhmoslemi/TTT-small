def project_to_feasible(v, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

### SLSQP function using symmetric representation.

```
def slsqp_optimize_sym(x0, n, beta, rng, max_iter=200):
    m = len(x0)
    target = n/4.0
    def obj(x):
        # using global beta variable per call
        h = sym_expand(x, n)
        corr = np.correlate(h, 1 - h, mode='full')
        bc = beta * corr
        mmax = np.max(bc)
        return (mmax + np.log(np.sum(np.exp(bc - mmax)))) / beta
    bounds = [(0.0, 1.0)] * m
    cons = {'type':'eq', 'fun': lambda x: np.sum(x) - target}
    res = minimize(obj, x0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol':1e-12, 'disp':False})
    # clip to [0,1]
    x = np.clip(res.x, 0.0, 1.0)
    # correct sum projection
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = project_to_feasible(x, target)
    return x
```

def perturb_h_full(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    eps = (rng.random() - 0.5) * 2 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h
```

def hill_climb_full(h_start, rng, max_iters=2000, start_step=0.3):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h_full(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it+1) % decay_every == 0:
            step = max(step * 0.85, 1e-5)
    return best_h, best_c
```

import time
import numpy as np
from scipy.optimize import minimize

# ------------------- helper utilities ------------------- #

def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_to_feasible(v, target):
    """
    Projection of v onto {0≤x≤1, Σx = target} via a simple bisection on the shift.
    """
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector_sym(m, target, rng):
    """Generate a random feasible vector for the symmetric parametrisation."""
    v = rng.random(m)
    v = v / v.sum() * target
    return project_to_feasible(v, target)


def sym_expand(x, n):
    """Expand a half‑vector x (length n/2) to a full symmetric step function."""
    m = len(x)
    h = np.empty(n)
    h[:m] = x
    h[-m:] = x[::-1]
    return h


def upsample_initial(to_n):
    """Linear interpolation of the provided initial_h_values to length to_n."""
    init = np.array(initial_h_values, dtype=float)
    m = init.size
    xs_old = np.linspace(0.0, 2.0, m, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, to_n, endpoint=False)
    return np.interp(xs_new, xs_old, init)


def slsqp_optimize_sym(x0, n, beta, rng, max_iter=200):
    """SLSQP optimisation of the smooth max‑approximation for a symmetric h."""
    m = len(x0)
    target = n / 4.0

    def obj(x):
        h = sym_expand(x, n)
        corr = np.correlate(h, 1.0 - h, mode="full")
        bc = beta * corr
        mmax = np.max(bc)
        # log‑sum‑exp in a numerically stable way
        return (mmax + np.log(np.sum(np.exp(bc - mmax)))) / beta

    bounds = [(0.0, 1.0)] * m
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    res = minimize(
        obj,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # final projection to exactly satisfy the sum constraint
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = project_to_feasible(x, target)
    return x


def multistart_sym_opt(n, rng, time_budget):
    """Multi‑start symmetric optimisation within a time budget."""
    m = n // 2
    target = n / 4.0
    best_h = None
    best_c = np.inf
    start = time.time()

    # Seed from the provided initial_h_values if available
    if "initial_h_values" in globals():
        init_full = upsample_initial(n)
        x0 = np.empty(m)
        for i in range(m):
            x0[i] = 0.5 * (init_full[i] + init_full[n - 1 - i])
        x0 = project_to_feasible(x0, target)
        x = x0
        for beta in (20, 100, 500, 2000):
            if time.time() - start > time_budget:
                break
            x = slsqp_optimize_sym(x, n, beta, rng, max_iter=250)
        h = sym_expand(x, n)
        c = compute_c5(h)
        best_h, best_c = h, c

    # Random restarts
    attempts = 0
    while time.time() - start < time_budget and attempts < 30:
        attempts += 1
        x = random_feasible_vector_sym(m, target, rng)
        for beta in (10, 50, 200, 800, 2000):
            if time.time() - start > time_budget:
                break
            x = slsqp_optimize_sym(x, n, beta, rng, max_iter=200)
        h = sym_expand(x, n)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
    return best_h, best_c


def perturb_h_full(h, rng, scale=0.2):
    """Transfer a tiny amount between two coordinates (preserving the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_full(h_start, rng, max_iters=2000, start_step=0.3):
    """Simple deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h_full(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.85, 1e-5)
    return best_h, best_c


# ------------------- main optimisation entry point ------------------- #

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # Candidate discretisations (even lengths for symmetry)
    candidate_ns = [180, 240, 300, 360, 480, 600]

    for n in candidate_ns:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining < 10.0:
            break

        # allocate a modest slice of the remaining budget to this n
        time_for_n = min(remaining * 0.5, max(5.0, min(180.0, n * 0.35)))

        # symmetric multi‑start optimisation
        h_sym, c_sym = multistart_sym_opt(n, rng, time_for_n)

        if h_sym is not None and c_sym < best_c:
            best_c, best_h, best_n = c_sym, h_sym.copy(), n

        # short hill‑climb refinement if time permits
        if time.time() - start_time < budget_s * 0.95 and best_h is not None:
            h_hc, c_hc = hill_climb_full(best_h, rng, max_iters=1500, start_step=0.25)
            if c_hc < best_c:
                best_c, best_h, best_n = c_hc, h_hc.copy(), n

    # final asymmetric hill‑climb on the best solution found (if time left)
    if best_h is not None and (budget_s - (time.time() - start_time)) > 5.0:
        h_final, c_final = hill_climb_full(best_h, rng, max_iters=3000, start_step=0.3)
        if c_final < best_c:
            best_c, best_h = c_final, h_final.copy()
            best_n = best_h.size

    # fallback if nothing succeeded
    if best_h is None:
        best_n = 120
        best_h = np.full(best_n, 0.5)
        best_c = compute_c5(best_h)

    # enforce feasibility exactly
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, best_n)
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
