def gradient_descent(h, rng, deadline, max_iters=2000, init_step=0.5):
    best_h = h.copy()
    best_c = compute_c5(best_h)
    step = init_step
    n = best_h.size
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = correlate_full(best_h)
        # shift index
        shift = int(np.argmax(conv) - (n - 1))
        # compute gradient for worst shift
        g = gradient_worst_shift(best_h, shift)
        # adjust direction to preserve sum
        avg = g.mean()
        d = -g + avg
        # compute max feasible step
        # avoid zero division
        pos_mask = d > 1e-12
        neg_mask = d < -1e-12
        if np.any(pos_mask):
            max_step_pos = np.min((1.0 - best_h)[pos_mask] / d[pos_mask])
        else:
            max_step_pos = np.inf
        if np.any(neg_mask):
            max_step_neg = np.min((-best_h)[neg_mask] / d[neg_mask])
        else:
            max_step_neg = np.inf
        max_alpha = min(max_step_pos, max_step_neg)
        # line search
        alpha = min(step, max_alpha)
        improved = False
        while alpha > 1e-8:
            new_h = best_h + alpha * d
            # Clip to [0,1] just in case numeric
            np.clip(new_h, 0.0, 1.0, out=new_h)
            # enforce exact sum, small correction
            target = n / 2.0
            diff = target - new_h.sum()
            if abs(diff) > 1e-12:
                # apply simple correction by distributing diff proportionally to slack
                # modify entries not at bounds
                slack_pos = (new_h < 1.0)
                slack_neg = (new_h > 0.0)
                # distribute diff equally among all slack entries
                # This may violate bounds; we can fallback to enforce_sum
                new_h = enforce_sum(new_h, target, rng)
            new_c = compute_c5(new_h)
            if new_c < best_c - 1e-12:
                best_h, best_c = new_h, new_c
                improved = True
                break
            else:
                alpha *= 0.5
        if not improved:
            step *= 0.7
            if step < 1e-6:
                break
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic feasibility utilities
# ----------------------------------------------------------------------
def enforce_sum(h, target, rng):
    """Adjust h to have sum == target while respecting 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


# ----------------------------------------------------------------------
# Correlation / objective
# ----------------------------------------------------------------------
def correlate_full(h):
    """Full linear correlation of h with (1-h) using FFT."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())  # next power of two
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Discrete C5 value for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)


# ----------------------------------------------------------------------
# Gradient of the worst shift
# ----------------------------------------------------------------------
def gradient_worst_shift(h, shift):
    """Gradient of f_k = Σ h_i (1 - h_{i+shift}) w.r.t. h."""
    n = h.shape[0]
    g = np.ones(n)
    if shift == 0:
        g -= 2.0 * h
        return g
    if shift > 0:
        k = shift
        g[: n - k] -= h[k:]
        g[k:] -= h[: n - k]
    else:
        k = -shift
        g[k:] -= h[: n - k]
        g[: n - k] -= h[k:]
    return g


def gradient_descent(h0, rng, deadline, max_iters=3000, init_step=0.5):
    """Deterministic descent on the worst shift (preserves sum)."""
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    step = init_step
    n = best_h.size
    for _ in range(max_iters):
        if time.time() >= deadline:
            break
        conv = correlate_full(best_h)
        shift = int(np.argmax(conv) - (n - 1))
        g = gradient_worst_shift(best_h, shift)
        avg = g.mean()
        d = -g + avg                 # direction with zero sum
        # maximal feasible step without leaving [0,1]
        pos = d > 1e-12
        neg = d < -1e-12
        max_alpha = np.inf
        if np.any(pos):
            max_alpha = min(max_alpha, np.min((1.0 - best_h)[pos] / d[pos]))
        if np.any(neg):
            max_alpha = min(max_alpha, np.min((-best_h)[neg] / d[neg]))
        alpha = min(step, max_alpha)
        improved = False
        while alpha > 1e-9:
            new_h = best_h + alpha * d
            # numerical safety
            new_h = np.clip(new_h, 0.0, 1.0)
            new_c = compute_c5(new_h)
            if new_c < best_c - 1e-12:
                best_h, best_c = new_h, new_c
                improved = True
                break
            alpha *= 0.5
        if not improved:
            step *= 0.7
            if step < 1e-6:
                break
    return best_h, best_c

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Log-sum-exp smoothing of max_k Σ_i x_i (1‑x_{i+k})."""
    conv = correlate_full(x)
    bc = CURRENT_BETA * conv
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=500):
    """SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:

            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

    if (
        best_h is not None
        and time.time() < deadline_main
        and best_h.size * 2 <= 800
    ):
        h_up = np.repeat(best_h, 2)
        n_up = h_up.size
        target_up = n_up / 2.0
        h_up = enforce_sum(h_up, target_up, rng)
        h_up_opt, c_up_opt = gradient_descent(
            h_up, rng, deadline_main, max_iters=5000, init_step=0.4
        )
        if c_up_opt < best_c:
            best_h, best_c = h_up_opt, c_up_opt

    # --------------------------------------------------------------
    # Final SLSQP refinements with increasing smoothness
    # --------------------------------------------------------------
    beta_list = [5, 20, 80, 200, 800, 2000]
    for beta in beta_list:
        if time.time() >= deadline_final:
            break
        try:
            h_ref = slsqp_optimize(best_h, beta, max_iter=500)
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_h, best_c = h_ref, c_ref
        except Exception:
            continue

    # --------------------------------------------------------------
    # Clean‑up and return
    # --------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
