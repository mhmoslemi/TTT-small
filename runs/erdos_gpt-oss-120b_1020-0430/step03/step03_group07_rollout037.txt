def project_onto_bounds_and_sum(h, target):
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Identify free indices where h not at bounds (or where we can still move)
    free = np.where((h > 1e-12) & (h < 1-1e-12))[0]
    if free.size == 0:
        # All at bounds; adjust by distributing to all (may cause violation)
        # We'll fallback to scaling, though diff should be near zero.
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
        # Re-adjust recursively
        return project_onto_bounds_and_sum(h, target)
    # Distribute diff equally among free indices
    per = diff / free.size
    # Limit addition to bounds
    for i in free:
        add = per
        if add > 0:
            add = min(add, 1.0 - h[i])
        else:
            add = max(add, -h[i])
        h[i] += add
    # Recompute diff recursively until small
    new_diff = target - h.sum()
    if abs(new_diff) > 1e-12:
        return project_onto_bounds_and_sum(h, target)
    return h
```

            break
        free = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
        if free.size == 0:
            # no free entries; adjust all (should be near target already)
            # distribute equally
            h += diff / h.size
            h = np.clip(h, 0.0, 1.0)
            diff = target - h.sum()
        else:
            # distribute diff equally among free
            per = diff / free.size
            # limit per to bounds
            for i in free:
                add = per
                if add > 0:
                    add = min(add, 1.0 - h[i])
                else:
                    add = max(add, -h[i])
                h[i] += add
            diff = target - h.sum()
    return h

def correlate_full(h):
    n = h.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2*n - 1]
    return conv

    # Compute correlation
    corr = correlate_full(h)

import time
import numpy as np

# -------------------------------------------------------------
# Helper functions
# -------------------------------------------------------------
def correlate_full(h):
    """Full linear correlation of h with (1-h).  Returns length 2n-1 array."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1.0 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)


def enforce_sum(h, target, rng):
    """Adjust h so that Σh == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_binary_vector(n, rng):
    """Binary feasible vector with exactly n/2 ones (n must be even)."""
    if n % 2 != 0:
        # fallback to a continuous random feasible vector
        h = np.full(n, 0.5)
        return enforce_sum(h, n / 2.0, rng)
    idx = rng.choice(n, n // 2, replace=False)
    h = np.zeros(n, dtype=float)
    h[idx] = 1.0
    return h


def subgradient_step(h, target, step, rng):
    """One subgradient descent step targeting the current maximal correlation."""
    n = h.size
    corr = correlate_full(h)
    max_idx = int(np.argmax(corr))
    shift = max_idx - (n - 1)          # shift k corresponding to max lag
    grad = np.ones_like(h)

    if shift >= 0:
        k = shift
        grad[: n - k] -= h[k:]
        grad[k:]      -= h[: n - k]
    else:
        k = -shift
        grad[k:]      -= h[: n - k]
        grad[: n - k] -= h[k:]

    new_h = h - step * grad
    new_h = np.clip(new_h, 0.0, 1.0)
    new_h = enforce_sum(new_h, target, rng)
    return new_h


def softmax_step(h, target, beta, step, rng):
    """One gradient step on the smooth log‑sum‑exp surrogate."""
    n = h.size
    corr = correlate_full(h)
    # softmax weights
    max_bc = beta * corr.max()
    w = np.exp(beta * corr - max_bc)
    w_sum = w.sum()
    if w_sum == 0.0:
        return h
    w /= w_sum

    w_rev = w[::-1]
    conv1 = np.convolve(w_rev, h, mode='full')
    conv2 = np.convolve(w, h, mode='full')
    offset = len(w) // 2          # = n-1
    grad = 1.0 - (conv1[offset:offset + n] + conv2[offset:offset + n])

    new_h = h - step * grad
    new_h = np.clip(new_h, 0.0, 1.0)
    new_h = enforce_sum(new_h, target, rng)
    return new_h


# -------------------------------------------------------------
# Main optimisation routine
# -------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    total_deadline = start_time + budget_s

    # ---- 1. Initial vector -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        # plain binary start (half ones)
        h0 = random_binary_vector(360, rng)

    n = h0.size
    target = n / 2.0
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = enforce_sum(h0, target, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ---- 2. Subgradient phase (coarse) ------------------------------------
    sub_deadline = start_time + budget_s * 0.45
    step = 0.12
    it = 0
    while time.time() < sub_deadline and it < 2500:
        h_new = subgradient_step(best_h, target, step, rng)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        # decay step gently
        step *= 0.9995
        it += 1

    # ---- 3. Smooth softmax gradient phase ---------------------------------
    beta_schedule = [5.0, 20.0, 80.0, 200.0, 800.0]
    soft_deadline = start_time + budget_s * 0.78
    for beta in beta_schedule:
        if time.time() >= soft_deadline:
            break
        step = 0.08 if beta < 200 else 0.04
        it = 0
        while time.time() < soft_deadline and it < 1800:
            h_new = softmax_step(best_h, target, beta, step, rng)
            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            step *= 0.9997
            it += 1

    # ---- 4. Optional up‑sampling to finer grid -----------------------------
    up_deadline = total_deadline - 2.0   # keep a small margin
    if time.time() < up_deadline and n * 2 <= 720:
        # double resolution by piecewise‑constant repetition
        h_up = np.repeat(best_h, 2)
        n_up = h_up.size
        target_up = n_up / 2.0
        h_up = enforce_sum(h_up, target_up, rng)

        # short refinement on the finer grid
        beta_fine = 2000.0
        step_fine = 0.02
        it = 0
        while time.time() < up_deadline and it < 3000:
            h_up = softmax_step(h_up, target_up, beta_fine, step_fine, rng)
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_c = c_up
                best_h = h_up.copy()
                n = n_up
            step_fine *= 0.9998
            it += 1

    # ---- 5. Final clean‑up -------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
