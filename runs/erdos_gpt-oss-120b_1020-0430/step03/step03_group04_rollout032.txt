def softmax_objective_and_gradient(h, beta):
    O = np.correlate(h, 1 - h, mode='full')
    max_shift = O.max()
    # compute weights:
    bc = beta * O
    max_bc = bc.max()
    exp_bc = np.exp(bc - max_bc)
    sum_exp = exp_bc.sum()
    w = exp_bc / sum_exp
    # gradient:
    n = len(h)
    g = np.zeros_like(h)
    # loops over shift index
    for idx, w_k in enumerate(w):
        k = idx - (n - 1)
        # compute contributions
        if k >= 0:
            # (1 - h[j+k]) term for j=0..n-1-k
            g[:n - k] += w_k * (1 - h[k:])
            # -h[j-k] term for j=k..n-1
            g[k:] -= w_k * h[:n - k]
        else: # k negative
            kk = -k
            # (1 - h[j - kk]) term for j=kk..n-1
            g[kk:] += w_k * (1 - h[:n - kk])

def softmax_obj_and_grad(h, beta):
    n = h.size
    O = np.correlate(h, 1 - h, mode='full')
    bc = beta * O
    maxb = bc.max()
    expb = np.exp(bc - maxb)
    sumexp = expb.sum()
    w = expb / sumexp   # weights sum to 1

    g = np.zeros_like(h)
    # shift loop
    for idx, wk in enumerate(w):
        k = idx - (n - 1)
        if wk == 0.0:
            continue
        if k >= 0:
            # indices 0..n-k-1 get (1 - h[i+k])
            g[:n - k] += wk * (1 - h[k:])
            # indices k..n-1 get -h[i - k]
            g[k:] -= wk * h[:n - k]
        else:
            kk = -k
            # indices kk..n-1 get (1 - h[i - kk])
            g[kk:] += wk * (1 - h[:n - kk])
            # indices 0..n-kk-1 get -h[i + kk]
            g[:n - kk] -= wk * h[kk:]
    # objective
    obj = (maxb + np.log(sumexp)) / beta
    return obj, g

def project_feasible(h, target, rng):
    # clip
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # random adjustments
        for _ in range(2000):
            i = rng.integers(0, h.size)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h

def softmax_obj_and_grad(h, beta):
    n = h.shape[0]
    O = np.correlate(h, 1 - h, mode='full')
    bc = beta * O
    maxb = bc.max()
    expb = np.exp(bc - maxb)
    sumexp = expb.sum()
    w = expb / sumexp
    g = np.zeros_like(h)
    # We need to loop over shift k for each weight
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        k = idx - (n - 1)
        if k >= 0:
            # g[0:n-k] += wk * (1 - h[k:])
            g[:n - k] += wk * (1 - h[k:])
            # g[k:n] -= wk * h[:n - k]
            g[k:] -= wk * h[:n - k]
        else:
            kk = -k
            g[kk:] += wk * (1 - h[:n - kk])
            g[:n - kk] -= wk * h[kk:]
    obj = (maxb + np.log(sumexp)) / beta
    return obj, g

import time
import numpy as np

# ------------------ basic utilities ------------------ #
def compute_c5(h):
    """Exact discrete C5 for step function h on [0,2] with uniform spacing."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def project_feasible(h, target, rng):
    """Clip to [0,1] and adjust to hit the exact sum `target`."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, h.size)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h


def random_feasible_vector(n, rng):
    """Uniform random vector in [0,1] with exact sum n/2."""
    h = rng.random(n)
    target = n / 2.0
    h = h * (target / h.sum())
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h


# ----------- soft‑max objective and gradient ----------- #
def softmax_obj_and_grad(h, beta):
    """Smooth max using log‑sum‑exp and its gradient."""
    n = h.shape[0]
    O = np.correlate(h, 1 - h, mode="full")
    bc = beta * O
    maxb = bc.max()
    expb = np.exp(bc - maxb)
    sumexp = expb.sum()
    w = expb / sumexp                     # normalized weights
    g = np.zeros_like(h)

    # loop over all shifts
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += wk * (1 - h[k:])
            g[k:]      -= wk * h[: n - k]
        else:
            kk = -k
            g[kk:] += wk * (1 - h[: n - kk])
            g[: n - kk] -= wk * h[kk:]

    obj = (maxb + np.log(sumexp)) / beta
    return obj, g


def gradient_descent_softmax(h, rng, betas, max_iters_per_beta=150, step0=0.6):
    """Gradient descent on the soft‑max surrogate, with feasibility projection."""
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(best_h)

    for beta in betas:
        for it in range(max_iters_per_beta):
            _, grad = softmax_obj_and_grad(best_h, beta)
            step = step0 * (0.95 ** it)
            cand = best_h - step * grad
            cand = project_feasible(cand, target, rng)
            cand_c = compute_c5(cand)
            if cand_c < best_c:
                best_c = cand_c
                best_h = cand
        step0 = max(step0 * 0.5, 1e-4)   # tighten step for larger betas
    return best_h, best_c


# ------------------- hill climbing ------------------- #
def hill_climb(h_start, rng, max_iters=2000, step=0.5):
    """Simple deterministic hill‑climber (move mass between two coordinates)."""
    n = h_start.size
    target = n / 2.0
    h = h_start.copy()
    best_c = compute_c5(h)

    cur_step = step
    decay_every = max(1, max_iters // 10)

    for it in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * cur_step
        eps = max(-max_eps, min(eps, max_eps))
        cand = h.copy()
        cand[i] -= eps
        cand[j] += eps
        cand_c = compute_c5(cand)
        if cand_c < best_c:
            best_c = cand_c
            h = cand
        if (it + 1) % decay_every == 0:
            cur_step = max(cur_step * 0.7, 1e-4)
    return h, best_c

            x_tgt = np.linspace(0.0, 2.0, n, endpoint=False)
            h_up = np.interp(x_tgt, x_src, parent)
            starts.append(project_feasible(h_up, target, rng))

        # 2) a couple of random continuous feasible vectors
        for _ in range(2):
            starts.append(random_feasible_vector(n, rng))

        # 3) random binary feasible vector
        h_bin = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h_bin[idx] = 1.0
        starts.append(h_bin)

        # 4) perturbation of the current global best if same size
        if best_n == n and best_h is not None:
            starts.append(best_h + 0.02 * rng.standard_normal(n))

        for h0 in starts:
            if time.time() - start_time > budget_s * 0.95:
                break

            h0 = project_feasible(h0, target, rng)

            # ---- stage A: soft‑max gradient descent ----
            h_cur, c_cur = gradient_descent_softmax(h0, rng, betas,
                                                   max_iters_per_beta=120,
                                                   step0=0.7)
            if c_cur < best_c:
                best_h, best_c, best_n = h_cur.copy(), c_cur, n

            # ---- stage B: deterministic hill climb ----
            h_cur, c_cur = hill_climb(best_h, rng, max_iters=800, step=0.5)
            if c_cur < best_c:
                best_h, best_c, best_n = h_cur.copy(), c_cur, n

            # ---- stage C: simulated annealing (time‑capped) ----
            remaining = budget_s - (time.time() - start_time)
            if remaining > 30.0:
                deadline_sa = time.time() + remaining * 0.5
                h_cur, c_cur = simulated_annealing(best_h, rng, deadline_sa)
                if c_cur < best_c:
                    best_h, best_c, best_n = h_cur.copy(), c_cur, n

            # ---- stage D: final hill climb ----
            if time.time() - start_time < budget_s * 0.99:
                h_cur, c_cur = hill_climb(best_h, rng, max_iters=1500, step=0.3)
                if c_cur < best_c:
                    best_h, best_c, best_n = h_cur.copy(), c_cur, n

    # ----------- optional up‑sampling refinement ----------- #
    if best_h is not None and time.time() - start_time < budget_s * 0.99:
        if best_n * 2 <= 300:
            n_up = best_n * 2
            x_src = np.linspace(0.0, 2.0, best_n, endpoint=False)
            x_tgt = np.linspace(0.0, 2.0, n_up, endpoint=False)
            h_up = np.interp(x_tgt, x_src, best_h)
            h_up = project_feasible(h_up, n_up / 2.0, rng)

            h_up, c_up = gradient_descent_softmax(h_up, rng, betas,
                                                 max_iters_per_beta=200,
                                                 step0=0.5)
            h_up, c_up = hill_climb(h_up, rng, max_iters=1200, step=0.4)
            if c_up < best_c:
                best_h, best_c, best_n = h_up, c_up, n_up

    # ------------------- final safety check ------------------- #
    if best_h is None:
        best_n = 180
        best_h = random_feasible_vector(best_n, rng)

    best_h = project_feasible(best_h, best_n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
