def softmax_gradient(h, beta):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    maxcorr = np.max(corr)
    w = np.exp(beta * (corr - maxcorr))
    w_sum = np.sum(w)
    w = w / w_sum
    # compute gradient
    g = np.zeros_like(h)
    # k from -(n-1) to (n-1)
    for kk in range(-(n-1), n):
        wk = w[kk + n - 1]
        if wk == 0: continue
        # compute 1 - h_{i+kk} - h_{i-kk}
        term = wk * (1 - np.roll(h, -kk) - np.roll(h, kk))
        g += term
    return g

def softmax_gradient(h, beta):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    maxcorr = np.max(corr)
    # evaluate exponent
    exp_vals = np.exp(beta * (corr - maxcorr))
    w = exp_vals / np.sum(exp_vals)
    g = np.zeros_like(h)
    # shift 0
    wk = w[n-1]  # index for shift zero
    g += wk * (1.0 - 2.0 * h)
    # Positive shifts
    for k in range(1, n):
        wk = w[n-1 + k] + w[n-1 - k]
        upper = n - k
        # term for 1 - h[i+k] for i in 0..upper-1
        g[:upper] += wk * (1.0 - h[k:])
        # term for -h[i] for i in k..n-1
        g[k:] -= wk * h[:upper]
    return g

import time
import numpy as np

# ---------- basic utilities ----------

def compute_c5(h):
    """Discrete C5 for step function h on [0,2] with uniform spacing."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def random_feasible_vector(n, rng):
    """Uniform random vector in [0,1] with exact sum n/2."""
    h = rng.random(n)
    target = n / 2.0
    scale = target / h.sum()
    h = np.clip(h * scale, 0.0, 1.0)
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            if add > 0:
                h[i] += add
                diff -= add
        else:
            sub = max(diff, -h[i])
            if sub < 0:
                h[i] += sub
                diff -= sub
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_simple(h_start, rng, max_iters=800, start_step=0.3):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


def softmax_gradient(h, beta):
    """Gradient of the soft‑max surrogate objective."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    maxc = np.max(corr)
    w = np.exp(beta * (corr - maxc))
    w /= w.sum()
    g = np.zeros_like(h)

    # shift 0
    wk = w[n - 1]
    g += wk * (1.0 - 2.0 * h)

    # positive shifts (combine +k and -k)
    for k in range(1, n):
        wk = w[n - 1 + k] + w[n - 1 - k]
        upper = n - k
        g[:upper] += wk * (1.0 - h[k:])
        g[k:] -= wk * h[:upper]
    return g


def gradient_descent_softmax(h0, rng, betas, steps_per_beta=60, lr_start=0.5):
    """Continuation on β with projected gradient descent."""
    h = np.clip(h0.copy(), 0.0, 1.0)
    n = h.size
    target = n / 2.0

    # enforce exact sum once
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(200):
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                if add > 0:
                    h[i] += add
                    diff -= add
            else:
                sub = max(diff, -h[i])
                if sub < 0:
                    h[i] += sub
                    diff -= sub
            if abs(diff) < 1e-12:
                break
        h = np.clip(h, 0.0, 1.0)

    best_c = compute_c5(h)

    for beta in betas:
        lr = lr_start
        for _ in range(steps_per_beta):
            g = softmax_gradient(h, beta)
            if np.allclose(g, 0):
                continue
            h_new = h - lr * g
            h_new = np.clip(h_new, 0.0, 1.0)

            diff = target - h_new.sum()
            if abs(diff) > 1e-12:
                for __ in range(200):
                    i = rng.integers(0, n)
                    if diff > 0:
                        add = min(diff, 1.0 - h_new[i])
                        if add > 0:

                    else:
                        sub = max(diff, -h_new[i])
                        if sub < 0:

                h_up = np.clip(h_up, 0.0, 1.0)
            # refine up‑sampled version
            h_up = gradient_descent_softmax(h_up, rng, betas[:4], steps_per_beta=40, lr_start=0.5)
            h_up, c_up = hill_climb_simple(h_up, rng, max_iters=600, start_step=0.2)
            if c_up < best_c:
                best_h, best_c, best_n = h_up.copy(), c_up, n_up

    # final safety checks
    if best_h is None:
        best_n = 180
        best_h = np.full(best_n, 0.5)

    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            i = rng.integers(0, best_n)
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()
            if abs(diff) < 1e-12:
                break

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
