```
def enforce_sum(h, target, rng):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter=10000
    it=0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
            if idx.size == 0: break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i])
            # Add some fraction to avoid overshoot
            add *= rng.random()
            h[i] += add
            diff -= add
        else: # diff<0
            idx = np.where(h > 0.0)[0]
            if idx.size == 0: break
            i = rng.choice(idx)
            sub = min(-diff, h[i])
            sub *= rng.random()
            h[i] -= sub
            diff += sub
        it+=1
    # final adjust if needed
    if abs(diff) > 1e-12:
        # adjust a random index directly
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h
```

def hill_climb_targeted(h_start, rng, max_iters=2000, start_step=0.5, target_sum=None):
    h = h_start.copy()
    n = h.size
    best_h = h.copy()
    corr = np.correlate(h, 1 - h, mode="full")
    best_c = np.max(corr) * (2.0 / n)

    step = start_step
    decay_interval = max(1, max_iters // 10)

    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr)
        max_corr = corr[idx]
        cur_c = max_corr * (2.0 / n)
        k = idx - (n - 1)
        # gradient
        grad = np.ones(n)
        if k >= 0:
            grad[:n - k] -= h[k:]
            grad[k:] -= h[:n - k]
        else:
            kk = -k
            grad[:n - kk] -= h[kk:]
            grad[kk:] -= h[:n - kk]

        new_h = perturb_targeted(h, grad, k, rng, scale=step)
        # new_h preserves sum, but may have tiny floating errors; enforce
        new_h = enforce_sum(new_h, target_sum if target_sum is not None else n/2.0, rng)
        corr_new = np.correlate(new_h, 1 - new_h, mode="full")
        new_c = np.max(corr_new) * (2.0 / n)

        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
        if new_c < cur_c:
            h = new_h

        if (it + 1) % decay_interval == 0:
            step = max(step * 0.85, 1e-6)
    return best_h, best_c

def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET
def _softmax_obj(x):
    corr = np.correlate(x, 1-x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # compute log-sumexp in a stable way
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
def slsqp_optimize(h0, beta, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = opt.minimize(_softmax_obj, h0, method="SLSQP", bounds=bounds,
                       constraints=cons,
                       options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # repair sum
    rng = np.random.default_rng()
    x = enforce_sum(x, CURRENT_TARGET, rng)
    return x

    return np.sum(x) - CURRENT_TARGET

def _softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def enforce_sum(h, target, rng):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    it = 0
    while abs(diff) > 1e-12 and it < 10_000:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
            if not idx.size:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i]) * rng.random()
            h[i] += add
            diff -= add
        else:
            idx = np.where(h > 0.0)[0]
            if not idx.size:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i]) * rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def perturb_random(h, rng, scale=0.2):

    nh[j] += eps

    # up‑sample to 720 points
    n0 = h0.size
    n = n0 * 2
    xs0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.interp(xs, xs0, h0)
    h = enforce_sum(np.clip(h, 0.0, 1.0), n / 2.0, rng)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    beta_schedule = [10.0, 50.0, 200.0, 800.0, 3000.0]

    # ---------- intensive hill‑climbing ----------
    while time.time() - start_time < budget_s * 0.65:
        if rng.random() < 0.7:
            start = perturb_random(best_h, rng, scale=0.3)
        else:
            start = random_feasible_vector(n, rng)

        h_loc, c_loc = hill_climb_targeted(start, rng, max_iters=3000,
                                           start_step=0.6, target_sum=n / 2.0)
        if c_loc < best_c:
            best_h, best_c = h_loc.copy(), c_loc

        # SLSQP refinements
        for beta in beta_schedule:
            if time.time() - start_time > budget_s * 0.95:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, max_iter=500)
            except Exception:
                continue
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_h, best_c = h_ref.copy(), c_ref

    # ---------- binary rounding ----------
    target_ones = n // 2
    order = np.argsort(-best_h)
    bin_h = np.zeros(n, dtype=float)
    bin_h[order[:target_ones]] = 1.0
    c_bin = compute_c5(bin_h)
    if c_bin < best_c:
        best_h, best_c = bin_h, c_bin

    # ---------- simulated annealing on binary ----------
    cur_h = best_h.copy()
    cur_c = best_c
    best_bin_h = cur_h.copy()
    best_bin_c = cur_c
    T = 0.001
    cooling = 0.9995

    while time.time() - start_time < budget_s * 0.99:
        ones = np.flatnonzero(cur_h > 0.5)
        zeros = np.flatnonzero(cur_h < 0.5)
        if not ones.size or not zeros.size:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        nh = cur_h.copy()
        nh[i] = 0.0
        nh[j] = 1.0
        new_c = compute_c5(nh)
        delta = new_c - cur_c
        if delta <= 0 or rng.random() < math.exp(-delta / T):
            cur_h, cur_c = nh, new_c
            if cur_c < best_bin_c:
                best_bin_c, best_bin_h = cur_c, cur_h.copy()
        T = max(T * cooling, 1e-8)

    if best_bin_c < best_c:
        best_h, best_c = best_bin_h, best_bin_c

    # ---------- final micro‑search ----------
    while time.time() - start_time < budget_s * 0.999:
        nh = perturb_random(best_h, rng, scale=0.02)
        c_new = compute_c5(nh)
        if c_new < best_c:
            best_h, best_c = nh, c_new

    best_h = enforce_sum(np.clip(best_h, 0.0, 1.0), best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
