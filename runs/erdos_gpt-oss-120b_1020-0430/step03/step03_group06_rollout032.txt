def project_onto_bounds_and_sum(v, target, max_iter=100):
    # Clip to [0,1]
    v = np.clip(v, 0, 1)
    # Compute deficit/excess
    diff = v.sum() - target
    # If diff close zero, done.
    # If diff > 0 (sum too large), we need to decrease some entries.
    # we can subtract diff/N from all entries; but that may break bounds, so iterate.
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        # Compute free indices where not at bound in direction of adjusting
        if diff > 0:
            free = v > 0
        else:
            free = v < 1
        n_free = free.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[free] -= adjust
        v = np.clip(v, 0, 1)
        diff = v.sum() - target
    return v
```

This simple water-filling will converge because each iteration reduces diff and respects bounds.

    # Clip to [0,1]
    v = np.clip(v, 0.0, 1.0)
    # Compute current sum
    cur = v.sum()
    if abs(cur - target) < 1e-12:
        return v
    # Use water-filling: adjust inside allowable range
    # We can use lambda to solve for threshold tau such that sum(min(max(v_i + tau, 0), 1)) = target
    # This is a 1D root-finding problem, as adjusting by same tau to all entries before clipping.
    # The final solution is v_i' = clip(v_i + tau, 0, 1).
    # The sum is monotonic in tau.
    # We can binary search tau in range [-1, 1] to meet target.
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        v_mid = np.clip(v + mid, 0.0, 1.0)
        s_mid = v_mid.sum()
        if s_mid > target:
            hi = mid
        else:
            lo = mid
    v = np.clip(v + (lo + hi) / 2, 0.0, 1.0)
    return v

def smooth_objective_and_grad(h, beta):
    n = h.size
    a = 1 - h
    C = np.correlate(h, a, mode='full')
    # compute max evaluation later...
    # compute weights
    # We may want to subtract max for stability
    maxC = np.max(C)
    expC = np.exp(beta * (C - maxC))
    w = expC / expC.sum()
    # gradient initialization
    grad = np.zeros_like(h)
    # loops over shift indices
    for s in range(- (n-1), n):
        w_s = w[s + (n-1)]
        if w_s == 0:
            continue
        shift = s
        if shift >= 0:
            # part1: (1 - h_{j+shift}) -> a[shift:]
            grad[:n-shift] += w_s * a[shift:]
            # part2: -h_{j-shift} -> -h[:n-shift]
            grad[shift:] -= w_s * h[:n-shift]
        else:
            # shift negative: shift = -d
            d = -shift
            # part1: (1 - h_{j+shift}) -> a[:-d] (since j+shift = j-d)
            grad[d:] += w_s * a[:n-d]
            # part2: -h_{j-shift} -> -h[d:] (j - shift = j + d)
            grad[:n-d] -= w_s * h[d:]
    # gradient of (logsumexp)/beta = sum_s w_s * dC_s/dh
    # plus if we want derivative of objective w.r.t. beta? not needed.
    return (maxC + np.log(np.exp(beta * (C - maxC)).sum())) / beta, grad
```

import time
import numpy as np

# --------------------------------------------------------------
# Core utilities
# --------------------------------------------------------------
def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / h.size))


def project_bounds_sum(v, target, tol=1e-12, max_iter=60):
    """Project v onto the box [0,1]^n while enforcing sum(v)=target."""
    v = np.asarray(v, dtype=float)
    lo, hi = -1.0, 1.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        vm = np.clip(v + mid, 0.0, 1.0)
        s = vm.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    v = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny corrective sweep if needed
    diff = v.sum() - target
    if abs(diff) > tol:
        free = (v > 0.0) & (v < 1.0)
        nfree = free.sum()
        if nfree:
            v[free] -= diff / nfree
            v = np.clip(v, 0.0, 1.0)
    return v


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    v = rng.random(n)
    return project_bounds_sum(v, n / 2.0)


def perturb(h, rng, magnitude=0.2):
    """Add small random noise to a subset of coordinates, then repair."""
    n = h.size
    k = max(1, int(0.05 * n))
    idx = rng.choice(n, size=k, replace=False)
    noise = (rng.random(k) - 0.5) * 2.0 * magnitude
    h2 = h.copy()
    h2[idx] += noise
    return project_bounds_sum(h2, n / 2.0)


# --------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient
# --------------------------------------------------------------
def smooth_objective_and_grad(h, beta):
    """Return (f, grad) for f = (1/β)·log∑exp(β·C) where C is overlap vector."""
    a = 1.0 - h
    C = np.correlate(h, a, mode="full")
    maxC = C.max()
    expC = np.exp(beta * (C - maxC))
    sumExp = expC.sum()
    w = expC / sumExp                     # softmax weights, sum to 1
    f = (maxC + np.log(sumExp)) / beta    # stable log‑sum‑exp
    n = h.size
    grad = np.zeros_like(h)

    # accumulate weighted gradients for each shift
    for idx, ws in enumerate(w):
        if ws == 0.0:
            continue
        shift = idx - (n - 1)            # true shift value
        if shift >= 0:
            grad[: n - shift] += ws * a[shift:]
            grad[shift:]      -= ws * h[: n - shift]
        else:
            d = -shift
            grad[d:]          += ws * a[: n - d]
            grad[: n - d]     -= ws * h[d:]
    return f, grad


# --------------------------------------------------------------
# Projected gradient descent on the smooth surrogate
# --------------------------------------------------------------
def gradient_descent(h0, rng, start_time, budget_s,
                    max_iters=4000,
                    beta_start=5.0, beta_end=800.0,
                    lr_start=0.3, lr_end=0.001):
    """Run a projected gradient descent with an increasing β schedule."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)

    for it in range(max_iters):
        if time.time() - start_time > budget_s * 0.97:
            break
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t
        lr = lr_start * (lr_end / lr_start) ** t

        _, grad = smooth_objective_and_grad(h, beta)
        h = h - lr * grad
        h = project_bounds_sum(h, target)

        if it % 25 == 0:
            c = compute_c5(h)
            if c < best_c:
                best_c, best_h = c, h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----------------------------------------------------------
    # Determine dimension and build initial candidates
    # ----------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_bounds_sum(h0, n / 2.0)
    else:
        n = 180
        h0 = random_feasible_vector(n, rng)

    target = n / 2.0
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx

    candidates = [h0]

    for f in range(1, 4):
        for fun in (np.sin, np.cos):
            h = 0.5 + 0.5 * fun(np.pi * f * xs)
            h = np.clip(h, 0.0, 1.0)
            candidates.append(project_bounds_sum(h, target))

    # linear ramps
    candidates.append(project_bounds_sum(xs / 2.0, target))
    candidates.append(project_bounds_sum(1.0 - xs / 2.0, target))

    for _ in range(5):
        candidates.append(random_feasible_vector(n, rng))

    best_h = None
    best_c = float("inf")
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # ----------------------------------------------------------
    # Stochastic restarts with gradient descent
    # ----------------------------------------------------------
    while time.time() - start_time < budget_s * 0.85:
        if best_h is not None and rng.random() < 0.6:
            start = perturb(best_h, rng, magnitude=0.15)
        else:
            start = random_feasible_vector(n, rng)

        h_opt, c_opt = gradient_descent(start, rng, start_time, budget_s,
                                        max_iters=5000,
                                        beta_start=5.0, beta_end=800.0,
                                        lr_start=0.3, lr_end=0.001)

        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()

    # ----------------------------------------------------------
    # Optional up‑sampling and refinement (if time permits)
    # ----------------------------------------------------------
    if time.time() - start_time < budget_s * 0.94 and n * 2 <= 360:
        h_up = np.repeat(best_h, 2)
        n_up = h_up.size
        # tiny random kick to unlock new degrees of freedom
        h_up = perturb(h_up, rng, magnitude=0.02)
        h_up_opt, c_up_opt = gradient_descent(h_up, rng, start_time, budget_s,
                                              max_iters=5000,
                                              beta_start=10.0, beta_end=1200.0,
                                              lr_start=0.2, lr_end=0.0005)
        if c_up_opt < best_c:
            best_c, best_h, n = c_up_opt, h_up_opt.copy(), n_up

    # ----------------------------------------------------------
    # Final clean‑up
    # ----------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_bounds_sum(best_h, n / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
