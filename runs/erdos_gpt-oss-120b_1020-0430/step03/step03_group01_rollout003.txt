def enforce_sum(h, target, rng, max_iter=1000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        # find indices that can be increased (if diff>0) or decreased (if diff<0)
        if diff > 0:
            mask = h < 1.0
            count = np.count_nonzero(mask)
            if count == 0: break
            delta = diff / count
            # cap each increase to not exceed 1
            inc = np.minimum(delta, 1.0 - h[mask])
            h[mask] += inc
        else:
            mask = h > 0.0
            count = np.count_nonzero(mask)
            if count == 0: break
            delta = diff / count
            dec = np.maximum(delta, -h[mask])
            h[mask] += dec
        diff = target - h.sum()
    # Final correction if any small residual
    if abs(diff) > 1e-12:
        # distribute arbitrarily across all entries
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_obj(x):
    global CURRENT_BETA
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # use log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum exact
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # adjust uniformly then clip
        x += diff / n
        x = np.clip(x, 0.0, 1.0)
        diff = target - x.sum()
        if abs(diff) > 1e-12:
            # small correction across all
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    add = min(diff, 1.0 - x[i])
                else:
                    add = max(diff, -x[i])
                x[i] += add
                diff = target - x.sum()

import time, numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i(1-x_{i+k}) with current beta."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def compute_c5(h):
    """Exact discrete C5 using FFT for speed."""
    n = h.shape[0]
    L = 2 * n - 1
    H = np.fft.fft(h, n=L)
    G = np.fft.fft((1 - h)[::-1], n=L)
    conv = np.fft.ifft(H * G).real
    return np.max(conv) * (2.0 / n)

def enforce_sum(h, target, rng, max_iter=500):
    """Project h onto [0,1] with Σh = target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            delta = diff / cnt
            inc = np.minimum(delta, 1.0 - h[mask])
            h[mask] += inc
        else:
            mask = h > 0.0
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            delta = diff / cnt
            dec = np.maximum(delta, -h[mask])
            h[mask] += dec
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)

def perturb(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def hill_climb_multi(h_start, rng, max_iters=2000, init_step=0.5, n_trials=5):
    """Multi‑trial hill‑climb using ε‑transfers."""
    h = h_start.copy()
    best_val = compute_c5(h)
    step = init_step
    n = h.size
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0:
                continue
            eps = (rng.random() - 0.5) * 2.0 * max_eps * step
            nh = h.copy()
            nh[i] -= eps
            nh[j] += eps
            val = compute_c5(nh)
            if val < best_val:
                h, best_val = nh, val
                improved = True
                break
        if not improved and rng.random() < 0.02:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                max_eps = min(h[i], 1.0 - h[j])
                if max_eps > 0:
                    eps = (rng.random() - 0.5) * 2.0 * max_eps * step * 2.0
                    nh = h.copy()
                    nh[i] -= eps
                    nh[j] += eps
                    val = compute_c5(nh)
                    if val < best_val:
                        h, best_val = nh, val
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)
    return h, best_val

def slsqp_optimize(h0, beta, max_iter=300):
    """Refine using smooth surrogate and SLSQP."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum (tiny drift)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x += diff / n
        x = np.clip(x, 0.0, 1.0)
        diff = CURRENT_TARGET - x.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
                x[i] += add
                diff = CURRENT_TARGET - x.sum()
    return x

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    # ---- initial vector -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 180
        h0 = random_feasible_vector(n, rng)

    best_h = h0
    best_c = compute_c5(best_h)

    # ---- main optimisation (≈75% of time) -----------------------------
    main_budget = budget_s * 0.75
    while time.time() - t0 < main_budget:
        start_vec = perturb(best_h, rng, scale=0.4) if rng.random() < 0.7 else random_feasible_vector(n, rng)
        h_opt, c_opt = hill_climb_multi(start_vec, rng, max_iters=3000, init_step=0.5, n_trials=6)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

        for beta in (5, 20, 80, 200, 500, 1000, 2000):
            if time.time() - t0 > main_budget:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, max_iter=400)
                c_ref = compute_c5(h_ref)
                if c_ref < best_c:
                    best_h, best_c = h_ref, c_ref
            except Exception:
                pass

    # ---- up‑sampling & fine tuning (if time left) ----------------------
    elapsed = time.time() - t0
    remaining = budget_s - elapsed
    if remaining > budget_s * 0.15:
        target_n = min(720, max(360, n * 2))
        if target_n != n:
            old_x = (np.arange(n) + 0.5) * (2.0 / n)
            new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
            h_up = np.interp(new_x, old_x, best_h)
            h_up = enforce_sum(h_up, target_n / 2.0, rng)

            up_start = time.time()
            up_budget = remaining * 0.85
            while time.time() - up_start < up_budget * 0.9:
                start_vec = perturb(h_up, rng, scale=0.3) if rng.random() < 0.6 else random_feasible_vector(target_n, rng)
                h_opt, c_opt = hill_climb_multi(start_vec, rng, max_iters=2500, init_step=0.4, n_trials=6)
                if c_opt < best_c:
                    best_h, best_c, n = h_opt, c_opt, target_n
                for beta in (5, 20, 80, 200, 500, 1000, 2000):
                    if time.time() - up_start > up_budget:
                        break
                    try:
                        h_ref = slsqp_optimize(best_h, beta, max_iter=400)
                        c_ref = compute_c5(h_ref)
                        if c_ref < best_c:
                            best_h, best_c = h_ref, c_ref
                    except Exception:
                        pass
            # ensure we use the latest n
            if n != target_n:
                n = target_n

    # ---- final clean‑up ------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
