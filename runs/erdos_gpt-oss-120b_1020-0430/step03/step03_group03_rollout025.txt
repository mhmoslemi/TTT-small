def simple_sa(h, rng, max_iter, init_temp, final_temp):
    n = len(h)
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_h = h.copy()
    cur_c = best_c
    # precompute gradient? Not needed
    for t in range(1, max_iter+1):
        T = init_temp * ((final_temp / init_temp) ** (t / max_iter))
        # propose: transfer epsilon between two random indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # compute max epsilon allowed
        max_eps = min(cur_h[i], 1 - cur_h[j])
        if max_eps <= 0:
            continue
        # choose epsilon scaled by temp factor
        eps = (rng.random() - 0.5) * 2 * max_eps * (T / init_temp) # maybe just random scale
        # propose new
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        # compute c5
        new_c = compute_c5(new_h)
        diff = new_c - cur_c
        if diff < 0 or rng.random() < math.exp(-diff / max(T, 1e-12)):
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
    return best_h, best_c

import time, math, numpy as np

# ---------- helper functions ----------
def compute_c5(h):
    """Discrete C5 for step vector h on [0,2] (dx = 2/n)."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)


def enforce_sum(h, target, rng):
    """Adjust h in‑place so that sum(h)=target while respecting 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random feasible h with 0≤h≤1 and sum=n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.5
        h[i] -= eps
        h[j] += eps
    enforce_sum(h, n / 2.0, rng)
    return h


def gradient_descent(h, max_iters=200, tol=1e-12):
    """Projected gradient descent on the worst‑shift objective."""
    n = h.size
    target = n / 2.0
    for _ in range(max_iters):
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = corr.max()
        c5 = max_corr * (2.0 / n)

        # shifts attaining the maximum (within a tiny tolerance)
        mask = np.abs(corr - max_corr) <= 1e-12
        shifts = np.where(mask)[0] - (n - 1)
        if shifts.size == 0:
            break

        # gradient of the worst shift(s)
        g = np.zeros(n)
        idx = np.arange(n)
        for s in shifts:
            # h_{i+s}
            idx_p = idx + s
            ok_p = (idx_p >= 0) & (idx_p < n)
            h_plus = np.zeros(n)
            h_plus[ok_p] = h[idx_p[ok_p]]

            # h_{i-s}
            idx_m = idx - s
            ok_m = (idx_m >= 0) & (idx_m < n)
            h_minus = np.zeros(n)
            h_minus[ok_m] = h[idx_m[ok_m]]

            g += 1.0 - (h_plus + h_minus)
        g /= len(shifts)

        # descent direction (sum‑zero)
        direction = -g
        direction -= direction.mean()
        if np.linalg.norm(direction) < tol:
            break

        # maximal step keeping 0≤h≤1
        pos = direction > 1e-12
        neg = direction < -1e-12
        tau_pos = np.min((1.0 - h[pos]) / direction[pos]) if np.any(pos) else np.inf
        tau_neg = np.min((-h[neg]) / direction[neg]) if np.any(neg) else np.inf
        tau = min(tau_pos, tau_neg)
        if not np.isfinite(tau) or tau <= 0:
            break
        tau *= 0.95  # stay strictly interior

        h_new = h + tau * direction
        h_new = np.clip(h_new, 0.0, 1.0)

        new_corr = np.correlate(h_new, 1 - h_new, mode="full")
        new_c5 = new_corr.max() * (2.0 / n)

        if new_c5 + 1e-12 < c5:
            h = h_new
            continue
        break

    enforce_sum(h, target, np.random.default_rng())
    return h, compute_c5(h)


def simple_sa(h0, rng, max_iter, init_temp, final_temp):
    """Lightweight simulated annealing on C5."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    n = h.size
    for it in range(1, max_iter + 1):
        T = init_temp * ((final_temp / init_temp) ** (it / max_iter))
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        # scale epsilon with temperature
        eps = (rng.random() - 0.5) * 2 * max_eps * (T / init_temp)
        cand = h.copy()
        cand[i] -= eps
        cand[j] += eps
        cand_c = compute_c5(cand)
        diff = cand_c - cur_c
        if diff < 0 or rng.random() < math.exp(-diff / max(T, 1e-12)):
            h, cur_c = cand, cand_c
            if cand_c < best_c:
                best_c, best_h = cand_c, cand.copy()
    return best_h, best_c
