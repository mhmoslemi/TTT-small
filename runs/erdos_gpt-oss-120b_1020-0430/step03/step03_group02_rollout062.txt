```
def worst_gradient(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr) - (n-1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad
```

def random_feasible_vector(n, rng):
    # generate uniform random in [0,1] then scale to target sum
    h = rng.random(n)
    target = n / 2
    h = h * target / h.sum()
    # clip and adjust diff
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # fix diff iteratively
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h
```

def annealing_search(h0, rng, max_iter, init_temp, final_temp):
    n = h0.size
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    T = init_temp
    if max_iter <=0:
        return best_h, best_c
    factor = (final_temp / init_temp) ** (1.0 / max_iter)
    for it in range(max_iter):
        # choose indices
        if rng.random() < 0.7:
            grad = worst_gradient(cur_h)
            i_pos = int(np.argmax(grad))
            i_neg = int(np.argmin(grad))
        else:
            i_pos = rng.integers(0, n)
            i_neg = rng.integers(0, n)
        max_transfer = min(cur_h[i_pos], 1.0 - cur_h[i_neg])
        if max_transfer <= 0:
            continue
        eps = rng.random() * max_transfer
        new_h = cur_h.copy()
        new_h[i_pos] -= eps
        new_h[i_neg] += eps
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
        T *= factor
    return best_h, best_c

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # compute softmax average
    # use log-sum-exp stable
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
```

def slsqp_optimize(h0, beta, rng, max_iter=200):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum exactly via redistribution
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

import time, numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# helper functions
# ------------------------------------------------------------
def compute_c5(h):
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, rng):
    """Adjust vector to satisfy Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Random h ∈[0,1]^n with Σh = n/2."""
    h = rng.random(n)
    target = n / 2.0
    h = h * target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h):
    """Gradient of overlap O_k for the shift achieving max_k."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def local_search(h0, rng, max_iter=3000, init_step=0.5, start_time=None, budget=None):
    """Monotone descent using the worst‑shift gradient."""
    n = h0.size
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    step = init_step
    for it in range(max_iter):
        if start_time is not None and time.time() - start_time > budget * 0.99:
            break
        grad = worst_gradient(h)
        i_pos = int(np.argmax(grad))
        i_neg = int(np.argmin(grad))
        max_transfer = min(h[i_pos], 1.0 - h[i_neg])
        if max_transfer <= 0:
            break
        eps = step * max_transfer
        h_new = h.copy()
        h_new[i_pos] -= eps
        h_new[i_neg] += eps
        c_new = compute_c5(h_new)
        if c_new < cur_c:
            h, cur_c = h_new, c_new
            if c_new < best_c:
                best_h, best_c = h_new.copy(), c_new
        if (it + 1) % 200 == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


def annealing_search(h0, rng, max_iter=2000, init_temp=0.005, final_temp=1e-7):
    """Simulated annealing on the transfer‑move neighbourhood."""
    n = h0.size
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    T = init_temp
    if max_iter <= 0:
        return best_h, best_c
    factor = (final_temp / init_temp) ** (1.0 / max_iter)
    for _ in range(max_iter):
        if rng.random() < 0.7:
            grad = worst_gradient(cur_h)
            i_pos = int(np.argmax(grad))
            i_neg = int(np.argmin(grad))
        else:
            i_pos = rng.integers(0, n)
            i_neg = rng.integers(0, n)
        max_transfer = min(cur_h[i_pos], 1.0 - cur_h[i_neg])
        if max_transfer <= 0:
            continue
        eps = rng.random() * max_transfer
        new_h = cur_h.copy()
        new_h[i_pos] -= eps
        new_h[i_neg] += eps
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
        T *= factor
    return best_h, best_c


# ------------------------------------------------------------
# SLSQP soft‑max refinement (mirrors the parent code)
# ------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ------------------------------------------------------------
# main entry point
# ------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    budget = budget_s

    best_h = None
    best_c = np.inf
    best_n = None

    # ------------------------------------------------------------------
    # use parent construction if present
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0, c0, h0.size

    # ------------------------------------------------------------------
    # candidate discretisations (larger n are allowed)
    # ------------------------------------------------------------------
    candidate_ns = [84, 96, 108, 120, 144, 168, 192, 240, 384]
    if best_n is not None:
        candidate_ns = [best_n] + [n for n in candidate_ns if n != best_n]

    for n in candidate_ns:
        if time.time() - start_time > 0.93 * budget:
            break

        restarts = 4 if n >= 240 else 3
        for r in range(restarts):
            # ------------------------------------------------------------
            # initialise h for this n
            # ------------------------------------------------------------
            if best_h is not None and best_n == n and r == 0:
                h = best_h.copy()
            elif best_h is not None and best_n is not None and n != best_n and r == 0:
                if n == best_n * 2:
                    h = np.repeat(best_h, 2)
                else:
                    x_old = np.linspace(0, 2, best_n, endpoint=False)
                    x_new = np.linspace(0, 2, n, endpoint=False)
                    h = np.interp(x_new, x_old, best_h)
                h = np.clip(h, 0.0, 1.0)
                h = enforce_sum(h, rng)
            else:
                h = random_feasible_vector(n, rng)

            # ------------------------------------------------------------
            # gradient‑directed local descent
            # ------------------------------------------------------------
            h_opt, c_opt = local_search(
                h,
                rng,
                max_iter=3000,
                init_step=0.5,
                start_time=start_time,
                budget=budget,
            )
            if c_opt < best_c:
                best_h, best_c, best_n = h_opt.copy(), c_opt, n

            # ------------------------------------------------------------
            # short annealing phase (if time permits)
            # ------------------------------------------------------------
            if time.time() - start_time < 0.90 * budget:
                h_ann, c_ann = annealing_search(
                    best_h, rng, max_iter=2000, init_temp=0.005, final_temp=1e-7
                )
                if c_ann < best_c:
                    best_h, best_c, best_n = h_ann.copy(), c_ann, n

    # ------------------------------------------------------------------
    # final SLSQP refinement on the best found construction
    # ------------------------------------------------------------------
    if best_h is not None and time.time() - start_time < 0.96 * budget:
        for beta in (20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0):
            if time.time() - start_time > 0.98 * budget:
                break
            best_h = slsqp_optimize(best_h, beta, rng, max_iter=300)
            best_c = compute_c5(best_h)

    # ------------------------------------------------------------------
    # final light hill‑climb using very small transfers
    # ------------------------------------------------------------------
    while time.time() - start_time < budget * 0.999 and best_h is not None:
        grad = worst_gradient(best_h)
        i_pos = int(np.argmax(grad))
        i_neg = int(np.argmin(grad))
        max_transfer = min(best_h[i_pos], 1.0 - best_h[i_neg])
        if max_transfer > 0:
            eps = max_transfer * 0.01
            nh = best_h.copy()
            nh[i_pos] -= eps
            nh[i_neg] += eps
            nc = compute_c5(nh)
            if nc < best_c:
                best_h, best_c = nh.copy(), nc

    # ------------------------------------------------------------------
    # final feasibility enforcement and return
    # ------------------------------------------------------------------
    if best_h is None:
        best_n = 84
        best_h = np.full(best_n, 0.5)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
