def enforce_sum(h, target, rng):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:  # need to increase some entries
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i])
            add *= rng.random()
            h[i] += add
            diff -= add
        else:  # need to decrease
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i])
            sub *= rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        if diff > 0:
            h[i] = min(1.0, h[i] + diff)
        else:
            h[i] = max(0.0, h[i] + diff)
    return h

import time
import numpy as np
from scipy.optimize import differential_evolution, LinearConstraint

# ---------- helpers ----------
def enforce_sum(h, target, rng):
    """Clip to [0,1] and adjust to make sum exactly target (within 1e-12)."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:                       # need to increase some entries
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i])
            add *= rng.random()
            h[i] += add
            diff -= add
        else:                              # need to decrease some entries
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i])
            sub *= rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:                  # fallback correction
        i = rng.integers(0, h.size)
        if diff > 0:
            h[i] = min(1.0, h[i] + diff)
        else:
            h[i] = max(0.0, h[i] + diff)
    return h

def random_feasible_vector(n, rng):
    h = rng.random(n)
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, n / 2.0, rng)
    return h

def compute_c5(h):
    """Exact discrete C₅ for a given step vector."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

# ---------- main ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # ----- low‑resolution stage (n = 180) -----
    n_low = 180
    target_low = n_low / 2.0

    # try to obtain a low‑resolution start from a possible parent construction
    h0_low = None
    if "initial_h_values" in globals():
        init_h = np.array(initial_h_values, dtype=float)
        if init_h.size == 360:                     # down‑sample by averaging pairs
            h0_low = (init_h[::2] + init_h[1::2]) * 0.5
            h0_low = enforce_sum(np.clip(h0_low, 0.0, 1.0), target_low, rng)
        elif init_h.size == n_low:
            h0_low = enforce_sum(np.clip(init_h, 0.0, 1.0), target_low, rng)

    # objective for DE (uses exact C₅)
    def obj_low(x):
        return compute_c5(x)

    bounds_low = [(0.0, 1.0)] * n_low
    lin_con_low = LinearConstraint(np.ones((1, n_low)), [target_low], [target_low])

    # callback to respect half of the total budget
    def callback_low(xk, convergence):
        if time.time() - start_time > budget_s * 0.5:
            return True
        return False

    # run DE on the low‑resolution problem
    result_low = differential_evolution(
        obj_low,
        bounds_low,
        constraints=(lin_con_low,),
        strategy="best1bin",
        maxiter=3000,
        popsize=5,
        mutation=(0.5, 1.0),
        recombination=0.7,
        seed=seed,
        updating="deferred",
        polish=False,
        callback=callback_low,
        init="latinhypercube",
    )
    h_low = result_low.x
    h_low = np.clip(h_low, 0.0, 1.0)
    h_low = enforce_sum(h_low, target_low, rng)

    # ----- up‑sample to high resolution (n = 360) -----
    n_high = n_low * 2
    target_high = n_high / 2.0
    xs_low = (np.arange(n_low) + 0.5) * (2.0 / n_low)
    xs_high = (np.arange(n_high) + 0.5) * (2.0 / n_high)
    h_high = np.interp(xs_high, xs_low, h_low)
    h_high = np.clip(h_high, 0.0, 1.0)
    h_high = enforce_sum(h_high, target_high, rng)

    best_h = h_high.copy()
    best_c = compute_c5(best_h)

    # ----- local refinement (random mass transfers) -----
    while time.time() - start_time < budget_s * 0.96:

        j = rng.integers(0, n_high)
        if i == j:
            continue
        if best_h[i] <= 1e-12 or best_h[j] >= 1.0 - 1e-12:
            continue
        max_eps = min(best_h[i], 1.0 - best_h[j])
        eps = max_eps * 0.1 * rng.random()
        new_h = best_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c_new = compute_c5(new_h)
        if c_new < best_c:

        # occasional larger random re‑initialisation of a small fraction
        if rng.random() < 0.01:

            if mask.any():

                if c_new2 < best_c:

    best_h = enforce_sum(best_h, target_high, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n_high)
