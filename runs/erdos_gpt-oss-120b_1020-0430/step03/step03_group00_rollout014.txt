def anneal_optimize(h0, rng, target_sum, time_budget):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    beta_list = [1.0, 5.0, 20.0, 80.0, 200.0, 800.0]
    for beta in beta_list:
        # step size based on beta
        step = 0.5 if beta <= 20 else 0.2
        # max its per beta approximate constant
        max_iter = 2000
        for it in range(max_iter):
            # time check
            if time.time() - start > time_budget:
                return best_h, best_c
            corr = np.correlate(h, 1 - h, mode="full")
            max_corr = np.max(corr)
            # compute softmax weights
            # to avoid overflow
            shifted = corr - max_corr
            exp_vals = np.exp(beta * shifted)
            sum_exp = np.sum(exp_vals)
            if sum_exp == 0:
                w = np.full_like(exp_vals, 1.0 / exp_vals.size)
            else:
                w = exp_vals / sum_exp
            n = h.size
            grad = np.zeros_like(h)
            # Compute weighted gradient
            for idx, wi in enumerate(w):
                if wi == 0:
                    continue
                k = idx - (n - 1)
                if k >= 0:
                    grad[:n - k] += wi * (1 - h[k:])
                    grad[k:] -= wi * h[:n - k]
                else:
                    kk = -k
                    grad[:n - kk] += wi * (1 - h[kk:])
                    grad[kk:] -= wi * h[:n - kk]
            # take step
            new_h = h - step * grad
            # enforce constraints
            new_h = np.clip(new_h, 0.0, 1.0)
            new_h = enforce_sum(new_h, target_sum, rng)
            # compute C5

import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def enforce_sum(h, target, rng):
    """Clip to [0,1] and adjust to make sum exactly target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:  # need to increase some entries
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i])
            add *= rng.random()
            h[i] += add
            diff -= add
        else:  # need to decrease
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i])
            sub *= rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        if diff > 0:
            h[i] = min(1.0, h[i] + diff)
        else:
            h[i] = max(0.0, h[i] + diff)
    return h

def random_feasible_vector(n, rng):
    """Uniform random start repaired to satisfy Σh = n/2."""
    h = rng.random(n)
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)

def perturb_random(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def compute_c5(h):
    """Discrete C5 = (2/n) * max correlation."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

# ---------- SLSQP refinement (softmax surrogate) ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return enforce_sum(x, CURRENT_TARGET, np.random.default_rng())

# ---------- Targeted hill climbing (used for polishing) ----------
def hill_climb_targeted(h_start, rng, max_iters, start_step, target_sum):
    h = h_start.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(best_h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr)
        max_corr = corr[idx]
        c_current = max_corr * (2.0 / n)
        k = idx - (n - 1)

        # gradient of the worst shift
        grad = np.ones(n)
        if k >= 0:
            grad[:n - k] -= h[k:]
            grad[k:] -= h[:n - k]
        else:
            kk = -k
            grad[:n - kk] -= h[kk:]
            grad[kk:] -= h[:n - kk]

        new_h = perturb_random(h, rng, scale=step)
        # move a bit in the direction suggested by gradient
        # (perturb_random already respects sum, we keep it)
        corr_new = np.correlate(new_h, 1 - new_h, mode="full")
        c_new = np.max(corr_new) * (2.0 / n)

        if c_new < best_c:
            best_c = c_new
            best_h = new_h.copy()
        if c_new < c_current:
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

# ---------- Deterministic annealing (softmax gradient descent) ----------
def anneal_optimize(h0, rng, target_sum, time_budget):
    """Smooth max minimization using softmax gradient descent."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    beta_list = [1.0, 5.0, 20.0, 80.0, 200.0, 800.0]
    n = h.size
    for beta in beta_list:
        # step size adapts with beta
        step = 0.5 if beta <= 20 else 0.2
        max_iter = 2000
        for it in range(max_iter):
            if time.time() - start > time_budget:
                return best_h, best_c
            corr = np.correlate(h, 1 - h, mode="full")
            max_corr = np.max(corr)
            shifted = corr - max_corr
            # softmax weights (stable)
            exp_vals = np.exp(beta * shifted)
            sum_exp = np.sum(exp_vals)
            if sum_exp == 0:
                w = np.full_like(exp_vals, 1.0 / exp_vals.size)
            else:
                w = exp_vals / sum_exp
            grad = np.zeros_like(h)
            # weighted gradient of all shifts
            for idx, wi in enumerate(w):
                if wi < 1e-15:
                    continue
                k = idx - (n - 1)
                if k >= 0:
                    grad[:n - k] += wi * (1 - h[k:])
                    grad[k:] -= wi * h[:n - k]
                else:
                    kk = -k
                    grad[:n - kk] += wi * (1 - h[kk:])
                    grad[kk:] -= wi * h[:n - kk]
            new_h = h - step * grad
            new_h = np.clip(new_h, 0.0, 1.0)
            new_h = enforce_sum(new_h, target_sum, rng)
            c_new = compute_c5(new_h)
            if c_new < best_c:
                best_c = c_new
                best_h = new_h.copy()
            c_curr = compute_c5(h)
            if c_new < c_curr or rng.random() < 0.001:
                h = new_h
            if (it + 1) % 200 == 0:
                step = max(step * 0.7, 1e-4)
        # after each beta, jitter a bit to escape plateaus
        h = perturb_random(best_h, rng, scale=0.1)
        h = enforce_sum(np.clip(h, 0.0, 1.0), target_sum, rng)
    return best_h, best_c

# ---------- main entry ----------
