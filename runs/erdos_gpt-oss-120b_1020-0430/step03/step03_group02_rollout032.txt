def enforce_sum(h, target, rng):
    diff = target - h.sum()
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(len(h))
        if diff > 0:
            # can add up to 1 - h[i]
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

    from scipy.optimize import minimize

    rng = np.random.default_rng(seed)

    best_c = math.inf
    best_n = None

        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        target = h0.size / 2.0

        c0 = compute_c5(h0)
        best_h = h0
        best_c = c0
        best_n = h0.size

    candidate_ns = [192, 256, 224]
    # ensure best_n is first

        candidate_ns = [best_n] + [n for n in candidate_ns if n != best_n]

        if time.time() > end_time:
            break
        target = n / 2.0

            h = best_h.copy()

            new_fit = fitness[:elite_num]

                            bestf = fitness[ci]

import time, math
import numpy as np
from scipy.optimize import minimize

# Utility functions
# -------------------------------------------------------------
def compute_c5(h):
    """Maximum overlap integral for a discrete step function."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng):
    """Adjust h to satisfy sum == target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    for _ in range(10000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add == 0.0:
            continue
        h[i] += add
        diff = target - h.sum()
    np.clip(h, 0.0, 1.0, out=h)
    return h


def random_binary_vector(n, rng):
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h


def worst_gradient(h):
    """Gradient of the worst‑shift overlap O_k = Σ h_i(1‑h_{i+k})."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        return 1.0 - 2.0 * h
    grad = np.ones_like(h)
    grad[: n - k] -= h[k:]
    grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h, grad, rng, scale=0.05):
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h_start, rng, max_iters=2000, step_scale=0.5):
    """Hill‑climb using the worst‑shift gradient."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        if it % decay == 0:
            step_scale = max(step_scale * 0.9, 1e-4)
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0.0:
                eps = rng.random() * max_eps * step_scale
                h[i] -= eps
                h[j] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
    return best_h, best_c

def slsqp_optimize(h0, beta, rng, max_iter=400):
    """Soft‑max SLSQP optimisation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:

            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    end_time = start + budget_s * 0.995  # safety margin

    best_h = None
    best_c = math.inf
    best_n = None

    # -----------------------------------------------------------------
    # initialise from existing construction if present
    # -----------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        target0 = h0.size / 2.0
        h0 = enforce_sum(h0, target0, rng)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0.copy(), c0, h0.size

    # candidate discretisations (order with current best first)
    candidate_ns = [192, 256, 224, 288]
    if best_n is not None:
        candidate_ns = [best_n] + [n for n in candidate_ns if n != best_n]

    # -----------------------------------------------------------------
    # optimisation over each resolution
    # -----------------------------------------------------------------
    for n in candidate_ns:
        if time.time() > end_time:
            break
        target = n / 2.0

        # initialise h for this n
        if best_n == n and best_h is not None:
            h = best_h.copy()
        elif best_h is not None:
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h = np.interp(x_new, x_old, best_h)
            h = np.clip(h, 0.0, 1.0)
            h = enforce_sum(h, target, rng)
        else:
            h = random_binary_vector(n, rng)

        # quick local refinement to obtain a reasonable start
        h, c = guided_hill_climb(h, rng, max_iters=1500, step_scale=0.5)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

        # ---------------------------------------------------------
        # Genetic‑algorithm global search
        # ---------------------------------------------------------
        pop_size = 60
        elite_num = 5
        mutation_rate = 0.9
        mutation_prob_gene = 0.2
        mut_range = 0.2

        # initialise population
        population = [h]
        while len(population) < pop_size:
            population.append(random_binary_vector(n, rng))

        fitness = [compute_c5(ind) for ind in population]

        max_gens = 400
        for gen in range(max_gens):
            if time.time() > end_time:
                break

            # sort by fitness (lower is better)
            order = np.argsort(fitness)
            population = [population[i] for i in order]
            fitness = [fitness[i] for i in order]

            # elite preservation
            new_pop = population[:elite_num]
            new_fit = fitness[:elite_num]

            # offspring creation
            while len(new_pop) < pop_size:
                # tournament selection
                def select():
                    cand = rng.choice(pop_size, size=3, replace=False)
                    best_idx = cand[0]
                    best_val = fitness[best_idx]
                    for ci in cand[1:]:
                        if fitness[ci] < best_val:
                            best_idx, best_val = ci, fitness[ci]
                    return population[best_idx]

                p1 = select()
                p2 = select()

                # blend crossover with extrapolation allowed
                alpha = rng.uniform(-0.5, 1.5)
                child = alpha * p1 + (1.0 - alpha) * p2

                # mutation
                if rng.random() < mutation_rate:
                    mask = rng.random(n) < mutation_prob_gene
                    delta = (rng.random(n) - 0.5) * 2.0 * mut_range
                    child[mask] += delta[mask]

                child = np.clip(child, 0.0, 1.0)
                child = enforce_sum(child, target, rng)

                child_fit = compute_c5(child)

                new_pop.append(child)
                new_fit.append(child_fit)

                if child_fit < best_c:
                    best_h, best_c, best_n = child.copy(), child_fit, n

            population, fitness = new_pop, new_fit

            # modest adaptation of mutation range
            if (gen + 1) % 50 == 0:
                mut_range = max(mut_range * 0.9, 0.02)

        # ---------------------------------------------------------
        # Post‑GA refinement on the current best of this resolution
        # ---------------------------------------------------------
        if best_n == n and best_h is not None:
            refined_h, refined_c = guided_hill_climb(best_h, rng, max_iters=2000, step_scale=0.4)
            if refined_c < best_c:
                best_h, best_c = refined_h.copy(), refined_c

        # ---------------------------------------------------------
        # Soft‑max SLSQP fine tuning (if time permits)
        # ---------------------------------------------------------
        if time.time() < end_time:
            cur_h = best_h if best_n == n else h
            for beta in (5, 20, 80, 200, 500, 1000):
                if time.time() > end_time:
                    break
                cur_h = slsqp_optimize(cur_h, beta, rng, max_iter=400)
                cur_c = compute_c5(cur_h)
                if cur_c < best_c:
                    best_h, best_c, best_n = cur_h.copy(), cur_c, n

        # ---------------------------------------------------------
        # Final short hill climb on this resolution
        # ---------------------------------------------------------
        if time.time() < end_time and best_n == n:
            cur_h, cur_c = guided_hill_climb(best_h, rng, max_iters=1500, step_scale=0.3)
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c

    # -----------------------------------------------------------------
    # Exhaust remaining budget with tiny gradient‑directed perturbations
    # -----------------------------------------------------------------
    while time.time() < end_time and best_h is not None:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # -----------------------------------------------------------------
    # final feasibility enforcement and return
    # -----------------------------------------------------------------
    if best_h is None:
        best_n = 192
        best_h = np.full(best_n, 0.5, dtype=float)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
