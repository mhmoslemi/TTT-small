def get_worst_shift(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    max_val = corr[max_idx]
    shift = max_idx - (n - 1)
    if shift < 0: shift = -shift
    return shift, max_val

def compute_weights_for_shift(h, shift):
    n = len(h)
    k = shift
    weight = np.zeros(n)
    if k == 0:
        weight = 2 * h  # derivative of s_0 = sum h_i^2: d/dh_i = 2 h_i
        # but correlation c_0 = sum h_i - sum h_i^2. Derivative of c_0 wrt h_i = 1 - 2 h_i = -(2h_i - 1)
        # Weight for s_0 is 2h_i. We'll use weight = 2h_i (positive, to increase s_0)
    else:
        # left part: i = 0..n-1-k
        left_indices = np.arange(0, n - k)
        weight[left_indices] += h[left_indices + k]
        # right part: i = k..n-1
        right_indices = np.arange(k, n)
        weight[right_indices] += h[right_indices - k]
    return weight

def local_search(h, rng, max_iters=50000, time_budget=None):
   n = len(h)
   best_h = h.copy()
   best_c = compute_c5(h)
   start = time.time()
   step = 0.1  # initial epsilon fraction of max allowed
   for it in range(max_iters):
        if time_budget is not None and time.time() - start > time_budget:
            break
        # occasional random move for exploration, say 5% of iterations
        if rng.random() < 0.05:
            # random swap: select i, j distinct
            i, j = rng.choice(n, size=2, replace=False)
            max_eps = min(best_h[i], 1 - best_h[j])
            if max_eps <= 0:
                continue
            eps = rng.random() * max_eps * step
            new_h = best_h.copy()
            new_h[i] -= eps
            new_h[j] += eps
        else:
            # targeted move
            shift, _ = get_worst_shift(best_h)
            weight = compute_weights_for_shift(best_h, shift)
            # donor: low weight
            # select from candidates where h>0
            candidate_idxs = np.where(best_h > 1e-12)[0]
            if candidate_idxs.size == 0:
                break
            # Sort candidate indices by weight
            low_idxs = candidate_idxs[np.argsort(weight[candidate_idxs])[:max(1, candidate_idxs.size // 10)]]
            donor = rng.choice(low_idxs)
            # recipient: high weight where h<1
            rec_candidates = np.where(best_h < 1 - 1e-12)[0]
            if rec_candidates.size == 0:
                break
            high_idxs = rec_candidates[np.argsort(-weight[rec_candidates])[:max(1, rec_candidates.size // 10)]]
            recipient = rng.choice(high_idxs)
            max_eps = min(best_h[donor], 1 - best_h[recipient])
            if max_eps <= 0:
                continue
            eps = rng.random() * max_eps * step
            new_h = best_h.copy()
            new_h[donor] -= eps
            new_h[recipient] += eps

        # compute new C5
        c_new = compute_c5(new_h)
        if c_new < best_c - 1e-12:  # improvement
            best_h = new_h
            best_c = c_new
            # maybe increase step a little
            step = min(step * 1.01, 0.5)
        else:
            # maybe decrease step
            step = max(step * 0.99, 1e-4)
   return best_h, best_c

def local_search(h0, rng, max_iters=20000, time_budget=None):

    best_c = compute_c5(h)
    start = time.time()
    step = 0.2
    for it in range(max_iters):
        if time_budget is not None and time.time() - start > time_budget:

        # 5% random exploratory move
        if rng.random() < 0.05:
            i, j = rng.choice(n, size=2, replace=False)
            max_eps = min(best_h[i], 1.0 - best_h[j])
            if max_eps <= 0:
                continue
            eps = rng.random() * max_eps * step
            new_h = best_h.copy()
            new_h[i] -= eps
            new_h[j] += eps
        else:
            # targeted move
            shift = get_worst_shift(best_h)
            weight = compute_weights_for_shift(best_h, shift)
            # donor: low weight and h>0
            donor_candidates = np.where(best_h > 1e-12)[0]
            if donor_candidates.size == 0:
                break
            low_weights = weight[donor_candidates]
            # sort and take bottom 20%
            cutoff = max(1, donor_candidates.size // 5)
            low_idx = donor_candidates[np.argpartition(low_weights, cutoff-1)[:cutoff]]

            high_idx = rec_candidates[np.argpartition(-high_weights, cutoff_rec-1)[:cutoff_rec]]

import time
import numpy as np
from scipy.optimize import minimize

# globals used by the SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def get_worst_shift(h: np.ndarray) -> int:
    """Return the absolute shift (non‑negative integer) that gives the maximal overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)
    return abs(shift)


def compute_weights_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """
    For a given (non‑negative) shift, compute for each index i the contribution
    partner_i = h[i+shift] (if defined) + h[i-shift] (if defined).
    This is exactly the gradient of the autocorrelation sum at that shift.
    """
    n = h.size
    k = shift
    weight = np.zeros(n)
    if k < n:
        left = np.arange(0, n - k)
        weight[left] += h[left + k]
        right = np.arange(k, n)
        weight[right] += h[right - k]
    return weight


def project_to_feasible(v: np.ndarray, target: float) -> np.ndarray:
    """Projection onto {0 ≤ h ≤ 1, Σh = target} via a simple bisection on the shift."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Fast generation of a random feasible vector (0≤h≤1, Σh=n/2)."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_to_feasible(v, n / 2.0)


def sum_eq_constraint(x: np.ndarray) -> float:
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x: np.ndarray) -> float:
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int = 300) -> np.ndarray:
    """Run a single SLSQP optimisation with the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # exact sum correction (project onto feasible hyperplane)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start: np.ndarray, rng: np.random.Generator, max_iters: int = 3000, start_step: float = 0.5):
    """Simple deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


def local_search(h0: np.ndarray, rng: np.random.Generator,
                 max_iters: int = 20000, time_budget: float = None):
    """Targeted hill‑climb that focuses on the worst shift."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    start = time.time()
    step = 0.2
    for it in range(max_iters):
        if time_budget is not None and time.time() - start > time_budget:
            break

        # occasional random exploratory move
        if rng.random() < 0.05:
            i, j = rng.choice(n, size=2, replace=False)
            max_eps = min(best_h[i], 1.0 - best_h[j])
            if max_eps <= 0:
                continue
            eps = rng.random() * max_eps * step
            new_h = best_h.copy()
            new_h[i] -= eps
            new_h[j] += eps
        else:
            # targeted move based on the current worst shift
            shift = get_worst_shift(best_h)
            weight = compute_weights_for_shift(best_h, shift)

            donor_cands = np.where(best_h > 1e-12)[0]
            if donor_cands.size == 0:
                break
            low_cut = max(1, donor_cands.size // 5)
            low_idx = donor_cands[np.argpartition(weight[donor_cands], low_cut - 1)[:low_cut]]
            donor = rng.choice(low_idx)

            rec_cands = np.where(best_h < 1.0 - 1e-12)[0]
            if rec_cands.size == 0:
                break
            high_cut = max(1, rec_cands.size // 5)
            high_idx = rec_cands[np.argpartition(-weight[rec_cands], high_cut - 1)[:high_cut]]
            recipient = rng.choice(high_idx)

            max_eps = min(best_h[donor], 1.0 - best_h[recipient])
            if max_eps <= 0:
                continue
            eps = rng.random() * max_eps * step
            new_h = best_h.copy()
            new_h[donor] -= eps
            new_h[recipient] += eps

        c_new = compute_c5(new_h)
        if c_new < best_c - 1e-12:
            best_h = new_h
            best_c = c_new
            step = min(step * 1.02, 0.5)
        else:
            step = max(step * 0.98, 1e-4)

    return best_h, best_c


def upsample_initial(to_n: int) -> np.ndarray:
    """Interpolate the global initial_h_values (if present) to length to_n."""
    init = np.array(initial_h_values, dtype=float)
    m = init.size
    xs_old = np.linspace(0.0, 2.0, m, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, to_n, endpoint=False)
    interp = np.interp(xs_new, xs_old, init)
    target = to_n / 2.0
    return project_to_feasible(interp, target)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # candidate discretisations (include the size of the supplied initial if any)
    candidate_ns = [240, 300, 360, 420]
    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        if n0 not in candidate_ns:
            candidate_ns = [n0] + candidate_ns

    for n in candidate_ns:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining < 5.0:
            break

        # allocate time for this size (capped to keep overall runtime reasonable)
        time_for_n = min(max(5.0, remaining * 0.3), 40.0)

        # build a small pool of starting vectors
        starts = []

        if "initial_h_values" in globals():
            init_vec = upsample_initial(n)
            starts.append(init_vec)

        # a few random starts
        num_rand = max(1, int(time_for_n // 20))
        for _ in range(num_rand):
            starts.append(random_feasible_vector(n, rng))

        per_start_time = time_for_n / len(starts)

        for init_vec in starts:
            sub_start = time.time()
            # 1) focused local search
            h_loc, c_loc = local_search(init_vec, rng,
                                         max_iters=50000,
                                         time_budget=per_start_time * 0.6)

            # 2) short SLSQP refinement (if time permits)
            try:
                for beta in (10.0, 100.0, 1000.0):
                    if time.time() - start_time > budget_s * 0.95:
                        break
                    h_loc = slsqp_optimize(h_loc, beta, rng, max_iter=200)
                    c_loc = compute_c5(h_loc)
            except Exception:
                pass

            # 3) final cheap hill‑climb
            try:
                if time.time() - start_time < budget_s * 0.98:
                    h_loc, c_loc = hill_climb(h_loc, rng, max_iters=1500, start_step=0.1)
            except Exception:
                pass

            if c_loc < best_c:
                best_c, best_h, best_n = c_loc, h_loc.copy(), n

            if time.time() - start_time > budget_s * 0.99:
                break

        if time.time() - start_time > budget_s * 0.99:
            break

    # fallback in case nothing succeeded
    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c = compute_c5(best_h)

    # final feasibility enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, best_n)
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
