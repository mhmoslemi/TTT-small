def smooth_obj_grad(h, beta):
    n = h.size
    conv_raw = correlate_full(h) * (2.0 / n)   # dx = 2/n
    max_raw = conv_raw.max()
    # compute stabilized softmax weights
    shifted = conv_raw - max_raw
    exp_shift = np.exp(beta * shifted)
    sumexp = exp_shift.sum()
    w = exp_shift / sumexp
    obj = max_raw + np.log(sumexp) / beta
    # compute gradient using w
    # w is length 2n-1, indexing shift k = (n-1) - p
    # We'll treat w as array aligned with conv index p (0..2n-2)
    # conv corresponds to each p, we need w in same order as conv.
    # grad = grad1 - grad2
    # grad1: convolution of (1 - h) with w_rev
    rev_w = w[::-1]
    # compute conv via FFT
    L1 = 1 << ((n + (2*n-1) - 1).bit_length())
    # we can compute using np.fft.rfft
    # for conv1:
    a1 = np.fft.rfft(np.concatenate([1 - h, np.zeros(L1 - n)]), L1)
    b1 = np.fft.rfft(np.concatenate([rev_w, np.zeros(L1 - (2*n-1))]), L1)
    conv1_full = np.fft.irfft(a1 * b1, L1)[: n + (2*n-1) - 1]
    grad1 = conv1_full[(n-1):(n-1 + n)]
    # conv2: h * w
    a2 = np.fft.rfft(np.concatenate([h, np.zeros(L1 - n)]), L1)
    b2 = np.fft.rfft(np.concatenate([w, np.zeros(L1 - (2*n-1))]), L1)
    conv2_full = np.fft.irfft(a2 * b2, L1)[: n + (2*n-1) - 1]
    grad2 = conv2_full[(n-1):(n-1 + n)]
    grad = grad1 - grad2
    # gradient of smooth_obj is grad, because obj = weighted sum; we used the weight method.
    return obj, grad

def fft_convolve(a,b):
    L = 1 << ((a.size + b.size - 1).bit_length())
    fa = np.fft.rfft(np.pad(a, (0, L - a.size)))
    fb = np.fft.rfft(np.pad(b, (0, L - b.size)))
    conv = np.fft.irfft(fa * fb, L)[: a.size + b.size - 1]
    return conv

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # set durations
    main_deadline = start_time + budget_s * 0.9
    upsample_deadline = start_time + budget_s * 0.99

    # Determine base n (if initial_h_values exists)
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n/2.0, rng)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # Predefine beta schedule and iter per beta
    beta_schedule = [0.5, 2.0, 5.0, 20.0, 80.0, 200.0, 800.0]
    iters_per_beta = 300  # adjust based on time

    # main loop: random restarts
    while time.time() < main_deadline:
        # choose start candidate
        if rng.random() < 0.6:
            start_h = perturb_h(best_h, rng, scale=0.3)
        else:
            start_h = random_feasible_vector(n, rng)
        # PGD
        cand_h, cand_c = projected_gd(start_h, beta_schedule, iters_per_beta, rng)
        # hill climb refinement
        cand_h2, cand_c2 = hill_climb(cand_h, rng, max_iters=1500, start_step=0.3)
        if cand_c2 < best_c:
            best_c = cand_c2
            best_h = cand_h2.copy()
        # occasional SLSQP
        if rng.random() < 0.2:
            for beta in beta_schedule:
                try:
                    h_opt = slsqp_optimize(best_h, beta, max_iter=250)

def fft_convolve(a,b):
    L = 1 << ((a.size + b.size - 1).bit_length())
    fa = np.fft.rfft(np.pad(a, (0, L - a.size)))
    fb = np.fft.rfft(np.pad(b, (0, L - b.size)))
    conv = np.fft.irfft(fa * fb, L)
    return conv[: a.size + b.size - 1]

import time
import numpy as np
from scipy.optimize import minimize

# -------------------------------------------------
# Basic projection / feasibility helpers
# -------------------------------------------------
def project_to_feasible(y, target):
    """Project y onto [0,1] with sum == target via a simple bisection."""
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        x = np.clip(y + mid, 0.0, 1.0)
        if x.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)


def enforce_sum(h, target, rng):
    """Fallback: random pair adjustments to repair the sum (used only rarely)."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Uniform random vector projected onto the feasible polytope."""
    y = rng.random(n)
    return project_to_feasible(y, n / 2.0)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


# -------------------------------------------------
# Correlation / convolution utilities
# -------------------------------------------------
def correlate_full(h):
    """Full linear correlation of h with (1‑h) using FFT (zero‑padding)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)


def fft_convolve(a, b):
    """Convolution via FFT (real input). Returns full length a+b‑1."""
    L = 1 << ((a.size + b.size - 1).bit_length())
    fa = np.fft.rfft(np.pad(a, (0, L - a.size)))
    fb = np.fft.rfft(np.pad(b, (0, L - b.size)))
    conv = np.fft.irfft(fa * fb, L)
    return conv[: a.size + b.size - 1]


# -------------------------------------------------
# Smoothed objective and its gradient
# -------------------------------------------------
def smooth_obj_grad(h, beta):
    """Log‑sum‑exp smoothing of the max overlap, plus its gradient."""
    n = h.size
    dx = 2.0 / n
    conv_raw = correlate_full(h) * dx                     # length 2n‑1
    max_raw = conv_raw.max()
    shifted = conv_raw - max_raw
    exp_shift = np.exp(beta * shifted)
    sumexp = exp_shift.sum()
    w = exp_shift / sumexp                                 # weights, sum 1
    # grad = dx * (conv((1‑h), w) – conv(h, w_rev))
    grad1 = fft_convolve(1 - h, w)[n - 1:n - 1 + n]       # (1‑h) * w
    grad2 = fft_convolve(h, w[::-1])[n - 1:n - 1 + n]    # h * w_rev
    grad = (grad1 - grad2) * dx
    obj = max_raw + np.log(sumexp) / beta
    return obj, grad


# -------------------------------------------------
# Projected gradient descent on the smoothed objective
# -------------------------------------------------
def projected_gd(h0, beta_schedule, iters_per_beta, rng):
    """Simple projected gradient descent with step‑size adaptation."""
    h = h0.copy()
    target = h.size / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)

    for beta in beta_schedule:
        step = 0.5
        for _ in range(iters_per_beta):
            obj, grad = smooth_obj_grad(h, beta)
            y = h - step * grad
            h_new = project_to_feasible(y, target)
            # evaluate new smoothed objective
            obj_new, _ = smooth_obj_grad(h_new, beta)
            if obj_new < obj:
                h = h_new
                step = min(step * 1.05, 1.0)
            else:
                step *= 0.5
            c = compute_c5(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        # small random nudge before next beta (helps escape plateaus)
        if rng.random() < 0.3:
            h = perturb_h(best_h, rng, scale=0.25)

    return best_h, best_c


# -------------------------------------------------
# Simple hill‑climb (greedy local improvement)
# -------------------------------------------------
def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


# -------------------------------------------------
# SLSQP smooth‑max refinement (reuse from previous code)
# -------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Log‑sum‑exp smoothing of maxₖ Σᵢ xᵢ (1‑xᵢ₊ₖ)."""
    conv = correlate_full(x) * (2.0 / x.size)
    bc = CURRENT_BETA * conv
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        x = enforce_sum(x, CURRENT_TARGET, rng)
    return x
