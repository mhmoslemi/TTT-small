def compute_c5_vec(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def anneal_binary(n, init_h, time_budget):
    start = time.time()
    h = init_h.copy()
    cur_c = compute_c5_vec(h)
    best_h = h.copy()
    best_c = cur_c

    # Determine number of iterations calibrate: try 200k or time.
    it = 0
    # temperature schedule: t = t0 * (t_end / t0) ** (elapsed / time_budget)
    t0 = 0.001  # maybe start
    t_end = 1e-6

    while time.time() - start < time_budget:
        elapsed = time.time() - start
        t = t0 * (t_end / t0) ** (elapsed / time_budget)
        # choose random swap between 1 and 0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h[i], h[j] = h[j], h[i]  # swap
        new_c = compute_c5_vec(h)
        delta = new_c - cur_c
        if delta < 0 or np.random.rand() < np.exp(-delta / t):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]
        it += 1
    return best_h, best_c

def anneal_h(n, h0, rng, max_time):
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    start = time.time()
    iteration = 0
    # set maximum iterations arbitrarily large, break on time
    while time.time() - start < max_time:
        # pick i with h[i] > 0 and j with h[j] < 1
        # to speed up, we can precompute indices; but we can just random sample until condition.
        i = rng.integers(0, n)
        if h[i] == 0:
            continue
        j = rng.integers(0, n)
        if h[j] == 1:
            continue
        # transfer epsilon

        if max_eps == 0:
            continue
        eps = rng.random() * max_eps * 0.5  # scale 0.5

        h[j] += eps

                best_h = h.copy()

import time
import math
import numpy as np
from scipy.optimize import minimize

# --------- Global placeholders for SLSQP (required because of no closures) ----------
_GLOBAL_TARGET = None
_GLOBAL_BETA = None


def _sum_eq_constraint(x):
    return np.sum(x) - _GLOBAL_TARGET


def _softmax_obj(x):
    # smooth max of the overlap integral (log‑sum‑exp)
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = _GLOBAL_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / _GLOBAL_BETA

def compute_c5(h):
    """Exact discrete C5 for step function h on [0,2] (dx = 2/len(h))."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def _project_to_feasible(v, target):
    """Project v onto {0≤h≤1, Σh = target} by shifting and clipping."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Fast random feasible vector (0≤h≤1, Σh = n/2)."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return _project_to_feasible(v, n / 2.0)


def _upsample_initial(to_n):
    """Linear interpolation of the global initial_h_values to length to_n."""
    init = np.array(initial_h_values, dtype=float)  # defined by the harness if present
    m = init.size
    xs_old = np.linspace(0.0, 2.0, m, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, to_n, endpoint=False)
    return np.interp(xs_new, xs_old, init)


# --------- Simulated annealing for continuous vectors ---------------------------
def _anneal_continuous(n, h0, rng, max_time):
    """Simulated annealing on the exact C5 (no surrogate)."""
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    start = time.time()
    it = 0
    while time.time() - start < max_time:
        i = rng.integers(0, n)
        if h[i] <= 0.0:
            continue
        j = rng.integers(0, n)
        if h[j] >= 1.0:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * 0.5  # moderate step size
        h[i] -= eps
        h[j] += eps
        new_c = compute_c5(h)
        elapsed = time.time() - start
        # simple linear cooling from 0.02 to 1e‑6
        T = 0.02 * (1.0 - elapsed / max_time) + 1e-6
        if new_c < cur_c or rng.random() < math.exp(-(new_c - cur_c) / T):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            h[i] += eps
            h[j] -= eps
        it += 1
    return best_h, best_c


# --------- Greedy hill climbing (continuous) ------------------------------------
def _hill_climb_continuous(h, rng, max_iters=2000):
    n = h.size
    cur_c = compute_c5(h)
    best_c = cur_c
    best_h = h.copy()
    step = 0.3
    for it in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * step
        if eps == 0.0:
            continue
        h[i] -= eps
        h[j] += eps
        new_c = compute_c5(h)
        if new_c < cur_c:
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i] += eps
            h[j] -= eps
        if (it + 1) % 200 == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


# --------- SLSQP refinement (soft‑max surrogate) --------------------------------
def _slsqp_refine(h0, rng, max_time):
    global _GLOBAL_TARGET, _GLOBAL_BETA
    n = h0.size
    _GLOBAL_TARGET = n / 2.0
    h = h0.copy()
    start = time.time()
    for beta in (10.0, 50.0, 200.0, 800.0, 2000.0):
        if time.time() - start > max_time:
            break
        _GLOBAL_BETA = beta
        bounds = [(0.0, 1.0)] * n
        cons = {"type": "eq", "fun": _sum_eq_constraint}
        res = minimize(
            _softmax_obj,
            h,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": 400, "ftol": 1e-12, "disp": False},
        )
        x = np.clip(res.x, 0.0, 1.0)

        # Exact projection onto the hyperplane Σh = target
        diff = _GLOBAL_TARGET - x.sum()
        if abs(diff) > 1e-12:
            for _ in range(2000):
                if abs(diff) < 1e-12:
                    break
                idx = rng.integers(0, n)
                add = min(diff, 1.0 - x[idx]) if diff > 0 else max(diff, -x[idx])
                x[idx] += add
                diff = _GLOBAL_TARGET - x.sum()
        h = x
    return h


# --------- Main entry point ------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # candidate discretisations
    candidate_ns = [84, 120, 168, 240, 300]
    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        if n0 not in candidate_ns:
            candidate_ns = [n0] + candidate_ns

    for n in candidate_ns:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining < 10.0:
            break

        # allocate time for this n (roughly half of remaining, capped)
        time_for_n = min(remaining * 0.5, 120.0)
        n_start = time.time()
        while time.time() - n_start < time_for_n:
            # ----- initialise starting vector ---------------------------------
            if "initial_h_values" in globals() and n == len(initial_h_values):
                h0 = np.array(initial_h_values, dtype=float)
                h0 = _project_to_feasible(h0, n / 2.0)
            else:
                h0 = random_feasible_vector(n, rng)

            # ----- simulated annealing (continuous) ---------------------------
            restart_time = time_for_n - (time.time() - n_start)
            if restart_time < 5.0:
                break
            anneal_time = min(30.0, restart_time * 0.8)
            h_loc, c_loc = _anneal_continuous(n, h0, rng, anneal_time)

            # ----- greedy hill climb -----------------------------------------
            if time.time() - start_time < budget_s * 0.94:
                h_loc, c_loc = _hill_climb_continuous(h_loc, rng, max_iters=2000)

            # ----- SLSQP refinement (if time permits) ------------------------
            if time.time() - start_time < budget_s * 0.97:
                slsqp_time = min(15.0, (budget_s - (time.time() - start_time)) * 0.3)
                h_loc = _slsqp_refine(h_loc, rng, slsqp_time)
                c_loc = compute_c5(h_loc)

            # ----- update global best -----------------------------------------
            if c_loc < best_c:
                best_c, best_h, best_n = c_loc, h_loc.copy(), n

        if time.time() - start_time > budget_s * 0.98:
            break

    # -------------------------------------------------------------------------
    if best_h is None:
        best_n = 84
        best_h = np.full(best_n, 0.5)

    # enforce hard feasibility (clipping + tiny mass correction)
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            idx = rng.integers(0, best_n)
            if diff > 0:
                add = min(diff, 1.0 - best_h[idx])
                best_h[idx] += add
                diff -= add
            else:
                sub = min(-diff, best_h[idx])
                best_h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
