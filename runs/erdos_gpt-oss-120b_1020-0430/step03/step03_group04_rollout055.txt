def gradient_for_shift(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        L = n - k
        # term 1: +1 for i < L
        g[:L] += 1.0
        # -h_{i+k}
        g[:L] -= h[k:]
        # -h_{i - k}
        if k > 0:
            g[k:] -= h[:n - k]
    else:
        # k negative: shift left
        k2 = -k
        L = n - k2
        # for i such that i + k >= 0 => i >= k2

        # So sum_{j=0}^{n- k2 -1} h_{j + k2} * (1 - h_j)

def gradient_exchange_climb(h, rng, max_iters=5000, init_step=0.5, decay=0.9995):
   step = init_step
   n = len(h)
   for it in range(max_iters):
        # compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr) - (n-1)
        # gradient for worst shift
        g = gradient_for_shift(h, idx)
        # Determine i_pos (largest positive gradient) and i_neg (most negative gradient)
        # We want to move epsilon from any i where decreasing h reduces corr => positive g (since dC/dh > 0 => decreasing reduces)
        # So i_pos = argmax(g) where g > 0 and h[i] > 0
        # i_neg = argmin(g) where g < 0 and h[i] < 1
        # If no such pair, break
        pos_candidates = np.where((g > 1e-12) & (h > 1e-12))[0]
        neg_candidates = np.where((g < -1e-12) & (h < 1 - 1e-12))[0]
        if pos_candidates.size == 0 or neg_candidates.size == 0:
            # fallback random perturb
            new_h = perturb_h(h, rng, scale=step)
            h = new_h
        else:
            i_pos = pos_candidates[np.argmax(g[pos_candidates])]
            i_neg = neg_candidates[np.argmin(g[neg_candidates])]
            # epsilon: step * min(...). Let's set epsilon = step * min(h[i_pos], 1 - h[i_neg])
            max_eps = min(h[i_pos], 1 - h[i_neg])
            if max_eps <= 0:
                # pick random perturb
                new_h = perturb_h(h, rng, scale=step)
                h = new_h

                eps = step * max_eps * rng.random()   # random fraction up to step*max_eps
                # Optionally also allow full max_eps
                new_h = h.copy()
                new_h[i_pos] -= eps
                new_h[i_neg] += eps
                h = new_h
        # Optionally optionally diffusion: after each iteration adapt step = max(step * decay, 1e-4)
   return h

def simulated_annealing(h0, rng, deadline):
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    # Temperature schedule: start T = max(0.01*cur_c, 1e-4)
    it = 0
    while time.time() < deadline and it < 20000:
        # step for perturbation: decaying
        step = max(0.5 * (0.9998 ** it), 1e-4)
        cand_h = perturb_h(cur_h, rng, scale=step)
        cand_c = compute_c5(cand_h)
        if cand_c < cur_c:
            cur_h, cur_c = cand_h, cand_c
            if cand_c < best_c:
                best_h, best_c = cand_h.copy(), cand_c
        else:
            T = max(0.01 * cur_c, 1e-4) * (0.9995 ** it) + 1e-12
            if rng.random() < np.exp(-(cand_c - cur_c) / T):
                cur_h, cur_c = cand_h, cand_c
        it += 1
    return best_h, best_c

import time, math, numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h):
    n = h.size
    return float(np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / n))

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

def random_feasible_vector(n, rng):
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

# ---------- gradient for a given shift ----------
def gradient_for_shift(h, k):
    n = h.size
    g = np.zeros(n)
    if k >= 0:
        L = n - k
        g[:L] = 1.0 - h[k:]
        if k:
            g[k:] -= h[:L]
    else:
        k2 = -k
        L = n - k2
        g[k2:] = 1.0 - h[:L]
        g[:L] -= h[k2:]
    return g

# ---------- directed hill‑climb ----------
def gradient_exchange_climb(h, rng, max_iters=1500, init_step=0.5, decay=0.9995):
    n = h.size
    step = init_step
    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode="full")
        k = np.argmax(corr) - (n - 1)
        g = gradient_for_shift(h, k)

        pos = np.where((g > 1e-12) & (h > 1e-12))[0]
        neg = np.where((g < -1e-12) & (h < 1 - 1e-12))[0]
        if pos.size and neg.size:
            i = pos[np.argmax(g[pos])]
            j = neg[np.argmin(g[neg])]
            max_eps = min(h[i], 1.0 - h[j])
            eps = step * max_eps * rng.random()
            h[i] -= eps
            h[j] += eps
        else:
            h = perturb_h(h, rng, scale=step)

        step = max(step * decay, 1e-5)
    return h

# ---------- simulated annealing ----------
def simulated_annealing(h0, rng, deadline):
    cur = h0.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    it = 0
    T0 = max(0.01 * cur_c, 1e-4)
    while time.time() < deadline and it < 20000:
        step = max(0.5 * (0.9998 ** it), 1e-4)
        cand = perturb_h(cur, rng, scale=step)
        cand_c = compute_c5(cand)
        if cand_c < cur_c:
            cur, cur_c = cand, cand_c
            if cand_c < best_c:
                best, best_c = cand.copy(), cand_c
        else:
            T = T0 * (0.9995 ** it) + 1e-12
            if rng.random() < math.exp(-(cand_c - cur_c) / T):
                cur, cur_c = cand, cand_c
        it += 1
    return best, best_c

# ---------- SLSQP with soft‑max ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj, h0, method="SLSQP", bounds=bounds, constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

# ---------- main driver ----------
