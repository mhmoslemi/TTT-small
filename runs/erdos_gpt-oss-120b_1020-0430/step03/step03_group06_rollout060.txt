def project_sum_box(v, target):
    # binary search for lambda
    lo = -max(v.max(), 0) - 1.0
    hi = 1.0 - v.min() + 1.0
    for _ in range(50):

        if s > target:

def gradient_descent_refine(h, rng, max_steps=5000, eta0=0.1):
    n = h.size
    target = n / 2.0
    for step_idx in range(max_steps):
        # compute full correlation O(k)
        O = np.correlate(h, 1 - h, mode='full')
        # worst shift index
        idx = np.argmax(O)
        k = idx - (n - 1)
        if k < 0:
            k = -k
        if k == 0:
            # shouldn't happen; break
            break
        # compute gradient for worst shift
        if k >= n:
            # impossible as max shift n-1
            k = n - 1
        # pos shift = h shifted right by k (zero fill)
        pos = np.concatenate([h[k:], np.zeros(k)])
        neg = np.concatenate([np.zeros(k), h[:n - k]])
        grad = 1.0 - pos - neg
        # step size schedule
        eta = eta0 * (1.0 - step_idx / max_steps)
        if eta < 1e-5:
            eta = 1e-5
        h_new = h - eta * grad
        h = project_sum_box(h_new, target)
    return h

    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            if add == 0.0:
                continue
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x

# Projection onto box [0,1] with exact sum (binary‑search lambda)
# --------------------------------------------------------------
def project_sum_box(v, target):
    lo = -max(v.max(), 0.0) - 1.0
    hi = 1.0 - v.min() + 1.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + ((lo + hi) / 2.0), 0.0, 1.0)


def gradient_descent_refine(h, rng, steps=5000, eta0=0.2):
    """Targeted descent on the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(steps):
        O = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(O)
        k = idx - (n - 1)
        k = abs(k)
        if k == 0:
            break
        # construct shifted copies (zero padded)
        pos = np.concatenate([h[k:], np.zeros(k)])
        neg = np.concatenate([np.zeros(k), h[: n - k]])
        grad = 1.0 - pos - neg
        eta = eta0 * (1.0 - it / steps)
        if eta < 1e-5:
            eta = 1e-5
        h = project_sum_box(h - eta * grad, target)
    return h

# Main entry point
# --------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ---- initial vector (use parent if present) ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n0 / 2.0, rng)
    else:
        n0 = 180
        h0 = np.full(n0, 0.5)
        h0 = enforce_sum(h0, n0 / 2.0, rng)

    # ---- build a small pool of promising starters ----
    candidates = [h0]
    xs = (np.arange(n0) + 0.5) * (2.0 / n0)

    # triangular shape (peak 1 at x=1)
    tri = np.clip(1 - np.abs(xs - 1.0), 0.0, 1.0)
    candidates.append(enforce_sum(tri, n0 / 2.0, rng))

    # low‑frequency sin / cos waves
    for f in (1, 2, 3):
        for fun in (np.sin, np.cos):
            w = 0.5 + 0.5 * fun(np.pi * f * xs)
            candidates.append(enforce_sum(w, n0 / 2.0, rng))

    # linear ramps
    candidates.append(enforce_sum(xs / 2.0, n0 / 2.0, rng))
    candidates.append(enforce_sum(1.0 - xs / 2.0, n0 / 2.0, rng))

    # a few random feasible vectors
    for _ in range(5):
        candidates.append(random_feasible_vector(n0, rng))

    best_h = None
    best_c = float("inf")
    for cand in candidates:
        c = compute_c5(cand)
        if c < best_c:
            best_c, best_h = c, cand.copy()

    # ---- optional up‑sampling to a finer grid (up to 720) ----
    target_n = 720
    factor = target_n // n0
    if factor * n0 == target_n and time.time() - start_time < budget_s * 0.9:
        h = np.repeat(best_h, factor)
        n = h.size
        h = enforce_sum(h, n / 2.0, rng)
    else:
        h = best_h
        n = n0

    best_h, best_c = h.copy(), compute_c5(h)

    # ---- main optimisation loop ----
    beta_list = [10.0, 40.0, 120.0, 400.0, 800.0, 2000.0, 4000.0]
    while time.time() - start_time < budget_s * 0.85:
        # start from a perturbed best or a fresh random point
        if rng.random() < 0.7:
            start = perturb_h(best_h, rng, scale=0.3)
        else:
            start = random_feasible_vector(n, rng)

        # aggressive hill‑climb
        remain = budget_s - (time.time() - start_time)
        iters = int(20000 * remain / budget_s)
        iters = max(iters, 2000)
        h_opt, c_opt = hill_climb(start, rng, max_iters=iters, start_step=0.5)

        # if we improved, run targeted gradient descent and SLSQP refinements
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            # gradient‑descent on worst shift
            h_opt = gradient_descent_refine(best_h, rng, steps=5000, eta0=0.2)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c:
                best_h, best_c = h_opt, c_opt
            # SLSQP sweeps with increasing β
            for beta in beta_list:
                if time.time() - start_time > budget_s * 0.95:
                    break
                h_s = slsqp_optimize(best_h, beta, max_iter=800)
                c_s = compute_c5(h_s)
                if c_s < best_c:
                    best_h, best_c = h_s, c_s

        # occasional extra fine hill‑climb on current best
        if time.time() - start_time < budget_s * 0.70:
            h_fine, c_fine = hill_climb(best_h, rng, max_iters=10000, start_step=0.2)
            if c_fine < best_c:
                best_h, best_c = h_fine, c_fine

    # ---- final clean‑up ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
