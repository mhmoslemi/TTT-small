import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic feasibility utilities
# ----------------------------------------------------------------------
def enforce_sum(h, target, rng):
    """Adjust h so that sum(h) == target while keeping 0 <= h <= 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    # start from uniform 0.5 and perform a sum‑preserving random walk
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h = enforce_sum(h, target, rng)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


# ----------------------------------------------------------------------
# Correlation and C5 evaluation
# ----------------------------------------------------------------------
def correlate_full(h):
    """Full linear correlation of h with (1‑h) using FFT."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())  # next power of two
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)
    return conv[: 2 * n - 1]


def compute_c5(h):
    """Discrete C₅ for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)

# Smooth surrogate for max (log‑sum‑exp) and SLSQP refinement
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    conv = correlate_full(x)
    bc = CURRENT_BETA * conv
    m = bc.max()
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=500):
    """SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def hill_climb_multi(h_start, rng, max_iters=2000, step=0.4, candidates=5):
    """Greedy multi‑candidate hill‑climb."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iters):
        improved = False
        for _ in range(candidates):
            new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, best_h = c, new_h.copy()
                improved = True
        if improved:
            h = best_h.copy()
        else:
            step = max(step * 0.9, 1e-6)
    return best_h, best_c


def reduce_max_lag(h, rng, max_iters=500, step=0.05):
    """Targeted moves that specifically lower the lag attaining max overlap."""
    h_cur = h.copy()
    best_c = compute_c5(h_cur)
    n = h_cur.size
    for _ in range(max_iters):
        conv = correlate_full(h_cur)
        idx = np.argmax(conv)
        lag = idx - (n - 1)  # actual shift value
        # gradient of the max‑lag term
        grad = np.zeros(n)
        if lag >= 0:
            # (1 - h[i+lag])
            grad[: n - lag] += (1 - h_cur[lag:])
            # -h[i-lag]
            grad[lag:] -= h_cur[: n - lag]
        else:
            l = -lag
            grad[l:] += (1 - h_cur[: n - l])
            grad[: n - l] -= h_cur[l:]
        pos = np.where(grad > 1e-12)[0]
        neg = np.where(grad < -1e-12)[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = rng.choice(pos)
        j = rng.choice(neg)
        eps = min(h_cur[i], 1.0 - h_cur[j], step)
        if eps <= 0:
            continue
        h_cur[i] -= eps
        h_cur[j] += eps
        c_new = compute_c5(h_cur)
        if c_new < best_c:
            best_c = c_new
        else:
            # revert – move was not beneficial
            h_cur[i] += eps
            h_cur[j] -= eps
    return h_cur, best_c
