def linear_convolve(a, b):
    L = 1 << ((a.size + b.size - 1).bit_length())
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    conv = np.fft.irfft(fa * fb, L)[: a.size + b.size - 1]
    return conv

import time, numpy as np

def correlate_full(h):
    n = h.size
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv

def compute_c5(h):
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)

def linear_convolve(a, b):
    L = 1 << ((a.size + b.size - 1).bit_length())
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    c = np.fft.irfft(fa * fb, L)[: a.size + b.size - 1]
    return c

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    h = np.full(n, 0.5)

        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)

            mb = m / (1 - beta1 ** it)
            vb = v / (1 - beta2 ** it)
            step = lr * mb / (np.sqrt(vb) + eps)
            h -= step
            np.clip(h, 0.0, 1.0, out=h)
            diff = target - h.sum()
            if abs(diff) > 1e-12:
                h = enforce_sum(h, target, rng)
            if it % 200 == 0:
                c = compute_c5(h)
                if c < best_c:
                    best_c, best_h = c, h.copy()
        # short hill‑climb refinement
        if time.time() < deadline:
            hc, cc = hill_climb(best_h, rng, max_iters=400, start_step=0.2)
            if cc < best_c:
                best_c, best_h = cc, hc.copy()
        h = best_h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.95  # reserve a bit for final clean‑up

    base_n = 180
    candidates = []

    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)
        if init.size >= base_n:
            factor = init.size // base_n
            down = init[::factor][:base_n]
            down = enforce_sum(np.clip(down, 0.0, 1.0), base_n / 2.0, rng)
            candidates.append(down)
    for _ in range(3):
        candidates.append(random_feasible_vector(base_n, rng))

    best_h, best_c = None, float("inf")
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    attempts = 3
    for _ in range(attempts):
        if time.time() >= deadline:
            break
        att_dead = time.time() + (deadline - time.time()) * 0.33
        start_h = perturb_h(best_h, rng, scale=0.3) if rng.random() < 0.6 else random_feasible_vector(base_n, rng)
        h_opt, c_opt = optimize_smooth(start_h, rng, att_dead)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()
        # extra local hill‑climb
        if time.time() < att_dead:
            hc, cc = hill_climb(best_h, rng, max_iters=600, start_step=0.15)
            if cc < best_c:
                best_c, best_h = cc, hc.copy()

    # optional up‑sampling if time permits
    if time.time() < deadline and best_h.size * 2 <= 720:
        up = np.repeat(best_h, 2)
        up = enforce_sum(np.clip(up, 0.0, 1.0), up.size / 2.0, rng)
        up_dead = deadline
        h_up, c_up = optimize_smooth(up, rng, up_dead)
        if c_up < best_c:
            best_c, best_h = c_up, h_up.copy()
        else:
            hc, cc = hill_climb(up, rng, max_iters=1200, start_step=0.2)
            if cc < best_c:
                best_c, best_h = cc, hc.copy()

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
