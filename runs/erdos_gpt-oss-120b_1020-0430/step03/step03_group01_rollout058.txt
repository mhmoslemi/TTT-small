def enforce_sum(h, target, rng, max_iter=1000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) <= 1e-12:
            break
        # choose random index where we can adjust
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # final correction
    if abs(diff) > 1e-12:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
    return h

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h
```

import time
import numpy as np
from scipy.optimize import minimize

# ---------- helpers ----------
CURRENT_BETA = 1.0
CURRENT_TARGET = 1.0


def enforce_sum(h, target, rng, max_iter=20):
    """Clip to [0,1] and adjust to hit the exact target sum."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # redistribute proportionally to free capacity
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            caps = 1.0 - h
            total = caps.sum()
            if total <= 0:
                break
            delta = diff * caps / total
        else:
            caps = h
            total = caps.sum()
            if total <= 0:
                break
            delta = diff * caps / total
        h += delta
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
    # tiny residual fix (random)
    if abs(diff) > 1e-12:
        for _ in range(200):
            i = rng.integers(0, h.size)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h


def random_feasible_vector(n, rng):
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)


def perturb_h(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def compute_c5(h):
    """Exact discrete C₅ for uniform grid on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def gradient_step(h, k, step, rng):
    n = h.size
    g = np.zeros_like(h)
    if k < n:
        g[: n - k] += h[k:]
        g[k:] += h[: n - k]
    # zero‑mean direction (preserves sum)
    d = g - np.mean(g)
    h_new = h + step * d
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = enforce_sum(h_new, n / 2.0, rng)
    return h_new


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


# ---------- main ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- initial vector ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 240
        h0 = random_feasible_vector(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ---- gradient ascent on current grid ----
    while time.time() - start < budget_s * 0.55:
        conv = np.correlate(best_h, best_h, mode="full")
        inner = conv[n - 1 :]          # non‑negative shifts
        k = int(np.argmin(inner))
        if k == 0:                     # nothing to improve, random tweak
            cand = perturb_h(best_h, rng, scale=0.3)
            cand = enforce_sum(cand, n / 2.0, rng)
            c_cand = compute_c5(cand)
            if c_cand < best_c:
                best_h, best_c = cand, c_cand
            continue

        step = 0.5
        improved = False
        while step > 1e-4:
            cand = gradient_step(best_h, k, step, rng)
            c_cand = compute_c5(cand)
            if c_cand < best_c - 1e-12:
                best_h, best_c = cand, c_cand
                improved = True
                break
            step *= 0.5
        if not improved:
            # occasional random move to escape plateaus
            cand = perturb_h(best_h, rng, scale=0.3)
            cand = enforce_sum(cand, n / 2.0, rng)
            c_cand = compute_c5(cand)
            if c_cand < best_c:
                best_h, best_c = cand, c_cand

    # ---- optional up‑sampling ----
    elapsed = time.time() - start
    if budget_s - elapsed > budget_s * 0.2:
        new_n = min(720, max(360, n * 2))
        old_x = (np.arange(n) + 0.5) * (2.0 / n)
        new_x = (np.arange(new_n) + 0.5) * (2.0 / new_n)
        cur_h = np.interp(new_x, old_x, best_h)
        cur_h = enforce_sum(cur_h, new_n / 2.0, rng)

        while time.time() - start < budget_s * 0.9:
            conv = np.correlate(cur_h, cur_h, mode="full")
            inner = conv[new_n - 1 :]
            k = int(np.argmin(inner))
            if k == 0:
                cand = perturb_h(cur_h, rng, scale=0.3)
                cand = enforce_sum(cand, new_n / 2.0, rng)
                c_cand = compute_c5(cand)
                if c_cand < best_c:
                    best_h, best_c, n = cand.copy(), c_cand, new_n
                cur_h = cand
                continue

            step = 0.5
            improved = False
            while step > 1e-4:
                cand = gradient_step(cur_h, k, step, rng)
                c_cand = compute_c5(cand)
                if c_cand < best_c - 1e-12:
                    best_h, best_c, n = cand.copy(), c_cand, new_n
                    cur_h = cand
                    improved = True
                    break
                step *= 0.5
            if not improved:
                cand = perturb_h(cur_h, rng, scale=0.3)
                cand = enforce_sum(cand, new_n / 2.0, rng)
                c_cand = compute_c5(cand)
                if c_cand < best_c:
                    best_h, best_c, n = cand.copy(), c_cand, new_n
                cur_h = cand

    # ---- final SLSQP refinement if time permits ----
    if time.time() - start < budget_s * 0.97:
        global CURRENT_BETA, CURRENT_TARGET
        CURRENT_TARGET = n / 2.0
        CURRENT_BETA = 2000.0
        bounds = [(0.0, 1.0)] * n
        cons = {"type": "eq", "fun": sum_eq_constraint}
        try:
            res = minimize(
                softmax_obj,
                best_h,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 200, "ftol": 1e-12, "disp": False},
            )
            x = np.clip(res.x, 0.0, 1.0)
            x = enforce_sum(x, n / 2.0, rng)
            c = compute_c5(x)
            if c < best_c:
                best_h, best_c = x, c
        except Exception:
            pass

    # ---- simple local hill‑climb (last pass) ----
    while time.time() - start < budget_s * 0.99:
        cand = perturb_h(best_h, rng, scale=0.1)
        cand = enforce_sum(cand, n / 2.0, rng)
        c_cand = compute_c5(cand)
        if c_cand < best_c:
            best_h, best_c = cand, c_cand

    # ---- clean up and return ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
