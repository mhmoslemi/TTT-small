def project_box_sum(v, target, max_iter=10):
    v = np.clip(v, 0.0, 1.0)
    # adjust sum iteratively
    for _ in range(max_iter):
        s = v.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to add diff to entries with v < 1
            mask = v < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            add = diff / cnt
            v[mask] += add
            # after addition, may exceed 1, clip again
            v = np.clip(v, 0.0, 1.0)
        else:
            # need to subtract diff from entries with v > 0
            mask = v > 0.0
            cnt = mask.sum()
            if cnt == 0:
                break
            sub = -diff / cnt
            v[mask] -= sub
            v = np.clip(v, 0.0, 1.0)
    # final
    return v

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum via projection box sum
    x = project_box_sum(x, CURRENT_TARGET)
    return x

def compute_c5(h):
    n = h.shape[0]
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n))

def project_box_sum(v, target):
    v = np.clip(v, 0.0, 1.0)
    for _ in range(10):
        s = v.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        adjust = diff / cnt
        v[mask] += adjust
        v = np.clip(v, 0.0, 1.0)
    return v

def random_feasible_vector(n, rng):
    v = rng.random(n)
    return project_box_sum(v, n/2.0)

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

def gradient_descent_opt(h0, n, rng, end_time):
    h = h0.copy()
    target = n/2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    step0 = 0.6
    decay = 0.9995
    it = 0
    while True:
        if time.time() > end_time:
            break
        # compute worst shift
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = idx
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1 - 2 * h
        else:
            start = max(0, -k)
            end = min(n, n - k)
            i_slice = slice(start, end)
            j_slice = slice(start + k, end + k)
            g[i_slice] += 1 - h[j_slice]
            g[j_slice] -= h[i_slice]
        step = step0 * (decay ** it)
        new_h = h - step * g
        new_h = project_box_sum(new_h, target)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        it += 1
    return best_h, best_c

def hill_climb_simple(h, rng, max_iters, start_step, end_time):
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        if time.time() > end_time:
            break
        cand = perturb_h(best_h, rng, scale=step)
        c = compute_c5(cand)
        if c < best_c:
            best_c, best_h = c, cand.copy()
            step = start_step
        else:
            step = max(step * 0.9, 1e-5)
    return best_h, best_c

# SLSQP functions with globals
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter, end_time):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    x = project_box_sum(x, CURRENT_TARGET)
    # ensure we haven't exceeded time
    if time.time() > end_time:
        pass
    return x

def project_box_sum(v, target):
    v = np.clip(v, 0.0, 1.0)
    for _ in range(10):
        s = v.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        adj = diff / cnt
        v[mask] += adj
        v = np.clip(v, 0.0, 1.0)
    return v

    v = rng.random(n)
    return project_box_sum(v, n/2.0)

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

def gradient_descent_opt(h0, n, rng, end_time, max_iters=5000):
    h = h0.copy()
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    step0 = 0.6
    decay = 0.9995
    it = 0
    while it < max_iters and time.time() < end_time:
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = idx
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1 - 2 * h
        else:
            start = max(0, -k)
            end = min(n, n - k)
            i_slice = slice(start, end)
            j_slice = slice(start + k, end + k)
            g[i_slice] += 1 - h[j_slice]
            g[j_slice] -= h[i_slice]
        step = step0 * (decay ** it)
        new_h = h - step * g
        new_h = project_box_sum(new_h, target)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        it += 1
    return best_h, best_c

def hill_climb_simple(h, rng, max_iters, start_step, end_time):
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        if time.time() > end_time:
            break
        cand = perturb_h(best_h, rng, scale=step)
        c = compute_c5(cand)
        if c < best_c:
            best_c, best_h = c, cand.copy()
            step = start_step
        else:
            step = max(step * 0.9, 1e-5)
    return best_h, best_c

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter, end_time):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta

                   constraints=cons, options={'maxiter': max_iter,
                                              'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    x = project_box_sum(x, CURRENT_TARGET)
    return x
