def subgradient_optimize(h0, rng, max_iters, alpha0):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    alpha = alpha0
    for t in range(1, max_iters+1):
        corr = np.correlate(h, 1 - h, mode='full')
        k = np.argmax(corr)
        g = gradient_for_shift(h, k - (h.size - 1))  # shift in range -(n-1)...+(n-1)
        # step size schedule: alpha = alpha0 / sqrt(t)
        step = alpha0 / np.sqrt(t)
        h_new = project_to_feasible(h - step * g, h.size/2)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c
```

def gradient_max_corr(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n - 1)  # zero shift corresponds to idx = n-1
    # build gradient using slices
    g = np.zeros_like(h)
    if shift >= 0:
        k = shift
        # first term: (1 - h[i + k]) for i in [0, n - k - 1]
        g[:n - k] += 1 - h[k:]
        # second term: -h[i - k] for i in [k, n-1]
        g[k:] -= h[:n - k]
    else:

        # shift negative: C(-k) = sum_{i=k}^{n-1} h[i] * (1 - h[i - k])

    return g

import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_to_feasible(v, target):
    """Project onto {0≤h≤1, Σh = target} via a bisection on the shift."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Fast random feasible vector (0≤h≤1, Σh=n/2)."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_to_feasible(v, n / 2.0)


def upsample_initial(to_n):
    """Linear interpolation of global initial_h_values to a new length."""
    init = np.array(initial_h_values, dtype=float)
    m = init.size
    xs_old = np.linspace(0.0, 2.0, m, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, to_n, endpoint=False)
    return np.interp(xs_new, xs_old, init)


# ---------- gradient for the active shift ----------
def gradient_max_corr(h):
    """Sub‑gradient of the maximal overlap term."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)  # signed shift in [-(n-1), n-1]
    g = np.zeros_like(h)
    if shift >= 0:
        k = shift
        # (1 - h[i+k]) term
        g[: n - k] += 1 - h[k:]
        # -h[i-k] term
        g[k:] -= h[: n - k]
    else:
        k = -shift
        # (1 - h[i-k]) term (shift negative)
        g[k:] += 1 - h[: n - k]
        # -h[i+k] term
        g[: n - k] -= h[k:]
    return g


# ---------- sub‑gradient descent ----------
def subgradient_optimize(h0, rng, max_iters, time_limit, target):
    """Projected sub‑gradient descent for the max‑overlap objective."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    a0 = 0.5  # initial step scale
    check_every = 500
    start = time.time()
    for t in range(1, max_iters + 1):
        g = gradient_max_corr(h)
        step = a0 / np.sqrt(t)
        h = project_to_feasible(h - step * g, target)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if t % check_every == 0 and (time.time() - start) > time_limit:
            break
    return best_h, best_c


# ---------- smooth‑max SLSQP refinement ----------
def softmax_obj(x, beta):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = beta * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta


def slsqp_optimize(h0, beta, rng, max_iter=200):
    """SLSQP on the smooth‑max surrogate."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    res = minimize(lambda x: softmax_obj(x, beta), h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x


# ---------- simple hill‑climb ----------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.2):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


# ---------- main driver ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # candidate discretisations (focus on high resolution)
    candidate_ns = [300]
    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        if n0 not in candidate_ns:
            candidate_ns = [n0] + candidate_ns

    for n in candidate_ns:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining < 20.0:
            break

        target = n / 2.0
        # allocate time slices
        sub_time = remaining * 0.6
        slsqp_time = remaining * 0.25
        hill_time = remaining * 0.15

        # --- sub‑gradient phase (multiple restarts) ---
        restarts = 2
        sub_time_per = sub_time / restarts
        for r in range(restarts):
            if r == 0 and "initial_h_values" in globals():
                h0 = upsample_initial(n)
                h0 = project_to_feasible(h0, target)
            else:
                h0 = random_feasible_vector(n, rng)
            h_sub, c_sub = subgradient_optimize(h0, rng, max_iters=30000,
                                                time_limit=sub_time_per,
                                                target=target)
            if c_sub < best_c:
                best_c, best_h, best_n = c_sub, h_sub.copy(), n

        # --- smooth‑max SLSQP refinement ---
        h_cur = best_h if best_n == n else (random_feasible_vector(n, rng) if "initial_h_values" not in globals() else project_to_feasible(upsample_initial(n), target))
        c_cur = compute_c5(h_cur)
        for beta in (10.0, 50.0, 200.0, 800.0, 2000.0):
            if time.time() - start_time > budget_s * 0.96:
                break
            # give each beta a modest budget (max 150 iterations)
            max_iter = 150
            h_cur = slsqp_optimize(h_cur, beta, rng, max_iter=max_iter)
            c_cur = compute_c5(h_cur)
            if c_cur < best_c:
                best_c, best_h, best_n = c_cur, h_cur.copy(), n

        # --- final hill‑climb polishing ---
        if time.time() - start_time < budget_s * 0.98:
            h_cur, c_cur = hill_climb(h_cur, rng, max_iters=3000, start_step=0.15)
            if c_cur < best_c:
                best_c, best_h, best_n = c_cur, h_cur.copy(), n

    # fallback safety
    if best_h is None:
        best_n = 120
        best_h = np.full(best_n, 0.5)

    # final feasibility enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, best_n)
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
