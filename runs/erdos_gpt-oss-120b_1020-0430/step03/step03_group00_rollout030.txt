def compute_c5_fft(h):
    n = h.size
    # compute convolution with reversed (1-h)
    a = h.astype(np.float64)
    b = (1.0 - h)[::-1].astype(np.float64)
    L = 1
    while L < n + n - 1:
        L <<= 1
    fa = np.fft.fft(a, L)
    fb = np.fft.fft(b, L)
    conv = np.fft.ifft(fa * fb).real
    corr = conv[:2*n-1]
    return np.max(corr) * (2.0 / n)

def random_feasible_vector(n, rng):

    h = enforce_sum_bounds(h, n/2.0, rng)

def upsample_to_n(old, n):
    """Linear interpolation of step function old (uniform points) to n points."""
    nold = old.size
    xs_old = (np.arange(nold) + 0.5) * (2.0 / nold)
    xs_new = (np.arange(n) + 0.5) * (2.0 / n)
    new = np.interp(xs_new, xs_old, old)
    return new

def compute_c5_fft(h):
    n = h.size
    a = h.astype(np.float64)
    b = (1.0 - h)[::-1].astype(np.float64)
    L = 1
    while L < 2 * n - 1:
        L <<= 1
    fa = np.fft.fft(a, L)
    fb = np.fft.fft(b, L)
    conv = np.fft.ifft(fa * fb).real
    corr = conv[:2*n - 1]
    return np.max(corr) * (2.0 / n)

def worst_shift_info(h):
    n = h.size
    # compute correlation via np.correlate (or FFT) and return index of max
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr)
    k = idx - (n - 1)
    return k, corr[idx]

def gradient_worst_shift(h, k):
    n = h.size
    grad = np.ones_like(h)
    if k >= 0:
        grad[:n - k] -= h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[:n - kk] -= h[kk:]
        grad[kk:] -= h[:n - kk]
    return grad

def hill_climb_targeted(h, rng, steps=5000, step_scale=0.2):
    # gradient based moves
    best_h = h.copy()
    best_c = compute_c5_fft(best_h)
    cur_h = h.copy()
    cur_c = best_c
    for it in range(steps):
        k, _ = worst_shift_info(cur_h)
        grad = gradient_worst_shift(cur_h, k)
        # find positive and negative gradient positions
        pos = np.where((grad > 1e-9) & (cur_h > 1e-12))[0]
        neg = np.where((grad < -1e-9) & (cur_h < 1 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            # random perturb
            i = rng.integers(cur_h.size)
            j = rng.integers(cur_h.size)
        else:
            i = rng.choice(pos)
            j = rng.choice(neg)
        max_eps = min(cur_h[i], 1 - cur_h[j])
        if max_eps <= 0:
            continue
        eps = max_eps * step_scale * rng.random()
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5_fft(new_h)
        # Accept if improved
        if new_c < cur_c:
            cur_h = new_h
            cur_c = new_c
        if new_c < best_c:
            best_h = new_h
            best_c = new_c
        # decay step_scale
        step_scale = max(step_scale * 0.9995, 0.001)
    return best_h, best_c

        if abs(diff) < 1e-12:

            h[i] += add
            diff -= add
        else:                               # need to decrease some entries
            idx = np.where(h > 1e-12)[0]

            sub = min(-diff, h[i]) * rng.random()
            h[i] -= sub

    diff = target - h.sum()
    if abs(diff) > 1e-12:
        i = rng.integers(h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    h = rng.random(n)
    return enforce_sum_bounds(h, n / 2.0, rng)

def upsample_to_n(old, n):
    """Linear interpolation from len(old) points on [0,2] to n points."""
    n_old = old.size
    xs_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    xs_new = (np.arange(n) + 0.5) * (2.0 / n)
    return np.interp(xs_new, xs_old, old)

def compute_c5_fft(h):
    """Exact discrete C₅ using FFT for speed."""
    n = h.size
    a = h.astype(np.float64)
    b = (1.0 - h)[::-1].astype(np.float64)
    L = 1
    while L < 2 * n - 1:
        L <<= 1
    fa = np.fft.fft(a, L)
    fb = np.fft.fft(b, L)
    conv = np.fft.ifft(fa * fb).real
    corr = conv[: 2 * n - 1]
    return np.max(corr) * (2.0 / n)

def worst_shift_info(h):
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr)
    k = idx - (h.size - 1)
    return k, corr[idx]

def gradient_worst_shift(h, k):
    n = h.size
    grad = np.ones_like(h)
    if k >= 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    else:
        kk = -k
        grad[: n - kk] -= h[kk:]
        grad[kk:] -= h[: n - kk]
    return grad

def hill_climb_targeted(h_start, rng, steps=2000, step_scale=0.2):
    """Gradient‑guided hill climbing on the worst shift."""
    best_h = h_start.copy()
    best_c = compute_c5_fft(best_h)
    cur_h = h_start.copy()
    cur_c = best_c
    for it in range(steps):
        k, _ = worst_shift_info(cur_h)
        grad = gradient_worst_shift(cur_h, k)
        pos = np.where((grad > 1e-9) & (cur_h > 1e-12))[0]
        neg = np.where((grad < -1e-9) & (cur_h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            i = rng.integers(cur_h.size)
            j = rng.integers(cur_h.size)
        else:
            i = rng.choice(pos)
            j = rng.choice(neg)
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0:
            continue
        eps = max_eps * step_scale * rng.random()
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5_fft(new_h)
        if new_c < cur_c:
            cur_h, cur_c = new_h, new_c
        if new_c < best_c:
            best_h, best_c = new_h, new_c
        step_scale = max(step_scale * 0.9995, 0.001)
    return best_h, best_c

# ---------- SLSQP refinement ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    n = x.size
    a = x.astype(np.float64)
    b = (1.0 - x)[::-1].astype(np.float64)
    L = 1
    while L < 2 * n - 1:
        L <<= 1
    fa = np.fft.fft(a, L)
    fb = np.fft.fft(b, L)
    conv = np.fft.ifft(fa * fb).real
    corr = conv[: 2 * n - 1]
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_refine(h, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return enforce_sum_bounds(x, CURRENT_TARGET, rng)

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    total_budget = budget_s

    # ----- base resolution (720) -----
    n_base = 720
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)
        if init.size != n_base:
            h = upsample_to_n(init, n_base)
        else:
            h = init.copy()
    else:
        h = random_feasible_vector(n_base, rng)

    h = enforce_sum_bounds(np.clip(h, 0.0, 1.0), n_base / 2.0, rng)
    best_h = h.copy()
    best_c = compute_c5_fft(best_h)

    # ----- simulated annealing on base grid -----
    max_it_base = 800_000
    start_temp = 0.01
    end_temp = 1e-6
    cur_h, cur_c = best_h.copy(), best_c

    for it in range(max_it_base):
        if time.time() - start_time > total_budget * 0.48:
            break
        i, j = rng.integers(0, n_base, size=2)
        if i == j:
            continue
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0:
            continue
        scale = 0.5 * (1 - it / max_it_base) + 0.01
        eps = max_eps * scale * rng.random()
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5_fft(new_h)
        temp = start_temp * ((end_temp / start_temp) ** (it / max_it_base))
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / temp):
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c

    # ----- targeted hill‑climb on base solution -----
    if time.time() - start_time < total_budget * 0.53:
        bh, bc = hill_climb_targeted(best_h, rng, steps=3000, step_scale=0.2)
        if bc < best_c:
            best_h, best_c = bh, bc

    # ----- up‑sample to high resolution (1440) -----
    n_high = 1440
    h_high = upsample_to_n(best_h, n_high)
    h_high = enforce_sum_bounds(np.clip(h_high, 0.0, 1.0), n_high / 2.0, rng)
    cur_h, cur_c = h_high.copy(), compute_c5_fft(h_high)
    best_h_high, best_c_high = cur_h.copy(), cur_c

    # ----- simulated annealing on high grid -----
    max_it_high = 500_000
    start_temp_h = 0.01
    end_temp_h = 1e-7

    for it in range(max_it_high):
        if time.time() - start_time > total_budget * 0.9:
            break
        i, j = rng.integers(0, n_high, size=2)
        if i == j:
            continue
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0:
            continue
        scale = 0.3 * (1 - it / max_it_high) + 0.005
        eps = max_eps * scale * rng.random()
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5_fft(new_h)
        temp = start_temp_h * ((end_temp_h / start_temp_h) ** (it / max_it_high))
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / temp):
            cur_h, cur_c = new_h, new_c
            if new_c < best_c_high:
                best_h_high, best_c_high = new_h.copy(), new_c

    # ----- SLSQP refinement (high grid) -----
    beta_schedule = [5.0, 20.0, 80.0, 200.0, 800.0, 2000.0]
    for beta in beta_schedule:
        if time.time() - start_time > total_budget * 0.97:
            break
        try:
            refined = slsqp_refine(best_h_high, beta, rng, max_iter=500)
        except Exception:
            continue
        c_ref = compute_c5_fft(refined)
        if c_ref < best_c_high:
            best_h_high, best_c_high = refined, c_ref

    # ----- final targeted hill‑climb on high grid -----
    if time.time() - start_time < total_budget * 0.99:
        bh, bc = hill_climb_targeted(best_h_high, rng, steps=5000, step_scale=0.1)
        if bc < best_c_high:
            best_h_high, best_c_high = bh, bc

    # ----- choose best overall -----
    if best_c_high < best_c:
        final_h, final_c = best_h_high, best_c_high
    else:
        final_h, final_c = best_h, best_c

    final_h = np.clip(final_h, 0.0, 1.0)
    final_h = enforce_sum_bounds(final_h, final_h.size / 2.0, rng)
    final_c = compute_c5_fft(final_h)

    return final_h.tolist(), float(final_c), int(final_h.size)
