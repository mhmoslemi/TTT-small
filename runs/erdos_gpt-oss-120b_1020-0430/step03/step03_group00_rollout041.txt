def slsqp_optimize_with_grad(h0, beta, max_iter, rng):
    n = h0.size
    target = n / 2.0
    bounds = [(0.0,1.0)]*n
    cons = {"type":"eq","fun":lambda x: np.sum(x) - target}
    def fun(x):
        corr = np.correlate(x, 1 - x, mode='full')
        m = np.max(beta * corr)
        exp_term = np.exp(beta * corr - m)
        sum_exp = np.sum(exp_term)
        obj = (m + np.log(sum_exp)) / beta
        return obj
    def jac(x):
        corr = np.correlate(x, 1 - x, mode='full')
        m = np.max(beta * corr)
        exp_term = np.exp(beta * corr - m)
        sum_exp = np.sum(exp_term)
        p = exp_term / sum_exp
        offset = n - 1
        p_rev = p[::-1]
        conv1 = np.convolve(p_rev, x, mode='full')
        conv2 = np.convolve(p, x, mode='full')
        g_pos = conv1[offset:offset+n]
        g_neg = conv2[offset:offset+n]
        grad = np.ones(n) - g_pos - g_neg
        return grad
    res = minimize(fun, h0, method='SLSQP', jac=jac, bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # ensure sum exactly
    x = enforce_sum(x, target, rng)
    return x

    diff = target - h.sum()

        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        xs0 = (np.arange(n0) + 0.5) * (2.0 / n0)
        xs = (np.arange(n) + 0.5) * (2.0 / n)
        h_interp = np.interp(xs, xs0, h0)
        h_interp = np.clip(h_interp, 0.0, 1.0)
        idx = np.argsort(-h_interp)
        cur_h = np.zeros(n, dtype=float)
        cur_h[idx[: n // 2]] = 1.0
    else:
        cur_h = np.zeros(n, dtype=float)
        cur_h[: n // 2] = 1.0
        rng.shuffle(cur_h)

    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    no_improve = 0
    binary_deadline = start + budget_s * 0.55

    # ----- binary hill‑climb (targeted swaps) -----
    while time.time() < binary_deadline:
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        idx_max = int(np.argmax(corr))
        k = idx_max - (n - 1)

        idxs = np.arange(n)
        valid = (idxs + k >= 0) & (idxs + k < n)
        shifted = np.zeros(n, dtype=float)
        shifted[valid] = cur_h[idxs[valid] + k]

        bad = (cur_h == 1.0) & (shifted == 0.0)   # 1 followed by 0 at shift k
        good = (cur_h == 0.0) & (shifted == 1.0)  # 0 followed by 1 at shift k

        if bad.any() and good.any():
            i = rng.choice(np.where(bad)[0])
            j = rng.choice(np.where(good)[0])
            cand = cur_h.copy()
            cand[i] = 0.0
            cand[j] = 1.0
        else:
            if rng.random() < 0.05:
                ones = np.where(cur_h == 1.0)[0]
                zeros = np.where(cur_h == 0.0)[0]
                if ones.size >= 2 and zeros.size >= 2:
                    i1, i2 = rng.choice(ones, size=2, replace=False)
                    j1, j2 = rng.choice(zeros, size=2, replace=False)
                    cand = cur_h.copy()
                    cand[i1] = cand[i2] = 0.0
                    cand[j1] = cand[j2] = 1.0
                else:
                    i = rng.choice(np.where(cur_h == 1.0)[0])
                    j = rng.choice(np.where(cur_h == 0.0)[0])
                    cand = cur_h.copy()
                    cand[i] = 0.0
                    cand[j] = 1.0
            else:
                i = rng.choice(np.where(cur_h == 1.0)[0])
                j = rng.choice(np.where(cur_h == 0.0)[0])
                cand = cur_h.copy()
                cand[i] = 0.0
                cand[j] = 1.0

        cand_c = compute_c5(cand)

        # accept improvement or occasional uphill step
        if cand_c < cur_c or rng.random() < 0.001:
            cur_h = cand
            cur_c = cand_c
            no_improve = 0 if cand_c < cur_c else no_improve + 1
        else:
            no_improve += 1

        if cand_c < best_c:
            best_c = cand_c
            best_h = cand.copy()

        if no_improve > 10000:
            cur_h = np.zeros(n, dtype=float)
            cur_h[: n // 2] = 1.0
            rng.shuffle(cur_h)
            cur_c = compute_c5(cur_h)
            no_improve = 0

    # ----- continuous refinement with smoothed max (SLSQP) -----
    cont_deadline = start + budget_s * 0.95
    beta_sched = [5, 20, 80, 200, 800, 2000, 5000]
    cur_h_float = best_h.copy()
    cur_c_float = best_c

    for beta in beta_sched:
        if time.time() > cont_deadline:
            break
        remaining = cont_deadline - time.time()
        # keep at least 100 iterations per beta, cap at 800
        max_iter = int(min(800, max(100, remaining / (len(beta_sched) - beta_sched.index(beta) + 1))))
        try:
            new_h = slsqp_optimize_with_grad(cur_h_float, beta, max_iter=max_iter, rng=rng)
        except Exception:
            continue
        new_c = compute_c5(new_h)
        if new_c < cur_c_float:
            cur_h_float = new_h
            cur_c_float = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()

    # ----- final micro‑search (tiny fractional moves) -----
    micro_deadline = start + budget_s * 0.99
    while time.time() < micro_deadline:
        ones = np.where(best_h == 1.0)[0]
        zeros = np.where(best_h == 0.0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        eps = 0.02 * rng.random()
        cand = best_h.copy()
        cand[i] = max(0.0, cand[i] - eps)
        cand[j] = min(1.0, cand[j] + eps)
        cand_c = compute_c5(cand)
        if cand_c < best_c:
            best_c = cand_c
            best_h = cand.copy()

    # ----- clean‑up and final evaluation -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
