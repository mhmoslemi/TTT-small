def gradient_descent_step(h, shift, step_scale=1.0):
    n = h.size
    k = shift
    # compute gradient of Ck (max)
    # compute h_shift_forward: h[i+k] for i where i+k < n else 0
    # compute h_shift_backward: h[i-k] for i where i-k >=0 else 0
    forward = np.zeros_like(h)
    backward = np.zeros_like(h)
    if k >=0:
        forward[:n-k] = h[k:]   # h[i+k] for i < n-k
        backward[k:] = h[:n-k]  # h[i-k] for i >= k
    else:
        # negative shift
        k_neg = -k
        forward[k_neg:] = h[:n-k_neg]   # i+k < n => i < n-k
        backward[:n-k_neg] = h[k_neg:]  # i-k >=0 => i >= k
    grad = (1.0 - forward) - backward
    d = -grad
    # zero sum
    d = d - np.mean(d)
    # step size limit
    pos = d > 0
    neg = d < 0
    max_step = np.inf
    if np.any(pos):
        max_step = min(max_step, np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        max_step = min(max_step, np.min((-h[neg]) / d[neg]))
    if max_step == np.inf:
        return h
    step = step_scale * max_step
    h_new = h + step * d
    # Clip again for safety
    h_new = np.clip(h_new, 0.0, 1.0)
    # enforce sum
    h_new = enforce_sum(h_new, h_new.size/2.0, rng)  # rng global needed
    return h_new
```

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h
```

import time
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Utility helpers
# --------------------------------------------------------------
def enforce_sum(h, target, rng):
    """Adjust h so that sum(h) == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Random vector in [0,1]^n with sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small ε from i to j (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def correlate_full(h):
    """Full linear correlation of h with (1-h) using FFT and zero padding."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())      # next power of two
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Discrete C5 for step function h (dx = 2 / n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)


def max_corr_shift(h):
    """Return shift (index offset) achieving maximal correlation and its value."""
    conv = correlate_full(h)
    idx = int(conv.argmax())
    shift = idx - (h.size - 1)
    return shift, conv[idx] * (2.0 / h.size)

# Gradient descent on worst shift
# --------------------------------------------------------------
def gradient_descent_step(h, shift, rng, step_scale=0.7):
    """Projected gradient step improving the worst correlation."""
    n = h.size
    k = shift
    forward = np.zeros_like(h)
    backward = np.zeros_like(h)

    if k >= 0:
        if n - k > 0:
            forward[: n - k] = h[k:]
        if k < n:
            backward[k:] = h[: n - k]
    else:
        kp = -k
        if kp < n:
            forward[kp:] = h[: n - kp]
        if n - kp > 0:
            backward[: n - kp] = h[kp:]

    grad = (1.0 - forward) - backward
    d = -grad
    d -= d.mean()                         # keep sum zero

    pos = d > 1e-12
    neg = d < -1e-12
    max_step = np.inf
    if np.any(pos):
        max_step = min(max_step, np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        max_step = min(max_step, np.min((-h[neg]) / d[neg]))
    if not np.isfinite(max_step) or max_step <= 0:
        return h
    step = step_scale * max_step
    nh = h + step * d
    nh = np.clip(nh, 0.0, 1.0)
    nh = enforce_sum(nh, n / 2.0, rng)
    return nh

def simulated_annealing(start_h, rng, deadline, T0=0.02, Tf=1e-5, max_iters=2000):
    h = start_h.copy()
    c = compute_c5(h)
    best_h, best_c = h.copy(), c
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        new_h = perturb_h(h, rng, scale=0.35)
        new_c = compute_c5(new_h)
        delta = new_c - c
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            h, c = new_h, new_c
            if c < best_c:
                best_h, best_c = h.copy(), c
    return best_h, best_c

def hill_climb(h_start, rng, max_iters=2000, start_step=0.4):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c

# SLSQP on smooth surrogate (log-sum-exp)
# --------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = correlate_full(x)
    bc = CURRENT_BETA * corr
    m = bc.max()
    # stable log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny correction of the sum (numerical drift)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        n = x.size
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x
