def hill_climb(h, rng, max_iter, step_frac=0.5):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(best_h)
    cur_h = h.copy()
    cur_c = best_c

    for it in range(max_iter):
        # pick source i where h[i] > 0, destination j where h[j] < 1
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j: continue
        if cur_h[i] <= 0 or cur_h[j] >= 1: continue
        # epsilon magnitude
        eps_max = min(cur_h[i], 1.0 - cur_h[j])
        # choose epsilon scaled by step_frac
        eps = eps_max * (rng.random() * step_frac)
        if eps <= 0: continue
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        if new_c < cur_c - 1e-12:  # improvement 
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
        # else maybe accept with small probability? Not needed.

    return best_h, best_c

def long_sa(h0, rng, budget, start_time):
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c

    # set parameters
    total_allowed = budget
    # schedule: geometric decay from init_temp to final_temp over total_allowed seconds
    init_temp = 0.05
    final_temp = 1e-7

    while True:
        elapsed = time.time() - start_time
        if elapsed >= total_allowed:
            break
        # compute T based on elapsed fraction
        t_frac = elapsed / total_allowed
        T = init_temp * ((final_temp / init_temp) ** t_frac)
        # propose
        if rng.random() < 0.7:
            cand = biased_perturb_h(cur_h, rng, scale=0.15)

def local_improve(h, rng, max_iter=5000):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(best_h)
    for _ in range(max_iter):
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k = np.argmax(corr) - (n-1)
        if k == 0:
            break
        # find index i where product is max for this shift
        i0 = max(0, -k)
        i1 = min(n, n - k)
        idx = np.arange(i0, i1)
        prod = best_h[idx] * (1 - best_h[idx + k])
        if prod.size == 0:
            break
        i = idx[np.argmax(prod)]
        j = i + k
        # Move epsilon from i (source) to j (dest)

import time, math, numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# basic utilities
# ------------------------------------------------------------
def compute_c5(h):
    """Discrete C5 for step function h (dx = 2/n)."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)

def enforce_sum(h, target, rng):
    """Adjust h in‑place so that sum(h)=target while 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    """Generate a random feasible h with sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    enforce_sum(h, n / 2.0, rng)
    return h

def perturb_h(h, rng, scale=0.2):
    """Transfer ε between two coordinates (preserves sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

def biased_perturb_h(h, rng, scale=0.2):
    """Bias transfer away from positions contributing most to the max overlap."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    k = np.argmax(corr) - (n - 1)               # shift giving the max
    i0 = max(0, -k)
    i1 = min(n, n - k)
    idx = np.arange(i0, i1)
    prod = h[idx] * (1 - h[idx + k])
    if prod.size == 0:
        return perturb_h(h, rng, scale)
    thr_hi = np.quantile(prod, 0.70)
    thr_lo = np.quantile(prod, 0.30)
    high = idx[prod > thr_hi]
    low = idx[prod < thr_lo]
    if high.size == 0 or low.size == 0:
        return perturb_h(h, rng, scale)
    i = rng.choice(high)
    j = rng.choice(low)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

# ------------------------------------------------------------
# smooth max (softmax) for SLSQP
# ------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp, scaled back
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=200):
    """SLSQP with smooth approximation of max."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {"type": "eq", "fun": sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

def long_sa(start_h, rng, allocated_s, start_time):
    cur_h = start_h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    init_temp = 0.05
    final_temp = 1e-7

    while True:
        now = time.time()
        elapsed = now - start_time
        if elapsed >= allocated_s:
            break
        frac = elapsed / allocated_s
        T = init_temp * ((final_temp / init_temp) ** frac)  # geometric cooling
        # adaptive move size
        scale = max(0.01, min(0.3, T * 5.0))
        if rng.random() < 0.7:
            cand = biased_perturb_h(cur_h, rng, scale=scale)
        else:
            cand = perturb_h(cur_h, rng, scale=scale)
        cand_c = compute_c5(cand)
        diff = cand_c - cur_c
        if diff < 0 or rng.random() < math.exp(-diff / max(T, 1e-12)):
            cur_h, cur_c = cand, cand_c
            if cur_c < best_c:

        if rng.random() < 0.001:

            except Exception:
                pass
    return best_h, best_c
