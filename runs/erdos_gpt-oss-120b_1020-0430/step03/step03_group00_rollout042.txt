Projection function:

def project_onto_simplex_bounds(v, target):
    # Euclidean projection onto bounded simplex of sum target and each component in [0,1].
    # Approach: Use algorithm from "Projection onto a Simplex" with upper bound.
    # We can transform variable to w = v - lamda and clip to [0,1] then adjust lambda.
    # Simpler iterative:
    w = np.clip(v, 0.0, 1.0)
    # compute diff = sum(w) - target
    # if diff close to 0: done
    # else adjust free variables.
    # Use while loop:
    while True:
        cur_sum = w.sum()
        diff = cur_sum - target
        if abs(diff) <= 1e-12:
            break
        # indices where w not at bounds
        free = np.where((w > 0) & (w < 1))[0]
        if free.size == 0:
            # all at bounds, cannot adjust; distribute diff uniformly then clip again
            w = np.clip(w - diff / len(w), 0.0, 1.0)
            continue
        delta = diff / free.size
        w[free] -= delta
        # clip again
        w = np.clip(w, 0.0, 1.0)
    return w

import time
import numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None


def project_onto_simplex_bounds(v, target):
    """Euclidean projection of v onto {0<=x_i<=1, sum x_i = target}."""
    # binary search for lambda such that sum_i clip(v_i - lam,0,1) = target
    lo, hi = -5.0, 5.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # tiny correction for numerical drift
    diff = w.sum() - target
    if abs(diff) > 1e-10:
        free = np.where((w > 1e-12) & (w < 1 - 1e-12))[0]
        if free.size:
            w[free] -= diff / free.size
            w = np.clip(w, 0.0, 1.0)
    return w


def random_feasible_vector(N, rng):
    """Uniform random point on the bounded simplex."""
    v = rng.random(N)
    v = v / v.sum() * (N / 2.0)
    return project_onto_simplex_bounds(v, N / 2.0)

def compute_c5(h):
    """Exact C5 for a discrete step function."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / h.size)


# ---------- smooth‑max objective & gradient ----------
def smooth_max_and_grad(h, beta, dx):
    """Log‑sum‑exp approximation of the max overlap and its gradient."""
    N = h.size
    O_raw = np.correlate(h, 1.0 - h, mode="full")  # without dx
    maxO = O_raw.max()
    # stable log‑sum‑exp
    exp_term = np.exp(beta * (O_raw - maxO))
    sum_exp = exp_term.sum()
    w = exp_term / sum_exp  # weights sum to 1

    # objective (scaled by dx)
    smooth_max = (maxO + np.log(sum_exp)) / beta
    obj = smooth_max * dx

    # gradient of smooth max (already scaled by dx)
    grad = np.full(N, dx)  # derivative of the linear sum part
    # w corresponds to shifts -(N-1) … (N-1)
    for idx, weight in enumerate(w):
        s = idx - (N - 1)          # actual shift
        L = N - abs(s)
        if L <= 0:

        if s >= 0:
            grad[:L] -= wdx * h[s:s + L]
            grad[s:s + L] -= wdx * h[:L]
        else:
            t = -s
            grad[t:t + L] -= wdx * h[:L]
            grad[:L] -= wdx * h[t:t + L]
    return obj, grad

    return np.sum(x) - CURRENT_TARGET

    smooth = lse / beta
    return smooth * (2.0 / N)

def hill_climb_targeted(h_start, rng, max_iters, start_step, target_sum):
    h = h_start.copy()
    N = h.size
    best_h = h.copy()
    best_c = compute_c5(best_h)
    step = start_step
    decay = max(1, max_iters // 10)

    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr)
        k = idx - (N - 1)
        # gradient of worst shift
        grad = np.ones(N)
        if k >= 0:
            grad[: N - k] -= h[k:]
            grad[k:] -= h[: N - k]
        else:
            kk = -k
            grad[: N - kk] -= h[kk:]
            grad[kk:] -= h[: N - kk]

        # move mass from positive to negative gradient positions
        pos = np.where((grad > 1e-9) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-9) & (h < 1.0 - 1e-12))[0]
        if pos.size and neg.size:
            i = rng.choice(pos)
            j = rng.choice(neg)
            max_eps = min(h[i], 1.0 - h[j])
            eps = max_eps * step * rng.random()
            new_h = h.copy()
            new_h[i] -= eps
            new_h[j] += eps
        else:
            new_h = h.copy()

        c_new = compute_c5(new_h)
        if c_new < best_c:
            best_c, best_h = c_new, new_h.copy()
        if c_new < compute_c5(h):
            h = new_h
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

    best_c = compute_c5(best_h)

    # ---- smooth‑max gradient descent ------------------------------------
    beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0, 800.0, 2000.0, 5000.0]
    for beta in beta_schedule:
        base_lr = 0.6 / np.sqrt(beta)
        max_iters = 600   # generous but limited; will be cut by time budget
        for it in range(max_iters):
            if time.time() - start_time > budget_s * 0.86:
                break
            obj, grad = smooth_max_and_grad(h, beta, dx)
            lr = base_lr / (1.0 + it * 0.001)
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_onto_simplex_bounds(h_new, N / 2.0)
            c_val = compute_c5(h_new)
            if c_val < best_c:
                best_c, best_h = c_val, h_new.copy()
            h = h_new

    # ---- local targeted hill‑climb ------------------------------------
    while time.time() - start_time < budget_s * 0.90:
        h_local, c_local = hill_climb_targeted(
            best_h, rng, max_iters=1200, start_step=0.4, target_sum=N / 2.0
        )
        if c_local < best_c:
            best_c, best_h = c_local, h_local.copy()
        break   # one pass is sufficient for the remaining time budget

    # ---- SLSQP refinement with higher β ---------------------------------
    for beta in [8000.0, 15000.0]:
        if time.time() - start_time > budget_s * 0.94:
            break
        try:
            h_ref = slsqp_optimize(best_h, beta, max_iter=500)
        except Exception:
            continue
        c_ref = compute_c5(h_ref)
        if c_ref < best_c:
            best_c, best_h = c_ref, h_ref.copy()

    # ---- final tiny random tweaks --------------------------------------
    while time.time() - start_time < budget_s * 0.99:
        i, j = rng.integers(N, size=2)
        if i == j:
            continue
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0.0:
            continue
        eps = max_eps * 0.02 * rng.random()
        h_new = best_h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()

    # ---- final clean‑up -------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_simplex_bounds(best_h, N / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(N)
