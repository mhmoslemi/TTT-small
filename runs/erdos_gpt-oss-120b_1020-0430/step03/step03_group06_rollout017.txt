def compute_c5(h):
    """Exact C5 for discretized h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng):
    """Adjust h in-place so that sum(h) == target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-12:
            continue
        h[i] += add
        diff = target - h.sum()
    # tiny final fix
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(1000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            if abs(add) < 1e-12:
                continue
            h[i] += add
            diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Create a random vector in [0,1]^n with sum = n/2."""
    v = rng.exponential(1.0, size=n)
    v = v / v.sum() * (n / 2.0)
    h = np.clip(v, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


# ----------------------------------------------------------------------
# SLSQP refinement (smooth surrogate)
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def _eq_sum(x):
    return np.sum(x) - CURRENT_TARGET


def _soft_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _eq_sum}
    res = minimize(_soft_obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # repair any tiny sum error
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            if abs(add) < 1e-12:
                continue
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ----------------------------------------------------------------------
# Deterministic descent targeting the worst shift
# ----------------------------------------------------------------------
def descent_max_shift(h, step_start, max_iters, rng, time_limit, start_time):
    n = h.size
    step = step_start
    best_c = compute_c5(h)
    no_imp = 0
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        corr = np.correlate(h, 1 - h, mode="full")
        idx = int(np.argmax(corr))
        k = idx - (n - 1)          # shift corresponding to worst lag
        # gradient g_i = 1 - h_{i+k} (or 1 if out of bounds)
        if k >= 0:
            g = np.empty(n)
            g[:n - k] = 1 - h[k:]
            g[n - k:] = 1.0
        else:
            kk = -k
            g = np.empty(n)
            g[kk:] = 1 - h[:n - kk]
            g[:kk] = 1.0
        i_max = int(np.argmax(g))
        i_min = int(np.argmin(g))
        if i_max == i_min:
            i_max = rng.integers(0, n)
            i_min = rng.integers(0, n)
            while i_min == i_max:
                i_min = rng.integers(0, n)
        eps = min(step, h[i_max], 1.0 - h[i_min])
        if eps > 0:
            h[i_max] -= eps
            h[i_min] += eps
        # occasional tiny random exchange
        if (it & 31) == 0 and rng.random() < 0.2:
            a = rng.integers(0, n)
            b = rng.integers(0, n)
            while b == a:
                b = rng.integers(0, n)
            eps2 = min(step * 0.5, h[a], 1.0 - h[b])
            if eps2 > 0:
                h[a] -= eps2
                h[b] += eps2
        new_c = compute_c5(h)
        if new_c < best_c - 1e-12:
            best_c = new_c
            step = max(step * 0.95, 1e-4)
            no_imp = 0
        else:
            step = min(step * 1.05, 0.1)
            no_imp += 1
        if no_imp >= 200:
            break
    return h, best_c


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # initialise
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 180
        h0 = random_feasible_vector(n, rng)
    target = n / 2.0
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    beta_list = [10.0, 40.0, 120.0, 400.0, 800.0]

    # main optimisation loop
    while time.time() - start_time < budget_s * 0.92:
        # start point: perturb best or random
        if rng.random() < 0.7:
            cur = best_h.copy()
            for _ in range(5):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                while j == i:
                    j = rng.integers(0, n)
                eps = min(0.05, cur[i], 1.0 - cur[j])
                if eps > 0:
                    cur[i] -= eps
                    cur[j] += eps
        else:
            cur = random_feasible_vector(n, rng)

        # deterministic descent on the worst shift
        cur, cur_c = descent_max_shift(cur, step_start=0.2,
                                       max_iters=2500,
                                       rng=rng,
                                       time_limit=budget_s * 0.92,
                                       start_time=start_time)

        # binary rounding checkpoint
        sorted_idx = np.argsort(-cur)
        bin_h = np.zeros_like(cur)
        bin_h[sorted_idx[:int(target)]] = 1.0
        bin_c = compute_c5(bin_h)
        if bin_c < cur_c:
            cur, cur_c = bin_h, bin_c

        # SLSQP refinement if promising
        if cur_c < best_c:
            for beta in beta_list:
                if time.time() - start_time > budget_s * 0.92:
                    break
                try:
                    refined = slsqp_optimize(cur, beta, max_iter=300)
                except Exception:
                    continue
                refined_c = compute_c5(refined)
                if refined_c < cur_c:
                    cur, cur_c = refined, refined_c
                if refined_c < best_c:
                    best_h, best_c = refined.copy(), refined_c
            if cur_c < best_c:
                best_h, best_c = cur.copy(), cur_c

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
