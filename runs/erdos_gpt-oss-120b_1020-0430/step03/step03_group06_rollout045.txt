    # ------------------- helpers -------------------
    def compute_c5(h):
        """Exact discrete C5 for a step function h (dx = 2/len(h))."""
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * (2.0 / h.size)

    def enforce_sum(h, target):
        """Adjust h so that Σh = target while 0≤h≤1 (simple iterative repair)."""
        diff = target - h.sum()
        n = h.size
        for _ in range(3000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            if abs(add) > 1e-12:
                h[i] += add
                diff = target - h.sum()
        return np.clip(h, 0.0, 1.0)

    def binary_from_continuous(x):
        """Return a binary vector with exactly half the entries set to 1, taken from the largest values of x."""
        n = x.size
        target = int(n // 2)
        idx = np.argsort(x)[::-1][:target]
        bin_vec = np.zeros_like(x)
        bin_vec[idx] = 1.0
        return bin_vec

    def hill_climb_swaps(start_h, max_iters, temp_start, temp_end):
        """Simulated‑annealing style swap hill‑climb on a binary vector."""
        h = start_h.copy()
        n = h.size
        c_h = compute_c5(h)
        best_h = h.copy()
        best_c = c_h
        for it in range(max_iters):
            # linear temperature schedule
            t = temp_start + (temp_end - temp_start) * (it / max_iters)
            # pick random 1 and 0 positions

            zeros = np.flatnonzero(h < 0.5)
            if ones.size == 0 or zeros.size == 0:
                break

            j = rng.choice(zeros)

            if accept:

                if c_h < best_c:

                    best_h = h.copy()

                        diff = target - x.sum()

            except Exception:
