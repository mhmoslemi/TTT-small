import time, numpy as np

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible(n, rng):
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, target, rng)

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_hill_climb(h, rng, max_iters=2000, step_scale=0.5):
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    scale = step_scale
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        i = np.argmax(grad)
        j = np.argmin(grad)
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0:
                eps = rng.random() * max_eps * scale
                h[i] -= eps
                h[j] += eps
        # enforce bounds
        h = np.clip(h, 0.0, 1.0)
        # enforce sum exactly
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # simple correction: adjust random index
            i_adj = rng.integers(0, n)
            add = min(diff, 1.0 - h[i_adj]) if diff > 0 else max(diff, -h[i_adj])
            h[i_adj] += add
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        if (it + 1) % decay == 0:
            scale = max(scale * 0.7, 1e-4)
    return best_h, best_c

def blend_crossover(p1, p2, rng, alpha_range=(-0.5, 1.5)):
    alpha = rng.uniform(*alpha_range)
    child = p1 * (1 - alpha) + p2 * alpha
    return child

def mutate(h, rng, rate=0.1, delta=0.2):
    n = h.size
    mask = rng.random(n) < rate
    if np.any(mask):
        h[mask] += rng.uniform(-delta, delta, size=np.sum(mask))
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # Determine base n
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        base_n = h0.size
    else:
        base_n = 192
        h0 = random_feasible(base_n, rng)

    # Ensure base_h feasible
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = enforce_sum(h0, base_n / 2.0, rng)
    c0 = compute_c5(h0)
    best_h, best_c, best_n = h0, c0, base_n

    # GA parameters
    pop_size = 40
    elite_frac = 0.1
    elite_cnt = max(1, int(pop_size * elite_frac))
    max_gen = 2000
    mutation_rate = 0.15
    mutation_delta = 0.2
    crossover_prob = 0.7
    tournament_k = 3

    # Initialize population
    pop = np.empty((pop_size, base_n), dtype=float)
    pop[0] = best_h
    for i in range(1, pop_size):
        pop[i] = random_feasible(base_n, rng)
    fitness = np.empty(pop_size, dtype=float)
    for i in range(pop_size):
        fitness[i] = compute_c5(pop[i])
        if fitness[i] < best_c:
            best_h, best_c = pop[i].copy(), fitness[i]
    generation = 0
    while time.time() - start < budget_s * 0.85 and generation < max_gen:
        # sort indices by fitness
        idx_sorted = np.argsort(fitness)
        new_pop = pop[idx_sorted[:elite_cnt]].copy()
        new_fitness = fitness[idx_sorted[:elite_cnt]].copy()
        # generate offspring
        while len(new_pop) < pop_size:
            # selection
            sel = rng.integers(0, pop_size, size=tournament_k)
            p1_idx = sel[np.argmin(fitness[sel])]
            sel = rng.integers(0, pop_size, size=tournament_k)
            p2_idx = sel[np.argmin(fitness[sel])]
            p1 = pop[p1_idx]
            p2 = pop[p2_idx]
            if rng.random() < crossover_prob:
                child = blend_crossover(p1, p2, rng)
            else:
                child = p1.copy()
            child = mutate(child, rng, rate=mutation_rate, delta=mutation_delta)
            # enforce bounds
            child = np.clip(child, 0.0, 1.0)
            child = enforce_sum(child, base_n / 2.0, rng)
            f = compute_c5(child)
            new_pop = np.vstack([new_pop, child])
            new_fitness = np.append(new_fitness, f)
            if f < best_c:
                best_h, best_c = child.copy(), f
        # truncate to pop_size
        pop = new_pop[:pop_size]
        fitness = new_fitness[:pop_size]
        # adapt mutation delta
        if generation % 100 == 0 and generation > 0:
            # reduce delta if improvement slowed
            if best_c >= c0 * 0.999:
                mutation_delta *= 0.9
            c0 = best_c
        generation += 1

    # Local refinement on best_h
    if time.time() - start < budget_s * 0.95:
        best_h, best_c = guided_hill_climb(best_h, rng, max_iters=3000, step_scale=0.5)

    # Upsample if time left
    if time.time() - start < budget_s * 0.98 and best_n * 2 <= 384:
        up_h = np.repeat(best_h, 2)
        up_n = up_h.size
        up_h = enforce_sum(up_h, up_n / 2.0, rng)
        # refine upsampled
        up_h, up_c = guided_hill_climb(up_h, rng, max_iters=2000, step_scale=0.5)
        if up_c < best_c:
            best_h, best_c, best_n = up_h, up_c, up_n

    # Final tiny random transfers
    while time.time() - start < budget_s * 0.999:
        grad = worst_gradient(best_h)
        i = np.argmax(grad)
        j = np.argmin(grad)
        if i != j:
            max_eps = min(best_h[i], 1.0 - best_h[j])
            if max_eps > 0:
                eps = rng.random() * max_eps * 0.02
                best_h[i] -= eps
                best_h[j] += eps
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
        c = compute_c5(best_h)
        if c < best_c:
            best_c = c

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)

import time, numpy as np

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible(n, rng):
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, target, rng)

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_hill_climb(h, rng, max_iters=2000, step_scale=0.5):
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    scale = step_scale
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            mx = min(h[i], 1.0 - h[j])
            if mx > 0:
                eps = rng.random() * mx * scale
                h[i] -= eps
                h[j] += eps
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, target, rng)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        if (it + 1) % decay == 0:
            scale = max(scale * 0.7, 1e-4)
    return best_h, best_c

def blend_crossover(p1, p2, rng, ar=(-0.5, 1.5)):
    a = rng.uniform(*ar)
    return p1 * (1 - a) + p2 * a

def mutate(v, rng, rate=0.15, delta=0.2):
    mask = rng.random(v.size) < rate
    if mask.any():
        v[mask] += rng.uniform(-delta, delta, size=mask.sum())
    return v

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h, best_c, best_n = None, np.inf, None

    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        base_n = h0.size
    else:
        base_n = 192
        h0 = random_feasible(base_n, rng)

    h0 = np.clip(h0, 0.0, 1.0)
    h0 = enforce_sum(h0, base_n / 2.0, rng)
    best_h, best_c, best_n = h0, compute_c5(h0), base_n

    pop_size = 40
    elite_cnt = max(1, int(pop_size * 0.1))
    max_gen = 2000
    mut_rate, mut_delta = 0.15, 0.2
    cross_prob = 0.7
    tour_k = 3

    pop = np.empty((pop_size, base_n), dtype=float)
    pop[0] = best_h
    for i in range(1, pop_size):
        pop[i] = random_feasible(base_n, rng)
    fitness = np.array([compute_c5(p) for p in pop])
    if fitness.min() < best_c:
        idx = fitness.argmin()
        best_h, best_c = pop[idx].copy(), fitness[idx]

    gen = 0
    while time.time() - start < budget_s * 0.85 and gen < max_gen:
        order = np.argsort(fitness)
        new_pop = [pop[i] for i in order[:elite_cnt]]
        new_fit = [fitness[i] for i in order[:elite_cnt]]
        while len(new_pop) < pop_size:
            idx = rng.integers(0, pop_size, size=tour_k)
            p1 = pop[idx[np.argmin(fitness[idx])]]
            idx = rng.integers(0, pop_size, size=tour_k)
            p2 = pop[idx[np.argmin(fitness[idx])]]
            child = blend_crossover(p1, p2, rng) if rng.random() < cross_prob else p1.copy()
            child = mutate(child, rng, rate=mut_rate, delta=mut_delta)
            child = np.clip(child, 0.0, 1.0)
            child = enforce_sum(child, base_n / 2.0, rng)
            f = compute_c5(child)
            new_pop.append(child)
            new_fit.append(f)
            if f < best_c:
                best_h, best_c = child.copy(), f
        pop = np.array(new_pop[:pop_size])
        fitness = np.array(new_fit[:pop_size])
        if gen % 100 == 0 and gen > 0:
            mut_delta *= 0.9
        gen += 1

    if time.time() - start < budget_s * 0.95:
        best_h, best_c = guided_hill_climb(best_h, rng, max_iters=3000, step_scale=0.5)

    if (time.time() - start < budget_s * 0.98) and best_n * 2 <= 384:
        up_n = best_n * 2
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, up_n, endpoint=False)
        up_h = np.interp(x_new, x_old, best_h)
        up_h = np.clip(up_h, 0.0, 1.0)
        up_h = enforce_sum(up_h, up_n / 2.0, rng)
        up_h, up_c = guided_hill_climb(up_h, rng, max_iters=2000, step_scale=0.5)
        if up_c < best_c:
            best_h, best_c, best_n = up_h, up_c, up_n

    while time.time() - start < budget_s * 0.999:
        grad = worst_gradient(best_h)
        i, j = int(np.argmax(grad)), int(np.argmin(grad))
        if i != j:
            mx = min(best_h[i], 1.0 - best_h[j])
            if mx > 0:
                eps = rng.random() * mx * 0.02
                best_h[i] -= eps
                best_h[j] += eps
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
        c = compute_c5(best_h)
        if c < best_c:
            best_c = c

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
