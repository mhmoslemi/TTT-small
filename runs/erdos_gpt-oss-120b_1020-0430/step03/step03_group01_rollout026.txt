def propose_targeted(h, rng, epsilon, target_n):

    # compute max lag index
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)  # index in range [0, 2n-2]
    lag = max_idx - (h.size - 1)
    # Only consider positive lag? Could also use negative.
    # If lag == 0, maybe pick second best.
    if lag == 0:
        # pick second max
        sorted_idx = np.argsort(corr)[::-1]
        for idx in sorted_idx:
            if idx != max_idx:
                lag = idx - (h.size - 1)
                break
    new_h = h.copy()
    n = h.size
    # choose i where i and i+lag within bounds
    if lag >= 0:
        possible = np.arange(n - lag)
        i = rng.choice(possible)
        j = i + lag
    else:
        lag_abs = -lag
        possible = np.arange(lag_abs, n)
        i = rng.choice(possible)
        j = i - lag_abs
    # max transfer = min(h[i], 1 - h[j])
    max_eps = min(new_h[i], 1 - new_h[j])
    if max_eps <= 0:
        return h  # no feasible move
    eps = min(epsilon, max_eps)
    new_h[i] -= eps
    new_h[j] += eps
    # already sum preserved.
    return new_h

import time
import numpy as np
from scipy.optimize import minimize

# global placeholders for surrogate optimization
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    # compute smooth approximation of max_k correlation
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log-sum-exp trick
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def compute_c5_and_lag(h):
    """Return (C5 value, lag where it is attained) for vector h."""
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr)
    max_corr = corr[idx]
    lag = idx - (h.size - 1)
    return max_corr * (2.0 / h.size), lag

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h so that sum is target while 0<=h<=1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        # small uniform correction
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    """Generate random h with values in [0,1] and sum=n/2."""
    h = np.full(n, 0.5)
    # random swaps to spread mass
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

def perturb_transfer(h, rng, epsilon):
    """Transfer ε from one index to another, preserving sum."""
    n = h.size
    i, j = rng.choice(n, size=2, replace=False)
    max_eps = min(h[i], 1.0 - h[j])
    eps = min(epsilon, max_eps)
    if eps <= 0:
        # try opposite direction
        max_eps_rev = min(h[j], 1.0 - h[i])
        eps_rev = min(epsilon, max_eps_rev)
        if eps_rev <= 0:
            return h.copy()
        new_h = h.copy()
        new_h[j] -= eps_rev
        new_h[i] += eps_rev
        return new_h
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def perturb_targeted_lag(h, lag, epsilon, rng):
    """Move ε from index i to i+lag (or i-lag) to reduce worst overlap."""
    n = h.size
    if lag == 0:
        return perturb_transfer(h, rng, epsilon)
    if lag > 0:
        i = rng.integers(0, n - lag)
        j = i + lag
    else:
        i = rng.integers(-lag, n)
        j = i + lag  # lag negative
    max_eps = min(h[i], 1.0 - h[j])
    eps = min(epsilon, max_eps)
    if eps <= 0:
        return h.copy()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def slsqp_optimize(h0, beta, max_iter=300):
    """Smooth max surrogate min via SLSQP."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x

def hill_climb(h, rng, iters=2000, n_trials=5, step=0.3):
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    for _ in range(iters):
        best_local_h = None
        best_local_c = cur_c
        for _ in range(n_trials):
            cand = perturb_transfer(cur_h, rng, step)
            c = compute_c5(cand)
            if c < best_local_c:
                best_local_c = c
                best_local_h = cand
        if best_local_h is not None:
            cur_h = best_local_h
            cur_c = best_local_c
            step = max(step * 0.99, 0.01)
    return cur_h, cur_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # --- initial vector ----------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    cur_h = h0.copy()
    cur_c, cur_lag = compute_c5_and_lag(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c

    # --- simulated annealing -----------------------------------------------
    max_time = budget_s * 0.75
    max_temp = 0.005
    min_temp = 0.0001

    while True:
        elapsed = time.time() - start
        if elapsed >= max_time:
            break
        # temperature schedule (linear)
        t_frac = elapsed / max_time
        temp = max_temp * (1 - t_frac) + min_temp * t_frac
        epsilon = 0.2 * (temp / max_temp) + 0.001

        if rng.random() < 0.55:
            # targeted move using current worst lag
            new_h = perturb_targeted_lag(cur_h, cur_lag, epsilon, rng)
        else:
            new_h = perturb_transfer(cur_h, rng, epsilon)

        new_c, new_lag = compute_c5_and_lag(new_h)

        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(temp, 1e-12)):
            cur_h, cur_c, cur_lag = new_h, new_c, new_lag
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()

    # --- local hill climbing ------------------------------------------------
    if time.time() - start < budget_s * 0.85:
        hc_h, hc_c = hill_climb(best_h, rng, iters=3000, n_trials=8, step=0.4)
        if hc_c < best_c:
            best_c, best_h = hc_c, hc_h.copy()

    # --- SLSQP refinement ---------------------------------------------------
    beta_seq = [100, 300, 1000, 3000, 5000]
    for beta in beta_seq:
        if time.time() - start > budget_s * 0.9:
            break
        try:
            refined = slsqp_optimize(best_h, beta, max_iter=500)
        except Exception:
            continue
        c_ref = compute_c5(refined)
        if c_ref < best_c:
            best_c, best_h = c_ref, refined.copy()

    # --- optional up‑sampling if time permits -------------------------------
    remaining = budget_s - (time.time() - start)
    if remaining > 20.0:
        target_n = min(720, max(480, n * 2))
        if target_n != n:
            old_x = (np.arange(n) + 0.5) * (2.0 / n)
            new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
            h_up = np.interp(new_x, old_x, best_h)
            h_up = enforce_sum(h_up, target_n / 2.0, rng)

            cur_c, cur_lag = compute_c5_and_lag(cur_h)
            up_start = time.time()
            max_up_time = remaining - 10.0
            while time.time() - up_start < max_up_time:
                elapsed = time.time() - up_start
                temp = max_temp * (1 - elapsed / max_up_time) + min_temp * (elapsed / max_up_time)
                epsilon = 0.1 * (temp / max_temp) + 0.001
                if rng.random() < 0.5:
                    new_h = perturb_targeted_lag(cur_h, cur_lag, epsilon, rng)
                else:
                    new_h = perturb_transfer(cur_h, rng, epsilon)
                new_c, new_lag = compute_c5_and_lag(new_h)
                if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(temp, 1e-12)):
                    cur_h, cur_c, cur_lag = new_h, new_c, new_lag

            # final SLSQP on up‑sampled

                except Exception:

                c_ref = compute_c5(ref)

from scipy.optimize import minimize

# globals for the smooth surrogate used by SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1‑x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5_and_lag(h):
    """Exact discrete C₅ and the lag where the maximum is attained."""
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr)
    max_corr = corr[idx]
    lag = idx - (h.size - 1)
    return max_corr * (2.0 / h.size), lag


def compute_c5(h):
    """Exact discrete C₅."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h so that Σh = target while keeping 0≤h≤1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        # uniform correction

def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    h = np.full(n, 0.5)

        j = rng.integers(0, n)

            diff = target - h.sum()
    return h


def perturb_transfer(h, rng, epsilon):
    """Transfer ε from a random index to another (preserves sum)."""
    n = h.size
    i, j = rng.choice(n, size=2, replace=False)
    max_eps = min(h[i], 1.0 - h[j])
    eps = min(epsilon, max_eps)
    if eps <= 0:
        # try opposite direction
        max_eps_rev = min(h[j], 1.0 - h[i])
        eps_rev = min(epsilon, max_eps_rev)
        if eps_rev <= 0:
            return h.copy()
        new_h = h.copy()
        new_h[j] -= eps_rev
        new_h[i] += eps_rev
        return new_h
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def perturb_targeted_lag(h, lag, epsilon, rng):
    """Move ε from i to i+lag (or i‑lag) to reduce the worst overlap."""
    n = h.size
    if lag == 0:
        return perturb_transfer(h, rng, epsilon)
    if lag > 0:
        i = rng.integers(0, n - lag)
        j = i + lag
    else:

        j = i + lag
    max_eps = min(h[i], 1.0 - h[j])
    eps = min(epsilon, max_eps)
    if eps <= 0:
        return h.copy()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def slsqp_optimize(h0, beta, max_iter=300):
    """Refine h via SLSQP on a smooth surrogate of the max‑overlap."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

def hill_climb(h, rng, iters=2000, n_trials=5, step=0.3):

    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    for _ in range(iters):
        best_local_h = None
        best_local_c = cur_c

                best_local_h = cand
        if best_local_h is not None:
            cur_h = best_local_h
            cur_c = best_local_c
            step = max(step * 0.99, 0.01)
    return cur_h, cur_c
