def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def max_shift_and_corr(h):

    k = idx - (h.size - 1)

def perturb_targeted_from_grad(h, grad, rng, scale=0.3):
    # picks i with grad>0 and h[i]>0; j with grad<0 and h[j]<1.
    pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
    neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
    if pos.size == 0 or neg.size == 0:
        return perturb_transfer(h, rng, max_scale=scale)
    i = rng.choice(pos)
    j = rng.choice(neg)
    max_eps = min(h[i], 1 - h[j])

        return perturb_transfer(h, rng, max_scale=scale)
    eps = max_eps * scale * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h in-place so that sum(h) == target while staying in [0,1]."""
    diff = target - h.sum()
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h


def compute_c5(h):
    """Discrete C5 for vector h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def perturb_transfer(h, rng, max_scale=0.5):
    """Transfer a small amount ε from i to j preserving the sum."""
    n = h.size
    pos = np.where(h > 1e-12)[0]
    neg = np.where(h < 1.0 - 1e-12)[0]
    if pos.size == 0 or neg.size == 0:
        return h.copy()
    i = rng.choice(pos)
    j = rng.choice(neg)
    if i == j:
        alt = neg[neg != i]
        if alt.size == 0:
            return h.copy()
        j = rng.choice(alt)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * max_scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def max_shift_and_corr(h):
    """Return shift k (as in h[i+k]) achieving maximal correlation and the value."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    k = idx - (n - 1)
    return k, corr[idx]


def grad_worst(h, k):
    """Gradient of f(k)=∑ h[i]*(1-h[i+k]) with respect to h."""
    n = h.size
    grad = np.zeros(n)
    if k >= 0:
        grad[: n - k] += 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    else:
        kk = -k
        grad[: n - kk] += 1.0 - h[kk:]
        grad[kk:] -= h[: n - kk]
    return grad


def perturb_targeted(h, grad, rng, scale=0.5):
    """Move mass from positive‑gradient to negative‑gradient coordinates."""
    pos = np.where((grad > 1e-9) & (h > 1e-12))[0]
    neg = np.where((grad < -1e-9) & (h < 1.0 - 1e-12))[0]
    if pos.size == 0 or neg.size == 0:
        return perturb_transfer(h, rng, max_scale=scale)
    i = rng.choice(pos)
    j = rng.choice(neg)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return perturb_transfer(h, rng, max_scale=scale)
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def hill_climb_targeted(h_start, rng, max_iters=4000, step=0.5):
    """Deterministic polishing using the worst‑shift gradient."""
    h = h_start.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(best_h)
    cur_c = best_c
    for it in range(max_iters):
        k, _ = max_shift_and_corr(h)
        grad = grad_worst(h, k)
        new_h = perturb_targeted(h, grad, rng, scale=step)
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c, best_h = new_c, new_h.copy()
        if new_c < cur_c:
            h, cur_c = new_h, new_c
        if (it + 1) % 800 == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def simulated_annealing(h_init, time_limit, rng):
    """Simple SA that respects the sum constraint by transferring mass."""
    h_cur = h_init.copy()
    cur_c = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_c = cur_c
    start = time.time()
    T0, T_end = 0.10, 1e-5
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        frac = elapsed / time_limit
        T = T0 * (T_end / T0) ** frac
        scale = min(0.5, T * 10.0)          # larger steps while hot
        h_new = perturb_transfer(h_cur, rng, max_scale=scale)
        new_c = compute_c5(h_new)
        delta = new_c - cur_c
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            h_cur, cur_c = h_new, new_c
            if new_c < best_c:
                best_c, best_h = new_c, h_new.copy()
    return best_h, best_c

    # ---- initial vector -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        # fallback: random feasible vector of length 360
        n0 = 360
        h0 = rng.random(n0)
        h0 = enforce_sum(h0, n0 / 2.0, rng)

    # ---- up‑sample to 720 points ----------------------------------------
    n_target = 720
    if h0.size != n_target:
        xs0 = (np.arange(h0.size) + 0.5) * (2.0 / h0.size)
        xs = (np.arange(n_target) + 0.5) * (2.0 / n_target)
        h = np.interp(xs, xs0, h0)
    else:
        h = h0.copy()
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, n_target / 2.0, rng)

    best_h, best_c = h.copy(), compute_c5(h)

    # ---- time budgeting -------------------------------------------------
    total_budget = budget_s
    elapsed = time.time() - start_time
    remaining = total_budget - elapsed
    # allocate ~70% of remaining time to SA
    sa_budget = max(0.0, remaining * 0.70)

    # ---- simulated annealing (multiple restarts) -----------------------
    if sa_budget > 5.0:
        n_restarts = max(1, int(sa_budget // 30))
        n_restarts = min(n_restarts, 5)
        per_restart = sa_budget / n_restarts
        for _ in range(n_restarts):
            # start from a slight perturbation of current best or a fresh random point
            if rng.random() < 0.7:
                start_vec = perturb_transfer(best_h, rng, max_scale=0.6)
            else:
                start_vec = rng.random(n_target)
                start_vec = enforce_sum(start_vec, n_target / 2.0, rng)
            cand_h, cand_c = simulated_annealing(start_vec, per_restart, rng)
            if cand_c < best_c:
                best_h, best_c = cand_h.copy(), cand_c

    # ---- deterministic hill‑climb polishing ---------------------------
    elapsed = time.time() - start_time
    if total_budget - elapsed > 8.0:
        pc_h, pc_c = hill_climb_targeted(best_h, rng, max_iters=4000, step=0.5)
        if pc_c < best_c:
            best_h, best_c = pc_h, pc_c

    # ---- SLSQP refinement (soft‑max surrogate) -------------------------
    elapsed = time.time() - start_time
    if total_budget - elapsed > 4.0:
        n = best_h.size
        target_sum = n / 2.0
        bounds = [(0.0, 1.0)] * n
        def sum_eq(x, target=target_sum):
            return np.sum(x) - target
        cons = {"type": "eq", "fun": sum_eq}
        beta_schedule = [50.0, 200.0, 800.0]
        for beta in beta_schedule:
            if time.time() - start_time > total_budget * 0.96:
                break
            def softmax_obj(x, beta_val=beta):
                corr = np.correlate(x, 1.0 - x, mode="full")
                bc = beta_val * corr
                m = np.max(bc)
                # log‑sum‑exp, scaled back by beta
                return (m + np.log(np.sum(np.exp(bc - m)))) / beta_val
            res = minimize(
                softmax_obj,
                best_h,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 300, "ftol": 1e-12, "disp": False},
            )
            x = np.clip(res.x, 0.0, 1.0)
            x = enforce_sum(x, target_sum, rng)
            c_val = compute_c5(x)
            if c_val < best_c:
                best_h, best_c = x.copy(), c_val

    # ---- final micro‑search (tiny random moves) ------------------------
    elapsed = time.time() - start_time
    while time.time() - start_time < total_budget * 0.99:
        trial = perturb_transfer(best_h, rng, max_scale=0.05)
        c_trial = compute_c5(trial)
        if c_trial < best_c:
            best_h, best_c = trial, c_trial

    # ---- final clean‑up -------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = float(compute_c5(best_h))

    return best_h.tolist(), final_c, int(best_h.size)
