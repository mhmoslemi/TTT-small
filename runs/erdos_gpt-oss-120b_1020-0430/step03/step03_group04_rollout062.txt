def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    # Ensure not exceed bounds
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h
```

def gradient_refine(h, rng, step=0.01, max_iter=30):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            # g_i += 1 - h_{i+k}
            g[:upper] += 1.0 - h[k:]
            # g_{i+k} -= h_i
            g[k:] -= h[:upper]
        # step size schedule
        step_sz = step * (0.8 ** it)  # decaying
        h = h - step_sz * g
        # clip to [0,1]
        np.clip(h, 0.0, 1.0, out=h)
        # enforce sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # redistribute diff greedily
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def smoothed_random(n, rng):
    # generate uniform random then smooth via moving average
    h = rng.random(n)
    k = max(3, n // 20)  # smoothing window size
    kernel = np.ones(k) / k
    h = np.convolve(h, kernel, mode='same')
    # clip, adjust sum to target
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):

            h[i] += add
            diff = target - h.sum()
            if abs(diff) < 1e-12:

```
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    # The SLSQP may occasionally fail to converge; set high max_iter.
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum if needed
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x
```

def simulated_annealing(h0, rng, deadline):
    n = h0.size
    target = n / 2.0
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    # initial temperature as 0.05 * cur_c (maybe)
    T0 = max(0.01 * cur_c, 1e-4)
    it = 0
    while time.time() < deadline:
        # adaptive step size decays slowly
        step = max(0.5 * (0.995 ** it), 5e-4)
        cand_h = perturb_h(cur_h, rng, scale=step)
        cand_c = compute_c5(cand_h)
        if cand_c < cur_c:
            accept = True
        else:
            T = T0 * (0.995 ** it) + 1e-12
            accept = rng.random() < np.exp(-(cand_c - cur_c) / T)
        if accept:
            cur_h, cur_c = cand_h, cand_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
        it += 1
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ---------- #
def compute_c5(h):
    """Exact discrete C5 for step function h on [0,2] with uniform spacing."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        cand = perturb_h(h, rng, scale=step)
        c = compute_c5(cand)
        if c < best_c - 1e-12:
            best_c, best_h, h = c, cand.copy(), cand
        if (it + 1) % decay == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


def gradient_refine(h, rng, step=0.01, max_iter=30):
    """Refine by moving against gradient of the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            up = n - k
            g[:up] += 1.0 - h[k:]
            g[k:] -= h[:up]
        step_sz = step * (0.8 ** it)
        h = h - step_sz * g
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


def random_feasible_vector(n, rng):
    """Uniform random vector in [0,1] with exact sum n/2."""
    h = rng.random(n)
    target = n / 2.0
    scale = target / h.sum()
    h *= scale
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h


def smoothed_random(n, rng):
    """Random vector smoothed by a moving average."""
    h = rng.random(n)
    k = max(3, n // 20)
    kernel = np.ones(k) / k
    h = np.convolve(h, kernel, mode="same")
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h


# ---------- SLSQP with soft‑max surrogate ---------- #
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x


# ---------- Simulated annealing ---------- #
def simulated_annealing(h0, rng, deadline):
    n = h0.size
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    T0 = max(0.01 * cur_c, 1e-4)
    it = 0
    while time.time() < deadline:
        step = max(0.5 * (0.995 ** it), 5e-4)
        cand_h = perturb_h(cur_h, rng, scale=step)
        cand_c = compute_c5(cand_h)
        if cand_c < cur_c:
            accept = True
        else:
            T = T0 * (0.995 ** it) + 1e-12
            accept = rng.random() < np.exp(-(cand_c - cur_c) / T)
        if accept:
            cur_h, cur_c = cand_h, cand_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
        it += 1
    return best_h, best_c
