def simulated_annealing(h0, rng, time_limit, start_temp, end_temp):
    h = h0.copy()
    c = compute_c5(h)
    start_time = time.time()
    while time.time() - start_time < time_limit:
        elapsed = time.time() - start_time
        fraction = elapsed / time_limit
        T = start_temp * (end_temp / start_temp) ** fraction
        # neighbor: randomly pick perturb or maybe block swap
        if rng.random() < 0.7:
            h_new = perturb_h(h, rng, scale=0.1)
        else:
            # larger perturb
            h_new = perturb_h(h, rng, scale=0.3)
        c_new = compute_c5(h_new)
        d = c_new - c
        if d < 0 or rng.random() < math.exp(-d / max(T,1e-12)):
            h, c = h_new, c_new
    return h, c

import numpy as np

def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)

def _triangular_h(n):
    """Return the triangular step function on [0,2] with n even points."""
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx                # mid‑point sample locations
    h = 1.0 - np.abs(xs - 1.0)                    # h(x)=1-|x-1|

    target = n / 2.0

    if abs(s - target) > 1e-12:

    return h
