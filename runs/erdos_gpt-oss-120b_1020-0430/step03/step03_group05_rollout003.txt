def compute_c5_fft(h):
    n = h.size
    y = 1.0 - h
    n_fft = 1 << (2*n-1).bit_length()
    H = np.fft.rfft(h, n_fft)

    # conv[k] corresponds to sum_i h[i] * y[k-i] where i in [0,n-1] and k-i in [0,n-1]
    return conv.max() * (2.0 / n)

def binary_local_search(n, rng, max_iters=20000, patience=2000):
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, n//2, replace=False).tolist()
    h[ones] = 1.0
    zeros = [i for i in range(n) if i not in set(ones)]  # using set for speed
    # better: create a set all indices and subtract set(ones)
    zeros = list(set(range(n)) - set(ones))
    # compute C5
    best_c = compute_c5_fft(h)
    best_h = h.copy()
    no_imp = 0
    for it in range(max_iters):
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        c = compute_c5_fft(h)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            # update lists
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            no_imp = 0
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
            no_imp += 1
        if no_imp >= patience:
            break
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utilities ----------
_nfft_cache = {}
def compute_c5(h):
    """Fast C5 using FFT (dx = 2/len(h))."""
    n = h.size
    n_fft = _nfft_cache.get(n)
    if n_fft is None:
        n_fft = 1 << ((2 * n - 1) - 1).bit_length()
        _nfft_cache[n] = n_fft
    H = np.fft.rfft(h, n=n_fft)
    Y = np.fft.rfft(1.0 - h, n=n_fft)
    conv = np.fft.irfft(H * Y, n=n_fft)[: 2 * n - 1]
    return conv.max() * (2.0 / n)


# ---------- SLSQP helpers ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def project_to_feasible(v, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def slsqp_optimize(h0, beta, rng, max_iter=800):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ---------- simple perturbation ----------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=5000, start_step=0.2):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


# ---------- binary local search ----------
def binary_local_search(n, rng, max_iters=30000, patience=3000):
    """Swap a 1 with a 0 to reduce C5. Works only for even n."""
    half = n // 2
    ones = rng.choice(n, half, replace=False).tolist()
    h = np.zeros(n, dtype=float)
    h[ones] = 1.0
    zeros = [i for i in range(n) if i not in set(ones)]
    best_c = compute_c5(h)
    best_h = h.copy()
    no_imp = 0
    for _ in range(max_iters):
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        c = compute_c5(h)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            no_imp = 0
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_imp += 1
        if no_imp >= patience:
            break
    return best_h, best_c


def random_feasible_vector(n, rng):
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_to_feasible(v, n / 2.0)
