def correlate_full(h):
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)

def smooth_obj_and_grad(h, beta):
    n = h.shape[0]
    d = correlate_full(h)
    maxd = d.max()
    shifted = d - maxd
    # Using beta*_shifted may overflow for large beta; but β up to 800 and shift up maybe up to ~0.4? It's fine.
    exp_beta = np.exp(beta * shifted)
    sum_exp = exp_beta.sum()
    w = exp_beta / sum_exp  # softmax weights shape 2n-1

    # compute convolution for grad
    # Length of conv = n + (2n-1) - 1 = 3n-2
    L = 1 << ((n + (2 * n - 1) - 1).bit_length())
    # conv1 of (1-h) with w
    f1 = np.fft.rfft(1 - h, L)
    fw = np.fft.rfft(w, L)
    conv1_full = np.fft.irfft(f1 * fw, L)[: (n + (2 * n - 1) - 1)]
    conv1 = conv1_full[n - 1 : n - 1 + n]

    # conv_rev of h with w_rev
    w_rev = w[::-1]
    f2 = np.fft.rfft(h, L)
    fw_rev = np.fft.rfft(w_rev, L)
    conv_rev_full = np.fft.irfft(f2 * fw_rev, L)[: (n + (2 * n - 1) - 1)]
    conv_rev = conv_rev_full[n - 1 : n - 1 + n]

    grad = conv1 - conv_rev

    # Smooth objective value
    f = (np.log(sum_exp) + maxd) / beta
    return f, grad

def smooth_obj_and_grad(h,beta):
    n = h.shape[0]
    d = correlate_full(h)
    maxd = d.max()
    shifted = d - maxd
    # Compute exp(beta * shifted) safely
    exp_vec = np.exp(beta * shifted)
    sum_exp = exp_vec.sum()
    w = exp_vec / sum_exp
    # fft length
    L = 1 << ((n + (2*n - 1) - 1).bit_length())
    # conv1
    f1 = np.fft.rfft(1 - h, L)
    fw = np.fft.rfft(w, L)
    conv1_full = np.fft.irfft(f1 * fw, L)[: (n + (2*n - 1) - 1)]
    conv1 = conv1_full[n-1 : n-1 + n]
    # conv_rev
    w_rev = w[::-1]
    f2 = np.fft.rfft(h, L)
    fw_rev = np.fft.rfft(w_rev, L)
    conv_rev_full = np.fft.irfft(f2 * fw_rev, L)[: (n + (2*n - 1) - 1)]
    conv_rev = conv_rev_full[n-1 : n-1 + n]
    grad = conv1 - conv_rev
    f = (np.log(sum_exp) + maxd) / beta
    return f, grad

    # final evaluation
    c = compute_c5(h)

import time
import numpy as np

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    n = h.size
    for _ in range(200):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            i = rng.integers(0, n)
            if h[i] >= 1.0:
                continue
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            i = rng.integers(0, n)
            if h[i] <= 0.0:
                continue
            add = max(diff, -h[i])
            h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    v = rng.random(n)
    v = v - v.mean() + 0.5
    v = np.clip(v, 0.0, 1.0)
    return enforce_sum(v, n / 2.0, rng)

def correlate_full(h):
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv

def compute_c5(h):
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)

def smooth_obj_and_grad(h, beta):
    n = h.shape[0]
    d = correlate_full(h)
    maxd = d.max()
    shifted = d - maxd
    exp_vec = np.exp(beta * shifted)
    sum_exp = exp_vec.sum()
    w = exp_vec / sum_exp  # softmax weights
    L = 1 << ((n + (2 * n - 1) - 1).bit_length())
    f1 = np.fft.rfft(1 - h, L)
    fw = np.fft.rfft(w, L)
    conv1_full = np.fft.irfft(f1 * fw, L)[: (n + (2 * n - 1) - 1)]
    conv1 = conv1_full[n - 1 : n - 1 + n]
    w_rev = w[::-1]
    f2 = np.fft.rfft(h, L)
    fw_rev = np.fft.rfft(w_rev, L)
    conv_rev_full = np.fft.irfft(f2 * fw_rev, L)[: (n + (2 * n - 1) - 1)]
    conv_rev = conv_rev_full[n - 1 : n - 1 + n]
    grad = conv1 - conv_rev
    f = (np.log(sum_exp) + maxd) / beta
    return f, grad
