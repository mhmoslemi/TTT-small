def perturb_transfer(h, rng, scale):
    n = h.shape[0]
    i = rng.integers(n)
    j = rng.integers(n)
    while i == j: j = rng.integers(n)
    # compute max amount you can move from i to j
    max_transfer = min(h[i], 1 - h[j])
    # also compute max amount move opposite direction (j -> i)
    max_transfer_rev = min(h[j], 1 - h[i])
    # combine to sample eps in [-max_transfer_rev, max_transfer]
    lo = -max_transfer_rev * scale
    hi = max_transfer * scale
    if lo >= hi:
        return h
    eps = rng.random() * (hi - lo) + lo
    # new values
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

import time, math
import numpy as np
from scipy.optimize import minimize

# globals for the smooth surrogate used by SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h so that Σh = target while keeping 0≤h≤1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()

        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount between two coordinates while preserving sum."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    while i == j:
        j = rng.integers(0, n)
    max_fwd = min(h[i], 1.0 - h[j])          # i → j
    max_bwd = min(h[j], 1.0 - h[i])          # j → i
    lo = -max_bwd * scale
    hi = max_fwd * scale
    if lo >= hi:
        return h.copy()
    eps = rng.random() * (hi - lo) + lo
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    # tiny numerical safety
    new_h[i] = min(max(new_h[i], 0.0), 1.0)
    new_h[j] = min(max(new_h[j], 0.0), 1.0)
    return new_h


def hill_climb_multi(h_start, rng, max_iters=2000, init_step=0.5, n_trials=5):
    """Simple hill‑climb with decreasing step size."""
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
                improved = True
                break
        if not improved and rng.random() < 0.02:
            new_h = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return h, best_c


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum (tiny drift)
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x


def simulated_annealing(start_h, rng, time_limit,
                        init_temp=0.005, decay=0.9999, scale_start=0.5):
    """SA that respects bounds and the sum constraint."""
    cur_h = start_h.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    temp = init_temp
    scale = scale_start
    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        new_h = perturb_h(cur_h, rng, scale=scale)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < math.exp(-delta / (temp + 1e-12)):
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
        temp *= decay
        it += 1
        if it % 5000 == 0:
            scale = max(scale * 0.9, 0.02)
        if temp < 1e-9:
            temp = init_temp
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------ #
    # 1. Initialise
    # ------------------------------------------------------------------ #
    init = globals().get("initial_h_values")
    if init is not None:
        h0 = np.array(init, dtype=float)
        n = h0.size
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    h0 = np.clip(h0, 0.0, 1.0)
    h0 = enforce_sum(h0, n / 2.0, rng)

    best_h, best_c = h0, compute_c5(h0)

    # ------------------------------------------------------------------ #
    # 2. Quick hill‑climbs from a few starts
    # ------------------------------------------------------------------ #
    h1, c1 = hill_climb_multi(best_h, rng, max_iters=4000,
                              init_step=0.5, n_trials=10)
    if c1 < best_c:
        best_h, best_c = h1, c1

    for _ in range(2):
        rand_h = random_feasible_vector(n, rng)
        h2, c2 = hill_climb_multi(rand_h, rng, max_iters=3000,
                                  init_step=0.5, n_trials=8)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # ------------------------------------------------------------------ #
    # 3. Simulated annealing on current grid
    # ------------------------------------------------------------------ #
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 30:
        sa_time = min(remaining * 0.5, 450.0)
        h_sa, c_sa = simulated_annealing(best_h, rng, sa_time,
                                         init_temp=0.005,
                                         decay=0.9999,
                                         scale_start=0.5)
        if c_sa < best_c:
            best_h, best_c = h_sa, c_sa

    # ------------------------------------------------------------------ #
    # 4. SLSQP refinement on the 360‑point grid
    # ------------------------------------------------------------------ #
    for beta in [20, 80, 200, 500, 1000, 2000]:
        if time.time() - start_time > budget_s * 0.90:
            break
        try:
            h_ref = slsqp_optimize(best_h, beta, max_iter=400)
        except Exception:
            continue
        c_ref = compute_c5(h_ref)
        if c_ref < best_c:
            best_h, best_c = h_ref, c_ref

    # ------------------------------------------------------------------ #
    # 5. Upsample to a finer grid (720 points) if time permits
    # ------------------------------------------------------------------ #
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 80 and best_h.size < 720:
        target_n = 720
        old_x = (np.arange(best_h.size) + 0.5) * (2.0 / best_h.size)
        new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
        h_up = np.interp(new_x, old_x, best_h)
        h_up = enforce_sum(h_up, target_n / 2.0, rng)

        if remaining > 50:
            sa_time2 = min(remaining * 0.6, 350.0)
            h_up_sa, c_up_sa = simulated_annealing(h_up, rng, sa_time2,
                                                   init_temp=0.003,
                                                   decay=0.99995,
                                                   scale_start=0.3)
            if c_up_sa < best_c:
                best_h, best_c = h_up_sa, c_up_sa
                n = target_n

        # brief SLSQP on the finer grid
        for beta in [20, 80, 200]:
            if time.time() - start_time > budget_s * 0.98:
                break
            try:
                h_ref2 = slsqp_optimize(best_h, beta, max_iter=200)
            except Exception:
                continue
            c_ref2 = compute_c5(h_ref2)
            if c_ref2 < best_c:
                best_h, best_c = h_ref2, c_ref2
                n = best_h.size

    # ------------------------------------------------------------------ #
    # 6. Final clean‑up
    # ------------------------------------------------------------------ #
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
