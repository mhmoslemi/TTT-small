    n = len(h)
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iters):
        if time.time() > time_limit:
            break
        # compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = corr.max()
        c = max_corr * (2.0 / n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        # schedule beta and lr
        prog = it / max_iters
        beta = 1.0 + (2000.0 - 1.0) * (prog**2)  # quadratic
        lr = 0.5 * (1.0 - prog) + 0.02  # from 0.5 to 0.02
        # compute weights
        w = np.exp(beta * (corr - max_corr))
        w_sum = w.sum()
        if w_sum == 0.0 or np.isinf(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient
        g = np.zeros_like(h)

                upper = n - t
                g[t:] += weight * (1.0 - h[:upper])
                g[:upper] -= weight * h[t:]

import time

# ----------------- basic utilities ----------------- #
def compute_c5(h):
    """Exact discrete C5 for step function h on [0,2] with uniform spacing."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def random_feasible_vector(n, rng):
    """Uniform random vector in [0,1] with exact sum n/2."""
    target = n / 2.0
    h = rng.random(n)
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def hill_climb(h_start, rng, max_iters=2500, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c

def softmax_descent(h, rng, max_iters, target, time_limit):
    """Gradient descent on a soft‑max surrogate of the worst shift."""
    n = h.size
    shifts = np.arange(2 * n - 1) - (n - 1)
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iters):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = corr.max()
        cur_c = max_corr * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        prog = it / max_iters
        beta = 1.0 + (2000.0 - 1.0) * (prog ** 2)
        lr = 0.5 * (1.0 - prog) + 0.02
        w = np.exp(beta * (corr - max_corr))
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        g = np.zeros_like(h)
        for idx, weight in enumerate(w):
            if weight == 0.0:

            if s == 0:

            elif s > 0:

                h[i] += add

                if abs(diff) < 1e-12:
                    break
    final_c = compute_c5(h)
    if final_c < best_c:
        best_c = final_c
        best_h = h.copy()
    return best_h, best_c

            add = min(diff, 1.0 - h_interp[i]) if diff > 0 else max(diff, -h_interp[i])
            h_interp[i] += add
            diff = target - h_interp.sum()
        start_vectors.append(h_interp)

    # 2) random feasible (continuous)
    start_vectors.append(random_feasible_vector(n, rng))

    # 3) random binary feasible
    h_bin = np.zeros(n)
    ones_idx = rng.choice(n, n // 2, replace=False)
    h_bin[ones_idx] = 1.0
    start_vectors.append(h_bin)

    # ----- optimise each start ----- #
    for base_h in start_vectors:
        if time.time() - start_time > budget_s * 0.75:
            break
        h = np.clip(base_h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(2000):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break

        # soft‑max descent
        max_iters = 5000
        time_limit = start_time + budget_s * 0.75
        h_opt, c_opt = softmax_descent(h, rng, max_iters, target, time_limit)

        if c_opt < best_c:
            best_h, best_c, best_n = h_opt.copy(), c_opt, n

    # ----- final hill‑climb refinement ----- #
    if best_h is not None and time.time() - start_time < budget_s * 0.85:
        h_ref, c_ref = hill_climb(best_h, rng, max_iters=6000, start_step=0.3)
        if c_ref < best_c:
            best_h, best_c, best_n = h_ref.copy(), c_ref, n

    # ----- safety checks ----- #
    if best_h is None:
        best_n = 100
        best_h = np.full(best_n, 0.5)

    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            i = rng.integers(0, best_n)
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()
            if abs(diff) < 1e-12:
                break

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
