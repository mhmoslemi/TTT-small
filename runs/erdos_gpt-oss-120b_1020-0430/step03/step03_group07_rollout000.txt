import time
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Basic utilities
# --------------------------------------------------------------
def enforce_sum(h, target, rng):
    """Adjust h (in‑place) so that Σh == target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    # simple random walk on the simplex
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    """Uniformly sampled point of [0,1]^n with Σ = n/2 (approximate)."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    enforce_sum(h, n / 2.0, rng)
    return h

def perturb_h(h, rng, scale=0.2):
    """Transfer ε from i to j (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def correlate_full(h):
    """Full linear correlation of h with (1‑h) via FFT."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv

def compute_c5(h):
    """Discrete C₅ for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)

# Smooth surrogate + SLSQP
# --------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = correlate_full(x)
    bc = CURRENT_BETA * corr
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=350):
    """SLSQP on the smoothed max‑overlap objective."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

# --------------------------------------------------------------
# Deterministic hill climbing
# --------------------------------------------------------------
def hill_climb(h_start, rng, max_iters=5000, start_step=0.4):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay = max_iters // 8
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h.copy()
            h = new_h
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c

# --------------------------------------------------------------
# Binary helpers + simulated annealing on discrete space
# --------------------------------------------------------------
def random_binary_vector(n, rng):
    """Uniform binary vector with exactly n/2 ones."""
    idx = rng.permutation(n)
    h = np.zeros(n, dtype=float)
    h[idx[: n // 2]] = 1.0
    return h

def binary_simulated_annealing(n, rng, deadline):
    """Swap‑based SA on binary vectors with exact half‑ones."""
    cur = random_binary_vector(n, rng)
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    init_T = cur_c * 0.3 + 1e-4
    final_T = cur_c * 1e-4 + 1e-6
    start = time.time()
    while time.time() < deadline:
        # temperature based on elapsed fraction
        frac = (time.time() - start) / (deadline - start)
        T = init_T * ((final_T / init_T) ** frac)
        # pick a random one and a random zero
        ones = np.where(cur > 0.5)[0]
        zeros = np.where(cur < 0.5)[0]
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # propose swap
        cur[i], cur[j] = 0.0, 1.0
        new_c = compute_c5(cur)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            cur_c = new_c
            if cur_c < best_c:
                best_c, best = cur_c, cur.copy()
        else:
            # revert
            cur[i], cur[j] = 1.0, 0.0
    return best, best_c

# Main entry point
# --------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s

    # ----- base dimension -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        enforce_sum(h0, n / 2.0, rng)
    else:
        n = 360
        h0 = np.full(n, 0.5)
        enforce_sum(h0, n / 2.0, rng)

    # ----- pool of simple candidates --------------------------------------
    candidates = [h0]

    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx

    for f in (1, 2, 3, 4):
        h = 0.5 + 0.5 * np.sin(np.pi * f * xs)
        candidates.append(enforce_sum(np.clip(h, 0.0, 1.0), n / 2.0, rng))
        h = 0.5 + 0.5 * np.cos(np.pi * f * xs)
        candidates.append(enforce_sum(np.clip(h, 0.0, 1.0), n / 2.0, rng))

    h = xs / 2.0
    candidates.append(enforce_sum(np.clip(h, 0.0, 1.0), n / 2.0, rng))
    h = 1.0 - xs / 2.0
    candidates.append(enforce_sum(np.clip(h, 0.0, 1.0), n / 2.0, rng))

    step = np.zeros(n)
    step[: n // 2] = 1.0
    candidates.append(enforce_sum(step, n / 2.0, rng))

    alt = np.tile([0.0, 1.0], n // 2)
    candidates.append(enforce_sum(alt, n / 2.0, rng))

    for _ in range(3):
        candidates.append(random_feasible_vector(n, rng))

    # ----- initial best ----------------------------------------------------
    best_h = min(candidates, key=compute_c5)
    best_c = compute_c5(best_h)

    # ----- binary simulated annealing phase --------------------------------
    deadline_bin = start_time + budget_s * 0.45
    if deadline_bin > deadline:
        deadline_bin = deadline
    bin_h, bin_c = binary_simulated_annealing(n, rng, deadline_bin)
    if bin_c < best_c:
        best_h, best_c = bin_h, bin_c

    # ----- continuous refinement (hill‑climb + SLSQP) -----------------------
    deadline_cont = start_time + budget_s * 0.85
    beta_list = [5.0, 20.0, 80.0, 200.0, 800.0, 2000.0]

    while time.time() < deadline_cont:
        # hill‑climb
        h_opt, c_opt = hill_climb(best_h, rng, max_iters=3000, start_step=0.35)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
        # SLSQP ladder
        for beta in beta_list:
            if time.time() >= deadline_cont:
                break
            try:
                h_opt = slsqp_optimize(best_h, beta, max_iter=400)
                c_opt = compute_c5(h_opt)
                if c_opt < best_c:
                    best_h, best_c = h_opt, c_opt
            except Exception:
                continue

    # ----- optional up‑sampling to finer grid -------------------------------
    if time.time() < deadline and n * 2 <= 800:
        up_factor = 2
        h_up = np.kron(best_h, np.ones(up_factor))
        n_up = h_up.size
        enforce_sum(h_up, n_up / 2.0, rng)

        deadline_up = start_time + budget_s * 0.98
        # hill‑climb on the finer grid
        h_up_opt, c_up_opt = hill_climb(h_up, rng, max_iters=5000, start_step=0.25)
        if c_up_opt < best_c:
            best_h, best_c, n = h_up_opt, c_up_opt, n_up

        # SLSQP on the finer grid
        for beta in beta_list:
            if time.time() >= deadline:
                break
            try:
                h_up_opt = slsqp_optimize(best_h, beta, max_iter=400)
                c_up_opt = compute_c5(h_up_opt)
                if c_up_opt < best_c:
                    best_h, best_c, n = h_up_opt, c_up_opt, h_up_opt.size
            except Exception:
                continue

    # ----- final cleanup ---------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
