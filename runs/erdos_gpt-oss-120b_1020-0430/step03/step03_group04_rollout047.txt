def gradient_descent(h, target_sum, rng, time_limit):

    n = len(h); best_h = h.copy(); best_c = compute_c5(h).

    step = 0.1

    decay = 0.9995 per iteration.

    max_iters = 200000 (or until time limit).

    for iter in range(max_iters):
        if time.time() > time_limit: break
        # compute gradient of max
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = np.max(corr)
        # get all shifts achieving max
        idxs = np.where(corr == max_val)[0]
        shifts = idxs - (n - 1)
        # compute gradient sum
        grad = np.zeros_like(h)
        for sh in shifts:
            sh_mod = sh % n
            # truncated indices using np.take with mode wrap
            # vectorized: grad += 1 - (np.take(h, (np.arange(n) + sh_mod) % n) + np.take(h, (np.arange(n) - sh_mod) % n))
            h_shift_plus = h[(np.arange(n) + sh_mod) % n]
            h_shift_minus = h[(np.arange(n) - sh_mod) % n]
            grad += 1 - (h_shift_plus + h_shift_minus)
        grad /= len(shifts)

        # gradient descent step
        h_new = h - step * grad

        # project onto [0,1]
        h_new = np.clip(h_new, 0.0, 1.0)

        # adjust sum to target
        diff = target_sum - h_new.sum()
        if abs(diff) > 1e-12:
            # simple redistribution loop
            for _ in range(2000):
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h_new[i])
                else:
                    add = max(diff, -h_new[i])
                h_new[i] += add
                diff = target_sum - h_new.sum()
                if abs(diff) < 1e-12:
                    break

        # evaluate
        c = compute_c5(h_new)

        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        # optionally, if c <= best_c, keep h_new for next step else revert?
        # We'll set h = h_new regardless to follow gradient path.
        h = h_new

        step *= decay
        if step < 1e-8:
            step = 1e-8

    return best_h, best_c

import time
import numpy as np
from scipy.signal import fftconvolve

# ---------- basic utilities ---------- #
def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a step function h on [0,2]."""
    n = h.shape[0]
    corr = fftconvolve(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def project_to_sum_box(v: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Clip to [0,1] and enforce exact sum by random redistribution."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    if abs(diff) > 1e-12:
        n = v.shape[0]
        for _ in range(2000):
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - v[i])
            else:
                add = max(diff, -v[i])
            v[i] += add
            diff = target - v.sum()
            if abs(diff) < 1e-12:
                break
    return v


def gradient_of_max_corr(h: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max_k Σ h_i(1‑h_{i+k}) w.r.t. h."""
    n = h.shape[0]
    corr = fftconvolve(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    idxs = np.where(np.abs(corr - max_val) < 1e-12)[0]
    shifts = (idxs - (n - 1)) % n           # shifts in 0..n‑1
    arr = np.arange(n)
    grad = np.zeros_like(h)
    for s in shifts:
        s = int(s)
        grad += 1.0 - (h[(arr + s) % n] + h[(arr - s) % n])
    grad /= len(shifts)
    return grad


def gradient_descent_max_corr(
    h0: np.ndarray,
    target_sum: float,
    rng: np.random.Generator,
    time_limit: float,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the max‑correlation objective."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = 0.12
    decay = 0.9995
    it = 0
    while time.time() < time_limit and it < 200000:
        grad = gradient_of_max_corr(h)
        h = project_to_sum_box(h - step * grad, target_sum, rng)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        step *= decay
        if step < 1e-8:
            step = 1e-8
        it += 1
    return best_h, best_c


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    n = h.shape[0]
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(
    h_start: np.ndarray,
    rng: np.random.Generator,
    max_iters: int = 1500,
    start_step: float = 0.6,
) -> tuple[np.ndarray, float]:
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c
