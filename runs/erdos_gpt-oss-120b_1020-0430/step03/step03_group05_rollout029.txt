def gradient_softmax(h, beta):
    n = h.shape[0]
    s = np.correlate(h, 1 - h, mode='full')
    max_s = np.max(s) # maybe for numerical stability: subtract for exp?
    w = np.exp(beta * (s - max_s))
    w_sum = w.sum()
    w /= w_sum
    g = np.ones_like(h)  # baseline term from 1
    # loop over shifts
    # shift indices from -(n-1) to n-1
    # offset = n-1
    offset = n - 1
    # For each shift k
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        k = idx - offset
        if k >= 0:
            # i from 0 to n-1-k: g[i] -= wk * h[i+k]
            g[:n-k] -= wk * h[k:]
            # i from k to n-1: g[i] -= wk * h[i-k]
            g[k:] -= wk * h[:n-k]
        else:
            kk = -k
            # i from kk to n-1: g[i] -= wk * h[i-kk] (since i+k = i-kk)
            g[kk:] -= wk * h[:n-kk]
            # i from 0 to n-kk-1: g[i] -= wk * h[i+kk] (since i-k = i+kk)
            g[:n-kk] -= wk * h[kk:]
    return g

def gradient_softmax(h, beta):
    n = h.shape[0]
    s = np.correlate(h, 1 - h, mode='full')
    # compute weights stable
    max_s = np.max(s)
    w = np.exp(beta * (s - max_s))
    w_sum = np.sum(w)
    w = w / w_sum
    g = np.ones_like(h)
    offset = n - 1
    for idx, wk in enumerate(w):

            g[k:] -= wk * h[:n-k]

import time
import numpy as np

def compute_c5(h):
    """Exact C5 for discrete step function."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_to_feasible(v, target):
    """Projection onto {0≤h≤1, Σh = target} by a simple bisection."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Draw a random vector with 0≤h≤1 and Σh = n/2."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_to_feasible(v, n / 2.0)


def upsample_initial(to_n):
    """Linear interpolation of the global initial_h_values (if present)."""
    init = np.array(initial_h_values, dtype=float)
    m = init.size
    xs_old = np.linspace(0.0, 2.0, m, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, to_n, endpoint=False)
    new = np.interp(xs_new, xs_old, init)
    return new


def perturb_h(h, rng, scale=0.2):
    """Transfer a small ε from one coordinate to another, preserving sum."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new


def gradient_softmax(h, beta):
    """Gradient of the smooth max (log‑sum‑exp) surrogate."""
    n = h.size
    s = np.correlate(h, 1 - h, mode="full")
    max_s = np.max(s)
    w = np.exp(beta * (s - max_s))
    w /= w.sum()
    g = np.ones_like(h)
    offset = n - 1
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        k = idx - offset
        if k >= 0:
            g[:n - k] -= wk * h[k:]
            g[k:] -= wk * h[: n - k]
        else:
            kk = -k
            g[kk:] -= wk * h[: n - kk]
            g[: n - kk] -= wk * h[kk:]
    return g


def gradient_max_shift(h, k):
    """Gradient of the single shift term S_k = Σ h_i(1‑h_{i+k})."""
    n = h.size
    g = np.ones(n)
    if k >= 0:
        g[: n - k] -= h[k:]
        g[k:] -= h[: n - k]
    else:
        kk = -k
        g[kk:] -= h[: n - kk]
        g[: n - kk] -= h[kk:]
    return g

    # final stochastic hill‑climb
    while time.time() - start < time_budget:
        new_h = perturb_h(cur_h, rng, scale=0.2)
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c, best_h = new_c, new_h.copy()
        if new_c < cur_c:
            cur_h, cur_c = new_h, new_c

    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------
    # Choose discretisation size; prefer the size of the supplied seed.
    # ------------------------------------------------------------
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        h0 = upsample_initial(n)
        h0 = project_to_feasible(h0, n / 2.0)
    else:
        n = 300
        h0 = random_feasible_vector(n, rng)

    best_h, best_c = h0.copy(), compute_c5(h0)

    # ------------------------------------------------------------
    # Repeated optimisation passes while time permits.
    # ------------------------------------------------------------
    while True:
        elapsed = time.time() - start
        remaining = budget_s - elapsed
        if remaining < 8.0:
            break
        # allocate at most 180 s per pass (more than enough for our loops)
        pass_time = min(remaining * 0.5, 180.0)

        # start from the current best most of the time, occasionally from scratch
        if rng.random() < 0.7:
            start_vec = best_h.copy()
        else:
            start_vec = random_feasible_vector(n, rng)

        h_opt, c_opt = single_optimization(start_vec, rng, pass_time)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()

    # ------------------------------------------------------------
    # Final feasibility enforcement and return.
    # ------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    target = n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        best_h = project_to_feasible(best_h, target)

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
