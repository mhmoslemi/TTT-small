def binary_simulated_annealing(h_bin, rng, max_iters, temp_start, temp_end):
    n = len(h_bin)
    curr_h = h_bin.copy()
    curr_c = compute_c5(curr_h.astype(float))
    best_h = curr_h.copy()
    best_c = curr_c
    for it in range(max_iters):
        # select swap
        ones = np.where(curr_h == 1)[0]
        zeros = np.where(curr_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # propose swap
        new_h = curr_h.copy()
        new_h[i] = 0
        new_h[j] = 1
        new_c = compute_c5(new_h.astype(float))
        # accept?
        if new_c < curr_c:
            accept = True
        else:
            # temperature
            t = temp_start * (temp_end / temp_start) ** (it / max_iters)
            accept = rng.random() < np.exp(-(new_c - curr_c) / t)
        if accept:
            curr_h = new_h
            curr_c = new_c
            if curr_c < best_c:
                best_c = curr_c
                best_h = curr_h.copy()
    return best_h, best_c
```

### perturb_h_continuous

```
def perturb_h_continuous(h, rng, scale):
    n = len(h)
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    # ensure not exceed bounds
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h
```

```
def continuous_simulated_annealing(h_start, rng, max_iters, temp_start, temp_end, scale_start):
    h = h_start.copy()
    curr_c = compute_c5(h)
    best_h = h.copy()
    best_c = curr_c
    for it in range(max_iters):
        t = temp_start * (temp_end / temp_start) ** (it / max_iters)
        scale = scale_start * (temp_end / temp_start) ** (it / max_iters)  # decreasing scale
        new_h = perturb_h_continuous(h, rng, scale)
        new_c = compute_c5(new_h)
        if new_c < curr_c or rng.random() < math.exp(-(new_c - curr_c) / max(t, 1e-12)):
            h = new_h
            curr_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()
    return best_h, best_c
```

def upsample_h(h, new_n):
    old_n = h.shape[0]
    old_x = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    new_x = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    h_new = np.interp(new_x, old_x, h)
    return h_new

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # use log-sum-exp trick for numerical stability
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

import time
import math
import numpy as np
from scipy.optimize import minimize

# Global vars for SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h to have sum == target while staying in [0,1]."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        # small uniform correction
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth surrogate for max_k Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum (tiny drift)
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x


def random_feasible_vector(n, rng):
    """Generate a random feasible vector (0≤h_i≤1, Σh_i=n/2)."""
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)


def perturb_h_continuous(h, rng, scale):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def continuous_simulated_annealing(h_start, rng, max_iters, temp_start, temp_end, scale_start):
    """SA on continuous variables."""
    h = h_start.copy()
    curr_c = compute_c5(h)
    best_h, best_c = h.copy(), curr_c
    for it in range(max_iters):
        t = temp_start * (temp_end / temp_start) ** (it / max_iters)
        scale = scale_start * (temp_end / temp_start) ** (it / max_iters)
        new_h = perturb_h_continuous(h, rng, scale)
        new_c = compute_c5(new_h)
        if new_c < curr_c or rng.random() < math.exp(-(new_c - curr_c) / max(t, 1e-12)):
            h, curr_c = new_h, new_c
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
    return best_h, best_c


def binary_simulated_annealing(h_bin, rng, max_iters, temp_start, temp_end):
    """SA on binary vectors (swap a 1 with a 0)."""
    h = h_bin.copy()
    curr_c = compute_c5(h.astype(float))
    best_h, best_c = h.copy(), curr_c
    n = h.size
    for it in range(max_iters):
        t = temp_start * (temp_end / temp_start) ** (it / max_iters)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        new_h = h.copy()
        new_h[i] = 0
        new_h[j] = 1
        new_c = compute_c5(new_h.astype(float))
        if new_c < curr_c or rng.random() < math.exp(-(new_c - curr_c) / max(t, 1e-12)):
            h, curr_c = new_h, new_c
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
    return best_h, best_c


def upsample_h(h, new_n):
    """Linear interpolation to a finer grid."""
    old_n = h.shape[0]
    old_x = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    new_x = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    return np.interp(new_x, old_x, h)


def run(seed=42, budget_s=1100, **kwargs):
    """Optimise a step function h on [0,2] to minimise the Erdős C₅ bound."""
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # 1. Base dimension and initial vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 180
        h0 = np.zeros(n)
        h0[rng.choice(n, size=n // 2, replace=False)] = 1.0

    # ------------------------------------------------------------------
    # 2. Initialise a binary candidate
    # ------------------------------------------------------------------
    b0 = (h0 >= 0.5).astype(int)
    cur_ones = b0.sum()
    target_ones = n // 2
    if cur_ones != target_ones:
        diff = target_ones - cur_ones
        if diff > 0:
            zeros = np.where(b0 == 0)[0]
            b0[rng.choice(zeros, size=diff, replace=False)] = 1
        else:
            ones = np.where(b0 == 1)[0]
            b0[rng.choice(ones, size=-diff, replace=False)] = 0

    best_bin_h = b0.copy()
    best_bin_c = compute_c5(best_bin_h.astype(float))

    # ------------------------------------------------------------------
    # 3. Phase 1 – binary simulated annealing
    # ------------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.35:
        remaining = budget_s - (time.time() - start_time)
        iters = max(800, int(remaining * 400))
        bin_h, bin_c = binary_simulated_annealing(
            best_bin_h, rng, iters, temp_start=0.4, temp_end=0.001
        )
        if bin_c < best_bin_c:
            best_bin_h, best_bin_c = bin_h, bin_c
        # occasional fresh restart
        if rng.random() < 0.05:
            rand_bin = np.zeros(n, dtype=int)
            rand_bin[rng.choice(n, size=n // 2, replace=False)] = 1
            best_bin_h, best_bin_c = rand_bin, compute_c5(rand_bin.astype(float))

    # ------------------------------------------------------------------
    # 4. Phase 2 – continuous simulated annealing
    # ------------------------------------------------------------------
    h_cont = best_bin_h.astype(float)
    while time.time() - start_time < budget_s * 0.65:
        remaining = budget_s - (time.time() - start_time)
        iters = max(800, int(remaining * 250))
        h_new, c_new = continuous_simulated_annealing(
            h_cont, rng, iters, temp_start=0.2, temp_end=0.001, scale_start=0.3
        )
        if c_new < best_bin_c:
            best_bin_c = c_new
            best_bin_h = h_new.copy()
        h_cont = h_new

    # ------------------------------------------------------------------
    # 5. Phase 3 – SLSQP refinement with increasing β
    # ------------------------------------------------------------------
    beta_list = [5, 20, 80, 200, 500, 1000, 2000]
    for beta in beta_list:
        if time.time() - start_time > budget_s * 0.90:
            break
        # try a few starts for each β
        for attempt in range(3):
            start_vec = best_bin_h if attempt == 0 else random_feasible_vector(n, rng)
            try:
                h_ref = slsqp_optimize(start_vec, beta, max_iter=600)
            except Exception:

            c_ref = compute_c5(h_ref)
            if c_ref < best_bin_c:
                best_bin_c = c_ref
                best_bin_h = h_ref.copy()

    # ------------------------------------------------------------------
    # 6. Optional up‑sampling (if time permits)
    # ------------------------------------------------------------------
    if time.time() - start_time < budget_s * 0.98:
        new_n = n * 2
        if new_n <= 720:
            h_up = upsample_h(best_bin_h, new_n)
            h_up = enforce_sum(h_up, new_n / 2.0, rng)

            # binary SA on up‑sampled (quick)
            b_up = (h_up >= 0.5).astype(int)
            cur_ones = b_up.sum()
            target_ones = new_n // 2
            if cur_ones != target_ones:
                diff = target_ones - cur_ones
                if diff > 0:
                    zeros = np.where(b_up == 0)[0]
                    b_up[rng.choice(zeros, size=diff, replace=False)] = 1

                    b_up[rng.choice(ones, size=-diff, replace=False)] = 0

                b_up, rng, 2000, temp_start=0.4, temp_end=0.001
            )
            if up_c < best_bin_c:
                best_bin_c, best_bin_h, n = up_c, up_h.astype(float), new_n

            # short SLSQP refinement on up‑sampled
            for beta in [20, 80, 200, 500, 1000]:
                if time.time() - start_time > budget_s:
                    break
                try:
                    h_ref = slsqp_optimize(best_bin_h, beta, max_iter=500)
                except Exception:
                    continue
                c_ref = compute_c5(h_ref)
                if c_ref < best_bin_c:
                    best_bin_c, best_bin_h, n = c_ref, h_ref.copy(), new_n

    # ------------------------------------------------------------------
    # 7. Final clean‑up
    # ------------------------------------------------------------------
    best_bin_h = np.clip(best_bin_h, 0.0, 1.0)
    best_bin_h = enforce_sum(best_bin_h, n / 2.0, rng)
    final_c = compute_c5(best_bin_h)

    return best_bin_h.tolist(), float(final_c), int(n)
