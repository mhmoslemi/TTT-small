def de_optimize_custom(n, rng, max_time, init_vec=None):

    target = n/2

    popsize = max(30, min(80, int(0.25 * n)))  # maybe 30-80

    pop = []

    if init_vec is not None:
        # ensure proper length and feasible
        vec = project_to_feasible(init_vec, target)
        pop.append(vec)

    while len(pop) < popsize:
        pop.append(random_feasible_vector(n, rng))

    pop = np.array(pop)
    scores = np.array([compute_c5(ind) for ind in pop])
    best_idx = scores.argmin()
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]

    F = 0.7
    CR = 0.8
    start = time.time()
    while time.time() - start < max_time:
        for i in range(popsize):
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, 3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            # Clip after mutation? We'll project later.
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(0, n)] = True
            trial = np.where(cross, mutant, pop[i])
            trial = project_to_feasible(trial, target)
            trial_score = compute_c5(trial)
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_h = trial.copy()
        # optional early stop if no improvement maybe.
    return best_h, best_score

import time
import numpy as np
from scipy.optimize import minimize

# -------------------- Utility functions --------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_to_feasible(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0≤h≤1, Σh = target} by shifting and clipping."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Fast generation of a random vector with 0≤h≤1 and Σh=n/2."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_to_feasible(v, n / 2.0)


def random_binary_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a binary vector with exactly n/2 ones."""
    half = n // 2
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=half, replace=False)
    h[idx] = 1.0
    return h


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_continuous(h_start: np.ndarray, rng: np.random.Generator,
                         time_limit: float, start_step: float = 0.3):
    """Simple hill‑climb (accept only improvements) with a time budget."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    it = 0
    start = time.time()
    while time.time() - start < time_limit:
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h, h = c, new_h.copy(), new_h
        it += 1
        if it % 200 == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


def hill_climb_binary(h: np.ndarray, rng: np.random.Generator, max_iters: int = 20000):
    """Swap‑based hill climb restricted to binary vectors."""
    h = h.astype(float)
    best = h.copy()
    best_c = compute_c5(best)
    n = h.size
    for _ in range(max_iters):
        ones = np.where(best == 1.0)[0]
        zeros = np.where(best == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        best[i], best[j] = 0.0, 1.0
        c = compute_c5(best)
        if c < best_c:
            best_c = c
        else:
            best[i], best[j] = 1.0, 0.0
    return best, best_c


# -------------------- SLSQP with smooth max --------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 800) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def slsqp_refine(start_vec: np.ndarray, rng: np.random.Generator,
                 time_limit: float):
    betas = [10.0, 50.0, 200.0, 800.0, 2000.0, 8000.0]
    h_cur = start_vec.copy()
    best_h = h_cur.copy()
    best_c = compute_c5(h_cur)
    start = time.time()
    for beta in betas:
        if time.time() - start >= time_limit:
            break
        try:
            h_cur = slsqp_optimize(h_cur, beta, rng, max_iter=800)
        except Exception:
            continue
        c = compute_c5(h_cur)
        if c < best_c:
            best_c, best_h = c, h_cur.copy()
    return best_h, best_c


# -------------------- Differential Evolution --------------------
def de_optimize_custom(n: int, rng: np.random.Generator,
                       max_time: float, init_vec: np.ndarray = None):
    target = n / 2.0
    popsize = max(30, min(80, int(0.25 * n)))
    pop = []

    if init_vec is not None:
        pop.append(project_to_feasible(init_vec, target))

    while len(pop) < popsize:
        pop.append(random_feasible_vector(n, rng))

    pop = np.array(pop)
    scores = np.array([compute_c5(ind) for ind in pop])
    best_idx = scores.argmin()
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]

    F, CR = 0.9, 0.8
    start = time.time()
    while time.time() - start < max_time:
        for i in range(popsize):
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, 3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(0, n)] = True
            trial = np.where(cross, mutant, pop[i])
            trial = project_to_feasible(trial, target)
            trial_score = compute_c5(trial)
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score, best_h = trial_score, trial.copy()
    return best_h, best_score


# -------------------- Upsampling initial solution --------------------
def upsample_initial(to_n: int) -> np.ndarray:
    init = np.array(initial_h_values, dtype=float)  # type: ignore
    m = init.size
    xs_old = np.linspace(0.0, 2.0, m, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, to_n, endpoint=False)
    return np.interp(xs_new, xs_old, init)


# -------------------- Main optimisation driver --------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Optimise a step function h on [0,2] to minimise the Erdős C5 bound."""
    rng = np.random.default_rng(seed)
    start_time = time.time()
    n = 300
    target = n / 2.0

    best_h = None
    best_c = np.inf

    # -------- Phase 1: random binary search --------
    phase1_end = start_time + min(240.0, budget_s * 0.2)
    while time.time() < phase1_end:
        h_bin = random_binary_vector(n, rng)
        c = compute_c5(h_bin)
        if c < best_c:
            best_c, best_h = c, h_bin.copy()
    # ---- optional binary hill‑climb ----
    if best_h is not None:
        h_bin2, c2 = hill_climb_binary(best_h, rng, max_iters=20000)
        if c2 < best_c:
            best_c, best_h = c2, h_bin2.copy()

    # -------- Phase 2: Differential Evolution --------
    elapsed = time.time() - start_time
    time_left = budget_s - elapsed
    if time_left > 30.0:
        de_time = min(time_left * 0.5, max(30.0, time_left - 60.0))
        init_vec = None
        if "initial_h_values" in globals():
            try:
                init_vec = upsample_initial(n)
                init_vec = project_to_feasible(init_vec, target)
            except Exception:
                init_vec = None
        h_de, c_de = de_optimize_custom(n, rng, de_time, init_vec=init_vec)
        if c_de < best_c:
            best_c, best_h = c_de, h_de.copy()

    # -------- Phase 3: SLSQP refinement (multiple restarts) --------
    elapsed = time.time() - start_time
    time_left = budget_s - elapsed
    if time_left > 10.0:
        slsqp_budget = time_left * 0.6
        slsqp_start = time.time()
        while time.time() - slsqp_start < slsqp_budget:
            # choose start point
            if rng.random() < 0.5 and best_h is not None:
                start_vec = best_h.copy()
            else:
                start_vec = random_feasible_vector(n, rng)
            # allocate a slice of time for this refinement
            slice_time = min(30.0, slsqp_budget - (time.time() - slsqp_start))
            h_ref, c_ref = slsqp_refine(start_vec, rng, slice_time)
            if c_ref < best_c:
                best_c, best_h = c_ref, h_ref.copy()
            if time.time() - slsqp_start > slsqp_budget - 5.0:
                break

    # -------- Phase 4: final continuous hill‑climb --------
    elapsed = time.time() - start_time
    time_left = budget_s - elapsed
    if time_left > 5.0 and best_h is not None:
        h_final, c_final = hill_climb_continuous(best_h, rng, time_left, start_step=0.3)
        if c_final < best_c:
            best_c, best_h = c_final, h_final.copy()

    # -------- Final feasibility enforcement --------
    if best_h is None:
        best_h = np.full(n, 0.5)
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)

        for _ in range(5000):

            best_h[i] += add
            diff = target - best_h.sum()
