    low = -1.0  # enough negative to push all to 0 (if v_i + low <=0)
    high = 1.0  # enough positive to push all to 1 (if v_i + high >=1)

    for _ in range(50):
        mid = (low + high) / 2.0
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            low = mid
        else:
            high = mid
    proj = np.clip(v + (low + high) / 2.0, 0.0, 1.0)
    # Ensure sum close to target (maybe rounding errors) adjust small diff
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        # distribute diff among non-saturated variables
        free = (proj > 0) & (proj < 1)
        if np.any(free):
            # adjust proportionally
            proj[free] += diff / free.sum()
            proj = np.clip(proj, 0.0, 1.0)  # second clip
    return proj

def softmax_descent(h0, target, deadline):
    h = h0.copy()
    n = h.size
    beta = 5.0
    alpha = 0.1
    best_h = h.copy()
    best_c = compute_c5(h)
    iter = 0
    while time.time() < deadline:
        # compute overlaps
        corr = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
        max_corr = corr.max()
        # weights
        a = beta * corr
        m = a.max()
        exp_a = np.exp(a - m)
        w = exp_a / exp_a.sum()
        # gradient
        w_rev = w[::-1]
        conv = np.convolve(w_rev, h, mode='full')
        grad = 1.0 - conv[n-1:n-1+n]
        # update
        new_h = project_onto_bounds_and_sum(h - alpha * grad, target)
        h = new_h
        c = max_corr
        if c < best_c:
            best_c = c
            best_h = h.copy()
        iter += 1
        # schedule
        if iter % 50 == 0:
            if beta < 1000:
                beta = min(beta * 2, 1000)
            alpha = max(alpha * 0.9, 0.001)
    return best_h

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    h_new = h.copy()
    h_new[i] -= eps
    h_new[j] += eps
    return h_new

import time
import numpy as np

# ----------------- basic utilities ----------------- #
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for step function h on [0,2] with uniform spacing."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector in [0,1] with exact sum n/2."""
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h


def project_onto_bounds_and_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto [0,1]^n intersect {sum = target} using bisection."""
    low = -np.max(v) - 1.0
    high = 1.0 - np.min(v) + 1.0
    for _ in range(40):
        mid = (low + high) * 0.5
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            low = mid
        else:
            high = mid
    proj = np.clip(v + (low + high) * 0.5, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        free = (proj > 0.0) & (proj < 1.0)
        if free.any():
            proj[free] += diff / free.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(
    h_start: np.ndarray,
    rng: np.random.Generator,
    max_iters: int = 1000,
    start_step: float = 0.5,
) -> tuple[np.ndarray, float]:
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


# --------------- soft‑max descent ----------------- #
def softmax_descent(
    h0: np.ndarray,
    target: float,
    deadline: float,
    rng: np.random.Generator,
) -> np.ndarray:
    """Projected gradient descent on log‑sum‑exp surrogate of max overlap."""
    h = h0.copy()
    n = h.size
    beta = 5.0
    alpha = 0.1
    best_h = h.copy()
    best_c = compute_c5(h)
    it = 0
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
        max_corr = float(corr.max())
        a = beta * corr
        m = a.max()
        exp_a = np.exp(a - m)
        w = exp_a / exp_a.sum()
        # gradient = 1 - sum_k w_k * h_{i+lag}
        w_rev = w[::-1]
        conv = np.convolve(w_rev, h, mode="full")
        grad = 1.0 - conv[n - 1 : n - 1 + n]
        h = project_onto_bounds_and_sum(h - alpha * grad, target)
        if max_corr < best_c:
            best_c = max_corr
            best_h = h.copy()
        it += 1
        if it % 50 == 0:
            if beta < 1000:
                beta = min(beta * 2.0, 1000.0)
            alpha = max(alpha * 0.9, 0.001)
    return best_h
