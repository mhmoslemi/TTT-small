import time, numpy as np, math
from scipy.optimize import minimize

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def project_to_feasible(v, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def random_feasible_vector(n, rng):
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_to_feasible(v, n / 2.0)

def upsample(h, n_new):
    m = h.size
    xs_old = np.linspace(0.0, 2.0, m, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    new = np.interp(xs_new, xs_old, h)
    # Clip then project to ensure sum equals target
    target = n_new / 2.0
    new = np.clip(new, 0.0, 1.0)
    return project_to_feasible(new, target)

def perturb_h(h, rng, scale):
    n = h.size
    i = rng.integers(n)
    # choose j != i
    j = rng.integers(n - 1)
    if j >= i:
        j += 1
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def sa_optimize(h0, rng, time_limit):
    start = time.time()
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    cur_h = h0.copy()
    cur_c = best_c
    n = h0.size
    # initial temperature relative to expected improvements
    init_temp = 0.001  # choose small
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        remaining = time_limit - elapsed
        # linear cooling
        T = init_temp * (remaining / time_limit)
        # schedule scale linearly decreasing
        scale = 0.5 * (remaining / time_limit) + 0.02
        new_h = perturb_h(cur_h, rng, scale)
        new_c = compute_c5(new_h)
        if new_c < cur_c or rng.random() < math.exp(-(new_c - cur_c) / max(T, 1e-12)):
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
    return best_h, best_c

# global placeholders for soft-max surrogate used by SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_refine(h0, rng, time_limit):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    target = n / 2.0
    # ensure feasible start (clip)
    h0 = np.clip(h0, 0.0, 1.0)
    # ensure sum matches via projection (simple bisection)
    h0 = project_to_feasible(h0, target)
    CURRENT_TARGET = target
    # allocate inner time loop but we just run sequentially with increasing beta
    # total time budget is time_limit; we will split equally among beta steps
    betas = [10.0, 50.0, 200.0, 800.0, 2000.0]
    per_beta = max(0.0, time_limit / len(betas))
    best_h = h0
    best_c = compute_c5(best_h)
    start_tot = time.time()
    for beta in betas:
        if time.time() - start_tot > time_limit:
            break
        CURRENT_BETA = beta
        bounds = [(0.0, 1.0)] * n
        cons = {"type": "eq", "fun": sum_eq_constraint}
        # set maxiter proportional to n maybe
        maxiter = max(200, int(2000 / n * n))
        res = minimize(
            softmax_obj, best_h, method='SLSQP', bounds=bounds,
            constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False}
        )
        x = np.clip(res.x, 0.0, 1.0)
        # correct sum exactly
        diff = target - x.sum()
        if abs(diff) > 1e-12:
            # simple projection via adding diff to random coordinates respecting bounds
            for _ in range(2000):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(n)
                add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
                x[i] += add
                diff = target - x.sum()
        best_h = x
        best_c = compute_c5(best_h)
    return best_h, best_c

def hill_climb(h0, rng, max_iters=2000, step=0.3):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    decay = 0.9
    for it in range(max_iters):
        new_h = perturb_h(h, rng, step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h.copy()
            h = new_h
        if (it + 1) % (max_iters // 10) == 0:
            step = max(step * decay, 1e-4)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Determine initial vector at coarse resolution
    n0 = 300
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size != n0:
            # if different size, interpolate to n0
            h0 = upsample(h0, n0)
    else:
        h0 = random_feasible_vector(n0, rng)
    # ensure feasibility
    h0 = project_to_feasible(h0, n0 / 2.0)
    # Allocate time budget
    total = budget_s
    # Phase 1: SA on n0
    t1 = total * 0.45
    h1, c1 = sa_optimize(h0, rng, t1)
    # Phase 2: SLSQP refine on n0
    t2 = total * 0.20
    h1, c1 = slsqp_refine(h1, rng, t2)
    # Phase 3: hill climb on n0
    t3 = total * 0.05
    h1, c1 = hill_climb(h1, rng, max_iters=2000, step=0.2)
    # Keep best so far
    best_h = h1
    best_c = c1
    best_n = n0
    # Phase 4: upscale to finer resolution if time remains
    elapsed = time.time() - start_time
    remaining = total - elapsed
    if remaining > 30:  # need enough time
        n1 = 550  # choose under 1000
        # upsample best_h to new n1
        h_up = upsample(best_h, n1)
        # SA on finer grid
        t_sa_fine = remaining * 0.5
        h_fine, c_fine = sa_optimize(h_up, rng, t_sa_fine)
        # SLSQP refine
        t_slsqp_fine = remaining * 0.3
        h_fine, c_fine = slsqp_refine(h_fine, rng, t_slsqp_fine)
        # final hill climb
        t_hc_fine = remaining * 0.2
        h_fine, c_fine = hill_climb(h_fine, rng, max_iters=3000, step=0.15)
        if c_fine < best_c:
            best_h, best_c, best_n = h_fine, c_fine, n1
    # Final feasibility enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(best_n)
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)

import time, math
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h):
    """Exact discrete C5 (dx = 2 / len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def project_to_feasible(v, target):
    """Project v onto {0≤x≤1, Σx = target} via a simple bisection on a shift."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def random_feasible_vector(n, rng):
    """Fast random feasible vector (0≤h≤1, Σh=n/2)."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_to_feasible(v, n / 2.0)

def upsample(h, n_new):
    """Linear interpolation of h to length n_new, then projection to feasibility."""
    m = h.size
    xs_old = np.linspace(0.0, 2.0, m, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    new = np.interp(xs_new, xs_old, h)
    new = np.clip(new, 0.0, 1.0)
    return project_to_feasible(new, n_new / 2.0)

def perturb_h(h, rng, scale):
    """Transfer a random ε from i to j (preserves sum, respects bounds)."""
    n = h.size
    i = rng.integers(n)
    j = rng.integers(n - 1)
    if j >= i:
        j += 1
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

# ---------- simulated annealing ----------
def sa_optimize(h0, rng, time_limit):
    """Simple SA (time‑controlled)."""
    start = time.time()
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    cur_h, cur_c = best_h.copy(), best_c
    n = h0.size
    init_temp = 0.02  # enough to accept occasional uphill moves
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        remaining = time_limit - elapsed
        T = init_temp * (remaining / time_limit)          # linear cooling
        scale = 0.5 * (remaining / time_limit) + 0.02   # large moves early
        new_h = perturb_h(cur_h, rng, scale)
        new_c = compute_c5(new_h)
        if new_c < cur_c or rng.random() < math.exp(-(new_c - cur_c) / max(T, 1e-12)):
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
    return best_h, best_c

# ---------- SLSQP refinement (smooth max surrogate) ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_refine(h0, rng, time_limit):
    """Run SLSQP with increasing beta, respecting the time budget."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    target = n / 2.0
    h = project_to_feasible(np.clip(h0, 0.0, 1.0), target)
    CURRENT_TARGET = target

    betas = [10.0, 50.0, 200.0, 800.0, 2000.0]
    start = time.time()
    for beta in betas:
        if time.time() - start >= time_limit:
            break
        CURRENT_BETA = beta
        bounds = [(0.0, 1.0)] * n
        cons = {"type": "eq", "fun": sum_eq_constraint}
        res = minimize(
            softmax_obj,
            h,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": 500, "ftol": 1e-12, "disp": False},
        )
        x = np.clip(res.x, 0.0, 1.0)
        # exact sum correction
        diff = target - x.sum()
        if abs(diff) > 1e-12:
            for _ in range(2000):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(n)
                add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
                x[i] += add
                diff = target - x.sum()
        h = x
    return h, compute_c5(h)

# ---------- simple hill‑climb ----------
def hill_climb(h0, rng, max_iters=2000, step=0.3):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = 0.9
    for it in range(max_iters):
        new_h = perturb_h(h, rng, step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h.copy()
            h = new_h
        if (it + 1) % (max_iters // 10) == 0:
            step = max(step * decay, 1e-4)
    return best_h, best_c

# ---------- main optimisation driver ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
