def gradient_smooth_max(h, beta):
    n = h.size
    a = h
    b = 1.0 - h
    corr = np.correlate(a, b, mode='full')
    max_corr = np.max(beta * corr) # actually we want for stability: shift = beta * corr - maxBeta
    # compute w_norm
    exp_terms = np.exp(beta * corr - np.max(beta * corr))
    w_norm = exp_terms / np.sum(exp_terms)
    # gradient initialization
    grad = np.zeros_like(h)
    # offset indices
    for offset in range(-(n-1), n):
        idx = offset + (n - 1)
        w = w_norm[idx]
        if w == 0.0:
            continue
        if offset >= 0:
            # b part
            grad[:n - offset] += w * b[offset:]
            # a part
            grad[offset:] -= w * a[:n - offset]
        else:
            off = -offset
            grad[off:] += w * b[:n - off]
            grad[:n - off] -= w * a[off:]
    return grad

def project_to_feasible(h, target):

    np.clip(h, 0.0, 1.0, out=h)
    # compute diff
    diff = target - h.sum()
    # while diff not close to zero and there are adjustable elements
    # we will distribute diff uniformly among adjustable entries
    # Use tolerance 1e-12 and max 10 iterations.
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # can increase entries <1
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        # amount to add per entry
        delta = diff / cnt
        # update entries but ensure bounds
        if diff > 0:
            # increase but not exceed 1
            inc = np.minimum(delta, 1.0 - h[mask])
            h[mask] += inc
        else:
            dec = np.minimum(-delta, h[mask])
            h[mask] -= dec
        diff = target - h.sum()
    # final clip (just in case)
    np.clip(h, 0.0, 1.0, out=h)
    return h

import time, math, numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2 / n)."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)


def enforce_sum(h, target, rng):
    """Adjust h in‑place so that sum(h)=target while respecting 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def project_to_feasible(h, target):
    """Fast projection onto [0,1] box with sum constraint."""
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        mask = h < 1.0 if diff > 0 else h > 0.0
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        delta = diff / cnt
        if diff > 0:
            inc = np.minimum(delta, 1.0 - h[mask])
            h[mask] += inc
        else:
            dec = np.minimum(-delta, h[mask])
            h[mask] -= dec
        diff = target - h.sum()
    np.clip(h, 0.0, 1.0, out=h)
    return h


def random_feasible_vector(n, rng):
    """Generate random h with 0≤h≤1 and sum=n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    enforce_sum(h, n / 2.0, rng)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer ε between two coordinates (preserves sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new


# ---------- smooth‑max optimisation (SLSQP) ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on smooth max‑approximation."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {"type": "eq", "fun": sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x


# ---------- gradient of smooth‑max ----------
def gradient_smooth_max(h, beta):
    """Gradient of the log‑sum‑exp smooth max (w.r.t h)."""
    n = h.size
    a = h
    b = 1.0 - h
    corr = np.correlate(a, b, mode="full")
    c = beta * corr
    maxc = np.max(c)
    expc = np.exp(c - maxc)
    w = expc / np.sum(expc)          # weight for each lag
    grad = np.zeros_like(h)

    # offsets run from -(n-1) .. (n-1)
    for offset in range(-(n - 1), n):
        w_i = w[offset + n - 1]
        if w_i == 0.0:
            continue
        if offset >= 0:
            grad[:n - offset] += w_i * b[offset:]
            grad[offset:] -= w_i * a[:n - offset]
        else:
            o = -offset
            grad[o:] += w_i * b[:n - o]
            grad[:n - o] -= w_i * a[o:]
    return grad


# ---------- projected gradient descent on smooth max ----------
def projected_gradient_descent(h0, beta, steps, step_size, target, rng):
    """Simple PGD on the smooth‑max surrogate."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)

    for it in range(steps):
        grad = gradient_smooth_max(h, beta)
        h -= step_size * grad
        np.clip(h, 0.0, 1.0, out=h)
        project_to_feasible(h, target)

        # occasional evaluation
        if (it & 31) == 0:
            cur_c = compute_c5(h)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    return best_h, best_c
