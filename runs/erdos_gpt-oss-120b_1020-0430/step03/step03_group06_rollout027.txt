def compute_c5(h):
    n = h.size
    b_rev = (1 - h)[::-1]
    size = 1 << (2*n - 1).bit_length()
    fft_a = np.fft.rfft(h, n=size)
    fft_b = np.fft.rfft(b_rev, n=size)
    conv = np.fft.irfft(fft_a * fft_b, n=size)[:2*n-1]
    return np.max(conv) * (2.0/n)

def propose_move(h):
    n = len(h)
    i, j = rng.integers(0, n, size=2)
    while j == i:
        j = rng.integers(0, n)
    max_decrease = h[i]  # amount we can subtract from i
    max_increase = 1 - h[j]  # amount we can add to j
    max_opposite = h[j]  # amount we can subtract from j (if moving opposite direction)
    max_opposite_inc = 1 - h[i]  # amount we can add to i (if moving opposite direction)
    # We can transfer amount from i to j (delta positive) up to min(max_decrease, max_increase)
    # Or from j to i (delta negative) up to min(max_opposite, max_opposite_inc)
    limit_pos = min(max_decrease, max_increase)
    limit_neg = min(max_opposite, max_opposite_inc)
    if limit_pos + limit_neg == 0: return h # cannot move
    # Sample delta uniformly within [-limit_neg, limit_pos]
    delta = (rng.random() * (limit_pos + limit_neg)) - limit_neg
    # Optionally scale delta by a temperature factor later
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    # Clipping (should be within bounds by construction)
    return h_new

def compute_c5(h: np.ndarray) -> float:
    """Exact C5 using FFT (full correlation)."""
    n = h.shape[0]

    return float(np.max(conv) * (2.0 / n))

def greedy_hill_climb(h_start: np.ndarray, n: int, target: float,
                      rng: np.random.Generator, max_iters: int = 20000):
    """Greedy descent: keep only improving two‑index moves."""
    h = h_start.copy()
    c = compute_c5(h)
    for _ in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        max_pos = min(h[i], 1.0 - h[j])   # move from i to j
        max_neg = min(h[j], 1.0 - h[i])   # move from j to i
        if max_pos + max_neg == 0:
            continue
        # try full move i→j
        if max_pos > 0:
            h1 = h.copy()
            h1[i] -= max_pos
            h1[j] += max_pos
            c1 = compute_c5(h1)
        else:
            c1 = float('inf')
        # try full move j→i
        if max_neg > 0:
            h2 = h.copy()
            h2[i] += max_neg
            h2[j] -= max_neg
            c2 = compute_c5(h2)
        else:
            c2 = float('inf')
        if c1 < c and c1 <= c2:
            h, c = h1, c1
        elif c2 < c:
            h, c = h2, c2
    return h, c

# Main entry point
# -------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----- initial vector -------------------------------------------------
    if "initial_h_values" in globals():
        h0_base = np.array(initial_h_values, dtype=float)
    else:
        n0 = 180
        xs = (np.arange(n0) + 0.5) * (2.0 / n0)
        h0_base = 0.5 + 0.5 * np.sin(np.pi * xs)   # smooth start
        h0_base = np.clip(h0_base, 0.0, 1.0)

    # up‑sample for finer search
    n = h0_base.size * 2
    h = np.repeat(h0_base, 2)
    target = n / 2.0
    h = enforce_sum(h, target, rng)

    best_h = h.copy()
    best_c = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_c = best_c

    # ----- simulated annealing on the up‑sampled grid --------------------
    max_iters = 500000
    T0, T_end = 0.005, 1e-5

    for it in range(max_iters):
        if time.time() - start_time > budget_s * 0.94:
            break
        frac = it / max_iters
        T = T0 * (1 - frac) + T_end * frac
        scale = max(T / T0, 0.02)

        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        max_pos = min(cur_h[i], 1.0 - cur_h[j])
        max_neg = min(cur_h[j], 1.0 - cur_h[i])
        if max_pos + max_neg == 0:
            continue
        delta = (rng.random() * (max_pos + max_neg) - max_neg) * scale
        if delta > max_pos:
            delta = max_pos
        if delta < -max_neg:
            delta = -max_neg
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_c = compute_c5(new_h)
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-12)):
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
        # occasional exact‑sum repair
        if (it + 1) % 10000 == 0:
            diff = target - cur_h.sum()
            if abs(diff) > 1e-12:
                idx = rng.integers(0, n)
                adj = np.clip(diff, -cur_h[idx], 1.0 - cur_h[idx])
                cur_h[idx] += adj
                cur_c = compute_c5(cur_h)
                if cur_c < best_c:
                    best_c, best_h = cur_c, cur_h.copy()

    # ----- greedy refinements + random restarts -------------------------
    while time.time() - start_time < budget_s * 0.97:
        h_ref, c_ref = greedy_hill_climb(best_h, n, target, rng, max_iters=5000)
        if c_ref < best_c:
            best_c, best_h = c_ref, h_ref.copy()
        # random restart
        v = random_feasible_vector(n, target, rng)
        h_rc, c_rc = greedy_hill_climb(v, n, target, rng, max_iters=3000)
        if c_rc < best_c:
            best_c, best_h = c_rc, h_rc.copy()

    # ----- optional second up‑sampling ------------------------------------
    if time.time() - start_time < budget_s * 0.45 and n * 2 <= 720:
        n2 = n * 2
        target2 = n2 / 2.0
        h2 = np.repeat(best_h, 2)
        h2 = enforce_sum(h2, target2, rng)
        cur_h, cur_c = h2.copy(), compute_c5(h2)
        best_h2, best_c2 = cur_h.copy(), cur_c
        remaining = budget_s - (time.time() - start_time)
        max_iters2 = int(min(200000, remaining * 1500))
        for it in range(max_iters2):
            if time.time() - start_time > budget_s * 0.94:
                break
            frac = it / max_iters2
            T = T0 * (1 - frac) + T_end * frac
            scale = max(T / T0, 0.02)
            i = rng.integers(0, n2)
            j = rng.integers(0, n2 - 1)
            if j >= i:
                j += 1
            max_pos = min(cur_h[i], 1.0 - cur_h[j])
            max_neg = min(cur_h[j], 1.0 - cur_h[i])
            if max_pos + max_neg == 0:
                continue
            delta = (rng.random() * (max_pos + max_neg) - max_neg) * scale
            if delta > max_pos:
                delta = max_pos
            if delta < -max_neg:
                delta = -max_neg
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
            new_c = compute_c5(new_h)
            if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-12)):
                cur_h, cur_c = new_h, new_c
                if new_c < best_c2:
                    best_c2, best_h2 = new_c, new_h.copy()
        # final greedy refinement on second level
        h2_ref, c2_ref = greedy_hill_climb(best_h2, n2, target2, rng, max_iters=5000)
        if c2_ref < best_c2:
            best_c2, best_h2 = c2_ref, h2_ref.copy()
        if best_c2 < best_c:
            best_c, best_h, n, target = best_c2, best_h2.copy(), n2, target2

    # ----- final clean‑up -------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
