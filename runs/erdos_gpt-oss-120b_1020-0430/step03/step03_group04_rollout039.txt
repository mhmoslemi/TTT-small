def pairwise_optimize(h, max_time, rng):
    n = h.size
    L = 2*n - 1
    corr = np.correlate(h, 1 - h, mode='full')
    cur_max = corr.max()
    rev = h[::-1]
    start = time.time()
    # parameters
    attempts = 0
    while time.time() - start < max_time:
        # sample random pair
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # compute feasible delta
        hi = h[i]; hj = h[j]
        low = max(-hj, hi - 1.0)
        high = min(hi, 1.0 - hj)
        if low >= high - 1e-12:
            continue
        # compute b
        b = np.zeros(L)
        # slice for j part1
        b[(n-1 - j):(n-1 - j + n)] += 1 - h
        # slice for j part2
        b[j:j+n] += -rev
        # slice for i part1
        b[(n-1 - i):(n-1 - i + n)] -= 1 - h
        # slice for i part2
        b[i:i+n] += rev
        # find optimal delta
        delta_opt = optimal_delta(corr, b, low, high)
        if abs(delta_opt) < 1e-12:
            continue
        # compute new max
        new_corr = corr + b * delta_opt
        new_max = new_corr.max()
        if new_max < cur_max - 1e-12:
            # accept move
            h[i] -= delta_opt
            h[j] += delta_opt
            corr = new_corr
            cur_max = new_max
            rev = h[::-1]  # update reversed view
        attempts += 1
        # maybe add early stop after many attempts without improvement
        if attempts > 50000:
            # break if no improvement for many attempts?
            break
    final_c = cur_max * (2.0 / n)
    return h, final_c

import time
import numpy as np
from scipy.optimize import minimize

# ------------------- basic utilities ------------------- #
def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2500, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.size

# ------------------- SLSQP with soft‑max surrogate ------------------- #
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:

            diff = CURRENT_TARGET - x.sum()

                break
    return x

def optimal_delta(base, slope, lo, hi, tol=1e-12):
    # convex piecewise‑linear minimisation of max(base + slope*delta)
    a_low = base + slope * lo
    m_low = a_low.max()
    mask = a_low == m_low
    s_min = slope[mask].min()
    s_max = slope[mask].max()
    if s_min <= 0 <= s_max:
        return lo
    a_hi = base + slope * hi
    m_hi = a_hi.max()
    mask = a_hi == m_hi
    s_min = slope[mask].min()
    s_max = slope[mask].max()
    if s_min <= 0 <= s_max:
        return hi
    # binary search for 0 in subgradient
    lo_, hi_ = lo, hi
    for _ in range(40):
        mid = (lo_ + hi_) * 0.5
        a_mid = base + slope * mid
        m_mid = a_mid.max()
        mask = a_mid == m_mid
        s_min = slope[mask].min()
        s_max = slope[mask].max()
        if s_min <= 0 <= s_max:
            return mid
        if s_min > 0:
            hi_ = mid
        else:
            lo_ = mid
    return (lo_ + hi_) * 0.5

    L = 2 * n - 1
    corr = np.correlate(h, 1 - h, mode="full")

    start = time.time()
