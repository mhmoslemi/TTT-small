import time, numpy as np
def compute_c5(h):
    corr = np.correlate(h, 1-h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def random_binary(n, rng):
    # choose n/2 ones
    idx = rng.choice(n, n//2, replace=False)
    h = np.zeros(n, dtype=np.float64)
    h[idx] = 1.0
    return h

def binary_from_continuous(v):
    # Convert continuous values to binary preserving sum
    n = v.size
    idx = np.argsort(v)[-n//2:]  # top half
    h = np.zeros(n, dtype=np.float64)
    h[idx] = 1.0
    return h

def hill_climb_bin(h, rng, max_iters=2000):
    n = h.size
    best = h.copy()
    best_c = compute_c5(best)
    step = 1  # swap
    for it in range(max_iters):
        # pick random 1 and 0
        ones = np.where(best == 1)[0]
        zeros = np.where(best == 0)[0]
        if len(ones)==0 or len(zeros)==0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        new = best.copy()
        new[i] = 0
        new[j] = 1
        c = compute_c5(new)
        if c < best_c:
            best = new
            best_c = c
    return best, best_c

def local_search(h, rng, steps=500):
    # Could also use simulated annealing.
    return hill_climb_bin(h, rng, max_iters=steps)

import time, numpy as np
from scipy.optimize import minimize

# Global variables for SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None

def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def _softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type':'eq','fun':_sum_eq_constraint}
    res = minimize(_softmax_obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter':max_iter,'ftol':1e-12,'disp':False})
    x = np.clip(res.x, 0.0, 1.0)
    # tiny repair to exact sum
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            if abs(add) < 1e-12:
                continue
            x[i] += add
            diff = CURRENT_TARGET - x.sum()

def hill_climb_continuous(h_start, rng, max_iters=2500, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(best_h)
    step = start_step
    n = h.size
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)
        max_pos = min(h[i], 1.0 - h[j])  # transfer from i to j
        max_neg = min(h[j], 1.0 - h[i])  # transfer from j to i
        if max_pos + max_neg == 0:
            continue
        eps = (rng.random() * (max_pos + max_neg)) - max_neg
        eps *= step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

def random_binary(n, rng):
    idx = rng.choice(n, n // 2, replace=False)
    h = np.zeros(n, dtype=np.float64)
    h[idx] = 1.0
    return h

def binary_from_continuous(v):
    n = v.size
    idx = np.argsort(v)[-n//2:]
    h = np.zeros(n, dtype=np.float64)
    h[idx] = 1.0
    return h

def repair_binary(vec, n, rng):
    target = n // 2
    total = int(vec.sum())
    if total > target:
        # need to flip (total-target) ones to zero
        excess = total - target
        ones = np.where(vec == 1)[0]
        if excess > 0 and len(ones) > 0:
            flip = rng.choice(ones, excess, replace=False)
            vec[flip] = 0.0
    elif total < target:
        deficit = target - total
        zeros = np.where(vec == 0)[0]
        if deficit > 0 and len(zeros) > 0:
            flip = rng.choice(zeros, deficit, replace=False)
            vec[flip] = 1.0
    return vec

def hill_climb_binary(h, rng, max_iters=2000):
    n = h.size
    best = h.copy()
    best_c = compute_c5(best)
    # Precompute indices sets?
    for _ in range(max_iters):
        ones = np.where(best == 1)[0]
        zeros = np.where(best == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        new = best.copy()
        new[i] = 0
        new[j] = 1
        c = compute_c5(new)
        if c < best_c:
            best = new
            best_c = c
    return best, best_c

def genetic_algorithm_binary(n, rng, time_limit):
    pop_size = 40
    elite_size = 4
    tournament_k = 3
    prob_crossover = 0.9
    prob_mutation = 0.3

    # initial population
    pop = [random_binary(n, rng) for _ in range(pop_size)]
    if 'initial_h_values' in globals():
        v0 = np.array(initial_h_values, dtype=float)
        if v0.size == n:
            h0 = binary_from_continuous(v0)
            pop[0] = h0.copy()
    fitness = [compute_c5(ind) for ind in pop]
    best_idx = int(np.argmin(fitness))
    best_h = pop[best_idx].copy()
    best_c = fitness[best_idx]

    start = time.time()
    while time.time() - start < time_limit:
        # sort indices
        sorted_idx = np.argsort(fitness)
        new_pop = [pop[i].copy() for i in sorted_idx[:elite_size]]
        # tournament selection function
        def tournament():
            cand = rng.choice(pop_size, tournament_k, replace=False)
            cand_f = [fitness[i] for i in cand]
            return pop[cand[np.argmin(cand_f)]]
        while len(new_pop) < pop_size:
            p1 = tournament()
            p2 = tournament()
            child = p1.copy()
            if rng.random() < prob_crossover:
                mask = rng.integers(0, 2, size=n, dtype=bool)
                child = np.where(mask, p1, p2).astype(np.float64)
                child = repair_binary(child, n, rng)
            if rng.random() < prob_mutation:
                swaps = rng.integers(1, 4)  # 1 to 3 swaps
                for _ in range(swaps):
                    ones = np.where(child == 1)[0]
                    zeros = np.where(child == 0)[0]
                    if len(ones) == 0 or len(zeros) == 0:
                        break
                    i = rng.choice(ones)
                    j = rng.choice(zeros)
                    child[i] = 0
                    child[j] = 1
                # Ensure balance
                child = repair_binary(child, n, rng)
            # local refinement via short hill climb
            child, _ = hill_climb_binary(child, rng, max_iters=120)
            new_pop.append(child)
        pop = new_pop[:pop_size]
        fitness = [compute_c5(ind) for ind in pop]
        cur_best_idx = int(np.argmin(fitness))
        cur_best_c = fitness[cur_best_idx]
        if cur_best_c < best_c:
            best_c = cur_best_c
            best_h = pop[cur_best_idx].copy()
    return best_h, best_c

def continuous_refine(h_start, rng, time_limit):
    start = time.time()
    best_h = h_start.copy()
    best_c = compute_c5(best_h)
    # beta schedule
    beta_list = [20.0, 80.0, 200.0, 600.0, 1200.0]
    # main loop
    while time.time() - start < time_limit:
        for beta in beta_list:
            if time.time() - start >= time_limit:
                break
            h_opt = slsqp_optimize(best_h, beta, max_iter=400)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt.copy()
        # hill climb
        if time.time() - start >= time_limit:
            break
        h_hc, c_hc = hill_climb_continuous(best_h, rng, max_iters=2500, start_step=0.5)
        if c_hc < best_c:
            best_c, best_h = c_hc, h_hc.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Final rounding: ensure within [0,1] and exact sum
    best_h_cont = np.clip(best_h_cont, 0.0, 1.0)

            diff = target_sum - best_h_cont.sum()

    final_c = compute_c5(best_h_cont)
    return best_h_cont.tolist(), float(final_c), int(n)

import time
import numpy as np
from scipy.optimize import minimize

# -----------------------------------------------------------
# Global helpers for the smooth surrogate used by SLSQP
# -----------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    """Smooth log‑sum‑exp surrogate of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


# -----------------------------------------------------------
# Evaluation
# -----------------------------------------------------------
def compute_c5(h):
    """Exact discrete C₅ for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


# -----------------------------------------------------------
# Continuous optimisation helpers
# -----------------------------------------------------------
def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny exact‑sum repair
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            if abs(add) < 1e-12:
                continue
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def hill_climb_continuous(h_start, rng, max_iters=2500, start_step=0.5):
    """Local hill‑climb on the exact C₅ (preserves the sum)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(best_h)
    step = start_step
    n = h.size
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)
        max_pos = min(h[i], 1.0 - h[j])
        max_neg = min(h[j], 1.0 - h[i])
        if max_pos + max_neg == 0:
            continue
        eps = (rng.random() * (max_pos + max_neg)) - max_neg
        eps *= step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


# -----------------------------------------------------------
# Binary utilities (0/1 vectors with exactly n/2 ones)
# -----------------------------------------------------------
def random_binary(n, rng):
    idx = rng.choice(n, n // 2, replace=False)
    h = np.zeros(n, dtype=np.float64)
    h[idx] = 1.0
    return h


def binary_from_continuous(v):
    n = v.size
    idx = np.argsort(v)[-n // 2 :]
    h = np.zeros(n, dtype=np.float64)
    h[idx] = 1.0
    return h


def repair_binary(vec, n, rng):
    """Force the vector to contain exactly n/2 ones."""
    target = n // 2
    total = int(vec.sum())
    if total > target:
        excess = total - target
        ones = np.where(vec == 1)[0]
        if excess > 0 and len(ones) > 0:
            flip = rng.choice(ones, excess, replace=False)
            vec[flip] = 0.0
    elif total < target:
        deficit = target - total
        zeros = np.where(vec == 0)[0]
        if deficit > 0 and len(zeros) > 0:
            flip = rng.choice(zeros, deficit, replace=False)
            vec[flip] = 1.0
    return vec


def hill_climb_binary(h, rng, max_iters=2000):
    """Swap‑based hill‑climb on binary vectors."""
    n = h.size
    best = h.copy()
    best_c = compute_c5(best)
    for _ in range(max_iters):
        ones = np.where(best == 1)[0]
        zeros = np.where(best == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        new = best.copy()
        new[i] = 0
        new[j] = 1
        c = compute_c5(new)
        if c < best_c:
            best = new
            best_c = c
    return best, best_c

def genetic_algorithm_binary(n, rng, time_limit):
    pop_size = 40
    elite_size = 4
    tournament_k = 3
    prob_crossover = 0.9
    prob_mutation = 0.3

    # initial population

    if "initial_h_values" in globals():

            pop[0] = binary_from_continuous(v0)

    fitness = [compute_c5(ind) for ind in pop]
    best_idx = int(np.argmin(fitness))
    best_h = pop[best_idx].copy()
    best_c = fitness[best_idx]

            cand_f = [fitness[i] for i in cand]
            return pop[cand[np.argmin(cand_f)]]

            p2 = tournament()
            child = p1.copy()
            if rng.random() < prob_crossover:

        if cur_best_c < best_c:

            best_h = pop[cur_best_idx].copy()
    return best_h, best_c

def continuous_refine(h_start, rng, time_limit):
