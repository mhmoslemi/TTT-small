import time, numpy as np, math

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def sum_eq_correction(h, target, rng):
    diff = target - h.sum()
    n = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    # initial random uniform
    h = rng.random(n)
    target = n / 2.0
    # scale to sum target
    h = h * (target / h.sum())
    # clip to 0-1
    np.clip(h, 0.0, 1.0, out=h)
    # correct sum
    h = sum_eq_correction(h, target, rng)
    return h

def worst_shift_index(h):
    # returns shift index k (>=0) that achieves max overlap
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    return abs(idx)

def guided_hill_climb(h, rng, max_iters=2000, step_scale=0.5):
    # similar to previous but using gradient approach
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    for it in range(max_iters):
        # compute worst gradient
        n = h.size
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        grad = np.ones_like(h)
        if k == 0:
            grad = 1 - 2*h
        else:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
        # perturb with guided transfer
        i = np.argmax(grad)
        j = np.argmin(grad)
        if i != j:
            max_eps = min(h[i], 1 - h[j])
            eps = rng.random() * max_eps * step_scale
            h[i] -= eps
            h[j] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        # Optionally reduce step_scale over time
        if (it+1) % (max_iters // 5 + 1) == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # compute softmax (log-sum-exp)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

import time, numpy as np
from scipy.optimize import minimize

# ---------- utility functions ----------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h in-place so that sum(h) == target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random vector with 0≤h≤1 and sum = n/2."""
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    np.clip(h, 0.0, 1.0, out=h)
    return enforce_sum(h, target, rng)


# ---------- gradient based hill climb ----------
def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k = Σ h_i (1‑h_{i+k})."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]      # -h_{i+k}
        grad[k:] -= h[: n - k]      # -h_{i‑k}
    return grad


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                     max_iters: int = 2000, step_scale: float = 0.5):
    """Iterative transfer from positive‑gradient to negative‑gradient indices."""
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    for it in range(max_iters):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            eps = rng.random() * max_eps * step_scale
            h[i] -= eps
            h[j] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        # slowly shrink step_scale
        if (it + 1) % (max_iters // 5 + 1) == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


# ---------- SLSQP with soft‑max surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 400) -> np.ndarray:
    """Soft‑max SLSQP refinement for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(_softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x
