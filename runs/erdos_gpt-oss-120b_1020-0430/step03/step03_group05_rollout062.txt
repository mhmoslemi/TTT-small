def simulated_annealing_binary(n, rng, time_limit, init_h=None):
    if init_h is None:
        # random balanced binary vector: choose n/2 ones uniformly
        idx = rng.permutation(n)[:n//2]
        h = np.zeros(n, dtype=np.int8)
        h[idx] = 1
    else:
        # binarize init_h (float)
        # choose top n/2 values as 1
        order = np.argsort(-init_h)  # descending
        h = np.zeros(n, dtype=np.int8)
        h[order[:n//2]] = 1

    # maintain a = 1 - h (int)
    a = 1 - h
    # correlation
    corr = np.correlate(h, a, mode='full')
    # max
    cur_c = np.max(corr) * (2.0 / n)

    # ones_arr and zeros_arr as fixed-length numpy arrays
    ones_arr = np.where(h==1)[0]
    zeros_arr = np.where(h==0)[0]

    idx_in_ones = np.full(n, -1, dtype=int)
    idx_in_zeros = np.full(n, -1, dtype=int)
    idx_in_ones[ones_arr] = np.arange(len(ones_arr))
    idx_in_zeros[zeros_arr] = np.arange(len(zeros_arr))

    offset = n - 1

    start = time.time()
    # Temperature schedule
    # initial temperature T0 maybe set as 0.01 * cur_c? but we need scale such that acceptance probability works.
    # We can start with relatively high temperature: maybe 0.02.
    # We'll decrease exponentially.
    # Compute number of moves approx = time_limit / avg_time_per_move. If avg per move maybe 0.0002 sec -> 5k moves per sec => 5e6 moves in 1000 secs.
    # We'll attempt using while loop.
    iteration = 0
    while time.time() - start < time_limit:
        iteration += 1
        # temperature schedule: linear decay
        t = max(0.001, (time_limit - (time.time() - start)) / time_limit)  # from 1 to 0
        # maybe scale factor * cur_c
        T = 0.05 * t  # adjust

        # random swap
        i1 = rng.integers(0, len(ones_arr))
        i2 = rng.integers(0, len(zeros_arr))
        p = ones_arr[i1]
        q = zeros_arr[i2]

        # compute delta
        # term1
        delta = np.zeros(2*n-1, dtype=np.int16)   # note int16 may overflow? n=300, contributions up to maybe ~100, safe. but also term2 may be up to n/2, ~150, still within int16 (max 32767). safe.
        # using a slice for term1
        delta[p:p+n] -= a
        delta[q:q+n] += a

        # term2
        idx_p = p - ones_arr + offset
        # filter in bounds
        mask = (idx_p >= 0) & (idx_p < 2*n-1)
        np.add.at(delta, idx_p[mask], 1)
        idx_q = q - ones_arr + offset
        mask = (idx_q >= 0) & (idx_q < 2*n-1)
        np.add.at(delta, idx_q[mask], -1)

        # term3
        delta[offset] -= 2
        d_shift = q - p
        delta[offset + d_shift] += 1
        delta[offset - d_shift] += 1

        # compute new max
        # compute new_corr = corr + delta
        new_corr = corr + delta
        new_c = np.max(new_corr) * (2.0 / n)

        # acceptance
        accept = False
        if new_c < cur_c:
            accept = True
        else:
            # probability
            # delta_c = new_c - cur_c >0
            # acceptance probability = exp(-delta_c / T)
            # avoid overflow: if T small, probability tiny.
            # Compute random float
            if T > 0:
                prob = np.exp(-(new_c - cur_c) / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            # update state
            h[p] = 0
            h[q] = 1
            a[p] = 1
            a[q] = 0
            corr = new_corr
            cur_c = new_c

            # update ones_arr and zeros_arr
            ones_arr[i1] = q
            zeros_arr[i2] = p
            idx_in_ones[p] = -1
            idx_in_ones[q] = i1
            idx_in_zeros[p] = i2
            idx_in_zeros[q] = -1

        # optionally early exit if cur_c <= target bound (0.3808)
        # but we still may want to find even lower.
    return h.astype(float), cur_c

def sa_optimize_binary(n, rng, time_limit, init_h=None):
    # initialize
    if init_h is None:
        # random
        idx = rng.permutation(n)[:n//2]
        h = np.zeros(n, dtype=np.int8)
        h[idx] = 1
    else:
        # binarize init_h into binary
        order = np.argsort(-init_h)  # descending
        h = np.zeros(n, dtype=np.int8)
        h[order[:n//2]] = 1

    a = 1 - h
    corr = np.correlate(h, a, mode='full')
    cur_max = int(corr.max())
    cur_c = cur_max * (2.0 / n)

    # arrays
    ones_arr = np.where(h == 1)[0]
    zeros_arr = np.where(h == 0)[0]
    idx_in_ones = np.full(n, -1, dtype=np.int32)
    idx_in_zeros = np.full(n, -1, dtype=np.int32)
    idx_in_ones[ones_arr] = np.arange(len(ones_arr))
    idx_in_zeros[zeros_arr] = np.arange(len(zeros_arr))

    best_h = h.copy()
    best_c = cur_c

    offset = n - 1
    target_max = int(np.floor(0.3808 * n / 2 + 1e-12))
    target_c = target_max * (2.0 / n)

    start = time.time()
    # preallocate delta array
    delta = np.empty(2*n-1, dtype=np.int32)

    while time.time() - start < time_limit:
        # random swap indices
        i1 = rng.integers(0, len(ones_arr))
        i2 = rng.integers(0, len(zeros_arr))
        p = ones_arr[i1]
        q = zeros_arr[i2]

        # compute delta
        delta.fill(0)
        # term1
        delta[p:p+n] -= a
        delta[q:q+n] += a
        # term2 using vectorized indices
        idx = p - ones_arr + offset
        mask = (idx >= 0) & (idx < 2*n-1)
        np.add.at(delta, idx[mask], 1)
        idx = q - ones_arr + offset
        mask = (idx >= 0) & (idx < 2*n-1)
        np.add.at(delta, idx[mask], -1)
        # term3
        delta[offset] -= 2
        shift = q - p
        delta[offset + shift] += 1
        delta[offset - shift] += 1

        # new correlation and max
        new_corr = corr + delta
        new_max = int(new_corr.max())
        new_c = new_max * (2.0 / n)

        # acceptance
        accept = False
        if new_c < cur_c:

        else:

            elapsed = time.time() - start
            remaining_frac = max(0.0, (time_limit - elapsed) / time_limit)

            h[q] = 1
            a[p] = 1
            a[q] = 0

            cur_c = new_c
            cur_max = new_max

            zeros_arr[i2] = p
            idx_in_ones[p] = -1
            idx_in_ones[q] = i1
            idx_in_zeros[p] = i2
            idx_in_zeros[q] = -1

                best_h = h.copy()
                # early stopping if best meets target
                if best_c <= target_c + 1e-12:
                    break

    return best_h.astype(float), best_c

import time
import numpy as np
from scipy.optimize import minimize

# ---------- helpers for continuous refinement ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact C5 for a step function (float h)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def slsqp_optimize(h0, beta, rng, max_iter=500):
    """One SLSQP refinement with the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def refine_slsqp(h0, rng, time_limit):
    """Run a quick cascade of SLSQP refinements."""
    start = time.time()
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    for beta in (10.0, 50.0, 200.0, 800.0, 2000.0):
        if time.time() - start > time_limit:
            break
        h1 = slsqp_optimize(best_h, beta, rng, max_iter=500)
        c1 = compute_c5(h1)
        if c1 < best_c:
            best_c, best_h = c1, h1.copy()
    return best_h, best_c


# ---------- binary simulated‑annealing optimizer ----------
def sa_optimize_binary(n, rng, time_limit, init_vals=None):
    """Binary (0/1) simulated‑annealing with greedy max‑reduction."""
    # initialise binary vector
    half = n // 2
    if init_vals is not None:
        order = np.argsort(-init_vals)  # descending
        h = np.zeros(n, dtype=np.int8)
        h[order[:half]] = 1
    else:
        idx = rng.permutation(n)[:half]
        h = np.zeros(n, dtype=np.int8)
        h[idx] = 1

    a = 1 - h
    corr = np.correlate(h, a, mode="full")
    cur_max = int(corr.max())
    cur_c = cur_max * (2.0 / n)

    # bookkeeping structures
    ones_arr = np.where(h == 1)[0]
    zeros_arr = np.where(h == 0)[0]
    idx_in_ones = np.full(n, -1, dtype=np.int32)
    idx_in_zeros = np.full(n, -1, dtype=np.int32)
    idx_in_ones[ones_arr] = np.arange(half)
    idx_in_zeros[zeros_arr] = np.arange(half)

    best_h = h.copy()
    best_c = cur_c

    offset = n - 1
    target_max = int(np.floor(0.3808 * n / 2.0 + 1e-12))
    target_c = target_max * (2.0 / n)

    delta = np.empty(2 * n - 1, dtype=np.int32)
    start = time.time()

    while time.time() - start < time_limit:
        # ----- try a greedy swap that surely reduces the current maximum -----
        max_idxs = np.where(corr == cur_max)[0]
        performed_greedy = False
        for max_idx in max_idxs:
            m = int(max_idx) - offset
            # bad ones: h[i]=1, h[i+m]=0
            i_plus = ones_arr + m
            mask = (i_plus >= 0) & (i_plus < n)
            bad_candidates = ones_arr[mask]
            bad = bad_candidates[h[i_plus[mask]] == 0]
            if bad.size == 0:
                continue
            # good zeros: h[j]=0, h[j+m]=1
            j_plus = zeros_arr + m
            mask2 = (j_plus >= 0) & (j_plus < n)
            good_candidates = zeros_arr[mask2]
            good = good_candidates[h[j_plus[mask2]] == 1]
            if good.size == 0:
                continue
            p = rng.choice(bad)
            q = rng.choice(good)
            i1 = idx_in_ones[p]
            i2 = idx_in_zeros[q]

            # compute delta for this specific swap
            delta.fill(0)
            delta[p:p + n] -= a
            delta[q:q + n] += a
            idx = p - ones_arr + offset
            mask = (idx >= 0) & (idx < 2 * n - 1)
            np.add.at(delta, idx[mask], 1)
            idx = q - ones_arr + offset
            mask = (idx >= 0) & (idx < 2 * n - 1)
            np.add.at(delta, idx[mask], -1)
            delta[offset] -= 2
            shift = q - p
            delta[offset + shift] += 1
            delta[offset - shift] += 1

            new_corr = corr + delta
            new_max = int(new_corr.max())
            new_c = new_max * (2.0 / n)

            # accept unconditionally (greedy improvement)
            h[p] = 0
            h[q] = 1
            a[p] = 1
            a[q] = 0
            corr = new_corr
            cur_max, cur_c = new_max, new_c

            # update bookkeeping
            ones_arr[i1] = q
            zeros_arr[i2] = p
            idx_in_ones[p] = -1
            idx_in_ones[q] = i1
            idx_in_zeros[p] = i2
            idx_in_zeros[q] = -1

            if cur_c < best_c - 1e-14:
                best_c, best_h = cur_c, h.copy()
                if best_c <= target_c + 1e-12:
                    return best_h.astype(float), best_c
            performed_greedy = True
            break  # only one greedy move per iteration

        if performed_greedy:
            continue

        # ----- random swap (simulated annealing) -----
        i1 = rng.integers(0, half)
        i2 = rng.integers(0, half)
        p = ones_arr[i1]
        q = zeros_arr[i2]

        delta.fill(0)
        delta[p:p + n] -= a
        delta[q:q + n] += a
        idx = p - ones_arr + offset
        mask = (idx >= 0) & (idx < 2 * n - 1)
        np.add.at(delta, idx[mask], 1)
        idx = q - ones_arr + offset
        mask = (idx >= 0) & (idx < 2 * n - 1)
        np.add.at(delta, idx[mask], -1)
        delta[offset] -= 2
        shift = q - p
        delta[offset + shift] += 1
        delta[offset - shift] += 1

        new_corr = corr + delta
        new_max = int(new_corr.max())
        new_c = new_max * (2.0 / n)

        accept = False
        if new_c < cur_c:
            accept = True
        else:
            # temperature proportional to distance from target and remaining budget
            elapsed = time.time() - start
            remaining = max(0.0, (time_limit - elapsed) / time_limit)
            T = max(1e-6, (cur_c - target_c) * remaining * 0.5)
            if T > 0 and rng.random() < np.exp(-(new_c - cur_c) / T):
                accept = True

        if accept:
            h[p] = 0
            h[q] = 1
            a[p] = 1
            a[q] = 0
            corr = new_corr
            cur_max, cur_c = new_max, new_c

            ones_arr[i1] = q
            zeros_arr[i2] = p
            idx_in_ones[p] = -1
            idx_in_ones[q] = i1
            idx_in_zeros[p] = i2
            idx_in_zeros[q] = -1

            if cur_c < best_c - 1e-14:
                best_c, best_h = cur_c, h.copy()
                if best_c <= target_c + 1e-12:
                    return best_h.astype(float), best_c

    return best_h.astype(float), best_c


# ---------- main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_all = time.time()

    # choose discretisation size
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        init_vals = np.array(initial_h_values, dtype=float)
    else:
        n = 300
        init_vals = None

    # budget split
    time_sa = budget_s * 0.85
    time_refine = budget_s * 0.12
    # SA optimisation (binary)
    h_bin, c5_bin = sa_optimize_binary(n, rng, time_sa, init_vals)

    # continuous refinement
    h_ref, c5_ref = refine_slsqp(h_bin, rng, time_refine)

    # pick the better of the two
    if c5_ref < c5_bin:
        best_h = h_ref
        best_c = c5_ref
    else:
        best_h = h_bin
        best_c = c5_bin

    # final feasibility enforcement (tiny rounding corrections)
    best_h = np.clip(best_h, 0.0, 1.0)
    target = n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-10:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
        diff = target - best_h.sum()
        if abs(diff) > 1e-12:
            for _ in range(2000):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
                best_h[i] += add
                diff = target - best_h.sum()

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
