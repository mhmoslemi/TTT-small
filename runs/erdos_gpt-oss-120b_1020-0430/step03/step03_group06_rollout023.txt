def refine_max_gap(h, max_iters, epsilon_max=0.05):
    for _ in range(max_iters):
        c = np.correlate(h, 1 - h, mode='full')
        max_c = c.max()
        idx = np.argmax(c)
        # compute lag k
        k = idx - (len(h)-1)
        # compute weight_i = 1 - h_{i+k} (or 1 if out of bounds)
        n = len(h)
        i_range = np.arange(n)
        i_plus_k = i_range + k
        # set weight to 1 for out-of-bounds
        weight = np.ones(n)
        mask = (i_plus_k >= 0) & (i_plus_k < n)
        weight[mask] = 1 - h[i_plus_k[mask]]
        # i candidates where h_i > 0
        eligible_i = np.where(h > 1e-12)[0]
        if eligible_i.size == 0:
            break
        # Among eligible_i, pick i with max weight
        i = eligible_i[np.argmax(weight[eligible_i])]
        # j candidates where h_j < 1
        eligible_j = np.where(h < 1 - 1e-12)[0]
        if eligible_j.size == 0:
            break
        j = eligible_j[np.argmin(weight[eligible_j])]
        # Determine transfer amount
        eps = min(h[i], 1 - h[j], epsilon_max)
        if eps <= 1e-12:
            break
        # Apply
        h[i] -= eps
        h[j] += eps
        new_c = np.max(np.correlate(h, 1 - h, mode='full'))
        if new_c < max_c - 1e-12:
            # improvement, continue
            continue
        else:
            # revert and reduce epsilon perhaps
            h[i] += eps
            h[j] -= eps
            # reduce epsilon_max
            epsilon_max *= 0.5
            if epsilon_max < 1e-9:
                break
    return h

import time
import numpy as np

# ---------- basic utilities ----------
def compute_c5(h):
    """Exact C5 for a step function h on the grid (dx = 2/n)."""
    n = h.size
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / n)


def enforce_sum(h, target, rng):
    """Adjust h so that Σh = target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    if abs(diff) < 1e-12:
        return np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-12:
            continue
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with Σ = n/2."""
    v = rng.exponential(1.0, size=n)
    v = v / v.sum() * (n / 2.0)
    h = np.clip(v, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def random_binary_vector(n, rng):
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h


def perturb_h(h, rng, scale=0.5):
    """Small random transfer between two positions (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    while j == i:
        j = rng.integers(0, n)
    max_pos = min(h[i], 1.0 - h[j])
    max_neg = min(h[j], 1.0 - h[i])
    if max_pos + max_neg == 0:
        return h.copy()
    eps = (rng.random() * (max_pos + max_neg)) - max_neg
    eps *= scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return np.clip(new_h, 0.0, 1.0)


# ---------- local search primitives ----------
def hill_climb(h_start, rng, max_iters=5000, start_step=0.5):
    """Simple greedy hill–climb for the exact C5."""
    h = h_start.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)
        max_pos = min(h[i], 1.0 - h[j])
        max_neg = min(h[j], 1.0 - h[i])
        if max_pos + max_neg == 0:
            continue
        eps = (rng.random() * (max_pos + max_neg)) - max_neg
        eps *= step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c_new = compute_c5(new_h)
        if c_new < cur_c:
            h = new_h
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = new_h.copy()
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


def binary_hill_climb(h_start, rng, max_iters=3000):
    """Hill–climb on binary vectors (swap a 1 with a 0)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    n = h.size
    for _ in range(max_iters):
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        c_new = compute_c5(h)
        if c_new < cur_c:
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c


def refine_max_gap(h, max_iters=2000, epsilon_max=0.1):
    """
    Greedy refinement: reduce the currently maximal lag by moving a small mass
    from a position that contributes heavily to that lag to a position that
    contributes little.
    """
    h = h.copy()
    n = h.size
    for _ in range(max_iters):
        corr = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_c = corr[max_idx]
        k = max_idx - (n - 1)

        # weight_i = 1 - h_{i+k} (or 1 if out of range)
        idx = np.arange(n) + k
        weight = np.ones(n)
        mask = (idx >= 0) & (idx < n)
        weight[mask] = 1.0 - h[idx[mask]]

        # choose source (large weight, h>0) and target (small weight, h<1)
        src_candidates = np.where(h > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = src_candidates[np.argmax(weight[src_candidates])]

        tgt_candidates = np.where(h < 1.0 - 1e-12)[0]
        if tgt_candidates.size == 0:
            break
        j = tgt_candidates[np.argmin(weight[tgt_candidates])]

        eps = min(h[i], 1.0 - h[j], epsilon_max)
        if eps <= 1e-12:
            break

        h[i] -= eps
        h[j] += eps
        new_c = np.max(np.correlate(h, 1 - h, mode="full"))
        if new_c < max_c - 1e-12:
            # improvement, keep change and continue
            continue
        # revert and shrink epsilon
        h[i] += eps
        h[j] -= eps
        epsilon_max *= 0.5
        if epsilon_max < 1e-12:
            break
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----- determine dimension and initial candidates -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size

        candidates = [h0]
    else:

    target = n / 2.0

    # deterministic analytic starters
    for f in (1, 2, 3):
        for fun in (np.sin, np.cos):
            h = 0.5 + 0.5 * fun(np.pi * f * xs)
            h = np.clip(h, 0.0, 1.0)
            candidates.append(enforce_sum(h, target, rng))
    candidates.append(enforce_sum(xs / 2.0, target, rng))
    candidates.append(enforce_sum(1.0 - xs / 2.0, target, rng))

    # random starters
    for _ in range(5):
        candidates.append(random_binary_vector(n, rng))
    for _ in range(5):
        candidates.append(random_feasible_vector(n, rng))

    # evaluate starters
    best_h = None
    best_c = float("inf")
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    global_h = best_h
    global_c = best_c

    # ----- main optimisation loop -----
    while time.time() - start_time < budget_s * 0.88:
        mode = rng.integers(0, 2)  # 0: continuous, 1: binary
        if mode == 0:
            # start from perturbed global best or a fresh random point
            if rng.random() < 0.7:
                start = perturb_h(global_h, rng, scale=0.3)
            else:
                start = random_feasible_vector(n, rng)
            h_opt, c_opt = hill_climb(start, rng, max_iters=6000, start_step=0.5)
        else:

            h_opt, c_opt = binary_hill_climb(start, rng, max_iters=4000)

        # targeted max‑gap refinement
        h_opt = refine_max_gap(h_opt, max_iters=2500, epsilon_max=0.08)
        c_opt = compute_c5(h_opt)

        if c_opt < global_c:
            global_c = c_opt
            global_h = h_opt.copy()

        # occasional deeper local search on the current global best
        if time.time() - start_time < budget_s * 0.70:
            h_fine, c_fine = hill_climb(global_h, rng, max_iters=8000, start_step=0.2)
            h_fine = refine_max_gap(h_fine, max_iters=2000, epsilon_max=0.04)
            c_fine = compute_c5(h_fine)
            if c_fine < global_c:
                global_c = c_fine
                global_h = h_fine.copy()

    # ----- optional up‑sampling -----
    if time.time() - start_time < budget_s * 0.96 and n * 2 <= 720:
        h_up = np.repeat(global_h, 2)          # simple duplication
        n_up = h_up.size
        # sum already correct (2 * target)
        h_up = refine_max_gap(h_up, max_iters=4000, epsilon_max=0.05)

        if c_up < global_c:
            global_h = h_up
            global_c = c_up
            n = n_up

    # ----- final clean‑up -----
    global_h = np.clip(global_h, 0.0, 1.0)
    global_h = enforce_sum(global_h, n / 2.0, rng)
    final_c = compute_c5(global_h)

    return global_h.tolist(), float(final_c), int(n)
