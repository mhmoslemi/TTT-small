def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    step = start_step
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i: j+=1
        # compute max epsilon positive (move from i to j)
        max_pos = min(h[i], 1.0 - h[j])
        max_neg = min(h[j], 1.0 - h[i])
        if max_pos + max_neg == 0: continue
        eps = (rng.random() * (max_pos + max_neg)) - max_neg
        eps *= step
        if eps == 0: continue
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        # clip not needed because eps limited, but use np.clip later
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it+1) % decay == 0:
            step = max(step * 0.7, 1e-6)
    return best_h, best_c

import time

def compute_c5(h):
    """Exact discrete C5 for a step function h on [0,2] with n points."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng):
    """Adjust h so that sum(h) == target while keeping 0<=h_i<=1."""
    diff = target - h.sum()
    n = h.size
    # simple redistribution loop
    for _ in range(5000):
        if abs(diff) < 1e-14:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-14:
            continue
        h[i] += add
        diff = target - h.sum()
    # final tiny correction (if needed)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        for _ in range(1000):
            if abs(diff) < 1e-14:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            if abs(add) < 1e-14:
                continue
            h[i] += add
            diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Return a random vector in [0,1]^n with sum == n/2."""
    v = rng.exponential(1.0, size=n)
    v = v / v.sum() * (n / 2.0)
    v = np.clip(v, 0.0, 1.0)
    return enforce_sum(v, n / 2.0, rng)

def hill_climb(h_start, rng, max_iters=3000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    step = start_step
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        max_pos = min(h[i], 1.0 - h[j])
        max_neg = min(h[j], 1.0 - h[i])
        if max_pos + max_neg == 0:
            continue
        eps = (rng.random() * (max_pos + max_neg)) - max_neg
        eps *= step
        if eps == 0.0:
            continue
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-6)
    return best_h, best_c

def binary_local_search(h_start, rng, max_iters=5000):
    h = h_start.copy()
    n = h.size
    best_c = compute_c5(h)
    for _ in range(max_iters):
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        c = compute_c5(h)
        if c < best_c:
            best_c = c
        else:
            h[i], h[j] = 1.0, 0.0
    return h, best_c

def simulated_annealing(h_start, rng, time_limit):
    h = h_start.copy()
    n = h.size
    c_cur = compute_c5(h)
    best_h = h.copy()
    best_c = c_cur
    start = time.time()
    T0 = 0.005  # initial temperature
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = T0 * (1.0 - frac)  # linear cooling
        # propose a move
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        max_pos = min(h[i], 1.0 - h[j])
        max_neg = min(h[j], 1.0 - h[i])
        if max_pos + max_neg == 0:
            continue
        eps = (rng.random() * (max_pos + max_neg)) - max_neg
        # occasional larger jump
        if rng.random() < 0.02:
            eps *= 2.0
        else:
            eps *= 0.5
        if eps == 0.0:
            continue
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c_new = compute_c5(new_h)
        # record best
        if c_new < best_c:
            best_c = c_new
            best_h = new_h.copy()
        # Metropolis acceptance
        if c_new < c_cur or rng.random() < np.exp(-(c_new - c_cur) / max(T, 1e-12)):
            h = new_h
            c_cur = c_new
    return best_h, best_c

        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 180
        h0 = np.full(n, 0.5)
        h0 = enforce_sum(h0, n / 2.0, rng)

    target_sum = n / 2.0

    # --------------------------------------------------------
    # Baseline evaluation
    # --------------------------------------------------------
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # --------------------------------------------------------
    # First hill‑climb from the provided start
    # --------------------------------------------------------
    if time.time() - start_time < budget_s * 0.70:
        h_opt, c_opt = hill_climb(best_h, rng, max_iters=4000, start_step=0.6)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # --------------------------------------------------------
    # Binary conversion + binary local search
    # --------------------------------------------------------
    if time.time() - start_time < budget_s * 0.65:
        # top‑half threshold
        thresh = np.partition(best_h, -int(target_sum))[ -int(target_sum) ]
        bin_h = (best_h >= thresh).astype(float)
        bin_h = enforce_sum(bin_h, target_sum, rng)  # guarantee exact half‑ones
        c_bin = compute_c5(bin_h)
        if c_bin < best_c:
            best_h, best_c = bin_h, c_bin
        # improve binary by swaps
        if time.time() - start_time < budget_s * 0.60:
            bin_h2, c_bin2 = binary_local_search(best_h, rng, max_iters=8000)
            if c_bin2 < best_c:
                best_h, best_c = bin_h2, c_bin2

    # --------------------------------------------------------
    # Simulated annealing on the continuous space
    # --------------------------------------------------------
    while time.time() - start_time < budget_s * 0.94:
        remaining = budget_s - (time.time() - start_time)
        # allocate a modest slice (no less than 20 s) for each SA run
        slice_time = min(30.0, max(20.0, remaining * 0.3))
        # start from current best with a small perturbation
        h_start = best_h.copy()
        # a tiny random jitter to diversify the trajectory
        for _ in range(3):
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            max_pos = min(h_start[i], 1.0 - h_start[j])
            max_neg = min(h_start[j], 1.0 - h_start[i])
            if max_pos + max_neg == 0:
                continue
            eps = (rng.random() * (max_pos + max_neg)) - max_neg
            eps *= 0.2
            h_start[i] -= eps
            h_start[j] += eps
        h_start = np.clip(h_start, 0.0, 1.0)
        h_start = enforce_sum(h_start, target_sum, rng)

        h_sa, c_sa = simulated_annealing(h_start, rng, slice_time)
        if c_sa < best_c:
            best_h, best_c = h_sa, c_sa
            # a quick fine‑tune with a few hill‑climb steps
            if time.time() - start_time < budget_s * 0.90:
                h_fine, c_fine = hill_climb(best_h, rng, max_iters=2000, start_step=0.3)
                if c_fine < best_c:
                    best_h, best_c = h_fine, c_fine

    # --------------------------------------------------------
    # Upsampling (if time permits)
    # --------------------------------------------------------
    if time.time() - start_time < budget_s * 0.98 and n * 2 <= 360:
        up_h = np.repeat(best_h, 2)
        up_n = up_h.size
        up_target = up_n / 2.0
        up_h = enforce_sum(up_h, up_target, rng)
        # a brief refinement on the finer grid
        if time.time() - start_time < budget_s * 0.98:
            up_h, up_c = hill_climb(up_h, rng, max_iters=3000, start_step=0.4)
            if up_c < best_c:
                best_h, best_c, n = up_h, up_c, up_n

    # Final sanity clip & sum enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
