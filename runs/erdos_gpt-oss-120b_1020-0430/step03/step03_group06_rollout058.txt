def hill_climb_targeted(h, rng, max_iters):
    h = h.copy()
    best_c = compute_c5(h)
    n = len(h)
    for it in range(max_iters):
        corr_full = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr_full)
        k = max_idx - (n-1)
        # get overlapping indices
        if k >= 0:
            i_range = np.arange(0, n - k)
            i2_range = i_range + k
        else:
            # negative shift
            i_range = np.arange(-k, n)
            i2_range = i_range + k  # actually i_range + k = i_range - |k|
        # compute contributions

        # pick i where contrib large
        # also pick j where maybe opposite (1 - h[j2]) small
        # random selection weighted
        # compute candidate moves
        i = rng.choice(i_range, p=contrib/np.sum(contrib + 1e-12))
        # for j choose from all indices not in i_range?
        # or from whole set weighted by opposite of contribution at shift k
        # compute opposite measure: (1 - h[j]) * h[j2] ???

        # We'll try moving epsilon from i to random j:
        j = rng.integers(0, n)
        if j == i:
            continue
        # compute allowable epsilon
        max_pos = min(h[i], 1 - h[j])
        max_neg = min(h[j], 1 - h[i])
        # we only want to move from i to j if h[i] > 0 and h[j] < 1

        max_move = max_pos

        new_h[i] -= eps
        new_h[j] += eps
        c = compute_c5(new_h)
        if c < best_c:
            h = new_h
            best_c = c
    return h, best_c

def max_corr_shift(h):
    # compute correlation of h with complement using FFT of real signals.
    # use convolution theorem: corr = ifft( fft(h) * conj(fft(1-h)) )
    # but careful about full convolution length.
    n = h.size
    m = 2*n - 1
    size = 1 << (m-1).bit_length()
    H = np.fft.rfft(np.pad(h, (0,size-n), 'constant'), n=size)
    complement = 1 - h
    C = np.fft.rfft(np.pad(complement, (0,size-n), 'constant'), n=size)
    conv = np.fft.irfft(H * np.conj(C), n=size)[:m]
    # This convolution yields linear convolution where conv[k] = sum_i h[i] * complement[k-i] (with complement zero-padded)
    # The index for shift = k - (n-1)
    idx = np.argmax(conv)
    max_val = conv[idx]
    shift = idx - (n - 1)
    return max_val, shift, conv

from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def _softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-12:
            continue
        h[i] += add
        diff = target - h.sum()
    # final clip
    return np.clip(h, 0.0, 1.0)

def random_feasible_vector(n, rng):
    v = rng.exponential(1.0, size=n)
    v = v / v.sum() * (n / 2.0)
    h = np.clip(v, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)

def max_corr_and_shift(h):
    n = h.size
    m = 2 * n - 1
    size = 1 << ((m - 1).bit_length())
    H = np.fft.rfft(np.concatenate([h, np.zeros(size - n)]))
    C = np.fft.rfft(np.concatenate([1 - h, np.zeros(size - n)]))
    conv = np.fft.irfft(H * np.conj(C), n=size)[:m]
    idx = int(np.argmax(conv))
    max_val = conv[idx]
    shift = idx - (n - 1)
    return max_val, shift

def compute_c5(h):
    max_val, _ = max_corr_and_shift(h)
    return max_val * (2.0 / h.size)

        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        max_pos = min(cur_h[i], 1.0 - cur_h[j])
        max_neg = min(cur_h[j], 1.0 - cur_h[i])
        if max_pos + max_neg == 0:
            continue
        eps = (rng.random() * (max_pos + max_neg)) - max_neg
        # scale by temperature to allow larger moves early?
        eps = eps * (0.5 + 0.5 * T)   # scale between 0.5 and 1?
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        # clip just in case of numeric rounding
        new_h = np.clip(new_h, 0.0, 1.0)
        # sum is preserved by construction (within floating errors)
        # maybe enforce sum a tiny amount
        # new_h = enforce_sum(new_h, n/2.0, rng)  # could be heavy each iteration

            cur_c = new_c

        # occasional perturb of best to diversify

            cur_h = cand
            cur_c = compute_c5(cur_h)
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# helpers for the smooth surrogate (used after stochastic search)
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


# ----------------------------------------------------------------------
# problem‑specific utilities
# ----------------------------------------------------------------------
def enforce_sum(h, target, rng):
    """Adjust h so that Σh = target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        if abs(add) < 1e-12:
            continue
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Uniform random point in the hypercube with Σh = n/2."""
    v = rng.exponential(1.0, size=n)
    v = v / v.sum() * (n / 2.0)
    h = np.clip(v, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def max_corr_and_shift(h):
    """
    Linear correlation of h with (1‑h).  Returns
    (max value, shift) where shift = index‑(n‑1).
    """
    n = h.size
    m = 2 * n - 1
    size = 1 << ((m - 1).bit_length())      # next power of two
    H = np.fft.rfft(np.concatenate([h, np.zeros(size - n)]))
    C = np.fft.rfft(np.concatenate([1 - h, np.zeros(size - n)]))
    conv = np.fft.irfft(H * np.conj(C), n=size)[:m]
    idx = int(np.argmax(conv))
    return conv[idx], idx - (n - 1)


def compute_c5(h):
    """Exact discrete C₅ for a step function h."""
    max_val, _ = max_corr_and_shift(h)
    return max_val * (2.0 / h.size)


def slsqp_optimize(h0, beta, max_iter=800):
    """Refine a point with the smooth log‑sum‑exp surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny exact‑sum repair
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            if abs(add) < 1e-12:
                continue
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ----------------------------------------------------------------------
# stochastic optimisation (simulated annealing)
# ----------------------------------------------------------------------
def simulated_annealing(start_h, rng, budget_s, start_time, fraction=0.45):
    """
    Run a simple SA on the exact C₅ objective.
    Returns the best vector found during the allocated time slice.
    """
    n = start_h.size
    target = n / 2.0
    cur_h = start_h.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c

    # linear temperature schedule: T goes from 1 to 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * fraction:
            break
        T = 1.0 - elapsed / (budget_s * fraction)       # 1 → 0
        # propose a transfer between two coordinates (preserves sum)
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        max_pos = min(cur_h[i], 1.0 - cur_h[j])  # move i→j
        max_neg = min(cur_h[j], 1.0 - cur_h[i])  # move j→i
        if max_pos + max_neg == 0:
            continue
        eps = (rng.random() * (max_pos + max_neg)) - max_neg
        # optionally scale by temperature to allow larger steps early
        eps = eps * (0.5 + 0.5 * T)
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        # tiny sum correction (floating‑point drift)
        diff = target - new_h.sum()
        if abs(diff) > 1e-12:
            # correct on a random non‑saturated index
            k = rng.integers(0, n)
            new_h[k] = np.clip(new_h[k] + diff, 0.0, 1.0)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / (0.001 + 0.02 * T)):
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
        # occasional jump back to best with a small random perturbation
        if rng.random() < 0.0005:
            pert = best_h + (rng.random(n) - 0.5) * 0.02
            pert = np.clip(pert, 0.0, 1.0)
            pert = enforce_sum(pert, target, rng)
            cur_h, cur_c = pert, compute_c5(pert)

    return best_h, best_c


# ----------------------------------------------------------------------
# main entry point
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------- initialise -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
    else:
        n0 = 180
        h0 = np.full(n0, 0.5)
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = enforce_sum(h0, n0 / 2.0, rng)

    # ------- first annealing stage (original resolution) ---------------
    best_h, best_c = simulated_annealing(h0, rng, budget_s, start_time, 0.45)

    # ------- optional up‑sampling ----------------------------------------
    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.90 and n0 * 2 <= 720:
        # double the resolution by repetition
        h_up = np.repeat(best_h, 2)
        n_up = h_up.size
        target_up = n_up / 2.0
        h_up = enforce_sum(h_up, target_up, rng)

        # anneal on the finer grid for the remaining budget
        best_h_up, best_c_up = simulated_annealing(
            h_up, rng, budget_s, start_time, 0.45
        )
        if best_c_up < best_c:
            best_h, best_c, n0 = best_h_up, best_c_up, n_up

    # ------- refinement with smooth surrogate -----------------------------
    beta_list = [200.0, 800.0, 2000.0, 5000.0]
    for beta in beta_list:
        if time.time() - start_time > budget_s * 0.97:
            break
        try:
            cand = slsqp_optimize(best_h, beta, max_iter=800)
        except Exception:
            continue
        cand_c = compute_c5(cand)
        if cand_c < best_c:
            best_h, best_c = cand, cand_c

    # ------- try pure binary rounding ------------------------------------
    int_target = int(round(best_h.size / 2.0))
    idx_desc = np.argsort(-best_h)
    bin_h = np.zeros_like(best_h)
    bin_h[idx_desc[:int_target]] = 1.0
    bin_c = compute_c5(bin_h)
    if bin_c < best_c:
        best_h, best_c = bin_h, bin_c

    # ------- final clean‑up -----------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
