def hill_climb_greedy(h_start, rng, max_iters, target_sum):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iters):
        corr = np.correlate(h, 1-h, mode='full')
        max_corr = corr.max()
        c_current = max_corr * (2.0 / h.size)
        if c_current < best_c:
            best_c = c_current
            best_h = h.copy()
        # find worst shift index
        idx = np.argmax(corr)
        k = idx - (h.size - 1)
        # compute grad
        n = h.size
        grad = np.ones(n)
        if k >= 0:
            grad[:n - k] -= h[k:]
            grad[k:] -= h[:n - k]
        else:
            kk = -k
            grad[:n - kk] -= h[kk:]
            grad[kk:] -= h[:n - kk]
        # Find pos and neg
        pos = np.where((grad > 1e-9) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-9) & (h < 1 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            # random perturb
            # transfer random amount using perturb_random
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0:
                continue
            eps = max_eps * rng.random()
            h[i] -= eps
            h[j] += eps
            continue
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        # compute max eps
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-12:
            # random perturb small
            eps = max_eps
        else:
            eps = max_eps
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c_new = compute_c5(new_h)
        if c_new <= c_current + 1e-12:
            h = new_h
            if c_new < best_c:
                best_c = c_new
                best_h = new_h.copy()
        else:
            # maybe try half eps
            half_eps = max_eps * 0.5
            if half_eps > 1e-12:
                new_h2 = h.copy()
                new_h2[i] -= half_eps
                new_h2[j] += half_eps
                c_new2 = compute_c5(new_h2)
                if c_new2 < c_current:
                    h = new_h2
                    if c_new2 < best_c:
                        best_c = c_new2
                        best_h = new_h2.copy()
        # optional early exit if best_c is already below target threshold ~0.3808
        # continue
    return best_h, best_c

def enforce_sum(h, target, rng):
    """Clip to [0,1] and adjust to make sum exactly target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i])
            add *= rng.random()
            h[i] += add
            diff -= add
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i])
            sub *= rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        if diff > 0:
            h[i] = min(1.0, h[i] + diff)
        else:
            h[i] = max(0.0, h[i] + diff)
    return h


def random_feasible_vector(n, rng):
    """Uniform random start, repaired to satisfy Σh = n/2."""
    h = rng.random(n)
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, n / 2.0, rng)
    return h


def perturb_random(h, rng, scale=0.2):
    """Transfer ε from i to j (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def compute_c5(h):
    """Discrete C₅ = max_k ∫ h(1−h(·+k))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)
