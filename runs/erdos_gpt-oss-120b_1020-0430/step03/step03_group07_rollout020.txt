def project_sum_bounds(h, target):
    # clip to [0,1]
    h_clipped = np.clip(h, 0, 1)
    diff = target - h_clipped.sum()
    # indices not at bounds
    free = (h_clipped > 0) & (h_clipped < 1)
    # if free empty (unlikely) just return h_clipped
    # allocate diff across free uniformly
    while abs(diff) > 1e-12 and np.any(free):
        free_idx = np.where(free)[0]
        m = free_idx.size
        delta = diff / m
        # update
        h_clipped[free_idx] += delta
        # enforce bounds again
        h_clipped = np.clip(h_clipped, 0, 1)
        diff = target - h_clipped.sum()
        free = (h_clipped > 0) & (h_clipped < 1)
    return h_clipped

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def enforce_sum(h, target, rng):
    """Project h onto the intersection of [0,1]^n and the hyperplane sum=h."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # water‑filling style loop
    while abs(diff) > 1e-12:
        free = (h > 0.0) & (h < 1.0)
        idx = np.where(free)[0]
        if idx.size == 0:
            # fall back to a random admissible move
            i = rng.integers(0, h.size)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
            continue
        delta = diff / idx.size
        # respect bounds per coordinate
        inc = np.minimum(1.0 - h[idx], np.maximum(-h[idx], delta))
        h[idx] += inc
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Uniform random point in [0,1]^n with prescribed sum=n/2."""
    h = rng.random(n)
    target = n / 2.0
    h = h * (target / h.sum())
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, target, rng)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def correlate_full(h):
    """Full linear correlation of h with (1‑h) using FFT (zero‑padding)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)


# ----------------------------------------------------------------------
# Soft‑max surrogate utilities
# ----------------------------------------------------------------------
def softmax_weights(conv, beta):
    """Log‑sum‑exp weights for a given β."""
    bc = beta * conv
    m = bc.max()
    exps = np.exp(bc - m)
    return exps / exps.sum()


def gradient_descent_opt(h0, rng, deadline):
    """Projected gradient descent on the smooth max surrogate."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h, best_c = h.copy(), compute_c5(h)

    beta = 5.0
    it = 0
    while time.time() < deadline and it < 5000:
        conv = correlate_full(h)
        cur_c = conv.max() * (2.0 / n)

        # soft‑max weights
        w = softmax_weights(conv, beta)
        w_rev = w[::-1]

        # convolution size for A,B (w length = 2n‑1)
        L = 1 << ((len(w) + n - 1).bit_length())
        ft_w = np.fft.rfft(w, L)
        ft_wrev = np.fft.rfft(w_rev, L)
        ft_h = np.fft.rfft(h, L)

        convB_full = np.fft.irfft(ft_w * ft_h, L)[: len(w) + n - 1]
        convA_full = np.fft.irfft(ft_wrev * ft_h, L)[: len(w) + n - 1]

        offset = n - 1
        A = convA_full[offset: offset + n]   # Σ_k w_k h[i+k]
        B = convB_full[offset: offset + n]   # Σ_k w_k h[i‑k]

        grad = 1.0 - (A + B)                 # ∇ of the smooth objective

        # step size linked to β (larger β ⇒ smaller steps)
        lr = 0.8 / np.sqrt(beta)

        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = enforce_sum(h_new, target, rng)

        conv_new = correlate_full(h_new)
        new_c = conv_new.max() * (2.0 / n)

        if new_c < cur_c:
            h = h_new
            if new_c < best_c:
                best_h, best_c = h_new.copy(), new_c
            # mild increase of step size after success
            lr = min(lr * 1.05, 2.0)
        else:
            # on failure shrink step size (implicitly via next lr)
            pass

        it += 1
        if it % 500 == 0:
            beta = min(beta * 1.5, 5000.0)

    return best_h, best_c

# Local refinement primitives (hill‑climb, SA, SLSQP)
# ----------------------------------------------------------------------
def hill_climb(h_start, rng, max_iters=1500, start_step=0.4):
    h = h_start.copy()
    best_h, best_c = h.copy(), compute_c5(h)
    step = start_step
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h.copy()
            h = new_h
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


def simulated_annealing(start_h, rng, deadline, T0=0.02, Tf=1e-5, max_iters=1800):
    cur_h, cur_c = start_h.copy(), compute_c5(start_h)
    best_h, best_c = cur_h.copy(), cur_c
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        new_h = perturb_h(cur_h, rng, scale=0.35)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
    return best_h, best_c


# ----------------------------------------------------------------------
# SLSQP with smooth surrogate (same idea as before)
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    conv = correlate_full(x)
    bc = CURRENT_BETA * conv
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny correcting step for the sum
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x
