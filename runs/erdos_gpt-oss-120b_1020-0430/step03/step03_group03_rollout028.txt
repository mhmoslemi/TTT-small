import time, math, numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / h.size)

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    enforce_sum(h, n / 2.0, rng)
    return h

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

def biased_perturb_h(h, rng, scale=0.2):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    k = np.argmax(corr)
    shift = k - (n - 1)
    i0 = max(0, -shift)
    i1 = min(n, n - shift)
    idx = np.arange(i0, i1)
    prod = h[idx] * (1 - h[idx + shift])
    if prod.size == 0:
        return perturb_h(h, rng, scale)
    thr_hi = np.quantile(prod, 0.70)
    thr_lo = np.quantile(prod, 0.30)
    high = idx[prod > thr_hi]
    low = idx[prod < thr_lo]
    if high.size == 0 or low.size == 0:
        return perturb_h(h, rng, scale)
    i = rng.choice(high)
    j = rng.choice(low)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

def simulated_annealing(h0, rng, max_iter, init_temp, final_temp):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    n = h.size
    if max_iter <= 0:
        return best_h, best_c
    log_ratio = math.log(final_temp / init_temp) if init_temp > 0 else 0.0
    for it in range(1, max_iter + 1):
        T = init_temp * math.exp(log_ratio * it / max_iter)
        if rng.random() < 0.7:
            cand = biased_perturb_h(h, rng, scale=0.3)
        else:
            cand = perturb_h(h, rng, scale=0.3)
        cand_c = compute_c5(cand)
        diff = cand_c - cur_c
        if diff < 0 or rng.random() < math.exp(-diff / max(T, 1e-12)):
            h, cur_c = cand, cand_c
            if cand_c < best_c:
                best_c, best_h = cand_c, cand.copy()
    return best_h, best_c

# ---------- SLSQP surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(softmax_obj, h0, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

def local_pair_improve(h, rng, max_iter=2000, eps=0.001):
    best_c = compute_c5(h)
    n = h.size
    for _ in range(max_iter):
        i_candidates = np.where(h > eps)[0]
        if i_candidates.size == 0:
            break
        j_candidates = np.where(h < 1 - eps)[0]
        if j_candidates.size == 0:
            break
        i = rng.choice(i_candidates)
        j = rng.choice(j_candidates)
        max_eps = min(h[i], 1 - h[j], eps)
        if max_eps <= 0:
            continue
        # try two eps values: max_eps and half
        for e in (max_eps, max_eps * 0.5):
            new_h = h.copy()
            new_h[i] -= e
            new_h[j] += e
            c = compute_c5(new_h)
            if c < best_c - 1e-12:
                h, best_c = new_h, c
                break
    return h, best_c

import time, math, numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Exact discrete C5 for step function h (dx=2/n)."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)

def enforce_sum(h, target, rng):
    """Adjust h in‑place so that sum(h)=target while respecting 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    """Generate random h with 0≤h≤1 and sum=n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    enforce_sum(h, n / 2.0, rng)
    return h

def perturb_h(h, rng, scale=0.2):
    """Transfer ε between two coordinates (preserves sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

def biased_perturb_h(h, rng, scale=0.2):
    """Bias transfer away from positions contributing most to the max overlap."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    k = np.argmax(corr)
    shift = k - (n - 1)
    i0 = max(0, -shift)
    i1 = min(n, n - shift)
    idx = np.arange(i0, i1)
    prod = h[idx] * (1 - h[idx + shift])
    if prod.size == 0:
        return perturb_h(h, rng, scale)
    thr_hi = np.quantile(prod, 0.70)
    thr_lo = np.quantile(prod, 0.30)
    high = idx[prod > thr_hi]
    low = idx[prod < thr_lo]
    if high.size == 0 or low.size == 0:
        return perturb_h(h, rng, scale)
    i = rng.choice(high)
    j = rng.choice(low)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

def simulated_annealing(h0, rng, max_iter, init_temp, final_temp):
    """Simple SA on C5 (lower is better)."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    n = h.size
    if max_iter <= 0:
        return best_h, best_c
    log_ratio = math.log(final_temp / init_temp) if init_temp > 0 else 0.0
    for it in range(1, max_iter + 1):
        T = init_temp * math.exp(log_ratio * it / max_iter)
        cand = biased_perturb_h(h, rng, scale=0.3) if rng.random() < 0.7 else perturb_h(h, rng, scale=0.3)
        cand_c = compute_c5(cand)
        diff = cand_c - cur_c
        if diff < 0 or rng.random() < math.exp(-diff / max(T, 1e-12)):
            h, cur_c = cand, cand_c
            if cand_c < best_c:
                best_c, best_h = cand_c, cand.copy()
    return best_h, best_c

# ---------- SLSQP surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on a smooth max‑approximation."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {"type": "eq", "fun": sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

def local_pair_improve(h, rng, max_iter=2000, eps=0.001):
    """Greedy pairwise mass transfer to lower exact C5."""
    best_c = compute_c5(h)
    n = h.size
    for _ in range(max_iter):
        i_cand = np.where(h > eps)[0]
        j_cand = np.where(h < 1 - eps)[0]
        if i_cand.size == 0 or j_cand.size == 0:
            break
        i = rng.choice(i_cand)
        j = rng.choice(j_cand)
        max_eps = min(h[i], 1 - h[j], eps)
        if max_eps <= 0:
            continue
        for e in (max_eps, max_eps * 0.5):
            new_h = h.copy()
            new_h[i] -= e
            new_h[j] += e
            c = compute_c5(new_h)
            if c < best_c - 1e-12:
                h, best_c = new_h, c
                break
    return h, best_c
