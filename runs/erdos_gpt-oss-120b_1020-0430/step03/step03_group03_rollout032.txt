    prod = h[idx] * (1 - h[idx + shift])
    if prod.size == 0:
        return perturb_h(h, rng, scale)
    thr_hi = np.quantile(prod, 0.70)
    thr_lo = np.quantile(prod, 0.30)
    high = idx[prod > thr_hi]
    low = idx[prod < thr_lo]
    if high.size == 0 or low.size == 0:
        return perturb_h(h, rng, scale)
    i = rng.choice(high)
    j = rng.choice(low)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new


# ---------- smooth surrogate ----------
def smooth_obj_and_grad(h, beta, target, penalty):
    """Log‑sum‑exp surrogate for max overlap and its gradient."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    beta_corr = beta * corr
    max_bc = np.max(beta_corr)
    exp_bc = np.exp(beta_corr - max_bc)
    sum_exp = np.sum(exp_bc)
    lse = max_bc + np.log(sum_exp)
    f = lse / beta + penalty * (np.sum(h) - target) ** 2
    w = exp_bc / sum_exp                     # weights for each shift
    grad = np.zeros_like(h)

    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        k = idx - (n - 1)
        if k >= 0:
            i_start = 0
            i_end = n - k
            grad[i_start:i_end] += wk * (1.0 - h[i_start + k:i_end + k])
            i2_start = k
            i2_end = n
            grad[i2_start:i2_end] -= wk * h[i2_start - k:i2_end - k]
        else:
            i_start = -k
            i_end = n
            grad[i_start:i_end] += wk * (1.0 - h[i_start + k:i_end + k])
            i2_start = 0
            i2_end = n + k
            grad[i2_start:i2_end] -= wk * h[i2_start - k:i2_end - k]

    if penalty != 0.0:
        grad += 2.0 * penalty * (np.sum(h) - target) * np.ones_like(h)
    return f, grad


def smooth_obj_and_grad_wrapper(x, beta, target, penalty):
    return smooth_obj_and_grad(x, beta, target, penalty)


def local_optimize(h0, beta_seq, target, rng):
    """Refine h by a cascade of L‑BFGS‑B runs on increasing β."""
    h = np.clip(h0, 0.0, 1.0)
    enforce_sum(h, target, rng)
    n = h.size
    bounds = [(0.0, 1.0)] * n
    penalty = 100.0
    for beta in beta_seq:
        res = minimize(
            smooth_obj_and_grad_wrapper,
            h,
            args=(beta, target, penalty),
            method="L-BFGS-B",
            jac=True,
            bounds=bounds,
            options={"maxiter": 200, "ftol": 1e-12, "disp": False},
        )
        h = np.clip(res.x, 0.0, 1.0)
        enforce_sum(h, target, rng)
    return h


def hill_climb_refine(h, rng, max_iter, scale=0.1):
    """Simple hill‑climbing using biased transfers."""
    best_h = h.copy()
    best_c = compute_c5(best_h)
    for _ in range(max_iter):
        cand = biased_perturb_h(best_h, rng, scale=scale)
        cand_c = compute_c5(cand)
        if cand_c < best_c:
            best_h, best_c = cand, cand_c
    return best_h, best_c


# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----- initial vector -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
    else:
        n = 180
        h0 = random_feasible_vector(n, rng)

    target = n / 2.0
    enforce_sum(h0, target, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    beta_seq = [10, 50, 200, 800, 2000, 8000, 20000]

    # ----- gradient‑based exploration -----
    grad_budget = budget_s * 0.75
    while time.time() - start_time < grad_budget:
        if rng.random() < 0.7:
            start = perturb_h(best_h, rng, scale=0.3)
        else:
            start = random_feasible_vector(n, rng)

        h_opt = local_optimize(start, beta_seq, target, rng)
        c_opt = compute_c5(h_opt)

        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            if best_c <= 0.3808:
                break

    # ----- final hill‑climbing refinement -----
    while time.time() - start_time < budget_s:
        cand = biased_perturb_h(best_h, rng, scale=0.08)
        cand_c = compute_c5(cand)
        if cand_c < best_c:
            best_h, best_c = cand, cand_c

    # ----- enforce constraints and return -----
    best_h = np.clip(best_h, 0.0, 1.0)
    enforce_sum(best_h, target, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
