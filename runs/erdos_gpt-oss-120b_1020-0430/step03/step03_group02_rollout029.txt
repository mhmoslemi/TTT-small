def enforce_sum(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # Clip first to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # adjust diff by random redistribution
    # limit attempts to avoid infinite loop
    attempts = 0
    max_attempts = 2000
    while abs(diff) > 1e-12 and attempts < max_attempts:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0:
            h[i] += add
            diff = target - h.sum()
        attempts += 1
    # if still not exact due to rounding, adjust last element
    if abs(diff) > 1e-12:
        # adjust any index that can accommodate diff
        for i in range(n):
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) <= 1e-12:
                    break
            elif diff < 0 and h[i] > 0.0:
                add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) <= 1e-12:
                    break
    return h

def guided_hill_climb(h_start, rng, max_iters=3000, step_scale=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)

        c = compute_c5(nh)

            best_h = nh.copy()
            h = nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

def ga_optimize(n, rng, time_limit):
    target = n/2.0
    pop_size = 30
    # init pop
    pop = [enforce_sum(rng.random(n), rng) for _ in range(pop_size)]
    best_h = None; best_c = np.inf

    # evaluate initial
    scores = [compute_c5(ind) for ind in pop]
    # update best
    idx_best = np.argmin(scores)
    best_h = pop[idx_best].copy()
    best_c = scores[idx_best]

    generation = 0
    while time.time() < time_limit and generation < 2000:
        # sort indices by score ascending
        sorted_idx = np.argsort(scores)
        pop = [pop[i] for i in sorted_idx]
        scores = [scores[i] for i in sorted_idx]
        # elite
        elite_num = max(2, pop_size // 5)
        new_pop = pop[:elite_num]
        # generate rest
        while len(new_pop) < pop_size:
            # select parents uniformly
            i, j = rng.integers(0, elite_num, size=2)
            p1, p2 = pop[i], pop[j]
            # crossover
            mask = rng.integers(0, 2, size=n).astype(bool)
            child = np.where(mask, p1, p2).astype(np.float64)
            # mutation: add Gaussian noise to some entries
            if rng.random() < 0.7:
                sigma = 0.05
                noise = rng.normal(scale=sigma, size=n)
                child += noise
            # clip and enforce
            child = np.clip(child, 0.0, 1.0)
            child = enforce_sum(child, rng)
            # evaluate
            score = compute_c5(child)
            new_pop.append(child)
        # replace pop
        pop = new_pop
        scores = [compute_c5(ind) for ind in pop]
        # update best
        idx_best = np.argmin(scores)
        if scores[idx_best] < best_c:
            best_c = scores[idx_best]; best_h = pop[idx_best].copy()
        generation += 1
        # early exit if bound achieved
        if best_c <= 0.3808:
            break
    return best_h, best_c
