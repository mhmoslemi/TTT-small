def perturb_v(v, rng, scale=0.1):
    m = v.size
    i = rng.integers(m)
    j = rng.integers(m)
    if i == j: return v.copy()
    max_eps = min(v[i], 1 - v[j])
    if max_eps <= 0: return v.copy()
    eps = (rng.random() - 0.5) * 2 * max_eps * scale
    eps = np.clip(eps, -max_eps, max_eps)
    new_v = v.copy()
    new_v[i] -= eps
    new_v[j] += eps
    return new_v
```

def hill_climb_v(v_start, rng, max_iters=2000, init_step=0.3, n_trials=5):
    v = v_start.copy()
    best_c = compute_c5(v)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            new_v = perturb_v(v, rng, scale=step)
            c = compute_c5(new_v)
            if c < best_c:
                best_c = c
                v = new_v
                improved = True
                break
        if not improved and rng.random() < 0.01:
            new_v = perturb_v(v, rng, scale=step*2)
            c = compute_c5(new_v)
            if c < best_c:
                best_c = c
                v = new_v
        if (it+1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return v, best_c
```

def enforce_sum_h(h):
    """Adjust h so that sum(h)=n/2 while staying in [0,1]."""
    target = h.size / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    rng = np.random.default_rng()
    N = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, N)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        # uniform correction as a last resort
        correction = diff / N
        h += correction
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(N):
                if abs(diff) < 1e-12:

                h[i] += add

    return h


def enforce_sum_v(v):

def compute_c5_h(h):
    """Exact discrete C5 for a full‑resolution vector h."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

    """C5 for the reduced‑dimension vector v (repeated to length n)."""
    h = np.repeat(v, block_len)[:n]

def random_feasible_vector_v(m, rng):

def perturb_v(v, rng, scale=0.2):
    """Transfer a small amount between two coordinates (preserves sum)."""
    m_ = v.size
    i = rng.integers(m_)
    j = rng.integers(m_)
    if i == j:
        return v.copy()
    max_eps = min(v[i], 1.0 - v[j])
    if max_eps <= 0.0:
        return v.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_v = v.copy()
    new_v[i] -= eps
    new_v[j] += eps
    return new_v


def hill_climb_v(v_start, rng, max_iters=2000, init_step=0.5, n_trials=5):
    """Simple hill‑climb on the reduced problem."""
    v = v_start.copy()
    best_c = compute_c5_from_v(v)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            new_v = perturb_v(v, rng, scale=step)
            c = compute_c5_from_v(new_v)
            if c < best_c:
                best_c = c
                v = new_v
                improved = True
                break
        if not improved and rng.random() < 0.02:
            new_v = perturb_v(v, rng, scale=step * 2.0)
            c = compute_c5_from_v(new_v)
            if c < best_c:
                best_c = c
                v = new_v
        if (it + 1) % decay_interval == 0:

    return v, best_c


def slsqp_optimize_v(v0, beta, max_iter=300):
    """SLSQP optimisation of the smooth max‑surrogate on the reduced space."""
    global CURRENT_BETA, n, block_len, m
    CURRENT_BETA = beta
    target = m / 2.0
    bounds = [(0.0, 1.0)] * m

    def sum_eq(x):
        return np.sum(x) - target

    def soft_obj(x):
        h = np.repeat(x, block_len)[:n]
        corr = np.correlate(h, 1 - h, mode="full")
        bc = CURRENT_BETA * corr
        mmax = np.max(bc)
        return (mmax + np.log(np.sum(np.exp(bc - mmax)))) * (2.0 / n) / CURRENT_BETA

    cons = {"type": "eq", "fun": sum_eq}
    res = minimize(
        soft_obj,
        v0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = enforce_sum_v(x)
    return x
