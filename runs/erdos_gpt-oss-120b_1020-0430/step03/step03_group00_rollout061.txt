def smooth_obj_and_grad(h, beta):
    # h is array shape (n,)
    n = h.shape[0]
    # compute corr using np.correlate
    corr = np.correlate(h, 1 - h, mode='full')
    # compute exponentials
    exp_term = np.exp(beta * (corr - np.max(corr))) # subtract max for stability
    sum_exp = np.sum(exp_term)
    f = (np.max(corr) + np.log(sum_exp)) / beta  # because log-sum-exp = max + log(sum(exp(beta*(c - max))))
    # compute p = exp(beta * (corr - max_corr)) / sum_exp
    p = exp_term / sum_exp  # probabilities
    # compute gradient:
    # p vector length M = 2n-1, index mapping: k from -(n-1) to n-1 => p_idx = k + n-1.
    # we need to compute grad_i = sum_k p_k * ((1 - h[i+k]) if i+k in bounds else 0) - sum_k p_k * (h[i-k] if i-k in bounds else 0)
    # Use broadcasting:
    k = np.arange(-(n-1), n)  # shape (M,)
    # pad h with zeros on both ends pad=n-1
    pad = n - 1
    hpad = np.pad(h, (pad, pad), mode='constant')
    # For term1: (1 - h[i+k])
    # Compute indices matrix i + k + pad for each i,k
    i_arr = np.arange(n)[:, None]  # shape (n,1)
    idx1 = i_arr + k + pad  # shape (n, M)
    term1 = np.sum(p * (1 - hpad[idx1]), axis=1)
    # For term2: h[i-k] = h[i - k]
    idx2 = i_arr - k + pad
    term2 = np.sum(p * hpad[idx2], axis=1)
    grad = term1 - term2
    return f, grad
```

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utilities ----------
def enforce_sum(h, target, rng):
    """Clip to [0,1] and adjust to make sum exactly target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i])
            add *= rng.random()
            h[i] += add
            diff -= add
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i])
            sub *= rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        if diff > 0:
            h[i] = min(1.0, h[i] + diff)
        else:
            h[i] = max(0.0, h[i] + diff)
    return h

def random_feasible_vector(n, rng):
    h = rng.random(n)
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, n / 2.0, rng)
    return h

def perturb_random(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def perturb_targeted(h, grad, shift, rng, scale=0.2):
    pos = np.where((grad > 1e-9) & (h > 1e-12))[0]
    neg = np.where((grad < -1e-9) & (h < 1.0 - 1e-12))[0]
    if pos.size == 0 or neg.size == 0:
        return perturb_random(h, rng, scale)
    i = rng.choice(pos)
    j = rng.choice(neg)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return perturb_random(h, rng, scale)
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

# ---------- hill climbing ----------
def hill_climb_targeted(h_start, rng, max_iters, start_step, target_sum):
    h = h_start.copy()
    n = h.size
    best_c = compute_c5(h)
    best_h = h.copy()
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr)
        max_corr = corr[idx]
        c_curr = max_corr * (2.0 / n)
        k = idx - (n - 1)

        # gradient of the worst shift
        grad = np.ones(n)
        if k >= 0:
            grad[:n - k] -= h[k:]
            grad[k:] -= h[:n - k]
        else:
            kk = -k
            grad[:n - kk] -= h[kk:]
            grad[kk:] -= h[:n - kk]

        new_h = perturb_targeted(h, grad, k, rng, scale=step)
        c_new = compute_c5(new_h)

        if c_new < best_c:
            best_c, best_h = c_new, new_h.copy()
        if c_new < c_curr:
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c

# ---------- smoothed SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    maxc = np.max(corr)
    # stable log-sum-exp
    bc = CURRENT_BETA * (corr - maxc)
    lse = maxc + np.log(np.sum(np.exp(bc))) / CURRENT_BETA
    return lse

def slsqp_optimize(h0, beta, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

# ---------- smooth objective with gradient ----------
def smooth_obj_and_grad(h, beta):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    maxc = np.max(corr)
    bc = beta * (corr - maxc)
    expb = np.exp(bc)
    sum_exp = np.sum(expb)
    f = maxc + np.log(sum_exp) / beta

    p = expb / sum_exp  # probabilities, length 2n-1
    k = np.arange(-(n - 1), n)  # shifts
    pad = n - 1
    hpad = np.pad(h, (pad, pad), mode="constant")
    i_arr = np.arange(n)[:, None]

    idx1 = i_arr + k + pad
    term1 = np.sum(p * (1.0 - hpad[idx1]), axis=1)

    idx2 = i_arr - k + pad
    term2 = np.sum(p * hpad[idx2], axis=1)

    grad = term1 - term2
    return f, grad

def lbfgsb_optimize(h0, beta, max_iter=200):
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def fun(x):
        f, g = smooth_obj_and_grad(x, beta)
        return f, g

    res = minimize(
        fun,
        h0,
        method="L-BFGS-B",
        jac=True,
        bounds=bounds,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = enforce_sum(x, target, np.random.default_rng())
    return x

# ---------- main entry ----------
