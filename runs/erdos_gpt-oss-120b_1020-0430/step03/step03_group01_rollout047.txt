def targeted_perturb(h, rng, lag, top_frac=0.1, scale=0.5):
    n = len(h)
    # compute contributions for positions where i+lag in range
    valid = np.arange(n - lag)
    contrib = h[valid] * (1 - h[valid + lag])
    if contrib.size == 0:
        return perturb_h(h, rng, scale)
    # sort indices by contrib descending
    k = max(1, int(len(valid) * top_frac))
    top_idx = valid[np.argpartition(-contrib, k-1)[:k]]
    bottom_idx = valid[np.argpartition(contrib, k-1)[:k]]
    # sample i from top_idx, j from bottom_idx (or any)
    i = rng.choice(top_idx)
    # choose j either from bottom_idx, or random elsewhere
    j = rng.choice(bottom_idx) if rng.random() < 0.7 else rng.integers(0, n)
    # Ensure i!=j
    if i == j:
        j = (j+1) % n
    # compute max eps for transfer from i to j
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h
```

def enforce_sum(h, target, rng, max_iter=5000):
    """Clip to [0,1] and adjust to meet exact sum `target`."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h
