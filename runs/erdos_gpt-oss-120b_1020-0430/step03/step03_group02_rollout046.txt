def softmax_grad(h, beta):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    exps = np.exp(beta * (corr - max_corr))
    sum_exps = np.sum(exps)
    probs = exps / sum_exps  # w
    # compute gradient
    grad = np.zeros_like(h)
    # center
    center = probs[n-1]
    grad += center * (1 - 2*h)
    for k in range(1, n):
        wk_pos = probs[n-1 + k]
        wk_neg = probs[n-1 - k]
        # positive shift contributions
        grad[:n-k] += wk_pos * (1 - h[k:])
        # negative shift contributions
        grad[k:] += wk_neg * (1 - h[:n-k])
    return (max_corr + np.log(sum_exps)/beta), grad

import time, numpy as np

# -----------------------------------------------------------
# Core utilities
# -----------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2] (dx=2/n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(x: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Exact projection of x onto {0≤x≤1, sum=x_target} using simple transfers."""
    diff = target - x.sum()
    n = x.size
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0.0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        if abs(add) < 1e-12:
            continue
        x[i] += add
        diff = target - x.sum()
    return x


def project_onto_box_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Projection onto [0,1]^n intersect sum=target via a binary search on a shift."""
    lo, hi = -1.0, 1.0
    for _ in range(40):
        lam = (lo + hi) * 0.5
        xp = np.clip(x + lam, 0.0, 1.0)
        s = xp.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random h with 0≤h≤1 and sum=n/2."""
    target = n / 2.0
    h = rng.random(n)
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, target, rng)


# -----------------------------------------------------------
# Gradient of smoothed max (softmax) objective
# -----------------------------------------------------------
def softmax_obj_and_grad(h: np.ndarray, beta: float):
    """Return (objective, gradient) of (1/β)log Σ exp(β·corr)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exp_shift = np.exp(beta * (corr - m))
    sum_exp = exp_shift.sum()
    probs = exp_shift / sum_exp                     # softmax weights

    # gradient accumulation
    grad = np.zeros_like(h)
    center = probs[n - 1]
    grad += center * (1.0 - 2.0 * h)

    for k in range(1, n):
        wk_pos = probs[n - 1 + k]
        wk_neg = probs[n - 1 - k]

        # shift +k
        grad[: n - k] += wk_pos * (1.0 - h[k:])
        # shift -k
        grad[k:] += wk_neg * (1.0 - h[: n - k])

    obj = m + np.log(sum_exp) / beta
    return obj, grad


# -----------------------------------------------------------
# Worst‑shift gradient (sub‑gradient of the true max)
# -----------------------------------------------------------
def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


# -----------------------------------------------------------
# Simple transfer perturbation guided by gradient signs
# -----------------------------------------------------------
def guided_perturb(h: np.ndarray, grad: np.ndarray, rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


# -----------------------------------------------------------
# Greedy hill‑climb using the worst‑shift gradient
# -----------------------------------------------------------
def guided_hill_climb(h_start: np.ndarray, rng: np.random.Generator,
                     max_iters: int = 2000, step_scale: float = 0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps > 0.0:
            eps = rng.random() * max_eps * step_scale
            h[i] -= eps
            h[j] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-5)
    return best_h, best_c
