import time
import numpy as np
from scipy.optimize import minimize

# ---------- utility functions ----------
def compute_c5(h):
    """Discrete C5 for step function h on [0,2] (dx = 2/len(h))."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / h.size)


def enforce_sum(h, target, rng):
    """In‑place adjustment to achieve sum(h)=target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random h with 0≤h≤1 and sum=h_target=n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    enforce_sum(h, n / 2.0, rng)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small ε between two coordinates (preserves the sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new


# ---------- SLSQP with smooth‑max ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ h_i(1‑h_{i+k})."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on the smooth‑max objective."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {"type": "eq", "fun": sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x


# ---------- main optimisation ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # number of discretisation points
    n = 240
    target_sum = n / 2.0

    # initial best solution
    best_h = random_feasible_vector(n, rng)
    best_c = compute_c5(best_h)

    # ---------- random‑restart deterministic annealing ----------
    betas = [10, 50, 200, 800, 3200, 12800, 51200]
    # leave ~100 s for final polishing
    while time.time() - start_time < budget_s - 100:
        # start point: perturb best or fresh random
        if rng.random() < 0.5:
            h0 = perturb_h(best_h, rng, scale=0.2)
        else:
            h0 = random_feasible_vector(n, rng)

        h_cur = h0
        for beta in betas:
            # modest iteration budget, larger for higher β
            if beta < 800:
                iters = 200
            elif beta < 12800:
                iters = 300
            else:
                iters = 500
            try:
                h_cur = slsqp_optimize(h_cur, beta, max_iter=iters)
            except Exception:
                break

        c_cur = compute_c5(h_cur)
        if c_cur < best_c:
            best_c, best_h = c_cur, h_cur.copy()
            # early exit if we already beat target
            if best_c < 0.3808:
                break

    # ---------- high‑β polishing ----------
    final_betas = [200000, 800000, 2000000]
    for beta in final_betas:
        if time.time() - start_time > budget_s - 30:
            break
        try:
            best_h = slsqp_optimize(best_h, beta, max_iter=800)
            best_c = compute_c5(best_h)
        except Exception:
            continue

    # ---------- greedy gradient‑descent on the worst shift ----------
    while time.time() - start_time < budget_s - 5:
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        k_max = np.argmax(corr)
        shift = k_max - (n - 1)

        idx = np.arange(n)
        i_plus = idx + shift
        i_minus = idx - shift
        w = np.zeros(n)
        mask = (i_plus >= 0) & (i_plus < n)
        w[mask] += best_h[i_plus[mask]]
        mask = (i_minus >= 0) & (i_minus < n)
        w[mask] += best_h[i_minus[mask]]

        eps_min = 1e-9
        high_cand = np.where(best_h > eps_min)[0]
        low_cand = np.where(best_h < 1.0 - eps_min)[0]
        if high_cand.size == 0 or low_cand.size == 0:
            break

        i_high = high_cand[np.argmax(w[high_cand])]
        i_low = low_cand[np.argmin(w[low_cand])]
        step = min(0.04, best_h[i_high], 1.0 - best_h[i_low])
        if step <= 0:
            break

        new_h = best_h.copy()
        new_h[i_high] -= step
        new_h[i_low] += step
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_h, best_c = new_h, new_c

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    enforce_sum(best_h, target_sum, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
