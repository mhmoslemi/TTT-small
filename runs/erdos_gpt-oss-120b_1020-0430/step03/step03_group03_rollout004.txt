def project_onto_capped_simplex(v, target, tol=1e-9, max_iter=100):
    # returns x s.t. 0 <= x_i <= 1 and sum x_i = target, minimizing ||x - v||^2.
    # Equivalent to water-filling.
    # Use binary search on lambda.
    # The function s(lambda) = sum(clip(v_i - lambda, 0, 1))
    # Need to find lambda where s(lambda) = target.
    lo = np.min(v - 1) - 1.0  # lower bound for lambda
    hi = np.max(v)  # upper bound
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            lo = lam
        else:
            hi = lam
    # after loop, take midpoint
    x = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    # adjust minor rounding error
    diff = target - x.sum()
    if abs(diff) > 1e-8:
        # distribute to indices not at bound
        free = np.where((x > 0) & (x < 1))[0]
        if free.size > 0:
            add = diff / free.size
            x[free] += add
            x = np.clip(x, 0.0, 1.0)
    return x

import time, math, numpy as np
from scipy.optimize import minimize

# ---------- core utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function h on [0,2] with dx=2/len(h)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_onto_capped_simplex(v: np.ndarray, target: float,
                                tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    """Projection of v onto {0<=x<=1, sum(x)=target} (Euclidean)."""
    lo = np.min(v) - 1.0          # for lambda low enough, all entries become 1
    hi = np.max(v)                # for lambda high enough, all become 0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-8:
        free = np.where((x > 0) & (x < 1))[0]
        if free.size:
            x[free] += diff / free.size
            x = np.clip(x, 0.0, 1.0)
    return x


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random point in [0,1]^n with sum = n/2."""
    v = rng.random(n)
    return project_onto_capped_simplex(v, n / 2.0)


def tournament_select(fitness: list, size: int, rng: np.random.Generator) -> int:
    """Return index of best fitness among `size` random contestants."""
    idx = rng.integers(0, len(fitness), size=size)
    best = idx[0]
    best_f = fitness[best]
    for i in idx[1:]:
        if fitness[i] < best_f:
            best = i
            best_f = fitness[i]
    return best


def produce_child(p1: np.ndarray, p2: np.ndarray, rng: np.random.Generator,
                  target: float, crossover_prob: float = 0.5,
                  mut_rate: float = 0.08, mut_scale: float = 0.2) -> np.ndarray:
    """Uniform crossover + simple mutation, then project to feasible set."""
    mask = rng.random(p1.size) < crossover_prob
    child = np.where(mask, p1, p2).copy()
    mut_mask = rng.random(p1.size) < mut_rate
    if mut_mask.any():
        deltas = (rng.random(mut_mask.sum()) - 0.5) * 2.0 * mut_scale
        child[mut_mask] += deltas
    child = np.clip(child, 0.0, 1.0)
    child = project_onto_capped_simplex(child, target)
    return child


# ---------- smooth‑max SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(x0: np.ndarray, beta: float, max_iter: int = 200) -> np.ndarray:
    """Smooth‑max SLSQP refinement."""
    global CURRENT_TARGET, CURRENT_BETA
    n = x0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {"type": "eq", "fun": sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(softmax_obj, x0, method="SLSQP", bounds=bounds,
                   constraints=cons, options={"maxiter": max_iter,
                                             "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    return project_onto_capped_simplex(x, CURRENT_TARGET)
