def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_h, best_c, best_n = None, np.inf, None

    def compute_c5(h):
        n = h.size
        return float(np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n))

    def enforce_sum_and_bounds(h, n):
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        # redistributive loop
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
        return h

    def random_binary(n):
        h = np.zeros(n)
        idx = rng.choice(n, n//2, replace=False)
        h[idx] = 1.0
        return h

    def perturb_transfer(h, scale):
        n = h.size
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            return h
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            return h
        eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
        eps = max(-max_eps, min(eps, max_eps))
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        return new_h

    def hill_climb(h, max_iters=3000, start_step=0.6):
        n = h.size
        step = start_step
        best_h = h.copy()
        best_c = compute_c5(h)
        # schedule: decay step every some iterations
        decay_iter = max(1, max_iters // 10)
        for it in range(max_iters):
            cand = perturb_transfer(best_h, scale=step)
            c = compute_c5(cand)
            if c < best_c - 1e-12:
                best_c = c
                best_h = cand
            if (it+1) % decay_iter == 0:
                step = max(step*0.9, 1e-5)
        return best_h, best_c

    def simulated_annealing(h, time_deadline):
        n = h.size
        cur_h = h.copy()
        cur_c = compute_c5(cur_h)
        best_h = cur_h.copy()
        best_c = cur_c
        it = 0
        # initial temperature estimate
        T0 = max(0.01 * cur_c, 1e-4)
        while time.time() < time_deadline:
            # geometric cooling
            T = T0 * (0.999 ** it) + 1e-12
            step = max(0.5 * (0.999 ** it), 5e-4)
            cand = perturb_transfer(cur_h, scale=step)
            cand_c = compute_c5(cand)
            if cand_c < cur_c or rng.random() < np.exp(-(cand_c - cur_c) / T):
                cur_h, cur_c = cand, cand_c
                if cur_c < best_c:
                    best_c, best_h = cur_c, cur_h.copy()
            it += 1
        return best_h, best_c

import time
import numpy as np

# ---------- utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for step function h on [0,2] (uniform spacing)."""
    n = h.size
    # full correlation of h with (1-h)
    return float(np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / n))


def enforce_sum_and_bounds(h: np.ndarray, n: int, rng: np.random.Generator) -> np.ndarray:
    """Clip to [0,1] and adjust to exact sum n/2."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    # simple greedy redistribution
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_binary_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random 0/1 vector with exactly n/2 ones."""
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h


def perturb_transfer(h: np.ndarray, rng: np.random.Generator, scale: float) -> np.ndarray:
    """Move a small amount ε from i to j while preserving the sum."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    # ε drawn uniformly in [-max_eps*scale, max_eps*scale]
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(
    h0: np.ndarray,
    rng: np.random.Generator,
    max_iters: int = 5000,
    start_step: float = 0.7,
) -> tuple[np.ndarray, float]:
    """Deterministic hill‑climb (accept only improvements) with step decay."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)

    for it in range(max_iters):
        cand = perturb_transfer(best_h, rng, scale=step)
        c = compute_c5(cand)
        if c < best_c - 1e-14:
            best_c = c
            best_h = cand
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-5)
    return best_h, best_c


def simulated_annealing(
    h0: np.ndarray,
    rng: np.random.Generator,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Simple SA respecting the sum constraint."""
    n = h0.size
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    it = 0
    T0 = max(0.01 * cur_c, 1e-4)

    while time.time() < deadline and it < 200_000:
        # geometric cooling
        T = T0 * (0.999 ** it) + 1e-12
        step = max(0.5 * (0.999 ** it), 5e-4)
        cand = perturb_transfer(cur_h, rng, scale=step)
        cand_c = compute_c5(cand)

        if cand_c < cur_c or rng.random() < np.exp(-(cand_c - cur_c) / T):
            cur_h, cur_c = cand, cand_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
        it += 1
    return best_h, best_c


def gradient_refine(
    h0: np.ndarray,
    rng: np.random.Generator,
    step: float = 0.02,
    max_iter: int = 12,
) -> np.ndarray:
    """Refine by moving against gradient of the worst shift."""
    h = h0.copy()
    n = h.size
    target = n / 2.0

    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)          # shift with maximal overlap (worst)
        k = abs(idx)
        g = np.zeros_like(h)

        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]          # contribution from (1 - h[i+k])
            g[k:] -= h[:upper]                # contribution from -h[i]

        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # cheap redistribution
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    best_h = None
    best_c = np.inf
    best_n = None

    # parent solution if supplied
    parent = None
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)

    # Main resolution (use the fairly high resolution 300)
    target_n = 300

    # Prepare an initial vector
    if parent is not None:
        # up‑sample / linear interpolate parent to target_n
        x_src = np.linspace(0.0, 2.0, parent.size, endpoint=False)
        x_tgt = np.linspace(0.0, 2.0, target_n, endpoint=False)
        h0 = np.interp(x_tgt, x_src, parent)
    else:
        # random binary feasible start
        h0 = random_binary_feasible(target_n, rng)

    h0 = enforce_sum_and_bounds(h0, target_n, rng)

    # A quick hill‑climb to get into a decent basin
    h, c = hill_climb(h0, rng, max_iters=4000, start_step=0.8)
    if c < best_c:
        best_h, best_c, best_n = h.copy(), c, target_n

    # Gradient refinement (few iterations)
    h = gradient_refine(h, rng, step=0.02, max_iter=12)
    c = compute_c5(h)
    if c < best_c:
        best_h, best_c, best_n = h.copy(), c, target_n

    # Simulated annealing using the remaining budget (up to ~70% of total)
    deadline_sa = start_time + budget_s * 0.85
    if time.time() < deadline_sa:
        h_sa, c_sa = simulated_annealing(h, rng, deadline_sa)
        if c_sa < best_c:
            best_h, best_c, best_n = h_sa.copy(), c_sa, target_n
        h = h_sa

    # Final intensive hill‑climb
    if time.time() - start_time < budget_s * 0.95:
        h, c = hill_climb(h, rng, max_iters=8000, start_step=0.5)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, target_n

    # One more gradient tidy‑up
    h = gradient_refine(best_h, rng, step=0.015, max_iter=8)
    final_c = compute_c5(h)

    # Ensure feasibility one last time
    h = enforce_sum_and_bounds(h, best_n, rng)
    final_c = compute_c5(h)

    # Return result
    return h.tolist(), float(final_c), int(best_n)
