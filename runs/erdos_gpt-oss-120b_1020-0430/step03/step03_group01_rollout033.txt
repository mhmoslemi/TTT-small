def slsqp_optimize(h0, beta, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - CURRENT_TARGET}
    def obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = CURRENT_BETA * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
    res = opt.minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum precisely
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        # simple correction: add diff uniformly then clip and renorm iteratively
        # We'll just call enforce_sum
        x = enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

def enforce_sum(h, target, rng, max_iter=5000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])

            add = max(diff, -h[i])

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h
```

def slsqp_optimize(h0, beta, max_iter=400):
    n = h0.size
    target = n/2.0
    bounds = [(0.0,1.0)]*n
    cons = {"type":"eq","fun": lambda x, target=target: np.sum(x)-target}
    def obj(x, beta=beta):
        corr = np.correlate(x, 1 - x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    res = opt.minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol':1e-12, 'disp':False})
    x = np.clip(res.x,0.0,1.0)
    # enforce exact sum
    x = enforce_sum(x, target, np.random.default_rng())
    return x
```

import time
import numpy as np
import scipy.optimize as opt

# ----------------- helper utilities -----------------
def compute_c5(h):
    """Exact discrete C5 for step function h on [0,2] with dx=2/len(h)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=5000):
    """Clip to [0,1] and adjust to hit the exact sum `target`."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        # uniform correction then re‑clip
        corr = diff / n
        h += corr
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random h with 0≤h_i≤1 and Σh_i = n/2."""
    target = n / 2.0
    h = rng.random(n)
    h = h / h.sum() * target
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, target, rng)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_multi(h_start, rng, max_iters=2000, init_step=0.5,
                     n_trials=8, time_limit=None):
    """Stochastic hill climbing with several trial moves per iteration."""
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay = 0.995
    for it in range(max_iters):
        if time_limit is not None and time.time() > time_limit:
            break
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
                improved = True
        if not improved:
            if rng.random() < 0.03:
                new_h = perturb_h(h, rng, scale=step * 3.0)
                c = compute_c5(new_h)
                if c < best_c:
                    best_c, h = c, new_h
        step = max(step * decay, 1e-4)
    return h, best_c


def slsqp_optimize(h0, beta, max_iter=400):
    """SLSQP minimisation of a smooth surrogate for max_k overlap."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x, t=target: np.sum(x) - t}

    def obj(x, b=beta):
        corr = np.correlate(x, 1.0 - x, mode="full")
        bc = b * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / b

    res = opt.minimize(
        obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum to avoid tiny drift
    x = enforce_sum(x, target, np.random.default_rng())
    return x


def fine_tune(h, rng, iters=500):
    """Very small‑scale hill climbing for final polishing."""
    step = 0.05
    best_c = compute_c5(h)
    for _ in range(iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, h = c, new_h
        step = max(step * 0.99, 1e-5)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.time()

    # ---- initialise -------------------------------------------------
    if "initial_h_values" in globals():

        n = h0.size

    else:
        n = 240
        h0 = random_feasible_vector(n, rng)

    # pool of diverse starters
    candidates = [h0]
    xs = (np.arange(n) + 0.5) * (2.0 / n)

    for f in range(1, 7):
        cand = 0.5 + 0.5 * np.sin(f * np.pi * xs)
        cand = np.clip(cand, 0.0, 1.0)
        candidates.append(enforce_sum(cand, n / 2.0, rng))
        cand = 0.5 + 0.5 * np.cos(f * np.pi * xs)

    # binary masks
