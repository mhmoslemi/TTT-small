def simulated_annealing(h, rng, max_iters, start_temp):
    cur = h.copy()
    cur_c = compute_c5(cur)
    temp = start_temp
    for it in range(max_iters):
        # linear cooling
        temp = start_temp * (1 - it / max_iters)
        if temp <= 0:
            break
        new_h = perturb_h(cur, rng, scale=0.5) # larger scale early
        new_c = compute_c5(new_h)
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / (temp + 1e-12)):
            cur, cur_c = new_h, new_c
    return cur, cur_c
```

def project_to_feasible(v, target):
    """Clip v to [0,1] and shift so sum(v)=target (bisection)."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid

    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)
