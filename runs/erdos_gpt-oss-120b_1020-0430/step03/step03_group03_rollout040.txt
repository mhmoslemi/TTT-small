def enforce_sum_proj(h, target, tol=1e-10, max_iter=100):
    # assume h already clipped to [0,1]
    lo = -np.max(h)  # shift such that all become 0 at most
    hi = 1.0 - np.min(h)  # shift such that all become 1 at most
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    new_h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # adjust rounding errors by distributing diff across interior entries if needed.
    diff = target - new_h.sum()
    if abs(diff) > 1e-12:
        # correct using small adjustments on interior positions
        interior = np.where((new_h > 0) & (new_h < 1))[0]
        if interior.size > 0:
            # spread diff equally
            inc = diff / interior.size
            new_h[interior] += inc
            # reclip
            new_h = np.clip(new_h, 0.0, 1.0)
            # final adjust if still off (unlikely)
            diff2 = target - new_h.sum()
            if abs(diff2) > 1e-12:
                # distribute to first interior
                new_h[interior[0]] += diff2
                new_h = np.clip(new_h, 0.0, 1.0)
    return new_h

def gradient_descent(h, steps, init_lr, decay, rng):
    best = h.copy()
    best_c = compute_c5(best)
    lr = init_lr
    for t in range(steps):
        # compute gradient based on max correlation indices
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = np.max(corr)
        max_idx = np.flatnonzero(np.abs(corr - max_val) < 1e-12)
        n = h.size
        grad = np.zeros_like(h)
        for idx in max_idx:
            k = idx - (n - 1)
            if k >= 0:
                grad[:n - k] += (1 - h[k:])
            else:
                grad[-k:] += (1 - h[:n + k])
        grad /= len(max_idx)
        # Step opposite gradient (since we minimize max, we subtract grad)
        h = h - lr * grad
        # Clip then enforce sum using proj
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum_proj(h, n / 2.0)
        # Optionally add small random noise to avoid stagnation?
        # compute C5
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best = h.copy()
        # decay lr
        lr *= decay
    return best, best_c

import time, math, numpy as np

# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function h on [0,2] with dx=2/len(h)."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)


def project_sum_to_target(h: np.ndarray, target: float, tol: float = 1e-12,
                          max_iter: int = 50) -> np.ndarray:
    """Project h onto the hyperplane sum(h)=target while keeping 0<=h<=1."""
    lo = -np.max(h)
    hi = 1.0 - np.min(h)
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    new_h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - new_h.sum()
    if abs(diff) > tol:
        interior = np.where((new_h > 0) & (new_h < 1))[0]
        if interior.size:
            i = interior[0]
            new_h[i] = np.clip(new_h[i] + diff, 0.0, 1.0)
    return new_h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector in [0,1]^n with exact sum n/2."""
    h = rng.random(n)
    return project_sum_to_target(h, n / 2.0)


# ---------- smooth max gradient ----------
def softmax_gradient_step(h: np.ndarray, beta: float, lr: float) -> np.ndarray:
    """
    One gradient‑descent step on the soft‑max approximation of max corr.
    h stays in [0,1] and sum(h)=n/2 after the step.
    """
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    # stable softmax weights
    exps = np.exp(beta * (corr - max_corr))
    w = exps / np.sum(exps)

    grad = np.ones_like(h)                     # Σ w_s * 1 = 1 for each entry
    for idx, ws in enumerate(w):
        if ws == 0.0:
            continue
        s = idx - (n - 1)
        if s >= 0:
            # subtract h[i+s]
            grad[: n - s] -= ws * h[s:]
            # subtract h[i-s]
            grad[s:] -= ws * h[: n - s]
        else:
            t = -s
            grad[t:] -= ws * h[: n - t]
            grad[: n - t] -= ws * h[t:]
    # gradient step (minimise)
    new_h = h - lr * grad
    new_h = np.clip(new_h, 0.0, 1.0)
    return project_sum_to_target(new_h, n / 2.0)


# ---------- optimisation schedule ----------
def run_schedule(start_h: np.ndarray, schedule, rng: np.random.Generator) -> np.ndarray:
    """Apply a series of (beta, steps, lr) blocks to start_h."""
    h = start_h.copy()
    for beta, steps, lr in schedule:
        for _ in range(steps):
            h = softmax_gradient_step(h, beta, lr)
    return h


# ---------- main entry point ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----- initialise -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_sum_to_target(h0, n / 2.0)
    else:
        n = 180
        h0 = random_feasible_vector(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ----- optimisation schedule (continuation) -----------------------
    schedule = [
        (5,    2000, 0.05),
        (20,   2000, 0.03),
        (80,   2000, 0.02),
        (320,  2000, 0.015),
        (1280, 2000, 0.01),
        (5120, 2000, 0.008),
        (20480,4000, 0.005)
    ]

    # ----- main search loop -------------------------------------------
    while time.time() - start_time < budget_s * 0.85:
        # create a start point (perturb best or fresh random)
        if rng.random() < 0.7:
            cand = best_h + rng.normal(scale=0.02, size=n)
        else:
            cand = random_feasible_vector(n, rng)
        cand = np.clip(cand, 0.0, 1.0)
        cand = project_sum_to_target(cand, n / 2.0)

        # run continuation on this start point
        cand_h = run_schedule(cand, schedule, rng)
        cand_c = compute_c5(cand_h)

        if cand_c < best_c:
            best_h, best_c = cand_h, cand_c

        # occasional quick refinement from current best
        if rng.random() < 0.15:
            tmp = best_h.copy()
            for _ in range(600):
                tmp = softmax_gradient_step(tmp, beta=80, lr=0.01)
            tmp_c = compute_c5(tmp)
            if tmp_c < best_c:
                best_h, best_c = tmp, tmp_c

    # ----- final intensive polishing ------------------------------------
    remaining = max(0.0, budget_s - (time.time() - start_time))
    if remaining > 1.0:
        # estimate one gradient step costs ~0.0003 s; allocate remaining time
        est_step = 3e-4
        steps = int(min(200000, max(5000, remaining / est_step)))
        beta = 32768.0
        lr = 0.001
        h = best_h.copy()
        for _ in range(steps):
            h = softmax_gradient_step(h, beta, lr)
        best_h = h

    # ----- final clean‑up ------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_to_target(best_h, n / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
