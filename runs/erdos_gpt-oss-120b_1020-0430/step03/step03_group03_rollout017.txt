def smooth_max_and_grad(v, beta):
    corr = np.convolve(v, 1 - v, mode='full')
    maxval = np.max(corr) # not needed
    # compute softmax probabilities
    # stable: subtract max beta*corr
    scaled = beta * corr
    max_scaled = np.max(scaled)
    exps = np.exp(scaled - max_scaled)
    Z = np.sum(exps)
    p = exps / Z  # shape 2n-1
    smooth = (max_scaled + np.log(Z)) / beta
    # gradient:

    return smooth, g

import time, math, numpy as np

# ---------- global parameters for smooth objective ----------
CURR_BETA = None
CURR_LAMB = None
CURR_TARGET = None

# ---------- helper functions ----------
def compute_c5(h):
    """Discrete C5 for step function h on [0,2] with uniform spacing."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)


def enforce_sum(h, target, rng):
    """Adjust h in‑place so that sum(h)=target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    enforce_sum(h, n / 2.0, rng)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer ε between two coordinates (preserves sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new


# ---------- smooth max objective ----------
def smooth_f(v):
    """Smooth approximation of max_{k} corr(k) using log‑sum‑exp."""
    beta = CURR_BETA
    lam = CURR_LAMB
    target = CURR_TARGET
    corr = np.convolve(v, 1 - v, mode='full')
    scaled = beta * corr
    m = np.max(scaled)
    exps = np.exp(scaled - m)
    Z = np.sum(exps)
    smooth = (m + np.log(Z)) / beta
    diff = np.sum(v) - target
    return smooth + lam * diff * diff


def smooth_grad(v):
    """Gradient of smooth_f with respect to v."""
    beta = CURR_BETA
    lam = CURR_LAMB
    target = CURR_TARGET
    corr = np.convolve(v, 1 - v, mode='full')
    scaled = beta * corr
    m = np.max(scaled)
    exps = np.exp(scaled - m)
    Z = np.sum(exps)
    p = exps / Z                       # softmax weights, length 2n‑1
    r = 1.0 - 2.0 * v                  # length n
    # g_k = Σ_j p_{k+j} * r_j
    windows = sliding_window_view(p, v.shape[0])   # shape (n,n)
    grad_smooth = windows @ r                     # shape (n,)
    diff = np.sum(v) - target
    return grad_smooth + 2.0 * lam * diff * np.ones_like(v)


# ---------- main run ----------
def run(seed=42, budget_s=1100, **kwargs):
    global CURR_BETA, CURR_LAMB, CURR_TARGET
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----- decide dimension and initial vector -----
    if "initial_h_values" in globals():
        h_small = np.array(initial_h_values, dtype=float)
        n_small = h_small.size
        n_target = max(240, n_small)               # richer resolution
        # linear interpolation to size n_target
        xs = (np.arange(n_small) + 0.5) * (2.0 / n_small)
        xt = (np.arange(n_target) + 0.5) * (2.0 / n_target)
        h0 = np.interp(xt, xs, h_small, left=0.0, right=0.0)
        h0 = np.clip(h0, 0.0, 1.0)
        enforce_sum(h0, n_target / 2.0, rng)
    else:
        n_target = 240
        h0 = random_feasible_vector(n_target, rng)

    n = n_target
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ----- optimisation schedule -----
    beta_schedule = [5, 20, 80, 300, 1200, 5000]
    main_deadline = budget_s * 0.85

    while time.time() - start_time < main_deadline:
        # generate a start point
        if rng.random() < 0.3:
            start = best_h + rng.normal(scale=0.02, size=n)
            start = np.clip(start, 0.0, 1.0)
        else:
            start = random_feasible_vector(n, rng)
        enforce_sum(start, n / 2.0, rng)

        cur = start.copy()
        for beta in beta_schedule:
            CURR_BETA = float(beta)
            CURR_LAMB = beta * 10.0               # penalty weight
            CURR_TARGET = n / 2.0
            bounds = [(0.0, 1.0)] * n

            # adapt iteration budget to remaining time
            remain = max(0.0, main_deadline - (time.time() - start_time))
            max_iter = 250 if remain > 2.0 else 50

            try:
                res = minimize(
                    smooth_f,
                    cur,
                    method='L-BFGS-B',
                    jac=smooth_grad,
                    bounds=bounds,
                    options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False},
                )
                cur = np.clip(res.x, 0.0, 1.0)
                enforce_sum(cur, n / 2.0, rng)
            except Exception:
                break

        cur_c = compute_c5(cur)
        if cur_c < best_c:
            best_h, best_c = cur, cur_c

    # ----- final polishing with higher betas -----
    final_betas = [10000, 20000, 50000]
    for beta in final_betas:
        if time.time() - start_time > budget_s:
            break
        CURR_BETA = float(beta)
        CURR_LAMB = beta * 10.0
        CURR_TARGET = n / 2.0
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(
                smooth_f,
                best_h,
                method='L-BFGS-B',
                jac=smooth_grad,
                bounds=bounds,
                options={'maxiter': 500, 'ftol': 1e-12, 'disp': False},
            )
            best_h = np.clip(res.x, 0.0, 1.0)
            enforce_sum(best_h, n / 2.0, rng)
            best_c = compute_c5(best_h)
        except Exception:
            pass

    # ----- simple hill‑climbing refinement -----
    for _ in range(2000):
        cand = perturb_h(best_h, rng, scale=0.02)
        cand_c = compute_c5(cand)
        if cand_c < best_c:
            best_h, best_c = cand, cand_c

    # ----- final sanity check -----
    best_h = np.clip(best_h, 0.0, 1.0)
    enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
