def gradient_descent(h, target_sum, max_iters, rng, time_budget):
    best_h = h.copy()
    best_c = compute_c5_fft(h)
    n = h.shape[0]
    # parameters
    alpha = 0.5
    for it in range(max_iters):
        # time check
        if time_budget is not None and (time.time() - start_time) > time_budget:
            break
        # compute corr and shift
        corr = corr_full_fft(best_h)
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        # gradient
        g = np.ones_like(best_h)
        if shift >= 0:
            k = shift
            if k < n:
                g[:n-k] -= best_h[k:]
                g[k:] -= best_h[:n-k]
        else:
            k = -shift
            if k < n:
                g[k:] -= best_h[:n-k]
                g[:n-k] -= best_h[k:]
        # line search: try step sizes decreasing
        improved = False
        # we can try multiple alphas: alpha * (0.5^j)
        step = alpha
        for _ in range(5):
            new_h = best_h - step * g
            # projection
            new_h = project_to_feasible(new_h, target_sum)
            c_new = compute_c5_fft(new_h)
            if c_new < best_c:
                best_h = new_h
                best_c = c_new
                improved = True
                break
            step *= 0.5
        if not improved:
            # optionally random perturb
            new_h = perturb_h(best_h, rng, scale=0.2)
            new_h = project_to_feasible(new_h, target_sum)
            c_new = compute_c5_fft(new_h)
            if c_new < best_c:
                best_h = new_h
                best_c = c_new
        # optionally adapt alpha based on success
    return best_h, best_c

def gradient_descent(h, target, rng, max_iters, time_limit, start_time):
    best_h = h.copy()
    best_c = compute_c5_fft(best_h)
    n = len(h)
    step_alpha = 0.5
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        # find shift
        corr = corr_full_fft(best_h)
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        # gradient
        g = np.ones(n)
        if shift >= 0:
            k = shift
            if k < n:
                g[:n-k] -= best_h[k:]
                g[k:] -= best_h[:n-k]
        else:
            k = -shift
            if k < n:
                g[k:] -= best_h[:n-k]
                g[:n-k] -= best_h[k:]
        # try step sizes
        improved = False
        step = step_alpha
        for _ in range(5):
            new_h = best_h - step * g
            new_h = project_to_feasible(new_h, target)
            c_new = compute_c5_fft(new_h)
            if c_new < best_c:

def corr_full_fft(h):
    n = h.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    a = np.fft.rfft(h, L)
    b = np.fft.rfft((1 - h)[::-1], L)
    conv = np.fft.irfft(a * b, L)

def project_to_feasible(v, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

import time
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the feasible set {0 ≤ h ≤ 1, sum(h) = target}
def project_to_feasible(v, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


# ----------------------------------------------------------------------
# Random feasible vector (sum = n/2, entries in [0,1])
def random_feasible_vector(n, rng):
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_to_feasible(v, n / 2.0)


# ----------------------------------------------------------------------
# Small transfer preserving sum
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


# ----------------------------------------------------------------------
# Full linear correlation via FFT (fast for larger n)
def corr_full_fft(h):
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())  # next power of two ≥ 2n‑1
    a = np.fft.rfft(h, L)
    b = np.fft.rfft((1.0 - h)[::-1], L)
    conv = np.fft.irfft(a * b, L)
    return conv[: 2 * n - 1]


def compute_c5_fft(h):
    n = h.shape[0]
    corr = corr_full_fft(h)
    return np.max(corr) * (2.0 / n)


# ----------------------------------------------------------------------
# Gradient of the correlation term that currently attains the maximum
def gradient_of_max_corr(h, shift):
    n = h.size
    g = np.ones(n, dtype=float)
    if shift >= 0:
        k = shift
        if k < n:
            g[: n - k] -= h[k:]          # term -h_{i+k}
            g[k:] -= h[: n - k]          # term -h_{i‑k}
    else:
        k = -shift
        if k < n:
            g[k:] -= h[: n - k]          # term -h_{i‑k}
            g[: n - k] -= h[k:]          # term -h_{i+k}
    return g


# ----------------------------------------------------------------------
# Projected gradient descent with simple random escapes
def projected_gradient_descent(h0, target, rng, max_iters, deadline):
    h = h0.copy()
    best_c = compute_c5_fft(h)
    step_alpha = 0.5
    n = h.size
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        corr = corr_full_fft(h)
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        g = gradient_of_max_corr(h, shift)

        # try a few step sizes
        improved = False
        step = step_alpha
        for _ in range(5):
            new_h = h - step * g
            new_h = project_to_feasible(new_h, target)
            c_new = compute_c5_fft(new_h)
            if c_new < best_c:
                h, best_c = new_h, c_new
                improved = True
                step_alpha = min(1.0, step_alpha * 1.05)
                break
            step *= 0.5

        if not improved:
            # random perturbation as escape move
            new_h = perturb_h(h, rng, scale=0.1)
            new_h = project_to_feasible(new_h, target)
            c_new = compute_c5_fft(new_h)
            if c_new < best_c:
                h, best_c = new_h, c_new
                step_alpha = max(0.01, step_alpha * 0.9)
    return h, best_c

# Simple hill‑climb (deterministic improvement)
def hill_climb_simple(h, rng, max_iters=2000):
    best_h = h.copy()
    best_c = compute_c5_fft(best_h)
    n = h.size
    step = 0.3
    for _ in range(max_iters):
        new_h = perturb_h(best_h, rng, scale=step)
        new_h = project_to_feasible(new_h, n / 2.0)
        c_new = compute_c5_fft(new_h)
        if c_new < best_c:
            best_h, best_c = new_h, c_new
            step = min(1.0, step * 1.05)
        else:
            step = max(0.01, step * 0.95)
    return best_h, best_c
