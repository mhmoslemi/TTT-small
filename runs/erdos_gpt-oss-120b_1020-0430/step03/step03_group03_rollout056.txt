import time, math, numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / h.size)

def compute_weights(a, beta):
    exp_arg = -beta * a
    max_arg = np.max(exp_arg)
    exp_vals = np.exp(exp_arg - max_arg)
    return exp_vals / np.sum(exp_vals)

def compute_gradient(f, w):
    n = f.size
    g = np.zeros_like(f)
    # loop over lags
    base = n - 1
    for idx, weight in enumerate(w):
        if weight == 0.0:
            continue
        ell = idx - base
        if ell >= 0:
            i_end = n - ell
            if i_end > 0:
                g[:i_end] += weight * f[ell:]
                g[ell:] += weight * f[:i_end]
        else:
            m = -ell
            i_len = n - m
            if i_len > 0:
                g[m:] += weight * f[:i_len]
                g[:i_len] += weight * f[m:]
    return g

def project_onto_box_sum(v, lo, hi, target):
    # binary search for lambda
    low = -1.0
    high = 1.0
    # expand interval
    while np.clip(v + low, lo, hi).sum() > target:
        low *= 2
    while np.clip(v + high, lo, hi).sum() < target:
        high *= 2
    for _ in range(60):
        mid = (low + high) / 2.0
        s = np.clip(v + mid, lo, hi).sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) / 2.0, lo, hi)

def linked_perturb(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:

    new[i] -= eps

import time, math, numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h):
    """Discrete C5 for step function h (dx = 2/n)."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)

def compute_weights(a, beta):
    """Soft‑min weights w ∝ exp(-β·a)."""
    exp_arg = -beta * a
    max_arg = np.max(exp_arg)
    exp_vals = np.exp(exp_arg - max_arg)
    return exp_vals / np.sum(exp_vals)

def compute_gradient(f, w):
    """∇_f  Σ_k w_k Σ_i f_i f_{i+k} = Σ_k w_k (f_{i+k}+f_{i-k})."""
    n = f.shape[0]
    g = np.zeros_like(f)
    base = n - 1                     # index of lag 0 in w
    for idx, wt in enumerate(w):
        if wt == 0.0:
            continue
        ell = idx - base
        if ell >= 0:
            end = n - ell
            if end > 0:
                g[:end] += wt * f[ell:]
                g[ell:] += wt * f[:end]
        else:
            m = -ell
            length = n - m
            if length > 0:
                g[m:] += wt * f[:length]
                g[:length] += wt * f[m:]
    return g

def project_onto_box_sum(v, lo, hi, target):
    """Project v onto [lo,hi]^n with Σ v_i = target."""
    low, high = -1.0, 1.0
    # expand interval until the sum straddles target
    while np.clip(v + low, lo, hi).sum() > target:
        low *= 2
    while np.clip(v + high, lo, hi).sum() < target:
        high *= 2

def linked_perturb(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def simulated_annealing(h0, rng, iters, init_temp, final_temp):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    n = h.size
    log_ratio = math.log(final_temp / init_temp) if init_temp > 0 else 0.0
    for it in range(1, iters + 1):
        T = init_temp * math.exp(log_ratio * it / iters)
        cand = linked_perturb(h, rng, scale=0.2)
        cand_c = compute_c5(cand)
        diff = cand_c - cur_c
        if diff < 0 or rng.random() < math.exp(-diff / max(T, 1e-12)):
            h, cur_c = cand, cand_c
            if cand_c < best_c:
                best_c, best_h = cand_c, cand.copy()
    return best_h, best_c

# ---------- main optimisation ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_end = start + budget_s

    # ----- initial step function -----
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.size
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum_h(h)
    else:
        n = 180
        h = random_feasible_h(n, rng)

    f = h - 0.5
    best_f = f.copy()
    best_c = compute_c5(h)

    # ----- gradient‑ascent phase (≈55 % budget) -----
    grad_end = start + budget_s * 0.55
    betas = [5, 15, 45, 135, 405, 1215, 3645]
    eta0 = 0.5
    step = 0
    while time.time() < grad_end:
        for beta in betas:
            for _ in range(3000):
                if time.time() >= grad_end:
                    break
                a = np.correlate(f, f, mode="full")
                w = compute_weights(a, beta)
                g = compute_gradient(f, w)
                eta = eta0 / np.sqrt(step + 1)
                f = f + eta * g
                np.clip(f, -0.5, 0.5, out=f)
                f -= f.mean()                     # keep zero‑mean approximately
                step += 1
                if step % 200 == 0:
                    h_tmp = f + 0.5
                    c = compute_c5(h_tmp)
                    if c < best_c:
                        best_c = c
                        best_f = f.copy()
            if time.time() >= grad_end:
                break

    # final projection of the best iterate
    best_f = project_onto_box_sum(best_f, -0.5, 0.5, 0.0)
    best_h = best_f + 0.5
    best_h = enforce_sum_h(best_h)
    best_c = compute_c5(best_h)

    # ----- SLSQP polishing (≈40 % remaining) -----
    polish_end = total_end - 30.0  # keep a short reserve for final SA
    while time.time() < polish_end:
        for beta in (2000, 8000, 20000):
            if time.time() >= polish_end:
                break
            try:
                refined = slsqp_optimize(best_h, beta, max_iter=250)
                rc = compute_c5(refined)
                if rc < best_c:
                    best_h, best_c = refined, rc
            except Exception:
                continue

    # ----- tiny final SA if time permits -----
    if time.time() < total_end:
        remaining = total_end - time.time()
        iters = int(min(50000, remaining * 2000))
        if iters > 0:
            cand_h, cand_c = simulated_annealing(best_h, rng, iters,
                                                init_temp=0.02,
                                                final_temp=1e-6)
            if cand_c < best_c:
                best_h, best_c = cand_h, cand_c

    # ----- clean up and return -----
    best_h = enforce_sum_h(np.clip(best_h, 0.0, 1.0))
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
