def softmax_obj_grad(h, beta):
    n = h.shape[0]
    O = np.correlate(h, 1 - h, mode='full')
    bc = beta * O
    max_bc = np.max(bc)
    exp_bc = np.exp(bc - max_bc)
    sum_exp = np.sum(exp_bc)
    w = exp_bc / sum_exp
    # Compute gradient
    g = np.ones_like(h)  # because sum w = 1
    # Use loops over shift values
    # Use local variables for speed
    for shift in range(-(n - 1), n):
        wk = w[shift + n - 1]
        if wk == 0:
            continue
        if shift == 0:
            g -= 2.0 * wk * h
        elif shift > 0:
            # subtract forward part
            # g[i] for i=0..n-shift-1, subtract wk * h[i+shift]
            g[:n - shift] -= wk * h[shift:]
            # subtract backward part
            g[shift:] -= wk * h[:n - shift]
        else:  # shift < 0
            k = -shift
            # subtract part for i>=k: g[i] -= wk * h[i - k] => g[k:] -= wk * h[:n - k]
            g[k:] -= wk * h[:n - k]
            # subtract part for i <= n - k -1: g[i] -= wk * h[i + k] => g[:n - k] -= wk * h[k:]
            g[:n - k] -= wk * h[k:]

    # Multiply by 1 (beta not needed)
    # Actually derivative of f is (1/sum_exp) * sum_k exp(beta O_k) * dO_k/dh = sum_k w_k * dO_k/dh, no additional factor. So gradient is as computed.

    # Return f and grad
    f = (max_bc + np.log(sum_exp)) / beta
    return f, g

def gradient_refine_extended(h, rng, step=0.02, max_iter=50):
    n = h.shape[0]
    target = n/2.0
    for it in range(max_iter):
        # compute correlation O and worst shift idx
        O = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(O) - (n - 1)  # shift k
        k = abs(idx)
        # compute gradient as described earlier for worst shift only
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            # For shift idx (could be negative or positive)
            # For idx >=0:
            if idx >= 0:
                upper = n - idx
                g[:upper] += 1.0 - h[idx:]   # +1 - h[i+idx]
                g[idx:] -= h[:upper]        # -h[i-idx]
            else:
                # idx < 0, shift negative
                kpos = -idx
                upper = n - kpos
                # gradient formula when shift negative:
                # O_{-k} = sum_i h_i (1 - h_{i - k})
                # derivative: for i >= k: 1 - h_{i - k} ; for i < n - k: -h_{i + k}
                g[kpos:] += 1.0 - h[:upper]
                g[:upper] -= h[kpos:]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        # clip and fix sum
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # perform redistribution similar to previous
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C₅ for step function h on [0,2] with uniform spacing."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h (in‑place) so that its sum equals target while staying in [0,1]."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    h = h.copy()
    for _ in range(2000):
        i = rng.integers(0, h.size)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector in [0,1] with exact sum n/2."""
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, target, rng)


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def hill_climb(
    h_start: np.ndarray,
    rng: np.random.Generator,
    max_iters: int = 1500,
    start_step: float = 0.6,
) -> tuple[np.ndarray, float]:
    """Simple deterministic hill‑climb (accept only strict improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_every = max(1, max_iters // 8)
    for it in range(max_iters):
        cand = perturb_h(h, rng, scale=step)
        c = compute_c5(cand)
        if c < best_c:
            best_c, best_h, h = c, cand.copy(), cand
        if (it + 1) % decay_every == 0:
            step = max(step * 0.85, 5e-4)
    return best_h, best_c


def gradient_refine_extended(
    h: np.ndarray,
    rng: np.random.Generator,
    step: float = 0.02,
    max_iter: int = 50,
) -> np.ndarray:
    """Sub‑gradient descent on the worst shift (more iterations than the original)."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)  # shift achieving the current max
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        elif idx > 0:
            upper = n - idx
            g[:upper] += 1.0 - h[idx:]          # + (1 - h[i+idx])
            g[idx:] -= h[:upper]                # - h[i]
        else:  # idx < 0
            kk = -idx
            upper = n - kk
            g[kk:] += 1.0 - h[:upper]           # + (1 - h[i-kk])
            g[:upper] -= h[kk:]                 # - h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # cheap redistribution
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


# -------------------------------------------------
# Soft‑max surrogate with analytic gradient
# -------------------------------------------------
def softmax_obj_grad(x: np.ndarray, beta: float) -> tuple[float, np.ndarray]:
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) with its gradient."""
    n = x.size
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = beta * corr
    m = np.max(bc)
    exp_bc = np.exp(bc - m)
    sum_exp = np.sum(exp_bc)
    w = exp_bc / sum_exp                     # weights, sum to 1
    # gradient g = Σ_k w_k * ∂O_k/∂x
    g = np.ones_like(x)                      # Σ_k w_k * 1
    for shift in range(-(n - 1), n):
        wk = w[shift + n - 1]
        if wk == 0.0:
            continue
        if shift == 0:
            g -= 2.0 * wk * x
        elif shift > 0:
            g[: n - shift] -= wk * x[shift:]
            g[shift:] -= wk * x[: n - shift]
        else:  # shift < 0
            k = -shift
            g[k:] -= wk * x[: n - k]
            g[: n - k] -= wk * x[k:]
    f = (m + np.log(sum_exp)) / beta
    return float(f), g


def slsqp_softmax_optimize(
    h0: np.ndarray,
    beta: float,
    rng: np.random.Generator,
    max_iter: int = 250,
) -> np.ndarray:
    """SLSQP optimisation of the soft‑max surrogate using the analytic gradient."""
    n = h0.size
    target = n / 2.0

    def fun(x):
        return softmax_obj_grad(x, beta)[0]

    def jac(x):
        return softmax_obj_grad(x, beta)[1]

    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x, t=target: np.sum(x) - t}
    res = minimize(
        fun,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        jac=jac,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return enforce_sum(x, target, rng)


# -------------------------------------------------
# Main driver
# -------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # -------------------------------------------------
    # Helper to check remaining time
    # -------------------------------------------------
    def time_left():
        return budget_s - (time.time() - start_time)

    # -------------------------------------------------
    # Candidate discretisations
    # -------------------------------------------------
    candidate_ns = [180]  # start with the size of the provided parent (if any)
    if time_left() > 200:
        candidate_ns.append(240)
    if time_left() > 350:
        candidate_ns.append(360)

    for n in candidate_ns:
        if time_left() < 20:
            break
        target = n / 2.0

        # ---- build diverse starting vectors ---------------------------------
        starts = []

        # parent (if same size)
        if "initial_h_values" in globals() and len(initial_h_values) == n:
            starts.append(np.array(initial_h_values, dtype=float))

        # interpolated parent (if sizes differ)
        if "initial_h_values" in globals():
            parent = np.array(initial_h_values, dtype=float)
            if parent.size != n:
                xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
                xt = np.linspace(0.0, 2.0, n, endpoint=False)
                starts.append(np.interp(xt, xs, parent))

        # random continuous feasible
        starts.append(random_feasible_vector(n, rng))

        # random binary feasible
        hb = np.zeros(n)
        hb[rng.choice(n, n // 2, replace=False)] = 1.0
        starts.append(hb)

        # low‑frequency sinusoid (smooth shape)
        t = np.arange(n)
        freq = 2
        hs = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * t / n)
        hs = enforce_sum(hs, target, rng)
        starts.append(hs)

        # perturbed current global best (if same n)
        if best_n == n and best_h is not None:
            starts.append(best_h + 0.02 * rng.standard_normal(n))

        # ---- quick hill‑climb on each start, keep the best -----------------
        local_best_h = None
        local_best_c = np.inf
        for base in starts:
            if time_left() < 10:
                break
            h = enforce_sum(np.clip(base, 0.0, 1.0), target, rng)
            h, c = hill_climb(h, rng, max_iters=1200, start_step=0.6)
            if c < local_best_c:
                local_best_c, local_best_h = c, h

        if local_best_h is None:
            continue

        # ---- gradual soft‑max SLSQP schedule --------------------------------
        beta_schedule = [0.5, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0]
        h = local_best_h
        for beta in beta_schedule:
            if time_left() < 3:
                break
            max_it = 180 if beta < 50 else 250
            h = slsqp_softmax_optimize(h, beta, rng, max_iter=max_it)

        # ---- extended gradient refinement -----------------------------------
        h = gradient_refine_extended(h, rng, step=0.02, max_iter=60)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

        # ---- final local hill‑climb ----------------------------------------
        if time_left() > 4:
            h, c = hill_climb(h, rng, max_iters=1800, start_step=0.4)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

    # -------------------------------------------------
    # Optional up‑sampling refinement (double resolution)
    # -------------------------------------------------
    if best_h is not None and time_left() > 30 and best_n * 2 <= 500:
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        target_up = n_up / 2.0
        h_up = enforce_sum(h_up, target_up, rng)

        for beta in [5.0, 20.0, 50.0, 200.0, 500.0]:
            if time_left() < 5:
                break
            h_up = slsqp_softmax_optimize(h_up, beta, rng, max_iter=200)

        h_up = gradient_refine_extended(h_up, rng, step=0.02, max_iter=40)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, n_up

    # -------------------------------------------------
    # Final safety checks
    # -------------------------------------------------
    if best_h is None:
        best_n = 180
        best_h = np.full(best_n, 0.5)
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    best_h = enforce_sum(best_h, target, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
