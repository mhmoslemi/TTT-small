def simulated_annealing(h0, rng, max_steps, temp_start, temp_end):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    curr_c = best_c
    n = h.size
    for step in range(max_steps):
        # temperature schedule
        t = temp_start * ((temp_end/temp_start) ** (step / max_steps))
        # propose move
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        # Determine max epsilon magnitude
        max_eps = min(h[i], 1 - h[j])
        max_eps_neg = min(h[j], 1 - h[i])
        # Choose direction random
        if rng.random() < 0.5:
            # move from i to j
            max_move = max_eps
            if max_move <= 0:
                continue
            eps = rng.random() * max_move
        else:
            # move from j to i
            max_move = max_eps_neg
            if max_move <= 0:
                continue
            eps = - rng.random() * max_move
        # Scale eps by t relative to initial? Actually we already limited by values; maybe also limit by factor of t
        # Actually we can multiply by (t / temp_start) to make larger moves early
        eps = eps * (t / temp_start) * 2.0  # scaling factor
        # Ensure after scaling remain in bounds:
        if eps > 0:
            eps = min(eps, max_eps)
        else:
            eps = - min(-eps, max_eps_neg)
        # Apply move
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global variables for the smooth surrogate used by SLSQP
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    """Smooth log‑sum‑exp surrogate of max_{k} Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C₅ for a step function h (dx = 2/len(h))."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def enforce_sum(h, target, rng):
    """Adjust h so that Σh = target while 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-12:
            continue
        h[i] += add
        diff = target - h.sum()
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(1000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            if abs(add) < 1e-12:
                continue
            h[i] += add
            diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def slsqp_optimize(h0, beta, max_iter=800):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            if abs(add) < 1e-12:
                continue
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def hill_climb_improve(h_start, rng, max_iters=5000, start_step=0.5):
    """Greedy hill‑climb: keep only improving moves."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        max_pos = min(h[i], 1.0 - h[j])
        max_neg = min(h[j], 1.0 - h[i])
        if max_pos + max_neg == 0:
            continue
        eps = (rng.random() * (max_pos + max_neg) - max_neg) * step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_h = np.clip(new_h, 0.0, 1.0)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def perturb_h(h, rng, scale=0.5):
    """Transfer a small amount between two coordinates (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)

    return np.clip(new_h, 0.0, 1.0)


def simulated_annealing(h0, rng, max_time):
    """Simple SA that respects the sum constraint."""
    start = time.time()
    end = start + max_time
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    curr_c = best_c
    n = h.size
    T_start = 0.02
    T_end = 0.0002
    while True:
        now = time.time()
        if now >= end:
            break
        frac = (now - start) / max_time
        t = T_start * (1.0 - frac) + T_end * frac
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        max_neg = min(h[j], 1.0 - h[i])

        if rng.random() < 0.5:
            eps = rng.random() * max_pos
