def reduce_worst_shift(h, rng, max_iters, top_k=10, alpha=0.5, time_deadline=None):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    it = 0
    while it < max_iters:
        if time_deadline is not None and time.time() >= time_deadline: break
        s, max_val, conv = worst_shift_and_contributions(best_h)
        # compute overlapping region
        if s >= 0:
            i_arr = np.arange(n - s)
        else:
            i_arr = np.arange(-s, n)
        # compute product contributions
        prod = best_h[i_arr] * (1 - best_h[i_arr + s])
        # choose top_k contributions
        if prod.size == 0:
            break
        k = min(top_k, prod.size)
        top_idx = np.argpartition(-prod, k-1)[:k]
        improved = False
        for idx in top_idx:
            i = i_arr[idx]
            j = i + s
            max_eps = min(best_h[i], 1 - best_h[j])
            if max_eps < 1e-12:
                continue
            eps = max_eps * alpha
            new_h = best_h.copy()
            new_h[i] -= eps
            new_h[j] += eps
            # ensure bounds
            # no need to enforce sum because epsilon shift preserves sum
            c_new = compute_c5(new_h)
            if c_new < best_c - 1e-9:
                best_h = new_h
                best_c = c_new
                improved = True
                break  # accept first improvement
        if not improved:
            # try random perturb
            new_h = perturb_h(best_h, rng, scale=0.2)
            c_new = compute_c5(new_h)
            if c_new < best_c:
                best_h = new_h
                best_c = c_new
            else:
                # maybe break if cannot improve after some attempts
                pass
        it += 1
    return best_h, best_c

def run(seed=42,budget_s=1100,**kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s*0.97

    # Determine n
    if 'initial_h_values' in globals():

        n = h0.size

            except Exception:
                continue

        # greedy refine

        if cur_c < best_c:

        # occasional extra greedy passes
        if rng.random() < 0.3:
            cur_h2, cur_c2 = reduce_worst_shift(best_h, rng, max_iters=1500, top_k=5, alpha=0.5, time_deadline=deadline)

def enforce_sum(h, target, rng):
    """Adjust h so that Σh == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Random vector in [0,1]^n with sum = n/2."""
    h = np.full(n, 0.5)

    """Transfer a small amount ε from i to j (preserves the sum)."""

def correlate_full(h):

    rev = (1 - h)[::-1]

    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]

def compute_c5(h):
