def correlation_fft(a, b):
    # return full correlation of a with b (i.e., np.correlate(a, b, mode='full'))
    n = a.size
    m = b.size
    L = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, L)
    # Note: correlation a with b = convolution a with reversed b
    br = b[::-1]
    fb = np.fft.rfft(br, L)
    conv = np.fft.irfft(fa * fb, L)[: n + m - 1]
    return conv

def pgd_step(h, beta, step):
    # compute gradient
    f = correlation_fft(h, 1 - h)
    maxf = f.max()
    exp_beta = np.exp(beta * (f - maxf))

import time
import numpy as np

# ------------------------------------------------------------
# Projection onto the box [0,1] and sum = target (capped simplex)
# ------------------------------------------------------------
def _project_sum_box(v: np.ndarray, target: float) -> np.ndarray:
    """Projection of v onto {x | 0<=x<=1, sum(x)=target}."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    # binary search for λ satisfying sum clip(v-λ,0,1)=target
    for _ in range(50):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)


# ------------------------------------------------------------
# Random feasible vector (box + sum)
# ------------------------------------------------------------
def _random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    v = rng.random(n)  # uniform in [0,1)
    return _project_sum_box(v, n / 2.0)


# ------------------------------------------------------------
# FFT based full correlation: np.correlate(a,b,mode='full')
# ------------------------------------------------------------
def _corr_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    n, m = a.size, b.size
    L = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b[::-1], L)
    conv = np.fft.irfft(fa * fb, L)
    return conv[: n + m - 1]


# ------------------------------------------------------------
# C5 value for a step function h (dx = 2/n)
# ------------------------------------------------------------
def _c5_of(h: np.ndarray) -> float:
    conv = _corr_fft(h, 1.0 - h)
    return conv.max() * (2.0 / h.size)


# ------------------------------------------------------------
# One projected gradient descent step on the smoothed max
# ------------------------------------------------------------
def _pgd_step(h: np.ndarray, beta: float, step: float,
              target: float, n: int) -> tuple[np.ndarray, float]:
    # soft‑max weights over all lags
    f = _corr_fft(h, 1.0 - h)                # length 2n‑1
    max_f = f.max()
    w = np.exp(beta * (f - max_f))
    w /= w.sum()

    Lw = w.size                               # = 2n‑1
    # g₁ = ∑ₖ wₖ h_{j+k}
    g1_full = _corr_fft(w, h)
    start = Lw - 1
    g1 = g1_full[start:start + n]

    # g₂ = ∑ₖ wₖ h_{j‑k} = correlation of reversed w with h
    w_rev = w[::-1]
    g2_full = _corr_fft(w_rev, h)
    g2 = g2_full[start:start + n]

    grad = 1.0 - (g1 + g2)                     # ∇S(h)
    v = h - step * grad
    h_new = _project_sum_box(v, target)
    return h_new, _c5_of(h_new)


# ------------------------------------------------------------
# Small local hill‑climb (purely combinatorial moves)
# ------------------------------------------------------------
def _hill_climb(h_start: np.ndarray, rng: np.random.Generator,
                time_deadline: float, max_iters: int = 8000,
                step_scale: float = 0.08) -> tuple[np.ndarray, float]:
    h = h_start.copy()
    best_h = h.copy()
    best_c = _c5_of(h)
    n = h.size
    for _ in range(max_iters):
        if time.time() >= time_deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * step_scale
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c = _c5_of(new_h)
        if c < best_c:
            best_c, best_h, h = c, new_h.copy(), new_h
    return best_h, best_c
