def slsqp_optimize(h0, beta, max_iter=500):
    n = h0.size
    target = n / 2.0
    def sum_eq(x):
        return np.sum(x) - target
    def softmax_obj(x):
        # x within [0,1] but may be slight outside due to numerical; we handle.
        x_clip = np.clip(x, 0.0, 1.0)
        corr = np.correlate(x_clip, 1 - x_clip, mode='full')
        bc = beta * corr
        m = np.max(bc)
        # compute log-sum-exp
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq}
    # Use 'SLSQP' method
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    # Clip and enforce sum
    x = np.clip(res.x, 0.0, 1.0)
    # Enforce exact sum using simple adjustment: compute diff = target - sum(x); adjust first element within bounds
    diff = target - np.sum(x)
    # Find index where we can adjust within [0,1].
    # To avoid complicated iteration, we can just add diff to the first component and clip.
    # But that could violate bounds. We'll implement a simple fix: add diff to the element with max distance from bounds.
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        # compute distances to bounds
        dist_low = x - 0.0
        dist_high = 1.0 - x
        # choose index where adding diff won't breach bounds: if diff>0 choose max dist_high, else max dist_low
        if diff > 0:
            idx = np.argmax(dist_high)
            add = min(diff, dist_high[idx])
            x[idx] += add
            diff -= add
        else:
            idx = np.argmax(dist_low)
            sub = min(-diff, dist_low[idx])
            x[idx] -= sub
            diff += sub

            x[idx] = min(1.0, x[idx] + diff)

def enforce_sum(h, target, rng):
    """Clip to [0,1] and adjust to make sum exactly target (float)."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 10000
    it = 0

    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:  # need to increase
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i])
            add *= rng.random()
            h[i] += add
            diff -= add
        else:        # need to decrease
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:

            sub = min(-diff, h[i])

def perturb_random(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

            if k >= 0:
                grad[:n - k] -= h[k:]
                grad[k:]    -= h[:n - k]
            else:
                kk = -k
                grad[:n - kk] -= h[kk:]
                grad[kk:]     -= h[:n - kk]
            grad_total += grad

        pos = np.where((grad_total > 1e-9) & (h > 1e-12))[0]
        neg = np.where((grad_total < -1e-9) & (h < 1.0 - 1e-12))[0]

        if pos.size == 0 or neg.size == 0:
            new_h = perturb_random(h, rng, scale=step)
        else:
            i = rng.choice(pos)
            j = rng.choice(neg)
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0.0:
                new_h = h.copy()
                c_new = c_current
                continue
            eps = max_eps * step * rng.random()
            if eps < 1e-12:
                continue
            new_h = h.copy()
            new_h[i] -= eps
            new_h[j] += eps

        c_new = compute_c5(new_h)

        if c_new < best_c:
            best_c = c_new
            best_h = new_h.copy()

        if c_new < c_current:
            h = new_h

        if (it + 1) % decay_interval == 0:
            step = max(step * decay_factor, 1e-5)

    return best_h, best_c

# SLSQP refinement on a smooth approximation of the max
# --------------------------------------------------------------
def slsqp_optimize(h0, beta, rng, max_iter=400):
    n = h0.size
    target = n / 2.0

    def sum_eq(x):
        return np.sum(x) - target

    def softmax_obj(x):
        x_clip = np.clip(x, 0.0, 1.0)
        corr = np.correlate(x_clip, 1.0 - x_clip, mode="full")
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # enforce exact sum (simple projection)
    diff = target - np.sum(x)
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            dist = 1.0 - x
            idx = np.argmax(dist)
            add = min(diff, dist[idx])
            x[idx] += add
            diff -= add
        else:
            dist = x
            idx = np.argmax(dist)
            sub = min(-diff, dist[idx])
            x[idx] -= sub
            diff += sub
    x = np.clip(x, 0.0, 1.0)
    if abs(diff) > 1e-9:
        i = rng.integers(0, n)
        if diff > 0:
            x[i] = min(1.0, x[i] + diff)
        else:
            x[i] = max(0.0, x[i] + diff)
    return x

# Main entry point required by the harness
# --------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Initialise low‑resolution vector (use parent if supplied)
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h_low = np.array(initial_h_values, dtype=float)
    else:
        n_low = 360
        h_low = np.zeros(n_low)
        ones = rng.choice(n_low, size=n_low // 2, replace=False)
        h_low[ones] = 1.0

    n_low = h_low.size
    # Upsample to double resolution (720 points)
    n = 2 * n_low
    xs_low = (np.arange(n_low) + 0.5) * (2.0 / n_low)
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.interp(xs, xs_low, h_low)
    target_sum = n / 2.0
    h = enforce_sum(np.clip(h, 0.0, 1.0), target_sum, rng)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Main optimisation phase (≈85 % of time budget)
    # ------------------------------------------------------------------
    main_limit = budget_s * 0.85
    beta_schedule = [5.0, 20.0, 80.0, 200.0, 800.0, 2000.0]

    while time.time() - start_time < main_limit:
        # pick a start point
        if rng.random() < 0.6:
            start = perturb_random(best_h, rng, scale=0.3)
        else:
            start = np.zeros(n)
            idx_one = rng.choice(n, size=n // 2, replace=False)
            start[idx_one] = 1.0
            # add a tiny random jitter to allow fractional moves
            start += rng.random(n) * 0.05
            start = enforce_sum(np.clip(start, 0.0, 1.0), target_sum, rng)

        # hill‑climb with multi‑gradient moves
        remaining = main_limit - (time.time() - start_time)
        max_iters = 2000
        h_loc, c_loc = multi_targeted_hill_climb(
            start,
            rng,
            max_iters=max_iters,
            start_step=0.5,
            target_sum=target_sum,
            top_shifts=3,
            time_limit=remaining,
            start_time=start_time,
        )
        if c_loc < best_c:
            best_h, best_c = h_loc.copy(), c_loc

        # SLSQP refinements with increasing β
        for beta in beta_schedule:
            if time.time() - start_time > main_limit:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, rng, max_iter=400)
            except Exception:
                continue
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_h, best_c = h_ref.copy(), c_ref

        # early exit if we have already beaten the target bound
        if best_c <= 0.3808:
            break

    # ------------------------------------------------------------------
    # Fine‑grained random search (≈14 % of time budget)
    # ------------------------------------------------------------------
    micro_limit = budget_s * 0.99
    while time.time() - start_time < micro_limit:
        nh = perturb_random(best_h, rng, scale=0.02)
        cn = compute_c5(nh)
        if cn < best_c:
            best_h, best_c = nh.copy(), cn
        if best_c <= 0.3808:
            break

    # ------------------------------------------------------------------
    # Optional binary rounding test (may improve the bound)
    # ------------------------------------------------------------------
    n = best_h.size
    target = n // 2
    idx = np.argpartition(-best_h, target)[:target]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    c_bin = compute_c5(h_bin)
    if c_bin < best_c:
        best_h, best_c = h_bin, c_bin

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
