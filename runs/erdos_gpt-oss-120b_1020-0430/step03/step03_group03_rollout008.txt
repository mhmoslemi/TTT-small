def enforce_sum(v, target, rng):
    diff = target - v.sum()
    n = len(v)
    # while diff not small
    max_iter = 1000
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        # Determine eligible indices
        if diff > 0:
            # indices where v < 1
            eligible = np.where(v < 1 - 1e-12)[0]
        else:
            eligible = np.where(v > 1e-12)[0]
        if eligible.size == 0:
            break
        # choose random eligible index
        i = rng.choice(eligible)
        cap = (1 - v[i]) if diff > 0 else v[i]
        add = min(abs(diff), cap)
        if diff < 0:
            add = -add
        v[i] += add
        diff = target - v.sum()
    return v

    n = h0.size

    bounds = [(0.0, 1.0)] * n
    # define smooth objective
    def obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        # scaled difference: but we want max, not min; we use softmax of correlation
        # Actually we want to minimize max of corr. So we approximate max using softmax
        # softmax(max_i a_i) = (1/beta) log(sum exp(beta a_i))
        # Since we minimize that, it approximates max.
        bc = CURRENT_BETA * corr
        m = np.max(bc)
        # compute softmax log-sum-exp
        return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
    res = minimize(obj, h0, method="SLSQP", bounds=bounds, constraints=cons,
                    options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

def compute_c5(h):
    """Discrete C5 for step function h on [0,2] with dx=2/len(h)."""
    n = h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)

def enforce_sum(v, target, rng):
    """Adjust v in‑place so that sum(v)=target while respecting 0≤v≤1."""
    diff = target - v.sum()
    n = v.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(v < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
        else:
            idx = np.where(v > 1e-12)[0]
            if idx.size == 0:
                break
        i = rng.choice(idx)
        cap = (1.0 - v[i]) if diff > 0 else v[i]
        add = min(abs(diff), cap)
        if diff < 0:
            add = -add
        v[i] += add
        diff = target - v.sum()
    return v

def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h≤1, sum=n/2)."""
    h = np.full(n, 0.5, dtype=float)
    for _ in range(n * 8):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

def epsilon_mutation(parent, rng, scale):
    """Transfer ε between two coordinates (preserves sum)."""
    n = parent.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return parent.copy()
    # try moving from i to j
    max_eps = min(parent[i], 1.0 - parent[j])
    if max_eps <= 0:
        # try opposite direction
        max_eps = min(parent[j], 1.0 - parent[i])
        if max_eps <= 0:
            return parent.copy()
        i, j = j, i
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    child = parent.copy()
    child[i] -= eps
    child[j] += eps
    return child

def slsqp_optimize(h0, beta, max_iter=300):
    """Smooth‑max SLSQP refinement."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - CURRENT_TARGET}
    bounds = [(0.0, 1.0)] * n

    def softmax_obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        bc = CURRENT_BETA * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

        enforce_sum(h0, n / 2.0, rng)
    else:
        n = 180
        h0 = random_feasible_vector(n, rng)

    target_sum = n / 2.0

    # ------------------------------------------------------------
    # GA parameters
    # ------------------------------------------------------------
    POP = 80                     # population size
    ELITE_FRAC = 0.25
    OFFSPRING_PER_ELITE = 5
    RANDOM_NEW = 12
    SIGMA_MAX = 0.30
    SIGMA_MIN = 0.001

    elite_cnt = max(1, int(POP * ELITE_FRAC))

    # initialise population
    pop_h = np.empty((POP, n), dtype=float)
    pop_h[0] = h0
    for i in range(1, POP):
        pop_h[i] = random_feasible_vector(n, rng)
    pop_c = np.array([compute_c5(v) for v in pop_h])
    order = np.argsort(pop_c)
    pop_h = pop_h[order]
    pop_c = pop_c[order]
    best_h = pop_h[0].copy()
    best_c = float(pop_c[0])

    # ------------------------------------------------------------
    # main GA loop (≈85% of budget)
    # ------------------------------------------------------------
    GA_TIME = budget_s * 0.85
    while time.time() - start < GA_TIME:
        prog = (time.time() - start) / GA_TIME
        sigma = SIGMA_MAX * (1.0 - prog) + SIGMA_MIN * prog

        # generate offspring
        offspring = []
        elite_parents = pop_h[:elite_cnt]
        for parent in elite_parents:
            for _ in range(OFFSPRING_PER_ELITE):
                child = epsilon_mutation(parent, rng, sigma)
                offspring.append(child)

        # add random newcomers
        for _ in range(RANDOM_NEW):
            offspring.append(random_feasible_vector(n, rng))

        offspring = np.stack(offspring, axis=0)
        off_c = np.array([compute_c5(v) for v in offspring])

        # combine and select next generation
        combined_h = np.concatenate([pop_h, offspring], axis=0)
        combined_c = np.concatenate([pop_c, off_c], axis=0)
        idx = np.argsort(combined_c)[:POP]
        pop_h = combined_h[idx]
        pop_c = combined_c[idx]

        if pop_c[0] < best_c:
            best_c = float(pop_c[0])
            best_h = pop_h[0].copy()

    # ------------------------------------------------------------
    # final SLSQP polishing (remaining budget)
    # ------------------------------------------------------------
    for beta in (200, 800, 2000, 8000, 20000):
        if time.time() - start > budget_s - 0.5:
            break
        try:
            refined = slsqp_optimize(best_h, beta, max_iter=400)
            rc = compute_c5(refined)
            if rc < best_c:
                best_c = rc
                best_h = refined.copy()
        except Exception:
            continue

    # final safety clip and sum correction
    best_h = np.clip(best_h, 0.0, 1.0)
    enforce_sum(best_h, target_sum, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
