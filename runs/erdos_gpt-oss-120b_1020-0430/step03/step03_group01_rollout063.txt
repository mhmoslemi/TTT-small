def enforce_sum(h, target, rng, max_iter=5000):

    h = np.clip(h, 0.0, 1.0)

    n = h.size

        if abs(diff) < 1e-12:

        h += corr

CURRENT_BETA = None

def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")

        i = rng.integers(0, n)

                dj = -1.0

            dj = 1.0
        elif max_eps_j_i > 0:

            dj = -1.0

    remain = budget_s - elapsed

        # try increasing β values
        for beta in (500, 2000, 5000):
            if remain < 5.0:
                break
            refined = slsqp_optimize(best_h, beta, max_iter=250)
            corr_ref = np.correlate(refined, 1 - refined, mode="full")
            raw_max_ref = corr_ref.max()
            if raw_max_ref < best_raw_max:
                best_h = refined
                best_raw_max = raw_max_ref
            # update remaining time
            elapsed = time.time() - start_time
            remain = budget_s - elapsed

    # 4. Ensure feasibility exactly
    n = best_h.size
    best_h = np.clip(best_h, 0.0, 1.0)
    target = n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for i in range(n):
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()
            if abs(diff) < 1e-12:
                break

    final_c = np.max(np.correlate(best_h, 1 - best_h, mode="full")) * (2.0 / n)
    return best_h.tolist(), float(final_c), int(n)
