def compute_c5(h):
   n = h.shape[0]
   # zero pad to length 2*n - 1
   N = 1
   while N < 2*n - 1:
       N <<= 1
   # compute FFT
   H = np.fft.rfft(h, N)
   B = np.fft.rfft(1 - h, N)
   corr = np.fft.irfft(H * np.conj(B), N)
   # take first 2*n-1 entries
   corr = corr[:2*n-1]
   max_corr = np.max(corr)
   return max_corr * (2.0 / n)

def subgradient_optimize(h0, rng, max_time):
    n = len(h0)
    target = n / 2.0
    h = np.clip(h0, 0, 1)
    # ensure sum exactly
    h = project_to_feasible(h, target)
    best_h = h.copy()
    best_c = compute_c5(h)
    step = 0.2   # initial step
    start = time.time()
    it = 0
    while time.time() - start < max_time:
        # compute correlation
        n = h.size
        # Use FFT compute
        N = 1
        while N < 2*n - 1:
            N <<= 1
        H = np.fft.rfft(h, N)
        B = np.fft.rfft(1 - h, N)
        corr = np.fft.irfft(H * np.conj(B), N)[:2*n-1]
        # get worst shift
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        # gradient
        grad = np.empty(n)
        # compute contributions vectorized:
        # For i - shift part: define arr1 = np.arange(n) - shift, mask1 = (0 <= arr1 < n)
        # For i + shift part: arr2 = np.arange(n) + shift, mask2 = (0 <= arr2 < n)
        idx1 = np.arange(n) - shift
        mask1 = (idx1 >= 0) & (idx1 < n)
        grad[mask1] = (1 - h[idx1[mask1]])
        # else 0
        idx2 = np.arange(n) + shift
        mask2 = (idx2 >= 0) & (idx2 < n)
        # subtract h at idx2
        grad[mask2] -= h[idx2[mask2]]
        # update
        # step size schedule
        # e.g., step = initial / sqrt(it+1)
        step_size = step / np.sqrt(it + 1)
        h_new = h - step_size * grad
        # project
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_feasible(h_new, target)
        h = h_new
        # evaluate
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        it += 1
        # occasional random perturb to escape local optimum
        if it % 200 == 0:
            if rng.random() < 0.1:
                # small random noise
                noise = rng.uniform(-0.1, 0.1, n)
                h = np.clip(h + noise, 0, 1)
                h = project_to_feasible(h, target)
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ---------- feasibility utilities ----------
def project_to_feasible(v, target):
    """Project v onto {0≤x≤1, sum(x)=target} by shifting and clipping."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def random_feasible_vector(n, rng):
    """Generate a random h with 0≤h≤1 and sum = n/2."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_to_feasible(v, n / 2.0)

def upsample_initial(to_n):
    """Linear interpolation of the global initial_h_values to length to_n."""
    init = np.array(initial_h_values, dtype=float)
    m = init.size
    xs_old = np.linspace(0.0, 2.0, m, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, to_n, endpoint=False)
    return np.interp(xs_new, xs_old, init)

# ---------- exact C5 evaluation ----------
def compute_c5(h):
    """Exact discrete C5 using np.correlate."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

# ---------- fast max-correlation and shift ----------
def max_corr_shift(h):
    """Return (max_corr, shift) where shift∈[-n+1,n-1] giving the max."""
    n = h.size
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    H = np.fft.rfft(h, N)
    B = np.fft.rfft(1 - h, N)
    corr = np.fft.irfft(H * np.conj(B), N)[: 2 * n - 1]
    idx = int(np.argmax(corr))
    return float(corr[idx]), idx - (n - 1)

# ---------- gradient for a given shift ----------
def gradient_for_shift(h, shift):
    """Subgradient of g(k)=∑ h[i+shift]*(1−h[i]) w.r.t. h."""
    n = h.size
    grad = np.zeros(n)
    idx1 = np.arange(n) - shift
    m1 = (idx1 >= 0) & (idx1 < n)
    grad[m1] = 1 - h[idx1[m1]]
    idx2 = np.arange(n) + shift
    m2 = (idx2 >= 0) & (idx2 < n)
    grad[m2] -= h[idx2[m2]]
    return grad

# ---------- subgradient optimisation ----------
def subgradient_optimize(h0, rng, max_time):
    """Projected sub‑gradient descent on the worst shift."""
    n = h0.size
    target = n / 2.0
    h = project_to_feasible(np.clip(h0, 0.0, 1.0), target)
    best_h = h.copy()
    best_c = compute_c5(h)
    step0 = 0.25
    start = time.time()
    it = 0
    while time.time() - start < max_time:
        max_corr, shift = max_corr_shift(h)
        grad = gradient_for_shift(h, shift)
        step = step0 / np.sqrt(it + 1)
        h = h - step * grad
        h = project_to_feasible(np.clip(h, 0.0, 1.0), target)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        it += 1
        # occasional random jitter to escape plateaus
        if it % 250 == 0 and rng.random() < 0.12:
            h = project_to_feasible(
                np.clip(h + rng.uniform(-0.08, 0.08, n), 0.0, 1.0), target
            )
    return best_h, best_c

# ---------- hill climbing ----------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb(h_start, rng, max_iters=2000, start_step=0.3):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h.copy()
            h = new_h
        if (it + 1) % decay == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c

# ---------- SLSQP refinement (soft‑max surrogate) ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

# ---------- main driver ----------
