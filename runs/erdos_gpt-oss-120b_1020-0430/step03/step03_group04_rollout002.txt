import time
import numpy as np
from scipy.optimize import minimize, differential_evolution

# -------------------------------------------------------------
# basic utilities (all top‑level, no closures)
# -------------------------------------------------------------
def compute_c5(h):
    """Exact discrete C5 for step function h on [0,2] with uniform spacing."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def fix_sum_and_bounds(h, target, rng):
    """Clip to [0,1] and redistribute mass to hit the exact target sum."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    n = h.size
    # deterministic sweep
    for i in range(n):
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
        if abs(diff) < 1e-12:
            break
    # random fallback if needed
    if abs(diff) > 1e-12:
        for _ in range(5000):
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2500, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


def gradient_refine(h, rng, step=0.02, max_iter=12):
    """Refine by moving against gradient of the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]      # + (1 - h[i+k])
            g[k:] -= h[:upper]            # - h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h = fix_sum_and_bounds(h, target, rng)
    return h


def random_feasible_vector(n, rng):
    """Uniform random vector in [0,1] with exact sum n/2."""
    h = rng.random(n)
    target = n / 2.0
    scale = target / h.sum()
    h *= scale
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h = fix_sum_and_bounds(h, target, rng)
    return h


def simulated_annealing(h0, rng, deadline):
    """Simple SA respecting the sum constraint."""
    n = h0.size
    target = n / 2.0
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    T0 = max(0.01 * cur_c, 1e-4)
    it = 0
    while time.time() < deadline and it < 8000:
        step = max(0.5 * (0.999 ** it), 5e-4)
        cand_h = perturb_h(cur_h, rng, scale=step)
        cand_c = compute_c5(cand_h)
        accept = False
        if cand_c < cur_c:
            accept = True
        else:
            T = T0 * (0.999 ** it) + 1e-12
            if rng.random() < np.exp(-(cand_c - cur_c) / T):
                accept = True
        if accept:
            cur_h, cur_c = cand_h, cand_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
        it += 1
    return best_h, best_c


# -------------------------------------------------------------
# DE helper class
# -------------------------------------------------------------
class DEObjective:
    def __init__(self, n, x_grid, x_ctrl, target, rng):
        self.n = n
        self.x_grid = x_grid
        self.x_ctrl = x_ctrl
        self.target = target
        self.rng = rng

    def __call__(self, y):
        y = np.clip(y, 0.0, 1.0)
        h = np.interp(self.x_grid, self.x_ctrl, y)
        h = np.clip(h, 0.0, 1.0)
        h = fix_sum_and_bounds(h, self.target, self.rng)
        return compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_h, best_c, best_n = None, np.inf, None

    parent = globals().get("initial_h_values", None)
    if parent is not None:
        parent = np.array(parent, dtype=float)

    candidate_ns = [72, 96, 108, 120, 144, 180, 210, 240]

    for n in candidate_ns:
        if time.time() - start_time > budget_s * 0.97:
            break

        target_sum = n / 2.0
        x_grid = np.linspace(0.0, 2.0, n, endpoint=False)
        m = max(6, n // 15)          # number of control points
        x_ctrl = np.linspace(0.0, 2.0, m, endpoint=False)

        # ---- DE optimisation ----
        de_obj = DEObjective(n, x_grid, x_ctrl, target_sum, rng)
        try:
            de_res = differential_evolution(
                de_obj,
                bounds=[(0.0, 1.0)] * m,
                maxiter=200,
                popsize=12,
                polish=False,
                seed=int(rng.integers(1_000_000)),
                updating='deferred',
                workers=1,
            )
            y_opt = np.clip(de_res.x, 0.0, 1.0)
            h_opt = np.interp(x_grid, x_ctrl, y_opt)
            h_opt = fix_sum_and_bounds(np.clip(h_opt, 0.0, 1.0), target_sum, rng)
            cur_h, cur_c = h_opt, compute_c5(h_opt)
        except Exception:
            cur_h = random_feasible_vector(n, rng)
            cur_c = compute_c5(cur_h)

        # incorporate parent if sizes match (might be already good)
        if parent is not None and parent.size == n:
            p_h = fix_sum_and_bounds(parent.copy(), target_sum, rng)
            p_c = compute_c5(p_h)
            if p_c < cur_c:
                cur_h, cur_c = p_h, p_c

        # ---- local refinement cascade ----
        cur_h = gradient_refine(cur_h, rng, step=0.02, max_iter=15)
        cur_c = compute_c5(cur_h)

        cur_h, cur_c = hill_climb(cur_h, rng, max_iters=2000, start_step=0.5)

        deadline_sa = start_time + budget_s * 0.98
        if time.time() < deadline_sa:
            cur_h, cur_c = simulated_annealing(cur_h, rng, deadline_sa)

        cur_h = gradient_refine(cur_h, rng, step=0.01, max_iter=10)
        cur_c = compute_c5(cur_h)

        if cur_c < best_c:
            best_h, best_c, best_n = cur_h.copy(), cur_c, n

        if time.time() - start_time > budget_s * 0.99:
            break

    # ----- safety net -----
    if best_h is None:
        best_n = 180
        best_h = np.full(best_n, 0.5)
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    best_h = fix_sum_and_bounds(best_h, target, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
