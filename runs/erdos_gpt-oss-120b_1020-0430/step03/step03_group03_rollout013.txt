import time, math, numpy as np
from scipy.optimize import minimize

# ------------------- basic helpers -------------------
def compute_c5(h):
    """Discrete C5 for step function h on [0,2] (dx=2/len(h))."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)

def enforce_sum(h, target, rng):
    """Adjust in‑place so that sum(h)=target while 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    """Generate a feasible h with average 0.5."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    enforce_sum(h, n / 2.0, rng)
    return h

def perturb_h(h, rng, scale=0.2):
    """Transfer ε between two coordinates (preserves sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

def biased_perturb_h(h, rng, scale=0.2):
    """Bias transfer away from positions contributing most to the max overlap."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    k = np.argmax(corr)
    shift = k - (n - 1)
    i0 = max(0, -shift)
    i1 = min(n, n - shift)
    idx = np.arange(i0, i1)
    prod = h[idx] * (1 - h[idx + shift])
    if prod.size == 0:
        return perturb_h(h, rng, scale)
    thr_hi = np.quantile(prod, 0.70)
    thr_lo = np.quantile(prod, 0.30)
    high = idx[prod > thr_hi]
    low = idx[prod < thr_lo]
    if high.size == 0 or low.size == 0:
        return perturb_h(h, rng, scale)
    i = rng.choice(high)
    j = rng.choice(low)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

# ------------------- SLSQP with smooth max -------------------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp trick
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=200):
    """Smooth‑max SLSQP refinement."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {"type": "eq", "fun": sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

# ------------------- Simulated annealing -------------------
def simulated_annealing(h0, rng, max_iter, init_temp, final_temp):
    """Simple SA on the true C5 (lower is better)."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    n = h.size
    log_ratio = math.log(final_temp / init_temp) if init_temp > 0 else 0.0
    for it in range(1, max_iter + 1):
        T = init_temp * math.exp(log_ratio * it / max_iter)
        cand = biased_perturb_h(h, rng, scale=0.3) if rng.random() < 0.7 else perturb_h(h, rng, scale=0.3)
        cand_c = compute_c5(cand)
        diff = cand_c - cur_c
        if diff < 0 or rng.random() < math.exp(-diff / max(T, 1e-12)):
            h, cur_c = cand, cand_c
            if cand_c < best_c:
                best_c, best_h = cand_c, cand.copy()
    return best_h, best_c

# ------------------- Differential Evolution -------------------
def differential_evolution(pop, fitness, target_sum, rng, time_limit):
    """Custom DE (rand/1/bin) with sum enforcement."""
    NP, n = pop.shape
    best_idx = np.argmin(fitness)
    best_h = pop[best_idx].copy()
    best_c = fitness[best_idx]
    start = time.time()
    while time.time() - start < time_limit:
        for i in range(NP):
            # choose three distinct others
            idxs = list(range(NP))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            F = 0.8
            mutant = pop[a] + F * (pop[b] - pop[c])
            np.clip(mutant, 0.0, 1.0, out=mutant)
            enforce_sum(mutant, target_sum, rng)
            # binomial crossover
            jrand = rng.integers(0, n)
            mask = rng.random(n) < 0.9
            mask[jrand] = True
            trial = pop[i].copy()
            trial[mask] = mutant[mask]
            enforce_sum(trial, target_sum, rng)
            trial_c = compute_c5(trial)
            if trial_c <= fitness[i]:
                pop[i] = trial
                fitness[i] = trial_c
                if trial_c < best_c:
                    best_c = trial_c
                    best_h = trial.copy()
        # occasional diversity injection
        if rng.random() < 0.05:
            worst = np.argmax(fitness)
            new = random_feasible_vector(n, rng)
            pop[worst] = new
            fitness[worst] = compute_c5(new)
            if fitness[worst] < best_c:
                best_c = fitness[worst]
                best_h = new.copy()
    return best_h, best_c

# ------------------- Main entry point -------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ---- initial vector and dimension ----

    NP = 30

    # ---- DE phase (≈60% of budget) ----
    de_time = budget_s * 0.6
    best_h, best_c = differential_evolution(pop, fitness, target_sum, rng, de_time)

    # ---- Simulated annealing phase (≈30% of budget) ----
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 5.0:
        max_iter = int(min(200000, remaining * 2000))
        best_h, best_c = simulated_annealing(best_h, rng, max_iter,
                                             init_temp=0.04, final_temp=1e-7)

    # ---- SLSQP polishing (remaining time) ----
    while time.time() - start_time < budget_s:
        for beta in (50, 200, 800, 2000, 8000, 20000):

            if rc < best_c:
                best_c = rc
                best_h = refined

    # ---- final cleanup ----
    best_h = np.clip(best_h, 0.0, 1.0)
    enforce_sum(best_h, target_sum, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
