def slsqp_refine(h, beta, max_iter=200):
    n = h.size
    target = n / 2.0
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        m = np.max(corr)
        return m
    # to smooth: use softmax
    def soft_obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        m = np.max(corr)
        # softmax using beta
        exps = np.exp(beta * (corr - m))
        return (m + np.log(np.sum(exps)) / beta)
    cons = {'type':'eq','fun':lambda x: np.sum(x) - target}
    bounds = [(0.0,1.0)]*n
    res = minimize(soft_obj, h, method='SLSQP', bounds=bounds, constraints=[cons], options={'maxiter':max_iter,'ftol':1e-12,'disp':False})
    x = np.clip(res.x,0,1)
    # enforce sum exactly
    diff = target - x.sum()
    # adjust as before
    for _ in range(1000):
        if abs(diff) < 1e-12: break
        i = rng.integers(0,n)

import time
import numpy as np

def compute_c5_detail(h):
    """Return (C5, shift, correlation) for step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr)
    return corr[idx] * dx, idx - (n - 1), corr


def enforce_sum(h, rng):
    """Project h onto the simplex Σh = n/2 while keeping 0≤h≤1."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple redistribution loop
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Random h with 0≤h≤1 and Σh=n/2."""
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, rng)


def worst_gradient_from_shift(h, k):
    """Gradient of the worst‐shift overlap for shift k≥0."""
    n = h.size
    grad = np.ones_like(h)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad[: n - k] = 1.0 - h[k:]      # d/dh_i for i in [0, n‑k‑1]
        grad[k:] -= h[: n - k]          # subtract contribution for i in [k, n‑1]
    return grad


def slsqp_refine(h, beta, rng, max_iter=300):
    """SLSQP refinement using a soft‑max approximation of the max."""
    n = h.size
    target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        m = np.max(corr)
        exps = np.exp(beta * (corr - m))
        return m + np.log(np.sum(exps)) / beta

    cons = ({"type": "eq", "fun": lambda x: np.sum(x) - target},)
    bounds = [(0.0, 1.0)] * n
    from scipy.optimize import minimize

    res = minimize(
        obj,
        h,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return enforce_sum(x, rng)
