def simulated_annealing_binary(h_init, rng, max_steps, time_limit):
    n = len(h_init)
    h = h_init.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    # maintain lists
    ones = [i for i, val in enumerate(h) if val == 1]
    zeros = [i for i, val in enumerate(h) if val == 0]
    start = time.time()
    T0 = 0.02
    T_end = 1e-6
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        # temperature
        # use geometric schedule
        T = T0 * ((T_end / T0) ** (step / max_steps))
        # pick random swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # propose swap
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1
        new_c = compute_c5(h_new)
        delta = new_c - cur_c
        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            # accept move
            h = h_new
            cur_c = new_c
            # update lists
            # swap i from ones to zeros, j from zeros to ones
            ones.remove(i)
            zeros.append(i)
            zeros.remove(j)
            ones.append(j)
            # update best
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        # optional early break if best_c below target
        if best_c < 0.3805:  # early stop if very good
            break
    return best_h, best_c

def slsqp_optimize(h0, beta, max_iter=250):
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type":"eq", "fun": lambda x: np.sum(x) - target}
    # we need softmax_obj which depends on global beta
    global CURRENT_BETA
    CURRENT_BETA = beta
    def obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = CURRENT_BETA * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
    res = minimize(obj, h0, method="SLSQP", bounds=bounds, constraints=cons,
                    options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum exactly
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # distribute diff over entries that can accommodate
        # simple linear correction
        x += diff / n
        x = np.clip(x, 0.0, 1.0)
        diff = target - x.sum()
        # if still off, adjust first element
        if abs(diff) > 1e-12:
            for i in range(n):
                add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
                x[i] += add
                diff = target - x.sum()
                if abs(diff) < 1e-12:
                    break
    return x

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb_multi(h_start, rng, max_iters=2000, init_step=0.5, n_trials=5):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c = c
                h = new_h
                improved = True
                break
        if not improved and rng.random() < 0.02:
            new_h = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(new_h)
            if c < best_c:
                best_c = c
                h = new_h
                improved = True
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
        # early exit if best_c very low
        if best_c < 0.3805:
            break
    return h, best_c

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_multi(h_start, rng, max_iters=2000, init_step=0.5, n_trials=5):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c = c
                h = new_h
                improved = True
                break
        if not improved and rng.random() < 0.02:
            new_h = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(new_h)
            if c < best_c:
                best_c = c
                h = new_h
                improved = True
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
        if best_c < 0.3805:
            break
    return h, best_c


def simulated_annealing_binary(h_init, rng, max_steps, time_limit):
    n = h_init.size
    h = h_init.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    ones = [i for i, v in enumerate(h) if v == 1]
    zeros = [i for i, v in enumerate(h) if v == 0]
    start = time.time()
    T0 = 0.02
    T_end = 1e-6
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        T = T0 * ((T_end / T0) ** (step / max_steps))
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1
        new_c = compute_c5(h_new)
        delta = new_c - cur_c
        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            h = h_new
            cur_c = new_c
            # update lists
            ones.remove(i)
            zeros.append(i)
            zeros.remove(j)
            ones.append(j)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        if best_c < 0.3805:
            break
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    init_vals = globals().get("initial_h_values")
    if init_vals is not None:
        init_float = np.array(init_vals, dtype=float)
        n = init_float.size
        idx = np.argsort(-init_float)
        h_bin = np.zeros(n, dtype=int)
        h_bin[idx[: n // 2]] = 1
    else:
        n = 180
        h_bin = np.zeros(n, dtype=int)
        h_bin[: n // 2] = 1
        rng.shuffle(h_bin)

    global_best_h = None
    global_best_c = np.inf

    # ------------------------------------------------------------------
    # Simulated annealing on binary vectors
    # ------------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.65:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 5.0:
            break

        # pick a start binary vector
        if rng.random() < 0.3:
            start_h = h_bin.copy()
            for _ in range(12):
                i, j = rng.integers(0, n, size=2)
                if start_h[i] != start_h[j]:
                    start_h[i], start_h[j] = start_h[j], start_h[i]
        else:
            start_h = np.zeros(n, dtype=int)
            start_h[: n // 2] = 1
            rng.shuffle(start_h)

        # allocate steps (rough estimate: 2e5 steps ≈ 80 s for n≈360)
        est_step_time = 0.0004
        max_steps = int(min(200000, remaining / est_step_time * 0.6))
        if max_steps < 2000:
            break

        h_sa, c_sa = simulated_annealing_binary(start_h, rng, max_steps, remaining * 0.6)
        if c_sa < global_best_c:
            global_best_c = c_sa
            global_best_h = h_sa.astype(float)

        # quick hill‑climb refinement on the real‑valued version
        if global_best_h is not None:
            h_hc, c_hc = hill_climb_multi(global_best_h, rng, max_iters=1500,
                                          init_step=0.3, n_trials=6)
            if c_hc < global_best_c:
                global_best_c = c_hc
                global_best_h = h_hc.copy()

            # SLSQP ladder
            for beta in (50, 200, 500, 1000):
                if time.time() - start_time > budget_s * 0.85:
                    break
                try:
                    h_ref = slsqp_optimize(global_best_h, beta, max_iter=300)
                    c_ref = compute_c5(h_ref)
                    if c_ref < global_best_c:
                        global_best_c = c_ref
                        global_best_h = h_ref.copy()
                except Exception:
                    continue

    # ------------------------------------------------------------------
    # Final clean‑up
    # ------------------------------------------------------------------
    if global_best_h is None:
        if init_vals is not None:
            global_best_h = np.array(initial_h_values, dtype=float)
        else:
            global_best_h = np.full(n, 0.5)
    # enforce bounds and exact sum
    global_best_h = np.clip(global_best_h, 0.0, 1.0)
    target = n / 2.0
    diff = target - global_best_h.sum()
    if abs(diff) > 1e-12:
        global_best_h += diff / n
        global_best_h = np.clip(global_best_h, 0.0, 1.0)
        diff = target - global_best_h.sum()
        if abs(diff) > 1e-12:
            # final tiny fix on a single coordinate
            i = int(np.argmin(global_best_h) if diff > 0 else np.argmax(global_best_h))
            add = min(diff, 1.0 - global_best_h[i]) if diff > 0 else max(diff, -global_best_h[i])
            global_best_h[i] += add

    final_c = compute_c5(global_best_h)
    return global_best_h.tolist(), float(final_c), int(n)
