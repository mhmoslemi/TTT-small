    idx_center = n - 1
    s = np.arange(-n+1, n, dtype=int)  # shape 2n-1

    delta_self = np.zeros_like(self_corr)

    # i - s
    mask_i_minus = (i - s >= 0) & (i - s < n)
    delta_self[mask_i_minus] += -eps * h[i - s[mask_i_minus]]

def local_search(h, n_iters, rng, step_scale_start=0.5):
    n = len(h)
    dx = 2.0 / n
    # compute correlations
    self_corr = np.correlate(h, h, mode='full')
    h_corr1 = np.correlate(h, np.ones(n), mode='full')
    O = h_corr1 - self_corr
    cur_c = np.max(O) * dx
    best_h = h.copy()
    best_c = cur_c
    step_scale = step_scale_start
    center = n - 1
    # pre-alloc s range
    s_range = range(-n+1, n)
    for it in range(n_iters):
        # compute worst shift
        idx_max = np.argmax(O)
        k = idx_max - center
        k_abs = k if k >= 0 else -k
        # gradient
        grad = np.ones(n)
        if k_abs == 0:
            grad -= 2 * h
        else:
            grad[: n - k_abs] -= h[k_abs:]
            grad[k_abs:] -= h[: n - k_abs]
        # choose i, j
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            # random pair
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 1e-12:
            # try opposite
            max_transfer = min(h[j], 1.0 - h[i])
            if max_transfer > 1e-12:
                i, j = j, i

import time, numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h):
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, rng):
    """Project h onto Σh = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, rng)


# ---------- soft‑max SLSQP refinement ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    bc = CURRENT_BETA * np.correlate(x, 1 - x, mode="full")
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return enforce_sum(x, rng)


# ---------- local search with incremental updates ----------
def local_search(h, rng, start_time, budget_s, max_iters=100000):
    n = h.size
    dx = 2.0 / n
    center = n - 1
    # initial correlations
    self_corr = np.correlate(h, h, mode="full")
    h_corr1 = np.correlate(h, np.ones(n), mode="full")
    O = h_corr1 - self_corr
    cur_c = np.max(O) * dx
    best_h = h.copy()
    best_c = cur_c
    step_scale = 0.5
    s_range = range(-n + 1, n)

    it = 0
    while it < max_iters:
        if time.time() - start_time > 0.94 * budget_s:
            break
        # worst shift & gradient
        idx_max = np.argmax(O)
        k = idx_max - center
        k_abs = -k if k < 0 else k
        grad = np.ones(n)
        if k_abs == 0:
            grad -= 2 * h
        else:
            grad[: n - k_abs] -= h[k_abs:]
            grad[k_abs:] -= h[: n - k_abs]

        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)

        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 1e-12:
            # try opposite direction
            max_transfer = min(h[j], 1.0 - h[i])
            if max_transfer > 1e-12:
                i, j = j, i
            else:
                # random feasible pair
                pos = np.where(h > 1e-12)[0]
                neg = np.where(h < 1 - 1e-12)[0]
                if pos.size == 0 or neg.size == 0:
                    break
                i = rng.choice(pos)
                j = rng.choice(neg)
                max_transfer = min(h[i], 1.0 - h[j])

        eps = max_transfer * step_scale * rng.random()
        if eps <= 0:
            it += 1
            continue

        # backup for possible revert
        sc_old = self_corr.copy()
        hc_old = h_corr1.copy()
        h_old = h.copy()

        # incremental update
        for s in s_range:
            idx = s + center
            # self‑corr linear part
            p = i - s
            if 0 <= p < n:
                self_corr[idx] += -eps * h[p]
            p = j - s
            if 0 <= p < n:
                self_corr[idx] += eps * h[p]
            p = i + s
            if 0 <= p < n:
                self_corr[idx] += -eps * h[p]
            p = j + s
            if 0 <= p < n:
                self_corr[idx] += eps * h[p]
            # quadratic part
            if s == 0:
                self_corr[idx] += 2 * eps * eps
            elif s == j - i:
                self_corr[idx] -= eps * eps
            elif s == i - j:
                self_corr[idx] -= eps * eps
            # h_corr1 linear part
            if 0 <= i + s < n:
                h_corr1[idx] -= eps
            if 0 <= j + s < n:
                h_corr1[idx] += eps

        h[i] -= eps
        h[j] += eps

        O = h_corr1 - self_corr
        new_c = np.max(O) * dx
        if new_c < cur_c - 1e-14:
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            self_corr = sc_old
            h_corr1 = hc_old
            h = h_old
            O = h_corr1 - self_corr

        # decay step scale
        step_scale = max(step_scale * 0.9995, 1e-4)
        it += 1
        # occasional time check
        if (it & 1023) == 0 and time.time() - start_time > 0.94 * budget_s:
            break
    return best_h, best_c


# ---------- main driver ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # use parent construction if available
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0, c0, h0.size

    # discretisation candidates (increase gradually)
    candidate_ns = [96, 120, 144, 168, 192, 216, 240, 256]
    if best_n is not None and best_n in candidate_ns:
        candidate_ns = [best_n] + [n for n in candidate_ns if n != best_n]

    for n in candidate_ns:
        if time.time() - start > 0.95 * budget_s:
            break

        # initialise h for this n
        if best_h is not None:
            if best_n == n:
                h = best_h.copy()
            else:
                x_old = np.linspace(0, 2, best_n, endpoint=False)
                x_new = np.linspace(0, 2, n, endpoint=False)
                h = np.interp(x_new, x_old, best_h)
                h = np.clip(h, 0.0, 1.0)
                h = enforce_sum(h, rng)
        else:
            h = random_feasible_vector(n, rng)

        # local search with incremental updates
        h, c = local_search(h, rng, start, budget_s, max_iters=120000)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

        # soft‑max SLSQP polishing
        for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0):
            if time.time() - start > 0.96 * budget_s:
                break
            maxit = 600 if beta >= 200 else 400
            h = slsqp_optimize(h, beta, rng, max_iter=maxit)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

        # a few extra refinements with smaller step sizes
        if time.time() - start < 0.97 * budget_s:
            h, c = local_search(h, rng, start, budget_s, max_iters=30000)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

    # optional up‑sampling if time allows
    if best_h is not None and time.time() - start < 0.99 * budget_s:
        if best_n * 2 <= 260:
            h_up = np.repeat(best_h, 2)
            h_up, c_up = local_search(h_up, rng, start, budget_s, max_iters=80000)
            if c_up < best_c:
                best_h, best_c, best_n = h_up.copy(), c_up, h_up.size
            for beta in (5.0, 20.0, 80.0, 200.0):
                if time.time() - start > budget_s:
                    break
                h_up = slsqp_optimize(best_h, beta, rng, max_iter=300)
                c_up = compute_c5(h_up)
                if c_up < best_c:
                    best_h, best_c, best_n = h_up.copy(), c_up, h_up.size

    # final feasibility enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
