def softmax_correlation_and_grad(h, beta):
    c = np.correlate(h, 1 - h, mode='full')
    maxc = np.max(c) # for log-sum-exp stability.
    exp_terms = np.exp(beta * (c - maxc))
    sum_exp = np.sum(exp_terms)
    f = (maxc + np.log(sum_exp)) / beta
    # compute probs
    p = exp_terms / sum_exp
    # compute gradient
    n = h.size
    grad = np.zeros(n)
    # shift from -(n-1) .. n-1
    for shift in range(-(n-1), n):
        w = p[shift + (n - 1)]
        if w == 0.0:
            continue
        i_start = max(0, -shift)
        i_end = min(n, n - shift)
        # vectorized: i = np.arange(i_start, i_end); j = i+shift
        i = np.arange(i_start, i_end)
        j = i + shift
        # use broadcasting for gradient contributions
        hi = h[i]
        hj = h[j]
        grad[i] += w * (1 - hj)
        grad[j] -= w * hi
    return f, grad

def project_to_feasible(v, target, tol=1e-9, max_iter=60):
    # Ensure box constraints after shifting lambda
    n = v.size
    # initial low/high for lambda
    low = -np.max(v) - 1.0
    high = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (low + high) / 2.0
        h = np.clip(v - lam, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) <= tol:
            return h
        if s > target:
            # need to increase lam to lower h
            low = lam
        else:
            high = lam
    # after loop, return h
    h = np.clip(v - (low + high) / 2.0, 0.0, 1.0)
    # Ensure sum exactly target by adjusting small residual
    # simple: if sum too low, distribute extra to elements not at 1
    diff = target - h.sum()
    if abs(diff) > 1e-8:
        # adjust proportionally to (1 - h) for positive diff, else reduce from h
        if diff > 0:
            mask = h < 1 - 1e-12
            inc = np.minimum(diff / np.maximum(mask.sum(), 1), 1 - h[mask])
            h[mask] += inc
        else:
            mask = h > 1e-12
            dec = np.minimum(-diff / np.maximum(mask.sum(), 1), h[mask])
            h[mask] -= dec
    return h

import time, math, numpy as np
from scipy.optimize import minimize

# ---------- helper functions ----------
def compute_c5(h):
    """Exact discrete C5 for step function h (dx=2/n)."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)


def enforce_sum(h, target, rng):
    """Adjust h in‑place so that sum(h)=target while respecting 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate random h with 0≤h≤1 and sum=n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    enforce_sum(h, n / 2.0, rng)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer ε between two coordinates (preserves sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new


def biased_perturb_h(h, rng, scale=0.2):
    """Bias transfer away from positions contributing most to the max overlap."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    k = np.argmax(corr)
    shift = k - (n - 1)
    i0 = max(0, -shift)
    i1 = min(n, n - shift)
    idx = np.arange(i0, i1)
    prod = h[idx] * (1 - h[idx + shift])
    if prod.size == 0:
        return perturb_h(h, rng, scale)
    thr_hi = np.quantile(prod, 0.70)
    thr_lo = np.quantile(prod, 0.30)
    high = idx[prod > thr_hi]
    low = idx[prod < thr_lo]
    if high.size == 0 or low.size == 0:
        return perturb_h(h, rng, scale)
    i = rng.choice(high)
    j = rng.choice(low)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new


# ---------- SLSQP surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on smooth max‑approximation."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {"type": "eq", "fun": sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x


# ---------- Simulated annealing ----------
def simulated_annealing(h0, rng, max_iter, init_temp, final_temp):
    """Simple SA on C5 (lower is better)."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    n = h.size
    log_ratio = math.log(final_temp / init_temp) if init_temp > 0 else 0.0
    for it in range(1, max_iter + 1):
        T = init_temp * math.exp(log_ratio * it / max_iter)
        cand = biased_perturb_h(h, rng, scale=0.3) if rng.random() < 0.7 else perturb_h(h, rng, scale=0.3)
        cand_c = compute_c5(cand)
        diff = cand_c - cur_c
        if diff < 0 or rng.random() < math.exp(-diff / max(T, 1e-12)):
            h, cur_c = cand, cand_c
            if cand_c < best_c:
                best_c, best_h = cand_c, cand.copy()
    return best_h, best_c


# ---------- Initialisation ----------
def get_initial_h(target_n, rng):
    """Return a feasible start vector of length target_n."""
    if "initial_h_values" in globals():
        h_parent = np.array(initial_h_values, dtype=float)
        parent_n = h_parent.size
        if target_n % parent_n == 0:
            factor = target_n // parent_n
            h = np.repeat(h_parent, factor)
            enforce_sum(h, target_n / 2.0, rng)
            return h
        else:
            # simple interpolation to target_n
            x_old = np.linspace(0.0, 2.0, parent_n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
            h = np.interp(x_new, x_old, h_parent)
            enforce_sum(h, target_n / 2.0, rng)
            return h
    # no parent: generate random
    return random_feasible_vector(target_n, rng)


# ---------- main run ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # target sample size – 360 (double typical 180)
    n = 360
    h0 = get_initial_h(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # allocate ~85% of time to SA + cheap SLSQP, rest to final polishing
    sa_time = budget_s * 0.85

    while time.time() - start_time < sa_time:
        # choose fresh start: perturb best or random
        start = perturb_h(best_h, rng, scale=0.3) if rng.random() < 0.6 else random_feasible_vector(n, rng)

        # estimate remaining time and allocate iterations
        remaining = max(0.0, sa_time - (time.time() - start_time))
        # about 0.7 ms per iteration (empirical)
        max_iter = int(min(50000, remaining * 1400))
        max_iter = max(5000, max_iter)

        cand_h, cand_c = simulated_annealing(
            start,
            rng,
            max_iter=max_iter,
            init_temp=0.05,      # slightly higher to explore larger space
            final_temp=1e-6,
        )

        if cand_c < best_c:
            best_h, best_c = cand_h.copy(), cand_c

        # occasional low‑beta SLSQP refinements
        for beta in (50, 200, 800):
            if time.time() - start_time > sa_time:
                break
            try:
                refined = slsqp_optimize(best_h, beta, max_iter=300)
                rc = compute_c5(refined)
                if rc < best_c:
                    best_h, best_c = refined, rc
            except Exception:
                continue

    # ---------- final polishing ----------
    for beta in (2000, 8000, 20000, 50000):
        if time.time() - start_time > budget_s:
            break
        try:
            best_h = slsqp_optimize(best_h, beta, max_iter=250)
            best_c = compute_c5(best_h)
        except Exception:
            continue

    # final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
