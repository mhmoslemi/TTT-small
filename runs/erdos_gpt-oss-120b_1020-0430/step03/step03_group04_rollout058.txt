def gradient_for_lag(h, k):
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k == 0:
        grad = 2 * h
    else:
        if k > 0:
            grad[:n - k] += h[k:]  # h[j+k] for j in 0..n-1-k
            grad[k:] += h[:n - k]  # h[j-k] for j in k..n-1
        else:
            k = -k
            grad[:n - k] += h[k:]  # same as positive lag
            grad[k:] += h[:n - k]
    return grad
```

def smart_transfer(h, grad, step, rng):
    n = len(h)
    # find candidates where we can transfer: h[i] > eps, h[j] < 1 - eps
    eps = 1e-12
    # get order of gradient values
    # we pick i with smallest grad among those with h[i] > some threshold
    # and j with largest grad among those with h[j] < 1 - threshold
    # To vectorize:
    mask_i = h > eps
    mask_j = h < 1 - eps
    if not np.any(mask_i) or not np.any(mask_j):
        return h, False
    # among mask_i, find argmin grad
    i_candidates = np.where(mask_i)[0]
    j_candidates = np.where(mask_j)[0]
    i = i_candidates[np.argmin(grad[i_candidates])]
    j = j_candidates[np.argmax(grad[j_candidates])]
    if grad[i] >= grad[j]:
        # no improvement direction
        return h, False
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h, False
    eps_transfer = min(step, max_eps)
    new_h = h.copy()
    new_h[i] -= eps_transfer
    new_h[j] += eps_transfer
    return new_h, True
```

    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    it = 0
    # initial temp perhaps based on cur_c
    T0 = cur_c * 0.1 + 1e-4
    while time.time() < deadline:

        # schedule
        T = T0 * (0.995 ** it)
        # generate candidate by random transfer (as before) or smart move with probability
        if rng.random() < 0.3:
            # smart move targeting worst lag
            k,_ = min_autocorr_shift(cur_h)
            grad = gradient_for_lag(cur_h, k)
            step = 0.2 * (0.999 ** it)

    return best_h, best_c

import time, numpy as np

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def worst_overlap_shift(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    return idx - (n - 1), corr[idx]

def gradient_worst_overlap(h, k):
    n = h.shape[0]
    g = np.zeros_like(h)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        upper = n - abs(k)
        if k > 0:
            # i in [0, upper-1], pair (i,i+k)
            g[:upper] += 1.0 - h[k:]          # derivative wrt h[i]
            g[k:]   -= h[:upper]               # derivative wrt h[i+k]
        else:
            # k negative, treat similarly using abs(k)
            kk = -k
            g[:upper] += 1.0 - h[kk:]          # derivative wrt h[i]
            g[kk:]   -= h[:upper]               # derivative wrt h[i+kk]
    return g

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def make_feasible(h, target, rng):

import time
import numpy as np

# --------------------------------------------------------------
# Core utilities
# --------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for step function h on [0,2] with uniform spacing."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def worst_overlap_shift(h: np.ndarray):
    """Shift k (relative to centre) that gives the maximal overlap."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return idx - (n - 1), corr[idx]


def gradient_worst_overlap(h: np.ndarray, k: int):
    """Gradient of the overlap for the worst shift (to be subtracted)."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        kk = abs(k)
        upper = n - kk
        if k > 0:
            # i from 0..upper-1 paired with i+kk
            g[:upper] += 1.0 - h[kk:]      # d/ d h[i]
            g[kk:]   -= h[:upper]          # d/ d h[i+kk]
        else:
            # k < 0  (use symmetry)
            g[:upper] += 1.0 - h[kk:]
            g[kk:]   -= h[:upper]
    return g


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2):
    """Transfer a small ε from i to j, preserving the total sum."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def make_feasible(h: np.ndarray, target: float, rng: np.random.Generator):
    """Adjust h in-place so that sum(h) == target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.shape[0]
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_descent(
    h: np.ndarray,
    rng: np.random.Generator,
    max_iters: int = 8000,
    step0: float = 0.4,
):

    n = h.shape[0]
    target = n / 2.0
    step = step0
    best_h = h.copy()
    best_c = compute_c5(h)

    for it in range(max_iters):
        k, _ = worst_overlap_shift(h)
        g = gradient_worst_overlap(h, k)

        pos = np.where(g > 1e-12)[0]
        neg = np.where(g < -1e-12)[0]

        if pos.size and neg.size:
            i = rng.choice(pos)
            j = rng.choice(neg)
            max_eps = min(h[i], 1.0 - h[j])
            eps = min(step, max_eps)
            new_h = h.copy()
            new_h[i] -= eps
            new_h[j] += eps
        else:
            # fallback to a random transfer
            new_h = perturb_h(h, rng, scale=step / target)

        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = make_feasible(new_h, target, rng)

        if c < best_c:

    return best_h, best_c

            h_a, c_a = gradient_descent(h, rng, max_iters=8000, step0=0.4)
            if c_a < best_c:
                best_h, best_c, best_n = h_a.copy(), c_a, n

            # Phase B – hill‑climb refinement
            h_b, c_b = hill_climb(h_a, rng, max_iters=3000, step0=0.5)
            if c_b < best_c:
                best_h, best_c, best_n = h_b.copy(), c_b, n

            # Phase C – simulated annealing with remaining time
            remaining = budget_s - (time.time() - start_time)
            if remaining > 2.0:
                deadline = time.time() + remaining * 0.8
                h_c, c_c = simulated_annealing(h_b, rng, deadline)
                if c_c < best_c:
                    best_h, best_c, best_n = h_c.copy(), c_c, n

            if time.time() - start_time > budget_s * 0.99:
                break
        if time.time() - start_time > budget_s * 0.99:
            break

    # ------------------------------------------------------------------
    # Final safety net
    # ------------------------------------------------------------------
    if best_h is None:
        best_n = 120
        best_h = np.full(best_n, 0.5)
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = make_feasible(best_h, best_n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
