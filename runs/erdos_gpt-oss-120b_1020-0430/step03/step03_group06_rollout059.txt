def max_shift_and_indices(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = corr.max()

def simulated_annealing(h_initial, rng, max_iters, T0=1.0, Tmin=0.001):
    h = h_initial.copy()
    n = len(h)
    m = n//2
    best_h = h.copy()
    # compute initial
    corr = np.correlate(h, 1 - h, mode='full')
    cur_max = int(corr.max())
    best_max = cur_max
    # schedule: temperature linear descent
    for it in range(1, max_iters+1):
        T = T0 + (Tmin - T0) * (it / max_iters)  # linearly from T0 to Tmin
        # compute max shift
        corr = np.correlate(h, 1 - h, mode='full')
        cur_max = int(corr.max())
        max_idx = int(corr.argmax())
        shift = max_idx - (n - 1)
        # if cur_max is already low target, break
        if cur_max <= target_O_max:
            # optional break early
            break
        # find swap
        swap = select_swap(h, shift, rng)
        if swap is None:
            # random swap
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
        else:
            i, j = swap
        # propose new h
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1
        # evaluate neighbor
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_max = int(corr_new.max())
        delta = new_max - cur_max
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            # accept
            h = h_new
            cur_max = new_max
            # update best if better
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        # maybe periodic restart if stuck for many iterations
    return best_h, best_max

def get_contributing_indices(h, shift):

    if shift >= 0:

        j = i + shift

        i = np.arange(-shift, n)
        j = i + shift

import time
import numpy as np

# ------------------------------------------------------------
# Core evaluation utilities
# ------------------------------------------------------------
def compute_c5(h):
    """Exact discrete C₅ for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def compute_O_max(h):
    """Maximum overlap O = max_k Σ_i h_i (1‑h_{i+k}) for binary h."""
    return int(np.max(np.correlate(h, 1 - h, mode="full")))


# ------------------------------------------------------------
# Simple binary constructions
# ------------------------------------------------------------
def random_binary_vector(n, m, rng):
    """Uniform random binary vector with exactly m ones."""
    h = np.zeros(n, dtype=np.int8)
    idx = rng.choice(n, size=m, replace=False)
    h[idx] = 1
    return h


# ------------------------------------------------------------
# Helpers for local moves
# ------------------------------------------------------------
def contributing_indices(h, shift):
    """Indices i where h[i]=1 and h[i+shift]=0 (valid i)."""
    n = h.shape[0]
    if shift >= 0:
        i = np.arange(0, n - shift)
        j = i + shift
    else:
        i = np.arange(-shift, n)
        j = i + shift
    mask = (h[i] == 1) & (h[j] == 0)
    return i[mask]


def select_swap(h, shift, rng):
    """Pick a (1→0,0→1) swap that is unlikely to increase O for *shift*."""
    n = h.shape[0]
    cand_i = contributing_indices(h, shift)
    if cand_i.size:
        i = rng.choice(cand_i)
    else:
        ones = np.where(h == 1)[0]
        if ones.size == 0:
            return None
        i = rng.choice(ones)

    zeros = np.where(h == 0)[0]
    if zeros.size == 0:
        return None

    # try a few zeros that would not create a new (1,0) pair for this shift
    for _ in range(5):
        j = rng.choice(zeros)
        jp = j + shift
        if 0 <= jp < n and h[jp] == 0:   # would create new bad pair
            continue
        break
    else:
        j = rng.choice(zeros)

    return int(i), int(j)


# ------------------------------------------------------------
# Main improvement routine (simulated annealing + greedy)
# ------------------------------------------------------------
def improve_binary(start_h, rng, max_iters, target_O=None):
    """Try to lower the maximum overlap O starting from *start_h*."""
    n = start_h.shape[0]
    best_h = start_h.copy()
    best_O = compute_O_max(best_h)

    cur_h = start_h.copy()
    cur_O = best_O

    for it in range(max_iters):
        if target_O is not None and best_O <= target_O:
            break

        # current max‑overlap shift
        corr = np.correlate(cur_h, 1 - cur_h, mode="full")
        cur_O = int(np.max(corr))
        shift = int(np.argmax(corr)) - (n - 1)

        # decide move type
        if rng.random() < 0.1:          # occasional larger move (2‑swap)
            ones = np.where(cur_h == 1)[0]
            zeros = np.where(cur_h == 0)[0]
            if len(ones) >= 2 and len(zeros) >= 2:
                i1, i2 = rng.choice(ones, size=2, replace=False)
                j1, j2 = rng.choice(zeros, size=2, replace=False)
                new_h = cur_h.copy()
                new_h[i1] = new_h[i2] = 0
                new_h[j1] = new_h[j2] = 1
            else:
                continue
        else:
            sw = select_swap(cur_h, shift, rng)
            if sw is None:
                continue
            i, j = sw
            new_h = cur_h.copy()
            new_h[i] = 0
            new_h[j] = 1

        new_O = compute_O_max(new_h)
        delta = new_O - cur_O

        # linear cooling schedule
        T = 1.0 * (1.0 - it / max_iters) + 0.001
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            cur_h = new_h
            cur_O = new_O
            if new_O < best_O:
                best_O = new_O
                best_h = new_h.copy()

    return best_h, best_O

        n_fine = h_fine.size
        m_fine = n_fine // 2
        remaining = budget_s - (time.time() - start_time)
        iters_fine = 12000 if remaining > 6 else max(2000, int(remaining * 2500))
        h_fine_opt, O_fine_opt = improve_binary(h_fine, rng, max_iters=iters_fine)
        c_fine = O_fine_opt * (2.0 / n_fine)
        if c_fine < best_c:
            best_c = c_fine
            best_h = h_fine_opt.copy()
            n = n_fine

    # --------------------------------------------------------
    # Final clean‑up and return
    # --------------------------------------------------------
    if best_h is None:
        # fallback – uniform half‑ones (should never happen)
        best_h = np.zeros(n, dtype=np.float64)
        best_h[:m] = 1.0

    best_h = np.clip(best_h, 0.0, 1.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
