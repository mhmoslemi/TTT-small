def project_box_sum(x, target, max_iter=100):
    # using simple method with while loops: adjust diff distributing equally among free.
    # each iteration updates x[free] += diff / len(free); then clip again; if any become bound, they become fixed.
    x = np.clip(x, 0.0, 1.0)
    free = np.full(x.shape, True, dtype=bool)
    for _ in range(max_iter):
        s = x.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # distribute diff among free entries
        nfree = free.sum()
        if nfree == 0:
            # can't distribute -> break
            break
        delta = diff / nfree
        x[free] += delta
        # after adding delta, some may violate bounds; fix them, update free mask
        # Clip again and adjust free mask
        new_free = (x > 0) & (x < 1)
        # if no change in free set, we may need to adjust step size. we simply clip.
        x = np.clip(x, 0.0, 1.0)
        # recompute free
        free = new_free
    # final correction: enforce exact sum by scaling residual over all free (or all) because rounding errors could remain.
    # Use simple scaling of all values (including limit?) unrealistic if some at bounds.
    # We'll do a final small adjustment:
    diff = target - x.sum()

def corr_linear_fft(h):
    n = h.size
    nfft = 1 << (2 * n - 1).bit_length()  # next power of two >= 2n-1 (maybe 2* n)

def corr_linear_fft(h):
    n = h.shape[0]
    nfft = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, nfft)
    G = np.fft.rfft(1.0 - h, nfft)
    conv = np.fft.irfft(H * G, nfft)

def gradient_softmax(h, beta):
    n = h.shape[0]
    corr = corr_linear_fft(h)
    z = beta * corr
    z_max = np.max(z)
    w = np.exp(z - z_max)
    w_sum = np.sum(w)
    w = w / w_sum
    # compute convs
    # find nfft for conv of w and h (linear convolution)
    nfft = 1 << ((3*n - 2).bit_length())
    H = np.fft.rfft(h, nfft)
    W = np.fft.rfft(w, nfft)
    conv1 = np.fft.irfft(W * H, nfft)
    # backward
    conv_backward = conv1[(n-1):(n-1)+n]
    w_rev = w[::-1]
    W_rev = np.fft.rfft(w_rev, nfft)
    conv2 = np.fft.irfft(W_rev * H, nfft)
    conv_forward = conv2[(n-1):(n-1)+n]
    grad = 1.0 - (conv_backward + conv_forward)
    return grad

def upsample_h(h_low, n_target):
    n_low = h_low.size
    if n_low == n_target:
        return h_low.copy()
    # Use linear interpolation on midpoints.

def enforce_sum(h, target, rng):
    # Clip to bounds
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # Use water-filling style
    max_iter = 10000
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i])
            # add random proportion to avoid deterministic
            add = add * rng.random()
            h[i] += add
            diff -= add
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:

import time
import numpy as np
from scipy.optimize import minimize

# ---------- projection / sum enforcement ----------
def enforce_sum(x, target, rng):
    """Clip to [0,1] and adjust to make the sum exactly target."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    max_iter = 1000
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0:  # need to increase some entries
            idx = np.where(x < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - x[i]) * rng.random()
            x[i] += add
            diff -= add
        else:        # need to decrease some entries
            idx = np.where(x > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, x[i]) * rng.random()
            x[i] -= sub
            diff += sub
    # final tiny correction if needed
    if abs(diff) > 1e-12:
        i = rng.integers(0, x.size)
        x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return np.clip(x, 0.0, 1.0)


# ---------- correlation utilities ----------
def corr_linear_fft(h):
    """Linear convolution of h with (1‑h) using FFT."""
    n = h.size
    nfft = 1 << ((2 * n - 1).bit_length())   # next power of two ≥ 2n‑1
    H = np.fft.rfft(h, nfft)
    G = np.fft.rfft(1.0 - h, nfft)
    conv = np.fft.irfft(H * G, nfft)
    return conv[: 2 * n - 1]


def compute_c5_fft(h):
    """Discrete C₅ for a step vector."""
    n = h.size
    corr = corr_linear_fft(h)
    return np.max(corr) * (2.0 / n)


# ---------- soft‑max gradient ----------
def gradient_softmax(h, beta):
    """Gradient of (1/β)·log‑sum‑exp(β·corr) w.r.t. h."""
    n = h.size
    corr = corr_linear_fft(h)
    z = beta * corr
    z_max = np.max(z)
    w = np.exp(z - z_max)
    w /= w.sum()                     # w[k] corresponds to shift k‑(n‑1)

    # convolution of w with h (linear) → needed for both forward/backward terms
    nfft = 1 << ((3 * n - 2).bit_length())   # ≥ 3n‑2
    H = np.fft.rfft(h, nfft)
    W = np.fft.rfft(w, nfft)
    conv1 = np.fft.irfft(W * H, nfft)  # w * h
    # backward part: Σ_k w_k·h[i‑k]
    conv_backward = conv1[(n - 1) : (n - 1) + n]

    w_rev = w[::-1]
    W_rev = np.fft.rfft(w_rev, nfft)
    conv2 = np.fft.irfft(W_rev * H, nfft)  # w_rev * h
    # forward part: Σ_k w_k·h[i+k]
    conv_forward = conv2[(n - 1) : (n - 1) + n]

    grad = 1.0 - (conv_backward + conv_forward)
    return grad


# ---------- subgradient (max‑over‑k) ----------
def subgradient_step(h, step, target, rng):
    n = h.size
    corr = corr_linear_fft(h)
    idx = np.argmax(corr)
    k = idx - (n - 1)

    grad = np.ones(n)
    if k >= 0:
        grad[: n - k] -= h[k:]
        grad[k:]      -= h[: n - k]
    else:
        kk = -k
        grad[: n - kk] -= h[kk:]
        grad[kk:]      -= h[: n - kk]

    y = h - step * grad
    return enforce_sum(y, target, rng)


# ---------- small random perturb ----------
def perturb_random(h, rng, scale=0.1):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    y = h.copy()
    y[i] -= eps
    y[j] += eps
    return y


# ---------- up‑sampling ----------
def upsample_h(h_low, n_target, rng):
    n_low = h_low.size
    if n_low == n_target:
        return h_low.copy()
    xs_low = (np.arange(n_low) + 0.5) * (2.0 / n_low)
    xs_target = (np.arange(n_target) + 0.5) * (2.0 / n_target)
    h_up = np.interp(xs_target, xs_low, h_low)
    return enforce_sum(h_up, n_target / 2.0, rng)


# ---------- SLSQP refinement ----------
def slsqp_refine(h0, beta, max_iter=400):
    n = h0.size
    target = n / 2.0

    def sum_eq(x):
        return np.sum(x) - target

    def soft_obj(x):
        corr = corr_linear_fft(x)
        z = beta * corr
        m = np.max(z)
        return (m + np.log(np.sum(np.exp(z - m)))) / beta

    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq}
    res = minimize(
        soft_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return enforce_sum(x, target, np.random.default_rng())
