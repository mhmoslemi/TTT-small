        h0 = np.array(initial_h_values, dtype=float)

def upscale(h, n_new, rng):
    n_old = h.size
    x_old = (np.arange(n_old) + 0.5) * (2.0/n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0/n_new)
    h_new = np.interp(x_new, x_old, h)
    # enforce sum exact
    return enforce_sum(h_new, n_new/2., rng)

def enforce_sum(h, target, rng):
    """Adjust h so that Σh == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-14:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Random point in [0,1]^n with sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)

        h[j] += eps

    return enforce_sum(h, n / 2.0, rng)


def perturb_h(h, rng, scale=0.2):

def correlate_full(h):
    """Full linear correlation of h with (1‑h) via FFT."""
    n = h.shape[0]

    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv

    conv = correlate_full(h)
