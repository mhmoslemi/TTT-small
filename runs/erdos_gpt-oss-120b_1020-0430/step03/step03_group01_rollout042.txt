def compute_c5_fft(h):
    n = h.size
    a_rev = h[::-1]
    b = 1 - h
    N = 2*n - 1
    Nfft = 1 << (N-1).bit_length()  # next power of two
    A = np.fft.rfft(a_rev, Nfft)
    B = np.fft.rfft(b, Nfft)
    conv = np.fft.irfft(A * B, Nfft)[:N]  # length N
    return np.max(conv) * (2.0 / n)

def block_transfer(h, rng, scale):
    n = len(h)
    # random block length L between 1 and n//4?
    L = rng.integers(1, n//4 + 1)
    start_src = rng.integers(0, n)
    start_dst = rng.integers(0, n)
    # get indices modulo n
    src_idx = (np.arange(L) + start_src) % n
    dst_idx = (np.arange(L) + start_dst) % n
    # compute total mass src, max possible increase at dst limited by 1-x
    src_mass = h[src_idx].sum()
    dst_capacity = (1 - h[dst_idx]).sum()
    # we can move at most min(src_mass, dst_capacity)
    max_move = min(src_mass, dst_capacity)
    # move mass fraction = scale * max_move
    move = rng.random() * max_move * scale
    # Proportionally reduce src and increase dst without violating bound.
    if src_mass > 0:
        src_frac = move / src_mass
        h_new = h.copy()
        h_new[src_idx] -= h[src_idx] * src_frac
        if dst_capacity > 0:
            dst_frac = move / dst_capacity
            h_new[dst_idx] += (1 - h[dst_idx]) * dst_frac
        else:
            return h  # no capacity
        # due to rounding may go out of bounds tiny, clip and project
        h_new = np.clip(h_new, 0.0, 1.0)
        return h_new
    else:
        return h.copy()

def project_to_box_simplex(y, target):
    """Project y onto intersection of box [0,1] and hyperplane sum=target."""
    # Clip
    x = np.clip(y, 0.0, 1.0)
    # Iteratively adjust sum.
    for _ in range(10):
        s = x.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # Determine which indices can move in direction of diff.
        if diff > 0:
            # can increase where x < 1
            mask = x < 1 - 1e-12
        else:
            mask = x > 1e-12
        if np.any(mask):
            # equally distribute diff among mask
            adjust = diff / mask.sum()
            # but respect bounds
            inc = np.minimum(adjust, 1 - x) * mask
            dec = np.maximum(adjust, -x) * mask
            # Actually combine using where to handle sign
            # Use sign of diff
            if diff > 0:
                x[mask] = np.minimum(x[mask] + diff / mask.sum(), 1.0)
            else:
                x[mask] = np.maximum(x[mask] + diff / mask.sum(), 0.0)
        else:
            break
    # Final fix
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # distribute small diff naive
        for i in range(len(x)):
            if diff == 0:

def project_to_box_simplex(y, target):
    low = -1.0
    high = 1.0
    # Actually y in [0,1] after clipping, we can set low = -1.0, high = 1.0.
    for _ in range(50):
        tau = (low + high) / 2.0
        x = np.clip(y - tau, 0.0, 1.0)
        s = x.sum()
        if s > target:
            low = tau
        else:
            high = tau
    x = np.clip(y - (low + high)/2.0, 0.0, 1.0)
    # Ensure sum exactly target by adjusting tiny diff
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # adjust by distributing to non-bound elements
        # find indices where 0 < x < 1
        mask = (x > 1e-12) & (x < 1-1e-12)
        if mask.any():
            adjust = diff / mask.sum()
            x[mask] = np.clip(x[mask] + adjust, 0.0, 1.0)
        else:
            # fallback naive
            for i in range(len(x)):
                if diff == 0:
                    break
                if diff > 0 and x[i] < 1:
                    add = min(diff, 1 - x[i])
                    x[i] += add
                    diff -= add
                elif diff < 0 and x[i] > 0:
                    sub = min(-diff, x[i])
                    x[i] -= sub
                    diff += sub
    return x

def block_transfer(h, rng, scale):
    n = h.size
    max_len = max(1, n // 4)
    L = rng.integers(1, max_len + 1)
    src_start = rng.integers(0, n)
    dst_start = rng.integers(0, n)
    src_idx = (np.arange(L) + src_start) % n
    dst_idx = (np.arange(L) + dst_start) % n
    src_mass = h[src_idx].sum()
    dst_capacity = (1.0 - h[dst_idx]).sum()
    max_move = min(src_mass, dst_capacity)
    if max_move <= 0:
        return h
    move = rng.random() * max_move * scale
    if src_mass > 0:
        src_frac = move / src_mass
        # subtract proportionally
        h_new = h.copy()
        h_new[src_idx] -= h[src_idx] * src_frac
        # increase destination proportionally
        if dst_capacity > 0:
            dst_frac = move / dst_capacity
            h_new[dst_idx] += (1.0 - h[dst_idx]) * dst_frac
        else:
            return h
        # Clip to [0,1] and project to sum
        h_new = np.clip(h_new, 0.0, 1.0)
        # Ensure sum unchanged (should be)
        # Minor numeric diff adjust
        target = n / 2.0
        diff = target - h_new.sum()
        if abs(diff) > 1e-12:
            h_new = project_to_box_simplex(h_new, target)
        return h_new
    else:
        return h

        for _ in range(n_trials):
            # choose move type
            if rng.random() < 0.8:
                cand = random_transfer(h, rng, scale=step)
            else:
                cand = block_transfer(h, rng, scale=step)
            c = compute_c5_fft(cand)
            if c < best_c:
                best_c = c
                h = cand
                improved = True

        if not improved:
            # occasional acceptance of worse move (simulated annealing)
            if rng.random() < 0.005:
                cand = random_transfer(h, rng, scale=step * 2.0)
                c = compute_c5_fft(cand)
                # Accept if not too much worse (within 0.1% maybe)
                if c < best_c * 1.001:
                    best_c, h = c, cand
        # Update step
        step = max(step - decay_rate, min_step)
    return h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# Global placeholders for SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None


def project_to_box_simplex(y, target):
    """Project y onto {0≤x_i≤1, Σx_i=target} using bisection on the offset τ."""
    low, high = -1.0, 1.0  # τ bounds: at low → all ones, at high → all zeros
    for _ in range(50):
        tau = (low + high) * 0.5
        x = np.clip(y - tau, 0.0, 1.0)
        s = x.sum()
        if s > target:
            low = tau
        else:
            high = tau
    x = np.clip(y - (low + high) * 0.5, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        if mask.any():
            adj = diff / mask.sum()
            x[mask] = np.clip(x[mask] + adj, 0.0, 1.0)
        else:
            # tiny fallback
            for i in range(len(x)):
                if diff == 0.0:
                    break
                if diff > 0.0 and x[i] < 1.0:
                    add = min(diff, 1.0 - x[i])
                    x[i] += add
                    diff -= add
                elif diff < 0.0 and x[i] > 0.0:
                    sub = min(-diff, x[i])
                    x[i] -= sub
                    diff += sub
    return x


def random_feasible_vector(n, rng):
    """Uniform random point on the capped simplex."""
    y = rng.random(n)
    return project_to_box_simplex(y, n / 2.0)


def compute_corr_fft(h):
    """Linear correlation of h with (1‑h) without the dx factor (full length)."""
    n = h.shape[0]
    a_rev = h[::-1]
    b = 1.0 - h
    N = 2 * n - 1
    Nfft = 1 << ((N - 1).bit_length())
    A = np.fft.rfft(a_rev, Nfft)
    B = np.fft.rfft(b, Nfft)
    conv = np.fft.irfft(A * B, Nfft)[:N]
    return conv


def compute_c5_fft(h):
    """Exact discrete C₅ for step function h using FFT."""
    n = h.size
    corr = compute_corr_fft(h)
    return np.max(corr) * (2.0 / n)


def sum_eq_constraint(x):
    """Equality constraint Σx_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth log‑sum‑exp surrogate of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = compute_corr_fft(x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = project_to_box_simplex(x, CURRENT_TARGET)
    return x


def random_transfer(h, rng, scale):
    """Transfer a fraction of mass between two coordinates (preserves sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    if rng.random() < 0.5:
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            return h.copy()
        eps = rng.random() * max_eps * scale
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
    else:
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0.0:
            return h.copy()
        eps = rng.random() * max_eps * scale
        new_h = h.copy()
        new_h[i] += eps
        new_h[j] -= eps
    return new_h


def block_transfer(h, rng, scale):
    """Move mass from a random source block to a random destination block."""
    n = h.size
    max_len = max(1, n // 4)
    L = rng.integers(1, max_len + 1)
    src_start = rng.integers(0, n)
    dst_start = rng.integers(0, n)
    src_idx = (np.arange(L) + src_start) % n
    dst_idx = (np.arange(L) + dst_start) % n
    src_mass = h[src_idx].sum()
    dst_cap = (1.0 - h[dst_idx]).sum()
    max_move = min(src_mass, dst_cap)
    if max_move <= 0.0:
        return h.copy()
    move = rng.random() * max_move * scale
    if src_mass > 0.0:
        src_frac = move / src_mass
        new_h = h.copy()
        new_h[src_idx] -= h[src_idx] * src_frac
        if dst_cap > 0.0:
            dst_frac = move / dst_cap
            new_h[dst_idx] += (1.0 - h[dst_idx]) * dst_frac
        new_h = np.clip(new_h, 0.0, 1.0)
        # tiny correction to keep exact sum
        target = n / 2.0
        diff = target - new_h.sum()
        if abs(diff) > 1e-12:
            new_h = project_to_box_simplex(new_h, target)
        return new_h
    return h.copy()


def hill_climb(start_h, rng, max_iters=2000, init_step=0.5, n_trials=5):
    """Simple hill‑climb with occasional block moves."""
    h = start_h.copy()
    best_c = compute_c5_fft(h)
    step = init_step
    min_step = 0.01
    decay = (init_step - min_step) / max_iters
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            cand = random_transfer(h, rng, scale=step) if rng.random() < 0.8 else block_transfer(h, rng, scale=step)
            c = compute_c5_fft(cand)
            if c < best_c:
                best_c, h = c, cand
                improved = True
                break
        if not improved:
            # occasional weak acceptance (simulated annealing style)
            if rng.random() < 0.005:
                cand = random_transfer(h, rng, scale=step * 2.0)
                c = compute_c5_fft(cand)
                if c < best_c * 1.001:
                    best_c, h = c, cand
        step = max(step - decay, min_step)
    return h, best_c


def upsample_h(h, new_n):
    """Linear interpolation to a finer grid and projection onto feasibility."""
    old_n = h.size
    old_x = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    new_x = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    h_new = np.interp(new_x, old_x, h)
    return project_to_box_simplex(h_new, new_n / 2.0)
