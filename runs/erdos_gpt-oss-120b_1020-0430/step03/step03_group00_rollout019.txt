def worst_shift_and_gradient(h):
    n=len(h)
    corr = np.correlate(h, 1-h, mode='full')
    idx = np.argmax(corr)
    max_corr = corr[idx]
    k = idx - (n-1)
    # compute gradient
    grad = np.ones(n)
    if k>=0:
        # overlapping region length = n-k
        grad[:n-k] -= h[k:]
        grad[k:] -= h[:n-k]
    else:
        kk = -k
        grad[:n-kk] -= h[kk:]
        grad[kk:] -= h[:n-kk]
    return k, max_corr, grad

def steepest_transfer(h, grad, rng, step_factor=1.0):
    # positive gradient indices with h>0
    pos = np.where((grad>0) & (h>1e-12))[0]

    h = h_start.copy()

    step_factor = init_step_factor

def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    from scipy.optimize import minimize

    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Determine target n
    target_n = 960  # choose 960 points

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        # Interpolate to target_n
        xs0 = (np.arange(n0) + 0.5) * (2.0 / n0)
        xs = (np.arange(target_n) + 0.5) * (2.0 / target_n)
        h = np.interp(xs, xs0, h0)
        # Clip and enforce sum
        h = np.clip(h, 0.0, 1.0)
        # enforce sum exactly target_n/2
        h = enforce_sum(h, target_n / 2.0, rng)
    else:
        # Random start
        h = random_feasible_vector(target_n, rng)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # Hill climbing phase
    while time.time() - start_time < budget_s * 0.6:
        # Choose start: sometimes perturb best, sometimes random
        if rng.random() < 0.5:
            start = perturb_random(best_h, rng, scale=0.2)
        else:

    # SLSQP refinement
    beta_schedule = [10.0, 50.0, 200.0, 800.0, 3200.0, 12800.0]
    for beta in beta_schedule:
        if time.time() - start_time > budget_s * 0.85:
            break

        except Exception:
            continue
        c_ref = compute_c5(h_ref)

            best_h, best_c = h_ref.copy(), c_ref

    # Final micro-search: targeted steepest moves

        # Attempt steepest move on worst shift
        k, max_corr, grad = worst_shift_and_gradient(best_h)
        pos_idx = np.where((grad > 0) & (best_h > 1e-12))[0]
        neg_idx = np.where((grad < 0) & (best_h < 1-1e-12))[0]
        if pos_idx.size and neg_idx.size:

import time

# ---------- utilities ----------
def enforce_sum(h, target, rng):
    """Clip h to [0,1] and adjust to make sum exactly target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i])
            add *= rng.random()
            h[i] += add
            diff -= add
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i])
            sub *= rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        if diff > 0:
            h[i] = min(1.0, h[i] + diff)
        else:
            h[i] = max(0.0, h[i] + diff)
    return h

def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum exactly n/2."""
    h = rng.random(n)
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, n / 2.0, rng)
    return h

def perturb_random(h, rng, scale=0.2):
    """Transfer a small amount from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def compute_c5(h):
    """Discrete C₅ = max correlation * (2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def worst_shift_and_gradient(h):
    """
    Return (k, max_corr, grad) where k is the shift (index offset from centre)
    giving the maximal overlap, max_corr the raw correlation value,
    and grad the gradient of that correlation w.r.t. each h[i].
    """
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr)
    max_corr = corr[idx]
    k = idx - (n - 1)
    grad = np.ones(n)
    if k >= 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    else:
        kk = -k
        grad[: n - kk] -= h[kk:]
        grad[kk:] -= h[: n - kk]
    return k, max_corr, grad

def hill_climb_targeted(h_start, rng, max_iters=2000, init_step=1.0):
    """
    Focus on the current worst shift and move mass from high‑gradient
    to low‑gradient coordinates.  Attempts a few candidate pairs each
    iteration and adapts the step size.
    """
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    step = init_step
    top_k = 3  # how many best gradient indices to try

    for _ in range(max_iters):
        _, _, grad = worst_shift_and_gradient(h)
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1 - 1e-12))[0]

        if pos.size and neg.size:
            # pick top‑k candidates
            pos_sel = pos[np.argsort(-grad[pos])[:top_k]]
            neg_sel = neg[np.argsort(grad[neg])[:top_k]]
            best_local_c = cur_c
            best_local_h = None

            for i in pos_sel:
                for j in neg_sel:
                    max_eps = min(h[i], 1.0 - h[j])
                    if max_eps <= 0.0:
                        continue
                    eps = max_eps * step
                    new_h = h.copy()
                    new_h[i] -= eps
                    new_h[j] += eps
                    c_new = compute_c5(new_h)
                    if c_new < best_local_c:
                        best_local_c = c_new
                        best_local_h = new_h

            if best_local_h is not None:
                new_h, new_c = best_local_h, best_local_c
            else:
                # no improving pair – make a small random transfer
                new_h = perturb_random(h, rng, scale=step * 0.5)
                new_c = compute_c5(new_h)
        else:
            # fall back to a random move
            new_h = perturb_random(h, rng, scale=step * 0.5)
            new_c = compute_c5(new_h)

        # record best ever
        if new_c < best_c:
            best_c, best_h = new_c, new_h.copy()

        # accept if improves current state
        if new_c < cur_c:
            h, cur_c = new_h, new_c
            step = min(step * 1.02, 1.0)   # slowly raise step after success
        else:
            step = max(step * 0.70, 1e-4)  # shrink step after failure

    return best_h, best_c

    beta_schedule = [10.0, 100.0, 800.0, 3200.0]
    for beta in beta_schedule:
        if time.time() - start_time > budget_s * 0.85:
            break
        try:
            h_ref = slsqp_optimize(best_h, beta, max_iter=300)
        except Exception:
            continue
        c_ref = compute_c5(h_ref)
        if c_ref < best_c:
            best_h, best_c = h_ref.copy(), c_ref

    # -------------------------------------------------
    # Phase 3 – final micro‑search (targeted steepest moves)
    # -------------------------------------------------
    while time.time() - start_time < budget_s * 0.98:
        k, _, grad = worst_shift_and_gradient(best_h)
        pos = np.where((grad > 1e-12) & (best_h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (best_h < 1 - 1e-12))[0]
        if pos.size and neg.size:
            i = pos[np.argmax(grad[pos])]
            j = neg[np.argmin(grad[neg])]
            max_eps = min(best_h[i], 1.0 - best_h[j])
            eps = max_eps * 0.5
            new_h = best_h.copy()
            new_h[i] -= eps
            new_h[j] += eps
            c_new = compute_c5(new_h)
            if c_new < best_c:
                best_h, best_c = new_h, c_new
                continue
        # fallback tiny random tweak
        new_h = perturb_random(best_h, rng, scale=0.02)
        c_new = compute_c5(new_h)
        if c_new < best_c:
            best_h, best_c = new_h, c_new

    # -------------------------------------------------
    # Clean‑up and final evaluation
    # -------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
