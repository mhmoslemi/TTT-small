def binary_hill_climb(h, max_iters, rng, time_limit):
    n = h.size
    m = int(round(n/2))
    # Ensure binary with exactly m ones
    # Use top m entries.
    idx_sorted = np.argsort(-h)
    h_bin = np.zeros_like(h, dtype=float)
    h_bin[idx_sorted[:m]] = 1.0
    best_c = compute_c5(h_bin)
    c_current = best_c
    T0 = 0.1
    for it in range(max_iters):
        if time.time() > time_limit:
            break
        # linear cooling
        T = T0 * (1 - it / max_iters)
        # random swap
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h_bin[i] = 0; h_bin[j] = 1
        c_new = compute_c5(h_bin)
        delta = c_new - c_current
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            # accept
            c_current = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_bin.copy()
        else:
            # revert
            h_bin[i] = 1; h_bin[j] = 0
    # return best found
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def compute_c5(h):
    """Exact discrete C5 for a step function h on [0,2] with dx = 2/len(h)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng):
    """Adjust h so that sum(h) = target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    # simple random walk adjustment
    for _ in range(10000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-12:
            continue
        h[i] += add
        diff = target - h.sum()
    # final tiny fix (deterministic)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # distribute remaining mass greedily
        if diff > 0:
            caps = 1.0 - h
            while diff > 1e-12 and caps.max() > 1e-12:
                i = np.argmax(caps)
                inc = min(diff, caps[i])
                h[i] += inc
                diff -= inc
                caps[i] = 1.0 - h[i]
        else:
            caps = h.copy()
            while diff < -1e-12 and caps.max() > 1e-12:
                i = np.argmax(caps)
                dec = min(-diff, caps[i])
                h[i] -= dec
                diff += dec
                caps[i] = h[i]
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    v = rng.exponential(1.0, size=n)
    v = v / v.sum() * (n / 2.0)
    h = np.clip(v, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def perturb_h(h, rng, scale=0.5):
    """Transfer a small amount between two coordinates while preserving the sum."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if j >= i:
        j += 1
    max_pos = min(h[i], 1.0 - h[j])
    max_neg = min(h[j], 1.0 - h[i])
    if max_pos + max_neg == 0:
        return h.copy()
    eps = (rng.random() * (max_pos + max_neg)) - max_neg
    eps *= scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return np.clip(nh, 0.0, 1.0)


# ----------------------------------------------------------------------
# Soft‑max surrogate for the max‑overlap objective (used by SLSQP)
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=800):
    """Run SLSQP on the smooth surrogate for a fixed β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny exact‑sum repair
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            if abs(add) < 1e-12:
                continue
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ----------------------------------------------------------------------
# Binary hill‑climb / simulated‑annealing
# ----------------------------------------------------------------------
def binary_hill_climb(h_cont, max_iters, rng, time_limit):
    """Start from a continuous vector, snap to binary with half ones,
       then perform simulated‑annealing hill‑climb on swaps."""
    n = h_cont.size
    m = int(round(n / 2))
    # snap to binary by keeping the m largest entries
    idx = np.argsort(-h_cont)[:m]
    h = np.zeros(n, dtype=float)
    h[idx] = 1.0
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    T0 = 0.08
    for it in range(max_iters):
        if time.time() > time_limit:
            break
        T = T0 * (1.0 - it / max_iters)
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        c_new = compute_c5(h)
        delta = c_new - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c
