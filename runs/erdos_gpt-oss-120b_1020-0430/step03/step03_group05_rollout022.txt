def binary_hill_climb(h, rng, max_time):
    n = len(h)
    best = h.copy()
    best_score = compute_c5(best)
    start = time.time()
    while time.time() - start < max_time:
        ones = np.where(best == 1)[0]
        zeros = np.where(best == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        # number of tries
        tries = min(100, len(ones)*len(zeros))
        # We'll sample try pairs randomly
        best_pair = None
        best_pair_score = best_score
        # We'll copy best array locally for each trial? We can compute by swapping,
        # but repeated compute_c5 on swapped arrays each time. We'll do inside loop.
        for _ in range(tries):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            # swap
            best[i] = 0
            best[j] = 1
            s = compute_c5(best)
            if s < best_pair_score:
                best_pair_score = s
                best_pair = (i,j)
            # revert
            best[i] = 1
            best[j] = 0
        # after trying, if improvement found, apply best pair
        if best_pair is not None:
            i,j = best_pair
            best[i] = 0
            best[j] = 1
            best_score = best_pair_score
        else:
            # no improvement after many tries; break (might be local optimum)
            break
    return best, best_score

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Helper functions ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth max using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Discrete C5 for step function h on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_to_feasible(v, target):
    """Project v onto {0≤h≤1, Σh = target} via bisection on a shift."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Uniform random vector with 0≤h≤1 and Σh = n/2."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_to_feasible(v, n / 2.0)


def random_binary_vector(n, rng):
    """Binary {0,1} vector with exactly n/2 ones."""
    half = n // 2
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=half, replace=False)
    h[idx] = 1.0
    return h


def slsqp_optimize(h0, beta, rng, max_iter=800):
    """SLSQP on the smooth max objective."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def de_optimize_continuous(n, rng, max_time, init_vec=None):
    """Simple DE on the exact C5 objective."""
    target = n / 2.0
    popsize = max(50, min(200, n // 2))
    pop = []
    if init_vec is not None:
        iv = np.clip(init_vec, 0.0, 1.0)
        pop.append(project_to_feasible(iv, target))
    while len(pop) < popsize:
        pop.append(random_feasible_vector(n, rng))
    pop = np.array(pop)
    scores = np.array([compute_c5(ind) for ind in pop])
    best_idx = scores.argmin()
    best = pop[best_idx].copy()
    best_score = scores[best_idx]

def binary_hill_climb(h, rng, max_time):
