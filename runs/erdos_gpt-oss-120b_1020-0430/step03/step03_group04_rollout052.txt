        # generate random feasible start
        h = random_feasible_vector(n, rng)
        # local optimization pipeline:
        # Stage A: hill climb
        h, c = hill_climb(h, rng, max_iters=1200, start_step=0.7)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n
        # Stage B: gradient refine with more iters
        h = gradient_refine(h, rng, step=0.025, max_iter=20)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n
        # Stage C: SLSQP with a schedule
        for beta in (10, 50, 200, 500):
            if time.time() - start >= budget_s * 0.96:
                break
            h = slsqp_optimize(h, beta, rng, max_iter=400)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n
        # Stage D: additional hill climb fine
        h, c = hill_climb(h, rng, max_iters=800, start_step=0.4)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n
        # Stage E: simulated annealing for some time
        deadline_sa = start + budget_s * 0.98
        if time.time() < deadline_sa:
            h_sa, c_sa = simulated_annealing(h, rng, deadline_sa)
            if c_sa < best_c:
                best_h, best_c, best_n = h_sa.copy(), c_sa, n
                h = h_sa
        # final hill climb short
        h, c = hill_climb(h, rng, max_iters=400, start_step=0.3)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n
    # after loop, final refinement on best solution with upsampling:
    if best_h is not None and time.time() - start < budget_s * 0.99:
        # attempt upsample by factor 2 if not too large
        if best_n * 2 <= 400:
            n_up = best_n * 2
            # upsample via interpolation
            x_src = np.linspace(0,2,best_n,endpoint=False)
            x_tgt = np.linspace(0,2,n_up,endpoint=False)
            h_up = np.interp(x_tgt, x_src, best_h)
            # enforce sum
            target_up = n_up/2
            diff = target_up - h_up.sum()
            if abs(diff)>1e-12:
                for _ in range(2000):
                    i = rng.integers(0,n_up)
                    add = min(diff,1.0-h_up[i]) if diff>0 else max(diff,-h_up[i])
                    h_up[i] += add
                    diff = target_up - h_up.sum()
                    if abs(diff) < 1e-12:
                        break
            # refine upsampled
            h_up, c_up = hill_climb(h_up, rng, max_iters=2000, start_step=0.6)
            for beta in (20,200,800):
                if time.time() - start >= budget_s*0.995:
                    break
                h_up = slsqp_optimize(h_up,beta,rng,max_iter=400)
            h_up = gradient_refine(h_up,rng,step=0.02,max_iter=15)
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, best_n = h_up, c_up, n_up

    # ensure feasibility
    if best_h is None:
        best_n = 120
        best_h = np.full(best_n,0.5)
    best_h = np.clip(best_h,0,1)
    target= best_n/2
    diff = target - best_h.sum()

        for _ in range(5000):
            i = rng.integers(0,best_n)
            add = min(diff,1.0-best_h[i]) if diff>0 else max(diff,-best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()
            if abs(diff) <1e-12:
                break
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Global variables for SLSQP surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None

# ---------- Utility functions ----------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for step function h on [0,2] with uniform spacing."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start: np.ndarray, rng: np.random.Generator,
               max_iters: int = 2000, start_step: float = 0.5) -> tuple[np.ndarray, float]:
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.02, max_iter: int = 12) -> np.ndarray:
    """Refine by moving against gradient of the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector in [0,1] with exact sum n/2."""
    h = rng.random(n)
    target = n / 2.0
    scale = target / h.sum()
    h *= scale
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h


def sum_eq_constraint(x: np.ndarray) -> float:
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x: np.ndarray) -> float:
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 400) -> np.ndarray:
    """Run a single SLSQP optimisation with the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x


def simulated_annealing(h0: np.ndarray, rng: np.random.Generator,
                        deadline: float) -> tuple[np.ndarray, float]:
    """Simple SA that respects the sum constraint."""
    n = h0.size
    target = n / 2.0
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    T0 = max(0.01 * cur_c, 1e-4)
    it = 0
    while time.time() < deadline and it < 12000:
        step = max(0.5 * (0.999 ** it), 5e-4)
        cand_h = perturb_h(cur_h, rng, scale=step)
        cand_c = compute_c5(cand_h)
        accept = False
        if cand_c < cur_c:
            accept = True
        else:
            T = T0 * (0.999 ** it) + 1e-12
            if rng.random() < np.exp(-(cand_c - cur_c) / T):
                accept = True
        if accept:
            cur_h, cur_c = cand_h, cand_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
        it += 1
    return best_h, best_c
