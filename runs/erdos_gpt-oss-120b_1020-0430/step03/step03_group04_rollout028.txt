def run(...):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_h, best_c, best_n = None, np.inf, None

    # Determine n's
    candidate_ns = [72, 96, 108, 120, 144, 180, 240]
    # We'll do DE on n0 = candidate_ns[0] (72)
    n0 = candidate_ns[0]

    # Prepare Fourier basis
    p = 3  # number of harmonics
    x = np.linspace(0.0, 2.0, n0, endpoint=False)
    cos_mat = np.cos(np.pi * np.arange(1, p+1)[:, None] * x)  # shape (p,n0)
    sin_mat = np.sin(np.pi * np.arange(1, p+1)[:, None] * x)

    def de_objective(coeffs):
        # compute h
        a = coeffs[:p]
        b = coeffs[p:]
        h = 0.5 + a @ cos_mat + b @ sin_mat
        # clip
        h = np.clip(h, 0.0, 1.0)
        # enforce sum via simple scaling (not random) to reduce penalty
        target = n0/2.0
        sum_h = h.sum()
        diff = target - sum_h
        # apply simple proportional correction if possible
        if abs(diff) > 1e-12:
            # adjust using available slack
            # compute slack: where h<1 if diff>0; where h>0 if diff<0
            # We'll just add diff/n0 uniformly then clip again
            h += diff / n0
            h = np.clip(h, 0.0, 1.0)
            # recompute diff; if still not zero, do simple random redistribution
            sum_h = h.sum()
            diff = target - sum_h
            if abs(diff) > 1e-6:
                # fallback random adjustments
                for _ in range(200):
                    i = rng.integers(0, n0)
                    add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                    h[i] += add
                    diff = target - h.sum()
                    if abs(diff) < 1e-12:
                        break
        c = compute_c5(h)
        penalty = 10.0 * abs(target - h.sum())
        return c + penalty

    # run DE
    bounds = [(-0.8, 0.8)] * (2*p)
    result = differential_evolution(de_objective, bounds, seed=seed, maxiter=50, popsize=10, polish=False)
    # get best coefficients
    best_coeffs = result.x
    # generate start h0 for n0
    a = best_coeffs[:p]; b = best_coeffs[p:]
    h0 = 0.5 + a @ cos_mat + b @ sin_mat
    h0 = np.clip(h0,0.0,1.0)
    h0 = repair_h_sum(h0, target=n0/2.0, rng=rng)
    # Evaluate and possibly hill climb
    best_h_local, best_c_local = hill_climb(h0, rng, max_iters=2000, start_step=0.5)
    if best_c_local < best_c:
        best_h, best_c, best_n = best_h_local, best_c_local, n0
    # now for each other n in candidate_ns:
    for n in candidate_ns[1:]:
        if time.time() - start_time > budget_s*0.9:
            break
        # Upsample best_h to n by interpolation
        x_src = np.linspace(0.0,2.0,best_n,endpoint=False)
        x_tgt = np.linspace(0.0,2.0,n,endpoint=False)
        h_up = np.interp(x_tgt, x_src, best_h)
        # repair sum exactly for n
        h_up = repair_h_sum(h_up, target=n/2.0, rng=rng)
        # local refinement
        h_up_ref, c_up = hill_climb(h_up, rng, max_iters=1500, start_step=0.5)
        # optionally SLSQP fine tuning (like earlier)
        for beta in (5,20,80,200,500):
            if time.time() - start_time > budget_s*0.95:
                break
            h_up_ref = slsqp_optimize(h_up_ref, beta, rng, max_iter=250)
            c_up = compute_c5(h_up_ref)
            if c_up < best_c:
                best_h, best_c, best_n = h_up_ref.copy(), c_up, n
        # gradient refine
        h_up_ref = gradient_refine(h_up_ref, rng, step=0.02, max_iter=12)
        c_up = compute_c5(h_up_ref)
        if c_up < best_c:
            best_h, best_c, best_n = h_up_ref.copy(), c_up, n

    # final safety check/clip
    if best_h is None:
        best_n = 120
        best_h = np.full(best_n,0.5)

    best_h = np.clip(best_h, 0.0, 1.0)
    # enforce sum again
    best_h = repair_h_sum(best_h, target=best_n/2.0, rng=rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)

import time
import numpy as np
from scipy.optimize import minimize, differential_evolution

DE_SIN = None

# ---------- utility functions ----------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for step function h on [0,2] with uniform spacing."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def repair_h_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Redistribute mass to hit the exact sum while staying in [0,1]."""
    diff = target - h.sum()
    if abs(diff) <= 1e-12:
        return h
    for _ in range(2000):
        i = rng.integers(0, h.size)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
    return h


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start: np.ndarray, rng: np.random.Generator,
               max_iters: int = 1500, start_step: float = 0.5) -> tuple:
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.02, max_iter: int = 12) -> np.ndarray:
    """Refine by moving against gradient of the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)          # signed shift
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]      # + (1 - h[i+k])
            g[k:] -= h[:upper]            # - h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


# ---------- SLSQP with soft‑max surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 300) -> np.ndarray:
    """Run a single SLSQP optimisation with the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x

def de_objective(coeffs: np.ndarray) -> float:
    """Objective for DE (C5 with heavy penalty for sum violation)."""
    global DE_N, DE_COS, DE_SIN, DE_RNG, DE_TARGET
    p = DE_COS.shape[0]
    a = coeffs[:p]
    b = coeffs[p:]
    h = 0.5 + a @ DE_COS + b @ DE_SIN
    h = np.clip(h, 0.0, 1.0)
    h = repair_h_sum(h, DE_TARGET, DE_RNG)
    c = compute_c5(h)
    penalty = 10.0 * abs(DE_TARGET - h.sum())
    return c + penalty
