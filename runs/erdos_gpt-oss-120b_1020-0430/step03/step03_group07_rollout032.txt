def perturb_h_symmetric(h, rng, scale=0.2):
    n = h.size
    half = n // 2
    i = rng.integers(0, half)
    j = rng.integers(0, half)
    if i == j:
        return h.copy()
    # max epsilon limited by bounds: h[i] can increase up to 1, decrease up to 0; similar for h[j]
    # We will transfer epsilon from i to j (i loses, j gains). So ε in [-max_down_i, max_up_j] etc.
    max_eps_from_i = h[i]  # cannot go below 0
    max_eps_to_j = 1.0 - h[j]  # cannot exceed 1
    max_eps = min(max_eps_from_i, max_eps_to_j)
    if max_eps <= 0:
        return h.copy()
    # random ε up to max_eps * scale  (scale fraction)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    # enforce symmetry:
    new_h[n-1-i] = 1.0 - new_h[i]
    new_h[n-1-j] = 1.0 - new_h[j]
    return new_h

import time
import numpy as np

# --------------------------------------------------------------
# Core correlation utilities (FFT based, fast for many calls)
# --------------------------------------------------------------
def correlate_full(h: np.ndarray) -> np.ndarray:
    """Full linear correlation of h with (1‑h) using FFT."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())  # next power of two
    fh = np.fft.rfft(h, L)
    rev = (1.0 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return float(conv.max() * (2.0 / h.size))


# --------------------------------------------------------------
# Symmetry handling (h(x)+h(2−x)=1)
# --------------------------------------------------------------
def symmetrize(h: np.ndarray) -> np.ndarray:
    """Project h onto the subspace h[i]=v, h[-i‑1]=1‑v."""
    n = h.size
    half = n // 2
    for i in range(half):
        v = (h[i] + (1.0 - h[n - 1 - i])) * 0.5
        h[i] = v
        h[n - 1 - i] = 1.0 - v
    return h


def random_symmetric_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random symmetric feasible vector (sum = n/2, 0≤h≤1)."""
    half = n // 2
    v = rng.random(half)  # uniform in [0,1)
    h = np.empty(n, dtype=float)
    h[:half] = v
    h[half:] = 1.0 - v[::-1]
    return h


def perturb_h_symmetric(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small amount ε between two positions in the first half."""
    n = h.size
    half = n // 2
    i = rng.integers(0, half)
    j = rng.integers(0, half)
    if i == j:
        return h.copy()
    # transfer ε from i to j (i loses, j gains)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    # enforce symmetry
    new_h[n - 1 - i] = 1.0 - new_h[i]
    new_h[n - 1 - j] = 1.0 - new_h[j]
    return new_h


# --------------------------------------------------------------
# Gradient of the smooth (log‑sum‑exp) surrogate of the max
# --------------------------------------------------------------
def gradient_softmax(beta: float, h: np.ndarray):
    """Return gradient of (1/β)·log∑exp(β·corr) w.r.t. h."""
    n = h.size
    conv = correlate_full(h)                   # length 2n‑1
    bc = beta * conv
    max_bc = bc.max()
    exp_bc = np.exp(bc - max_bc)
    w = exp_bc / exp_bc.sum()                  # softmax weights, sum to 1

    # convolution of weights with (1‑h)
    A = np.convolve(w, 1.0 - h, mode='full')   # length 3n‑2
    grad = A[n - 1 : n - 1 + n] - w[n - 1] * h
    return grad, conv


def gradient_descent_step(h: np.ndarray, beta: float, step: float, rng: np.random.Generator) -> np.ndarray:
    """One projected gradient step on the smooth surrogate."""
    grad, _ = gradient_softmax(beta, h)
    # simple back‑tracking: we keep the step size given
    h_new = h - step * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = symmetrize(h_new)
    return h_new


# --------------------------------------------------------------
# Local refinement (greedy hill‑climb respecting symmetry)
# --------------------------------------------------------------
def hill_climb_symmetric(start: np.ndarray,
                         rng: np.random.Generator,
                         max_iters: int = 800,
                         start_step: float = 0.2):
    h = start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)

    for it in range(max_iters):
        new_h = perturb_h_symmetric(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.8, 1e-5)
    return best_h, best_c
