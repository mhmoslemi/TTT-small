def worst_gradient(h):
    n = h.shape[0]
    conv = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(conv)  # index in conv
    k = idx - (n - 1)   # shift (pos or neg)
    if k < 0:
        k = -k
    # compute gradient
    g = np.ones_like(h)
    if k == 0:
        g = 1.0 - 2.0 * h
    else:
        g[: n - k] -= h[k:]
        g[k:] -= h[: n - k]
    return g, k, conv
```

def guided_step(h, grad, rng, eps):
    # move from max_grad to min_grad
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    # ensure we can transfer epsilon
    transfer = min(eps, h[i], 1.0 - h[j])
    if transfer <= 0:
        return h
    nh = h.copy()
    nh[i] -= transfer
    nh[j] += transfer
    return nh
```

def slsqp_optimize(h0, beta, rng, max_iter=600):
    n = h0.size
    target = n/2.0
    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        # softmax on beta * corr
        bc = beta * corr
        m = np.max(bc)
        # compute softmax in numerically stable way
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    def eq_constraint(x):
        return np.sum(x) - target
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': eq_constraint}
    res = minimize(objective, h0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum exactly via redistribution
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # simple redistribution
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x

    diff = target - h.sum()

    for _ in range(2000):

        if s <= 0:
            break
        h = h - s * grad
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h = enforce_sum(h, target, rng)
    return h


def slsqp_optimize(h0, beta, rng, max_iter=600):
    """Soft‑max SLSQP refinement for a given β."""
    n = h0.size
    target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    def eqc(x):
        return np.sum(x) - target

    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": eqc}
    res = minimize(
        obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # exact sum correction (quick redistribution)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x


def binary_swap_hill_climb(h0, target, rng, max_iters):
    """Swap 1↔0 entries to improve max overlap (binary search)."""
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    n = h0.size
    for _ in range(max_iters):
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        nh = best_h.copy()
        nh[i], nh[j] = 0.0, 1.0
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h = c, nh
    return best_h, best_c


# --------------------------------------------------------------
# main optimisation driver
# --------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c = np.inf
    best_n = None
    TARGET_BOUND = 0.3808

    # ----- initialise from parent if present -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        n0 = h0.size
        h0 = enforce_sum(h0, n0 / 2.0, rng)
        best_h, best_c, best_n = h0.copy(), compute_c5(h0), n0

    # ----- candidate discretisations -----
    cand = [96, 108, 120, 144, 168, 192, 216, 240, 256]
    if best_n is not None:
        cand = [best_n] + [n for n in cand if n != best_n]

    # ----- main search over n -----
    for n in cand:
        if time.time() - start > 0.94 * budget_s:
            break
        target = n / 2.0
        inits = []

        # interpolation from best known
        if best_h is not None:
            if best_n != n:
                x_old = np.linspace(0, 2, best_n, endpoint=False)
                x_new = np.linspace(0, 2, n, endpoint=False)
                h_interp = np.interp(x_new, x_old, best_h)
                h_interp = np.clip(h_interp, 0.0, 1.0)
                h_interp = enforce_sum(h_interp, target, rng)
                inits.append(h_interp)
            else:
                inits.append(best_h.copy())

        # random binary
        if n % 2 == 0:
            h_bin = np.zeros(n)
            h_bin[rng.choice(n, n // 2, replace=False)] = 1.0
            inits.append(h_bin)

        # random fractional
        h_rand = rng.random(n)
        h_rand *= target / h_rand.sum()
        h_rand = np.clip(h_rand, 0.0, 1.0)
        h_rand = enforce_sum(h_rand, target, rng)
        inits.append(h_rand)

        # ----- process each initial vector -----
        for h0 in inits:
            if time.time() - start > 0.96 * budget_s:
                break

            # guided hill‑climb
            h, c = hill_climb_fractional(
                h0, target, rng, max_iters=8000, start_eps=0.5, decay=0.999
            )
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n
                if best_c < TARGET_BOUND:
                    break

            # increasing‑β SLSQP refinement
            for beta in (10.0, 50.0, 200.0, 800.0, 3000.0, 8000.0):
                if time.time() - start > 0.97 * budget_s or best_c < TARGET_BOUND:
                    break
                h = slsqp_optimize(h, beta, rng, max_iter=800 if beta < 800 else 1200)
                c = compute_c5(h)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n
                    if best_c < TARGET_BOUND:
                        break

            # fine gradient refinement
            if time.time() - start < 0.98 * budget_s and best_c >= TARGET_BOUND:
                h = gradient_refine(h, rng, step=0.01, max_iter=10)
                c = compute_c5(h)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n

            # short final hill‑climb
            if time.time() - start < 0.99 * budget_s and best_c >= TARGET_BOUND:
                h, c = hill_climb_fractional(
                    h, target, rng, max_iters=3000, start_eps=0.3, decay=0.995
                )
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n

            if best_c < TARGET_BOUND:
                break
        if best_c < TARGET_BOUND:
            break

    # ----- binary rounding + swap hill‑climb (if time permits) -----
    if best_h is not None:
        n = best_n
        target = n / 2.0
        h_bin = (best_h > 0.5).astype(float)

        # exact sum fix by flipping entries closest to 0.5
        cur = h_bin.sum()
        diff = int(target - cur)
        if diff != 0:
            dist = np.abs(best_h - 0.5)
            if diff > 0:
                cand = np.where(h_bin == 0)[0]
                idx = cand[np.argsort(dist[cand])[:diff]]
                h_bin[idx] = 1.0
            else:
                cand = np.where(h_bin == 1)[0]
                idx = cand[np.argsort(dist[cand])[: -diff]]
                h_bin[idx] = 0.0

        c_bin = compute_c5(h_bin)
        if c_bin < best_c:
            best_h, best_c = h_bin.copy(), c_bin

        # binary swap local search
        time_left = budget_s - (time.time() - start)
        if time_left > 0.0 and best_c >= TARGET_BOUND:
            max_swaps = int(min(200000, max(0, time_left * 2000 / n)))
            if max_swaps > 0:
                h_bin2, c_bin2 = binary_swap_hill_climb(best_h, target, rng, max_swaps)
                if c_bin2 < best_c:
                    best_h, best_c = h_bin2.copy(), c_bin2
                    best_n = n

    # ----- optional up‑sampling refinement (if time) -----
    if best_h is not None and time.time() - start < 0.999 * budget_s:
        if best_n * 2 <= 300:
            h_up = np.repeat(best_h, 2)
            n_up = h_up.size
            target_up = n_up / 2.0
            h_up, _ = hill_climb_fractional(
                h_up, target_up, rng, max_iters=4000, start_eps=0.3, decay=0.997
            )
            for beta in (20.0, 200.0, 2000.0):
                if time.time() - start > budget_s:
                    break
                h_up = slsqp_optimize(h_up, beta, rng, max_iter=600)
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, best_n = h_up.copy(), c_up, n_up

    # ----- final enforce and return -----
    if best_h is None:
        best_n = 96
        best_h = np.full(best_n, 0.5)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_n / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
