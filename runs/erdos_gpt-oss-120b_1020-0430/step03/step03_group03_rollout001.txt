    h = np.clip(h,0.0,1.0)
    # Compute current sum
    cur = h.sum()
    if abs(cur - target) < 1e-12:
        return h

    # Use iterative approach: while diff significant:
    #   Identify indices that are not at bounds (0<hi<1).
    #   If diff>0: we need to increase; allocate equally among free indices but limited by upper bound.
    #   If diff<0: decrease similarly.
    # This is water-filling.
    n = len(h)
    free = np.arange(n)
    # Derive loops
    diff = target - cur
    max_iter=100
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        free = np.where((h>0.0) & (h<1.0))[0]
        if free.size == 0:
            # No free variable left; adjust the first within bounds
            # fallback: add diff to h[0] clamped.

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # compute log sum exp (stable)
    # bc - m yields negatives.
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    bounds = [(0.0,1.0)] * n
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter,
                                              'ftol': 1e-12,'disp':False})
    x = np.clip(res.x,0.0,1.0)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

def perturb_two_indices(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    # sample epsilon within [-max_eps, max_eps]
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    h_new = h.copy()
    h_new[i] -= eps
    h_new[j] += eps
    return h_new

def tournament(pop, fitness, rng, tsize=5):
    indices = rng.integers(0,len(pop), size=tsize)
    best_idx = indices[0]
    best_f = fitness[best_idx]
    for idx in indices[1:]:
        f = fitness[idx]
        if f < best_f:
            best_f = f
            best_idx = idx
    return pop[best_idx]

    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:

            h[i] += add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()

def random_feasible_vector(n, rng):
    h = np.full(n, 0.5)
    # random swap-like adjustments
    for _ in range(n * 5):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    enforce_sum(h, n / 2.0, rng)
    return h

def perturb_two_indices(h, rng, scale=0.2):

    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:

    h_new = h.copy()
    h_new[i] -= eps
    h_new[j] += eps
    return h_new

def tournament(pop, fitness, rng, tsize=5):
    indices = rng.integers(0, len(pop), size=tsize)
    best = indices[0]
    best_f = fitness[best]
    for idx in indices[1:]:
        f = fitness[idx]
        if f < best_f:
            best_f = f
            best = idx
    return pop[best]

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x
