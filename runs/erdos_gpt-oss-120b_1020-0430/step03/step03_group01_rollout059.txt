def compute_grad(h, beta):
    O = np.correlate(h, 1 - h, mode='full') * (2.0 / h.size)   # O shape 2n-1
    # compute softmax weights
    max_O = O.max()
    exp_beta_O = np.exp(beta * (O - max_O))
    w = exp_beta_O / exp_beta_O.sum()
    # convolution
    n = h.size
    M = 2 * n - 1
    L = 1 << ((M + n - 1 - 1).bit_length())  # next power of two for length M+n-1
    
    # compute FFT of h padded
    H = np.fft.fft(h, L)
    W = np.fft.fft(w, L)  # note: w length M, padded to L
    conv = np.fft.ifft(H * W).real  # full conv length L (circular); but linear conv first M+n-1 entries are correct.

def compute_loss_and_grad(h, beta):
    n = h.size
    # compute O
    corr = np.correlate(h, 1 - h, mode='full')
    O = corr * (2.0 / n)
    # softmax
    max_O = O.max()
    exp_beta = np.exp(beta * (O - max_O))
    sum_exp = exp_beta.sum()
    loss = max_O + np.log(sum_exp) / beta
    w = exp_beta / sum_exp
    # compute term1 and term2 via FFT
    M = 2 * n - 1
    L = 1 << ((M + n - 1 - 1).bit_length())  # nearest power of two >= M + n -1 = 3n-2
    H = np.fft.fft(h, L)

import time
import numpy as np

# ---------- helper functions ----------

def project_box_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Projection of x onto the box [0,1]^n with exact sum = target."""
    lo, hi = -10.0, 10.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        p = np.clip(x - mid, 0.0, 1.0)
        s = p.sum()
        if s < target:
            hi = mid
        else:
            lo = mid
    return np.clip(x - lo, 0.0, 1.0)


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector projected onto the feasible set."""
    x = rng.random(n)
    return project_box_sum(x, n / 2.0)


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small amount ε from one index to another (preserves sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def compute_loss_and_grad(h: np.ndarray, beta: float):
    """Soft‑max surrogate loss and its gradient w.r.t. h."""
    n = h.size
    # overlap values O_k
    corr = np.correlate(h, 1.0 - h, mode="full")
    O = corr * (2.0 / n)

    # log‑sum‑exp loss
    max_O = O.max()
    exp_beta = np.exp(beta * (O - max_O))
    sum_exp = exp_beta.sum()
    loss = max_O + np.log(sum_exp) / beta
    w = exp_beta / sum_exp                     # soft‑max weights, length 2n‑1

    # convolution terms for the gradient
    M = 2 * n - 1
    L = 1 << ((M + n - 2).bit_length())        # next power of two ≥ 3n‑2
    H = np.fft.fft(h, L)
    W = np.fft.fft(w, L)
    conv = np.fft.ifft(H * W).real
    term1 = conv[(n - 1):(n - 1 + n)]

    w_rev = w[::-1]
    W_rev = np.fft.fft(w_rev, L)
    conv_rev = np.fft.ifft(H * W_rev).real
    term2 = conv_rev[(n - 1):(n - 1 + n)]

    grad = 1.0 - term1 - term2
    return loss, grad


def gradient_descent(h: np.ndarray, beta: float, max_iters: int, target: float, rng: np.random.Generator):
    """Simple projected gradient descent on the soft‑max surrogate."""
    step = 0.5
    loss, grad = compute_loss_and_grad(h, beta)
    for _ in range(max_iters):
        # backtracking line search
        for _ in range(15):
            new_h = project_box_sum(h - step * grad, target)
            new_loss, new_grad = compute_loss_and_grad(new_h, beta)
            if new_loss <= loss - 1e-9 * step:
                h = new_h
                loss = new_loss
                grad = new_grad
                step = min(step * 1.2, 1.0)
                break
            step *= 0.5
        else:
            break
    return h, loss


def hill_climb_simple(h_start: np.ndarray, rng: np.random.Generator, max_iters: int = 1500, step: float = 0.4):
    """Straightforward hill‑climb on the exact C₅."""
    h = h_start.copy()
    best_c = compute_c5(h)
    for _ in range(max_iters):
        cand = perturb_h(h, rng, scale=step)
        c = compute_c5(cand)
        if c < best_c:
            h = cand
            best_c = c
            step = min(step * 1.1, 0.7)
        else:
            step *= 0.98
    return h, best_c


# ---------- main driver ----------

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # initial dimension
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        h = np.array(initial_h_values, dtype=float)
        h = np.clip(h, 0.0, 1.0)
        h = project_box_sum(h, n / 2.0)
    else:
        n = 360
        h = random_feasible_vector(n, rng)

    target_sum = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(best_h)

    # schedule of β values (soft‑max temperature)
    beta_schedule = [5, 20, 80, 200, 500, 1000, 1500, 2000]

    # main optimisation loop
    while time.time() - start_time < budget_s * 0.78:
        # occasional random restart
        if rng.random() < 0.15:
            h = random_feasible_vector(n, rng)

        for beta in beta_schedule:
            if time.time() - start_time > budget_s * 0.78:
                break
            h, _ = gradient_descent(h, beta, max_iters=400, target=target_sum, rng=rng)
            c = compute_c5(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()

            # brief local hill‑climb to escape flat regions
            if rng.random() < 0.35:
                h_tmp, c_tmp = hill_climb_simple(best_h, rng, max_iters=800, step=0.35)
                if c_tmp < best_c:
                    best_c = c_tmp
                    best_h = h_tmp.copy()
                    h = h_tmp

        # perturb current best before next outer iteration
        h = perturb_h(best_h, rng, scale=0.25)

    # optional up‑sampling if time permits
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > budget_s * 0.12:
        target_n = min(720, max(360, n * 2))
        if target_n != n:
            old_x = (np.arange(n) + 0.5) * (2.0 / n)
            new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
            h_up = np.interp(new_x, old_x, best_h)
            h_up = project_box_sum(h_up, target_n / 2.0)

            for beta in beta_schedule:
                if time.time() - start_time > budget_s:
                    break
                h_up, _ = gradient_descent(h_up, beta, max_iters=300, target=target_n / 2.0, rng=rng)

            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, n = h_up, c_up, target_n

    # final clean‑up
    best_h = project_box_sum(best_h, n / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
