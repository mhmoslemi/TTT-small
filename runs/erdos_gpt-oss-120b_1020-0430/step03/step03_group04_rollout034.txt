import time
import numpy as np
from scipy.optimize import minimize

# ---------- utility functions ----------

def compute_c5(h):
    """Discrete C5 for step function h on [0,2] (uniform spacing)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def fix_sum(h, target, rng, max_iter=2000):
    """Adjust h (in [0,1]) so that sum(h)=target while staying feasible."""
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, h.size)
        if diff > 0.0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add == 0.0:
            continue
        h[i] += add
        diff = target - h.sum()
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount between two coordinates (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        cand = perturb_h(h, rng, scale=step)
        c = compute_c5(cand)
        if c < best_c:
            best_c = c
            best_h = cand.copy()
            h = cand
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


def gradient_refine(h, rng, step=0.02, max_iter=20):
    """Iterative gradient descent on the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            up = n - k
            g[:up] += 1.0 - h[k:]          # + (1 - h[i+k])
            g[k:] -= h[:up]                # - h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        h = fix_sum(h, target, rng)
    return h


def random_feasible_vector(n, rng):
    """Uniform random vector in [0,1] with exact sum n/2."""
    h = rng.random(n)
    target = n / 2.0
    scale = target / h.sum()
    h *= scale
    h = np.clip(h, 0.0, 1.0)
    h = fix_sum(h, target, rng)
    return h


# ---------- SLSQP with smooth max ----------

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = fix_sum(x, CURRENT_TARGET, rng)
    return x


# ---------- Simulated annealing ----------

def simulated_annealing(h0, rng, deadline):
    n = h0.size
    target = n / 2.0
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    T0 = max(0.01 * cur_c, 1e-4)
    it = 0
    while time.time() < deadline and it < 8000:
        step = max(0.5 * (0.999 ** it), 5e-4)
        cand_h = perturb_h(cur_h, rng, scale=step)
        cand_c = compute_c5(cand_h)
        if cand_c < cur_c:
            accept = True
        else:
            T = T0 * (0.999 ** it) + 1e-12
            accept = rng.random() < np.exp(-(cand_c - cur_c) / T)
        if accept:
            cur_h, cur_c = cand_h, cand_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
        it += 1
    return best_h, best_c


# ---------- Core refinement routine ----------

def refine_h(start_h, rng, deadline):
    n = start_h.size
    best_h = start_h.copy()
    best_c = compute_c5(best_h)

    # stage 1: hill‑climb
    if time.time() < deadline:
        h, c = hill_climb(best_h, rng, max_iters=1500, start_step=0.6)
        if c < best_c:
            best_h, best_c = h, c

    # stage 2: gradient descent
    if time.time() < deadline:
        h = gradient_refine(best_h, rng, step=0.02, max_iter=20)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h, c

    # stage 3: SLSQP with increasing beta
    for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0):
        if time.time() >= deadline:
            break
        try:
            h = slsqp_optimize(best_h, beta, rng,
                               max_iter=400 if beta < 200 else 800)
        except Exception:
            continue
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h, c

    # stage 4: simulated annealing
    if time.time() < deadline:
        h_sa, c_sa = simulated_annealing(best_h, rng, deadline)
        if c_sa < best_c:
            best_h, best_c = h_sa, c_sa

    # stage 5: final hill‑climb
    if time.time() < deadline:
        h, c = hill_climb(best_h, rng, max_iters=2000, start_step=0.4)
        if c < best_c:
            best_h, best_c = h, c

    return best_h, best_c
