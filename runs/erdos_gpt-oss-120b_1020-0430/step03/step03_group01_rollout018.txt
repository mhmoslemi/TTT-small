def enforce_sum(h, target, rng, max_iter=5000):
    """Clip to [0,1] and adjust to exact sum = target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-14:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:

        # final uniform correction, may violate bounds – fix again
        corr = diff / n
        h += corr

def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def compute_c5_exact(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")

def gradient_at_lag(h, L):
    """Gradient of C(L) = Σ h_i (1-h_{i+L}) w.r.t each h_i (linear padding)."""
    N = h.size
    grad = np.zeros(N)
    if L >= 0:
        i1 = np.arange(N - L)
        grad[i1] += 1.0 - h[i1 + L]
        i2 = np.arange(L, N)
        grad[i2] -= h[i2 - L]
    else:
        L2 = -L
        i1 = np.arange(L2, N)
        grad[i1] += 1.0 - h[i1 - L2]
        i2 = np.arange(0, N - L2)
        grad[i2] -= h[i2 + L2]
    return grad

def gradient_move(h, rng, step):
    """Move epsilon from a positive‑gradient index to a negative‑gradient one."""
    N = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    lag = np.argmax(corr) - (N - 1)
    grad = gradient_at_lag(h, lag)

    pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
    neg = np.where((grad < -1e-12) & (h < 1 - 1e-12))[0]

    j = rng.choice(neg)

    if max_move <= 0.0:
        return None
    new_h = h.copy()
    new_h[i] -= max_move
    new_h[j] += max_move

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=600):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # enforce exact sum (tiny numerical drift)
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-14:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-14:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # --------------------------------------------------------------
    # 1. Initial candidate(s)
    # --------------------------------------------------------------
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
        candidates = [h0]
    else:
        n = 360
        candidates = []

    # add a few random starts
    for _ in range(5):
        candidates.append(random_feasible_vector(n, rng))

    best_h = None
    best_c = np.inf
    for h in candidates:
        c = compute_c5_exact(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # --------------------------------------------------------------
    # 2. Main optimisation loop
    # --------------------------------------------------------------
    beta_schedule = [200, 500, 1000, 2000, 5000, 10000]
    while time.time() - start_time < budget_s * 0.85:
        # pick start point
        if rng.random() < 0.8:
            start_vec = perturb_h(best_h, rng, scale=0.3)
        else:
            start_vec = random_feasible_vector(n, rng)

        # hill‑climb
        h_opt, c_opt = hill_climb_extended(
            start_vec, rng, max_iters=5000, step_init=0.5, grad_prob=0.3
        )
        if c_opt < best_c:
            best_h, best_c = h_opt.copy(), c_opt

        # SLSQP refinements
        for beta in beta_schedule:
            if time.time() - start_time > budget_s * 0.85:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, max_iter=800)
            except Exception:
                continue
            c_ref = compute_c5_exact(h_ref)
            if c_ref < best_c:
                best_h, best_c = h_ref.copy(), c_ref

    # --------------------------------------------------------------
    # 3. Optional up‑sampling (if we have enough time left)
    # --------------------------------------------------------------
    elapsed = time.time() - start_time
    if time.time() - start_time < budget_s * 0.95:
        target_n = n * 2
        old_x = (np.arange(n) + 0.5) * (2.0 / n)
        new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
        h_up = np.interp(new_x, old_x, best_h)
        h_up = enforce_sum(h_up, target_n / 2.0, rng)

        # hill‑climb on fine grid
        h_up_opt, c_up_opt = hill_climb_extended(
            h_up, rng, max_iters=8000, step_init=0.2, grad_prob=0.3
        )
        # SLSQP refinements on fine grid
        for beta in beta_schedule:
            if time.time() - start_time > budget_s:
                break
            try:
                h_ref = slsqp_optimize(h_up_opt, beta, max_iter=800)
            except Exception:
                continue
            c_ref = compute_c5_exact(h_ref)
            if c_ref < best_c:
                best_h, best_c, n = h_ref.copy(), c_ref, target_n

    # --------------------------------------------------------------
    # 4. Final clean‑up and return
    # --------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5_exact(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
