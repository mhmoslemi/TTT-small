def project_sum_bounds(x, target):
    # x assumed possibly out of bounds.
    x = np.clip(x, 0.0, 1.0)
    # iterative water-filling
    # first raise lower bound deficiency:
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # we will adjust using capacity
    # If diff > 0, need to add; we can compute capacities = 1 - x. Sum caps = total capacity.
    # If diff <= total capacity, we can distribute proportionally.
    # else diff > capacity (should not happen because target <= n)
    if diff > 0:
        caps = 1.0 - x
        total_cap = caps.sum()
        if diff > total_cap:
            # Should not happen; set all to 1
            x = np.ones_like(x)
            return x
        # Distribute uniformly:
        # While diff > tolerance: allocate to those with capacity > 0.
        # We'll allocate in one pass: delta = diff / num_nonzero
        # but after adding delta, some may exceed 1; need to iterate
        # Implement water-filling loop:
        while diff > 1e-12:
            free = caps > 1e-12
            if not np.any(free):
                break
            num = free.sum()
            delta = diff / num
            # apply delta up to caps
            add = np.minimum(delta, caps)
            x[free] += add[free]
            caps = 1.0 - x
            diff = target - x.sum()
        return x
    else:
        # diff < 0: need to subtract
        diff = -diff
        caps = x
        total_cap = caps.sum()
        if diff > total_cap:
            x = np.zeros_like(x)
            return x
        while diff > 1e-12:
            free = caps > 1e-12
            if not np.any(free):
                break
            num = free.sum()
            delta = diff / num
            sub = np.minimum(delta, caps)
            x[free] -= sub[free]
            caps = x
            diff = target - x.sum()
        return x

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(x0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = x0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {"type": "eq", "fun": sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(softmax_obj, x0, method="SLSQP", bounds=bounds,
                   constraints=cons, options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # ensure exact sum
    x = project_sum_bounds(x, CURRENT_TARGET)
    return x

    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    n = x.size
    # We'll adjust using water-filling algorithm.
    # Approach: while diff not ~0:
    #   if diff > 0: need to add to variables not at upper bound (x < 1)
    #   else need to subtract from variables not at lower bound (x > 0)
    #   amount per variable = diff / count_free
    #   clamp to bounds and recompute diff.
    # Guaranteed to converge in few iterations (max n steps).
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = np.where(x < 1 - 1e-12)[0]
            if free.size == 0:
                break
            delta = diff / free.size
            # add delta but not exceed 1
            add = np.minimum(delta, 1 - x[free])
            x[free] += add
        else:
            free = np.where(x > 1e-12)[0]
            if free.size == 0:
                break
            delta = diff / free.size
            sub = np.minimum(-delta, x[free])  # delta negative; -delta positive
            x[free] -= sub

    if abs(diff) > 1e-12:

            idx = np.argmax(1 - x)  # farthest from upper bound

        else:
            idx = np.argmin(x)  # smallest value
            x[idx] = max(0.0, x[idx] + diff)  # diff negative
    # clip final
    x = np.clip(x, 0.0, 1.0)

def local_descent(h, rng, max_iters=20000):
    h = h.copy()
    best_c = compute_c5(h)
    n = len(h)
    for _ in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        if h[i] == h[j]:
            continue
        # swap
        h[i], h[j] = h[j], h[i]
        new_c = compute_c5(h)
        if new_c < best_c:
            best_c = new_c
            # keep swap
        else:
            # revert
            h[i], h[j] = h[j], h[i]
    return h, best_c

    corr = np.correlate(child, 1 - child, mode="full")

import time, math
import numpy as np
from scipy.optimize import minimize

# -------------------------------------------------
# basic evaluation
# -------------------------------------------------
def compute_c5(h):
    """Discrete C5 for a step function sampled on [0,2]."""
    h = np.asarray(h, dtype=float)
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / h.size)


def compute_c5_batch(pop):
    """Vectorised version for a population (rows are candidates)."""
    m = pop.shape[0]
    res = np.empty(m, dtype=np.float64)
    for i in range(m):
        res[i] = compute_c5(pop[i])
    return res


# -------------------------------------------------
# feasibility utilities
# -------------------------------------------------
def enforce_sum(h, target, rng):
    """
    Adjust h in‑place so that sum(h)=target while keeping 0≤h≤1.
    Simple random transfer method – fast enough for the final polishing.
    """
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            if add <= 0:
                continue
            h[i] += add
        else:
            sub = min(-diff, h[i])
            if sub <= 0:
                continue
            h[i] -= sub
        diff = target - h.sum()
    return h


def random_binary_vector(n, target, rng):
    """Binary vector of length n with exactly target ones."""
    vec = np.zeros(n, dtype=np.uint8)
    vec[:target] = 1
    rng.shuffle(vec)
    return vec


# -------------------------------------------------
# genetic operators (binary representation)
# -------------------------------------------------
def crossover(parent1, parent2, target, rng):
    """Uniform crossover followed by sum correction."""
    n = parent1.size
    mask = rng.random(n) < 0.5
    child = np.where(mask, parent1, parent2).astype(np.uint8)
    cur = int(child.sum())
    diff = cur - target
    if diff > 0:
        idx = np.where(child == 1)[0]
        if idx.size:
            to_zero = rng.choice(idx, size=diff, replace=False)
            child[to_zero] = 0
    elif diff < 0:
        idx = np.where(child == 0)[0]
        if idx.size:
            to_one = rng.choice(idx, size=-diff, replace=False)
            child[to_one] = 1
    return child


def generic_swap_mutation(child, swap_rate, rng):
    """Swap a few 1↔0 positions – keeps the exact sum."""
    n = child.size
    num_swaps = int(round(swap_rate * n))
    for _ in range(num_swaps):
        ones = np.where(child == 1)[0]
        zeros = np.where(child == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        child[i] = 0
        child[j] = 1
    return child


# -------------------------------------------------
# local greedy descent (binary)
# -------------------------------------------------
def local_descent(h_bin, rng, max_iters=20000):
    """Zero‑temperature hill climbing by random swaps."""
    h = h_bin.copy()
    best_c = compute_c5(h)
    n = h.size
    for _ in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]          # perform swap
        c = compute_c5(h)
        if c < best_c:
            best_c = c                     # keep change
        else:
            h[i], h[j] = h[j], h[i]      # revert
    return h, best_c


# -------------------------------------------------
# smooth max optimisation (SLSQP)
# -------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(x0, beta, max_iter=300):
    """SLSQP refinement of the smooth‐max objective."""
    global CURRENT_TARGET, CURRENT_BETA
    n = x0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {"type": "eq", "fun": sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        softmax_obj,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- problem size -------------------------------------------------
    n = 240                     # even length → target = n/2
    target = n // 2

    # ---- GA parameters ------------------------------------------------
    pop_size = 60
    elite_size = 5
    swap_rate = 0.08            # fraction of positions swapped per mutation
    tournament_sz = 5

    # ---- initialise population ----------------------------------------
    pop = np.empty((pop_size, n), dtype=np.uint8)
    for i in range(pop_size):
        pop[i] = random_binary_vector(n, target, rng)

    # If a parent construction exists, embed a derived binary version
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)
        if init.size != n:
            # linear resampling to the required resolution
            x_old = np.linspace(0, 2, init.size, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            init = np.interp(x_new, x_old, init)
        init = np.clip(init, 0.0, 1.0)
        # keep the top‑target entries as 1
        idx = np.argpartition(-init, target - 1)[:target]
        bin_init = np.zeros(n, dtype=np.uint8)
        bin_init[idx] = 1
        pop[0] = bin_init

    fitness = compute_c5_batch(pop.astype(float))
    best_idx = int(np.argmin(fitness))
    best_h = pop[best_idx].copy()
    best_c = fitness[best_idx]

    # ---- GA main loop -------------------------------------------------
    ga_time = budget_s * 0.70
    while time.time() - start < ga_time:
        # elite preservation
        elite_idx = np.argsort(fitness)[:elite_size]
        new_pop = np.empty_like(pop)
        new_pop[:elite_size] = pop[elite_idx]

        # fill the rest
        for i in range(elite_size, pop_size):
            # tournament selection for two parents
            cand = rng.integers(0, pop_size, size=tournament_sz)
            p1 = pop[cand[np.argmin(fitness[cand])]]
            cand = rng.integers(0, pop_size, size=tournament_sz)
            p2 = pop[cand[np.argmin(fitness[cand])]]

            child = crossover(p1, p2, target, rng)
            child = generic_swap_mutation(child, swap_rate, rng)
            new_pop[i] = child

        # evaluate
        new_fit = compute_c5_batch(new_pop.astype(float))

        # combine and select the best pop_size individuals
        combined = np.vstack([pop, new_pop])
        comb_fit = np.concatenate([fitness, new_fit])
        sel = np.argsort(comb_fit)[:pop_size]
        pop = combined[sel]
        fitness = comb_fit[sel]

        # keep track of the global best
        cur_best = int(np.argmin(fitness))
        if fitness[cur_best] < best_c - 1e-12:
            best_c = fitness[cur_best]
            best_h = pop[cur_best].copy()

    # ---- optional binary hill climbing ---------------------------------
    remaining = budget_s - (time.time() - start)
    if remaining > 6.0:
        max_it = int(min(50000, remaining * 1000))
        best_h, best_c = local_descent(best_h, rng, max_iters=max_it)

    # ---- continuous polishing with SLSQP --------------------------------
    remaining = budget_s - (time.time() - start)
    beta_schedule = (50, 200, 800, 2000, 8000, 20000)
    while remaining > 3.0:
        for beta in beta_schedule:
            if time.time() - start >= budget_s - 2.0:
                break
            try:
                refined = slsqp_optimize(best_h.astype(float), beta, max_iter=300)
                rc = compute_c5(refined)
                if rc < best_c:
                    best_c = rc
                    best_h = refined.copy()
            except Exception:
                continue
        remaining = budget_s - (time.time() - start)

    # ---- final sanity checks -------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, np.random.default_rng())
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
