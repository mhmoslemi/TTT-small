import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# utilities
# ----------------------------------------------------------------------
def compute_c5(h):
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(v, rng):
    """Project v onto the affine set sum(v)=len(v)/2 while staying in [0,1]."""
    n = v.size
    target = n / 2.0
    diff = target - v.sum()
    # simple random redistribution
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - v[i]) if diff > 0 else max(diff, -v[i])
        v[i] += add
        diff = target - v.sum()
    return v


def worst_gradient(h):
    """Gradient of the worst shift O_k = Σ h_i(1‑h_{i+k})."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)

    grad = np.zeros_like(h)
    if k == 0:                                            # O_0 = 0, gradient = 1‑2h
        grad[:] = 1.0 - 2.0 * h
    else:
        # for i < n‑k : dO_k/dh_i = 1 - h_{i+k}
        grad[: n - k] = 1.0 - h[k:]
        # for i ≥ k   : -h_{i‑k}
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h, grad, rng, scale=0.5):
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h, rng, max_iters=8000, step_scale=0.8):
    """Hill‑climb driven by the worst‑shift gradient."""
    n = h.size
    best_h = h.copy()
    cur_c = compute_c5(h)
    best_c = cur_c
    scale = step_scale
    decay = max(1, max_iters // 10)

    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=scale)
        c_new = compute_c5(nh)
        if c_new < cur_c:
            h = nh
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        if (it + 1) % decay == 0:
            scale = max(scale * 0.7, 1e-5)
    return best_h, best_c


def slsqp_optimize(h0, beta, rng, max_iter=400):
    """Soft‑max SLSQP refinement for a given β."""
    n = h0.size
    target = n / 2.0

    def soft_obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        # soft‑max of β·corr
        m = np.max(beta * corr)
        return (m + np.log(np.sum(np.exp(beta * corr - m)))) / beta

    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        soft_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = target - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = target - x.sum()
    return x


def gradient_refine(h, rng, step=0.02, max_iter=6):
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)

        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] = 1.0 - h[k:]
            g[k:] -= h[: n - k]

        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        diff = n / 2.0 - h.sum()
        for _ in range(500):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = n / 2.0 - h.sum()
    return h


def random_feasible(n, rng):
    """Generate a random vector in [0,1]^n with Σ = n/2."""
    h = rng.random(n)
    h *= (n / 2.0) / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, rng)

            # ------------------------------------------------------------------
            if time.time() - start < budget_s * 0.98:
                h, c = guided_hill_climb(h, rng, max_iters=3000, step_scale=0.4)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n

    # ------------------------------------------------------------------
    # final local random nudges while budget remains
    # ------------------------------------------------------------------
    while time.time() - start < budget_s * 0.999 and best_h is not None:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # ------------------------------------------------------------------
    # final feasibility enforcement and return
    # ------------------------------------------------------------------
    if best_h is None:
        best_n = 96
        best_h = np.full(best_n, 0.5)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
