def adam_optimize(h0, rng, betas, iters_per_beta, lr=0.05):

- Initialize m = zeros, v = zeros, t = 0.

- For each beta in betas: for i in range(iters_beta):

   - v = beta2*v + (1-beta2)*(g**2)

   - v_hat = v / (1 - beta2**t)

   - update = lr * m_hat / (np.sqrt(v_hat) + eps)

   - h = h - update

   - project to feasible.

- Return final h.

import time
import numpy as np

# ---------- basic utilities ----------
def compute_c5(h):
    """Exact C5 for discrete step function."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_to_feasible(v, target, tol=1e-12):
    """Project onto {0≤h≤1, Σh = target} via bisection on a uniform shift."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Generate a random vector with 0≤h≤1 and Σh = n/2."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_to_feasible(v, n / 2.0)


# ---------- gradient of soft‑max ----------
def softmax_weights(corr, beta):
    """Soft‑max weights for a correlation vector."""
    bc = beta * corr
    m = np.max(bc)
    w = np.exp(bc - m)
    w /= w.sum()
    return w


def grad_softmax(h, w):
    """∇ of the soft‑max surrogate; w must sum to 1."""
    n = h.shape[0]
    grad = np.ones(n)          # constant term from Σ w = 1
    off = n - 1
    for idx, weight in enumerate(w):
        if weight < 1e-14:
            continue
        s = idx - off
        if s >= 0:
            if n - s > 0:
                grad[:n - s] -= weight * h[s:]
                grad[s:] -= weight * h[:n - s]
        else:
            s = -s
            if n - s > 0:
                grad[s:] -= weight * h[:n - s]
                grad[:n - s] -= weight * h[s:]
    return grad


# ---------- optimisation core ----------
def optimize_one_start(h0, rng, betas, iters_per_beta, base_lr, max_time=None):
    """Projected gradient descent on the soft‑max surrogate."""
    h = h0.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_c = compute_c5(h)

    start = time.time()
    for beta, n_iter in zip(betas, iters_per_beta):
        lr = base_lr / (np.sqrt(beta) + 1.0)
        for it in range(n_iter):
            # time guard
            if max_time is not None and (time.time() - start) > max_time:
                return best_h, best_c
            corr = np.correlate(h, 1.0 - h, mode="full")
            w = softmax_weights(corr, beta)
            g = grad_softmax(h, w)
            h = project_to_feasible(h - lr * g, n / 2.0)
            # occasional exact evaluation
            if (it & 15) == 0:
                c = compute_c5(h)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
    return best_h, best_c


# ---------- local hill‑climb ----------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c

    return best_h.tolist(), float(final_c), int(best_n)
