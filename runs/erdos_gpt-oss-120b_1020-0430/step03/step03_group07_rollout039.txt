def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        # symmetrize
        new_h = symmetrize(new_h)
        # ensure bounds (symmetrize ensures that)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-6)
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global variables for L-BFGS-B callbacks
# ----------------------------------------------------------------------
CURRENT_BETA = 1.0
CURRENT_PENALTY = 0.0
CURRENT_TARGET = 0.0

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def enforce_sum(h, target, rng):
    """Adjust h in-place so that sum(h) == target while keeping 0 <= h <= 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Random vector in [0,1]^n with sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def symmetrize(h):
    """Enforce mirror symmetry h[i]=h[-i-1]."""
    return (h + h[::-1]) * 0.5


def convolve_fft(a, b):
    """Full convolution a * b using FFT (real inputs)."""
    n = a.shape[0]
    m = b.shape[0]
    L = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    conv = np.fft.irfft(fa * fb, L)[: n + m - 1]
    return conv


def correlate_full(h):
    """Full linear correlation of h with (1-h) using FFT."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Discrete C5 = max_k ∫ h(x)(1-h(x+k)) dx."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)


# ----------------------------------------------------------------------
# L-BFGS-B with smooth max (log‑sum‑exp) and sum‑penalty
# ----------------------------------------------------------------------
def lbfgsb_obj_and_grad(x):
    """Return (f, grad) for L‑BFGS‑B. Uses global CURRENT_* variables."""
    h = x
    corr = correlate_full(h)                     # sum_i h_i (1-h_{i+lag})
    bc = CURRENT_BETA * corr
    max_bc = np.max(bc)
    exp_bc = np.exp(bc - max_bc)                # stable
    sum_exp = np.sum(exp_bc)
    # smooth approximation of max
    f = (max_bc + np.log(sum_exp)) / CURRENT_BETA
    # sum‑penalty (quadratic)
    sdiff = h.sum() - CURRENT_TARGET
    f += CURRENT_PENALTY * sdiff * sdiff

    w = exp_bc / sum_exp                         # softmax weights
    a = 1.0 - h
    conv1 = convolve_fft(w, a)
    conv2 = convolve_fft(w[::-1], h)
    off = len(w) - 1
    grad_corr = conv1[off:off + len(h)] - conv2[off:off + len(h)]
    grad = grad_corr + 2.0 * CURRENT_PENALTY * sdiff * np.ones_like(h)
    return f, grad


def lbfgsb_optimize(h0, max_iter=250):
    """Run L‑BFGS‑B on the current smooth objective."""
    n = h0.size
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        lbfgsb_obj_and_grad,
        h0,
        method="L-BFGS-B",
        jac=True,
        bounds=bounds,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny sum correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x


# ----------------------------------------------------------------------
# Simple hill‑climb (greedy local search)
# ----------------------------------------------------------------------
def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        new_h = symmetrize(new_h)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-6)
    return best_h, best_c
