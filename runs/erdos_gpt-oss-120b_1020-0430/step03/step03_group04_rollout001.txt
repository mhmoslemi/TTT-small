def simulated_annealing_time(h0, rng, deadline):
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    it = 0
    while time.time() < deadline:
        # dynamic step scale: start high, decay
        step_scale = max(0.5 * (0.9995 ** it), 5e-5)
        cand_h = perturb_h(cur_h, rng, scale=step_scale)
        cand_c = compute_c5(cand_h)
        if cand_c < cur_c:
            cur_h, cur_c = cand_h, cand_c
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
        else:
            # acceptance probability
            T = max(1e-5, cur_c * 0.05 * (0.999 ** it))
            if rng.random() < np.exp(-(cand_c - cur_c) / T):
                cur_h = cand_h
                cur_c = cand_c
        it += 1

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------- utilities ----------------------- #
def compute_c5(h):
    """Exact discrete C5 for step function h on [0,2] with uniform spacing."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def random_binary_vector(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

def random_continuous_vector(n, rng):
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 8)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.85, 1e-5)
    return best_h, best_c

def gradient_refine(h, rng, step=0.02, max_iter=20):
    """Iterative refinement against the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]       # + (1 - h[i+k])
            g[k:] -= h[:upper]            # - h[i]
        step_sz = step * (0.7 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def simulated_annealing(h0, rng, deadline):
    """Time‑bounded SA that respects the sum constraint."""
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    it = 0
    while time.time() < deadline:
        step_scale = max(0.5 * (0.9995 ** it), 5e-5)
        cand_h = perturb_h(cur_h, rng, scale=step_scale)
        cand_c = compute_c5(cand_h)
        if cand_c < cur_c:
            cur_h, cur_c = cand_h, cand_c
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
        else:
            T = max(1e-5, cur_c * 0.03 * (0.999 ** it))
            if rng.random() < np.exp(-(cand_c - cur_c) / T):
                cur_h, cur_c = cand_h, cand_c
        it += 1
    return best_h, best_c

# ----------------------- main driver ----------------------- #

    start_time = time.time()
    best_h = None
    best_c = np.inf

    # construct start vectors

    # 2) random binary feasible
    start_vectors.append(random_binary_vector(target_n, rng))

    # 4) contiguous block (first half ones)
    block = np.zeros(target_n)

    # 5) alternating pattern (even indices)
    alt = np.zeros(target_n)
    alt[::2] = 1.0

    # allocate time for each start vector (roughly equal)
