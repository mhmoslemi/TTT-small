def smooth_c5_and_grad(h, beta):
    N = h.size
    dx = 2.0 / N
    # autocorrelation
    H = np.fft.fft(h)
    auto = np.fft.ifft(H * np.conj(H)).real
    # compute weights
    min_auto = auto.min()
    shifted = auto - min_auto
    # avoid overflow
    w = np.exp(-beta * shifted)
    Z = w.sum()
    if Z == 0:
        w = np.full_like(w, 1.0 / w.size)
        Z = 1.0
    w = w / Z
    # gradient
    W = np.fft.fft(w)
    conv = np.fft.ifft(H * W).real
    corr = np.fft.ifft(H * np.conj(W)).real
    grad = -dx * (conv + corr)
    # surrogate C5
    c_sur = dx * (N/2.0 + np.log(Z) / beta)
    return c_sur, grad

import time
import numpy as np

# --------------------------------------------------------------
# Projection onto the intersection of a hyperplane and box [0,1]
# --------------------------------------------------------------
def proj_sum_box(v, target):
    """Project v onto {x : 0≤x≤1, sum(x)=target} using a simple bisection."""
    lo, hi = -1.0, 1.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        if x.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


# --------------------------------------------------------------
# Random feasible vector (quick)
# --------------------------------------------------------------
def random_feasible_vector(N, rng):
    v = rng.random(N)
    return proj_sum_box(v, N / 2.0)


# --------------------------------------------------------------
# Exact C5 from autocorrelation
# --------------------------------------------------------------
def compute_c5(h):
    N = h.size
    dx = 2.0 / N
    H = np.fft.fft(h)
    auto = np.fft.ifft(H * np.conj(H)).real
    min_auto = auto.min()
    return dx * (N / 2.0 - min_auto)


# --------------------------------------------------------------
# Smooth surrogate and its gradient (log‑sum‑exp smoothing)
# --------------------------------------------------------------
def smooth_c5_and_grad(h, beta):
    N = h.size
    dx = 2.0 / N
    H = np.fft.fft(h)

    # autocorrelation of h
    auto = np.fft.ifft(H * np.conj(H)).real

    # weights for the soft‑max of -auto
    min_auto = auto.min()
    shifted = auto - min_auto          # makes the largest exponent zero
    w = np.exp(-beta * shifted)
    Z = w.sum()
    if Z == 0.0:
        w = np.full_like(w, 1.0 / w.size)
        Z = 1.0
    w /= Z

    # gradient: -dx*(conv + corr)
    W = np.fft.fft(w)
    conv = np.fft.ifft(H * W).real
    corr = np.fft.ifft(H * np.conj(W)).real
    grad = -dx * (conv + corr)

    # surrogate value of C5
    c_sur = dx * (N / 2.0 + np.log(Z) / beta)
    return c_sur, grad


# --------------------------------------------------------------
# Upsample a step function by an integer factor (linear interp, periodic)
# --------------------------------------------------------------
def upsample_h(h, factor):
    N = h.size
    N2 = N * factor
    dx_old = 2.0 / N
    xs_old = (np.arange(N) + 0.5) * dx_old
    xs_ext = np.concatenate((xs_old, xs_old + 2.0))
    h_ext = np.concatenate((h, h))
    dx_new = 2.0 / N2
    xs_new = (np.arange(N2) + 0.5) * dx_new
    h_new = np.interp(xs_new, xs_ext, h_ext)
    h_new = np.clip(h_new, 0.0, 1.0)
    return proj_sum_box(h_new, N2 / 2.0)


# --------------------------------------------------------------
# Simple hill‑climb refinement (pairwise transfers)
# --------------------------------------------------------------
def hill_climb_refine(h, rng, max_iters=3000):
    N = h.size
    target = N / 2.0
    best_h = h.copy()
    best_c = compute_c5(best_h)
    for _ in range(max_iters):
        i = rng.integers(0, N)
        j = rng.integers(0, N)
        if i == j:
            continue
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.2
        new_h = best_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_h = proj_sum_box(new_h, target)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h
    return best_h, best_c


# --------------------------------------------------------------
# Projected gradient descent on the smooth surrogate
# --------------------------------------------------------------
def gradient_descent(start_h, beta, max_iters, step0, target, deadline):
    h = start_h.copy()
    best_h = h.copy()
    best_c = compute_c5(best_h)
    N = h.size
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        _, grad = smooth_c5_and_grad(h, beta)

        # diminishing step size
        frac = it / max_iters
        step = step0 * (1.0 - frac) + 1e-4

        h_new = proj_sum_box(h - step * grad, target)

        c_exact = compute_c5(h_new)
        if c_exact < best_c:
            best_c, best_h = c_exact, h_new.copy()
        h = h_new
    return best_h, best_c, h
