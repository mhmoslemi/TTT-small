def de_optimize(n, max_time):
    target = n/2
    popsize = max(30, int(0.05*n))   # maybe 30 or bigger
    pop = []
    # initial guess from initial_h_values if available
    if 'initial_h_values' in globals():
        init = upsample_initial(n)
        init = project_feasible(init, target)
        pop.append(init)
    while len(pop) < popsize:
        pop.append(random_feasible(n))
    pop = np.vstack(pop)
    scores = np.array([compute_c5(ind) for ind in pop])
    best_idx = np.argmin(scores)
    best = pop[best_idx].copy()
    best_score = scores[best_idx]

    F = 0.5
    CR = 0.9
    start = time.time()
    # generate until time out
    while time.time() - start < max_time:
        for i in range(popsize):
            # mutation using best/1 strategy
            # choose two distinct random indices not equal i
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2 = rng.choice(idxs, 2, replace=False)
            mutant = best + F * (pop[r1] - pop[r2])
            # crossover
            cross_mask = rng.random(n) < CR
            if not np.any(cross_mask):
                cross_mask[rng.integers(0, n)] = True
            trial = np.where(cross_mask, mutant, pop[i])
            trial = project_feasible(trial, target)
            trial_score = compute_c5(trial)
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best = trial.copy()
                    best_score = trial_score
    return best, best_score
```

def lbfgsb_refine(h0, betas, rng):
    n = h0.size
    target = n/2
    bounds = [(0.0,1.0)]*n
    def obj(x, beta):
        # ensure sum constraint via penalty
        sum_diff = x.sum() - target
        penalty = 1e3 * sum_diff*sum_diff
        corr = np.fft.irfft(np.fft.rfft(x) * np.fft.rfft(1-x), n=2*n-1)[:2*n-1]
        bc = beta * corr
        m = bc.max()
        # log-sum-exp stable
        lse = m + np.log(np.exp(bc - m).sum())
        return lse / beta + penalty
    x = np.clip(h0, 0.0, 1.0)
    for beta in betas:
        res = minimize(lambda x: obj(x, beta), x, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 200, 'ftol': 1e-12})
        x = np.clip(res.x, 0.0, 1.0)
        # project to exact sum after each beta
        diff = target - x.sum()
        if abs(diff) > 1e-12:
            # adjust using simple distribution among indices that are not saturated
            # use loop similar to earlier
            for _ in range(5000):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
                x[i] += add
                diff = target - x.sum()
    return x

def gradient_step(h, lr):
    n = h.size
    corr_full = np.fft.irfft(np.fft.rfft(h) * np.fft.rfft(1 - h), n=1 << ((2*n-1).bit_length()))[:2*n-1]
    # find argmax
    idx = np.argmax(corr_full)
    k = idx - (n - 1)  # shift
    grad = np.empty(n, dtype=float)
    if k >= 0:
        # vectorized
        # compute contributions
        # part1: i <= n-1-k => 1 - h[i+k]
        # part2: i >= k => -h[i - k]
        # Use slicing to fill efficiently.
        grad[:] = 0.0
        if k < n:
            grad[:n - k] += 1 - h[k:]
            grad[k:] -= h[:n - k]
    else:
        p = -k
        grad[:] = 0.0
        if p < n:
            grad[p:] += 1 - h[:n - p]
            grad[:n - p] -= h[p:]
    # step
    h_new = h - lr * grad
    # clip
    h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

import time
import numpy as np
from scipy.optimize import minimize

# globals for SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1‑x_{i+k}) via log‑sum‑exp."""
    n = x.size
    N = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(x, n=N)
    G = np.fft.rfft(1 - x, n=N)
    corr = np.fft.irfft(H * G, n=N)[: 2 * n - 1]
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def project_to_feasible(v, target):
    """Project onto {0≤h≤1, Σh=target} by a bisection on a uniform shift."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n, rng):
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_to_feasible(v, n / 2.0)


def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=3000, start_step=0.5):
    h = h_start.copy()
    best = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        cand = perturb_h(h, rng, scale=step)
        c = compute_c5(cand)
        if c < best_c:
            best_c, best, h = c, cand.copy(), cand
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best, best_c


def compute_c5(h):
    """Exact C5 for a discretised step function."""
    n = h.size
    N = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=N)
    G = np.fft.rfft(1 - h, n=N)
    corr = np.fft.irfft(H * G, n=N)[: 2 * n - 1]
    return np.max(corr) * (2.0 / n)


def upsample_initial(to_n):
    init = np.array(initial_h_values, dtype=float)
    m = init.size
    xs_old = np.linspace(0.0, 2.0, m, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, to_n, endpoint=False)
    return np.interp(xs_new, xs_old, init)


def de_optimize(n, rng, max_time):
    target = n / 2.0
    popsize = max(20, min(80, int(0.05 * n)))
    pop = []

    if "initial_h_values" in globals():
        init_vec = upsample_initial(n)
        init_vec = project_to_feasible(init_vec, target)
        pop.append(init_vec)

    while len(pop) < popsize:
        pop.append(random_feasible_vector(n, rng))

    pop = np.array(pop)
    scores = np.array([compute_c5(ind) for ind in pop])
    best_idx = scores.argmin()
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]

    F, CR = 0.5, 0.9
    start = time.time()
    while time.time() - start < max_time:
        for i in range(popsize):
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2 = rng.choice(idxs, 2, replace=False)
            mutant = best_h + F * (pop[r1] - pop[r2])
            cross = rng.random(n) < CR
            if not cross.any():
                cross[rng.integers(0, n)] = True
            trial = np.where(cross, mutant, pop[i])
            trial = project_to_feasible(trial, target)
            s = compute_c5(trial)
            if s < scores[i]:
                pop[i] = trial
                scores[i] = s
                if s < best_score:
                    best_score, best_h = s, trial.copy()
    return best_h, best_score


def slsqp_optimize(h0, beta, rng, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def gradient_step(h, lr):
    """One projected gradient step targeting the worst shift."""
    n = h.size
    N = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=N)
    G = np.fft.rfft(1 - h, n=N)
    corr = np.fft.irfft(H * G, n=N)[: 2 * n - 1]
    idx = np.argmax(corr)
    k = idx - (n - 1)  # shift (may be negative)

    grad = np.zeros_like(h)
    if k >= 0:
        if k < n:
            grad[: n - k] += 1 - h[k:]
            grad[k:] -= h[: n - k]
    else:
        p = -k
        if p < n:
            grad[p:] += 1 - h[: n - p]
            grad[: n - p] -= h[p:]
    h_new = h - lr * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    return h_new


def grad_descent_optimize(h0, rng, max_time):
    target = h0.size / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    lr = 0.2
    start = time.time()
    while time.time() - start < max_time:
        h_new = gradient_step(h, lr)
        h_new = project_to_feasible(h_new, target)
        c = compute_c5(h_new)
        if c < best_c:
            best_c, best_h, h = c, h_new.copy(), h_new
            lr = min(lr * 1.1, 0.5)
        else:
            lr = max(lr * 0.6, 1e-4)
        if rng.random() < 0.01:
            h = perturb_h(h, rng, scale=0.3)
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # choose a working resolution
    n = 600 if budget_s > 800 else 300
    target = n / 2.0

    # --- Global search (DE) ---
    de_time = min(budget_s * 0.6, max(30.0, budget_s - 300.0))
    h_cur, c_cur = de_optimize(n, rng, de_time)

    # --- Local refinement (SLSQP surrogate) ---
    for beta in (10, 50, 200, 800, 2000):
        if time.time() - start_time > budget_s * 0.95:
            break
        h_cur = slsqp_optimize(h_cur, beta, rng, max_iter=500)
        c_cur = compute_c5(h_cur)

    # --- Gradient descent targeting worst shift ---
    remaining = budget_s - (time.time() - start_time)
    if remaining > 20.0:
        gd_time = remaining * 0.2
        h_cur, c_cur = grad_descent_optimize(h_cur, rng, gd_time)

    # --- Final hill‑climb ---
    if time.time() - start_time < budget_s * 0.99:
        h_cur, c_cur = hill_climb(h_cur, rng, max_iters=2500, start_step=0.3)

    # enforce feasibility precisely
    h_cur = np.clip(h_cur, 0.0, 1.0)
    diff = target - h_cur.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h_cur[i]) if diff > 0 else max(diff, -h_cur[i])
            h_cur[i] += add
            diff = target - h_cur.sum()

    final_c = compute_c5(h_cur)
    return h_cur.tolist(), float(final_c), int(n)
