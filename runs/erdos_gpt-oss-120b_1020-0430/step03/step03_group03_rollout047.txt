import time, math, numpy as np
from scipy.optimize import minimize

# --------------------- helpers ---------------------

def compute_c5(h):
    """Discrete C5 for step function h on [0,2] with dx = 2/n."""
    n = h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)


def enforce_sum(h, target, rng):
    """Adjust h in‑place so that sum(h)=target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector with 0≤h≤1 and sum=n/2."""
    h = np.full(n, 0.5)
    # spread mass by pairwise transfers
    for _ in range(n * 5):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    enforce_sum(h, n / 2.0, rng)
    return h


def shift_product_info(h, shift):
    """Return indices i and product h[i]*(1-h[i+shift]) for a given shift."""
    n = h.size
    if shift >= 0:
        idx = np.arange(0, n - shift)
        prod = h[idx] * (1.0 - h[idx + shift])
    else:
        s = -shift
        idx = np.arange(s, n)
        prod = h[idx] * (1.0 - h[idx - s])
    return idx, prod


def targeted_transfer(h, rng, shift, scale=0.5):
    """Move mass from a high‑product to a low‑product position for a given shift."""
    idx, prod = shift_product_info(h, shift)
    if prod.size == 0:
        # fallback to a simple random transfer
        i = rng.integers(0, h.size)
        j = rng.integers(0, h.size)
        while i == j:
            j = rng.integers(0, h.size)
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            return h.copy()
        eps = max_eps * scale * rng.random()
        nh = h.copy()
        nh[i] -= eps
        nh[j] += eps
        return nh

    # high‑product (close to 1) and low‑product (close to 0) sets
    hi_thr = np.quantile(prod, 0.75)
    lo_thr = np.quantile(prod, 0.25)
    high = idx[prod >= hi_thr]
    low = idx[prod <= lo_thr]
    if high.size == 0 or low.size == 0:
        # fallback to random transfer
        i = rng.integers(0, h.size)
        j = rng.integers(0, h.size)
        while i == j:
            j = rng.integers(0, h.size)
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            return h.copy()
        eps = max_eps * scale * rng.random()
        nh = h.copy()
        nh[i] -= eps
        nh[j] += eps
        return nh

    i = rng.choice(high)
    j = rng.choice(low)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def random_transfer(h, rng, scale=0.3):
    """Transfer ε between two coordinates (preserves sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


# --------- smooth‑max SLSQP (same as previous) ---------

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # soft‑max (log‑sum‑exp) stabilised
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on a smooth approximation of the max‑overlap."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {"type": "eq", "fun": sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x


# --------------------- main run ---------------------

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # ----- initialise vector -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        enforce_sum(h0, n / 2.0, rng)
    else:
        n = 180
        h0 = random_feasible_vector(n, rng)

    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c

    # ----- simulated annealing with targeted transfers -----
    # use ~80 % of total budget for SA
    sa_budget = budget_s * 0.82
    T0, Tf = 0.05, 1e-6
    while time.time() - start < sa_budget:
        elapsed = time.time() - start
        frac = elapsed / sa_budget
        T = T0 * (Tf / T0) ** frac

        # location of current worst shift
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        kmax = int(np.argmax(corr))
        shift = kmax - (n - 1)

        # propose a candidate
        if rng.random() < 0.65:
            cand_h = targeted_transfer(cur_h, rng, shift, scale=0.6)
        else:
            cand_h = random_transfer(cur_h, rng, scale=0.4)

        cand_c = compute_c5(cand_h)
        delta = cand_c - cur_c

        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            cur_h, cur_c = cand_h, cand_c
            if cand_c < best_c:
                best_h, best_c = cand_h.copy(), cand_c

        # occasional rejuvenation from a fresh random point
        if rng.random() < 0.002:
            cur_h = random_feasible_vector(n, rng)
            cur_c = compute_c5(cur_h)

    # ----- final polishing with SLSQP (high‑beta) -----
    remaining = budget_s - (time.time() - start)
    beta_list = [50, 200, 800, 2000, 8000, 20000, 80000, 200000]
    for beta in beta_list:
        if time.time() - start >= budget_s:
            break
        # allocate a modest iteration budget per beta
        iters = 500
        try:
            refined = slsqp_optimize(best_h, beta, max_iter=iters)
            rc = compute_c5(refined)
            if rc < best_c:
                best_h, best_c = refined, rc
        except Exception:
            continue

    # final clipping and sum enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
