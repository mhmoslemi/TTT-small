def targeted_annealing(h0, rng, max_iter, init_temp, final_temp):
    h = h0.copy()
    current_c = compute_c5(h)
    best_h, best_c = h.copy(), current_c
    for it in range(1, max_iter+1):
        T = init_temp * ((final_temp/init_temp)**(it / max_iter))
        # compute current max shift
        corr = np.correlate(h, 1 - h, mode='full')
        k = np.argmax(corr) - (h.size - 1)
        # compute product
        if k >= 0:
            prod = h[:h.size - k] * (1 - h[k:])
        else:
            prod = h[-k:] * (1 - h[:h.size + k])
        # high product indices (top 20%)
        thresh = np.quantile(prod, 0.8) if prod.size>0 else 0
        high_idxs = np.where(prod >= thresh)[0]
        # choose source i from high_idxs or random
        if high_idxs.size > 0 and rng.random() < 0.8:
            i = rng.choice(high_idxs)
        else:
            i = rng.integers(0, h.size)
        # choose dest j random where h[j] < 1
        # ensure not same as i
        for _ in range(5):
            j = rng.integers(0, h.size)
            if j != i and h[j] < 1.0:
                break
        # compute max epsilon
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            cand_h = h
        else:
            eps = max_eps * (rng.random() * 0.5 + 0.25) # 0.25-0.75 of max
            cand = h.copy()
            cand[i] -= eps
            cand[j] += eps
            cand_c = compute_c5(cand)
            d = cand_c - current_c
            if d < 0 or rng.random() < math.exp(-d / max(T,1e-12)):
                h = cand
                current_c = cand_c
                if cand_c < best_c:
                    best_c, best_h = cand_c, cand.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # Determine base n
    if 'initial_h_values' in globals():
        base = np.array(initial_h_values, dtype=float)
        n0 = base.size
        upscale = 2 if n0 * 2 <= 800 else 1
        if upscale > 1:
            h0 = np.repeat(base, upscale)
        else:
            h0 = base.copy()
    else:
        n0 = 200
        h0 = random_feasible_vector(n0, rng)
        upscale = 1
    n = h0.size
    enforce_sum(h0, n/2.0, rng)
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    # allocate SA time
    sa_time = budget_s * 0.80
    while time.time() - start < sa_time:
        # restart: start from best or random
        if rng.random() < 0.7:
            start_h = best_h.copy()
            # add small random perturbation to diversify
            start_h = perturb_h(start_h, rng, scale=0.2)
        else:
            start_h = random_feasible_vector(n, rng)
        # allocate iter count
        remaining = sa_time - (time.time() - start)
        # approximate iteration count: 1500 evaluations per second (empirically)
        max_iter = int(min(50000, max(2000, remaining * 1500)))
        cand_h, cand_c = targeted_annealing(start_h, rng, max_iter,
                                            init_temp=0.04, final_temp=1e-7)

        # occasional SLSQP refinement if time permits

                break
            try:

def targeted_annealing(h0, rng, max_iter, init_temp, final_temp):
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()

    n = h.size
    for it in range(1, max_iter+1):

            thr = np.quantile(prod_arr, 0.75)
            high_idxs = np.where(prod_arr >= thr)[0]
        else:
            high_idxs = np.arange(n) # fallback
        # choose source i: high idx offset by shift? Need actual index in h.
        # For prod_arr indices i correspond to h[i] (if k >=0) else same.
        i = None
        if high_idxs.size > 0 and rng.random() < 0.9:
            # pick high idx
            idx = rng.choice(high_idxs)
            i = idx
        else:
            i = rng.integers(0, n)
        # ensure i has some mass to move
        if h[i] <= 1e-12:
            i = rng.integers(0, n)
        # choose dest j where h[j] < 1
        # Prefer location where product is low for same shift
        # compute product low indices threshold
        if prod_arr.size > 0:
            thr_low = np.quantile(prod_arr, 0.25)
            low_idxs = np.where(prod_arr <= thr_low)[0]
        else:
            low_idxs = np.arange(n)
        # adjust j index for shift offset: same as i?
        # We'll just pick random j.
        j = rng.integers(0, n)

        while j == i:

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    # adjust random element to fix diff, within bounds
    # We can try repeated attempts
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        i = rng.integers(0, n)
        # possible add: up to 1 - h[i] if diff>0, or down to -h[i] if diff<0
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-12:
            attempts += 1
            continue
        h[i] += add
        diff = target - h.sum()
        attempts = 0
    return h

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    # objective as earlier: softmax approx of max correlation
    def softmax_obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        bc = CURRENT_BETA * corr
        m = np.max(bc)
        # compute log-sum-exp
        return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
    def sum_eq(x):
        return np.sum(x) - CURRENT_TARGET
    cons = {'type':'eq','fun': sum_eq}
    bounds = [(0.0,1.0)]*n
    res = minimize(softmax_obj, h0, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter,'ftol':1e-12,'disp':False})
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

def greedy_descent(h, rng, steps):
    h = h.copy()
    cur_c = compute_c5(h)
    n = h.size
    no_improve = 0
    for s in range(steps):
        corr = np.correlate(h, 1 - h, mode='full')

import time, math, numpy as np
from scipy.optimize import minimize

# ---------- utilities ----------
def compute_c5(h):
    """Discrete C5 for step function h on [0,2] with dx = 2/n."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)

def enforce_sum(h, target, rng):
    """Adjust h in‑place so that sum(h)=target while 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 2000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-12:
            attempts += 1
            continue
        h[i] += add
        diff = target - h.sum()
        attempts = 0
    return h

def random_feasible_vector(n, rng):
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def perturb_h(h, rng, scale=0.2):
    """Transfer ε between two coordinates (preserves sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = max_eps * (rng.random() * scale)
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

# ---------- SLSQP with softmax max ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    """Smooth max minimisation with SLSQP."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {"type": "eq", "fun": sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

# ---------- Targeted annealing ----------
def targeted_annealing(h0, rng, max_iter, init_temp, final_temp):
    """SA that biases moves using the currently maximal overlap shift."""
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    n = h.size
    for it in range(1, max_iter + 1):
        T = init_temp * ((final_temp / init_temp) ** (it / max_iter))
        corr = np.correlate(h, 1 - h, mode="full")
        k = np.argmax(corr) - (n - 1)

        if k >= 0:
            prod_arr = h[: n - k] * (1 - h[k:])
            idx_arr = np.arange(0, n - k)
        else:
            prod_arr = h[-k:] * (1 - h[: n + k])
            idx_arr = np.arange(-k, n)

        # choose source from high‑product region
        if prod_arr.size and rng.random() < 0.85:
            thr_hi = np.quantile(prod_arr, 0.75)
            high = idx_arr[prod_arr >= thr_hi]
            if high.size:
                i = rng.choice(high)
            else:
                i = rng.integers(0, n)
        else:
            i = rng.integers(0, n)

        # ensure i has mass to give
        if h[i] <= 1e-12:
            i = rng.integers(0, n)

        # destination (random, respecting bounds)
        j = rng.integers(0, n)
        while j == i or h[j] >= 1.0:
            j = rng.integers(0, n)

        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = max_eps * (rng.random() * 0.6 + 0.2)  # 20‑80 % of max_eps
        cand = h.copy()
        cand[i] -= eps
        cand[j] += eps
        cand_c = compute_c5(cand)
        diff = cand_c - cur_c
        if diff < 0 or rng.random() < math.exp(-diff / max(T, 1e-12)):
            h, cur_c = cand, cand_c
            if cand_c < best_c:
                best_c, best_h = cand_c, cand.copy()
        if it % 500 == 0:
            enforce_sum(h, n / 2.0, rng)
    return best_h, best_c

# ---------- greedy descent ----------
def greedy_descent(h, rng, steps):
    h = h.copy()
    cur_c = compute_c5(h)
    n = h.size
    no_improve = 0
    for _ in range(steps):
        corr = np.correlate(h, 1 - h, mode="full")
        k = np.argmax(corr) - (n - 1)

        if k >= 0:
            prod_arr = h[: n - k] * (1 - h[k:])
            idx_arr = np.arange(0, n - k)
        else:
            prod_arr = h[-k:] * (1 - h[: n + k])
            idx_arr = np.arange(-k, n)

        if prod_arr.size == 0:
            break
        i = idx_arr[np.argmax(prod_arr)]
        j = rng.integers(0, n)
        while j == i or h[j] >= 1.0:
            j = rng.integers(0, n)

        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 1e-12:
            continue
        eps = max_eps * 0.5
        cand = h.copy()
        cand[i] -= eps
        cand[j] += eps
        cand_c = compute_c5(cand)
        if cand_c < cur_c - 1e-12:
            h, cur_c = cand, cand_c
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 2000:
            break
    return h, cur_c

# ---------- main run ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # initialise vector, possibly up‑sampled
    if "initial_h_values" in globals():
        base = np.array(initial_h_values, dtype=float)
        upscale = 2 if base.size * 2 <= 800 else 1
        h0 = np.repeat(base, upscale) if upscale > 1 else base.copy()
    else:
        base = None
        upscale = 1
        h0 = random_feasible_vector(200, rng)

    n = h0.size
    enforce_sum(h0, n / 2.0, rng)

    best_h, best_c = h0.copy(), compute_c5(h0)

    # ------- simulated annealing phase -------
    sa_budget = budget_s * 0.78
    while time.time() - start < sa_budget:
        if rng.random() < 0.7:
            start_h = perturb_h(best_h, rng, scale=0.25)
        else:
            start_h = random_feasible_vector(n, rng)

        remaining = max(0.0, sa_budget - (time.time() - start))
        max_iter = int(min(18000, max(3000, remaining * 1800)))
        cand_h, cand_c = targeted_annealing(
            start_h, rng, max_iter, init_temp=0.045, final_temp=1e-7
        )
        if cand_c < best_c:
            best_h, best_c = cand_h.copy(), cand_c

        # occasional SLSQP polishing
        for beta in (120, 480, 1920):
            if time.time() - start > sa_budget:
                break
            try:
                refined = slsqp_optimize(best_h, beta, max_iter=180)
                rc = compute_c5(refined)
                if rc < best_c:
                    best_h, best_c = refined, rc
            except Exception:
                continue

    # ------- greedy local improvement -------
    time_left = budget_s - (time.time() - start)
    if time_left > 5.0:
        steps = int(min(12000, max(2000, time_left * 2000)))
        best_h, best_c = greedy_descent(best_h, rng, steps)

    # ------- final high‑beta SLSQP polishing -------
    for beta in (8000, 24000, 72000):
        if time.time() - start > budget_s:
            break
        try:
            best_h = slsqp_optimize(best_h, beta, max_iter=250)
            best_c = compute_c5(best_h)
        except Exception:
            continue

    best_h = np.clip(best_h, 0.0, 1.0)
    enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
