import time
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Utility functions
# --------------------------------------------------------------
def compute_c5(h):
    """Exact discrete C₅ for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_sum_bounds(h, target):
    """Clip to [0,1] and then adjust to hit the target sum."""
    h = np.clip(h, 0.0, 1.0)
    diff = h.sum() - target
    # simple iterative correction; usually converges in a few passes
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        offset = diff / h.size
        h = np.clip(h - offset, 0.0, 1.0)
        diff = h.sum() - target
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1] with sum = n/2."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    v = np.clip(v, 0.0, 1.0)
    v = project_sum_bounds(v, n / 2.0)
    return v


def perturb_h(h, rng, scale=0.5):
    """Transfer a small amount between two coordinates (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    while j == i:
        j = rng.integers(0, n)
    max_pos = min(h[i], 1.0 - h[j])
    max_neg = min(h[j], 1.0 - h[i])
    if max_pos + max_neg == 0:
        return h.copy()
    eps = (rng.random() * (max_pos + max_neg)) - max_neg
    eps *= scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


# --------------------------------------------------------------
# Smooth surrogate for SLSQP (log‑sum‑exp of the shifted overlaps)
# --------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=500):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny repair
    if abs(CURRENT_TARGET - x.sum()) > 1e-12:
        x = project_sum_bounds(x, CURRENT_TARGET)
    return x

# Main optimisation routine
# --------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # ----------------------------------------------------------
    # Base vector (either provided or a uniform start)
    # ----------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_sum_bounds(h0, n0 / 2.0)
    else:
        n0 = 180
        h0 = np.full(n0, 0.5)
        h0 = project_sum_bounds(h0, n0 / 2.0)

    # ----------------------------------------------------------
    # Upsample for higher resolution (factor 2 → 360 points)
    # ----------------------------------------------------------
    factor = 2
    n = n0 * factor
    h = np.repeat(h0, factor)
    h = project_sum_bounds(h, n / 2.0)
    target_sum = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ----------------------------------------------------------
    # Helper to keep the global best updated
    # ----------------------------------------------------------
    def maybe_update(candidate):
        nonlocal best_h, best_c
        c = compute_c5(candidate)
        if c < best_c:
            best_c = c
            best_h = candidate.copy()
        return c

    # ----------------------------------------------------------
    # Local hill‑climb (transfer ε between two positions)
    # ----------------------------------------------------------
    def hill_climb_local(start_h, max_iters, start_step):
        h_loc = start_h.copy()
        best_local = h_loc.copy()
        best_local_c = compute_c5(h_loc)
        step = start_step
        decay = max(1, max_iters // 10)
        for it in range(max_iters):
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            max_pos = min(h_loc[i], 1.0 - h_loc[j])
            max_neg = min(h_loc[j], 1.0 - h_loc[i])
            if max_pos + max_neg == 0:
                continue
            eps = (rng.random() * (max_pos + max_neg)) - max_neg
            eps *= step
            new_h = h_loc.copy()
            new_h[i] -= eps
            new_h[j] += eps
            c_new = compute_c5(new_h)
            if c_new < best_local_c:
                best_local_c = c_new
                best_local = new_h.copy()
                h_loc = new_h
            if (it + 1) % decay == 0:
                step = max(step * 0.7, 1e-5)
        return best_local, best_local_c

    # ----------------------------------------------------------
    # Phase 1 – aggressive hill‑climbing (≈50 % of the time)
    # ----------------------------------------------------------
    while time.time() - start < budget_s * 0.5:
        if rng.random() < 0.6:
            start_h = perturb_h(best_h, rng, scale=0.3)
        else:
            start_h = random_feasible_vector(n, rng)
        h_opt, c_opt = hill_climb_local(start_h, max_iters=4000, start_step=0.5)
        maybe_update(h_opt)

    # ----------------------------------------------------------
    # Phase 2 – subgradient descent on the worst shift
    # ----------------------------------------------------------
    base_step = 0.6
    it = 0
    h_curr = best_h.copy()
    while time.time() - start < budget_s * 0.8:
        # worst shift
        corr = np.correlate(h_curr, 1 - h_curr, mode="full")
        vals = corr * (2.0 / n)
        idx = np.argmax(vals)
        k = idx - (n - 1)  # signed offset
        # gradient of the worst‑shift objective
        g = np.full(n, 2.0 / n)
        if k >= 0:
            nk = n - k
            g[:nk] -= (2.0 / n) * h_curr[k:]
            g[k:] -= (2.0 / n) * h_curr[:nk]
        else:
            kk = -k
            nk = n - kk
            g[kk:] -= (2.0 / n) * h_curr[:nk]
            g[:nk] -= (2.0 / n) * h_curr[kk:]
        step = base_step / np.sqrt(it + 1)
        h_new = h_curr - step * g
        h_new = project_sum_bounds(h_new, target_sum)
        c_new = compute_c5(h_new)
        maybe_update(h_new)
        # accept if we improved; otherwise keep with a tiny probability
        if c_new < compute_c5(h_curr) or rng.random() < 0.01:
            h_curr = h_new
        it += 1

    # ----------------------------------------------------------
    # Phase 3 – refinement with smooth surrogate + SLSQP
    # ----------------------------------------------------------
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0:
        for beta in [200, 800, 2000, 5000]:
            if time.time() - start > budget_s * 0.98:
                break
            h_opt = slsqp_optimize(best_h, beta, max_iter=800)
            maybe_update(h_opt)

        # final quick hill‑climb
        if time.time() - start < budget_s * 0.99:
            h_opt, c_opt = hill_climb_local(best_h, max_iters=3000, start_step=0.2)
            maybe_update(h_opt)

    # ----------------------------------------------------------
    # Final clean‑up
    # ----------------------------------------------------------
    best_h = project_sum_bounds(np.clip(best_h, 0.0, 1.0), best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
