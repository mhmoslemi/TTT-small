    cur_c = best_c

        # select random 1 and 0 indices
        # choose i where h[i]==1
        # choose j where h[j]==0

        i = rng.choice(ones)
        j = rng.choice(zeros)

        c = compute_c5(h)
        delta = c - cur_c

def correlate_full(h):
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def enforce_sum(h, target, rng):
    """Adjust h so that Σh == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_binary_feasible(n, rng):
    """Return a binary 0/1 vector of length n with exactly n/2 ones."""
    arr = np.zeros(n, dtype=float)
    arr[: n // 2] = 1.0
    rng.shuffle(arr)
    return arr


def binary_from_initial(init_h, rng):
    """Round an initial real vector to binary while respecting the sum."""
    n = init_h.size
    target = n / 2.0
    bin_h = (init_h > 0.5).astype(float)
    cur = bin_h.sum()
    diff = int(target - cur)
    if diff > 0:
        zeros = np.flatnonzero(bin_h == 0)
        ids = rng.choice(zeros, size=diff, replace=False)
        bin_h[ids] = 1.0
    elif diff < 0:
        ones = np.flatnonzero(bin_h == 1)
        ids = rng.choice(ones, size=-diff, replace=False)
        bin_h[ids] = 0.0
    return bin_h


def correlate_full(h):
    """Full linear correlation of h with (1‑h) using FFT (zero‑padding)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Exact discrete C₅ for a step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.shape[0])


# ----------------------------------------------------------------------
# SLSQP optimisation (smooth approximation of the max)
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Log‑sum‑exp smoothing of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = correlate_full(x)
    bc = CURRENT_BETA * corr
    m = bc.max()
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=400):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-14, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny correction of the sum
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ----------------------------------------------------------------------
# Binary simulated annealing (swap moves)
# ----------------------------------------------------------------------
def binary_sa(start_h, rng, max_iters=4000, T0=0.02, Tf=1e-6):
    """Swap‑based SA for binary vectors preserving the exact half‑sum."""
    h = start_h.copy()
    n = h.size
    # maintain mutable Python lists of indices
    ones = [int(i) for i in np.flatnonzero(h)]
    zeros = [int(i) for i in np.flatnonzero(1 - h)]
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    for it in range(max_iters):
        # temperature schedule (geometric)
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        # pick a random 1 and a random 0
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        # propose swap
        h[i], h[j] = 0.0, 1.0
        ones[i_idx] = j
        zeros[j_idx] = i
        new_c = compute_c5(h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = 1.0, 0.0
            ones[i_idx] = i
            zeros[j_idx] = j
    return best_h, best_c
