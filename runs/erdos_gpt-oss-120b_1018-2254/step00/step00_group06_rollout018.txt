def project_onto_sum_box(v, target):
    # v is ndarray.
    # find lambda such that sum(clip(v+lam,0,1)) = target
    lo = -2.0
    hi = 2.0
    # Ensure target feasible: target between 0 and n.
    for _ in range(50):
        lam = (lo+hi)/2
        w = np.clip(v + lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo+hi)/2, 0.0, 1.0)

def project_onto_sum_box(v, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) / 2
        w = np.clip(v + lam, 0.0, 1.0)
        s = w.sum()

            hi = lam
        else:
            lo = lam

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    C5 = np.max(corr) * dx
    return C5, corr

def shift_zero(arr, shift):
    n = arr.shape[0]
    out = np.zeros_like(arr)
    if shift >= 0:
        out[:n - shift] = arr[shift:]
    else:
        out[-shift:] = arr[:n + shift]
    return out

# Helper: project a vector onto the hyperplane sum=target with box [0,1]
def _project_sum_box(v: np.ndarray, target: float) -> np.ndarray:
    """Return the Euclidean projection of v onto {0<=x<=1, sum(x)=target}."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        w = np.clip(v + lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


# ------------------------------------------------------------
# Helper: compute C5 and the full correlation array
def _c5_and_corr(h: np.ndarray):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx), corr


# ------------------------------------------------------------
# Helper: shift an array with zero padding (no wrap‑around)
def _shift_zero(arr: np.ndarray, shift: int) -> np.ndarray:
    n = arr.shape[0]
    out = np.zeros_like(arr)
    if shift >= 0:
        out[: n - shift] = arr[shift:]
    else:
        out[-shift:] = arr[: n + shift]
    return out


# ------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a step function h, its C5 bound, and number of samples."""
    rng = np.random.default_rng(seed)

    # -----------------------------------------------------------------
    # Initialise h
    try:
        h = np.array(initial_h_values, dtype=float)  # type: ignore[name-defined]
    except NameError:
        # fallback: 200 points, half ones half zeros, shuffled
        n = 200
        h = np.zeros(n, dtype=float)
        h[: n // 2] = 1.0
        rng.shuffle(h)

    n = h.shape[0]
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    h = _project_sum_box(h, target_sum)

    best_h = h.copy()
    best_c5, _ = _c5_and_corr(best_h)

    start_time = time.time()
    step = 0.3
    min_step = 1e-7
    it = 0

    # -----------------------------------------------------------------
    while True:
        if time.time() - start_time >= budget_s:
            break

        c5_cur, corr = _c5_and_corr(h)
        if c5_cur < best_c5 - 1e-12:
            best_c5 = c5_cur
            best_h = h.copy()
            if best_c5 <= 0.3808:
                break  # reached target bound

        # Lag that attains the maximum correlation
        lag = int(np.argmax(corr) - (n - 1))

        # Gradient of the maximal term (sub‑gradient of the max)
        shift_plus = _shift_zero(h, lag)
        shift_minus = _shift_zero(h, -lag)
        dx = 2.0 / n
        grad = dx * (1.0 - (shift_plus + shift_minus))

        # --------------------------------------------------------------
        # Backtracking line search on the projected sub‑gradient step
        accepted = False
        trial_step = step
        for _ in range(20):
            h_new = h - trial_step * grad
            h_new = _project_sum_box(h_new, target_sum)
            c5_new, _ = _c5_and_corr(h_new)
            if c5_new < c5_cur:
                accepted = True
                break
            trial_step *= 0.5
            if trial_step < min_step:
                break

        if accepted:
            h = h_new
            step = trial_step * 1.2  # modestly increase after success
        else:
            step *= 0.5
            # occasional random perturbation to escape shallow minima
            if rng.random() < 0.06:
                i, j = rng.choice(n, size=2, replace=False)
                delta = (rng.random() - 0.5) * 0.08  # ≈[-0.04,0.04]
                tmp = h.copy()
                tmp[i] = np.clip(tmp[i] + delta, 0.0, 1.0)
                actual_delta = tmp[i] - h[i]
                tmp[j] = np.clip(tmp[j] - actual_delta, 0.0, 1.0)
                tmp = _project_sum_box(tmp, target_sum)
                c5_tmp, _ = _c5_and_corr(tmp)
                if c5_tmp < best_c5:
                    best_c5 = c5_tmp
                    best_h = tmp.copy()
                if c5_tmp < c5_cur:
                    h = tmp

        it += 1

        # Periodic full restart using a fresh binary pattern (optional)
        if it % 8000 == 0:
            new_h = np.zeros(n, dtype=float)
            new_h[: n // 2] = 1.0
            rng.shuffle(new_h)
            h = _project_sum_box(new_h.astype(float), target_sum)

    final_c5, _ = _c5_and_corr(best_h)
    return best_h.tolist(), float(final_c5), int(n)
