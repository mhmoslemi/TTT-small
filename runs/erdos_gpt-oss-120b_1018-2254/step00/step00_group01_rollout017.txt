import time
import numpy as np

def _project_bounds_sum(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project vector v onto the set {0 <= x_i <= 1, sum(x) = target_sum}.
    Uses a simple bisection on the threshold λ such that
    sum( clip(v_i - λ, 0, 1) ) = target_sum.
    """
    lo = -np.max(v) - 2.0   # all entries become 1
    hi = np.max(v) + 2.0    # all entries become 0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)

def _c5_from_h(h: np.ndarray) -> float:
    """
    Compute the C5 value for a candidate h.
    Falls back to a manual correlation if evaluate_erdos_solution is not usable.
    """
    n = h.shape[0]
    try:
        # Provided helper returns a tuple, first entry is the bound.
        return evaluate_erdos_solution(h, n)[0]
    except Exception:
        full = np.correlate(h, 1.0 - h, mode='full')
        return np.max(full) * (2.0 / n)

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] with ∫h = 1 and 0 ≤ h ≤ 1
    to minimize C5 = max_k ∫ h(x)(1−h(x+k))dx.

    Returns:
        (h_values, c5_bound, n_points)
    """
    np.random.seed(seed)

    # --- initialise ---------------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64).copy()
        n_points = h.size
    else:
        n_points = 101                     # default modest size
        h = np.full(n_points, 0.5, dtype=np.float64)

    target_sum = n_points / 2.0
    h = _project_bounds_sum(h, target_sum)

    cur_c5 = _c5_from_h(h)
    best_h = h.copy()
    best_c5 = cur_c5

    start_time = time.time()
    it = 0
    last_improve_it = 0

    # --- optimization loop --------------------------------------------------
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break
        if best_c5 < 0.3808:               # target reached, stop early
            break

        # worst shift (max correlation) for current h
        full_corr = np.correlate(h, 1.0 - h, mode='full')
        idx_max = int(np.argmax(full_corr))
        k = idx_max - (n_points - 1)       # shift value (negative allowed)

        # subgradient of the worst term
        grad = np.zeros_like(h)
        if k >= 0:
            grad[:n_points - k] += 1.0 - h[k:]
            grad[k:] += -h[:n_points - k]
        else:
            kk = -k
            grad[kk:] += 1.0 - h[:n_points - kk]
            grad[:n_points - kk] += -h[kk:]

        # decide move type
        if np.random.rand() < 0.3:
            # stochastic perturbation
            cand = h + np.random.normal(scale=0.08, size=n_points)
        else:
            # projected subgradient step with decaying step size
            step = 0.12 / (1.0 + it * 0.0015)
            cand = h - step * grad

        # enforce feasibility
        cand = _project_bounds_sum(cand, target_sum)

        # evaluate candidate
        cand_c5 = _c5_from_h(cand)

        # accept if it improves the current point
        if cand_c5 < cur_c5:
            h = cand
            cur_c5 = cand_c5
        else:
            # occasional uphill acceptance (simulated annealing spirit)
            if it % 500 == 0 and np.random.rand() < 0.07:
                h = cand
                cur_c5 = cand_c5

        # track the best ever found
        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = cand.copy()
            last_improve_it = it

        # stagnation restart
        if it - last_improve_it > 30000:
            h = best_h + np.random.normal(scale=0.05, size=n_points)
            h = _project_bounds_sum(h, target_sum)
            cur_c5 = _c5_from_h(h)
            last_improve_it = it

        it += 1

    return (best_h, float(best_c5), n_points)
