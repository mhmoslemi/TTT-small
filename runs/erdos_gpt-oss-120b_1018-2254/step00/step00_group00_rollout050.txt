def repair_sum(v, target):
    # Clip
    v = np.clip(v, 0.0, 1.0)
    # Compute diff
    diff = target - v.sum()
    # If diff is near zero, return
    if abs(diff) < 1e-12:
        return v
    # iterative adjustment
    # Set tolerance
    for _ in range(100):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Need to increase some entries not at 1
            free = np.where(v < 1.0 - 1e-15)[0]
        else:
            free = np.where(v > 1e-15)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        # cap delta to respect bounds
        if diff > 0:
            max_increase = np.min(1.0 - v[free])
            if delta > max_increase:
                # set those entries to 1 and recompute diff
                v[free] = np.where(v[free] + delta > 1.0, 1.0, v[free] + delta)
                # alternative: set all to 1 and adjust
                # Actually easier: fill increment to max possible for each
                inc = np.minimum(delta, 1.0 - v[free])
                v[free] += inc
            else:
                v[free] += delta
        else:
            max_decrease = np.min(v[free])
            if -delta > max_decrease:
                dec = np.minimum(-delta, v[free])
                v[free] -= dec
            else:
                v[free] += delta  # delta negative
        # Clip again
        v = np.clip(v, 0.0, 1.0)
        diff = target - v.sum()
    # final adjustment for residual diff
    if abs(diff) > 1e-9:
        # adjust one element that can accommodate
        # iterate over indices
        for i in range(v.size):
            if diff > 0 and v[i] < 1.0:
                cap = 1.0 - v[i]
                adj = min(cap, diff)
                v[i] += adj
                diff -= adj
            elif diff < 0 and v[i] > 0.0:
                cap = v[i]
                adj = min(cap, -diff)
                v[i] -= adj
                diff += adj
            if abs(diff) < 1e-12:
                break
    # Final clip
    v = np.clip(v, 0.0, 1.0)
    # Ensure sum is exact within tolerance
    # If still not exact adjust first element
    residual = target - v.sum()
    if abs(residual) > 1e-12:
        # adjust first element within bounds
        term = v[0] + residual

import time

def _evaluate(h):
    """Return C5 for a discretized h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _repair_sum(v, target):
    """Clip to [0,1] and adjust to satisfy sum(v)=target (exactly)."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    if abs(diff) < 1e-12:
        return v
    # iterative redistribution
    for _ in range(100):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = np.where(v < 1.0 - 1e-15)[0]
        else:
            free = np.where(v > 1e-15)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        if diff > 0:
            inc = np.minimum(delta, 1.0 - v[free])
            v[free] += inc
        else:
            dec = np.minimum(-delta, v[free])
            v[free] -= dec
        v = np.clip(v, 0.0, 1.0)
        diff = target - v.sum()
    # final tiny adjustment
    if abs(diff) > 1e-9:
        for i in range(v.size):
            if diff > 0 and v[i] < 1.0:
                cap = 1.0 - v[i]
                adj = min(cap, diff)
                v[i] += adj
                diff -= adj
            elif diff < 0 and v[i] > 0.0:
                cap = v[i]
                adj = min(cap, -diff)
                v[i] -= adj
                diff += adj
            if abs(diff) < 1e-12:
                break
    # ensure exactness (tiny residual)
    residual = target - v.sum()
    if abs(residual) > 1e-12:
        v[0] = max(0.0, min(1.0, v[0] + residual))
    return v

def _random_feasible(n):
    """Generate a random feasible vector of length n."""
    v = np.random.rand(n)
    return _repair_sum(v, n / 2.0)

def _mutate(vec, step):

    n = vec.shape[0]
    new = vec.copy()
    # mutate ~20% of entries (at least one)
    k = max(1, int(0.2 * n))

    new[idx] += np.random.normal(scale=step, size=k)
    new = np.clip(new, 0.0, 1.0)
    return _repair_sum(new, n / 2.0)
