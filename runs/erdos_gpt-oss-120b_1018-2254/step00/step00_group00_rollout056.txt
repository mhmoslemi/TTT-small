import numpy as np
import random
import math
import time

def evaluate_h(h):
    """Return C5 bound for a candidate h."""
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def worst_shift_and_gradient(h):
    """Gradient of the worst (maximum) overlap shift."""
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)
    grad = np.zeros_like(h)
    if shift >= 0:
        s = shift
        grad[:n - s] += 1 - h[s:]
        grad[s:] -= h[:n - s]
    else:
        t = -shift
        grad[t:] += 1 - h[:n - t]
        grad[:n - t] -= h[t:]
    return grad

def project_onto_bounds_and_sum(v, target, lower=0.0, upper=1.0):
    """Project vector v onto the box [lower,upper]^n intersect {sum = target}."""
    v = np.clip(v, lower, upper)
    total = np.sum(v)
    if abs(total - target) <= 1e-12:
        return v
    lo, hi = -1.0, 1.0
    # expand search interval until it brackets the solution
    while np.sum(np.clip(v + lo, lower, upper)) > target:
        lo *= 2.0
    while np.sum(np.clip(v + hi, lower, upper)) < target:
        hi *= 2.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, lower, upper))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, lower, upper)

def random_feasible(n, target):
    """Generate a random feasible vector of length n."""
    v = np.random.rand(n)
    return project_onto_bounds_and_sum(v, target)

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h that lowers the Erdős C5 bound."""
    np.random.seed(seed)
    random.seed(seed)

    # Determine problem size and initial vector
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        n = init.shape[0]
    else:
        n = 45
        init = np.random.rand(n)

    target_sum = n / 2.0
    init = project_onto_bounds_and_sum(init, target_sum)

    # Initial population (small) and best tracking
    best_h = init.copy()
    best_val = evaluate_h(best_h)

    for _ in range(7):
        cand = random_feasible(n, target_sum)
        val = evaluate_h(cand)
        if val < best_val:
            best_val = val
            best_h = cand.copy()

    cur_h = best_h.copy()
    cur_val = best_val

    start_time = time.time()
    iteration = 0
    # Main stochastic projected gradient loop
    while time.time() - start_time < budget_s * 0.95:
        iteration += 1

        # Gradient step toward decreasing the worst shift
        grad = worst_shift_and_gradient(cur_h)
        lr = 0.05 / (1.0 + iteration / 5000.0)
        sigma = 0.05 / (1.0 + iteration / 5000.0)

        proposal = cur_h - lr * grad + sigma * np.random.randn(n)
        proposal = project_onto_bounds_and_sum(proposal, target_sum)
        prop_val = evaluate_h(proposal)

        # Accept if improvement, otherwise Metropolis-like acceptance
        if prop_val < cur_val:
            cur_h, cur_val = proposal, prop_val
        else:
            temp = 0.01 / (1.0 + iteration / 5000.0)
            prob = math.exp((cur_val - prop_val) / max(temp, 1e-12))
            if random.random() < prob:
                cur_h, cur_val = proposal, prop_val

        # Update global best
        if prop_val < best_val:
            best_val, best_h = prop_val, proposal.copy()

        # Occasional exploration resets

            if random.random() < 0.2:
                cur_h, cur_val = best_h.copy(), best_val
            else:
                cur_h = random_feasible(n, target_sum)
                cur_val = evaluate_h(cur_h)

    # Final recompute to ensure consistency
    best_val = evaluate_h(best_h)
    return best_h, float(best_val), n
