def propose_move(h, rng, step):
    N = len(h)
    i, j = rng.choice(N, size=2, replace=False)
    delta = rng.uniform(-step, step)
    # ensure delta does not push outside [0,1]
    # h_i_new = h[i] + delta, h_j_new = h[j] - delta
    # constraints: 0 <= h_i_new <= 1; 0 <= h_j_new <=1.
    # So allowed delta must satisfy:
    # -h[i] <= delta <= 1 - h[i] (to keep h_i within)
    # -(1 - h[j]) <= delta <= h[j] (to keep h_j within)
    # Intersection of those intervals.
    lo = max(-h[i], -(1 - h[j]))
    hi = min(1 - h[i], h[j])
    if lo > hi:
        # cannot move; return None
        return None
    # Clip delta to [lo, hi]
    delta = max(lo, min(hi, delta))
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    # sum remains unchanged, bounds satisfied.
    return new_h

def propose_move(h, rng, max_delta):
    N = len(h)
    i, j = rng.integers(N, size=2, endpoint=False)
    # ensure i != j
    while i == j:
        j = rng.integers(N)
    # compute allowed delta range
    lo = max(-h[i], -(1 - h[j]))
    hi = min(1 - h[i], h[j])
    if lo >= hi:
        return None
    delta = rng.uniform(-max_delta, max_delta)
    # clip delta to feasible range
    delta = min(max(delta, lo), hi)
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    return new_h
```

import time
import numpy as np

def _project_to_constraints(v, target_sum, max_iter=60):
    """Project vector v onto {0<=h<=1, sum(h)=target_sum}."""
    lo, hi = -1e6, 1e6
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def _compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated via correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def _propose_move(h, rng, max_delta):
    """Return a new feasible h by moving mass between two indices."""
    n = h.shape[0]
    i, j = rng.integers(n, size=2)
    while i == j:
        j = rng.integers(n)
    # feasible delta range so that both entries stay in [0,1]
    lo = max(-h[i], -(1.0 - h[j]))
    hi = min(1.0 - h[i], h[j])
    if lo >= hi:
        return None
    delta = rng.uniform(-max_delta, max_delta)
    # clip to feasible interval
    delta = max(lo, min(hi, delta))
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    return new_h

def _random_feasible(rng, n, target_sum):
    """Generate a random vector in [0,1]^n with prescribed sum."""
    v = rng.random(n)
    return _project_to_constraints(v, target_sum)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    # Determine problem size and an initial feasible vector
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)
        n = init.shape[0]
        target = n / 2.0
        init = np.clip(init, 0.0, 1.0)
        if not np.isclose(init.sum(), target):
            init = _project_to_constraints(init, target)
        best_h = init.copy()
        best_c5 = _compute_c5(best_h)
    else:
        n = 70
        target = n / 2.0
        best_h = _random_feasible(rng, n, target)
        best_c5 = _compute_c5(best_h)

    # Core stochastic hill‑climbing loop
    cur_h = best_h.copy()
    cur_c5 = best_c5
    start = time.time()
    max_delta_initial = 0.3
    it = 0
    # schedule for reducing step size
    while time.time() - start < budget_s:
        elapsed = time.time() - start
        remaining = budget_s - elapsed
        # linear decay of max_delta
        max_delta = max_delta_initial * (1.0 - elapsed / budget_s) + 0.001
        cand = _propose_move(cur_h, rng, max_delta)
        if cand is None:
            continue
        cand_c5 = _compute_c5(cand)
        it += 1
        # Greedy acceptance
        if cand_c5 < cur_c5:
            cur_h, cur_c5 = cand, cand_c5
            if cand_c5 < best_c5:
                best_h, best_c5 = cand.copy(), cand_c5
                # early exit if good enough
                if best_c5 <= 0.3808:
                    break
        else:
            # occasional uphill move (annealing)
            temperature = max(0.001, remaining / budget_s * 0.02)
            prob = np.exp(-(cand_c5 - cur_c5) / temperature)
            if rng.random() < prob:
                cur_h, cur_c5 = cand, cand_c5

        # periodic random restarts to diversify search
        if it % 5000 == 0:
            # 30 % chance to jump to a fresh random point
            if rng.random() < 0.3:
                cur_h = _random_feasible(rng, n, target)
                cur_c5 = _compute_c5(cur_h)
            else:
                cur_h = best_h.copy()
                cur_c5 = best_c5

    # Ensure final vector respects constraints exactly
    best_h = np.clip(best_h, 0.0, 1.0)
    if not np.isclose(best_h.sum(), n / 2.0):
        best_h = _project_to_constraints(best_h, n / 2.0)
    best_c5 = _compute_c5(best_h)
    return best_h, float(best_c5), int(n)
