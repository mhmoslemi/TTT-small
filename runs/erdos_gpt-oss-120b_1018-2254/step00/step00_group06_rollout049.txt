def repair_vector(v, n):
    # ensure bounds [0,1] and sum = n/2
    target = n/2.0
    x = np.clip(v, 0.0, 1.0)
    # We'll adjust via algorithm described
    total = x.sum()
    diff = target - total
    # tolerance
    max_iter = 1000
    it = 0
    while abs(diff) > 1e-8 and it < max_iter:
        if diff > 0:
            idx = np.where(x < 1 - 1e-12)[0]
            if len(idx) == 0:
                break
            i = np.random.choice(idx)
            add = min(diff, 1 - x[i])
            x[i] += add
            diff -= add
        else:
            idx = np.where(x > 1e-12)[0]
            if len(idx) == 0:
                break
            i = np.random.choice(idx)
            sub = min(-diff, x[i])
            x[i] -= sub
            diff += sub
        it += 1
    # final small adjustment
    if abs(diff) > 1e-12:
        # find index that can handle diff
        if diff > 0:
            idx = np.where(x < 1 - 1e-12)[0]
        else:
            idx = np.where(x > 1e-12)[0]
        if len(idx) > 0:

        delta = target - total

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def mutate_vector(v, mutation_rate=0.01, sigma=0.1):
    # Gaussian mutation per gene with prob mutation_rate
    mask = np.random.rand(v.shape[0]) < mutation_rate
    if np.any(mask):
        noise = np.random.randn(mask.sum()) * sigma
        v = v.copy()
        v[mask] += noise
    return v

import time
import numpy as np

def repair_vector(v, n):
    """Project vector v onto the hypercube [0,1]^n with sum = n/2."""
    target = n / 2.0
    x = np.clip(v, 0.0, 1.0)
    diff = target - x.sum()
    # Simple iterative redistribution
    max_iter = 1000
    it = 0
    while abs(diff) > 1e-10 and it < max_iter:
        if diff > 0:
            # need to increase some entries
            idx = np.where(x < 1 - 1e-12)[0]
            if idx.size == 0:
                break
            i = np.random.choice(idx)
            inc = min(diff, 1.0 - x[i])
            x[i] += inc
            diff -= inc
        else:
            # need to decrease some entries
            idx = np.where(x > 1e-12)[0]
            if idx.size == 0:
                break
            i = np.random.choice(idx)
            dec = min(-diff, x[i])
            x[i] -= dec
            diff += dec
        it += 1
    # final tiny correction on a single coordinate if needed
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(x < 1 - 1e-12)[0]
        else:
            idx = np.where(x > 1e-12)[0]
        if idx.size:
            i = idx[0]
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    # ensure exact sum by adjusting first element (tiny rounding)
    total = x.sum()
    if abs(total - target) > 1e-9:
        delta = target - total
        x[0] = np.clip(x[0] + delta, 0.0, 1.0)
    return x

def compute_c5(h):
    """Return the overlap constant for vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def mutate_vector(v, mutation_rate=0.01, sigma=0.1):
    """Gaussian mutation on each coordinate with given probability."""
    mask = np.random.rand(v.shape[0]) < mutation_rate
    if np.any(mask):
        noise = np.random.randn(mask.sum()) * sigma
        v = v.copy()
        v[mask] += noise
    return v
