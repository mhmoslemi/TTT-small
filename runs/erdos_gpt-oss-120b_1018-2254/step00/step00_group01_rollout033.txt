def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode='full') * dx

def random_pair_move(h, rng, step_scale):
    n = len(h)
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    # compute feasible delta interval
    hi, hj = h[i], h[j]
    lo = max(-hi, hj - 1.0)
    hi_delta = min(1.0 - hi, hj)
    # restrict move magnitude
    max_move = step_scale
    lo = max(lo, -max_move)

    delta = rng.uniform(lo, hi_delta)

def target_move(h, shift, rng, max_step):
    n = len(h)
    # compute region indices for shift
    if shift >= 0:
        region = range(0, n - shift)
    else:
        region = range(-shift, n)
    region = list(region)
    # If region empty or full, fallback to random move.
    if not region or len(region) == n:
        return None
    # pick i from region
    i = rng.choice(region)
    # pick j from outside region
    complement = [idx for idx in range(n) if idx not in region]
    if not complement:
        return None
    j = rng.choice(complement)
    hi, hj = h[i], h[j]
    # compute feasible positive delta (increase i, decrease j)
    lo = max(0.0, -hj)  # but we want positive delta, and ensure j - delta >=0.
    # Actually delta must be >0, <= min(1-hi, hj)
    max_delta = min(1.0 - hi, hj)
    if max_delta <= 0:
        return None
    # restrict delta magnitude
    delta_max = min(max_delta, max_step)
    if delta_max <= 0:
        return None
    delta = rng.uniform(0.0, delta_max)
    new_h = h.copy()

    return new_h

    j = rng.integers(0, n-1)
    while j == i:
        j = rng.integers(0, n)
    hi, hj = h[i], h[j]

import time
import numpy as np

def compute_c5_and_shift(h):
    n = h.shape[0]
    dx = 2.0 / n
    auto = np.correlate(h, h, mode='full') * dx
    min_idx = int(np.argmin(auto))
    min_val = auto[min_idx]
    c5 = 1.0 - min_val
    shift = min_idx - (n - 1)
    return c5, shift

def pairwise_move(h, rng, max_step):
    n = h.shape[0]
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    while j == i:
        j = rng.integers(0, n)
    hi = h[i]
    hj = h[j]
    lo = max(-hi, hj - 1.0)
    hi_delta = min(1.0 - hi, hj)
    # limit magnitude
    lo = max(lo, -max_step)
    hi_delta = min(hi_delta, max_step)
    if lo >= hi_delta:
        return None
    delta = rng.uniform(lo, hi_delta)
    new_h = h.copy()
    new_h[i] = hi + delta
    new_h[j] = hj - delta
    if new_h[i] < -1e-12 or new_h[i] > 1.0 + 1e-12 or new_h[j] < -1e-12 or new_h[j] > 1.0 + 1e-12:
        return None
    return new_h

def target_move(h, shift, rng, max_step):
    n = h.shape[0]
    if shift >= 0:
        region_start = 0
        region_end = n - shift
    else:
        region_start = -shift
        region_end = n
    region_len = region_end - region_start
    if region_len <= 0 or region_len == n:
        return None
    i_offset = rng.integers(0, region_len)
    i = region_start + i_offset
    # choose j from complement
    left_len = region_start
    right_len = n - region_end
    if left_len + right_len == 0:
        return None
    if left_len == 0:
        j = rng.integers(region_end, n)
    elif right_len == 0:
        j = rng.integers(0, region_start)
    else:

    lo = max(-hi, hj - 1.0)
    hi_delta = min(1.0 - hi, hj)
    lo = max(lo, -max_step)
    hi_delta = min(hi_delta, max_step)
    if lo >= hi_delta:
        return None
    delta = rng.uniform(lo, hi_delta)
    new_h = h.copy()
    new_h[i] = hi + delta
    new_h[j] = hj - delta
    if new_h[i] < -1e-12 or new_h[i] > 1.0 + 1e-12 or new_h[j] < -1e-12 or new_h[j] > 1.0 + 1e-12:
        return None
    return new_h

def target_move(h, shift, rng, max_step):
    n = h.shape[0]
    if shift >= 0:
        region_start = 0
        region_end = n - shift
    else:
        region_start = -shift
        region_end = n
    region_len = region_end - region_start
    if region_len <= 0 or region_len == n:
        return None
    i = region_start + rng.integers(0, region_len)
    left_len = region_start
    right_len = n - region_end
    if left_len + right_len == 0:
        return None
    if left_len == 0:
        j = rng.integers(region_end, n)
    elif right_len == 0:
        j = rng.integers(0, region_start)
    else:
        if rng.random() < left_len / (left_len + right_len):
            j = rng.integers(0, region_start)
        else:
            j = rng.integers(region_end, n)
    hi = h[i]
    hj = h[j]
    max_delta = min(1.0 - hi, hj)  # increase i, decrease j
    if max_delta <= 0:
        return None
    delta = rng.uniform(0.0, min(max_delta, max_step))
    new_h = h.copy()
    new_h[i] = hi + delta
    new_h[j] = hj - delta
    if new_h[i] < -1e-12 or new_h[i] > 1.0 + 1e-12 or new_h[j] < -1e-12 or new_h[j] > 1.0 + 1e-12:
        return None
    return new_h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=np.float64).copy()
        n = h0.shape[0]
    else:
        n = 50
        h0 = np.full(n, 0.5, dtype=np.float64)

    target_sum = n / 2.0
    cur_sum = h0.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        diff = target_sum - cur_sum
        h0[0] = np.clip(h0[0] + diff, 0.0, 1.0)

    cur_h = h0.copy()
    cur_c5, cur_shift = compute_c5_and_shift(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    start_time = time.time()
    T0 = 0.01
    Tmin = 1e-7
    max_step_initial = 0.5
    min_step = 1e-4
    target_move_prob = 0.2

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.98:
            break
        frac = elapsed / budget_s
        max_step = max_step_initial * (1.0 - frac) + min_step * frac
        T = T0 * (1.0 - frac) + Tmin * frac

        if rng.random() < target_move_prob:
            new_h = target_move(cur_h, cur_shift, rng, max_step)
        else:
            new_h = pairwise_move(cur_h, rng, max_step)

        if new_h is None:
            continue

        new_c5, new_shift = compute_c5_and_shift(new_h)

        if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / max(T, 1e-12)):
            cur_h = new_h
            cur_c5 = new_c5
            cur_shift = new_shift
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()

    final_sum = best_h.sum()
    diff = target_sum - final_sum
    if abs(diff) > 1e-9:
        best_h[0] = np.clip(best_h[0] + diff, 0.0, 1.0)

    return best_h.tolist(), float(best_c5), n
