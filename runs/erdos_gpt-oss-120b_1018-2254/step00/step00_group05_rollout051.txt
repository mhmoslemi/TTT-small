def shift_array(arr, shift):
    # shift positive: shift right (increase index), fill zeros at start.
    # returns arr_shifted[i] = arr[i-shift] if i-shift in bounds else 0
    N = arr.shape[0]
    if shift > 0:

import numpy as np
import time

def project_onto_box_sum(v, target):
    lo = -np.max(v) - 1.0  # ensure sum > target
    hi = np.max(v) + 1.0   # ensure sum < target
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def shift_arr(arr, s):
    N = arr.shape[0]
    if s > 0:
        out = np.empty_like(arr)
        out[:s] = 0.0
        out[s:] = arr[:N - s]
        return out
    elif s < 0:
        ss = -s
        out = np.empty_like(arr)
        out[-ss:] = 0.0
        out[:N - ss] = arr[ss:]
        return out
    else:
        return arr.copy()

def compute_c5(h):
    N = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / N)

def grad_of_max_corr(h):
    N = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (N - 1)
    grad = np.ones(N) - shift_arr(h, shift) - shift_arr(h, -shift)
    return grad * (2.0 / N)

def upsample_h(h, new_n):
    old_n = h.shape[0]
    if old_n == new_n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    target_sum = None
    # Determine number of points
    base_n = 200
    if 'initial_h_values' in globals():
        init_h_raw = initial_h_values.astype(np.float64)
        n0 = init_h_raw.shape[0]
        # Use max of given and base for better resolution
        n_points = max(base_n, n0)
        # Upsample if needed
        h0 = upsample_h(init_h_raw, n_points)
    else:
        n_points = base_n
        h0 = rng.random(n_points)
    # Enforce box and sum constraints via projection
    target_sum = n_points / 2.0
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = project_onto_box_sum(h0, target_sum)

    best_h = h0.copy()
    best_score = compute_c5(best_h)

    h = h0.copy()
    start = time.time()
    iter_no = 0
    no_improve = 0
    # Hyperparameters
    step0 = 0.5
    while True:
        now = time.time()
        if now - start > budget_s * 0.97:
            break
        # compute gradient of max corr
        grad = grad_of_max_corr(h)
        step = step0 / np.sqrt(iter_no + 1)
        h_new = h - step * grad
        h_new = project_onto_box_sum(h_new, target_sum)
        new_score = compute_c5(h_new)
        cur_score = compute_c5(h)
        # simple acceptance: if better accept, else maybe accept with prob
        if new_score < cur_score:
            h = h_new
        else:
            # probabilistic acceptance (simulated annealing)
            temp = max(0.01, 0.1 * (1 - (now - start) / budget_s))
            if rng.random() < np.exp(-(new_score - cur_score) / temp):
                h = h_new
        # update best
        if new_score < best_score:
            best_score = new_score
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        # restart if stuck
        if no_improve >= 200:
            # random restart keeping best as reference
            h = project_onto_box_sum(rng.random(n_points), target_sum)
            no_improve = 0
        iter_no += 1

    # final projection and evaluate
    best_h = project_onto_box_sum(np.clip(best_h, 0.0, 1.0), target_sum)
    final_score = compute_c5(best_h)
    return best_h, float(final_score), n_points

import numpy as np
import time

def project_onto_box_sum(v, target):
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def shift_arr(arr, s):
    N = arr.shape[0]
    if s > 0:
        out = np.empty_like(arr)
        out[:s] = 0.0
        out[s:] = arr[:N - s]
        return out
    elif s < 0:
        ss = -s
        out = np.empty_like(arr)
        out[-ss:] = 0.0
        out[:N - ss] = arr[ss:]
        return out
    else:
        return arr.copy()

def compute_c5(h):
    N = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / N)

def grad_and_max_corr(h):
    N = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (N - 1)
    grad = np.ones(N) - shift_arr(h, shift) - shift_arr(h, -shift)
    # gradient of the C5 bound (scaled by dx)
    return grad * (2.0 / N), corr[idx] * (2.0 / N)

def upsample_h(h, new_n):
    old_n = h.shape[0]
    if old_n == new_n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    base_n = 200

    if 'initial_h_values' in globals():
        init_raw = np.asarray(initial_h_values, dtype=np.float64)
        n0 = init_raw.shape[0]
        n_points = max(base_n, n0)
        h = upsample_h(init_raw, n_points)
    else:
        n_points = base_n
        h = rng.random(n_points)

    target_sum = n_points / 2.0
    h = np.clip(h, 0.0, 1.0)
    h = project_onto_box_sum(h, target_sum)

    best_h = h.copy()
    best_score = compute_c5(best_h)
    cur_score = best_score

    start = time.time()
    iter_no = 0
    stagnation = 0
    step0 = 0.5

    while True:
        if time.time() - start > budget_s * 0.96:
            break

        grad, _ = grad_and_max_corr(h)
        step = step0 / np.sqrt(iter_no + 1)
        h_new = h - step * grad
        h_new = project_onto_box_sum(h_new, target_sum)
        new_score = compute_c5(h_new)

        if new_score < cur_score:
            h = h_new
            cur_score = new_score
            stagnation = 0
        else:
            elapsed = time.time() - start
            temp = max(0.001, 0.2 * (1 - elapsed / budget_s))
            if rng.random() < np.exp(-(new_score - cur_score) / temp):
                h = h_new
                cur_score = new_score
                stagnation = 0
            else:
                stagnation += 1

        if cur_score < best_score:
            best_score = cur_score
            best_h = h.copy()

        if stagnation >= 300:
            h = project_onto_box_sum(rng.random(n_points), target_sum)
            cur_score = compute_c5(h)
            stagnation = 0

        iter_no += 1

    best_h = project_onto_box_sum(np.clip(best_h, 0.0, 1.0), target_sum)
    final_score = compute_c5(best_h)
    return best_h, float(final_score), n_points
