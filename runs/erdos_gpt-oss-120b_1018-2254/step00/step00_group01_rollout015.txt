def _project_to_feasible(v: np.ndarray, target: float) -> np.ndarray:
    """Project vector v onto [0,1] box with sum == target."""
    v = np.clip(v, 0.0, 1.0)
    # iterative water-filling
    for _ in range(200):
        s = v.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        mask = (v > 1e-12) & (v < 1.0 - 1e-12)
        free = np.where(mask)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    # final tiny adjustment if needed
    diff = target - v.sum()
    if abs(diff) > 1e-12:
        for i in range(len(v)):
            if diff == 0:

                diff -= add
            elif diff < 0 and v[i] > 0.0:
                add = max(diff, -v[i])

                prob = math.exp(-(new_c5 - cur_c5) / temp)
                accept = rng.random() < prob
            else:
                accept = False

        if accept:
            h_current = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()

    return (best_h, float(best_c5), n_points)
