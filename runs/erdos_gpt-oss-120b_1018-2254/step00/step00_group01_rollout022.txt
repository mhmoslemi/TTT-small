    new_h[j] -= delta

import time
import numpy as np

def compute_c5(h):
    """Return the C5 value for a given step function array h."""
    N = h.shape[0]
    dx = 2.0 / N
    # correlation between h and 1-h (full mode)
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def shift_vec(v, s):
    """Shift vector v by s places with zero padding (no wraparound)."""
    N = v.shape[0]
    out = np.empty_like(v)
    if s >= 0:
        out[:N - s] = v[s:]
        out[N - s:] = 0.0
    else:
        s = -s
        out[s:] = v[:N - s]
        out[:s] = 0.0
    return out

def gradient_for_shift(h, shift):
    """Gradient of overlap O_shift w.r.t. h (for a fixed shift)."""
    return shift_vec(h, shift) + shift_vec(h, -shift)

def enforce_sum(h, target):
    """Adjust h in-place so that its sum equals target while staying in [0,1]."""
    cur = np.sum(h)
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    N = h.size
    # Simple greedy adjust
    for i in range(N):
        if diff > 0:
            avail = 1.0 - h[i]
            inc = min(avail, diff)
            if inc > 0:
                h[i] += inc
                diff -= inc
        else:
            avail = h[i]
            dec = min(avail, -diff)
            if dec > 0:
                h[i] -= dec
                diff += dec
        if abs(diff) < 1e-12:
            break
    return h

def random_feasible(N):
    """Create a random feasible binary/half‑binary vector of length N with sum N/2."""
    h = np.zeros(N, dtype=np.float64)
    ones = N // 2
    h[:ones] = 1.0
    if N % 2 == 1:
        h[ones] = 0.5
        start = ones + 1
    else:
        start = ones
    np.random.shuffle(h)
    return h
