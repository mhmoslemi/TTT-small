def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, random
    start = time.time()
    random.seed(seed)
    np.random.seed(seed)
    # choose n_points
    n = 200   # or maybe 150?
    dx = 2.0 / n
    target_sum = n/2.0
    # define helper functions
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return corr.max()
    def project_onto_constraints(v):
        # Clip and adjust to meet sum target
        # We'll use bisection method for lambda
        lo, hi = -1.0, 1.0
        for _ in range(50):
            mid = (lo + hi) / 2.0
            h = np.clip(v - mid, 0.0, 1.0)
            s = h.sum()
            if s > target_sum:
                lo = mid
            else:
                hi = mid
        h = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
        # after bisection may be slightly off due to rounding; adjust
        # adjust by adding small constant to meet sum exactly while respecting bounds
        diff = target_sum - h.sum()
        if abs(diff) > 1e-10:
            # distribute diff uniformly to non-saturated indices
            idx = np.where((h > 0) & (h < 1))[0]
            if idx.size > 0:
                h[idx] += diff / idx.size
                h = np.clip(h, 0.0, 1.0)
            # else fallback: adjust first index with capacity
            # final correction: re-project if needed
            if abs(h.sum() - target_sum) > 1e-6:
                # simple scaling of all non-zero values
                scale = target_sum / h.sum()
                h = np.clip(h * scale, 0.0, 1.0)
                # final clip may disturb sum again; iterate once more
                h = np.clip(h, 0.0, 1.0)
                # final enforce sum by adding diff to some non-full index
                diff = target_sum - h.sum()
                for i in range(len(h)):
                    if diff == 0:
                        break
                    add = min(1.0 - h[i], max(-h[i], diff))
                    h[i] += add
                    diff -= add
        return h
    # optionally use initial_h_values if defined
    try:
        init = initial_h_values.copy()
        # Check length; if not matching n, we can resample via interpolation
    except NameError:
        init = None
    # If we have init, we need to adjust to current n
    if init is not None and len(init) != n:
        # interpolate to new size
        x_old = np.linspace(0, 2, len(init), endpoint=False) + 1.0/len(init) # or use simple...
        x_new = np.linspace(0, 2, n, endpoint=False) + 1.0/n
        from scipy.interpolate import interp1d
        f = interp1d(x_old, init, kind='linear', fill_value='extrapolate')
        init = f(x_new)
    if init is None:
        # create shape: ramp-length r=0.5 perhaps?
        r = 0.5
        h = np.zeros(n)
        x = (np.arange(n) + 0.5) * dx  # sample at midpoints
        L = 1.0 - r
        for i, xi in enumerate(x):
            d = abs(xi - 1.0)
            if d <= L/2:
                h[i] = 1.0
            elif d <= L/2 + r:
                h[i] = 1.0 - (d - L/2) / r
            else:
                h[i] = 0.0
        init = project_onto_constraints(h)
    else:
        init = project_onto_constraints(np.array(init))
    # Keep best solution
    best_h = init.copy()
    best_c5 = compute_c5(best_h)
    # simple hill climbing loop
    max_iters = 20000
    iteration = 0
    while time.time() - start < budget_s * 0.95 and iteration < max_iters:
        iteration += 1
        # propose move: choose two indices i,j, delta.
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # compute max transferable amount while maintaining bounds
        # Transfer eps from i to j
        max_transfer = min(best_h[i], 1.0 - best_h[j])
        if max_transfer <= 0:
            continue
        eps = np.random.rand() * max_transfer * 0.5  # random fraction of max
        # create new candidate
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        # candidate already satisfies bounds and sum constraint exactly
        c5 = compute_c5(cand)
        if c5 < best_c5:
            best_h = cand
            best_c5 = c5
        # else maybe accept with probability? skip.
        # Also occasionally random restart
        if iteration % 2000 == 0:
            # random restart: generate random vector
            v = np.random.rand(n)
            cand_restart = project_onto_constraints(v)
            c5r = compute_c5(cand_restart)
            if c5r < best_c5:
                best_h = cand_restart
                best_c5 = c5r
    # final forced projection to ensure sum
    best_h = project_onto_constraints(best_h)
    return (best_h, float(best_c5), n)

def project_onto_constraints(v):
    lo = -1.0
    hi = 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        h = np.clip(v - mid, 0.0, 1.0)
        s = h.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid

    # ensure sum exactly target via small adjustment
    diff = target_sum - h.sum()
    if abs(diff) > 1e-8:
        # distribute diff to elements not at bounds
        free = (h > 0) & (h < 1)
        count = free.sum()
        if count > 0:
            h[free] += diff / count
            # clip again
            h = np.clip(h, 0.0, 1.0)
        else:
            # fallback: scale all values proportionally, then clip
            h = h * (target_sum / h.sum())
            h = np.clip(h, 0.0, 1.0)
    return h

import time
import random
import numpy as np

def _project_onto_constraints(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project vector v onto the feasible set:
       0 <= h[i] <= 1 and sum(h) == target_sum."""
    lo, hi = -1.0, 2.0  # safely enclose the solution
    for _ in range(50):
        mid = (lo + hi) / 2.0
        h = np.clip(v - mid, 0.0, 1.0)
        s = h.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # tiny adjustment to hit the exact sum (avoid drift)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-9:
        free = (h > 0) & (h < 1)
        cnt = free.sum()
        if cnt > 0:
            h[free] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
        else:
            # fallback scaling
            h *= target_sum / h.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def _correlation_max(h: np.ndarray, dx: float) -> tuple:
    """Return (max_correlation, shift) where shift is the integer offset
       corresponding to the maximal value."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    max_val = float(corr[idx] * dx)
    shift = idx - (len(h) - 1)  # positive shift => h paired with left part of (1-h)
    return max_val, shift

def _neighbor_sum(h: np.ndarray, s: int) -> np.ndarray:
    """For a given integer shift s, return array G_i = h_{i-s} + h_{i+s}
       with zero padding outside the interval."""
    n = h.shape[0]
    if s >= 0:
        left = np.concatenate((np.zeros(s), h[: n - s]))
        right = np.concatenate((h[s:], np.zeros(s)))
    else:
        s_abs = -s
        left = np.concatenate((h[s_abs:], np.zeros(s_abs)))          # i-s = i+|s|
        right = np.concatenate((np.zeros(s_abs), h[: n - s_abs]))   # i+s = i-|s|
    return left + right

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a step function h on [0,2] with ∫h=1 that minimizes
       max_k ∫ h(x)(1‑h(x+k))dx.
       Returns (h_array, best_c5, n_points)."""
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    time_limit = budget_s * 0.95  # keep a safety margin

    # ------------------------------------------------------------------
    # Determine problem size and starting point
    # ------------------------------------------------------------------
    try:
        init = np.array(initial_h_values, copy=True)  # may be defined by harness
    except NameError:
        init = None

    if init is not None:

        dx = 2.0 / n
        target = n / 2.0

    else:
        # default size (small enough for fast iterations)
        n = 200
        dx = 2.0 / n
        target = n / 2.0
        h = np.full(n, 0.5)  # uniform feasible start

    # compute initial objective
    best_h = h.copy()
    best_c5, best_shift = _correlation_max(best_h, dx)

    # ------------------------------------------------------------------
    # Helper to possibly replace current best with a new candidate
    # ------------------------------------------------------------------
    def _update_candidate(cand):

        val, sh = _correlation_max(cand, dx)
        if val < best_c5:
            best_h = cand
            best_c5 = val
            best_shift = sh
        return val, sh

    # ------------------------------------------------------------------
    # Main directed hill‑climbing loop

    it = 0
    no_improve = 0
    while time.time() - start_time < time_limit:
        it += 1

        # compute worst shift (the one giving the current maximum correlation)
        cur_c5, cur_shift = _correlation_max(best_h, dx)

        # neighbour sum G_i = h_{i-s} + h_{i+s}
        G = _neighbor_sum(best_h, cur_shift)

        # choose a low‑score index with positive mass to give away
        low_mask = best_h > 1e-12
        if np.any(low_mask):

            G_low[~low_mask] = np.inf

            G_high[~high_mask] = -np.inf
