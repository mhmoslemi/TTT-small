import time
from scipy import optimize as opt

_target_sum = None

def compute_c5(h):
    N = h.shape[0]
    dx = 2.0 / N
    # correlation of h with 1-h
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_onto_simplex_box(x, target, lower=0.0, upper=1.0):
    # Clip to box first
    x = np.clip(x, lower, upper)
    # Binary search for lambda such that sum(clipped(x - lambda)) == target
    lo = -10.0
    hi = 10.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        y = np.clip(x - lam, lower, upper)
        s = y.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    y = np.clip(x - (lo + hi) / 2.0, lower, upper)

    return np.sum(x) - _target_sum

def local_opt(x0):
    N = x0.shape[0]
    bounds = [(0.0, 1.0)] * N

import time
from scipy import optimize as opt

_target_sum = None  # will be set in run()


def compute_c5(h):
    """Return max_{k} ∫ h(x)(1‑h(x+k)) dx for the discretised step function."""
    N = h.shape[0]
    dx = 2.0 / N
    # correlation of h with (1‑h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def project_onto_simplex_box(x, target, lower=0.0, upper=1.0):
    """
    Project a vector onto the box [lower,upper]^n intersected with the hyperplane
    sum(x)=target using a simple water‑filling (binary search) algorithm.
    """
    x = np.clip(x, lower, upper)
    lo, hi = -10.0, 10.0  # wide enough to force extremes
    for _ in range(60):

        y = np.clip(x - lam, lower, upper)
        s = y.sum()

            lo = lam
        else:
            hi = lam
    y = np.clip(x - (lo + hi) / 2.0, lower, upper)
    return y


def constraint_sum_eq(x):
    """Equality constraint: sum(x) == _target_sum."""
    return np.sum(x) - _target_sum


def local_opt(x0):
    """Run a short SLSQP refinement starting from a feasible point."""
    N = x0.shape[0]
    bounds = [(0.0, 1.0)] * N
