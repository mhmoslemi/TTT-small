def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(v, target):
    # binary search for lambda such that sum(clamp(v+lambda,0,1)) = target
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

    v = np.random.rand(n)

import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(v, target):
    """Project vector v onto the box [0,1]^n while enforcing sum(v)=target."""
    lo, hi = -1.0, 1.0          # enough to drive all entries to 0 or 1
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def random_feasible(n, target):
    """Return a random vector in [0,1]^n with prescribed sum = target."""
    v = np.random.rand(n)
    return project_to_feasible(v, target)

def get_initial_h(n):
    """Interpolate the global initial_h_values (if present) to length n."""
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.shape[0]
        if n0 == n:
            h = h0.copy()
        else:
            x_old = np.linspace(0.0, 2.0, n0)
            x_new = np.linspace(0.0, 2.0, n)
            h = np.interp(x_new, x_old, h0)
        return project_to_feasible(h, n / 2.0)
    # fallback: random feasible vector
    return random_feasible(n, n / 2.0)

def objective_de(x, target):
    """Penalized objective for differential evolution."""
    c5 = compute_c5(x)
    pen = 1e3 * (np.sum(x) - target) ** 2
    return c5 + pen

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    best_h = None
    best_c5 = np.inf

    # start from the supplied construction if available
    if 'initial_h_values' in globals():
        try:
            h0 = np.array(initial_h_values, dtype=float)
            h0 = project_to_feasible(h0, h0.shape[0] / 2.0)
            c0 = compute_c5(h0)
            best_h, best_c5 = h0, c0
        except Exception:
            pass

    n_list = [45, 90, 180]

    for n in n_list:
        if time.time() - start > budget_s * 0.98:
            break

        target = n / 2.0
        bounds = [(0.0, 1.0)] * n

        # initial guess (interpolated if possible)
        init_h = get_initial_h(n)

        # DE parameters tuned per dimension
        if n <= 45:
            maxiter, popsize = 800, 15
        elif n <= 90:
            maxiter, popsize = 600, 12
        else:
            maxiter, popsize = 400, 10

        # --- global search ----------------------------------------------------
        try:
            de_res = opt.differential_evolution(
                func=lambda x: objective_de(x, target),
                bounds=bounds,
                maxiter=maxiter,
                popsize=popsize,
                polish=False,
                init='latinhypercube',
                seed=seed,
                updating='deferred',
                disp=False,
                tol=1e-6,
                strategy='best1bin',
            )
            de_x = de_res.x
        except Exception:
            de_x = init_h

        de_x = project_to_feasible(de_x, target)

        # --- local refinement -------------------------------------------------
        eq_cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}
        try:
