def project_onto_simplex_with_upper(v, target):
    # v: np.ndarray, any real numbers
    # target: desired sum, should be between 0 and n
    n = len(v)
    # clip to [0,1]
    v = np.clip(v, 0.0, 1.0)
    # compute sum
    cur_sum = np.sum(v)
    if abs(cur_sum - target) < 1e-12:
        return v
    # Use iterative water-filling until convergence
    # We'll allow small tolerance.
    max_iter = 1000
    for _ in range(max_iter):
        # Identify free indices (strictly interior)
        free = (v > 1e-12) & (v < 1 - 1e-12)
        free_idx = np.where(free)[0]
        if free_idx.size == 0:
            # No free variables; need to adjust by moving mass from ones to zeros
            # Distribute deficit to zeros or excess to ones
            # fallback: just return v (should already sum target). but may not.
            break
        # compute needed adjustment
        adjustment = (target - cur_sum) / free_idx.size
        v[free_idx] += adjustment
        # clip again
        v = np.clip(v, 0.0, 1.0)
        new_sum = np.sum(v)
        if abs(new_sum - target) < 1e-12:
            break
        cur_sum = new_sum
    # final fudge: if sum not exact, distribute remaining uniformly among non-blocked
    diff = target - np.sum(v)
    if abs(diff) > 1e-9:
        # distribute to indices not at bound in direction of diff sign
        if diff > 0:
            candidates = np.where(v < 1 - 1e-12)[0]
        else:
            candidates = np.where(v > 1e-12)[0]
        if candidates.size > 0:
            inc = diff / candidates.size
            v[candidates] += inc
            v = np.clip(v, 0.0, 1.0)
    # final clip
    return v

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Return the Erdős C5 value for a step function sampled in `h`."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_simplex_upper(v, target, tol=1e-12, max_iter=1000):
    """
    Euclidean projection of vector `v` onto the set:
        {x | 0 <= x_i <= 1, sum(x) = target}
    """
    v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) <= tol:
        return v
    for _ in range(max_iter):
        if s > target:
            free = v > tol
        else:
            free = v < 1.0 - tol
        idx = np.nonzero(free)[0]
        if idx.size == 0:
            break
        delta = (target - s) / idx.size
        v[idx] += delta
        v = np.clip(v, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) <= tol:
            break
    # Final tiny adjustment if needed
    diff = target - v.sum()
    if abs(diff) > 1e-9:
        if diff > 0:
            cand = np.where(v < 1.0 - tol)[0]
        else:
            cand = np.where(v > tol)[0]
        if cand.size:
            v[cand] += diff / cand.size
            v = np.clip(v, 0.0, 1.0)
    return v

def run(seed=42, budget_s=1100, **kwargs):

    """

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    try:

        # enforce feasibility
        h_init = project_onto_simplex_upper(h_init, n_init / 2.0)
        best_h = h_init.copy()
        best_c5 = compute_c5(best_h)
        best_n = n_init
    except Exception:
        # fallback: uniform 0.5 (automatically feasible)
        best_n = 45
        best_h = np.full(best_n, 0.5, dtype=float)
        best_c5 = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Candidate resolutions (more points may allow smaller overlap)
    # ------------------------------------------------------------------
    candidate_ns = sorted({best_n, 90, 180})
    for n in candidate_ns:
        if time.time() - start_time > budget_s:
            break

        # Build a feasible starting point for this resolution
        if n == best_n:
            h0 = best_h.copy()
        else:
            # Interpolate the current best onto the new grid
            x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(x_new, x_old, best_h)
            h0 = project_onto_simplex_upper(h0, n / 2.0)

        # --------------------------------------------------------------
