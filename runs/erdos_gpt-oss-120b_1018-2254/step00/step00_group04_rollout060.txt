def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time, math
    rng = np.random.default_rng(seed)
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 80
    target = n_points / 2.0
    dx = 2.0 / n_points

    # repair function
    def repair(h):
        h = np.clip(h, 0.0, 1.0)
        # water-filling to sum target
        total = h.sum()
        # adjust using simple scaling and clipping iteratively
        for _ in range(20):
            diff = target - total
            if abs(diff) < 1e-12:
                break
            # compute free indices
            free = (h > 0) & (h < 1)
            n_free = free.sum()
            if n_free == 0:
                # adjust all proportionally
                h += diff / n_points
                h = np.clip(h, 0.0, 1.0)
                total = h.sum()
                continue
            delta = diff / n_free
            h[free] += delta
            h = np.clip(h, 0.0, 1.0)
            total = h.sum()
        # final minor correction to enforce exactly sum
        total = h.sum()
        diff = target - total
        if abs(diff) > 1e-9:
            # distribute diff across all entries proportionally ensuring bounds
            for i in range(n_points):
                if diff == 0:
                    break
                # can add diff to any entry not at extreme
                if diff > 0 and h[i] < 1.0:
                    add = min(diff, 1.0 - h[i])
                    h[i] += add
                    diff -= add
                elif diff < 0 and h[i] > 0.0:
                    sub = min(-diff, h[i])
                    h[i] -= sub
                    diff += sub
        return h

    # define compute_c5
    def compute_c5(h):
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx

    # init candidate
    if 'initial_h_values' in globals():
        cur = repair(np.array(initial_h_values, dtype=float))
    else:
        cur = repair(rng.random(n_points))
    cur_score = compute_c5(cur)
    best = cur.copy()
    best_score = cur_score

    start = time.time()
    # parameters for SA
    max_iter = 10000000
    # initial temperature
    T0 = cur_score * 0.5
    # schedule parameters
    t = 0
    while True:
        elapsed = time.time() - start
        if elapsed > budget_s - 0.1:  # leave little margin
            break
        t += 1
        # temperature decrease
        T = max(1e-8, T0 * (1 - elapsed / budget_s))
        # propose mutation
        sigma = 0.1 * (1 - elapsed / budget_s) + 0.01  # reduce over time
        # random subset
        new = cur.copy()
        # mutate a fraction of entries
        idx = rng.choice(n_points, size=max(1, int(0.2 * n_points)), replace=False)
        new[idx] += rng.normal(0, sigma, size=idx.shape[0])
        new = repair(new)

            cur_score = compute_c5(cur)

def run(seed=42, budget_s=1100, **kwargs):

    target = n_points / 2.0
    dx = 2.0 / n_points

    def repair(h):
        # Clip to bounds
        h = np.clip(h, 0.0, 1.0)
        total = h.sum()
        # iterative water-filling
        for _ in range(50):
            diff = target - total
            if abs(diff) < 1e-12:
                break
            free = (h > 0) & (h < 1)
            n_free = free.sum()
            if n_free == 0:
                # distribute uniformly
                h += diff / n_points
                h = np.clip(h, 0.0, 1.0)
                total = h.sum()
                continue
            delta = diff / n_free
            h[free] += delta
            h = np.clip(h, 0.0, 1.0)
            total = h.sum()
        # final fine adjustment if needed
        total = h.sum()
        diff = target - total
        if abs(diff) > 1e-9:
            # distribute to any index not at bound
            for i in range(n_points):
                if diff == 0:
                    break
                if diff > 0 and h[i] < 1.0:
                    add = min(diff, 1.0 - h[i])
                    h[i] += add
                    diff -= add
                elif diff < 0 and h[i] > 0.0:
                    sub = min(-diff, h[i])
                    h[i] -= sub
                    diff += sub
        return h

    def compute_c5(h):
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx

    # Base shape initial guess
    i = np.arange(n_points, dtype=float)

    candidates.append(base_h)

    candidates.append(repair(rng.random(n_points)))
