    # move delta up to min(1-h[a], h[b]) (so that sum unchanged)
    max_up = 1 - h[a]
    max_down = h[b]

import time

def _eval_h(h: np.ndarray) -> float:
    """Return C5 for a candidate h."""
    n = h.shape[0]
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)

def _construct_h(half_idx, ones_set, n):
    """Build the full h-array from representation."""
    h = np.zeros(n, dtype=float)
    if ones_set:
        h[list(ones_set)] = 1.0
    if half_idx is not None:
        h[half_idx] = 0.5
    return h

    """Search for a step function h that minimizes the Erdős C5 constant."""
    start = time.time()
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # Determine problem size and starting point
    # ----------------------------------------------------------------------
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        target_sum = n / 2.0
        cur = h0.sum()
        if not np.isclose(cur, target_sum):
            diff = target_sum - cur
            for i in range(n):
                if 0.0 <= h0[i] + diff <= 1.0:
                    h0[i] += diff
                    break
        best_h = h0
        best_c5 = _eval_h(best_h)
    else:
        n = 47
        best_h = None
        best_c5 = np.inf

    target_sum = n / 2.0
    # Number of unit entries when we force a binary+half representation
    if n % 2 == 1:
        num_ones = (n - 1) // 2          # one half will be present
    else:
        num_ones = int(target_sum)       # pure binary

    # ----------------------------------------------------------------------
    # Phase 1: Random sampling (≈70% of time budget)
    # ----------------------------------------------------------------------
    time_random_end = start + budget_s * 0.70
    while time.time() < time_random_end:
        perm = rng.permutation(n)
        if n % 2 == 1:
            half_idx = int(perm[0])
            ones = perm[1:num_ones + 1]
        else:
            half_idx = None
            ones = perm[:num_ones]
        h = np.zeros(n, dtype=float)
        h[ones] = 1.0
        if half_idx is not None:
            h[half_idx] = 0.5
        c5 = _eval_h(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
            if best_c5 <= 0.3808:
                break

    # ----------------------------------------------------------------------
    # Fallback if nothing was generated
    # ----------------------------------------------------------------------
    if best_h is None:
        h = np.zeros(n, dtype=float)
        h[:num_ones] = 1.0
        if n % 2 == 1:
            h[num_ones] = 0.5
        best_h = h
        best_c5 = _eval_h(best_h)

    # ----------------------------------------------------------------------
    # Phase 2: Deterministic local improvement (exhaustive swaps)
    # ----------------------------------------------------------------------
    remaining = budget_s - (time.time() - start)
    if remaining > 2.0:
        # Build the discrete representation (set of ones and optional half)
        if n % 2 == 1:
            half_candidates = np.where(np.isclose(best_h, 0.5, atol=1e-12))[0]
            half_idx = int(half_candidates[0]) if half_candidates.size else None
        else:
            half_idx = None
        ones_set = set(np.where(best_h > 0.5)[0].tolist())

        # Ensure correct cardinalities
        while len(ones_set) > num_ones:
            ones_set.pop()
        while len(ones_set) < num_ones:
            pool = set(range(n)) - ones_set - ({half_idx} if half_idx is not None else set())
            if not pool:
                break
            ones_set.add(rng.choice(list(pool)))
        if n % 2 == 1 and half_idx is None:
            pool = set(range(n)) - ones_set
            half_idx = rng.choice(list(pool))

        # Local search loop
        improved = True
        while improved and (time.time() - start) < budget_s - 0.1:
            improved = False
            # Derive zero indices (excluding half if present)
            if half_idx is None:
                zeros = set(range(n)) - ones_set
            else:
                zeros = set(range(n)) - ones_set - {half_idx}

            # 1) Swap a 1 with a 0
            for i in list(ones_set):
                for j in list(zeros):
                    new_ones = ones_set.copy()
                    new_ones.remove(i)
                    new_ones.add(j)
                    h_new = _construct_h(half_idx, new_ones, n)
                    c5_new = _eval_h(h_new)
                    if c5_new < best_c5 - 1e-12:
                        best_c5 = c5_new
                        best_h = h_new
                        ones_set = new_ones
                        improved = True
                        break
                if improved:
                    break
            if improved:
                continue

            # 2) Move the half to a zero position
            if half_idx is not None:
                for j in list(zeros):
                    h_new = _construct_h(j, ones_set, n)
                    c5_new = _eval_h(h_new)
                    if c5_new < best_c5 - 1e-12:
                        best_c5 = c5_new
                        best_h = h_new
                        half_idx = j
                        zeros = set(range(n)) - ones_set - {half_idx}
                        improved = True
                        break
                if improved:
                    continue

            # 3) Swap half with a one (half becomes 1, that 1 becomes 0.5)
            if half_idx is not None:
                for i in list(ones_set):
                    new_ones = ones_set.copy()
                    new_ones.remove(i)
                    new_ones.add(half_idx)  # old half becomes a one
                    h_new = _construct_h(i, new_ones, n)  # new half at i
                    c5_new = _eval_h(h_new)
                    if c5_new < best_c5 - 1e-12:
                        best_c5 = c5_new
                        best_h = h_new
                        half_idx = i
                        ones_set = new_ones
                        improved = True
                        break

    # ----------------------------------------------------------------------
    # Return the best discovered configuration
    # ----------------------------------------------------------------------
    return (best_h, best_c5, n)
