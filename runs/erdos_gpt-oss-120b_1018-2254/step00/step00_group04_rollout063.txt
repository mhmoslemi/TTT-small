        # If no improvement, try a small random perturbation
        if not improved:
            cand = project(h_best + 0.02 * (rng.random(n_points) - 0.5))
            cand_c5 = compute_c5(cand)
            if cand_c5 < best_c5:
                h_best, best_c5 = cand, cand_c5

        # Early exit if we've already beaten the target bound
        if best_c5 < 0.3808:
            break

    # Final safety projection and evaluation
    h_best = project(h_best)
    best_c5 = compute_c5(h_best)

    return h_best, best_c5, n_points
