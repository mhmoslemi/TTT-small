def random_pairwise_move(h, max_step=0.2):
    n = len(h)
    i, j = random.sample(range(n), 2)
    hi = h[i]
    hj = h[j]
    lb = max(-hi, hj - 1.0)
    ub = min(1.0 - hi, hj)
    if ub <= lb:
        # no feasible move (shouldn't happen unless both at bounds etc.)
        return h.copy()
    # choose delta within bounds; sample uniform ~ maybe small magnitude
    # If max_step < ub - lb, restrict to [-max_step, max_step], else do full.
    # We'll sample delta = random.uniform(max(lb, -max_step), min(ub, max_step))
    low = max(lb, -max_step)
    high = min(ub, max_step)
    if low >= high:
        delta = random.uniform(lb, ub)
    else:
        delta = random.uniform(low, high)
    h_new = h.copy()
    h_new[i] = hi + delta
    h_new[j] = hj - delta
    return h_new

def compute_c5(h):

def random_binary_h(n):
    target = n // 2
    h = np.zeros(n, dtype=float)
    idx = np.random.choice(n, target, replace=False)
    h[idx] = 1.0
    return h

def mutate(h, max_step=0.2, n_moves=2):
    n = len(h)
    h_new = h.copy()
    for _ in range(n_moves):
        i, j = random.sample(range(n), 2)
        hi = h_new[i]
        hj = h_new[j]
        # compute bounds
        lb = max(-hi, hj - 1.0)
        ub = min(1.0 - hi, hj)
        if ub <= lb:
            continue
        low = max(lb, -max_step)
        high = min(ub, max_step)
        if low >= high:
            delta = random.uniform(lb, ub)
        else:
            delta = random.uniform(low, high)
        h_new[i] = hi + delta
        h_new[j] = hj - delta
    return h_new

def random_binary_h(n):

def mutate(h, max_step=0.2, n_moves=2):
    n = h.shape[0]
    h_new = h.copy()
    for _ in range(n_moves):
        i, j = random.sample(range(n), 2)
        hi = h_new[i]
        hj = h_new[j]
        lb = max(-hi, hj - 1.0)
        ub = min(1.0 - hi, hj)
        if ub <= lb:
            continue
        low = max(lb, -max_step)
        high = min(ub, max_step)
        if low >= high:
            delta = random.uniform(lb, ub)
        else:
            delta = random.uniform(low, high)
        h_new[i] = hi + delta
        h_new[j] = hj - delta
    return h_new
