try:
    from scipy.optimize import minimize
    SCIPY_AVAILABLE = True
except Exception:
    SCIPY_AVAILABLE = False

def project_onto_hyperplane_box(v, target_sum, low=0.0, high=1.0, max_iter=50):
    lo = -2.0
    hi = 2.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(v + lam, low, high)
        s = x.sum()

            lo = lam
        else:
            hi = lam
    return np.clip(v + (lo + hi) / 2.0, low, high)

def compute_correlation_bound(h, dx):
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

            return corr[N-1:]  # length N

import time
import numpy as np

# Attempt to import scipy optimizer; proceed without it if unavailable.
try:
    from scipy.optimize import minimize
    _SCIPY = True
except Exception:  # pragma: no cover
    _SCIPY = False


def _project_onto_box_hyperplane(v: np.ndarray, target_sum: float,
                                 low: float = 0.0, high: float = 1.0,
                                 max_iter: int = 60) -> np.ndarray:
    """
    Project vector v onto the intersection of the box [low, high]^n
    and the hyperplane sum(x) == target_sum (Euclidean projection).
    """
    lo = -np.max(v) - 1.0
    hi = (high - np.min(v)) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(v + lam, low, high)
        s = x.sum()
        if s < target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v + (lo + hi) / 2.0, low, high)


def _correlation_bound(h: np.ndarray, dx: float) -> float:
    """Compute C5 = max_k∫h(x)(1‑h(x+k))dx via discrete correlation."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return (h_values, c5_bound, n_points) for the Erdős minimum overlap problem."""
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Starting point

    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)  # copy

    dx = 2.0 / n

    # Ensure feasibility of the starting vector
    if not np.isclose(h.sum(), target_sum):
        h = _project_onto_box_hyperplane(h, target_sum)

        try:
            return evaluate_erdos_solution(h_vec)   # pre‑imported by the harness
        except Exception:                           # pragma: no cover
            return _correlation_bound(h_vec, dx)

    best_h = h.copy()
    best_c5 = _eval(best_h)

    # ------------------------------------------------------------------
    # Optional SLSQP refinement (maximising the minimal autocorrelation)
    # ------------------------------------------------------------------
    if _SCIPY:
        def _s_vals(vec):
            """Positive‑lag autocorrelations s_k = Σ h_i h_{i+k}."""
            full = np.correlate(vec, vec, mode="full")
            return full[n - 1 :]

            # x = [h_0,…,h_{n-1}, t]; maximise t  → minimise -t
            return -x[-1]

            if c5_val < best_c5:
                best_h = h_opt.copy()
                best_c5 = float(c5_val)

    # ------------------------------------------------------------------
    # Simple stochastic hill‑climbing to polish the best solution
    # ------------------------------------------------------------------
    while time.time() - start_time < budget_s - 0.3:
        # choose two distinct indices and transfer a random amount of mass
        i, j = rng.choice(n, size=2, replace=False)
        if best_h[i] > best_h[j]:
            donor, recv = i, j
        else:
            donor, recv = j, i
        max_transfer = min(best_h[donor], 1.0 - best_h[recv])
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer
        h_new = best_h.copy()
        h_new[donor] -= delta
        h_new[recv] += delta
        new_c5 = _eval(h_new)
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = float(new_c5)

    return (best_h, float(best_c5), n)
