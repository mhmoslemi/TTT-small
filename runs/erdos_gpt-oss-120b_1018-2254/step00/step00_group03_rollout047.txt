        if abs(delta) < 1e-12:
            break

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_onto_feasible(x, target):
    # Clip to [0,1]
    x = np.clip(x, 0.0, 1.0)
    # Water-filling to match sum target
    # Use iterative approach
    max_iter = 1000
    for _ in range(max_iter):
        s = x.sum()
        delta = target - s
        if abs(delta) < 1e-12:
            break
        if delta > 0:
            # increase where possible
            slack = 1.0 - x
            idx = np.where(slack > 1e-12)[0]
            if idx.size == 0:
                break
            inc = delta / idx.size
            # if inc exceeds slack for any, fill those and continue
            excess = inc - slack[idx]
            if np.any(excess > 0):
                # fill those
                to_fill = idx[excess > 0]
                fill_amount = slack[to_fill].sum()
                x[to_fill] = 1.0
                # reduce delta by amount filled
                delta -= fill_amount
                continue
            # otherwise inc fits
            x[idx] += inc
        else:
            # decrease where possible
            slack = x
            idx = np.where(slack > 1e-12)[0]
            if idx.size == 0:
                break
            dec = (-delta) / idx.size
            excess = dec - slack[idx]
            if np.any(excess > 0):
                to_zero = idx[excess > 0]
                fill_amount = slack[to_zero].sum()
                x[to_zero] = 0.0
                delta += fill_amount
                continue
            x[idx] -= dec
    # final small adjust via scaling if still off due to round-off
    s = x.sum()
    if abs(s - target) > 1e-8:
        # distribute small residual proportionally among non-boundary entries
        # compute free indices not at bounds
        free_idx = np.where((x > 1e-12) & (x < 1.0 - 1e-12))[0]
        if free_idx.size > 0:
            adjust = (target - s) / free_idx.size
            x[free_idx] += adjust
            x = np.clip(x, 0.0, 1.0)
    # final clip
    x = np.clip(x, 0.0, 1.0)
    # final normalization if slight mismatch due to rounding
    s = x.sum()
    if s != 0:
        x *= (target / s)
        x = np.clip(x, 0.0, 1.0)
    # enforce sum exactly by small correction on first element
    # compute diff
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # adjust first element within bounds
        i = 0
        for i in range(x.shape[0]):
            if (diff > 0 and x[i] < 1.0) or (diff < 0 and x[i] > 0.0):
                step = min(abs(diff), 1.0 - x[i]) if diff > 0 else min(abs(diff), x[i])
                x[i] += np.sign(diff) * step
                break
    return x

import time
import random
import numpy as np

def compute_c5(h):
    """Maximum overlap integral approximation for step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_onto_feasible(x, target):
    """Clip to [0,1] and adjust sum to exactly target using water-filling."""
    x = np.clip(x, 0.0, 1.0)
    max_iter = 1000
    for _ in range(max_iter):
        s = x.sum()
        delta = target - s
        if abs(delta) < 1e-12:
            break
        if delta > 0:
            slack = 1.0 - x
            idx = np.where(slack > 1e-12)[0]
            if idx.size == 0:
                break
            inc = delta / idx.size
            excess = inc - slack[idx]
            if np.any(excess > 0):
                to_fill = idx[excess > 0]
                fill_amount = slack[to_fill].sum()
                x[to_fill] = 1.0
                delta -= fill_amount
                continue
            x[idx] += inc
        else:
            slack = x
            idx = np.where(slack > 1e-12)[0]
            if idx.size == 0:
                break
            dec = (-delta) / idx.size
            excess = dec - slack[idx]
            if np.any(excess > 0):
                to_zero = idx[excess > 0]
                fill_amount = slack[to_zero].sum()
                x[to_zero] = 0.0
                delta += fill_amount
                continue
            x[idx] -= dec

    s = x.sum()
    diff = target - s
    if abs(diff) > 1e-12:

        for i in range(x.shape[0]):

                break
