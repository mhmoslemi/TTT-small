def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # using np.correlate
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_simplex_with_bounds(h, target_sum):
    # simple approach: clip 0-1, then adjust sum by adding constant offset while staying in bounds.
    # We'll implement iterative scaling:
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    # If sum is zero (unlikely), set uniform.
    if current_sum == 0:
        return np.full_like(h, target_sum / len(h))
    # Scale to adjust sum so that proportionally adjust interior values
    # We use scaling then clipping again until sum close.
    for _ in range(10):
        scale = target_sum / np.sum(h)
        h = h * scale
        # after scaling, clip, but scaling may cause over/under
        h = np.clip(h, 0, 1)
        # compute sum after clipping
        s = np.sum(h)
        if abs(s - target_sum) < 1e-8:
            break
        # adjust with a small offset based on difference
        # compute diff = target_sum - s, distribute equally but respect bounds
        diff = target_sum - s
        # compute adjustable indices not at bound
        mask = (h > 0) & (h < 1)
        if np.any(mask):
            # distribute diff equally among adjustable entries
            delta = diff / np.sum(mask)
            h[mask] += delta
            # clip again
            h = np.clip(h, 0, 1)
        else:
            # if all at bounds, break
            break
    # final clip
    h = np.clip(h, 0, 1)
    # final scaling if needed
    s = np.sum(h)
    if abs(s - target_sum) > 1e-6:
        # adjust uniformly small to fix rounding errors, ignoring bounds for now
        # but ensure no bound violation by distributing small adjustment
        diff = target_sum - s
        # just add diff/n to each element then clip
        h += diff / len(h)
        h = np.clip(h, 0, 1)
    return h

            if len(starts) > 1:

                h0 = _project_feasible(h0 + noise, target_sum)

            bounds = [(0.0, 1.0)] * n
            cons = {"type": "eq", "fun": lambda h, t=target_sum: np.sum(h) - t}

            def objective(x, n=n):
                return np.max(np.correlate(x, 1.0 - x, mode="full")) * (2.0 / n)

            res = opt.minimize(
                objective,
                h0,
                method="SLSQP",
                bounds=bounds,
                constraints=[cons],
                options={"maxiter": 400, "ftol": 1e-12, "disp": False},
            )
            h_opt = np.clip(res.x, 0.0, 1.0)
            h_opt = _project_feasible(h_opt, target_sum)

            # evaluate (prefer harness helper if present)
            if eval_func:
                try:
                    score = eval_func(h_opt)
                except Exception:
                    score = objective(h_opt)
            else:
                score = objective(h_opt)

            if score < best_score:
                best_score, best_h, best_n = float(score), h_opt.copy(), n

            # --- simple hill‑climbing refinement ---------------------------------
            max_hc_iters = 2000
            it = 0
            while it < max_hc_iters and time.time() - start_time < budget_s * 0.90:
                i, j = rng.integers(0, n, size=2)
                if i == j:
                    it += 1
                    continue
                max_inc = 1.0 - h_opt[i]
                max_dec = h_opt[j]
                if max_inc <= 0.0 or max_dec <= 0.0:
                    it += 1
                    continue
                delta = rng.uniform(0.0, min(max_inc, max_dec, 0.1))
                h_new = h_opt.copy()
                h_new[i] += delta
                h_new[j] -= delta
                if eval_func:
                    try:
                        new_score = eval_func(h_new)
                    except Exception:
                        new_score = objective(h_new)
                else:
                    new_score = objective(h_new)
                if new_score < score - 1e-9:
                    h_opt, score = h_new, new_score
                    if score < best_score:
                        best_score, best_h, best_n = float(score), h_opt.copy(), n
                it += 1
            # ---------------------------------------------------------------------

        if time.time() - start_time > budget_s * 0.95:
            break

    # Fallback if nothing succeeded (should never happen)
    if best_h is None:
        n = 70
        target_sum = n / 2.0
        h = np.zeros(n)
        h[: int(target_sum)] = 1.0
        best_h = h
        best_n = n
        best_score = _c5_value(best_h)

    return best_h.tolist(), float(best_score), int(best_n)
