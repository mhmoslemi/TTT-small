    def compute_c5(h):

        return float(np.max(corr) * (2.0 / n_points))
    def gradient_of_max_shift(h):

import time
import numpy as np

# -------------------- Helper functions --------------------

def _project_box_sum(y, target_sum, low=0.0, high=1.0):
    """Project y onto the box [low, high]^n with equality sum = target_sum.
    Uses a simple bisection on the Lagrange multiplier."""
    lo = low - np.max(y)
    hi = high - np.min(y)
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(y + lam, low, high)
        s = x.sum()
        if s < target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(y + (lo + hi) * 0.5, low, high)


def _compute_c5(h, n):
    """Return the overlap constant C5 for a given vector h (length n)."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _gradient_of_max_shift(h, n):
    """Gradient of f_k = sum_i h_i*(1-h_{i+k}) at the shift achieving max."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))          # index of worst shift
    shift = idx - (n - 1)               # signed shift amount
    grad = np.ones(n)                   # contribution from sum_i h_i
    # forward part: subtract h_{j+shift}
    if shift >= 0:
        grad[: n - shift] -= h[shift:]
    else:
        grad[-shift:] -= h[: n + shift]
    # backward part: subtract h_{j-shift}
    s2 = -shift
    if s2 >= 0:
        grad[: n - s2] -= h[s2:]
    else:
        grad[-s2:] -= h[: n + s2]
    return grad


def _line_search_best(h, grad, n, target_sum, steps):
    """Try a list of step sizes, return the best improved candidate."""
    best_h = h
    best_val = _compute_c5(h, n)
    improved = False
    for step in steps:
        cand = _project_box_sum(h - step * grad, target_sum)
        cand_val = _compute_c5(cand, n)
        if cand_val < best_val - 1e-12:
            best_h, best_val = cand, cand_val
            improved = True
            break   # accept the first improvement (larger steps first)
    return best_h, best_val, improved


def _random_perturb(h, sigma, rng, n, target_sum):
    """Add Gaussian noise and re‑project."""
    cand = _project_box_sum(h + sigma * rng.standard_normal(n), target_sum)
    return cand, _compute_c5(cand, n)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to lower the Erdős C5 constant.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)

    # ----- initial vector -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 70
        h0 = rng.random(n)

    target = n / 2.0
    h0 = _project_box_sum(h0, target)

    # ----- bookkeeping ----------------------------------------------------
    best_h = h0.copy()
    best_val = _compute_c5(best_h, n)

    start = time.time()
    time_limit = budget_s * 0.95

    # parameters
    max_iters = 20000
    step_candidates = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
    sigma0 = 0.25
    stagnation = 0

    it = 0
    while it < max_iters and (time.time() - start) < time_limit:
        # ----- gradient descent on the worst shift ------------------------
        grad = _gradient_of_max_shift(best_h, n)
        new_h, new_val, improved = _line_search_best(best_h, grad, n, target, step_candidates)

        if improved:
            best_h, best_val = new_h, new_val
            stagnation = 0
        else:
            # ----- random perturbation ------------------------------------
            elapsed = time.time() - start
            frac = elapsed / time_limit
            sigma = sigma0 * (1.0 - frac)   # decay sigma
            cand_h, cand_val = _random_perturb(best_h, sigma, rng, n, target)
            if cand_val < best_val:
                best_h, best_val = cand_h, cand_val
                stagnation = 0
            else:
                stagnation += 1

        # occasional restart if stuck for a long while
        if stagnation >= 2000:
            # keep the best found so far, start exploring from a fresh random point
            best_h = _project_box_sum(rng.random(n), target)
            best_val = _compute_c5(best_h, n)
            stagnation = 0

        it += 1

    # final projection to guarantee feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_box_sum(best_h, target)

    final_val = _compute_c5(best_h, n)

    return best_h, final_val, n
