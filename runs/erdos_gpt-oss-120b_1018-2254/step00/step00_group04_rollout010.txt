    n=len(h)

    idx = np.random.choice(n, num_change, replace=False)
    # Perturbation:

    new[idx] += delta
    # maybe also a random reset:
    # 10% chance set random values in [0,1]
    if np.random.rand() < 0.1:
        new[idx] = np.random.rand(num_change)

    shift = idx - (n - 1)  # integer shift: positive shift means h[i] with (1-h[i+shift])
    if shift >= 0:

        a = h[i_range]
        b = 1 - h[i_range + shift]

        i_range = np.arange(-shift, n)
        a = h[i_range]
        b = 1 - h[i_range + shift]  # shift negative => i+shift < i_range start
    contrib = a * b
    # pick index i among i_range with max contrib
    # to avoid local minima, also sometimes sample weighted randomly
    if rng.random() < 0.7:

        i_local = i_range[np.argmax(contrib)]
    else:
        # random weighted by contrib + epsilon

import time

def _project_feasible(v, target):
    """Project vector v onto {0≤x≤1, sum(x)=target}."""
    h = np.clip(v, 0.0, 1.0)
    diff = target - h.sum()
    for _ in range(100):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = h < 1 - 1e-12
        else:
            free = h > 1e-12
        if not np.any(free):
            break
        cnt = free.sum()
        delta = diff / cnt
        if diff > 0:
            inc = np.minimum(delta, 1 - h[free])
            h[free] += inc
        else:
            dec = np.minimum(-delta, h[free])
            h[free] -= dec
        diff = target - h.sum()
    return h

        a = h[i_range]
        b = 1 - h[i_range + shift]

        a = h[i_range]

        # Accept if better or with a temperature‑scaled probability.
        if cand_c < cur_c or rng.random() < np.exp((cur_c - cand_c) / max(T, 1e-12)):
            cur_h, cur_c = cand_h, cand_c

        if cand_c < best_c:
            best_h, best_c = cand_h, cand_c
            if best_c <= 0.3805:      # early stop if we already beat the target
                break

    # Final evaluation to guarantee consistency.
    best_c = evaluate_erdos_solution(best_h)
    return best_h, best_c, n
