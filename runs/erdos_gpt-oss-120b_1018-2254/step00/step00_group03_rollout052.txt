import math

def compute_c5(h):
    """Return the discretized C5 value for step function h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def generate_binary_vector(n):
    """Generate a binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = np.random.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

def project_to_feasible(v):
    """Clamp to [0,1] and adjust sum to n/2 by small redistributions."""
    n = len(v)
    target = n / 2.0
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - target) < 1e-12:
        return v
    # Simple iterative redistribution
    for _ in range(1000):
        diff = target - v.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # increase some entries that are <1
            idxs = np.where(v < 1.0 - 1e-12)[0]
            if len(idxs) == 0:
                break
            inc = min(diff / len(idxs), 0.01)
            delta = np.minimum(inc, 1.0 - v[idxs])
            v[idxs] += delta
        else:
            # decrease some entries that are >0
            idxs = np.where(v > 1e-12)[0]
            if len(idxs) == 0:
                break
            dec = min(-diff / len(idxs), 0.01)
            delta = np.minimum(dec, v[idxs])
            v[idxs] -= delta
    return np.clip(v, 0.0, 1.0)
