import numpy as np
import time
import math

def compute_c5_full(h):
    """Full convolution of h with (1‑h)."""
    return np.correlate(h, 1 - h, mode='full')

def compute_c5(h):
    """Maximum overlap integral C₅."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(compute_c5_full(h)) * dx

def compute_c5_and_shift(h):
    """Return (C₅, shift giving the maximum, full correlation array)."""
    n = h.shape[0]
    corr = compute_c5_full(h)
    idx = np.argmax(corr)
    shift = idx - (n - 1)          # shift in index units (positive → right)
    return corr[idx] * (2.0 / n), shift, corr

    """Gradient of the worst‑shift term wrt each entry of h."""
    n = h.shape[0]
    if shift >= 0:
        partner = np.arange(n) + shift
        valid = partner < n
        grad = np.where(valid, 1 - h[partner], 1.0)
    else:
        partner = np.arange(n) + shift
        valid = partner >= 0
        grad = np.where(valid, 1 - h[partner], 1.0)
    return grad

    """Return a random feasible vector (23 ones, one 0.5, rest 0 for odd n)."""

        while abs(diff) > 1e-12:
            if diff > 0:
                cand = np.where(h < 1 - 1e-12)[0]
                if cand.size == 0:
                    break
                i = cand[0]
                delta = min(diff, 1 - h[i])
                h[i] += delta
                diff -= delta
            else:
                cand = np.where(h > 1e-12)[0]
                if cand.size == 0:
                    break
                i = cand[0]
                delta = min(-diff, h[i])
                h[i] -= delta
                diff += delta

    best_h = h.copy()
    best_c5 = compute_c5(h)

    start = time.time()
    no_improve = 0

    # --- main optimisation loop --------------------------------------------
    while True:
        elapsed = time.time() - start
        if elapsed > budget_s * 0.97:
            break

        c_cur, shift, _ = compute_c5_and_shift(h)

        # update best solution
        if c_cur < best_c5 - 1e-12:
            best_c5 = c_cur
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # gradient for the worst shift
        grad = gradient_for_shift(h, shift)

        donor_candidates = np.where(h > 1e-12)[0]
        receiver_candidates = np.where(h < 1 - 1e-12)[0]
        if donor_candidates.size == 0 or receiver_candidates.size == 0:
            break

        donor_scores = grad[donor_candidates] * h[donor_candidates]
        donor = donor_candidates[np.argmax(donor_scores)]

        receiver_scores = (1 - grad[receiver_candidates]) * (1 - h[receiver_candidates])
        receiver = receiver_candidates[np.argmax(receiver_scores)]

        if donor == receiver:
            donor = np.random.choice(donor_candidates)
            receiver = np.random.choice(receiver_candidates)
            while donor == receiver and receiver_candidates.size > 1:
                receiver = np.random.choice(receiver_candidates)

        max_delta = min(0.05, h[donor], 1.0 - h[receiver])
        if max_delta <= 1e-12:
            donor = np.random.choice(donor_candidates)
            receiver = np.random.choice(receiver_candidates)
            while donor == receiver:
                receiver = np.random.choice(receiver_candidates)
            max_delta = min(0.05, h[donor], 1.0 - h[receiver])

        delta = 0.5 * max_delta
        new_h = h.copy()
        new_h[donor] -= delta
        new_h[receiver] += delta

        c_new = compute_c5(new_h)

        # simulated‑annealing acceptance
        T = max(0.0005, (budget_s - elapsed) / budget_s * 0.02)
        accept = False
        if c_new < c_cur:
            accept = True
        else:
            if np.random.rand() < math.exp(-(c_new - c_cur) / T):
                accept = True
        if accept:
            h = new_h

        # occasional random restart if stuck
        if no_improve >= 50000:
            h = random_feasible(n)
            no_improve = 0

    return (best_h, best_c5, n)
