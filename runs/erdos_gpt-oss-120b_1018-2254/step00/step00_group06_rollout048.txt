import time
import random
import numpy as np

# ---------- Helper Functions ----------
def compute_c5(h: np.ndarray) -> float:
    """Return the overlap integral C5 for a given step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(v: np.ndarray, target: float) -> np.ndarray:
    """Project vector v onto the box [0,1]^n with sum(v)=target."""
    v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) < 1e-12:
        return v
    # Reduce sum if necessary
    if s > target:
        v = v * (target / s)
        v = np.clip(v, 0.0, 1.0)
        return v
    # Increase sum: find tau >= 0 so that sum(min(1, v+tau)) = target
    low, high = 0.0, 1.0
    for _ in range(50):
        mid = (low + high) * 0.5
        w = np.minimum(1.0, v + mid)
        if w.sum() < target:
            low = mid
        else:
            high = mid
    v = np.minimum(1.0, v + high)
    # Tiny correction to hit the target exactly
    diff = target - v.sum()
    if abs(diff) > 1e-12:
        v[0] += diff
        v[0] = min(max(v[0], 0.0), 1.0)
    return v


def tournament_select(pop, fit, k=3):
    """Select one individual from pop using tournament of size k."""
    idx = np.random.choice(len(pop), k, replace=False)
    best = idx[np.argmin(fit[idx])]
    return pop[best]


# ---------- Main Optimization ----------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    # Determine problem size and initial vector
    try:
        n_points = len(initial_h_values)  # provided by harness if any
        init_vec = np.array(initial_h_values, dtype=float)
    except NameError:
        n_points = 95
        init_vec = np.random.rand(n_points)

    target_sum = n_points / 2.0

    # Create initial population
    pop_size = 120
    population = []
    # include the supplied construction (if any)
    population.append(project_feasible(init_vec, target_sum))
    while len(population) < pop_size:
        cand = np.random.rand(n_points)
        population.append(project_feasible(cand, target_sum))

    fitness = np.array([compute_c5(ind) for ind in population])
    best_idx = int(np.argmin(fitness))
    best_h = population[best_idx].copy()
    best_score = float(fitness[best_idx])

    start_time = time.time()
    elapsed = 0.0
    generation = 0
    max_generations = 2000
    mutation_rate = 0.01
    mutation_std = 0.1
    crossover_prob = 0.7
    tournament_k = 3
    target_goal = 0.3808

    # GA loop
    while elapsed < budget_s and generation < max_generations:
        generation += 1
        new_pop = [best_h.copy()]  # elitism

        while len(new_pop) < pop_size:
            # parent selection
            p1 = tournament_select(population, fitness, k=tournament_k)
            p2 = tournament_select(population, fitness, k=tournament_k)

            # crossover
            if random.random() < crossover_prob:
                alpha = random.random()
                child = alpha * p1 + (1.0 - alpha) * p2
            else:
                child = p1.copy()

            # mutation
            mask = np.random.rand(n_points) < mutation_rate
            if mask.any():
                child[mask] += np.random.normal(0.0, mutation_std, size=mask.sum())

            # repair to feasibility
            child = project_feasible(child, target_sum)
            new_pop.append(child)

        # evaluate new population
        fitness = np.array([compute_c5(ind) for ind in new_pop])

        # update best
        idx = int(np.argmin(fitness))
        if fitness[idx] < best_score:
            best_score = float(fitness[idx])
            best_h = new_pop[idx].copy()
            if best_score <= target_goal:
