def project_to_capped_simplex(v, target):
    n = v.shape[0]
    # Use binary search for lambda
    # Lower bound where all clip to 1:
    low = -np.max(v) - 1.0   # ensures v - low > 1
    high = np.max(v) + 1.0   # ensures v - high < 0
    for _ in range(50):
        lam = (low + high) / 2.0
        w = np.clip(v - lam, 0, 1)
        s = w.sum()
        if s > target:
            low = lam
        else:
            high = lam
    # after loops, choose w = clip(v - high,0,1) maybe.
    w = np.clip(v - high, 0, 1)
    # due to rounding errors, may not sum exactly; adjust final correction:
    # compute diff = target - w.sum(); if |diff| small ignore
    # else add diff to any element not at bound.
    diff = target - w.sum()
    if abs(diff) > 1e-10:
        # distribute diff across indices with 0 < w < 1
        mask = (w > 0) & (w < 1)
        if mask.any():
            w[mask] += diff / mask.sum()
            # ensure still within bounds after addition; clip again
            w = np.clip(w, 0, 1)
        else:
            # If no interior points, just add diff to one element within bounds.
            # Choose any index where w<1 if diff>0 else w>0.
            if diff > 0:
                idx = np.argmax(w < 1)
                w[idx] = min(1, w[idx] + diff)
            else:
                idx = np.argmax(w > 0)
                w[idx] = max(0, w[idx] + diff)
    return w

import numpy as np
import time, math

def compute_c5(h):
    """Return C5 for a step vector h on [0,2] with uniform spacing."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h), full mode
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_capped_simplex(v, target_sum, max_iter=60, eps=1e-12):
    """Project v onto the hyperplane sum=x with box constraints 0<=x_i<=1."""
    n = v.shape[0]
    # lower bound lambda where all entries clip to 1 -> sum = n
    lo = -np.max(v) - 5.0
    hi = np.max(v) + 5.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
        if hi - lo < eps:
            break
    w = np.clip(v - hi, 0.0, 1.0)
    # final tiny correction to hit the target sum (avoid drift)
    diff = target_sum - w.sum()
    if abs(diff) > 1e-10:
        mask = (w > 0.0) & (w < 1.0)
        if mask.any():
            w[mask] += diff / mask.sum()
            w = np.clip(w, 0.0, 1.0)
        else:
            # fallback: distribute to any non‑saturated coordinate
            idx = np.argmax(w < 1.0) if diff > 0 else np.argmax(w > 0.0)
            w[idx] = np.clip(w[idx] + diff, 0.0, 1.0)
    return w

def optimize_h(n, h_init, time_limit, rng):
    """Hill‑climbing / simulated‑annealing optimizer for C5."""
    target_sum = n / 2.0
    h = project_to_capped_simplex(h_init, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_c5 = best_c5

    start = time.time()
    # initial temperature proportional to observed value
    T0 = max(best_c5 * 0.05, 1e-4)

    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        # linear cooling
        T = max(1e-12, T0 * (1.0 - elapsed / time_limit))

        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j:
            continue
        # decide direction of transfer
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_delta = min(h[src], 1.0 - h[dst])
        if max_delta <= 0.0:
            continue
        # scale delta with temperature for finer moves later
        delta = rng.random() * max_delta
        if T < best_c5 * 0.02:
            # when cold, limit step size
            delta *= T / (best_c5 * 0.02)
            delta = min(delta, max_delta)
        # perform move
        h[src] -= delta
        h[dst] += delta

        new_c5 = compute_c5(h)
        improve = new_c5 < cur_c5
        if improve or rng.random() < math.exp(-(new_c5 - cur_c5) / T):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert
            h[src] += delta
            h[dst] -= delta

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):

    start_all = time.time()

    # enforce exact integral constraint
