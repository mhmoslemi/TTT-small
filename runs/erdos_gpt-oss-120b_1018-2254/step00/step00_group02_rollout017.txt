def random_swap_move(h, max_delta):
    n = len(h)
    i, j = np.random.choice(n, size=2, replace=False)
    # choose delta that does not violate bounds
    # current hi, hj
    hi = h[i]
    hj = h[j]
    # delta can be positive or negative, moving from i to j (or opposite)
    # we pick direction: either move from i to j or from j to i randomly; choose direction=False => from i to j else opposite
    direction = np.random.rand() < 0.5
    if direction:
        # move amount from i to j: hi decreases, hj increases
        max_possible = min(hi, 1 - hj, max_delta)
        if max_possible <= 0:
            return None
        delta = np.random.rand() * max_possible
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
    else:
        # move from j to i
        max_possible = min(hj, 1 - hi, max_delta)
        if max_possible <= 0:
            return None
        delta = np.random.rand() * max_possible
        new_h = h.copy()
        new_h[j] -= delta
        new_h[i] += delta
    return new_h

import time
import numpy as np
from scipy.optimize import differential_evolution, minimize, NonlinearConstraint

def _evaluate_erdos(h):
    """Compute C5 = max_k ∫ h(x)(1‑h(x+k))dx via discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def _make_sum_constraint(n):
    """Return a NonlinearConstraint enforcing sum(h) = n/2."""
    return NonlinearConstraint(lambda x: np.sum(x) - n / 2.0, 0.0, 0.0)

def _project_to_simplex(h, target_sum):
    """
    Adjust a vector to have the exact target sum while staying in [0,1].
    Simple scaling + uniform correction; small residuals are ignored.
    """
    cur_sum = np.sum(h)
    if cur_sum > 0:
        h = h * (target_sum / cur_sum)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        add = residual / h.size
        h = np.clip(h + add, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Return a step‑function approximation h, its C5 bound, and the number of points.
    The algorithm starts from the optional global `initial_h_values`, improves it
    with differential evolution + SLSQP, then up‑samples and refines further.
    """
    start = time.time()
    def _time_left():
        return budget_s - (time.time() - start)

    # ----------------------------------------------------------------------
    # 1. Obtain an initial feasible vector.
    # ----------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h_cur = np.array(initial_h_values, dtype=float)
    else:
        # default: 45 samples, uniform 0.5 (integral = 1)
        h_cur = np.full(45, 0.5, dtype=float)

    n0 = h_cur.shape[0]
    h_cur = _project_to_simplex(h_cur, n0 / 2.0)

    best_h = h_cur
    best_c5 = _evaluate_erdos(best_h)

    # ----------------------------------------------------------------------
    # 2. Global optimisation on the base resolution (n0 points).
    # ----------------------------------------------------------------------
    if _time_left() > 5.0:                       # leave some slack for later steps
        n = n0
        bounds = [(0.0, 1.0)] * n
        sum_con = _make_sum_constraint(n)
        max_iter = 60                               # modest number of generations

        # Try DE with an explicit equality constraint; fall back to a
        # penalised version if the constraint interface is unsupported.
        try:
            de_res = differential_evolution(
                _evaluate_erdos,
                bounds,
                constraints=[sum_con],
                maxiter=max_iter,
                popsize=10,
                seed=seed,
                polish=False,
                disp=False,
            )
        except Exception:  # pragma: no cover
            # penalty version – large coefficient forces feasibility
            def _penalised(x):
                return _evaluate_erdos(x) + 1e6 * abs(np.sum(x) - n / 2.0)
            de_res = differential_evolution(
                _penalised,
                bounds,
                maxiter=max_iter,
                popsize=10,
                seed=seed,
                polish=False,
                disp=False,
            )
        # Local refinement with SLSQP (handles the equality constraint)
        try:
            loc_res = minimize(
                _evaluate_erdos,
                de_res.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[sum_con],
                options={'maxiter': 300, 'ftol': 1e-12},
            )
            cand_h = loc_res.x
        except Exception:  # pragma: no cover
            cand_h = de_res.x

        cand_c5 = _evaluate_erdos(cand_h)
        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = cand_h

    # ----------------------------------------------------------------------
    # 3. Iteratively up‑sample and locally refine.
    # ----------------------------------------------------------------------
    for factor in (2, 3):
        if _time_left() <= 2.0:

        base_h = best_h
        n_base = base_h.shape[0]
        n = n_base * factor

        # Linear interpolation onto the finer grid
        x_old = np.linspace(0.0, 2.0, n_base)
        x_new = np.linspace(0.0, 2.0, n)

        # Force the sum constraint (small numerical drift after interpolation)
