import numpy as np
import time

def _project_to_sum(v, target):
    """Project vector v onto the box [0,1]^n while enforcing sum(v)=target.
    Solves min ||x-v||^2 s.t. 0<=x<=1, sum(x)=target via bisection on the
    Lagrange multiplier."""
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny residual fix
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = np.where((x > 0.0) & (x < 1.0))[0]
        if free.size:
            inc = diff / free.size
            x[free] += inc
            x = np.clip(x, 0.0, 1.0)
    return x

def _c5_of(h, dx):
    """Return the overlap constant C5 for a discrete step function h."""
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
