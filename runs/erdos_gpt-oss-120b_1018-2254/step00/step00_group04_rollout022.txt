def project_feasible(h):

    n = h.shape[0]
    target = n / 2.0
    # Clip initial values
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # Determine free indices for adjusting
        if diff > 0:
            free = np.where(h < 1.0 - 1e-12)[0]
        else:
            free = np.where(h > 1e-12)[0]
        if free.size == 0:
            break
        inc = diff / free.size
        # Apply inc but ensure no overflow/underflow
        h[free] += inc
        # Clip again

    return h

def small_mutate(h, max_step):
    n = h.shape[0]
    # choose two distinct indices
    i, j = random.sample(range(n), 2)
    # Determine feasible delta
    # Positive delta means transfer from i to j: reduce i, increase j
    max_pos = min(max_step, h[i], 1.0 - h[j])
    max_neg = min(max_step, h[j], 1.0 - h[i])  # negative transfer from j to i
    # If both are zero, return unchanged copy
    if max_pos <= 0 and max_neg <= 0:
        return h.copy()
    # Choose delta uniformly from [-max_neg, max_pos]
    d = random.uniform(-max_neg, max_pos)
    new_h = h.copy()
    new_h[i] -= d
    new_h[j] += d
    return new_h

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / h.shape[0]
    return float(np.max(corr) * dx)

def project_feasible(h):
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # iterative projection
    for _ in range(10):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = np.where(h < 1.0 - 1e-12)[0]
        else:
            free = np.where(h > 1e-12)[0]
        if free.size == 0:
            break
        inc = diff / free.size
        h[free] += inc
        h = np.clip(h, 0.0, 1.0)
    # final tiny fix
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-12:
        for i in range(n):
            if 0.0 <= h[i] + diff <= 1.0:
                h[i] += diff
                break
    return h

def small_mutate(h, max_step):

    i, j = random.sample(range(n), 2)
    # feasible delta
    max_pos = min(max_step, h[i], 1.0 - h[j])
    max_neg = min(max_step, h[j], 1.0 - h[i])
    if max_pos <= 0 and max_neg <= 0:
        return h.copy()
    d = random.uniform(-max_neg, max_pos)
    new_h = h.copy()
    new_h[i] -= d
    new_h[j] += d

import time
import random
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a discrete step function."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / h.shape[0]
    return float(np.max(corr) * dx)

def project_feasible(h):
    """Clip to [0,1] and enforce sum(h) == n/2."""
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Simple iterative water‑filling
    for _ in range(10):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = np.where(h < 1.0 - 1e-12)[0]
        else:
            free = np.where(h > 1e-12)[0]
        if free.size == 0:
            break
        inc = diff / free.size
        h[free] += inc
        h = np.clip(h, 0.0, 1.0)
    # Final tiny fix if needed
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-12:
        for i in range(n):
            if 0.0 <= h[i] + diff <= 1.0:
                h[i] += diff
                break
    return h

def small_mutate(h, max_step):
    """Transfer a small amount of mass between two coordinates."""
    n = h.shape[0]
    i, j = random.sample(range(n), 2)
    # Positive d moves mass from i to j
    max_pos = min(max_step, h[i], 1.0 - h[j])
    # Negative d moves mass from j to i
    max_neg = min(max_step, h[j], 1.0 - h[i])
    if max_pos <= 0 and max_neg <= 0:
        return h.copy()
    d = random.uniform(-max_neg, max_pos)
    new_h = h.copy()
    new_h[i] -= d
    new_h[j] += d
    return new_h

def random_feasible_vector(n):
    """Generate a random feasible h of length n."""
    v = np.random.rand(n)
    target = n / 2.0
    v = v * (target / np.sum(v))
    v = project_feasible(v)
    return v
