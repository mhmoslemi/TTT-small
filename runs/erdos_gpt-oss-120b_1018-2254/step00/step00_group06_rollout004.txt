        # generate offsprings

                candidates = random.sample(range(pop_size), tournament_k)
                bestc = min(candidates, key=lambda i: fitness[i])
                return population[bestc]
            parent1 = tournament()
            if random.random() < crossover_rate:
                parent2 = tournament()
                # single point crossover
                cp = random.randint(1, n_points-1)

def repair(v):

    target = n * 0.5
    v = np.clip(v, 0.0, 1.0)
    current = v.sum()
    diff = target - current
    if abs(diff) < 1e-12:
        return v
    # adjust by adding diff to elements not at bound
    # We'll iterate
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        # indices can absorb diff
        # if diff > 0, need to increase values; find those < 1
        # if diff < 0, need to decrease; find those > 0
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        idx = np.where(mask)[0]
        if len(idx) == 0:
            break
        # distribute equally
        inc = diff / len(idx)
        v[idx] += inc
        v = np.clip(v, 0.0, 1.0)
        diff = target - v.sum()

import numpy as np
import random
import time

def compute_c5(h, dx):
    """Return max overlap integral C5 for a given step function h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def repair(v):
    """Project vector v onto the feasible set: entries in [0,1] and average 0.5."""
    n = v.shape[0]
    target = n * 0.5
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    # distribute diff iteratively among non‑boundary entries
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        idx = np.where(mask)[0]
        if idx.size == 0:
            break
        inc = diff / idx.size
        v[idx] += inc
        v = np.clip(v, 0.0, 1.0)
        diff = target - v.sum()
    # final tidy‑up: adjust a single element if needed
    diff = target - v.sum()
    if abs(diff) > 1e-12:
        for i in range(n):
            if diff > 0 and v[i] < 1.0:
                add = min(diff, 1.0 - v[i])
                v[i] += add
                diff -= add
            elif diff < 0 and v[i] > 0.0:
                sub = min(-diff, v[i])
                v[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return np.clip(v, 0.0, 1.0)

def mutate(v, prob=0.01, sigma=0.1):
    """Gaussian mutation on each gene with given probability."""
    mask = np.random.rand(v.shape[0]) < prob
    if np.any(mask):
        v[mask] += np.random.normal(0.0, sigma, size=mask.sum())
    return v

def run(seed=42, budget_s=1100, **kwargs):
    """Genetic algorithm to lower the Erdős C5 bound."""
    random.seed(seed)
    np.random.seed(seed)

    start = time.time()
    # Determine problem size
    if 'initial_h_values' in globals():
        init_vec = np.array(initial_h_values, dtype=float)
        n_points = init_vec.shape[0]
        init_vec = repair(init_vec.copy())
    else:
        n_points = 95
        init_vec = None

    dx = 2.0 / n_points
    pop_size = 120
    crossover_rate = 0.7
    mut_prob = 0.02
    mut_sigma = 0.1
    tournament_k = 3

    # ----- Initialise population -----
    population = []
    if init_vec is not None:
        population.append(init_vec)
    while len(population) < pop_size:
        cand = np.random.rand(n_points)
        cand = repair(cand)
        population.append(cand)

    fitness = [compute_c5(ind, dx) for ind in population]
    best_idx = int(np.argmin(fitness))
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]

    # ----- Evolution loop -----
    generation = 0
    max_gen = 1000
    while time.time() - start < budget_s * 0.95 and generation < max_gen:
        # Elitism: preserve current best
        new_pop = [best_h.copy()]

        def tournament():
            choices = random.sample(range(pop_size), tournament_k)
            return population[min(choices, key=lambda i: fitness[i])]

        while len(new_pop) < pop_size:
            p1 = tournament()
            if random.random() < crossover_rate:
                p2 = tournament()
                cp = random.randint(1, n_points - 1)
                child1 = np.concatenate([p1[:cp], p2[cp:]])
                child2 = np.concatenate([p2[:cp], p1[cp:]])
                child1 = mutate(child1, mut_prob, mut_sigma)
                child2 = mutate(child2, mut_prob, mut_sigma)
                child1 = repair(child1)
                child2 = repair(child2)
                new_pop.append(child1)
                if len(new_pop) < pop_size:
                    new_pop.append(child2)
            else:
                child = p1.copy()
                child = mutate(child, mut_prob, mut_sigma)
                child = repair(child)
                new_pop.append(child)

        fitness = [compute_c5(ind, dx) for ind in new_pop]
        cur_best = int(np.argmin(fitness))
        if fitness[cur_best] < best_c5:
            best_c5 = fitness[cur_best]
            best_h = new_pop[cur_best].copy()
            if best_c5 <= 0.3808:
                break

        population = new_pop
        generation += 1

    # ----- Simple hill‑climbing refinement -----
    while time.time() - start < budget_s * 0.98:
        v = best_h.copy()
        i, j = random.sample(range(n_points), 2)

            c = compute_c5(v, dx)
            if c < best_c5:
                best_c5 = c
                best_h = v
