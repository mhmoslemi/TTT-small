def project_feasible(x, target_sum):
    x = np.clip(x, 0.0, 1.0)
    max_iter = 10 * len(x)
    for _ in range(max_iter):
        cur_sum = x.sum()
        diff = target_sum - cur_sum
        if abs(diff) < 1e-9:
            break
        # free indices where not at bound
        free = np.where((x > 0 + 1e-12) & (x < 1 - 1e-12))[0]
        if free.size == 0:
            # cannot adjust without breaking bounds; distribute diff to all and then clip
            # simple approach: add diff/len(x) to all and clip again
            x += diff / len(x)
            x = np.clip(x, 0.0, 1.0)
            continue
        adjust = diff / free.size
        x[free] += adjust
        # clip after adjustment
        x = np.clip(x, 0.0, 1.0)
    # final adjust for small error
    cur_sum = x.sum()
    diff = target_sum - cur_sum
    if abs(diff) > 1e-8:
        # if small diff remains, adjust uniformly but ensure bounds
        # add diff / len(x) to each
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
    return x

def perturb_swap(h, step):
   n = len(h)
   i, j = np.random.choice(n, size=2, replace=False)
   # choose delta
   max_decrease_i = h[i]  # max we can subtract from i (>=0)
   max_increase_j = 1 - h[j]  # max we can add to j
   max_delta_pos = min(max_increase_j, max_decrease_i)  # max positive delta we can add to j
   max_increase_i = 1 - h[i]
   max_decrease_j = h[j]
   max_delta_neg = -min(max_increase_i, max_decrease_j)  # max negative delta we can add (i.e., moving mass opposite way)
   # total allowed delta range: [max_delta_neg, max_delta_pos]
   # sample delta from uniform in that range scaled by step factor (maybe step is maximum magnitude)
   # Actually step is scale for range: set delta = step * (random.uniform(-1, 1)) and then clip to allowed range.
   delta = step * (np.random.rand() * 2 - 1)
   # now clip delta to allowed range
   delta = np.clip(delta, max_delta_neg, max_delta_pos)
   # apply
   new_h = h.copy()
   new_h[i] -= delta
   new_h[j] += delta
   return new_h

def project_to_sum_and_bounds(v, target_sum):
    # Clip to [0,1] first
    x = np.clip(v, 0.0, 1.0)
    # adjust sum by redistributing overflow/underflow
    # We'll use a simple water-filling algorithm
    # Compute current sum
    cur_sum = x.sum()
    diff = target_sum - cur_sum
    # tolerance
    tol = 1e-12
    # If diff is zero, done
    if abs(diff) <= tol:
        return x
    # We'll iteratively adjust
    # Use while loop, adjust values that are not at bounds
    # We'll use a simple method: for diff >0, increase values proportionally where x<1.
    # for diff<0, decrease where x>0.
    for _ in range(10 * len(x)):
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(x < 1 - 1e-12)[0]
            if free.size == 0:
                break
            inc = diff / free.size
            # compute possible increment without exceeding bound
            delta = np.minimum(1 - x[free], inc)
            x[free] += delta
        else:
            free = np.where(x > 1e-12)[0]
            if free.size == 0:
                break
            dec = diff / free.size
            delta = np.minimum(x[free], -dec)  # dec negative, -dec positive
            x[free] -= delta
        cur_sum = x.sum()
        diff = target_sum - cur_sum
    # final small correction:
    cur_sum = x.sum()
    diff = target_sum - cur_sum
    if abs(diff) > tol:
        # distribute remaining diff across any indices not at bound
        # naive loop
        for i in range(len(x)):
            if diff > 0 and x[i] < 1:
                add = min(diff, 1 - x[i])
                x[i] += add
                diff -= add
            elif diff < 0 and x[i] > 0:
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
            if abs(diff) <= tol:
                break
    # final clip
    x = np.clip(x, 0.0, 1.0)
    return x

import numpy as np
import time

def compute_c5(h):
    """Return C5 for the step function h."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_to_sum_and_bounds(v, target_sum):
    """Project vector v onto the hyperplane sum=v_target with box [0,1]."""
    x = np.clip(v, 0.0, 1.0)
    cur_sum = x.sum()
    diff = target_sum - cur_sum
    tol = 1e-12
    # iterative water‑filling
    for _ in range(10 * len(x)):
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(x < 1.0 - 1e-12)[0]
            if free.size == 0:
                break
            inc = diff / free.size
            delta = np.minimum(1.0 - x[free], inc)
            x[free] += delta
        else:
            free = np.where(x > 1e-12)[0]
            if free.size == 0:
                break
            dec = diff / free.size
            delta = np.minimum(x[free], -dec)
            x[free] -= delta
        cur_sum = x.sum()
        diff = target_sum - cur_sum
    # final tiny correction if needed
    cur_sum = x.sum()
    diff = target_sum - cur_sum
    if abs(diff) > tol:
        for i in range(len(x)):
            if diff > 0 and x[i] < 1.0:
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
            elif diff < 0 and x[i] > 0.0:
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
            if abs(diff) <= tol:
                break
    return np.clip(x, 0.0, 1.0)

def random_swap_move(h, step):
    """Swap a random amount of mass between two indices while respecting bounds."""
    n = len(h)
    i, j = np.random.choice(n, size=2, replace=False)
    # delta: amount moved from i to j (h_i decreases, h_j increases)
    lo = max(-h[j], h[i] - 1.0)        # lower bound for delta
    hi = min(h[i], 1.0 - h[j])         # upper bound for delta
    if lo >= hi:
        return h.copy()
    delta = np.random.uniform(-step, step)
    delta = np.clip(delta, lo, hi)
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h

def targeted_swap_move(h, step):
    """Perform a directed swap based on the currently worst shift."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    s = int(np.argmax(corr) - (n - 1))  # shift causing maximal overlap
    if s == 0:
        return random_swap_move(h, step)
    if s > 0:
        i_start, i_end = 0, n - s
    else:
        i_start, i_end = -s, n
    if i_end <= i_start:
        return random_swap_move(h, step)
    idx = np.arange(i_start, i_end)
    terms = h[idx] * (1.0 - h[idx + s])
    i = idx[int(np.argmax(terms))]
    j = i + s
    # amount that can be moved without violating bounds
    delta = min(step, h[i], 1.0 - h[j])
    if delta <= 0:
        return random_swap_move(h, step)
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    # initialise h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n = 47
        h = np.random.rand(n)
        h = project_to_sum_and_bounds(h, n / 2.0)

    n = len(h)
    target_sum = n / 2.0

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    cur_h = best_h.copy()
    cur_c5 = best_c5

    # parameters
    max_step = 0.5
    min_step = 0.005
    init_temp = 0.005
    target_move_prob = 0.2      # fraction of iterations using directed move
    restart_interval = 120000   # iters without improvement before a fresh restart
    last_improve = 0
    it = 0

    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.98:
            break
        progress = elapsed / budget_s
        step = max_step * (1 - progress) + min_step * progress

        # generate candidate
        if np.random.rand() < target_move_prob:
            cand = targeted_swap_move(cur_h, step)
        else:
            cand = random_swap_move(cur_h, step)

        cand_c5 = compute_c5(cand)
        it += 1

        # deterministic improvement
        if cand_c5 + 1e-12 < cur_c5:
            cur_h = cand
            cur_c5 = cand_c5
            if cand_c5 < best_c5 - 1e-12:
                best_h = cand.copy()
                best_c5 = cand_c5
                last_improve = it
        else:
            # simulated‑annealing style acceptance
            temp = init_temp * (1 - progress)
            if temp > 0:
                prob = np.exp(-(cand_c5 - cur_c5) / (temp + 1e-12))
                if np.random.rand() < prob:
                    cur_h = cand
                    cur_c5 = cand_c5

        # occasional full restart if stagnating
        if it - last_improve > restart_interval:
            rnd = np.random.rand(n)
            cur_h = project_to_sum_and_bounds(rnd, target_sum)
            cur_c5 = compute_c5(cur_h)
            last_improve = it

    # final verification
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n)
