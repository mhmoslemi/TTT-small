    high = np.max(y) + 1.0

    sum_low = np.clip(y - low, lo, hi).sum()
    sum_high = np.clip(y - high, lo, hi).sum()

    # expand bounds if needed
    while sum_low < target:
        low -= 10.0
        sum_low = np.clip(y - low, lo, hi).sum()
    while sum_high > target:
        high += 10.0
        sum_high = np.clip(y - high, lo, hi).sum()

    for _ in range(max_iter):
        mid = (low + high) / 2.0
        x = np.clip(y - mid, lo, hi)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            low = mid
        else:
            high = mid
    # return final approximation
    return np.clip(y - (low + high) / 2.0, lo, hi)


def compute_c5(h, dx):
    """Maximum overlap integral for a discretised step function."""
    return (np.correlate(h, 1.0 - h, mode="full") * dx).max()


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] (sampled at n points) to minimize
    C5 = max_k ∫ h(x)(1 - h(x+k)) dx.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)

    # Initial vector (if supplied) or fallback uniform half‑density.
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        # fallback: 42 points, each 0.5 gives correct integral.
        h0 = np.full(42, 0.5, dtype=float)

    n = h0.size
    target_sum = n / 2.0
    dx = 2.0 / n

    # Ensure feasibility of the starting point.
    if not (abs(h0.sum() - target_sum) < 1e-10 and np.all((h0 >= 0) & (h0 <= 1))):
        h0 = project_onto_sum_and_box(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)

    # Current state for simulated annealing.
    cur_h = best_h.copy()
    cur_c5 = best_c5

    start_time = time.time()
    deadline = start_time + budget_s * 0.95

    # Parameters
    step = 0.5                     # max mass moved in a swap
    prob_global = 0.15            # probability of a global Gaussian perturbation
    T0 = 5e-3                     # initial temperature for SA
    it = 0

    while time.time() < deadline:
        it += 1
        # Annealing temperature decays linearly to ~0.
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        T = max(1e-7, (1.0 - frac) * T0)

        # ------- propose a new candidate ----------
        if rng.random() < prob_global:
            # Global Gaussian perturbation followed by projection.
            sigma = step * 0.5
            y = cur_h + sigma * rng.standard_normal(n)
            cand_h = project_onto_sum_and_box(y, target_sum)
        else:
            # Pairwise mass transfer (swap) that preserves the sum.
            cand_h = cur_h.copy()
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # Randomly decide direction of transfer.
            if rng.random() < 0.5:
                src, dst = i, j
            else:
                src, dst = j, i
            max_delta = min(step, cand_h[src], 1.0 - cand_h[dst])
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta
            cand_h[src] -= delta
            cand_h[dst] += delta

        # ------- evaluate -------
        cand_c5 = compute_c5(cand_h, dx)

        # ------- acceptance (simulated annealing) -------
        if cand_c5 <= cur_c5 or rng.random() < np.exp(-(cand_c5 - cur_c5) / T):
            cur_h = cand_h
            cur_c5 = cand_c5

        # ------- update best found -------
        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = cand_h

        # ------- adapt step size -------
        if it % 2000 == 0:
            step = max(step * 0.9, 1e-4)
            # occasional random restart to potentially escape plateaus
            if rng.random() < 0.03:
                rnd = rng.random(n)
                rnd = project_onto_sum_and_box(rnd, target_sum)
                rnd_c5 = compute_c5(rnd, dx)
                cur_h, cur_c5 = rnd, rnd_c5
                if rnd_c5 < best_c5:
                    best_c5, best_h = rnd_c5, rnd

    # Final projection to guarantee feasibility (tiny numerical drift may happen).
    best_h = project_onto_sum_and_box(best_h, target_sum)
    best_c5 = compute_c5(best_h, dx)

    return (best_h, float(best_c5), int(n))
