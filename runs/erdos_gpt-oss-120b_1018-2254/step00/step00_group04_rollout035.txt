import time, math, random
import numpy as np

# Correlation based C5 evaluation (same as the provided evaluator)
def _c5_value(h):
    dx = 2.0 / h.shape[0]
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

# Swap two entries while keeping the total sum unchanged.
def _swap_move(h):
    n = h.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    hi, hj = h[i], h[j]
    # delta added to i, subtracted from j
    upper = min(1.0 - hi, hj)      # max positive delta
    lower = -min(hi, 1.0 - hj)     # max negative delta
    if upper <= lower:  # no feasible move
        return h
    delta = np.random.uniform(lower, upper)
    h_new = h.copy()
    h_new[i] = hi + delta
    h_new[j] = hj - delta
    return h_new

# Try to equalise a pair (i, i+shift) to raise their product.
def _equalise_pair(h, i, j):
    hi, hj = h[i], h[j]
    if hi == hj:
        return h
    if hi > hj:
        max_move = min((hi - hj) / 2.0, hi, 1.0 - hj)
        if max_move <= 0:
            return h
        delta = np.random.uniform(0.0, max_move)
        h_new = h.copy()
        h_new[i] = hi - delta
        h_new[j] = hj + delta
    else:  # hj > hi
        max_move = min((hj - hi) / 2.0, hj, 1.0 - hi)
        if max_move <= 0:
            return h
        delta = np.random.uniform(0.0, max_move)
        h_new = h.copy()
        h_new[i] = hi + delta
        h_new[j] = hj - delta
    return h_new
