    h2 = h.copy()
    # choose random number of swaps
    n_swaps = max(1, int(0.05 * n))
    for _ in range(n_swaps):

        if i == j: continue
        # compute max transfer based on direction
        # randomly decide direction: transfer from i to j or opposite
        if np.random.rand() < 0.5:
            # move from i to j
            max_transfer = min(h2[i], 1.0 - h2[j])
        else:
            max_transfer = min(h2[j], 1.0 - h2[i])

        if max_transfer <= 0:

        delta = np.random.rand() * max_transfer * magnitude
        h2[i] -= delta
        h2[j] += delta

    # due to clip, sum may change; adjust to target sum using project_to_constraints
    h2 = project_to_constraints(h2, target_sum)

import numpy as np, time

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_to_constraints(v, target_sum, max_iter=10):
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        total = v.sum()
        diff = target_sum - total
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            inc = 1.0 - v
            room = inc.sum()
            if room == 0:
                break
            v += diff * inc / room
        else:
            dec = v
            room = dec.sum()
            if room == 0:
                break
            v += diff * dec / room
        v = np.clip(v, 0.0, 1.0)
    # Final fix tiny residual
    residual = target_sum - v.sum()
    if abs(residual) > 1e-12:
        free = np.where((v > 0) & (v < 1))[0]
        if free.size:
            i = free[0]
            v[i] = np.clip(v[i] + residual, 0.0, 1.0)
    return v

def gradient_max_corr(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    shift = int(np.argmax(corr) - (n - 1))
    g = np.zeros_like(h)
    if shift >= 0:
        # term (1 - h_{j+shift})
        idx1 = np.arange(n - shift)
        g[idx1] = 1.0 - h[idx1 + shift]
        # term -h_{j-shift}
        idx2 = np.arange(shift, n)
        g[idx2] -= h[idx2 - shift]
    else:
        s = -shift
        idx1 = np.arange(s, n)
        g[idx1] = 1.0 - h[idx1 - s]
        idx2 = np.arange(n - s)
        g[idx2] -= h[idx2 + s]
    return g, shift

def perturb(h, rng, magnitude=0.5):
    n = h.shape[0]
    h2 = h.copy()
    n_swaps = max(1, int(0.05 * n))
    for _ in range(n_swaps):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # decide direction
        if rng.random() < 0.5:
            max_transfer = min(h2[i], 1.0 - h2[j])
            if max_transfer <= 0:
                continue
            delta = rng.random() * max_transfer * magnitude
            h2[i] -= delta
            h2[j] += delta
        else:
            max_transfer = min(h2[j], 1.0 - h2[i])
            if max_transfer <= 0:
                continue
            delta = rng.random() * max_transfer * magnitude
            h2[j] -= delta
            h2[i] += delta
    h2 = np.clip(h2, 0.0, 1.0)
    target_sum = h.shape[0] / 2.0
    h2 = project_to_constraints(h2, target_sum)
    return h2

def cos2_shape(n):
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    h = np.cos(np.pi * x / 2) ** 2
    target_sum = n / 2.0
    return project_to_constraints(h, target_sum)

    start = time.time()

import numpy as np, time

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_to_constraints(v, target_sum, max_iter=10):
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        total = v.sum()
        diff = target_sum - total
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            inc = 1.0 - v
            room = inc.sum()
            if room == 0:
                break
            v += diff * inc / room
        else:
            dec = v
            room = dec.sum()
            if room == 0:
                break
            v += diff * dec / room
        v = np.clip(v, 0.0, 1.0)
    residual = target_sum - v.sum()
    if abs(residual) > 1e-12:
        free = np.where((v > 0) & (v < 1))[0]
        if free.size:
            i = free[0]
            v[i] = np.clip(v[i] + residual, 0.0, 1.0)
    return v
