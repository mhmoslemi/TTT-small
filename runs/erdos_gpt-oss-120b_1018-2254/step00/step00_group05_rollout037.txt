    exp_vals = np.exp(alpha * f - max_f)
    w = exp_vals / np.sum(exp_vals)  # shape 2N-1
    # compute gradient
    grad = np.zeros(N, dtype=np.float64)
    for s in range(-N+1, N):
        w_s = w[s + N-1]
        shift = s
        if shift >=0:
            # forward term: i such that i+shift < N
            # For i from 0 to N-1-shift:

            # grad[i] -= w_s * h[i-shift] * dx
            i_end = N - shift

            # backward part
            i_start = shift
            i_range2 = slice(i_start, N)

def project_box_simplex(v, target_sum, lower=0.0, upper=1.0, tol=1e-9, max_iter=50):
    lo = np.min(v) - upper  # lower bound for lambda (because v_i - lambda could exceed upper)
    hi = np.max(v) - lower  # upper bound for lambda
    # Actually as lambda increases, v_i - lambda decreases, so we want lambda such that sum_i clamp(v_i - lambda) = target.
    # Start lo = -inf? We'll use binary search on lambda.
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, lower, upper)
        s = np.sum(x)
        if np.abs(s - target_sum) < tol:
            return x
        if s > target_sum:
            # need to increase lambda to reduce sum
            lo = lam
        else:
            hi = lam
    # final
    return np.clip(v - (lo+hi)/2.0, lower, upper)

    if abs(diff) > tol:
        free = (x > lower + 1e-12) & (x < upper - 1e-12)
        if free.any():
            i = np.where(free)[0][0]
            x[i] = np.clip(x[i] + diff, lower, upper)
        else:
            x += diff / len(x)
            x = np.clip(x, lower, upper)
    return x

def random_feasible(N, target, rng):
    v = rng.random(N)
    return project_to_box_simplex(v, target)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # initialise h
    try:
        h = np.array(initial_h_values, dtype=np.float64)
        N = h.shape[0]
    except Exception:
        N = 70
        h = np.full(N, 0.5, dtype=np.float64)

    target_sum = N * 0.5
    dx = 2.0 / N
    h = project_to_box_simplex(h, target_sum)

    best_h = h.copy()
    best_score = compute_c5(best_h, dx)

    # Adam state
    m = np.zeros_like(h)
    v_sq = np.zeros_like(h)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    start = time.time()
    it = 0
    # early stop target (slightly below record)
    goal = 0.3808

    while True:
        now = time.time()
        if now - start > budget_s * 0.95:
            break
        # correlation
        y = 1.0 - h
        corr = np.correlate(h, y, mode='full') * dx        # length 2N-1
        # soft‑max weights approximating the max
        frac = (now - start) / budget_s
        alpha = 5.0 + 25.0 * frac                         # 5 → 30
        maxa = np.max(alpha * corr)
        w = np.exp(alpha * corr - maxa)
        w /= w.sum()

        # gradient of the smoothed max
        grad = np.zeros_like(h)
        for shift in range(-N + 1, N):
            ws = w[shift + N - 1]
            if ws == 0.0:
                continue
            if shift >= 0:
                if shift < N:
                    grad[:N - shift] += ws * (1.0 - h[shift:]) * dx
                    grad[shift:]    -= ws * h[:N - shift] * dx
            else:
                t = -shift
                grad[t:]   += ws * (1.0 - h[:N - t]) * dx
                grad[:N - t] -= ws * h[t:] * dx

        # Adam update
        lr = 0.5 * (1.0 - frac) + 0.05                     # decay from 0.55 → 0.05
        m = beta1 * m + (1.0 - beta1) * grad
        v_sq = beta2 * v_sq + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v_sq / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        h_new = project_to_box_simplex(h_new, target_sum)

        # evaluate
        score = compute_c5(h_new, dx)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
            h = h_new
        else:
            # occasional Metropolis acceptance
            temp = 0.01 * (1.0 - frac) + 0.001
            if rng.random() < np.exp(-(score - best_score) / temp):
                h = h_new

        # small random pair‑swap perturbations every few iterations
        if it % 200 == 0 and it > 0:
            i, j = rng.choice(N, size=2, replace=False)
            max_move = min(h[i], 1.0 - h[j])
            if max_move > 0:
                delta = rng.random() * max_move * 0.5
                h_pert = h.copy()
                h_pert[i] -= delta
                h_pert[j] += delta
                h_pert = project_to_box_simplex(h_pert, target_sum)
                s_pert = compute_c5(h_pert, dx)
                if s_pert < best_score:
                    best_score, best_h = s_pert, h_pert.copy()
                    h = h_pert

        # restart if stagnated
        if it % 5000 == 0 and it > 0:
            h = random_feasible(N, target_sum, rng)

        if best_score <= goal:
            break

        it += 1

    best_h = project_to_box_simplex(best_h, target_sum)
    final_score = compute_c5(best_h, dx)
    return best_h, final_score, N
