def mutate(h, step):
    n = len(h)
    i, j = np.random.choice(n, 2, replace=False)
    # compute max delta positive we can add to i while keeping <=1, and subtract from j while staying >=0.
    max_add_i = 1.0 - h[i]
    max_sub_j = h[j]  # amount we can subtract from j (i.e., increase i)
    # delta_up = min(max_add_i, max_sub_j, step)
    # also consider negative direction: we could subtract from i and add to j.
    # random choose direction sign.
    direction = np.random.choice([1, -1])
    if direction == 1:
        max_delta = min(max_add_i, max_sub_j, step)
        if max_delta <= 0:
            return h  # cannot move
        delta = np.random.uniform(0, max_delta)
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        return h_new
    else:
        # opposite
        max_delta = min(h[i], 1.0 - h[j], step)
        if max_delta <= 0:
            return h
        delta = np.random.uniform(0, max_delta)
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        return h_new
```

import time
import random
import math
import numpy as np

def compute_c5(h):
    """Return max_k ∫ h(x)*(1-h(x+k)) dx using discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h) in full mode gives the required sums
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair_h(h):
    """Clip to [0,1] and adjust to hit exact target sum = n/2."""
    n = h.size
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if abs(cur - target) < 1e-12:
        return h
    if cur > target:  # need to decrease
        excess = cur - target
        idx = np.argsort(-h)  # largest first
        for i in idx:
            dec = min(h[i], excess)
            h[i] -= dec
            excess -= dec
            if excess <= 1e-12:
                break
    else:               # need to increase
        deficit = target - cur
        idx = np.argsort(h)   # smallest first
        for i in idx:
            inc = min(1.0 - h[i], deficit)
            h[i] += inc
            deficit -= inc
            if deficit <= 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

def mutate(h, step):
    """Generate a new candidate by moving mass between two coordinates."""
    n = h.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    hi, hj = h[i], h[j]

    # direction +: increase i, decrease j
    max_inc_i = 1.0 - hi
    max_dec_j = hj
    delta_pos = min(step, max_inc_i, max_dec_j)

    # direction -: decrease i, increase j
    max_dec_i = hi
    max_inc_j = 1.0 - hj
    delta_neg = min(step, max_dec_i, max_inc_j)

    if delta_pos <= 0 and delta_neg <= 0:
        return h  # no feasible move

    h_new = h.copy()
    if delta_pos > 0 and delta_neg > 0:
        if random.random() < 0.5:
            delta = random.uniform(0.0, delta_pos)
            h_new[i] += delta
            h_new[j] -= delta
        else:
            delta = random.uniform(0.0, delta_neg)
            h_new[i] -= delta
            h_new[j] += delta
    elif delta_pos > 0:
        delta = random.uniform(0.0, delta_pos)
        h_new[i] += delta
        h_new[j] -= delta
    else:
        delta = random.uniform(0.0, delta_neg)
        h_new[i] -= delta
        h_new[j] += delta
    return h_new

def random_initial(n):
    """Create a random feasible h with sum = n/2."""
    target = n / 2.0
    h = np.random.rand(n)
    # Scale to target sum
    h = h / h.sum() * target
    # Clip and final repair
    h = repair_h(h)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Optimize the Erdős C₅ bound and return (h_values, c5_bound, n_points)."""
    random.seed(seed)
    np.random.seed(seed)
    start = time.time()

    # ---- initial solution -------------------------------------------------
    if 'initial_h_values' in globals():
        h_cur = np.array(initial_h_values, dtype=float)
    else:
        # fallback: moderate size if no parent is provided
        h_cur = random_initial(200)

    n = h_cur.shape[0]
    target_sum = n / 2.0
    # safety repair
    h_cur = repair_h(h_cur)

    # ---- bookkeeping ------------------------------------------------------
    best_h = h_cur.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h_cur.copy()
    cur_c5 = best_c5

    # ---- annealing schedule ------------------------------------------------
    T0 = 0.015          # initial temperature
    T_end = 1e-5        # final temperature
    step0 = 0.5         # initial max delta
    step_end = 0.001    # final max delta

    # main loop
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.97:
            break

        frac = elapsed / budget_s
        T = T0 * (1.0 - frac) + T_end * frac
        step = step0 * (1.0 - frac) + step_end * frac

        # propose a neighbour
        cand_h = mutate(cur_h, step)
        cand_c5 = compute_c5(cand_h)

        # acceptance test
        if cand_c5 < cur_c5:
            accept = True
        else:
            # Metropolis criterion
            prob = math.exp(-(cand_c5 - cur_c5) / (T + 1e-12))
            accept = random.random() < prob

        if accept:
            cur_h = cand_h
            cur_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = cand_h.copy()

    # Final repair to guarantee feasibility
    best_h = repair_h(best_h)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
