    n = h.shape[0]
    size = 1 << (int(np.ceil(np.log2(2*n - 1))))  # next power of two
    H = np.fft.rfft(h, n=size)

    corr = np.fft.irfft(H * np.conj(G), n=size)
    corr = corr[:2*n - 1]  # linear correlation
    return np.max(corr) * (2.0 / n)

import numpy as np
import time
import math

def _c5_val(h):
    """Return the C5 value using the provided evaluator if present,
    otherwise fall back to a direct np.correlate implementation."""
    try:
        # evaluate_erdos_solution is provided in the harness environment
        return evaluate_erdos_solution(h)
    except NameError:
        # direct computation matching the specification
        n = h.shape[0]
        dx = 2.0 / n
        return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _feasible_delta(h, i, j, rng, step):
    """Return a feasible delta to add to h[i] and subtract from h[j],
    respecting the bounds [0,1] and the current step size."""
    low = max(-h[i], h[j] - 1.0)
    high = min(1.0 - h[i], h[j])
    if low >= high:
        return None
    # propose a delta scaled by the current step size
    delta = (rng.random() - 0.5) * step
    # clip to the feasible interval
    delta = max(low, min(high, delta))
    if abs(delta) < 1e-12:
        return None
    return delta


def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function minimizing the Erdős C5 overlap bound."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise h
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.shape[0]
    else:
        n = 200                     # default size if no parent provided
        h = np.full(n, 0.5, dtype=float)

    # Ensure the integral constraint exactly: sum(h) == n/2
    target_sum = n / 2.0
    cur_sum = h.sum()
    if not np.isclose(cur_sum, target_sum, atol=1e-12):
        # Simple projection: scale and then clip, iterating a few times
        for _ in range(5):
            h = np.clip(h * (target_sum / h.sum()), 0.0, 1.0)
            # adjust any residual error by adding to a random index
            err = target_sum - h.sum()
            if abs(err) < 1e-12:
                break
            idx = rng.integers(0, n)
            h[idx] = np.clip(h[idx] + err, 0.0, 1.0)

    # ------------------------------------------------------------------
    # Optimisation parameters
    # ------------------------------------------------------------------
    best_h = h.copy()
    best_c5 = _c5_val(best_h)

    # If the initial construction already beats the target, return early
    target_c5 = 0.3808
    if best_c5 <= target_c5:
        return best_h.tolist(), float(best_c5), int(n)

    # Simulated‑annealing style search
    cur_h = h.copy()
    cur_c5 = best_c5

    # Temperature and step schedules
    temp = 0.03
    temp_decay = 0.9995
    step = 0.5
    step_decay = 0.9997

    no_improve = 0
    max_no_improve = 150000

    start_time = time.time()
    time_limit = budget_s - 0.5   # leave a tiny safety margin

    while time.time() - start_time < time_limit:
        # pick two distinct positions
        i, j = rng.choice(n, size=2, replace=False)
        delta = _feasible_delta(cur_h, i, j, rng, step)
        if delta is None:
            continue

        # Apply the move
        cur_h[i] += delta
        cur_h[j] -= delta

        new_c5 = _c5_val(cur_h)

        # Acceptance test (simulated annealing)
        accept = False
        if new_c5 < cur_c5:
            accept = True
        else:

            prob = math.exp(-(new_c5 - cur_c5) / max(temp, 1e-12))
            if rng.random() < prob:

        if accept:

            # Record best solution seen so far

                best_c5 = new_c5

            cur_h[j] += delta
            no_improve += 1
