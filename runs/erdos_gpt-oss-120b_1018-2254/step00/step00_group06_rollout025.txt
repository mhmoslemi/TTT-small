import numpy as np
import time
import random

def project_bounds_sum(v, target):
    # binary search for lambda
    low = np.min(v) - 1.0
    high = np.max(v) + 1.0
    for _ in range(60):
        lam = (low + high) / 2.0
        x = np.clip(v - lam, 0, 1)
        s = np.sum(x)
        if s > target:
            low = lam
        else:
            high = lam
    x = np.clip(v - (low+high)/2.0, 0, 1)
    # adjust residual
    diff = target - np.sum(x)
    if abs(diff) > 1e-8:
        # allocate diff to non-bound elements
        free = (x > 1e-12) & (x < 1-1e-12)
        if np.any(free):
            inc = diff / np.count_nonzero(free)
            x[free] += inc
            x = np.clip(x, 0, 1)
        else:
            # all at bounds, distribute to a random index within bounds capacity
            # find index with capacity
            if diff > 0:
                capacity = 1 - x
                idx = np.argmax(capacity)
                delta = min(diff, capacity[idx])
                x[idx] += delta
            else:
                capacity = x
                idx = np.argmax(capacity)
                delta = min(-diff, capacity[idx])
                x[idx] -= delta
    return x

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def random_feasible(n, target):
    v = np.random.rand(n)
    return project_bounds_sum(v, target)

    idx = np.random.choice(len(pop), size=k, replace=False)
    best_i = idx[0]
    best_f = fitness[best_i]
    for i in idx[1:]:
        if fitness[i] < best_f:
            best_i = i
            best_f = fitness[i]
    return pop[best_i]

    n = parent1.shape[0]
    cut = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:cut], parent2[cut:]])
    child2 = np.concatenate([parent2[:cut], parent1[cut:]])
    return child1, child2

    mask = np.random.rand(h.shape[0]) < mutation_rate
    noise = np.random.normal(0, sigma, size=h.shape[0])
    h = h + mask * noise
    return h

    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    s = idx - (len(h) - 1)
    return s, corr[idx]

def gradient_wrt_shift(h, s):
    n = h.shape[0]
    grad = np.zeros_like(h)
    # term1: (1 - h[i+s]) for i where i+s in bounds
    if s >= 0:
        i = np.arange(0, n - s)
        grad[i] = 1 - h[i + s]
        # term2: -h[i - s] for i where i-s in bounds => i from s to n-1
        i2 = np.arange(s, n)
        grad[i2] -= h[i2 - s]
    else:
        # s negative, let t = -s >0 for symmetry
        t = -s
        i = np.arange(t, n)  # i such that i+s >=0 -> i >= t
        grad[i] = 1 - h[i + s]  # i+s = i - t
        i2 = np.arange(0, n - t)
        grad[i2] -= h[i2 - s]   # i2 - s = i2 + t
    return grad

def gradient_descent_refine(h, target, max_iter, time_limit, start_time):
    best = h.copy()
    best_val = compute_c5(best)
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        s, cur_corr = worst_shift_and_corr(h)
        grad = gradient_wrt_shift(h, s)
        if np.allclose(grad, 0):
            break
        # step size decreasing
        step = 0.1 / (1 + it*0.01)
        h_new = h - step * grad
        h_new = project_bounds_sum(h_new, target)
        val = compute_c5(h_new)
        if val < best_val:
            best_val = val
            best = h_new.copy()
            h = h_new
        else:
            # occasionally accept if improvement not found; use small probability
            if np.random.rand() < 0.01:
                h = h_new
    return best, best_val
