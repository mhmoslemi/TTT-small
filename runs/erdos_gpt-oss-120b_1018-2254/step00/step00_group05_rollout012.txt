    # adjust via scaling and clipping loop
    for _ in range(10):
        total = h.sum()
        if abs(total - target) < 1e-12:
            break
        factor = target / total if total != 0 else 0
        h = np.clip(h * factor, 0, 1)
    # If still not exact, adjust with pairwise moves
    diff = target - h.sum()
    if abs(diff) > 1e-10:

                diff -= delta

import time
import math
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a given step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project_to_target_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Scale and adjust a vector so that its entries lie in [0,1] and sum to target."""
    total = h.sum()
    if total == 0.0:
        h[:] = target / h.size
        return h
    factor = target / total
    h = np.clip(h * factor, 0.0, 1.0)
    residual = target - h.sum()
    attempts = 0
    max_attempts = 2000
    n = h.size
    while abs(residual) > 1e-12 and attempts < max_attempts:
        idx = rng.integers(n)
        if residual > 0.0:
            cap = 1.0 - h[idx]
            delta = min(cap, residual)
        else:
            cap = h[idx]
            delta = -min(cap, -residual)
        if delta != 0.0:
            h[idx] += delta
            residual -= delta
        attempts += 1
    if abs(residual) > 1e-12:
        # Final tiny correction on a safe element
        if residual > 0.0:
            h[0] = min(1.0, h[0] + residual)
        else:
            h[0] = max(0.0, h[0] + residual)
    return h

    Optimize a step function h on [0,2] to minimise the Erdős C5 overlap constant.

        Corresponding C5 value.

        Number of discretisation points.

        h = rng.random(70)
    n = h.size
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    h = _project_to_target_sum(h, target_sum, rng)

    # Ensure exact feasibility (tiny tweak)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0.0:
            h[0] = min(1.0, h[0] + diff)
        else:
            h[0] = max(0.0, h[0] + diff)

    # ------------------------------------------------------------------
    # Simulated‑annealing style optimisation
    # ------------------------------------------------------------------
    curr_h = h.copy()
    curr_c5 = compute_c5(curr_h)
    best_h = curr_h.copy()
    best_c5 = curr_c5

    start_time = time.time()
    # Parameters (tuned for speed / quality)
    epsilon = 0.02          # max absolute mass moved between two coordinates

    Tmin = 1e-6

        hi, hj = curr_h[i], curr_h[j]
