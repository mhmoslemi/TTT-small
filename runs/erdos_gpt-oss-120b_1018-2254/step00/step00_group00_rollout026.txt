    # compute sum.

        diff = target - current_sum

            break

def propose_move(h, sigma):
    n = len(h)
    i, j = np.random.randint(n), np.random.randint(n)
    while i == j:
        j = np.random.randint(n)
    delta = np.random.randn() * sigma
    # compute bounds for delta
    # when moving from i to j: new_i = h_i - delta, new_j = h_j + delta
    # Constraint: 0 <= h_i - delta <= 1 => delta ∈ [h_i - 1, h_i]
    # Constraint: 0 <= h_j + delta <= 1 => delta ∈ [-h_j, 1 - h_j]
    lower = max(h[i] - 1.0, -h[j])
    upper = min(h[i], 1.0 - h[j])
    if lower > upper:
        # no feasible delta move; just return h unchanged
        return h.copy()
    delta = np.clip(delta, lower, upper)
    cand = h.copy()
    cand[i] -= delta
    cand[j] += delta
    return cand

def project_onto_bounds_sum(v, target):
    # v: numpy array
    # Project onto [0,1] box and linear equality sum = target
    # Use iterative water-filling
    w = np.clip(v, 0.0, 1.0)
    # Compute sum
    # We'll adjust using free indices method.
    # set tolerance
    tol = 1e-10
    while True:
        current = w.sum()
        diff = target - current
        if abs(diff) <= tol:
            break
        free = (w > 0) & (w < 1)
        n_free = free.sum()
        if n_free == 0:
            # cannot adjust; break
            break
        delta = diff / n_free
        w[free] += delta
        # Clip again to bounds
        w = np.clip(w, 0.0, 1.0)
    return w

import time
import math
import random
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)*(1-h(x+k)) dx using discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)


def project_onto_bounds_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of v onto [0,1]^n ∩ {sum = target}."""
    w = np.clip(v, 0.0, 1.0)
    tol = 1e-12
    while True:
        cur = w.sum()
        diff = target - cur
        if abs(diff) <= tol:
            break
        free = (w > 0.0) & (w < 1.0)
        n_free = free.sum()
        if n_free == 0:
            # cannot adjust any further; break to avoid infinite loop
            break
        delta = diff / n_free
        w[free] += delta
        w = np.clip(w, 0.0, 1.0)
    return w


def propose_move(h: np.ndarray, sigma: float) -> np.ndarray:
    """Generate a new feasible h by moving a small amount of mass between two indices."""
    n = h.shape[0]
    i, j = np.random.randint(n), np.random.randint(n)
    while i == j:
        j = np.random.randint(n)

    # delta will be transferred from i to j (i loses, j gains)
    delta = np.random.randn() * sigma

    # feasible interval for delta so that both entries stay in [0,1]
    lower = max(h[i] - 1.0, -h[j])          # h[i]-delta >=0, h[j]+delta >=0
    upper = min(h[i], 1.0 - h[j])           # h[i]-delta <=1, h[j]+delta <=1
    if lower > upper:
        return h.copy()
    delta = np.clip(delta, lower, upper)

    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new


def optimize_h(h_start: np.ndarray, time_budget: float) -> tuple:
    """Simulated‑annealing style local search for lower C5."""
    n = h_start.shape[0]
    target_sum = n / 2.0

    best_h = h_start.copy()
    best_score = compute_c5(best_h)
    curr_h = best_h.copy()
    curr_score = best_score

    start = time.time()
    # initial schedule parameters
    sigma_start, sigma_end = 0.20, 0.01
    temp_start, temp_end = 0.05, 0.001

    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        sigma = sigma_start * (1 - frac) + sigma_end * frac
        temperature = temp_start * (1 - frac) + temp_end * frac

        cand_h = propose_move(curr_h, sigma)
        cand_score = compute_c5(cand_h)

        # accept if better or with Metropolis probability
        if cand_score < curr_score or random.random() < math.exp(-(cand_score - curr_score) / max(temperature, 1e-12)):
            curr_h = cand_h
            curr_score = cand_score

        if curr_score < best_score:
            best_h = curr_h.copy()
            best_score = curr_score

    return best_h, best_score
