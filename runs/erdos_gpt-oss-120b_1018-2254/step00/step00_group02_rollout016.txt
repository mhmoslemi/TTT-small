def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    # obtain initial h if provided
    try:
        init = np.array(initial_h_values, dtype=np.float64)
        n0 = init.size

        n = 180

    # ensure feasibility
    target_sum = n / 2.0
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        scale = target_sum / cur_sum
        h = h * scale
        # clip and correct slight error
        np.clip(h, 0.0, 1.0, out=h)
        cur_sum = h.sum()
        diff = cur_sum - target_sum
        if abs(diff) > 1e-9:
            for i in range(n):
                if 0 <= h[i] - diff <= 1:
                    h[i] -= diff
                    break

    dx = 2.0 / n

    def compute_c5(v):
        corr = np.correlate(v, 1.0 - v, mode='full') * dx
        return float(np.max(corr))

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    start = time.time()
    max_iter = 2000000
    T_start = 0.02
    T_end = 1e-4
    step_start = 0.2
    step_end = 0.001
    restart_interval = 300000
    last_improve = 0

    for it in range(max_iter):
        if time.time() - start > budget_s * 0.95:
            break

        # schedule
        frac = it / max_iter
        T = T_start * (1.0 - frac) + T_end * frac
        max_step = step_start * (1.0 - frac) + step_end * frac

        # pick two distinct indices
        i = np.random.randint(n)
        j = np.random.randint(n-1)
        if j >= i:
            j += 1

        hi, hj = h[i], h[j]
        low = max(-hj, hi - 1.0)
        high = min(hi, 1.0 - hj)
        if low >= high:
