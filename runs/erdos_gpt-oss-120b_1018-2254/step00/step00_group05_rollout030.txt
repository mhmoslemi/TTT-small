def solve_milp(n, time_limit):
    import cvxpy as cp
    h = cp.Variable(n, boolean=True)
    M = cp.Variable()
    constraints = [cp.sum(h) == n/2]
    # Ensure h in [0,1] implicitly Boolean.
    # For each shift...
    for s in range(-(n-1), n):
        if s == 0:
            continue
        start = max(0, -s)
        end = min(n-1, n-1 - s)
        L = end - start + 1
        if L <= 0:
            continue
        w = cp.Variable(L, boolean=True)
        # linking constraints
        idx_i = np.arange(start, end+1)
        idx_j = idx_i + s
        # vectorized constraints
        constraints.append(w <= h[idx_i])
        constraints.append(w <= h[idx_j])
        constraints.append(w >= h[idx_i] + h[idx_j] - 1)
        # correlation expression
        corr = cp.sum(h[idx_i]) - cp.sum(w)
        constraints.append(corr <= M)
    # Ensure M >=0 perhaps
    constraints.append(M >= 0)
    prob = cp.Problem(cp.Minimize(M), constraints)
    # Use solver with time limit:
    prob.solve(solver=cp.GLPK_MI, glpk={'tmlim': time_limit}) # but not sure about syntax.

import numpy as np, time, math

def _c5_of_h(h: np.ndarray) -> float:
    """Compute the overlap constant C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full") * dx)
