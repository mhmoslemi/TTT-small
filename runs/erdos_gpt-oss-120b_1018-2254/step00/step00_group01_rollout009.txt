import numpy as np
import time
import math

def _project_to_box_sum(x: np.ndarray, target: float) -> np.ndarray:
    """
    Project vector x onto the box [0,1]^n intersect {sum = target}.
    The solution is y_i = clip(x_i - λ, 0, 1) for a suitable λ.
    """
    lo = np.min(x) - 1.0          # ensures all entries become 1 when λ ≤ lo
    hi = np.max(x)                # ensures all entries become 0 when λ ≥ hi
    for _ in range(80):           # sufficient for double precision
        lam = (lo + hi) * 0.5
        y = np.clip(x - lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)


def _compute_c5(h: np.ndarray) -> float:
    """C5 = max_k ∫ h(x)(1−h(x+k))dx  (discretized)."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(conv.max() * dx)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a step function h on [0,2] with n points that minimizes C5.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()

    # Determine number of points and an initial feasible vector.
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = h0.shape[0]
        # Project in case the provided parent does not satisfy the sum exactly.
        target_sum = n / 2.0
        h0 = _project_to_box_sum(h0, target_sum)
    else:
        # Default size when no parent is supplied.
        n = 47
        target_sum = n / 2.0
        # Random feasible start.
        rnd = np.random.rand(n)
        h0 = _project_to_box_sum(rnd, target_sum)

    target_sum = n / 2.0
    best_h = h0
    best_c5 = _compute_c5(best_h)

    # Local search state.
    cur_h = best_h.copy()
    cur_c5 = best_c5

    step = 0.2                # typical magnitude for two‑index moves
    min_step = 0.001
    max_step = 0.3
    no_improve = 0            # counter for stagnation
    restart_interval = 4000   # iterations after which a random restart is tried

    iteration = 0
    while True:
        if time.time() - start_time >= budget_s:
            break
        iteration += 1

        # Occasionally perform a global random perturbation.
        if np.random.rand() < 0.03:
            cand = cur_h + step * (np.random.rand(n) - 0.5) * 2.0
            cand = _project_to_box_sum(cand, target_sum)
        else:
            # Two‑index mass transfer, guaranteed to keep feasibility.
            i, j = np.random.choice(n, 2, replace=False)
            # Transfer from i to j (i loses, j gains)
            max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
            if max_delta <= 0.0:
                # try opposite direction
                max_delta = min(step, cur_h[j], 1.0 - cur_h[i])
                if max_delta <= 0.0:
                    continue
                i, j = j, i
            delta = np.random.rand() * max_delta
            cand = cur_h.copy()
            cand[i] -= delta
            cand[j] += delta

        # Evaluate candidate.
        cand_c5 = _compute_c5(cand)

        # Accept if better.
        if cand_c5 < cur_c5:
            cur_h = cand
            cur_c5 = cand_c5
            no_improve = 0
            # Update global best if applicable.
            if cand_c5 < best_c5:
                best_h = cand.copy()
                best_c5 = cand_c5
        else:
            no_improve += 1

        # Adaptive step‑size control.
        if iteration % 500 == 0:
            if no_improve > 2000:
                step = max(step * 0.7, min_step)
                no_improve = 0
            else:
                step = min(step * 1.05, max_step)

        # Periodic random restart to escape possible deep local minima.
        if iteration % restart_interval == 0:
            rnd = np.random.rand(n)
            cur_h = _project_to_box_sum(rnd, target_sum)
            cur_c5 = _compute_c5(cur_h)

    return (best_h, best_c5, n)
