def project_to_feasible(x):
    n = len(x)
    target_sum = n / 2.0  # since sum must equal n/2
    # Clip to [0,1]
    y = np.clip(x, 0.0, 1.0)
    # Compute difference
    diff = target_sum - y.sum()
    # Water-filling iterative
    # Use tolerance.
    # While diff appreciable:
    #   free = (y > 0) & (y < 1)
    #   if free sum zero: break
    #   adjust = diff / free.sum()
    #   y[free] += adjust
    #   clip again
    #   recompute diff
    # Ensure diff small
    # At end may produce small residual due to rounding; adjust one element.
    # 
    # We'll iterate upto 100 times.
    for _ in range(100):
        if abs(diff) < 1e-9:
            break
        free = (y > 0) & (y < 1)
        n_free = free.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        # Clip adjust to avoid exceed bounds
        # for each free index, applying adjust may push out of [0,1]; handle by limiting
        # Instead of simple adjust, we apply adjust and then clip after loop. 
        y[free] += adjust
        # Clip
        y = np.clip(y, 0.0, 1.0)
        diff = target_sum - y.sum()
    # If still diff not zero, adjust first element within bounds
    if abs(diff) > 1e-8:
        # adjust any element not at bound:
        idxs = np.where((y > 0) & (y < 1))[0]
        if idxs.size > 0:
            i = idxs[0]
            new_val = y[i] + diff
            new_val = min(max(new_val, 0.0), 1.0)
            y[i] = new_val
        else:
            # All at bounds, but sum not match; distribute via small adjustments
            # adjust any element within [0,1].
            i = np.argmin(y) if diff < 0 else np.argmax(y)
            y[i] = min(max(y[i] + diff, 0.0), 1.0)
    return y

def project_to_feasible(x):
    n = x.shape[0]
    target = n / 2.0
    # Clip
    x = np.clip(x, 0.0, 1.0)
    # iterative water-filling
    for _ in range(100):
        cur_sum = x.sum()
        diff = target - cur_sum
        if abs(diff) < 1e-12:
            break
        free = (x > 0.0) & (x < 1.0)
        count_free = int(free.sum())
        if count_free == 0:
            # adjust a single element within bounds
            # find element that can be adjusted
            for idx in range(n):
                if (diff > 0 and x[idx] < 1.0) or (diff < 0 and x[idx] > 0.0):
                    x[idx] = min(max(x[idx] + diff, 0.0), 1.0)
                    break
            break
        delta = diff / count_free
        # Apply delta only to free entries, but ensure within bounds after addition
        x[free] += delta
        # Clip again
        x = np.clip(x, 0.0, 1.0)
    # final correction if needed
    cur_sum = x.sum()
    diff = target - cur_sum
    if abs(diff) > 1e-10:
        # adjust first element that can absorb diff
        for idx in range(n):
            new_val = x[idx] + diff
            if 0.0 <= new_val <= 1.0:
                x[idx] = new_val
                break
    return x

import time
import math
import numpy as np

def _dx(n):
    return 2.0 / n

def _correlation(h):
    """Full linear correlation of h with (1-h)."""
    return np.correlate(h, 1.0 - h, mode='full')

def _max_overlap(h, dx):
    """Return max_{k} integral = max_correlation * dx."""
    corr = _correlation(h)
    return np.max(corr) * dx

def _max_overlap_with_shift(h, dx):
    """Return (max value, shift index, full correlation array)."""
    corr = _correlation(h)
    idx = int(np.argmax(corr))
    max_val = corr[idx]
    shift = idx - (len(h) - 1)          # shift = k / dx, integer shift
    return max_val * dx, shift, corr

def _project_to_feasible(x):
    """Project x onto [0,1]^n with sum(x)=n/2 using water-filling."""
    n = x.shape[0]
    target = n / 2.0
    x = np.clip(x, 0.0, 1.0)
    for _ in range(100):
        cur = x.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        free = (x > 0.0) & (x < 1.0)
        n_free = int(free.sum())
        if n_free == 0:
            # adjust a single feasible entry
            for i in range(n):
                if (diff > 0 and x[i] < 1.0) or (diff < 0 and x[i] > 0.0):
                    x[i] = min(max(x[i] + diff, 0.0), 1.0)
                    break
            break
        delta = diff / n_free
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
    # final tiny correction
    cur = x.sum()
    diff = target - cur
    if abs(diff) > 1e-10:
        for i in range(n):
            new_val = x[i] + diff
            if 0.0 <= new_val <= 1.0:
                x[i] = new_val
                break
    return x

def _random_feasible(n):
    """Generate a random vector in [0,1]^n with exact sum n/2."""
    r = np.random.rand(n)
    return _project_to_feasible(r)

def _enforce_sum_exact(h):
    """Adjust a tiny amount to make sum exactly n/2 while staying within bounds."""
    n = h.shape[0]
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    for i in range(n):
        new_val = h[i] + diff
        if 0.0 <= new_val <= 1.0:
            h[i] = new_val
            break
    return h

def _random_swap(h, max_step):
    """Propose a feasible swap of mass between two positions."""
    n = h.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    # delta added to i, subtracted from j
    lower = max(-h[i], h[j] - 1.0)
    upper = min(1.0 - h[i], h[j])
    # limit by max_step
    lower = max(lower, -max_step)
    upper = min(upper, max_step)
    if lower >= upper:
        return None, None, None
    delta = np.random.uniform(lower, upper)
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    return new_h, i, j

def _targeted_swap(h, shift, max_step):
    """Swap mass focusing on indices that contribute most to the worst shift."""
    n = h.shape[0]
    contrib = np.empty(n, dtype=np.float64)

    if shift >= 0:
        s = shift
        # overlapping region
        contrib[:n - s] = h[:n - s] * (1.0 - h[s:])
        # out‑of‑range region (where shifted index is outside)
        contrib[n - s:] = h[n - s:]
    else:
        s = -shift
        contrib[s:] = h[s:] * (1.0 - h[:n - s])
        contrib[:s] = h[:s]

    donor = int(np.argmax(contrib))
    receiver = int(np.argmin(contrib))
    if donor == receiver:
        return None, None, None

    # we want to move mass from donor (high contribution) to receiver (low)
    i, j = receiver, donor
    # amount to add to i, subtract from j
    upper = min(1.0 - h[i], h[j])          # positive direction only
    if upper <= 0.0:
        return None, None, None
    delta = np.random.uniform(0.0, min(max_step, upper))
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    return new_h, i, j

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    margin = 2.0                      # safety margin before hard timeout

    # ----- initial vector -------------------------------------------------
    init = globals().get('initial_h_values')
    if init is not None:
        h = np.array(init, dtype=np.float64)
        n = h.shape[0]
        h = _project_to_feasible(h)   # guarantee feasibility
    else:
        n = 45
        h = _random_feasible(n)

    dx = _dx(n)
    best_h = h.copy()
    best_score = _max_overlap(best_h, dx)

    current_h = h.copy()
    current_score = best_score

    # ----- search parameters ---------------------------------------------
    T0 = 0.001                         # initial temperature for SA
    max_step = 0.5                     # max delta per swap
    no_improve = 0
    restart_limit = 5000
    iteration = 0

    while True:
        now = time.time()
        if now - start > budget_s - margin:
            break

        # evaluate current worst shift
        cur_score, cur_shift, _ = _max_overlap_with_shift(current_h, dx)

        # decide mutation type
        if np.random.rand() < 0.35 and cur_shift != 0:
            new_h, i, j = _targeted_swap(current_h, cur_shift, max_step)
        else:
            new_h, i, j = _random_swap(current_h, max_step)

        # if mutation could not be performed, skip
        if new_h is None:
            continue

        new_score, _, _ = _max_overlap_with_shift(new_h, dx)

        # Acceptance criteria (simulated annealing)
        if new_score < current_score:
            accept = True
        else:
            elapsed = now - start
            frac = elapsed / budget_s
            temperature = T0 * (1.0 - frac)
            # avoid division by zero
            if temperature > 0:
                prob = math.exp(-(new_score - current_score) / temperature)
                accept = np.random.rand() < prob
            else:
                accept = False

        if accept:
            current_h = new_h
            current_score = new_score
            no_improve = 0
        else:
            no_improve += 1

        # record best ever found
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_h = new_h.copy()
            no_improve = 0

        # temperature and step annealing
        iteration += 1
        max_step = max(max_step * 0.9999, 0.001)

        # occasional restart to escape stagnation
        if no_improve > restart_limit:
            current_h = _random_feasible(n)
            current_score = _max_overlap(current_h, dx)
            no_improve = 0

    # final correction to exact sum
    best_h = _enforce_sum_exact(best_h)
    best_score = _max_overlap(best_h, dx)

    return (best_h.tolist(), float(best_score), n)
