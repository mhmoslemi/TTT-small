import numpy as np
import time

# Helper: project a vector onto the box [0,1] with a given sum (target)
def _project_to_box_sum(v, target):
    # find lambda such that sum(clip(v+lambda,0,1)) = target
    lo = -np.max(v) - 1.0      # ensures all zero after clipping
    hi = 1.0 - np.min(v) + 1.0  # ensures all one after clipping
    for _ in range(50):
        lam = (lo + hi) * 0.5
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    x = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny adjustment to hit the target exactly
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        # adjust first non‑boundary element if possible
        free = np.where((x > 0) & (x < 1))[0]
        if free.size:
            i = free[0]
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
        else:
            # all at bounds – adjust first element within allowable range
            i = 0
            cap = (1.0 - x[i]) if diff > 0 else x[i]
            x[i] = x[i] + np.clip(diff, -cap, cap)
    return x

# Compute C5 for a given step function h
def _c5_value(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

# Tournament selection
def _tournament(pop, fit, size):
    idx = np.random.choice(len(pop), size, replace=False)
    best = idx[0]
    best_f = fit[best]
    for i in idx[1:]:
        if fit[i] < best_f:
            best = i
            best_f = fit[i]
    return pop[best]

# Crossover (single point)
def _crossover(p1, p2):
    n = p1.shape[0]
    pt = np.random.randint(1, n)
    child = np.concatenate([p1[:pt], p2[pt:]])
    return child

# Mutation with Gaussian noise on a subset of entries
def _mutate(v, rate, sigma, target):
    mask = np.random.rand(v.shape[0]) < rate
    noise = np.random.randn(v.shape[0]) * sigma
    v2 = v + mask * noise
    # projection restores feasibility
    return _project_to_box_sum(v2, target)

def run(seed=42, budget_s=1100, **kwargs):
    # --------------------------------------------------------------------
    # Setup
    # --------------------------------------------------------------------
    np.random.seed(seed)
    start = time.time()
    # Determine problem size
    init_h = globals().get('initial_h_values', None)
    if init_h is not None:
        init_h = np.asarray(init_h, dtype=float)
        n = init_h.shape[0]
    else:
        n = 70
    target_sum = n / 2.0
    dx = 2.0 / n

    # Evaluation function (use provided helper if available)
    eval_fn = globals().get('evaluate_erdos_solution', None)
    def _evaluate(v):
        if eval_fn is not None:
            return float(eval_fn(v))
        return _c5_value(v, dx)

    # --------------------------------------------------------------------
    # GA hyper‑parameters
    # --------------------------------------------------------------------
    pop_size = 30
    tournament_sz = 3
    crossover_prob = 0.9
    mutation_rate = 0.3
    sigma = 0.12
    max_gens = 1000
    patience = 80   # generations without improvement before early stop

    # --------------------------------------------------------------------
    # Initial population
    # --------------------------------------------------------------------
    population = []
    if init_h is not None:
        population.append(_project_to_box_sum(init_h, target_sum))
    while len(population) < pop_size:
        rnd = np.random.rand(n)
        population.append(_project_to_box_sum(rnd, target_sum))

    fitness = np.array([_evaluate(ind) for ind in population])
    best_idx = int(np.argmin(fitness))
    best_h = population[best_idx].copy()
    best_f = float(fitness[best_idx])
    no_improve = 0

    # --------------------------------------------------------------------
    # Evolution loop
    # --------------------------------------------------------------------
    gen = 0
    while gen < max_gens:
        # Time check
        if time.time() - start > budget_s * 0.95:
            break

        new_pop = [best_h]  # elitism: keep the best
        while len(new_pop) < pop_size:
            # parent selection
            p1 = _tournament(population, fitness, tournament_sz)
            p2 = _tournament(population, fitness, tournament_sz)

            # crossover
            if np.random.rand() < crossover_prob:
                child = _crossover(p1, p2)
            else:
                child = p1.copy()

            # mutation
            child = _mutate(child, mutation_rate, sigma, target_sum)
            new_pop.append(child)

        population = new_pop
        fitness = np.array([_evaluate(ind) for ind in population])

        # update best
        cur_best_idx = int(np.argmin(fitness))
        cur_best_f = float(fitness[cur_best_idx])
        if cur_best_f + 1e-12 < best_f:  # improvement
            best_f = cur_best_f
            best_h = population[cur_best_idx].copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= patience:
            break

        gen += 1

    # Final evaluation (ensuring feasibility)
    best_h = _project_to_box_sum(best_h, target_sum)
    best_f = _evaluate(best_h)

    return best_h, best_f, n
