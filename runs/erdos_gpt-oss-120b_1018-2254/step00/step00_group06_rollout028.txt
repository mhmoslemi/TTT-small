def tournament_select(pop, fit, rng, k=3):
    indices = rng.integers(0, len(pop), size=k)
    best = indices[0]
    best_fit = fit[best]
    for i in indices[1:]:
        if fit[i] < best_fit:
            best = i
            best_fit = fit[i]
    return best
```

def crossover(p1, p2, rng):
    n = len(p1)
    point = rng.integers(1, n) # inclusive start but exclusive end
    child1 = np.concatenate([p1[:point], p2[point:]])
    child2 = np.concatenate([p2[:point], p1[point:]])
    return child1, child2
```

def repair(h, target=None):
    if target is None:
        n = len(h)
        target = n / 2.0
    # clip
    np.clip(h, 0.0, 1.0, out=h)
    # adjust sum
    s = h.sum()
    diff = target - s
    # if diff is very small, just return
    if abs(diff) < 1e-12:
        return
    # choose indices that can be adjusted in direction of diff
    if diff > 0:
        # need increase -> indices with h < 1
        indices = np.where(h < 1.0)[0]
        if len(indices) == 0:
            return
        # distribute diff equally across eligible indices up to capacity
        # compute capacities
        capacities = 1.0 - h[indices]
        total_cap = capacities.sum()
        if total_cap < diff:
            # cannot fully satisfy diff, set to max and adjust remainder negative later
            # set those to 1
            h[indices] = 1.0
            diff_remaining = diff - total_cap
            # now need to decrease somewhere? This is unlikely.
            # fallback simple: subtract diff_remaining uniformly.
            # This won't happen often; just continue subtract from any >0.
            # For simplicity, do a loop to adjust any.
            # Not needed for typical diff.
        else:
            # allocate proportionally (or equally)
            # prop = capacities / total_cap * diff
            # add to each
            add = diff * capacities / total_cap
            h[indices] += add
    else:
        # diff < 0: need decrease -> indices with h > 0
        indices = np.where(h > 0.0)[0]
        if len(indices) == 0:
            return
        capacities = h[indices]  # amount we can decrease
        total_cap = capacities.sum()
        if total_cap < -diff:
            # cannot reduce enough
            h[indices] = 0.0
            # leftover diff cannot be fixed; ignore.
        else:
            sub = (-diff) * capacities / total_cap
            h[indices] -= sub
    # after adjustment, ensure numerical within bounds via clip
    np.clip(h, 0.0, 1.0, out=h)
    # recompute sum; if still off a little due to rounding, adjust a random index
    s = h.sum()
    diff = target - s
    if abs(diff) > 1e-12:
        # adjust a random index within bounds
        if diff > 0:
            # need increase; find index where h < 1
            idx = np.where(h < 1.0)[0]
            if len(idx) > 0:
                i = rng.choice(idx)
                # safe amount:
                inc = min(diff, 1.0 - h[i])
                h[i] += inc
        else:
            idx = np.where(h > 0.0)[0]
            if len(idx) > 0:
                i = rng.choice(idx)
                dec = min(-diff, h[i])
                h[i] -= dec
    # final clip
    np.clip(h, 0.0, 1.0, out=h)

    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def tournament_select(fitness, rng, k=3):
    indices = rng.integers(0, len(fitness), size=k)
    best = indices[0]
    best_fit = fitness[best]
    for idx in indices[1:]:
        if fitness[idx] < best_fit:
            best = idx
            best_fit = fitness[idx]
    return best
```

def crossover(p1, p2, rng):
    n = len(p1)
    # choose crossover point not at ends
    point = rng.integers(1, n)
    child1 = np.concatenate([p1[:point], p2[point:]])
    child2 = np.concatenate([p2[:point], p1[point:]])
    return child1, child2

                c2 = p2.copy()

            mutate(c2, rng)
            # Repair
            repair(c1, target, rng)
            repair(c2, target, rng)

            if len(new_pop) < popsize:

        # Update best

    def tournament(fit):
        idx = rng.integers(0, len(fit), size=3)
        best = idx[0]
        best_val = fit[best]
        for i in idx[1:]:
            if fit[i] < best_val:
                best = i
                best_val = fit[i]
        return best

    def crossover(p1, p2):
        pt = rng.integers(1, n)
        c1 = np.concatenate([p1[:pt], p2[pt:]])
        c2 = np.concatenate([p2[:pt], p1[pt:]])
        return c1, c2

    def mutate(ind):
        mask = rng.random(n) < 0.01
        if not np.any(mask):
            return
        noise = rng.normal(0.0, 0.1, size=n)
        ind[mask] += noise[mask]

    # ------------------------------------------------------------------
    # Main GA loop
    # ------------------------------------------------------------------
    start = time.time()
    while time.time() - start < budget_s:
        new_pop = [best_h.copy()]  # elitism
        while len(new_pop) < popsize:
            p1 = population[tournament(fitness)]
            p2 = population[tournament(fitness)]
            if rng.random() < 0.7:
                c1, c2 = crossover(p1, p2)
            else:
                c1 = p1.copy()
                c2 = p2.copy()
            mutate(c1)
            mutate(c2)
            repair(c1)
            repair(c2)
            new_pop.append(c1)
            if len(new_pop) < popsize:
                new_pop.append(c2)
        fitness = np.array([compute_c5(ind) for ind in new_pop])
        cur_best = int(np.argmin(fitness))
        cur_f = float(fitness[cur_best])
        if cur_f < best_f:
            best_f = cur_f
            best_h = new_pop[cur_best].copy()
            if best_f <= 0.3808:
                population = new_pop
                break
        population = new_pop

    # ------------------------------------------------------------------
    # Local refinement (pairwise transfer)
    # ------------------------------------------------------------------
    while time.time() - start < budget_s:
        h_new = best_h.copy()
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if rng.random() < 0.5:
            max_delta = min(h_new[i], 1.0 - h_new[j])
            if max_delta <= 0:
                continue
            delta = rng.random() * max_delta * 0.5
            h_new[i] -= delta
            h_new[j] += delta
        else:
            max_delta = min(h_new[j], 1.0 - h_new[i])
            if max_delta <= 0:
                continue
            delta = rng.random() * max_delta * 0.5
            h_new[j] -= delta
            h_new[i] += delta
        f_new = compute_c5(h_new)
        if f_new < best_f:
            best_f = float(f_new)
            best_h = h_new.copy()
            if best_f <= 0.3808:
                break

    # ------------------------------------------------------------------
    # Finalisation
    # ------------------------------------------------------------------
    repair(best_h)
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), float(final_c5), n)
