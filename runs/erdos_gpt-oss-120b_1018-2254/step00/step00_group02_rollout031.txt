def correlation_max(h):
    n = len(h)
    dx = 2.0 / n
    m = 1 << ((2*n - 1).bit_length())

    # compute fft of h and (1-h) reversed.
    h_fft = np.fft.rfft(h, n=m)
    g_rev = (1 - h)[::-1]
    g_fft = np.fft.rfft(g_rev, n=m)
    conv = np.fft.irfft(h_fft * g_fft, n=m)[:2*n-1]

def anneal(h, budget_s, start_time):
    n = len(h)
    dx = 2/n
    current_h = h.copy()
    current_val = compute_c5(current_h)
    best_h = current_h.copy()
    best_val = current_val

    iteration = 0
    # schedule parameters
    while time.time() - start_time < budget_s * 0.9:
        iteration += 1
        # random swap
        ones = np.where(current_h == 1)[0]
        zeros = np.where(current_h == 0)[0]
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # swap
        current_h[i] = 0
        current_h[j] = 1
        new_val = compute_c5(current_h)
        # temperature schedule: start high ~ 0.1 maybe; second half low.
        elapsed = time.time() - start_time
        frac = elapsed / (budget_s * 0.9)
        T = max(0.001, 0.1 * (1 - frac))
        # decide acceptance
        if new_val < current_val or np.random.rand() < np.exp(-(new_val - current_val) / T):

            if new_val < best_val:
                best_val = new_val
                best_h = current_h.copy()

            # revert
            current_h[i] = 1
            current_h[j] = 0

        if new_val < current_val or random.random() < np.exp((current_val - new_val) / (T + 1e-12)):

import time
import random
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x) (1-h(x+k)) dx for the discretized h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_feasible(x: np.ndarray, target_sum: float) -> np.ndarray:
    """Project x onto the box [0,1] while enforcing sum(x)=target_sum."""
    # Find lambda so that sum(clip(x-lambda,0,1)) = target_sum.
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(x - lam, 0.0, 1.0)

def local_search(h_start: np.ndarray, time_limit: float) -> tuple:
    """Simple simulated‑annealing style search that respects the constraints."""
    n = h_start.shape[0]
    target_sum = n / 2.0
    h = h_start.copy()
    current_val = compute_c5(h)
    best_h = h.copy()
    best_val = current_val

    start = time.time()
    max_delta_initial = 0.15  # initial maximal transfer amount

    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        # temperature and step size decay linearly
        T = 0.12 * (1.0 - frac) + 1e-4
        max_delta = max_delta_initial * (1.0 - frac) + 1e-5

        # --- directed move targeting the worst shift (largest overlap) ---
        if random.random() < 0.68:
            # autocorrelation (h with itself)
            auto = np.correlate(h, h, mode="full") * (2.0 / n)
            center = n - 1
            # ignore the zero shift (center) – we need the smallest autocorrelation
            tail = auto[center + 1 :]
            if tail.size == 0:
                continue
            s = int(np.argmin(tail)) + 1          # positive shift with minimal I_s
            # compute contribution weights for this shift
            w = np.zeros(n)
            if s < n:
                w[: n - s] += h[s:]               # forward neighbours
                w[s:] += h[: n - s]               # backward neighbours
            inc_idx = np.where(h < 1.0)[0]
            dec_idx = np.where(h > 0.0)[0]
            if inc_idx.size == 0 or dec_idx.size == 0:
                continue
            i = inc_idx[np.argmax(w[inc_idx])]
            j = dec_idx[np.argmin(w[dec_idx])]
            max_inc = min(max_delta, 1.0 - h[i])
            max_dec = min(max_delta, h[j])
            move = min(max_inc, max_dec)
            if move <= 0.0:
                continue
            delta = random.random() * move
            h[i] += delta
            h[j] -= delta
            new_val = compute_c5(h)
            accept = new_val < current_val or random.random() < np.exp((current_val - new_val) / (T + 1e-12))
            if accept:
                current_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best_h = h.copy()
            else:
                h[i] -= delta
                h[j] += delta
            continue

        # --- random transfer move ---
        i, j = random.sample(range(n), 2)          # i will receive, j will give
        # ensure direction respects bounds
        if h[i] >= 1.0 or h[j] <= 0.0:
            i, j = j, i
            if h[i] >= 1.0 or h[j] <= 0.0:
                continue
        max_inc = min(max_delta, 1.0 - h[i])
        max_dec = min(max_delta, h[j])
        move = min(max_inc, max_dec)
        if move <= 0.0:
            continue
        delta = random.random() * move
        h[i] += delta
        h[j] -= delta
        new_val = compute_c5(h)
        accept = new_val < current_val or random.random() < np.exp((current_val - new_val) / (T + 1e-12))
        if accept:
            current_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            h[i] -= delta
            h[j] += delta

        if abs(h.sum() - target_sum) > 1e-8:
            h = project_to_feasible(h, target_sum)
            current_val = compute_c5(h)
            if current_val < best_val:
                best_val = current_val
                best_h = h.copy()

    return best_h, best_val

        n_init = init_vec.size
        # aim for a finer discretisation (multiple of the original size)
        n_points = max(180, (n_init // 5) * 5)           # keep it multiple of 5 for convenience
        if n_points < 180:
            n_points = 180
    except Exception:
        init_vec = None
        n_points = 200

    # Build initial h
    if init_vec is not None:
        # linear interpolation to the desired resolution
        xs_src = np.linspace(0.0, 2.0, init_vec.size, endpoint=False)
        xs_tgt = np.linspace(0.0, 2.0, n_points, endpoint=False)
        h0 = np.interp(xs_tgt, xs_src, init_vec)
        # enforce the exact integral constraint
        h0 = project_to_feasible(h0, n_points / 2.0)
    else:
        # start from a random balanced binary pattern
        h0 = np.zeros(n_points, dtype=float)
        ones = np.random.choice(n_points, n_points // 2, replace=False)
        h0[ones] = 1.0

    # allocate time for the optimisation phase
    start_time = time.time()
    remaining = budget_s - (start_time - start_time)  # small safety subtraction
    search_time = max(0.0, remaining * 0.94)          # keep a small margin for final steps

    best_h, best_val = local_search(h0, search_time)

    # final safety: ensure feasibility
    best_h = project_to_feasible(best_h, n_points / 2.0)
    best_val = compute_c5(best_h)

    return best_h, best_val, n_points
