import numpy as np
import random
import time
import math

def compute_c5(h: np.ndarray, dx: float) -> float:
    """Return max_k ∫ h(x)(1-h(x+k)) dx approximated by correlation."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_constraints(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the box [0,1]^n intersected with
    the hyperplane sum(x)=target_sum.
    """
    n = v.shape[0]
    lo, hi = -1.0, 1.0  # enough to drive all entries to 0 or 1
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            hi = lam
        else:
            lo = lam
    proj = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # Minor residual correction if clipping saturated some entries
    diff = proj.sum() - target_sum
    if abs(diff) > 1e-9:
        free = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        cnt = int(free.sum())
        if cnt > 0:
            proj[free] -= diff / cnt
            np.clip(proj, 0.0, 1.0, out=proj)
    return proj
