def project_sum_and_bounds(x, target_sum):
    # Clip to [0,1]
    x = np.clip(x, 0.0, 1.0)
    # Compute difference
    diff = target_sum - np.sum(x)
    # While not small:
    for _ in range(10):
        # Find indices not at bounds
        free = (x > 0) & (x < 1)
        nfree = np.count_nonzero(free)
        if nfree == 0:
            break
        delta = diff / nfree
        x[free] += delta
        # Clip again
        over = x > 1
        under = x < 0
        if over.any() or under.any():
            # set them at bound
            x[over] = 1
            x[under] = 0
        diff = target_sum - np.sum(x)
        if abs(diff) < 1e-12:
            break
    return x

def optimize_h(h0, budget_time, start_time):

        f = 2*h - 1

        # autocorr_f
        ac = np.correlate(f, f, mode='full')

        # find minimal value
        min_idx = np.argmin(ac)

        # compute shift k
        k = min_idx - (n - 1)

        if k < 0: k = -k # absolute

        # compute gradient w.r.t h

        # shift plus

def project_h_bounds_sum(x, target_sum, max_iter=10):
    # Clip to [0,1] and adjust to meet sum constraint by water-filling.
    x = np.clip(x, 0.0, 1.0)
    diff = target_sum - x.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        free = (x > 0.0) & (x < 1.0)
        nfree = int(free.sum())

            # all at bounds, cannot fix further
            break
        delta = diff / nfree
        x[free] += delta
        # enforce bounds again
        x = np.clip(x, 0.0, 1.0)
        diff = target_sum - x.sum()
    # If still off due to all bound situation, distribute minimal adjustment to avoid violation.

import numpy as np
import time

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def evaluate_c5(h):
    """Return the Erdős C₅ value for a step function h."""
    try:
        # pre‑imported helper (may be a fast implementation)
        return evaluate_erdos_solution(h)
    except Exception:
        n = len(h)
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full')
        return float(np.max(corr) * dx)


def project_h_bounds_sum(x, target_sum, max_iter=12):
    """
    Project vector x onto the box [0,1]^n while forcing sum(x)=target_sum.
    Simple water‑filling iteration; sufficient for modest dimensions.
    """
    x = np.clip(x, 0.0, 1.0)
    diff = target_sum - x.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        free = (x > 0.0) & (x < 1.0)
        nfree = int(free.sum())
        if nfree == 0:
            # nothing adjustable – break to avoid division by zero
            break
        delta = diff / nfree
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
        diff = target_sum - x.sum()
    # final safety – slight scaling if still off (rare)
    if abs(x.sum() - target_sum) > 1e-9:
        scale = target_sum / x.sum()
        x = np.clip(x * scale, 0.0, 1.0)
        diff = target_sum - x.sum()
        if abs(diff) > 1e-12:
            # distribute leftover to any non‑tight entry
            idx = np.where((x > 0.0) & (x < 1.0))[0]
            if idx.size:
                x[idx[0]] = np.clip(x[idx[0]] + diff, 0.0, 1.0)
    return x


def upsample_solution(h, n_target):
    """Linearly up‑sample a step function to a new resolution."""
    n_curr = len(h)
    if n_curr == n_target:
        return h.copy()
    x_curr = np.linspace(0.0, 2.0, n_curr, endpoint=False)
    x_tgt = np.linspace(0.0, 2.0, n_target, endpoint=False)
    h_interp = np.interp(x_tgt, x_curr, h)
