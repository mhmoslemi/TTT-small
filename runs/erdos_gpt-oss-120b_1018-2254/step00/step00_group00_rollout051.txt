def project_to_capped_simplex(v, target):
    # v is a numpy array
    # project onto {x | 0 <= x_i <= 1, sum x = target}
    # We'll use iterative method:
    x = np.clip(v, 0, 1)
    for _ in range(10):
        total = x.sum()
        if abs(total - target) < 1e-8:
            break
        diff = target - total
        # Determine indices where x can be moved
        if diff > 0:
            # increase some components not at upper bound
            free = (x < 1 - 1e-12)
        else:
            free = (x > 1e-12)
        if not np.any(free):
            # cannot adjust; break
            break
        # distribute diff equally among free component
        inc = diff / free.sum()
        # apply incremental update
        x[free] += inc
        # clamp again to [0,1]
        x = np.clip(x, 0, 1)
    # after iteration, may still have small error; adjust final by adding diff to first free component (if any)
    total = x.sum()
    if abs(total - target) > 1e-6:
        # adjust small diff by adding to the component with capacity
        if total < target:
            # increase a component not at 1
            idx = np.where(x < 1)[0][0]
            x[idx] += target - total
        else:
            idx = np.where(x > 0)[0][0]
            x[idx] -= total - target
        x = np.clip(x, 0, 1)
    return x
```

    def repair(h):

        x = np.clip(h, 0.0, 1.0)
        # iterative water-filling
        for _ in range(10):
            total = x.sum()
            diff = target_sum - total
            if abs(diff) < 1e-9:

            if diff > 0:

            else:
                free = x > 0.0

import time
import random
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h (as a numpy array), its C5 bound, and the number of points."""
    start_time = time.time()
    rng = np.random.RandomState(seed)
    py_random = random.Random(seed)

    # ------------------------------------------------------------------
    # Helper: projection onto the capped simplex {0<=x_i<=1, sum x_i = target}
    # ------------------------------------------------------------------
    def project_to_capped_simplex(v, target):
        x = np.clip(v, 0.0, 1.0)
        # iterative water-filling
        for _ in range(100):
            total = x.sum()
            diff = target - total
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                free = x < 1.0 - 1e-12
                capacities = 1.0 - x[free]
            else:
                free = x > 1e-12
                capacities = x[free]
            if not np.any(free):
                break
            cap_sum = capacities.sum()
            if cap_sum < abs(diff) - 1e-12:
                # saturate all free variables
                if diff > 0:
                    x[free] = 1.0
                else:
                    x[free] = 0.0
                continue
            # distribute proportionally to capacity
            allocation = diff * capacities / cap_sum
            x[free] += allocation
            x = np.clip(x, 0.0, 1.0)
        # final tiny adjustment if needed
        total = x.sum()
        diff = target - total
        if abs(diff) > 1e-8:
            if diff > 0:
                idxs = np.where(x < 1.0)[0]
            else:
                idxs = np.where(x > 0.0)[0]
            if idxs.size:
                idx = idxs[0]
                x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
        return x

    # ------------------------------------------------------------------
    # Evaluate C5 for a candidate h
    # ------------------------------------------------------------------
    def evaluate(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return corr.max() * (2.0 / len(h))

    # ------------------------------------------------------------------
    # Obtain initial vector if provided
    # ------------------------------------------------------------------
    try:
        init_h_raw = np.array(initial_h_values, dtype=float)
        N = init_h_raw.shape[0]
        init_h = project_to_capped_simplex(init_h_raw, N / 2.0)
    except Exception:
        # No initial vector available – use a default size
        N = 45
        init_h = None

    target_sum = N / 2.0

    # ------------------------------------------------------------------
    # Population initialization
    # ------------------------------------------------------------------
    POP_SIZE = 40
    population = []
    if init_h is not None:
        population.append(init_h)
    while len(population) < POP_SIZE:
        cand = rng.rand(N)
        cand = project_to_capped_simplex(cand, target_sum)
        population.append(cand)

    scores = [evaluate(h) for h in population]
    best_idx = int(np.argmin(scores))
    best_h = population[best_idx].copy()
    best_score = scores[best_idx]

    # ------------------------------------------------------------------
    # GA parameters
    # ------------------------------------------------------------------
    MUT_PROB = 0.2      # probability per gene to mutate
    MUT_SIGMA = 0.05    # standard deviation of Gaussian mutation
    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # Main evolutionary loop
    # ------------------------------------------------------------------
    generation = 0
    while time.time() - start_time < budget_s:
        generation += 1
        new_pop = [best_h.copy()]   # elitism: keep current best

        # tournament selection helper
        def tournament():
            idxs = py_random.sample(range(POP_SIZE), 3)
            best_local = idxs[0]
            best_val = scores[best_local]
            for i in idxs[1:]:
                if scores[i] < best_val:
                    best_local = i
                    best_val = scores[i]
            return population[best_local]

        while len(new_pop) < POP_SIZE:
            p1 = tournament()
            p2 = tournament()
            # blend crossover
            alphas = rng.rand(N)
            child = alphas * p1 + (1.0 - alphas) * p2
            # mutation
            mask = rng.rand(N) < MUT_PROB
            if np.any(mask):
                child[mask] += rng.randn(mask.sum()) * MUT_SIGMA
            # repair to feasibility
            child = project_to_capped_simplex(child, target_sum)
            new_pop.append(child)

        # evaluate new population
        new_scores = [evaluate(h) for h in new_pop]

        # update best
        idx_best = int(np.argmin(new_scores))
        if new_scores[idx_best] < best_score:
            best_score = new_scores[idx_best]
            best_h = new_pop[idx_best].copy()

        # replace population
        population = new_pop
        scores = new_scores

        # early stop if we already beat the target
        if best_score <= 0.3808:
            break

    # ------------------------------------------------------------------
    # Local refinement (pairwise exchange) using remaining time
    # ------------------------------------------------------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 0.5:
        # simple hill climbing that preserves the sum exactly
        while time.time() - start_time < budget_s:
            i, j = py_random.sample(range(N), 2)
            # propose moving a small amount from i to j
            delta = rng.normal(scale=0.02)
            if best_h[i] - delta < 0.0 or best_h[j] + delta > 1.0:

                cand = best_h + rng.randn(N) * 0.01
                cand = np.clip(cand, 0.0, 1.0)
                cand = project_to_capped_simplex(cand, target_sum)

                cand[j] += delta

                cand = np.clip(cand, 0.0, 1.0)
                # no need for full projection – sum unchanged
            score = evaluate(cand)
            if score < best_score:
                best_score = score
                best_h = cand

    # final safety repair and final score
    best_h = project_to_capped_simplex(best_h, target_sum)
    best_score = evaluate(best_h)
    return best_h, best_score, N
