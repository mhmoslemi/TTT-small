def project_to_sum(v, target):
    lo = -1.0
    hi = 1.0
    for _ in range(30):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo+hi)/2.0, 0.0, 1.0)

def evaluate_h(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def random_feasible(n):
    v = np.random.rand(n)
    # Clip not necessary

def hill_climb(initial_h, budget_s):

        new = best.copy()

def simulated_annealing(initial, budget_s):
    start = time.time()
    n = len(initial)
    target = n/2
    cur = initial.copy()
    cur_score = evaluate_h(cur)
    best = cur.copy()
    best_score = cur_score
    max_temp = 0.02  # adjust based on scale
    while time.time() - start < budget_s * 0.9:
        # propose
        new = cur.copy()
        # choose two indexes
        i, j = np.random.choice(n, 2, replace=False)
        # delta magnitude
        delta = (np.random.rand() - 0.5) * 0.4  # up to +/-0.2

        delta = max(delta, -new[i])  # ensure i >=0

        if new[j] - delta > 1.0:

        new[j] -= delta
        # after these bound checks, sum unchanged.
        # evaluate
        new_score = evaluate_h(new)
        # compute temperature based on elapsed fraction

            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score
    return best, best_score

import numpy as np, math, time

def project_to_sum(v, target):
    # Ensure v are within [0,1] already? We'll clip within loop.
    lo = -1.0
    hi = 1.0
    for _ in range(30):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

def evaluate_h(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def random_feasible(n):
    v = np.random.rand(n)
    target = n / 2.0
    return project_to_sum(v, target)

def propose_move(h):
    n = h.shape[0]
    new = h.copy()
    t = np.random.randint(1, 4)  # number of transfers
    for _ in range(t):

        # allowed delta range ensuring bounds after transfer
        lower = max(-new[i], new[j] - 1.0)
        upper = min(1.0 - new[i], new[j])

        new[i] += delta

    target = n / 2.0

    if abs(np.sum(new) - target) > 1e-8:
