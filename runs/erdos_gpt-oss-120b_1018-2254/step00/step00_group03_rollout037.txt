    v = np.clip(v, 0.0, 1.0)
    # Use iterative method to adjust sum to target_sum.
    # If sum is close enough, done.
    tolerance = 1e-8
    cur_sum = np.sum(v)
    # If cur_sum == target_sum within tol, return
    # else adjust values that are not at bounds.
    # Use while loop with at most few iterations
    for _ in range(100):
        diff = target_sum - cur_sum
        if abs(diff) <= tolerance:
            break
        # Find indices that are not at lower bound when diff>0, not at upper bound when diff<0
        if diff > 0:
            free = (v < 1.0 - tolerance)
        else:
            free = (v > tolerance)
        free_idx = np.where(free)[0]
        if len(free_idx) == 0:
            # cannot adjust
            break
        # Distribute diff equally among free indices
        delta = diff / len(free_idx)
        # Update but respect bounds
        v_new = v.copy()
        for idx in free_idx:
            v_new[idx] = np.clip(v[idx] + delta, 0.0, 1.0)
        # compute new sum
        cur_sum = np.sum(v_new)
        v = v_new
    # final clip
    v = np.clip(v, 0.0, 1.0)
    # If still off by small tolerance, normalize last residual uniformly onto non-bound entries
    cur_sum = np.sum(v)
    if abs(cur_sum - target_sum) > 1e-6:
        # Simple fallback: scale all values and re-clip
        scale = target_sum / cur_sum if cur_sum != 0 else 0.0
        v = np.clip(v * scale, 0.0, 1.0)
    return v
```

def adjust_binary_sum(bin_vec, target_ones):

def binary_hillclimb(bin_h, target_ones, time_limit):
    best_h = bin_h.copy()
    best_val = evaluate_erdos_solution(best_h)
    cur_h = bin_h.copy()
    cur_val = best_val
    n = len(bin_h)
    # precompute ones and zeros indices
    while time.time() < time_limit:
        # choose random indices
        ones_idx = np.where(cur_h == 1)[0]
        zeros_idx = np.where(cur_h == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)
        # perform swap
        cur_h[i] = 0
        cur_h[j] = 1
        val = evaluate_erdos_solution(cur_h)
        if val < cur_val:
            cur_val = val
            if val < best_val:
                best_val = val
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] = 1
            cur_h[j] = 0
        # occasional random restart to avoid stagnation
        if np.random.rand() < 0.001:
            # random binary fresh
            cur_h = np.zeros(n)
            cur_h[:target_ones] = 1
            np.random.shuffle(cur_h)
            cur_val = evaluate_erdos_solution(cur_h)
            if cur_val < best_val:
                best_val = cur_val
                best_h = cur_h.copy()
    return best_h, best_val

def continuous_anneal(h, cur_val, target_sum, time_limit):
    best_h = h.copy()
    best_val = cur_val
    cur_h = h.copy()
    cur_val = cur_val
    n = len(h)
    start_temp = 0.05
    temp = start_temp
    it = 0
    while time.time() < time_limit:
        i, j = np.random.randint(n), np.random.randint(n)
        if i == j:
            continue
        # choose direction randomly
        if np.random.rand() < 0.5:
            donor, receiver = i, j
        else:
            donor, receiver = j, i
        max_transfer = min(cur_h[donor], 1.0 - cur_h[receiver])
        if max_transfer <= 0:
            continue
        eps = max_transfer * np.random.rand()
        new_h = cur_h.copy()
        new_h[donor] -= eps
        new_h[receiver] += eps
        new_val = evaluate_erdos_solution(new_h)
        delta = new_val - cur_val
        if delta < 0 or np.random.rand() < math.exp(-delta / temp):
            cur_h = new_h
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = cur_h.copy()
        it += 1
        # schedule temperature
        if it % 1000 == 0:
            temp *= 0.9
            # optionally refresh temp if becomes too low
            if temp < 1e-6:
                temp = start_temp
    return best_h, best_val
```

import time, math, numpy as np

# ---------- helper utilities ----------
def project_feasible(v, target):
    """Clip to [0,1] and adjust sum to target using simple water‑filling."""
    v = np.clip(v, 0.0, 1.0)
    eps = 1e-9
    for _ in range(100):
        cur = v.sum()
        diff = target - cur
        if abs(diff) <= eps:
            break
        if diff > 0:
            free = (v < 1.0 - eps)
        else:
            free = (v > eps)
        free_idx = np.where(free)[0]
        if free_idx.size == 0:
            break
        delta = diff / free_idx.size
        v[free_idx] = np.clip(v[free_idx] + delta, 0.0, 1.0)
    # final tiny correction
    cur = v.sum()
    if abs(cur - target) > 1e-6:
        scale = target / cur if cur != 0 else 0.0
        v = np.clip(v * scale, 0.0, 1.0)
    return v

    """Force a binary vector to contain exactly N ones."""

    if cur > N:
        idx = np.where(b == 1)[0]
        to_zero = np.random.choice(idx, cur - N, replace=False)
        b[to_zero] = 0
    elif cur < N:
        idx = np.where(b == 0)[0]
        to_one = np.random.choice(idx, N - cur, replace=False)
        b[to_one] = 1
    return b

def compute_c5(h):
    """Wrapper for the provided evaluator."""
    return evaluate_erdos_solution(h)

# ---------- binary simulated annealing ----------
def binary_sa_one_run(n, N, end_time):
    """Single SA run on binary vectors, returns best (h,val) found before end_time."""
    # start from a random feasible binary vector
    cur = np.zeros(n, dtype=float)
    cur[:N] = 1.0
    np.random.shuffle(cur)
    cur_val = compute_c5(cur)
    best_h = cur.copy()
    best_val = cur_val
    it = 0
    # temperature schedule parameters
    init_temp = 0.2 * cur_val if cur_val > 0 else 0.01
    final_temp = 1e-6
    while time.time() < end_time:
        # pick random 1 and 0 positions
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # propose swap
        cur[i] = 0.0
        cur[j] = 1.0
        new_val = compute_c5(cur)
        # temperature based on iteration count
        temp = max(final_temp, init_temp * (0.9995 ** it))
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if np.random.rand() < math.exp(-(new_val - cur_val) / temp):
                accept = True
        if accept:
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = cur.copy()
        else:
            # revert swap
            cur[i] = 1.0
            cur[j] = 0.0
        it += 1
        # occasional restart to avoid stagnation
        if it % 2000 == 0 and np.random.rand() < 0.02:
            cur = np.zeros(n, dtype=float)
            cur[:N] = 1.0
            np.random.shuffle(cur)
            cur_val = compute_c5(cur)
    return best_h, best_val

def binary_phase(n, N, end_time):
    """Run several SA attempts until the time budget expires."""
    best_h = None
    best_val = float('inf')
    while time.time() < end_time:
        # allocate a small chunk (max 8 s) for each attempt
        chunk_end = min(end_time, time.time() + 8.0)
        cand_h, cand_val = binary_sa_one_run(n, N, chunk_end)
        if cand_val < best_val:
            best_val = cand_val
            best_h = cand_h.copy()
        # early exit if we already beat the target bound
        if best_val <= 0.3808:
            break
    return best_h, best_val

# ---------- continuous annealing ----------
def continuous_anneal(h, target_sum, end_time):
    """Mass‑transfer annealing on a continuous vector that keeps the sum exact."""
    cur = h.copy()
    cur_val = compute_c5(cur)
    best_h = cur.copy()
    best_val = cur_val
    it = 0
    init_temp = 0.02 * cur_val if cur_val > 0 else 0.01
    final_temp = 1e-7
    n = len(cur)
    while time.time() < end_time:
        i, j = np.random.randint(n), np.random.randint(n)
        if i == j:
            continue
        # random direction of transfer
        if np.random.rand() < 0.5:
            donor, receiver = i, j
        else:
            donor, receiver = j, i
        max_transfer = min(cur[donor], 1.0 - cur[receiver])
        if max_transfer <= 1e-12:
            continue
        eps = max_transfer * np.random.rand()
        new = cur.copy()
        new[donor] -= eps
        new[receiver] += eps
        new_val = compute_c5(new)
        temp = max(final_temp, init_temp * (0.9999 ** it))
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if np.random.rand() < math.exp(-(new_val - cur_val) / temp):
                accept = True
        if accept:
            cur = new
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = cur.copy()
        it += 1
        if it % 5000 == 0:
            # occasional small random tweak to avoid plateaus
            idx = np.random.randint(n)
            cur[idx] = np.clip(cur[idx] + (np.random.rand() - 0.5) * 0.02, 0.0, 1.0)
            cur = project_feasible(cur, target_sum)
            cur_val = compute_c5(cur)
    return best_h, best_val

# ---------- main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    # Determine problem size and an initial feasible vector
    init_vec = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        init_vec = np.array(initial_h_values, dtype=float)
    if init_vec is None:
        # fallback: 70 points, constant 0.5 (satisfies sum exactly)
        n = 70
        init_vec = np.full(n, 0.5, dtype=float)
    else:
        n = len(init_vec)

    target_sum = n / 2.0   # exact integral requirement

    # Force feasibility of the initial vector
    cur = project_feasible(init_vec, target_sum)
    best_h = cur.copy()
    best_val = compute_c5(best_h)

    # ---------- phase 1 : binary optimization ----------
    time_elapsed = time.time() - start
    remaining = budget_s - time_elapsed
    # allocate roughly half the budget to the binary search
    binary_deadline = start + remaining * 0.48
    if n % 2 == 0:
        N = n // 2
        # start from a binary version of the current best vector
        bin_start = (best_h > 0.5).astype(float)
        bin_start = adjust_binary_sum(bin_start, N)
        # Run binary phase
        bin_best, bin_val = binary_phase(n, N, binary_deadline)
        if bin_val < best_val:
            best_val, best_h = bin_val, bin_best.copy()
    else:
        # odd length – skip pure binary search
        pass

    # ---------- phase 2 : continuous refinement ----------
    time_elapsed = time.time() - start
    remaining = budget_s - time_elapsed
    cont_deadline = start + remaining - 0.05  # tiny safety margin
    if time.time() < cont_deadline:
        cont_best, cont_val = continuous_anneal(best_h, target_sum, cont_deadline)
        if cont_val < best_val:
            best_val, best_h = cont_val, cont_best.copy()

    # Final projection to guarantee constraints (tiny tolerance)
    best_h = project_feasible(best_h, target_sum)

    # Return as plain Python list for the harness
    return best_h.tolist(), float(best_val), int(n)
