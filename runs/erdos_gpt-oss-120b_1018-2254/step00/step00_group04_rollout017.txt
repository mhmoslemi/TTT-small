def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def random_binary(n, rng):

    h[ones] = 1.0
    return h

def compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1‑h(x+k))dx using the discrete definition."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_feasible(h: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and adjust a few entries so that sum(h) == target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # need to increase some components
        for i in np.where(h < 1.0)[0]:
            inc = min(1.0 - h[i], diff)
            h[i] += inc
            diff -= inc
            if diff <= 0:
                break
    else:
        diff = -diff
        # need to decrease some components
        for i in np.where(h > 0.0)[0]:
            dec = min(h[i], diff)
            h[i] -= dec
            diff -= dec

def hill_climb_binary(h: np.ndarray, max_iters: int, rng: np.random.Generator):
    """Greedy swap‑search on binary vectors (preserves sum exactly)."""

    ones = list(np.where(best == 1)[0])
    zeros = list(np.where(best == 0)[0])
    for _ in range(max_iters):

            break
