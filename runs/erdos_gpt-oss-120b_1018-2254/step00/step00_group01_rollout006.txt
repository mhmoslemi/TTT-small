def project_to_capped_simplex(v, target_sum, bound=1.0, max_iter=100, tol=1e-9):
    # v: np.array
    # target_sum: desired sum
    # bound: upper bound (1)
    # returns projected vector
    lower = np.min(v) - bound
    upper = np.max(v)
    for i in range(max_iter):
        lam = (lower + upper) / 2.0
        w = np.clip(v - lam, 0, bound)
        s = w.sum()
        if abs(s - target_sum) <= tol:
            return w
        if s > target_sum:
            lower = lam
        else:
            upper = lam
    return np.clip(v - lam, 0, bound)
```

import numpy as np
import time
import math

def _project_to_capped_simplex(v, target_sum, bound=1.0, max_iter=100, tol=1e-9):
    """
    Project vector v onto the capped simplex:
        0 <= w_i <= bound,  sum(w) = target_sum
    using a binary search on the Lagrange multiplier.
    """
    v = np.asarray(v, dtype=np.float64)
    lower = np.min(v) - bound     # ensures all entries clipped to bound at low end
    upper = np.max(v)             # ensures all entries clipped to 0 at high end
    for _ in range(max_iter):
        lam = (lower + upper) * 0.5
        w = np.clip(v - lam, 0.0, bound)
        s = w.sum()
        if abs(s - target_sum) <= tol:
            return w
        if s > target_sum:
            lower = lam
        else:
            upper = lam
    # final projection (may be off by tol)
    w = np.clip(v - lam, 0.0, bound)
    # tiny correction if needed
    diff = target_sum - w.sum()
    if abs(diff) > tol:
        free = (w > 0.0) & (w < bound)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0.0, bound)
    return w

def _compute_c5(h, n):
    """Exact discrete C5 for a step function sampled at n points."""
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx

def _random_feasible(n, target_sum):
    """Generate a random feasible vector (0-1 entries, correct sum)."""
    rnd = np.random.rand(n)
    return _project_to_capped_simplex(rnd, target_sum)
