    """Return a step function h on [0,2] (as a numpy array), its C5 bound, and n_points."""
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # Determine problem size and initial vector
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        n = int(len(initial_h_values))
        cur_h = np.array(initial_h_values, dtype=float).copy()
    else:
        # fallback: use a binary+half pattern with n = 101 (odd) – small enough
        n = 101
        target_sum = n / 2.0
        ones = int(math.floor(target_sum))
        half = target_sum - ones
        cur_h = np.zeros(n, dtype=float)
        idx_one = np.random.choice(n, ones, replace=False)
        cur_h[idx_one] = 1.0
        if half > 0:
            remaining = np.setdiff1d(np.arange(n), idx_one)
            cur_h[np.random.choice(remaining)] = half

    # Ensure the exact integral constraint (sum = n/2)
    target_sum = n / 2.0
    cur_sum = cur_h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        diff = target_sum - cur_sum
        for i in range(n):
            new_val = cur_h[i] + diff
            if 0.0 <= new_val <= 1.0:
                cur_h[i] = new_val
                break

    dx = 2.0 / n

    def compute_c5(arr):
        """C5 = max_k ∫ h(x)(1−h(x+k))dx, discretised."""
        return np.max(np.correlate(arr, 1.0 - arr, mode='full')) * dx

    # ------------------------------------------------------------------
    # Simulated annealing / local search
    # ------------------------------------------------------------------
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val

    start_time = time.time()
    max_no_improve = 10000          # after this many non‑improving steps, restart
    no_improve_cnt = 0

    while time.time() - start_time < budget_s * 0.95:
        # pick two distinct indices
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        hi, hj = cur_h[i], cur_h[j]

        # decide direction of weight transfer
        if hi > 0.0 and hj < 1.0:
            src, dst, max_delta = i, j, min(hi, 1.0 - hj)
        elif hj > 0.0 and hi < 1.0:
            src, dst, max_delta = j, i, min(hj, 1.0 - hi)
        else:
            continue  # no feasible move

        if max_delta <= 0.0:
            continue

        # step size shrinks as time passes
        elapsed = time.time() - start_time
        frac = max(0.0, 1.0 - elapsed / budget_s)          # 1 at start, 0 at end
        max_delta_scaled = max_delta * (0.5 + 0.5 * frac)   # between 0.5·max and max
        delta = np.random.rand() * max_delta_scaled

        cand = cur_h.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand_val = compute_c5(cand)

        # ---- accept as the new current state (SA) ----
        if cand_val < cur_val:
            cur_h, cur_val = cand, cand_val
        else:
            # exponential temperature schedule
            T0, T1 = 0.01, 0.0005
            T = T0 * (T1 / T0) ** (elapsed / budget_s)
            if np.random.rand() < math.exp(-(cand_val - cur_val) / T):
                cur_h, cur_val = cand, cand_val

        # ---- update best solution found ----
        if cand_val < best_val - 1e-12:
            best_h, best_val = cand.copy(), cand_val
            no_improve_cnt = 0
        else:
            no_improve_cnt += 1

        # ---- stagnation handling ----
        if no_improve_cnt >= max_no_improve:
            # random binary+half restart
            ones = int(math.floor(target_sum))
            half = target_sum - ones
            restart = np.zeros(n, dtype=float)
            idx_one = np.random.choice(n, ones, replace=False)
            restart[idx_one] = 1.0
            if half > 0.0:
                remaining = np.setdiff1d(np.arange(n), idx_one)
                restart[np.random.choice(remaining)] = half
            cur_h = restart
            cur_val = compute_c5(cur_h)
            no_improve_cnt = 0
            if cur_val < best_val:
                best_h, best_val = cur_h.copy(), cur_val

    # ------------------------------------------------------------------
    # Final correction to satisfy the integral exactly
    # ------------------------------------------------------------------
    final_sum = best_h.sum()
    diff = target_sum - final_sum
    if abs(diff) > 1e-12:
        if diff > 0.0:
            for idx in range(n):
                if best_h[idx] + diff <= 1.0:
                    best_h[idx] += diff
                    break
        else:
            for idx in range(n):
                if best_h[idx] + diff >= 0.0:
                    best_h[idx] += diff
                    break

    final_c5 = compute_c5(best_h)
    return best_h, float(final_c5), int(n)
