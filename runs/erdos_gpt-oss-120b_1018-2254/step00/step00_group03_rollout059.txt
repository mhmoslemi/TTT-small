def gradient_f(h, lag):
    n = len(h)

        # derivative w.r.t. h[i-p] term -h[i]
        grad[:m] -= h[p:]
    return grad

import numpy as np
import time
import random

def _c5_raw_and_lag(h):
    """Return raw max correlation value, its lag, and the full correlation vector."""
    n = len(h)
    conv = np.correlate(h, 1 - h, mode='full')
    idx = int(np.argmax(conv))
    return conv[idx], idx - (n - 1), conv

def _gradient_f(h, lag):
    """Gradient of the raw max‑correlation term w.r.t. h for a given lag."""
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        m = n - lag
        g[:m] += (1 - h[lag:])          # ∂/∂h_i term, i in [0,m)
        g[lag:] -= h[:m]               # ∂/∂h_{i+lag} term
    else:
        p = -lag
        m = n - p
        g[p:] += (1 - h[:m])           # ∂/∂h_i term, i in [p,n)
        g[:m] -= h[p:]                 # ∂/∂h_{i-p} term
    return g

def _project_sum_bounds(h, target):
    """Clip to [0,1] and adjust sum to target by distributing the residue."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    tol = 1e-12
    for _ in range(10):
        if abs(diff) < tol:
            break
        free = (h > tol) & (h < 1 - tol)
        cnt = free.sum()
        if cnt == 0:
            break
        delta = diff / cnt
        h[free] += delta
        # re‑clip any that left the box
        over_low = h < 0.0
        over_high = h > 1.0
        if over_low.any():
            diff += h[over_low].sum()
            h[over_low] = 0.0
        if over_high.any():
            diff -= (h[over_high] - 1.0).sum()
            h[over_high] = 1.0
        diff = target - h.sum()
    return h

def _optimize(h_start, time_budget):
    """Local search using gradient‑guided mass transfers."""
    start = time.time()
    h = np.array(h_start, dtype=float)
    n = len(h)
    target = n / 2.0
    # enforce exact sum initially
    h = _project_sum_bounds(h, target)

    best_h = h.copy()
    best_raw, best_lag, _ = _c5_raw_and_lag(h)
    no_improve = 0
    MAX_STAGNANT = 5000

    while time.time() - start < time_budget:
        # gradient for current worst lag
        grad = _gradient_f(h, best_lag)
        pos = int(np.argmax(grad))
        neg = int(np.argmin(grad))

        # try a directed transfer
        if grad[pos] > 0 and grad[neg] < 0:
            delta_max = min(h[pos], 1.0 - h[neg])
            if delta_max > 0:
                delta = 0.5 * delta_max
                cand = h.copy()
                cand[pos] -= delta
                cand[neg] += delta
                new_raw, new_lag, _ = _c5_raw_and_lag(cand)
                if new_raw < best_raw - 1e-12:
                    h = cand
                    best_raw, best_lag = new_raw, new_lag
                    if new_raw < best_raw:
                        best_h = cand.copy()
                    no_improve = 0
                    continue

        # fallback: random feasible transfer
        i, j = random.randrange(n), random.randrange(n)
        if i != j:
            eps_max = min(h[i], 1.0 - h[j])
            if eps_max > 0:
                eps = random.random() * eps_max
                cand = h.copy()
                cand[i] -= eps
                cand[j] += eps
                new_raw, new_lag, _ = _c5_raw_and_lag(cand)
                if new_raw < best_raw - 1e-12:
                    h = cand
                    best_raw, best_lag = new_raw, new_lag
                    best_h = cand.copy()
                    no_improve = 0
                    continue

        # no improvement this round
        no_improve += 1
        if no_improve >= MAX_STAGNANT:
            # small random perturbation to escape plateau
            h = best_h.copy()
            for _ in range(5):
                i, j = random.randrange(n), random.randrange(n)
                if i == j:
                    continue
                eps_max = min(h[i], 1.0 - h[j])
                if eps_max > 0:
                    eps = random.random() * eps_max
                    h[i] -= eps
                    h[j] += eps
            best_raw, best_lag, _ = _c5_raw_and_lag(h)
            no_improve = 0

    dx = 2.0 / n
    return best_h, best_raw * dx, n

    np.random.seed(seed)
    random.seed(seed)

    # gather a few starting vectors
    starts = []

    # 1) use the parent construction if provided

        # recompute exact bound
        best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * (2.0 / best_n)

    return best_h, best_c5, best_n
