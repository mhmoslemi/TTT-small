def project_onto_box_simplex(v, S):
    # ensure S within [0, n*upper]
    n = len(v)
    lower = 0.0; upper = 1.0
    # Clip is not needed inside; we use function clip after shift.
    # Use binary search on lambda.
    # For lambda large positive -> all v_i - lambda <= 0 => sum=0.
    # For lambda large negative -> v_i - lambda >= upper => sum=n*upper.
    # So lambda range: [-max(v)+upper, max(v)]
    lo = -np.max(v) - 1.0  # ensure enough negative to push sum to n*upper
    hi = np.max(v) + 1.0   # ensure enough positive to push sum to 0
    # binary search
    for _ in range(50):
        lam = (lo + hi) / 2
        y = np.clip(v - lam, lower, upper)
        s = np.sum(y)
        if s > S:
            # need to increase lambda to reduce sum
            lo = lam
        else:
            hi = lam
    y = np.clip(v - (lo + hi) / 2, lower, upper)
    return y

def project_onto_box_simplex(v, S):
    # v: np.array shaped (n,)
    # returns y satisfying 0<=y_i<=1 and sum(y)=S.
    # Use bisection for lambda.
    n = v.shape[0]
    lower = 0.0
    upper = 1.0
    # define function to compute sum after applying lambda
    def sum_clip(lam):
        return np.sum(np.clip(v - lam, lower, upper))
    # We need find lam s.t. sum_clip(lam) = S.
    # Determine bounds for lam:
    lam_low = -np.max(v) - 1.0  # for large negative, v - lam is large positive -> clipped to upper 1 => sum = n*1
    lam_high = np.max(v) + 1.0   # for large positive, v - lam negative => clipped to 0 => sum = 0
    # Ensure that sum_clip(lam_low) >= S >= sum_clip(lam_high)
    # But sum_clip(lam_low) ~ n, sum_clip(lam_high) ~ 0.
    # Bisection
    for _ in range(50):
        lam_mid = (lam_low + lam_high) / 2.0
        s = sum_clip(lam_mid)
        if s > S:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    y = np.clip(v - (lam_low + lam_high) / 2.0, lower, upper)
    # To correct small residual due to numeric, adjust with simple fix:
    diff = S - np.sum(y)
    if abs(diff) > 1e-8:
        # Distribute diff among non-saturated components
        free = np.where((y > lower + 1e-12) & (y < upper - 1e-12))[0]
        if free.size > 0:
            y[free] += diff / free.size
            y = np.clip(y, lower, upper)
    return y

import numpy as np
import time

def project_onto_box_simplex(v, S):
    v = np.asarray(v, dtype=float)
    # Bisection on lambda
    lower = -np.max(v) - 1.0
    upper = np.max(v) + 1.0
    for _ in range(60):
        lam = (lower + upper) / 2.0
        y = np.clip(v - lam, 0.0, 1.0)
        s = np.sum(y)
        if s > S:
            lower = lam
        else:
            upper = lam
    y = np.clip(v - (lower + upper) / 2.0, 0.0, 1.0)
    # Fix small discrepancy
    diff = S - np.sum(y)
    if abs(diff) > 1e-8:
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        free = np.where(mask)[0]
        if free.size > 0:
            y[free] += diff / free.size
            y = np.clip(y, 0.0, 1.0)
    return y

def gradient_for_shift(h, lag):
    n = h.shape[0]
    g = np.ones(n, dtype=float)
    if lag >= 0:
        l = lag
        if l < n:
            # subtract h[i+l] for i=0..n-l-1
            g[:n - l] -= h[l:]
            # subtract h[i-l] for i=l..n-1
            g[l:] -= h[:n - l]
    else:
        l = -lag
        if l < n:
            # subtract h[i-l] for i=l..n-1
            g[l:] -= h[:n - l]
            # subtract h[i+l] for i=0..n-l-1
            g[:n - l] -= h[l:]
    return g

def evaluate_fitness(h, dx):
    # Returns C5 bound
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

    start = time.time()
    np.random.seed(seed)

    # Determine dimension
    try:
        init_h = np.array(initial_h_values, dtype=float)
        n = init_h.size

        init_h = None
        n = 45
        have_init = False

    S = n / 2.0
    dx = 2.0 / n

    # helper inside run uses earlier defined functions
    POP_SIZE = 80
    TOURN_SIZE = 4
    MUT_PROB = 0.2
    MUT_SIGMA = 0.15
    CROSS_RATE = 0.5

    pop = []
    if have_init:
        p = project_onto_box_simplex(init_h, S)
        pop.append(p)
    else:

        pop.append(p)

        vec = project_onto_box_simplex(np.random.rand(n), S)
        pop.append(vec)

    fitness = [evaluate_fitness(ind, dx) for ind in pop]
    best_idx = int(np.argmin(fitness))
    best_h = pop[best_idx].copy()
    best_f = fitness[best_idx]

    while time.time() - start < budget_s - 0.5:

        # select parent1
        idxs = np.random.choice(POP_SIZE, TOURN_SIZE, replace=False)

        # optionally crossover
        if np.random.rand() < CROSS_RATE:

        child = project_onto_box_simplex(child, S)

        if child_f < fitness[worst_idx]:

            fitness[worst_idx] = child_f

import numpy as np
import time

def project_onto_box_simplex(v, S):
    """Projection of v onto {0<=x_i<=1, sum(x)=S}."""
    v = np.asarray(v, dtype=float)
    # Bisection on lambda to satisfy sum constraint
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        y = np.clip(v - lam, 0.0, 1.0)
        s = y.sum()
        if s > S:
            lo = lam
        else:
            hi = lam
    y = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # Fix possible tiny sum error
    diff = S - y.sum()
    if abs(diff) > 1e-10:
        free = np.where((y > 1e-12) & (y < 1.0 - 1e-12))[0]
        if free.size:
            y[free] += diff / free.size
            y = np.clip(y, 0.0, 1.0)
        else:
            # fallback: distribute to all entries safely
            y += diff / len(y)
            y = np.clip(y, 0.0, 1.0)
    return y

def gradient_for_shift(h, lag):
    """Gradient of the worst-shift overlap w.r.t. h."""
    n = h.shape[0]
    g = np.ones(n, dtype=float)
    if lag >= 0:
        l = lag
        if l < n:
            g[:n - l] -= h[l:]          # term h_{i+lag}
            g[l:]      -= h[:n - l]     # term h_{i-lag}
    else:
        l = -lag
        if l < n:
            g[l:]      -= h[:n - l]     # term h_{i-lag}
            g[:n - l]  -= h[l:]         # term h_{i+lag}
    return g

def evaluate_fitness(h, dx):
    """C5 bound of a candidate."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    np.random.seed(seed)

    # --- problem size -------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)  # provided by harness
        n = init_h.size
        have_init = True
    except Exception:
        init_h = None
        n = 45               # fallback size
        have_init = False

    S = n / 2.0               # sum constraint
    dx = 2.0 / n

    # --- GA parameters ------------------------------------------------
    POP_SIZE   = 80
    TOURN_SIZE = 4
    MUT_PROB   = 0.20          # fraction of genes mutated
    MUT_SIGMA  = 0.15          # Gaussian mutation std
    CROSS_RATE = 0.5

    # --- initial population -------------------------------------------
    pop = []
    if have_init:
        pop.append(project_onto_box_simplex(init_h, S))
    else:
        pop.append(project_onto_box_simplex(np.random.rand(n), S))
    while len(pop) < POP_SIZE:
        pop.append(project_onto_box_simplex(np.random.rand(n), S))

    fitness = [evaluate_fitness(ind, dx) for ind in pop]
    best_idx = int(np.argmin(fitness))
    best_h = pop[best_idx].copy()
    best_f = fitness[best_idx]

    # --- main GA loop -------------------------------------------------
    while time.time() - start < budget_s - 0.5:
        # tournament selection for parent1
        idxs = np.random.choice(POP_SIZE, TOURN_SIZE, replace=False)
        p1 = pop[int(idxs[np.argmin([fitness[i] for i in idxs])])]

        # possibly crossover
        if np.random.rand() < CROSS_RATE:
            idxs2 = np.random.choice(POP_SIZE, TOURN_SIZE, replace=False)
            p2 = pop[int(idxs2[np.argmin([fitness[i] for i in idxs2])])]
            child = (p1 + p2) * 0.5
        else:
            child = p1.copy()

        # mutation
        num_mut = max(1, int(MUT_PROB * n))
        mut_idx = np.random.choice(n, num_mut, replace=False)
        child[mut_idx] += np.random.randn(num_mut) * MUT_SIGMA
        child = np.clip(child, 0.0, 1.0)
        child = project_onto_box_simplex(child, S)

        child_f = evaluate_fitness(child, dx)

        # replace worst if better
        worst_idx = int(np.argmax(fitness))
        if child_f < fitness[worst_idx]:
            pop[worst_idx] = child
            fitness[worst_idx] = child_f
            if child_f < best_f:
                best_f = child_f
                best_h = child.copy()

        # early stopping if already very good
        if best_f <= 0.3808 and (time.time() - start) > budget_s * 0.3:
            break

    # --- local refinement (gradient descent on worst shift) ----------
    step = 0.02
    while time.time() - start < budget_s - 0.1:
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        cur_f = np.max(corr) * dx
        max_idx = int(np.argmax(corr))
        lag = max_idx - (n - 1)
        grad = gradient_for_shift(best_h, lag)

        new_h = best_h - step * grad
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = project_onto_box_simplex(new_h, S)

        new_f = evaluate_fitness(new_h, dx)
        if new_f < cur_f - 1e-9:
            best_h = new_h
            best_f = new_f
            step = min(step * 1.1, 0.08)   # modest step increase
        else:
            step *= 0.5
            if step < 1e-6:
                break

    final_c5 = evaluate_fitness(best_h, dx)
    return (best_h, float(final_c5), n)
