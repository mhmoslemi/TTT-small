        # Use algorithm of projecting onto simplex with bounds.
        # We'll treat each iteration: clip to [0,1], compute diff = S - sum, distribute diff uniformly among non-bound entries.
        h = np.clip(h, 0.0, 1.0)
        max_iter = 100
        for _ in range(max_iter):
            cur_sum = h.sum()
            diff = S - cur_sum
            if abs(diff) < 1e-12:

            # indices not at bounds
            free = (h > 0.0 + 1e-12) & (h < 1.0 - 1e-12)

                free = np.ones_like(h, dtype=bool)
            # Distribute diff equally among free entries
            delta = diff / free.sum()
            h[free] += delta
            h = np.clip(h, 0.0, 1.0)

        return np.clip(h, 0.0, 1.0)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    # simple iterative projection onto hyperplane with bounds
    # Use algorithm like water-filling
    # Equivalent to solving min ||h' - h||^2 subject to sum(h') = target_sum and bounds.
    # Use iterative scaling.
    for _ in range(100):
        cur_sum = h.sum()
        diff = target_sum - cur_sum
        if abs(diff) < 1e-12:
            break
        # indices not at bounds
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.any():
            delta = diff / free.sum()
            h[free] += delta
            h = np.clip(h, 0.0, 1.0)
        else:
            # all at bounds; distribute to all then clip.
            delta = diff / len(h)
            h += delta
            h = np.clip(h, 0.0, 1.0)
    return h
```

import numpy as np
from scipy.optimize import minimize

# Global to be read by constraint functions
CURRENT_N = None

def raw_max_corr(h: np.ndarray) -> float:
    """Maximum of the raw overlap correlation (no dx factor)."""
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")))

def c5_from_raw(raw: float, n: int) -> float:
    """Convert raw correlation to the C5 value using dx = 2/n."""
    return raw * (2.0 / n)

def project_h(h: np.ndarray, target_sum: float) -> np.ndarray:
    """Project vector onto [0,1]^n and enforce sum = target_sum (water-filling)."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        free = (h > 0.0) & (h < 1.0)
        if free.any():
            delta = diff / free.sum()
            h[free] += delta
            h = np.clip(h, 0.0, 1.0)
        else:
            # all at bounds, redistribute uniformly then clip
            delta = diff / h.size
            h += delta
            h = np.clip(h, 0.0, 1.0)
    return h

# Objective for SLSQP: minimise the raw correlation bound t
def _obj(x):
    return x[-1]

# Equality: sum(h) = n/2
def _eq_con(x):
    if CURRENT_N is None:
        return 0.0
    h = x[:-1]
    return np.sum(h) - (CURRENT_N / 2.0)

# Inequalities: for every shift, t >= raw correlation
def _ineq_con(x):
    if CURRENT_N is None:
        return np.array([0.0])
    h = x[:-1]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr  # vector of length 2n-1, each must be >=0

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Optimise a step function for the Erdős C5 constant."""
    import time
    rng = np.random.default_rng(seed)

    # ----- initialise ----------
    init = globals().get("initial_h_values")
    if init is not None:
        h = np.array(init, dtype=float)
        n = h.size
    else:
        n = 70
        h = np.zeros(n, dtype=float)
        half = n // 2
        h[:half] = 1.0
        rng.shuffle(h)

    global CURRENT_N
    CURRENT_N = n
    target = n / 2.0
    h = project_h(h, target)

    # ----- evaluate initial solution ----------
    best_raw = raw_max_corr(h)
    best_c5 = c5_from_raw(best_raw, n)
    best_h = h.copy()

    # ----- local SLSQP refinement ----------
    x0 = np.concatenate([best_h, [best_raw]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "eq", "fun": _eq_con},
        {"type": "ineq", "fun": _ineq_con},
    ]
    try:
        res = minimize(
            _obj,
            x0,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options={"maxiter": 250, "ftol": 1e-9, "disp": False},
        )
        if res.success:
            h_opt = np.clip(res.x[:-1], 0.0, 1.0)
            h_opt = project_h(h_opt, target)
            raw_opt = raw_max_corr(h_opt)
            c5_opt = c5_from_raw(raw_opt, n)
            if c5_opt < best_c5:
                best_h, best_c5, best_raw = h_opt, c5_opt, raw_opt
    except Exception:
        pass

    # ----- greedy random walk (budget aware) ----------
    cur_h = best_h.copy()
    cur_raw = best_raw
    start = time.time()
    while time.time() - start < budget_s * 0.94:
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        # direction 1: increase i, decrease j
        inc_i = 1.0 - cur_h[i]
        dec_j = cur_h[j]
        # direction 2: increase j, decrease i
        inc_j = 1.0 - cur_h[j]
        dec_i = cur_h[i]

        if rng.random() < 0.5:
            delta = min(inc_i, dec_j)
            if delta <= 0.0:
                continue
            delta *= rng.random()
            new_h = cur_h.copy()
            new_h[i] += delta
            new_h[j] -= delta
        else:
            delta = min(inc_j, dec_i)
            if delta <= 0.0:
                continue
            delta *= rng.random()
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta

        cand_raw = raw_max_corr(new_h)
        if cand_raw < cur_raw:
            cur_h, cur_raw = new_h, cand_raw
        if cand_raw < best_raw:
            best_h, best_raw = new_h.copy(), cand_raw

    # final projection and C5 computation
    best_h = project_h(best_h, target)
    best_c5 = c5_from_raw(raw_max_corr(best_h), n)

    return (best_h.tolist(), float(best_c5), n)
