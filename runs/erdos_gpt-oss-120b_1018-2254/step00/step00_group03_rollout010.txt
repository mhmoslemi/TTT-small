def project_box_sum(v, target, low=0.0, high=1.0, tol=1e-9, max_iter=100):
    # binary search lambda
    lo = -1e6
    hi = 1e6
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, low, high)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
        if hi - lo < tol:
            break
    return np.clip(v - lam, low, high)

import time, math, numpy as np

def project_box_sum(v, target, low=0.0, high=1.0, tol=1e-9, max_iter=100):
    lo = -1e6
    hi = 1e6
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, low, high)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
        if hi - lo < tol:
            break
    return np.clip(v - lam, low, high)

def compute_c5(h, dx):
    # correlation for all shifts
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    start = time.time()

def project_box_sum(v, target, low=0.0, high=1.0, tol=1e-9, max_iter=100):
    lo = -1e6
    hi = 1e6
    for _ in range(max_iter):

        proj = np.clip(v - lam, low, high)
        s = proj.sum()

            lo = lam
        else:
            hi = lam

def compute_c5(h, dx):
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

import time, math, numpy as np

def project_box_sum(v, target, low=0.0, high=1.0, tol=1e-9, max_iter=100):
    """Project v onto the box [low,high] with an exact sum equal to target."""
    lo, hi = -1e6, 1e6
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, low, high)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
        if hi - lo < tol:
            break
    return np.clip(v - lam, low, high)

def compute_c5(h, dx):
    """Evaluate the Erdős C₅ value for a step function h."""
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # --- initialise ---------------------------------------------------------
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n = init.size
        target = n / 2.0
        h = project_box_sum(init, target)
    else:
        n = 70
        target = n / 2.0
        h = rng.random(n)
        h = project_box_sum(h, target)

    dx = 2.0 / n
    cur_c = compute_c5(h, dx)
    best_h = h.copy()
    best_c = cur_c

    start = time.time()
    max_no_improve = 5000
    no_improve = 0

    # --- main optimisation loop --------------------------------------------
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.98:
            break

        # current objective and worst shift
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr)
        cur_c = corr[idx] * dx
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()

        # gradient of the worst‑shift overlap
        s = idx - (n - 1)
        g = np.ones_like(h)
        if s >= 0:
            g[:n - s] -= h[s:]
            g[s:]   -= h[:n - s]
        else:
            t = -s
            g[t:]   -= h[:n - t]
            g[:n - t] -= h[t:]

        # step size schedule (decreasing)
        remaining = budget_s - elapsed
        alpha0 = 0.02

        alpha = max(min_alpha, alpha0 * max(0.0, remaining / budget_s))

        # propose a projected gradient step
        v = h - alpha * g
        h_new = project_box_sum(v, target)

        new_c = compute_c5(h_new, dx)

        # acceptance rule (greedy with tiny uphill probability)
        if new_c < cur_c:
            h = h_new
            cur_c = new_c
            no_improve = 0
        else:

            T = T0 * max(0.0, remaining / budget_s)
            if T > 0.0 and rng.random() < math.exp(-(new_c - cur_c) / T):

            else:
                h = rng.random(n)
                h = project_box_sum(h, target)
            cur_c = compute_c5(h, dx)
            no_improve = 0

    # final projection and evaluation of the best solution found
    best_h = project_box_sum(best_h, target)
    best_c = compute_c5(best_h, dx)

    return (best_h.tolist(), float(best_c), n)
