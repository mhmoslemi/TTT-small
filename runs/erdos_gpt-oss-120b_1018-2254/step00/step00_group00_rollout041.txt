def project_to_feasible(v, target_sum):
    n = v.shape[0]
    # Clip to [0,1]
    v = np.clip(v, 0.0, 1.0)
    # If sum already target, return
    s = v.sum()
    if abs(s - target_sum) < 1e-12:
        return v
    # Simple bisection on lambda
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s2 = proj.sum()
        if s2 > target_sum:
            lo = mid
        else:
            hi = mid
    # final
    proj = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    # adjust any tiny rounding deviations
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-10:

        if free.any():
            adjust = diff / free.sum()
            proj[free] += adjust
            proj = np.clip(proj, 0.0, 1.0)
        # else ignore (should not happen)
    return proj

def run(seed=42, budget_s=1100, **kwargs):
    import time
    import numpy as np

    np.random.seed(seed)

    # ---------- helper functions ----------
    def compute_c5(vec):
        """C5 = max_k ∫ h(x)(1‑h(x+k))dx, discretized."""
        n = vec.shape[0]
        dx = 2.0 / n
        conv = np.correlate(vec, 1.0 - vec, mode="full")
        return float(np.max(conv) * dx)

    def project_to_feasible(v, target_sum):
        """Clip to [0,1] and adjust so that sum(v) == target_sum."""
        v = np.clip(v, 0.0, 1.0)
        cur = v.sum()
        if abs(cur - target_sum) < 1e-12:
            return v
        # bisection on a scalar shift λ : sum(clip(v-λ,0,1)) = target_sum
        lo, hi = -1.0, 1.0
        for _ in range(60):
            mid = (lo + hi) * 0.5
            proj = np.clip(v - mid, 0.0, 1.0)
            s = proj.sum()
            if s > target_sum:
                lo = mid
            else:
                hi = mid
        proj = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
        # tiny residual correction
        diff = target_sum - proj.sum()
        if abs(diff) > 1e-9:
            free = (proj > 1e-12) & (proj < 1.0 - 1e-12)
            if free.any():
                adjust = diff / free.sum()
                proj[free] += adjust
                proj = np.clip(proj, 0.0, 1.0)
        return proj

    def propose_move(h, target_sum):
        """Swap a small amount between two entries while preserving sum."""
        n = h.shape[0]
        new_h = h.copy()
        i, j = np.random.choice(n, size=2, replace=False)
        hi, hj = new_h[i], new_h[j]
        lower = max(-hi, hj - 1.0)
        upper = min(1.0 - hi, hj)
        if lower >= upper:
            return new_h
        delta = np.random.normal(scale=0.2)
        delta = np.clip(delta, lower, upper)
        new_h[i] = hi + delta
        new_h[j] = hj - delta
        np.clip(new_h, 0.0, 1.0, out=new_h)
        # Fix tiny drift in the sum
        drift = target_sum - new_h.sum()
        if abs(drift) > 1e-12:
            free = (new_h > 1e-12) & (new_h < 1.0 - 1e-12)
            if free.any():
                idx = np.random.choice(np.where(free)[0])
                if drift > 0:
                    add = min(drift, 1.0 - new_h[idx])
                else:
                    add = max(drift, -new_h[idx])
                new_h[idx] += add
                new_h[idx] = max(0.0, min(1.0, new_h[idx]))
            else:
                new_h = project_to_feasible(new_h, target_sum)
        return new_h

    # ---------- initialization ----------
    if "initial_h_values" in globals():
        base = np.array(initial_h_values, dtype=float)
        n0 = base.size
        factor = 8
        n = n0 * factor
        if n % 2:
            factor += 1
            n = n0 * factor
        h = np.repeat(base, factor)
        if h.size != n:  # safety resize
            h = np.resize(h, n)
        target = n / 2.0
        h = project_to_feasible(h, target)
    else:
        # fallback random start
        n = 200
        if n % 2:
            n += 1
        target = n / 2.0
        rand_vec = np.random.rand(n)
        h = project_to_feasible(rand_vec, target)

    # ---------- local search ----------
    best_h = h.copy()
    best_score = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_score = best_score

    max_iter = 20000
    it = 0

        cand = propose_move(cur_h, target)
        cand_score = compute_c5(cand)

        if cand_score < cur_score:
            cur_h, cur_score = cand, cand_score
            if cand_score < best_score:

        else:

            elapsed = time.time() - start
            T = max(1e-5, (budget_s - elapsed) / budget_s)  # linear cooling
            if np.random.rand() < np.exp(-(cand_score - cur_score) / T):

        it += 1

    # final safeguard projection and re‑evaluation
    best_h = project_to_feasible(best_h, target)
    final_score = compute_c5(best_h)

    return (best_h.tolist(), float(final_score), int(best_h.size))
