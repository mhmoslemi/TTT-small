import numpy as np, time, math

def compute_c5(h, dx):
    c = np.correlate(h, 1 - h, mode='full')
    return np.max(c) * dx

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.1

    # Determine initial vector length and start.
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        n = init.size
        # Ensure values in [0,1]
        init = np.clip(init, 0.0, 1.0)
        # Convert to binary by thresholding at 0.5 (if needed)
        # ensure exact half ones
        h = (init > 0.5).astype(float)
        target_ones = n // 2
        cur_ones = int(h.sum())
        if cur_ones != target_ones:
            # adjust by random flips
            idx = np.arange(n)
            # if too many ones
            if cur_ones > target_ones:
                excess = cur_ones - target_ones
                ones_idx = np.flatnonzero(h == 1)
                flip = rng.choice(ones_idx, size=excess, replace=False)
                h[flip] = 0.0
            else:
                deficit = target_ones - cur_ones
                zeros_idx = np.flatnonzero(h == 0)
                flip = rng.choice(zeros_idx, size=deficit, replace=False)
                h[flip] = 1.0
    else:
        n = 80
        h = np.zeros(n, dtype=float)
        ones_idx = rng.choice(n, size=n//2, replace=False)
        h[ones_idx] = 1.0

    dx = 2.0 / n
    cur_c5 = compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = cur_c5

    # lists of indices
    ones = list(np.flatnonzero(h == 1))
    zeros = list(np.flatnonzero(h == 0))

    max_temp = 0.02
    # restart after stagnation
    stagnation_limit = 5000
    stagnation = 0

    while time.time() < deadline:
        # sample swap
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        # perform swap
        h[i] = 0.0
        h[j] = 1.0

        new_c5 = compute_c5(h, dx)
        delta = new_c5 - cur_c5
        # temperature linearly decreasing
        remaining = deadline - time.time()
        if remaining < 0:
            break
        temp = max_temp * (remaining / (budget_s - 0.1))
        accept = False
        if delta <= 0:
            accept = True
        else:
            # avoid overflow
            if temp > 1e-12:

            zeros.append(i)
            zeros.remove(j)
            ones.append(j)
            # update best

                best_h = h.copy()

            else:

def _compute_c5(h, dx):
    """Maximum overlap integral for a discretized step function."""
    # correlation of h with (1-h), then take the maximum and scale by dx
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx
