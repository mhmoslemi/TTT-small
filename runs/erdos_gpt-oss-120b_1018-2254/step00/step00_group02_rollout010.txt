def project_to_feasible(h, target):
    # projection onto box [0,1] and sum constraint via iterative water-filling
    # start with clipping
    h = np.clip(h, 0.0, 1.0)
    # compute current sum
    s = np.sum(h)
    # if sum already equals target within tolerance, return
    # else adjust
    # Perform water-filling loop up to 100 iterations
    for _ in range(100):
        diff = target - s
        if abs(diff) < 1e-9:
            break
        # indices that can be adjusted (not at bound)
        if diff > 0:
            free = np.where(h < 1.0)[0]
        else:
            free = np.where(h > 0.0)[0]
        if free.size == 0:
            # cannot adjust further
            break
        delta = diff / free.size
        h[free] += delta
        # clip again
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
    return h

def hill_climb_optimize(h, max_iters, step_scale):
    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    n = len(h)
    for _ in range(max_iters):
        # pick two indices i!=j
        i, j = rng.integers(n, size=2)
        if i == j: continue
        # choose delta uniformly in [-step, step] (maybe adapt)
        delta = (rng.random() * 2 - 1) * step_scale
        if delta == 0: continue
        # check bounds
        hi = current_h[i]
        hj = current_h[j]
        if hi + delta < 0 or hi + delta > 1: continue
        if hj - delta < 0 or hj - delta > 1: continue
        # propose new h
        h_new = current_h.copy()
        h_new[i] = hi + delta
        h_new[j] = hj - delta
        c5_new = compute_c5(h_new)
        if c5_new < current_c5:
            current_h = h_new
            current_c5 = c5_new
    return current_h, current_c5

def project_to_feasible(h, target, max_iter=100):
    # Ensure bounds
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # Determine which indices can be moved
        if diff > 0:

        else:
            free = np.where(h > 0.0)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # Final adjustment after loop to correct small residual
    # Slightly adjust using simple scaling if needed
    # ensure final sum close to target
    s = np.sum(h)

import time
import numpy as np
from scipy.optimize import differential_evolution

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def project_to_feasible(h, target, max_iter=100):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = np.where(h < 1.0)[0]
        else:
            free = np.where(h > 0.0)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(old_h, n_new):
    x_old = np.linspace(0.0, 2.0, old_h.shape[0])
    x_new = np.linspace(0.0, 2.0, n_new)
    h_new = np.interp(x_new, x_old, old_h)
    return project_to_feasible(h_new, n_new / 2.0)

def hill_climb_optimize(h0, rng, max_iters=2000, step_scale=0.1):
    h = h0.copy()
    n = h.shape[0]
    cur = compute_c5(h)
    for _ in range(max_iters):
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        delta = (rng.random() * 2.0 - 1.0) * step_scale
        if delta == 0.0:
            continue
        hi, hj = h[i], h[j]
        if hi + delta < 0.0 or hi + delta > 1.0 or hj - delta < 0.0 or hj - delta > 1.0:
            continue
        h[i] = hi + delta
        h[j] = hj - delta
        new = compute_c5(h)
        if new < cur:
            cur = new
        else:
            h[i] = hi
            h[j] = hj
    return h, cur

def de_objective(x, n, target, penalty_weight):
    diff = np.sum(x) - target
    c5 = compute_c5(x)
    return c5 + penalty_weight * diff * diff

def de_optimize(n, init_h, rng, maxiter=30, popsize=8, penalty_weight=1e3):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    seed = int(rng.integers(1 << 30))
    result = differential_evolution(
        de_objective,
        bounds,
        args=(n, target, penalty_weight),
        maxiter=maxiter,
        popsize=popsize,
        polish=False,
        seed=seed,
        disp=False,
    )
    x = project_to_feasible(result.x, target)
    return hill_climb_optimize(x, rng, max_iters=2000, step_scale=0.05)
