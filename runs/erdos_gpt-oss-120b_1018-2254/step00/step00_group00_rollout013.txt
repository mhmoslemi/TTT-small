def project_to_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    # water-filling loop
    while True:
        sum_h = h.sum()
        diff = target - sum_h
        if abs(diff) < 1e-10:
            break
        # find free indices not at bounds in direction we need to move
        if diff > 0:
            free = np.where(h < 1.0 - 1e-12)[0]
        else:
            free = np.where(h > 1e-12)[0]
        if free.size == 0:
            # cannot satisfy sum due to bounds; just break
            break
        # amount to add per free variable
        delta = diff / free.size
        # apply delta but clamp
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
        # loop again if needed
    return h

def gradient_descent_refine(h, max_iters=200, step=0.1):
    n = len(h)
    for i in range(max_iters):
        # compute correlation for all shifts and find worst shift
        dx = 2.0 / n
        corr = np.correlate(h, 1 - h, mode='full')
        # convert to actual values: overlap integral for each shift = corr[idx] * dx
        # find index of max corr
        idx = np.argmax(corr)
        # shift corresponds to k = idx - (n - 1)
        k = idx - (n - 1)
        # compute gradient: for j in range(n):
        # gradient_j = 1 - (h[j+k] if 0 <= j+k < n else 0) - (h[j-k] if 0 <= j-k < n else 0)
        grad = np.ones(n)
        if k >= 0:
            grad[:n - k] -= h[k:]
            grad[k:] -= h[:n - k]
        else:
            # negative k: shift left
            k_abs = -k
            grad[k_abs:] -= h[:n - k_abs]
            grad[:n - k_abs] -= h[k_abs:]
        # step
        h = h - step * grad
        # project back
        h = project_to_box_sum(h, target_sum)
    return h

    h = np.clip(h, 0.0, 1.0)

import numpy as np

# ------------------------------------------------------------
# Helper functions

def project_to_box_sum(x, target_sum, max_iter=1000, tol=1e-9):
    """
    Project vector x onto the feasible set:
        0 <= x_i <= 1  and  sum(x) = target_sum
    Uses a simple water-filling iterative scheme.
    """
    x = np.clip(x, 0.0, 1.0)
    for _ in range(max_iter):
        cur_sum = x.sum()
        diff = target_sum - cur_sum
        if abs(diff) < tol:
            break
        if diff > 0:
            free = x < 1.0 - 1e-12
        else:
            free = x > 1e-12
        idx = np.where(free)[0]
        if idx.size == 0:
            break
        delta = diff / idx.size
        x[idx] += delta
        x = np.clip(x, 0.0, 1.0)
    return x


def evaluate_overlap(h):
    """
    Compute C5 for a given discrete step function h.
    """
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def uniform_crossover(p1, p2):
    """Uniform crossover producing a child from two parents."""
    mask = np.random.rand(p1.shape[0]) < 0.5
    child = np.where(mask, p1, p2)
    return child.astype(float)


def mutate(x, rate, sigma):
    """Gaussian mutation on a fraction of genes."""
    mask = np.random.rand(x.shape[0]) < rate
    noise = np.random.randn(x.shape[0]) * sigma
    y = x.copy()
    y[mask] += noise[mask]
    return y


def tournament_selection(scores, size):
    """Return index of the best individual among a random subset."""
    participants = np.random.choice(len(scores), size, replace=False)
    best = participants[0]
    best_score = scores[best]
    for p in participants[1:]:
        if scores[p] < best_score:
            best = p
            best_score = scores[p]
    return best


def gradient_descent_refine(h, target_sum, max_iters=200, step=0.05):
    """
    Simple projected sub‑gradient descent that reduces the worst overlap.
    """
    n = h.shape[0]
    for _ in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr)
        k = idx - (n - 1)          # shift corresponding to max overlap
        grad = np.ones(n)

        if k >= 0:
            # Terms h_j+k
            grad[:n - k] -= h[k:]
            # Terms h_j‑k
            grad[k:] -= h[:n - k]
        else:
            k_abs = -k
            grad[k_abs:] -= h[:n - k_abs]
            grad[:n - k_abs] -= h[k_abs:]

        h = h - step * grad
        h = project_to_box_sum(h, target_sum)
    return h
