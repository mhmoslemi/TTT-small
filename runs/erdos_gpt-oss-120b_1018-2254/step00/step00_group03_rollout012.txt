def adjust_sum(u, target):
    diff = target - u.sum()
    if abs(diff) < 1e-12:
        return u
    # Try to add diff to an element not at bound
    for i in range(len(u)):
        if diff > 0 and u[i] < 1:
            max_inc = 1 - u[i]
            add = min(diff, max_inc)
            u[i] += add
            diff -= add
        elif diff < 0 and u[i] > 0:
            max_dec = u[i]
            sub = min(-diff, max_dec)
            u[i] -= sub
            diff += sub
        if abs(diff) < 1e-12:

    # If still diff, may not be possible due to rounding. ignore.
    return u

def tournament_selection(pop, size, rng):

import time
import numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # correlation h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

# --- binary utilities ---------------------------------------------------------
def binary_from_continuous(v, target_ones, rng):
    """Convert a continuous vector to a binary vector with exactly target_ones ones."""
    bin_vec = (v >= 0.5).astype(float)
    cur = int(bin_vec.sum())
    if cur > target_ones:
        idx = np.where(bin_vec == 1)[0]
        to_flip = rng.choice(idx, size=cur - target_ones, replace=False)
        bin_vec[to_flip] = 0.0
    elif cur < target_ones:
        idx = np.where(bin_vec == 0)[0]
        to_flip = rng.choice(idx, size=target_ones - cur, replace=False)
        bin_vec[to_flip] = 1.0
    return bin_vec

def thue_morse_binary(n, target_ones, rng):
    """Generate a Thue‑Morse binary sequence of length n, then adjust to target_ones."""
    seq = np.fromiter((i.bit_count() % 2 for i in range(n)), dtype=float, count=n)
    cur = int(seq.sum())
    if cur > target_ones:
        idx = np.where(seq == 1)[0]
        to_flip = rng.choice(idx, size=cur - target_ones, replace=False)
        seq[to_flip] = 0.0
    elif cur < target_ones:
        idx = np.where(seq == 0)[0]
        to_flip = rng.choice(idx, size=target_ones - cur, replace=False)
        seq[to_flip] = 1.0
    return seq

def random_binary_vector(n, target_ones, rng):
    idx = rng.choice(n, size=target_ones, replace=False)
    vec = np.zeros(n, dtype=float)
    vec[idx] = 1.0
    return vec

def crossover_binary(p1, p2, target_ones, rng):
    """Uniform crossover then projection to exact number of ones."""
    mask = rng.random(p1.shape[0]) < 0.5
    child = np.where(mask, p1, p2).astype(float)
    cur = int(child.sum())
    if cur > target_ones:
        idx = np.where(child == 1)[0]
        to_flip = rng.choice(idx, size=cur - target_ones, replace=False)
        child[to_flip] = 0.0
    elif cur < target_ones:
        idx = np.where(child == 0)[0]
        to_flip = rng.choice(idx, size=target_ones - cur, replace=False)
        child[to_flip] = 1.0
    return child

def mutate_binary(h, mutation_rate, rng):
    """Perform a number of random 1↔0 swaps proportional to mutation_rate."""
    n = h.shape[0]
    n_swaps = max(1, int(mutation_rate * n))
    for _ in range(n_swaps):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
    return h

def tournament_select(pop, k, rng):
    """Select the best of k randomly chosen individuals."""
    idx = rng.integers(0, len(pop), size=k)
    best = pop[idx[0]]
    for i in idx[1:]:
        if pop[i]['c5'] < best['c5']:
            best = pop[i]
    return best

# --- continuous utilities ------------------------------------------------------
def project_to_bounds_and_sum(v, target_sum):
    """Project v onto [0,1]^n with sum==target_sum via simple bisection."""
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        u = np.clip(v - mid, 0.0, 1.0)
        s = u.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny fix to hit the exact sum
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-12:
        # adjust a non‑boundary element
        for i in range(len(proj)):
            if (diff > 0 and proj[i] < 1.0) or (diff < 0 and proj[i] > 0.0):
                adj = min(abs(diff), 1.0 - proj[i]) if diff > 0 else min(-diff, proj[i])
                proj[i] += np.sign(diff) * adj
                break
    return proj

def random_continuous_vector(n, target_sum, rng):
    v = rng.random(n)
    return project_to_bounds_and_sum(v, target_sum)

def mutate_continuous(h, rate, sd, target_sum, rng):
    mask = rng.random(h.shape[0]) < rate
    noise = rng.normal(0.0, sd, size=h.shape[0])
    h_new = h.copy()
    h_new[mask] += noise[mask]
    return project_to_bounds_and_sum(h_new, target_sum)

def crossover_continuous(p1, p2, rng):
    alpha = rng.random(p1.shape[0])
    return alpha * p1 + (1 - alpha) * p2

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function yielding a low C₅ bound."""
    start = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------ init
    has_initial = 'initial_h_values' in globals()
    if has_initial:
        init_h = np.asarray(initial_h_values, dtype=float)
    else:
        # fallback: 70 points, average 0.5
        init_h = random_continuous_vector(70, 35.0, rng)

    n = init_h.shape[0]
    target_sum = n / 2.0
    binary_mode = (n % 2 == 0)

    pop_size = 30
    tournament_k = 3
    mutation_rate = 0.08   # fraction of indices to swap (binary) or per‑gene prob (cont.)
    mutation_sd = 0.15     # only for continuous
    max_gen = 100000  # large; we will stop by time

    # ------------------------------------------------------------------ population
    population = []

    if binary_mode:
        target_ones = int(target_sum)
        # 1) transformed initial (if any)
        if has_initial:
            bin_initial = binary_from_continuous(init_h, target_ones, rng)
            population.append({'h': bin_initial, 'c5': compute_c5(bin_initial)})
        # 2) a Thue‑Morse seed
        tm = thue_morse_binary(n, target_ones, rng)
        population.append({'h': tm, 'c5': compute_c5(tm)})
        # fill remainder with random binary vectors
        while len(population) < pop_size:
            h = random_binary_vector(n, target_ones, rng)
            population.append({'h': h, 'c5': compute_c5(h)})
    else:
        # continuous mode
        # include initial if feasible
        if has_initial:
            proj = project_to_bounds_and_sum(init_h, target_sum)
            population.append({'h': proj, 'c5': compute_c5(proj)})
        # random rest
        while len(population) < pop_size:
            h = random_continuous_vector(n, target_sum, rng)
            population.append({'h': h, 'c5': compute_c5(h)})

    # keep population sorted (best first)
    population.sort(key=lambda x: x['c5'])
    best = population[0]

    # ------------------------------------------------------------------ main loop
    time_limit = start + budget_s * 0.93  # reserve a bit for final polishing
    gen = 0
    while time.time() < time_limit and gen < max_gen:
        # selection
        p1 = tournament_select(population, tournament_k, rng)
        p2 = tournament_select(population, tournament_k, rng)
        # crossover & mutation
        if binary_mode:
            child_h = crossover_binary(p1['h'], p2['h'], target_ones, rng)
            child_h = mutate_binary(child_h, mutation_rate, rng)
        else:
            child_h = crossover_continuous(p1['h'], p2['h'], rng)
            child_h = mutate_continuous(child_h, mutation_rate, mutation_sd, target_sum, rng)

        child_c5 = compute_c5(child_h)
        # replace worst if better
        if child_c5 < population[-1]['c5']:
            population[-1] = {'h': child_h, 'c5': child_c5}
            population.sort(key=lambda x: x['c5'])
            best = population[0]
        gen += 1

    # ------------------------------------------------------------------ local refinement
    # Simple hill‑climbing swaps for binary; small Gaussian tweaks for continuous.
    finish_time = start + budget_s - 0.05
    if binary_mode:
        h_cur = best['h'].copy()
        c5_cur = best['c5']
        while time.time() < finish_time:
            # perform a handful of random swaps and accept improvements
            for _ in range(5):
                ones = np.where(h_cur == 1)[0]
                zeros = np.where(h_cur == 0)[0]
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = rng.choice(ones)
                j = rng.choice(zeros)
                h_new = h_cur.copy()
                h_new[i] = 0.0
                h_new[j] = 1.0
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur:
                    h_cur, c5_cur = h_new, c5_new
            # occasional break to check time
            if time.time() >= finish_time:
                break
        best_h = h_cur
        best_c5 = c5_cur
    else:
        h_cur = best['h'].copy()
        c5_cur = best['c5']
        while time.time() < finish_time:
            # tiny Gaussian perturbation then project
            h_new = mutate_continuous(h_cur, 0.02, 0.05, target_sum, rng)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur:
                h_cur, c5_cur = h_new, c5_new
        best_h = h_cur
        best_c5 = c5_cur

    # final sanity: ensure constraints
    best_h = np.clip(best_h, 0.0, 1.0)
    if binary_mode:
        # enforce exact number of ones (tiny rounding errors may appear)
        cur_ones = int(round(best_h.sum()))
        if cur_ones != target_ones:
            # adjust
            diff = target_ones - cur_ones
            if diff > 0:
                zeros = np.where(best_h == 0)[0]
                idx = rng.choice(zeros, size=diff, replace=False)
                best_h[idx] = 1.0
            elif diff < 0:
                ones = np.where(best_h == 1)[0]
                idx = rng.choice(ones, size=-diff, replace=False)
                best_h[idx] = 0.0
    else:
        best_h = project_to_bounds_and_sum(best_h, target_sum)

    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n)
