def hill_climb_binary(h, n, max_iters, start_time, budget_s):
    N = n // 2
    ones = np.flatnonzero(h == 1)
    zeros = np.flatnonzero(h == 0)
    best_h = h.copy()
    best_minB = compute_min_corr(h)
    best_c5 = 1.0 - (2.0 / n) * best_minB

    cur_h = h.copy()
    cur_minB = best_minB
    cur_c5 = best_c5

    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95: break
        # pick random indices
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # swap
        cur_h[i] = 0
        cur_h[j] = 1
        # compute new auto
        new_minB = compute_min_corr(cur_h)
        new_c5 = 1.0 - (2.0 / n) * new_minB
        # acceptance
        if new_c5 < cur_c5: # improvement
            # accept
            cur_minB = new_minB
            cur_c5 = new_c5
            # update ones, zeros lists
            ones = np.where(cur_h == 1)[0]
            zeros = np.where(cur_h == 0)[0]
        else:
            # maybe revert based on simulated annealing probability
            # compute temperature
            fraction = elapsed / budget_s
            T = max(1e-6, 0.01 * (1 - fraction))
            if np.random.rand() < np.exp(-(new_c5 - cur_c5) / T):
                # accept worse
                cur_minB = new_minB
                cur_c5 = new_c5
                ones = np.where(cur_h == 1)[0]
                zeros = np.where(cur_h == 0)[0]
            else:
                # revert swap
                cur_h[i] = 1
                cur_h[j] = 0
        # update global best if improved
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = cur_h.copy()

def hill_climb(initial_h, n, start_time, budget_s):
    best_h = initial_h.copy()
    best_minB = min_B_of_h(best_h)
    best_c5 = 1.0 - (2.0 / n) * best_minB
    cur_h = best_h.copy()
    cur_minB = best_minB
    cur_c5 = best_c5

    ones = [int(i) for i in np.flatnonzero(cur_h == 1)]
    zeros = [int(i) for i in np.flatnonzero(cur_h == 0)]
    # target modify attempts defined
    iteration = 0
    no_improve = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
        iteration += 1
        # random swap
        i = random.choice(ones)
        j = random.choice(zeros)
        # perform swap
        cur_h[i] = 0
        cur_h[j] = 1
        # compute new minB
        new_minB = min_B_of_h(cur_h)
        new_c5 = 1.0 - (2.0 / n) * new_minB
        # acceptance based on improvement or annealing
        if new_c5 < cur_c5 - 1e-12:
            # improvement
            cur_minB = new_minB
            cur_c5 = new_c5
            # update ones/zeros
            ones.remove(i)
            zeros.append(i)
            zeros.remove(j)
            ones.append(j)
            no_improve = 0
            # update best
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = cur_h.copy()
                best_minB = new_minB
        else:
            # maybe accept worse
            # compute temperature
            fraction = elapsed / budget_s
            T = max(1e-6, 0.02 * (1.0 - fraction))
            prob = np.exp(-(new_c5 - cur_c5) / T)
            if np.random.rand() < prob:
                # accept worse
                cur_minB = new_minB
                cur_c5 = new_c5
                # update ones/zeros
                ones.remove(i)
                zeros.append(i)
                zeros.remove(j)
                ones.append(j)
                no_improve = 0
            else:
                # revert
                cur_h[i] = 1
                cur_h[j] = 0
                no_improve += 1
        # if stuck for many attempts, reinitialize random h
        if no_improve >= 5000:
            # reinitialize random binary with N ones
            cur_h = np.zeros(n, dtype=int)
            ones_idx = np.random.choice(n, n // 2, replace=False)
            cur_h[ones_idx] = 1
            cur_minB = min_B_of_h(cur_h)
            cur_c5 = 1.0 - (2.0 / n) * cur_minB
            ones = [int(i) for i in np.flatnonzero(cur_h==1)]
            zeros = [int(i) for i in np.flatnonzero(cur_h==0)]
            no_improve = 0
    return best_h, best_c5
