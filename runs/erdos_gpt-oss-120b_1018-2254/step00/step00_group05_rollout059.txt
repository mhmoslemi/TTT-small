def propose_move(h, max_step, rng):
    n = len(h)
    i, j = rng.choice(n, size=2, replace=False)
    # Determine feasible delta range: 
    # deltamax = min(max_step, 1 - h[i]) # can't exceed 1

    delta = rng.uniform(low, high)

import time
import math
import numpy as np

def _project_box_sum(v, target):
    """Project v onto the set {0<=x_i<=1, sum x_i = target}."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _gradient_shift(h, shift):
    n = h.shape[0]
    grad = np.ones(n, dtype=np.float64)
    if shift >= 0:
        # term h[i-shift]
        grad[shift:] -= h[:n - shift]
        # term h[i+shift]
        grad[:n - shift] -= h[shift:]
    else:
        s = -shift
        grad[:n - s] -= h[s:]
        grad[s:] -= h[:n - s]
    return grad

def _max_corr(h, dx):
    # np.correlate returns linear correlation (full mode)
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialize problem size and target sum
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n = h0.size
    else:
        n = 70
        h0 = None
    target_sum = n / 2.0
    dx = 2.0 / n

    # ------------------------------------------------------------------
    # Initial h: use supplied values if present, otherwise random.
    # Then enforce box and sum constraints.
    # ------------------------------------------------------------------
    if h0 is not None:
        h = np.clip(h0, 0.0, 1.0)
        h = _project_box_sum(h, target_sum)
    else:
        # start with a random binary vector (exactly target_sum ones)
        h = np.zeros(n, dtype=np.float64)
        ones = rng.choice(n, size=int(target_sum), replace=False)
        h[ones] = 1.0

    cur_val = _max_corr(h, dx)
    best_h = h.copy()
    best_val = cur_val

    start = time.time()
    # parameters for the sub‑gradient (simulated‑annealing) loop
    lr0 = 1.0          # initial step size
    T0 = 5.0           # initial temperature for stochastic acceptance
    max_no_improve = 5000
    no_improve_cnt = 0

    while True:
        now = time.time()
        if now - start > budget_s - 0.2:
            break

        # --------------------------------------------------------------
        # evaluate current worst shift and its sub‑gradient
        # --------------------------------------------------------------
        corr_full = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(corr_full))
        shift = idx - (n - 1)          # shift corresponding to the max entry
        grad = _gradient_shift(h, shift)

        # --------------------------------------------------------------
        # step size and temperature decay (linear in elapsed time)
        # --------------------------------------------------------------
        elapsed = now - start
        frac = elapsed / budget_s
        lr = lr0 * (1.0 - frac) + 0.005   # keep a tiny floor step
        T = max(1e-5, T0 * (1.0 - frac))

        # --------------------------------------------------------------
        # propose a move
        # --------------------------------------------------------------
        y = h - lr * grad
        y = _project_box_sum(y, target_sum)   # stay feasible

        new_val = _max_corr(y, dx)
        delta = new_val - cur_val

        # accept if better or with Metropolis probability
        if delta < 0 or rng.random() < math.exp(-delta / T):
            h = y
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
                no_improve_cnt = 0
            else:
                no_improve_cnt += 1
        else:
            no_improve_cnt += 1

        # --------------------------------------------------------------
        # occasional diversification
        # --------------------------------------------------------------
        if no_improve_cnt >= max_no_improve:
            # perturb the current best a little and continue
            h = best_h + 0.2 * rng.standard_normal(n)
            h = np.clip(h, 0.0, 1.0)
            h = _project_box_sum(h, target_sum)
            cur_val = _max_corr(h, dx)
            no_improve_cnt = 0

    # final validation (project once more to be safe)
    best_h = _project_box_sum(best_h, target_sum)
    final_val = _max_corr(best_h, dx)

    return best_h, final_val, n
