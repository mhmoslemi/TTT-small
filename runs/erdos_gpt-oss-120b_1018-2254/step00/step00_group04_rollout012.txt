import numpy as np
import time
import math

def _compute_c5(h: np.ndarray) -> float:
    """Return max_{shift} ∫ h(x)(1-h(x+shift))dx approximated by correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()
    total_budget = float(budget_s)

    # ------------------------------------------------------------------

            h_current = np.clip(init, 0.0, 1.0).copy()
        else:
            raise ValueError
    except Exception:
        # no valid parent: pick a modest even size and start from a binary pattern
        n = 100
        if n % 2:
            n += 1
        target_sum = n / 2.0

        def _random_binary():
            h = np.zeros(n, dtype=float)
            idx = rng.choice(n, n // 2, replace=False)
            h[idx] = 1.0
            return h

        h_current = _random_binary()

    # ------------------------------------------------------------------
    # Helper to keep exact sum after the whole optimisation
    # ------------------------------------------------------------------
    def _fix_sum(vec: np.ndarray) -> np.ndarray:

    # ------------------------------------------------------------------
