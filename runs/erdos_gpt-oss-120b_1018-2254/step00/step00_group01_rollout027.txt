import numpy as np, time

def _project_bounds_sum(v, target):
    lo = 0.0
    hi = 1.0
    # binary search theta
    lower = lo - np.max(v) - 1.0
    upper = hi - np.min(v) + 1.0
    for _ in range(60):
        mid = (lower + upper) * 0.5
        proj = np.clip(v + mid, lo, hi)
        s = proj.sum()
        if s < target:
            lower = mid
        else:
            upper = mid
    theta = (lower + upper) * 0.5
    return np.clip(v + theta, lo, hi)

def _c5_manual(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

    try:
        res = evaluate_erdos_solution(h, h.shape[0])
        if isinstance(res, (list, tuple)):
            return float(res[0])
        return float(res)
    except Exception:
        return _c5_manual(h)

def _gradient_maxlag(h):
    n = h.shape[0]
    # correlation without factor
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = int(np.argmax(corr))

    if shift >= 0:
        i = np.arange(0, n - shift)
        j = i + shift
        grad[i] += (1.0 - h[j])
        grad[j] -= h[i]
    else:
        t = -shift
        i = np.arange(t, n)
        j = i - t
        grad[i] += (1.0 - h[j])
        grad[j] -= h[i]
    return grad

import numpy as np
import time

def _project_bounds_sum(v, target):
    """Project vector v onto [0,1]^n intersect {sum = target}."""
    lo, hi = 0.0, 1.0
    # Find theta such that sum(clip(v+theta,0,1)) = target
    lower = lo - np.max(v) - 1.0
    upper = hi - np.min(v) + 1.0
    for _ in range(60):
        mid = (lower + upper) * 0.5
        proj = np.clip(v + mid, lo, hi)
        s = proj.sum()
        if s < target:
            lower = mid
        else:
            upper = mid
    theta = (lower + upper) * 0.5
    return np.clip(v + theta, lo, hi)

def _c5_manual(h):
    """Fallback C5 computation if evaluate_erdos_solution is unavailable."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def _evaluate(h):
    """Compute C5 using the provided evaluator or the manual fallback."""
    try:
        res = evaluate_erdos_solution(h, h.shape[0])
        if isinstance(res, (list, tuple)):
            return float(res[0])
        return float(res)
    except Exception:
        return _c5_manual(h)

def _gradient_maxlag(h):
    """Subgradient of the worst (max) correlation term."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = int(np.argmax(corr))
    shift = max_idx - (n - 1)               # integer lag attaining max
    grad = np.zeros_like(h)
    if shift >= 0:
        i = np.arange(0, n - shift)
        j = i + shift
        grad[i] += (1.0 - h[j])
        grad[j] -= h[i]
    else:
        t = -shift
        i = np.arange(t, n)
        j = i - t
        grad[i] += (1.0 - h[j])
        grad[j] -= h[i]
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] with low Erdős overlap constant C5."""
    start = time.time()
    np.random.seed(seed)

    # Determine problem size and initial vector
    if 'initial_h_values' in globals():
        n = int(initial_h_values.shape[0])
        h = np.copy(initial_h_values)
    else:
        n = 47
        h = np.full(n, 0.5)

    target_sum = n * 0.5                     # because integral must be 1 on [0,2]
    h = _project_bounds_sum(h, target_sum)

    best_h = h.copy()
    best_c5 = _evaluate(best_h)

    cur_h = best_h.copy()
    cur_c5 = best_c5

    step_scale = 0.07                         # initial perturbation magnitude
    min_step = 0.001
    last_improve = 0
    it = 0

    while True:
        elapsed = time.time() - start
        if elapsed > budget_s * 0.98:
            break

        # ---------- propose a new candidate ----------
        if np.random.rand() < 0.25:
            # gradient descent on the worst lag
            g = _gradient_maxlag(cur_h)
            norm = np.linalg.norm(g)
            if norm > 0:
                delta = -step_scale * (g / norm)
            else:
                delta = np.zeros_like(cur_h)
            cand = cur_h + delta
        else:
            # random perturbation of all coordinates
            delta = (np.random.rand(n) - 0.5) * 2.0 * step_scale
            cand = cur_h + delta

        # enforce feasibility
        cand = _project_bounds_sum(cand, target_sum)

            cur_c5 = best_c5
            step_scale = max(min_step, step_scale * 0.5)
            last_improve = it

        it += 1

    return (best_h, best_c5, n)
