import time
import numpy as np

def compute_c5(h):
    N = h.shape[0]
    dx = 2.0 / N
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def adjust_sum_to_target(h, target):
    """Adjust vector h in-place so that sum(h) == target while keeping 0<=h<=1."""
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # adjust greedily
    if diff > 0:
        # Need to increase some entries that are <1
        for idx in np.where(h < 1.0)[0]:
            cap = 1.0 - h[idx]
            add = min(cap, diff)
            h[idx] += add
            diff -= add
            if diff <= 1e-12:
                break
    else:
        # Need to decrease some entries that are >0
        for idx in np.where(h > 0.0)[0]:
            cap = h[idx]
            sub = min(cap, -diff)
            h[idx] -= sub
            diff += sub
            if diff >= -1e-12:
                break
    return h

def simulated_annealing(initial_h, time_budget, target_sum):
    N = initial_h.shape[0]
    current_h = initial_h.copy()
    current_score = compute_c5(current_h)
    best_h = current_h.copy()
    best_score = current_score

    start = time.time()
    iteration = 0
    max_iter = 10_000_000

    # Parameters for temperature schedule
    # Starting temperature set proportionally to initial score
    T0 = max(0.05, current_score * 0.5)

    while iteration < max_iter:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break

        # select i (can increase) and j (can decrease)
        # Filter indices to avoid infeasible moves
        inc_candidates = np.where(current_h < 1.0 - 1e-12)[0]
        dec_candidates = np.where(current_h > 1e-12)[0]

        if inc_candidates.size == 0 or dec_candidates.size == 0:
            break

        i = inc_candidates[np.random.randint(inc_candidates.size)]
        j = dec_candidates[np.random.randint(dec_candidates.size)]

        # ensure i != j
        if i == j:
            continue

        # feasible delta
        max_inc = 1.0 - current_h[i]
        max_dec = current_h[j]
        max_delta = min(max_inc, max_dec)
        if max_delta <= 0:
            continue

        # draw delta (scaled down by factor to keep steps moderate)
        delta = max_delta * np.random.rand() * 0.5
        if delta <= 0:
            continue

        # propose move
        current_h[i] += delta
        current_h[j] -= delta

        new_score = compute_c5(current_h)
        dE = new_score - current_score

        # temperature schedule (linear cooling)
        T = T0 * (1.0 - iteration / max_iter)

            if new_score < best_score:
                best_score = new_score
                best_h = current_h.copy()
        else:
            # revert
            current_h[i] -= delta
            current_h[j] += delta

        iteration += 1

import time
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a given step function."""
    N = h.shape[0]
    dx = 2.0 / N
    # correlation of h with (1-h); mode='full' yields linear correlation
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))


def _adjust_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust values in‑place so that sum(h) == target while keeping 0 ≤ h ≤ 1."""
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # need to increase some entries
        for idx in np.where(h < 1.0)[0]:
            inc = min(1.0 - h[idx], diff)
            h[idx] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    else:
        # need to decrease some entries
        for idx in np.where(h > 0.0)[0]:
            dec = min(h[idx], -diff)
            h[idx] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return h


def _simulated_annealing(start_h: np.ndarray, time_budget: float, target_sum: float):
    """Simple SA that preserves the linear constraint by pairwise transfers."""
    N = start_h.shape[0]
    cur_h = start_h.copy()
    cur_score = _compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score

    start = time.time()
    max_iter = 10_000_000
    # initial temperature scaled from the starting score
    T0 = max(0.05, cur_score * 0.5)

    for it in range(max_iter):
        if time.time() - start >= time_budget:
            break

        # indices that can be increased / decreased
        inc_idx = np.where(cur_h < 1.0 - 1e-12)[0]
        dec_idx = np.where(cur_h > 1e-12)[0]
        if inc_idx.size == 0 or dec_idx.size == 0:
            break

        j = dec_idx[np.random.randint(dec_idx.size)]

        max_dec = cur_h[j]

        delta = max_delta * np.random.rand() * 0.5

        cur_h[j] -= delta

            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            # reject – revert

            cur_h[j] += delta

    # ------------------------------------------------------------------
    start = time.time()
    remaining = max(0.0, budget_s - (time.time() - start) - 0.2)  # tiny margin
    best_h, _ = _simulated_annealing(h0, remaining, target)

    # ------------------------------------------------------------------
    # 3. Final sanity fix – enforce the constraints exactly.
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _adjust_sum(best_h, target)

    final_c5 = _compute_c5(best_h)

    return best_h, final_c5, N
