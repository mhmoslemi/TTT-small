def project_to_box_and_sum(v, target):
    n = len(v)
    # lambda bounds
    lo = -np.max(v) - 1.0   # ensures v+lo <= 0, sum = 0
    hi = 1.0 - np.min(v) + 1.0 # ensures v+hi >=1, sum = n
    # binary search
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v + lam, 0, 1)
        s = x.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(v + (lo+hi)/2.0, 0, 1)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return corr.max() * dx

def projection(v, target_sum):
    # find lambda that makes sum clamp(v+lambda,[0,1]) == target_sum
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s < target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v + (lo+hi)/2.0, 0.0, 1.0)

def worst_shift_and_grad(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    s = idx - (n - 1)
    if s >= 0:
        left = np.concatenate((np.zeros(s), h[:n-s]))
        right = np.concatenate((h[s:], np.zeros(s)))

        left = np.concatenate((h[s_abs:], np.zeros(s_abs)))
        right = np.concatenate((np.zeros(s_abs), h[:n-s_abs]))
    grad = 1.0 - (left + right)
    return s, grad

def optimize(n, init_h, time_limit):

    lr = 0.5
    it = 0

        # gradient descent

import numpy as np, time, random

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return corr.max() * dx

def project_onto_sum_box(v, target):
    # Find lambda such that sum(clamp(v + lam,0,1)) == target
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

def worst_shift_and_gradient(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    s = idx - (n - 1)
    if s >= 0:
        left = np.concatenate((np.zeros(s), h[:n - s]))
        right = np.concatenate((h[s:], np.zeros(s)))
    else:
        s_abs = -s
        left = np.concatenate((h[s_abs:], np.zeros(s_abs)))
        right = np.concatenate((np.zeros(s_abs), h[:n - s_abs]))
    grad = 1.0 - (left + right)
    return s, grad

def gradient_descent_optimize(h0, time_budget):
    n = h0.shape[0]
    target = n / 2.0
    h = project_onto_sum_box(h0.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    lr = 0.5
    iteration = 0
    while time.time() - start < time_budget:
        s, grad = worst_shift_and_gradient(h)
        # step
        h = h - lr * grad
        # keep feasibility
        h = project_onto_sum_box(h, target)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # decay lr
        lr *= 0.9995
        iteration += 1
        if iteration % 200 == 0:
            # occasional random perturb
            h = best_h + np.random.randn(n) * 0.01
            h = project_onto_sum_box(h, target)
            lr = 0.5
    return best_h, best_c5

def binary_from_continuous(h):
    n = h.shape[0]
    k = n // 2
    # indices of largest k entries
    idx = np.argpartition(h, -k)[-k:]
    b = np.zeros_like(h)
    b[idx] = 1.0
    return b

def binary_hill_climb(b0, time_budget):
    n = b0.shape[0]
    b = b0.copy()
    best_b = b.copy()
    best_c5 = compute_c5(b)
    start = time.time()
    # precompute list of indices for ones and zeros
    # For efficiency, maintain sets
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]
    while time.time() - start < time_budget:
        # select random i in ones, j in zeros
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # swap
        b[i], b[j] = b[j], b[i]
        # evaluate
        c5 = compute_c5(b)
        if c5 < best_c5:
            best_c5 = c5
            best_b = b.copy()
            # update lists
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
        else:
            # revert
            b[i], b[j] = b[j], b[i]
        # optionally break early if perfect? Not needed
    return best_b, best_c5

def upsample_vector(v, n_target):
    n_src = v.shape[0]
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_tgt = np.linspace(0, 2, n_target, endpoint=False)
    v_new = np.interp(x_tgt, x_src, v)
    return v_new

def run(seed=42, budget_s=1100, **kwargs):

            # convert to binary and hill climb
            b0 = binary_from_continuous(h_opt)
            # ensure sum exactly n/2

            b_opt, c5_b = binary_hill_climb(b0, portion * 0.5)  # half time
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = b_opt.copy()
                best_n = n
            # also check binary version from original init
            b_init = binary_from_continuous(project_onto_sum_box(h0, n/2.0))
            b_opt2, c5_b2 = binary_hill_climb(b_init, portion * 0.5)
            if c5_b2 < best_c5:
                best_c5 = c5_b2
                best_h = b_opt2.copy()
                best_n = n
            # early exit if we already beat target
            if best_c5 <= 0.3808:
                break
        if best_c5 <= 0.3808:
            break
    # fallback if no solution found
    if best_h is None:
        # simple balanced binary

        b = np.zeros(n)
        b[:n//2] = 1.0
        np.random.shuffle(b)
        best_h = b
        best_n = n

    # final projection (ensure feasibility)
    best_h = project_onto_sum_box(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, best_n

import numpy as np
import time
import random

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)*(1-h(x+k)) dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def project_onto_sum_box(v, target):
    """Euclidean projection of v onto {x∈[0,1]^n : sum(x)=target}."""
    lo = -np.max(v) - 1.0          # makes all v+lo ≤ 0  → sum 0
    hi = 1.0 - np.min(v) + 1.0     # makes all v+hi ≥ 1  → sum n
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

def worst_shift_and_gradient(h):
    """Find shift with maximal overlap and its gradient w.r.t. h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    s = idx - (n - 1)               # shift (negative … positive)
    if s >= 0:
        left = np.concatenate((np.zeros(s), h[:n - s]))
        right = np.concatenate((h[s:], np.zeros(s)))
    else:
        s_abs = -s
        left = np.concatenate((h[s_abs:], np.zeros(s_abs)))
        right = np.concatenate((np.zeros(s_abs), h[:n - s_abs]))
    grad = 1.0 - (left + right)
    return s, grad

def gradient_descent_optimize(h0, time_budget):
    """Simple gradient descent on the max‑overlap objective."""
    n = h0.shape[0]
    target = n / 2.0
    h = project_onto_sum_box(h0.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    lr = 0.5
    it = 0
    while time.time() - start < time_budget:
        _, grad = worst_shift_and_gradient(h)
        h = h - lr * grad
        h = project_onto_sum_box(h, target)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        lr *= 0.9995
        it += 1
        if it % 200 == 0:
            # occasional stochastic perturbation to escape plateaus
            h = best_h + np.random.randn(n) * 0.01
            h = project_onto_sum_box(h, target)
            lr = 0.5
    return best_h, best_c5

def binary_from_continuous(h):
    """Return a binary vector with exactly n/2 ones (largest entries set to 1)."""
    n = h.shape[0]
    k = n // 2
    idx = np.argpartition(h, -k)[-k:]
    b = np.zeros_like(h)
    b[idx] = 1.0
    return b

def binary_hill_climb(b0, time_budget):
    """Greedy swap‑based hill climbing for a binary vector."""
    n = b0.shape[0]
    b = b0.copy()
    best_b = b.copy()
    best_c5 = compute_c5(b)
    start = time.time()
    while time.time() - start < time_budget:
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        b[i], b[j] = b[j], b[i]          # propose swap
        c5 = compute_c5(b)
        if c5 < best_c5:
            best_c5 = c5
            best_b = b.copy()
        else:
            b[i], b[j] = b[j], b[i]      # revert
    return best_b, best_c5

def upsample_vector(v, n_target):
    """Linear interpolation of a vector to a larger grid."""
    n_src = v.shape[0]
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_tgt = np.linspace(0.0, 2.0, n_target, endpoint=False)
    return np.interp(x_tgt, x_src, v)
