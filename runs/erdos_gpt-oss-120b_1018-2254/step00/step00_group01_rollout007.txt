import numpy as np
import time

def _project_feasible(x, target_sum, tol=1e-9, max_iter=200):
    """
    Project vector x onto the feasible set:
    0 <= x_i <= 1 and sum(x) = target_sum.
    """
    x = np.clip(x, 0.0, 1.0)
    for _ in range(max_iter):
        s = x.sum()
        diff = target_sum - s
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = x < 1.0
        else:
            free = x > 0.0
        idx = np.where(free)[0]
        if idx.size == 0:
            break
        delta = diff / idx.size
        x[idx] += delta
        x = np.clip(x, 0.0, 1.0)
    # final tiny correction if needed
    s = x.sum()
    if abs(s - target_sum) > 1e-6:
        # distribute remaining error uniformly (might slightly violate bounds)
        correction = (target_sum - s) / x.size
        x += correction
        x = np.clip(x, 0.0, 1.0)
    return x

def _random_feasible(n, target_sum):
    """Generate a random feasible vector of length n."""
    x = np.random.rand(n)
    return _project_feasible(x, target_sum)

def _compute_c5(h, n):
    """Return the overlap value C5 for a given step function."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] with values in [0,1] and integral 1
    that minimizes C5 = max_k ∫ h(x)(1-h(x+k)) dx.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)

    # Determine dimension and initial vector
    if 'initial_h_values' in globals():
        init_arr = np.asarray(initial_h_values, dtype=float)
        n = init_arr.size
        target = n / 2.0
        curr_h = _project_feasible(init_arr, target)
    else:
        n = 47
        target = n / 2.0
        curr_h = _random_feasible(n, target)

    # Initialise best solution
    best_h = curr_h.copy()
    best_c5 = _compute_c5(best_h, n)

    # Simulated annealing state
    curr_c5 = best_c5
    start_time = time.time()
    no_improve = 0
    # early‑exit target (slightly better than the current record)
    early_target = 0.3808

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.97:
            break
        if best_c5 <= early_target:
            break

        frac = elapsed / budget_s
        # Temperature and step size slowly decay
        temp = max(0.001, 0.04 * (1.0 - frac) + 0.001)   # ~0.041 → 0.001
        step = max(0.02, 0.5 * (1.0 - frac) + 0.02)     # ~0.52 → 0.02

        # Propose a new candidate
        if np.random.rand() < 0.001:   # occasional full restart
            cand = _random_feasible(n, target)
        else:
            noise = (np.random.rand(n) - 0.5) * 2.0 * step
            cand = curr_h + noise
        cand = _project_feasible(cand, target)

        cand_c5 = _compute_c5(cand, n)

        # Accept according to simulated‑annealing criterion
        if cand_c5 < curr_c5 or np.random.rand() < np.exp(-(cand_c5 - curr_c5) / temp):
            curr_h = cand
            curr_c5 = cand_c5

        # Update global best
        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = cand.copy()
            no_improve = 0
        else:
            no_improve += 1

        # If stuck for a while, jump back to the current best
        if no_improve >= 5000:
            curr_h = best_h.copy()
            curr_c5 = best_c5
            no_improve = 0

    return (best_h, float(best_c5), n)
