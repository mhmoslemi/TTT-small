def project_onto_capped_simplex(v, s, rng=None):
    # v: np.array, shape (n,)
    # s: target sum (float)
    # returns projection onto [0,1] hyperplane
    n = v.size
    # Initially, clip to [0,1] to get feasible values; compute sum:
    x = np.clip(v, 0, 1)
    sum_x = x.sum()
    # If sum_x == s, done.
    if abs(sum_x - s) < 1e-12:
        return x
    # If sum_x > s: need to reduce sum, can via "water-filling" negative lambda
    # If sum_x < s: need to increase sum, positive lambda.
    # We'll use lambda search:
    # Use binary search on lambda such that sum_i clip(v_i + lambda, 0, 1) = s.
    lo = -1.0
    hi = 1.0
    # Expand bounds as needed:
    while np.sum(np.clip(v + hi, 0, 1)) < s:
        hi *= 2
        if hi > 1e6:
            break
    while np.sum(np.clip(v + lo, 0, 1)) > s:
        lo *= 2
        if lo < -1e6:
            break
    for _ in range(100):
        mid = (lo + hi) / 2.0
        xmid = np.clip(v + mid, 0, 1)
        sum_mid = xmid.sum()
        if sum_mid < s:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo+hi)/2.0, 0, 1)

def project_onto_capped_simplex(v, s):
    lo = -1e6
    hi = 1e6
    # binary search for lambda such that sum_i clip(v_i+lambda,0,1)=s

        else:
