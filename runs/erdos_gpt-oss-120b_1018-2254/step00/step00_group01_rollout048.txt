        hi, hj = current_h[i], current_h[j]
        max_plus = min(1.0 - hi, hj)   # positive delta allowed
        max_minus = min(hi, 1.0 - hj)  # negative delta magnitude allowed
        if max_plus == 0 and max_minus == 0:
            continue

        raw_delta = (rng.random() * 2 - 1) * step
        delta = np.clip(raw_delta, -max_minus, max_plus)
        if abs(delta) < 1e-12:
            continue

        new_h = current_h.copy()
        new_h[i] = hi + delta
        new_h[j] = hj - delta
        new_h[i] = min(max(new_h[i], 0.0), 1.0)
        new_h[j] = min(max(new_h[j], 0.0), 1.0)

        new_c5 = compute_c5(new_h, n)

        # Acceptance test
        if new_c5 < current_c5:
            accept = True
        else:
            prob = math.exp(-(new_c5 - current_c5) / T) if T > 0 else 0.0
            accept = rng.random() < prob

        if accept:
            current_h = new_h
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
                last_improve = it

        # If stuck for a long while, perturb from the best solution
        if it - last_improve > 200_000:
            current_h = best_h.copy()
            current_c5 = best_c5
            for _ in range(5):
                i2, j2 = rng.choice(n, size=2, replace=False)
                hi2, hj2 = current_h[i2], current_h[j2]
                max_plus2 = min(1.0 - hi2, hj2)
                max_minus2 = min(hi2, 1.0 - hj2)
                if max_plus2 == 0 and max_minus2 == 0:
                    continue
                d2 = (rng.random() * 2 - 1) * 0.1
                d2 = np.clip(d2, -max_minus2, max_plus2)
                current_h[i2] = hi2 + d2
                current_h[j2] = hj2 - d2
                current_h[i2] = min(max(current_h[i2], 0.0), 1.0)
                current_h[j2] = min(max(current_h[j2], 0.0), 1.0)
            current_c5 = compute_c5(current_h, n)
            last_improve = it

    return (best_h, best_c5, n)
