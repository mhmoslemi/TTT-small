import time
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return the C5 value for a discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


def _proj_box_sum(v: np.ndarray, target_sum: float, iters: int = 60) -> np.ndarray:
    """
    Euclidean projection of `v` onto the set {x | 0<=x_i<=1, sum(x)=target_sum}.
    Uses a simple bisection on the threshold tau such that
    x_i = clip(v_i - tau, 0, 1) has the required sum.
    """
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(iters):
        tau = (lo + hi) * 0.5
        x = np.clip(v - tau, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = tau
        else:
            hi = tau
    tau = (lo + hi) * 0.5
    x = np.clip(v - tau, 0.0, 1.0)
    return x


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a step function on [0,2] that lowers the C5 overlap constant."""
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------ #
    # initialise -------------------------------------------------------- #
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        # fallback: 45 points, half 1's half 0's (shuffled)
        n0 = 45
        h0 = np.zeros(n0, dtype=float)
        h0[: n0 // 2] = 1.0
        rng.shuffle(h0)

    n = h0.shape[0]
    target = n / 2.0
    h0 = _proj_box_sum(h0, target)

    cur_h = h0.copy()
    cur_val = _compute_c5(cur_h)

    best_h = cur_h.copy()
    best_val = cur_val

    # ------------------------------------------------------------------ #
    # hill‑climbing on the simplex -------------------------------------- #
    step = 0.20                # maximal amount transferred in one move
    iters_without_improve = 0
    total_iters = 0

    while True:
        # time guard
        if time.time() - start > budget_s * 0.95:
            break

        # pick two distinct indices
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        hi = cur_h[i]
        hj = cur_h[j]

        # feasible delta range for moving mass from i to j
        lower = max(-hj, hi - 1.0)
        upper = min(hi, 1.0 - hj)

        # limit by current step size
        lower = max(lower, -step)
        upper = min(upper, step)

        if lower >= upper:
            continue

        delta = rng.uniform(lower, upper)

        cand = cur_h.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_val = _compute_c5(cand)

        if cand_val < cur_val - 1e-14:
            cur_h = cand
            cur_val = cand_val
            iters_without_improve = 0
            if cand_val < best_val:
                best_h = cand.copy()
                best_val = cand_val
        else:
            iters_without_improve += 1

        total_iters += 1

        # scheduled decay of step size
        if total_iters % 2000 == 0:
            step *= 0.95
            if iters_without_improve > 5000:
                cur_h = best_h.copy()
                cur_val = best_val
                iters_without_improve = 0

    best_h = _proj_box_sum(best_h, target)
