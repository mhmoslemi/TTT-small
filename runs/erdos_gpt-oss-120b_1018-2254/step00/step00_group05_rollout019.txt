import numpy as np, time, math

def c5_of_h(h):
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode='full')
    return float(conv.max() * dx)

def gradient_for_shift(h, shift):
    n = h.shape[0]
    g = np.ones(n, dtype=float)
    idx = np.arange(n)
    idx1 = idx + shift
    mask1 = (idx1 >= 0) & (idx1 < n)
    g[mask1] -= h[idx1[mask1]]
    idx2 = idx - shift
    mask2 = (idx2 >= 0) & (idx2 < n)
    g[mask2] -= h[idx2[mask2]]
    return g

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    # initial construction
    try:
        init_h = np.array(initial_h_values, dtype=float)
        n = init_h.size
        target_sum = n / 2.0
        cur_sum = init_h.sum()
        if abs(cur_sum - target_sum) > 1e-12:
            scale = target_sum / cur_sum
            init_h = np.clip(init_h * scale, 0.0, 1.0)
            diff = target_sum - init_h.sum()
            if abs(diff) > 1e-12:
                mask = (init_h > 0) & (init_h < 1)
                if mask.any():
                    init_h[mask] += diff / mask.sum()
                    init_h = np.clip(init_h, 0.0, 1.0)
                else:
                    init_h[0] = np.clip(init_h[0] + diff, 0.0, 1.0)
    except NameError:
        n = 70
        init_h = np.full(n, 0.5, dtype=float)
        target_sum = n / 2.0

    best_h = init_h.copy()
    best_c5 = c5_of_h(best_h)

    cur_h = best_h.copy()
    cur_c5 = best_c5

    T0 = 5e-3        # initial temperature
    T_min = 1e-6     # final temperature

    # Main simulated‑annealing loop
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.95:
            break

        frac = elapsed / (budget_s * 0.95)   # 0..1
        T = T0 * (T_min / T0) ** frac        # exponential cooling

        # max‑transfer fraction shrinks with time
        max_frac = max(0.5 * (1 - frac), 0.01)

        # decide indices and direction
        i = np.random.randint(0, n)
        j = np.random.randint(0, n-1)
        if j >= i:
            j += 1

        # compute worst shift for guided move
        conv = np.correlate(cur_h, 1 - cur_h, mode='full')
        worst_idx = int(np.argmax(conv))
        shift = worst_idx - (n - 1)

        # gradient for that shift
        g = gradient_for_shift(cur_h, shift)

        # pick decreasing/increasing indices based on gradient sign
        pos = np.where(g > 1e-12)[0]
        neg = np.where(g < -1e-12)[0]

        if pos.size and neg.size:
            i = np.random.choice(pos)
            j = np.random.choice(neg)
            max_eps = min(cur_h[i], 1.0 - cur_h[j])
        else:
            # fallback random direction
            if np.random.rand() < 0.5:
                max_eps = min(1.0 - cur_h[i], cur_h[j])
                # i will increase, j will decrease
                i, j = i, j
            else:
                max_eps = min(1.0 - cur_h[j], cur_h[i])
                i, j = j, i

        if max_eps <= 0:
            continue

        eps = max_eps * max_frac * np.random.rand()
        cur_h[i] += eps
        cur_h[j] -= eps

        new_c5 = c5_of_h(cur_h)
        delta = new_c5 - cur_c5

                best_h = cur_h.copy()
        else:

            cur_h[i] -= eps
            cur_h[j] += eps

    # final correction of sum constraint
    target_sum = n / 2.0
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        free = np.where((best_h > 0) & (best_h < 1))[0]

            best_h[0] = np.clip(best_h[0] + diff, 0.0, 1.0)

    best_c5 = c5_of_h(best_h)
    return best_h, best_c5, n

import numpy as np
import time
import math

def c5_of_h(h):
    """Return C₅ for a step function sampled by array h."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode='full')
    return float(conv.max() * dx)

def gradient_for_shift(h, shift):
    """Sub‑gradient of Cₖ (k = shift) w.r.t. h."""
    n = h.shape[0]
    g = np.ones(n, dtype=float)
    idx = np.arange(n)
    i1 = idx + shift
    mask1 = (i1 >= 0) & (i1 < n)
    g[mask1] -= h[i1[mask1]]
    i2 = idx - shift
    mask2 = (i2 >= 0) & (i2 < n)
    g[mask2] -= h[i2[mask2]]
    return g

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    # ----- initial construction ------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)   # provided by harness
        n = init_h.size
        target_sum = n / 2.0
        cur_sum = init_h.sum()
        if abs(cur_sum - target_sum) > 1e-12:
            scale = target_sum / cur_sum
            init_h = np.clip(init_h * scale, 0.0, 1.0)
            diff = target_sum - init_h.sum()
            if abs(diff) > 1e-12:
                mask = (init_h > 0) & (init_h < 1)
                if mask.any():
                    init_h[mask] += diff / mask.sum()
                    init_h = np.clip(init_h, 0.0, 1.0)
                else:
                    init_h[0] = np.clip(init_h[0] + diff, 0.0, 1.0)
    except NameError:
        n = 70
        init_h = np.full(n, 0.5, dtype=float)
        target_sum = n / 2.0

    best_h = init_h.copy()
    best_c5 = c5_of_h(best_h)

    cur_h = best_h.copy()
    cur_c5 = best_c5

    # ----- simulated annealing / guided random walk ----------------------------
    T0 = 5e-3          # start temperature
    T_min = 1e-6       # final temperature
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.95:
            break

        frac = elapsed / (budget_s * 0.95)          # 0 … 1
        T = T0 * (T_min / T0) ** frac               # exponential cooling
        max_frac = max(0.5 * (1 - frac), 0.01)       # max transfer fraction

        # worst shift of the current profile
        conv = np.correlate(cur_h, 1 - cur_h, mode='full')
        worst_idx = int(np.argmax(conv))
        shift = worst_idx - (n - 1)

        # gradient for that shift (guides mass transfer)
        g = gradient_for_shift(cur_h, shift)
        pos = np.where(g > 1e-12)[0]   # decreasing is desirable
        neg = np.where(g < -1e-12)[0]  # increasing is desirable

        # choose indices for a feasible transfer
        if pos.size and neg.size:
            dec = np.random.choice(pos)      # will lose mass
            inc = np.random.choice(neg)      # will gain mass
            max_eps = min(cur_h[dec], 1.0 - cur_h[inc])
            inc_idx, dec_idx = inc, dec
        else:
            # fallback: random pair, random direction
            i = np.random.randint(0, n)
            j = np.random.randint(0, n - 1)
            if j >= i:
                j += 1
            if np.random.rand() < 0.5:
                inc_idx, dec_idx = i, j
            else:
                inc_idx, dec_idx = j, i
            max_eps = min(1.0 - cur_h[inc_idx], cur_h[dec_idx])

        if max_eps <= 0:
            continue

        eps = max_eps * max_frac * np.random.rand()
        cur_h[inc_idx] += eps
        cur_h[dec_idx] -= eps

        if delta < 0 or np.random.rand() < math.exp(-delta / (T + 1e-15)):

        else:
            # reject move
            cur_h[inc_idx] -= eps
            cur_h[dec_idx] += eps

    # ----- final feasibility correction ----------------------------------------
    target_sum = n / 2.0
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        free = np.where((best_h > 0) & (best_h < 1))[0]
        if free.size:
            best_h[free[0]] += diff
            best_h[free[0]] = np.clip(best_h[free[0]], 0.0, 1.0)
        else:
            best_h[0] = np.clip(best_h[0] + diff, 0.0, 1.0)

    best_c5 = c5_of_h(best_h)
    return best_h, best_c5, n
