def neighbor_move(h, step_max):
    n = h.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    hi = h[i]
    hj = h[j]
    # Compute feasible delta range
    low = -min(hi, 1 - hj)
    high = min(1 - hi, hj)
    # If both zero wide, just return same h (skip).
    if high <= low:
        return h.copy()
    # limit delta magnitude by step_max
    delta_low = max(low, -step_max)
    delta_high = min(high, step_max)
    if delta_low >= delta_high:
        # tough but maybe use small delta at center
        delta = 0.0
    else:
        delta = np.random.uniform(delta_low, delta_high)
    h_new = h.copy()
    h_new[i] = hi + delta
    h_new[j] = hj - delta
    return h_new

def project_onto_bounds_and_sum(x, target, tol=1e-12, max_iter=100):
    low = np.min(x) - 1.0
    high = np.max(x) + 1.0

import time
import math
import numpy as np

# ---------- Helper functions ----------

def _project_bounds_sum(x: np.ndarray, target: float, tol: float = 1e-12, max_iter: int = 100) -> np.ndarray:
    """
    Project vector x onto the feasible set:
        0 <= y_i <= 1  for all i,
        sum(y) == target.
    Uses a binary search on the Lagrange multiplier.
    """
    low = np.min(x) - 1.0   # ensures all entries become 1 after clipping
    high = np.max(x) + 1.0  # ensures all entries become 0 after clipping
    for _ in range(max_iter):
        lam = (low + high) * 0.5
        y = np.clip(x - lam, 0.0, 1.0)
        s = y.sum()
        if abs(s - target) <= tol:
            return y
        if s > target:
            low = lam       # need larger lambda to reduce sum
        else:
            high = lam
    # final approximation
    y = np.clip(x - (low + high) * 0.5, 0.0, 1.0)
    return y


def _random_feasible(n: int, target: float) -> np.ndarray:
    """Generate a random vector in [0,1]^n with prescribed sum `target`."""
    x = np.random.rand(n)
    return _project_bounds_sum(x, target)


def _neighbor_move(h: np.ndarray, step_max: float) -> np.ndarray:
    """
    Produce a feasible neighbor by moving a mass delta between two coordinates.
    The total sum stays unchanged and bounds are preserved.
    """
    n = h.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    hi, hj = h[i], h[j]

    # allowable delta that keeps both entries in [0,1]
    delta_low = -min(hi, 1.0 - hj)   # transfer from i to j
    delta_high = min(1.0 - hi, hj)   # transfer from j to i

    # enforce step size limits
    delta_low = max(delta_low, -step_max)
    delta_high = min(delta_high, step_max)

    if delta_low >= delta_high:
        return h.copy()

    delta = np.random.uniform(delta_low, delta_high)
    h_new = h.copy()
    h_new[i] = hi + delta
    h_new[j] = hj - delta
    return h_new


def _c5_value(h: np.ndarray) -> float:
    """Compute the overlap constant C5 for a step function h."""
    # evaluate_erdos_solution is provided by the harness; its first entry is C5.
    return evaluate_erdos_solution(h, h.shape[0])[0]
