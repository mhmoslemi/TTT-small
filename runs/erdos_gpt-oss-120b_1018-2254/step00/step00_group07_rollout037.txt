def project_to_bounds_sum(v, target, lower=0.0, upper=1.0, max_iter=1000, tol=1e-9):
    h = np.clip(v, lower, upper)
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) < tol:
            break
        # Determine which indices can move in direction to reduce diff
        # If s > target, need to decrease some h; choose those not at lower bound.
        # If s < target, need to increase some h; choose those not at upper bound.
        if diff > 0:
            # need to subtract
            # set capacity = h - lower
            cap = h - lower
            can_move = cap > 1e-12
            total_cap = cap[can_move].sum()
            if total_cap <= 0:
                break
            delta = diff / total_cap
            # subtract proportionally to capacity
            h[can_move] -= delta * cap[can_move]
        else:
            # diff < 0 -> need to increase
            cap = upper - h
            can_move = cap > 1e-12
            total_cap = cap[can_move].sum()
            if total_cap <= 0:
                break
            delta = -diff / total_cap
            h[can_move] += delta * cap[can_move]
        # Clip again for numerical
        h = np.clip(h, lower, upper)
    return h

import cvxpy as cp

def project_to_bounds_sum(v, target):
    n = len(v)
    h = cp.Variable(n)
    objective = cp.Minimize(cp.sum_squares(h - v))
    constraints = [h >= 0, h <= 1, cp.sum(h) == target]
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True) # maybe default.
    return h.value

def hill_climb(h_start, target_sum, dx, time_limit, rng):
    cur_h = h_start.copy()
    cur_c5 = compute_c5(cur_h, dx)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    n = len(cur_h)
    small_step = 0.03
    large_step = 0.2
    start_time = time.time()
    # Temperature param for simulated annealing
    start_temp = 0.005  # small
    while True:
        elapsed = time.time() - start_time
        if elapsed >= time_limit:
            break
        # random indices
        i, j = rng.choice(n, 2, replace=False)
        step = large_step if rng.random() < 0.1 else small_step
        delta_max = min(step, cur_h[i], 1.0 - cur_h[j])
        if delta_max <= 1e-12:
            continue
        delta = rng.random() * delta_max
        h_new = cur_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        c5_new = compute_c5(h_new, dx)
        # Simulated annealing acceptance
        temp = start_temp * (1 - elapsed / time_limit)
        if c5_new < cur_c5 - 1e-12 or rng.random() < np.exp(-(c5_new - cur_c5) / max(temp, 1e-12)):
            cur_h = h_new
            cur_c5 = c5_new

                best_h = cur_h.copy()
                best_c5 = cur_c5
                # early exit if good enough
                if best_c5 <= 0.3808:
                    break
    return best_h, best_c5

import scipy.optimize as opt

DX = 1.0

def compute_c5(h):
    # uses global DX
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * DX

def project_feasible(h, target, lower=0.0, upper=1.0, max_iter=200, tol=1e-9):
    h = np.clip(h, lower, upper)
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) < tol:
            break
        if diff > 0:
            # need to decrease some values not at lower bound
            free = h > lower + 1e-12
            if not free.any():
                break
            cap = h[free] - lower
            total_cap = cap.sum()
            if total_cap <= 0:
                break
            delta = diff / total_cap
            h[free] -= delta * cap
        else:
            free = h < upper - 1e-12
            if not free.any():
                break
            cap = upper - h[free]
            total_cap = cap.sum()
            if total_cap <= 0:
                break
            delta = -diff / total_cap
            h[free] += delta * cap
        h = np.clip(h, lower, upper)
    h = np.clip(h, lower, upper)
    return h

class SumEquality:
    def __init__(self, target):
        self.target = target
    def __call__(self, h):
        return float(np.sum(h) - self.target)

import time
import numpy as np
import scipy.optimize as opt

# Global variable used by the objective to avoid closures
DX = 1.0  # will be set inside run()


def compute_c5(h):
    """Return max_{k} ∫ h(x)*(1-h(x+k)) dx approximated on the grid."""
    # uses the global DX as the step size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * DX


def project_feasible(h, target, lower=0.0, upper=1.0, max_iter=200, tol=1e-9):
    """
    Euclidean projection of h onto the intersection of the box [lower,upper]^n
    and the hyperplane sum(h)=target.
    """
    h = np.clip(h, lower, upper)
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) < tol:
            break
        if diff > 0:
            free = h > lower + 1e-12
            if not free.any():
                break
            cap = h[free] - lower
            total = cap.sum()
            if total <= 0:
                break
            delta = diff / total
            h[free] -= delta * cap
        else:
            free = h < upper - 1e-12
            if not free.any():
                break
            cap = upper - h[free]
            total = cap.sum()
            if total <= 0:
                break
            delta = -diff / total
            h[free] += delta * cap
        h = np.clip(h, lower, upper)
    return np.clip(h, lower, upper)


class SumEquality:
    """Callable for the equality constraint sum(h)=target."""
    def __init__(self, target):
        self.target = target

    def __call__(self, h):
        return float(np.sum(h) - self.target)


def hill_climb(h_start, target, time_limit, rng):
    """
    Simple simulated‑annealing hill‑climber that preserves the sum exactly.
    Moves a small amount of mass from one index to another.
    """
    h = h_start.copy()
    c5 = compute_c5(h)
    best_h, best_c5 = h.copy(), c5
    n = len(h)
    small_step = 0.03
    large_step = 0.20
    start = time.time()
    start_temp = 0.005  # annealing temperature at time 0

    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        i, j = rng.choice(n, 2, replace=False)
        step = large_step if rng.random() < 0.1 else small_step
        delta_max = min(step, h[i], 1.0 - h[j])
        if delta_max <= 0:
            continue
        delta = rng.random() * delta_max
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        c5_new = compute_c5(h_new)

        # temperature linearly decays to 0
        temp = start_temp * (1.0 - elapsed / time_limit)
        accept = False
        if c5_new < c5 - 1e-12:
            accept = True
        else:
            if rng.random() < np.exp(-(c5_new - c5) / max(temp, 1e-12)):
                accept = True
        if accept:
            h, c5 = h_new, c5_new
            if c5 < best_c5:
                best_h, best_c5 = h.copy(), c5
                if best_c5 <= 0.3808:
                    return best_h, best_c5
    return best_h, best_c5


def slsqp_optimize(h_start, target, max_iter=200):
    """
    Refine a feasible vector with SLSQP while respecting the box and sum constraints.
    """
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    constraints = {'type': 'eq', 'fun': SumEquality(target)}
    res = opt.minimize(
        compute_c5,
        h_start,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options={"maxiter": max_iter, "ftol": 1e-9, "disp": False},
    )
    if res.success:
        h_opt = project_feasible(res.x, target)
        return h_opt, compute_c5(h_opt)
    # fallback – return the starting point unchanged
    return h_start, compute_c5(h_start)


def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points) where:
      - h_values is a list of length n_points containing the step function values,
      - c5_bound is the achieved upper bound for C₅,
      - n_points is the number of discretisation points.
    """
    global DX
    rng = np.random.default_rng(seed)

    # Determine the problem size and an initial feasible vector
    try:
        init_h = initial_h_values.astype(float)  # provided by the harness
        n = int(init_h.shape[0])
    except Exception:  # pragma: no cover
        n = 80
        init_h = np.zeros(n)
        init_h[: n // 2] = 1.0

    target_sum = n / 2.0
    DX = 2.0 / n  # step size for the integral approximation

    # Ensure feasibility of the starting vector
    init_h = project_feasible(init_h, target_sum)
    best_h = init_h.copy()
    best_c5 = compute_c5(best_h)

    overall_start = time.time()

    # ---- Phase 1: stochastic hill climbing ------------------------------
    time_left = budget_s - (time.time() - overall_start)
    if time_left > 0:
        hill_time = time_left * 0.70
        best_h, best_c5 = hill_climb(best_h, target_sum, hill_time, rng)

    # ---- Phase 2: SLSQP refinements ------------------------------------
    while time.time() - overall_start < budget_s * 0.95:
        # Produce a new random feasible start point
        rnd = rng.random(n)
        rnd = project_feasible(rnd, target_sum)

        h_opt, c5_opt = slsqp_optimize(rnd, target_sum, max_iter=200)

        if c5_opt + 1e-12 < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            if best_c5 <= 0.3808:
                break

    # Final safety projection (should already be feasible)
    best_h = project_feasible(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return (best_h.tolist(), float(best_c5), n)
