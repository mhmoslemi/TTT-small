def _c5_of(h):
    """Compute C₅ = max_k ∫ h(x)(1−h(x+k))dx using the prescribed discrete formula."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx
