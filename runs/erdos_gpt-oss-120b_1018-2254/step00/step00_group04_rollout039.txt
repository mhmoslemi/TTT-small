    s = h.sum()

def compute_c5(h):
    n = len(h)
    # pad to length 2*n-1 for 'full' convolution
    nfft = 2**int(np.ceil(np.log2(2*n-1)))
    # compute FFT of h and reversed (1-h)
    H = np.fft.rfft(h, n=nfft)
    rev = (1 - h)[::-1]
    R = np.fft.rfft(rev, n=nfft)
    conv = np.fft.irfft(H * R, n=nfft)[:2*n-1]  # length full
    # conv is correlation (full)
    return np.max(conv) * (2.0 / n)

            init = np.asarray(initial_h_values, dtype=float)
        else:
            if 'initial_h_values' in globals():
                old = np.asarray(initial_h_values, dtype=float)
                xs_old = np.linspace(0.0, 2.0, len(old), endpoint=False)
                xs_new = np.linspace(0.0, 2.0, n, endpoint=False)
                init = np.interp(xs_new, xs_old, old)
            else:
                init = np.random.rand(n)
        init = repair_h(init)

        # bounds for DE (box constraints)
        bounds = [(0.0, 1.0)] * n

        # iteration budget depending on size
        if n <= 50:
            maxiter = 200
        elif n <= 80:
            maxiter = 150
        else:
            maxiter = 100

        # run global search
        try:
            result = differential_evolution(
                objective_func,
                bounds,
                maxiter=maxiter,
                popsize=15,
                mutation=(0.5, 1),
                recombination=0.7,
                seed=seed,
                polish=True,
                init='latinhypercube',
                callback=de_callback,
                disp=False,
                tol=1e-6,
            )
        except Exception:
            result = None

        # evaluate DE solution
        if result is not None and getattr(result, 'success', False):
            cand = repair_h(result.x)
            cand_val = compute_c5(cand)
            if cand_val < best_val:
                best_val, best_h, best_n = cand_val, cand, n

            # refine with local search if time permits
            remaining = budget_s - (time.time() - start_time)
            if remaining > 5.0:
                ls_time = min(remaining * 0.3, 20.0)
                cand2, cand2_val = local_search(cand, ls_time)
                if cand2_val < best_val:
                    best_val, best_h, best_n = cand2_val, cand2, n

        # also consider the initial feasible guess itself
        guess_val = compute_c5(init)
        if guess_val < best_val:
            best_val, best_h, best_n = guess_val, init, n

    # fallback safety
    if best_h is None:
        n = 42
        best_h = np.full(n, 0.5)
        best_h = repair_h(best_h)
        best_val = compute_c5(best_h)
        best_n = n

    return best_h, best_val, best_n
