def project_to_feasible(x, sum_target):
    # clip to [0,1] then apply correction iteratively
    x = np.clip(x, 0, 1)
    for _ in range(10):
        diff = sum_target - x.sum()
        if abs(diff) < 1e-8:
            break
        # distribute uniformly to all entries not at bounds
        free = (x > 0) & (x < 1)
        n_free = np.count_nonzero(free)
        if n_free == 0:
            # can't adjust further; break
            break
        delta = diff / n_free
        x[free] += delta
        # clip again
        x = np.clip(x, 0, 1)
    # final adjustment: if still not exactly sum_target due to rounding, adjust by adding diff to random free index
    diff = sum_target - x.sum()
    if abs(diff) > 1e-8:
        # Add diff to random index
        idx = np.random.choice(np.arange(len(x)))
        x[idx] = np.clip(x[idx] + diff, 0, 1)
    # Ensure final sum matches within tolerance
    return x
```

import numpy as np
import time
from scipy.optimize import minimize

def project_to_feasible(x, sum_target):
    x = np.clip(x, 0.0, 1.0)
    for _ in range(10):
        diff = sum_target - x.sum()
        if abs(diff) < 1e-10:
            break
        free = (x > 0) & (x < 1)
        n_free = np.count_nonzero(free)
        if n_free == 0:
            break
        delta = diff / n_free
        x[free] += delta
        # Clip again
        x = np.clip(x, 0.0, 1.0)
    # Final small fix
    diff = sum_target - x.sum()
    if abs(diff) > 1e-8:
        # adjust a random free element
        idxs = np.where((x > 0) & (x < 1))[0]
        if idxs.size == 0:
            idxs = np.arange(len(x))
        idx = np.random.choice(idxs)
        x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
    # Ensure final sum close
    # Might slightly be off due to rounding; we accept tolerance
    return x

def evaluate_h(h):
    # Evaluate C5 using pre-imported function
    return evaluate_erdos_solution(h)

def hill_climb(start_h, budget_s, start_time):
    h = start_h.copy()
    best_val = evaluate_h(h)
    n = len(h)
    while time.time() - start_time < budget_s * 0.95:
