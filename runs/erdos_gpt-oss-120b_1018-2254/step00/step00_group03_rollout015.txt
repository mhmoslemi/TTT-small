        if ones > target_ones:
            # Too many ones: randomly turn some ones to zero

            to_one = np.random.choice(idx_zeros, size=target_ones - ones, replace=False)
            binary[to_one] = 1
        return binary
    else:
        return binary_initial(n)

    cur_h = start_h.copy()
    cur_val = evaluate(cur_h)

    best_val = cur_val

    # Temperature schedule: start temperature = 0.1? maybe based on current_val
    T0 = 0.05  # could be based on cur_val * 0.2
    # Determine total allowed iterations based on time
    # while time elapsed < max_time:
    while True:
        now = time.time()
        elapsed = now - start_time
        if elapsed >= max_time:
            break
        # propose swap
        # compute indices of ones and zeros each time
        ones_idx = np.where(cur_h > 0.5)[0]
        zeros_idx = np.where(cur_h < 0.5)[0]

        i = np.random.choice(ones_idx)

        # swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_val = evaluate(cur_h)
        delta = new_val - cur_val

            # Keep track best

                best_h = cur_h.copy()

            # revert swap
            cur_h[i] = 1.0
            cur_h[j] = 0.0

import time, numpy as np, math, random

def evaluate(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_binary_to_target(arr, target_ones):
    # arr binary or not; adjust to exactly target_ones ones
    # ensures values 0/1
    arr = np.where(arr >= 0.5, 1.0, 0.0)
    ones = int(arr.sum())
    if ones > target_ones:
        idx = np.where(arr == 1)[0]
        to_zero = np.random.choice(idx, size=ones - target_ones, replace=False)
        arr[to_zero] = 0.0
    elif ones < target_ones:
        idx = np.where(arr == 0)[0]
        to_one = np.random.choice(idx, size=target_ones - ones, replace=False)
        arr[to_one] = 1.0
    return arr

def initial_binary(n):
    # Use initial_h_values if available for n
    if 'initial_h_values' in globals():
        try:
            h0 = np.array(initial_h_values, copy=True)
        except Exception:
            h0 = None
        if h0 is not None and len(h0) == n:
            # threshold and adjust
            target = n // 2
            arr = project_binary_to_target(h0, target)
            return arr
    # fallback random
    arr = np.zeros(n, dtype=float)
    arr[:n//2] = 1.0
    np.random.shuffle(arr)
    return arr

def anneal(start_h, max_time, start_time):
    n = len(start_h)
    cur_h = start_h.copy()
    cur_val = evaluate(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val
    # initial temperature as a fraction of best_val or fixed
    T0 = 0.05  # may adapt
    # Quick precompute indices list for speed, but can compute each iteration
    while True:
        now = time.time()
        elapsed = now - start_time
        if elapsed >= max_time:
            break
        # propose swap
        ones = np.where(cur_h > 0.5)[0]
        zeros = np.where(cur_h < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_val = evaluate(cur_h)
        delta = new_val - cur_val
        # temperature linearly decreasing
        progress = elapsed / max_time
        T = T0 * (1.0 - progress)
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 1e-9:
                accept = np.random.rand() < math.exp(-delta / T)
        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] = 1.0
            cur_h[j] = 0.0
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, random, math
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    best_overall_h = None
    best_overall_val = float('inf')
    best_n = None
    target_val = 0.3808
    n_list = [80, 70, 60, 50, 40]
    # allocate equal time
    remaining_budget = budget_s
    # We'll allocate dynamic per n, ensuring we don't exceed total budget
    for idx, n in enumerate(n_list):
        # compute remaining time budget
        time_now = time.time()
        elapsed_total = time_now - start_time
        remaining_time = max(0.0, budget_s - elapsed_total)
        if remaining_time <= 0:
            break
        # allocate a fraction of remaining time: evenly distribute among remaining n's
        n_remaining = len(n_list) - idx
        time_alloc = remaining_time / n_remaining
        # Run multiple restarts within allocated time
        # We'll loop until time for this n is used
        n_start_time = time.time()
        while True:
            # check time left for this n
            t_now = time.time()
            elapsed_n = t_now - n_start_time
            if elapsed_n >= time_alloc:
                break
            # generate initial binary
            init_h = initial_binary(n)
            # allocate inner time for anneal: maybe half of remaining for n
            inner_max_time = min(time_alloc - elapsed_n, time_alloc * 0.9)
            # run anneal
            h_candidate, val_candidate = anneal(init_h, inner_max_time, time.time())
            # Update best overall
            if val_candidate < best_overall_val:
                best_overall_val = val_candidate
                best_overall_h = h_candidate.copy()
                best_n = n
                # early exit if target met
                if best_overall_val <= target_val:
                    # return immediate
                    return (best_overall_h, best_overall_val, best_n)
            # possible break condition: if inner anneal didn't improve for many restarts
            # omitted for simplicity
        # continue next n
    # After loop, return best found
    # Ensure best values exist
    if best_overall_h is None:
        # fallback: return simple uniform
        n = n_list[0]
        h = np.zeros(n, dtype=float)
        h[:n//2] = 1.0
        return (h, evaluate(h), n)
    return (best_overall_h, best_overall_val, best_n)

import time
import math
import random
import numpy as np

def _evaluate(h: np.ndarray) -> float:
    """Return C5 for a step function given by array h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def _project_to_target(arr: np.ndarray, target_ones: int) -> np.ndarray:
    """Force a binary array to have exactly target_ones entries equal to 1."""
    bin_arr = (arr >= 0.5).astype(float)
    cur = int(bin_arr.sum())
    if cur > target_ones:
        idx = np.where(bin_arr == 1)[0]
        to_zero = np.random.choice(idx, cur - target_ones, replace=False)
        bin_arr[to_zero] = 0.0
    elif cur < target_ones:
        idx = np.where(bin_arr == 0)[0]
        to_one = np.random.choice(idx, target_ones - cur, replace=False)
        bin_arr[to_one] = 1.0
    return bin_arr


def _initial_binary(n: int) -> np.ndarray:
    """Create an initial binary vector of length n with exactly n/2 ones."""
    target = n // 2
    # Use the provided construction if it exists and matches the size.
    if "initial_h_values" in globals():
        try:
            h0 = np.array(initial_h_values, copy=True)
            if h0.shape[0] == n:
                return _project_to_target(h0, target)
        except Exception:
            pass
    # Fallback: random balanced vector.
    arr = np.zeros(n, dtype=float)
    arr[:target] = 1.0
    np.random.shuffle(arr)
    return arr


def _anneal(start_h: np.ndarray, time_limit: float) -> tuple:
    """
    Simulated annealing on a binary vector to minimise C5.
    Returns the best vector found and its C5 value.
    """
    deadline = time.time() + time_limit
    cur_h = start_h.copy()
    cur_val = _evaluate(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val

    # Initial temperature proportional to current objective.
    T0 = cur_val * 0.5 if cur_val > 0 else 0.1

    n = cur_h.shape[0]
    while time.time() < deadline:
        # indices of current 1's and 0's
        ones = np.where(cur_h > 0.5)[0]
        zeros = np.where(cur_h < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break

        i = np.random.choice(ones)
        j = np.random.choice(zeros)

        # propose swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_val = _evaluate(cur_h)
        delta = new_val - cur_val

        # linear cooling
        progress = (deadline - time.time()) / time_limit
        T = T0 * progress  # temperature goes to 0 as we approach deadline

        accept = False
        if delta < 0:
            accept = True
        elif T > 1e-12:
            accept = np.random.rand() < math.exp(-delta / T)

        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i] = 1.0
            cur_h[j] = 0.0

    return best_h, best_val


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function with a low Erdős C5 bound."""
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    best_h = None
    best_val = float("inf")
    best_n = None
    target = 0.3808

    # Try a few resolutions, larger resolutions usually give tighter bounds.
    n_list = [80, 70, 60, 50, 40]

    for idx, n in enumerate(n_list):
        elapsed = time.time() - start
        remaining = budget_s - elapsed
        if remaining <= 0:
            break

        # Allocate remaining time equally among the remaining resolutions.
        remaining_res = len(n_list) - idx
        time_for_n = remaining / remaining_res

        # Initialise and optimise.
        init_h = _initial_binary(n)
        h_opt, val_opt = _anneal(init_h, time_for_n)

        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt
            best_n = n

        if best_val <= target:
            break

    # Fallback in the unlikely case nothing was found.
    if best_h is None:
        best_n = n_list[-1]
        best_h = _initial_binary(best_n)
        best_val = _evaluate(best_h)

    # Ensure the returned vector satisfies the constraints exactly.
    # For binary vectors this is guaranteed by construction.
    return (best_h.tolist(), float(best_val), int(best_n))
