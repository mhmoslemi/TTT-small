def random_feasible_h(n, t=None):
    if t is None:
        t = n/2.0
    # Ensure 0 <= t <= n
    h = np.empty(n)
    remaining = t
    for i in range(n-1):
        # lower bound ensure remaining can be filled with remaining entries not exceeding 1 each.
        lo = max(0.0, remaining - (n - i - 1) * 1.0)
        hi = min(1.0, remaining)

        hi = lo + np.random.rand() * (hi - lo) # uniform in [lo, hi]

        val = lo + np.random.rand() * (hi - lo)

        remaining -= val
    h[-1] = remaining
    return h

def random_feasible_h(n, target_sum=None):
    if target_sum is None:
        target_sum = n / 2.0
    h = np.empty(n, dtype=float)
    remaining = target_sum
    for i in range(n-1):
        lo = max(0.0, remaining - (n - i - 1) * 1.0)
        hi = min(1.0, remaining)
        # ensure lo <= hi
        if hi < lo:
            lo, hi = hi, lo
        # sample uniformly in [lo, hi]
        val = lo + np.random.rand() * (hi - lo)
        h[i] = val
        remaining -= val
    h[-1] = remaining
    # Clip just in case numeric extra
    h = np.clip(h, 0.0, 1.0)
    return h

def adjust_to_sum(h, target_sum):
    # Adjust vector h (numpy array) to exactly sum to target_sum within bounds [0,1]
    # Simple iterative projection: clip and distribute difference
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    # tolerance
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            cap = np.sum(1.0 - h[mask])
            if cap <= 0:
                break
            add = diff * (1.0 - h[mask]) / cap
            h[mask] += add
        else:
            mask = h > 0.0
            cap = np.sum(h[mask])
            if cap <= 0:
                break
            sub = diff * (h[mask]) / cap
            h[mask] += sub  # diff negative, so sub negative => decreasing
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
    return h

import numpy as np
import time
import math

def evaluate_c5(h):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k))dx via discrete correlation."""
    func = globals().get('evaluate_erdos_solution')
    if func is not None:
        try:
            return float(func(np.asarray(h, dtype=float)))
        except Exception:
            pass
    h_arr = np.asarray(h, dtype=float)
    n = h_arr.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h_arr, 1.0 - h_arr, mode='full')
    return dx * np.max(corr)


def random_feasible_h(n, target_sum=None):
    """Generate a random vector in [0,1]^n whose entries sum exactly to target_sum."""
    if target_sum is None:
        target_sum = n / 2.0
    h = np.empty(n, dtype=float)
    remaining = target_sum
    for i in range(n - 1):
        lo = max(0.0, remaining - (n - i - 1) * 1.0)
        hi = min(1.0, remaining)
        val = lo + np.random.rand() * (hi - lo)
        h[i] = val
        remaining -= val
    h[-1] = remaining
    return np.clip(h, 0.0, 1.0)


def adjust_to_sum(h, target_sum):
    """Project h onto the box [0,1] while forcing the sum to target_sum."""
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0.0:
            mask = h < 1.0
            cap = np.sum(1.0 - h[mask])
            if cap <= 0.0:
                break
            add = diff * (1.0 - h[mask]) / cap
            h[mask] += add
        else:
            mask = h > 0.0
            cap = np.sum(h[mask])
            if cap <= 0.0:
                break
            sub = diff * (h[mask]) / cap
            h[mask] += sub
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
    return h


def hill_climb_once(h0, max_steps, start_time, budget_s):
    """Simple hill‑climbing that preserves the sum constraint."""
    h = h0.copy()
    n = h.shape[0]
    cur_val = evaluate_c5(h)
    best_h = h.copy()
    best_val = cur_val
    for _ in range(max_steps):
        if time.time() - start_time >= budget_s:
            break
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        # Choose donor (decrease) and receiver (increase)
        if np.random.rand() < 0.5:
            donor, receiver = i, j
        else:
            donor, receiver = j, i
        max_delta = min(h[donor], 1.0 - h[receiver])
        if max_delta <= 0.0:
            continue
        delta = np.random.rand() * max_delta
        h[donor] -= delta
        h[receiver] += delta
        new_val = evaluate_c5(h)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert move
            h[donor] += delta
            h[receiver] -= delta
    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h that gives a low C5 bound."""
    np.random.seed(seed)
    start_time = time.time()
    target_bound = 0.3808

    best_h = None
    best_val = math.inf
    best_n = None

    # Use the provided initial construction if present
    init_h = None
    if 'initial_h_values' in globals():
        init_h = np.array(initial_h_values, dtype=float)
        val0 = evaluate_c5(init_h)
        best_h, best_val, best_n = init_h.copy(), val0, len(init_h)
        if best_val <= target_bound:
            return best_h.tolist(), best_val, best_n

    n_candidates = [45, 90, 180, 360]

    for n in n_candidates:
        if time.time() - start_time >= budget_s:
            break
        target_sum = n / 2.0

        start_vectors = []

        # Interpolated version of the parent construction
        if init_h is not None:
            if len(init_h) == n:
                start_vectors.append(init_h.copy())
            else:
                x_old = np.linspace(0.0, 2.0, len(init_h))
                x_new = np.linspace(0.0, 2.0, n)
                interp = np.interp(x_new, x_old, init_h)
                interp = adjust_to_sum(interp, target_sum)
                start_vectors.append(interp)

        # Uniform constant 0.5 (feasible by construction)
        start_vectors.append(np.full(n, 0.5, dtype=float))

        # A few random feasible starts
        for _ in range(2):
            start_vectors.append(random_feasible_h(n, target_sum))

        # Determine step budget per start (roughly scaled with n)
        max_steps = 2500 if n <= 90 else 1500

        for vec in start_vectors:
            if time.time() - start_time >= budget_s:
                break
            # Ensure sum constraint exactly
            vec = adjust_to_sum(vec, target_sum)

            local_h, local_val = hill_climb_once(vec, max_steps, start_time, budget_s)

            if local_val < best_val:
                best_h, best_val, best_n = local_h.copy(), local_val, n
                if best_val <= target_bound:
                    return best_h.tolist(), best_val, best_n

    # Fallback in the unlikely event no solution was found
    if best_h is None:
        n = n_candidates[0]
        best_h = random_feasible_h(n, n / 2.0)
        best_val = evaluate_c5(best_h)
        best_n = n

    return best_h.tolist(), best_val, best_n
