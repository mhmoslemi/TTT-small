def project_onto_bounds_and_sum(v, S):
    lo = -2.0
    hi = 2.0
    # Expand bounds if needed
    while np.sum(np.clip(v - lo, 0, 1)) < S:
        lo *= 2
    while np.sum(np.clip(v - hi, 0, 1)) > S:
        hi *= 2
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0, 1)
        s = x.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0, 1)

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx, corr

def run(seed=42, budget_s=1100, **kwargs):

    np.random.seed(seed)

    # Determine initial vector

        dx = 2.0 / n
        xs = (np.arange(n) + 0.5) * dx
        h = np.sin(np.pi * xs / 2.0) ** 2
    target_sum = n / 2.0
    h = project_onto_bounds_and_sum(h, target_sum)
    dx = 2.0 / n

    # initial evaluation
    cur_score, _ = compute_c5(h, dx)
    best_h = h.copy()
    best_score = cur_score

    start = time.time()
    elapsed = 0.0

    step = 0.5
    min_step = 1e-5
    max_step = 1.0
    patience = 5000
    no_improve = 0

    while elapsed < budget_s - 0.5:
        # compute worst shift index
        corr_vec = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr_vec)
        k = max_idx - (n - 1)  # shift
        # subgradient of the max term
        g = np.zeros_like(h)
        i_start = max(0, -k)
        i_end = n - max(0, k)

def project_onto_bounds_and_sum(v, S):
    lo = -2.0

    # expand bounds if needed
    while np.sum(np.clip(v - lo, 0, 1)) < S:

        hi *= 2.0

        else:
