                np.clip(pert, 0.0, 1.0, out=pert)
                if abs(pert.sum() - target_sum) > 1e-9:
                    pert *= (target_sum / pert.sum())
                    np.clip(pert, 0.0, 1.0, out=pert)
            current_h = pert
            current_score = compute_c5(current_h)
            it_since_best = 0

    # ---- final tidy‑up -------------------------------------------------------
    final_h = best_h.copy()
    diff = target_sum - final_h.sum()
    if abs(diff) > 1e-12:
        free = np.where((final_h > 0) & (final_h < 1))[0]
        if free.size:
            final_h[free[0]] = np.clip(final_h[free[0]] + diff, 0.0, 1.0)
        else:
            final_h[0] = np.clip(final_h[0] + diff, 0.0, 1.0)
        np.clip(final_h, 0.0, 1.0, out=final_h)

    final_score = compute_c5(final_h)

    return (final_h.tolist(), float(final_score), n_points)
