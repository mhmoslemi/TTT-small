def enforce_constraints(h):
    n = len(h)
    target = n/2
    # Clip
    h = np.clip(h, 0.0, 1.0)
    sum_h = h.sum()
    diff = target - sum_h
    # while diff not small
    eps = 1e-12
    max_iter = 10
    for _ in range(max_iter):
        if abs(diff) < eps:
            break
        if diff > 0:
            cap = 1.0 - h
            total_cap = cap.sum()
            if total_cap < 1e-12:
                break
            # allocate proportionally
            inc = np.minimum(cap, cap * diff / total_cap)  # this is just cap * diff/total_cap; ensures sum inc = diff
            inc = cap * diff / total_cap
            h += inc
        else:
            cap = h
            total_cap = cap.sum()
            if total_cap < 1e-12:
                break
            dec = cap * diff / total_cap  # diff negative
            h += dec
        sum_h = h.sum()
        diff = target - sum_h
    # final clip for safety
    h = np.clip(h, 0.0, 1.0)
    # In case still off due to rounding, adjust a random index
    sum_h = h.sum()
    if abs(sum_h - target) > 1e-8:
        # adjust first entry within bounds
        i = 0
        delta = target - sum_h
        # safe adjust
        hi = h[i]
        new_val = np.clip(hi + delta, 0.0, 1.0)
        h[i] = new_val
    return h

import numpy as np
import time
import random
import math

def _compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)*(1-h(x+k)) dx for discrete h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _enforce_constraints(h: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and adjust to exact sum = n/2 while respecting bounds."""
    n = h.shape[0]
    target = n / 2.0
    # Clip first
    h = np.clip(h, 0.0, 1.0)
    # Adjust sum by distributing the deficit/surplus proportionally
    for _ in range(10):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            cap = 1.0 - h
            total = cap.sum()
            if total <= 0:
                break
            inc = cap * (diff / total)
            h += inc
        else:
            cap = h
            total = cap.sum()
            if total <= 0:
                break
            dec = cap * (diff / total)  # diff negative
            h += dec
    # Final safety clipping
    h = np.clip(h, 0.0, 1.0)
    # If still off by a tiny amount, fix first element
    diff = target - h.sum()
    if abs(diff) > 1e-8:
        i = 0
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) improving the Erdős C5 bound."""
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()

    # ---- initial construction ------------------------------------------------

    except NameError:

        n_base = 45
        base_h = np.full(n_base, 0.5)

    # Choose a working resolution (multiple of the base size for easy interpolation)
    n_points = max(90, n_base * 2)          # 90 is a comfortable small size
    # Interpolate base solution onto the new grid
    x_old = np.linspace(0.0, 2.0, n_base, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
    h_cur = np.interp(x_new, x_old, base_h)
    h_cur = _enforce_constraints(h_cur)

    # ---- keep the best seen --------------------------------------------------
    best_h = h_cur.copy()
    best_score = _compute_c5(best_h)

    # Current state for the annealing walk
    cur_h = h_cur.copy()
    cur_score = best_score

    # ---- Simulated annealing parameters --------------------------------------
    init_T = 0.12                     # initial temperature
    min_T = 1e-5
    iteration = 0
    # Main optimisation loop
    while time.time() - start_time < budget_s * 0.98:
        elapsed = time.time() - start_time
        # Linear cooling schedule
        T = max(min_T, init_T * (1.0 - elapsed / budget_s))

        # Propose a move that preserves the sum exactly
        i, j = np.random.randint(0, n_points, size=2)
        if i == j:
            continue

        # Determine feasible delta range for h[i] += delta, h[j] -= delta
        lower = -min(cur_h[i], 1.0 - cur_h[j])   # most negative delta allowed

        if upper <= lower:

        raw_delta = np.random.uniform(lower, upper)
