def project_to_bounds_sum(y, lower=0., upper=1., target_sum=None):
    if target_sum is None:
        target_sum = np.sum(y)
    # binary search λ
    lo = -1e6
    hi = 1e6
    # but we can refine: lo = min(y - upper) ; hi = max(y - lower)
    lo = np.min(y - upper) - 1.0

def project_to_bounds_sum(v, target_sum, lower=0.0, upper=1.0):
    # binary search for lambda (shift) to meet sum constraint after clipping.
    # v is a numpy array.
    lo = np.min(v - upper) - 1.0
    hi = np.max(v - lower) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, lower, upper)
        s = np.sum(x)
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, lower, upper)

import time
import math
import numpy as np

def evaluate_c5(h):
    """Return the discretized C5 value for a step function h."""
    n = h.shape[0]
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_to_bounds_sum(v, target_sum, lower=0.0, upper=1.0):
    """
    Euclidean projection of vector v onto the box [lower,upper]^n
    intersected with the hyperplane sum(x)=target_sum.
    Implemented via a binary search for the Lagrange multiplier.
    """
    lo = np.min(v - upper) - 1.0
    hi = np.max(v - lower) + 1.0
    for _ in range(60):  # enough for double precision
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, lower, upper)
        s = np.sum(x)
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, lower, upper)


def compute_subgradient(h):
    """
    Subgradient of max_k ∫ h(x)(1-h(x+k))dx at the current maximising shift.
    Implements the derivative of the discrete correlation term.
    """
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    imax = int(np.argmax(corr))
    shift = imax - (n - 1)          # integer shift producing the maximal value
    grad = np.zeros_like(h)

    if shift >= 0:
        if shift < n:
            # term where h[i] appears as the first factor
            idx1 = np.arange(0, n - shift)
            grad[idx1] += (1.0 - h[idx1 + shift])
            # term where h[i] appears as the second factor inside (1 - h[⋅])
            idx2 = np.arange(shift, n)
            grad[idx2] += -h[idx2 - shift]
    else:
        s = -shift
        if s < n:
            # positive shift version for negative indices (symmetry)
            idx_i = np.arange(s, n)                # i where i-s >= 0
            grad[idx_i] += (1.0 - h[idx_i - s])
            idx_j = np.arange(0, n - s)            # j = i-s
            grad[idx_j] += -h[idx_j + s]
    return grad


def local_optimize(start_h, target_sum, max_iter, deadline, rng):
    """
    Gradient‑based local improvement starting from start_h.
    Returns the best (h, C5) found before *deadline*.
    """
    h = start_h.copy()
    c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = c5

    for it in range(max_iter):
        if time.time() >= deadline:
            break
        grad = compute_subgradient(h)
        step = 0.4 * (1.0 - it / max_iter)          # linear decay
        new_h = h - step * grad
        new_h = project_to_bounds_sum(new_h, target_sum)
        new_c5 = evaluate_c5(new_h)

        if new_c5 < c5:
            h, c5 = new_h, new_c5
        else:
            # occasional uphill move (simulated annealing)
            temp = 0.02 * (1.0 - it / max_iter)
            prob = math.exp(-(new_c5 - c5) / (temp + 1e-12))
            if rng.random() < prob:
                h, c5 = new_h, new_c5

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_total = time.time()

    best_h = None
    best_c5 = float('inf')
    best_n = None

    n_list = [40, 50, 60, 70, 80]          # grid resolutions to explore
    prev_h = None
    prev_n = None

    for idx, n in enumerate(n_list):
        elapsed = time.time() - start_total
        remaining_total = budget_s - elapsed
        if remaining_total <= 5.0:
            break

        remaining_n = len(n_list) - idx
        time_per_n = max(5.0, remaining_total / remaining_n)
        local_deadline = time.time() + time_per_n
        target_sum = n / 2.0

        # First candidate (interpolated from previous resolution if available)
        if prev_h is not None and prev_n is not None:
            x_old = np.linspace(0.0, 2.0, prev_n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            candidate = np.interp(x_new, x_old, prev_h)
            candidate = project_to_bounds_sum(candidate, target_sum)
        elif 'initial_h_values' in globals() and len(initial_h_values) == n:
            candidate = project_to_bounds_sum(np.array(initial_h_values, dtype=float), target_sum)
        else:
            candidate = project_to_bounds_sum(rng.random(n), target_sum)

        # Run a short local optimisation on the candidate
        cand_h, cand_c5 = local_optimize(candidate, target_sum, max_iter=200,
                                         deadline=local_deadline, rng=rng)
        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = cand_h.copy()
            best_n = n

        best_local_h = cand_h
        best_local_c5 = cand_c5

        # Random restarts until the slice’s time quota expires
        while time.time() < local_deadline:
            start_vec = rng.random(n)
            start_vec = project_to_bounds_sum(start_vec, target_sum)
            h_opt, c5_opt = local_optimize(start_vec, target_sum, max_iter=200,
                                           deadline=local_deadline, rng=rng)
            if c5_opt < best_local_c5:
                best_local_c5 = c5_opt
                best_local_h = h_opt.copy()
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n

            # Early exit if we already beat the target bound
            if best_c5 <= 0.3808:
                break

        # Keep the best solution from this resolution for the next finer grid
        prev_h, prev_n = best_local_h, n

        # Global early exit
        if best_c5 <= 0.3808:
            break

    # Fallback (should never happen)
    if best_h is None:
        n = n_list[0]
        best_h = np.full(n, 0.5)
        best_n = n
        best_c5 = evaluate_c5(best_h)

    # Return the best found construction
    return (best_h, float(best_c5), int(best_n))
