import numpy as np
import time
import math

def _project_capped_simplex(v, target_sum, max_iter=50, tol=1e-9):
    # Project onto {x | 0<=x_i<=1, sum x_i = target_sum}
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if abs(s - target_sum) <= tol:
            return x
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _random_feasible(n, target_sum):
    v = np.random.rand(n)
    return _project_capped_simplex(v, target_sum)

def _evaluate(h, n):
    # evaluate_erdos_solution is assumed to be imported
    # returns tuple where first element is the bound
    return evaluate_erdos_solution(h, n)[0]

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    # Determine problem size
    if 'initial_h_values' in globals() and isinstance(initial_h_values, np.ndarray):
        n = int(initial_h_values.shape[0])
    else:
        n = 47  # fallback size
    target_sum = n / 2.0
    dx = 2.0 / n

    # Initialize
    if 'initial_h_values' in globals() and isinstance(initial_h_values, np.ndarray):
        h0 = _project_capped_simplex(initial_h_values.astype(float), target_sum)
    else:
        h0 = _random_feasible(n, target_sum)
    best_h = h0.copy()
    best_c5 = _evaluate(best_h, n)
    cur_h = best_h.copy()
    cur_c5 = best_c5

    # Simulated annealing parameters
    T0 = 0.05
    T_end = 1e-5
    sigma0 = 0.12
    sigma_end = 0.008

    start = time.time()
    deadline = start + budget_s - 0.05  # tiny safety margin

    # Main loop
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / budget_s  # 0..1
        # Exponential cooling for temperature and step size
        T = T0 * ((T_end / T0) ** frac)
        sigma = sigma0 * ((sigma_end / sigma0) ** frac)

        # Propose new candidate
        cand = cur_h + sigma * (np.random.rand(n) - 0.5) * 2.0
        cand = _project_capped_simplex(cand, target_sum)

        # Evaluate
        cand_c5 = _evaluate(cand, n)

        # Acceptance rule
        if cand_c5 < cur_c5 or np.random.rand() < math.exp((cur_c5 - cand_c5) / max(T, 1e-12)):
            cur_h = cand
            cur_c5 = cand_c5

        # Track global best
        if cand_c5 < best_c5:
            best_h = cand.copy()
            best_c5 = cand_c5

        # Occasionally reset to the best to avoid drift
        if np.random.rand() < 0.001:
            cur_h = best_h.copy()
            cur_c5 = best_c5

    # Ensure feasibility of returned vector (tiny correction if needed)
    # Clip again and re-project to satisfy sum exactly
    best_h = _project_capped_simplex(np.clip(best_h, 0.0, 1.0), target_sum)

    # Compute final bound one more time in case projection changed it
    best_c5 = _evaluate(best_h, n)

    return (best_h, float(best_c5), n)

import numpy as np
import time
import math

def _project_capped_simplex(v, target_sum, max_iter=50, tol=1e-9):
    """Project v onto {x | 0<=x_i<=1, sum x_i = target_sum}."""
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if abs(s - target_sum) <= tol:
            return x
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
