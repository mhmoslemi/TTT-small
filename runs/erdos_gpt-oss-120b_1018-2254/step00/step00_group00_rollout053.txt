def repair_sum(x):
    n = len(x)
    target = n / 2.0
    x = np.clip(x, 0.0, 1.0)
    s = np.sum(x)
    diff = target - s
    # small tolerance
    if abs(diff) < 1e-12:
        return x
    # We will adjust sequentially
    # Simple linear scan
    i = 0
    while abs(diff) > 1e-12 and i < n:
        if diff > 0:
            # Need to increase
            avail = 1.0 - x[i]
            if avail > 0:
                delta = min(avail, diff)
                x[i] += delta
                diff -= delta
        else:
            # Need to decrease
            avail = x[i]
            if avail > 0:
                delta = min(avail, -diff)
                x[i] -= delta
                diff += delta
        i += 1
    # after loop diff may still be small but non-zero due to float
    if abs(diff) > 1e-12:
        # adjust first element within bounds
        # adjust by diff clipped
        x[0] = np.clip(x[0] + diff, 0.0, 1.0)

                diff -= delta
        else:
            avail = x[i]
            if avail > 0:
                delta = min(avail, -diff)
                x[i] -= delta
                diff += delta
        i += 1
    if abs(diff) > 1e-12:
        # Final tiny correction on first element
        x[0] = np.clip(x[0] + diff, 0.0, 1.0)
    return x

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] with ∫h = 1 to minimize
    C5 = max_k ∫ h(x)(1-h(x+k)) dx.

    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Initialise h
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
        h0 = np.clip(h0, 0.0, 1.0)
        cur_h = _repair_sum(h0)
    else:
        n = 45  # fallback size (odd is okay)
        cur_h = rng.random(n)
        cur_h = _repair_sum(cur_h)

    cur_score = _evaluate(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score

    # Simulated‑annealing parameters
    step_max, step_min = 0.5, 0.01
    T_max, T_min = 0.05, 1e-4
    no_improve = 0

    while time.time() - start_time < budget_s - 0.5:
        elapsed = time.time() - start_time
        remaining = max(0.0, (budget_s - elapsed) / budget_s)

        # linearly anneal step size and temperature
        step = step_min + (step_max - step_min) * remaining
        T = T_min + (T_max - T_min) * remaining

        # propose a move that preserves the sum
        a, b = rng.integers(n, size=2, replace=False)
        # positive delta: move from a to b
        delta_pos = min(step, cur_h[a], 1.0 - cur_h[b])
        # negative delta: move from b to a
        delta_neg = -min(step, cur_h[b], 1.0 - cur_h[a])

        if delta_pos <= delta_neg:
            continue  # no feasible direction

        delta = rng.uniform(delta_neg, delta_pos)
        new_h = cur_h.copy()
        new_h[a] -= delta
        new_h[b] += delta
        # repair numerical drift
        new_h = _repair_sum(new_h)

        new_score = _evaluate(new_h)

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / (T + 1e-12)):
            cur_h = new_h
            cur_score = new_score
            no_improve = 0
        else:
            no_improve += 1

        if cur_score < best_score:
            best_h = cur_h.copy()
            best_score = cur_score
            no_improve = 0

        # occasional random restart if stagnating
        if no_improve > 5000:
            cur_h = rng.random(n)
            cur_h = _repair_sum(cur_h)
            cur_score = _evaluate(cur_h)
            no_improve = 0

    return best_h, best_score, n
