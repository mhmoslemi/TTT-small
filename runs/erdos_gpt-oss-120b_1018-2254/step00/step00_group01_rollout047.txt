import time
import math
import random

def _compute_c5(h, dx):
    """Return max correlation integral for vector h."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def _make_random_feasible(n_points):
    """Generate a random feasible vector (values in [0,1], sum = n/2)."""
    target = n_points / 2.0
    n_one = int(np.floor(target))
    remainder = target - n_one
    h = np.zeros(n_points, dtype=float)
    # choose indices for ones
    if n_one > 0:
        idx_one = np.random.choice(n_points, n_one, replace=False)
        h[idx_one] = 1.0
    else:
        idx_one = np.array([], dtype=int)
    # place remainder if needed
    if remainder > 1e-12:
        avail = np.setdiff1d(np.arange(n_points), idx_one, assume_unique=True)
        i_half = np.random.choice(avail)
        h[i_half] = remainder
    return h

def _adjust_sum_to_target(h, target):
    """Adjust a single element to make the sum exactly target (tiny correction)."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:

    # try to add/subtract diff to a feasible entry
