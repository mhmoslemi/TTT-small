def gradient_for_shift(h, r):
    n = h.shape[0]
    # compute term1 and term2 as described using slicing.

    if r >= 0:
        # term1[j] = 1 - h[j+r] for j+r<n else 0
        term1 = np.empty(n, dtype=float)
        term1[:n - r] = 1.0 - h[r:]
        term1[n - r:] = 0.0
        # term2[j] = h[j-r] for j>=r else 0
        term2 = np.empty(n, dtype=float)
        term2[:r] = 0.0
        term2[r:] = h[:n - r]
    else:
        s = -r
        term1 = np.empty(n, dtype=float)
        term1[:s] = 0.0
        term1[s:] = 1.0 - h[:n - s]
        term2 = np.empty(n, dtype=float)
        term2[:n - s] = h[s:]
        term2[n - s:] = 0.0

def gradient_for_shift(h, shift):
    n = len(h)
    g = np.empty_like(h)
    if shift >= 0:
        # compute term1: 1 - h[shift:]
        # shape n-shift
        term1 = np.empty(n, dtype=float)
        term1[:n-shift] = 1.0 - h[shift:]
        term1[n-shift:] = 0.0
        term2 = np.empty(n, dtype=float)
        term2[:shift] = 0.0
        term2[shift:] = h[:n-shift]
        g[:] = term1 - term2
    else:
        s = -shift
        term1 = np.empty(n, dtype=float)
        term1[:s] = 0.0
        term1[s:] = 1.0 - h[:n-s]
        term2 = np.empty(n, dtype=float)
        term2[:n-s] = h[s:]
        term2[n-s:] = 0.0
        g[:] = term1 - term2
    return g

def project_onto_box_and_sum(y, target):
    # Range for lambda
    lo = -np.max(y) - 2.0 # ensures y+lo <= -2 => clip to 0
    hi = 1.0 - np.min(y) + 2.0 # ensures y+hi >= 1 leads to clip 1
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(y + mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(y + (lo + hi) / 2.0, 0.0, 1.0)
    # Adjust small deviation
    diff = target - x.sum()
    if abs(diff) > 1e-8:
        # adjust interior
        interior = (x > 0) & (x < 1)
        cnt = interior.sum()
        if cnt > 0:
            # distribute diff evenly
            adj = diff / cnt
            x[interior] += adj
            # Clip again
            x = np.clip(x, 0.0, 1.0)
            # After this, sum may not be exact but close
    return x

import time, math, random

def project_onto_box_and_sum(y, target):
    """Euclidean projection of y onto {0<=x<=1, sum(x)=target}."""
    lo = -np.max(y) - 2.0
    hi = 1.0 - np.min(y) + 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(y + mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny adjustment for exact sum
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        interior = (x > 0.0) & (x < 1.0)
        cnt = interior.sum()
        if cnt > 0:
            adj = diff / cnt
            x[interior] += adj
            x = np.clip(x, 0.0, 1.0)
    return x

def gradient_for_shift(h, shift):
    """Subgradient of the overlap integral at the shift attaining the maximum."""
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        r = shift
        # term1 = 1 - h[i+r] (zero when i+r out of range)
        term1 = np.empty(n, dtype=float)
        term1[:n - r] = 1.0 - h[r:]
        term1[n - r:] = 0.0
        # term2 = h[i-r] (zero when i-r out of range)
        term2 = np.empty(n, dtype=float)
        term2[:r] = 0.0
        term2[r:] = h[:n - r]
    else:
        s = -shift
        term1 = np.empty(n, dtype=float)
        term1[:s] = 0.0
        term1[s:] = 1.0 - h[:n - s]
        term2 = np.empty(n, dtype=float)
        term2[:n - s] = h[s:]
        term2[n - s:] = 0.0
    g[:] = term1 - term2
    return g

def compute_c5(vec):
    n = vec.shape[0]
    conv = np.correlate(vec, 1.0 - vec, mode='full')
    return conv.max() * (2.0 / n)

            if cand_c5 < best_c5:
                best_c5, best_h = cand_c5, cand.copy()

        step = step_loc * 1.1
        step *= decay

    # ------------------------------------------------------------------
    # Random hill‑climbing with mass‑exchange moves for remaining time
    cur_h = best_h.copy()
    cur_c5 = best_c5
    temp_start = 0.02
    temp_end = 0.0005

    while time.time() - start < budget_s - 0.5:
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # decide direction i->j or j->i
        if np.random.rand() < 0.5:
            max_delta = min(cur_h[i], 1.0 - cur_h[j])
            if max_delta <= 0:
                continue
            delta = np.random.rand() * max_delta * 0.5
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        else:
            max_delta = min(cur_h[j], 1.0 - cur_h[i])
            if max_delta <= 0:
                continue
            delta = np.random.rand() * max_delta * 0.5
            new_h = cur_h.copy()
            new_h[j] -= delta
            new_h[i] += delta

        new_c5 = compute_c5(new_h)

        # temperature schedule linear in elapsed fraction
        frac = (time.time() - start) / budget_s
        temp = temp_start * (1 - frac) + temp_end * frac

        if new_c5 < cur_c5 or np.random.rand() < math.exp(-(new_c5 - cur_c5) / (temp + 1e-12)):
            cur_h, cur_c5 = new_h, new_c5
            if new_c5 < best_c5:
                best_c5, best_h = new_c5, new_h.copy()

    # final projection to guarantee feasibility
    best_h = project_onto_box_and_sum(best_h, target_sum)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
