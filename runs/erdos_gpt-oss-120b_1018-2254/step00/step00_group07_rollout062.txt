def subgradient_optimize(h, iters):
    for t in range(iters):
        corr = np.correlate(h, h, mode='full') * dx
        min_corr = corr.min()
        shift_idx = corr.argmin()
        shift = shift_idx - (n - 1)
        # gradient of corr[shift] wrt h:
        g = np.zeros_like(h)
        if shift >= 0:
            g[:n-shift] += h[shift:]
            g[shift:] += h[:n-shift]
        else:
            s = -shift
            g[s:] += h[:n-s]
            g[:n-s] += h[s:]
        # remove mean to maintain sum:
        g = g - g.mean()
        # compute step size: maybe 0.2 / sqrt(t+1)
        step = 0.5 / np.sqrt(t+1) # adjust factor
        # Determine max feasible step
        # find max alpha such that 0 <= h + alpha*g <= 1 elementwise
        # For each i, if g_i > 0, alpha <= (1 - h_i)/g_i; if g_i < 0, alpha <= h_i/(-g_i)
        # Compute all bounds and take minimum:
        alpha_pos = [(1 - h[i]) / g[i] if g[i] > 0 else np.inf for i in range(n)]
        alpha_neg = [h[i] / (-g[i]) if g[i] < 0 else np.inf for i in range(n)]
        max_alpha = min(min(alpha_pos), min(alpha_neg))

        step = min(step, 0.9 * max_alpha)
        if step <= 0:

        h = h + step * g
        # clip to avoid numeric noise

def run(seed=42, budget_s=1100, **kwargs):
    import time
    import numpy as np
    from scipy.optimize import minimize

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size and initial vector
    # ------------------------------------------------------------------
    try:
        init = initial_h_values  # provided by the harness
        if isinstance(init, np.ndarray):
            n = init.shape[0]
            h_start = init.astype(np.float64).copy()
        else:
            raise NameError
    except Exception:
        n = 200                     # fallback size (under 1000)
        h_start = np.full(n, 0.5, dtype=np.float64)

    target_sum = n / 2.0            # because ∫h = 1 and length = 2
    dx = 2.0 / n

    # ------------------------------------------------------------------
    # Ensure feasibility of the start vector (bounds + sum)
    # ------------------------------------------------------------------
    def enforce_sum(vec):
        """Project onto the hyperplane sum(vec)=target_sum while staying in [0,1]."""
        cur = vec.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            return vec
        # distribute the difference uniformly, then clip and repeat a few times
        vec += diff / n
        np.clip(vec, 0.0, 1.0, out=vec)
        for _ in range(5):
            cur = vec.sum()
            diff = target_sum - cur
            if abs(diff) < 1e-12:
                break
            # pick a random index that can absorb the remaining diff
            idx = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - vec[idx])
                vec[idx] += add
                diff -= add
            else:
                sub = min(-diff, vec[idx])
                vec[idx] -= sub
                diff += sub
        return vec

    h_start = enforce_sum(h_start)

    # ------------------------------------------------------------------
    # Helper: evaluation of C5 for a given vector
    # ------------------------------------------------------------------
    def compute_c5(vec):
        """Return max_k ∫ h(x)(1-h(x+k)) dx using discrete correlation."""
        # The correlation is exactly np.correlate(vec, 1-vec, mode='full')*dx
        # but we can replace with 1 - correlation of vec with itself for stability.
        corr = np.correlate(vec, vec, mode='full') * dx
        # C(k) = 1 - corr(k)
        return 1.0 - corr.min()

    # ------------------------------------------------------------------
    # Initial best solution
    # ------------------------------------------------------------------
    best_h = h_start.copy()
    best_c5 = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Optimization helpers
    # ------------------------------------------------------------------
    bounds = [(0.0, 1.0)] * n
    constraints = ({
        'type': 'eq',
        'fun': lambda x: np.sum(x) - target_sum,
    },)

    def objective(x):
        # objective is the quantity we want to minimise: C5
        return compute_c5(x)

    # ------------------------------------------------------------------
    # First deterministic local refinement (SLSQP)
    # ------------------------------------------------------------------
    try:
        res = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 200, 'ftol': 1e-9, 'disp': False},
        )
        if res.success:
            cand = enforce_sum(np.clip(res.x, 0.0, 1.0))
            cand_c5 = compute_c5(cand)
            if cand_c5 < best_c5:
                best_c5, best_h = cand_c5, cand
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Simple hill‑climbing move: exchange mass between two coordinates
    # ------------------------------------------------------------------
    def hill_climb(h, steps, step_size):
        nonlocal best_h, best_c5
        for _ in range(steps):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            # feasible delta range that preserves bounds
            low = max(-h[i], h[j] - 1.0)
            high = min(1.0 - h[i], h[j])
            if low >= high:
                continue
            delta = rng.uniform(low, high)
            # limit magnitude to keep moves small
            if abs(delta) > step_size:
                delta = step_size if delta > 0 else -step_size
            h[i] += delta
            h[j] -= delta
            cur_c5 = compute_c5(h)
            if cur_c5 < best_c5:
                best_c5, best_h = cur_c5, h.copy()
            else:
                # revert
                h[i] -= delta
                h[j] += delta

    # ------------------------------------------------------------------
    # Main search loop with time budget
    # ------------------------------------------------------------------
    start_time = time.time()
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 0.5:      # leave a tiny safety margin
            break

        # start from current best, perturb a bit, then run a short SLSQP
        x0 = best_h.copy()
        # small random mass swaps to diversify
        for _ in range(30):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            low = max(-x0[i], x0[j] - 1.0)
            high = min(1.0 - x0[i], x0[j])
            if low >= high:
                continue
            d = rng.uniform(low, high)
            x0[i] += d
            x0[j] -= d

        # short SLSQP refinement
        try:
            res = minimize(
                objective,
                x0,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 120, 'ftol': 1e-8, 'disp': False},
            )
            if res.success:
                cand = enforce_sum(np.clip(res.x, 0.0, 1.0))
                cand_c5 = compute_c5(cand)
                if cand_c5 < best_c5:
                    best_c5, best_h = cand_c5, cand
        except Exception:
            pass

        # a few hill‑climb steps to squeeze out remaining improvement
        hill_climb(best_h.copy(), steps=250, step_size=0.07)

    # ------------------------------------------------------------------
    # Final projection to ensure feasibility (tiny numerical drift may occur)
    # ------------------------------------------------------------------
    best_h = enforce_sum(np.clip(best_h, 0.0, 1.0))

    # ------------------------------------------------------------------
    # Final bound (use the provided evaluator if available)
    # ------------------------------------------------------------------
    try:
        final_c5 = evaluate_erdos_solution(best_h)
    except Exception:
        final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
