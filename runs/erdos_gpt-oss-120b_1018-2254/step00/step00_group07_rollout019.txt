import time
import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # Determine number of points and initial vector
    try:
        h0 = np.array(initial_h_values, copy=True)
        n_points = h0.shape[0]
    except NameError:
        n_points = 80
        h0 = np.zeros(n_points)
        h0[: n_points // 2] = 1.0

    target_sum = n_points / 2.0

    # Projection onto the box [0,1] and the equality sum constraint
    def project(x):
        x = np.clip(x, 0.0, 1.0)
        for _ in range(100):
            s = x.sum()
            if np.isclose(s, target_sum, atol=1e-12):
                break
            diff = target_sum - s
            if diff > 0:
                mask = x < 1.0 - 1e-12
            else:
                mask = x > 1e-12
            cnt = mask.sum()
            if cnt == 0:
                break
            delta = diff / cnt
            x[mask] += delta
            x = np.clip(x, 0.0, 1.0)
        # final tiny correction
        residual = target_sum - x.sum()
        if abs(residual) > 1e-10:
            if residual > 0:
                mask = x < 1.0 - 1e-10
            else:
                mask = x > 1e-10
            cnt = mask.sum()
            if cnt:
                x[mask] += residual / cnt
                x = np.clip(x, 0.0, 1.0)
        return x

    # Objective wrapper
    def obj(h):
        return evaluate_erdos_solution(np.asarray(h))

    # Equality constraint for SLSQP
    eq_cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}
    bounds = [(0.0, 1.0)] * n_points

    # Initial best candidate
    best_h = project(h0)
    best_val = float(obj(best_h))

    # Keep a small list of top candidates (sorted by value, low to high)
    top_k = 5
    top_candidates = [(best_h.copy(), best_val)]

    def insert_candidate(vec, val):
        # insert keeping list sorted, size at most top_k
        if len(top_candidates) < top_k or val < top_candidates[-1][1]:
            pos = 0
            while pos < len(top_candidates) and top_candidates[pos][1] <= val:
                pos += 1
            top_candidates.insert(pos, (vec.copy(), val))
            if len(top_candidates) > top_k:
                top_candidates.pop()

    # Random exploration phase (≈50% of budget)
    explore_time = budget_s * 0.5
    batch = 400  # candidates per batch
    while time.time() - start < explore_time:
        # generate a batch of uniform random vectors
        rand_batch = rng.random((batch, n_points))
        for i in range(batch):
            cand = project(rand_batch[i])
            val = float(obj(cand))
            if val < best_val:
                best_val = val
                best_h = cand.copy()
            insert_candidate(cand, val)
            if time.time() - start >= explore_time:
                break

    # Local refinement with SLSQP for each top candidate
    refine_time = budget_s * 0.9
    for vec, _ in top_candidates:
        if time.time() - start > refine_time:
            break
        # Run SLSQP from this start point
        res = opt.minimize(
            obj,
            vec,
            method='SLSQP',
            bounds=bounds,
            constraints=[eq_cons],
            options={'maxiter': 400, 'ftol': 1e-9, 'disp': False}
        )
        h_opt = project(res.x)
        val_opt = float(obj(h_opt))
        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()

    # Final fallback: ensure we have a feasible vector
    best_h = project(best_h)
    best_val = float(obj(best_h))

    return best_h, best_val, n_points
