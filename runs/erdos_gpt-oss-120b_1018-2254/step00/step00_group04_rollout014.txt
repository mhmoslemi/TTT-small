def mutate(h, max_delta):

    i, j = np.random.choice(N, 2, replace=False)

            if diff > 0 and up_cap > 0:
                h += diff * slack_up / up_cap
            elif diff < 0 and down_cap > 0:
                h += diff * slack_down / down_cap
            h = np.clip(h, 0.0, 1.0)

    # --------------------------------------------------------------
    def mutate(h, max_delta):
        N = h.shape[0]
