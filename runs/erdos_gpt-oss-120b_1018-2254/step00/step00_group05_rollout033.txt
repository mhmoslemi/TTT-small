import time
import math

def _score(h, dx):
    """Return max_k ∫ h(x)(1-h(x+k)) dx approximated on the grid."""
    # np.correlate returns the discrete correlation (no dx), multiply after max
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] that minimizes the Erdős C₅ bound.

    Returns:
        h_values (np.ndarray): sampled values of h on a uniform grid.
        c5_bound (float): the achieved upper bound for C₅.
        n_points (int): number of samples (grid size).
    """

    start_time = time.time()

        h_start = np.array(initial_h_values, dtype=np.float64)
    except Exception:

        h_start = np.full(n, 0.5, dtype=np.float64)

    target_sum = n / 2.0          # because ∫h = 1  ⇒  sum(h)*dx = 1,  dx=2/n
    cur_sum = h_start.sum()

    np.clip(h_start, 0.0, 1.0, out=h_start)

    diff = target_sum - h_start.sum()

    dx = 2.0 / n
