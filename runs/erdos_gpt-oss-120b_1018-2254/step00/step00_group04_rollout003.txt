def project_h(v, target):
    # Clip to [0,1]
    v = np.clip(v, 0.0, 1.0)
    # Adjust sum to target using water-filling
    # We'll iterate up to 100 times
    for _ in range(100):
        cur_sum = np.sum(v)
        diff = target - cur_sum
        if abs(diff) < 1e-12:
            break
        # Identify free entries (not at bounds)
        free = (v > 0) & (v < 1)
        n_free = np.count_nonzero(free)
        if n_free == 0:
            # cannot adjust further, break
            break

import numpy as np
import time
import random

def project_h(v, target):
    # Clip to [0,1] and adjust sum to target.
    v = np.clip(v, 0.0, 1.0)
    for _ in range(100):
        cur = np.sum(v)
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        free = (v > 0) & (v < 1)
        n_free = np.count_nonzero(free)
        if n_free == 0:
            break
        delta = diff / n_free
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    # Minor correction if needed
    cur = np.sum(v)
    diff = target - cur
    if abs(diff) > 1e-9:
        # distribute diff to any non‑full/empty entries
        for i in range(len(v)):
            if (diff > 0 and v[i] < 1) or (diff < 0 and v[i] > 0):
                avail = (1 - v[i]) if diff > 0 else -v[i]
                take = min(abs(diff), abs(avail))
                v[i] += np.sign(diff) * take
                diff -= np.sign(diff) * take
                if abs(diff) < 1e-12:
                    break
    return v

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_c5_and_shift(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    overlaps = corr * dx
    idx = np.argmax(overlaps)
    shift = idx - (n - 1)
    return overlaps[idx], shift

def gradient_worst_shift(h, shift):
    n = len(h)
    g = np.ones(n)
    if shift >= 0:
        # subtract h[i+shift]
        g[:n-shift] -= h[shift:]
        # subtract h[i-shift]
        g[shift:] -= h[:n-shift]
    else:
        s = -shift
        # subtract h[i+shift] = h[i-s] for i>=s
        g[s:] -= h[:n-s]
        # subtract h[i-shift] = h[i+s] for i<=n-1-s
        g[:n-s] -= h[s:]
    return g

def gradient_step(h, shift, alpha, target):
    g = gradient_worst_shift(h, shift)
    h_new = h - alpha * g
    h_new = project_h(h_new, target)
    return h_new

def random_swap_mutation(h, max_delta=0.05):
    n = len(h)
    i, j = np.random.choice(n, size=2, replace=False)
    delta = (np.random.rand() - 0.5) * 2.0 * max_delta
    dv = np.zeros(n)
    dv[i] = delta
    dv[j] = -delta
    h_new = h + dv
    target = n / 2.0
    h_new = project_h(h_new, target)
    return h_new

    np.random.seed(seed)
    random.seed(seed)
    # initialise h
    if 'initial_h_values' in globals():

        n = len(h)

    else:

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    current_h = h.copy()
    current_c5 = best_c5

    start = time.time()
    # Parameter schedule
    T0 = 0.02    # initial temperature
    alpha = 0.1  # gradient step size
    max_delta = 0.08

        if elapsed >= budget_s * 0.95:

        if iter_count % 20 == 0:

            cand = gradient_step(current_h, shift, alpha, target)
        else:
            cand = random_swap_mutation(current_h, max_delta)

        cand_c5 = compute_c5(cand)

        if cand_c5 < current_c5 or np.random.rand() < np.exp(-(cand_c5 - current_c5) / (T + 1e-12)):

            max_delta = max(0.01, max_delta * 0.95)

    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, n)
