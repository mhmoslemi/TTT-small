def project_to_feasible(h):
    # Clip values to [0,1] then adjust to meet sum constraint.
    np.clip(h, 0, 1, out=h)
    total = h.sum()
    target = len(h) / 2.0
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    # Adjust by distributing diff uniformly across entries not at bounds?
    # Simpler: add diff/N to all entries then clip again; iterate.
    # But more stable: Since we maintain sum exactly during moves, this may rarely used.
    # We'll implement simple fix: compute additive needed, add diff to first element within bounds.
    for i in range(len(h)):

            diff -= delta

            diff += delta

    # Clip to [0,1] then adjust sum to target preserving bounds
    np.clip(h, 0.0, 1.0, out=h)
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    # Distribute diff greedily
    if diff > 0:
        # need to increase sum
        # find indices where h[i] < 1
        idx = np.where(h < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - h[i])
            h[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    else:
        # need to decrease sum
        idx = np.where(h > 0.0)[0]
        for i in idx:
            dec = min(-diff, h[i])
            h[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return h

        start_vec = best_h.copy() if restart == 0 else random_binary(n)
        remaining = time_limit - elapsed
        # allocate a modest, time‑scaled iteration budget per restart
        max_iters = int(min(40000, max(2000, remaining * 30)))
        cand_h, cand_val = anneal(start_vec, dx, max_iters, start_time, time_limit)
        if cand_val < best_val:
            best_val = cand_val
            best_h = cand_h.copy()
        restart += 1

    # ----- final feasibility enforcement and return -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_target_sum(best_h, n / 2.0)
    return best_h.tolist(), float(best_val), n
