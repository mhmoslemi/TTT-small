def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')

def steepest_descent_binary(h, dx, start_time, budget_s):
    n = h.shape[0]
    cur_h = h.copy()
    cur_val = compute_c5(cur_h, dx)
    # Pre-calc indices sets
    while True:
        best_improve = 0.0
        best_swap = None
        # get indexes
        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        # For each possible swap compute new val
        for i in ones:
            for j in zeros:
                # Swap
                new_h = cur_h.copy()
                new_h[i] = 0.0
                new_h[j] = 1.0
                new_val = compute_c5(new_h, dx)

        cur_h = new_h
        cur_val = new_val
        # time check

def random_real_solution(n):

    # Using np.random.gamma

    # Now h entries may exceed 1; but with target_sum = n/2, average = 0.5; max of gamma may exceed 2; but improbable: need to clip.

    # We'll fix:
    while True:
        over = h > 1.0
        if not np.any(over):
            break
        excess = h[over] - 1.0
        h[over] = 1.0
        # distribute excess to those <1 proportionally
        under = h < 1.0
        if not np.any(under):
            break
        total_under = np.sum(h[under])
        if total_under == 0:
            break
        # subtract scaled excess from under entries equally
        # Actually we need to reduce sum to keep target; we can subtract excess sum equally from under entries
        excess_sum = np.sum(excess)

        # ensure we don't go below 0

        h[under] -= reduction
        # if any go below 0, clip and repeat

    return h

    if alt.sum() != n / 2:
        alt = np.zeros(n)
        alt[: n // 2] = 1.0
    patterns.append(alt)

    patterns.append(best_h.copy())
    for _ in range(5):
        patterns.append(random_binary_solution(n))

    # 4️⃣ improve each pattern with local search
    for pat in patterns:
        if time.time() - start > budget_s * 0.94:
            break
        pat = np.clip(pat, 0.0, 1.0)
        target = n / 2.0
        cur = pat.sum()
        if abs(cur - target) > 1e-9 and cur > 0:
            pat = pat * (target / cur)
            pat = np.clip(pat, 0.0, 1.0)
            cur = pat.sum()
            diff = target - cur
            if abs(diff) > 1e-12:
                for i in range(n):
                    if diff > 0 and pat[i] + diff <= 1:
                        pat[i] += diff
                        break
                    if diff < 0 and pat[i] + diff >= 0:
                        pat[i] += diff
                        break
        pat_h, pat_val = local_search(pat, dx, start, budget_s)
        if pat_val < best_val:
            best_h, best_val = pat_h, pat_val

    # 5️⃣ random binary restarts until budget runs out
    while time.time() - start < budget_s * 0.9:
        rand_pat = random_binary_solution(n)
        h_opt, val_opt = local_search(rand_pat, dx, start, budget_s)
        if val_opt < best_val:
            best_h, best_val = h_opt, val_opt

    # 6️⃣ final evaluation via the provided helper (if any)
    try:
        final_val, _ = evaluate_erdos_solution(best_h)
        final_val = float(final_val)
    except Exception:
        final_val = compute_c5(best_h, dx)

    return best_h.tolist(), final_val, n
