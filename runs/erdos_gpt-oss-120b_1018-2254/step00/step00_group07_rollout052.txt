            x[idx] += add

            cap = x
            idx = np.argmax(cap)
            sub = min(-diff, cap[idx])

            if interior.size:
                idx = interior[0]
                x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
        return x

    # ------------------------------------------------------------------
    # Initialise the current best solution.
    # ------------------------------------------------------------------
    if h0 is None:
        # random feasible start if no parent exists
        y = np.random.rand(n)
        h0 = project_to_box_sum(y, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Simple stochastic improvement loop.
    # ------------------------------------------------------------------
    start_time = time.time()
    target_c5 = 0.3808  # early‑stop target

    while time.time() - start_time < budget_s - 0.5:
        # perturb the current best solution modestly
        y = best_h + np.random.normal(scale=0.05, size=n)
        cand = project_to_box_sum(y, target_sum)
        cand_c5 = compute_c5(cand)

        if cand_c5 < best_c5:
            best_h, best_c5 = cand, cand_c5
            if best_c5 <= target_c5:
                break

        # occasional random restart to escape local traps
        if np.random.rand() < 0.01:
            y = np.random.rand(n)
            cand = project_to_box_sum(y, target_sum)
            cand_c5 = compute_c5(cand)
            if cand_c5 < best_c5:
                best_h, best_c5 = cand, cand_c5
                if best_c5 <= target_c5:
                    break

    # Return the solution in the required format.
    return (best_h.tolist(), float(best_c5), n)
