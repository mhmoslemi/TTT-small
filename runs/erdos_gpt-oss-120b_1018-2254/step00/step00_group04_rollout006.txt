def project_onto_simplex_box(v, target_sum, lower, upper):
    # Clip to bounds
    v = np.clip(v, lower, upper)
    # Compute current sum
    cur = np.sum(v)
    if abs(cur - target_sum) < 1e-8:
        return v
    # Water-filling: For diff >0 need to increase some entries not at upper
    # We'll implement iterative approach:
    diff = target_sum - cur
    # We'll treat potentials: indices that can be increased if diff>0 else decreased
    # We can use while loop.
    # Safety limit: 100 iterations.
    for _ in range(100):
        if diff > 0:
            free = v < upper - 1e-12
            if not np.any(free):
                break
            inc = diff / np.count_nonzero(free)
            # add inc to free entries
            v[free] += inc
            # clip again and recompute diff
            v = np.clip(v, lower, upper)
        elif diff < 0:
            free = v > lower + 1e-12
            if not np.any(free):
                break
            dec = diff / np.count_nonzero(free)  # diff negative
            v[free] += dec  # dec negative
            v = np.clip(v, lower, upper)
        cur = np.sum(v)
        diff = target_sum - cur
        if abs(diff) < 1e-10:
            break
    return v

def project_onto_simplex_box(v, target_sum, lower=0.0, upper=1.0):
    # Clip
    v = np.clip(v, lower, upper)
    cur_sum = v.sum()
    # If already target, return
    if abs(cur_sum - target_sum) < 1e-10:
        return v
    # Use water-filling
    # We'll iterate
    for _ in range(100):
        diff = target_sum - v.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # increase free entries (not at upper)
            free = v < upper - 1e-12
            if not np.any(free):
                break
            add = diff / free.sum()
            v[free] += add
            v = np.clip(v, lower, upper)
        else:
            # decrease free entries (not at lower)
            free = v > lower + 1e-12
            if not np.any(free):
                break
            sub = diff / free.sum()
            v[free] += sub
            v = np.clip(v, lower, upper)
    # final clip
    return np.clip(v, lower, upper)

import numpy as np
import time
import random
import math

def _compute_c5(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)*(1-h(x+k)) dx approximated on a grid."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return conv.max() * dx

def _project_onto_simplex_box(v: np.ndarray, target_sum: float,
                              lower: float = 0.0, upper: float = 1.0) -> np.ndarray:
    """Project v onto the box [lower,upper] while enforcing sum(v)=target_sum."""
    # Clip to bounds first.
    v = np.clip(v, lower, upper)
    cur = v.sum()
    if abs(cur - target_sum) < 1e-12:
        return v
    # Water‑filling style iterative correction.
    for _ in range(100):
        diff = target_sum - v.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = v < upper - 1e-12
            if not np.any(free):
                break
            inc = diff / free.sum()
            v[free] += inc
            v = np.clip(v, lower, upper)
        else:  # diff < 0
            free = v > lower + 1e-12
            if not np.any(free):
                break
            dec = diff / free.sum()   # diff negative, dec negative
            v[free] += dec
            v = np.clip(v, lower, upper)
    return np.clip(v, lower, upper)

    best_h = _project_onto_simplex_box(best_h, N / 2.0, 0.0, 1.0)
    final_score = _compute_c5(best_h)

    return best_h, final_score, N
