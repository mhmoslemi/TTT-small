def _c5_value(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)*(1-h(x+k))dx for discrete h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

                        continue
                delta = rng.random() * max_delta
            else:
                # fallback to random pair
                i = rng.integers(n)
                j = rng.integers(n - 1)
                if j >= i:
                    j += 1
                max_delta = min(cur_h[i], 1.0 - cur_h[j])
                if max_delta <= 0.0:
                    it += 1
                    continue
                delta = rng.random() * max_delta
        else:                     # pure random transfer
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            max_delta = min(cur_h[i], 1.0 - cur_h[j])
            if max_delta <= 0.0:
                # try opposite direction
                max_delta = min(cur_h[j], 1.0 - cur_h[i])
                if max_delta <= 0.0:
                    it += 1
                    continue
                i, j = j, i
            delta = rng.random() * max_delta

        # apply move
        cur_h[i] -= delta
        cur_h[j] += delta

        new_c5 = _c5_value(cur_h)

        # Metropolis acceptance
        if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / (T + 1e-12)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] += delta
            cur_h[j] -= delta

        # cooling & bookkeeping
        T *= decay
        it += 1

        # occasional restart from the best found (helps escaping plateaus)
        if it % 5000 == 0:
            cur_h = best_h.copy()
            cur_c5 = best_c5

    # final bound computed from the best vector
    final_c5 = _c5_value(best_h)

    return best_h.tolist(), final_c5, n
