def subgradient_descent(h_initial, n, max_iters, time_budget, start_time):
    h = np.copy(h_initial)
    best_h = h.copy()
    best_obj = objective(h, n)
    for it in range(max_iters):
        if time.time() - start_time > time_budget:
            break
        # compute correlation
        corr = np.correlate(h, 1-h, mode='full')
        max_corr = corr.max()
        # store best
        cur_obj = max_corr * (2.0/n)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_h = h.copy()
        # find indices where corr equals max (worst shifts)
        worst_shifts = np.where(corr == max_corr)[0]
        # choose one shift
        s_idx = worst_shifts[0]  # Choose first for gradient
        s = s_idx - (n-1)
        # compute gradient for shift s
        grad = np.zeros_like(h)
        if s >= 0:
            # j + s in range: j=0..n-1-s
            grad[:n-s] += (1 - h[s:])
            # j - s: j=s..n-1
            grad[s:] -= h[:n-s]
        else:
            t = -s
            # j + s valid for j=t..n-1
            grad[t:] += (1 - h[:n-t])
            # j - s valid for j=0..n-1-t
            grad[:n-t] -= h[t:]
        # step size schedule
        step = 0.5 / np.sqrt(it+1)  # simple
        # gradient descent, negative gradient
        h_new = h - step * grad
        # projection to feasible region
        h = project_onto_box_sum(h_new, target_sum=n/2.0, lower=0., upper=1.)
    return best_h, best_obj

    best = h.copy()
    best_val = objective(best, n)
    cur = h.copy()
    cur_val = best_val

    for it in range(max_iters):

            break
        # temperature schedule
        T = T0 * ((T_final / T0) ** (it / max_iters))

        if i == j:

import time
import numpy as np

def _objective(h):
    """C5 value for a vector h (assumes uniform spacing on [0,2])."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)

def _project_sum_box(v, target, lower=0.0, upper=1.0, tol=1e-9, max_iter=100):
    """Project v onto the intersection of box [lower,upper] and sum(v)=target."""
    lo = -np.max(v) - 10.0
    hi = (upper - np.min(v)) + 10.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        proj = np.clip(v + mid, lower, upper)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
        if hi - lo < tol:
            break
    return np.clip(v + (lo + hi) * 0.5, lower, upper)

def _gradient_shift(h, shift):
    """Gradient of corr(shift) = sum_i h_i (1 - h_{i+shift}) w.r.t. h."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        g[: n - shift] += 1.0 - h[shift:]
        g[shift:] -= h[: n - shift]
    else:
        t = -shift
        g[t:] += 1.0 - h[: n - t]
        g[: n - t] -= h[t:]
    return g

def _subgradient_descent(h0, max_iters, start, time_limit):
    """Simple sub‑gradient descent on max correlation."""
    n = h0.shape[0]
    h = h0.copy()
    best_h = h.copy()

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = corr.max()
        cur_val = max_corr * (2.0 / n)
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
        # pick a worst shift (first occurrence)
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        grad = _gradient_shift(h, shift)
        step = 0.7 / np.sqrt(it + 1.0)
        h_new = h - step * grad
        h = _project_sum_box(h_new, target=n / 2.0, lower=0.0, upper=1.0)
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # ---- initial feasible vector ------------------------------------------------
    if "initial_h_values" in globals():
        h_init = np.asarray(initial_h_values, dtype=float)
    else:
        # fallback: uniform half‑density
        n0 = 70
        h_init = np.full(n0, 0.5, dtype=float)
    n0 = h_init.shape[0]
    h_init = _project_sum_box(h_init, target=n0 / 2.0, lower=0.0, upper=1.0)

    best_h = h_init.copy()
    best_val = _objective(best_h)
    best_n = n0

    # ---- resolution schedule (keep small to stay fast) -------------------------
    levels = [n0]
    if n0 * 2 <= 300:
        levels.append(n0 * 2)
    if n0 * 3 <= 300:
        levels.append(n0 * 3)

    for n in levels:
        if time.time() - start_time > budget_s * 0.95:
            break

        if n == n0:
            base_h = h_init
        else:
            # up‑sample previous best solution linearly
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            base_h = np.interp(x_new, x_old, best_h)
            base_h = _project_sum_box(base_h, target=n / 2.0, lower=0.0, upper=1.0)

        # ----- multiple restarts at this resolution --------------------------------
        restarts = 4
        for r in range(restarts):
            if time.time() - start_time > budget_s * 0.95:
                break
            if r == 0:
                h_start = base_h.copy()
            else:
                rnd = np.random.rand(n)
                h_start = _project_sum_box(rnd, target=n / 2.0, lower=0.0, upper=1.0)

            # allocate a modest number of iterations (scaled with size)
            remaining = budget_s - (time.time() - start_time)
            max_iters = int(3000 * n0 / n) + 500  # more steps for smaller n
            h_opt, val_opt = _subgradient_descent(h_start, max_iters, start_time,
                                                  time_limit=remaining)
            if val_opt < best_val:
                best_val = val_opt
                best_h = h_opt.copy()
                best_n = n

    # final projection (safety) and evaluation
    best_h = _project_sum_box(best_h, target=best_n / 2.0, lower=0.0, upper=1.0)
    final_c5 = _objective(best_h)

    return best_h, final_c5, best_n
