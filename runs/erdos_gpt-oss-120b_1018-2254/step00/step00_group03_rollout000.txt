def project_box_simplex(v, target):
    # v is any real vector; returns u such that sum(u)=target, 0<=u<=1
    lo = -1.0
    hi = 1.0
    # Expand bounds appropriately
    # Ensure we find lo, hi where sum at lo <= target <= sum at hi
    # Compute sums at extremes
    def sum_at(l):
        u = v + l
        u = np.clip(u, 0, 1)
        return u.sum()
    # Expand hi until sum_at(hi) >= target
    while sum_at(hi) < target:
        hi *= 2
    while sum_at(lo) > target:
        lo *= 2
    for _ in range(50):
        mid = (lo+hi)/2
        s = sum_at(mid)
        if s < target:
            lo = mid
        else:
            hi = mid
    # return final
    u = v + (lo+hi)/2
    u = np.clip(u,0,1)
    return u

def crossover(p1, p2):
    n = len(p1)
    mask = np.random.rand(n) < 0.5
    child = np.where(mask, p1, p2)
    # repair to ensure sum = target_sum
    current_sum = child.sum()
    diff = int(current_sum - target_sum)
    if diff > 0:
        # too many ones, turn diff ones to zero
        ones_idx = np.where(child == 1)[0]
        to_flip = np.random.choice(ones_idx, diff, replace=False)
        child[to_flip] = 0
    elif diff < 0:
        zeros_idx = np.where(child == 0)[0]
        to_flip = np.random.choice(zeros_idx, -diff, replace=False)
        child[to_flip] = 1
    return child

        return float(np.max(corr) * dx)
