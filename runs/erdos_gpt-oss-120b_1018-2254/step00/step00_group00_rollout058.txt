            else:
                high = mid

    def evaluate(h):
        # h is np.ndarray
        # Might need to ensure float64
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    if 'initial_h_values' in globals():
        best = np.array(initial_h_values, dtype=np.float64)
        # ensure feasibility: project if needed
        best = project(best)

import numpy as np
import math


def project_vector(v, target):
    """Project onto {0<=x<=1, sum(x)=target} via a simple water‑filling binary search."""
    low = -np.max(v)          # guarantees v+low <= 0  → all zeros
    high = 1.0 - np.min(v)    # guarantees v+high >= 1 → all ones
    for _ in range(50):       # binary search for the Lagrange multiplier
        mid = (low + high) * 0.5
        h = np.clip(v + mid, 0.0, 1.0)
        s = h.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, 0.0, 1.0)


def evaluate_overlap(h, dx):
    """C5 value for a given step function."""
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx


def gradient_worst_shift(h):
    """Return the shift attaining the maximal overlap and its sub‑gradient."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')

        s = -r
        L = n - s

        grad[:L] -= h[s:]
    return r, grad
