def project_to_constraints(v, target_sum):
    # v: np.ndarray
    n = len(v)
    x = cp.Variable(n)
    objective = cp.Minimize(cp.sum_squares(x - v))
    constraints = [x >= 0, x <= 1, cp.sum(x) == target_sum]
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=False)
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        # fallback via simple method
        v = np.clip(v, 0, 1)
        cur = v.sum()
        # adjust with scaling
        if cur != 0:
            v = v * (target_sum / cur)
        v = np.clip(v, 0, 1)
        # final adjust via iterative simple.
        return v
    return x.value

import numpy as np
import random
import time
import math

def _c5_bound(h: np.ndarray) -> float:
    """Return the C5 bound for a step function sampled as array h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with 1-h (full mode includes zero padding)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] (sampled) to minimize the
    Erdős C5 overlap bound.

    Returns (h_values, c5_bound, n_points).
    """
    # ------------------------------------------------------------------
    # Setup
    np.random.seed(seed)
    random.seed(seed)

    # Determine problem size and starting vector
    try:
        # Use the provided initial construction if present
        init_arr = np.array(initial_h_values, dtype=float)
        n_points = init_arr.size
        # Ensure n_points is even; otherwise fall back to a fresh even size
        if n_points % 2 != 0:
            raise ValueError
        target_sum = n_points // 2  # integer because n_points is even
        # Produce a binary start by taking the largest entries
        order = np.argsort(-init_arr)          # descending order
        start_h = np.zeros(n_points, dtype=float)
        start_h[order[:target_sum]] = 1.0
    except Exception:
        # No usable initial_h_values – build a fresh problem of moderate size
        n_points = 200                     # even, gives finer resolution
        target_sum = n_points // 2
        start_h = np.zeros(n_points, dtype=float)
        # Two end blocks (a classic starting shape)
        left_len = target_sum // 2
        right_len = target_sum // 2
        extra = target_sum - left_len - right_len
        start_h[:left_len] = 1.0
        start_h[-right_len:] = 1.0
        if extra:
            start_h[left_len] = 1.0

    # ------------------------------------------------------------------
    # Helper to keep index lists up‑to‑date
    def _indices(h):
        ones = list(np.where(h == 1.0)[0])
        zeros = list(np.where(h == 0.0)[0])
        return ones, zeros

    current = start_h.copy()
    cur_c5 = _c5_bound(current)
    best_h = current.copy()
    best_c5 = cur_c5
    ones_idx, zeros_idx = _indices(current)

    # Annealing schedule constants
    max_temp = 0.5          # start temperature
    min_temp = 1e-5         # final temperature
    start_time = time.time()
    # ------------------------------------------------------------------
    # Main optimisation loop
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        # Early stop if we already achieved the target bound
        if best_c5 <= 0.3808:
            break

        frac = elapsed / budget_s
        temp = max_temp * (1.0 - frac) + min_temp

        # Propose a swap of a 1 and a 0 (maintains feasibility)
        i = random.choice(ones_idx)
        j = random.choice(zeros_idx)

        neighbor = current.copy()
        neighbor[i] = 0.0
        neighbor[j] = 1.0

        n_c5 = _c5_bound(neighbor)
        delta = n_c5 - cur_c5

        # Metropolis acceptance
        if delta <= 0.0 or random.random() < math.exp(-delta / temp):
            # accept move
            current = neighbor
            cur_c5 = n_c5
            # update index containers
            ones_idx.remove(i)
            zeros_idx.append(i)
            zeros_idx.remove(j)
            ones_idx.append(j)

            # record best
            if n_c5 < best_c5:
                best_c5 = n_c5
                best_h = neighbor.copy()

    # ------------------------------------------------------------------
    # Return result
    return best_h, best_c5, n_points
