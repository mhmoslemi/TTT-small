    h0 = np.clip(h0, 0.0, 1.0)
    cur_sum = h0.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        # Adjust the first element to fix the sum while staying in bounds
        diff = cur_sum - target_sum
        adj = min(max(-h0[0], -diff), 1.0 - h0[0])
        h0[0] += adj
        # If still off, distribute tiny correction uniformly (unlikely)
        residual = h0.sum() - target_sum
        if abs(residual) > 1e-12:
            h0 += -residual / n

    best_h = h0.copy()
    best_c5 = _c5_of(best_h)

    # If the helper evaluator is available, use it for a more accurate bound
    try:
        ev = evaluate_erdos_solution(best_h)
        best_c5 = min(best_c5, float(ev))
    except Exception:
        pass

    # -------------------------------------------------------------
    # Stochastic hill‑climbing with simulated‑annealing acceptance
    # -------------------------------------------------------------
    start_time = time.time()
    max_elapsed = budget_s - 5.0      # keep a small safety margin
    it = 0

    while True:
        elapsed = time.time() - start_time
        if elapsed > max_elapsed:
            break

        # adaptive step size: shrink as time passes
        step_scale = max(0.005, 0.2 * (1.0 - elapsed / budget_s))
        temperature = max(0.0, 0.1 * (1.0 - elapsed / budget_s))

        # pick two distinct indices
        i, j = np.random.choice(n, size=2, replace=False)

        # propose a symmetric perturbation that preserves the sum
        delta = (np.random.rand() * 2.0 - 1.0) * step_scale

        # clamp delta to keep both entries in [0,1]
        hi = best_h[i] + delta
        hj = best_h[j] - delta

        if hi < 0.0:
            delta = -best_h[i]
        elif hi > 1.0:
            delta = 1.0 - best_h[i]

        hi = best_h[i] + delta
        hj = best_h[j] - delta

        if hj < 0.0:
            delta = best_h[j]
        elif hj > 1.0:
            delta = best_h[j] - 1.0

        hi = best_h[i] + delta
        hj = best_h[j] - delta

        if not (0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0):
            continue

        candidate = best_h.copy()
        candidate[i] = hi
        candidate[j] = hj

        cand_c5 = _c5_of(candidate)

        # accept if improvement, otherwise with annealed probability
        if cand_c5 < best_c5:
            best_h = candidate
            best_c5 = cand_c5
        else:
            if temperature > 0.0:
                prob = math.exp(-(cand_c5 - best_c5) / temperature)
                if np.random.rand() < prob:
                    best_h = candidate

        it += 1

    # -------------------------------------------------------------
    # Final local refinement using SLSQP (if SciPy is available)
    # -------------------------------------------------------------
    leftover = budget_s - (time.time() - start_time)
    if minimize is not None and leftover > 1.0:
        # define constraint and bounds
        cons = {"type": "eq", "fun": lambda x: np.sum(x) - target_sum}
        bounds = [(0.0, 1.0)] * n

        try:
            res = minimize(
                _c5_of,
                best_h,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 200, "ftol": 1e-12, "disp": False},
            )
            if res.success and res.fun < best_c5:
                best_h = np.clip(res.x, 0.0, 1.0)
                # re‑enforce the sum exactly (tiny numerical drift)
                diff = best_h.sum() - target_sum
                if abs(diff) > 1e-12:
                    # shift a single component to correct the sum
                    for idx in range(n):
                        if (diff > 0 and best_h[idx] - diff >= 0.0) or (
                            diff < 0 and best_h[idx] - diff <= 1.0
                        ):
                            best_h[idx] -= diff
                            break
                best_c5 = _c5_of(best_h)
        except Exception:
            pass

    # -------------------------------------------------------------
    # Final check with the provided evaluator (if any) and final fix‑up
    # -------------------------------------------------------------
    try:
        ev = evaluate_erdos_solution(best_h)
        best_c5 = min(best_c5, float(ev))
    except Exception:
        pass

    # Ensure exact sum constraint before returning
    total = best_h.sum()
    if abs(total - target_sum) > 1e-12:
        diff = total - target_sum
        for idx in range(n):
            if (diff > 0 and best_h[idx] - diff >= 0.0) or (
                diff < 0 and best_h[idx] - diff <= 1.0
            ):
                best_h[idx] -= diff
                break
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h.tolist(), float(best_c5), n
