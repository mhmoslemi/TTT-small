def propose_transfer(h, max_transfer):
    # pick source and target
    source = np.random.choice(n)
    target = np.random.choice(n)
    if source == target:
        return None
    # compute allowed transfer
    max_up = 1 - h[target]
    max_down = h[source]
    max_allowed = min(max_up, max_down, max_transfer)
    if max_allowed <= 1e-12:
        return None
    delta = np.random.uniform(0, max_allowed)
    h_new = h.copy()
    h_new[source] -= delta
    h_new[target] += delta
    return h_new
```

def project_to_feasible(v, target_sum, low=0.0, high=1.0, max_iter=1000, tol=1e-12):
    v = np.clip(v, low, high)
    for _ in range(max_iter):
        s = v.sum()
        diff = target_sum - s
        if abs(diff) <= tol:
            break
        # Determine free indices where v not at bounds
        free = (v > low + tol) & (v < high - tol)
        num_free = free.sum()
        if num_free == 0:
            # cannot adjust; distribute equally naive
            v += diff / len(v)
            v = np.clip(v, low, high)
        else:
            delta = diff / num_free
            v[free] += delta
            v = np.clip(v, low, high)
    return v

import numpy as np
import time
import random
import math

def _project_to_feasible(v, target_sum, low=0.0, high=1.0, max_iter=200, tol=1e-12):
    """Project vector v onto the box [low,high] and sum constraint sum(v)=target_sum."""
    v = np.clip(v, low, high)
    for _ in range(max_iter):
        s = v.sum()
        diff = target_sum - s
        if abs(diff) <= tol:
            break
        free = (v > low + tol) & (v < high - tol)
        nfree = free.sum()
        if nfree == 0:
            # fallback: distribute diff uniformly and clip again
            v += diff / len(v)
            v = np.clip(v, low, high)
        else:
            delta = diff / nfree
            v[free] += delta
            v = np.clip(v, low, high)
    return v

def _random_feasible(n, target_sum):
    v = np.random.rand(n)
    v = v * (target_sum / v.sum())
    v = _project_to_feasible(v, target_sum)
    return v

def _compute_c5(h, dx):
    # correlation of h with (1-h); max element times dx
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _propose_transfer(h, max_transfer):
    """Transfer a small amount of mass from a positive entry to a non‑saturated entry."""
    n = h.shape[0]
    src_idx = np.where(h > 1e-12)[0]
    tgt_idx = np.where(h < 1.0 - 1e-12)[0]
    if src_idx.size == 0 or tgt_idx.size == 0:
        return None
    src = np.random.choice(src_idx)
    tgt = np.random.choice(tgt_idx)
    amount = min(h[src], 1.0 - h[tgt], max_transfer)
    if amount <= 1e-12:
        return None
    delta = np.random.uniform(0.0, amount)
    h_new = h.copy()
    h_new[src] -= delta
    h_new[tgt] += delta
    return h_new

def _propose_targeted_transfer(h, max_transfer):
    """Reduce the contribution of the currently worst lag by moving mass along that lag."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    lag = int(np.argmax(corr) - (n - 1))  # lag where C5 is attained
    if lag >= 0:
        i_vals = np.arange(n - lag)
        contrib = h[i_vals] * (1.0 - h[i_vals + lag])
        src = i_vals[np.argmax(contrib)]
        tgt = src + lag
    else:
        lag_abs = -lag
        i_vals = np.arange(lag_abs, n)
        contrib = h[i_vals] * (1.0 - h[i_vals - lag_abs])
        src = i_vals[np.argmax(contrib)]
        tgt = src - lag_abs
    amount = min(h[src], 1.0 - h[tgt], max_transfer)
    if amount <= 1e-12:
        return None
    delta = np.random.uniform(0.0, amount)
    h_new = h.copy()
    h_new[src] -= delta
    h_new[tgt] += delta
    return h_new

    The return tuple is (h_values, c5_bound, n_points).
