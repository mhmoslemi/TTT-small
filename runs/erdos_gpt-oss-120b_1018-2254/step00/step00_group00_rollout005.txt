import numpy as np
import time

def evaluate_overlap(h):
    """Return C5 value for a given step function h (array)."""
    N = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / N)


def project_simplex(v, target):
    """
    Project vector v onto the set {x | 0 <= x_i <= 1, sum(x)=target}.
    Uses a simple bisection on the shift t in x = clip(v - t,0,1).
    """
    v = np.asarray(v, dtype=float)
    lo, hi = -1.0, 2.0
    for _ in range(60):
        t = (lo + hi) * 0.5
        proj = np.clip(v - t, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = t
        else:
            hi = t
    return np.clip(v - t, 0.0, 1.0)


def tournament(pop, scores, rng, size=3):
    """Return index of the best individual among `size` random contestants."""
    idx = rng.integers(0, pop.shape[0], size=size)
    best = idx[0]
    best_score = scores[best]
    for i in idx[1:]:
        if scores[i] < best_score:
            best = i
            best_score = scores[i]
    return best
