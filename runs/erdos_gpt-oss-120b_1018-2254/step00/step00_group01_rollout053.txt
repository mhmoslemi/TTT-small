def anneal_opt(start_h, dx, target_sum, time_limit, rng):
    n = len(start_h)
    h = start_h.copy()
    best_h = h.copy()
    cur_val = compute_c5(h, dx)
    best_val = cur_val
    start_time = time.time()
    # set initial temperature
    T0 = 0.02 # adjust
    # total allowed iterations maybe unlimited; continue while time < limit
    while time.time() - start_time < time_limit:
        elapsed = time.time() - start_time
        T = T0 * (1 - elapsed / time_limit)
        # propose move
        i, j = rng.integers(n, size=2)
        if i == j: continue
        hi, hj = h[i], h[j]
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue
        delta = (rng.random() * (high - low) + low)
        # create new candidate
        new_h = h.copy()
        new_h[i] = hi + delta
        new_h[j] = hj - delta
        new_val = compute_c5(new_h, dx)
        delta_val = new_val - cur_val
        if delta_val < 0 or rng.random() < np.exp(-delta_val / max(T, 1e-12)):
            h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
    return best_h, best_val

    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return h
    # find indices not at bound
    for i in range(len(h)):

            h[i] += add
            diff -= add

import time
import numpy as np

def _c5(h, dx):
    """Return max_{k} ∫ h(x)(1-h(x+k)) dx approximated by discrete correlation."""
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _fix_sum(h, target):
    """Project h onto [0,1] box while enforcing sum == target (small residuals are fixed)."""
    cur = np.sum(h)
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # adjust entries that are not at a bound
    for i in range(len(h)):
        if diff > 0 and h[i] < 1.0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        elif diff < 0 and h[i] > 0.0:
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
        if abs(diff) < 1e-12:
            break
    return h

def _random_feasible(n, target, rng):
    """Create a feasible vector of length n with sum == target (binary plus one fractional)."""
    whole = int(np.floor(target))
    frac = target - whole
    h = np.zeros(n, dtype=float)
    perm = rng.permutation(n)
    h[perm[:whole]] = 1.0
    if frac > 0:
        h[perm[whole]] = frac
    return h

def _anneal(start, dx, target, time_budget, rng):
    """Simulated‑annealing style search starting from *start*."""
    n = len(start)
    h = start.copy()
    cur_val = _c5(h, dx)
    best_h = h.copy()
    best_val = cur_val
    start_time = time.time()
    T0 = 0.08  # initial temperature

    while time.time() - start_time < time_budget:
        elapsed = time.time() - start_time
        T = T0 * (1.0 - elapsed / time_budget)  # linear cooling
        # propose a feasible pair move
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        hi, hj = h[i], h[j]
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue
        delta = rng.random() * (high - low) + low
        new_h = h.copy()
        new_h[i] = hi + delta
        new_h[j] = hj - delta
        new_val = _c5(new_h, dx)
        dval = new_val - cur_val
        if dval < 0.0 or rng.random() < np.exp(-dval / max(T, 1e-12)):
            h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()

    best_h = _fix_sum(best_h, target)
    best_val = _c5(best_h, dx)
    return best_h, best_val
