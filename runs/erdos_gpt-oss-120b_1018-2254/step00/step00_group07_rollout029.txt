import time
import numpy as np
import scipy.optimize as opt

# Global placeholders – will be set in run()
N_POINTS = None
TARGET_SUM = None
DX = None
BOUNDS = None
CONSTRAINTS = None

def objective_t(x):
    """Objective: minimize the last variable t."""
    return x[-1]

def sum_eq_constraint(x):
    """Equality: sum of h equals TARGET_SUM."""
    return np.sum(x[:N_POINTS]) - TARGET_SUM

def corr_ineq_constraint(x):
    """Inequality: t >= correlation of h with (1-h)."""
    h = x[:N_POINTS]
    t = x[-1]
    c = np.correlate(h, 1 - h, mode="full") * DX
    return t - c

def project_to_sum_bounds(v, target_sum):
    """Project a vector onto the box [0,1] with a prescribed sum."""
    v = np.clip(v, 0.0, 1.0)
    n = v.shape[0]
    diff = target_sum - np.sum(v)
    for _ in range(100):
        if abs(diff) < 1e-12:
            break
        v = v + diff / n
        v = np.clip(v, 0.0, 1.0)
        diff = target_sum - np.sum(v)
    return v

def generate_random_feasible(n, target_sum, rng):
    """Create a random feasible h (box constraints and sum)."""
    v = rng.random(n)
    return project_to_sum_bounds(v, target_sum)

def run(seed=42, budget_s=1100, **kwargs):
    global N_POINTS, TARGET_SUM, DX, BOUNDS, CONSTRAINTS
    start = time.time()
    rng = np.random.default_rng(seed)

    # ---- obtain initial h -------------------------------------------------
    try:
        init = initial_h_values  # provided by the harness
        n = int(init.shape[0])
        init = np.asarray(init, dtype=float).copy()
    except NameError:
        n = 80
        init = np.full(n, 0.5, dtype=float)

    N_POINTS = n
    TARGET_SUM = n / 2.0
    DX = 2.0 / n

    # Ensure feasibility of the starting vector
    init = project_to_sum_bounds(init, TARGET_SUM)

    # ---- optimizer configuration ------------------------------------------
    BOUNDS = [(0.0, 1.0) for _ in range(n)] + [(0.0, 1.0)]
    CONSTRAINTS = [
        {"type": "eq", "fun": sum_eq_constraint},
        {"type": "ineq", "fun": corr_ineq_constraint},
    ]

    # ---- evaluate initial solution -----------------------------------------
    best_h = init.copy()
    best_score = evaluate_erdos_solution(best_h)

    # ---- single SLSQP run from the current best ----------------------------
    t_guess = best_score
    x0 = np.concatenate([best_h, [t_guess]])
    res = opt.minimize(
        fun=objective_t,
        x0=x0,
        method="SLSQP",
        bounds=BOUNDS,
        constraints=CONSTRAINTS,
        options={"maxiter": 600, "ftol": 1e-9, "disp": False},
    )
    if res.success:
        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        h_opt = project_to_sum_bounds(h_opt, TARGET_SUM)
        score = evaluate_erdos_solution(h_opt)
        if score < best_score:
            best_h, best_score = h_opt, score

    # ---- additional random restarts ---------------------------------------
    max_restarts = 10
    for _ in range(max_restarts):
        if time.time() - start > budget_s * 0.95:
            break
        guess = generate_random_feasible(n, TARGET_SUM, rng)
        t_guess = evaluate_erdos_solution(guess)
        x0 = np.concatenate([guess, [t_guess]])
        res = opt.minimize(
            fun=objective_t,
            x0=x0,
            method="SLSQP",
            bounds=BOUNDS,
            constraints=CONSTRAINTS,

    best_h = project_to_sum_bounds(best_h, TARGET_SUM)
    final_score = evaluate_erdos_solution(best_h)

    return (best_h, final_score, n)
