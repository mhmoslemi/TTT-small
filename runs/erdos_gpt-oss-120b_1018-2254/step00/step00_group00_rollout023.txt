    return float(np.max(corr) * (2.0 / h.size))

def _random_feasible_h(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a random feasible h: values in [0,1], sum == n/2."""
    h = np.zeros(n, dtype=np.float64)
    # integer part: floor(n/2) ones
    k = n // 2
    ones_idx = rng.choice(n, size=k, replace=False)
    h[ones_idx] = 1.0
    target = n / 2.0
    cur = h.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # place the remaining fractional mass on a zero entry
        zero_idx = rng.choice(np.where(h == 0)[0])
        h[zero_idx] = diff
    return h

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Determine number of points and initial vector
    n = kwargs.get("n_points")
    if n is None:
        try:
            h_cur = np.array(initial_h_values, copy=True)
            n = h_cur.size
        except NameError:
            n = 45
            h_cur = _random_feasible_h(n, rng)
    else:
        n = int(n)
        try:
            if len(initial_h_values) == n:
                h_cur = np.array(initial_h_values, copy=True)
            else:
                h_cur = _random_feasible_h(n, rng)
        except NameError:
            h_cur = _random_feasible_h(n, rng)

    # Enforce exact sum constraint (tiny numeric drift)
    target_sum = n / 2.0
    cur_sum = h_cur.sum()
    if abs(cur_sum - target_sum) > 1e-10:
        diff = target_sum - cur_sum
        h_cur[0] = np.clip(h_cur[0] + diff, 0.0, 1.0)

    # Initial scores & best solution
    best_h = h_cur.copy()
    best_score = _c5_score(best_h)
    cur_score = best_score

    # Simulated‑annealing parameters
    start_T = best_score if best_score > 0 else 0.5
    end_T = 1e-6
    stagnation = 0
    max_stagnation = 2000               # restart after this many non‑improving steps
    target_bound = 0.3808                # desired upper bound

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.97:
            break

        # Linear temperature schedule
        prog = elapsed / budget_s
        T = start_T * (1.0 - prog) + end_T * prog

        # Propose a two‑index move preserving the sum
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        low = max(-h_cur[i], -(1.0 - h_cur[j]))
        high = min(1.0 - h_cur[i], h_cur[j])
        if low >= high:
            continue
        delta = rng.uniform(low, high)

        # Apply move
        h_cur[i] += delta
        h_cur[j] -= delta
        new_score = _c5_score(h_cur)

        # Acceptance test
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / (T + 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_cur.copy()
                stagnation = 0
                if best_score <= target_bound:
                    break
            else:
                stagnation += 1
        else:
            # Revert move
            h_cur[i] -= delta
            h_cur[j] += delta
            stagnation += 1

        # Restart if stuck
        if stagnation >= max_stagnation:
            h_cur = _random_feasible_h(n, rng)
            cur_score = _c5_score(h_cur)
            stagnation = 0

    # Final evaluation to avoid stale rounding errors
    final_score = _c5_score(best_h)
    return best_h.tolist(), float(final_score), n
