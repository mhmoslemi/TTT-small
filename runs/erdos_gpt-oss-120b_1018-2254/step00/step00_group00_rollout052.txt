    y = np.clip(x, 0, 1)

    # Compute current sum
    s = y.sum()
    if abs(s - target) < 1e-9:
        return y
    # Determine direction
    # We'll use algorithm: while (s != target) and there are free variables
    # For increasing: need to add diff = target - s >0. We'll distribute diff equally across entries not at 1.
    # For decreasing: subtract diff across entries not at 0.
    # But we must not exceed bounds.
    # Use loop:
    free = np.ones_like(y, dtype=bool)
    diff = target - s
    # max iterations maybe 100
    for _ in range(100):
        if abs(diff) < 1e-9:
            break
        free_idxs = np.where(free)[0]
        if free_idxs.size == 0:
            break
        # Compute feasible amount to add/subtract per free variable
        # For increase: each free can accept at most (1 - y[i])
        # For decrease: each free can give at most y[i]
        if diff > 0:
            capacities = 1 - y[free_idxs]
            total_capacity = capacities.sum()
            if total_capacity < 1e-12:
                break
            # allocate proportionally (or equally)
            add = diff * capacities / total_capacity  # proportionally to capacity
            y[free_idxs] += add
            # Clip any overshoot
            y = np.clip(y, 0, 1)
        else:

            total_capacity = capacities.sum()
            if total_capacity < 1e-12:
                break

        s = y.sum()
        diff = target - s
        # Update free mask: those at bounds cannot be adjusted further

def project_bounds_sum(v, target):
    # Clip to [0,1]
    x = np.clip(v, 0.0, 1.0)
    n = len(x)
    # iterate to satisfy sum constraint
    for _ in range(100):
        s = x.sum()
        diff = target - s
        if abs(diff) < 1e-9:
            break
        # Determine indices that can be adjusted
        if diff > 0:
            # Need to increase sum; consider indices where x < 1
            free_idx = np.where(x < 1.0 - 1e-12)[0]
            if free_idx.size == 0:
                break
            # Max possible increase per index
            capacities = 1.0 - x[free_idx]
            total_capacity = capacities.sum()
            if total_capacity <= 0:
                break
            # Distribute diff proportionally to capacities
            # use scaling factor gamma = diff / total_capacity
            # Actually we can add to each free_idx: add = diff * (capacities / total_capacity)
            add = diff * (capacities / total_capacity)
            x[free_idx] += add
            # Clip any overshoot due to numerical issues
            x = np.clip(x, 0.0, 1.0)
        else:
            # Need to decrease sum; consider indices where x > 0
            free_idx = np.where(x > 1e-12)[0]
            if free_idx.size == 0:
                break
            capacities = x[free_idx]  # amount we can remove
            total_capacity = capacities.sum()
            if total_capacity <= 0:
                break
            sub = diff * (capacities / total_capacity)  # diff negative => sub negative
            x[free_idx] += sub
            x = np.clip(x, 0.0, 1.0)
    # final enforce sum due to possible rounding: compute final diff & adjust smallest slack
    s = x.sum()
    diff = target - s
    if abs(diff) > 1e-6:
        # Distribute diff to any index not at bound
        # find indices with some slack
        slack_up = np.where(x < 1.0 - 1e-8)[0]
        slack_down = np.where(x > 1e-8)[0]
        if diff > 0 and slack_up.size > 0:
            i = slack_up[0]
            x[i] = min(1.0, x[i] + diff)
        elif diff < 0 and slack_down.size > 0:
            i = slack_down[0]
            x[i] = max(0.0, x[i] + diff)
    # final clip
    return np.clip(x, 0.0, 1.0)

    lo = -1.0

        return np.clip(v + lam, 0.0, 1.0).sum()

import numpy as np
import time

def _project_bounds_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto [0,1]^n with sum(v)=target using a binary search on a
    common offset (Lagrange multiplier)."""
    lo, hi = -1.0, 1.0

    def _s(lam):
        return np.clip(v + lam, 0.0, 1.0).sum()

    # expand the interval until the target lies between the sums
    while _s(lo) > target:
        lo *= 2.0
    while _s(hi) < target:
        hi *= 2.0

    for _ in range(60):  # enough for double precision
        mid = (lo + hi) * 0.5
        s = _s(mid)
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(v + lam, 0.0, 1.0)

            old_x = np.linspace(0.0, 2.0, base_vec.size, endpoint=False)
            new_x = np.linspace(0.0, 2.0, n_points, endpoint=False)
            parent = np.interp(new_x, old_x, base_vec)
        parent = _project_bounds_sum(parent, target_sum)

    # --- GA parameters ------------------------------------------------
    POP_SIZE = 80
    MUT_RATE = 0.2          # per‑gene mutation probability
    MUT_SIGMA = 0.15        # mutation strength
    CROSS_RATE = 0.5        # blending weight range (uniform)

    # --- population initialization ------------------------------------
    pop = []
    if parent is not None:
        pop.append(parent)
    while len(pop) < POP_SIZE:
        pop.append(_random_feasible(rng, n_points, target_sum))

    scores = np.array([_evaluate_h(ind, dx) for ind in pop])
    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_score = float(scores[best_idx])

    # --- main evolutionary loop ---------------------------------------
    time_limit = start + budget_s * 0.93
    while time.time() < time_limit:
        new_pop = [best_h.copy()]                     # elitism
        while len(new_pop) < POP_SIZE:
            # tournament selection
            ids = rng.choice(POP_SIZE, size=3, replace=False)
            p1 = pop[ids[int(np.argmin(scores[ids]))]]
            ids = rng.choice(POP_SIZE, size=3, replace=False)
            p2 = pop[ids[int(np.argmin(scores[ids]))]]

            # uniform blending crossover
            alpha = rng.random(n_points)
            child = alpha * p1 + (1.0 - alpha) * p2

            # mutation
            mut_mask = rng.random(n_points) < MUT_RATE
            if mut_mask.any():
                child[mut_mask] += rng.normal(0.0, MUT_SIGMA, size=mut_mask.sum())

            # repair to feasibility
            child = _project_bounds_sum(child, target_sum)
            new_pop.append(child)

        pop = new_pop
        scores = np.array([_evaluate_h(ind, dx) for ind in pop])
        cur_best = int(np.argmin(scores))
        cur_score = float(scores[cur_best])
        if cur_score < best_score:
            best_score = cur_score
            best_h = pop[cur_best].copy()
            if best_score <= 0.3808:
                break

    # --- local refinement ---------------------------------------------
    refine_limit = start + budget_s * 0.97
    while time.time() < refine_limit:
        cand = best_h + rng.normal(0.0, 0.05, size=n_points)
        cand = _project_bounds_sum(cand, target_sum)
        cand_score = _evaluate_h(cand, dx)
        if cand_score < best_score:
            best_score = cand_score
            best_h = cand
            if best_score <= 0.3808:
                break

    # final safety projection and evaluation
    best_h = _project_bounds_sum(best_h, target_sum)
    final_c5 = _evaluate_h(best_h, dx)

    return best_h, final_c5, n_points
