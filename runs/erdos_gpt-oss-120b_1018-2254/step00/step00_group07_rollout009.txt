def targeted_move(h, lag, step, rng):
    n = len(h)
    if lag == 0:
        return h, False
    # Determine possible i range:
    i_min = max(0, -lag)
    i_max = min(n-1, n-1 - lag)
    if i_max < i_min:
        return h, False
    i = rng.integers(i_min, i_max+1)
    j = i + lag
    # Determine max increase for both
    inc_max = min(1.0 - h[i], 1.0 - h[j])
    if inc_max <= 0:
        return h, False
    # Choose donor
    donor_candidates = np.where(h > 1e-8)[0]
    donor_candidates = donor_candidates[(donor_candidates != i) & (donor_candidates != j)]
    if donor_candidates.size == 0:
        return h, False
    d = rng.choice(donor_candidates)
    # Max decrease from donor to provide 2*delta
    dec_max = h[d] / 2.0
    if dec_max <= 0:
        return h, False
    delta = min(step, inc_max, dec_max)
    if delta <= 0:
        return h, False
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] += delta
    h_new[d] -= 2*delta
    # Clip small negative due to floating errors
    h_new = np.clip(h_new, 0.0, 1.0)
    return h_new, True
```

import time
import numpy as np

def compute_c5(h: np.ndarray, dx: float) -> float:
    """Return max_k ∫ h(x)(1-h(x+k)) dx approximated on the grid."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * dx


def max_lag_gradient(h: np.ndarray):
    """Gradient of the correlation at the lag attaining the current maximum."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx_max = int(np.argmax(corr))
    lag = idx_max - (n - 1)

    # grad_i = (1 - h[i+lag]) if i+lag in range else 0
    #          - h[i-lag]          if i-lag in range else 0
    grad = np.zeros_like(h)
    idx = np.arange(n)

    j = idx + lag
    mask = (j >= 0) & (j < n)
    grad[mask] += 1.0 - h[j[mask]]

    j2 = idx - lag
    mask2 = (j2 >= 0) & (j2 < n)
    grad[mask2] -= h[j2[mask2]]

    return grad, lag


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise h
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        # fallback: half ones, half zeros, shuffled
        n_default = 80
        h = np.zeros(n_default, dtype=np.float64)
        h[: n_default // 2] = 1.0
        rng.shuffle(h)

    n = h.size
    target_sum = n / 2.0
    # Project onto the feasible box-simplex (clamp + fix sum)
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if abs(cur - target_sum) > 1e-12:
        # adjust a single free coordinate
        free = np.where((h > 0) & (h < 1))[0]
        if free.size == 0:
            free = np.arange(n)
        i = free[0]
        h[i] += target_sum - cur
        h[i] = np.clip(h[i], 0.0, 1.0)

    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    start_time = time.time()
    it = 0
    # step size schedule (large at start, small near timeout)
    max_step = 0.04
    min_step = 0.001

    while time.time() - start_time < budget_s - 0.2:
        it += 1
        elapsed = time.time() - start_time
        step = max_step * (1.0 - elapsed / budget_s) + min_step

        # ------------------------------------------------------------------
        # Decide move type
        # ------------------------------------------------------------------
        move_type = rng.random()
        # 0-0.5 : directed gradient move (if possible)
        # 0.5-0.8 : simple pair transfer
        # 0.8-1.0 : targeted move based on max lag
        if move_type < 0.5:
            # Gradient directed transfer
            grad, lag = max_lag_gradient(h)
            # indices where decreasing h would lower the max overlap
            donors = np.where((grad > 1e-12) & (h > 1e-12))[0]
            receivers = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
            if donors.size and receivers.size:
                d = rng.choice(donors)
                r = rng.choice(receivers)
                delta = min(step, h[d], 1.0 - h[r])
                if delta > 0:
                    h_new = h.copy()
                    h_new[d] -= delta
                    h_new[r] += delta
                    c5_new = compute_c5(h_new, dx)
                    if c5_new < best_c5 - 1e-9:
                        h = h_new
                        best_h = h_new.copy()
                        best_c5 = c5_new
                    continue

        if move_type < 0.8:
            # Simple stochastic mass transfer between two random positions
            i, j = rng.choice(n, size=2, replace=False)
            # direction i -> j
            delta_max = min(1.0 - h[j], h[i])
            if delta_max > 1e-12:
                delta = rng.uniform(0.0, min(delta_max, step))
                h_new = h.copy()
                h_new[i] -= delta
                h_new[j] += delta
                c5_new = compute_c5(h_new, dx)
                if c5_new < best_c5 - 1e-9:
                    h = h_new
                    best_h = h_new.copy()
                    best_c5 = c5_new
                continue
            # try opposite direction
            delta_max = min(1.0 - h[i], h[j])
            if delta_max > 1e-12:
                delta = rng.uniform(0.0, min(delta_max, step))
                h_new = h.copy()
                h_new[j] -= delta
                h_new[i] += delta
                c5_new = compute_c5(h_new, dx)
                if c5_new < best_c5 - 1e-9:
                    h = h_new
                    best_h = h_new.copy()
                    best_c5 = c5_new
                continue

        # Targeted move focusing on the current worst shift
        corr = np.correlate(h, 1 - h, mode="full")
        idx_max = int(np.argmax(corr))
        lag = idx_max - (n - 1)
        if lag != 0:
            i_min = max(0, -lag)
            i_max = min(n - 1, n - 1 - lag)
            if i_max >= i_min:
                i = rng.integers(i_min, i_max + 1)
                j = i + lag
                inc_max = min(1.0 - h[i], 1.0 - h[j])
                if inc_max > 1e-12:
                    # choose a donor distinct from i,j with enough mass
                    candidates = np.where((h > 1e-12) & (np.arange(n) != i) & (np.arange(n) != j))[0]
                    if candidates.size:
                        d = rng.choice(candidates)
                        dec_max = h[d] / 2.0
                        delta = min(step, inc_max, dec_max)
                        if delta > 0:
                            h_new = h.copy()
                            h_new[i] += delta
                            h_new[j] += delta
                            h_new[d] -= 2.0 * delta
                            # tidy numeric noise
                            h_new = np.clip(h_new, 0.0, 1.0)
                            c5_new = compute_c5(h_new, dx)
                            if c5_new < best_c5 - 1e-9:
                                h = h_new
                                best_h = h_new.copy()
                                best_c5 = c5_new
                                continue

        # Occasionally inject a noisy restart around the best found
        if it % 7000 == 0 and rng.random() < 0.12:
            noise = rng.normal(scale=0.03, size=n)
            h_new = best_h + noise
            h_new = np.clip(h_new, 0.0, 1.0)
            # re‑project onto the exact sum constraint
            cur = h_new.sum()
            diff = target_sum - cur
            # distribute the correction onto a free coordinate
            free = np.where((h_new > 1e-12) & (h_new < 1.0 - 1e-12))[0]
            if free.size == 0:
                free = np.arange(n)
            idx = free[0]
            h_new[idx] += diff
            h_new[idx] = np.clip(h_new[idx], 0.0, 1.0)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                h = h_new
                best_h = h_new.copy()
                best_c5 = c5_new

    # ------------------------------------------------------------------
    # Final sanity projection
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    cur = best_h.sum()
    if abs(cur - target_sum) > 1e-10:
        # adjust a non‑boundary entry if possible
        adj_idx = np.where((best_h > 1e-12) & (best_h < 1.0 - 1e-12))[0]
        if adj_idx.size == 0:
            adj_idx = np.arange(n)
        i = adj_idx[0]
        best_h[i] += target_sum - cur
        best_h[i] = np.clip(best_h[i], 0.0, 1.0)

    final_c5 = compute_c5(best_h, dx)
    return best_h, final_c5, n
