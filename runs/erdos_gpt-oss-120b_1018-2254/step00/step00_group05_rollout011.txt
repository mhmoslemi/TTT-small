def project_onto_box_sum(v, S, lower=0.0, upper=1.0, max_iter=100, tol=1e-9):
    lo = -max(v) - 1.0   # ensure all clipped at lower bound
    hi = upper - min(v) + 1.0 # ensure all at upper
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        y = np.clip(v + lam, lower, upper)
        sum_y = y.sum()
        if abs(sum_y - S) < tol:
            return y
        if sum_y > S:
            hi = lam
        else:
            lo = lam
    # after loop do final clip
    return np.clip(v + (lo+hi)/2, lower, upper)

def subgradient_step(h, step_size):
    N = h.shape[0]
    corr = np.correlate(h, 1-h, mode='full')
    s_idx = np.argmax(corr)
    shift = s_idx - (N-1)
    idx = np.arange(N)
    # term1: (1 - h[idx+shift]) if valid else 0
    pos = idx + shift
    term1 = np.zeros(N)
    mask1 = (pos >= 0) & (pos < N)
    term1[mask1] = 1 - h[pos[mask1]]
    # term2: h[idx - shift] if valid else 0
    neg = idx - shift
    term2 = np.zeros(N)
    mask2 = (neg >= 0) & (neg < N)
    term2[mask2] = h[neg[mask2]]
    grad = term1 - term2
    # descent
    h_new = h - step_size * grad
    # projection
    h_proj = project_onto_box_sum(h_new, N/2.0)
    return h_proj, corr.max()

def random_binary(N):
    # produce random binary vector with exactly N/2 ones
    indices = np.arange(N)
    ones = np.random.choice(N, N//2, replace=False)
    h = np.zeros(N, dtype=float)
    h[ones] = 1.0
    return h

def binary_hill_climb(h0, max_moves, max_time=None):
    # local search with random swaps
    h = h0.copy()
    N = len(h)
    current_raw = np.max(np.correlate(h, 1-h, mode='full'))
    best_h = h.copy()
    best_raw = current_raw
    start = time.time()
    attempts = 0
    while attempts < max_moves:
        # optionally check time
        if max_time is not None and time.time() - start > max_time:
            break
        # select random ones and zeros positions
        idx_one = np.flatnonzero(h > 0.5) # positions where h==1
        idx_zero = np.flatnonzero(h < 0.5)
        if len(idx_one) == 0 or len(idx_zero) == 0:
            break
        i = np.random.choice(idx_one)
        j = np.random.choice(idx_zero)
        # swap
        h[i], h[j] = 0.0, 1.0
        new_raw = np.max(np.correlate(h, 1-h, mode='full'))
        # accept if better
        if new_raw < current_raw:
            current_raw = new_raw
            # update best
            if new_raw < best_raw:
                best_raw = new_raw
                best_h = h.copy()
        else:
            # reject: swap back
            h[i], h[j] = 1.0, 0.0
        attempts += 1
    return best_h, best_raw

def project_onto_box_sum(v, target, lower=0.0, upper=1.0, max_iter=50, tol=1e-9):
    lo = -np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        y = np.clip(v + lam, lower, upper)
        s = y.sum()
        if abs(s - target) < tol:

import numpy as np
import time
import random
import math

def _project_box_sum(v: np.ndarray, target: float, lower: float = 0.0, upper: float = 1.0,
                     max_iter: int = 60, tol: float = 1e-9) -> np.ndarray:
    lo = -np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        y = np.clip(v + lam, lower, upper)
        s = y.sum()
        if abs(s - target) < tol:
            return y
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) / 2.0, lower, upper)


def _subgradient_step(h: np.ndarray, step: float) -> tuple:
    N = h.shape[0]
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (N - 1)   # shift value (can be negative)
    shift = idx
    ar = np.arange(N)
    # term (1 - h[i+shift])
    pos = ar + shift
    term1 = np.zeros(N, dtype=h.dtype)
    mask1 = (pos >= 0) & (pos < N)
    term1[mask1] = 1.0 - h[pos[mask1]]
    # term h[i-shift]
    neg = ar - shift
    term2 = np.zeros(N, dtype=h.dtype)
    mask2 = (neg >= 0) & (neg < N)
    term2[mask2] = h[neg[mask2]]
    grad = term1 - term2
    h_new = h - step * grad
    h_proj = _project_box_sum(h_new, N / 2.0)
    return h_proj, np.max(corr)


def _gradient_descent(h0: np.ndarray, max_iter: int, step0: float) -> tuple:
    h = h0.copy()
    best_h = h.copy()
    best_raw = np.max(np.correlate(h, 1 - h, mode="full"))
    step = step0
    for _ in range(max_iter):
        h, cur_raw = _subgradient_step(h, step)
        step *= 0.98
        if cur_raw < best_raw:
            best_raw = cur_raw
            best_h = h.copy()
    return best_h, best_raw


def _random_binary(N: int) -> np.ndarray:
    h = np.zeros(N, dtype=np.float64)
    ones = np.random.choice(N, N // 2, replace=False)
    h[ones] = 1.0
    return h


def _thue_morse(N: int) -> np.ndarray:
    seq = np.array([0.0])
    while seq.size < N:
        seq = np.concatenate([seq, 1.0 - seq])
    return seq[:N]


def _binary_simulated_annealing(h0: np.ndarray, max_iter: int,
                               time_limit: float = None) -> tuple:
    """Swap a 1 with a 0, accept with Metropolis rule."""
    N = h0.shape[0]
    cur = h0.copy()
    cur_raw = np.max(np.correlate(cur, 1 - cur, mode="full"))
    best = cur.copy()
    best_raw = cur_raw
    start = time.time()
    # temperature schedule – start high enough to allow uphill moves
    temp0 = N * 0.25
    for it in range(max_iter):
        if time_limit is not None and (time.time() - start) >= time_limit:
            break
        temp = temp0 * (1.0 - it / max_iter)
        ones = np.flatnonzero(cur > 0.5)
        zeros = np.flatnonzero(cur < 0.5)
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        cur[i], cur[j] = 0.0, 1.0
        new_raw = np.max(np.correlate(cur, 1 - cur, mode="full"))
        delta = new_raw - cur_raw
        if delta <= 0 or random.random() < math.exp(-delta / (temp + 1e-12)):
            cur_raw = new_raw
            if new_raw < best_raw:
                best_raw = new_raw
                best = cur.copy()
        else:
            cur[i], cur[j] = 1.0, 0.0
    return best, best_raw


def _c5_bound(h: np.ndarray) -> tuple:
    N = h.shape[0]
    raw = np.max(np.correlate(h, 1 - h, mode="full"))
    bound = raw * (2.0 / N)
    return bound, raw
