```python
import numpy as np, time

def compute_c5(h, dx):
    # correlation
    # note: using full mode yields length 2*n-1, representing shift from -(n-1) to (n-1)
    # max over all
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def binary_random_balanced(rng, n):
    # generate binary array with exactly n/2 ones
    indices = rng.choice(n, n//2, replace=False)
    h = np.zeros(n)
    h[indices] = 1.0
    return h

def hill_climb(start_h, rng, max_iter, dx, time_limit):
    h = start_h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    n = len(h)
    # precompute positions of ones and zeros
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    # loop
    for it in range(max_iter):
        # pick random swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        # compute new c5
        c5 = compute_c5(h, dx)
        # if improvement or accept?
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            # update ones/zeros lists
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
        # optionally early exit if time
        if time.time() >= time_limit:
            break
    return best_h, best_c5
```

import numpy as np, time, math

def max_overlap(h):
    # Compute correlation O_k = np.correlate( h, 1 - h, mode='full')
    # O_k is integer count of mismatches (h_i=1 & shifted zero)
    # Return max(O_k) (as float scaled by dx)
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / len(h))

def hill_climb(start_h, rng, time_limit):
    # Simulated annealing style
    h = start_h.copy()
    n = len(h)
    current_score = max_overlap(h)
    best_h = h.copy()
    best_score = current_score
    
    # Precompute positions of 1s and 0s for swap
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    
    # temperature schedule
    T0 = 1.0
    # total allowed moves maybe compute the number of iterations given time
    start = time.time()
    while time.time() < time_limit:
        # maybe gradually lower temperature
        # use linear schedule by proportion of elapsed time
        elapsed = time.time() - start
        frac = elapsed / (time_limit - start)  # 0-1
        T = T0 * (1 - frac)  # linearly decreasing
        # choose random swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = max_overlap(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            # accept with probability exp(-delta / T)
            if T > 0:
                if rng.random() < math.exp(-delta / T):
                    accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                # update ones and zeros
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h < 0.5)[0]
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_score

import numpy as np
import time
import math

def _dx(n):
    return 2.0 / n

def _c5_score(h):
    n = h.shape[0]
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * _dx(n)

def _random_balanced(rng, n):
    """binary vector of length n with exactly n/2 ones."""
    idx = rng.choice(n, n // 2, replace=False)
    h = np.zeros(n, dtype=float)
    h[idx] = 1.0
    return h

def _local_optimize(h, rng, deadline):
    """ hill‑climb with exhaustive single‑swap search until no improvement or time runs out """
    n = h.shape[0]
    best_score = _c5_score(h)
    while time.time() < deadline:
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        # random order to avoid deterministic loops
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for i in ones:
            for j in zeros:
                # try swap i (1) <-> j (0)
                h[i] = 0.0
                h[j] = 1.0
                new_score = _c5_score(h)
                if new_score < best_score - 1e-12:
                    best_score = new_score
                    improved = True
                    break
                # revert swap
                h[i] = 1.0
                h[j] = 0.0
            if improved:
                break
        if not improved:
            break
    return h.copy(), best_score
