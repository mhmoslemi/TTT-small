def project_box_sum(v, target, tol=1e-9, max_iter=100):
    low = -np.max(v) - 2  # ensure low enough that v - low > 1 for all.
    high = np.max(v) + 2  # ensure high enough that v - high < 0 for all.
    # Actually we can set low = -5, high = 5
    for _ in range(max_iter):
        lam = (low + high) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            # lam too low (shift left), increase lam
            low = lam
        else:
            high = lam
    # Return final projection
    return np.clip(v - lam, 0, 1)

import numpy as np
import scipy.optimize as opt
import time

def project_to_sum(v, target):
    # Project v onto [0,1] box and sum = target
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -10.0, 10.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def soft_obj(v, alpha=30.0):
    # Smooth approximation to max of -corr (i.e., -min autocorrelation)
    corr = np.correlate(v, v, mode='full')
    x = -alpha * corr
    max_x = x.max()
    # stable log-sum-exp
    return max_x + np.log(np.exp(x - max_x).sum())

def triangular_candidate(n):
    idx = np.arange(n)
    center = (n - 1) / 2.0
    w = (n - 1) / 2.0
    h = np.maximum(0.0, 1.0 - np.abs(idx - center) / w)
    return h  # sum is already n/2

    if init is not None:
        h0 = project_to_sum(init.astype(float), target_sum)
        update_best(h0)

    h_tri = triangular_candidate(n_points)
    h_tri = project_to_sum(h_tri, target_sum)
    update_best(h_tri)

    h_uni = np.full(n_points, 0.5)
    h_uni = project_to_sum(h_uni, target_sum)
    update_best(h_uni)

    # ----------------------------------------------------------------------
    # Optimization settings
    bounds = [(0.0, 1.0)] * n_points
    cons = {'type': 'eq', 'fun': sum_constraint, 'args': (target_sum,)}
    opt_opts = {'maxiter': 250, 'ftol': 1e-9}
    alpha_val = 30.0  # smoothness parameter

    # ----------------------------------------------------------------------
    # Main stochastic restart loop
    restart = 0
    while True:
        if time.time() - start >= budget_s * 0.90:
            break
        if restart == 0 and init is not None:
            x0 = h0.copy()
        else:
            rnd = rng.random(n_points)
            x0 = project_to_sum(rnd, target_sum)

        res = opt.minimize(
            soft_obj,
            x0,
            args=(alpha_val,),
            method='SLSQP',
            bounds=bounds,
            constraints=[cons],
            options=opt_opts,
        )
        h_opt = project_to_sum(res.x, target_sum)
        update_best(h_opt)
        restart += 1

    # ----------------------------------------------------------------------
    # Simple pairwise hill climbing from the best found solution
    if best_h is None:
        best_h = h_uni.copy()
        best_c5 = compute_c5(best_h, dx)

    while True:
        if time.time() - start >= budget_s * 0.98:
            break
        i, j = rng.integers(0, n_points, size=2)
        if i == j:
            continue
        delta = (rng.random() - 0.5) * 0.1  # ±0.05
        if not (0.0 <= best_h[i] + delta <= 1.0):
            continue
        if not (0.0 <= best_h[j] - delta <= 1.0):
            continue
        new_h = best_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        c5_new = compute_c5(new_h, dx)
        if c5_new < best_c5:
            best_h = new_h
            best_c5 = c5_new

    # Final projection to guarantee exact feasibility
    best_h = project_to_sum(best_h, target_sum)
    final_c5 = compute_c5(best_h, dx)

    return (best_h, final_c5, n_points)
