def compute_gradient(h, k):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros_like(h)
    if k >= 0:
        # indices i with i + k < n:
        i = np.arange(0, n - k)
        g[i] += dx * (1 - h[i + k])
        # indices i with i - k >= 0 => i >= k:
        j = np.arange(k, n)
        # For these i, i - k in range.
        g[j] += dx * (-h[j - k])
    else:
        d = -k
        # i + k in range => i >= d:
        i = np.arange(d, n)
        g[i] += dx * (1 - h[i - d])
        # i - k >=0 => i + d < n => i in [0, n-d-1]:
        j = np.arange(0, n - d)
        g[j] += dx * (-h[j + d])
    return g

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    # Determine n_points: maybe choose 180
    n = 180
    # Initialize h
    if 'initial_h_values' in globals():
        # If length differs, resample, else use it
        init = initial_h_values
        n0 = init.shape[0]
        if n0 != n:
            # Interpolate
            x0 = np.linspace(0,2, n0, endpoint=False)
            x = np.linspace(0,2, n, endpoint=False)
            h = np.interp(x, x0, init)
            # Clip to [0,1] and adjust sum
            h = np.clip(h, 0, 1)
            # modify to have exactly n/2 sum: we can round to binary after scaling
            # However we can just random binary init
        else:

        # Even n, half ones binary
        h = np.zeros(n)
        h[:n//2] = 1
        np.random.shuffle(h)
    # Ensure sum equals n/2
    # If non binary, adjust scaling? Let's use binary (0/1) for even n.
    # Compute best
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5
    iterations = 0

    # Precompute indices of ones and zeros for faster pick
    # We'll update each iteration after a flip.
    while time.time() - start < budget_s - 0.5:
        # Get indices
        ones_idx = np.where(cur_h == 1)[0]
        zeros_idx = np.where(cur_h == 0)[0]
        if len(ones_idx)==0 or len(zeros_idx)==0:
            # Should not happen
            break
        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)
        # swap
        cur_h[i] = 0
        cur_h[j] = 1
        new_c5 = compute_c5(cur_h)
        # accept if better or with prob
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            # update best if better
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
        else:
            # reject - revert swap
            cur_h[i] = 1
            cur_h[j] = 0
        iterations += 1

    # At end, return best solution
    return best_h, best_c5, n

import numpy as np, time, math

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return corr.max() * dx, corr.argmax()

def project(h, target_sum):
    # project onto [0,1] and sum = target_sum
    # We'll use bisection for lambda
    n = len(h)
    lo = -1.0
    hi = 1.0
    # Actually lambda range may be [-max(h), max(1 - h)]
    # Use wide range:
    low = -np.max(h) - 1.0
    high = np.max(1 - h) + 1.0
    for _ in range(50):
        lam = (low + high) / 2.0
        x = np.clip(h - lam, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            low = lam
        else:
            high = lam
    return np.clip(h - (low+high)/2.0, 0.0, 1.0)

def random_transfer(h, max_delta):
    n = len(h)
    # sample i with h[i] > 0
    i = np.random.randint(n)
    attempts = 0
    while h[i] <= 1e-12 and attempts < 10:
        i = np.random.randint(n)
        attempts += 1
    if h[i] <= 1e-12:
        return None
    j = np.random.randint(n)
    attempts = 0
    while h[j] >= 1.0 - 1e-12 and attempts < 10:
        j = np.random.randint(n)
        attempts += 1
    if h[j] >= 1.0 - 1e-12:
        return None
    delta = np.random.rand() * min(max_delta, h[i], 1.0 - h[j])
    if delta <= 0:
        return None
    return i, j, delta

import time
import math
import numpy as np

def _compute_c5_and_shift(h):
    """Return (C5 value, integer shift) for the current h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(corr))
    max_val = corr[max_idx]
    c5 = max_val * dx
    shift = max_idx - (n - 1)
    return c5, shift

def _swap_mismatch(h, shift):
    """Try to swap two entries that reduce the overlap for the given shift.
    Returns tuple of swapped indices (i, j) if a swap was performed, else None."""
    n = h.shape[0]
    d = abs(shift)
    if d == 0:
        return None
    if shift > 0:
        left_idx = np.arange(0, n - d)
        right_idx = left_idx + d
    else:  # shift < 0
        left_idx = np.arange(d, n)
        right_idx = left_idx - d
    left = h[left_idx]
    right = h[right_idx]
    # positions where left is 1 and right is 0
    mism_left = np.where((left > 0.5) & (right < 0.5))[0]
    # positions where left is 0 and right is 1
    mism_right = np.where((left < 0.5) & (right > 0.5))[0]
    if mism_left.size == 0 or mism_right.size == 0:
        return None
    i_loc = np.random.choice(mism_left)
    j_loc = np.random.choice(mism_right)
    i_abs = left_idx[i_loc]
    j_abs = left_idx[j_loc]
    h[i_abs], h[j_abs] = h[j_abs], h[i_abs]
    return i_abs, j_abs

def _random_swap_binary(h):
    """Swap a random 1 with a random 0. Returns swapped indices or None."""
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    if ones.size == 0 or zeros.size == 0:
        return None
    i = np.random.choice(ones)
    j = np.random.choice(zeros)
    h[i], h[j] = h[j], h[i]
    return i, j

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function on [0,2] with a small C₅ value."""
    np.random.seed(seed)
    start_time = time.time()
    # choose a modest size (≤1k) – bigger n makes the bound easier to improve
    n = 200                       # even, half‑ones = 100
    target_sum = n / 2.0

    # ----------------------------------------------------------------------
    # initialise h
    # ----------------------------------------------------------------------
    try:
        init = initial_h_values    # provided by the harness, may be None
    except NameError:
        init = None

    if init is not None:
        n0 = int(init.shape[0])
        if n0 != n:
            x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
            x = np.linspace(0.0, 2.0, n, endpoint=False)
            h_interp = np.interp(x, x0, init)
        else:
            h_interp = init.copy()
        # Turn the interpolated values into a binary (or near‑binary) vector
        order = np.argsort(h_interp)[::-1]
        h = np.zeros(n, dtype=np.float64)
        half = n // 2
        h[order[:half]] = 1.0
        if n % 2 == 1:
            # add the fractional remainder required by the integral constraint
            remainder = target_sum - half
            h[order[half]] = remainder
    else:
        # simple random binary start
        h = np.zeros(n, dtype=np.float64)
        half = n // 2
        h[:half] = 1.0
        np.random.shuffle(h)

    # ----------------------------------------------------------------------
    # main stochastic optimisation loop
    # ----------------------------------------------------------------------
    best_h = h.copy()
    best_c5, _ = _compute_c5_and_shift(best_h)
    last_improve_iter = 0
    iter_counter = 0
    restart_interval = 25000        # re‑initialise if progress stalls
    max_temp = 0.008                # initial temperature for simulated‑annealing

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 0.5:
            break

        c5, shift = _compute_c5_and_shift(h)

        # keep global best
        if c5 + 1e-12 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improve_iter = iter_counter

        # ------------------------------------------------------------------
        # try a targeted swap that directly reduces the worst shift
        # ------------------------------------------------------------------
        performed = _swap_mismatch(h, shift)
        if performed is not None:
            i_sw, j_sw = performed
            new_c5, _ = _compute_c5_and_shift(h)
            if new_c5 <= c5:

        # ------------------------------------------------------------------
        performed = _random_swap_binary(h)
        if performed is not None:
            i_sw, j_sw = performed
            new_c5, _ = _compute_c5_and_shift(h)
            if new_c5 <= c5:
                # improvement – keep it
                continue
            else:
                # possibly accept a worse move
                temperature = max(1e-6, max_temp * (1.0 - elapsed / budget_s))
                prob = math.exp(-(new_c5 - c5) / temperature)
                if np.random.rand() < prob:

                else:

        # occasional restart if we have not seen improvement for a while

    # final sanity projection (ensures exact sum & box constraints)
    # ----------------------------------------------------------------------
    # Project onto [0,1]∩{sum=h_target} using a simple water‑filling bisection.
    def _project(v, target):

            s = x.sum()
            if s > target:

            else:
                hi = lam
        return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

    best_h = _project(best_h, target_sum)
    best_c5, _ = _compute_c5_and_shift(best_h)

    return best_h, best_c5, n
