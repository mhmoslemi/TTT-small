import numpy as np, time, math

def evaluate_overlap(h):
    h = np.asarray(h, dtype=float)
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_onto_constraints(x, target_sum):
    # Clip
    x = np.clip(x, 0.0, 1.0)
    # Adjustment with water-filling
    # Use while loop with tolerance
    for _ in range(100):
        current_sum = x.sum()
        diff = target_sum - current_sum
        if abs(diff) < 1e-12:
            break
        # Identify free indices where pushing by diff could be feasible
        # Determine which indices not at bound in direction of diff
        if diff > 0:
            # need increase sum: can increase entries <1
            free = x < 1.0 - 1e-12
        else:
            # need decrease sum
            free = x > 1e-12
        n_free = np.count_nonzero(free)
        if n_free == 0:
            # cannot adjust; break
            break
        delta = diff / n_free
        # Adjust but ensure not exceed bounds
        # We'll add delta to free entries, then clip
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
    # final clipping
    return np.clip(x, 0.0, 1.0)
```

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, copy=True, dtype=float)

    else:
        n = 45

    target_sum = n / 2.0
    h0 = project_onto_constraints(h0, target_sum)
    best_h = h0.copy()
    best_c5 = evaluate_overlap(best_h)

    current_h = best_h.copy()
    current_c5 = best_c5

    start_time = time.time()
    # SA parameters
    T_initial = 0.1
    T_final = 1e-6
    # slowly exponential cooling per iteration: T = T_initial * (T_final/T_initial)^(iter/iter_max)
    # Determine iteration_max based on time budget; guess 1e6 iterations?
    # We'll compute cooling factor each iteration: T *= 0.9999 etc.
    T = T_initial
    cooling_rate = 0.9995  # per iteration

    # step size for delta
    step_initial = 0.2
    step = step_initial
    step_decay = 0.9999

    iteration = 0
    # We'll set a small safety margin to stop early.
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.99:  # stop with 1% buffer
            break
        # propose new candidate
        new_h = current_h.copy()
        # random i, j
        i, j = rng.integers(0, n, size=2)
        # ensure i != j
        while j == i:
            j = rng.integers(0, n)
        # random delta
        delta = (rng.random() - 0.5) * 2 * step  # uniform in [-step, step]
        # apply delta: increase h[i] and decrease h[j]
        new_h[i] += delta
        new_h[j] -= delta
        # project onto constraints
        new_h = project_onto_constraints(new_h, target_sum)
        # evaluate
        new_c5 = evaluate_overlap(new_h)
        # Acceptance decision
        delta_c5 = new_c5 - current_c5
        if delta_c5 < 0 or rng.random() < math.exp(-delta_c5 / max(T, 1e-12)):
            current_h = new_h
            current_c5 = new_c5
            # update best if improved
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
        # cooling
        T = max(T * cooling_rate, T_final)
        step = max(step * step_decay, 1e-4)

        # occasional random restart to avoid trap
        iteration += 1
        if iteration % 10000 == 0:
            if rng.random() < 0.1:  # 10% chance to restart from random feasible
                rand_h = rng.random(n)
                rand_h = project_onto_constraints(rand_h, target_sum)
                current_h = rand_h
                current_c5 = evaluate_overlap(current_h)

    # final ensure best meets constraints (fix sum)
    best_h = project_onto_constraints(best_h, target_sum)
    best_c5 = evaluate_overlap(best_h)

    return best_h, best_c5, n

def solve_binary_milp(N, time_limit):
    import cvxpy as cp, numpy as np
    # create variables
    h = cp.Variable(N, boolean=True)
    w = cp.Variable((N, N), boolean=True)
    M = cp.Variable()
    constraints = [cp.sum(h) == N/2]
    # AND constraints
    for i in range(N):
        for j in range(N):
            constraints.append(w[i,j] <= h[i])
            constraints.append(w[i,j] <= h[j])
            constraints.append(w[i,j] >= h[i] + h[j] - 1)
    # shift constraints
    for s in range(-N+1, N):
        sum_terms = [w[i, i+s] for i in range(N) if 0 <= i+s < N]
        constraints.append(cp.sum(cp.hstack(sum_terms)) >= M)
    # maybe M >= 0
    constraints.append(M >= 0)
    # objective: maximize M
    obj = cp.Maximize(M)
    prob = cp.Problem(obj, constraints)
    # Solve with optional time limit
    try:
        prob.solve(solver=cp.GLPK_MI, warm_start=True, eps_abs=1e-6, # set options?
                  # pass time limit if solver supports:
                  )
    except Exception:
        try:
            prob.solve(solver=cp.CBC, warm_start=True)
        except Exception:
            # fallback then None
            return None, None
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        return None, None
    h_val = np.array(h.value).astype(float)

def projected_gradient_descent(h_init, time_budget):
    start = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = evaluate_overlap(h)
    step = 0.1
    step_decay = 0.9995
    while time.time() - start < time_budget:
        # compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        shift = max_idx - (len(h)-1)
        # gradient
        grad = np.zeros_like(h)
        N = len(h)
        if shift >= 0:
            # part1
            grad[:N-shift] += 1 - h[shift:]
            # part2
            grad[shift:] -= h[:N-shift]
        else:
            d = -shift
            grad[d:] += 1 - h[:N-d]
            grad[:N-d] -= h[d:]
        # step
        h_new = h - step * grad
        h_new = project_onto_constraints(h_new, N/2)
        c5_new = evaluate_overlap(h_new)
        # accept if improve else maybe reject
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new
        else:
            # maybe accept with small probability
            if np.random.rand() < 0.01:
                h = h_new
        # decay step
        step = max(step * step_decay, 1e-4)
        # occasional random perturb:
        if np.random.rand() < 0.001:

import numpy as np
import time

def evaluate_overlap(h):
    """
    Compute C5 for a discretized step function h on [0,2].
    h: 1‑D array of length n, values in [0,1] and sum(h)=n/2.
    Returns max_k ∫ h(x)(1−h(x+k))dx approximated on the grid.
    """
    h = np.asarray(h, dtype=float)
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_onto_constraints(x, target_sum, tol=1e-9, max_iter=200):
    """
    Project a vector onto the feasible set:
        0 ≤ x_i ≤ 1  and  sum(x) = target_sum.
    Uses simple water‑filling: clip then distribute the residual.
    """
    x = np.clip(x, 0.0, 1.0)
    for _ in range(max_iter):
        cur_sum = x.sum()
        diff = target_sum - cur_sum
        if abs(diff) < tol:
            break
        # indices that can move in the needed direction
        if diff > 0:
            free = x < 1.0 - 1e-12
        else:
            free = x > 1e-12
        n_free = np.count_nonzero(free)
        if n_free == 0:
            break
        delta = diff / n_free
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
    return x

def _gradient_for_shift(h, shift):
    """
    Subgradient of f_shift(h)=∑ h_i(1−h_{i+shift}) at the point h.
    Returns a vector of length n.
    """
    n = h.size
    grad = np.zeros(n, dtype=float)
    if shift >= 0:
        # part where i+shift exists
        grad[:n-shift] += 1.0 - h[shift:]
        # part where i-shift exists
        grad[shift:] -= h[:n-shift]
    else:
        d = -shift
        grad[d:] += 1.0 - h[:n-d]
        grad[:n-d] -= h[d:]
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # -------------------------------------------------
    # Initial feasible vector
    # -------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = h0.size
    else:
        n = 45                     # fallback size
        h0 = rng.random(n)
    target_sum = n / 2.0
    h0 = project_onto_constraints(h0, target_sum)

    best_h = h0.copy()
    best_c5 = evaluate_overlap(best_h)
    curr_h = best_h.copy()
    curr_c5 = best_c5

    # -------------------------------------------------
    # Search parameters
    # -------------------------------------------------
    start_time = time.time()
    step = 0.12                 # initial step size for gradient moves
    step_decay = 0.9996
    prob_grad = 0.45            # probability to perform a gradient step
    prob_restart = 0.001        # occasional random restart
    prob_random = 0.005         # chance to accept a worse random move

    # -------------------------------------------------
    # Main optimisation loop
    # -------------------------------------------------
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.97:   # keep a safety margin
            break

        if rng.random() < prob_grad:
            # ---- Gradient descent on the active shift ----
            corr = np.correlate(curr_h, 1.0 - curr_h, mode='full')
            max_idx = int(np.argmax(corr))
            shift = max_idx - (n - 1)

            grad = _gradient_for_shift(curr_h, shift)

            # step opposite to the subgradient (we minimise)
            new_h = curr_h - step * grad
        else:
            # ---- Random perturbation around the current best ----
            perturb = (rng.random(n) - 0.5) * 2.0 * step
            new_h = best_h + perturb

        # Project back to feasibility
        new_h = project_onto_constraints(new_h, target_sum)

        new_c5 = evaluate_overlap(new_h)

        # Acceptance logic
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            curr_h = new_h.copy()
            curr_c5 = new_c5
        else:
            # keep the current point if it improved, otherwise occasionally
            # jump to a worse point to escape plateaus
            if new_c5 < curr_c5 or rng.random() < prob_random:
                curr_h = new_h.copy()
                curr_c5 = new_c5

        # Decay step size gradually
        step = max(step * step_decay, 1e-5)

        # Rare occasional restart from a fresh random feasible point
        if rng.random() < prob_restart:
            rand_h = rng.random(n)
            rand_h = project_onto_constraints(rand_h, target_sum)
            rand_c5 = evaluate_overlap(rand_h)
            if rand_c5 < best_c5:
                best_h = rand_h.copy()
                best_c5 = rand_c5
                curr_h = rand_h.copy()
                curr_c5 = rand_c5

    # Final projection to clean up any tiny numerical drift
    best_h = project_onto_constraints(best_h, target_sum)
    best_c5 = evaluate_overlap(best_h)

    return best_h, best_c5, n
