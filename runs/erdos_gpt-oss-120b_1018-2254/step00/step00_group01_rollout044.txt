import numpy as np

def _compute_c5(h):
    """Return the C5 value for a candidate h."""
    try:
        # Most common signature: (h, n_points) -> (c5, ...)

def _feasible_delta(h, i, j, max_step):
    """
    Compute a feasible delta to transfer from index i to j while
    respecting 0≤h≤1 and |delta|≤max_step.
    Returns None if no feasible move.
    """
    # delta will be subtracted from h[i] and added to h[j]
    lower = max(-(1.0 - h[i]), -h[j])   # minimal delta (most negative)
    upper = min(h[i], 1.0 - h[j])       # maximal delta (most positive)
    if lower >= upper:
        return None
    # restrict magnitude to max_step
    bound = min(max_step, max(abs(lower), abs(upper)))
    delta = np.random.uniform(-bound, bound)
    # Clip to feasible interval
    delta = max(lower, min(upper, delta))
    return delta

def _adjust_sum(arr, target):
    """Adjust the first entry (if possible) to make the sum exactly target."""
    cur = arr.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return arr
    # Find an index we can adjust without violating bounds
    for idx in range(len(arr)):
        new_val = np.clip(arr[idx] + diff, 0.0, 1.0)
        if not np.isclose(new_val, arr[idx]):
            arr[idx] = new_val
            break
    return arr
