def compute_c5(vec):

        else:
            hi = lam

    v = np.random.rand(n)
    v = project_to_capped_simplex(v, target)

import numpy as np
import time
import math

def compute_c5(vec: np.ndarray) -> float:
    """Return the Erdős overlap constant for a discrete h."""
    n = vec.shape[0]
    corr = np.correlate(vec, 1.0 - vec, mode="full")
    return float(corr.max() * (2.0 / n))


def _project_to_capped_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project v onto the set {x : 0 <= x_i <= 1, sum(x) = target}.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lo, 0.0, 1.0)


def _random_feasible(n: int, target: float) -> np.ndarray:
    """Generate a random feasible vector (bounds + sum)."""
    v = np.random.rand(n)
    return _project_to_capped_simplex(v, target)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a step function h with a low C₅ value.

    Returns:
        h_values : np.ndarray of shape (n_points,)
        c5_bound : float, the achieved C₅ upper bound
        n_points : int, number of samples used
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # initialise h
    # ------------------------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=np.float64).ravel()
        n = h0.size
    except Exception:                         # no supplied construction
        n = 200                                 # a modest even size
        h0 = np.random.rand(n)

    target_sum = n / 2.0
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = _project_to_capped_simplex(h0, target_sum)

    current_h = h0.copy()
    current_c5 = compute_c5(current_h)

    best_h = current_h.copy()
    best_c5 = current_c5

    start_time = time.time()

    # ------------------------------------------------------------------
    # optimisation parameters
    # ------------------------------------------------------------------
    max_temp = 0.2          # initial Metropolis temperature
    min_temp = 1e-6
    max_step = 1.0          # maximal relative step size
    min_step = 0.02         # minimal step size (to keep precision)
    restart_prob = 1e-4     # chance to jump to a fresh random state

    # ------------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------------
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        # linear cooling schedules
        temp = max(min_temp, max_temp * (1.0 - elapsed / budget_s))
        step_scale = max(min_step, max_step * (1.0 - elapsed / budget_s))

        # ------------------------------------------------------------------
        # propose a feasible move that preserves the sum exactly
        # ------------------------------------------------------------------
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        if np.random.rand() < 0.5:
            # move mass from j to i
            max_inc_i = 1.0 - current_h[i]
            max_dec_j = current_h[j]
            max_delta = min(max_inc_i, max_dec_j)
            if max_delta <= 0.0:
                continue
            delta = max_delta * np.random.rand() * step_scale
            cand = current_h.copy()
            cand[i] += delta
            cand[j] -= delta
        else:
            # move mass from i to j
            max_inc_j = 1.0 - current_h[j]
            max_dec_i = current_h[i]
            max_delta = min(max_inc_j, max_dec_i)
            if max_delta <= 0.0:

            delta = max_delta * np.random.rand() * step_scale
            cand = current_h.copy()
            cand[i] -= delta

        np.clip(cand, 0.0, 1.0, out=cand)

        # accept / reject the move

            current_c5 = cand_c5
            if cand_c5 < best_c5:
