        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            if best_c5 <= 0.3808:
                break

    # --- Local refinement via hill climbing ----------------------------------
    def hill_move(h):
        i, j = rng.choice(n, size=2, replace=False)
        max_inc = 1.0 - h[i]
        max_dec = h[j]
        if max_inc <= 0 or max_dec <= 0:
            # try opposite direction
            max_inc = 1.0 - h[j]
            max_dec = h[i]
            if max_inc <= 0 or max_dec <= 0:
                return h, False
            i, j = j, i
        delta_max = min(max_inc, max_dec)
        delta = delta_max * (rng.random() * 0.5)   # move up to half of the available margin
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        return h_new, True

    while time.time() - start_time < budget_s - 1:
        h_new, ok = hill_move(best_h)
        if not ok:
            continue
        c5_new = compute_c5(h_new)
        if c5_new + 1e-9 < best_c5:
            best_c5 = c5_new
            best_h = h_new
            if best_c5 <= 0.3808:
                break

    return (best_h.tolist(), float(best_c5), n)
