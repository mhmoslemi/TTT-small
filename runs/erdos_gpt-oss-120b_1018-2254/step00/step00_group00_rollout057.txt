def repair_sum(h):
    n = h.shape[0]
    target = n / 2.0
    # Clip to [0, 1] to avoid small overshoot
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = target - cur
    # if diff approx zero:
    if abs(diff) < 1e-12:
        return h
    # adjust
    if diff > 0:
        # Need to increase sum by diff
        # Determine indices where h < 1
        while diff > 1e-10:
            idx = np.where(h < 1 - 1e-12)[0]
            if idx.size == 0:
                break
            i = np.random.choice(idx)
            inc = min(diff, 1 - h[i])
            h[i] += inc
            diff -= inc
    else:
        # diff < 0, need to reduce sum
        while diff < -1e-10:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = np.random.choice(idx)
            dec = min(-diff, h[i])
            h[i] -= dec
            diff += dec
    # After loop, diff should be near zero.
    # Finally reclip and adjust slight rounding
    h = np.clip(h, 0.0, 1.0)
    # If still off due to rounding, adjust one component (first) by remainder
    final_diff = target - np.sum(h)
    if abs(final_diff) > 1e-10:
        # adjust first element within bounds
        i = 0
        if final_diff > 0:
            # increase up to 1
            add = min(final_diff, 1 - h[i])
            h[i] += add
        else:
            sub = min(-final_diff, h[i])
            h[i] -= sub
    return h

import time
import numpy as np

def _repair_sum(h):
    """Project h onto the simplex sum(h)=n/2 with box constraints [0,1]."""
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # Increase or decrease sum by moving mass between coordinates
    if diff > 0:
        # need to add mass
        while diff > 1e-12:
            idx = np.where(h < 1 - 1e-12)[0]
            if idx.size == 0:
                break
            i = np.random.choice(idx)
            inc = min(diff, 1 - h[i])
            h[i] += inc
            diff -= inc
    else:
        # need to remove mass
        while diff < -1e-12:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = np.random.choice(idx)
            dec = min(-diff, h[i])
            h[i] -= dec
            diff += dec
    # Final tiny correction on first element if needed
    final_diff = target - np.sum(h)
    if abs(final_diff) > 1e-12:
        i = 0
        if final_diff > 0:
            add = min(final_diff, 1 - h[i])
            h[i] += add
        else:
            sub = min(-final_diff, h[i])
            h[i] -= sub
    return np.clip(h, 0.0, 1.0)


def _propose_move(h, max_delta):
    """Move a small amount of mass from one coordinate to another."""
    n = h.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    # feasible delta range (keep both within [0,1])
    lower = max(-h[i], -(1 - h[j]))
    upper = min(1 - h[i], h[j])
    if lower >= upper:
        return h.copy()
    delta = np.random.uniform(lower, upper)
    # limit magnitude
    if delta > max_delta:
        delta = max_delta
    elif delta < -max_delta:
        delta = -max_delta
        if delta < lower:
            delta = lower
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    return new_h


def _evaluate_and_corr(h):
    """Return C5 value and the full correlation array."""
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * (2.0 / n), corr
