import time
import numpy as np
import scipy.optimize

def _objective(h):
    """Objective: C5 value for a candidate h."""
    # Prefer the provided evaluator if available
    if "evaluate_erdos_solution" in globals():
        try:
            return evaluate_erdos_solution(h)
        except Exception:
            pass
    # Fallback: direct computation
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode="full")) * dx

def _sum_eq_constraint(h):
    """Equality constraint: sum(h) must equal n/2."""
    return np.sum(h) - h.shape[0] / 2.0

def _perturb(parent, rng, sigma):
    """Generate a perturbed copy of parent, roughly respecting the sum."""
    h = parent + rng.normal(0.0, sigma, size=parent.shape)
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s != 0:
        h = h * ((parent.shape[0] / 2.0) / s)
    return np.clip(h, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Build initial feasible vector
    if "initial_h_values" in globals():
        parent = np.asarray(initial_h_values, dtype=float).copy()
        n = parent.size
    else:
        n = 80
        parent = np.full(n, 0.5, dtype=float)  # uniform feasible
    target_sum = n / 2.0

    # Force feasibility (may be already satisfied)
    cur_sum = parent.sum()
    if cur_sum != 0 and abs(cur_sum - target_sum) > 1e-12:
        parent = parent * (target_sum / cur_sum)
    parent = np.clip(parent, 0.0, 1.0)

    # ------------------------------------------------------------------
    # Optimization settings
    bounds = [(0.0, 1.0) for _ in range(n)]
    constraints = [{"type": "eq", "fun": _sum_eq_constraint}]
    max_iter = 500
    ftol = 1e-9

    # ------------------------------------------------------------------
    # Track best solution
    best_h = parent.copy()
    best_val = _objective(best_h)

    # Number of random restarts (including the unperturbed start)
    max_starts = 12
    sigma = 0.2

    for idx in range(max_starts):
        # Time‑budget guard
        if time.time() - start_time > budget_s * 0.95:
            break

        if idx == 0:
            x0 = best_h.copy()
        else:
            x0 = _perturb(parent, rng, sigma)

        res = scipy.optimize.minimize(
            _objective,
            x0,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options={"maxiter": max_iter, "ftol": ftol, "disp": False},
        )

        cand = np.clip(res.x, 0.0, 1.0)
        # Re‑enforce the sum exactly (tiny drift may remain)
        s = cand.sum()
        if s != 0 and abs(s - target_sum) > 1e-9:
            cand = cand * (target_sum / s)
            cand = np.clip(cand, 0.0, 1.0)

        cand_val = _objective(cand)
        if cand_val < best_val:
            best_val = cand_val
            best_h = cand.copy()

    # Final feasibility tweak
    total = best_h.sum()
    if total != 0 and abs(total - target_sum) > 1e-12:
        best_h = best_h * (target_sum / total)
    best_h = np.clip(best_h, 0.0, 1.0)

    # Return: array of h-values, its C5 bound, and number of samples
    return best_h, float(best_val), n
