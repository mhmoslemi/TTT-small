import numpy as np, time, math
from scipy.optimize import minimize

def compute_c5(h):
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def project_01_sum(v, target):
    # implemented bisection maybe
    # Could also use simple algorithm: shift = (np.sum(v) - target) / N; w = np.clip(v - shift, 0, 1); but this may not converge if after clipping sum still off. Use iterative fix.
    # We'll implement robust bisection.
    lo = np.min(v - 1)
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        w = np.clip(v - mid, 0, 1)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v - (lo+hi)/2.0, 0, 1)
    # Ensure sum close to target; adjust with small addition to avoid rounding errors.
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        # distribute diff over non-bound positions
        free_idx = (w > 1e-12) & (w < 1 - 1e-12)
        if free_idx.any():
            w[free_idx] += diff / free_idx.sum()
            w = np.clip(w, 0, 1)
        else:
            # If no free, adjust all
            w += diff / len(w)
            w = np.clip(w, 0, 1)
    return w

def random_feasible(n, target, rng):
    # sample uniform on simplex [0,1]^n with sum=target.
    # Use Dirichlet or random sample then project
    v = rng.random(n)  # random in [0,1)
    w = project_01_sum(v, target)
    return w

def hill_climb(h_initial, max_iters, rng):
    h = h_initial.copy()
    best = h.copy()
    best_val = compute_c5(h)
    N = len(h)
    target_sum = N/2
    for _ in range(max_iters):
        # Choose i from indices with h[i] > 0; j from indices with h[j] < 1
        # Use random picks
        i = rng.integers(N)
        # ensure h[i] > small epsilon else re-pick
        if h[i] < 1e-8:
            continue
        j = rng.integers(N)
        if h[j] > 1 - 1e-8:
            continue
        max_delta = min(h[i], 1 - h[j])
        if max_delta <= 0:
            continue
        # delta random between 0 and max_delta
        delta = rng.random() * max_delta
        # Apply transfer
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        # No need to project as sum unchanged and bounds enforced
        val = compute_c5(h_new)
        if val < best_val - 1e-12:
            h = h_new
            best = h_new
            best_val = val
        # else maybe accept with small probability to escape local minima?
        # For simplicity, reject.
    return best, best_val

def project_01_sum(v, target):
    lo = np.min(v - 1)
    hi = np.max(v)
    # bisection
    for _ in range(50):
        mid = (lo + hi) / 2.0
        w = np.clip(v - mid, 0, 1)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v - (lo+hi)/2.0, 0, 1)
    # adjust to have exact sum due to numeric tolerance
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        # add diff uniformly to non-saturated variables
        free = (w > 1e-12) & (w < 1 - 1e-12)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0, 1)
        else:
            # If all variables are saturated, adjust all equally (should not happen)
            w += diff / len(w)
            w = np.clip(w, 0, 1)
    return w

def simulated_annealing(start_h, max_iters, rng):
    n = len(start_h)
    target_sum = n / 2.
    cur_h = start_h.copy()
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val
    # schedule parameters
    init_step = 0.2
    init_temp = 0.005   # small value because objective changes small maybe < 0.5.
    for it in range(max_iters):
        # compute decreasing step and temp
        frac = it / max_iters
        step = init_step * (1 - frac) + 0.001  # at least small step
        temp = init_temp * (1 - frac) + 1e-6

        v = cur_h + rng.normal(scale=step, size=n)

import time, math
import numpy as np

def _c5_value(h):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k)) dx approximated by correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h) using zero padding
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def _project_01_sum(v, target):
    """Project v onto the box [0,1] with sum(v)=target (bounded simplex)."""
    lo = np.min(v - 1)          # all entries will be clamped to 1 if lambda <= lo
    hi = np.max(v)              # all entries will be clamped to 0 if lambda >= hi
    for _ in range(60):
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny adjustment for rounding errors
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        free = (w > 1e-12) & (w < 1.0 - 1e-12)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0.0, 1.0)
        else:
            w += diff / w.size
            w = np.clip(w, 0.0, 1.0)
    return w

def _simulated_annealing(start_h, iters, rng):
    """Simple SA that respects the sum constraint via projection."""
    n = start_h.shape[0]
    target = n / 2.0
    cur = start_h.copy()
    cur_val = _c5_value(cur)
    best = cur.copy()
    best_val = cur_val
    # linear schedules
    init_step = 0.2
    init_temp = 0.02
    for it in range(iters):
        frac = it / iters
        step = init_step * (1.0 - frac) + 1e-4
        temp = init_temp * (1.0 - frac) + 1e-6
        cand = cur + rng.normal(scale=step, size=n)
        cand = _project_01_sum(cand, target)
        cand_val = _c5_value(cand)
        delta = cand_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / temp):
            cur, cur_val = cand, cand_val
            if cur_val < best_val:
                best, best_val = cur.copy(), cur_val
    return best, best_val

def _grid_start(n, rng):
    """Create a few deterministic two‑level candidates (edge‑high, interior‑low)."""
    best_h = np.full(n, 0.5)
    best_val = _c5_value(best_h)
    target = n / 2.0
    for p in np.linspace(0.02, 0.48, 24):
        edge_len = int(round(p * n / 2.0))
        if edge_len * 2 >= n:

        h = np.full(n, alpha)

        h = _project_01_sum(h, target)
        val = _c5_value(h)
        if val < best_val:
            best_h, best_val = h, val
    return best_h
