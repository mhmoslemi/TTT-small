def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, cvxpy as cp
    start = time.time()
    rng = np.random.default_rng(seed)

    # Determine n_points
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 70
        h0 = np.zeros(n)
        idx = rng.choice(n, n//2, replace=False)
        h0[idx] = 1.0

    # Ensure feasibility
    desired_sum = n / 2.0
    # adjust sum by scaling if necessary
    cur_sum = np.sum(h0)
    if not np.isclose(cur_sum, desired_sum):
        # simple scaling
        if cur_sum > 0:
            h0 = h0 * (desired_sum / cur_sum)
        else:
            # set half entries to 1
            h0[:int(desired_sum)] = 1.0
        h0 = np.clip(h0, 0.0, 1.0)
        # adjust final sum with correction
        diff = desired_sum - np.sum(h0)
        if abs(diff) > 1e-12:
            # adjust first non-bound element
            non_bounds = np.where((h0 > 0) & (h0 < 1))[0]
            if non_bounds.size > 0:
                h0[non_bounds[0]] += diff
            else:
                h0[0] = np.clip(h0[0] + diff, 0.0, 1.0)

    # Linear programming via McCormick
    # Build model
    h = cp.Variable(n)
    t = cp.Variable()
    constraints = [h >= 0, h <= 1, cp.sum(h) == desired_sum, t >= 0]

    # Precompute list of constraints per shift, to avoid large Python loops overhead
    # We'll generate for each k.
    for k in range(n):
        len_k = n - k
        if len_k <= 0:
            continue
        h_i = h[:len_k]
        h_j = h[k:]
        # z variables for this shift
        z = cp.Variable(len_k)
        constraints += [z <= h_i,
                        z <= h_j,
                        z >= h_i + h_j - 1]
        Ck = cp.sum(h_i) - cp.sum(z)
        constraints += [Ck <= t]

    # Objective
    objective = cp.Minimize(t)

    # Solve with a fast LP solver
    prob = cp.Problem(objective, constraints)
    # choose solver; ECOS is default LP solver
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:
        prob.solve(solver=cp.SCS, verbose=False)

    # Gather solution
    if h.value is None:
        # fallback to initial
        best_h = h0.copy()
    else:
        best_h = np.array(h.value).reshape(n)
        # Clip numeric errors
        best_h = np.clip(best_h, 0.0, 1.0)
        # Adjust sum precisely
        cur_sum = np.sum(best_h)
        diff = desired_sum - cur_sum
        if abs(diff) > 1e-8:
            # distribute diff proportionally to entries not at bounds
            free = np.where((best_h > 0) & (best_h < 1))[0]
            if free.size > 0:
                best_h[free] += diff / free.size
                best_h = np.clip(best_h, 0.0, 1.0)
            else:
                best_h[0] = np.clip(best_h[0] + diff, 0.0, 1.0)

    # Compute bound
    def evaluate(vec):
        corr = np.correlate(vec, 1 - vec, mode='full')
        max_corr = np.max(corr)
        return max_corr * (2.0 / len(vec))

    c5 = evaluate(best_h)

    # If time remains, perform local search refinement
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    if remaining > 5.0:
        # simple hill climbing
        current_h = best_h.copy()
        current_val = c5
        # set temperature scaling based on current_val
        T = max(1e-4, current_val * 0.1)
        cooling = 0.999
        # number of moves per second approximated: each evaluate O(n^2); we limit loop counts
        end_time = start + budget_s * 0.95
        n = len(current_h)
        while time.time() < end_time:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            hi = current_h[i]
            hj = current_h[j]
            max_inc = 1.0 - hi
            max_dec = hj
            if max_inc <= 0 or max_dec <= 0:
                # try opposite direction
                max_inc = 1.0 - hj
                max_dec = hi
                if max_inc <= 0 or max_dec <= 0:
                    continue
                i, j = j, i
                hi, hj = hj, hi
            # delta magnitude
            max_delta = min(max_inc, max_dec)
            # propose delta scaled by temperature
            delta = max_delta * (rng.random() ** 2) * (T / (current_val + 1e-9) + 0.1)
            # apply
            new_h = current_h.copy()
            new_h[i] += delta
            new_h[j] -= delta
            new_val = evaluate(new_h)
            if new_val < current_val or rng.random() < np.exp(-(new_val - current_val) / max(T, 1e-12)):
                current_h = new_h
                current_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best_h = new_h.copy()
            T = max(T * cooling, 1e-6)

        c5 = best_val

    return best_h, c5, n

import time
import numpy as np

def _evaluate_corr(h):
    """C5 value for a discrete step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def _fix_binary_solution(y, n):
    """Round a near‑binary array to exact 0/1 while keeping the sum = n/2."""
    y_int = np.rint(y).astype(int)
    target = n // 2
    cur = int(y_int.sum())
    diff = target - cur
    if diff > 0:
        # need more 1s
        zeros = np.where(y_int == 0)[0]
        if zeros.size:
            flip = np.random.choice(zeros, min(diff, zeros.size), replace=False)
            y_int[flip] = 1
    elif diff < 0:
        # too many 1s
        ones = np.where(y_int == 1)[0]
        if ones.size:
            flip = np.random.choice(ones, min(-diff, ones.size), replace=False)
            y_int[flip] = 0
    return y_int.astype(float)


def _milp_binary_opt(n, time_limit):
    """Try a mixed‑integer linear program for a binary optimum."""
    try:
        import cvxpy as cp
    except Exception:
        return None, None

    # binary indicator vector
    y = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)                     # maximal overlap count
    constraints = [
        cp.sum(y) == n // 2,
        t >= 0
    ]

    # constraints for each positive shift (k=0 gives zero overlap automatically)
    for k in range(1, n):
        m = n - k
        y_i = y[:m]
        y_j = y[k:]
        w = cp.Variable(m, boolean=True)             # AND of the two bits
        constraints += [
            w <= y_i,
            w <= y_j,
            w >= y_i + y_j - 1
        ]
        mismatches = cp.sum(y_i) - cp.sum(w)          # count of 1→0 pairs for shift k
        constraints.append(mismatches <= t)

    prob = cp.Problem(cp.Minimize(t), constraints)

    # try a few solvers; the first that works wins
    solvers = ['ECOS_BB', 'CBC', 'GLPK_MI', 'SCIP']
    for s in solvers:
        try:
            prob.solve(solver=getattr(cp, s), verbose=False, max_iters=200000)
            if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
                h = _fix_binary_solution(y.value, n)
                return h, float(t.value)
        except Exception:
            continue
        if time.time() - start_time > time_limit:
            break
    return None, None


def _local_search(h, budget, start):
    """Simple simulated‑annealing improvement for a real‑valued vector."""
    n = h.shape[0]
    best_h = h.copy()
    best_val = _evaluate_corr(best_h)
    cur_h = best_h.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    T = best_val * 0.5 + 1e-3
    cooling = 0.999
    end = start + budget * 0.95

    while time.time() < end:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        hi, hj = cur_h[i], cur_h[j]
        max_inc = 1.0 - hi
        max_dec = hj
        if max_inc <= 0 or max_dec <= 0:
            # try opposite direction
            max_inc = 1.0 - hj
            max_dec = hi
            if max_inc <= 0 or max_dec <= 0:
                continue
            i, j = j, i
            hi, hj = hj, hi
        delta = min(max_inc, max_dec) * rng.random()
        # propose move
        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        new_val = _evaluate_corr(new_h)
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12)):
            cur_h, cur_val = new_h, new_val
            if new_val < best_val:
                best_h, best_val = new_h.copy(), new_val
        T = max(T * cooling, 1e-6)
    return best_h, best_val
