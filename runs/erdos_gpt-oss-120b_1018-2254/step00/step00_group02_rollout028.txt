import time, numpy as np

def compute_c5(h):
   # h array of floats in [0,1] (binary)
   # compute correlation h and 1-h via np.convolve
   conv = np.convolve(h, (1 - h)[::-1], mode='full')
   # multiply by dx
   dx = 2.0 / len(h)
   return np.max(conv) * dx

def hill_climb(h, max_iters, time_limit):
   best_h = h.copy()
   best_c5 = compute_c5(best_h)
   n = len(h)
   start_time = time.time()
   for it in range(max_iters):
       if time.time() - start_time > time_limit:
           break
       # create neighbor by swapping two positions with opposite bits
       # choose indices i where h[i]==1 and j where h[j]==0
       ones = np.where(best_h > 0.5)[0]
       zeros = np.where(best_h <= 0.5)[0]
       if len(ones) == 0 or len(zeros) == 0:
           break
       i = np.random.choice(ones)
       j = np.random.choice(zeros)
       candidate = best_h.copy()
       candidate[i] = 0.0
       candidate[j] = 1.0
       cand_c5 = compute_c5(candidate)
       if cand_c5 < best_c5 - 1e-12: # improvement
           best_h = candidate
           best_c5 = cand_c5
   return best_h, best_c5

import time
import numpy as np
from scipy.signal import fftconvolve

# Helper to compute the C5 value for a given step function h
def _c5_of(h):
    n = h.size
    conv = fftconvolve(h, (1 - h)[::-1], mode='full')
    return np.max(conv) * (2.0 / n)

def _random_balanced(n):
    """
    Produce a length-n vector in [0,1] with sum == n/2.
    For even n the vector is binary (exactly n/2 ones).
    For odd n it has floor(n/2) ones, one fractional entry, rest zeros.
    """
    target = n / 2.0
    h = np.zeros(n, dtype=float)
    ones = int(np.floor(target))
    if ones > 0:
        idx_one = np.random.choice(n, ones, replace=False)
        h[idx_one] = 1.0
    remaining = target - ones
    if remaining > 1e-12:
        # place the fractional remainder on a zero position
        zero_candidates = np.setdiff1d(np.arange(n), np.where(h == 1)[0])
        idx_frac = np.random.choice(zero_candidates)
        h[idx_frac] = remaining
    return h

def _anneal_one(start_h, max_iters, time_limit):
    """
    Simulated annealing on a binary (or binary+fractional) vector.
    Swaps a 1 with a 0 each move, preserving the sum constraint.
    """
    n = start_h.size
    cur = start_h.copy()
    best = cur.copy()
    cur_val = _c5_of(cur)
    best_val = cur_val
    start = time.time()
    T0 = 0.5

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        # indices of 1s and 0s (ignore any exact 0.5 value)
        ones = np.where(cur > 0.5)[0]
        zeros = np.where(cur < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        cand = cur.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        cand_val = _c5_of(cand)

        # linear cooling schedule
        T = T0 * (1.0 - it / max_iters)
        if cand_val < cur_val or np.random.rand() < np.exp(-(cand_val - cur_val) / (T + 1e-12)):
            cur = cand
            cur_val = cand_val
            if cand_val < best_val:
                best = cand.copy()
                best_val = cand_val
    return best, best_val
