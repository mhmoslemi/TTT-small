def project_to_constraints(v, target_sum):
    # First scale to match sum approx
    scaling = target_sum / v.sum()
    h = v * scaling
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # Adjust sum residual iteratively
    for _ in range(10):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-9:
            break
        # Determine indices that can be increased or decreased
        if diff > 0:
            # increase some entries not at 1
            mask = h < 1.0 - 1e-12
            if not np.any(mask):
                break
            # distribute diff equally among them
            inc = diff / np.count_nonzero(mask)
            h[mask] += inc
            # clamp again
            h = np.clip(h, 0.0, 1.0)
        else:
            # decrease entries not at 0
            mask = h > 1e-12
            if not np.any(mask):
                break
            dec = diff / np.count_nonzero(mask)  # diff negative
            h[mask] += dec  # diff is negative
            h = np.clip(h, 0.0, 1.0)
    # In case residual still > tolerance, just return h anyway.
    return h

import numpy as np
import time
import math

def compute_c5(h):
    """Return max overlap integral C5 for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_to_constraints(v, target):
    """Project vector v onto the box [0,1]^n intersect sum(v)=target."""
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    # binary search for lambda
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h = np.clip(v - mid, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            return h
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - mid, 0.0, 1.0)

def upsample_h(src, n_target):
    """Linearly interpolate src to length n_target and enforce constraints."""
    src = np.asarray(src, dtype=float)
    n_src = src.shape[0]
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_tgt = np.linspace(0.0, 2.0, n_target, endpoint=False)
    tgt = np.interp(x_tgt, x_src, src)
    return project_to_constraints(tgt, n_target / 2.0)
