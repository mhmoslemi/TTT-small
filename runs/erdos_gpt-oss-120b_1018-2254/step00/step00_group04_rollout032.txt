def propose_neighbor(h):
    n = len(h)
    i, j = np.random.choice(n, 2, replace=False)
    # direction: move from i to j (i loses, j gains)
    max_delta = min(h[i], 1 - h[j])
    if max_delta <= 0:
        # try opposite direction if possible
        max_delta = min(h[j], 1 - h[i])
        if max_delta <= 0:
            return h.copy() # no move
        # invert direction: move from j to i
        i, j = j, i
    # pick delta uniform in (0, max_delta)
    delta = max_delta * np.random.rand()
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)
    # Determine n_points
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_init = np.array(initial_h_values, dtype=float)
    else:
        n_points = 42
        # generate random binary half ones
        h_init = rng.choice([0.0,1.0], size=n_points, p=[0.5,0.5])
        # adjust sum (rarely needed)
        half = n_points//2
        # ensure exactly half ones
        idx = np.arange(n_points)
        rng.shuffle(idx)
        h_init[:] = 0.0
        h_init[idx[:half]] = 1.0

    # Ensure feasible: sum = n/2
    target_sum = n_points / 2.0
    # If not exactly target, we need to adjust by moving mass between entries
    # Use simple scaling but preserving bounds and sum
    # We'll implement a projection method:
    def project_sum_one(h):
        # Adjust to exact sum while staying within [0,1]
        total = h.sum()
        diff = target_sum - total
        if abs(diff) < 1e-12:
            return h
        # Distribute diff uniformly to entries not at bound.
        # Compute free entries
        free = (h > 0) & (h < 1)
        num_free = free.sum()
        if num_free == 0:
            # If all at bounds, can't adjust; fallback scaling
            h = h * (target_sum / total)
            h = np.clip(h,0,1)
            return h
        # Add diff equally
        h[free] += diff / num_free
        # Clip again
        h = np.clip(h, 0.0, 1.0)
        # Recursively correct if still off (rare)
        if abs(h.sum() - target_sum) > 1e-9:
            # simple fallback scaling for final correction
            h = h * (target_sum / h.sum())
            h = np.clip(h, 0.0, 1.0)
        return h

    dx = 2.0 / n_points

        return np.max(corr) * dx

    while elapsed() < budget_s * 0.95:  # leave margin

        donor = i

def project_sum_one(h, target_sum):
    total = h.sum()
    diff = target_sum - total
    if abs(diff) < 1e-12:
        return h
    # Identify free indices where we can add/subtract without hitting bounds
    # We'll use simple iterative approach
    # Use water-filling: add diff equally to free entries, repeat if some hit bounds
    # We'll loop at most few times.

import numpy as np
import time

def compute_c5(h, dx):
    """Return max_{k} ∫ h(x)(1-h(x+k)) dx for discrete h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_sum_one(h, target):
    """Adjust h to satisfy sum(h)=target while keeping 0≤h≤1."""
    h = h.copy()
    for _ in range(10):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 0) & (h < 1)
        n_free = free.sum()
        if n_free == 0:
            break
        delta = diff / n_free
        # avoid crossing bounds
        over_up = (h[free] + delta > 1)
        over_down = (h[free] + delta < 0)
        if over_up.any():
            idx = np.where(free)[0][over_up]
            h[idx] = 1.0
            continue
        if over_down.any():
            idx = np.where(free)[0][over_down]
            h[idx] = 0.0
            continue
        h[free] += delta
    # final fallback scaling
    total = h.sum()
    if abs(total - target) > 1e-9:
        factor = target / total if total != 0 else 0.0
        h = np.clip(h * factor, 0.0, 1.0)
        # tiny correction if still off
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            free = (h > 0) & (h < 1)
            n_free = free.sum()
            if n_free > 0:

    return h
