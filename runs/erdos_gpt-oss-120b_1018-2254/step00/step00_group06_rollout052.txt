def random_feasible(n, total, rng):
    x = np.zeros(n)
    remaining = total
    idxs = list(range(n))
    rng.shuffle(idxs)
    for i in idxs:
        if remaining <= 0:
            break
        max_assign = min(1.0, remaining)
        # assign a random amount between 0 and max_assign
        # use uniform random
        assign = rng.random() * max_assign
        x[i] = assign
        remaining -= assign
    # if remaining > small tolerance after loop due to rounding, distribute.
    if remaining > 1e-8:
        # Add remainder to random index with capacity left
        for i in idxs:
            capacity = 1.0 - x[i]
            if capacity > 1e-12:
                add = min(capacity, remaining)
                x[i] += add
                remaining -= add
                if remaining <= 1e-12:
                    break
    return x

    # returns vector y
    # Use binary search for lambda.
    lo = -np.max(v) - 1  # ensure all entries >1 after subtract.
    hi = np.max(v) + 1    # ensure all entries <0 after subtract.
    # Ensure that sum at lo >= total
    # At lo: v - lo huge positive -> clamp to high => sum = n*high = n
    # This holds because hi bound is high (1). So sum at lo = n*high >= total.
    # At hi: v - hi very negative -> clamp to low => sum = 0 <= total.
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        y = np.clip(v - mid, low, high)
        s = y.sum()
        if s > total:
            lo = mid
        else:
            hi = mid

import numpy as np
import time

# ---------- Helper functions ----------
def project_to_capped_simplex(v, total, low=0.0, high=1.0, max_iter=60, tol=1e-10):
    """
    Euclidean projection of vector v onto the set:
        low <= x_i <= high  and  sum(x) = total
    """
    # Binary search on lambda for x_i = clip(v_i - lambda, low, high)
    lo = -np.max(v) - 1.0   # ensures all entries become > high -> clipped to high
    hi = np.max(v) + 1.0    # ensures all entries become < low -> clipped to low

    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, low, high)
        s = x.sum()
        if s > total:
            lo = mid
        else:
            hi = mid
        if hi - lo < tol:
            break
    # Final projection
    x = np.clip(v - lo, low, high)
    # Minor adjustment to hit the sum exactly (numeric safety)
    diff = total - x.sum()
    if abs(diff) > 1e-12:
        # distribute diff proportionally to non‑bound entries
        free = (x > low + 1e-12) & (x < high - 1e-12)
        cnt = free.sum()
        if cnt > 0:
            adjust = diff / cnt
            x[free] += adjust
            x = np.clip(x, low, high)
    return x

def random_feasible(n, total, rng):
    """Generate a random vector in [0,1]^n with prescribed sum=total."""
    v = rng.random(n)               # uniform in [0,1)
    return project_to_capped_simplex(v, total)

def compute_c5(h):
    """C5 value for a candidate step function h (array of length n)."""
    n = h.shape[0]
    dx = 2.0 / n
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def tournament_select(pop, fitness, k, rng):
    """Select one individual via tournament of size k."""
    idx = rng.choice(pop.shape[0], size=k, replace=False)
    best = idx[np.argmin(fitness[idx])]
    return pop[best]

def crossover(p1, p2, rng, rate=0.7):
    """Single‑point crossover."""
    if rng.random() < rate:
        cp = rng.integers(1, len(p1))
        c1 = np.concatenate((p1[:cp], p2[cp:]))
        c2 = np.concatenate((p2[:cp], p1[cp:]))
    else:
        c1 = p1.copy()
        c2 = p2.copy()
    return c1, c2

def mutate(x, rng, prob=0.01, sigma=0.1):
    """Gaussian mutation applied per coordinate with probability prob."""
    mask = rng.random(x.shape) < prob
    if not np.any(mask):
        return x.copy()
    noise = rng.normal(scale=sigma, size=x.shape)
    y = x.copy()
    y[mask] += noise[mask]
    return y
