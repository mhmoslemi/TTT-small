                best_score = candidate_score
                best_solution = candidate
                realized_score = candidate_score
            print(f"No improvement at bexp: {iteration}: retained {best_score}")
        
        # Budget check: Ample margin left?
        if iteration == 4:  # Final iteration for reacting to budget impact
            break
    
    # Determine niches: by energy levels
    levels, niche_points = stratify_solutions_by_energy(best_solution, n_points, 3)
    return best_solution, best_score, n_points
