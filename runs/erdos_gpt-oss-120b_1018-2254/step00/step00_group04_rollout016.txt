def repair_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        s = h.sum()
        if abs(s - target_sum) < 1e-9:
            break
        factor = target_sum / s if s != 0 else 0.0
        h = h * factor
        h = np.clip(h, 0.0, 1.0)
    # residual
    s = h.sum()
    diff = target_sum - s
    if abs(diff) > 1e-8:
        # try to distribute diff to free entries
        free = np.where((h > 0) & (h < 1))[0]
        if free.size > 0:
            # add diff to first free; if exceeds bound, clip and adjust residual.
            idx = free[0]
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        else:
            # adjust all entries proportionally
            h = np.clip(h + diff/len(h), 0.0, 1.0)
        # after adjustment maybe still off, but small
        # final clip
        h = np.clip(h, 0.0, 1.0)
        # final rescale if still off due to clipping
        s = h.sum()
        if abs(s - target_sum) > 1e-6:
            # distribute among all entries not at bound
            free = np.where((h > 0) & (h < 1))[0]
            if free.size > 0:
                # equally adjust
                adj = (target_sum - s) / free.size
                h[free] += adj
                # clip again
                h = np.clip(h, 0.0, 1.0)
    return h

def repair_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    # iterative scaling
    for _ in range(20):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        if s == 0:
            # distribute uniformly
            h[:] = target_sum / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            factor = target_sum / s
            h = h * factor
            h = np.clip(h, 0.0, 1.0)
    # final adjustment if needed
    s = h.sum()
    diff = target_sum - s
    if abs(diff) > 1e-10:
        # distribute diff to non-bound entries if possible
        free = np.where((h > 0) & (h < 1))[0]

            h[free] += per

        else:
            # fallback: adjust first element
            h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h

import time
import random
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return the maximal overlap integral for a given step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)

def repair_h(h: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project a vector onto the box [0,1]^n intersected with the hyperplane sum = target_sum.
    Simple iterative scaling + final redistribution.
    """
    h = np.clip(h, 0.0, 1.0)
    # iterative scaling to approach the target sum
    for _ in range(20):
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
        if s == 0.0:
            h[:] = target_sum / h.size
        else:
            factor = target_sum / s
            h *= factor
        h = np.clip(h, 0.0, 1.0)

    # final tiny correction if needed
    s = h.sum()
    diff = target_sum - s
    if abs(diff) > 1e-10:
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size:
            per = diff / free.size
            h[free] += per
            h = np.clip(h, 0.0, 1.0)
        else:
            # fallback: adjust first coordinate (should be rare)
            h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h
