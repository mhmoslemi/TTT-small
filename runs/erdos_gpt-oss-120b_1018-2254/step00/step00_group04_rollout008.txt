import numpy as np, time, random, math

def repair_h(h, target):
    h = np.clip(h, 0.0, 1.0)
    # iterative scaling/clip
    for _ in range(50):
        total = h.sum()
        diff = target - total
        if abs(diff) < 1e-12:
            break
        # scale factor
        if total != 0:
            factor = target / total
            h = h * factor
        h = np.clip(h, 0.0, 1.0)
    # final diff adjust small via one element
    diff = target - h.sum()
    if abs(diff) > 1e-9:
        # adjust a non-boundary element
        if diff > 0:
            idxs = np.where(h < 1)[0]
        else:
            idxs = np.where(h > 0)[0]
        if idxs.size > 0:
            i = idxs[0]
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def evaluate(h):
    # call pre-imported function
    return evaluate_erdos_solution(h)

def mutate(h, sigma):
    # Add Gaussian noise
    new = h + sigma * np.random.randn(h.shape[0])
    # repair
    target = h.shape[0] / 2.0
    new = repair_h(new, target)
    return new

def random_feasible(n):
    # generate random uniform, then repair
    h = np.random.rand(n)
    target = n/2.0
    h = repair_h(h, target)
    return h

    best_h = current.copy()
    best_c = evaluate(best_h)
    # parameters

    # schedule sigma decay based on time

def repair_h(vec, target):
    """Clip vector to [0,1] and enforce sum = target (exactly, within tolerance)."""
    # Clip to bounds
    vec = np.clip(vec, 0.0, 1.0)
    # Iteratively enforce the sum constraint via scaling + clipping
    for _ in range(50):
        total = vec.sum()
        diff = target - total
        if abs(diff) <= 1e-12:
            break
        if total > 0:
            factor = target / total
            vec *= factor
        vec = np.clip(vec, 0.0, 1.0)

    diff = target - vec.sum()
    if abs(diff) > 1e-12:

        if cand.size:
            i = cand[0]
            vec[i] = np.clip(vec[i] + diff, 0.0, 1.0)
    return vec
