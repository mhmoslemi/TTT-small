def compute_c5(h):
    dx = 2.0 / len(h)
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

    g = np.correlate(x, x, mode='full')
    # exp(-alpha * g)
    exps = np.exp(-alpha * g)
    Z = np.sum(exps)
    # avoid zero division
    if Z == 0:
        return np.inf
    return np.log(Z) / alpha

import numpy as np
import time
import math
from scipy.optimize import minimize

# ---------- utility functions ----------
def compute_c5(h):
    """Maximum overlap integral for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx


def smooth_obj(x, alpha):
    """Smooth approximation of max_s ( - sum_i x_i x_{i+s} )."""
    g = np.correlate(x, x, mode='full')
    a = -alpha * g
    a_max = np.max(a)
    exp_shift = np.exp(a - a_max)          # stable scaling
    Z = np.sum(exp_shift)
    return (a_max + np.log(Z)) / alpha


def smooth_obj_grad(x, alpha):
    """Gradient of the smooth objective."""
    n = len(x)
    offset = n - 1
    g = np.correlate(x, x, mode='full')
    a = -alpha * g
    a_max = np.max(a)
    exp_shift = np.exp(a - a_max)
    Z = np.sum(exp_shift)
    w = exp_shift / Z                      # weights for each shift
    grad = np.zeros_like(x)
    for i in range(n):
        total = 0.0
        for s_idx, ws in enumerate(w):
            s = s_idx - offset
            j = i + s
            if 0 <= j < n:
                total += ws * x[j]
            k = i - s
            if 0 <= k < n:
                total += ws * x[k]
        grad[i] = -total
    return grad


def objective_wrapper(x, alpha, lam, target):
    """Full objective: smooth term + penalty for sum constraint."""
    return smooth_obj(x, alpha) + lam * (np.sum(x) - target) ** 2


def gradient_wrapper(x, alpha, lam, target):
    """Gradient of the full objective."""
    return smooth_obj_grad(x, alpha) + 2.0 * lam * (np.sum(x) - target)


def project_sum(x, target, rng):
    """Clip to [0,1] and adjust to meet the exact sum constraint."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - np.sum(x)
    n = len(x)
    for _ in range(20):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idxs = np.where(x < 1.0)[0]
            if len(idxs) == 0:
                break
            i = rng.choice(idxs)
            inc = min(diff, 1.0 - x[i])
            x[i] += inc
            diff -= inc
        else:
            idxs = np.where(x > 0.0)[0]
            if len(idxs) == 0:
                break
            i = rng.choice(idxs)
            dec = min(-diff, x[i])
            x[i] -= dec
            diff += dec
    return x


def random_feasible(n, target, rng):
    """Generate a random feasible vector (0≤x≤1, sum = target)."""
    x = rng.random(n)
    x = np.clip(x, 0.0, 1.0)
    diff = target - np.sum(x)
    x += diff / n
    return project_sum(x, target, rng)


def hill_climb(start, target, dx, rng, time_limit):
    """Simple hill‑climbing refinement."""
    x = start.copy()
    n = len(x)
    best = x.copy()
    best_val = compute_c5(best)
    start_t = time.time()
    it = 0
    while time.time() - start_t < time_limit:
        i, j = rng.choice(n, size=2, replace=False)
        max_pos = min(x[i], 1.0 - x[j])
        max_neg = min(x[j], 1.0 - x[i])
        if max_pos <= 0 and max_neg <= 0:
            continue
        if max_pos > 0 and max_neg > 0:
            delta = rng.uniform(0.0, max_pos) if rng.random() < 0.5 else -rng.uniform(0.0, max_neg)
        elif max_pos > 0:
            delta = rng.uniform(0.0, max_pos)
        else:
            delta = -rng.uniform(0.0, max_neg)
        x_new = x.copy()
        x_new[i] -= delta
        x_new[j] += delta
        val = compute_c5(x_new)
        if val < best_val:
            best_val = val
            best = x_new.copy()
        # accept only improvements (simple greedy)
        cur_val = compute_c5(x)
        if val < cur_val:
            x = x_new
        it += 1
        if it % 5000 == 0:
            # occasional re‑perturbation around the best found so far
            x = best + rng.normal(scale=0.01, size=n)
            x = np.clip(x, 0.0, 1.0)
            x = project_sum(x, target, rng)
    return best, best_val

            )
            x_curr = np.clip(res.x, 0.0, 1.0)

        time_for_hc = min(5.0, remaining * 0.3)
        hc_h, hc_val = hill_climb(x_curr, target, dx, rng, time_for_hc)
        if hc_val < best_c5:
            best_c5 = hc_val

    if final_rem > 2.0:

        if final_val < best_c5:

            best_h = final_h.copy()

    # ensure feasibility exactly
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum(best_h, target, rng)

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), float(final_c5), n)
