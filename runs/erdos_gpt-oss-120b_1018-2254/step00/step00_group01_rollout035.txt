import numpy as np
import time

def _compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k)) dx for discrete samples."""
    n = h.shape[0]
    # correlation with no reversal (np.correlate implements cross‑correlation)
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)

def _project_bounds_sum(v, target, tol=1e-12, max_iter=100):
    """
    Euclidean projection of v onto the set {x | 0≤x≤1, sum(x)=target}
    using a simple bisection on the Lagrange multiplier λ:
        x_i = clip(v_i - λ, 0, 1)
    """
    lo = -np.max(v) - 1.0          # guarantees all entries become 1
    hi = np.max(v) + 1.0           # guarantees all entries become 0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    # final fallback
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
