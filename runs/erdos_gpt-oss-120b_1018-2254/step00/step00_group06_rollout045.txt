def project_to_box_and_sum(v, target):
    n = len(v)
    # Clip initially to [0,1]
    x = np.clip(v, 0.0, 1.0)
    # Compute current sum
    s = x.sum()
    # If close to target, return.
    if abs(s - target) < 1e-12:
        return x
    # We'll iterate.
    # indices that are not at bounds
    free = np.ones(n, dtype=bool)
    # We'll free to adjust.
    # compute lambda iteratively.
    for _ in range(n):
        s = x[free].sum()
        cnt = free.sum()
        if cnt == 0:
            break
        lambda_ = (target - s) / cnt
        x[free] += lambda_
        # now handle bound violations
        lower_viol = (x < 0) & free
        upper_viol = (x > 1) & free
        if not (lower_viol.any() or upper_viol.any()):
            break
        x[lower_viol] = 0.0
        x[upper_viol] = 1.0
        free = free & (~lower_viol) & (~upper_viol)
    # final clip
    np.clip(x, 0.0, 1.0, out=x)
    # Possibly adjust a bit to meet target (due to floating errors)
    # Use simple linear adjustment on indices not at bounds.
    residual = target - x.sum()
    if abs(residual) > 1e-9:
        # indices not at bound
        adjustable = (x > 0) & (x < 1)
        cnt = adjustable.sum()
        if cnt > 0:
            x[adjustable] += residual / cnt
            np.clip(x, 0.0, 1.0, out=x)
    return x

def run(seed=42, budget_s=1100, **kwargs):
    import time
    import numpy as np
    np.random.seed(seed)
    start = time.time()
    # Determine n_points, init.
    if 'initial_h_values' in globals():
        init = globals()['initial_h_values']
        init = np.asarray(init, dtype=float)
        n_points = init.shape[0]
    else:
        init = None
        n_points = 95
    target_sum = n_points / 2.0

    pop_size = 150
    tournament_k = 3
    crossover_rate = 0.7
    mutation_rate = 0.01
    mutation_sigma = 0.1
    max_generations = 500

    pop = initialize_population(pop_size, n_points, target_sum, init)
    fitness = evaluate_population(pop)
    best_idx = np.argmin(fitness)
    best_h = pop[best_idx].copy()
    best_f = fitness[best_idx]

    generation = 0
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s - 0.1:
            break
        if generation >= max_generations:
            break

        new_pop = np.empty_like(pop)
        new_pop[0] = best_h  # elitism
        for i in range(1, pop_size):
            # parent selection
            p1_idx = tournament_select(fitness, tournament_k)
            p2_idx = tournament_select(fitness, tournament_k)
            child = crossover(pop[p1_idx], pop[p2_idx], crossover_rate)
            child = mutate(child, mutation_rate, mutation_sigma)
            child = project_to_box_and_sum(child, target_sum)
            new_pop[i] = child

        pop = new_pop
        fitness = evaluate_population(pop)
        # update best
        cur_best_idx = np.argmin(fitness)
        cur_best_f = fitness[cur_best_idx]
        if cur_best_f < best_f:
            best_f = cur_best_f
            best_h = pop[cur_best_idx].copy()
        generation += 1

    # local random search in remaining time
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s - 0.1:
            break
        # generate random perturbation
        cand = best_h + np.random.randn(n_points) * 0.05
        cand = project_to_box_and_sum(cand, target_sum)
        c = compute_c5(cand)
        if c < best_f:
            best_f = c
            best_h = cand.copy()

    return (best_h, float(best_f), int(n_points))

    x = np.clip(v, 0.0, 1.0)
    # iterative water-filling with bounds
    free = np.ones(n, dtype=bool)
    for _ in range(n):
        s = x[free].sum()
        cnt = free.sum()
        if cnt == 0:
            break
        lam = (target - s) / cnt
        x[free] += lam
        below = (x < 0) & free
        above = (x > 1) & free
        if not (below.any() or above.any()):
            break
        x[below] = 0.0
        x[above] = 1.0
        free = free & (~below) & (~above)
    # final clip and small residual fix
    np.clip(x, 0.0, 1.0, out=x)
    residual = target - x.sum()
    if abs(residual) > 1e-12:
        adjustable = (x > 0) & (x < 1)
        cnt = adjustable.sum()
        if cnt > 0:
            x[adjustable] += residual / cnt
            np.clip(x, 0.0, 1.0, out=x)
    return x

def compute_c5(h):
    """Return max overlap integral for step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def evaluate_population(pop):
    """Compute C5 for each individual in pop (shape: (m, n))."""
    m, n = pop.shape
    dx = 2.0 / n
    scores = np.empty(m)
    for i in range(m):
        corr = np.correlate(pop[i], 1.0 - pop[i], mode='full')
        scores[i] = np.max(corr) * dx
    return scores

def tournament_select(fitness, k):
    """Select index of best individual among k random competitors."""
    cand = np.random.choice(len(fitness), k, replace=False)
    return cand[np.argmin(fitness[cand])]

def crossover(p1, p2, cr):
    """Single-point crossover with probability cr."""
    if np.random.rand() < cr:
        pt = np.random.randint(1, len(p1))
        child = np.empty_like(p1)
        child[:pt] = p1[:pt]
        child[pt:] = p2[pt:]
        return child
    return p1.copy()

def mutate(child, mr, sigma):
    """Gaussian mutation per gene with probability mr."""
    mask = np.random.rand(child.size) < mr
    if mask.any():
        child[mask] += np.random.randn(mask.sum()) * sigma
    return child

def initialize_population(pop_size, n, target, init):
    """Create initial population; first entry is (projected) init if given."""
    pop = np.empty((pop_size, n))
    idx = 0
    if init is not None:
        pop[0] = project_to_box_and_sum(np.asarray(init, dtype=float), target)
        idx = 1
    while idx < pop_size:
        rand_vec = np.random.rand(n)
        pop[idx] = project_to_box_and_sum(rand_vec, target)
        idx += 1
    return pop

def run(seed=42, budget_s=1100, **kwargs):
    import time
    np.random.seed(seed)
    start = time.time()

    # Determine problem size and possible initial solution
    if 'initial_h_values' in globals():
        init_vals = np.asarray(globals()['initial_h_values'], dtype=float)
        n_points = init_vals.shape[0]
    else:
        init_vals = None
        n_points = 95

    target_sum = n_points / 2.0

    # GA parameters
    pop_size = 150
    tournament_k = 3
    crossover_rate = 0.7
    mutation_rate = 0.01
    mutation_sigma = 0.1
    max_gens = 500

    pop = initialize_population(pop_size, n_points, target_sum, init_vals)
    fitness = evaluate_population(pop)

    best_idx = np.argmin(fitness)
    best_h = pop[best_idx].copy()
    best_f = fitness[best_idx]

    gen = 0
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s - 0.1 or gen >= max_gens:
            break

        new_pop = np.empty_like(pop)
        new_pop[0] = best_h  # elitism

        for i in range(1, pop_size):
            p1 = pop[tournament_select(fitness, tournament_k)]
            p2 = pop[tournament_select(fitness, tournament_k)]
            child = crossover(p1, p2, crossover_rate)
            child = mutate(child, mutation_rate, mutation_sigma)
            child = project_to_box_and_sum(child, target_sum)
            new_pop[i] = child

        pop = new_pop
        fitness = evaluate_population(pop)

        cur_best = np.argmin(fitness)
        cur_fit = fitness[cur_best]
        if cur_fit < best_f:
            best_f = cur_fit
            best_h = pop[cur_best].copy()
        gen += 1

    # Local random refinement within remaining time
    while time.time() - start < budget_s - 0.1:
        cand = best_h + np.random.randn(n_points) * 0.05
        cand = project_to_box_and_sum(cand, target_sum)
        c_val = compute_c5(cand)
        if c_val < best_f:
            best_f = c_val
            best_h = cand.copy()

    return (best_h, float(best_f), int(n_points))
