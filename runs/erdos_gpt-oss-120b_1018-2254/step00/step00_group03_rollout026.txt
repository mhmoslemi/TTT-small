    # clip to [0,1]
    x = np.clip(x, 0, 1)
    target = target_sum
    # water-filling
    # while sum != target within tolerance:
    for _ in range(10):
        s = x.sum()
        diff = target - s
        if abs(diff) < 1e-8:
            break
        free = (x > 0) & (x < 1)
        count = free.sum()
        if count == 0:
            # cannot adjust, break
            break
        increment = diff / count
        x[free] += increment
        # clip again

    # final scaling if still off:
    s = x.sum()
    if abs(s - target) > 1e-6:
        # fallback: do scaling
        if s > 0:

            x = np.clip(x, 0, 1)

        v = np.clip(v, 0.0, 1.0)

        for _ in range(10):
            s = v.sum()
            diff = target_sum - s
            if abs(diff) < 1e-12:
                break
            # find free indices
            free = (v > 0) & (v < 1)
            cnt = free.sum()
            if cnt == 0:
                # adjust any variable within bounds
                idx = np.random.randint(n)
                # adjust within bounds
                step = np.clip(diff, -v[idx], 1 - v[idx])
                v[idx] += step
                continue
            # distribute diff equally among free
            v[free] += diff / cnt
            v = np.clip(v, 0.0, 1.0)
        # final correction
        s = v.sum()
        if abs(s - target_sum) > 1e-9:
