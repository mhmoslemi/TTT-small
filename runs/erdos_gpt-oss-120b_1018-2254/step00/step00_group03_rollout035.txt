def project_onto_box_simplex(v, target_sum):
    # binary search for lambda
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(30):
        lam = (lo + hi) / 2.0
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            hi = lam
        else:
            lo = lam
    proj = np.clip(v + (lo + hi)/2.0, 0.0, 1.0)
    # Might be slight error; adjust by scaling small diff
    # Ensure sum exactly target -> adjust by adding small delta uniformly within bounds
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-12:
        # distribute diff proportionally to slackness
        # indices where 0<proj<1
        slack = (proj > 0) & (proj < 1)
        if slack.any():
            proj[slack] += diff / slack.sum()
            proj = np.clip(proj, 0.0, 1.0)  # ensure bounds again
    return proj

def optimize_h(initial_h, dx, time_limit):
    n = len(initial_h)
    target_sum = n/2.0
    h = np.clip(initial_h, 0.0, 1.0)
    h = project_onto_box_simplex(h, target_sum)
    best_h = h.copy()
    best_val = compute_c5(h, dx)
    start = time.time()
    iteration = 0
    # track no_improve counter
    no_improve = 0
    while True:
        # time check
        if time.time() - start > time_limit:
            break
        # compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        cur_val = corr[max_idx] * dx
        d = max_idx - (n - 1)
        # gradient
        grad = gradient_max_c5(h, d)
        # compute step size
        alpha = 0.5 / np.sqrt(iteration + 1)
        # line search try up to 10 backtracks
        improved = False
        for _ in range(10):
            cand = project_onto_box_simplex(h - alpha * grad, target_sum)
            cand_val = compute_c5(cand, dx)
            if cand_val < cur_val - 1e-9:
                h = cand
                cur_val = cand_val
                improved = True
                break
            else:
                alpha *= 0.5
        if not improved:
            # maybe random perturb to escape
            # add small random noise
            noise = (np.random.rand(n) - 0.5) * 0.01
            cand = project_onto_box_simplex(h + noise, target_sum)
            cand_val = compute_c5(cand, dx)
            if cand_val < cur_val:
                h = cand
                cur_val = cand_val
                improved = True
        # update best
        if cur_val < best_val - 1e-9:
            best_val = cur_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        # stop if no improvement for many steps
        if no_improve > 2000:
            break
        iteration += 1
    return best_h, best_val

import time
import random
import numpy as np

# Helper: compute C5 for a given h (dx = 2/n)
def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

# Gradient of the worst shift (the argmax of the correlation)
def gradient_max_c5(h, shift):
    n = h.shape[0]
    grad = np.zeros_like(h)
    if shift >= 0:
        if shift < n:
            grad[: n - shift] += 1.0 - h[shift:]
            grad[shift:] -= h[: n - shift]
    else:
        k = -shift
        if k < n:
            grad[k:] += 1.0 - h[: n - k]
            grad[: n - k] -= h[k:]
    return grad

# Projection onto the intersection of [0,1]^n and the hyperplane sum = target
def project_onto_box_simplex(v, target):
    n = v.shape[0]
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(30):
        lam = (lo + hi) / 2.0
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    proj = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        free = (proj > 0.0) & (proj < 1.0)
        if free.any():
            proj[free] += diff / free.sum()
            proj = np.clip(proj, 0.0, 1.0)
        else:
            # fallback: adjust the smallest element
            i = int(np.argmin(proj))
            proj[i] = np.clip(proj[i] + diff, 0.0, 1.0)
    return proj

# Local gradient descent on the max‑overlap objective
def optimize_h(init_h, dx, time_limit):
    n = init_h.shape[0]
    target = n / 2.0
    h = np.clip(init_h, 0.0, 1.0)
    h = project_onto_box_simplex(h, target)
    best_h = h.copy()
    best_val = compute_c5(h, dx)
    start = time.time()
    it = 0
    no_improve = 0
    while True:
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1 - h, mode="full")
        idx = int(np.argmax(corr))
        cur_val = float(corr[idx] * dx)
        shift = idx - (n - 1)
        grad = gradient_max_c5(h, shift)
        alpha = 0.5 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(10):
            cand = project_onto_box_simplex(h - alpha * grad, target)
            cand_val = compute_c5(cand, dx)
            if cand_val < cur_val - 1e-12:
                h = cand
                cur_val = cand_val
                improved = True
                break
            alpha *= 0.5
        if not improved:
            # small random jitter as escape move
            noise = (np.random.rand(n) - 0.5) * 0.01
            cand = project_onto_box_simplex(h + noise, target)
            cand_val = compute_c5(cand, dx)
            if cand_val < cur_val:
                h = cand
                cur_val = cand_val
                improved = True
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 2000:
            break
        it += 1
    return best_h, best_val

# Build a binary vector (0/1) from a continuous one by keeping the largest entries
def binary_from_continuous(h):
    n = h.shape[0]
    target = int(round(n / 2.0))
    idx = np.argsort(h)[::-1]
    bin_h = np.zeros_like(h)
    bin_h[idx[:target]] = 1.0
    return bin_h

# Simple hill‑climbing on binary vectors: swaps a 1 with a 0 to lower the max‑overlap
def hill_climb_binary(bin_h, dx, time_limit):
    n = bin_h.shape[0]
    cur_val = compute_c5(bin_h, dx)
    best_h = bin_h.copy()
    best_val = cur_val
    start = time.time()
    while time.time() - start < time_limit:
        ones = np.where(bin_h == 1)[0]
        zeros = np.where(bin_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        bin_h[i] = 0.0
        bin_h[j] = 1.0
        new_val = compute_c5(bin_h, dx)
        if new_val < best_val:
            best_val = new_val
            best_h = bin_h.copy()
            # keep the improvement
        else:
            # revert swap
            bin_h[i] = 1.0
            bin_h[j] = 0.0
        if best_val <= 0.3808:
            break
    return best_h, best_val

# Main entry point
def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)

    # Determine starting size and vector
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n = 70
        h0 = np.random.rand(n)
        h0 = project_onto_box_simplex(h0, n / 2.0)

    n = int(h0.shape[0])
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_onto_box_simplex(np.clip(h0, 0.0, 1.0), target_sum)

    total_time = float(budget_s)
    start_total = time.time()
    opt_time = total_time * 0.90  # reserve a little for final checks
    max_restarts = 6
    time_per_restart = opt_time / max_restarts

    best_h = h0.copy()
    best_val = compute_c5(best_h, dx)

    for r in range(max_restarts):
        elapsed = time.time() - start_total
        remaining = total_time - elapsed - 0.5
        if remaining <= 0:
            break
        t_limit = min(time_per_restart, remaining)
        if r == 0:
            init = best_h.copy()
        else:
            rand_vec = np.random.rand(n)
            init = project_onto_box_simplex(rand_vec, target_sum)
        h_opt, val_opt = optimize_h(init, dx, t_limit)
        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()
        if best_val <= 0.3808:
            break

    # Try a binary rounding + local search
    bin_start = binary_from_continuous(best_h)
    elapsed = time.time() - start_total
    remaining = total_time - elapsed - 0.3
    if remaining > 0.1:
        bin_opt, bin_val = hill_climb_binary(bin_start.copy(), dx, remaining)
        if bin_val < best_val:
            best_val = bin_val
            best_h = bin_opt.copy()

    # Final projection to ensure feasibility
    best_h = project_onto_box_simplex(np.clip(best_h, 0.0, 1.0), target_sum)
    final_val = compute_c5(best_h, dx)
    return (best_h, float(final_val), n)
