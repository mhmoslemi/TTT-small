def project(v):
    # lambda bisection
    lower=0.0; upper=1.0; target=S
    # binary search
    lo = -np.max(v) - 1.0
    hi = 1 - np.min(v) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < S:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    # adjust residual
    diff = S - proj.sum()
    if abs(diff) > 1e-12:
        # distribute diff over free indices
        free = np.where((proj > 0) & (proj < 1))[0]
        if free.size > 0:
            inc = diff / free.size
            proj[free] += inc
            # clip again
            proj = np.clip(proj, 0.0, 1.0)
            # global adjust if still off
            diff2 = S - proj.sum()
            if abs(diff2) > 1e-8:
                # adjust uniformly across all entries not at bound as much as possible
                # simple adjust using scaling
                scale = (S) / proj.sum()
                proj = np.clip(proj * scale, 0.0, 1.0)
                # final correction
                # may still be off slight due to clipping; we can iterate.
                diff3 = S - proj.sum()
                if abs(diff3) > 1e-8:
                    # distribute small residual
                    for i in range(len(proj)):
                        if diff3 == 0: break
                        if diff3 > 0 and proj[i] < 1.0:
                            add = min(diff3, 1.0 - proj[i])
                            proj[i] += add
                            diff3 -= add
                        elif diff3 < 0 and proj[i] > 0.0:
                            sub = min(-diff3, proj[i])
                            proj[i] -= sub
                            diff3 += sub
    return proj

def project_to_bounds_sum(v, target):
    # clip to bounds
    v = np.clip(v, 0.0, 1.0)
    # compute excess
    total = v.sum()
    if np.isclose(total, target):
        return v
    # Use lambda method to adjust while preserving bounds
    # We'll solve for lambda such that sum(min(1, max(0, v + lam))) = target
    # Bisection.
    lam_low = -1.0
    lam_high = 1.0
    # Expand bounds if needed
    while np.sum(np.clip(v + lam_low, 0.0, 1.0)) > target:
        lam_low *= 2
    while np.sum(np.clip(v + lam_high, 0.0, 1.0)) < target:
        lam_high *= 2
    for _ in range(50):
        lam_mid = (lam_low + lam_high) / 2.0
        s = np.sum(np.clip(v + lam_mid, 0.0, 1.0))
        if s < target:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    v = np.clip(v + (lam_low + lam_high) / 2.0, 0.0, 1.0)

        mask = (v > 0) & (v < 1)
        if mask.any():
            v[mask] += diff / mask.sum()
            v = np.clip(v, 0.0, 1.0)

import time
import numpy as np
from scipy.optimize import differential_evolution

def _project_to_bounds_sum(v, target, max_iter=50):
    """Project vector v onto [0,1]^n with sum == target."""
    v = np.clip(v, 0.0, 1.0)
    if np.isclose(v.sum(), target):
        return v
    # Find lambda such that sum(clipped(v+lambda)) == target
    lam_low, lam_high = -1.0, 1.0
    # Expand bounds if needed
    while np.sum(np.clip(v + lam_low, 0.0, 1.0)) > target:
        lam_low *= 2.0
    while np.sum(np.clip(v + lam_high, 0.0, 1.0)) < target:
        lam_high *= 2.0
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = np.sum(np.clip(v + lam_mid, 0.0, 1.0))
        if s < target:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    v = np.clip(v + (lam_low + lam_high) * 0.5, 0.0, 1.0)
    diff = target - v.sum()
    if abs(diff) > 1e-12:
        mask = (v > 0.0) & (v < 1.0)
        if np.any(mask):
            v[mask] += diff / mask.sum()
            v = np.clip(v, 0.0, 1.0)
    return v

def _c5_value(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=np.float64)
    else:
        n = 95
        h0 = np.full(n, 0.5, dtype=np.float64)

    target_sum = n / 2.0
    h0 = _project_to_bounds_sum(h0, target_sum)

    best_h = h0.copy()
    best_c5 = _c5_value(best_h)

    # Fourier basis (first few harmonics)
    M = 5
    idx = np.arange(n)
    cos_basis = np.stack([np.cos(2.0 * np.pi * (m + 1) * idx / n) for m in range(M)], axis=0)

    def objective(params):
        h = 0.5 + params @ cos_basis
        h = np.clip(h, 0.0, 1.0)
        h = _project_to_bounds_sum(h, target_sum)
        c5 = _c5_value(h)
        # modest penalty if coefficients become too large (optional)
        l1 = np.sum(np.abs(params))
        if l1 > 0.5:
            c5 += 10.0 * (l1 - 0.5)
        return c5

    bounds = [(-0.5, 0.5) for _ in range(M)]

    def de_stop(xk, convergence):
        return (time.time() - start) > budget_s * 0.9

    result = differential_evolution(
        objective,
        bounds,
        seed=seed,
        polish=False,
        callback=de_stop,
        maxiter=200,
        updating="deferred",
        init="random",
    )

    # Build candidate from DE result
    h_candidate = 0.5 + result.x @ cos_basis
    h_candidate = np.clip(h_candidate, 0.0, 1.0)
    h_candidate = _project_to_bounds_sum(h_candidate, target_sum)
    cand_c5 = _c5_value(h_candidate)
    if cand_c5 < best_c5:
        best_h, best_c5 = h_candidate, cand_c5

    # Simple balanced two‑index hill climbing for final refinement
    step = 0.02
    while (time.time() - start) < budget_s * 0.95:
        i, j = rng.choice(n, size=2, replace=False)
        hi, hj = best_h[i], best_h[j]

        # feasible delta keeping both in [0,1]
        delta_low = max(-step, -hi, -(1.0 - hj))
        delta_high = min(step, 1.0 - hi, hj)
        if delta_low >= delta_high:

        delta = rng.uniform(delta_low, delta_high)

        trial = best_h.copy()
        trial[i] = hi + delta
        trial[j] = hj - delta
        # sum is preserved by construction, bounds already respected
        trial_c5 = _c5_value(trial)
        if trial_c5 < best_c5:
            best_h, best_c5 = trial, trial_c5
            step = min(step * 1.05, 0.1)
        else:
            step *= 0.99

    return best_h, best_c5, n
