def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    import time
    # Determine n_points
    try:
        h_initial = np.array(initial_h_values, dtype=np.float64)
        n = len(h_initial)
    except NameError:
        n = 45
        h_initial = None

    target_sum = n / 2.0
    dx = 2.0 / n

    # Define evaluation function
    def evaluate_vec(h):
        # assume h is numpy array
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Projection to feasible set
    def project_feasible(h):
        # Clip bounds
        h = np.clip(h, 0.0, 1.0)
        current = h.sum()
        diff = target_sum - current
        # Adjust entries not at bound
        if abs(diff) > 1e-12:
            # get indices with slack
            # we will distribute diff proportionally to slack capacity
            # For positive diff, we can increase entries not at 1.
            # For negative diff, decrease entries not at 0.
            if diff > 0:
                # Increase
                # compute capacities
                caps = 1.0 - h
                total_cap = caps.sum()
                if total_cap > 0:
                    # allocate proportionally
                    inc = diff * caps / total_cap
                    h += inc
                    # might exceed 1 due to rounding; clamp again
                else:
                    # all at 1, cannot increase, just leave
                    pass
            else:
                # Decrease
                caps = h  # capacity to decrease to zero
                total_cap = caps.sum()
                if total_cap > 0:
                    dec = (-diff) * caps / total_cap
                    h -= dec
            # clip again
            h = np.clip(h, 0.0, 1.0)
            # after adjustments sum could be off due to round; small correction
            # compute residual
            residual = target_sum - h.sum()
            if abs(residual) > 1e-9:
                # distribute residual to any non-bound entries if possible
                # find any index with room
                for idx in range(n):
                    if residual > 0 and h[idx] < 1.0:
                        add = min(residual, 1.0 - h[idx])
                        h[idx] += add
                        residual -= add
                    elif residual < 0 and h[idx] > 0.0:
                        sub = min(-residual, h[idx])
                        h[idx] -= sub
                        residual += sub
                    if abs(residual) < 1e-12:
                        break
        # final clip
        return np.clip(h, 0.0, 1.0)

    # Initialize h_current
    if h_initial is not None:
        h_current = project_feasible(h_initial)
    else:
        # random binary + half initialization
        ones = int(np.floor(target_sum))
        half = target_sum - ones
        h_current = np.zeros(n)
        perm = np.random.permutation(n)
        if ones > 0:
            h_current[perm[:ones]] = 1.0
        if half > 1e-9:
            h_current[perm[ones]] = half
        # already feasible

    # Ensure sum exact
    h_current = project_feasible(h_current)
    best_h = h_current.copy()
    best_val = evaluate_vec(best_h)

    start_time = time.time()
    # SA parameters
    temp = 0.05
    min_temp = 1e-6
    decay = 0.9995
    max_step = 0.1
    iter_count = 0
    no_improve = 0
    # loop until time budget near exhaustion
    while True:

        if elapsed >= budget_s * 0.95:
            break
        # propose move
        # we attempt limited attempts to find feasible move
        for _ in range(5):
            i, j = np.random.choice(n, size=2, replace=False)
            # allowed delta

            lo = max(-h_current[i], h_current[j] - 1.0)  # decrease i, increase j? Actually delta defined as + to i, - to j.
            # coefficient for delta is added to i, sub from j.
            # So delta must satisfy both constraints.
            if lo >= hi:
                continue
            delta = np.random.uniform(-max_step, max_step)
            delta = max(lo, min(hi, delta))
            if abs(delta) < 1e-12:
                continue
            h_new = h_current.copy()
            h_new[i] += delta
            h_new[j] -= delta
            # Ensure feasible within bounds (should by construction)
            # Evaluate
            val_new = evaluate_vec(h_new)
            break
        else:

            continue
        # accept?

        delta_val = val_new - cur_val
        if delta_val < 0 or np.exp(-delta_val / max(temp, 1e-12)) > np.random.rand():
            h_current = h_new
            cur_val = val_new
            if cur_val < best_val:

                best_h = h_current.copy()
                no_improve = 0
            else:

import time

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h (as an array), its C5 bound, and the number of points."""
    np.random.seed(seed)

    # ------------------------------------------------------------
    # Determine problem size and initial vector
    # ------------------------------------------------------------
    try:
        h_initial = np.array(initial_h_values, dtype=np.float64)
        n = int(len(h_initial))
    except Exception:
        n = 45
        h_initial = None

    target_sum = n / 2.0
    dx = 2.0 / n

    # ------------------------------------------------------------
    # Helper: evaluate C5 for a vector h
    # ------------------------------------------------------------
    def evaluate_vec(h):
        # correlation of h with 1-h, full mode
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr) * dx

    # ------------------------------------------------------------
    # Helper: project any vector onto the feasible set
    # ------------------------------------------------------------
    def project_feasible(h):
        h = np.clip(h, 0.0, 1.0)
        curr = h.sum()
        diff = target_sum - curr
        if abs(diff) > 1e-12:
            if diff > 0:
                # need to increase some entries (capacity = 1 - h)
                cap = 1.0 - h
                tot = cap.sum()
                if tot > 0:
                    inc = diff * cap / tot
                    h += inc
                else:
                    pass
            else:
                # need to decrease some entries (capacity = h)
                cap = h
                tot = cap.sum()
                if tot > 0:
                    dec = (-diff) * cap / tot
                    h -= dec
                else:
                    pass
            h = np.clip(h, 0.0, 1.0)
            # small residual fix (loop at most n times)
            resid = target_sum - h.sum()
            if abs(resid) > 1e-12:
                for idx in range(n):
                    if resid > 0 and h[idx] < 1.0:
                        add = min(resid, 1.0 - h[idx])
                        h[idx] += add
                        resid -= add
                    elif resid < 0 and h[idx] > 0.0:
                        sub = min(-resid, h[idx])
                        h[idx] -= sub
                        resid += sub
                    if abs(resid) < 1e-12:
                        break
        return np.clip(h, 0.0, 1.0)

    # ------------------------------------------------------------
    # Initialise current solution
    # ------------------------------------------------------------
    if h_initial is not None:

        h_current = np.zeros(n, dtype=np.float64)
        perm = np.random.permutation(n)
        if ones > 0:

    h_current = project_feasible(h_current)

    while True:
