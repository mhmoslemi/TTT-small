    s = h.sum()

        return h
    # Use scaling + clipping iteratively
    for _ in range(10):
        if s == 0:
            # all zeros, add equally
            h = np.full_like(h, target_sum / len(h))
            break
        scale = target_sum / s
        h = np.clip(h * scale, 0.0, 1.0)

def repair_h(h, target_sum):
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if np.isclose(s, target_sum, atol=1e-12):
        return h
    # Compute deficit
    diff = target_sum - s
    # If diff > 0, we need to increase some entries not at 1
    # If diff < 0, we need to decrease entries not at 0
    # We'll distribute uniformly among free indices then clip again and repeat
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # indices where h < 1
            free = np.where(h < 1.0)[0]
            if len(free) == 0:
                break
            inc = diff / len(free)
            # add inc but not exceed 1
            h[free] = np.minimum(h[free] + inc, 1.0)
        else:
            # diff < 0: need to reduce
            free = np.where(h > 0.0)[0]
            if len(free) == 0:
                break
            dec = (-diff) / len(free)
            h[free] = np.maximum(h[free] - dec, 0.0)
        s = h.sum()
        diff = target_sum - s
    # final clip
    h = np.clip(h, 0.0, 1.0)
    # Ensure sum close
    # If still off due to rounding, adjust a random free index
    s = h.sum()
    if not np.isclose(s, target_sum, atol=1e-8):
        diff = target_sum - s
        # adjust first free entry
        idx = np.where((h > 0) & (h < 1))[0]
        if len(idx) == 0:
            # find any index not at bound anyway
            idx = np.arange(len(h))
        i = idx[0]
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h
```

def repair_h(h, target_sum):
    """Clip h to [0,1] and adjust its sum to target_sum (exact within tolerance)."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if np.isclose(s, target_sum, atol=1e-12):
        return h
    # Iteratively distribute deficit/excess among non‑boundary entries
    for _ in range(10):
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Need to increase some entries (those < 1)
            free = np.where(h < 1.0)[0]
            if free.size == 0:
                break
            inc = diff / free.size
            h[free] = np.minimum(h[free] + inc, 1.0)
        else:
            # Need to decrease some entries (those > 0)
            free = np.where(h > 0.0)[0]
            if free.size == 0:
                break
            dec = (-diff) / free.size
            h[free] = np.maximum(h[free] - dec, 0.0)
        s = h.sum()
    # Final tiny adjustment if needed
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        idx = np.where((h > 0) & (h < 1))[0]
        if idx.size == 0:
            idx = np.arange(h.size)
        i = idx[0]
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def compute_c5(h):
    """Return max_k ∫ h(x) (1 - h(x+k)) dx for discrete step function h."""
    n = h.size
    dx = 2.0 / n
    # Linear correlation via convolution with reversed complement
    conv = np.convolve(h, (1.0 - h)[::-1], mode='full')
    return np.max(conv) * dx

def tournament_select(pop, scores, tour_size=3):
    """Select one individual from pop using tournament selection."""
    candidates = np.random.choice(len(pop), tour_size, replace=False)
    best = candidates[np.argmin(scores[candidates])]
    return pop[best]

def genetic_optimize(h_initial, n_points, target_sum, time_budget):
    """Simple GA to reduce C5."""
    population_size = 80
    elitism = 2
    mutation_rate = 0.2          # probability per gene
    sigma = 0.2                  # stddev of Gaussian mutation
    # Initialise population
    population = [repair_h(np.random.rand(n_points), target_sum)
                  for _ in range(population_size)]
    population[0] = repair_h(h_initial, target_sum)  # seed with provided guess
    best_h = h_initial.copy()
    best_score = compute_c5(best_h)

    start = time.time()
    while time.time() - start < time_budget:
        # Evaluate
        scores = np.empty(population_size)
        for i, h in enumerate(population):
            scores[i] = compute_c5(h)

        # Update global best
        idx_min = int(np.argmin(scores))
        if scores[idx_min] < best_score:
            best_score = scores[idx_min]
            best_h = population[idx_min].copy()

        # Elitism
        elite_idxs = scores.argsort()[:elitism]
        new_pop = [population[i].copy() for i in elite_idxs]

        # Generate offspring
        while len(new_pop) < population_size:
            parent_a = tournament_select(population, scores)
            parent_b = tournament_select(population, scores)

            # Uniform crossover
            mask = np.random.rand(n_points) < 0.5
            child = np.where(mask, parent_a, parent_b)

            # Mutation
            if np.random.rand() < 0.9:
                mut_mask = np.random.rand(n_points) < mutation_rate
                if mut_mask.any():
                    child[mut_mask] += np.random.normal(scale=sigma,
                                                        size=mut_mask.sum())

            # Repair to feasibility
            child = repair_h(child, target_sum)
            new_pop.append(child)

        population = new_pop
        # Cool down mutation magnitude slowly
        sigma = max(0.01, sigma * 0.995)

    return best_h, best_score
