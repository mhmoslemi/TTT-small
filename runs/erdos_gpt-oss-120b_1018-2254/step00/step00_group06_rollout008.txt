def project_onto_bounds_and_sum(x, target_sum):

    lo = -np.max(x) - 1.0  # enough lower bound to push all to 0

        else:
            hi = mid

    # clip sum tolerance
    s = proj.sum()
    if abs(s - target_sum) > 1e-8:

        diff = target_sum - s
        # find indices not at bounds
        mask = (proj > 0.0 + 1e-12) & (proj < 1.0 - 1e-12)
        count = np.count_nonzero(mask)
        if count > 0:
            proj[mask] += diff / count
            proj = np.clip(proj, 0.0, 1.0)

def project_onto_bounds_and_sum(x, target_sum):
    lo = -np.max(x) - 1.0
    hi = 1.0 - np.min(x) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        clipped = np.clip(x + mid, 0.0, 1.0)
        s = clipped.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)

        cnt = np.count_nonzero(mask)
        if cnt > 0:
            proj[mask] += diff / cnt
            proj = np.clip(proj, 0.0, 1.0)
    return proj

import numpy as np
import time

def _project_bounds_sum(x, target):
    lo = -np.max(x) - 1.0
    hi = 1.0 - np.min(x) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        clipped = np.clip(x + mid, 0.0, 1.0)
        s = clipped.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    s = proj.sum()
    if abs(s - target) > 1e-8:
        diff = target - s
        mask = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt:
            proj[mask] += diff / cnt
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def _c5_max(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _smooth_gradient(h, gamma=5.0):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(gamma * (corr - m))
    w /= w.sum()
    g = np.zeros_like(h)
    for idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        lag = idx - (n - 1)
        if lag >= 0:
            g[: n - lag] += wi * (1.0 - h[lag:])
            g[lag:] -= wi * h[: n - lag]
        else:
            L = -lag
            g[L:] += wi * (1.0 - h[: n - L])
            g[: n - L] -= wi * h[L:]
    return g

def _optimize(start, target_sum, time_limit):
    n = start.shape[0]
    h = _project_bounds_sum(start, target_sum)
    best_h, best_val = h.copy(), _c5_max(h)
    alpha, gamma = 0.1, 5.0
    start_t = time.time()
    while time.time() - start_t < time_limit:
        g = _smooth_gradient(h, gamma)
        h_new = _project_bounds_sum(h - alpha * g, target_sum)
        val_new = _c5_max(h_new)
        if val_new < best_val:
            best_val, best_h = val_new, h_new.copy()
        cur_val = _c5_max(h)
        if val_new <= cur_val:
            h = h_new
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha *= 0.5
        if alpha < 1e-9:
            break
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n = 95
        h0 = rng.random(n)
    target = n / 2.0
    start_total = time.time()
    def _remaining():
        return max(0.0, budget_s - (time.time() - start_total))
    best_h, best_val = None, np.inf
    # primary run from provided construction
    if 'initial_h_values' in globals():
        alloc = _remaining() * 0.6
        h_opt, v_opt = _optimize(h0, target, alloc)
        if v_opt < best_val:
            best_h, best_val = h_opt, v_opt
    # random restarts
    while _remaining() > 2.0:
        chunk = min(5.0, _remaining() * 0.3)
        h_start = rng.random(n)
        h_start = _project_bounds_sum(h_start, target)
        h_opt, v_opt = _optimize(h_start, target, chunk)
        if v_opt < best_val:
            best_h, best_val = h_opt, v_opt
    if best_h is None:
        best_h = _project_bounds_sum(h0, target)
        best_val = _c5_max(best_h)
    else:
        best_h = _project_bounds_sum(best_h, target)
        best_val = _c5_max(best_h)
    return best_h, best_val, n
