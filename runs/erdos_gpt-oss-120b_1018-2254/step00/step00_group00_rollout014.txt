def repair_initial(h):
    # Clip bounds
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target = n / 2
    diff = target - current_sum
    if abs(diff) > 1e-8:

        for idx in range(n):
            max_add = 1 - h[idx]
            min_add = -h[idx]
            add = np.clip(diff, min_add, max_add)
            h[idx] += add
            diff -= add

import time, math, random
import numpy as np

def evaluate_overlap(h):
    """Return C5 for step function h (numpy array)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def repair_vector(h):
    """Clamp to [0,1] and enforce sum_i h_i = n/2."""
    n = h.size
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = target - cur
    if abs(diff) > 1e-12:
        # distribute diff to entries with room
        for idx in range(n):
            if diff == 0.0:
                break
            if diff > 0.0:
                room = 1.0 - h[idx]
                add = min(room, diff)
                h[idx] += add
                diff -= add
            else:
                room = -h[idx]          # most we can subtract (negative)
                sub = max(room, diff)   # diff is negative
                h[idx] += sub
                diff -= sub
    h = np.clip(h, 0.0, 1.0)

    cur = np.sum(h)

        for idx in range(n):
            if 0.0 < h[idx] < 1.0:
                h[idx] += adj
                break
        h = np.clip(h, 0.0, 1.0)
