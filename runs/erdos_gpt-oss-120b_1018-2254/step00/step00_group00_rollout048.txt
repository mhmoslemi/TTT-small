def repair_sum(h, target):

    s = h.sum()
    diff = target - s
    # tolerance
    eps = 1e-12
    max_iter = 1000
    it=0
    while abs(diff) > eps and it < max_iter:
        if diff > 0:
            # find indices where h < 1
            caps = 1.0 - h
            total_cap = caps.sum()
            if total_cap < eps:
                break
            # allocate proportionally to capacity
            add = np.minimum(caps, diff * caps / total_cap)
            h += add
        else:

import time
import numpy as np

# ----- Helper functions -----------------------------------------------------

def evaluate(h):
    """Return C5 value for discretized step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_to_sum(v, target):
    """Project vector v onto box [0,1] with sum(v)=target via a shift+clip."""
    # Binary search for λ such that sum(clip(v+λ,0,1)) = target
    lo, hi = -1.0, 1.0
    # Expand bounds if needed
    while np.clip(v + lo, 0.0, 1.0).sum() > target:
        lo -= 1.0
    while np.clip(v + hi, 0.0, 1.0).sum() < target:
        hi += 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(v + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def repair_sum(h, target):
    """Enforce 0≤h≤1 and sum(h)=target."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) <= 1e-12:
        return h
    # Use projection by shifting uniformly then clipping
    return project_to_sum(h, target)


def uniform_crossover(p1, p2):
    """Blend two parents gene‑wise with random weights."""
    alpha = np.random.rand(p1.size)
    return alpha * p1 + (1.0 - alpha) * p2


def gaussian_mutation(v, sigma=0.1):
    """Additive Gaussian noise."""
    return v + np.random.randn(v.size) * sigma


def tournament_select(pop, fit, k=3):
    """Return a parent chosen by k‑tournament."""
    idx = np.random.choice(len(pop), k, replace=False)
    best = idx[0]
    best_f = fit[best]
    for i in idx[1:]:
        if fit[i] < best_f:
            best = i
            best_f = fit[i]
    return pop[best]


def local_search(start_h, target_sum, time_budget, start_time):
    """Simple hill‑climbing with adaptive step size."""
    best_h = start_h.copy()
    best_score = evaluate(best_h)
    step = 0.2
    while time.time() - start_time < time_budget:
        # perturb a random coordinate
        cand = best_h.copy()
        i = np.random.randint(cand.size)
        delta = (np.random.rand() - 0.5) * 2.0 * step
        cand[i] = np.clip(cand[i] + delta, 0.0, 1.0)
        cand = repair_sum(cand, target_sum)
        sc = evaluate(cand)
        if sc < best_score:
            best_score = sc
            best_h = cand
            step = max(step * 0.9, 1e-4)
        else:
            step = min(step * 1.05, 0.5)
    return best_h, best_score
