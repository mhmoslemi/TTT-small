def project_box_sum(v, target):
    # v: 1d np.ndarray
    # Returns w with same shape, 0<=w<=1, sum(w)=target
    # Use bisection on tau
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        tau = (lo + hi) * 0.5
        w = np.clip(v - tau, 0, 1)
        s = w.sum()
        if abs(s - target) < 1e-12:
            return w
        if s > target:
            lo = tau
        else:
            hi = tau
    return np.clip(v - tau, 0, 1)

import time, random, math
import numpy as np

def _compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k)) dx approximated on a grid."""
    N = len(h)
    dx = 2.0 / N
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def _project_box_sum(v, target):
    """Project v onto the box [0,1] with sum(v)=target."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        tau = (lo + hi) * 0.5
        w = np.clip(v - tau, 0.0, 1.0)
        s = w.sum()
        if abs(s - target) < 1e-12:
            return w
        if s > target:
            lo = tau
        else:
            hi = tau
    return np.clip(v - tau, 0.0, 1.0)

def _initial_h(N, seed):
    """Construct a feasible initial h, using the provided initial_h_values if present."""
    random.seed(seed)
    np.random.seed(seed)
    try:
        base = np.array(initial_h_values, dtype=float)  # provided by harness
        N0 = base.size
        if N0 == N:
            h = base.copy()
        else:
            x0 = np.linspace(0.0, 2.0, N0, endpoint=False)
            x = np.linspace(0.0, 2.0, N, endpoint=False)
            h = np.interp(x, x0, base)
    except NameError:
        h = np.random.rand(N)
    h = _project_box_sum(h, N / 2.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise the Erdős C5 overlap constant.
    Returns (h_values, c5_bound, n_points).
    """
    start = time.time()
    rng = random.Random(seed)
    np.random.seed(seed)

    # Determine resolution: upsample initial data a few times but stay modest.
    try:
        base_len = len(initial_h_values)
    except NameError:
        base_len = 45
    factor = 4
    N = int(min(base_len * factor, 200))  # keep N <= 200 for speed
    if N < 30:
        N = 30

    # Initialise current and best solutions
    h_cur = _initial_h(N, seed)
    best_h = h_cur.copy()
    best_val = _compute_c5(best_h)

    # Main optimisation loop
    no_improve = 0
    # parameters for adaptive step size
    while time.time() - start < budget_s * 0.95:
        elapsed = time.time() - start
        t_frac = elapsed / budget_s
        step_scale = max(0.001, 0.5 * (1.0 - t_frac))  # shrink over time

        # correlation for current h (also gives current C5)
        corr = np.correlate(h_cur, 1 - h_cur, mode='full')
        cur_val = corr.max() * (2.0 / N)
        # decide move type
        if rng.random() < 0.7:
