import numpy as np
import time

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def repair(v, target):
    # Ensure v within [0,1] and sum=v target
    v = np.clip(v, 0.0, 1.0)
    max_iter = 1000
    tol = 1e-12
    for _ in range(max_iter):
        s = v.sum()
        diff = target - s
        if abs(diff) <= tol:
            break
        free = (v > tol) & (v < 1.0 - tol)
        free_count = free.sum()
        if free_count == 0:
            # adjust a single element if possible
            idx = np.argmin(np.abs(v - 0.5))  # find closest to interior
            v[idx] = np.clip(v[idx] + diff, 0.0, 1.0)
            break
        adj = diff / free_count
        v[free] += adj
        v = np.clip(v, 0.0, 1.0)
    # final fix for numeric drift
    diff = target - v.sum()
    if abs(diff) > tol:
        # apply to any non‑boundary entry
        for i in range(v.shape[0]):
            if 0.0 <= v[i] + diff <= 1.0:
                v[i] += diff
                break
    return v

def generate_random_feasible(n, target):
    v = np.random.rand(n)
    return repair(v, target)

def mutate(v, mutation_rate, sigma, target):
    n = v.shape[0]
    mask = np.random.rand(n) < mutation_rate
    if mask.any():
        v = v.copy()
        v[mask] += np.random.randn(mask.sum()) * sigma
    return repair(v, target)

def crossover(p1, p2, pc):
    if np.random.rand() < pc:
        pt = np.random.randint(1, p1.shape[0])
        c1 = np.concatenate([p1[:pt], p2[pt:]])
        c2 = np.concatenate([p2[:pt], p1[pt:]])
        return c1, c2
    else:
        return p1.copy(), p2.copy()

def tournament(pop, scores, k=3):
    indices = np.random.choice(len(pop), k, replace=False)
    best = indices[0]
    best_score = scores[best]
    for idx in indices[1:]:
        if scores[idx] < best_score:
            best = idx
            best_score = scores[idx]
    return pop[best]

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    # Determine problem size
    if 'initial_h_values' in globals():
        init_h = np.array(initial_h_values, dtype=np.float64)
        n = init_h.shape[0]
    else:
        n = 95
        init_h = generate_random_feasible(n, n/2.0)
    target_sum = n / 2.0

    # Ensure initial feasibility
    init_h = repair(init_h, target_sum)

    pop_size = 120
    population = [generate_random_feasible(n, target_sum) for _ in range(pop_size - 1)]
    population.append(init_h.copy())

    scores = [compute_c5(ind) for ind in population]
    best_idx = int(np.argmin(scores))
    best_h = population[best_idx].copy()
    best_score = scores[best_idx]

    start = time.time()
    generation = 0
    pc = 0.7
    mutation_rate = 0.02
    sigma = 0.05

    while time.time() - start < budget_s:
        generation += 1
        new_pop = [best_h.copy()]  # elitism
        while len(new_pop) < pop_size:
            p1 = tournament(population, scores, k=3)
            p2 = tournament(population, scores, k=3)
            c1, c2 = crossover(p1, p2, pc)
            c1 = mutate(c1, mutation_rate, sigma, target_sum)
            c2 = mutate(c2, mutation_rate, sigma, target_sum)
            new_pop.append(c1)
            if len(new_pop) < pop_size:
                new_pop.append(c2)
        new_scores = [compute_c5(ind) for ind in new_pop]
        # update best
        for h, s in zip(new_pop, new_scores):
            if s < best_score:
                best_score = s
                best_h = h.copy()
                if best_score <= 0.3808:
                    # reached target bound, stop early
                    population = new_pop
                    scores = new_scores
                    break
        population = new_pop
        scores = new_scores
        if time.time() - start >= budget_s - 0.5:
            break

    # Local refinement
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    if remaining > 1.0:
        # simple hill climbing with small steps
        steps = int(remaining * 8)  # about 8 evaluations per second
        for _ in range(steps):
            cand = mutate(best_h, mutation_rate=0.05, sigma=0.02, target=target_sum)
            cand_score = compute_c5(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_h = cand.copy()
            if time.time() - start > budget_s - 0.1:
                break

    return (best_h, best_score, n)

import numpy as np
import time

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def repair(v, target):
    v = np.clip(v, 0.0, 1.0)
    tol = 1e-12
    for _ in range(1000):
        s = v.sum()
        diff = target - s
        if abs(diff) <= tol:
            break
        free = (v > tol) & (v < 1.0 - tol)
        cnt = free.sum()
        if cnt == 0:
            idx = np.argmin(np.abs(v - 0.5))
            v[idx] = np.clip(v[idx] + diff, 0.0, 1.0)
            break
        adj = diff / cnt
        v[free] += adj
        v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    if abs(diff) > tol:
        for i in range(v.shape[0]):
            if 0.0 <= v[i] + diff <= 1.0:
                v[i] += diff
                break
    return v

def generate_random_feasible(n, target):
    v = np.random.rand(n)
    return repair(v, target)

def mutate(v, mutation_rate, sigma, target):
    n = v.shape[0]
    mask = np.random.rand(n) < mutation_rate
    if mask.any():
        v = v.copy()
        v[mask] += np.random.randn(mask.sum()) * sigma
    return repair(v, target)

def crossover(p1, p2, pc):
    if np.random.rand() < pc:
        pt = np.random.randint(1, p1.shape[0])
        c1 = np.concatenate([p1[:pt], p2[pt:]])
        c2 = np.concatenate([p2[:pt], p1[pt:]])
        return c1, c2
    return p1.copy(), p2.copy()

def tournament(pop, scores, k=3):
    idx = np.random.choice(len(pop), k, replace=False)
    best = idx[0]
    best_score = scores[best]
    for i in idx[1:]:
        if scores[i] < best_score:
            best = i
            best_score = scores[i]
    return pop[best]

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    if "initial_h_values" in globals():
        init_h = np.array(initial_h_values, dtype=np.float64)
        n = init_h.shape[0]
    else:
        n = 95
        init_h = generate_random_feasible(n, n / 2.0)
    target_sum = n / 2.0
    init_h = repair(init_h, target_sum)

    pop_size = 120
    population = [generate_random_feasible(n, target_sum) for _ in range(pop_size - 1)]
    population.append(init_h.copy())

    scores = [compute_c5(ind) for ind in population]
    best_idx = int(np.argmin(scores))
    best_h = population[best_idx].copy()
    best_score = scores[best_idx]

    start = time.time()
    pc = 0.7
    mut_rate = 0.02
    sigma = 0.05

    while time.time() - start < budget_s:
        new_pop = [best_h.copy()]  # elitism
        while len(new_pop) < pop_size:
            p1 = tournament(population, scores, k=3)
            p2 = tournament(population, scores, k=3)
            c1, c2 = crossover(p1, p2, pc)
            c1 = mutate(c1, mut_rate, sigma, target_sum)
            c2 = mutate(c2, mut_rate, sigma, target_sum)
            new_pop.append(c1)
            if len(new_pop) < pop_size:
                new_pop.append(c2)
        new_scores = [compute_c5(ind) for ind in new_pop]

        for h, s in zip(new_pop, new_scores):
            if s < best_score:
                best_score = s
                best_h = h.copy()
                if best_score <= 0.3808:
                    break
        population = new_pop
        scores = new_scores

        if time.time() - start >= budget_s - 0.5:
            break

    # local refinement
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    if remaining > 1.0:
        steps = int(remaining * 8)
        for _ in range(steps):
            cand = mutate(best_h, mutation_rate=0.05, sigma=0.02, target=target_sum)
            cand_score = compute_c5(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_h = cand.copy()
            if time.time() - start > budget_s - 0.1:
                break

    return (best_h, best_score, n)
