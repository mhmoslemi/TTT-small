import time
import numpy as np

def _project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project vector v onto the set {0 <= x_i <= 1, sum(x)=target}."""
    # binary search for Lagrange multiplier lambda so that sum(clamp(v - λ,0,1)) = target
    lo, hi = -1.0, 2.0
    for _ in range(40):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # tiny residual correction (helps when floating error leaves sum off by ~1e-12)
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        free = np.where((w > 1e-12) & (w < 1.0 - 1e-12))[0]
        if free.size:
            w[free[0]] += diff
    return w

def _c5_of(h: np.ndarray) -> float:
    """Compute C₅ for the given step values."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))

def _beatty_h(n: int, alpha: float) -> np.ndarray:
    """Generate a 0‑1 pattern from a Beatty sequence and project to feasibility."""
    idx = np.arange(n, dtype=np.float64)
    vals = (idx * alpha) % 1.0
    h = (vals < 0.5).astype(np.float64)   # binary pattern
    h = _project_box_sum(h, n / 2.0)
    return h

    best_h = None
