def projected_gradient_descent(h, max_iters, time_budget):
    start = time.time()
    best_h = h.copy()
    best_val = compute_c5(h)
    alpha0 = 0.1  # initial step size
    for it in range(max_iters):
        if time.time() - start > time_budget:
            break
        conv = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(conv)
        k = max_idx - (len(h) - 1)  # shift (positive or negative)
        # compute gradient for shift k
        n = len(h)
        g = np.ones_like(h)
        if k >= 0:
            g[:n-k] -= h[k:]  # for i from 0 to n-k-1, subtract h[i+k]
            g[k:] -= h[:n-k]  # for i from k to n-1, subtract h[i-k]
        else:
            kk = -k
            g[kk:] -= h[:n-kk]
            g[:n-kk] -= h[kk:]
        # step size schedule: alpha = alpha0 / sqrt(it+1)
        alpha = alpha0 / np.sqrt(it + 1)
        h_new = h - alpha * g
        # project to constraints
        h_new = project_to_constraints(h_new, target_sum)
        c5_new = compute_c5(h_new)

        if c5_new < compute_c5(h):
            h = h_new
        else:
            # maybe keep old h, or occasionally accept for exploration:
            # with probability exp(-(c5_new - old)/temp)
            # use temperature decaying

def improve_h(h, target_sum, time_limit, best_val):
    start = time.time()
    val = compute_c5(h)
    best_h_local = h.copy()
    best_val_local = val
    n = len(h)
    alpha0 = 0.5
    iter = 0
    while time.time() - start < time_limit:

import numpy as np
import time
import random

def compute_c5(h):
    """Return C5 for a given step function h sampled on n points."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h), zero‑padded
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_to_constraints(v, target, tol=1e-12, max_iter=500):
    """Project vector v onto the box [0,1]^n while enforcing sum(v)=target."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        s = v.sum()
        diff = target - s
        if abs(diff) <= tol:
            break
        free = (v > 0.0) & (v < 1.0)
        n_free = int(free.sum())
        if n_free == 0:                     # all at bounds, distribute uniformly
            v += diff / v.size
            v = np.clip(v, 0.0, 1.0)
            continue
        delta = diff / n_free
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    # final tiny correction
    s = v.sum()
    if abs(s - target) > 1e-9:
        v += (target - s) / v.size
        v = np.clip(v, 0.0, 1.0)
    return v

        # worst shift
        max_idx = int(np.argmax(corr))
        k = max_idx - (n - 1)      # shift amount, can be negative
        # gradient of the active term
        g = np.ones_like(h)
        if k > 0:
            g[:n - k] -= h[k:]
            g[k:]   -= h[:n - k]
        elif k < 0:
            p = -k
            g[p:]   -= h[:n - p]
            g[:n - p] -= h[p:]
        else:
            g -= 2.0 * h
        # step size decays gently
        alpha = 0.5 / (1.0 + it * 0.001)
        h_new = h - alpha * g
        h_new = project_to_constraints(h_new, target_sum)
        new_val = compute_c5(h_new)
        # record best ever
        if new_val < best_val:
            best_val = new_val
            best_h = h_new.copy()
        # acceptance
        if new_val < cur_val or random.random() < 0.01:
            h = h_new
            cur_val = new_val
            corr = np.correlate(h, 1.0 - h, mode="full")
        # early exit if we already beat the target
        if best_val < stop_val:
            break
        it += 1
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function minimizing the Erdős C5 integral."""
    np.random.seed(seed)
    random.seed(seed)

    # Discretisation size – modestly larger than the provided 42 points.
    n = 200
    target_sum = n / 2.0                 # ensures ∫ h = 1

    # Initialise from the supplied construction when available.
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n0 = init.shape[0]
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h0 = np.interp(x_new, x0, init)
    else:
        h0 = np.random.rand(n)

    h0 = project_to_constraints(h0, target_sum)

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    start_time = time.time()

    # Primary intensive descent.
    primary_budget = budget_s * 0.6
    h_opt, val_opt = descent_improve(best_h, target_sum, primary_budget, 0.3808)
    if val_opt < best_val:
        best_h, best_val = h_opt, val_opt

    # Secondary quick restarts for diversification.
    while time.time() - start_time < budget_s:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 2.0:
            break
        h_rand = np.random.rand(n)
        h_rand = project_to_constraints(h_rand, target_sum)
        slice_budget = min(5.0, remaining * 0.3)
        h_loc, val_loc = descent_improve(h_rand, target_sum, slice_budget, best_val)
        if val_loc < best_val:
            best_h, best_val = h_loc, val_loc

    # Final projection to guarantee feasibility and recompute exact bound.
    best_h = project_to_constraints(best_h, target_sum)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
