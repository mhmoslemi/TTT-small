def repair_sum(h):
    target = len(h)/2
    # Clip to [0,1] first
    np.clip(h, 0, 1, out=h)
    # Compute sum
    s = np.sum(h)
    diff = target - s
    max_iter=10
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to increase
            room = 1 - h
            capacity = np.sum(room)
            if capacity < 1e-12:
                break
            inc = diff * room / capacity
            h += inc
        else:
            # diff < 0: need to decrease
            room = h
            capacity = np.sum(room)
            if capacity < 1e-12:
                break
            dec = -diff * room / capacity
            h -= dec
        # clip again
        np.clip(h, 0, 1, out=h)
        s = np.sum(h)
        diff = target - s
    return h
```

        h *= target / s
    return h

    dx = 2.0 / n

    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def repair_sum(h: np.ndarray) -> np.ndarray:
    """Adjust h so that sum(h) == len(h)/2 while keeping 0≤h≤1."""
    n = h.size
    target = n / 2.0
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    # iterative proportional adjustment
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            cap = room.sum()
            if cap <= 0:
                break
            inc = diff * room / cap
            h += inc
        else:
            room = h
            cap = room.sum()
            if cap <= 0:
                break
            dec = -diff * room / cap
            h -= dec
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
    # tiny residual correction
    if abs(diff) > 1e-12:
        idx = np.where((h > 0) & (h < 1))[0]
        if idx.size:
            adj = diff / idx.size
            h[idx] += adj
        else:
            # fallback: scale all entries
            h *= target / h.sum()
    np.clip(h, 0.0, 1.0, out=h)
    return h
