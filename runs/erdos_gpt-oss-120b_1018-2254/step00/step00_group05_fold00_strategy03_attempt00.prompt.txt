    h_clip = np.clip(h, 0.001, 1-0.001)
    total = np.sum(h_clip)
    desired = n/2.0
    h_adjusted = h_clip * (desired/total)
    h_adjusted = np.clip(h_adjusted, 0.001, 0.999)

def crossover(p1, p2, seed=None):
