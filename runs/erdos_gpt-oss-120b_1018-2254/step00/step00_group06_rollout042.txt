def compute_c5_fft(h):
    n = len(h)
    m = 2*n
    H = np.fft.rfft(h, m)
    B = np.fft.rfft(1 - h, m)
    corr = np.fft.irfft(H * B, m)
    # correlation length = 2*n-1 (centered)
    # lags from -(n-1) to n-1 inclusive correspond to corr[n-1 : 2*n-1]
    max_corr = np.max(corr[n-1: 2*n-1])
    return dx * max_corr
```

def project_h(h):
    # Clip to 0,1
    h = np.clip(h, 0.0, 1.0)
    target = n_points/2.0
    total = h.sum()
    if total == target:
        return h
    # Adjust by adding delta to all entries while staying in bounds.
    # Use bias = (target - total) / n_points
    # Then shift each element by bias, then clip again. Repeat a few times.
    for _ in range(5):
        bias = (target - h.sum()) / n_points
        h += bias
        h = np.clip(h, 0.0, 1.0)
        # after clipping, sum may still differ, but loop again
    # final fix: if still off, adjust one element within bounds
    diff = target - h.sum()
    if abs(diff) > 1e-9:
        # find index where adding diff wouldn't exceed bounds
        # else adjust proportionally.
        for i in range(len(h)):
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-9:
                break
    return h

import numpy as np
import time
import random
import math

def _project_to_feasible(v, target, tol=1e-12, max_iter=60):
    """Project vector v onto the box [0,1] and the hyperplane sum(v)=target."""
    lo, hi = -2.0, 2.0
    # expand bounds until the clipped sum brackets the target
    while np.clip(v + lo, 0.0, 1.0).sum() > target:
        lo -= 1.0
    while np.clip(v + hi, 0.0, 1.0).sum() < target:
        hi += 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

def _max_corr_and_lag(h):
    """Return (max_corr, lag) for correlation of h with (1-h)."""
    n = h.shape[0]
    m = 2 * n
    H = np.fft.rfft(h, m)
    B = np.fft.rfft(1.0 - h, m)
    corr = np.fft.irfft(H * B, m)[:2 * n - 1]
    idx = int(np.argmax(corr))
    lag = idx - (n - 1)
    return float(corr[idx]), int(lag)

def _gradient_for_lag(h, lag):
    """Gradient of Σ_i h_i h_{i+lag} w.r.t. h (ignoring sum constraint)."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if lag >= 0:
        d = lag
        g[: n - d] += h[d:]
        g[d:] += h[: n - d]
    else:
        d = -lag
        g[d:] += h[: n - d]
        g[: n - d] += h[d:]
    return g

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    # starting vector
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # fallback: uniform 0.5 on 95 points
        n = 95
        h = np.full(n, 0.5, dtype=float)

    n = h.shape[0]
    target = n / 2.0
    dx = 2.0 / n

    # enforce feasibility
    h = _project_to_feasible(h, target)

    max_corr, lag = _max_corr_and_lag(h)
    cur_score = dx * max_corr
    best_h = h.copy()
    best_score = cur_score

    start = time.time()
    budget = float(budget_s)

    # annealing / step‑size schedule parameters
    eta0 = 0.9
    eta_min = 1e-4
    T0 = 0.02
    Tf = 1e-5
    it = 0
    while True:
        elapsed = time.time() - start
        if elapsed >= budget * 0.98:
            break
        it += 1
        frac = elapsed / budget
        # temperature for occasional uphill moves
        T = T0 * (1.0 - frac) + Tf * frac
        # step size for gradient moves
        eta = eta_min + (eta0 - eta_min) * (1.0 - frac)

        # Decide move type
        if random.random() < 0.75:  # gradient move
            g = _gradient_for_lag(h, lag)
            cand = h + eta * g
        else:  # stochastic perturbation
            sigma = 0.05 * (1.0 - frac) + 1e-4
            cand = h + sigma * np.random.randn(n)

        cand = _project_to_feasible(cand, target)
        cand_corr, cand_lag = _max_corr_and_lag(cand)
        cand_score = dx * cand_corr

        # acceptance rule (simulated annealing)
        if cand_score < cur_score or math.exp(-(cand_score - cur_score) / (T + 1e-16)) > random.random():
            h = cand
            cur_score = cand_score
            lag = cand_lag
            if cand_score < best_score:

                best_h = cand.copy()

        # occasional localized two‑point exchange for extra exploration
        if it % 1200 == 0:
