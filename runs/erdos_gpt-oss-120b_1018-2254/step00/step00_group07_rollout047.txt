def project_to_feasible(h, target_sum):
    for _ in range(10):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        h += diff / len(h)
    h = np.clip(h, 0.0, 1.0)
    # final adjust to exactly sum: compute residual and adjust on indices not at bounds.
    s = h.sum()
    diff = target_sum - s
    if abs(diff) > 1e-12:
        # distribute across indices with slack
        slack = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
        if slack.size > 0:
            incr = diff / slack.size
            h[slack] += incr
        # clip again
        h = np.clip(h, 0.0, 1.0)
    return h

import time
import numpy as np
from scipy.optimize import minimize

def all_correlations(h, dx):
    """Return full correlation array of h with (1-h) using zero padding."""
    N = h.shape[0]
    out = np.empty(2 * N - 1, dtype=float)
    # shifts from -(N-1) … N-1
    for idx, s in enumerate(range(-(N - 1), N)):
        if s >= 0:
            total = np.dot(h[: N - s], 1.0 - h[s:])
        else:
            k = -s
            total = np.dot(h[k:], 1.0 - h[: N - k])
        out[idx] = total * dx
    return out


def max_corr(h, dx):
    """Maximum overlap value C5 for a given h."""
    return np.max(all_correlations(h, dx))


def project_to_feasible(x, target_sum):
    """Project x onto box [0,1]^n and the hyperplane sum(x)=target_sum."""
    x = np.asarray(x, dtype=float).copy()
    # iterative simple projection
    for _ in range(10):
        x = np.clip(x, 0.0, 1.0)
        s = x.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        x += diff / x.size
    x = np.clip(x, 0.0, 1.0)
    # final tiny correction on non‑boundary entries
    s = x.sum()
    diff = target_sum - s
    if abs(diff) > 1e-9:
        slack = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        if slack.size:
            incr = diff / slack.size
            x[slack] += incr
            x = np.clip(x, 0.0, 1.0)
    return x


class SumConstraint:
    """Equality ∑h = target_sum."""
    def __init__(self, N, target):
        self.N = N
        self.target = target

    def __call__(self, x):
        return np.sum(x[: self.N]) - self.target


class CorrConstraint:
    """Inequality t >= correlation at a given shift."""
    def __init__(self, shift, N, dx):
        self.shift = shift
        self.N = N
        self.dx = dx

    def __call__(self, x):
        h = x[: self.N]
        t = x[self.N]
        s = self.shift
        if s >= 0:
            total = np.dot(h[: self.N - s], 1.0 - h[s:])
        else:
            k = -s
            total = np.dot(h[k:], 1.0 - h[: self.N - k])
        return t - total * self.dx


def objective(x):
    """Minimize the auxiliary variable t."""
    return x[-1]


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # ----- Determine problem size and initial vector -----
    try:

    constraints = []
    constraints.append({"type": "eq", "fun": SumConstraint(N, target_sum)})

    bounds = [(0.0, 1.0)] * N + [(0.0, None)]

    best_t = max_corr(best_h, dx)

    start_time = time.time()
    attempt = 0
