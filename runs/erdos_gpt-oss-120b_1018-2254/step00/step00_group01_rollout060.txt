    x = np.clip(v, 0.0, 1.0)

    s = x.sum()

    residual = target - s

    # We'll iteratively adjust.
    # This algorithm chooses free indices where adjustments possible.
    # Make while loop:
    max_iter = 1000

        # or decreased if residual <0 (x >0)
        if residual > 0:

    x = np.clip(v, 0.0, 1.0)

import time
import numpy as np

def _project_bounds_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and adjust to meet the exact sum `target`."""
    x = np.clip(v, 0.0, 1.0)
    residual = target - x.sum()
    if abs(residual) < 1e-12:
        return x
    # Iteratively distribute the residual over free coordinates
    for _ in range(20):
        if residual > 0:
            free = np.where(x < 1.0 - 1e-12)[0]
        else:
            free = np.where(x > 0.0 + 1e-12)[0]
        if free.size == 0:
            break
        delta = residual / free.size
        if delta > 0:
            adj = np.minimum(delta, 1.0 - x[free])
        else:
            adj = np.maximum(delta, -x[free])
        x[free] += adj
        residual = target - x.sum()
        if abs(residual) < 1e-12:
            break
    return np.clip(x, 0.0, 1.0)


def _c5_of(h: np.ndarray, dx: float) -> float:
    """Compute the overlap constant C5 for a given step function."""
    # full correlation of h with (1-h)
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős overlap constant C5.

    Returns
    -------
    (h_values, c5_bound, n_points)
        h_values : 1‑D numpy array of length n_points, values in [0,1] and sum = n_points/2
        c5_bound : float, the achieved upper bound on C5
        n_points : int, the number of samples used (length of h_values)
    """
    start = time.time()
    np.random.seed(seed)

    # ----- initialise ---------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=np.float64).copy()
    else:
        n = 47
        v = np.random.rand(n)
        h = _project_bounds_sum(v, n / 2.0)

    n = len(h)
    target_sum = n / 2.0
    dx = 2.0 / n
    h = _project_bounds_sum(h, target_sum)

    best_h = h.copy()
    best_c5 = _c5_of(best_h, dx)
    cur_h = h.copy()
    cur_c5 = best_c5

    # ----- optimisation parameters -------------------------------------------
    temp = 0.02               # initial temperature for Metropolis acceptance
    decay = 0.99995           # exponential cooling per iteration
    no_improve = 0            # counter for stagnation

    # ----- main loop ---------------------------------------------------------
    while time.time() - start < budget_s * 0.95:
        # 70 % of the time try a move that targets the current worst shift
        if np.random.rand() < 0.7:
            corr_full = np.correlate(cur_h, 1.0 - cur_h, mode="full")
            max_idx = int(np.argmax(corr_full))
            shift = max_idx - (n - 1)
            if shift != 0:
                # indices i such that i+shift stays in range
                if shift > 0:
                    i_cand = np.arange(0, n - shift)
                else:
                    i_cand = np.arange(-shift, n)
                contrib = cur_h[i_cand] * (1.0 - cur_h[i_cand + shift])
                total = contrib.sum()
                if total > 0:
                    probs = contrib / total
                    i = np.random.choice(i_cand, p=probs)
                    j = i + shift
                    max_delta = min(cur_h[i], 1.0 - cur_h[j])
                    if max_delta > 0:
                        delta = np.random.rand() * max_delta
                        new_h = cur_h.copy()
                        new_h[i] -= delta
                        new_h[j] += delta
                        new_c5 = _c5_of(new_h, dx)
                        if (new_c5 < cur_c5) or (np.random.rand() <
                                                 np.exp(-(new_c5 - cur_c5) / max(temp, 1e-12))):
                            cur_h, cur_c5 = new_h, new_c5
                            no_improve = 0
                        else:
                            no_improve += 1
                        if new_c5 < best_c5:
                            best_c5, best_h = new_c5, new_h.copy()
                        temp *= decay
                        continue

        # ----- fallback: random pair transfer ---------------------------------
        i, j = np.random.choice(n, size=2, replace=False)
        hi, hj = cur_h[i], cur_h[j]
        delta_min = max(-hi, hj - 1.0)
        delta_max = min(1.0 - hi, hj)
        if delta_min >= delta_max:
            continue
        delta = np.random.uniform(delta_min, delta_max)
        if abs(delta) < 1e-12:
            continue
        new_h = cur_h.copy()
        new_h[i] = hi + delta
        new_h[j] = hj - delta
        new_c5 = _c5_of(new_h, dx)

        if (new_c5 < cur_c5) or (np.random.rand() <
                                 np.exp(-(new_c5 - cur_c5) / max(temp, 1e-12))):

    best_c5 = _c5_of(best_h, dx)
