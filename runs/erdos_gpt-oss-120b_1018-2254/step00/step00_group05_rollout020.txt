    g = np.full(N, dx)  # sum of w =1 term
    for s in range(-(N-1), N):
        lam_s = lam[s + mid]
        if lam_s == 0.0:
            continue
        if s >= 0:
            g[:N-s] -= lam_s * dx * h[s:]
        else:
            sp = -s
            g[sp:] -= lam_s * dx * h[:N-sp]
    return g

def random_feasible(N, rng):
    h = np.zeros(N, dtype=float)
    ones = rng.choice(N, N//2, replace=False)
    h[ones] = 1.0
    return h

def evaluate_score(h, dx):
    return np.max(np.correlate(h, 1 - h, mode='full') * dx)

def optimize_h(N, dx, target_sum, budget_s, start_time, rng):
    best_h = None
    best_score = np.inf
    alpha = 80.0
    init_lr = 0.6
    max_iters_per_restart = 300
    # Evaluate initial if exists

    while time.time() - start_time < budget_s * 0.95:

            grad = compute_gradient(h, w, dx)
            # decaying step

            # take gradient step and project

        # maybe after each restart also do a quick hill climbing? skip

import numpy as np
import time

def _proj_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project vector v onto the intersection of [0,1]^n and sum(v)=target."""
    lo, hi = -1e6, 1e6
    for _ in range(50):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _corr(h: np.ndarray, dx: float) -> np.ndarray:
    """Full linear correlation of h with (1-h), scaled by dx."""
    return np.correlate(h, 1.0 - h, mode="full") * dx


def _score(h: np.ndarray, dx: float) -> float:
    """Maximum overlap integral C5 for a given h."""
    return np.max(_corr(h, dx))


def _gradient(h: np.ndarray, w: np.ndarray, dx: float) -> np.ndarray:
    """
    Gradient of the smoothed max objective with respect to h.
    w is the normalized weight vector of length 2*n-1 (sum(w)=1).
    """
    n = h.shape[0]
    # combine symmetric weights: lambda_s = w_s + w_{-s}
    lam = w + w[::-1]                     # length 2*n-1
    mid = n - 1
    g = np.full(n, dx)                    # base term from sum_i w_i * 1
    for s in range(-(n - 1), n):
        lam_s = lam[s + mid]
        if lam_s == 0.0:
            continue
        if s >= 0:
            # h_{k+s} contributes for k = 0 .. n-1-s
            g[: n - s] -= lam_s * dx * h[s:]
        else:
            sp = -s
            # h_{k+s} with s negative contributes for k = sp .. n-1
            g[sp:] -= lam_s * dx * h[: n - sp]
    return g


def _random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random binary vector with exactly n/2 ones (n assumed even)."""
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h


def _optimize(n: int, dx: float, target_sum: float, budget_s: float, start: float,
              rng: np.random.Generator) -> tuple:
    """Main optimization loop returning best (h,score)."""
    best_h = None
    best_score = np.inf

    # include parent solution if available
    try:
        parent = np.asarray(initial_h_values, dtype=float)
        if parent.shape[0] == n:
            parent = np.clip(parent, 0.0, 1.0)
            parent = _proj_box_sum(parent, target_sum)
            sc = _score(parent, dx)
            if sc < best_score:
                best_score, best_h = sc, parent.copy()
    except Exception:
        pass

    alpha = 80.0          # smoothing parameter for max
    init_lr = 0.6         # initial learning rate
    max_iters = 300       # per restart
    while time.time() - start < budget_s * 0.95:
        # start from a fresh random feasible point
        h = _random_feasible(n, rng)
        for it in range(max_iters):
            now = time.time()
            if now - start >= budget_s * 0.95:
                break
            corr = _corr(h, dx)
            cur = corr.max()
            if cur < best_score:
                best_score, best_h = cur, h.copy()
            # smoothed weights for the max
            maxc = corr.max()
            exp_terms = np.exp(alpha * (corr - maxc))
            w = exp_terms / exp_terms.sum()
            # gradient step
            grad = _gradient(h, w, dx)
            lr = init_lr * (1.0 - it / max_iters)    # linear decay
            h = _proj_box_sum(h - lr * grad, target_sum)
        # optional quick hill‑climb on the best found so far
        if best_h is not None:
            # try a few random swaps aiming to improve best_h directly
            hb = best_h.copy()
            for _ in range(50):
                # pick two indices of differing side of 0.5
                idx1 = rng.integers(n)
                idx2 = rng.integers(n)
                if idx1 == idx2:
                    continue
                hb_new = hb.copy()
                hb_new[idx1], hb_new[idx2] = hb_new[idx2], hb_new[idx1]
                hb_new = _proj_box_sum(hb_new, target_sum)
                snew = _score(hb_new, dx)
                if snew < best_score:
                    best_score, best_h = snew, hb_new.copy()
                    hb = hb_new
    # final safeguard: ensure feasibility of the returned vector
    if best_h is None:
        best_h = _random_feasible(n, rng)
    best_h = _proj_box_sum(best_h, target_sum)
    best_score = _score(best_h, dx)
    return best_h, best_score


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a step function h, its C5 bound, and number of samples."""
