def adjust_sum(h, target):
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Index list where can adjust
    if diff > 0:
        # Need to increase sum
        while diff > 1e-12:
            # find indices with h[i] < 1 - 1e-12
            candidates = np.where(h < 1 - 1e-12)[0]
            if len(candidates) == 0:
                break
            i = np.random.choice(candidates)
            inc = min(diff, 1 - h[i])
            h[i] += inc
            diff -= inc
    else:
        # diff < 0 decrease sum
        while diff < -1e-12:
            candidates = np.where(h > 1e-12)[0]
            if len(candidates) == 0:
                break
            i = np.random.choice(candidates)
            dec = min(-diff, h[i])

            diff += dec
    return h

import time
import numpy as np

# ---- Helper utilities --------------------------------------------------------

def _target_sum(n: int) -> float:
    """Exact target sum of h-values."""
    return n / 2.0


def _ensure_feasible(vec: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and adjust to exact target sum."""
    vec = np.clip(vec, 0.0, 1.0)
    diff = target - vec.sum()
    # Simple redistribution of the remainder
    if abs(diff) > 1e-12:
        # Increase where slack exists, or decrease where value > 0
        if diff > 0:
            candidates = np.where(vec < 1.0 - 1e-12)[0]
        else:
            candidates = np.where(vec > 1e-12)[0]
        # If nothing feasible, just return (should not happen)
        if candidates.size == 0:
            return vec
        # Distribute diff greedily
        while abs(diff) > 1e-12 and candidates.size > 0:
            i = candidates[0]
            if diff > 0:
                inc = min(diff, 1.0 - vec[i])
                vec[i] += inc
                diff -= inc
                if vec[i] >= 1.0 - 1e-12:
                    candidates = candidates[1:]
            else:
                dec = min(-diff, vec[i])
                vec[i] -= dec
                diff += dec
                if vec[i] <= 1e-12:
                    candidates = candidates[1:]
    return vec


def _c5_score(vec: np.ndarray, dx: float) -> float:
    """Compute the C5 value for a given vector."""
    conv = np.correlate(vec, 1.0 - vec, mode="full")
    return dx * np.max(conv)


def _random_feasible(n: int, target: float, rng: np.random.Generator) -> np.ndarray:
    """Generate a simple feasible vector: many 0/1 with at most one fractional entry."""
    low = int(np.floor(target))
    frac = target - low
    v = np.zeros(n, dtype=np.float64)
    if low > 0:
        v[:low] = 1.0
    if frac > 1e-12 and low < n:
        v[low] = frac
    rng.shuffle(v)
    return v


def _max_lag(vec: np.ndarray) -> int:
    """Return the lag (negative to positive) that attains the maximal correlation."""
    conv = np.correlate(vec, 1.0 - vec, mode="full")
    idx = int(np.argmax(conv))
    n = vec.shape[0]
    return idx - (n - 1)


def _gradient_step(vec: np.ndarray, lag: int, dx: float) -> np.ndarray:
    """One projected gradient step for the lag that currently dominates."""
    n = vec.shape[0]
    grad = np.ones_like(vec)
    if lag >= 0:
        grad[: n - lag] -= vec[lag:]
        grad[lag:] -= vec[: n - lag]
    else:
        a = -lag
        grad[a:] -= vec[: n - a]
        grad[: n - a] -= vec[a:]
    # Gradient of the objective (dx multiplied)
    grad *= dx
    # Simple step size
    step = 0.1
    new = vec - step * grad
    new = np.clip(new, 0.0, 1.0)
    return new


def _refine(vec: np.ndarray, dx: float, target: float, rng: np.random.Generator,
            time_limit: float) -> tuple:
    """Iterative gradient refinement until time expires."""
    start = time.time()
    best = vec.copy()
    best_val = _c5_score(best, dx)
    cur = vec.copy()
    while time.time() - start < time_limit:
        lag = _max_lag(cur)
        nxt = _gradient_step(cur, lag, dx)
        # enforce exact sum after the gradient move
        nxt = _ensure_feasible(nxt, target)
        cur_val = _c5_score(nxt, dx)
        if cur_val < best_val:
            best, best_val = nxt, cur_val
        # simple deterministic descent: keep the better of cur and nxt
        cur = nxt if cur_val > best_val else best
    return best, best_val

        # -----------------------------------------------------------------
        if rng.random() < 0.5:
            i, j = rng.choice(n, size=2, replace=False)
            neigh = cur_h.copy()
            neigh[i], neigh[j] = neigh[j], neigh[i]
        else:
            i, j = rng.choice(n, size=2, replace=False)
            max_transfer = min(cur_h[i], 1.0 - cur_h[j])
            if max_transfer <= eps_min:
                continue
            eps = rng.random() * max_transfer
            neigh = cur_h.copy()
            neigh[i] -= eps
            neigh[j] += eps

        neigh_score = _c5_score(neigh, dx)

        # -----------------------------------------------------------------
        # Acceptance rule (simulated annealing)
        # -----------------------------------------------------------------
        if neigh_score < cur_score:
            accept = True
        else:
            elapsed = time.time() - start_time
            temp_frac = max(0.0, 1.0 - elapsed / max_time)  # 1 → 0
            # scale temperature by current best_score to get a reasonable magnitude
            prob = np.exp(-(neigh_score - cur_score) /
                          (temp_frac * best_score + eps_min))
            accept = rng.random() < prob

        if accept:
            cur_h, cur_score = neigh, neigh_score
            no_improve = 0
            if neigh_score < best_score - 1e-12:
                best_h, best_score = neigh.copy(), neigh_score
        else:
            no_improve += 1

        # -----------------------------------------------------------------
        # Stagnation handling – restart from a fresh random feasible point
        # -----------------------------------------------------------------
        if no_improve >= restart_threshold:
            cur_h = _random_feasible(n, target, rng)
            cur_score = _c5_score(cur_h, dx)
            no_improve = 0

    # -------------------------------------------------------------------------
    # 3. Local gradient refinement (uses remaining time budget)
    # -------------------------------------------------------------------------
    remaining = max_time - (time.time() - start_time)
    if remaining > 0.2:
        # allocate at most 0.5 * remaining to refinement; leave a small buffer
        ref_time = min(0.5 * remaining, 5.0)
        best_h, best_score = _refine(best_h, dx, target, rng, ref_time)

    # -------------------------------------------------------------------------
    # 4. Final projection onto exact constraints
    # -------------------------------------------------------------------------
    best_h = _ensure_feasible(best_h, target)
    final_score = _c5_score(best_h, dx)

    return best_h, final_score, n
