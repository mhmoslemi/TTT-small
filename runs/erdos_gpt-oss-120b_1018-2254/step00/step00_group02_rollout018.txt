def _compute_c5(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)*(1-h(x+k))dx for a discretized h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation without padding (np.correlate with mode='full')
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)
