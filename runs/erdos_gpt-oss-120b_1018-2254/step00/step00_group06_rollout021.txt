def project_to_simplex_bounds(y, target):
    lo = -np.max(y)  # ensure y+λ <=0 => clamp to 0

        s = proj.sum()
        if s < target:

        else:

import numpy as np
import time

def _project_to_bounds_and_sum(x, target):
    """Project x onto the set {0<=y_i<=1, sum(y)=target}."""
    lo = -np.max(x)
    hi = 1.0 - np.min(x)
    for _ in range(30):
        mid = (lo + hi) * 0.5
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)

def _c5_score(h):
    """Return the C5 value for a candidate h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _tournament_select(pop, fitness, k=3):
    """Return a copy of a parent selected by tournament."""
    idx = np.random.choice(pop.shape[0], size=k, replace=False)
    best = idx[np.argmin(fitness[idx])]
    return pop[best].copy()

def _uniform_crossover(p1, p2, rate=0.8):
    """Single‑point style uniform crossover."""
    if np.random.rand() < rate:
        mask = np.random.rand(p1.shape[0]) < 0.5
        child = np.where(mask, p1, p2)
        return child
    return p1.copy()

def _mutate(x, prob=0.02, sigma=0.1):
    """Gaussian mutation applied per gene with given probability."""
    mask = np.random.rand(x.shape[0]) < prob
    if np.any(mask):
        x[mask] += np.random.normal(0.0, sigma, size=mask.sum())
    return x

def _initial_population(popsize, n, target, base=None):
    pop = np.empty((popsize, n), dtype=float)
    start = 0
    if base is not None:
        pop[0] = base.copy()
        start = 1
    for i in range(start, popsize):
        rnd = np.random.rand(n)
        pop[i] = _project_to_bounds_and_sum(rnd, target)
    return pop
