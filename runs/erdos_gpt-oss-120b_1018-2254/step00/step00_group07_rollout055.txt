def solve_optimal_binary(n, time_limit):
    h = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    constraints = [cp.sum(h) == n//2, t >= 0]
    shift_vars = {}
    shift_to_vars = {}
    for k in range(-(n-1), n):
        if k == 0:
            continue
        var_list = []
        for i in range(n):
            j = i + k
            if j < 0 or j >= n:
                continue
            y = cp.Variable(boolean=True)
            shift_vars[(i,k)] = y
            var_list.append(y)
            # linking constraints
            constraints.append(y <= h[i])
            constraints.append(y <= 1 - h[j])
            constraints.append(y >= h[i] + (1 - h[j]) - 1)
        if var_list:
            shift_to_vars[k] = var_list
            # sum constraint
            constraints.append(cp.sum(var_list) <= t)
    # Solve
    prob = cp.Problem(cp.Minimize(t), constraints)
    # set a time limit for solver
    try:
        prob.solve(solver=cp.CBC, timeLimit=time_limit)
    except Exception as e:
        # maybe solver fails; fall back to default solve
        prob.solve(solver=cp.CBC)
    # extract solution
    h_val = np.array([h[i].value for i in range(n)], dtype=float)
    t_val = t.value
    return h_val, float(t_val)

import time, math, numpy as np

def compute_c5(h, dx):
    # h is array of floats (0..1)
    # correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def random_binary(n, rng):
    # generate a binary vector with exactly n/2 ones
    ones = n // 2
    h = np.zeros(n, dtype=int)
    idx = rng.choice(n, ones, replace=False)
    h[idx] = 1
    return h

def thue_morse(n):
    # return binary Thue-Morse sequence of length n (0/1)
    seq = np.empty(n, dtype=int)
    for i in range(n):
        seq[i] = bin(i).count('1') % 2
    return seq

def solve_optimal_binary(n, time_limit):
    try:
        import cvxpy as cp
    except Exception:
        return None, None
    # binary decision variables
    h = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    constraints = [cp.sum(h) == n // 2, t >= 0]
    # for each shift k (excluding 0)
    for k in range(-n + 1, n):
        if k == 0:
            continue
        pair_vars = []
        for i in range(n):
            j = i + k
            if j < 0 or j >= n:
                continue
            y = cp.Variable(boolean=True)
            # y = h[i] * (1 - h[j])
            constraints.append(y <= h[i])
            constraints.append(y <= 1 - h[j])
            constraints.append(y >= h[i] - h[j])
            pair_vars.append(y)
        if pair_vars:
            constraints.append(cp.sum(pair_vars) <= t)
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CBC, max_seconds=time_limit)
    except Exception:
        # solver may not accept max_seconds, try without it
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            return None, None
    if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        return None, None
    # extract solution
    h_val = np.array([h[i].value for i in range(n)], dtype=float)
    t_val = t.value
    return h_val, t_val

            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.astype(float).copy()
        else:
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]   # revert
        it += 1

    # ------------------------------------------------------------------
    # 4) Final tidy‑up: enforce exact binary nature and integral
    if best_h is None:
        best_h = random_binary(n, rng).astype(float)
    best_h = np.rint(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)

    cur_sum = best_h.sum()
    target = n / 2
    if abs(cur_sum - target) > 1e-6:
        diff = int(round(target - cur_sum))
        if diff > 0:
            zeros = np.where(best_h < 0.5)[0]
            if len(zeros) >= diff:
                idx = rng.choice(zeros, diff, replace=False)
                best_h[idx] = 1.0
        elif diff < 0:
            ones = np.where(best_h > 0.5)[0]
            if len(ones) >= -diff:
                idx = rng.choice(ones, -diff, replace=False)
                best_h[idx] = 0.0

    final_c5 = compute_c5(best_h, dx)
    return (best_h, float(final_c5), n)
