    n = len(h)
    i, j = np.random.choice(n, size=2, replace=False)
    direction = np.random.choice([1, -1])
    if direction == 1:

def hill_climb(n, max_iters, start_h=None):
    if start_h is None:
        # random binary half ones
        idx = np.random.permutation(n)
        h = np.zeros(n)
        h[idx[:n//2]] = 1.0
    else:
        h = start_h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    # Precompute indices lists
    for it in range(max_iters):
        # pick random 1 and 0 indices
        ones = np.where(h==1)[0]
        zeros = np.where(h==0)[0]
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # swap
        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:  # improvement
            h = h_new
            best_h = h_new
            best_c5 = c5_new
        # else keep original h unchanged
    return best_h, best_c5
```

def make_feasible_continuous(h):

    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    diff = target - total
    # if diff is small, try to adjust a random index
    if abs(diff) > 1e-12:
        # find indices where we can add/subtract
        if diff > 0:
            idx = np.where(h < 1 - 1e-12)[0]
            if idx.size > 0:
                i = np.random.choice(idx)
                add = min(diff, 1 - h[i])
                h[i] += add
                diff -= add
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size > 0:
                i = np.random.choice(idx)
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub

def propose_move_continuous(h, step):
    n = len(h)
    # choose two distinct indices
    i, j = np.random.choice(n, size=2, replace=False)
    # randomly decide inc/dec
    if np.random.rand() < 0.5:
        inc, dec = i, j
    else:
        inc, dec = j, i
    max_inc = min(step, 1.0 - h[inc])
    max_dec = min(step, h[dec])
    max_delta = min(max_inc, max_dec)
    if max_delta <= 0:
        return None
    delta = np.random.uniform(0, max_delta)
    h_new = h.copy()
    h_new[inc] += delta
    h_new[dec] -= delta
    return h_new

        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        k_idx = np.argmax(corr)
        k = k_idx - (n - 1)
        if k >= 0:
            i_range = np.arange(0, n - k)
            i_shift = i_range + k
        else:
            i_range = np.arange(-k, n)
            i_shift = i_range + k  # k is negative
        h_i = cur_h[i_range]
        h_s = cur_h[i_shift]
        mism_i = i_range[(h_i == 1) & (h_s == 0)]
        opp_i = i_range[(h_i == 0) & (h_s == 1)]
        moved = False
        if mism_i.size and opp_i.size:
            i = np.random.choice(mism_i)
            j = np.random.choice(opp_i)
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
            new_val = compute_c5(cur_h)
            if new_val < cur_val - 1e-12:
                cur_val = new_val
                moved = True
                if new_val < best_val:
                    best_val = new_val
                    best_h = cur_h.copy()
            else:
                cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        if not moved:
            ones_idx = np.where(cur_h == 1)[0]
            zeros_idx = np.where(cur_h == 0)[0]
            if ones_idx.size == 0 or zeros_idx.size == 0:
                break
            i = np.random.choice(ones_idx)
            j = np.random.choice(zeros_idx)
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
            new_val = compute_c5(cur_h)
            if new_val < cur_val - 1e-12:
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best_h = cur_h.copy()
            else:
                cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()

    # Base solution (continuous) from provided initial_h_values if any
    try:
        init = np.array(initial_h_values, dtype=float)
        n0 = init.shape[0]
        base_h = make_feasible_continuous(init)
    except Exception:
        n0 = 70
        base_h = np.full(n0, 0.5, dtype=float)
        base_h = make_feasible_continuous(base_h)

    best_h = base_h.copy()
    best_c5 = compute_c5(best_h)
    best_n = n0

    # Continuous local search (about half of time budget)
    time_remaining = budget_s - (time.time() - start_time)
    cont_budget = min(budget_s * 0.5, time_remaining * 0.9)
    h_opt, c5_opt = hill_climb_continuous(best_h, cont_budget)
    if c5_opt < best_c5:
        best_h, best_c5, best_n = h_opt, c5_opt, n0

    # Binary search on several grid sizes
    elapsed = time.time() - start_time
    time_left = max(0.0, budget_s - elapsed)
    n_candidates = [60, 70, 80, 90, 100]
    if n0 not in n_candidates:
        n_candidates = [n0] + n_candidates

    target_c5 = 0.3808
    for n in n_candidates:
        if time.time() - start_time >= budget_s * 0.95:
            break
        target_ones = n // 2
        per_n_start = time.time()
        # allow roughly 30% of remaining time for this n
        per_n_budget = min(time_left * 0.3, budget_s - (time.time() - start_time))
        # Perform a few random restarts
        while time.time() - start_time < budget_s * 0.95:
            # random binary vector
            h_bin = np.zeros(n, dtype=float)
            idx = np.random.permutation(n)[:target_ones]
            h_bin[idx] = 1.0
            h_bin_opt, c5_bin = hill_climb_binary_targeted(h_bin, max_iters=2000)
            if c5_bin < best_c5:
                best_h, best_c5, best_n = h_bin_opt, c5_bin, n
                if best_c5 <= target_c5:
                    return (best_h, float(best_c5), int(best_n))
            # stop this n if its allocated budget exhausted
            if time.time() - per_n_start > per_n_budget:
                break
        # update remaining time estimate
        elapsed = time.time() - start_time
        time_left = max(0.0, budget_s - elapsed)

    # Ensure final solution satisfies constraints (clip and sum adjust)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = make_feasible_continuous(best_h)
    best_c5 = compute_c5(best_h)
    return (best_h, float(best_c5), int(best_n))
