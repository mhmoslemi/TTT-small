        k = int(target_sum)
        idx = np.argpartition(v, -k)[-k:]   # indices of the k largest entries
        h = np.zeros(n, dtype=float)
        h[idx] = 1.0
        return h

    # ----- initial candidate --------------------------------------------
    if have_init:
        start_h = binary_from(init)
    else:
        start_h = binary_from(rng.random(n))

    best_h = start_h.copy()
    best_c5 = compute_c5(best_h)

    if have_init:
        init_c5 = compute_c5(init)
        if init_c5 < best_c5:
            best_h = init.copy()
            best_c5 = init_c5

    # ----- binary hill‑climbing ----------------------------------------
    current_h = best_h.copy()
    current_c5 = best_c5

    start_time = time.time()
    deadline = start_time + budget_s * 0.95

    max_no_improve = 2000 * n
    no_improve = 0

    while time.time() < deadline:
        ones = np.where(current_h == 1.0)[0]
        zeros = np.where(current_h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        new_h = current_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_c5 = compute_c5(new_h)

        if new_c5 <= current_c5:
            current_h = new_h
            current_c5 = new_c5
            no_improve = 0
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
        else:
            no_improve += 1

        if no_improve >= max_no_improve:
            current_h = binary_from(rng.random(n))
            current_c5 = compute_c5(current_h)
            no_improve = 0

    # ----- continuous refinement (projected gradient descent) ----------
    remaining = deadline - time.time()
    if remaining > 5.0:
        h = best_h.copy()
        gamma = 0.02          # smoothing for the soft‑min
        lr = 0.5              # learning rate
        shifts = np.arange(-(n - 1), n)   # full correlation lags

        for it in range(2000):
            if time.time() >= deadline:
                break

            # auto‑correlation g_k = dx * Σ h_i h_{i‑k}
            corr = np.correlate(h, h, mode='full')
            g = corr * dx

            # soft‑min weights
            exps = np.exp(-g / gamma)
            w = exps / np.sum(exps)

            # gradient of the smooth objective
            grad = np.zeros_like(h)
            for idx, k in enumerate(shifts):
                wk = w[idx] * dx
                if k >= 0:
                    if k < n:
                        grad[:n - k] -= wk * h[k:]      # term h_{i+k}
                        grad[k:]     -= wk * h[:n - k]  # term h_{i‑k}
                else:
                    kk = -k
                    grad[kk:]   -= wk * h[:n - kk]
                    grad[:n-kk] -= wk * h[kk:]

            # gradient step
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0.0, 1.0)

            # projection onto the capped simplex (sum == target_sum)
            s = np.sum(h_new)
            diff = target_sum - s
            if abs(diff) > 1e-8:
                free = (h_new > 1e-12) & (h_new < 1.0 - 1e-12)
                if np.any(free):
                    delta = diff / np.count_nonzero(free)
                    h_new[free] += delta
                    h_new = np.clip(h_new, 0.0, 1.0)
                s = np.sum(h_new)
                if abs(s - target_sum) > 1e-6:
                    h_new *= target_sum / s
                    h_new = np.clip(h_new, 0.0, 1.0)

            h = h_new
            cur_c5 = compute_c5(h)
            if cur_c5 < best_c5:
                best_h = h.copy()
                best_c5 = cur_c5

    # ----- final clean‑up ------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    cur_sum = np.sum(best_h)
    if abs(cur_sum - target_sum) > 1e-6:
        best_h *= target_sum / cur_sum
        best_h = np.clip(best_h, 0.0, 1.0)

    return (best_h, float(best_c5), n)
