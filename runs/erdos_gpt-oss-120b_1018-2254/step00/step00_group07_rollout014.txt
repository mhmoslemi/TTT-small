def compute_gradient(h, lag):
    n = len(h)
    grad = np.ones(n)
    if lag >= 0:
        # term h_{k+lag}
        grad[ : n - lag] -= h[lag:]  # h_{k+lag} for k up to n-lag-1
        # term h_{k-lag}
        grad[lag:] -= h[: n - lag]
    else:
        L = -lag
        # for lag negative: h_{k+lag} = h_{k - L}
        grad[L:] -= h[:n - L]
        # h_{k - lag} = h_{k + L}
        grad[: n - L] -= h[L:]
    return grad

        else:

import numpy as np
import time
import random

def compute_c5(h):
    """Maximum overlap integral for a given step function."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_onto_simplex_box(v, target, lower=0.0, upper=1.0, max_iter=60):
    """
    Euclidean projection of vector v onto the set
    {h | lower <= h_i <= upper, sum(h) = target}.
    Uses binary search on the Lagrange multiplier.
    """
    lo = v.min() - upper    # ensures all entries are clipped to upper -> sum = n
    hi = v.max() - lower    # ensures all entries are clipped to lower -> sum = 0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        h = np.clip(v - mid, lower, upper)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, lower, upper)

def compute_gradient(h, lag):
    """Gradient of the worst overlap (max lag) w.r.t. the vector h."""
    n = h.size
    grad = np.ones(n)
    if lag >= 0:
        L = lag
        if L < n:
            grad[:n - L] -= h[L:]          # -h_{k+L}
            grad[L:]      -= h[:n - L]     # -h_{k-L}
    else:
        L = -lag
        if L < n:
            grad[L:]      -= h[:n - L]     # -h_{k+lag}=h_{k-L}
            grad[:n - L]  -= h[L:]         # -h_{k-lag}=h_{k+L}
    return grad

    Returns (h_values, c5_bound, n_points).
    """
    # reproducibility
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------

    target_sum = n / 2.0            # because ∫h = 1 ⇔ sum(h)*dx = 1, dx=2/n
    h = project_onto_simplex_box(h, target_sum)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    iteration = 0

    # ------------------------------------------------------------------
    # main optimization loop
    # ------------------------------------------------------------------
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.97:   # keep a safety margin
            break

        # determine the lag attaining the current worst overlap
        corr = np.correlate(h, 1.0 - h, mode='full')
        overlaps = corr * (2.0 / n)
        max_idx = int(np.argmax(overlaps))
        lag = max_idx - (n - 1)

        # gradient of that worst term
        grad = compute_gradient(h, lag)

        # step size that shrinks linearly to a small floor
        t_frac = elapsed / budget_s
        alpha = 0.5 * (1.0 - t_frac) + 0.001

        # gradient step followed by projection onto the feasible set
        h_new = project_onto_simplex_box(h - alpha * grad, target_sum)
