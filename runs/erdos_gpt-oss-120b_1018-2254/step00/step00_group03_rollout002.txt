    lo = -1e6

    def compute_c5(h):
        """Return max_k ∫ h(x)(1-h(x+k)) dx for discretized h."""
        n = h.size
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full")
        return float(np.max(corr) * dx)

    def project_onto_sum_and_bounds(v, target):
        """
        Project vector v onto the feasible set:
        0 <= w_i <= 1, sum w_i = target.
        Uses a simple bisection on the offset λ such that
        w_i = clip(v_i + λ, 0, 1) yields the desired sum.
        """
        lo, hi = -2.0, 2.0  # sufficiently wide interval for λ
        for _ in range(60):
            mid = (lo + hi) * 0.5
            w = np.clip(v + mid, 0.0, 1.0)
            s = w.sum()
            if s > target:
                hi = mid
            else:
                lo = mid
        return np.clip(v + lo, 0.0, 1.0)

    def make_random_feasible(n):
        """Generate a random feasible h for length n."""
        # start from uniform random numbers
        v = np.random.rand(n)
        target = n / 2.0
        # scale to hit the target sum approximately
        v *= target / v.sum()
        # now project onto the box and sum constraints
        return project_onto_sum_and_bounds(v, target)
