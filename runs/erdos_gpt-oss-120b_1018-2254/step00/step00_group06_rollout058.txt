def tournament_select(population, scores, k=3):
    idx = rng.integers(len(population))
    best = idx
    best_score = scores[idx]
    for _ in range(k-1):
        j = rng.integers(len(population))
        if scores[j] < best_score:
            best = j
            best_score = scores[j]
    return population[best]

def _project_to_sum_and_box(v, target):
    """Project v onto the feasible set {x | 0<=x_i<=1, sum x = target}."""
    lo = -np.max(v) - 1.0          # ensures all v+lo <= 0 -> sum 0
    hi = 1.0 - np.min(v) + 1.0     # ensures all v+hi >= 1 -> sum n
    for _ in range(60):
