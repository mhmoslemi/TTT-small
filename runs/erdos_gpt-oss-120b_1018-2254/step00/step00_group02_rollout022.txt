def random_transfer(h, step_factor, rng, target):
    n = len(h)
    i, j = rng.choice(n, size=2, replace=False)
    # direction: move from i to j
    max_move = min(h[i], 1.0 - h[j])
    if max_move <= 0:

        max_move = min(h[j], 1.0 - h[i])
        if max_move <= 0:
            return None

    # delta magnitude:

    new_h = h.copy()

    new_h[j] += delta
    # maybe rounding errors: we clip just in case

        for _ in range(60):

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time

    rng = np.random.default_rng(seed)

    # ----- helper: projection onto [0,1] and sum constraint -----
    def project_to_sum_and_bounds(v, target):
        # binary search on lambda to enforce sum constraint after clipping
        lo = -np.max(v)          # ensures v+lo <= 0
        hi = 1.0 - np.min(v)     # ensures v+hi >= 1
        for _ in range(60):
            mid = (lo + hi) / 2.0
            proj = np.clip(v + mid, 0.0, 1.0)
            s = proj.sum()
            if s < target:
                lo = mid
            else:
                hi = mid
        return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

    # ----- helper: compute C5 for a vector -----
    def compute_c5(vec):
        nloc = len(vec)
        dx = 2.0 / nloc
        corr = np.correlate(vec, 1.0 - vec, mode='full')
        return corr.max() * dx

    # ----- helper: random transfer move preserving sum -----
    def random_transfer(cur, step_factor, target):
        nloc = len(cur)
        i, j = rng.choice(nloc, size=2, replace=False)
        # move from i to j
        max_move = min(cur[i], 1.0 - cur[j])
        if max_move <= 0:
            # try opposite direction
            max_move = min(cur[j], 1.0 - cur[i])
            if max_move <= 0:
                return None
            i, j = j, i
        delta = rng.random() * max_move * step_factor
        if delta <= 0.0:
            return None
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = np.clip(new, 0.0, 1.0)
        new = project_to_sum_and_bounds(new, target)
        return new

    # ----- initialise h -----
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.shape[0]
        # upscale to a finer grid for more flexibility
        n = max(180, n0 * 4)  # aim for ~180 points or more
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x_new, x0, h0)
    else:
        n = 180
        h = np.full(n, 0.5, dtype=float)

    target_sum = n / 2.0
    # ensure feasibility
    h = project_to_sum_and_bounds(h, target_sum)

    # ----- main optimization loop -----
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    cur_h = h.copy()
    cur_c5 = best_c5

    start = time.time()
    it = 0
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.97:
            break

        # current worst shift

        idx_max = int(corr_full.argmax())

            if best_c5 <= 0.3808:

                pass

        # schedule step size (large early, shrink later)
        frac = 1.0 - elapsed / budget_s
        step = 0.5 * frac

        # --- gradient-like improvement on the worst lag ---
        if lag != 0:
            neighbor = np.zeros_like(cur_h)
            if lag > 0:
                neighbor[:n - lag] += cur_h[lag:]
                neighbor[lag:] += cur_h[:n - lag]
            else:  # lag < 0
                lag_abs = -lag
                neighbor[:n - lag_abs] += cur_h[lag_abs:]
                neighbor[lag_abs:] += cur_h[:n - lag_abs]
        else:
            neighbor = 2.0 * cur_h  # self-pair case

        # move opposite to the sub‑gradient (increase autocorrelation)
        cand_h = cur_h + step * (2.0 / n) * neighbor
        cand_h = np.clip(cand_h, 0.0, 1.0)

    best_h = project_to_sum_and_bounds(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return (best_h, float(best_c5), n)
