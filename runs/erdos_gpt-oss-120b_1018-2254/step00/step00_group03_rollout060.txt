import time, numpy as np, random, math

def evaluate(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_onto_constraints(v, target_sum):
    v = np.clip(v, 0.0, 1.0)
    current = v.sum()
    # water-filling loop
    for _ in range(100):
        if abs(current - target_sum) <= 1e-12:
            break
        diff = target_sum - current
        free = np.where((v > 0) & (v < 1))[0]
        if len(free) == 0:
            # cannot adjust further
            break
        delta = diff / len(free)
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
        current = v.sum()
    # final fine tuning
    err = target_sum - v.sum()
    if abs(err) > 1e-9:
        # adjust a random index within bounds
        idx = np.random.randint(len(v))
        v[idx] = min(1.0, max(0.0, v[idx] + err))
    # ensure final = target_sum by scaling small residual if needed
    # small adjustment to spread error
    return v

import time
import random
import math
import numpy as np

# ---------- helper functions (top level) ----------

def evaluate(h, dx):
    """Return C5 value for a step function given by vector h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_onto_constraints(v, target_sum):
    """Clip v to [0,1] and adjust it so that sum(v) == target_sum."""
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    # water-filling style adjustment
    for _ in range(100):
        if abs(cur - target_sum) <= 1e-12:
            break
        diff = target_sum - cur
        free = np.where((v > 0.0) & (v < 1.0))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
        cur = v.sum()
    # tiny residual fix
    err = target_sum - v.sum()
    if abs(err) > 1e-9:
        idx = np.random.randint(v.size)
        v[idx] = min(1.0, max(0.0, v[idx] + err))
    return v


def thue_morse_bits(n):
    """Generate the first n bits of the Thue–Morse sequence as floats."""
    bits = np.fromiter((bin(i).count('1') % 2 for i in range(n)), dtype=float, count=n)
    return bits


def random_binary(n):
    """Uniform random binary vector with exactly n/2 ones (floor if n odd)."""
    h = np.zeros(n, dtype=float)
    k = n // 2
    ones = np.random.choice(n, k, replace=False)
    h[ones] = 1.0
    return h


def propose_move(cur_h, n, s_max, step_max):
    """Generate a feasible neighboring vector."""
    eps = 1e-12
    # try a targeted move that directly reduces the current maximal shift
    if np.random.rand() < 0.3:
        i_start = max(0, -s_max)
        i_end = min(n, n - s_max)
        if i_end > i_start:
            hi = cur_h[i_start:i_end]
            hj = cur_h[i_start + s_max:i_end + s_max]
            i_cand = np.where((hi > eps) & (hj < 1 - eps))[0]
            j_cand = np.where((hi < 1 - eps) & (hj > eps))[0]
            if i_cand.size and j_cand.size:
                i_loc = np.random.choice(i_cand)
                j_loc = np.random.choice(j_cand)
                i = i_start + i_loc
                j = i_start + j_loc
                max_delta = min(cur_h[i], 1.0 - cur_h[j])
                if max_delta > eps:
                    delta = np.random.uniform(0.0, min(max_delta, step_max))
                    new_h = cur_h.copy()
                    new_h[i] -= delta
                    new_h[j] += delta
                    return new_h
    # fallback: generic redistribution move
    i_cand = np.where(cur_h > eps)[0]
    j_cand = np.where(cur_h < 1 - eps)[0]
    if i_cand.size == 0 or j_cand.size == 0:
        return None
    i = np.random.choice(i_cand)
    j = np.random.choice(j_cand)
    max_delta = min(cur_h[i], 1.0 - cur_h[j])
    if max_delta <= eps:
        return None
    delta = np.random.uniform(0.0, min(max_delta, step_max))
    new_h = cur_h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h


def local_search(start_h, n, target_sum, dx, time_budget):
    """Simulated‑annealing style refinement of a feasible vector."""
    cur_h = start_h.copy()
    cur_score = evaluate(cur_h, dx)
    best_h = cur_h.copy()
    best_score = cur_score

    start = time.time()
    T_start, T_end = 0.02, 0.0001
    step_max_initial = 0.5

    while time.time() - start < time_budget:
        # compute current maximal shift
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        max_idx = int(np.argmax(corr))
        s_max = max_idx - (n - 1)

        # temperature and step size decay linearly
        elapsed = time.time() - start
        remaining = max(time_budget - elapsed, 0.0)
        T = T_end + (T_start - T_end) * (remaining / time_budget) if time_budget > 0 else T_end
        step_max = step_max_initial * (remaining / time_budget) if time_budget > 0 else step_max_initial

        # propose a move
        new_h = propose_move(cur_h, n, s_max, step_max)
        if new_h is None:
            continue

        new_score = evaluate(new_h, dx)
        dscore = new_score - cur_score
        if dscore < 0.0 or np.random.rand() < math.exp(-dscore / T):
            cur_h = new_h
            cur_score = new_score
            if cur_score < best_score:
                best_h = cur_h.copy()
                best_score = cur_score

    return best_h, best_score


# ---------- main entry point ----------

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) within the given time budget."""
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    total_budget = budget_s

    # dimension (use parent's size if available)
    init = globals().get('initial_h_values')
    n = len(init) if init is not None else 64

    target_sum = n / 2.0
    dx = 2.0 / n

    # ----- build initial candidates -----
    candidates = []

    if init is not None:
        h0 = np.array(init, dtype=float)
        h0 = project_onto_constraints(h0, target_sum)
        candidates.append(h0)

    h_tm = thue_morse_bits(n)
    h_tm = project_onto_constraints(h_tm, target_sum)
    candidates.append(h_tm)

    for _ in range(3):
        candidates.append(random_binary(n))

    best_h = None
    best_score = float('inf')

    # helper for remaining time
    def remaining_time():
        return total_budget - (time.time() - start_time)

    # ----- local search on the prepared candidates -----
    per_cand = min(8.0, remaining_time() / max(1, len(candidates)))
    for cand in candidates:
        if remaining_time() < 2.0:
            break
        allot = min(per_cand, remaining_time() * 0.5)
        h_loc, score_loc = local_search(cand, n, target_sum, dx, allot)
        if score_loc < best_score:
            best_score = score_loc
            best_h = h_loc

    # ----- random restarts until budget exhausted -----
    while remaining_time() > 5.0:
        rnd = np.random.rand(n)
        rnd = project_onto_constraints(rnd, target_sum)
        allot = min(15.0, remaining_time() * 0.5)
        h_loc, score_loc = local_search(rnd, n, target_sum, dx, allot)
        if score_loc < best_score:
            best_score = score_loc
            best_h = h_loc

    # fallback if nothing was found
    if best_h is None:
        best_h = random_binary(n)
        best_h = project_onto_constraints(best_h, target_sum)
        best_score = evaluate(best_h, dx)

    # final safety projection and evaluation
    best_h = project_onto_constraints(best_h, target_sum)
    best_score = evaluate(best_h, dx)

    return (best_h, float(best_score), n)
