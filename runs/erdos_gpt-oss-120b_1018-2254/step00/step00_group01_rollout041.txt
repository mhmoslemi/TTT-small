```
def project_to_feasible(v, target_sum):
    lo = -1.0
    hi = 2.0
    for _ in range(50):
        lam = (lo + hi) / 2
        h = np.clip(v - lam, 0, 1)
        s = h.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    # final h
    h = np.clip(v - (lo+hi)/2, 0, 1)
    # adjust small numeric error by distributing difference
    diff = target_sum - h.sum()
    if abs(diff) > 1e-8:
        # add diff uniformly to entries not at bounds
        free = (h > 0) & (h < 1)
        if free.any():
            h[free] += diff / free.sum()
            # clip again to be safe
            h = np.clip(h, 0, 1)
        else:
            # if no free entries, adjust randomly within bounds
            idx = np.random.choice(len(h))
            h[idx] = np.clip(h[idx] + diff, 0, 1)
    return h
```

import numpy as np
import time

def _project_to_feasible(v, target):
    lo, hi = -1.0, 2.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        h = np.clip(v - lam, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    h = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-8:
        mask = (h > 0.0) & (h < 1.0)
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
        else:
            i = np.random.randint(len(h))
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def _c5_of_h(h, dx):
    return np.max(np.correlate(h, 1 - h, mode='full') * dx)

    # linear ramp – satisfies sum and bounds exactly
