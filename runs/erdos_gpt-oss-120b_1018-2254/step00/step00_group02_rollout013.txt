import numpy as np, time

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    S = np.correlate(h, h, mode='full')
    t_min = np.min(S)
    return 1.0 - dx * t_min

def random_feasible_vector(n, rng):
    target = n / 2.0
    while True:
        xs = rng.random(n-1)
        s = xs.sum()
        last = target - s
        if 0 <= last <= 1:
            return np.concatenate([xs, [last]])

def random_binary_feasible_vector(n, rng):
    target = n / 2.0
    int_part = int(np.floor(target))
    frac = target - int_part
    h = np.zeros(n)
    if int_part > 0:
        idx = rng.choice(n, size=int_part, replace=False)
        h[idx] = 1.0
        # get remaining indices for fraction
        if frac > 0:
            remaining = np.setdiff1d(np.arange(n), idx, assume_unique=True)
            if remaining.size > 0:
                k = rng.choice(remaining)
                h[k] = frac
    else:
        # only fractional part
        k = rng.integers(n)
        h[k] = frac
    return h

import time
import numpy as np


def compute_c5(h: np.ndarray) -> float:
    """Return the C5 value for a step function represented by h."""
    n = h.shape[0]
    dx = 2.0 / n
    # autocorrelation of h with itself
    s = np.correlate(h, h, mode="full")
    t_min = np.min(s)
    # C = 1 - dx * min_k sum_i h_i h_{i+k}
    return 1.0 - dx * t_min


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector in [0,1]^n with sum exactly n/2."""
    target = n / 2.0
    while True:
        xs = rng.random(n - 1)
        s = xs.sum()
        last = target - s
        if 0.0 <= last <= 1.0:
            return np.concatenate([xs, [last]])


def random_binary_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a binary‑ish feasible vector (0/1 entries, one fractional if needed)."""
    target = n / 2.0
    int_part = int(np.floor(target))
    frac = target - int_part
    h = np.zeros(n)
    if int_part > 0:
        idx = rng.choice(n, size=int_part, replace=False)
        h[idx] = 1.0
        if frac > 1e-12:
            remaining = np.setdiff1d(np.arange(n), idx, assume_unique=True)
            if remaining.size:
                k = rng.choice(remaining)
                h[k] = frac
    else:
        k = rng.integers(n)
        h[k] = frac
    return h


def propose_move(h: np.ndarray, rng: np.random.Generator, max_step: float):
    """Return a new vector after moving a random amount between two indices."""
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    # delta such that h[i]-delta and h[j]+delta stay in [0,1]
    lower = max(h[i] - 1.0, -h[j])
    upper = min(h[i], 1.0 - h[j])
    lower = max(lower, -max_step)
    upper = min(upper, max_step)
    if lower >= upper:
        return None
    delta = rng.uniform(lower, upper)
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h
