    if s < target:

import numpy as np
import time
import math

def evaluate_h(h):
    """Return the C5 bound for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_capped_simplex(v, target):
    """
    Project vector v onto the set {x : sum(x)=target, 0<=x_i<=1}.
    Uses bisection on the Lagrange multiplier.
    """
    lo, hi = -1.0, 2.0
    for _ in range(40):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def tournament_select(pop, fitness, k, rng):
    """Return a candidate from pop chosen by k‑tournament."""
    idx = rng.choice(len(pop), size=k, replace=False)
    best = idx[0]
    best_val = fitness[best]
    for i in idx[1:]:
        if fitness[i] < best_val:
            best = i
            best_val = fitness[i]
    return pop[best]

def crossover(parent1, parent2, rng, target):
    """Arithmetic crossover followed by projection."""
    alpha = rng.random()
    child = alpha * parent1 + (1.0 - alpha) * parent2
    return project_to_capped_simplex(child, target)

def mutate(child, rng, target, step_max):
    """Two transfer moves plus optional Gaussian noise, then project."""
    n = child.shape[0]
    # transfer moves
    for _ in range(2):
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        max_inc = min(step_max, 1.0 - child[i])
        max_dec = min(step_max, child[j])
        max_tr = min(max_inc, max_dec)
        if max_tr > 0.0:
            delta = rng.random() * max_tr
            child[i] += delta
            child[j] -= delta
    # occasional Gaussian perturbation
    if rng.random() < 0.2:
        child += rng.normal(scale=0.05, size=n)
    child = np.clip(child, 0.0, 1.0)
    return project_to_capped_simplex(child, target)

def triangular_shape(n):
    """Symmetric triangle peaking at 1, scaled to satisfy the sum constraint."""
    idx = np.arange(n, dtype=float)
    x = idx / (n - 1.0) if n > 1 else np.zeros_like(idx)
    tri = np.maximum(0.0, 1.0 - 2.0 * np.abs(x - 0.5))
    return tri

    """Genetic search for a low C5 bound.

    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # Determine problem size and initial vector (if provided)
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    except NameError:

        h0 = None

    target = n / 2.0  # required sum of h values

    # --- Build initial population -------------------------------------------------
    pop_size = 200
    population = []

    # 1) use the supplied construction if present
    if h0 is not None:
        population.append(project_to_capped_simplex(h0, target))

    # 2) a binary baseline (half ones, remainder zero / fractional)
    k = int(math.floor(target))
    r = target - k
    base = np.zeros(n, dtype=float)
    if k > 0:
        base[:k] = 1.0
    if r > 0 and k < n:
        base[k] = r
    rng.shuffle(base)
    population.append(project_to_capped_simplex(base, target))

    # 3) a symmetric triangular shape
    tri = triangular_shape(n)
    population.append(project_to_capped_simplex(tri, target))

    # 4) fill the rest with random feasible vectors
    while len(population) < pop_size:
        rand_vec = rng.random(n)
        population.append(project_to_capped_simplex(rand_vec, target))

    # evaluate initial fitness
    fitness = [evaluate_h(ind) for ind in population]
    best_idx = int(np.argmin(fitness))
    best_h = population[best_idx].copy()
    best_score = fitness[best_idx]

    # --- GA parameters -----------------------------------------------------------
    max_generations = 2000
    tournament_k = 3
    crossover_prob = 0.7
    step_max = 0.2

    generation = 0
    while time.time() - start < budget_s and generation < max_generations:
        generation += 1
        new_pop = [best_h]                     # elitism
        # generate rest of the new population
        while len(new_pop) < pop_size:
            # parent selection
            p1 = tournament_select(population, fitness, tournament_k, rng)
            p2 = tournament_select(population, fitness, tournament_k, rng)

            # crossover
            if rng.random() < crossover_prob:

            else:
                child = p1.copy()

            # mutation
            child = mutate(child, rng, target, step_max)
            new_pop.append(child)

        # evaluate new generation
        new_fitness = [evaluate_h(ind) for ind in new_pop]

        # update best
        cur_best_idx = int(np.argmin(new_fitness))
        cur_best_score = new_fitness[cur_best_idx]
        if cur_best_score < best_score:
            best_score = cur_best_score
            best_h = new_pop[cur_best_idx].copy()

        population = new_pop
        fitness = new_fitness

        # early exit if target achieved
        if best_score <= 0.3808:

    # final safety projection
    best_h = project_to_capped_simplex(best_h, target)
    best_score = evaluate_h(best_h)

    return (best_h, float(best_score), n)
