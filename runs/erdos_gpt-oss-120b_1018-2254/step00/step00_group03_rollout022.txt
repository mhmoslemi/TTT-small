            # revert
            h[i] = 1.0
            h[j] = 0.0

        # ----- every few thousand steps try a targeted improvement -------------
        if it % 10000 == 0 and it > 0:
            # autocorrelation of h with itself
            a = np.correlate(h, h, mode='full')
            a[n - 1] = np.inf                       # ignore the perfect overlap
            worst_idx = int(np.argmin(a))
            shift = abs(worst_idx - (n - 1))

            if 0 < shift < n:
                # try to create a new coincident pair at this shift
                candidates = [p for p in range(n - shift)
                              if h[p] == 0.0 and h[p + shift] == 0.0]
                if candidates:
                    p = np.random.choice(candidates)
                    q = np.random.choice(list(ones))

                    # tentative swap
                    h[p] = 1.0
                    h[q] = 0.0
                    cand_c5 = _c5_value(h)

                    if cand_c5 < cur_c5:
                        # accept improvement
                        zeros.remove(p)
                        ones.add(p)
                        ones.remove(q)
                        zeros.add(q)
                        cur_c5 = cand_c5
                        if cand_c5 < best_c5:
                            best_c5 = cand_c5
                            best_h = h.copy()
                    else:
                        # revert
                        h[p] = 0.0
                        h[q] = 1.0

    # ------------------------------------------------------------------
    # return the best solution found
    # ------------------------------------------------------------------
    return best_h.tolist(), float(best_c5), n
```
