def project_to_sum_and_bounds(x, target):
    # x arbitrary (no guarantee in [0,1]), we want to find lambda such that after clipping x+lam to [0,1], sum = target.
    # Use binary search.
    lo = -1.0
    hi = 1.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        y = np.clip(x + lam, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(x + lo, 0.0, 1.0)

But note s monotonic with lam, and we ensure that at lam=-1 sum may be 0, at lam=1 sum may be n (or less), so target is within range.

def compute_c5(h):
    n = h.shape[0]
    return np.max(np.correlate(h, 1 - h, mode='full') * (2.0 / n))

def project_to_sum_and_bounds(x, target):
    # binary search for lambda such that sum(clip(x+lam, 0,1)) = target
    lo = -1.0
    hi = 1.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        y = np.clip(x + lam, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(x + lo, 0.0, 1.0)

def gradient_of_lag(h, lag):
    n = h.shape[0]
    dx = 2.0 / n
    g = np.zeros_like(h)
    if lag >= 0:
        l = lag
        g[:n-l] += (1.0 - h[l:])
        g[l:] -= h[:n-l]
    else:
        k = -lag
        g[k:] += (1.0 - h[:n-k])
        g[:n-k] -= h[k:]
    return g * dx

def find_max_lag(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    idx = np.argmax(corr)
    lag = idx - (n - 1)
    return lag, corr[idx]

def random_feasible_initial(n):

import numpy as np
import time
import math

def compute_c5(h):
    """Maximum overlap C₅ for a given step vector h."""
    n = h.shape[0]
    return float(np.max(np.correlate(h, 1 - h, mode="full") * (2.0 / n)))

def project_to_sum_and_bounds(x, target):
    """Project x onto the box [0,1] while enforcing sum(x)=target."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        y = np.clip(x + lam, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(x + lo, 0.0, 1.0)

def gradient_of_lag(h, lag):
    """Subgradient of the active lag term."""
    n = h.shape[0]
    dx = 2.0 / n
    g = np.zeros_like(h)
    if lag >= 0:
        l = lag
        g[: n - l] += (1.0 - h[l:])
        g[l:] -= h[: n - l]
    else:
        k = -lag
        g[k:] += (1.0 - h[: n - k])
        g[: n - k] -= h[k:]
    return g * dx

def find_max_lag(h):
    """Return lag producing the maximal overlap and the value."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    idx = int(np.argmax(corr))
    lag = idx - (n - 1)
    return lag, float(corr[idx])

def three_level_candidate(n):
    """Search the simple 1 / 0.5 / 0 pattern for the best C₅."""
    best_val = 1e9
    best_h = None
    for cnt in range(n // 2 + 1):
        h = np.full(n, 0.5, dtype=float)
        if cnt > 0:
            h[:cnt] = 1.0
            h[-cnt:] = 0.0
        val = compute_c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
    return best_h, best_val

def upsample_initial(arr, n):
    """Linear interpolation of a coarse vector to length n."""
    m = arr.shape[0]
    if m == n:
        return arr.copy()
    x_old = np.linspace(0.0, 2.0, m, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_feasible(n):
    """Random feasible vector (sum = n/2, values in [0,1])."""
    h = np.random.rand(n)
    return project_to_sum_and_bounds(h, n / 2.0)
