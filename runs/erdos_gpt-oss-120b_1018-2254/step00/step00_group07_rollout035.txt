def project_h(y, target_sum):
    # Clip to [0,1] after adding lambda via bisection
    low = -np.max(y) - 1.0
    high = 1.0 - np.min(y) + 1.0
    for _ in range(50):
        mid = (low + high) / 2.0
        x = np.clip(y + mid, 0, 1)
        s = x.sum()
        if s > target_sum:
            high = mid
        else:
            low = mid
    x = np.clip(y + low, 0, 1)
    # final correction
    diff = target_sum - x.sum()
    # adjust while diff > tolerance
    for _ in range(10):
        if abs(diff) < 1e-10:
            break
        free = (x > 0) & (x < 1)
        n_free = np.count_nonzero(free)
        if n_free == 0:
            break
        delta = diff / n_free
        x[free] += delta
        # clip and recompute diff
        x = np.clip(x, 0, 1)
        diff = target_sum - x.sum()
    return x
```

def build_trapezoid(a, N):
    # a fraction of total length in units of sample points? a is length of each slanted side.
    # However a is in continuous length units in [0,1] (max slanted side length).
    # We'll compute continuous x values.
    xs = np.linspace(0, 2, N, endpoint=False)  # sample points
    H = 1.0 / (2.0 - a)
    h = np.empty(N)
    # left region: x < a
    left = xs < a
    h[left] = xs[left] * (H / a) if a>0 else 0.0
    # middle region
    middle = (xs >= a) & (xs <= 2 - a)
    h[middle] = H
    # right region
    right = xs > 2 - a
    h[right] = (2 - xs[right]) * (H / a) if a>0 else 0.0
    return h
```

import time
import numpy as np

def _project_to_box_simplex(y: np.ndarray, target_sum: float) -> np.ndarray:
    """Project y onto the set {x: 0<=x<=1, sum(x)=target_sum} using a bisection on the shift."""
    low = -np.max(y) - 1.0
    high = 1.0 - np.min(y) + 1.0
    for _ in range(50):
        mid = (low + high) * 0.5
        x = np.clip(y + mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            high = mid
        else:
            low = mid
    x = np.clip(y + low, 0.0, 1.0)
    # final tiny correction to hit the sum exactly
    diff = target_sum - x.sum()
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        free = (x > 0.0) & (x < 1.0)
        n_free = free.sum()
        if n_free == 0:
            break
        delta = diff / n_free
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
        diff = target_sum - x.sum()
    return x


def _c5_value(h: np.ndarray, dx: float) -> float:
    """Compute C5 = max_k ∫ h(x)(1‑h(x+k))dx using discrete correlation."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _trapezoid_shape(a: float, N: int) -> np.ndarray:
    """Create a trapezoidal step function with side length a (0≤a≤1)."""
    if a <= 1e-12:
        return np.full(N, 0.5)
    xs = np.linspace(0.0, 2.0, N, endpoint=False)
    H = 1.0 / (2.0 - a)
    h = np.empty(N)
    left = xs < a
    h[left] = xs[left] * (H / a)
    middle = (xs >= a) & (xs <= 2.0 - a)
    h[middle] = H
    right = xs > 2.0 - a
    h[right] = (2.0 - xs[right]) * (H / a)
    return h


def _cosine_shape(A: float, N: int) -> np.ndarray:
    """Create h(x)=0.5 + A*cos(pi*x) on [0,2)."""
    xs = np.linspace(0.0, 2.0, N, endpoint=False)
    return 0.5 + A * np.cos(np.pi * xs)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    np.random.seed(seed)
    N = 200                     # number of discretisation points
    dx = 2.0 / N
    target = N / 2.0            # required sum of h-values

    best_h = None
    best_c5 = np.inf

    def consider(y: np.ndarray):
        nonlocal best_h, best_c5
        h = _project_to_box_simplex(y.astype(float), target)
        c5 = _c5_value(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # 1. initial construction, if provided
    try:
        consider(np.asarray(initial_h_values, dtype=float))
    except NameError:
        pass

    # 2. trivial uniform candidate
    consider(np.full(N, 0.5))

    # 3. systematic grid over trapezoids
    for a in np.linspace(0.0, 1.0, 41):
        consider(_trapezoid_shape(a, N))

    # 4. systematic grid over cosine amplitudes
    for A in np.linspace(0.0, 0.5, 51):
        consider(_cosine_shape(A, N))

    start = time.time()
    # 5. stochastic improvement loop
    no_improve = 0
    iteration = 0
    while time.time() - start < budget_s * 0.94:
        elapsed = time.time() - start
        # occasional full random restart
        if best_h is None or no_improve > 800:
            y = np.random.rand(N)
        else:
            # perturb current best with a shrinking Gaussian step
            sigma = 0.25 * (1.0 - elapsed / (budget_s * 0.94))
            y = best_h + np.random.normal(scale=sigma, size=N)
        prev_best = best_c5
        consider(y)
        if best_c5 < prev_best:
            no_improve = 0
        else:
            no_improve += 1
        iteration += 1
        if iteration > 30000:   # safety guard
            break

    return (best_h, float(best_c5), N)
