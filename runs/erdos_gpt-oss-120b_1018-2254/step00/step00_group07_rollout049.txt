def project_to_box_and_sum(v, target, max_iter=100):
    lo = -np.max(v) - 1  # ensure v+lo <= -1 -> after clip = 0 sum 0
    hi = 1 - np.min(v) + 1  # ensure v+hi >= 2 -> after clip = 1 sum = N
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        y = np.clip(v + mid, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    y = np.clip(v + (lo+hi)/2.0, 0.0, 1.0)
    # To correct small numeric discrepancy, adjust one element:
    diff = target - y.sum()
    if abs(diff) > 1e-10:
        # find index that can be adjusted without violating bounds
        # If diff>0 => need to increase; look for y<1
        # If diff<0 => need to decrease; look for y>0
        if diff > 0:
            idx = np.where(y < 1 - 1e-12)[0]
            if len(idx) > 0:
                i = idx[0]
                y[i] = min(1.0, y[i] + diff)
        else:
            idx = np.where(y > 1e-12)[0]
            if len(idx) > 0:
                i = idx[0]
                y[i] = max(0.0, y[i] + diff)  # diff negative
    return y

import time
import math
import numpy as np
import scipy.optimize as opt

# Global placeholder for the equality constraint (sum of h)
GLOBAL_TARGET = None


def eq_sum_constraint(x):
    """Equality constraint: sum(x) == GLOBAL_TARGET."""
    return np.sum(x) - GLOBAL_TARGET


def project_to_box_and_sum(v, target, max_iter=60):
    """
    Euclidean projection of a vector onto the set {x | 0<=x_i<=1, sum(x)=target}.
    Implements a simple bisection on the Lagrange multiplier.
    """
    # Initial bounds for lambda: ensure all entries are clipped to 0 or 1.
    lo = -np.max(v) - 1.0          # at lo, v+lo <= -1 => all clipped to 0, sum=0
    hi = 1.0 - np.min(v) + 1.0     # at hi, v+hi >= 2 => all clipped to 1, sum=N
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        y = np.clip(v + mid, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    y = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # Small correction for numerical drift
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(y < 1.0 - 1e-12)[0]
        else:
            idx = np.where(y > 1e-12)[0]
        if idx.size > 0:
            i = idx[0]
            y[i] = np.clip(y[i] + diff, 0.0, 1.0)
    return y


def compute_c5(h, dx):
    """Return the Erdős C5 value for a discrete step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def generate_random_feasible(n, target, rng):
    """Draw a random point from the feasible cube–simplex intersection."""
    # Dirichlet draws give a random point on the simplex (sum=1)
    v = rng.dirichlet(np.ones(n))
    v *= target
    return project_to_box_and_sum(v, target)


def local_optimize(x0, dx, target, maxiter=250):
    """Run a short SLSQP local optimization starting from x0."""
    global GLOBAL_TARGET
    GLOBAL_TARGET = target
    x0 = project_to_box_and_sum(x0, target)

    def obj_func(x):
        return compute_c5(x, dx)

    constraints = {"type": "eq", "fun": eq_sum_constraint}
    bounds = [(0.0, 1.0)] * len(x0)

    res = opt.minimize(
        obj_func,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=[constraints],
        options={"maxiter": maxiter, "ftol": 1e-9, "disp": False},
    )
    x_opt = project_to_box_and_sum(res.x, target)
    return x_opt, compute_c5(x_opt, dx), res

                cur_h = fresh
                cur_c5 = fresh_c5

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős C5 constant.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size and initial vector
    try:
        init_h = np.array(initial_h_values, dtype=float)
        n = init_h.size
    except Exception:
        n = 80
        init_h = np.full(n, 0.5)  # sum = n/2 => integral = 1

    target_sum = n / 2.0
    dx = 2.0 / n

    # Ensure feasibility of the initial candidate
    init_h = project_to_box_and_sum(init_h, target_sum)
    best_h = init_h.copy()
    best_c5 = compute_c5(best_h, dx)

    # Build a diverse set of starting points for local optimization
    start_points = []

    # 1. Pure sinusoidal pattern (satisfies constraints analytically)
    sin_h = 0.5 + 0.5 * np.cos(2.0 * np.pi * np.arange(n) / n)
    sin_h = project_to_box_and_sum(sin_h, target_sum)
    start_points.append(sin_h)

    # 2. The provided initial construction (if different)
    if not np.allclose(sin_h, init_h):
        start_points.append(init_h)

    # 3. A few random feasible vectors
    for _ in range(5):
        start_points.append(generate_random_feasible(n, target_sum, rng))

    # Local refinement using SLSQP on each start point
    for sp in start_points:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 15.0:
            break
        # Number of SLSQP iterations limited by remaining time
        maxiter = 250
        opt_h, opt_c5, _ = local_optimize(sp, dx, target_sum, maxiter=maxiter)
        if opt_c5 < best_c5:
            best_c5 = opt_c5
            best_h = opt_h.copy()

    # After local runs, apply a longer hill‑climbing phase
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 2.0:
        # Roughly allocate up to 200 000 iterations; actual count limited by time
        max_iters = 200_000
        best_h, best_c5 = hill_climb_iterative(
            best_h,
            best_c5,
            dx,
            target_sum,
            rng,
            max_iters,
            start_time,
            budget_s,
        )

    # Final projection and evaluation to guarantee feasibility
    best_h = project_to_box_and_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h, dx)

    return best_h.tolist(), float(best_c5), n
