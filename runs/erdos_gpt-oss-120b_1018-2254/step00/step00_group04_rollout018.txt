import time, numpy as np

def random_swap_move(h, step=0.05):
    n = len(h)
    # choose two distinct indices i (decrease) and j (increase)
    i, j = random.sample(range(n), 2)
    # Determine allowable transfer delta: h[i] can reduce at most h[i]; h[j] can increase at most 1 - h[j]
    max_transfer = min(h[i], 1.0 - h[j])
    if max_transfer <= 0:
        return h
    # Choose delta randomly up to max_transfer
    delta = random.random() * max_transfer  # uniform
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h

def random_feasible(n):

    target = n/2
    h = np.zeros(n)
    indices = np.arange(n)
    np.random.shuffle(indices)
    remaining = target
    for idx in indices:
        max_val = min(1.0, remaining)
        # assign random between 0 and max_val
        val = np.random.rand() * max_val
        h[idx] = val
        remaining -= val
        if remaining <= 0:
            break
    # If some remaining > 0 after assign to all indices, distribute remainder equally?
    if remaining > 1e-12:
        # distribute among all indices where not filled to 1.
        # Add small epsilon to each not at 1.
        mask = h < 1.0
        incr_per = remaining / np.count_nonzero(mask)
