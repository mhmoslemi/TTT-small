            n_restart = 2

        # ----- optimisation loops --------------------------------------------------
        for _ in range(n_restart):
            if time.time() - start > budget_s:
                break

            # jitter the base vector to explore other basins
            jitter = rng.normal(scale=0.02, size=N)
            h_start = np.clip(h0 + jitter, 0.0, 1.0)
            h_start = _project_sum(h_start, target_sum)

            # local smooth optimisation (if scipy available)
            if _HAS_SCIPY:
                elapsed = time.time() - start
                remain = budget_s - elapsed
                max_iter = 500
                if remain > 200:
                    max_iter = 2000
                elif remain > 80:
                    max_iter = 1000
                else:
                    max_iter = 300
                try:
                    res = minimize(
                        _soft_obj,
                        h_start,
                        method="L-BFGS-B",
                        bounds=[(0.0, 1.0)] * N,
                        options={"maxiter": max_iter, "disp": False},
                    )
                    h_opt = _project_sum(np.clip(res.x, 0.0, 1.0), target_sum)
                except Exception:
                    h_opt = h_start.copy()
            else:
                h_opt = h_start.copy()

            cur_c5 = _c5_of(h_opt)

            # ----- simple hill‑climbing refinement ---------------------------------
            def _hill_climb(cur_h, cur_val, steps):
                h_cur = cur_h.copy()
                best_val = cur_val
                for _ in range(steps):
                    if time.time() - start > budget_s:
                        break
                    i = rng.integers(N)
                    j = rng.integers(N)
                    if i == j:
                        continue
                    hi, hj = h_cur[i], h_cur[j]
                    max_delta = min(hi, 1.0 - hj)
                    if max_delta <= 0.0:
                        continue
                    delta = rng.random() * max_delta
                    h_new = h_cur.copy()
                    h_new[i] -= delta
                    h_new[j] += delta
                    new_val = _c5_of(h_new)
                    if new_val < best_val:
                        h_cur = h_new
                        best_val = new_val
                return h_cur, best_val

            rem = budget_s - (time.time() - start)
            if rem > 200:
                steps = 5000
            elif rem > 80:
                steps = 2000
            else:
                steps = 500
            h_ref, c5_ref = _hill_climb(h_opt, cur_c5, steps)

            # update global best
            if c5_ref < best_c5:
                best_c5, best_h, best_n = c5_ref, h_ref.copy(), N

        # also try pure hill‑climb from the (projected) base vector
        if time.time() - start < budget_s:
            base_c5 = _c5_of(h0)
            rem = budget_s - (time.time() - start)
            steps = 3000 if rem > 150 else 1000
            h_base_ref, c5_base_ref = _hill_climb(h0, base_c5, steps)
            if c5_base_ref < best_c5:
                best_c5, best_h, best_n = c5_base_ref, h_base_ref.copy(), N

    # ----- fallback if nothing succeeded -----------------------------------------
    if best_h is None:
        if "initial_h_values" in globals():
            best_h = np.array(initial_h_values, dtype=float)
            best_n = best_h.shape[0]
        else:
            best_n = 2
            best_h = np.full(best_n, 0.5, dtype=float)
        best_h = _project_sum(best_h, best_n / 2.0)
        best_c5 = _c5_of(best_h)
    else:
        # ensure final constraints are satisfied
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = _project_sum(best_h, best_n / 2.0)
        best_c5 = _c5_of(best_h)

    return (best_h, float(best_c5), int(best_n))
