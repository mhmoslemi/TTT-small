import numpy as np
import time
import math

def _compute_c5(h):
    """Compute the C5 value for a given step function h."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    return float(np.max(conv) * dx)


def _propose_move(h, max_step):
    """Generate a new candidate by transferring a small amount between two indices."""
    n = h.shape[0]
    i, j = np.random.choice(n, 2, replace=False)

    # allowed delta so that both entries stay in [0,1]
    allowed_low = max(-h[i], -(1.0 - h[j]))
    allowed_high = min(1.0 - h[i], h[j])
    if allowed_low >= allowed_high:
        return None

    # intersect with the magnitude bound
    low = max(allowed_low, -max_step)
    high = min(allowed_high, max_step)
    if low >= high:
        # fallback to the full feasible interval
        low, high = allowed_low, allowed_high
        if low >= high:
            return None

    delta = np.random.uniform(low, high)
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    # Safety clipping (should already be satisfied)
    new_h[i] = min(max(new_h[i], 0.0), 1.0)
    new_h[j] = min(max(new_h[j], 0.0), 1.0)
    return new_h


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the Erdős overlap constant C5.

    Returns:
        (h_values, c5_bound, n_points)
    """
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Initialise h
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_points = h0.shape[0]
        # Ensure feasibility
        h0 = np.clip(h0, 0.0, 1.0)
        target_sum = n_points / 2.0
        cur_sum = h0.sum()
        if abs(cur_sum - target_sum) > 1e-12:
            # Simple fix: spread the needed correction over the first free entry
            diff = target_sum - cur_sum
            for idx in range(n_points):
                if diff > 0 and h0[idx] < 1.0:
                    add = min(diff, 1.0 - h0[idx])
                    h0[idx] += add
                    diff -= add
                elif diff < 0 and h0[idx] > 0.0:
                    sub = min(-diff, h0[idx])
                    h0[idx] -= sub
                    diff += sub
                if abs(diff) < 1e-12:
                    break
    else:
        # Fallback: a simple binary construction with a possible fractional entry
        n_points = 200
        target_sum = n_points / 2.0
        h0 = np.zeros(n_points, dtype=float)
        ones = int(np.floor(target_sum))
        frac = target_sum - ones
        perm = np.random.permutation(n_points)
        h0[perm[:ones]] = 1.0
        if frac > 0:
            h0[perm[ones]] = frac

    # ------------------------------------------------------------------
    # Local search (simulated annealing / hill climbing)
    # ------------------------------------------------------------------
    best_h = h0.copy()
    best_c5 = _compute_c5(best_h)
    cur_h = best_h.copy()
    cur_c5 = best_c5

    # Annealing schedule parameters
    max_step_start = 0.10
    min_step = 1e-5
    temp_start = 0.010
    temp_end = 1e-6

    # Main optimisation loop
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.97:
            break

        frac = elapsed / budget_s
        # step size shrinks linearly
        max_step = max_step_start * (1.0 - frac) + min_step
        # temperature shrinks linearly
        temp = temp_start * (1.0 - frac) + temp_end

        cand = _propose_move(cur_h, max_step)
        if cand is None:
            continue

        cand_c5 = _compute_c5(cand)

        if cand_c5 < cur_c5:
            cur_h = cand
            cur_c5 = cand_c5
            if cand_c5 < best_c5:
                best_h = cand.copy()
                best_c5 = cand_c5
        else:
            # Accept worse solutions with a probability that decays with temperature
            if temp > 0.0 and np.random.rand() < math.exp(-(cand_c5 - cur_c5) / temp):
                cur_h = cand
                cur_c5 = cand_c5

    # ------------------------------------------------------------------
    # Final polishing – enforce constraints exactly
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    target_sum = best_h.shape[0] / 2.0
    cur_sum = best_h.sum()
    diff = target_sum - cur_sum
    if abs(diff) > 1e-12:
        # Distribute the remaining discrepancy greedily
        for idx in range(best_h.shape[0]):
            if diff > 0 and best_h[idx] < 1.0:
                add = min(diff, 1.0 - best_h[idx])
                best_h[idx] += add
                diff -= add
            elif diff < 0 and best_h[idx] > 0.0:
                sub = min(-diff, best_h[idx])
                best_h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break

    final_c5 = _compute_c5(best_h)

    return (best_h, final_c5, best_h.shape[0])
