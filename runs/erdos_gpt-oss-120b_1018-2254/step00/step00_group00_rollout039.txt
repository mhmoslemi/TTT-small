def make_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    # If s is close enough, exit
    if abs(s - target) < 1e-9:
        return h
    # adjust uniformly
    # We'll use simple iterative approach.
    for _ in range(10):
        shift = (target - s) / n
        h += shift
        # Clip again
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-9:
            break
    # final normalization: might still be off. As last resort, scale proportionally but keep within bounds.
    if abs(s - target) > 1e-6:
        # Simple fallback: distribute remaining diff over entries not at bound.
        diff = target - s
        # indices where we can add (if diff>0) or subtract (if diff<0)
        if diff > 0:
            idx = np.where(h < 1)[0]
        else:
            idx = np.where(h > 0)[0]
        # distribute equally
        if len(idx) > 0:
            inc = diff / len(idx)
            h[idx] += inc
            h = np.clip(h, 0.0, 1.0)
    # final clip
    return h

def project_onto_sum_bounds(v, target):

    x = np.clip(v, 0.0, 1.0)
    # Simple iterative fix
    for _ in range(10):
        s = x.sum()
        if abs(s - target) < 1e-9:
            break
        # compute diff
        diff = target - s
        # distribute diff uniformly to all entries not at bounds (0 or 1)
        # Determine adjustable indices
        if diff > 0:
            # need to increase sum: can increase entries with x < 1
            idx = np.where(x < 1)[0]
        else:
            # need to decrease sum: can decrease entries with x > 0
            idx = np.where(x > 0)[0]
        if len(idx) == 0:
            break
        # amount per index
        inc = diff / len(idx)
        x[idx] += inc
        # ensure bounds
        x = np.clip(x, 0.0, 1.0)

    s = x.sum()
    if abs(s - target) > 1e-8:

def project_sum_bounds(h, target):
    x = np.clip(h, 0.0, 1.0)
    # Use simple iterative adjustment
    for _ in range(20):
        s = x.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(x < 1)[0]
        else:
            idx = np.where(x > 0)[0]
        if len(idx) == 0:
            break
        inc = diff / len(idx)
        x[idx] += inc
        x = np.clip(x, 0.0, 1.0)
    # final small fix
    s = x.sum()
    diff = target - s
    if abs(diff) > 1e-9:
        # adjust a single element
        for i in range(len(x)):
            if diff > 0 and x[i] < 1:
                adj = min(diff, 1 - x[i])
                x[i] += adj
                diff -= adj
            elif diff < 0 and x[i] > 0:
                adj = min(-diff, x[i])
                x[i] -= adj
                diff += adj
            if abs(diff) < 1e-12:
                break
    return x

def targeted_move(h, shift, scale):
    n = len(h)
    if shift >= 0:
        # compare h[i] and h[i+shift] for i in [0, n-shift-1]
        diff = h[:n-shift] - h[shift:]
        viable = np.where(diff > 1e-12)[0]
        if viable.size == 0:
            return None
        i = np.random.choice(viable)
        src = i
        tgt = i + shift
    else:
        d = -shift
        diff = h[d:] - h[:n-d]
        viable = np.where(diff > 1e-12)[0]
        if viable.size == 0:
            return None
        i = np.random.choice(viable)
        src = i + d
        tgt = i
    # allowed transfer
    allowed = min(h[src], 1.0 - h[tgt])
    if allowed <= 1e-12:
        return None
    # delta proportional to scale
    delta = allowed * scale * np.random.rand()
    # ensure small but positive
    if delta < 1e-15:
        delta = allowed * scale * 0.5
    new_h = h.copy()
    new_h[src] -= delta
    new_h[tgt] += delta
    return new_h

Now random_move:

def random_move(h, scale):
    n = len(h)
    i = np.random.randint(0, n)
    j = np.random.randint(0, n-1)
    if j >= i:
        j += 1
    hi = h[i]
    hj = h[j]
    max_pos = min(1.0 - hi, hj)
    max_neg = min(hi, 1.0 - hj)
    total_range = max_pos + max_neg
    if total_range <= 1e-12:
        return None
    # sample delta uniformly in allowed range
    # generate r in [-max_neg, max_pos]
    r = np.random.rand() * total_range - max_neg
    # scale
    delta = r * scale
    # ensure within bounds
    delta = max(-max_neg, min(max_pos, delta))
    if abs(delta) < 1e-15:
        return None
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    return new_h

import numpy as np
import math
import time
import random

def compute_c5(h, dx):
    """Return the maximal overlap integral C5 for a step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_sum_bounds(v, target):
    """Project vector v onto the box [0,1]^n while enforcing sum(v)=target."""
    x = np.clip(v, 0.0, 1.0)
    # simple iterative redistribution
    for _ in range(30):
        s = x.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(x < 1.0)[0]
        else:
            idx = np.where(x > 0.0)[0]
        if idx.size == 0:
            break
        inc = diff / idx.size
        x[idx] += inc
        x = np.clip(x, 0.0, 1.0)
    # final tiny fix on a single coordinate if needed
    s = x.sum()
    diff = target - s
    if abs(diff) > 1e-10:
        for i in range(len(x)):
            if diff > 0 and x[i] < 1.0:
                adj = min(diff, 1.0 - x[i])
                x[i] += adj
                diff -= adj
            elif diff < 0 and x[i] > 0.0:
                adj = min(-diff, x[i])
                x[i] -= adj
                diff += adj
            if abs(diff) < 1e-12:
                break
    return x

def targeted_move(h, shift, scale):
    """Move mass from a larger entry towards a smaller entry along the worst shift."""
    n = len(h)
    if shift >= 0:
        diff = h[: n - shift] - h[shift:]
        candidates = np.where(diff > 1e-12)[0]
        if candidates.size == 0:
            return None
        i = np.random.choice(candidates)
        src, tgt = i, i + shift
    else:
        d = -shift
        diff = h[d:] - h[: n - d]
        candidates = np.where(diff > 1e-12)[0]
        if candidates.size == 0:
            return None
        i = np.random.choice(candidates)
        src, tgt = i + d, i
    allowed = min(h[src], 1.0 - h[tgt])
    if allowed <= 1e-12:
        return None
    delta = allowed * scale * np.random.rand()
    if delta < 1e-15:
        delta = allowed * scale * 0.5
    new_h = h.copy()
    new_h[src] -= delta
    new_h[tgt] += delta
    return new_h

def random_move(h, scale):
    """Transfer a random amount between two positions while preserving the sum."""
    n = len(h)
    i = np.random.randint(0, n)
    j = np.random.randint(0, n - 1)
    if j >= i:
        j += 1
    hi, hj = h[i], h[j]
    max_pos = min(1.0 - hi, hj)      # increase i, decrease j
    max_neg = min(hi, 1.0 - hj)      # decrease i, increase j
    total = max_pos + max_neg
    if total <= 1e-12:
        return None
    # pick a delta uniformly in the feasible interval and then scale it
    r = np.random.rand() * total - max_neg   # in [-max_neg, max_pos]
    delta = r * scale
    delta = max(-max_neg, min(max_pos, delta))
    if abs(delta) < 1e-15:
        return None
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    return new_h

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] with reduced C5."""
    np.random.seed(seed)
    random.seed(seed)

    # Determine problem size and initial vector
    if "initial_h_values" in globals():
        init_arr = np.array(initial_h_values, dtype=float)
        n = init_arr.shape[0]
        target_sum = n / 2.0
        h = project_sum_bounds(init_arr, target_sum)
    else:
        n = 45
        target_sum = n / 2.0
        h = np.full(n, 0.5, dtype=float)          # already feasible
        h = project_sum_bounds(h, target_sum)

    dx = 2.0 / n
    best_h = h.copy()
    best_score = compute_c5(best_h, dx)

    curr_h = h.copy()
    curr_score = best_score

    start = time.time()
    T0 = 0.01               # base temperature
    prob_targeted = 0.7
    no_improve = 0
    max_no_improve = 20000

    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s:
            break

        # cooling factor (linear)
        factor = max(0.0, 1.0 - elapsed / budget_s)
        T = factor * T0

        # identify the shift producing the current maximum overlap
        corr = np.correlate(curr_h, 1.0 - curr_h, mode="full")
        shift = int(np.argmax(corr) - (n - 1))

        # propose a move
        if np.random.rand() < prob_targeted:
            new_h = targeted_move(curr_h, shift, factor)
            if new_h is None:
                new_h = random_move(curr_h, factor)
        else:
            new_h = random_move(curr_h, factor)

        if new_h is None:
            continue

        new_score = compute_c5(new_h, dx)

        # acceptance test (simulated annealing)
        accept = False
        if new_score < curr_score - 1e-12:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_score - curr_score) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            curr_h = new_h
            curr_score = new_score
            no_improve = 0
        else:
            no_improve += 1

        # track global best
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_h = new_h.copy()
            no_improve = 0

        # occasional restart if stuck
        if no_improve >= max_no_improve:
            rand_vec = np.random.rand(n)
            rand_vec = project_sum_bounds(rand_vec, target_sum)
            curr_h = rand_vec
            curr_score = compute_c5(curr_h, dx)
            no_improve = 0

    # final cleanup
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_bounds(best_h, n / 2.0)
    best_score = compute_c5(best_h, dx)
    return best_h, best_score, n
