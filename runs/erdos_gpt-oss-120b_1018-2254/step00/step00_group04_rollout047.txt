def neighbor_h(h, max_delta=0.1):
    h_new = h.copy()
    n = len(h)
    i, j = random.sample(range(n), 2)
    # Compute max possible delta up and down:
    # delta <= min(max_delta, 1 - h[i]) to keep <=1
    # delta >= -max_delta, also >= -h[i]
    # For j: to keep >=0, delta >= -h[j] (since we subtract delta from j)
    # delta <= max_delta, also <= h[j] (since subtracting)
    # Actually we want to increase h[i] by delta and decrease h[j] by delta, so sum unchanged.
    # So delta must satisfy:
    # h[i] + delta in [0,1] => delta in [-h[i], 1 - h[i]]
    # h[j] - delta in [0,1] => delta in [h[j]-1, h[j]]
    # Intersection with [-max_delta, max_delta].
    delta_low = max(-max_delta, -h_new[i], h_new[j]-1)
    delta_high = min(max_delta, 1 - h_new[i], h_new[j])
    if delta_low > delta_high:
        # cannot adjust with this max_delta, try smaller random indices
        return h_new
    delta = random.uniform(delta_low, delta_high)
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

    start_time = time.time()

def project_to_box_sum(v, target):
    # bisection lambda
    low = -2.0  # enough lower bound
    high = 2.0
    for _ in range(60):
        lam = (low + high) / 2.0
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s < target:
            low = lam
        else:
            high = lam
    x = np.clip(v + (low + high) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        free = (x > 0) & (x < 1)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

import time
import math
import random
import numpy as np

def _project_to_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project vector v onto the intersection of [0,1]^n and sum(v)=target."""
    low, high = -2.0, 2.0
    for _ in range(60):
        lam = (low + high) / 2.0
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s < target:
            low = lam
        else:
            high = lam
    x = np.clip(v + (low + high) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 0.0) & (x < 1.0)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def _repair_h(h: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce the integral constraint exactly."""
    n = h.shape[0]
    target = n / 2.0
    return _project_to_box_sum(h, target)

def _random_feasible(n: int) -> np.ndarray:
    """Generate a random feasible vector of length n."""
    v = np.random.rand(n)
    return _repair_h(v)

def _initial_solution(n: int) -> np.ndarray:
    """Return an initial feasible solution, using the supplied parent if available."""
    if 'initial_h_values' in globals():
        arr = np.array(initial_h_values, dtype=float)
        if arr.shape[0] != n:
            x_old = np.linspace(0.0, 2.0, arr.shape[0], endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            arr = np.interp(x_new, x_old, arr)
        arr = np.clip(arr, 0.0, 1.0)
        return _repair_h(arr)
    return _random_feasible(n)

def _compute_c5(h: np.ndarray) -> float:
    """Compute the overlap constant C5 for discrete h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _neighbor_h(h: np.ndarray, max_delta: float) -> np.ndarray:
    """Create a neighbor of h: either a swap or a small pairwise adjustment."""
    n = h.shape[0]
    if random.random() < 0.5:
        i, j = random.sample(range(n), 2)
        h_new = h.copy()
        h_new[i], h_new[j] = h[j], h[i]
        return h_new
    i, j = random.sample(range(n), 2)
    hi, hj = h[i], h[j]
    delta_low = max(-max_delta, -hi, hj - 1.0)
    delta_high = min(max_delta, 1.0 - hi, hj)
    if delta_low >= delta_high:
        # fallback to a simple swap if no feasible delta
        h_new = h.copy()
        h_new[i], h_new[j] = h[j], h[i]
        return h_new
    delta = random.uniform(delta_low, delta_high)
    h_new = h.copy()
    h_new[i] = hi + delta
    h_new[j] = hj - delta
    return h_new

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    # Problem size
    n = len(initial_h_values) if 'initial_h_values' in globals() else 84

    # Initial feasible vector
    h_curr = _initial_solution(n)
    best_h = h_curr.copy()
    best_val = _compute_c5(best_h)
    curr_val = best_val

    start_time = time.time()
    max_temp = 0.02
    min_temp = 1e-5
    max_delta0 = 0.5
    stagnation = 0

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        frac = elapsed / budget_s
        T = max_temp * (1.0 - frac) + min_temp * frac
        max_delta = max_delta0 * (1.0 - frac) + 0.02 * frac

        h_next = _neighbor_h(h_curr, max_delta)
        next_val = _compute_c5(h_next)

        if next_val < curr_val:
            accept = True
        else:
            accept = T > 0.0 and random.random() < math.exp(-(next_val - curr_val) / T)

        if accept:
            h_curr = h_next
            curr_val = next_val
            stagnation = 0
        else:
            stagnation += 1

        if curr_val < best_val:
            best_val = curr_val
            best_h = h_curr.copy()

        if stagnation >= 5000:
            h_curr = _random_feasible(n)
            curr_val = _compute_c5(h_curr)
            stagnation = 0

    best_h = _repair_h(best_h)
    best_val = _compute_c5(best_h)
    return best_h.tolist(), float(best_val), n
