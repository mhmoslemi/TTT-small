```
def random_move(h, step=0.01):
    n = len(h)
    i, j = np.random.choice(n, 2, replace=False)
    # amount to transfer from i to j
    max_transfer = min(step, h[i], 1 - h[j])
    if max_transfer <= 1e-12:
        return h  # cannot move
    delta = np.random.uniform(0, max_transfer)
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new
```

    lo = -np.max(v)  # v+lo <=0 => all zeros
    hi = 1.0 - np.min(v)  # v+hi >=1 => all ones

            lo = mid
        else:
            hi = mid

def compute_c5(h: np.ndarray) -> float:
    """Return the C5 value for a step function given by samples h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    """
    Project vector v onto the intersection of the box [0,1]^n and the affine hyperplane
    sum(x) = target.  Returns a new array with the same shape as v.

    lo = -np.max(v)          # ensures all entries become 0 after clipping

                continue
            delta = np.random.rand() * max_transfer
            h_new = current_h.copy()
            h_new[i] -= delta
            h_new[j] += delta

        # evaluate candidate
        new_c5 = compute_c5(h_new)

        # metropolis acceptance
        if new_c5 < current_c5 or np.random.rand() < math.exp(-(new_c5 - current_c5) / T):
            current_h = h_new
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
                if best_c5 < 0.3808:      # reached target, stop early
                    break

    # ----- post‑processing: periodic smoothing ----------------------------------
    kernel = np.array([0.25, 0.5, 0.25], dtype=float)   # doubly‑stochastic
    smooth_h = best_h.copy()
    for _ in range(5):
        smooth_h = smooth_periodic(smooth_h, kernel)
        # small numeric drift may appear; enforce box constraints and exact sum
        smooth_h = np.clip(smooth_h, 0.0, 1.0)
        smooth_h = project_onto_simplex_box(smooth_h, target_sum)
        cand_c5 = compute_c5(smooth_h)
        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = smooth_h.copy()

    # final safety projection and evaluation
    best_h = project_onto_simplex_box(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
