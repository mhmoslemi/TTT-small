    corr = np.correlate(h, 1 - h, mode='full')

def project_to_sum_and_bounds(h, target):
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(10):
            mask = (h > 0.0) & (h < 1.0)
            if not mask.any():
                break
            delta = diff / mask.sum()
            h[mask] += delta
            np.clip(h, 0.0, 1.0, out=h)
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    if abs(h.sum() - target) > 1e-8:
        idx = np.random.randint(0, h.shape[0])
        need = target - h.sum()
        h[idx] = np.clip(h[idx] + need, 0.0, 1.0)
    return h

def initialize_h(n):
    try:
        parent = initial_h_values
    except NameError:
        parent = None
    if parent is not None:
        p = np.array(parent, dtype=float)
        if p.shape[0] == n:
            h0 = p.copy()
        else:
            xp = np.linspace(0, 2, p.shape[0], endpoint=False)
            x = np.linspace(0, 2, n, endpoint=False)
            h0 = np.interp(x, xp, p)
    else:
        h0 = np.random.rand(n)
    return project_to_sum_and_bounds(h0, n / 2.0)

def simulated_annealing(start_h, max_seconds):
    n = start_h.shape[0]
    target = n / 2.0
    curr = start_h.copy()
    best = curr.copy()
    curr_score = evaluate_c5(curr)
    best_score = curr_score
    T0 = 0.04
    stagnate = 0
    t0 = time.time()
    while True:
        if time.time() - t0 >= max_seconds:
            break
        i = np.random.randint(0, n)
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1
        hi, hj = curr[i], curr[j]
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue
        delta = np.random.uniform(low, high)
        new = curr.copy()
        new[i] = hi + delta
        new[j] = hj - delta
        new_score = evaluate_c5(new)
        elapsed = time.time() - t0
        T = T0 * max(0.0, 1.0 - elapsed / max_seconds)
        if new_score < curr_score or (T > 1e-12 and np.random.rand() < math.exp(-(new_score - curr_score) / T)):
            curr = new
            curr_score = new_score
            stagnate = 0
            if new_score < best_score:
                best = new.copy()
                best_score = new_score
        else:
            stagnate += 1
            if stagnate >= 5000:
                pert = best.copy()
                m = max(1, n // 20)
                idxs = np.random.choice(n, size=m, replace=False)
                for k in idxs:
                    pert[k] = np.clip(pert[k] + np.random.uniform(-0.05, 0.05), 0.0, 1.0)
                pert = project_to_sum_and_bounds(pert, target)
                curr = pert
                curr_score = evaluate_c5(curr)
                stagnate = 0
    return best, best_score

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    total = float(budget_s)
    start = time.time()
    # Coarse phase
    n1 = 80
    t1 = min(60.0, total * 0.12)
    h1 = initialize_h(n1)
    best1, _ = simulated_annealing(h1, t1)
    # Fine phase
    n2 = 200
    x1 = np.linspace(0, 2, n1, endpoint=False)
    x2 = np.linspace(0, 2, n2, endpoint=False)
    h2 = np.interp(x2, x1, best1)
    h2 = project_to_sum_and_bounds(h2, n2 / 2.0)
    elapsed = time.time() - start
    remaining = max(1.0, total - elapsed - 0.5)
    best2, _ = simulated_annealing(h2, remaining)
    best2 = project_to_sum_and_bounds(best2, n2 / 2.0)
    c5 = evaluate_c5(best2)
    return (best2, c5, n2)
