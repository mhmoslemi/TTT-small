def project_to_feasible(x, S):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo+hi)/2
        y = np.clip(x - mid, 0, 1)
        total = y.sum()
        if total > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - (lo+hi)/2, 0, 1)
    # adjust any small residual via scaling? near enough.
    # Ensure sum exactly S by adjusting a small subset.
    diff = S - y.sum()
    if abs(diff) > 1e-12:
        # distribute diff over non-bound entries
        # find indices not at 0 or 1
        free = np.where((y>0) & (y<1))[0]
        if free.size > 0:
            # add diff equally
            y[free] += diff / free.size
            y = np.clip(y, 0, 1)
        else:
            # adjust ones or zeros limited
            # add diff to one random element within bounds
            i = np.random.randint(len(y))
            y[i] = np.clip(y[i] + diff, 0, 1)
    # final clip if needed
    y = np.clip(y, 0, 1)
    # final scale if still off
    if abs(y.sum() - S) > 1e-8:
        # adjust proportionally on free entries
        diff = S - y.sum()
        # distribute among all entries scaled proportionally?
        # use simple addition to first element:
        y[0] = np.clip(y[0] + diff, 0, 1)
    return y

import time
import random
import numpy as np

def compute_c5(h, dx):
    """Return max overlap integral for a given step function h."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_feasible(x, target_sum):
    """
    Euclidean projection of vector x onto the box [0,1]^n intersected with the hyperplane sum = target_sum.
    Uses bisection on the Lagrange multiplier λ for y_i = clip(x_i - λ, 0, 1).
    """
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    λ = (lo + hi) / 2.0
    y = np.clip(x - λ, 0.0, 1.0)
    # Small residual correction (unlikely needed)
    diff = target_sum - y.sum()
    if abs(diff) > 1e-12:
        free = np.where((y > 0.0) & (y < 1.0))[0]
        if free.size > 0:
            y[free] += diff / free.size
            y = np.clip(y, 0.0, 1.0)
        else:
            # adjust a random index as a fallback
            i = random.randrange(len(y))
            y[i] = np.clip(y[i] + diff, 0.0, 1.0)
    return y

def generate_candidate(N, floor_S, frac):
    """Generate a random feasible step function with values 0, 1 and possibly one fractional entry."""
    idx = np.arange(N)
    np.random.shuffle(idx)
    h = np.zeros(N, dtype=np.float64)
    if floor_S > 0:
        h[idx[:floor_S]] = 1.0
    if frac > 1e-12:
        h[idx[floor_S]] = frac
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize the Erdős minimum overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    random.seed(seed)

    # Determine problem size
    if 'initial_h_values' in globals():
        init_arr = np.array(initial_h_values, dtype=np.float64)
        N = init_arr.shape[0]
    else:
        N = 200  # default moderate size if no parent provided

    target_sum = N / 2.0
    floor_S = int(np.floor(target_sum))
    frac = target_sum - floor_S
    dx = 2.0 / N

    # Initialise best solution (use provided parent if feasible)
    best_h = None
    best_val = np.inf

    if 'initial_h_values' in globals():
        init_h = np.clip(init_arr, 0.0, 1.0)
        init_h = project_to_feasible(init_h, target_sum)
        val = compute_c5(init_h, dx)
        best_h = init_h.copy()
        best_val = val

    # If no feasible parent, start from a random candidate
    if best_h is None:
        best_h = generate_candidate(N, floor_S, frac)
        best_val = compute_c5(best_h, dx)

    # Current state for simulated annealing
    cur_h = best_h.copy()
    cur_val = best_val

    start_time = time.time()
    end_time = start_time + budget_s * 0.99  # slight safety margin
    total_time = end_time - start_time
    T0 = 0.05  # initial temperature for Metropolis acceptance
    steps_without_improve = 0
    RESTART_THRESHOLD = 20000

    while time.time() < end_time:
        # Choose a random swap between a non‑zero entry and a zero entry
        nonzero_idx = np.where(cur_h > 0.0)[0]
        zero_idx = np.where(cur_h == 0.0)[0]
        if nonzero_idx.size == 0 or zero_idx.size == 0:
            # degenerate case: re‑initialise
            cur_h = generate_candidate(N, floor_S, frac)
            cur_val = compute_c5(cur_h, dx)
            steps_without_improve = 0
            continue

        i = np.random.choice(nonzero_idx)
        j = np.random.choice(zero_idx)
        neighbor = cur_h.copy()
        neighbor[i], neighbor[j] = neighbor[j], neighbor[i]

        neighbor_val = compute_c5(neighbor, dx)
        delta = neighbor_val - cur_val

        # Linear temperature schedule
        elapsed = time.time() - start_time
        T = max(1e-8, T0 * (1.0 - elapsed / total_time))

        accept = False
        if delta < 0.0:
            accept = True
        else:
            # Metropolis criterion
            if np.random.rand() < np.exp(-delta / T):
                accept = True

        if accept:
            cur_h = neighbor
            cur_val = neighbor_val
            steps_without_improve = 0 if delta < 0.0 else steps_without_improve + 1
            if neighbor_val < best_val:
                best_h = neighbor.copy()
                best_val = neighbor_val
                steps_without_improve = 0
        else:
            steps_without_improve += 1

        # Periodic restart if stuck
        if steps_without_improve >= RESTART_THRESHOLD:
            cur_h = generate_candidate(N, floor_S, frac)
            cur_val = compute_c5(cur_h, dx)
            steps_without_improve = 0

    # Return as a plain list for compatibility
    return best_h.tolist(), float(best_val), int(N)
