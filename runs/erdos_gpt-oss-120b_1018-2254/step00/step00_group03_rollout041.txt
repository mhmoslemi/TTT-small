def evaluate_c5(h, dx):
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def project_to_feasible(v, target_sum):
    v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    eps = 1e-12
    if abs(s - target_sum) <= eps:
        return v
    # adjust
    if s > target_sum:
        surplus = s - target_sum
        # reduce from largest entries
        for i in np.argsort(-v):
            if surplus <= eps:
                break
            dec = min(v[i], surplus)
            v[i] -= dec
            surplus -= dec
    else:
        deficit = target_sum - s
        # increase smallest entries
        for i in np.argsort(v):
            if deficit <= eps:
                break
            inc = min(1.0 - v[i], deficit)
            v[i] += inc
            deficit -= inc
    # final small correction if needed
    s2 = v.sum()
    if abs(s2 - target_sum) > 1e-9:
        diff = target_sum - s2
        # distribute diff to a feasible entry
        for i in range(len(v)):
            if 0 <= v[i] + diff <= 1:
                v[i] += diff
                break
    return v

def propose_move(h, max_delta, rng):
    n = len(h)
    # attempt a few random pairs
    for _ in range(10):
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        avail_from = h[i]
        avail_to = 1.0 - h[j]

        epsilon = epsilon * rng.random()
        if epsilon <= 1e-12:
            continue
        h_new = h.copy()
        h_new[i] = h[i] - epsilon

    v = np.clip(v, 0.0, 1.0)

            v[i] -= dec
            excess -= dec

    """Return a step‑function h, its C5 bound, and number of sample points."""
    rng = np.random.default_rng(seed)

    # number of points – use parent if present, else fallback to 70
    try:
        n = int(len(initial_h_values))
    except NameError:
        n = 70

    dx = 2.0 / n
    target_sum = n / 2.0

    # ----- generate a diverse set of initial candidates -----
    candidates = []

    # 1. sinusoidal family (including powers)
    x = (np.arange(n) + 0.5) * dx
    base = np.sin(np.pi * x / 2.0) * (np.pi / 4.0)   # scaled so that integral ≈1
    base = np.clip(base, 0.0, 1.0)
    for p in np.linspace(0.5, 2.0, 7):
        h = (np.sin(np.pi * x / 2.0) ** p) * (np.pi / 4.0)
        h = np.clip(h, 0.0, 1.0)
        h = _project_to_feasible(h, target_sum)
        candidates.append(h)

    # 2. pure triangle (the classic construction)
    tri = np.clip(1.0 - np.abs(x - 1.0), 0.0, 1.0)
    tri = _project_to_feasible(tri, target_sum)
    candidates.append(tri)

    # 3. uniform half‑height (feasible by construction)
    uni = np.full(n, 0.5)
    uni = _project_to_feasible(uni, target_sum)
    candidates.append(uni)

    # 4. a few random feasible vectors
    for _ in range(5):
        rand = rng.random(n)
        rand = rand / rand.sum() * target_sum
        rand = _project_to_feasible(rand, target_sum)
        candidates.append(rand)

    # 5. parent construction if available
    try:
        parent = np.asarray(initial_h_values, dtype=float)
        if parent.shape[0] == n:
            parent = _project_to_feasible(parent, target_sum)
            candidates.append(parent)
    except NameError:
        pass

    # pick the best starter
    best_start = None
    best_val = np.inf
    for h in candidates:
        val = _evaluate_c5(h, dx)
        if val < best_val:
            best_val = val
            best_start = h

    # ----- optimisation phase -----
    start_time = time.time()
    remaining = max(0.0, budget_s - (time.time() - start_time))
    if remaining > 1.0:
        best_h, best_c5 = _optimize(best_start, target_sum, dx, remaining, rng, target_bound=0.3808)
    else:
        best_h = best_start
        best_c5 = best_val

    return (best_h.tolist(), float(best_c5), n)
