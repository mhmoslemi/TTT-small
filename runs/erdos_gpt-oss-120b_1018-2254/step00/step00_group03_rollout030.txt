                if new_c5 < cur_c5 - 1e-12:
                    cur_c5 = new_c5
                    no_improve = 0
                    if new_c5 < best_c5:
                        best_c5, best_h = new_c5, cur_h.copy()
                        if best_c5 < 0.3808:
                            return best_h.tolist(), float(best_c5), n
                else:
                    cur_h[i], cur_h[j] = 1.0, 0.0
                    no_improve += 1
        # end inner hill‑climb
    # end Phase 1

    # Phase 2: continuous sub‑gradient descent from the best binary vector
    if time.time() < deadline - 5.0:
        cur_h = best_h.astype(float).copy()
        cur_c5 = best_c5
        total_time = deadline - start_time
        while time.time() < deadline - 1.0:
            # worst lag and its gradient
            corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
            max_idx = int(np.argmax(corr))
            lag = max_idx - (n - 1)

            g = np.ones(n, dtype=float)
            if lag >= 0:
                i_f = np.arange(0, n - lag)
                j_f = i_f + lag
                g[i_f] -= cur_h[j_f]
                g[j_f] -= cur_h[i_f]
            else:
                lag_abs = -lag
                i_f = np.arange(lag_abs, n)
                j_f = i_f + lag
                g[i_f] -= cur_h[j_f]
                g[j_f] -= cur_h[i_f]

            # step size shrinks linearly with remaining time
            remaining = deadline - time.time()
            step = 0.2 * (remaining / total_time)
            new_h = cur_h - step * g
            new_h = _project_sum_box(new_h, target_sum)

            new_c5 = _compute_c5(new_h)
            if new_c5 < cur_c5 - 1e-12:
                cur_h, cur_c5 = new_h, new_c5
                if new_c5 < best_c5:
                    best_c5, best_h = new_c5, new_h.copy()
                    if best_c5 < 0.3808:
                        break
        # end Phase 2

    # Final output
    return best_h.tolist(), float(best_c5), n
