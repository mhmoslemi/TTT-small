import numpy as np
import time

def compute_corr_and_shift(h):
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (len(h) - 1)
    max_corr = corr[idx]
    return corr, shift, max_corr

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def gradient_for_shift(h, shift):
    n = len(h)
    g = np.ones(n)
    if shift >= 0:
        i = np.arange(0, n - shift)
        g[i] -= h[i + shift]
        if shift > 0:
            i2 = np.arange(shift, n)
            g[i2] -= h[i2 - shift]
    else:
        s = -shift
        i = np.arange(s, n)
        g[i] -= h[i + shift]   # h[i - s]
        i2 = np.arange(0, n - s)
        g[i2] -= h[i2 - shift] # h[i + s]
    return g

def project_simple(x, target):
    # Clip to [0,1] then adjust sum to target
    x = np.clip(x, 0.0, 1.0)
    diff = x.sum() - target
    # simple iterative correction
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        # Determine indices that can be adjusted
        if diff > 0:
            mask = x > 0.0  # can reduce
        else:
            mask = x < 1.0  # can increase
        nfree = np.count_nonzero(mask)
        if nfree == 0:
            break
        adjust = diff / nfree
        x[mask] -= adjust
        x = np.clip(x, 0.0, 1.0)
        diff = x.sum() - target
    return x

def optimize_h(initial_h, max_iter, time_limit):
    h = initial_h.copy()
    n = len(h)
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    step = 0.5
    start = time.time()
    # main loop
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        # compute shift with max correlation
        _, shift, _ = compute_corr_and_shift(h)
        # gradient

        # proposal

        new_c5 = compute_c5(h_new)

            h = np.random.rand(n)
            h = project_simple(h, target_sum)
            cur_c5 = compute_c5(h)
            step = 0.5
    return best_h, best_c5

import time
import numpy as np

# ---------- helper functions ----------

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx for a discretised h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def max_shift_and_corr(h):
    """Return (shift, max_corr) where shift is the integer offset (‑(n‑1)…n‑1)
    giving the maximal correlation, and max_corr is the unscaled maximum."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)
    return shift, float(corr[idx])


def gradient_for_shift(h, shift):
    """Sub‑gradient of the correlation for a given shift."""
    n = h.shape[0]
    g = np.ones(n, dtype=np.float64)

    if shift == 0:
        g -= 2.0 * h
    elif shift > 0:
        # term h[i+shift]
        i = np.arange(0, n - shift)
        g[i] -= h[i + shift]
        # term h[i-shift]
        i2 = np.arange(shift, n)
        g[i2] -= h[i2 - shift]
    else:  # shift < 0
        s = -shift
        # term h[i+shift] = h[i-s]
        i = np.arange(s, n)
        g[i] -= h[i - s]
        # term h[i-shift] = h[i+s]
        i2 = np.arange(0, n - s)
        g[i2] -= h[i2 + s]
    return g


def project_onto_box_and_sum(x, target):
    """Project x onto {0 ≤ z ≤ 1, Σz = target}."""
    x = np.clip(x, 0.0, 1.0)
    diff = x.sum() - target

    # iterative equal‑share correction
    for _ in range(12):
        if abs(diff) < 1e-12:
            break
        if diff > 0.0:
            mask = x > 0.0
        else:
            mask = x < 1.0
        nfree = np.count_nonzero(mask)
        if nfree == 0:
            break
        adjust = diff / nfree
        x[mask] -= adjust
        x = np.clip(x, 0.0, 1.0)
        diff = x.sum() - target

    # final tiny tweak (if necessary)
    residual = target - x.sum()
    if abs(residual) > 1e-12:
        free = np.where((x > 1e-12) & (x < 1.0 - 1e-12))[0]
        if free.size == 0:
            free = np.arange(len(x))
        x[free[0]] = np.clip(x[free[0]] + residual, 0.0, 1.0)
    return x


# ---------- optimisation core ----------

def optimise_h(initial_h, time_limit):
    """Gradient‑descent with random restarts, returns best (h, C5)."""
    n = initial_h.shape[0]
    target_sum = n / 2.0
    h = project_onto_box_and_sum(initial_h, target_sum)

    best_h = h.copy()
    best_c5 = compute_c5(h)

    step = 0.5
    start = time.time()

    while time.time() - start < time_limit:
        # dominant shift
        shift, _ = max_shift_and_corr(h)

        # gradient descent step
        grad = gradient_for_shift(h, shift)
        h_candidate = project_onto_box_and_sum(h - step * grad, target_sum)
        cand_c5 = compute_c5(h_candidate)

        if cand_c5 < best_c5 - 1e-12:
            # accept improvement
            h = h_candidate
            best_h = h_candidate.copy()
            best_c5 = cand_c5
            step = min(step * 1.03, 2.0)
        else:
            # reject or stay
            step *= 0.5

        # restart if step shrinks too much
        if step < 1e-9:
            h = np.random.rand(n)
            h = project_onto_box_and_sum(h, target_sum)
            step = 0.5

    return best_h, best_c5
