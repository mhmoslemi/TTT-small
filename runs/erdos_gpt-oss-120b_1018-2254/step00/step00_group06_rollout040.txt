import time
import math
import numpy as np

def compute_c5(h):
    """Return the C₅ value for a given vector h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def gradient_of_correlation(h, lag):
    """Gradient of the correlation at a specific lag (the lag that attains max)."""
    n = h.shape[0]
    g = np.ones(n, dtype=np.float64)
    if lag >= 0:
        if lag < n:
            # subtract h[i+lag] for i where i+lag exists
            g[: n - lag] -= h[lag:]
            # subtract h[i-lag] for i where i-lag exists
            g[lag:] -= h[: n - lag]
    else:  # lag < 0
        l = -lag
        if l < n:
            g[l:] -= h[: n - l]       # i+lag = i-l
            g[: n - l] -= h[l:]       # i-lag = i+l
    return g

def project_feasible(v, target_sum):
    """
    Euclidean projection of vector v onto the box [0,1]^n intersected with
    the hyperplane sum(v) = target_sum.
    """
    w = np.clip(v, 0.0, 1.0)
    # Simple water‑filling loop; n is tiny, so a few passes suffice.
    for _ in range(50):
        s = w.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = w < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            inc = diff / cnt
            w[mask] += inc
            w = np.clip(w, 0.0, 1.0)
        else:
            mask = w > 0.0
            cnt = mask.sum()
            if cnt == 0:
                break
            dec = (-diff) / cnt
            w[mask] -= dec
            w = np.clip(w, 0.0, 1.0)
    # Tiny residual fix (should be very small)
    residual = target_sum - w.sum()
    if abs(residual) > 1e-12:
        for i in range(w.shape[0]):
            if 0.0 <= w[i] + residual <= 1.0:
                w[i] += residual
                break
    return w

def random_feasible(n):
    """Generate a random feasible vector (0/1 with a possible 0.5 entry)."""
    target = n / 2.0
    floor = int(math.floor(target))
    h = np.zeros(n, dtype=np.float64)
    if floor > 0:
        idx = np.random.choice(n, floor, replace=False)
        h[idx] = 1.0
    remainder = target - floor
    if remainder > 1e-12:
        remaining = np.setdiff1d(np.arange(n), idx) if floor > 0 else np.arange(n)
        i = np.random.choice(remaining)
        h[i] = remainder
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize the step function for the Erdős C₅ problem.

    Returns
    -------
    h_values : np.ndarray
        The best found vector h (length n_points).
    c5_bound : float
        Corresponding C₅ value.
    n_points : int
        Number of discretization points.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------
    # 1. Determine problem size and initial feasible vector.
    # ------------------------------------------------------------
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if init.size == 0:
            init = None
    else:
        init = None

    if init is not None:
        n = init.shape[0]
        target_sum = n / 2.0
        h_cur = project_feasible(init, target_sum)
    else:
        # default size used in the description when no parent is provided
        n = 95
        target_sum = n / 2.0
        h_cur = random_feasible(n)

    # ------------------------------------------------------------
    # 2. Helper values
    # ------------------------------------------------------------
    dx = 2.0 / n
    best_h = h_cur.copy()
    best_score = compute_c5(best_h)

    # ------------------------------------------------------------
    # 3. Optimization loop (sub‑gradient descent + occasional restart)
    # ------------------------------------------------------------
    start_time = time.time()
    step_initial = 0.30   # initial step magnitude
    step_min = 0.001      # final step magnitude
    max_no_improve = 5000     # steps before a stochastic reset
    no_improve_cnt = 0
    iteration = 0

    # function to obtain current C5 and lag of worst overlap
    def current_state(vec):
        corr = np.correlate(vec, 1.0 - vec, mode="full")
        max_idx = int(np.argmax(corr))
        lag = max_idx - (n - 1)
        return corr[max_idx] * dx, lag, corr

    # Main loop
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 0.05:
            break

        # Current C5 and lag
        cur_score, cur_lag, _ = current_state(h_cur)

        # Sub‑gradient step
        grad = gradient_of_correlation(h_cur, cur_lag)
        step = max(step_initial * (1.0 - elapsed / budget_s), step_min)
        h_candidate = h_cur - step * grad
        h_candidate = project_feasible(h_candidate, target_sum)

        cand_score = compute_c5(h_candidate)

        # Accept if improvement
        if cand_score < cur_score - 1e-12:
            h_cur = h_candidate
            no_improve_cnt = 0
            cur_score = cand_score
            if cand_score < best_score - 1e-12:
                best_h = h_candidate.copy()
                best_score = cand_score
        else:
            no_improve_cnt += 1

        # Stochastic restart if stuck
        if no_improve_cnt >= max_no_improve:
            # perturb the current best a little and re‑project
            perturb = best_h + np.random.normal(scale=0.05, size=n)
            h_cur = project_feasible(perturb, target_sum)
            no_improve_cnt = 0

        # Early exit if we already beat the target bound
        if best_score <= 0.3808:
            break

        iteration += 1

    # Final safety projection (ensures constraints exactly)
    best_h = project_feasible(best_h, n / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, n
