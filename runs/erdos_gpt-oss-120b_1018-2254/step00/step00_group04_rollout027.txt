def repair_h(h):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    target = len(h) / 2.0
    if abs(total - target) < 1e-10:
        return h
    # compute scaling factor
    factor = target / total
    h = h * factor
    # after scaling may violate bounds due to scaling up values >1. Clip again repeatedly until sum matches and state within bounds.
    for _ in range(10):
        h = np.clip(h, 0.0, 1.0)
        total = np.sum(h)
        if abs(total - target) < 1e-10:
            break
        factor = target / total
        h = h * factor
    # final small correction for any residual error
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # add diff to a random index that is not at bounds
        # find all indices where adding diff would stay in bounds
        indices = np.where((h + diff >= 0) & (h + diff <= 1))[0]
        if len(indices) == 0:
            # fallback: adjust first index within bound (set to bound)
            # adjust by distributing diff across many indices
            # simply add diff to the element with largest capacity.
            capacities = np.where(diff > 0, 1 - h, h)  # capacity for positive diff is up to 1, for negative diff is up to h 
            idx = np.argmax(capacities)
        else:
            idx = random.choice(indices)
        h[idx] += diff
    return h

import numpy as np
import random
import time
import math

# ----------------------------------------------------------------------
# Helper: enforce 0≤h_i≤1 and sum(h)=n/2 exactly
def repair_h(h):
    h = np.clip(h, 0.0, 1.0)
    n = h.shape[0]
    target = n / 2.0
    total = h.sum()
    if abs(total - target) < 1e-12:
        return h
    # simple scaling loop – keeps values inside [0,1]
    for _ in range(10):
        factor = target / total
        h = h * factor
        h = np.clip(h, 0.0, 1.0)
        total = h.sum()
        if abs(total - target) < 1e-12:
            break
    # tiny residual correction
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # find a index that can absorb the diff without breaking bounds
        if diff > 0:
            caps = 1.0 - h
        else:
            caps = h
        idxs = np.where(caps > 1e-12)[0]
        if idxs.size == 0:
            idx = np.argmax(caps)  # safest fallback
        else:
            idx = random.choice(idxs.tolist())
        h[idx] += diff
    return h

# ----------------------------------------------------------------------
# Compute C5 for a given h (max correlation with 1‑h)
def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx

# ----------------------------------------------------------------------

    start = time.time()
