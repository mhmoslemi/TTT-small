import time
import numpy as np
from scipy.optimize import minimize

def _c5_objective(h):
    """Compute the overlap constant C5 for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _project_onto_capped_simplex(x, target, max_iter=100, tol=1e-12):
    """
    Euclidean projection of x onto the set { y | 0<=y_i<=1, sum(y)=target }.
    Uses a binary search on the Lagrange multiplier λ for the shift x-λ.
    """
    x = np.asarray(x, dtype=float)
    lower = -np.max(x) - 1.0      # sum at lower bound == n (all ones)
    upper =  np.max(x) + 1.0      # sum at upper bound == 0 (all zeros)

    for _ in range(max_iter):
        lam = (lower + upper) / 2.0
        y = np.clip(x - lam, 0.0, 1.0)
        s = y.sum()
        if abs(s - target) <= tol:
            return y
        if s > target:
            lower = lam
        else:
            upper = lam
    # final interpolation
    lam = (lower + upper) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def _random_feasible(n, target, rng):
    """Generate a random vector in [0,1]^n with prescribed sum."""
    x = rng.random(n)
    return _project_onto_capped_simplex(x, target)

def run(seed=42, budget_s=1100, **kwargs):
    """Return a high‑quality step function for the Erdős C₅ problem.

    Returns
    -------
    h_values : np.ndarray
        The optimized values of h on the uniform grid.
    c5_bound : float
        The achieved upper bound C₅ = max_k ∫ h(x)(1−h(x+k))dx.
    n_points : int
        Number of discretisation points (length of h_values).
    """
    rng = np.random.default_rng(seed)

    # -------------------------------------------------
    # Initialise h
    # -------------------------------------------------
    try:
        init = np.asarray(initial_h_values, dtype=float)  # provided by harness
        n = init.size
        target_sum = n / 2.0
        h0 = _project_onto_capped_simplex(init, target_sum)
    except NameError:
        # No parent construction – start from a random feasible vector
        n = 80
        target_sum = n / 2.0
        h0 = _random_feasible(n, target_sum, rng)

    best_h = h0.copy()
    best_c5 = _c5_objective(best_h)

    # -------------------------------------------------
    # Optimisation settings
    # -------------------------------------------------
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},)
    bounds = [(0.0, 1.0)] * n
    maxiter_opt = 150   # SLSQP internal iterations per restart

    start_time = time.time()
    it = 0

    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 2.0:   # keep a safe margin
            break

        # Choose a start point
        if it == 0:
            x0 = best_h
        else:
            if it % 4 == 0:
                # completely fresh random start
                x0 = _random_feasible(n, target_sum, rng)
            else:
                # perturb current best
                x0 = best_h + rng.normal(scale=0.08, size=n)
                x0 = np.clip(x0, 0.0, 1.0)
                x0 = _project_onto_capped_simplex(x0, target_sum)

        # Run a local optimisation (SLSQP handles bounds + equality)
        res = minimize(_c5_objective, x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter_opt, 'ftol': 1e-9, 'disp': False})

        if res.success:

            cand = _project_onto_capped_simplex(cand, target_sum)
            cand_c5 = _c5_objective(cand)

        # occasional pure random evaluation (no local optimisation) – cheap
        if it % 20 == 0:

            rand_c5 = _c5_objective(rand_h)
            if rand_c5 < best_c5:
                best_c5 = rand_c5
                best_h = rand_h

        it += 1

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_onto_capped_simplex(best_h, target_sum)

    best_c5 = _c5_objective(best_h)

    return (best_h, float(best_c5), n)
