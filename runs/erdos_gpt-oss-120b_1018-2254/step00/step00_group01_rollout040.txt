def project_to_sum(v, target):
    # ensures 0 <= v <= 1, sum = target
    # Using water-filling bisection
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

    """Maximum overlap integral for discretized h."""
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

    """Gradient of the overlap for a fixed shift."""
    n = h.shape[0]
    deriv = np.zeros(n)
    if shift >= 0:
        s = shift
        deriv[: n - s] += 1.0 - h[s:]
        deriv[s:] -= h[: n - s]
    else:
        s = -shift
        deriv[s:] += 1.0 - h[: n - s]
        deriv[: n - s] -= h[s:]
    return deriv

def project_to_sum(v, target):
    """Project v onto {0≤x≤1, sum(x)=target} using bisection."""
    lo, hi = -1.0, 1.0

        x = np.clip(v + mid, 0.0, 1.0)

        else:

    iteration = 0
    T0 = 0.01
    cooling = 0.9995
    target_bound = 0.3808

    while time.time() - start < budget_s * 0.95:
        corr = np.correlate(cur_h, 1 - cur_h, mode="full")
        max_idx = int(np.argmax(corr))
        cur_c5 = float(corr[max_idx] * dx)
        shift = max_idx - (n - 1)

        deriv = derivative_for_shift(cur_h, shift)
        i = int(np.argmax(deriv))
        j = int(np.argmin(deriv))
        guided = False

        if deriv[i] > 0 and deriv[j] < 0:
            max_transfer = min(cur_h[i], 1.0 - cur_h[j])
            if max_transfer > 0:
                temp = T0 * (cooling ** iteration)
                factor = max(0.01, temp / T0)
                t = rng.random() * max_transfer * factor
                if t > 0:
                    new_h = cur_h.copy()
                    new_h[i] -= t
                    new_h[j] += t
                    guided = True

        if not guided:
            i, j = random_pair(rng, n)
            low, high = move_bounds(cur_h, i, j)
            if low >= high:
                iteration += 1
                continue
            t = rng.random() * (high - low) + low
            new_h = cur_h.copy()
            new_h[i] += t
            new_h[j] -= t

        np.clip(new_h, 0.0, 1.0, out=new_h)
        new_h = project_to_sum(new_h, target_sum)

        new_c5 = compute_c5(new_h, dx)
        delta = new_c5 - cur_c5
        temp = T0 * (cooling ** iteration)
        accept = delta < 0 or rng.random() < math.exp(-delta / (temp + 1e-12))
        if accept:
            cur_h = new_h
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
                if best_c5 <= target_bound:
                    break

        iteration += 1

    best_h = project_to_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h, dx)
    return (best_h, float(best_c5), n)
