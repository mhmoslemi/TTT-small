def project_onto_bounds_and_sum(v, target, lower=0.0, upper=1.0, tol=1e-9):
    lo = lower - np.max(v)
    hi = upper - np.min(v)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(v + mid, lower, upper)
        s = proj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    proj = np.clip(v + (lo + hi) / 2.0, lower, upper)
    # Due to rounding, adjust final small error:
    diff = target - proj.sum()
    if abs(diff) > 1e-8:
        # distribute diff across non-bounded entries
        idx = np.where((proj > lower + 1e-12) & (proj < upper - 1e-12))[0]
        if idx.size > 0:
            proj[idx[0]] += diff
            proj[idx[0]] = np.clip(proj[idx[0]], lower, upper)
    return proj

            hi = mid
        else:
            lo = mid

    diff = target - proj.sum()
    if abs(diff) > 1e-8:

        if np.any(free):
            idx = np.where(free)[0][0]
            proj[idx] += diff
            proj[idx] = min(max(proj[idx], lower), upper)
    return proj

import numpy as np
import time
import math
import random

def compute_c5(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)*(1-h(x+k)) dx discretized."""
    n = h.size
    dx = 2.0 / n
    # np.correlate computes sum_i a[i] * b[i+shift] (full mode includes zero padding)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_bounds_and_sum(v: np.ndarray, target: float,
                                lower: float = 0.0, upper: float = 1.0,
                                tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    """Project onto the box [lower,upper] while enforcing sum(v)=target."""
    lo = lower - np.max(v)
    hi = upper - np.min(v)
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        proj = np.clip(v + mid, lower, upper)
        s = proj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    proj = np.clip(v + (lo + hi) / 2.0, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        free = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if np.any(free):
            idx = np.where(free)[0][0]
            proj[idx] = np.clip(proj[idx] + diff, lower, upper)
    return proj

def refine_solution(h: np.ndarray, cur_c5: float, budget: float) -> tuple:
    """Local stochastic refinement respecting constraints."""
    start = time.time()
    n = h.size
    target = n / 2.0
    cur_h = h.copy()
    cur_val = cur_c5
    best_h = cur_h.copy()
    best_val = cur_val
    while time.time() - start < budget:
        elapsed = time.time() - start
        frac = elapsed / budget
        # annealing temperature and max step decay
        T = 0.02 * (1 - frac) + 1e-6
        max_step = 0.2 * (1 - frac) + 1e-4

        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        inc_i = 1.0 - cur_h[i]
        dec_i = cur_h[i]
        inc_j = 1.0 - cur_h[j]
        dec_j = cur_h[j]

        delta_low = max(-dec_i, -inc_j, -max_step)
        delta_high = min(inc_i, dec_j, max_step)
        if delta_low >= delta_high:
            continue
        delta = np.random.uniform(delta_low, delta_high)

        cand_h = cur_h.copy()
        cand_h[i] += delta
        cand_h[j] -= delta

        cand_val = compute_c5(cand_h)
        if cand_val < cur_val or random.random() < math.exp(-(cand_val - cur_val) / max(T, 1e-12)):
            cur_h = cand_h
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_h = cand_h.copy()
                if best_val <= 0.3808:
                    break
    return best_h, best_val
