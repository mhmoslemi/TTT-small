import time
import numpy as np

def compute_c5_general(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_to_sum_bounds(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    diff = target - s
    for _ in range(10):
        free = (h > 0) & (h < 1)
        n_free = free.sum()
        if n_free == 0:
            # adjust a random entry within bounds
            idx = np.random.randint(len(h))
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
            break
        delta = diff / n_free
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
    return h

def random_feasible_h(n, rng):
    h = rng.rand(n)
    target = n / 2.0
    return project_to_sum_bounds(h, target)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    # initial vector
    if 'initial_h_values' in globals():

import time
import numpy as np

def compute_c5_general(h, dx):
    """Maximum overlap integral for a given step function h."""
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_to_sum_bounds(h, target):
    """Clip h to [0,1] and adjust it to satisfy sum(h)=target."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    diff = target - s
    for _ in range(10):
        free = (h > 0) & (h < 1)
        n_free = int(free.sum())
        if n_free == 0:
            # deterministic fallback: adjust the first entry
            idx = 0
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
            break
        delta = diff / n_free
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
    return h

def random_feasible_h(n, rng):
    """Generate a random feasible h with sum(h)=n/2."""
    h = rng.rand(n)
    return project_to_sum_bounds(h, n / 2.0)

    rng = np.random.RandomState(seed)

    # ------------------------------------------------------------------ #
    #   Initial guess

    n = h0.shape[0]
    dx = 2.0 / n
    target_sum = n / 2.0
    h = project_to_sum_bounds(h0, target_sum)

    best_h = h.copy()
    best_c5 = compute_c5_general(best_h, dx)
    cur_h = h.copy()
    cur_c5 = best_c5

    # ------------------------------------------------------------------ #
