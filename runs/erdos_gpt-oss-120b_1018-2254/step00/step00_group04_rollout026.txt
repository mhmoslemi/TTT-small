def project_onto_box_simplex(x, target_sum):
    y = np.clip(x, 0, 1)
    total = y.sum()
    if np.isclose(total, target_sum):
        return y
    # while loop with max iterations maybe 10
    for _ in range(10):
        diff = target_sum - total
        if diff > 0:
            # need to increase some entries
            free = (y < 1)
            if not free.any():
                break
            inc = diff / free.sum()
            y[free] = np.minimum(y[free] + inc, 1)
        else:
            # need to decrease some entries
            free = (y > 0)
            if not free.any():
                break
            dec = -diff / free.sum()
            y[free] = np.maximum(y[free] - dec, 0)
        total = y.sum()
        if abs(total - target_sum) < 1e-9:
            break
    return y

def project_box_sum(x, target_sum, max_iter=10, eps=1e-12):
    # Project x onto [0,1] and sum constraint sum = target_sum
    y = np.clip(x, 0.0, 1.0)
    cur = y.sum()
    if abs(cur - target_sum) <= eps:
        return y
    for _ in range(max_iter):
        diff = target_sum - cur

            # need to increase some entries not at upper bound

import time

def _project_box_sum(x, target_sum, max_iter=10, eps=1e-12):
    """Project onto the box [0,1]^n with exact sum = target_sum."""
    y = np.clip(x, 0.0, 1.0)
    cur = y.sum()
    if abs(cur - target_sum) <= eps:
        return y
    for _ in range(max_iter):
        diff = target_sum - cur
        if diff > 0:
            free = y < 1.0 - eps
            if not np.any(free):
                break
            inc = diff / free.sum()
            y[free] = np.minimum(y[free] + inc, 1.0)
        else:
            free = y > eps
            if not np.any(free):
                break
            dec = -diff / free.sum()
            y[free] = np.maximum(y[free] - dec, 0.0)
        cur = y.sum()
        if abs(cur - target_sum) <= eps:
            break
    diff = target_sum - cur
    if abs(diff) > eps:
        if diff > 0:
            cand = np.where(y < 1.0 - eps)[0]
        else:
            cand = np.where(y > eps)[0]
        if cand.size:
            idx = cand[0]
            y[idx] = np.clip(y[idx] + diff, 0.0, 1.0)
