def compute_c5(vec):
    n = vec.shape[0]
    dx = 2.0 / n
    # correlation compute
    return (np.correlate(vec, 1.0 - vec, mode='full') * dx).max()

    # Determine initial vector

        np.random.shuffle(h)

import numpy as np
import math

def compute_c5(vec: np.ndarray) -> float:
    """Compute the Erdős C5 value for a discrete step function."""
    n = vec.shape[0]
    dx = 2.0 / n
    return (np.correlate(vec, 1.0 - vec, mode="full") * dx).max()

def _fix_sum(vec: np.ndarray) -> np.ndarray:
    """Adjust a vector in [0,1] so that its sum equals len(vec)/2 exactly."""
    n = vec.shape[0]
    target = n / 2.0
    diff = target - vec.sum()
    if abs(diff) < 1e-12:
        return vec
    idxs = np.random.permutation(n)
    for idx in idxs:
        if diff > 0:
            cap = 1.0 - vec[idx]
            move = min(diff, cap)
        else:
            cap = vec[idx]
            move = max(diff, -cap)
        if abs(move) > 1e-12:
            vec[idx] += move
            diff -= move
            if abs(diff) < 1e-12:
                break
    return vec

                new_h = cur_h.copy()
                new_h[i] -= delta
                new_h[j] += delta
        else:
            # --- targeted move on the currently worst offset -------------------------
            corr = np.correlate(cur_h, 1.0 - cur_h, mode="full") * dx
            max_idx = int(np.argmax(corr))
            offset = max_idx - (n - 1)

            # If the worst offset is zero, fall back to a random exchange
            if offset == 0:
                i = np.random.randint(n)
                j = np.random.randint(n - 1)
                if j >= i:
                    j += 1
                max_delta = min(1.0 - cur_h[i], cur_h[j])
                if max_delta <= 0:
                    continue
                delta = max_delta * np.random.rand() * max(0.01, 1.0 / (1 + iteration / 5000))
                new_h = cur_h.copy()
                new_h[i] += delta
                new_h[j] -= delta
            else:
                # Determine admissible index range for the offset
                if offset > 0:
                    i_start, i_end = 0, n - offset - 1
                else:
                    i_start, i_end = -offset, n - 1
                if i_end < i_start:
                    continue
                i0 = np.random.randint(i_start, i_end + 1)
                j = i0 + offset

                # Try moving mass from i0 to j (decrease i0, increase j)
                max_delta = min(cur_h[i0], 1.0 - cur_h[j])
                if max_delta > 0:
                    delta = max_delta * np.random.rand() * max(0.01, 1.0 / (1 + iteration / 5000))
                    new_h = cur_h.copy()
                    new_h[i0] -= delta
                    new_h[j] += delta
                else:
                    # opposite direction
                    max_delta = min(cur_h[j], 1.0 - cur_h[i0])
                    if max_delta <= 0:
                        continue
                    delta = max_delta * np.random.rand() * max(0.01, 1.0 / (1 + iteration / 5000))
                    new_h = cur_h.copy()
                    new_h[i0] += delta
                    new_h[j] -= delta

        # ------------------------------------------------------------------
        # Evaluate candidate
        new_c5 = compute_c5(new_h)

        # Acceptance (greedy + simulated annealing)
        if new_c5 < cur_c5:
            cur_h = new_h
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
        else:
            temp = T0 * max(0.0, (budget_s - elapsed) / budget_s) + Tmin
            if temp > 0.0:
                prob = math.exp(-(new_c5 - cur_c5) / temp)
                if prob > np.random.rand():
                    cur_h = new_h
                    cur_c5 = new_c5

        iteration += 1

    # ----------------------------------------------------------------------
    # Final polishing: exact sum enforcement
    best_h = _fix_sum(best_h)
    final_c5 = compute_c5(best_h)

    return best_h.tolist(), float(final_c5), best_h.shape[0]
