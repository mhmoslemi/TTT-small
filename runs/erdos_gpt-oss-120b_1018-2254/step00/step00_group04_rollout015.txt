def propose_targeted_move(h, shift, rng):
    n = len(h)
    # generate random i such that i and i+shift within bounds
    if shift >= 0:
        i_candidates = np.arange(n - shift)
    else:
        # shift negative
        i_candidates = np.arange(-shift, n)
    if i_candidates.size == 0:
        return None
    i = rng.choice(i_candidates)
    j = i + shift
    # both i and j indices within [0,n)
    # check capacity to increase both
    inc_cap = min(1.0 - h[i], 1.0 - h[j])
    if inc_cap <= 0:
        return None
    # find a donor index k with h[k] > 1e-12
    donor_candidates = np.where(h > 1e-12)[0]
    # exclude i and j
    donor_candidates = donor_candidates[(donor_candidates != i) & (donor_candidates != j)]
    if donor_candidates.size == 0:
        return None
    k = rng.choice(donor_candidates)
    # we need to take 2*delta from k
    max_delta = min(inc_cap, h[k] / 2.0)
    if max_delta <= 0:
        return None
    delta = rng.random() * max_delta
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] += delta
    new_h[k] -= 2.0 * delta
    return new_h
```

def propose_random_move(h, rng):
    n = len(h)
    i, j = rng.integers(0, n, size=2)
    while j == i:
        j = rng.integers(0, n)
    direction = rng.choice([-1, 1])
    if direction == 1:
        max_delta = min(1.0 - h[i], h[j])
    else:
        max_delta = min(h[i], 1.0 - h[j])
    if max_delta <= 0:
        return None
    delta = rng.random() * max_delta
    new_h = h.copy()
    if direction == 1:
        new_h[i] += delta
        new_h[j] -= delta
    else:
        new_h[i] -= delta
        new_h[j] += delta
    return new_h

import time
import math
import random
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1‑h(x+k)) dx for the discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    # np.correlate gives the discrete cross‑correlation (zero‑padded)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _worst_shift(h: np.ndarray):
    """
    Return the integer shift (relative to the centre of the full correlation)
    that yields the current maximal overlap, together with the raw correlation
    array (not scaled by dx).
    """
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)                # centre index corresponds to shift 0
    return shift, corr


def _propose_random_move(h: np.ndarray, rng: np.random.Generator):
    """Generate a feasible neighbour by moving a small amount between two entries."""
    n = h.shape[0]
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n - 1))
    if j >= i:
        j += 1                              # ensure i != j
    # direction = +1 : increase i, decrease j; -1 opposite
    direction = 1 if rng.random() < 0.5 else -1
    if direction == 1:
        max_delta = min(1.0 - h[i], h[j])
    else:
        max_delta = min(h[i], 1.0 - h[j])
    if max_delta <= 0.0:
        return None
    delta = rng.random() * max_delta
    new_h = h.copy()
    if direction == 1:
        new_h[i] += delta
        new_h[j] -= delta
    else:
        new_h[i] -= delta
        new_h[j] += delta
    return new_h


def _propose_targeted_move(h: np.ndarray, shift: int,
                           rng: np.random.Generator):
    """
    Create a move that explicitly tries to increase the inner product
    for the given worst shift (i.e. to lower the maximal overlap).
    The move increases two cooperating entries by the same amount
    and compensates by decreasing another entry.
    """
    if shift == 0:
        return None
    n = h.shape[0]
    # valid i such that i+shift stays inside the array
    if shift > 0:
        i_candidates = np.arange(n - shift)
    else:
        i_candidates = np.arange(-shift, n)
    if i_candidates.size == 0:
        return None
    i = int(rng.choice(i_candidates))
    j = i + shift                     # partner index
    # capacity to raise both without violating the upper bound
    inc_cap = min(1.0 - h[i], 1.0 - h[j])
    if inc_cap <= 0.0:
        return None
    # pick a donor index that can afford to lose mass
    donors = np.where(h > 1e-12)[0]
    donors = donors[(donors != i) & (donors != j)]
    if donors.size == 0:
        return None
    k = int(rng.choice(donors))
    # we need to take 2*delta from k
    max_delta = min(inc_cap, h[k] / 2.0)
    if max_delta <= 0.0:
        return None
    delta = rng.random() * max_delta
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] += delta
    new_h[k] -= 2.0 * delta
    return new_h
