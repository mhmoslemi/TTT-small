def project_to_bounds_and_sum(v, target_sum=0.5):
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target_sum - total
    # distribution among free variables
    for _ in range(100):
        free = (v > 0) & (v < 1)
        free_count = free.sum()
        if free_count == 0 or abs(diff) < 1e-12:
            break
        delta = diff / free_count
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)

import time
import numpy as np

def _project_to_bounds_and_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and then adjust to achieve the exact target sum."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    # simple iterative redistribution
    for _ in range(100):
        if abs(diff) < 1e-12:
            break
        free = (v > 0) & (v < 1)
        cnt = free.sum()
        if cnt == 0:
            break
        delta = diff / cnt
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
        diff = target - v.sum()
    return v

def _c5_of(h: np.ndarray) -> float:
    """Return the C5 value for a step function sampled by h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)

def _random_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a binary (0/1) vector of length n with exactly n/2 ones."""
    v = np.zeros(n, dtype=np.float64)
    ones = rng.choice(n, size=n // 2, replace=False)
    v[ones] = 1.0
    return v
