    corr = np.correlate(h, 1-h, mode='full')
    # find index of max correlation: index offset = k + (n-1)
    idx_max = np.argmax(corr)
    k = idx_max - (len(h) - 1)  # shift can be negative
    n = len(h)
    g = np.zeros_like(h)
    # for j in range(n):
    # compute contributions
    # term1: (1 - h[j+k]) if j+k in [0,n-1]
    # term2: -h[j-k] if j-k in [0,n-1]
    if k >= 0:
        # term1 for j upto n-1-k
        term1_indices = np.arange(0, n - k)
        g[term1_indices] += (1 - h[term1_indices + k])
        # term2 for j >= k
        term2_indices = np.arange(k, n)

def proj_capped_simplex(v, N, max_iter=100, tol=1e-9):
    # Clip to [0,1]
    u = np.clip(v, 0, 1)
    total = u.sum()
    if abs(total - N) <= tol:
        return u
    # Determine bounds for lambda: lambda low can be -1 (so u_i - lambda up to >1) but after clip to 1.
    # But we can set low = -1, high = 1 (since u_i in [0,1]; subtract lambda > 1 pushes down to <=0).
    low = -1.0
    high = 1.0
    # adjust high so sum <= N at high: for very large lambda, u_i - lambda <=0 => sum =0 < N.
    # For low -1, u_i - (-1) = u_i+1 <=2 => clip to 1 => sum = n (since each term becomes min(1, u_i+1) = 1). So sum = n >= N.
    # So lambda in [-1,1] works.
    for _ in range(max_iter):
        lam = (low + high) / 2.0
        x = np.minimum(1, np.maximum(0, u - lam))
        s = x.sum()
        if abs(s - N) <= tol:
            return x
        if s > N:
            low = lam
        else:
            high = lam
    # Return final
    return np.minimum(1, np.maximum(0, u - (low+high)/2))

import numpy as np
import random
import time

def compute_c5(h):
    """Return max_k ∫ h(x)(1‑h(x+k)) dx for a discrete step function."""
    h = np.asarray(h, dtype=float)
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_onto_capped_simplex(v, N, tol=1e-10, max_iter=60):
    """
    Euclidean projection of vector v onto the set
        {x | 0 ≤ x_i ≤ 1, sum x_i = N }.
    """
    u = np.clip(v, 0.0, 1.0)
    if abs(u.sum() - N) <= tol:
        return u
    lo, hi = -1.0, 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.minimum(1.0, np.maximum(0.0, u - lam))
        s = x.sum()
        if abs(s - N) <= tol:
            return x
        if s > N:
            lo = lam
        else:
            hi = lam
    return np.minimum(1.0, np.maximum(0.0, u - (lo + hi) / 2.0))

def hill_climb(start_h, deadline):
    """Greedy binary hill‑climbing with random restarts."""
    n = start_h.shape[0]
    N = n // 2
    best = start_h.copy()
    best_val = compute_c5(best)
    current = best.copy()
    cur_val = best_val
    no_improve = 0
    max_no_improve = 2000

    while time.time() < deadline:
        ones = np.where(current == 1)[0]
        zeros = np.where(current == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        cand = current.copy()
        cand[i] = 0
        cand[j] = 1
        val = compute_c5(cand)
        if val < cur_val:
            current, cur_val = cand, val
            if val < best_val:
                best, best_val = cand.copy(), val
                no_improve = 0
            else:
                no_improve = 0

            if no_improve >= max_no_improve:

                    current = np.zeros(n, dtype=int)
                    current[idx] = 1
                else:
                    current = best.copy()
                cur_val = compute_c5(current)
                no_improve = 0
