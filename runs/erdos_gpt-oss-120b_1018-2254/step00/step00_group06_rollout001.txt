    n = len(h)

def _compute_c5(h: np.ndarray) -> float:
    """Return the C5 bound for a given step function sampled on [0,2]."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with 1-h, mode='full' gives all integer shifts
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project_onto_feasible(y: np.ndarray, target_sum: float) -> np.ndarray:
    """Project y onto the convex set {0<=x<=1, sum(x)=target_sum}."""
