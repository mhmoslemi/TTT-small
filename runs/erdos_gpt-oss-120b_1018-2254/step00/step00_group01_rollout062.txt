def project_to_feasible(v, target_sum):
    # Ensure bounds:
    v = np.clip(v, 0.0, 1.0)
    # Iteratively adjust to match target sum:
    # compute sum
    s = v.sum()
    # difference
    diff = target_sum - s
    # tolerance
    tol = 1e-12
    max_iter = 10
    for _ in range(max_iter):
        if abs(diff) < tol:
            break
        # find interior indices where moving does not violate bounds
        # For diff>0, we need to increase some entries; available capacity is sum(1 - v_i) over interior?
        interior = np.where((v > 0) & (v < 1))[0]
        if len(interior) == 0:
            # no interior; adjust by distributing diff proportionally among all with capacity
            # For diff>0, add to those <1; for diff<0, subtract from those >0
            if diff > 0:
                idxs = np.where(v < 1)[0]
                caps = 1 - v[idxs]
                total_cap = caps.sum()
                if total_cap <= 0:
                    break
                delta = diff * caps / total_cap  # distribute proportionally
                v[idxs] += delta
                v = np.clip(v, 0, 1)
            else:
                idxs = np.where(v > 0)[0]
                caps = v[idxs]
                total_cap = caps.sum()
                if total_cap <= 0:

                delta = diff * caps / total_cap
                v[idxs] += delta
                v = np.clip(v, 0, 1)

            # Uniform distribution across interior
            delta = diff / len(interior)
            v[interior] += delta
            v = np.clip(v, 0, 1)

def enforce_sum_and_bounds(v, target_sum):

def enforce_sum_and_bounds(v, target):
    v = np.clip(v, 0.0, 1.0)
    # enforce sum using water-filling algorithm.
    # Compute excess = target - current sum.
    # While error > tolerance:
    #   Find indices that are not at bound (0 < v_i < 1). If none, break.
    #   Adjust equally: delta = excess / len(free)
    #   v[free] += delta
    #   Clip again (some may hit bounds), loop.
    total = v.sum()
    diff = target - total
    tol = 1e-10
    max_iter = 10
    for _ in range(max_iter):
        if abs(diff) < tol:
            break
        # free indices where movement possible
        if diff > 0:
            free = np.where(v < 1 - tol)[0]

import numpy as np
import time
import math
import random

def compute_c5(h: np.ndarray, dx: float) -> float:
    """Return the overlap bound C5 for a step function h."""
    # correlation of h with (1 - h) over full range
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def enforce_sum_and_bounds(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto the set {0 ≤ v_i ≤ 1, sum(v) = target_sum}."""
    v = np.clip(v, 0.0, 1.0)
    tol = 1e-12
    max_iter = 10
    for _ in range(max_iter):
        diff = target_sum - v.sum()
        if abs(diff) < tol:
            break
        # indices that can still move in the needed direction
        if diff > 0:
            free = np.where(v < 1 - tol)[0]
        else:
            free = np.where(v > tol)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    # final tiny correction if needed
    diff = target_sum - v.sum()
    if abs(diff) > tol:
        free = np.where((v > tol) & (v < 1 - tol))[0]
        if free.size > 0:
            idx = free[0]
            v[idx] = np.clip(v[idx] + diff, 0.0, 1.0)
        else:
            # distribute proportionally to capacity
            pos = np.where(v < 1 - tol)[0]
            neg = np.where(v > tol)[0]
            if diff > 0 and pos.size:
                caps = 1 - v[pos]
                v[pos] += diff * caps / caps.sum()
            elif diff < 0 and neg.size:
                caps = v[neg]
                v[neg] += diff * caps / caps.sum()
        v = np.clip(v, 0.0, 1.0)
    return v

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    # set RNG
    np.random.seed(seed)
    random.seed(seed)

    # Determine problem size and initial vector
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 47
        # start from a simple feasible vector (average 0.5)
        h0 = np.full(n, 0.5, dtype=float)

    target_sum = n / 2.0          # sum constraint (since dx = 2/n)
    dx = 2.0 / n

    # ensure feasibility of the start
    h0 = enforce_sum_and_bounds(h0, target_sum)

    cur_h = h0.copy()
    cur_c5 = compute_c5(cur_h, dx)

    best_h = cur_h.copy()
    best_c5 = cur_c5

    start_time = time.time()
    iteration = 0
    last_improve = 0
    max_no_improve = 20000               # restart after this many iterations without improvement
    base_alpha = 0.02                     # initial step size for gradient moves
    T0 = 0.01                             # initial temperature for simulated annealing

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 0.5:
            break
        iteration += 1

        # ---- gradient step targeting the shift with smallest autocorrelation ----
        corr_self = np.correlate(cur_h, cur_h, mode="full")
        center = n - 1
        # ignore zero shift when searching for the worst (minimal) autocorrelation
        temp = corr_self.copy()
        temp[center] = np.inf
        min_idx = int(np.argmin(temp))
        shift = min_idx - center
        s_abs = abs(shift)

        if s_abs == 0:
            grad = 2.0 * cur_h
        else:
            left = np.concatenate((np.zeros(s_abs), cur_h[:-s_abs]))
            right = np.concatenate((cur_h[s_abs:], np.zeros(s_abs)))
            grad = left + right

        # decreasing step size
        alpha = base_alpha * (1.0 - elapsed / budget_s)
        cand = cur_h + alpha * grad
        cand = enforce_sum_and_bounds(cand, target_sum)

        cand_c5 = compute_c5(cand, dx)

        # ---- acceptance (simulated annealing) ----
        temp_sched = T0 * (1.0 - elapsed / budget_s)
        accept = False
        if cand_c5 < cur_c5:
            accept = True
        else:
            if temp_sched > 1e-12:
                prob = math.exp(-(cand_c5 - cur_c5) / temp_sched)
                if random.random() < prob:
                    accept = True
        if accept:
            cur_h = cand
            cur_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = cand.copy()
                last_improve = iteration

        # ---- occasional random pairwise perturbation to diversify search ----
        if iteration % 500 == 0:
            i, j = np.random.choice(n, size=2, replace=False)
            max_inc = min(0.1, 1.0 - cur_h[i])
            max_dec = min(0.1, cur_h[j])
            max_delta = min(max_inc, max_dec)
            if max_delta > 1e-12:
                delta = random.random() * max_delta
                # increase i, decrease j

                cand2[i] += delta
                cand2[j] -= delta
                cand2 = enforce_sum_and_bounds(cand2, target_sum)

            cur_h = best_h + noise
            cur_h = enforce_sum_and_bounds(cur_h, target_sum)
            cur_c5 = compute_c5(cur_h, dx)
            last_improve = iteration

    # final polishing
    best_h = enforce_sum_and_bounds(best_h, target_sum)
    best_c5 = compute_c5(best_h, dx)

    return (best_h, best_c5, n)
