import time, math
import numpy as np

def _project_to_box_sum(v, target):
    """Project vector v onto [0,1]^n intersect sum(v)=target using a simple bisection."""
    lo, hi = -2.0, 2.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def _compute_c5(h):
    """Return the C5 value for a step function sampled by array h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def _initial_h(n):
    """Construct an initial feasible h of length n, using the provided parent if possible."""
    if "initial_h_values" in globals():
        old = np.asarray(initial_h_values, dtype=float)
        m = old.size
        if m != n:
            old_x = np.arange(m) * (2.0 / m)
            new_x = np.arange(n) * (2.0 / n)
            new = np.interp(new_x, old_x, old, left=old[0], right=old[-1])
        else:
            new = old.copy()
        return _project_to_box_sum(new, n / 2.0)
    # fallback: uniform 0.5
    return np.full(n, 0.5, dtype=float)
