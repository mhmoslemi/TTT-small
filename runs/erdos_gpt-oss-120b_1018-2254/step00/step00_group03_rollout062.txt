    # clip
    x = np.clip(x, 0, 1)
    # adjust sum via scaling of interior
    # compute deficit/excess
    deficit = target - np.sum(x)
    if abs(deficit) < 1e-12:
        return x
    # Identify indices not at bounds.
    free = (x > 0) & (x < 1)
    if np.any(free):
        # Add equal share to free entries, but ensure they stay within bounds
        # iterative approach:
        while True:
            free_indices = (x > 0) & (x < 1)
            n_free = np.sum(free_indices)
            if n_free == 0:
                break
            add = deficit / n_free
            x[free_indices] += add
            # handle clipping again
            x = np.clip(x, 0, 1)
            new_deficit = target - np.sum(x)
            if abs(new_deficit - deficit) < 1e-12:
                break
            deficit = new_deficit
        # if still deficit not zero due to rounding, adjust by small random redistributions

        # all are at bounds, adjust uniformly maybe by small random.
        # fallback: assign random feasible vector.

import time
import numpy as np

# The harness provides `initial_h_values` and `evaluate_erdos_solution`.
# `evaluate_erdos_solution` returns the C₅ value for a given vector.

def _project_to_constraints(x, target):
    """
    Project vector x onto the box [0,1]^n while forcing the sum to be `target`.
    Uses a simple water‑filling (binary search on a shift) because the sum
    after clipping is monotone in the shift.
    """
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)
    return y


def _c5_circular(h):
    """
    Circular version of the overlap integral:
    max_k sum_i h[i] * (1 - h[(i+k) % n]) * dx,
    where dx = 2 / n.
    """
    n = h.shape[0]
    fh = np.fft.fft(h)
    f1mh = np.fft.fft(1.0 - h)
    corr = np.real(np.fft.ifft(fh * np.conj(f1mh)))
    return np.max(corr) * (2.0 / n)


def _initial_vector(n, init_h):
    """
    Build a starting vector of length n.
    If `init_h` (the global parent) is available, interpolate it;
    otherwise use a smooth sin² shape which already has the correct
    integral (because its average is 0.5).
    """
    if init_h is not None:
        # Upsample / downsample the parent to length n by linear interpolation.
        m = init_h.shape[0]
        if m == n:
            h = init_h.astype(float).copy()
        else:
            x_old = np.linspace(0.0, 2.0, m, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h = np.interp(x_new, x_old, init_h)
    else:
        # sin² shape: average 0.5, values in [0,1].
        x = np.arange(n) * (2.0 / n)
        h = np.sin(np.pi * x / 2.0) ** 2
    # Clip and project to exact sum = n/2.
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    if abs(h.sum() - target) > 1e-12:
        h = _project_to_constraints(h, target)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] with n points
    to minimise the Erdős overlap constant C₅.
    Returns (list_of_h, C5_value, n_points).
    """
    np.random.seed(seed)

    # Global optional parent construction.
    try:
        parent = np.asarray(initial_h_values, dtype=float)
    except NameError:
        parent = None

    # Choose a modest size (≤1000) – 200 gives good resolution.
    n = 200

    # Start from the parent if it exists, otherwise a smooth shape.
    h = _initial_vector(n, parent)

    # Best solution so far.
    best_h = h.copy()
    best_c = _c5_circular(best_h)

    # Timing.
    start = time.time()
    limit = budget_s * 0.95  # safety margin

    # Hyper‑parameters.
    eps = 0.03                # maximal mass moved per step
    no_improve = 0
    restart_patience = 2000   # iterations without improvement before a restart

    while time.time() - start < limit:
        # Circular correlation for all shifts.
        fh = np.fft.fft(h)
        f1mh = np.fft.fft(1.0 - h)
        corr = np.real(np.fft.ifft(fh * np.conj(f1mh)))
        max_idx = int(np.argmax(corr))                # shift giving worst C₅
        # Gradient of the worst shift expression.
        h_plus = np.roll(h, -max_idx)
        h_minus = np.roll(h, max_idx)
        grad = 1.0 - (h_plus + h_minus)                # positive -> h should decrease

        # Candidates for decreasing and increasing.
        dec_idx = np.where((grad > 1e-12) & (h > 1e-12))[0]
        inc_idx = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]

        moved = False
        if dec_idx.size and inc_idx.size:
            i = np.random.choice(dec_idx)
            j = np.random.choice(inc_idx)
            max_delta = min(eps, h[i], 1.0 - h[j])
            if max_delta > 1e-12:
                delta = max_delta * np.random.rand()
                h_new = h.copy()
                h_new[i] -= delta
                h_new[j] += delta
                c_new = _c5_circular(h_new)
                if c_new < best_c - 1e-12:
                    best_h, best_c, h = h_new, c_new, h_new
                    no_improve = 0
                    moved = True

        if not moved:

            i = np.random.randint(n)
            j = (i + np.random.randint(1, n)) % n
            max_delta = min(eps, h[i], 1.0 - h[j])
            if max_delta > 1e-12:
                delta = max_delta * np.random.rand()
                h_new = h.copy()
                h_new[i] -= delta
                h_new[j] += delta
                c_new = _c5_circular(h_new)

    best_h = np.clip(best_h, 0.0, 1.0)

    target = n / 2.0
    if abs(total - target) > 1e-9:

    # Use the official evaluator for the reported bound.
    c5_bound = float(evaluate_erdos_solution(best_h))

    return best_h.tolist(), c5_bound, n
