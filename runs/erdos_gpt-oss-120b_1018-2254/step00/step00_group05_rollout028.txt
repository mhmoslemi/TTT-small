        h0 = np.clip(h0, 0.0, 1.0)
        s = np.sum(h0)
        if s > 0:

        else:

    def corr_scaled(h):
        return np.correlate(h, 1.0 - h, mode='full') * dx

    # smooth maximum with gradient
    def smooth_max_and_grad(h, beta):
        c = corr_scaled(h)
        maxc = np.max(c)
        shifted = c - maxc               # for numerical stability
        exp_term = np.exp(beta * shifted)
        sum_exp = np.sum(exp_term)
        p = exp_term / sum_exp           # softmax weights over shifts
        smooth_val = maxc + np.log(sum_exp) / beta

        Nloc = h.shape[0]
        grad_raw = np.zeros_like(h)
        # accumulate contributions for each shift
        for s in range(-Nloc + 1, Nloc):
            ps = p[s + Nloc - 1]
            if s >= 0:
                # first factor: h[i] part

                grad_raw[t:] += ps * (1.0 - h[:Nloc - t])
                grad_raw[:Nloc - t] += -ps * h[t:]

        grad = grad_raw * dx

        return np.sum(h) - N / 2.0
    def eq_jac(h):
        return np.ones_like(h)

    betas = [12.0, 30.0, 60.0, 120.0, 240.0]

    # ------------------------------------------------------------------
    if best_h is None:
        best_h = h0.copy()
        best_val = np.max(corr_scaled(best_h))

    # Final projection onto the feasible set
    best_h = np.clip(best_h, 0.0, 1.0)
    total = np.sum(best_h)
    if total != 0:
        best_h = best_h * (N / 2.0) / total

    final_c5 = np.max(corr_scaled(best_h))

    return best_h, final_c5, N
