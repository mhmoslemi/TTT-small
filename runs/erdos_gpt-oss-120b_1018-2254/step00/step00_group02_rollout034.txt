def random_swap_move(h, rng):
    n = len(h)
    i, j = rng.choice(n, size=2, replace=False)
    hi, hj = h[i], h[j]
    # Determine feasible range for hi'
    lo = max(0.0, hi + hj - 1.0)
    hi_ = min(1.0, hi + hj)
    if lo >= hi_:  # No feasible move (i.e., both 0 or both 1)
        return None, None, None
    new_hi = lo + rng.random() * (hi_ - lo)
    new_hj = hi + hj - new_hi
    # If move does not change values maybe skip
    if np.isclose(new_hi, hi) and np.isclose(new_hj, hj):
        return None, None, None
    h_new = h.copy()
    h_new[i] = new_hi
    h_new[j] = new_hj
    return h_new, i, j
```

def hill_climb(h0, rng, max_iters, no_improve_limit):
    best_h = h0.copy()
    best_val = compute_c5(best_h)
    no_improve = 0
    for it in range(max_iters):
        # generate move
        i, j = rng.integers(0, len(h0), size=2, endpoint=False)
        if i == j: continue
        hi, hj = best_h[i], best_h[j]
        lo = max(0, hi + hj - 1)
        hi_up = min(1, hi + hj)
        if lo >= hi_up: continue
        new_hi = lo + rng.random() * (hi_up - lo)
        new_hj = hi + hj - new_hi
        # optionally continue only if difference > 1e-9
        if abs(new_hi - hi) < 1e-12: continue

        # Create new h candidate
        cand = best_h.copy()
        cand[i] = new_hi
        cand[j] = new_hj

        val = compute_c5(cand)
        if val < best_val - 1e-12:
            best_h = cand
            best_val = val
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= no_improve_limit:
            break
    return best_h, best_val
