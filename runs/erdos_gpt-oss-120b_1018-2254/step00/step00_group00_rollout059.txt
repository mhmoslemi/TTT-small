def project_box_sum(x, lower, upper, target_sum, tol=1e-8, max_iter=100):
    # Find λ such that sum(clamp(x+λ, lower, upper)) = target_sum
    # Use binary search.
    # The feasible region ensures λ lower bound = lower - max(x) perhaps.
    # Since lower=0 and upper=1, we can set λ_low = -1.0, λ_high = 1.0.
    # But to ensure coverage, we can expand maybe [-2,2].
    lam_low = -2.0
    lam_high = 2.0
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) / 2.0
        proj = np.clip(x + lam_mid, lower, upper)
        s = proj.sum()
        if abs(s - target_sum) <= tol:
            return proj
        if s < target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    # after loop, return proj at mid
    return np.clip(x + lam_mid, lower, upper)

But we need to ensure monotonic relationship: as λ increases, sum of projection increases because adding to x+λ moves values up, after clipping at lower bound zero, increasing λ eventually adds to those which were at zero but still within bound; after hitting upper bound one, further λ increases doesn't increase sum. So monotonic until saturates at n*1 = n. Our target is n/2 so not saturating.

Thus binary search works.

def project_box_sum(x, lower, upper, target_sum, tol=1e-9, max_iter=100):
    # Ensure x is numpy array
    x = np.asarray(x, dtype=float)
    # lam bounds
    lam_low = lower - np.max(x) - 1.0  # ensure low enough
    lam_high = upper - np.min(x) + 1.0  # ensure high enough

    # Ensure that with lam_low sum <= target, lam_high sum >= target
    # We'll compute sums to guarantee.
    # Actually lam_low can be -2, lam_high = 2 as generic.

    for _ in range(max_iter):

        proj = np.clip(x + lam_mid, lower, upper)

            return proj
        if s < target_sum:

        else:
            lam_high = lam_mid
    # after loops, return proj

import numpy as np
import time
import random

def project_box_sum(x, lower, upper, target_sum, tol=1e-9, max_iter=100):
    """Project vector x onto the intersection of box constraints and a sum constraint."""
    x = np.asarray(x, dtype=float)
    # set lambda bounds
    lam_low = -2.0
    lam_high = 2.0
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) / 2.0
        proj = np.clip(x + lam_mid, lower, upper)
        s = proj.sum()
        if abs(s - target_sum) <= tol:
            return proj
        if s < target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(x + lam_mid, lower, upper)

def random_feasible(n, target_sum):
    """Generate a random vector in [0,1]^n with given sum."""
    x = np.random.rand(n)
    return project_box_sum(x, 0.0, 1.0, target_sum)

def evaluate_overlap(h):
    """Return C5 value for step function h."""
    # h is 1‑D numpy array
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * (2.0 / h.shape[0])

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)

    # Determine problem size from initial_h_values if present
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        target_sum = n / 2.0
        # Ensure feasibility
        h0 = project_box_sum(h0, 0.0, 1.0, target_sum)
    else:
        n = 45
        target_sum = n / 2.0
        h0 = None

    # GA parameters
    pop_size = 50
    elite_size = 2
    tournament_size = 3
    p_crossover = 0.9
    p_mutate = 0.25
    mut_sigma = 0.08  # mutation strength

    # Build initial population
    population = []
    if h0 is not None:
        population.append(h0)
    while len(population) < pop_size:
        population.append(random_feasible(n, target_sum))

    # Evaluate initial population
    scores = [evaluate_overlap(ind) for ind in population]

    best_idx = int(np.argmin(scores))
    best_h = population[best_idx].copy()
    best_score = scores[best_idx]

    start_time = time.time()

    # Helper: tournament selection
    def tournament_select():
        idxs = np.random.choice(pop_size, size=tournament_size, replace=False)
        best_local = idxs[0]
        best_local_score = scores[best_local]
        for i in idxs[1:]:
            if scores[i] < best_local_score:
                best_local = i
                best_local_score = scores[i]
        return population[best_local]

    # Main GA loop
    while True:
        # Time check
        if time.time() - start_time > budget_s * 0.94:
            break

        # Create new population
        new_pop = []

        # Elitism: copy the best elite_size individuals
        elite_idxs = np.argsort(scores)[:elite_size]
        for idx in elite_idxs:
            new_pop.append(population[idx].copy())

        # Fill the rest
        while len(new_pop) < pop_size:
            p1 = tournament_select()
            p2 = tournament_select()

            # Crossover
            if np.random.rand() < p_crossover:
                mask = np.random.rand(n) < 0.5
                child = np.where(mask, p1, p2)
            else:
                child = p1.copy()

            # Mutation
            if np.random.rand() < p_mutate:
                child = child + np.random.randn(n) * mut_sigma

            # Repair to satisfy constraints
            child = project_box_sum(child, 0.0, 1.0, target_sum)

            new_pop.append(child)

        # Replace population
        population = new_pop
        scores = [evaluate_overlap(ind) for ind in population]

        # Update global best
        cur_best_idx = int(np.argmin(scores))
        cur_best_score = scores[cur_best_idx]
        if cur_best_score < best_score:
            best_score = cur_best_score
            best_h = population[cur_best_idx].copy()

    # Local refinement with simple hill climbing (remaining time)
    remaining = budget_s - (time.time() - start_time)
    if remaining > 0.5:
        end_time = time.time() + remaining * 0.9
        while time.time() < end_time:
            # Perturb a small subset of entries
            trial = best_h.copy()
            # Choose between 1 and 5 indices to modify
            k = max(1, int(n * 0.05))
            ids = np.random.choice(n, size=k, replace=False)
            # Add Gaussian noise
            trial[ids] += np.random.randn(k) * (mut_sigma * 0.5)
            trial = project_box_sum(trial, 0.0, 1.0, target_sum)
            val = evaluate_overlap(trial)
            if val < best_score:
                best_score = val
                best_h = trial

    return (best_h, best_score, n)

import numpy as np
import time
import random

def project_box_sum(x, lower, upper, target_sum, tol=1e-9, max_iter=100):
    """Project a vector onto the box [lower,upper] and a fixed L1 sum."""
    x = np.asarray(x, dtype=float)
    lam_low, lam_high = -2.0, 2.0  # bounds guarantee sum below/above target
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) / 2.0
        proj = np.clip(x + lam_mid, lower, upper)

        if abs(s - target_sum) <= tol:
            return proj
        if s < target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(x + lam_mid, lower, upper)
