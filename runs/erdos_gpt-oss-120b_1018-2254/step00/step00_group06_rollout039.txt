import numpy as np
import time
import math

def project_onto_sum_and_bounds(v, target, eps=1e-9, max_iter=200):
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
        if hi - lo < eps:
            break
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-8:
        free = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        if len(free) > 0:
            x[free] += diff / len(free)
            x = np.clip(x, 0.0, 1.0)
        else:
            # distribute across all
            x += diff / len(x)
            x = np.clip(x, 0.0, 1.0)
    return x

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    max_corr = np.max(corr)
    c5 = max_corr * dx
    prod_corr = np.correlate(h, h, mode='full')
    idx = prod_corr.argmin()
    lag = idx - (n - 1)
    return c5, lag

def grad_worst_lag(h, lag):
    n = h.shape[0]
    g = np.zeros_like(h)
    if lag >= 0:
        k = lag
        if k < n:
            g[:n - k] += h[k:]
            g[k:] += h[:n - k]
    else:
        k = -lag
        if k < n:
            g[k:] += h[:n - k]
            g[:n - k] += h[k:]
    return g

    import numpy as np, time, math
    rng = np.random.default_rng(seed)

    init = globals().get('initial_h_values')
    if init is not None:
        h = np.array(init, dtype=np.float64).copy()

    else:

    target = n / 2.0

    h = project_onto_sum_and_bounds(h, target)
    best_h = h.copy()

    iter_count = 0
    last_improve = 0

    # parameters
    alpha_start = 0.4
    alpha_end = 0.001
    T_start = 0.02
    T_end = 0.0001
    # main loop
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        frac = elapsed / budget_s
        alpha = alpha_start * (1 - frac) + alpha_end * frac
        T = T_start * (1 - frac) + T_end * frac

        c5, lag = compute_c5(h)

        # move type
        if rng.random() < 0.7:
            g = grad_worst_lag(h, lag)
            g_norm = np.linalg.norm(g)
            if g_norm > 0:
                g = g / g_norm
            cand = h + alpha * g
            cand = project_onto_sum_and_bounds(cand, target)
        else:
            # random pair swap move
            i, j = rng.choice(n, size=2, replace=False)
            # max amount transferable limited by bounds
            max_transfer = min(alpha, h[i], 1.0 - h[j])
            min_transfer = -min(alpha, h[j], 1.0 - h[i])
            delta = rng.uniform(min_transfer, max_transfer)
            cand = h.copy()

            cand[j] += delta
            # due to delta bounds, cand respects [0,1] and sum unchanged

            best_h = cand.copy()
            last_improve = iter_count

        if cand_c5 < c5:
            h = cand
        else:
            # probability

                accept_prob = math.exp(-(cand_c5 - c5) / T)
                if rng.random() < accept_prob:
                    h = cand
        iter_count += 1

        # occasional random restart if stagnated
        if iter_count - last_improve > 20000:
            # restart with random feasible point
            h = rng.random(n)
            h = project_onto_sum_and_bounds(h, target)
            last_improve = iter_count

    # final refinement: small step gradient ascent
    for _ in range(5000):
        c5, lag = compute_c5(best_h)
        g = grad_worst_lag(best_h, lag)
        g_norm = np.linalg.norm(g)
        if g_norm == 0:
            break
        step = 1e-4 / g_norm
        cand = best_h + step * g
        cand = project_onto_sum_and_bounds(cand, target)
        cand_c5, _ = compute_c5(cand)
        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = cand

import numpy as np
import time
import math

def _project_onto_sum_and_bounds(v, target, eps=1e-9, max_iter=200):
    """Projection of v onto the set {x: 0<=x<=1, sum(x)=target}."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
        if hi - lo < eps:
            break
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-8:
        free = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        if free.size:
            x[free] += diff / free.size
            x = np.clip(x, 0.0, 1.0)
        else:
            x += diff / x.size
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_c5(h):
    """Return C5 value (with dx applied) and the lag attaining the worst overlap."""
    n = h.shape[0]
    # correlation with complement
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    c5 = np.max(corr) * dx

    prod_corr = np.correlate(h, h, mode='full')
    idx = prod_corr.argmin()
    lag = idx - (n - 1)
