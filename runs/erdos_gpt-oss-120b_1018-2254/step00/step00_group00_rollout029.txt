                    h += inc * room
                else:

                if total > 0:

                    h += dec * room
                else:

            return arr
        if diff > 0:
            room = 1 - arr
        else:
            room = arr

import numpy as np
import time

def _compute_c5(arr):
    n = arr.shape[0]
    dx = 2.0 / n
    corr = np.correlate(arr, 1.0 - arr, mode="full")
    return np.max(corr) * dx

def _repair_sum(arr):
    arr = np.clip(arr, 0.0, 1.0)
    n = arr.shape[0]
    target = n / 2.0
    cur = arr.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return arr
    if diff > 0:
        room = 1.0 - arr
    else:
        room = arr
    total_room = room.sum()
    if total_room > 0:
        factor = diff / total_room
        arr += factor * room
    arr = np.clip(arr, 0.0, 1.0)
    # tiny correction if still off
    cur = arr.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        free = (arr > 1e-12) & (arr < 1.0 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            arr[idx] += diff / idx.size
            arr = np.clip(arr, 0.0, 1.0)
    return arr

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # Target resolution
    n_target = 180

    # Initialize from provided construction if possible
    base = globals().get("initial_h_values", None)
    if base is not None:
        base = np.asarray(base, dtype=float)
        n_base = base.shape[0]
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_target = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h = np.interp(x_target, x_base, base)
    else:
        # Random start
        h = rng.random(n_target)
    h = _repair_sum(h)

    n = h.shape[0]
    best_h = h.copy()
    best_score = _compute_c5(best_h)

    max_iters = 10**7
    it = 0
    no_improve = 0

    while time.time() - start < budget_s and it < max_iters:
        it += 1

        # correlation and worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr))
        cur_score = corr[max_idx] * (2.0 / n)

        # keep best
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # restart if stuck
        if no_improve > 20000:
            h = rng.random(n)
            h = _repair_sum(h)
            no_improve = 0
            continue

        shift = max_idx - (n - 1)
        i_start = max(0, -shift)
        i_end = min(n, n - shift)
        if i_end > i_start:
            idxs = np.arange(i_start, i_end)
            contrib = h[idxs] * (1.0 - h[idxs + shift])
            mask = contrib > 1e-12
            if mask.any():
                good = idxs[mask]
                i_sel = rng.choice(good)
                j_sel = i_sel + shift
                max_delta = min(h[i_sel], 1.0 - h[j_sel])
                if max_delta > 1e-12:
                    elapsed = time.time() - start
                    frac = elapsed / budget_s
                    step = max(0.01, 0.5 * (1.0 - frac))
                    delta = rng.uniform(0.0, max_delta) * step
                    h[i_sel] -= delta
                    h[j_sel] += delta
                    continue  # go to next iteration

        # fallback random exchange move
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_inc_i = 1.0 - h[i]
        max_dec_i = h[i]
        max_inc_j = 1.0 - h[j]
        max_dec_j = h[j]
        low = max(-max_dec_i, -max_inc_j)
        high = min(max_inc_i, max_dec_j)
        if low >= high:
            continue
        elapsed = time.time() - start
        frac = elapsed / budget_s
        step = max(0.01, 0.5 * (1.0 - frac))
        delta = rng.uniform(low, high) * step
        h2 = h.copy()
        h2[i] += delta
        h2[j] -= delta
        new_score = _compute_c5(h2)
        if new_score < cur_score:
            h = h2

        # occasional global perturbation

            pert = rng.normal(scale=0.01, size=n)
            h = np.clip(h + pert, 0.0, 1.0)
            h = _repair_sum(h)

    best_h = _repair_sum(best_h)
    best_score = _compute_c5(best_h)
    return best_h, best_score, best_h.shape[0]
