Implementation of a move:

```
def propose_move(h, rng):
    n = h.shape[0]
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    hi = h[i]
    hj = h[j]
    # allowable delta so that hi+delta in [0,1] and hj-delta in [0,1]
    delta_low = max(-hi, hj - 1)   # hi+delta >=0 => delta >= -hi ; hj - delta <=1 => delta >= hj-1
    delta_high = min(1-hi, hj)     # hi+delta <=1 => delta <= 1-hi ; hj - delta >=0 => delta <= hj
    if delta_low >= delta_high:
        return h, None
    delta = rng.uniform(delta_low, delta_high)
    h_new = h.copy()
    h_new[i] = hi + delta
    h_new[j] = hj - delta
    return h_new, (i, j, delta)
```

def interpolate_h(h_old, n_new):
    n_old = h_old.shape[0]
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    # adjust sum to target:
    target = n_new / 2.0
    current = h_new.sum()
    if current != 0:
        h_new = h_new * (target / current)
    # clip to [0,1] and adjust again
    h_new = np.clip(h_new, 0.0, 1.0)
    # after clipping, sum may deviate, adjust using simple algorithm
    diff = target - h_new.sum()
    if abs(diff) > 1e-12:
        # distribute diff to first index that can accommodate
        # iterate indices to adjust
        for i in range(n_new):
            if diff > 0:
                inc = min(diff, 1.0 - h_new[i])
                h_new[i] += inc
                diff -= inc
            else:
                dec = min(-diff, h_new[i])
                h_new[i] -= dec
                diff += dec
            if abs(diff) < 1e-12:
                break
    return h_new
```

import time
import math
import numpy as np
from scipy.optimize import minimize, differential_evolution


def evaluate_c5(h):
    """Return the overlap integral C5 for a step function given by array h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)


def adjust_sum_to_target(h, target):
    """Modify h in‑place so that its sum equals target while keeping 0≤h≤1."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    for i in range(h.shape[0]):
        if diff > 0 and h[i] < 1.0:
            inc = min(diff, 1.0 - h[i])
            h[i] += inc
            diff -= inc
        elif diff < 0 and h[i] > 0.0:
            dec = min(-diff, h[i])
            h[i] -= dec
            diff += dec
        if abs(diff) < 1e-12:
            break
    return h


def random_binary_feasible(n, rng):
    """Generate a feasible 0/1 (and possibly a 0.5) vector with sum = n/2."""
    h = np.zeros(n, dtype=float)
    half = n // 2
    idx = rng.permutation(n)
    h[idx[:half]] = 1.0
    if n % 2 == 1:
        h[idx[half]] = 0.5
    return h


def interpolate_h(h_old, n_new):
    """Linear interpolation of a coarser vector to a finer resolution."""
    n_old = h_old.shape[0]
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    target = n_new / 2.0
    cur = h_new.sum()
    if cur != 0.0:
        h_new *= (target / cur)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = adjust_sum_to_target(h_new, target)
    return h_new


class SumConstraint:
    """Equality constraint forcing the sum of variables to a target."""
    def __init__(self, target_sum):
        self.target = target_sum

    def __call__(self, x):
        return np.sum(x) - self.target


class DEObjective:
    """Penalty‑augmented objective for differential evolution."""
    def __init__(self, target_sum, penalty):
        self.target = target_sum
        self.penalty = penalty

    def __call__(self, x):
        return evaluate_c5(x) + self.penalty * abs(np.sum(x) - self.target)


def run(seed=42, budget_s=1100, **kwargs):
    """Optimise a step function to minimise the Erdős C₅ overlap constant."""
    start_time = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = math.inf

    # obtain global initial construction if the harness provides it
    init_global = globals().get('initial_h_values', None)
    if init_global is not None:
        init_global = np.asarray(init_global, dtype=float)

    # resolutions to explore (smaller first for speed)
    target_ns = [45, 90, 180]

    for n in target_ns:
        # Respect overall time budget
        if time.time() - start_time > budget_s * 0.95:
            break

        target_sum = n / 2.0

        # Build a list of starting points
        starts = []

        if init_global is not None:
            if init_global.shape[0] == n:
                starts.append(np.clip(init_global.copy(), 0.0, 1.0))
            else:
                starts.append(interpolate_h(init_global, n))

        # supplement with random binary feasible vectors
        while len(starts) < 6:
            starts.append(random_binary_feasible(n, rng))

        # Local optimisation with SLSQP for each start
        bounds = [(0.0, 1.0)] * n
        eq_cons = {'type': 'eq', 'fun': SumConstraint(target_sum)}

        for h0 in starts:
            if time.time() - start_time > budget_s * 0.95:
                break

            h0 = np.clip(h0, 0.0, 1.0)
            h0 = adjust_sum_to_target(h0, target_sum)

            try:
                res = minimize(
                    evaluate_c5,
                    h0,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=[eq_cons],
                    options={'maxiter': 500, 'ftol': 1e-12, 'disp': False}
                )
                if res.success:
                    cand = adjust_sum_to_target(np.clip(res.x, 0.0, 1.0), target_sum)
                    cand_c5 = evaluate_c5(cand)
                    if cand_c5 < best_c5:
                        best_c5 = cand_c5
                        best_h = cand.copy()
            except Exception:
                pass

        # Global search with differential evolution if time permits
        if time.time() - start_time > budget_s * 0.90:
            continue

        # penalty weight: larger than expected C5 to enforce equality
        penalty_weight = max(10.0, (best_c5 if best_c5 != math.inf else 1.0) * 10.0)
        de_obj = DEObjective(target_sum, penalty_weight)

        try:
            de_res = differential_evolution(
                de_obj,
                bounds=bounds,
                seed=seed,
                maxiter=50,
                popsize=10,
                polish=False,
                updating='deferred'
            )
            x = np.clip(de_res.x, 0.0, 1.0)
            x = adjust_sum_to_target(x, target_sum)
            cand_c5 = evaluate_c5(x)
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = x.copy()
        except Exception:
            pass

    # Fallback if optimisation produced nothing feasible
    if best_h is None:
        n = 45
        best_h = random_binary_feasible(n, rng)
        best_c5 = evaluate_c5(best_h)

    return best_h, float(best_c5), int(best_h.shape[0])
