import numpy as np
import time

def _c5_value(h: np.ndarray) -> float:
    """Return the C5 value for a step vector h."""
    N = h.shape[0]
    dx = 2.0 / N
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))


def _worst_shift(h: np.ndarray):
    """Return the integer lag giving the maximal C5 contribution and the raw correlation."""
    N = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    lag = idx - (N - 1)          # integer shift (positive → right)
    return lag, corr


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ----- initialise -------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=np.float64)   # use provided start
    else:
        # fallback: simple half‑ones pattern (never used in the actual challenge)
        N = 100
        h = np.zeros(N, dtype=np.float64)
        h[: N // 2] = 1.0
        rng.shuffle(h)

    N = h.shape[0]
    target_sum = N / 2.0
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        diff = target_sum - cur_sum
        for i in range(N):
            new = h[i] + diff
            if 0.0 <= new <= 1.0:
                h[i] = new
                break
        else:
            h = np.clip(h, 0.0, 1.0)
            cur_sum = h.sum()
            diff = target_sum - cur_sum
            h[0] = np.clip(h[0] + diff, 0.0, 1.0)

    best_h = h.copy()
    best_c5 = _c5_value(h)
    current_c5 = best_c5

    # ----- optimisation loop -----------------------------------------
    start = time.time()
    limit = budget_s - 0.5          # safety margin
    step_max = 0.1
    tol = 1e-12

    while time.time() - start < limit:
        lag, _ = _worst_shift(h)

        # overlapping index pairs for this lag
        if lag >= 0:
            i_idx = np.arange(N - lag)
            j_idx = i_idx + lag
        else:
            i_idx = np.arange(-lag, N)
            j_idx = i_idx + lag               # lag is negative

        diff_vals = h[i_idx] - h[j_idx]
        cand = np.where(diff_vals > tol)[0]

        if cand.size == 0:
            # no obvious improving pair – pick a random one
            i = rng.integers(N)
            j = rng.integers(N)
            if i == j:
                continue
        else:
            sel = rng.choice(cand)
            i = i_idx[sel]
            j = j_idx[sel]

        # feasible transfer from i to j (preserves total mass)
        delta_max = min(h[i], 1.0 - h[j])
        if delta_max <= tol:
            continue
        delta = rng.random() * min(step_max, delta_max)
        if delta <= tol:
            continue

        # apply provisional move
        h[i] -= delta
        h[j] += delta

        new_c5 = _c5_value(h)

        # ---- simulated‑annealing acceptance ----
        elapsed = time.time() - start
        frac = elapsed / limit
        T = 0.02 * (1.0 - frac) + 1e-4   # temperature decreases linearly

        accept = False
        if new_c5 <= current_c5 + tol:
            accept = True
        else:
            prob = np.exp(-(new_c5 - current_c5) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            current_c5 = new_c5
            if new_c5 < best_c5 - 1e-8:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert the move
            h[i] += delta
            h[j] -= delta

        # early exit if we already beat the target bound
        if best_c5 <= 0.3808:
            break

    return best_h, best_c5, N
