        else:

            if np.random.rand() < math.exp(-(new_c5 - current_c5) / T):

                current_c5 = new_c5

            candidate = best_h + noise
            # clip + project to sum

            current_c5 = compute_c5(candidate)
            iter_no_improve = 0

    # final best
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n

import numpy as np
import time
import math
import random

def compute_c5(vec):
    """Maximum overlap C5 for a given step function vector."""
    n = vec.shape[0]
    corr = np.correlate(vec, 1.0 - vec, mode='full')
    return np.max(corr) * (2.0 / n)

def random_binary_vector(n):
    """Return a binary vector of length n with exactly n/2 ones."""
    v = np.zeros(n, dtype=float)
    idx = np.random.choice(n, n // 2, replace=False)
    v[idx] = 1.0
    return v

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h minimizing the Erdős C5 bound.

    Returns:
        h_values (np.ndarray): the best step function found.
        c5_bound (float): the corresponding C5 value.
        n_points (int): length of the discretisation.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    max_time = budget_s * 0.95  # keep a safety margin

    # ------------------------------------------------------------------
    # Initialise problem size and starting vector
    # ------------------------------------------------------------------
    try:
        init_vec = np.asarray(initial_h_values, dtype=float)
        n = init_vec.shape[0]
        # Clip to bounds and enforce sum exactly n/2 (with a small projection)
        init_vec = np.clip(init_vec, 0.0, 1.0)
        target_sum = n / 2.0
        cur_sum = init_vec.sum()
        if not np.isclose(cur_sum, target_sum):
            # simple scaling then re‑clip
            init_vec *= target_sum / cur_sum
            init_vec = np.clip(init_vec, 0.0, 1.0)
            # final adjustment to meet the sum exactly
            diff = target_sum - init_vec.sum()
            if abs(diff) > 1e-12:
                free = np.where((init_vec > 0) & (init_vec < 1))[0]
                if free.size:
                    init_vec[free[0]] += diff
                else:
                    init_vec[0] += diff
    except Exception:
        # fallback: use n = 70 binary vector
        n = 70
        init_vec = random_binary_vector(n)

    best_h = init_vec.copy()
    best_c5 = compute_c5(best_h)

    current_h = best_h.copy()
    current_c5 = best_c5

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
