def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    start = time.time()
    np.random.seed(seed)

    # get initial h if exists
    try:
        init_h = np.array(initial_h_values, dtype=float)
        N = init_h.shape[0]
    except NameError:
        # default N maybe 45
        N = 45
        init_h = None

    # if init_h is provided but not length matches 45, fine.
    target_sum = N / 2.0
    dx = 2.0 / N

    def evaluate(h):
        # compute max correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    def repair(h):
        # ensure shape np array copy
        h = np.array(h, dtype=float)
        # clip
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            return h
        # adjust with water-filling
        if s > target_sum:
            delta = s - target_sum
            # sort indices by descending value (largest first) for reduction
            order = np.argsort(-h)
            for i in order:
                if delta <= 0:
                    break
                max_reduce = h[i]  # can reduce to 0
                if max_reduce <= 0:
                    continue
                red = min(max_reduce, delta)
                h[i] -= red
                delta -= red
        else:
            delta = target_sum - s
            order = np.argsort(h)  # smallest first increase
            for i in order:
                if delta <= 0:
                    break
                max_inc = 1.0 - h[i]
                if max_inc <= 0:
                    continue
                inc = min(max_inc, delta)
                h[i] += inc
                delta -= inc
        # final correction
        diff = target_sum - h.sum()
        if abs(diff) > 1e-9:
            # try to adjust a non-boundary entry
            # find indices with room
            candidates = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
            if len(candidates) == 0:
                # adjust any entry within small epsilon
                i = np.argmin(np.abs(h - 0.5))
                h[i] = np.clip(h[i] + diff, 0.0, 1.0)
            else:
                i = candidates[0]
                h[i] = np.clip(h[i] + diff, 0.0, 1.0)
        # ensure final clip
        np.clip(h, 0.0, 1.0, out=h)
        return h

    # Create initial population
    pop_size = 30
    population = []

    # Add initial if exists
    if init_h is not None:
        init_h = repair(init_h)
        population.append(init_h)

    # Add a simple balanced vector
    # 1's for first floor(N/2), 0's for rest, half entry
    bal = np.zeros(N, dtype=float)
    half = N // 2
    bal[:half] = 1.0
    # the remaining entries are 0; make one entry 0.5 if needed
    # We need sum = target_sum = N/2 = half + 0.5 because N is odd.
    # So set bal[half] = 0.5 if N odd
    if N % 2 == 1:
        bal[half] = 0.5
    else:
        # For even N, just set half entries to 1 (already sum = N/2)
        pass
    bal = repair(bal)
    population.append(bal)

    # Fill rest of population with random individuals
    while len(population) < pop_size:
        rand_vec = np.random.rand(N)  # uniform in [0,1]
        ind = repair(rand_vec)
        population.append(ind)

    # Evaluate population
    obj_vals = np.array([evaluate(ind) for ind in population])

    # Keep track of best
    best_idx = np.argmin(obj_vals)
    best_h = population[best_idx]
    best_obj = obj_vals[best_idx]

    # GA parameters
    max_generations = 10000  # will break early by time
    tournament_size = 3
    mutation_rate = 0.25  # per gene
    sigma = 0.05  # Gaussian std

    # Generation loop
    gen = 0
    while True:
        current_time = time.time()
        elapsed = current_time - start
        if elapsed >= budget_s:
            break
        # optional early break if some generation limit
        if gen >= max_generations:
            break

        # produce next generation
        new_population = []

        # elitism: keep best
        new_population.append(best_h.copy())

        while len(new_population) < pop_size:
            # tournament selection for two parents
            # select parent1
            participants = np.random.choice(pop_size, tournament_size, replace=False)
            p1_idx = participants[np.argmin(obj_vals[participants])]
            # select parent2
            participants = np.random.choice(pop_size, tournament_size, replace=False)
            p2_idx = participants[np.argmin(obj_vals[participants])]
            parent1 = population[p1_idx]
            parent2 = population[p2_idx]

            # crossover: arithmetic convex combination
            alpha = np.random.rand()  # scalar
            child = alpha * parent1 + (1 - alpha) * parent2
            # Since parents sum to target_sum, child sum = target_sum automatically.

            # mutation: per gene probability
            mask = np.random.rand(N) < mutation_rate
            if np.any(mask):
                noise = np.random.normal(0.0, sigma, size=N)
                child = child + mask * noise
                # clip and repair
                child = repair(child)
            else:
                # ensure numeric stability
                child = repair(child)

            new_population.append(child)

        # evaluate new population
        population = new_population
        obj_vals = np.array([evaluate(ind) for ind in population])

        # update best
        idx = np.argmin(obj_vals)
        if obj_vals[idx] < best_obj:
            best_obj = obj_vals[idx]
            best_h = population[idx].copy()

        gen += 1

    # final return
    # ensure best_h meets constraints precisely (repair)
    best_h = repair(best_h)
    best_obj = evaluate(best_h)
    return best_h, best_obj, N

def local_search(h, max_iter):
    best_h = h.copy()
    best_val = evaluate(best_h)
    for it in range(max_iter):
        if time.time() - start >= budget_s:
            break
        # pick two indices
        i, j = np.random.choice(N, 2, replace=False)
        # max possible delta = min(1 - h[i], h[j]) for moving mass from j to i (increase i)
        # We'll try both directions.
        # Propose delta = random fraction of possible range
        # direction: increase i, decrease j
        delta_max = min(1 - best_h[i], best_h[j])
        if delta_max > 0:
            delta = np.random.rand() * delta_max * 0.5  # small fraction
            new_h = best_h.copy()
            new_h[i] += delta
            new_h[j] -= delta
            # evaluate
            val = evaluate(new_h)
            if val < best_val - 1e-9:  # improvement

                best_val = val
        # also try opposite direction: increase j, decrease i

            delta = np.random.rand() * delta_max * 0.5

            new_h[j] += delta
            val = evaluate(new_h)
            if val < best_val - 1e-9:
                best_h = new_h
                best_val = val
    return best_h, best_val

import numpy as np
import time

def evaluate_overlap(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair(h, target_sum):
    # clip
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h
    if s > target_sum:
        delta = s - target_sum
        order = np.argsort(-h)  # largest first
        for i in order:
            if delta <= 0:
                break
            max_reduce = h[i]  # can reduce to 0
            red = min(max_reduce, delta)
            h[i] -= red
            delta -= red
    else:
        delta = target_sum - s
        order = np.argsort(h)  # smallest first
        for i in order:
            if delta <= 0:
                break
            max_inc = 1.0 - h[i]
            inc = min(max_inc, delta)
            h[i] += inc
            delta -= inc
    # final correction
    diff = target_sum - h.sum()
    if abs(diff) > 1e-9:
        # find a non-boundary index
        cand = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
        if cand.size > 0:
            i = cand[0]
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
        else:
            # adjust the first element within bounds
            i = np.argmin(np.abs(h - 0.5))
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def evaluate_overlap(h, dx):

    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def repair(h, target_sum):

        return h
