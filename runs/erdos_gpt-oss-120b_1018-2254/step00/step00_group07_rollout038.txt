def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time

    start_time = time.time()
    np.random.seed(seed)

    # Determine problem size and initial parent if available
    if 'initial_h_values' in globals():
        try:
            n_points = int(len(initial_h_values))
            parent = np.array(initial_h_values, dtype=float)
        except Exception:
            n_points = 80
            parent = np.full(80, 0.5, dtype=float)
    else:
        n_points = 80
        parent = np.full(n_points, 0.5, dtype=float)

    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # equals 1 / dx

    # ---------- Helper functions ----------
    def project_capped_simplex(v):
        """Project v onto the box [0,1]^n intersect {sum = target_sum}."""
        lo = np.min(v) - 1.0
        hi = np.max(v)
        for _ in range(60):
            lam = (lo + hi) * 0.5
            x = np.clip(v - lam, 0.0, 1.0)
            s = x.sum()
            if s > target_sum:
                lo = lam
            else:
                hi = lam
        return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

    def compute_c5(h):
        """Return the C5 value for a given vector h."""
        corr = np.correlate(h, 1.0 - h, mode='full')
        return dx * corr.max()

    # ---------- Seed generation ----------
    seeds = []
    # parent solution (if provided)
    if 'initial_h_values' in globals():
        seeds.append(parent.copy())
    # uniform half-half solution
    seeds.append(np.full(n_points, 0.5, dtype=float))
    # a few random feasible vectors
    for _ in range(5):
        rnd = np.random.rand(n_points)
        rnd = project_capped_simplex(rnd)
        seeds.append(rnd)

    best_h = None
    best_val = float('inf')

    # ---------- Hill climbing optimizer ----------
    def hill_climb(h0, deadline):
        """Simple hill climbing that transfers mass between indices to lower max overlap."""
        h = h0.copy()
        best_local_h = h.copy()
        best_local_val = compute_c5(h)
        epsilon = 0.01
        n = n_points
        while time.time() < deadline:
            # current correlation and worst shift
            corr = np.correlate(h, 1.0 - h, mode='full')
            max_idx = int(corr.argmax())
            cur_val = corr[max_idx] * dx
            # record best seen
            if cur_val < best_local_val:
                best_local_val = cur_val
                best_local_h = h.copy()
            # shift corresponding to max_idx
            shift = max_idx - (n - 1)
            if shift == 0:
                # skip zero shift (product is h*(1-h) which cannot be lowered by transfer)
                continue
            # valid index range for i where i+shift is inside [0, n)
            if shift > 0:
                i_start, i_end = 0, n - shift
            else:
                i_start, i_end = -shift, n
            if i_start >= i_end:
                continue
            i_range = np.arange(i_start, i_end)
            j_range = i_range + shift
            prod = h[i_range] * (1.0 - h[j_range])
            total = prod.sum()
            if total > 0:
                probs = prod / total
                chosen = np.random.choice(len(i_range), p=probs)
                i = i_range[chosen]
                j = j_range[chosen]
            else:
                i = np.random.randint(i_start, i_end)
                j = i + shift
            # amount to move from i to j
            delta = epsilon
            delta = min(delta, h[i])
            delta = min(delta, 1.0 - h[j])
            if delta <= 0:
                continue
            # propose move
            h[i] -= delta
            h[j] += delta
            # evaluate new max overlap
            new_corr = np.correlate(h, 1.0 - h, mode='full')
            new_val = new_corr.max() * dx
            # accept only if not worsening
            if new_val <= cur_val + 1e-12:
                # keep change
                pass
            else:
                # revert
                h[i] += delta
                h[j] -= delta
        return best_local_h, best_local_val

    # ---------- Main optimization loop ----------
    for idx, seed_h in enumerate(seeds):
        remaining = budget_s - (time.time() - start_time)
        if remaining <= 5.0:
            break
        # allocate time slice for this seed
        per_seed = min(30.0, remaining * 0.5) if (len(seeds) - idx) > 1 else max(0.0, remaining - 0.5)
        deadline = time.time() + per_seed
        # hill climbing from this seed
        h_opt, val_opt = hill_climb(seed_h, deadline)
        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()
        # optional SLSQP refinement if SciPy is present
        try:
            from scipy.optimize import minimize
            bounds = [(0.0, 1.0)] * n_points
            eq_cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
            opts = {'maxiter': 200, 'ftol': 1e-9, 'disp': False}
            res = minimize(compute_c5, h_opt, method='SLSQP',
                           bounds=bounds, constraints=[eq_cons], options=opts)
            if res.success:
                fin_val = compute_c5(res.x)
                if fin_val < best_val:
                    best_val = fin_val
                    best_h = res.x.copy()
        except Exception:
            pass
        if time.time() - start_time > budget_s - 0.5:
            break

    # Fallback if nothing found
    if best_h is None:
        best_h = np.full(n_points, 0.5, dtype=float)
        best_val = compute_c5(best_h)

    # Final projection to guarantee feasibility (sum and bounds)
    best_h = project_capped_simplex(best_h)
    best_val = compute_c5(best_h)

    return (best_h, best_val, n_points)
