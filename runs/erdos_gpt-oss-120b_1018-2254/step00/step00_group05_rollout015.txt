def compute_c5(h, dx):

    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    # Determine initial h
    try:
        h0 = np.array(initial_h_values, dtype=np.float64)
        n = h0.shape[0]
    except NameError:
        n = 70
        # start with binary random half ones
        h0 = np.zeros(n, dtype=np.float64)
        indices = rng.choice(n, size=n//2, replace=False)
        h0[indices] = 1.0
    # ensure bounds and sum
    h0 = np.clip(h0, 0.0, 1.0)
    desired_sum = n / 2.0
    current_sum = h0.sum()
    if not np.isclose(current_sum, desired_sum):
        # simple scaling then re-projection
        # Use linear projection onto the hyperplane with box constraints
        # We'll adjust by adding offset to all entries then clip iteratively
        offset = (desired_sum - current_sum) / n
        h0 = np.clip(h0 + offset, 0.0, 1.0)
        # after clipping, adjust again if needed
        current_sum = h0.sum()
        # if still off due to clipping, distribute residual greedily
        residual = desired_sum - current_sum
        if abs(residual) > 1e-12:
            # add/subtract residual from random non-bound entries
            free = np.where((h0 > 0) & (h0 < 1))[0]
            if free.size == 0:
                # fallback to binary construction
                h0 = np.zeros(n)
                idx = rng.choice(n, size=n//2, replace=False)
                h0[idx] = 1.0
            else:
                # distribute residual equally (might be fractional)
                adjustment = residual / free.size
                h0[free] += adjustment
                h0 = np.clip(h0, 0.0, 1.0)
                # final correction to enforce sum exactly by adjusting one free element
                diff = desired_sum - h0.sum()
                if abs(diff) > 1e-12:
                    # adjust first free element within bounds
                    i = free[0]
                    h0[i] += diff
                    h0[i] = min(max(h0[i], 0.0), 1.0)
    # size and dx
    dx = 2.0 / n
    # compute initial score
    cur_h = h0.copy()
    cur_c5 = compute_c5(cur_h, dx)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    # SA parameters
    start_time = time.time()
    # schedule
    step_start = 0.05
    step_end = 0.0005
    temp_start = 0.02
    temp_end = 1e-5
    # iteration loop
    it = 0
    while True:
        now = time.time()
        elapsed = now - start_time
        if elapsed >= budget_s * 0.95:
            break
        # compute schedules
        frac = elapsed / budget_s
        step = step_start * (1.0 - frac) + step_end * frac
        temp = temp_start * (1.0 - frac) + temp_end * frac
        # propose move: pick two distinct indices

        new_h[i] = hi + delta
        new_h[j] = hj - delta
        # compute score

    return float(np.max(corr) * dx)
