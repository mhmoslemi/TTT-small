```
def hill_climb(x_initial, time_limit, rng):
    x = x_initial.copy()
    n = len(x)
    best_x = x.copy()
    best_min = compute_min_overlap(x)
    cur_min = best_min
    # maintain lists
    ones = list(np.where(x == 1)[0])
    zeros = list(np.where(x == 0)[0])
    start = time.time()
    # scaling for temperature
    # For simulated annealing schedule: T decays linearly to near zero.
    # We'll use total iterations budget maybe 1e6, but we base on time.
    # We'll compute fraction = (time.time() - start) / time_limit
    # temperature = T_start * (1 - fraction) + T_end * fraction
    # Accept prob if new_min >= cur_min: accept.
    # else accept with probability exp((new_min - cur_min)/ T)
    T_start = 2.0
    T_end = 0.001
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= time_limit:
            break
        # select random swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # apply swap
        x[i] = 0
        x[j] = 1
        # evaluate min_overlap
        new_min = compute_min_overlap(x)
        # acceptance
        accept = False
        if new_min >= cur_min:
            accept = True
        else:
            # temperature schedule
            frac = elapsed / time_limit
            T = T_start * (1 - frac) + T_end * frac
            # compute acceptance probability
            # delta negative: new_min - cur_min
            prob = math.exp((new_min - cur_min) / T) if T > 0 else 0.0
            if rng.random() < prob:
                accept = True
        if accept:
            cur_min = new_min
            # update ones/zeros lists
            # i was 1 -> 0; move i to zeros
            # j was 0 -> 1; move j to ones
            # Remove i from ones list (existing)
            # Let's use list remove: O(n) but small
            ones.remove(i)
            zeros.append(i)
            zeros.remove(j)
            ones.append(j)
            # update best
            if cur_min > best_min:
                best_min = cur_min
                best_x = x.copy()
        else:
            # revert swap
            x[i] = 1
            x[j] = 0
        # maybe periodic check for time; loop continues.
    return best_x, best_min
```

                best_x = x.copy()
        else:
            # revert swap
            x[i] = 1
            x[j] = 0

import time, math, numpy as np

def compute_min_overlap(x: np.ndarray) -> int:
    """Return the minimum overlap sum_i x[i] * x[i+k] over all positive shifts k."""
    n = x.shape[0]
    min_val = None
    for k in range(1, n):
        # dot of the overlapping parts
        val = int(np.dot(x[: n - k], x[k:]))
        if min_val is None or val < min_val:
            min_val = val
            if min_val == 0:
                break
    return min_val if min_val is not None else 0


def hill_climb(x_initial: np.ndarray, time_limit: float, rng: np.random.Generator):
    """Binary hill‑climbing with simulated annealing."""
    n = x_initial.shape[0]
    x = x_initial.copy()
    cur_min = compute_min_overlap(x)
    best_x = x.copy()
    best_min = cur_min

    # positions of 1s and 0s (as Python lists for O(1) swaps)
    ones = [int(i) for i in np.where(x == 1)[0]]
    zeros = [int(i) for i in np.where(x == 0)[0]]

    start = time.time()
    T_start, T_end = 2.0, 0.001

    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        if not ones or not zeros:
            break

        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]

        # propose swap
        x[i] = 0
        x[j] = 1
        new_min = compute_min_overlap(x)

        accept = False
        if new_min >= cur_min:
            accept = True
        else:
            frac = elapsed / time_limit
            T = T_start * (1 - frac) + T_end * frac
            prob = math.exp((new_min - cur_min) / T) if T > 0 else 0.0
            if rng.random() < prob:
                accept = True

        if accept:
            cur_min = new_min
            # update the index lists
            ones[i_idx] = j
            zeros[j_idx] = i
            if cur_min > best_min:
                best_min = cur_min
                best_x = x.copy()
        else:
            # revert swap
            x[i] = 1
            x[j] = 0

    return best_x, best_min


def continuous_refine(h_start: np.ndarray, time_limit: float, rng: np.random.Generator):
    """Simple continuous local search that preserves the integral."""
    n = h_start.shape[0]
    target_sum = n / 2.0
    h = np.clip(h_start.astype(float), 0.0, 1.0)

    # enforce exact sum by scaling then tiny correction
    total = h.sum()
    if total > 0:
        h *= target_sum / total
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:

        for _ in range(10):
            idx = rng.integers(n)
            can_inc = 1.0 - h[idx]
            can_dec = h[idx]
            if diff > 0 and can_inc > 1e-12:

                h[idx] += inc
                diff -= inc
            elif diff < 0 and can_dec > 1e-12:
                dec = min(can_dec, -diff)
                h[idx] -= dec
                diff += dec
            if abs(diff) < 1e-12:

    cur_c5 = float(evaluate_erdos_solution(h))

    start = time.time()
    T_start, T_end = 0.1, 0.001

            best_c5 = cand_c5
            best_h = cand_h.astype(float)
            best_min = cand_min

    # ------------------------------------------------------------------
    # Optional continuous refinement on the best binary solution
    # ------------------------------------------------------------------
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed - 0.1
    if remaining > 0.5:
        refined_h, refined_c5 = continuous_refine(best_h, remaining, rng)
        if refined_c5 < best_c5:
            best_c5 = refined_c5
            best_h = refined_h

    # ------------------------------------------------------------------
    # Final safety projection (ensure sum constraint and bounds)
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    target_sum = n / 2.0
    total = best_h.sum()
    if not np.isclose(total, target_sum, atol=1e-9):
        # simple scaling then clipping fix
        best_h *= target_sum / total
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h.astype(float), float(best_c5), int(n)
