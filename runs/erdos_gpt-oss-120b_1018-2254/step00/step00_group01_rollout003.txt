import time
import math
import random
import numpy as np

def compute_c5(h):
    """Return max overlap integral for discretized h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_to_sum_and_bounds(y, target):
    """
    Project vector y onto the feasible set:
    0 <= x_i <= 1 and sum(x) = target.
    The solution has the form x_i = clip(y_i + lam, 0, 1) for some lam.
    """
    # bounds for lambda ensuring full 0 or full 1 clipping
    lo = -np.max(y)          # all entries <=0 after shift -> clipped to 0
    hi = 1.0 - np.min(y)     # all entries >=1 after shift -> clipped to 1
    # binary search for lambda making sum = target
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(y + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(y + lam, 0.0, 1.0)

def random_feasible_vector(n, target):
    """Generate a random vector in [0,1]^n with given sum via projection."""
    y = np.random.rand(n)
    return project_to_sum_and_bounds(y, target)

def run(seed=42, budget_s=1100, **kwargs):
    # ------------------- Setup -------------------
    np.random.seed(seed)
    random.seed(seed)

    # Obtain initial vector if provided
    init = globals().get('initial_h_values')
    if init is not None:
        h = np.asarray(init, dtype=float)
        n = h.shape[0]
    else:
        n = 47                     # fallback size
        h = random_feasible_vector(n, n / 2.0)

    target_sum = n / 2.0

    # Ensure feasibility (projection may be needed due to rounding)
    if not np.isclose(h.sum(), target_sum, atol=1e-12):
        h = project_to_sum_and_bounds(h, target_sum)

    # Initial evaluation
    cur_h = h.copy()
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    # Parameters for stochastic search
    start_time = time.time()
    max_step_initial = 0.3   # initial magnitude of perturbations
    min_step = 0.0005        # minimal step size near the end
    init_temp = 0.02         # initial temperature for simulated annealing
    no_improve = 0
    RESTART_LIMIT = 50000    # iterations without improvement before restart

    # Main optimization loop
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.99:
            break

        # Adaptive step size and temperature
        progress = elapsed / budget_s
        step_scale = max_step_initial * (1.0 - progress) + min_step
        temperature = init_temp * (1.0 - progress) + 1e-9

        # Choose move type
        if np.random.rand() < 0.6:
            # Gaussian perturbation + projection
            delta = np.random.randn(n)
            delta -= delta.mean()                # enforce zero-sum direction
            delta *= step_scale
            y = cur_h + delta
            new_h = project_to_sum_and_bounds(y, target_sum)
        else:
            # Pairwise transfer move
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            # Amount that can be moved from i to j while staying feasible
            max_delta = min(cur_h[i], 1.0 - cur_h[j])
            if max_delta <= 0.0:
                continue
            delta = max_delta * np.random.rand()
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta

        new_c5 = compute_c5(new_h)

        # Acceptance rule (simulated annealing)
        if new_c5 < cur_c5 or np.random.rand() < math.exp((cur_c5 - new_c5) / temperature):
            cur_h = new_h
            cur_c5 = new_c5
            # Record best solution

                best_c5 = new_c5
                best_h = new_h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Restart if stagnation

            cur_h = random_feasible_vector(n, target_sum)
            cur_c5 = compute_c5(cur_h)
            no_improve = 0

    # Final evaluation using provided helper if available (more accurate)
    if 'evaluate_erdos_solution' in globals():

        except Exception:
            # fallback to our own evaluation
            best_c5 = compute_c5(best_h)

    return best_h.tolist(), best_c5, n
