def project_simplex_cap(v, target_sum):
    import cvxpy as cp
    x = cp.Variable(len(v))
    objective = cp.Minimize(cp.sum_squares(x - v))
    constraints = [cp.sum(x) == target_sum,
                   x >= 0,
                   x <= 1]
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True)
    return np.array(x.value).flatten()

def project_to_simplex_cap(v, target_sum):
    # ensure target sum within feasible range (0 <= target_sum <= n)
    if target_sum < 0 or target_sum > len(v):
        raise ValueError
    low = -np.max(v) - 1.0
    high = np.max(v) + 1.0
    for _ in range(50):
        mid = (low + high) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            low = mid
        else:
            high = mid
    # final projection
    x = np.clip(v - (low + high) / 2.0, 0.0, 1.0)
    # due to rounding may not equal target_sum precisely; adjust slight scaling
    # compute diff:
    diff = x.sum() - target_sum

import time
import numpy as np
from scipy.optimize import minimize

def _project_to_feasible(v, target_sum):
    """Project vector v onto the simplex {x: 0<=x_i<=1, sum x_i = target_sum}."""
    n = v.shape[0]
    target_sum = float(target_sum)
    # Binary search for threshold theta
    low = -np.max(v) - 1.0
    high = np.max(v) + 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            low = mid
        else:
            high = mid
    theta = (low + high) * 0.5
    x = np.clip(v - theta, 0.0, 1.0)
    # Small correction for numerical drift
    diff = x.sum() - target_sum
    if abs(diff) > 1e-8:
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        if free.any():
            x[free] -= diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
    return x
