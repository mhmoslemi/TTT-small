def upsample_solution(h, n_target):
    n = len(h)
    x_old = np.linspace(0, 2, n, endpoint=False)
    x_new = np.linspace(0, 2, n_target, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    # scale to desired sum
    desired_sum = n_target / 2.0
    current_sum = h_new.sum()
    if current_sum > 0:
        h_new = h_new * (desired_sum / current_sum)
    # ensure bounds

    # after clipping may have slight sum error; re-scale again

    if current_sum > 0:

import numpy as np
import time
import math
import random

def evaluate_c5(h):
    """Compute C5 for a discretised step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation between h and (1-h); np.correlate does zero‑padding
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_sum(h, target):
    """Scale h to have sum exactly target and keep it inside [0,1]."""
    total = h.sum()
    if total == 0:
        h = np.full_like(h, target / h.shape[0])
    else:
        h = h * (target / total)
    h = np.clip(h, 0.0, 1.0)

    # tiny uniform correction if the scaling+clipping moved the sum a bit
    total = h.sum()
    if abs(total - target) > 1e-8:
        diff = target - total
        h += diff / h.shape[0]
        h = np.clip(h, 0.0, 1.0)
    return h

def local_optimize(h0, time_limit):
    """Simple simulated‑annealing style optimizer respecting the constraints."""

    target = n / 2.0
    cur_h = project_sum(np.copy(h0), target)
    cur_score = evaluate_c5(cur_h)
