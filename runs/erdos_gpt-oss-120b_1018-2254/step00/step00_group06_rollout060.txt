    h = np.full(n, 0.5, dtype=np.float64)
    for _ in range(steps * n):

def repair_h(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    tolerance = 1e-12
    max_iter = 1000
    free = np.ones(n, dtype=bool)  # initially all considered free
    for _ in range(max_iter):
        if abs(diff) < tolerance:
            break
        free_idx = np.where(free)[0]
        if free_idx.size == 0:
            # can't adjust, break
            break
        delta = diff / free_idx.size
        # Update
        h[free_idx] += delta
        # Clip and update free mask
        # after adding delta, some may cross bounds
        # For those >1 set to 1 and mark not free
        # For those <0 set to 0 and mark not free
        over_mask = (h > 1.0)
        under_mask = (h < 0.0)
        h[over_mask] = 1.0
        h[under_mask] = 0.0
        # Update free mask: exclude those at bounds
        free[over_mask | under_mask] = False
        diff = target - np.sum(h)
    # If not exact, adjust smallest tolerance by scaling all?
    # final ensure sum close
    # If sum off due to floating errors, adjust one free variable if exists

                h[i] += diff

def run(seed=42, budget_s=1100, **kwargs):
    import time, random, numpy as np
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    # Determine n_points
    if 'initial_h_values' in globals():
        init_h = np.array(initial_h_values, dtype=np.float64)
        n_points = init_h.shape[0]
    else:
        n_points = 95
        init_h = np.full(n_points, 0.5)
    target = n_points / 2.0
    dx = 2.0 / n_points
    # repair init_h
    init_h = repair_h(init_h)
    # population parameters
    pop_size = 200
    generations = 0
    max_generations = 10000  # high, but will break by time
    # create initial population
    population = [init_h]
    for _ in range(pop_size - 1):
        h = random_feasible_h(n_points, steps=5, step_size=0.5)
        population.append(h)
    # evaluate fitness list
    fitness = [compute_c5(ind, dx) for ind in population]
    best_idx = int(np.argmin(fitness))
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    # GA loop
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s:
            break
        if generations >= max_generations:
            break
        generations += 1
        new_pop = []
        # elitism: keep best
        new_pop.append(best_h.copy())
        # generate rest individuals
        while len(new_pop) < pop_size:
            # tournament selection
            p1 = tournament_selection(population, fitness, k=3)
            p2 = tournament_selection(population, fitness, k=3)
            # crossover
            if random.random() < 0.7:
                cut = random.randint(1, n_points - 1)
                child1 = np.concatenate((p1[:cut], p2[cut:]))
                child2 = np.concatenate((p2[:cut], p1[cut:]))
            else:
                child1 = p1.copy()
                child2 = p2.copy()
            # mutation
            child1 = mutate(child1, mutation_rate=0.01, sigma=0.1, n=n_points)
            child2 = mutate(child2, mutation_rate=0.01, sigma=0.1, n=n_points)
            # repair
            child1 = repair_h(child1)
            child2 = repair_h(child2)
            # add to new_pop
            new_pop.append(child1)
            if len(new_pop) < pop_size:
                new_pop.append(child2)
        # evaluate new pop
        population = new_pop
        fitness = [compute_c5(ind, dx) for ind in population]
        # update best
        cur_best_idx = int(np.argmin(fitness))
        cur_best_c5 = fitness[cur_best_idx]
        if cur_best_c5 < best_c5:
            best_c5 = cur_best_c5
            best_h = population[cur_best_idx].copy()
    # final results
    return best_h.tolist(), float(best_c5), n_points

def tournament_selection(pop, fitness, k=3):
    indices = np.random.choice(len(pop), size=k, replace=False)
    best = indices[0]
    best_f = fitness[best]
    for i in indices[1:]:
        if fitness[i] < best_f:
            best = i
            best_f = fitness[i]
    return pop[best]
```

def mutate(h, mutation_rate=0.01, sigma=0.1, n=None):
    if n is None:
        n = h.shape[0]
    mask = np.random.rand(n) < mutation_rate
    if np.any(mask):
        h = h.copy()
        h[mask] += np.random.normal(0.0, sigma, size=np.count_nonzero(mask))
    return h

import time, random, numpy as np

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def repair_h(h):
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    eps = 1e-12
    free = np.ones(n, dtype=bool)
    for _ in range(1000):
        if abs(diff) <= eps:
            break
        free_idx = np.where(free)[0]
        if free_idx.size == 0:
            break
        delta = diff / free_idx.size
        h[free_idx] += delta
        over = h > 1.0
        under = h < 0.0
        if np.any(over):
            h[over] = 1.0
            free[over] = False
        if np.any(under):
            h[under] = 0.0
            free[under] = False
        diff = target - np.sum(h)
    # last resort adjust smallest free variable
    diff = target - np.sum(h)
    if abs(diff) > 1e-8:
        for i in range(n):
            if 0.0 < h[i] < 1.0:
                h[i] = np.clip(h[i] + diff, 0.0, 1.0)
                break
    return h

def random_feasible_h(n):
    h = np.random.rand(n)
    return repair_h(h)

def tournament_selection(pop, fit, k=3):
    idx = np.random.choice(len(pop), size=k, replace=False)
    best = idx[0]
    best_f = fit[best]
    for i in idx[1:]:
        if fit[i] < best_f:
            best = i
            best_f = fit[i]
    return pop[best]

def mutate(h, rate=0.01, sigma=0.1):
    n = h.shape[0]
    mask = np.random.rand(n) < rate
    if not np.any(mask):
        return h
    h = h.copy()
    h[mask] += np.random.normal(0.0, sigma, size=np.count_nonzero(mask))
    return h

def hill_climb(h, dx, max_iter=5000, time_limit=None):
    start = time.time()
    best = h.copy()
    best_c5 = compute_c5(best, dx)
    n = len(best)
    step = 0.3
    for _ in range(max_iter):
        if time_limit is not None and time.time() - start > time_limit:
            break
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # feasible delta moving from i to j
        max_pos = min(step, best[i], 1.0 - best[j])
        max_neg = -min(step, 1.0 - best[i], best[j])
        delta = np.random.uniform(max_neg, max_pos)
        if delta == 0.0:
            continue
        new = best.copy()
        new[i] -= delta
        new[j] += delta
        new_c5 = compute_c5(new, dx)
        if new_c5 < best_c5 - 1e-12:
            best = new
            best_c5 = new_c5
    return best, best_c5

        # allow hill climb for remaining time
        max_iter = int(min(20000, remaining * 500))
        best_h, best_c5 = hill_climb(best_h, dx, max_iter=max_iter, time_limit=remaining)
    return best_h.tolist(), float(best_c5), n

import time
import random
import numpy as np

def compute_c5(h, dx):
    """Maximum overlap integral for discrete step function."""
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def repair_h(h):
    """Project h onto [0,1]^n with sum exactly n/2."""
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    eps = 1e-12
    free = np.ones(n, dtype=bool)  # variables not on a bound
    for _ in range(1000):
        if abs(diff) <= eps:
            break
        idx = np.where(free)[0]
        if idx.size == 0:
            break
        delta = diff / idx.size
        h[idx] += delta
        over = h > 1.0
        under = h < 0.0
        if np.any(over):
            h[over] = 1.0
            free[over] = False
        if np.any(under):
            h[under] = 0.0
            free[under] = False
        diff = target - np.sum(h)
    # Final tiny correction if needed
    diff = target - np.sum(h)
    if abs(diff) > 1e-8:
        for i in range(n):
            if 0.0 < h[i] < 1.0:
                h[i] = np.clip(h[i] + diff, 0.0, 1.0)
                break
    return h

def random_feasible_h(n):
    """Generate a random feasible vector."""
    h = np.random.rand(n)
    return repair_h(h)

def tournament_selection(pop, fit, k=3):
    """Select the best individual among k random participants."""
    idx = np.random.choice(len(pop), size=k, replace=False)
    best = idx[0]
    best_f = fit[best]
    for i in idx[1:]:
        if fit[i] < best_f:
            best = i
            best_f = fit[i]
    return pop[best]

def mutate(h, rate=0.01, sigma=0.1):
    """Gaussian mutation on a subset of genes."""
    n = h.shape[0]
    mask = np.random.rand(n) < rate
    if not np.any(mask):
        return h
    h = h.copy()
    h[mask] += np.random.normal(0.0, sigma, size=np.count_nonzero(mask))
    return h

def hill_climb(h, dx, max_iter=5000, time_limit=None):
    """Simple pairwise mass-transfer hill climbing."""
    start = time.time()
    best = h.copy()
    best_c5 = compute_c5(best, dx)
    n = len(best)
    step = 0.3
    for _ in range(max_iter):
        if time_limit is not None and time.time() - start > time_limit:
            break
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        max_pos = min(step, best[i], 1.0 - best[j])
        max_neg = -min(step, 1.0 - best[i], best[j])
        delta = np.random.uniform(max_neg, max_pos)
        if delta == 0.0:
            continue
        new = best.copy()
        new[i] -= delta
        new[j] += delta
        new_c5 = compute_c5(new, dx)
        if new_c5 < best_c5:
            best, best_c5 = new, new_c5
    return best, best_c5

    np.random.seed(seed)
    random.seed(seed)
    start = time.time()

    # Determine problem size
    if 'initial_h_values' in globals():

        except Exception:
            init = None

        generation += 1

        new_pop = [best_h.copy()]

        while len(new_pop) < pop_size:

            p2 = tournament_selection(population, fitness, k=3)
            if random.random() < 0.7:
