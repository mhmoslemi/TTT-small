import numpy as np
import time
from scipy.optimize import minimize

_target_sum = None  # placeholder for equality constraint

def compute_c5_maxcorr(h):
    n = len(h)
    if n == 0:
        return np.inf
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def sum_eq_constraint(x):
    return np.sum(x) - _target_sum

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    np.random.seed(seed)
    # Obtain initial h.
    try:
        h0 = np.array(initial_h_values, copy=False)
    except NameError:
        # fallback: 45 points with half ones.
        n = 45
        h0 = np.zeros(n, dtype=float)
        half = n // 2
        h0[:half] = 1.0
        if n % 2 == 1:
            # add 0.5 to meet sum n/2
            h0[half] = 0.5
    # Ensure constraints: clip and adjust sum
    h0 = np.clip(h0, 0.0, 1.0)
    n = len(h0)
    target = n / 2.0
    # Adjust sum to target: simple proportional scaling may break bounds,
    # we'll use simple offset method.
    cur_sum = h0.sum()
    diff = target - cur_sum
    if abs(diff) > 1e-12:
        # distribute diff onto elements not at bounds.
        free_idx = np.where((h0 > 0) & (h0 < 1))[0]
        if len(free_idx) > 0:
            delta = diff / len(free_idx)
            h0[free_idx] += delta
            # clip again
            h0 = np.clip(h0, 0.0, 1.0)
        # recompute diff and adjust first element if needed
        cur_sum = h0.sum()
        diff = target - cur_sum
        if abs(diff) > 1e-12:
            # adjust first element
            h0[0] = np.clip(h0[0] + diff, 0.0, 1.0)
    # compute initial bound using evaluate_erdos_solution if available
    try:
        init_c5 = float(evaluate_erdos_solution(h0))
    except NameError:
        init_c5 = float(compute_c5_maxcorr(h0))
    best_h = h0.copy()
    best_c5 = init_c5
    # Time left?
    elapsed = time.time() - start
    if budget_s - elapsed > 5.0:
        # perform local refinement using SLSQP
        global _target_sum
        _target_sum = target
        bounds = [(0.0, 1.0)] * n
        # Options: limit maxiter by time?
        # We'll compute maxiter based on remaining budget: maybe 200 iterations fine.
        options = {'maxiter': 200, 'ftol': 1e-10}
        res = minimize(compute_c5_maxcorr, best_h, method='SLSQP',
                       bounds=bounds, constraints=[{'type': 'eq', 'fun': sum_eq_constraint}],
                       options=options)
        if res.success:
            cand = np.clip(res.x, 0.0, 1.0)
            # enforce sum exactly
            cur = cand.sum()
            diff = target - cur
            if abs(diff) > 1e-12:
                cand[0] = np.clip(cand[0] + diff, 0.0, 1.0)
            cand_c5 = float(compute_c5_maxcorr(cand))
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = cand
    # final
    return (best_h.tolist(), float(best_c5), len(best_h))

import time
import numpy as np
from scipy.optimize import minimize

# Global placeholder for the equality constraint (sum of h == n/2)
_target_sum = None


def compute_c5_maxcorr(h):
    """Maximum overlap integral for a discretised step function h."""
    n = len(h)
    if n == 0:
        return np.inf
    dx = 2.0 / n
    # correlation of h with (1 - h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


def sum_eq_constraint(x):
    """Equality constraint: sum(x) == target_sum."""
    return np.sum(x) - _target_sum


def enforce_sum_and_bounds(h, target):
    """Clip to [0,1] and adjust to satisfy the exact sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target - cur
    # distribute the difference over free entries (not at a bound)
    if abs(diff) > 1e-12:
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size > 0:
            delta = diff / free.size
            h[free] += delta
            h = np.clip(h, 0.0, 1.0)
        # final correction on the first element
        cur = h.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function for the Erdős minimum overlap problem.

    Returns
    -------
    tuple
        (h_values, c5_bound, n_points)
    """
    start_time = time.time()
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # 1. Obtain an initial feasible vector.
    # ------------------------------------------------------------------
    try:
        # Provided by the harness; may be a list or ndarray.
        h0 = np.array(initial_h_values, copy=False, dtype=float)
    except NameError:
        # Fallback: 45‑point simple construction with exactly half ones.
        n_fallback = 45
        h0 = np.zeros(n_fallback, dtype=float)
        half = n_fallback // 2
        h0[:half] = 1.0
        if n_fallback % 2 == 1:
            h0[half] = 0.5

    n = len(h0)
    target_sum = n / 2.0
    h0 = enforce_sum_and_bounds(h0, target_sum)

    # ------------------------------------------------------------------
    # 2. Compute the initial bound (use harness helper if available).
    # ------------------------------------------------------------------
    try:
        init_c5 = float(evaluate_erdos_solution(h0))
    except NameError:
        init_c5 = float(compute_c5_maxcorr(h0))

    best_h = h0.copy()
    best_c5 = init_c5

    # ------------------------------------------------------------------
    # 3. Quick local refinement with SLSQP (if time permits).
    # ------------------------------------------------------------------
    elapsed = time.time() - start_time
    time_left = budget_s - elapsed
    if time_left > 5.0:
        # Prepare global constraint data.
        global _target_sum
        _target_sum = target_sum

        bounds = [(0.0, 1.0)] * n
