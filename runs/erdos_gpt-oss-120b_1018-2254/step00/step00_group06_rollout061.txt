def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)

    # initial solution if provided
    try:
        init = np.array(initial_h_values, dtype=float)

        n = 100
        init = np.full(n, 0.5, dtype=float)

    target_sum = n / 2.0
    dx = 2.0 / n

    def compute_c5(h):
        return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

    # ensure feasibility of init
    init = np.clip(init, 0.0, 1.0)
    s = init.sum()
    if not np.isclose(s, target_sum):
        init = init * (target_sum / s)
        init = np.clip(init, 0.0, 1.0)
        diff = target_sum - init.sum()
        if abs(diff) > 1e-12:
            i = np.random.randint(n)
            init[i] = min(max(init[i] + diff, 0.0), 1.0)

    pop_size = 120
    pop = [init.copy()]

    # create a diverse initial population by random feasible transfers
    for _ in range(pop_size - 1):
        h = init.copy()
        moves = np.random.randint(5, 21)
        for __ in range(moves):
            i, j = np.random.choice(n, 2, replace=False)
            src, dst = (i, j) if np.random.rand() < 0.5 else (j, i)
            max_delta = min(h[src], 1.0 - h[dst])
            if max_delta <= 0.0:
                continue
            delta = np.random.rand() * max_delta
            h[src] -= delta
            h[dst] += delta
        pop.append(h)

    scores = [compute_c5(h) for h in pop]
    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]

    start = time.time()
    tournament_k = 3
    mutation_prob = 0.9
    generation = 0

    # Genetic algorithm loop
    while time.time() - start < budget_s * 0.9:
        new_pop = [best_h.copy()]  # elitism
        while len(new_pop) < pop_size:
            # tournament selection
            idxs = np.random.choice(pop_size, tournament_k, replace=False)
            parent1 = pop[int(idxs[np.argmin([scores[i] for i in idxs])])]
            idxs2 = np.random.choice(pop_size, tournament_k, replace=False)
            parent2 = pop[int(idxs2[np.argmin([scores[i] for i in idxs2])])]
            # arithmetic crossover (preserves sum)
            child = (parent1 + parent2) * 0.5
            # mutation by random feasible transfers
            if np.random.rand() < mutation_prob:
                n_moves = np.random.randint(1, 4)
                for _ in range(n_moves):
                    i, j = np.random.choice(n, 2, replace=False)
                    src, dst = (i, j) if np.random.rand() < 0.5 else (j, i)
                    max_delta = min(child[src], 1.0 - child[dst])
                    if max_delta <= 0.0:

                    child[src] -= delta

        generation += 1

    # Local hill‑climbing refinement
    h_cur = best_h.copy()
    cur_score = best_score
    while time.time() - start < budget_s * 0.99:
        i, j = np.random.choice(n, 2, replace=False)
        max_delta = min(h_cur[i], 1.0 - h_cur[j])
        if max_delta <= 0.0:

        h_cur[j] += delta

            if new_score < best_score:

            h_cur[j] -= delta

    diff = target_sum - best_h.sum()

        if abs(remaining) > 1e-12:
            best_h += remaining / n
            best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_score, n
