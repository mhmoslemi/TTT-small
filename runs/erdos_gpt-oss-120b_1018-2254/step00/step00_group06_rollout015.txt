def repair_candidate(h, target_sum):
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur == 0:
        # just assign uniform values that sum to target
        h = np.full_like(h, target_sum / len(h))
        return h
    # scale to target sum while maintaining bounds using iterative approach
    # simple linear scaling and then clip again until sum approx target
    # Because scaling may produce out-of-bound values; we can adopt bouncing
    # Simple iterative: iterate up to 5 times:
    for _ in range(5):
        cur = h.sum()
        # adjust scaling factor
        scale = target_sum / cur if cur != 0 else 0
        h = h * scale
        h = np.clip(h, 0.0, 1.0)
    # final fix small residual
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) > 1e-8:

import time
import numpy as np

def _compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # correlation h * (1-h) as described
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(conv) * dx)


def _repair_candidate(h, target_sum):
    """Clip to [0,1] and enforce sum == target_sum (within floating tolerance)."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur == 0.0:
        h[:] = target_sum / h.size
        return h

    # Scale to hit the target sum, then clip again
    scale = target_sum / cur
    h *= scale
    h = np.clip(h, 0.0, 1.0)

    # After clipping the sum may be off; correct proportionally
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        if diff > 0:
            slack = 1.0 - h
            total_slack = slack.sum()
            if total_slack > 0:
                h += slack * (diff / total_slack)
        else:
            capacity = h
            total_cap = capacity.sum()
            if total_cap > 0:
                h -= capacity * ((-diff) / total_cap)
        h = np.clip(h, 0.0, 1.0)

    # Final tiny adjustment on a single feasible index
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        idxs = np.where(((diff > 0) & (h < 1.0)) | ((diff < 0) & (h > 0.0)))[0]
        if idxs.size:
            i = idxs[0]
            if diff > 0:
                h[i] += min(diff, 1.0 - h[i])
            else:
                h[i] -= min(-diff, h[i])
    return h


def _tournament(pop, fitness, k, rng):
    """Return a member of pop selected by k‑size tournament (lower fitness wins)."""
    idxs = rng.choice(len(pop), size=k, replace=False)
    best = idxs[0]
    best_fit = fitness[best]
    for i in idxs[1:]:
        if fitness[i] < best_fit:
            best = i
            best_fit = fitness[i]
    return pop[best]


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # Determine problem size -------------------------------------------------
    try:

    target_sum = n_points / 2.0

    # GA parameters ---------------------------------------------------------
    pop_size = 80
    pc = 0.7            # crossover probability
    pm = 0.3            # overall mutation probability per child
    gene_mut_rate = 0.02

    target_c5 = 0.3808

    # ----------------------------------------------------------------------
    # Initialise population
    pop = []
    if init is not None:

        pop.append(init)
    while len(pop) < pop_size:
        h = rng.random(n_points)
        h = _repair_candidate(h, target_sum)
        pop.append(h)

    # Evaluate initial fitness

    best_idx = int(np.argmin(fitness))

    best_f = fitness[best_idx]

    while True:

        if elapsed > budget_s * 0.95:

        if best_f <= target_c5:
