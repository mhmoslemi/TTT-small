    if s == 0:
        # fill with target sum uniformly
        h = np.full(n, target_sum / n)
    else:
        h = h * (target_sum / s)
        # after scaling may violate [0,1]; clip and then redistribute remainder
        h = np.clip(h, 0.0, 1.0)
        # fix sum
        diff = target_sum - h.sum()
        # distribute diff by adding to positions with capacity
        # simple loop over indices
        i = 0
        while abs(diff) > 1e-12 and i < n:
            cap = (1.0 - h[i]) if diff > 0 else h[i]
            if cap > 0:
                add = min(abs(diff), cap)
                add = add if diff > 0 else -add
                h[i] += add
                diff -= add
            i += 1
        # If still diff left due to rounding, adjust first element
        if abs(diff) > 1e-12:
            h[0] += diff
    # Now h should be feasible exactly within float tolerance
    dx = 2.0 / n
    def compute_c5(vec):
        corr = np.correlate(vec, 1.0 - vec, mode='full')
        return np.max(corr) * dx
    cur_h = h.copy()
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    start = time.time()
    end_time = start + budget_s - 0.2  # small margin
    # annealing parameters
    T0 = 0.02  # initial temperature
    T_end = 1e-5
    max_transfer_initial = 0.1
    # main loop
    while time.time() < end_time:
        elapsed = time.time() - start
        frac = elapsed / (budget_s - 0.2)
        T = T0 * (T_end / T0) ** frac
        max_transfer = max_transfer_initial * (1.0 - frac) + 0.001
        # pick two distinct indices
        i, j = np.random.choice(n, 2, replace=False)
        # store original
        hi, hj = cur_h[i], cur_h[j]
        # decide direction of transfer
        # we want to move mass from the larger to the smaller to possibly flatten
        if hi > hj:
            # move from i to j
            delta_max = min(max_transfer, hi, 1.0 - hj)
            if delta_max <= 0:
                continue
            delta = np.random.rand() * delta_max
            cur_h[i] -= delta
            cur_h[j] += delta
        else:
            # move from j to i
            delta_max = min(max_transfer, hj, 1.0 - hi)
            if delta_max <= 0:
                continue
            delta = np.random.rand() * delta_max
            cur_h[i] += delta
            cur_h[j] -= delta
        new_c5 = compute_c5(cur_h)
        # acceptance
        if new_c5 < cur_c5 or np.random.rand() < np.exp(-(new_c5 - cur_c5) / (T + 1e-12)):
            cur_c5 = new_c5
            # update best
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i], cur_h[j] = hi, hj
        # optional occasional reheat?
        # continue
    # final check: ensure best_h meets constraints (clipping)
    best_h = np.clip(best_h, 0.0, 1.0)
    # enforce sum exactly target (small adjustment)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-9:
        # adjust first element within bounds
        for idx in range(n):
            cap = (1.0 - best_h[idx]) if diff > 0 else best_h[idx]
            if cap > 0:
                adj = min(abs(diff), cap)
                adj = adj if diff > 0 else -adj
                best_h[idx] += adj
                diff -= adj
                if abs(diff) < 1e-12:
                    break
    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, n)

import random

    def project_to_feasible(vec, n, target_sum):
        """Clip to [0,1] and adjust sum to target_sum."""
        v = np.clip(vec, 0.0, 1.0)
        s = v.sum()
        if s == 0.0:
            return np.full(n, target_sum / n)
        # scale to match target sum, then clip again
        v = v * (target_sum / s)
        v = np.clip(v, 0.0, 1.0)
        # fix any remaining sum error by distributing it greedily
        diff = target_sum - v.sum()
