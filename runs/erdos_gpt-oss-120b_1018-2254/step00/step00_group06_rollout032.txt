def gradient_max_overlap(h):
    O = np.correlate(h, 1-h, mode='full')
    # Not scaling dx yet because gradient doesn't need scaling (dx factor is constant)
    idx_max = np.argmax(O)
    shift = idx_max - (len(h) - 1)
    n = len(h)
    g = np.zeros_like(h)
    if shift >= 0:
        # Overlap length = n - shift
        # indices i = 0..n-shift-1
        i = slice(0, n-shift)
        j = slice(shift, n)
        g[i] += 1 - h[j]  # 1 - h_{i+shift}
        g[j] += -h[i]     # -h_i
    else:
        s = -shift
        # Overlap indices i = s..n-1

        j = slice(0, n-s)

        g[j] += -h[i]
    return g, O[idx_max] * (2.0/n)  # compute scaled C5

def project_onto_box_sum(v, s, tol=1e-8, max_iter=100):
    n = v.shape[0]
    low = -np.max(v) - 1.0
    high = 1.0 - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (low + high) / 2.0
        proj = np.clip(v + lam, 0, 1)
        total = proj.sum()
        if abs(total - s) <= tol:
            return proj
        if total < s:
            low = lam
        else:
            high = lam
    # final approx
    proj = np.clip(v + (low + high) / 2.0, 0, 1)
    # adjust residual
    residual = s - proj.sum()
    if abs(residual) > tol:
        # find indices not at bounds
        free = (proj > 1e-9) & (proj < 1 - 1e-9)
        if free.any():
            # distribute residual equally
            proj[free] += residual / free.sum()
            proj = np.clip(proj, 0, 1)
        else:
            # adjust first element
            proj[0] = np.clip(proj[0] + residual, 0, 1)
    return proj

def gradient_max_overlap(h):
    O = np.correlate(h, 1 - h, mode='full')
    idx_max = O.argmax()
    n = h.shape[0]
    shift = idx_max - (n - 1)
    g = np.zeros_like(h)
    if shift >= 0:
        # overlapping length n - shift
        L = n - shift
        # indices
        g[:L] += 1 - h[shift:]  # for i
        g[shift:] += -h[:L]     # for j
    else:
        s = -shift
        L = n - s
        # i = s..n-1, j = 0..L-1
        g[s:] += 1 - h[:L]
        g[:L] += -h[s:]
    return g, O[idx_max]  # raw O, not scaled

import time
import numpy as np

def _project_onto_box_sum(v, target_sum, tol=1e-9, max_iter=100):
    """Project vector v onto the set {x | 0<=x<=1, sum(x)=target_sum}."""
    n = v.shape[0]
    # initial bounds for lambda
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target_sum) <= tol:
            return proj
        if s < target_sum:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    # tiny residual correction
    residual = target_sum - proj.sum()
    if abs(residual) > tol:
        free = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        if free.any():
            proj[free] += residual / free.sum()
            proj = np.clip(proj, 0.0, 1.0)
        else:
            proj[0] = np.clip(proj[0] + residual, 0.0, 1.0)
    return proj

def _compute_c5(h, dx):
    """Return the overlap constant C5 for a given h and grid spacing dx."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def _gradient_max_overlap(h):
    """
    Compute gradient of the raw (unscaled) max overlap term.
    Returns (gradient, raw_max_overlap_value).
    """
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = corr.argmax()
    n = h.shape[0]
    shift = idx - (n - 1)          # positive => right shift
    g = np.zeros_like(h)

    if shift >= 0:
        L = n - shift
        # first part: i from 0..L-1
        g[:L] += 1.0 - h[shift:]
        # second part: j = i+shift
        g[shift:] += -h[:L]
    else:
        s = -shift
        L = n - s
        g[s:] += 1.0 - h[:L]
        g[:L] += -h[s:]

    return g, corr[idx]

    # Work while enough time remains (leave ~30% for refinement)
    while time.time() - start_time < budget_s * 0.3:
        idx_one = rng.choice(n, size=n // 2, replace=False)
        h_bin = np.zeros(n, dtype=float)
        h_bin[idx_one] = 1.0
        c5_val = _compute_c5(h_bin, dx)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_bin.copy()
        # occasional interpolation of the current best to create a new start
        if rng.random() < 0.01:
            h = best_h.copy()
        else:
            # small random perturbation around current best
            eps = rng.normal(scale=0.02, size=n)
            h = best_h + eps
            h = _project_onto_box_sum(h, target_sum)

    # ------------------------------------------------------------------
    # Phase 2: gradient‑descent refinement
    # ------------------------------------------------------------------
    h = best_h.copy()
    step_alpha = 0.3   # base step size factor
    iteration = 0

    def time_left():
        return budget_s - (time.time() - start_time)

    while time_left() > 0.5:
        # compute gradient of the raw max overlap term
        g_raw, raw_max = _gradient_max_overlap(h)
        curr_c5 = raw_max * dx
        g = g_raw * dx
        norm_g = np.linalg.norm(g)
        if norm_g < 1e-12:
            # stagnated – restart randomly
            v = rng.random(n)
            h = _project_onto_box_sum(v, target_sum)
            continue

        # descent direction (unit)
        direction = -g / (norm_g + 1e-12)
        improved = False

        # simple backtracking line search
        for factor in (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125):
            step = step_alpha * factor * direction
            h_candidate = h + step
            h_candidate = _project_onto_box_sum(h_candidate, target_sum)
            cand_c5 = _compute_c5(h_candidate, dx)
            if cand_c5 < curr_c5 - 1e-12:
                h = h_candidate
                curr_c5 = cand_c5
                improved = True
                if cand_c5 < best_c5:
                    best_c5 = cand_c5
                    best_h = h_candidate.copy()
                # modestly increase step size after a successful move
                step_alpha = min(step_alpha * 1.05, 1.0)
                break

        if not improved:
            # reduce the base step size when no progress
            step_alpha *= 0.7
            if step_alpha < 1e-4:
                # occasional random shake to escape plateaus
                eps = rng.normal(scale=0.01, size=n)
                h_candidate = h + eps
                h_candidate = _project_onto_box_sum(h_candidate, target_sum)
                cand_c5 = _compute_c5(h_candidate, dx)
                if cand_c5 < best_c5:
                    best_c5 = cand_c5
                    best_h = h_candidate.copy()
                    h = h_candidate
                else:
                    # full random restart
                    v = rng.random(n)
                    h = _project_onto_box_sum(v, target_sum)
                    step_alpha = 0.3

        # sporadic global perturbations
        if rng.random() < 0.005:
            eps = rng.normal(scale=0.005, size=n)
            h_candidate = h + eps
            h_candidate = _project_onto_box_sum(h_candidate, target_sum)
            cand_c5 = _compute_c5(h_candidate, dx)
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = h_candidate.copy()
                h = h_candidate

        # periodic full restart to explore new basins
        if iteration % 500 == 0 and iteration > 0:
            v = rng.random(n)
            h = _project_onto_box_sum(v, target_sum)
            step_alpha = 0.3

        iteration += 1

    # Final evaluation (safety)
    best_c5 = _compute_c5(best_h, dx)

    return best_h, best_c5, n
