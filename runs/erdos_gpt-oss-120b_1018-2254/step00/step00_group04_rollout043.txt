import time
import numpy as np

def compute_c5(h):
    """Return the C5 value for a given step function h (vector)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))

def max_shift_and_corr(h):
    """Return the shift (as integer index offset) that gives the maximal overlap
    and the corresponding correlation value."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)
    return shift, float(corr[idx])

def gradient_for_shift(h, s):
    """Subgradient of the objective for a fixed shift s."""
    n = h.shape[0]
    dx = 2.0 / n
    if s >= 0:
        # term1 = 1 - h[i+s] where defined, else 1
        term1 = np.empty(n)
        term1[: n - s] = 1.0 - h[s:]
        term1[n - s :] = 1.0
        # term2 = h[i-s] where defined, else 0
        term2 = np.empty(n)
        term2[:s] = 0.0
        term2[s:] = h[: n - s]
    else:
        k = -s
        term1 = np.empty(n)
        term1[:k] = 1.0
        term1[k:] = 1.0 - h[: n - k]
        term2 = np.empty(n)
        term2[: n - k] = h[k:]
        term2[n - k :] = 0.0
    return dx * (term1 - term2)

def project_onto_box_sum(v, target):
    """Project vector v onto the set {0 <= x_i <= 1, sum x_i = target}
    using a simple bisection on the Lagrange multiplier."""
    lo, hi = -1.0, 2.0  # bounds wide enough to contain the solution
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize the step function h for the Erdős C5 overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    # ------------------------------------------------------------------
    # Determine problem dimension
    # ------------------------------------------------------------------
    init = globals().get("initial_h_values", None)
    if init is not None:
        try:
            init_arr = np.asarray(init, dtype=float).ravel()
            n0 = init_arr.size
        except Exception:
            init_arr = None
            n0 = None
    else:
        init_arr = None
        n0 = None

    # Choose a modest size; larger n reduces the final bound via the 2/n factor.
    n = 200
    if n0 is not None and n0 <= 500:
        n = max(100, n0)          # respect a provided reasonable size
    if n % 2 == 1:
        n += 1                    # enforce even n for exact sum constraint
    weight = n // 2               # required sum of h

    # ------------------------------------------------------------------
    # Initial guess for h
    # ------------------------------------------------------------------
    if init_arr is not None and init_arr.shape[0] == n:
        h = np.clip(init_arr, 0.0, 1.0)
        h = project_onto_box_sum(h, weight)
    else:
        # Smooth sinusoidal start: h_i = sin^2(pi * (i+0.5) / n)
        x = np.arange(n) + 0.5
        h = np.sin(np.pi * x / n) ** 2
        h = np.clip(h, 0.0, 1.0)
        h = project_onto_box_sum(h, weight)

    best_h = h.copy()
