def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_onto_constraints(v, S):
    # projection onto [0,1] and sum=S using bisection
    low = -10.0
    high = 10.0
    # ensure that at low sum >= S, at high sum <= S
    # adapt bounds until condition satisfied.
    for _ in range(100):
        # shift
        mid = (low + high) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        sum_x = x.sum()
        if sum_x > S:
            low = mid
        else:
            high = mid
    return np.clip(v - (low + high) / 2.0, 0.0, 1.0)

def random_feasible(n, S, rng):

import numpy as np
import time
import math

def compute_c5(h):
    """Direct evaluation of C5 for a given sample vector."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def evaluate_h(h):
    """Use the supplied helper if available, else fall back."""
    try:
        return evaluate_erdos_solution(h)
    except NameError:
        return compute_c5(h)

def project_onto_sum_bounds(v, target_sum):
    """
    Euclidean projection of vector v onto the set
    {x | 0 <= x_i <= 1, sum_i x_i = target_sum}.
    Implemented via a simple bisection on the offset lambda.
    """
    low, high = -10.0, 10.0     # guarantees sum(high) = 0, sum(low) = n
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            low = mid
        else:
            high = mid
    x = np.clip(v - (low + high) * 0.5, 0.0, 1.0)

    # Minor correction for tiny numerical drift
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        free = np.where((x > 0.0) & (x < 1.0))[0]
        if free.size == 0:
            free = np.arange(x.shape[0])
        i = free[0]
        x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def random_feasible(n, target_sum, rng):
    """Generate a random point in the feasible polytope."""
    v = rng.random(n)               # uniform in [0,1)
    return project_onto_sum_bounds(v, target_sum)

def perturb(base, sigma, rng):
    """Add Gaussian noise."""
    return base + sigma * rng.standard_normal(base.shape)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # Determine number of points and initial vector
    try:
        init = np.array(initial_h_values, dtype=float)
        n = init.shape[0]
        current = init.copy()
    except NameError:
        n = 47
        current = np.full(n, 0.5, dtype=float)

    target_sum = n / 2.0
    current = project_onto_sum_bounds(current, target_sum)

    best_h = current.copy()
    best_score = evaluate_h(best_h)
    current_score = best_score

    start = time.time()
    Tmax = 0.05               # initial temperature
    sigma_max = 0.30
    sigma_min = 0.005

    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.95:

        cand_score = evaluate_h(candidate)

                best_score = cand_score
