def random_feasible(n, target_sum, rng):
    # generate random in [0,1] then project onto simplex with box.
    # one simple way: generate uniform [0,1] vector, then linearly scale to have sum target_sum.
    # But after scaling, values may be outside [0,1]; we can then clip and renorm iteratively.
    # We'll implement simple iterative projection: use while loop up to some iterations.
    x = rng.random(n)
    # scale to target sum
    x = x * target_sum / x.sum()
    # clip to bounds
    x = np.clip(x, 0.0, 1.0)
    # iterative correction
    for _ in range(10):
        s = x.sum()
        if abs(s - target_sum) < 1e-9:
            break
        diff = target_sum - s
        free = (x > 0) & (x < 1)
        nfree = free.sum()
        if nfree == 0:
            # adjust all with smallest magnitude diff fix
            # distribute diff to all entries (may exceed bound)
            x = x + diff / n
            x = np.clip(x, 0.0, 1.0)
        else:
            adjust = diff / nfree
            x[free] = np.clip(x[free] + adjust, 0.0, 1.0)
    # final adjust
    s = x.sum()
    if abs(s - target_sum) > 1e-8:
        # assign diff to first free
        diff = target_sum - s
        for i in range(n):
            if 0 <= x[i] + diff <= 1:
                x[i] += diff
                break
    return x
