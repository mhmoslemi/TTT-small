def project_onto_sum_and_bounds(v, target, lower=0., upper=1., tol=1e-9):
    lo = -np.max(v) - 1  # lower bound lambda may be negative large
    hi = np.max(v) + 1   # high bound lambda
    for _ in range(60):
        lam = (lo + hi) / 2
        w = np.clip(v - lam, lower, upper)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo+hi)/2, lower, upper)

import time
import numpy as np


def _project_to_sum_and_bounds(v, target, lower=0.0, upper=1.0, max_iter=60):
    """Projection of v onto the box [lower,upper] with a fixed sum `target`."""
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, lower, upper)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, lower, upper)


def _c5_and_idx(arr, n_points, nfft):
    """Return (C5 value, index of the worst shift) for the given step function."""
    h_fft = np.fft.rfft(arr, n=nfft)
    g_rev = (1.0 - arr)[::-1]
    g_fft = np.fft.rfft(g_rev, n=nfft)
    conv = np.fft.irfft(h_fft * g_fft, n=nfft)[: 2 * n_points - 1]
    idx = int(conv.argmax())
    dx = 2.0 / n_points
    return conv[idx] * dx, idx


def _targeted_move(h, shift):
    """Move a small amount of mass in the direction that reduces the worst shift."""
    n = h.shape[0]
    if shift > 0:
        max_i = n - shift
        feasible = np.where((h[:max_i] > 1e-12) & (h[shift:] < 1.0 - 1e-12))[0]
        if feasible.size == 0:
            return None
        i = np.random.choice(feasible)
        delta_max = min(h[i], 1.0 - h[i + shift])
        if delta_max <= 0:
            return None
        delta = np.random.rand() * delta_max
        new_h = h.copy()
        new_h[i] -= delta
        new_h[i + shift] += delta
        return new_h
    if shift < 0:
        d = -shift
        feasible = np.where((h[d:] > 1e-12) & (h[: n - d] < 1.0 - 1e-12))[0]
        if feasible.size == 0:
            return None
        offset = np.random.choice(feasible)
        i = offset + d
        delta_max = min(h[i], 1.0 - h[i - d])
        if delta_max <= 0:
            return None
        delta = np.random.rand() * delta_max
        new_h = h.copy()
        new_h[i] -= delta
        new_h[i - d] += delta
        return new_h
    return None  # shift == 0


def _random_swap_move(h):
    """Swap a random amount of mass between two coordinates while preserving the sum."""
    n = h.shape[0]
    for _ in range(5):
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        if np.random.rand() < 0.5:
            delta_max = min(1.0 - h[i], h[j])
            if delta_max <= 0:
                continue
            delta = np.random.rand() * delta_max
            new_h = h.copy()
            new_h[i] += delta
            new_h[j] -= delta
            return new_h
        else:
            delta_max = min(1.0 - h[j], h[i])
            if delta_max <= 0:
                continue
            delta = np.random.rand() * delta_max
            new_h = h.copy()
            new_h[i] -= delta
            new_h[j] += delta
            return new_h
    return None


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # ---- problem size -------------------------------------------------
    n_points = 200                     # < 1000 as requested
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # ---- initial construction -----------------------------------------
    if "initial_h_values" in globals():
        init = initial_h_values
        n0 = len(init)
        x0 = np.arange(n0) * (2.0 / n0)
        x = np.arange(n_points) * dx
        h = np.interp(x, x0, init)
    else:
        h = np.random.rand(n_points)

    h = _project_to_sum_and_bounds(h, target_sum)

    # ---- pre‑computed FFT size ----------------------------------------
    nfft = 1 << ((2 * n_points - 1).bit_length())

    # ---- current / best states ----------------------------------------
    cur_h = h.copy()
    cur_c5, worst_idx = _c5_and_idx(cur_h, n_points, nfft)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    start = time.perf_counter()
    time_limit = budget_s * 0.97

    no_improve = 0
    while time.perf_counter() - start < time_limit:
        shift = worst_idx - (n_points - 1)

        # try a targeted move first
        cand = _targeted_move(cur_h, shift)
        if cand is None:
            cand = _random_swap_move(cur_h)
        if cand is None:
            continue

        new_c5, new_worst = _c5_and_idx(cand, n_points, nfft)

        # simple greedy acceptance with a tiny annealing tail
        if new_c5 < cur_c5:
            accept = True
        else:
            elapsed = time.perf_counter() - start
            temp = 0.01 * max(1e-12, 1.0 - elapsed / budget_s)
            accept = np.random.rand() < np.exp(-(new_c5 - cur_c5) / temp)

        if accept:
            cur_h = cand
            cur_c5 = new_c5
            worst_idx = new_worst
            no_improve = 0 if new_c5 < cur_c5 else no_improve + 1
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cand.copy()
        else:
            no_improve += 1

        # occasional perturbation if stuck
        if no_improve >= 5000:
            perturb = best_h + np.random.normal(scale=0.02, size=n_points)
            perturb = np.clip(perturb, 0.0, 1.0)
            cur_h = _project_to_sum_and_bounds(perturb, target_sum)
            cur_c5, worst_idx = _c5_and_idx(cur_h, n_points, nfft)
            no_improve = 0

    # ---- final projection & evaluation --------------------------------
    best_h = _project_to_sum_and_bounds(best_h, target_sum)

    try:
        val = evaluate_erdos_solution(best_h, n_points)
        if isinstance(val, (list, tuple, np.ndarray)):
            best_c5 = float(val[0])
        else:
            best_c5 = float(val)
    except Exception:
        best_c5, _ = _c5_and_idx(best_h, n_points, nfft)

    return best_h, float(best_c5), n_points
