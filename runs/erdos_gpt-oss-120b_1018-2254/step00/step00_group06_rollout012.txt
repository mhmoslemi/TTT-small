import time
import numpy as np

def _project_to_bounds_and_sum(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project vector v onto the intersection of the box [0,1]^n and the affine
    hyperplane sum(x)=target_sum using a simple bisection on a scalar offset.
    """
    # Clip to box first
    lo, hi = -2.0, 2.0  # wide enough bounds for the offset
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            hi = lam
        else:
            lo = lam
    x = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # Fix tiny residual sum error
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        free = np.where((x > 1e-12) & (x < 1.0 - 1e-12))[0]
        if free.size == 0:
            # pick any index not at its bound
            idx = 0 if diff > 0 else -1
        else:
            idx = free[0]
        x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
    return x

def _compute_c5(h: np.ndarray) -> float:
    """Return the C5 value for a discrete h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def _gradient_wrt_h(h: np.ndarray, shift: int) -> np.ndarray:
    """
    Sub‑gradient of the worst overlap term c_k = Σ h_i(1‑h_{i+k}) w.r.t. h,
    where shift = k (positive ⇒ right shift).
    """
    n = h.shape[0]
    g = np.ones(n, dtype=float)

    if shift >= 0:
        k = shift
        if k < n:
            # term -h_{i+k}
            g[: n - k] -= h[k:]
            # term -h_{i‑k}
            if k > 0:
                g[k:] -= h[: n - k]
    else:
        kp = -shift
        if kp < n:
            # term -h_{i+k} = -h_{i‑kp}
            g[kp:] -= h[: n - kp]
            # term -h_{i‑k} = -h_{i+kp}
            g[: n - kp] -= h[kp:]
    return g

    """
    Optimize a step function h on [0,2] with ∫h=1, 0≤h≤1 to minimize C5.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- initialise ---------------------------------------------------------
    if "initial_h_values" in globals():
        arr = np.asarray(initial_h_values, dtype=float).ravel()
        n = arr.size
        h = arr.copy()
    else:
        n = 95
        h = np.full(n, 0.5, dtype=float)

    target_sum = n / 2.0
    h = _project_to_bounds_and_sum(h, target_sum)

    best_h = h.copy()
    best_c5 = _compute_c5(best_h)

    # ---- optimisation parameters -------------------------------------------
    max_no_improve = 5000          # iterations before a random restart
    no_improve = 0

    init_step = 0.20               # step size at start
    final_step = 0.005             # step size at end

    # ------------------------------------------------------------------------
    while time.time() - start < budget_s:
        # evaluate current worst lag
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        max_idx = int(np.argmax(corr))
        cur_c5 = (2.0 / n) * float(corr[max_idx])

        # update best record if needed
        if cur_c5 < best_c5 - 1e-12:
