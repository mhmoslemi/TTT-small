import numpy as np

def _compute_c5(vec: np.ndarray) -> float:
    """Return the overlap bound C5 for a given discretized function."""
    corr = np.correlate(vec, 1.0 - vec, mode="full")
    return float(np.max(corr) * (2.0 / vec.size))

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] (discretised) to minimise the
    Erdős overlap constant C5.

    Returns
    -------
    h_values : list[float]
        Optimised samples of h.
    c5_bound : float
        The achieved upper bound C5.
    n_points : int
        Number of discretisation points.
    """

    # Initialise h
    # ----------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Default: half ones, half zeros, shuffled
        n = 100
        h = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)])
        rng.shuffle(h)

    n = h.size
    target_sum = n / 2.0

    # Force exact sum while keeping bounds
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        diff = target_sum - cur_sum
        # Try to patch a single entry
        for idx in range(n):
            if 0.0 <= h[idx] + diff <= 1.0:
                h[idx] += diff
                break
        else:
            # Scale globally (rare)
            factor = target_sum / cur_sum
            h = np.clip(h * factor, 0.0, 1.0)
            diff = target_sum - h.sum()
            if abs(diff) > 1e-12:
                for idx in range(n):
                    if 0.0 <= h[idx] + diff <= 1.0:
                        h[idx] += diff
                        break

    h = np.clip(h, 0.0, 1.0)

    # Hill‑climbing search
    # ----------------------------------------------------------------------
    best_h = h.copy()

        idxs = np.where((best_h + diff >= 0.0) & (best_h + diff <= 1.0))[0]
        if idxs.size > 0:
            best_h[idxs[0]] += diff
        else:
            # fallback scaling (should be very rare)
            factor = target_sum / best_h.sum()
            best_h = np.clip(best_h * factor, 0.0, 1.0)
            diff = target_sum - best_h.sum()
            if abs(diff) > 1e-12:
                best_h[0] = min(max(best_h[0] + diff, 0.0), 1.0)

    final_c5 = _compute_c5(best_h)

    return best_h.tolist(), float(final_c5), int(n)
