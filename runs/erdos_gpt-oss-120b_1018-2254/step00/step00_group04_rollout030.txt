def project_to_bounds_sum(v, target):
    # Project onto [0,1] and sum=target.
    # Use algorithm from Duchi et al. projection onto simplex with upper bound.
    # We can use cvxopt or simpler iterative waterfilling.
    # We'll implement simple iterative:
    v = np.clip(v, 0, 1)
    # while sum not close
    for _ in range(100):
        total = np.sum(v)
        diff = target - total
        if abs(diff) < 1e-10:
            break
        # Determine free indices where 0 < v < 1
        free = (v > 0) & (v < 1)
        free_count = np.count_nonzero(free)
        if free_count == 0:
            # can't adjust. distribute diff uniformly (may violate bounds) but break
            v += diff / len(v)
            v = np.clip(v, 0, 1)
            continue
        delta = diff / free_count
        # add delta to free entries, then clip (some may hit bounds)
        v[free] += delta
        v = np.clip(v, 0, 1)
    return v

import numpy as np

def _evaluate_c5(h: np.ndarray, dx: float) -> float:
    """Return max overlap C5 for a step function sampled by h."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * dx


def _project_to_bounds_sum(v: np.ndarray, target: float, max_iter: int = 200) -> np.ndarray:
    """Project v onto the box [0,1] while enforcing sum(v)=target."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        total = v.sum()
        diff = target - total
        if abs(diff) < 1e-12:
            break
        free = (v > 0.0) & (v < 1.0)
        cnt = int(free.sum())

            v = np.clip(v, 0.0, 1.0)

        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    return v
