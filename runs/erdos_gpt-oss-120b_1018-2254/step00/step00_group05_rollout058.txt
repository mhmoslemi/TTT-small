import time, math, random
import numpy as np

def _compute_corr_raw(h):
    """Raw correlation values (without dx scaling)."""
    return np.correlate(h, 1.0 - h, mode='full')

def _c5_from_raw(raw_max, n):
    """Convert raw max correlation to C5 using dx = 2/n."""
    return raw_max * (2.0 / n)

def _project_to_sum(v, target, eps=1e-12):
    """Project v onto [0,1]^n with exact sum = target via water‑filling."""
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def _initial_feasible(n):
    """Generate a feasible start vector when no parent is provided."""
    # start from a smooth sine shape, then project to the simplex
    x = np.linspace(0.0, 2.0, n)
    vec = np.sin(np.pi * x / 2.0) ** 2
    return _project_to_sum(vec, n / 2.0)

def _prepare_candidates(arr):
    """Return a list of feasible candidate vectors based on a base array."""
    n = arr.size
    target = n / 2.0
    cand = []
    # raw candidate (maybe already feasible)
    cand.append(_project_to_sum(arr, target))
    # sine shape
    x = np.linspace(0.0, 2.0, n)
    cand.append(_project_to_sum(np.sin(np.pi * x / 2.0) ** 2, target))
    # cosine‑squared shape
    cand.append(_project_to_sum((1.0 + np.cos(np.pi * x / 2.0)) / 2.0, target))
    # triangular shape
    tri = np.where(x < 1.0, x, 2.0 - x)
    cand.append(_project_to_sum(tri, target))
    # binary half‑ones shuffled
    bin_vec = np.zeros(n)
    bin_vec[:n // 2] = 1.0
    np.random.shuffle(bin_vec)
    cand.append(_project_to_sum(bin_vec, target))
    return cand

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] with ∫h=1 that minimizes C₅.

    Returns:
        (h_values, c5_bound, n_points)
    """
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Initialise h
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64).copy()
        n = h0.size
        h0 = _project_to_sum(h0, n / 2.0)
    else:
        n = 200                     # reasonable size when no parent is given
        h0 = _initial_feasible(n)

    # ------------------------------------------------------------------
    # Choose the best of a few analytic starters
    # ------------------------------------------------------------------
    candidates = _prepare_candidates(h0)
    best_h = None
    best_raw = np.inf
    for cand in candidates:
        raw = _compute_corr_raw(cand).max()
        if raw < best_raw:
            best_raw = raw
            best_h = cand.copy()
    h = best_h.copy()
    best_raw = _compute_corr_raw(h).max()
    best_h = h.copy()

    # ------------------------------------------------------------------
    # Simulated‑annealing / targeted improvement loop
    # ------------------------------------------------------------------
    start = time.time()
    total_time = budget_s * 0.95  # leave a small safety margin

    # pre‑compute constants
    target_sum = n / 2.0
    max_iter = 10 ** 7  # generous upper bound; time check will stop earlier

    for iteration in range(max_iter):
        now = time.time()
        elapsed = now - start
        if elapsed >= total_time:
            break

        # temperature schedule (linear cooling)
        t_frac = elapsed / total_time
        temp = (1.0 - t_frac) + 1e-4  # temperature in [~1,~1e-4]

        # compute current raw max correlation and the worst shift
        corr = _compute_corr_raw(h)
        raw_max = corr.max()
        worst_idx = int(np.argmax(corr))
        offset = worst_idx - (n - 1)          # integer shift (may be negative)

        # ------------------------------------------------------------------

                accept = False
                if new_raw < best_raw:
                    accept = True
                else:

                        accept = True

                    if new_raw < best_raw:
                        best_raw = new_raw
                        best_h = h_new.copy()

    # ------------------------------------------------------------------
    # Final clean‑up: ensure exact feasibility
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_to_sum(best_h, target_sum)
    final_raw = _compute_corr_raw(best_h).max()
    final_c5 = _c5_from_raw(final_raw, n)

    return best_h, final_c5, n
