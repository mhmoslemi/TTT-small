def project_onto_constraints(v, target):
    # Clip to [0,1] after shift with lambda
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(50):
        mid = (lo + hi)/2
        w = np.clip(v - mid, 0, 1)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v - (lo+hi)/2, 0, 1)
    # Ensure sum close to target (due to rounding), adjust small remainder
    # if needed, distribute diff across non-binding entries
    diff = target - w.sum()

            w = np.clip(w, 0.0, 1.0)
        else:
            # Uniform adjustment
            w += diff / n
            w = np.clip(w, 0.0, 1.0)
    return w

import numpy as np
import time
import random

def shift_forward(h, k):
    """Return array a where a[i] = h[i+k] if inside bounds else 0."""
    n = h.shape[0]
    if k >= 0:
        if k >= n:
            return np.zeros_like(h)
        out = np.empty_like(h)
        out[: n - k] = h[k:]
        out[n - k :] = 0.0
        return out
    else:
        m = -k
        if m >= n:
            return np.zeros_like(h)
        out = np.empty_like(h)
        out[m:] = h[: n - m]
        out[:m] = 0.0
        return out


def compute_c5(h, dx):
    """Maximum overlap integral C5 for a discrete h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def project_onto_constraints(v, target):
    """Project v onto {0<=w_i<=1, sum w_i = target} (Euclidean)."""
    n = v.shape[0]
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

    diff = target - w.sum()
    if abs(diff) > 1e-12:
        mask = (w > 1e-12) & (w < 1.0 - 1e-12)
        if mask.any():
            w[mask] += diff / mask.sum()
            w = np.clip(w, 0.0, 1.0)
        else:
            w += diff / n
            w = np.clip(w, 0.0, 1.0)
    return w


def subgradient(h):
    """Subgradient of max_k C(k) at current h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = np.argmax(corr)
    k = max_idx - (n - 1)
    hp = shift_forward(h, k)
    hm = shift_forward(h, -k)
    return 1.0 - (hp + hm)


def random_feasible(n, target):
    """Generate a random feasible h (0<=h_i<=1, sum = target)."""
    v = np.random.rand(n)
    return project_onto_constraints(v, target)


def local_search(initial_h, target, dx, max_iters=2000, max_time=None):
    """Projected subgradient descent from initial_h."""
    h = initial_h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, dx)
    start = time.time()
    for it in range(max_iters):
        if max_time is not None and (time.time() - start) > max_time:
            break
        grad = subgradient(h)
        step = 0.2 / np.sqrt(it + 1.0)  # diminishing step
        h = project_onto_constraints(h - step * grad, target)
        c = compute_c5(h, dx)
        if c < best_c:
            best_c = c
            best_h = h.copy()
            if best_c <= 0.3808:
                break
    return best_h, best_c
