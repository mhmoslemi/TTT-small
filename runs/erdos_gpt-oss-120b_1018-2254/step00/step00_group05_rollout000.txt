    cur = h.sum()

        from scipy.optimize import minimize

        # smoothed option: run a few restarts with random jitter
        for _ in range(5):
            if time.time() - start_time > time_limit:
                break
            # random jitter around the current best
            jitter = np.random.uniform(-0.08, 0.08, size=n)
            x0 = best_h + jitter
            x0 = np.clip(x0, 0.0, 1.0)
            x0 = _project_to_sum(x0, target_sum)

            constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},)
            bounds = [(0.0, 1.0)] * n

            res = minimize(lambda x: _evaluate_c5(x, dx),
                           x0,
                           method='SLSQP',
                           bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': 250, 'ftol': 1e-12, 'disp': False})
            if res.success:
                cand = _project_to_sum(np.clip(res.x, 0.0, 1.0), target_sum)
                cand_score = _evaluate_c5(cand, dx)
                if cand_score < best_score:
                    best_h, best_score = cand, cand_score
    except Exception:
        # SciPy not present or optimisation failed – continue with hill climbing
        pass

    # ------------------------------------------------------------------
    # Simple randomized hill‑climbing / pairwise exchange
    # ------------------------------------------------------------------
    while time.time() - start_time < time_limit:
        # pick two distinct indices
        i, j = np.random.choice(n, size=2, replace=False)
        hi, hj = best_h[i], best_h[j]

        # feasible delta interval that preserves bounds
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue
        delta = np.random.uniform(low, high)

        h_new = best_h.copy()
        h_new[i] = hi + delta
        h_new[j] = hj - delta
        # (bounds already respected by construction)

        new_score = _evaluate_c5(h_new, dx)
        if new_score < best_score:
            best_h, best_score = h_new, new_score

    # ------------------------------------------------------------------
    # Final enforcement of constraints (tiny numerical drift)
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_to_sum(best_h, target_sum)
    best_score = _evaluate_c5(best_h, dx)

    return best_h.tolist(), float(best_score), n
