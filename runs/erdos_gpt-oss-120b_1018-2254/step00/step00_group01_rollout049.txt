    if abs(diff) < 1e-12:
        return h
    # try to fix with a single component
    for i in range(h.size):
        new_val = h[i] + diff
        if 0.0 <= new_val <= 1.0:
            h[i] = new_val
            return h
    # fallback: spread uniformly then clip
    h += diff / h.size
    np.clip(h, 0.0, 1.0, out=h)
    # minor residual correction
    total = h.sum()
    diff = target - total
    if abs(diff) > 1e-12:
        h[0] = min(max(h[0] + diff, 0.0), 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound, and the number of samples."""
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Initialise h
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n = h.shape[0]
    else:
        n = 47
        h = np.zeros(n, dtype=np.float64)
        # 23 ones, 23 zeros, one 0.5 (sum = n/2)
        idx = np.arange(n)
        np.random.shuffle(idx)
        h[idx[:23]] = 1.0
        h[idx[23]] = 0.5
    np.clip(h, 0.0, 1.0, out=h)
    target_sum = n / 2.0
    _enforce_sum(h, target_sum)

    # ------------------------------------------------------------------
    # Simulated‑annealing style local search
    # ------------------------------------------------------------------
    best_h = h.copy()
    best_c5 = _compute_c5(best_h)

    cur_h = h.copy()
    cur_c5 = best_c5

    # SA schedule parameters
    T0 = 0.02                # initial temperature
    decay = 5e-6             # exponential decay rate
    base_step = 0.2          # maximal magnitude for a “small” move
    max_no_improve = 20000   # stagnation trigger
    no_improve = 0
    it = 0
    start = time.time()

    while True:
        # time limit
        if time.time() - start > budget_s - 0.2:
            break

        # temperature
        T = T0 * math.exp(-decay * it)

        # decide move type
        large_move = np.random.rand() < 0.1   # 10% chance for a maximal move
        i, j = np.random.randint(0, n, size=2)
        while j == i:
            j = np.random.randint(0, n)

        hi, hj = cur_h[i], cur_h[j]

        if large_move:
            # transfer as much as possible in a random direction
            if np.random.rand() < 0.5:
                delta = min(1.0 - hi, hj)          # i up, j down
            else:
                delta = -min(hi, 1.0 - hj)         # i down, j up
        else:
            # small feasible move, limited by step size and bounds
            max_pos = min(1.0 - hi, hj)      # largest positive delta
            max_neg = -min(hi, 1.0 - hj)     # most negative delta (<=0)

            # bound delta magnitude by a temperature‑scaled step
            step_abs = base_step * (T / T0 + 0.01)   # keep >0
            max_step = min(step_abs, max_pos) if max_pos > 0 else 0.0
            min_step = max(-step_abs, max_neg) if max_neg < 0 else 0.0

            if min_step >= max_step:
                continue
            delta = np.random.rand() * (max_step - min_step) + min_step

        if delta == 0.0:
            continue

        # Apply proposal
        cur_h[i] = hi + delta
        cur_h[j] = hj - delta

        new_c5 = _compute_c5(cur_h)

        # Metropolis acceptance
        if new_c5 < cur_c5 or (T > 0 and np.random.rand() < math.exp(-(new_c5 - cur_c5) / T)):
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = cur_h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # revert
            cur_h[i] = hi
            cur_h[j] = hj
            no_improve += 1

        it += 1

        # Stagnation handling
        if no_improve >= max_no_improve:
            # perturb a few pairs around the best solution
            cur_h = best_h.copy()
            cur_c5 = best_c5
            for _ in range(5):
                a, b = np.random.randint(0, n, size=2)
                while b == a:
                    b = np.random.randint(0, n)
                ha, hb = cur_h[a], cur_h[b]
                maxp = min(1.0 - ha, hb)
                maxn = -min(ha, 1.0 - hb)
                if maxp == maxn == 0.0:
                    continue
                d = np.random.rand() * (maxp - maxn) + maxn
                cur_h[a] = ha + d
                cur_h[b] = hb - d
            cur_c5 = _compute_c5(cur_h)
            no_improve = 0

        # Early exit if target achieved
        if best_c5 < 0.3808:
            break

    # ------------------------------------------------------------------
    # Finalise solution
    # ------------------------------------------------------------------
    best_h = _enforce_sum(best_h, target_sum)

    try:
        final_c5 = evaluate_erdos_solution(best_h, n)[0]
    except Exception:
        final_c5 = _compute_c5(best_h)

    return (best_h, float(final_c5), n)
