def compute_gradient(h, shift):
    N = len(h)
    g = np.zeros_like(h)
    i = np.arange(N)
    j1 = i + shift
    mask1 = (j1 >= 0) & (j1 < N)
    g[mask1] = 1 - h[j1[mask1]]
    j2 = i - shift
    mask2 = (j2 >= 0) & (j2 < N)
    g[mask2] += -h[j2[mask2]]
    return g

import time
import numpy as np

def proj_box_sum(v, target):
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

def max_corr_shift(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    idx = int(np.argmax(corr))
    max_val = corr[idx] * dx
    shift = idx - (h.shape[0] - 1)
    return max_val, shift

def grad_for_shift(h, shift):
    N = h.shape[0]
    g = np.zeros_like(h)
    i = np.arange(N)
    j_plus = i + shift
    mask_plus = (j_plus >= 0) & (j_plus < N)
    g[mask_plus] = 1.0 - h[j_plus[mask_plus]]
    j_minus = i - shift
    mask_minus = (j_minus >= 0) & (j_minus < N)
    g[mask_minus] += -h[j_minus[mask_minus]]
    return g

def optimize_one(init_h, max_iters, step0, rng, dx, target):

import time
import numpy as np

def _proj_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the box [0,1]^n while enforcing sum(v)=target."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)


def _max_corr_shift(h: np.ndarray, dx: float):
    """Return (C5 value, shift index) for the current h."""
    corr = np.correlate(h, 1 - h, mode="full")
    idx = int(np.argmax(corr))
    max_val = corr[idx] * dx
    shift = idx - (h.shape[0] - 1)
    return max_val, shift


def _grad_for_shift(h: np.ndarray, shift: int):
    """Subgradient of the overlap at the shift attaining the maximum."""
    N = h.shape[0]
    g = np.zeros_like(h)
    i = np.arange(N)
    j_plus = i + shift
    mask_plus = (j_plus >= 0) & (j_plus < N)
    g[mask_plus] = 1.0 - h[j_plus[mask_plus]]
    j_minus = i - shift
    mask_minus = (j_minus >= 0) & (j_minus < N)
    g[mask_minus] += -h[j_minus[mask_minus]]
    return g


def _optimize_one(init_h: np.ndarray, max_iters: int, step0: float,
                  rng: np.random.Generator, dx: float, target: float):
    """Projected gradient search starting from init_h."""
    h = init_h.copy()
    best_h = h.copy()
    best_c5, _ = _max_corr_shift(h, dx)
    step = step0
    for _ in range(max_iters):
        c5, shift = _max_corr_shift(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        g = _grad_for_shift(h, shift)
        h_new = _proj_box_sum(h - step * g, target)
        new_c5, _ = _max_corr_shift(h_new, dx)
