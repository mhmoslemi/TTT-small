def project_box_sum(v, target):
    left = np.min(v) - 1.0
    right = np.max(v)
    for _ in range(50):
        mid = (left + right) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            left = mid
        else:
            right = mid
    return np.clip(v - (left+right)/2.0, 0.0, 1.0)

def local_search(h_init, time_limit):
    n = len(h_init)
    target = n/2
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_c5 = best_c5
    step = 0.1
    start = time.time()
    while time.time() - start < time_limit:
        # propose
        v = cur_h + rng.normal(scale=step, size=n)
        h_new = project_box_sum(v, target)
        c5_new = compute_c5(h_new)
        # accept if better

            cur_c5 = c5_new
            # update best
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

        # reduce step gradually
        step *= 0.9999
        # if step becomes too small, reset step
        if step < 1e-6:
            step = 0.01
    return best_h, best_c5

import time
import numpy as np

def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project vector v onto the feasible set:
        0 <= h_i <= 1  and  sum(h) = target
    using a simple bisection on the shift lambda such that
    h_i = clip(v_i - lambda, 0, 1).
    """
    left = np.min(v) - 1.0  # forces all entries to 1 (sum = n)
    right = np.max(v)       # forces all entries to 0 (sum = 0)
    for _ in range(50):
        mid = (left + right) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            left = mid
        else:
            right = mid
    lambda_opt = (left + right) * 0.5
    return np.clip(v - lambda_opt, 0.0, 1.0)

def compute_c5(arr: np.ndarray) -> float:
    """
    Compute the Erdős overlap constant C5 for a discrete step function.
    """
    n = arr.shape[0]
    dx = 2.0 / n
    corr = np.correlate(arr, 1.0 - arr, mode='full')
    return float(corr.max() * dx)

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Initialise a feasible starting vector (real-valued)
    # ------------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)  # provided by harness
        n = init_h.shape[0]
    except Exception:
        # fallback – 70 points, constant 0.5 (satisfies sum = n/2)
        n = 70

    target_sum = n / 2.0
    if not np.isclose(init_h.sum(), target_sum):
        init_h = project_box_sum(init_h, target_sum)

    # Real-valued current solution
    real_h = init_h.copy()
    real_c5 = compute_c5(real_h)

    # Best solution seen so far
    best_h = real_h.copy()
    best_c5 = real_c5

    # ------------------------------------------------------------------
    # Initialise a binary candidate (if n is even)
    # ------------------------------------------------------------------
    use_binary = (n % 2 == 0)
    if use_binary:
        target_ones = n // 2
        # start from rounded version of init_h
        bin_h = (init_h > 0.5).astype(float)
        cur_ones = int(bin_h.sum())
        if cur_ones > target_ones:
            excess = cur_ones - target_ones
            idx = np.flatnonzero(bin_h == 1.0)
            flip = rng.choice(idx, size=excess, replace=False)
            bin_h[flip] = 0.0
        elif cur_ones < target_ones:
            deficit = target_ones - cur_ones
            idx = np.flatnonzero(bin_h == 0.0)
            flip = rng.choice(idx, size=deficit, replace=False)
            bin_h[flip] = 1.0
        cur_bin_h = bin_h.copy()
        cur_bin_c5 = compute_c5(cur_bin_h)
        if cur_bin_c5 < best_c5:
            best_c5 = cur_bin_c5
            best_h = cur_bin_h.copy()
    else:
        cur_bin_h = None
        cur_bin_c5 = None

    # ------------------------------------------------------------------
    # Simulated‑annealing style local search
    # ------------------------------------------------------------------
    T0 = 0.10          # initial temperature
    step0 = 0.20       # initial Gaussian step size
    step_min = 0.005   # asymptotic step size

    while True:
        now = time.time()
        elapsed = now - start_time
        if elapsed >= budget_s:
            break
        frac = elapsed / budget_s

        # Linear cooling
        T = T0 * (1.0 - frac) + 1e-12
        step = step0 * (1.0 - frac) + step_min

        # Real‑valued update

        v = real_h + rng.normal(scale=step, size=n)

                j = idx_zero[rng.integers(idx_zero.size)]
                new_bin = cur_bin_h.copy()
                new_bin[i] = 0.0
                new_bin[j] = 1.0
                new_c5 = compute_c5(new_bin)

        best_h = project_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n)
