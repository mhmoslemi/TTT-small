def run(seed=42,budget_s=1100,**kwargs):
    import numpy as np, time, math, random
    random.seed(seed)
    np.random.seed(seed)
    start = time.time()
    n = 120 # choose moderate size.
    dx = 2.0/n
    # initial
    if 'initial_h_values' in globals() and len(initial_h_values)>0:
        h0 = np.interp(np.linspace(0,2, n, endpoint=False), np.linspace(0,2,len(initial_h_values), endpoint=False), initial_h_values)
    else:
        # start with half ones half zeros random arrangement
        h0 = np.zeros(n)
        # choose n/2 indices to be 1
        idx = np.random.choice(n, n//2, replace=False)
        h0[idx] = 1.0
    # project onto constraints: must sum to n/2 exactly. after interpolation may not.
    target_sum = n/2.0
    h = project_to_sum_box(h0, target_sum)
    best_h = h.copy()
    best_c = compute_c5(best_h)
    # Simulated annealing loop
    T0 = 0.1
    Tmin = 1e-5
    # compute number of iterations based on time left
    iteration = 0
    while time.time() - start < budget_s * 0.95: # keep ~5% margin
        # temperature schedule
        frac = (time.time() - start) / (budget_s*0.95)
        T = max(Tmin, T0 * (1 - frac))
        # propose move: pick two indices i!=j
        i,j = np.random.randint(0,n,2)
        if i==j:
            continue
        # compute allowable delta range
        lower = max(-h[i], h[j] - 1.0) # ensure h[i]+δ ≥0, h[j]-δ ≤1
        upper = min(1.0 - h[i], h[j]) # ensure h[i]+δ ≤1, h[j]-δ ≥0
        if lower >= upper:
            continue
        delta = (upper - lower) * np.random.rand() + lower
        # small delta scaling maybe
        # apply
        h[i] += delta
        h[j] -= delta
        # evaluate
        c = compute_c5(h)
        # acceptance
        if c < best_c or np.random.rand() < math.exp(-(c - best_c) / max(T, 1e-12)):
            if c < best_c:
                best_c = c
                best_h = h.copy()
            # keep current h (already modified)
        else:
            # revert changes
            h[i] -= delta
            h[j] += delta
        iteration += 1
    return best_h, best_c, n

import numpy as np
import time
import math
import random

def _project_to_sum_box(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto the box [0,1]^n while fixing the sum to target via bisection."""
    lo, hi = -1.0, 1.0
    # 50 iterations give ~1e-15 accuracy on the shift
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(x + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)


def _c5_of(h: np.ndarray) -> float:
    """Return the C5 value for a step function represented by h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function that gives a low C5 bound."""
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()

    # ----------------------------------------------------------------------
    # Prepare initial vector and discretisation size
    # ----------------------------------------------------------------------
    try:
        init = np.array(initial_h_values, copy=False)
        if init.ndim != 1:
            init = init.ravel()
        base_n = init.shape[0]
    except Exception:
        init = None
        base_n = 45

    # Choose a moderate resolution (>=180, capped at 500)
    n = max(180, base_n * 4)
    if n > 500:
        n = 500
    target_sum = n / 2.0

    if init is not None:
        old_x = np.linspace(0.0, 2.0, base_n, endpoint=False)
        new_x = np.linspace(0.0, 2.0, n,   endpoint=False)
        h = np.interp(new_x, old_x, init)
    else:
        h = np.random.rand(n)

    h = _project_to_sum_box(h, target_sum)
    best_h = h.copy()
    best_c = _c5_of(best_h)

    # ----------------------------------------------------------------------
    # Main optimisation loop (sub‑gradient descent + simulated annealing)
    # ----------------------------------------------------------------------
    no_improve = 0
    max_no_improve = 5000  # restart after stagnation
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.97:
            break

        # current correlation and worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = int(np.argmax(corr))
        c_cur = float(corr[max_idx] * (2.0 / n))
        shift = max_idx - (n - 1)   # shift in [-n+1, n-1]

        # sub‑gradient that increases overlap at the worst shift
        d = np.zeros_like(h)
        if shift == 0:
            d = 2.0 * h                     # derivative of sum h_i^2
        elif shift > 0:
            s = shift
            d[: n - s] += h[s:]              # h[i+shift]
            d[s:]      += h[: n - s]         # h[i-shift]
        else:
            s = -shift
            d[s:]      += h[: n - s]         # h[i-shift]
            d[: n - s] += h[s:]              # h[i+shift]

        # step size that decays with time
        frac = elapsed / (budget_s * 0.97)
        alpha = 0.02 * (1.0 - frac) + 0.001

        h_new = h + alpha * d
        h_new = _project_to_sum_box(h_new, target_sum)

        # evaluate new candidate
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c_new = float(np.max(corr_new) * (2.0 / n))

        # acceptance rule (always accept improvement)
        if c_new < c_cur:
            accept = True
        else:
            # simulated‑annealing style acceptance
            T = 0.02 * (1.0 - frac) + 1e-6
            prob = math.exp(-(c_new - c_cur) / T)
            accept = random.random() < prob

        if accept:
            h = h_new
            c_cur = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # occasional random restart if stuck

            noise = 0.1 * (np.random.rand(n) - 0.5)
            h = best_h + noise
            h = _project_to_sum_box(h, target_sum)
            no_improve = 0

    # final projection & evaluation (ensures exact sum)
    best_h = _project_to_sum_box(best_h, target_sum)
    best_c = _c5_of(best_h)

    return best_h, float(best_c), n
