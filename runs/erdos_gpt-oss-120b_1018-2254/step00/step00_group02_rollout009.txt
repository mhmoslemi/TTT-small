import numpy as np

def _c5_from_h(h):
    """Return the C5 value for a discretised step function h."""
    n = h.size
    dx = 2.0 / n
    # auto‑correlation of h with itself (full mode)
    auto = np.correlate(h, h, mode='full')
    min_auto = auto.min()
    return 1.0 - dx * min_auto

    # ------------------------------------------------------------------
    # Setup and initial construction
    # ------------------------------------------------------------------

    start_time = time.time()

        n = h.size

            # scale uniformly, then clip and correct any leftover mass
            h *= (target_sum / cur_sum)
            h = np.clip(h, 0.0, 1.0)
            diff = target_sum - h.sum()
            if abs(diff) > 1e-12:
                # distribute the tiny residual to the first feasible entry
                for idx in range(n):
                    if diff > 0 and h[idx] < 1.0:
                        add = min(diff, 1.0 - h[idx])
                        h[idx] += add
                        break
                    if diff < 0 and h[idx] > 0.0:
                        sub = min(-diff, h[idx])
                        h[idx] -= sub
                        break
