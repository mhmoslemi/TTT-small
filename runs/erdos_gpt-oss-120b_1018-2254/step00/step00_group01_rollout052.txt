import numpy as np

def _compute_c5(h, dx):
    """Return max_k ∫ h(x)*(1-h(x+k)) dx for discretized h."""
    # correlation of h with (1-h), full mode, multiplied by dx
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _project_sum_box(v, target):
    """
    Project vector v onto the box [0,1]^n while enforcing sum(v)=target.
    Uses a simple water-filling / iterative clipping method.
    """
    w = np.clip(v, 0.0, 1.0)
    for _ in range(50):
        excess = w.sum() - target
        if abs(excess) < 1e-12:
            break
        free = (w > 0.0) & (w < 1.0)
        if not np.any(free):
            # all at bounds but sum not correct -- distribute proportionally
            w = np.clip(w - excess / len(w), 0.0, 1.0)
            break
        adj = excess / np.count_nonzero(free)
        w[free] -= adj
        w = np.clip(w, 0.0, 1.0)
    return w

def _random_h(rng, n, target):
    """Generate a random feasible h (values in [0,1], sum = target)."""
    v = rng.random(n)
    return _project_sum_box(v, target)
