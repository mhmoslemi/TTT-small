def compute_gradient(vec, lag):
    n = len(vec)
    if lag >= 0:
        # shift forward array (i+lag)
        shift_plus = np.concatenate([np.zeros(lag), vec[:n - lag]])
        # shift backward (i-lag)
        shift_minus = np.concatenate([vec[lag:], np.zeros(lag)])

    grad = 1.0 - (shift_plus + shift_minus)
    return grad

import numpy as np
import time

def _project_to_sum_and_box(v, target):
    """Project v onto {w: sum(w)=target, 0<=w_i<=1} using bisection."""
    lo = -np.max(np.abs(v)) - 2.0
    hi = np.max(np.abs(v)) + 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _c5_and_corr(h):
    """Return C5 bound and full correlation array for vector h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx, corr


def _gradient(h, lag):
    """Gradient of max correlation w.r.t. h for a given lag."""
    n = h.size
    if lag >= 0:
        shift_plus = np.concatenate([np.zeros(lag), h[: n - lag]])
        shift_minus = np.concatenate([h[lag:], np.zeros(lag)])
    else:
        l = -lag
        shift_plus = np.concatenate([np.zeros(l), h[: n - l]])
        shift_minus = np.concatenate([h[l:], np.zeros(l)])
    return 1.0 - (shift_plus + shift_minus)


def _random_feasible(n, target):
    v = np.random.rand(n)
    return _project_to_sum_and_box(v, target)
