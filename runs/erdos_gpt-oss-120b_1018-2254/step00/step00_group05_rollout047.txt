import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return the C5 value for a discretized step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h); mode='full' gives all integer shifts
    max_corr = np.max(np.correlate(h, 1.0 - h, mode='full'))
    return max_corr * dx

def project_to_constraints(v: np.ndarray) -> np.ndarray:
    """
    Project vector v onto the feasible set:
      0 <= x_i <= 1   and   sum(x) = len(v)/2.
    Simple iterative redistribution; sufficient for modest sizes.
    """
    n = v.shape[0]
    target = n / 2.0
    x = np.clip(v, 0.0, 1.0)
    cur = np.sum(x)
    # If already satisfied, return
    if abs(cur - target) <= 1e-12:
        return x
    diff = target - cur
    # Perform a few redistribution passes
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Need to add mass: distribute to non‑full entries proportionally
            cap = 1.0 - x
            total_cap = np.sum(cap)
            if total_cap == 0:
                break
            add = np.minimum(cap, diff * cap / total_cap)
            x += add
        else:
            # Need to remove mass: take from non‑zero entries proportionally
            cap = x
            total_cap = np.sum(cap)
            if total_cap == 0:
                break
            sub = np.minimum(cap, -diff * cap / total_cap)
            x -= sub
        cur = np.sum(x)
        diff = target - cur
    # Final tiny adjustment on a single element if needed
    cur = np.sum(x)
    diff = target - cur
    if abs(diff) > 1e-12:
        idx = np.argmax(x) if diff < 0 else np.argmin(x)
        x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
    return x

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return (h_values, c5_bound, n_points) within the time budget."""
    np.random.seed(seed)
    start = time.time()

    # ---- initialise candidate -------------------------------------------------

        n = h0.shape[0]

    else:
        n = 200                     # fallback grid size
        h0 = np.full(n, 0.5, dtype=np.float64)   # uniform function satisfies sum constraint

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_c5 = best_c5

    max_iter = 2_000_000
