def optimize_h(initial_h, n_iter, time_limit):
    h = initial_h.copy()
    best_h = h.copy()
    best_val = evaluate_c5(h, dx)
    start = time.time()
    iter = 0
    while time.time() - start < time_limit and iter < n_iter:
        # compute correlation
        c = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(c)
        s = idx - (len(h) - 1)  # shift offset
        # compute gradient
        n = len(h)
        grad = np.ones(n)  # base 1
        if s >= 0:
            # forward term
            forward = np.zeros(n)
            forward[:n - s] = h[s:]
            # backward term
            backward = np.zeros(n)
            backward[s:] = h[:n - s]
        else:
            s_abs = -s
            forward = np.zeros(n)
            forward[s_abs:] = h[:n - s_abs]
            backward = np.zeros(n)
            backward[:n - s_abs] = h[s_abs:]
        grad -= (forward + backward)
        # update step size
        # we could use schedule: lr = step0 / sqrt(iter+1)
        lr = 0.5 / np.sqrt(iter + 1)  # maybe adjust
        y = h - lr * grad
        # project onto box+sum
        target = n * 0.5
        h = project_box_sum(y, target)
        # evaluate
        val = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if val < best_val:
            best_val = val
            best_h = h.copy()
        iter += 1
    return best_h, best_val

def project_box_sum(y, target):
    # Clip to box after adjusting by lambda (add offset)
    lo = -np.max(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(40):
        lam = (lo + hi) / 2.0
        proj = np.clip(y - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(y - (lo+hi)/2.0, 0.0, 1.0)
    # Ensure sum close to target (due to rounding). May adjust small residual.
    # Adjust by distributing residual equally among free variables.
    diff = target - proj.sum()

        # add diff proportionally to elements not at bounds
        mask = (proj > 1e-12) & (proj < 1 - 1e-12)
        num = mask.sum()
        if num > 0:
            proj[mask] += diff / num
            proj = np.clip(proj, 0.0, 1.0)
    return proj

import numpy as np, time, math, random

def project_box_sum(y, target):
    # Bisection for lambda
    lo = -np.max(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        proj = np.clip(y - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    # adjust residual
    diff = target - proj.sum()
    if abs(diff) > 1e-10:
        # distribute diff equally among non-boundary entries
        mask = (proj > 1e-12) & (proj < 1 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            proj[mask] += diff / cnt
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def compute_c5(h, dx):
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

import numpy as np, time, math, random

def _project_box_sum(y, target):
    lo = -np.max(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        proj = np.clip(y - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        mask = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt:
            proj[mask] += diff / cnt
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def _c5_value(h, dx):
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def run(seed=42, budget_s=1100, **kwargs):

    random.seed(seed)

    dx = 2.0 / n
    cur_val = _c5_value(h, dx)
    best_val = cur_val
    best_h = h.copy()
    it = 0
    # main optimization loop
    while time.time() - start < budget_s * 0.95:
        corr = np.correlate(h, 1 - h, mode='full')
        idx = int(np.argmax(corr))

        # forward and backward shifted versions (zero padding)
        if s >= 0:
            forward = np.empty_like(h)
            forward[:n - s] = h[s:]
            forward[n - s:] = 0.0
            backward = np.empty_like(h)
            backward[s:] = h[:n - s]
            backward[:s] = 0.0
        else:
            s_abs = -s
            forward = np.empty_like(h)
            forward[s_abs:] = h[:n - s_abs]
            forward[:s_abs] = 0.0
            backward = np.empty_like(h)

def _project_box_sum(y, target):
    """Projection of y onto the box [0,1]^n with sum constraint = target."""
    lo = -np.max(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(50):

        proj = np.clip(y - lam, 0.0, 1.0)

    return proj

    return np.max(np.correlate(h, 1 - h, mode='full')) * dx
