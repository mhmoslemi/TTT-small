import time
import numpy as np

def _project_onto_capped_simplex(v, S):
    """Euclidean projection of v onto the set {0≤y≤1, sum(y)=S}."""
    # Bisection on lambda such that y = clip(v - lambda, 0, 1) satisfies sum(y)=S.
    lower = np.min(v - 1.0)          # guarantees all y_i = 1
    upper = np.max(v)                # guarantees all y_i = 0
    for _ in range(60):
        mid = (lower + upper) * 0.5
        y = np.clip(v - mid, 0.0, 1.0)
        tot = y.sum()
        if tot > S:
            lower = mid
        else:
            upper = mid
    lam = (lower + upper) * 0.5
    y = np.clip(v - lam, 0.0, 1.0)
    diff = S - y.sum()
    if abs(diff) > 1e-12:  # fix round‑off
        if diff > 0:
            caps = 1.0 - y
            idx = np.argmax(caps)
            inc = min(diff, caps[idx])
            y[idx] += inc
        else:
            caps = y
            idx = np.argmax(caps)
            dec = min(-diff, caps[idx])
            y[idx] -= dec
    return y

def _random_feasible(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    v = rng.random(n)
    return _project_onto_capped_simplex(v, S=n/2)

def _c5_bound(h):
    """Compute the overlap bound for a step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _mutate(h, rate, sigma, rng):
    """Gaussian mutation followed by projection."""
    m = h.copy()
    mask = rng.random(m.size) < rate
    if np.any(mask):
        m[mask] += rng.normal(scale=sigma, size=mask.sum())
    m = np.clip(m, 0.0, 1.0)
    return _project_onto_capped_simplex(m, S=m.size/2)

def _crossover(p1, p2, rng):
    """Single‑point crossover with projection."""
    n = p1.size
    pt = rng.integers(1, n)
    child = np.empty(n, dtype=np.float64)
    child[:pt] = p1[:pt]
    child[pt:] = p2[pt:]
    return _project_onto_capped_simplex(child, S=n/2)

def _tournament(pop, scores, k, rng):
    """Select the best individual among k random choices."""
    idx = rng.integers(0, len(pop), size=k)
    best = idx[0]
    best_score = scores[best]
    for i in idx[1:]:
        if scores[i] < best_score:
            best = i
            best_score = scores[i]
    return pop[best], best_score
