def anneal(h0, c5_0, max_iters, total_time_end, rng):
    h_current = h0.copy()
    c5_current = c5_0
    best_h = h_current.copy()
    best_c5 = c5_current
    n = len(h0)
    target_sum = n / 2.0
    # set parameters
    T0 = 0.02  # initial temperature may be small because objective typical 0.4 magnitude; we need acceptance of slightly worse moves.
    T_end = 1e-5
    for it in range(max_iters):
        # time check
        if time.time() > total_time_end:
            break
        # temperature
        T = T0 * ((max_iters - it) / max_iters)
        # propose move
        i = rng.randint(0, n)
        j = rng.randint(0, n-1)
        if j >= i:
            j += 1
        # direction: from i (donor) to j (receiver)
        # compute max delta
        max_decrease = h_current[i]
        max_increase = 1.0 - h_current[j]
        max_move = min(max_decrease, max_increase)
        if max_move <= 0:
            continue
        # delta magnitude: fraction of max_move scaled by temperature maybe
        # random uniform in [0, max_move]
        delta = rng.random() * max_move
        # apply
        h_new = h_current.copy()
        h_new[i] -= delta
        h_new[j] += delta
        # evaluate
        c5_new = compute_c5(h_new)
        # acceptance
        if c5_new < c5_current or rng.random() < math.exp(-(c5_new - c5_current) / (T + 1e-12)):
            h_current = h_new
            c5_current = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
    return best_h, best_c5
