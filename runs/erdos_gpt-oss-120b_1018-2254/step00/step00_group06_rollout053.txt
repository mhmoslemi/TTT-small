import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return the overlap constant C5 for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with its complement
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_onto_box_simplex(v: np.ndarray, S: float, tol: float = 1e-12, max_iter: int = 100) -> np.ndarray:
    """
    Project vector v onto the feasible set:
        0 <= x_i <= 1  and  sum(x) = S
    using a simple bisection on the Lagrange multiplier.
    """
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - S) <= tol:
            return proj
        if s > S:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)

    # ----------------------------------------------------------------------
    return best_h, float(best_c5), n
