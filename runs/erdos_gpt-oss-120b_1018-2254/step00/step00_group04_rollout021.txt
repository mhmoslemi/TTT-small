import time
import numpy as np

def compute_c5(h):
    """Return C5 for a given h-array."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def random_swap_move(h, rng):
    """Transfer a small amount from a positive entry to a negative one,
    preserving the total sum and the box constraints."""
    n = h.shape[0]
    for _ in range(5):
        i = rng.integers(n)
        if h[i] <= 1e-12:
            continue
        j = rng.integers(n)
        if i == j or h[j] >= 1.0 - 1e-12:
            continue
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 1e-12:
            continue
        frac = 0.01 + 0.99 * rng.random()
        delta = max_delta * frac
        nh = h.copy()
        nh[i] -= delta
        nh[j] += delta
        return nh
    return None

def improve_step(h, k_shift, rng):
    """Move mass from a large contribution site to its partner for the
    currently worst shift."""
    n = h.shape[0]
    if k_shift >= 0:
        length = n - k_shift
        if length <= 0:
            return None
        head = h[:length]
        tail = h[k_shift:k_shift + length]
        contrib = head * (1.0 - tail)
        i_local = int(np.argmax(contrib))
        i = i_local
        j = i + k_shift
    else:
        kp = -k_shift
        length = n - kp
        if length <= 0:
            return None
        head = h[kp:]
        tail = h[:length]
        contrib = head * (1.0 - tail)
        i_local = int(np.argmax(contrib))
        i = i_local + kp
        j = i - kp
    max_delta = min(h[i], 1.0 - h[j])
    if max_delta <= 1e-12:
        return None
    frac = 0.01 + 0.99 * rng.random()
    delta = max_delta * frac
    nh = h.copy()
    nh[i] -= delta
    nh[j] += delta
    return nh

def simulated_annealing(start_h, time_limit, rng):
    """Simple SA with targeted moves for the Erdős C5 problem."""
    n = start_h.shape[0]
    cur_h = start_h.copy()
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    start = time.time()
    target = 0.3808
    no_improve = 0
    while time.time() - start < time_limit:
        # correlation to locate the worst shift
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        max_idx = int(np.argmax(corr))
        k_shift = max_idx - (n - 1)

        # propose a move
        if rng.random() < 0.6:
            nh = improve_step(cur_h, k_shift, rng)
        else:
            nh = random_swap_move(cur_h, rng)
        if nh is None:
            continue

        new_c5 = compute_c5(nh)

        if new_c5 < cur_c5:
            cur_h, cur_c5 = nh, new_c5
            no_improve = 0
            if new_c5 < best_c5:
                best_h, best_c5 = nh.copy(), new_c5
                if best_c5 <= target:
                    break
        else:
            # occasional uphill acceptance
            elapsed = time.time() - start
            T = max(0.001, 0.01 * (1.0 - elapsed / time_limit))
            if rng.random() < np.exp(-(new_c5 - cur_c5) / T):
                cur_h, cur_c5 = nh, new_c5
            no_improve += 1

        # if stagnating, perturb around best solution
        if no_improve > 5000:
            perturb = 0.02 * (rng.random(n) - 0.5)
            nh = best_h + perturb
            nh = np.clip(nh, 0.0, 1.0)
            # re‑balance the sum exactly
            S = n / 2.0
            diff = nh.sum() - S
            free = (nh > 1e-12) & (nh < 1.0 - 1e-12)
            if free.any():
                nh[free] -= diff / free.sum()
            else:
                nh -= diff / n
            nh = np.clip(nh, 0.0, 1.0)
            cur_h, cur_c5 = nh, compute_c5(nh)
            no_improve = 0
    return best_h, best_c5
