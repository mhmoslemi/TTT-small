    # v: array shape (n,)
    # target: float between 0 and n (actually n/2).
    # returns y in [0,1] such that sum(y)=target.
    # Use bisection on tau.
    n = v.shape[0]
    # Set bounds: tau_low = min(v) - 1 (ensures v_i - tau >= 1 for all)
    # tau_high = max(v) (ensures v_i - tau <= 0 for all)
    tau_low = np.min(v) - 1.0
    tau_high = np.max(v)
    # If target <= 0: return zeros
    # If target >= n: return ones
    # else bisection
    for _ in range(50):
        tau = (tau_low + tau_high) / 2.0
        y = np.clip(v - tau, 0, 1)
        s = np.sum(y)
        if s < target:
            tau_high = tau
        else:
            tau_low = tau
    y = np.clip(v - tau, 0, 1)
    # Due to numerical may be off; adjust small residual.
    diff = target - np.sum(y)
    if abs(diff) > 1e-8:
        # distribute diff over entries not at bounds.
        free_idx = np.where((y > 1e-12) & (y < 1 - 1e-12))[0]
        if len(free_idx) > 0:
            y[free_idx] += diff / len(free_idx)
            y = np.clip(y, 0, 1)
    return y

import numpy as np
import time

def proj_capped_simplex(v, target):
    """Euclidean projection of v onto the hypercube [0,1] with sum = target."""
    n = v.shape[0]
    if target <= 0:
        return np.zeros_like(v)
    if target >= n:
        return np.ones_like(v)
    # Bounds for the dual variable tau
    tau_low = np.min(v) - 1.0      # makes all entries >=1 -> clipped to 1
    tau_high = np.max(v)           # makes all entries <=0 -> clipped to 0
    # Bisection
    for _ in range(60):
        tau = (tau_low + tau_high) * 0.5
        y = np.clip(v - tau, 0.0, 1.0)
        s = np.sum(y)
        if s < target:
            tau_high = tau
        else:
            tau_low = tau
    y = np.clip(v - tau, 0.0, 1.0)
    # Minor correction for numerical error
    diff = target - np.sum(y)
    if abs(diff) > 1e-8:
        free = (y > 1e-12) & (y < 1.0 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            y[idx] += diff / idx.size
            np.clip(y, 0.0, 1.0, out=y)
    return y

def random_feasible(n, target):
    """Generate a random vector in [0,1]^n with prescribed sum."""
    v = np.random.rand(n)
    return proj_capped_simplex(v, target)

        h0 = np.array(initial_h_values, dtype=float)
        n_points = h0.shape[0]
    except Exception:
        # fallback: use a modest size if no parent is supplied
        n_points = 100
        h0 = None

    target_sum = n_points / 2.0

    # Initial feasible solution
    if h0 is None:
        cur_h = random_feasible(n_points, target_sum)
    else:
        cur_h = proj_capped_simplex(h0, target_sum)

    # Initial evaluations
    try:
        cur_c5 = float(evaluate_erdos_solution(cur_h, n_points)[0])
    except Exception:
        cur_c5 = float('inf')
    best_h = cur_h.copy()
    best_c5 = cur_c5

    start_time = time.time()
    no_improve = 0
    iteration = 0

    while time.time() - start_time < budget_s:
        iteration += 1
        elapsed = time.time() - start_time
        fraction = elapsed / budget_s
        # Temperature schedule: decays linearly from 0.2 to 0.001
        temp = max(0.001, 0.2 * (1.0 - fraction))

        # Propose a new candidate by adding bounded noise
        perturb = (np.random.rand(n_points) - 0.5) * 2.0 * temp
        cand = cur_h + perturb
        cand = proj_capped_simplex(cand, target_sum)

        # Evaluate candidate
        try:
            cand_c5 = float(evaluate_erdos_solution(cand, n_points)[0])
        except Exception:
            continue

        # Metropolis acceptance
        delta = cand_c5 - cur_c5
        if delta < 0.0 or np.random.rand() < np.exp(-delta / temp):
            cur_h = cand
            cur_c5 = cand_c5

        # Track best ever found
        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = cand.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Stagnation handling: random restart after many non‑improving steps
        if no_improve >= 5000:
            cur_h = random_feasible(n_points, target_sum)
            try:
                cur_c5 = float(evaluate_erdos_solution(cur_h, n_points)[0])
            except Exception:
                cur_c5 = float('inf')
            no_improve = 0

    return (best_h, best_c5, n_points)
