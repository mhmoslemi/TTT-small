```
import numpy as np, time, random

def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full') * dx)

def initial_binary(n, seed):
    np.random.seed(seed)
    # start with n/2 ones random positions.
    h = np.zeros(n)
    # if n is even, n/2 exactly; else handle rounding.
    num_ones = n // 2
    indices = np.random.choice(n, num_ones, replace=False)
    h[indices] = 1.0
    return h

def local_search(h, time_budget, start_time):
    best_h = h.copy()
    best_val = compute_c5(best_h)
    n = len(h)
    # loop until time budget
    while time.time() - start_time < time_budget:
        # propose move: swap two indices (i where h[i]==0, j where h[j]==1)
        zeros = np.where(best_h == 0)[0]
        ones = np.where(best_h == 1)[0]
        if len(zeros)==0 or len(ones)==0:
            break
        i = np.random.choice(zeros)
        j = np.random.choice(ones)
        # perform swap
        new_h = best_h.copy()
        new_h[i] = 1.0
        new_h[j] = 0.0
        new_val = compute_c5(new_h)
        # if improvement, accept
        if new_val < best_val - 1e-12:
            best_h = new_h
            best_val = new_val
        # optionally accept with probability (not required)
    return best_h, best_val
```

        s = y.sum()

        else:
            hi = lam

    n = h.shape[0]
    g = np.zeros_like(h)
    if s >= 0:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    else:
        t = -s
        g[t:] += 1.0 - h[:n - t]
        g[:n - t] -= h[t:]
    return g

def _compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def _smooth_max_obj(h, gamma):

    dx = 2.0 / n

    w = np.exp(gamma * (corr - m))
    return m + np.log(np.sum(w)) / gamma

                best_h, best_val = cand, val

    # simple pairwise hill climbing until budget expires
    while time.time() - start < 0.98 * budget_s:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        max_inc = min(1.0 - best_h[i], best_h[j])
        if max_inc <= 0:
            continue
        delta = np.random.rand() * max_inc
        h_new = best_h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        val = _compute_c5(h_new)
        if val < best_val:
            best_h, best_val = h_new, val

    best_h = _project_onto_capped_simplex(best_h, target)
    final_val = _compute_c5(best_h)
    return best_h, float(final_val), int(n)
