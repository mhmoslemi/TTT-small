import time
import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint

def _c5_of_h(h):
    """Return C5 for a given step function vector h."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


def _project_to_feasible(h, target_sum):
    """Clip to [0,1] and adjust to satisfy the exact sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if np.isclose(cur, target_sum, atol=1e-12):
        return h
    # simple redistribution of the residual on non‑bound entries
    diff = target_sum - cur
    mask = (h > 0.0) & (h < 1.0)
    if np.any(mask):
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    else:
        # fallback – adjust first element only
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] with integral 1 that
    minimizes the Erdős overlap bound C5.

    Returns:
        (list of h values, best C5 found, number of points)
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----- initial construction -----
    try:
        # Use the supplied construction if present.
        h0 = np.array(initial_h_values, dtype=float)  # noqa: F821
    except NameError:
        # No supplied vector – fall back to a simple half‑ones pattern.
        n0 = 45
        h0 = np.zeros(n0, dtype=float)
        h0[: n0 // 2] = 1.0
        if n0 % 2:
            h0[n0 // 2] = 0.5
    n = h0.size
    target = n / 2.0
    h0 = _project_to_feasible(h0, target)

    best_h = h0.copy()
    best_c5 = _c5_of_h(best_h)
    cur_h = best_h.copy()
    cur_c5 = best_c5

    # ----- simulated annealing -----
    T_start = 5e-3  # initial temperature
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
        # linear cooling
        progress = elapsed / (budget_s * 0.95)
        T = max(T_start * (1.0 - progress), 1e-8)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        if rng.random() < 0.5:
            # move mass from i to j
            delta_max = min(cur_h[i], 1.0 - cur_h[j])
            if delta_max <= 0.0:
                continue
            delta = rng.random() * delta_max
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        else:
            # move mass from j to i
            delta_max = min(cur_h[j], 1.0 - cur_h[i])
            if delta_max <= 0.0:
                continue
            delta = rng.random() * delta_max
            new_h = cur_h.copy()
            new_h[i] += delta
            new_h[j] -= delta

        new_c5 = _c5_of_h(new_h)
        diff = new_c5 - cur_c5
        if diff < 0.0 or rng.random() < np.exp(-diff / T):
            cur_h = new_h
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()

    # ----- local refinement with SLSQP -----
    bounds = Bounds(0.0, 1.0)
    lin_con = LinearConstraint(np.ones((1, n)), [target], [target])

    try:
        res = minimize(
            _c5_of_h,
            best_h,
            method="SLSQP",
            bounds=bounds,
            constraints=[lin_con],
            options={"maxiter": 500, "ftol": 1e-12, "disp": False},
        )
        if res.success and res.fun < best_c5:
            best_h = _project_to_feasible(res.x, target)
            best_c5 = _c5_of_h(best_h)
    except Exception:
        # If the optimizer fails, keep the previous best.
        pass

    # ----- final feasibility clean‑up -----
    best_h = _project_to_feasible(best_h, target)
    final_c5 = _c5_of_h(best_h)

    return best_h.tolist(), float(final_c5), n
