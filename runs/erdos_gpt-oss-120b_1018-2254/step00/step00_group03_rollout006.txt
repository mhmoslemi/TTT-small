def propose_by_pair_swap(h, step):
    n = h.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = (np.random.rand() - 0.5) * 2 * step

    final_c5 = compute_c5(best_h)

    return (best_h.tolist(), float(final_c5), n_points)
