def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return np.max(c) * dx

import numpy as np
import time

def repair(vec: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1] and adjust to achieve the exact target sum."""
    np.clip(vec, 0.0, 1.0, out=vec)
    diff = target_sum - vec.sum()
    if abs(diff) < 1e-12:
        return vec
    if diff > 0:
        free = 1.0 - vec
    else:
        free = vec
    total_free = free.sum()
    if total_free == 0.0:
        return vec
    vec += diff * free / total_free
    np.clip(vec, 0.0, 1.0, out=vec)
    return vec

def random_feasible_vector(n: int, target_sum: float) -> np.ndarray:
    """Generate a random vector in [0,1]^n whose entries sum to target_sum."""
    vec = np.random.rand(n)
    s = vec.sum()
    if s == 0.0:
        vec[0] = target_sum
    else:
        vec *= target_sum / s
    np.clip(vec, 0.0, 1.0, out=vec)
    return repair(vec, target_sum)

def tournament_select(pop, fitnesses, size=3):
    """Select one individual from the population via tournament."""
    idxs = np.random.choice(len(pop), size=size, replace=False)
    best_idx = idxs[np.argmin(fitnesses[idxs])]
    return pop[best_idx]

def compute_c5(h: np.ndarray) -> float:
    """Compute the C5 bound for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    # Determine problem size and initial vector (if provided)
    try:
        init = np.array(initial_h_values, dtype=np.float64)

        init = None
        n = 95

    target_sum = n / 2.0
    pop_size = 200

    # Initialise population
    pop = []
    if init is not None:
        vec = repair(init.copy(), target_sum)
        pop.append(vec)
    while len(pop) < pop_size:
        pop.append(random_feasible_vector(n, target_sum))

    fitnesses = np.array([compute_c5(ind) for ind in pop])
    best_idx = np.argmin(fitnesses)
    best = pop[best_idx].copy()
    best_f = fitnesses[best_idx]

    # GA parameters
    tournament_size = 3
    child_mut_prob = 0.5
    gene_mut_rate = 0.02
    gene_mut_sigma = 0.2

    # Main optimisation loop
    while True:
        if time.time() - start >= budget_s - 0.05:
            break
        # Parent selection
        p1 = tournament_select(pop, fitnesses, tournament_size)
        p2 = tournament_select(pop, fitnesses, tournament_size)

        # Convex combination crossover (preserves sum)
        alpha = np.random.rand()
        child = alpha * p1 + (1.0 - alpha) * p2

        # Mutation
        if np.random.rand() < child_mut_prob:
            mask = np.random.rand(n) < gene_mut_rate
            if mask.any():

                child[mask] += noise[mask]

        # Repair after mutation
