import numpy as np
import time

def compute_c5(h):
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_capped_simplex(v, target):
    # Project onto [0,1] hypercube and sum = target using water-filling.
    # Binary search for lambda.
    lo = np.min(v - 1)  # ensures all vi - lo >= 1
    hi = np.max(v)      # ensures all vi - hi <= 0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        y = np.clip(v - lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def ensure_feasible(h):
    h = np.clip(h, 0.0, 1.0)
    target = h.shape[0] / 2.0
    if abs(h.sum() - target) > 1e-12:
        h = project_capped_simplex(h, target)
    return h

def propose_pair_transfer(h, max_step, rng):
    N = h.shape[0]
    i, j = rng.choice(N, size=2, replace=False)
    # Randomly decide direction of transfer
    if rng.random() < 0.5:
        donor, recv = i, j
    else:
        donor, recv = j, i
    # allowable amount
    max_transfer = min(max_step, h[donor], 1.0 - h[recv])
    if max_transfer <= 0.0:
        return h  # no feasible move
    delta = rng.random() * max_transfer
    new_h = h.copy()
    new_h[donor] -= delta
    new_h[recv] += delta
    return new_h

def propose_global_perturb(h, max_step, rng):
    target = h.shape[0] / 2.0
    pert = rng.normal(scale=max_step, size=h.shape)
    new = h + pert
    new = np.clip(new, 0.0, 1.0)
    # adjust sum to target via projection
    new = project_capped_simplex(new, target)
    return new

    rng = np.random.default_rng(seed)
    # Determine number of points

    cur_h = best_h.copy()
    cur_val = best_val

    final_step = 1e-4
    init_temp = 0.005
    final_temp = 1e-7

    prob_global = 0.02

    while True:
        now = time.perf_counter()
        elapsed = now - start
        if elapsed >= budget_s * 0.95:

        progress = elapsed / budget_s
        # decreasing step and temperature
        step = init_step * (1 - progress) + final_step * progress
        temp = init_temp * (1 - progress) + final_temp * progress

    return h

import time
import numpy as np

def _c5_of(h):
    """Return the C5 value for a sampled function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * float(np.max(corr))

def _project_capped_simplex(v, target):
    """
    Project v onto the hypercube [0,1]^n while enforcing sum(v)=target.
    Uses water‑filling (binary search on a Lagrange multiplier).
    """
    lo = np.min(v - 1.0)          # all entries become 1 after shift <= lo
    hi = np.max(v)                # all entries become 0 after shift >= hi
    for _ in range(50):
        lam = (lo + hi) * 0.5
        y = np.clip(v - lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def _ensure_feasible(h):
    """Clip to [0,1] and project to exact sum=n/2."""
    n = h.shape[0]
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target) > 1e-12:
        h = _project_capped_simplex(h, target)
    return h

def _subgradient(h, shift):
    """
    Sub‑gradient of the maximal correlation term for a given integer shift.
    shift ∈ {-(n‑1), …, n‑1}.
    """
    n = h.shape[0]
    g = np.zeros(n, dtype=float)
    s = shift
    # overlapping indices where both i and i+s lie in [0,n)
    start = max(0, -s)
    end = min(n, n - s)          # exclusive
    if start < end:
        g[start:end] = 1.0
        g[start:end] -= h[start + s : end + s]
    # subtract contributions where i is the right hand side of a product
    idx = np.arange(n)
    mask = (idx - s >= 0) & (idx - s < n)
    g[mask] -= h[idx[mask] - s]
    return g

def _random_feasible(n, rng):
    """Return a random feasible vector of length n (0≤h≤1, sum=n/2)."""
    target = n * 0.5
    v = rng.random(n)
    return _project_capped_simplex(v, target)
