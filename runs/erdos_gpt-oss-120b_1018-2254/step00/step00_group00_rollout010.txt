    return new_h

import time
import numpy as np

def evaluate_overlap(h):
    """Return C₅ for a sampled step function h."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(x, target):
    """Project x onto the set {0≤y≤1, sum(y)=target} using a simple bisection."""
    low, high = -2.0, 2.0
    # expand interval until the target sum lies between the clipped sums
    while np.clip(x + low, 0.0, 1.0).sum() > target:
        low *= 2.0
    while np.clip(x + high, 0.0, 1.0).sum() < target:
        high *= 2.0
    for _ in range(60):
        mid = (low + high) * 0.5
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(x + (low + high) * 0.5, 0.0, 1.0)

def random_feasible(n, target):
    """Generate a random feasible vector of length n."""
    x = np.random.rand(n)
    return project_to_feasible(x, target)

def targeted_candidate(h, step):
    """Try to reduce the current worst‑lag overlap by moving mass from a
    high‑contributing 1‑position to a low‑contributing 0‑position."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(corr)
    lag = max_idx - (n - 1)               # shift that attains the max
    if lag >= 0:
        i_idx = np.arange(0, n - lag)
        j_idx = i_idx + lag
    else:
        lag_abs = -lag
        i_idx = np.arange(lag_abs, n)
        j_idx = i_idx - lag_abs
    contrib = h[i_idx] * (1.0 - h[j_idx])
    if contrib.size == 0:
        return None
    k = np.argmax(contrib)
    i, j = i_idx[k], j_idx[k]

    delta = min(max_dec_i, max_inc_j)
    if delta <= 0:
        return None
    delta *= np.random.rand()
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h
