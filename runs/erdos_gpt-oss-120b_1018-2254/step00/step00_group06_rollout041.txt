def project_feasible(v, target_sum):
    # find λ s.t sum(clipped(v + λ)) = target_sum
    lo = -np.max(v) - 1  # ensure v+λ <=0 for all at low bound.
    hi = 1 - np.min(v) + 1 # ensure v+λ >=1 for all at high bound.
    for _ in range(50):
        mid = (lo + hi) / 2
        projected = np.clip(v + mid, 0, 1)
        s = projected.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    projected = np.clip(v + (lo+hi)/2, 0, 1)
    # after clipping sum should be close to target; due to floating errors may not be exact.
    # We can then adjust by distributing remaining error uniformly over non-saturated indices, but it's fine.
    # Could also rescale by adjusting small epsilon.
    return projected

import numpy as np
import time
import math
import random

def project_feasible(v: np.ndarray, target_sum: float, tol: float = 1e-12, max_iter: int = 100) -> np.ndarray:
    """
    Project vector v onto the set {x in [0,1]^n : sum(x) = target_sum}.
    Uses bisection on the shift lambda.
    """
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
        if hi - lo < tol:
            break
    proj = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction if rounding error left the sum slightly off
    diff = proj.sum() - target_sum
    if abs(diff) > 1e-9:
        free = (proj > 1e-12) & (proj < 1 - 1e-12)
        if free.any():
            proj[free] -= diff / free.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def random_feasible(n: int, target_sum: float) -> np.ndarray:
    """Generate a random feasible vector of length n."""
    v = np.random.rand(n)
    return project_feasible(v, target_sum)

def evaluate_h(h: np.ndarray, dx: float):
    """
    Compute C5 for a given h and return (C5, index_of_max_lag).
    """
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    return float(corr[idx] * dx), idx
