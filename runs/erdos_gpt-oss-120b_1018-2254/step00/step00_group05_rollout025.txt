import numpy as np
import time

def _project_to_simplex_bounds(v: np.ndarray, total: float) -> np.ndarray:
    """Project vector v onto the set {0<=x_i<=1, sum(x)=total}."""
    lo = v.min() - 1.0        # yields sum=N (all clipped to 1)
    hi = v.max()              # yields sum=0 (all clipped to 0)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        y = np.clip(v - mid, 0.0, 1.0)
        s = y.sum()
        if s > total:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - hi, 0.0, 1.0)
    # tiny correction to hit the exact total (tolerance)
    diff = total - y.sum()
    if abs(diff) > 1e-10:
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, 0.0, 1.0)
        else:
            y += diff / len(y)
            y = np.clip(y, 0.0, 1.0)
    return y


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretised) that minimizes
    C5 = max_k ∫ h(x)(1-h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_score = np.inf
    best_n = None

    # Determine which grid sizes to try
    have_initial = 'initial_h_values' in globals()
    n_candidates = []
    if have_initial:
        n_candidates.append(len(initial_h_values))
    # add a few extra sizes for diversification
    n_candidates.extend([100, 200])
    # unique preserving order
    seen = set()
    n_candidates = [x for x in n_candidates if not (x in seen or seen.add(x))]

    for N in n_candidates:
        # limit restarts per size
        restarts = 0
        while time.time() - start_time < budget_s * 0.95:
            # ---- initialise a feasible vector h ----
            if have_initial and N == len(initial_h_values):
                h = np.asarray(initial_h_values, dtype=np.float64).copy()
                h = np.clip(h, 0.0, 1.0)
                h = _project_to_simplex_bounds(h, N / 2.0)
            else:
                rnd = np.random.rand(N)
                h = _project_to_simplex_bounds(rnd, N / 2.0)

            # ---- optimisation parameters ----
            lr = 0.5                     # learning rate
            beta = 5.0                   # softmax temperature (larger -> sharper max)
            max_iters = 3000
