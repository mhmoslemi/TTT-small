def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, target_sum):
    # clip to [0,1] then adjust sum
    h = np.clip(h, 0.0, 1.0)
    # Use iterative water-filling
    for _ in range(20):
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:

        free = (h > 0) & (h < 1)
        nfree = free.sum()
        if nfree == 0:
            # distribute uniformly then clip
            h += diff / len(h)

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_h(h, target_sum):
    # Clip to bounds
    h = np.clip(h, 0.0, 1.0)
    # iterative projection onto sum constraint
    for _ in range(30):
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        # Identify free indices (not at bounds)
        free = (h > 0) & (h < 1)
        nfree = free.sum()
        if nfree == 0:
            # adjust uniformly, then clip
            h += diff / len(h)
        else:
            inc = diff / nfree
            h[free] += inc

    cur = h.sum()
    if abs(cur - target_sum) > 1e-10:
        h = h * (target_sum / cur)
        h = np.clip(h, 0.0, 1.0)

import numpy as np
import time

def compute_c5(h):
    """Return C5 for a discrete step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h); np.correlate implements cross‑correlation with reversal of the second argument
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_h(h, target_sum):
    """Project a vector onto the feasible set:
       0 ≤ h_i ≤ 1 and sum(h) = target_sum.
    """
    h = np.clip(h, 0.0, 1.0)
    # iterative water‑filling to enforce the sum constraint while respecting bounds
    for _ in range(40):
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        free = (h > 0.0) & (h < 1.0)
        nfree = free.sum()
        if nfree == 0:
            h += diff / h.size
        else:
            inc = diff / nfree
            h[free] += inc
        h = np.clip(h, 0.0, 1.0)
    # final tiny scaling if needed
    cur = h.sum()
    if abs(cur - target_sum) > 1e-10:
        h = h * (target_sum / cur)
        h = np.clip(h, 0.0, 1.0)
    return h
