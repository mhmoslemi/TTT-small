    n = len(v)

    theta = (cssv[rho] - s) / (rho + 1)

def compute_c5(h):

    return dx * np.max(corr)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    # Determine n
    try:
        h0 = np.array(initial_h_values, dtype=float)

import time
import numpy as np
import scipy.optimize as opt

def _compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def _random_feasible_binary(n):
    h = np.zeros(n)
    half = n // 2
    h[:half] = 1.0
    if n % 2:
        h[half] = 0.5
    np.random.shuffle(h)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Initialise from parent construction if present
    # ------------------------------------------------------------------
    try:
        h0 = np.asarray(initial_h_values, dtype=float).ravel()
        n = h0.shape[0]
        # minor correction to guarantee feasibility
        target_sum = n / 2.0
        diff = target_sum - h0.sum()
        if abs(diff) > 1e-9:
            # adjust a non‑boundary entry if possible
            for i in range(n):
                if 0.0 < h0[i] < 1.0:
                    h0[i] += diff
                    break
        h0 = np.clip(h0, 0.0, 1.0)
        best_h = h0.copy()
        best_c5 = _compute_c5(best_h)
    except Exception:
        # no parent; fall back to a simple balanced binary pattern
        n = 50  # even length for convenience
        best_h = _random_feasible_binary(n)
        best_c5 = _compute_c5(best_h)

    # ------------------------------------------------------------------
    # Helper: penalty objective for DE (all but last component are free)
    # ------------------------------------------------------------------
    target_sum = n / 2.0
    de_dim = n - 1
    if de_dim > 0:
        def _de_obj(x):
            last = target_sum - np.sum(x)
            if last < 0.0 or last > 1.0:
                return 1e6 + abs(last) * 1e6
            h = np.empty(n, dtype=float)
            h[:-1] = x
            h[-1] = last
            return _compute_c5(h)

    # ------------------------------------------------------------------
    # Global search with differential evolution (if time permits)
    # ------------------------------------------------------------------
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    if de_dim > 0 and remaining > 30.0:
        # pick a modest number of generations based on remaining time
        maxiter = 200 if remaining > 100 else 80
        bounds_de = [(0.0, 1.0)] * de_dim
        try:
            de_res = opt.differential_evolution(
                _de_obj,
                bounds_de,
                maxiter=maxiter,
                polish=False,

            if de_res.success:

        return _compute_c5(x)
