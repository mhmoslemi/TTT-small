    start = time.time()

    # ---------- base data ----------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n_points = h0.shape[0]
    else:
        n_points = 80
        h0 = np.full(n_points, 0.5, dtype=float)

    target_sum = n_points / 2.0

    # ensure feasibility of the initial vector
    s0 = np.sum(h0)
    if not np.isclose(s0, target_sum):
        h0 = h0 + (target_sum - s0) / n_points
        h0 = np.clip(h0, 0.0, 1.0)
        # simple projection onto the hyperplane if clipping disturbed the sum
        for _ in range(5):
            s = np.sum(h0)
            if np.isclose(s, target_sum, atol=1e-12):
                break
            diff = target_sum - s
            free = (h0 > 0.0) & (h0 < 1.0)
            if not np.any(free):
                break
            h0[free] += diff / np.sum(free)
            h0 = np.clip(h0, 0.0, 1.0)

    # ---------- helpers ----------
    def eval_h(h):
        return evaluate_erdos_solution(np.asarray(h, dtype=float))

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    eq_cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}

    # ---------- baseline ----------
    best_h = h0.copy()
    best_score = eval_h(best_h)

    rng = np.random.RandomState(seed)

    # ---------- multi‑restart SLSQP ----------
    max_restarts = 8
    for r in range(max_restarts):
        if time.time() - start > budget_s * 0.9:
            break
        if r == 0:
            x0 = best_h.copy()
        else:
            noise = rng.normal(scale=0.2, size=n_points)
            x0 = h0 + noise
            x0 = np.clip(x0, 0.0, 1.0)

        try:
            res = minimize(
                eval_h,
                x0,
                method='SLSQP',
                bounds=bounds,
                constraints=[eq_cons],
                options={'maxiter': 350, 'ftol': 1e-8, 'disp': False},
            )
        except Exception:
            continue

        cand = np.clip(res.x, 0.0, 1.0)
        # enforce the sum constraint more strictly if needed
        sc = np.sum(cand)
        if not np.isclose(sc, target_sum, atol=1e-6):
            cand = cand + (target_sum - sc) / n_points
            cand = np.clip(cand, 0.0, 1.0)

        cand_score = eval_h(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_h = cand.copy()
            if best_score < 0.3808:
                break

    # ---------- simple coordinate hill‑climb ----------
    max_iters = 2000
    step_max = 0.05
    n = n_points
    while time.time() - start < budget_s * 0.95 and max_iters > 0:
        max_iters -= 1
        i, j = rng.choice(n, size=2, replace=False)
        delta_max = min(step_max, best_h[i], 1.0 - best_h[j])
        if delta_max <= 0:
            continue
        delta = rng.uniform(0.0, delta_max)
        if delta == 0.0:
            continue
        new_h = best_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_score = eval_h(new_h)
        if new_score < best_score:
            best_score = new_score
            best_h = new_h
            if best_score < 0.3808:
                break

    # ---------- final feasibility fix ----------
    best_h = np.clip(best_h, 0.0, 1.0)
    s = np.sum(best_h)
    if not np.isclose(s, target_sum, atol=1e-5):
        best_h = best_h + (target_sum - s) / n_points
        best_h = np.clip(best_h, 0.0, 1.0)

    final_score = eval_h(best_h)
    return (best_h, float(final_score), n_points)
