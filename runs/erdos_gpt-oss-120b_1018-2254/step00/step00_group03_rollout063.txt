        # else reject move

    # after loop, ensure best_h feasible
    best_h = project_to_feasible(best_h)
    best_c5 = evaluate(best_h)
    return (best_h, best_c5, n)

def _evaluate(h):
    """Return the C5 bound for a step function represented by array h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode="full")) * dx


def _project_feasible(v):
    """
    Project a vector v onto the feasible set:
      0 <= v_i <= 1   and   sum(v) = n/2  (where n = len(v)).
    The routine uses a simple iterative redistribution on non‑boundary entries.
    """
    n = v.shape[0]
    target = n * 0.5
    v = np.clip(v, 0.0, 1.0)

        v[free] += adjust
        v = np.clip(v, 0.0, 1.0)
        diff = target - v.sum()
        if abs(diff) < 1e-12:
            break
    return v
