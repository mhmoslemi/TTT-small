def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    # correlation returns array of length 2*n - 1
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def initial_guess(n):
    global initial_h_values
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
        if n == n0:
            return h0.copy()
        # Interpolate
        x_old = np.linspace(0, 2, n0, endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        h_new = np.interp(x_new, x_old, h0)
        # Normalize sum to n/2 (preserve integral)
        target_sum = n / 2.0
        cur_sum = np.sum(h_new)
        if cur_sum != 0:
            h_new = h_new * (target_sum / cur_sum)
        # Clip to [0,1] just in case
        h_new = np.clip(h_new, 0.0, 1.0)
        # After clipping, adjust sum again if needed: simple rescale again (but may go over)
        cur_sum = np.sum(h_new)
        if cur_sum != target_sum:
            # distribute remaining uniformly where feasible
            diff = target_sum - cur_sum
            # add diff/n equally but ensure bounds
            # simplistic: add diff/n to all entries, then clip again
            h_new += diff / n
            h_new = np.clip(h_new, 0.0, 1.0)
            # final rescale to target sum (small error)
            cur_sum = np.sum(h_new)
            if cur_sum > 0:

    else:

def project_capped_simplex(v, s):
    # Input v: numpy array, s target sum, bounds [0,1]
    # Ensure feasible region: 0 <= s <= len(v)
    if s < 0:
        raise ValueError
    if s > len(v):
        raise ValueError
    # Binary search for lambda
    # initial bounds for lambda: low = -np.max(v) - 1 (ensures all clipped to 1)
    # high = np.max(v) + 1 (ensures all clipped to 0)
    low = -np.max(v) - 1
    high = np.max(v) + 1
    # tolerance maybe 1e-12
    for _ in range(100):
        mid = (low + high) / 2.0
        # compute projection for given lambda
        proj = np.clip(v - mid, 0, 1)
        cur_sum = proj.sum()
        if cur_sum > s:
            low = mid
        else:
            high = mid
    proj = np.clip(v - (low + high) / 2.0, 0, 1)
    # due to floating errors, adjust final sum by small scaling
    # final sum may be slightly off; adjust uniformly within bounds:
    cur_sum = proj.sum()
    if not np.isclose(cur_sum, s, atol=1e-8):
        # distribute diff proportionally among non-boundary indices
        diff = s - cur_sum
        # find indices not at 0 or 1
        free = (proj > 0) & (proj < 1)
        if free.any():
            # add diff / count to each free element
            adjustment = diff / free.sum()
            proj[free] += adjustment
            # clip again just in case
            proj = np.clip(proj, 0, 1)
        else:
            # if all at bounds, just scale the whole vector (unlikely)
            proj = proj * (s / cur_sum) if cur_sum != 0 else proj
    return proj

import time
import numpy as np
from scipy import optimize as opt

def compute_c5(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)*(1-h(x+k)) dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(dx * np.max(corr))

def project_capped_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the set {x | 0<=x_i<=1, sum x_i = s}.
    Uses a simple bisection on the offset λ such that sum clip(v-λ,0,1)=s.
    """
    n = v.size
    if s <= 0:
        return np.zeros_like(v)
    if s >= n:
        return np.ones_like(v)

    # initial bracketing for λ
    lo = -np.max(v) - 1.0   # makes v-λ >1 => all clipped to 1
    hi = np.max(v) + 1.0    # makes v-λ <0 => all clipped to 0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        cur = proj.sum()
        if cur > s:
            lo = mid
        else:
            hi = mid
    λ = (lo + hi) / 2.0
    proj = np.clip(v - λ, 0.0, 1.0)
    # tiny adjustment to hit the exact sum (avoid rounding drift)
    diff = s - proj.sum()
    if abs(diff) > 1e-10:
        free = (proj > 0.0) & (proj < 1.0)
        if free.any():
            proj[free] += diff / free.sum()
            proj = np.clip(proj, 0.0, 1.0)
        else:
            # fallback uniform scaling (should be rare)
            if proj.sum() > 0:
                proj *= s / proj.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

class C5PenaltyObjective:
    """Penalty objective: C5 plus quadratic penalty for sum constraint."""
    __slots__ = ("target", "penalty")
    def __init__(self, target_sum: float, penalty: float = 1e4):
        self.target = target_sum
        self.penalty = penalty
    def __call__(self, x: np.ndarray) -> float:
        c5 = compute_c5(x)
        diff = float(x.sum() - self.target)
        return c5 + self.penalty * diff * diff

def upsample_h(old_h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a step function to a new number of points."""
    old_n = old_h.shape[0]
    if old_n == new_n:
        return old_h.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    new_h = project_capped_simplex(new_h, new_n / 2.0)
    return new_h

def local_improve(h: np.ndarray, iters: int, step: float = 0.1) -> tuple:
    """
    Simple swap‑move hill‑climbing that preserves the total sum.
    Randomly transfers a small amount δ between two coordinates.
    """
    n = h.size
    best = h.copy()
    best_val = compute_c5(best)
    for _ in range(iters):
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        δ = (np.random.rand() - 0.5) * step
        hi = best[i] + δ
        hj = best[j] - δ
        if not (0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0):
            continue
        cand = best.copy()
        cand[i] = hi
        cand[j] = hj
        val = compute_c5(cand)
        if val < best_val:
            best = cand
            best_val = val
    return best, best_val

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    best_h = None
    best_c5 = np.inf
    best_n = None

    # candidate grid sizes (more points give finer control)
    n_candidates = [45, 60, 80, 100, 120]

    for n in n_candidates:
        # time guard
        if time.time() - start_time > budget_s - 5.0:
            break

        target_sum = n / 2.0

        # ----- initial feasible guess -----
        try:
            # use the parent construction if present
            init_h = np.array(initial_h_values, dtype=float)
            init_h = upsample_h(init_h, n)
        except Exception:
            init_h = np.full(n, 0.5)  # perfectly feasible
        init_h = project_capped_simplex(init_h, target_sum)

        # ----- differential evolution (global search) -----
        obj = C5PenaltyObjective(target_sum, penalty=1e4)
        bounds = [(0.0, 1.0)] * n

        # population: include the current guess
        popsize = max(5, n // 15)
        population = np.random.rand(popsize, n)
        population[0] = init_h  # seed with feasible guess

        # adapt iteration count to remaining budget
        remaining = budget_s - (time.time() - start_time)
        maxiter = 150 if remaining > 300 else 50

        try:
            de_res = opt.differential_evolution(
                obj,
                bounds,
                strategy="best1bin",
                popsize=popsize,
                maxiter=maxiter,
                polish=False,
                init=population,
                updating="deferred",
                seed=seed + n,
                atol=0,
                disp=False,
            )
            de_best = de_res.x
        except Exception:
            de_best = init_h.copy()

        # ----- local refinement (gradient‑free) -----
        # L‑BFGS‑B with bounds, still using the penalty objective
        try:
            loc_res = opt.minimize(
                obj,
                de_best,
                method="L-BFGS-B",
                bounds=bounds,
                options={"maxfun": 4000, "ftol": 1e-9},
            )
            cand = loc_res.x if loc_res.success else de_best
        except Exception:
            cand = de_best

        # enforce exact feasibility
        cand = project_capped_simplex(cand, target_sum)

        # ----- cheap hill‑climbing refinement -----
        # allocate a modest amount of iterations based on remaining time
        remaining = budget_s - (time.time() - start_time)
        if remaining > 2.0:
            iters = int(min(3000, remaining * 200))  # ~200 it/s estimate
            cand, _ = local_improve(cand, iters, step=0.2)
            cand = project_capped_simplex(cand, target_sum)

        cur_c5 = compute_c5(cand)

        if cur_c5 < best_c5 - 1e-9:
            best_c5 = cur_c5
            best_h = cand.copy()
            best_n = n

        # early stop if we already beat the target bound
        if best_c5 <= 0.3808:
            break

    # fallback if nothing succeeded
    if best_h is None:
        best_n = 45
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    # final safety projection
    best_h = project_capped_simplex(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)

    return (best_h, float(best_c5), int(best_n))
