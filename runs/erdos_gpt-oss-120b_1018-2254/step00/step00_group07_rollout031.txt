import numpy as np
import time
from scipy.optimize import minimize

# globals that will be set in run()
_GLOBAL_N = None               # number of discretization points
_GLOBAL_TARGET_SUM = None      # required sum of h values (n/2)
_GLOBAL_DX = None              # step size 2/n

def compute_c5(x):
    """Return the Erdős C5 value for a given step function vector x."""
    # correlation of x with (1-x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    return float(np.max(corr) * _GLOBAL_DX)

def eq_constraint_func(x):
    """Equality constraint: sum(x) = target sum."""
    return float(np.sum(x) - _GLOBAL_TARGET_SUM)

def project_to_feasible(v):
    """Project a vector onto the box [0,1]^n and the hyperplane sum=v_target."""
    v = np.clip(v, 0.0, 1.0)
    target = _GLOBAL_TARGET_SUM
    s = np.sum(v)
    if s == 0.0:
        # uniform fill if everything is zero
        return np.full_like(v, target / _GLOBAL_N)
    # first scale to hit the target sum (may violate bounds)
    factor = target / s
    v = v * factor
    # iterate small adjustments to respect bounds while preserving the sum
    for _ in range(10):
        v = np.clip(v, 0.0, 1.0)
        s = np.sum(v)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        free = (v > 0.0) & (v < 1.0)
        cnt = np.count_nonzero(free)
        if cnt == 0:
            # cannot distribute further; break
            break
        # distribute the remaining difference equally among free entries
        v[free] += diff / cnt
    return np.clip(v, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h that lowers the C5 constant."""
    rng = np.random.default_rng(seed)

    # Determine number of points and initialise h
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
        n = h0.size
    else:
        n = 80
        h0 = np.full(n, 0.5, dtype=np.float64)  # uniform starter

    # Set globals for the helper functions
    global _GLOBAL_N, _GLOBAL_TARGET_SUM, _GLOBAL_DX
    _GLOBAL_N = n
    _GLOBAL_TARGET_SUM = n / 2.0
    _GLOBAL_DX = 2.0 / n

    # Make the starter feasible
    h0 = project_to_feasible(h0)

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # timing
    start_time = time.time()
    def time_left():
        return budget_s - (time.time() - start_time)

    # bounds for SLSQP
    bounds = [(0.0, 1.0)] * n
    eq_cons = {"type": "eq", "fun": eq_constraint_func}

    # Main optimisation loop
    restart = 0
    while time_left() > 5.0 and restart < 500:
        restart += 1

        # Create a perturbed initial guess from the current best
        x0 = best_h.copy()
        num_moves = rng.integers(5, 15)
        for _ in range(num_moves):
            i, j = rng.choice(n, size=2, replace=False)
            max_delta = min(x0[i], 1.0 - x0[j])
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta * 0.5
            x0[i] -= delta
            x0[j] += delta
        x0 = project_to_feasible(x0)

        # SLSQP local optimisation
        try:
            res = minimize(
                compute_c5,
                x0,
                method="SLSQP",
                bounds=bounds,
                constraints=[eq_cons],
                options={"maxiter": 200, "ftol": 1e-9, "disp": False},
            )
        except Exception:
            continue

        if not res.success:
            continue

        cand = project_to_feasible(res.x)
        cand_val = compute_c5(cand)

        if cand_val < best_val:
            best_h, best_val = cand, cand_val
            if best_val <= 0.3808:
                break

        # Simple hill‑climbing refinement on the candidate
        cur_h = cand.copy()
        cur_val = cand_val
        for _ in range(30):
            if time_left() < 1.0:
                break
            i, j = rng.choice(n, size=2, replace=False)
            max_delta = min(cur_h[i], 1.0 - cur_h[j])
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta * 0.5
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
            new_val = compute_c5(new_h)
            if new_val < cur_val:
                cur_h, cur_val = new_h, new_val

        if cur_val < best_val:
            best_h, best_val = cur_h, cur_val
            if best_val <= 0.3808:
                break

    # Final projection to guarantee feasibility and recompute exact value
    best_h = project_to_feasible(best_h)
    final_val = compute_c5(best_h)

    return (best_h, float(final_val), n)
