def project_onto_bounds_and_sum(y, S):
    import cvxpy as cp
    n = len(y)
    x = cp.Variable(n)
    prob = cp.Problem(cp.Minimize(cp.sum_squares(x - y)),
                      [cp.sum(x) == S,
                       x >= 0,
                       x <= 1])
    prob.solve(solver=cp.ECOS, warm_start=True)
    if x.value is None:
        # fallback simple scaling
        x_proj = np.clip(y, 0, 1)
        cur = x_proj.sum()
        if cur > 0:
            x_proj = x_proj * (S / cur)
            x_proj = np.clip(x_proj, 0, 1)
        return x_proj
    return np.array(x.value).flatten()
```

        h = start_h.copy()
        cur_c5 = compute_c5(h)
        # SA parameters

        cooling = 0.9995

            # propose move
            i = rng.integers(n)
            j = rng.integers(n-1)

                h_new[i] += delta
                h_new[j] -= delta

import time, math
import numpy as np

def _c5_of(h: np.ndarray) -> float:
    """Return the C5 overlap value for a given discrete h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h) using zero padding
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _make_feasible(h: np.ndarray) -> np.ndarray:
    """Project onto the set {0<=h<=1, sum(h)=n/2} (simple heuristic)."""
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur != 0:
        h *= target / cur
    h = np.clip(h, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        mask = (h > 0.0) & (h < 1.0)
        if mask.any():
            h[mask] += residual / mask.sum()
        else:
            h[:] = 0.5
    return np.clip(h, 0.0, 1.0)


def _random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector satisfying the constraints."""
    y = rng.random(n)
    target = n / 2.0
    if y.sum() == 0:
        y = np.full(n, 0.5)
    y = y * (target / y.sum())
    y = np.clip(y, 0.0, 1.0)
    cur = y.sum()
    if abs(cur - target) > 1e-12:
        mask = (y > 0.0) & (y < 1.0)
        if mask.any():
            y[mask] += (target - cur) / mask.sum()
        else:
            y[:] = 0.5
    return np.clip(y, 0.0, 1.0)

    """
    Optimize the step function h to obtain a low C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    def _anneal(start_h: np.ndarray):
        nonlocal best_h, best_c5
        h = start_h.copy()
        cur_c5 = _c5_of(h)

        # annealing parameters (tuned for speed/quality)
        temp = 0.05
        cooling = 0.9995
        step = 0.2
        it = 0

        while True:
            # time‑budget check
            if time.time() - start_time > budget_s * 0.95:
                break

            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            if rng.random() < 0.5:
                # move mass from j to i
                max_move = min(step, 1.0 - h[i], h[j])
                if max_move <= 0.0:
                    continue

                h_new = h.copy()
                h_new[i] += delta
                h_new[j] -= delta
            else:
                # move mass from i to j
                max_move = min(step, h[i], 1.0 - h[j])
