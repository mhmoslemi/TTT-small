import numpy as np
import time

def _compute_c5(h, dx):
    """Return max overlap integral C5 for a discretized h."""
    # full correlation of h with (1-h)
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def _project_to_constraints(h, target_sum, rng):
    """Adjust h (in [0,1]) to have exactly target_sum while staying in bounds."""
    cur = h.sum()
    if cur == 0:
        # distribute uniformly
        h[:] = target_sum / len(h)
        return h
    # scale proportionally
    h *= target_sum / cur
    # clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # correct any remaining small error
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # try to add diff to a non‑boundary entry
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            free = np.arange(len(h))
        idx = rng.choice(free)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] giving a low C5 bound.

    Returns:
        h_values (np.ndarray): sampled values of h, length n_points.
        c5_bound (float): achieved upper bound for C5.
        n_points (int): number of discretization points.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initial construction
    # ------------------------------------------------------------------
    init = globals().get("initial_h_values", None)
    if init is not None:
        h = np.array(init, dtype=float).copy()
        n = h.size
    else:
        n = 80
        h = np.full(n, 0.5, dtype=float)
        # tiny random perturbation

        h = np.clip(h, 0.0, 1.0)

    target_sum = n / 2.0
    h = _project_to_constraints(h, target_sum, rng)

    dx = 2.0 / n
    current_c5 = _compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = current_c5

    start_time = time.time()
    max_no_improve = 1500
    no_improve = 0
    # temperature parameters for occasional uphill moves
    T_start = 0.02
    T_end = 1e-4

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:  # safety margin
            break

        # ------------------------------------------------------------------

        # ------------------------------------------------------------------
