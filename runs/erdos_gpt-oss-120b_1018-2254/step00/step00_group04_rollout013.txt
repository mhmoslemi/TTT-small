def project_to_constraints(h, target_sum):
    # Clip to [0,1] and then adjust to satisfy equality via simple iterative water-filling.

    h = np.clip(h, 0.0, 1.0)
    # compute sum deficiency
    sum_h = h.sum()
    # If sum close to target, done.
    if abs(sum_h - target_sum) < 1e-8:
        return h
    # Compute needed adjustment
    # Use iterative scaling:
    # While not converged:
    for _ in range(100):
        delta = target_sum - sum_h
        # Identify free indices (not at bound) that can receive addition/subtraction.
        if delta > 0:
            free = (h < 1 - 1e-12)
        else:
            free = (h > 1e-12)
        free_idx = np.where(free)[0]
        if free_idx.size == 0:
            break
        # distribute delta uniformly
        add = delta / free_idx.size
        # Limit addition to keep within [0,1]
        for idx in free_idx:

def project_to_constraints(h, target):
    h = np.clip(h, 0, 1)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Compute free mask
    free = np.ones_like(h, dtype=bool)
    # For each iteration:
    while True:
        free_idx = np.where(free)[0]
        if free_idx.size == 0:
            break
        # Allocate diff evenly over free variables
        alloc = diff / free_idx.size
        # Apply allocation, but clip within [0,1]
        h_new = h.copy()
        h_new[free_idx] += alloc
        # Fix those that go out-of-bounds
        over = h_new > 1
        under = h_new < 0
        # For indices that exceed bound, set to bound, adjust diff accordingly, and mark them not free in next iteration
        # Compute actual updates:
        delta = np.zeros_like(h)
        delta[free_idx] = alloc
        # compute adjustments where exceed bound:
        exceeded_up = (h_new > 1) & free
        exceeded_down = (h_new < 0) & free
        # modifications:
        delta[exceeded_up] = 1 - h[exceeded_up]
        delta[exceeded_down] = -h[exceeded_down]
        h[free] += delta[free]
        # recompute diff
        diff = target - h.sum()
        # update free mask
        free = (h > 0) & (h < 1)
        if abs(diff) < 1e-12:
            break
        # if diff small else continue
    return h

def optimize_with_slsqp(h0):
    n = len(h0)
    target = n/2.0
    bounds = [(0.0, 1.0)] * n
    constraints = ({'type':'eq', 'fun': lambda x: np.sum(x) - target,
                    'jac': lambda x: np.ones_like(x)})
    # objective returns c5
    def obj(x):
        c5 = np.max(np.correlate(x, 1 - x, mode='full')) * (2.0 / n)
        return c5
    # gradient
    def jac(x):
        # compute c5 and gradient using shift with max value
        dx = 2.0 / n
        conv = np.correlate(x, 1 - x, mode='full') * dx
        max_idx = int(np.argmax(conv))
        k = max_idx - (n - 1)
        g = np.full(n, dx)
        if k != 0:
            # first term i+k in range
            if k > 0:
                i1 = np.arange(0, n - k)
                g[i1] -= dx * x[i1 + k]
                i2 = np.arange(k, n)
                g[i2] -= dx * x[i2 - k]
            else:
                # k negative
                kp = -k
                # i + k = i - kp
                i1 = np.arange(kp, n)  # because i >= kp to have i - kp >=0
                g[i1] -= dx * x[i1 + k]  # i + k = i - kp
                i2 = np.arange(0, n - kp)
                g[i2] -= dx * x[i2 - k]  # i - k = i + kp
        # gradient of equality constraint not added; optimizer handles constraints
        return g
    # call minimize
    res = scipy.optimize.minimize(obj, h0, method='SLSQP', jac=jac,
                                 bounds=bounds,
                                 constraints=constraints,
                                 options={'maxiter': 200, 'ftol': 1e-9, 'disp': False})
    return res.x, res.fun, res.success

def project_to_constraints(h, target):
    """Clip h to [0,1] and project onto sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    # compute current sum
    sum_h = h.sum()
    if abs(sum_h - target) < 1e-12:
        return h
    # iterative water-filling
    # We'll use while loop limit
    max_iter = 1000
    for _ in range(max_iter):
        diff = target - sum_h
        # indices not at bounds
        free = (h > 0) & (h < 1)
        n_free = free.sum()
        if n_free == 0:
            # cannot adjust further
            break
        # allocate equally
        delta = diff / n_free
        # compute new values, enforce bounds in this step
        h_new = h.copy()
        # apply delta to free indices
        h_new[free] += delta
        # handle overflows
        over = h_new > 1
        under = h_new < 0
        # if over, set to 1 and adjust diff accordingly
        if np.any(over):
            # compute actual change applied to each over element: from old h to 1
            actual_delta = 1 - h[over]
            diff -= actual_delta.sum()
            h_new[over] = 1
        if np.any(under):
            actual_delta = -h[under]  # negative amount removed
            diff -= actual_delta.sum()
            h_new[under] = 0
        # any remaining diff distributed among remaining free (excluding those now at bound)
        h = h_new
        sum_h = h.sum()
        if abs(sum_h - target) < 1e-12:
            break
    # final tweak: if still off slightly, scale
    sum_h = h.sum()
    diff = target - sum_h
    if abs(diff) > 1e-8:
        # distribute uniformly (small)
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        # final scaling to adjust sum exactly if needed
        sum_h = h.sum()
        if abs(sum_h - target) > 1e-8:
            scale = target / sum_h
            h = np.clip(h * scale, 0.0, 1.0)
    return h

# Helper: evaluate C5 for a given step function h (discretized)
def _evaluate(h):
    n = h.size
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx

# Project vector onto [0,1] box and enforce sum_i h_i = target (exactly)
def _project_to_constraints(h, target):
    h = np.clip(h, 0.0, 1.0)
    sum_h = h.sum()
    if abs(sum_h - target) < 1e-12:
        return h
    # Simple water-filling style projection
    max_iter = 1000
    for _ in range(max_iter):
        diff = target - sum_h
        free = (h > 0.0) & (h < 1.0)
        n_free = free.sum()
        if n_free == 0:
            break
        delta = diff / n_free
        # apply delta to all free entries
        h_new = h.copy()
        h_new[free] += delta
        # enforce bounds and recompute diff
        over = h_new > 1.0
        under = h_new < 0.0
        if np.any(over):
            delta_over = 1.0 - h[over]    # amount actually added
            h_new[over] = 1.0
            diff -= delta_over.sum()
        if np.any(under):
            delta_under = -h[under]      # amount actually removed
            h_new[under] = 0.0
            diff -= delta_under.sum()
        h = h_new
        sum_h = h.sum()
        if abs(sum_h - target) < 1e-12:
            break
    # Final tiny adjustment if needed
    sum_h = h.sum()
    diff = target - sum_h
    if abs(diff) > 1e-8:
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        sum_h = h.sum()
        if abs(sum_h - target) > 1e-8:

            h = np.clip(h * scale, 0.0, 1.0)

# Gradient of the max-overlap objective (subgradient using the current maximizing lag)
def _gradient(x):
    n = x.size
    dx = 2.0 / n
    conv = np.correlate(x, 1.0 - x, mode='full') * dx
    max_idx = int(np.argmax(conv))
    k = max_idx - (n - 1)          # lag achieving the max
    g = np.full(n, dx)            # d/dx of sum_i x_i term
    if k != 0:
        if k > 0:
            i1 = np.arange(0, n - k)
            g[i1] -= dx * x[i1 + k]
            i2 = np.arange(k, n)
            g[i2] -= dx * x[i2 - k]
        else:
            kp = -k
            i1 = np.arange(kp, n)
            g[i1] -= dx * x[i1 + k]      # i+k
            i2 = np.arange(0, n - kp)
            g[i2] -= dx * x[i2 - k]      # i-k
    return g

# Objective for scipy.optimize (returns C5)
def _objective(x):
    n = x.size
    dx = 2.0 / n
    conv = np.correlate(x, 1.0 - x, mode='full') * dx
    return np.max(conv)

# Run a single SLSQP optimization (if scipy is available)
def _slsqp_optimize(h0):
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    constraints = ({
        'type': 'eq',
        'fun': lambda x: np.sum(x) - target,
        'jac': lambda x: np.ones_like(x)
    })
    if opt is None:
        # SciPy not available: just return the starting point
        return h0, _evaluate(h0), False
    res = opt.minimize(_objective, h0, method='SLSQP',
                       jac=_gradient,
                       bounds=bounds,
                       constraints=[constraints],
                       options={'maxiter': 200, 'ftol': 1e-9, 'disp': False})
    # In rare cases optimizer may return infeasible h; enforce constraints
    h_opt = _project_to_constraints(res.x, target)
    return h_opt, _evaluate(h_opt), res.success

# Simple hill‑climbing that preserves the sum exactly by transferring mass
def _hill_climb(h0, time_limit, start_time):
    h = h0.copy()
    best_h = h.copy()
    best_val = _evaluate(h)
    n = h.size
    while time.time() - start_time < time_limit:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # transfer from j to i (ensure i can increase, j can decrease)
        max_inc = 1.0 - h[i]
        max_dec = h[j]
        max_transfer = min(max_inc, max_dec)
        if max_transfer <= 0.0:
            continue
        delta = np.random.rand() * max_transfer
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        val = _evaluate(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new
            h = h_new
    return best_h, best_val
