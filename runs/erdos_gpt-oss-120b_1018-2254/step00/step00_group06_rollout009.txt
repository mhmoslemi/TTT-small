import time
import math
import numpy as np

def _c5_value(vec: np.ndarray) -> float:
    """Compute C5 = max_k ∫ h(x)(1-h(x+k))dx for a discretized vector."""
    corr = np.correlate(vec, 1.0 - vec, mode='full')
    dx = 2.0 / vec.size
    return float(np.max(corr) * dx)


def _project_to_feasible(v: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Clip to [0,1] and enforce sum(v) = n/2 exactly."""
    np.clip(v, 0.0, 1.0, out=v)
    n = v.size
    target = n / 2.0
    diff = target - v.sum()
    # simple redistribution loop
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            candidates = np.where(v < 1.0)[0]
            if candidates.size == 0:
                break
            idx = rng.choice(candidates)
            inc = min(1.0 - v[idx], diff)
            v[idx] += inc
            diff -= inc
        else:
            candidates = np.where(v > 0.0)[0]
            if candidates.size == 0:
                break
            idx = rng.choice(candidates)
            dec = min(v[idx], -diff)
            v[idx] -= dec
            diff += dec
    # tiny residual correction
    residual = target - v.sum()
    if abs(residual) > 1e-12:
        if 0.0 <= v[0] + residual <= 1.0:
            v[0] += residual
        else:
            scale = target / v.sum()
            v *= scale
            np.clip(v, 0.0, 1.0, out=v)
    return v

    """Return a step function minimizing the Erdős C5 overlap bound.

        c5_bound : float, the achieved C5 bound
        n_points : int, number of discretization samples
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------ #
    # Initialise h

    h = _project_to_feasible(h, rng)

    # ------------------------------------------------------------------ #

    iteration = 0

    while True:
        # time guard
        if time.time() - start_time > budget_s - 0.05:
            break
        iteration += 1
        elapsed = time.time() - start_time
        temperature = init_temp * max(0.0, 1.0 - elapsed / budget_s)

        # -------------------------------------------------------------- #
        # Choose move type: targeted (70%) or random (30%)
        if rng.random() < 0.7:
