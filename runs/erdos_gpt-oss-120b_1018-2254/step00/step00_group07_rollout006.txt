    max_steps_per_restart = 4000
    T0 = 0.01  # initial temperature

    # Main exploratory loop – stops before the deadline to keep margin
    while time.time() - start_time < budget_s * 0.85:
        # Random binary start (exactly half 1s)
        h = np.zeros(n, dtype=float)
        idx_one = rng.choice(n, size=n // 2, replace=False)
        h[idx_one] = 1.0

        score = evaluate_erdos_solution(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()

        # Simulated‑annealing swaps
        for step in range(max_steps_per_restart):
            if time.time() - start_time >= budget_s * 0.85:
                break
            T = T0 * (1 - step / max_steps_per_restart)

            # pick a 1 and a 0 to swap
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)

            # perform swap
            h[i] = 0.0
            h[j] = 1.0
            new_score = evaluate_erdos_solution(h)
            delta = new_score - score

            accept = False
            if delta < 0:
                accept = True
            elif T > 0 and rng.random() < np.exp(-delta / T):
                accept = True

            if accept:
                score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_h = h.copy()
            else:
                # revert swap
                h[i] = 1.0
                h[j] = 0.0

        # Early exit if we have already beaten the target
        if best_score < 0.3808:
            break

    # Final local refinement with SLSQP if time permits
    remaining = budget_s - (time.time() - start_time)
    if remaining > 5.0:
        bounds = [(0.0, 1.0)] * n
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        try:
            res = minimize(
                objective,
                best_h,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-9}
            )
            if res.success:
                refined = np.clip(res.x, 0.0, 1.0)
                refined_score = evaluate_erdos_solution(refined)
                if refined_score < best_score:
                    best_h = refined
                    best_score = float(refined_score)
        except Exception:
            # In case the optimizer crashes for any reason, keep current best.
            pass

    return best_h, float(best_score), n
