def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    np.random.seed(seed)
    start = time.time()
    # Determine n_points and initialize
    try:
        init_arr = np.array(initial_h_values, dtype=float)
        n_points = init_arr.size
        has_init = True
    except Exception:
        init_arr = None
        n_points = 45
        has_init = False
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Projection function
    def project(v):
        # v: np.ndarray
        # Clip not needed inside, but do it anyway
        # Binary search for lambda to enforce sum target
        low = -1.0
        high = 2.0
        for _ in range(60):
            lam = (low + high) * 0.5
            clipped = np.clip(v - lam, 0.0, 1.0)
            s = clipped.sum()
            if s > target_sum:
                low = lam
            else:
                high = lam

        return project(v)

    # Evaluate
    def evaluate_vec(vec):
        corr = np.correlate(vec, 1.0 - vec, mode='full')
        return np.max(corr) * dx

    # Random feasible
    def random_feasible():

import time
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    # Set random seed for reproducibility
    np.random.seed(seed)
    start = time.time()

    # ----------------------------------------------------------------------
    # Determine problem size and initial vector (if provided)
    # ----------------------------------------------------------------------
    try:
        init_arr = np.array(initial_h_values, dtype=float)  # pre‑imported if exists
        n_points = init_arr.size
        has_init = True
    except Exception:
        init_arr = None
        n_points = 45           # default size when no parent is given
        has_init = False

    target_sum = n_points / 2.0          # because ∫h = 1 on [0,2]
    dx = 2.0 / n_points

    # ----------------------------------------------------------------------
    # Helper: projection onto the set {0 ≤ h ≤ 1, sum(h)=target_sum}
    # ----------------------------------------------------------------------
    def project_to_sum(v):
        # v is already clipped to [0,1]; we find λ so that sum(clamp(v-λ,0,1))=target_sum
        low, high = -1.0, 2.0          # guarantees sums at extremes are n_points and 0
        for _ in range(60):
            lam = (low + high) * 0.5
            clipped = np.clip(v - lam, 0.0, 1.0)
            s = clipped.sum()
            if s > target_sum:
                low = lam
            else:
                high = lam
        lam = (low + high) * 0.5
        return np.clip(v - lam, 0.0, 1.0)

    def repair(v):
        v = np.clip(v, 0.0, 1.0)
        return project_to_sum(v)

    # ----------------------------------------------------------------------
    # Objective evaluation
    # ----------------------------------------------------------------------
    def evaluate_vec(vec):
        corr = np.correlate(vec, 1.0 - vec, mode='full')
        return np.max(corr) * dx

    # ----------------------------------------------------------------------
    # Random feasible vector generator
    # ----------------------------------------------------------------------
    def random_feasible():
        v = np.random.rand(n_points)
        return repair(v)

    # ----------------------------------------------------------------------
    # Initialize population
    # ----------------------------------------------------------------------
    POP_SIZE = 50
    population = []
    if has_init:
        population.append(repair(init_arr.copy()))
    while len(population) < POP_SIZE:
        population.append(random_feasible())

    fitness = [evaluate_vec(ind) for ind in population]
    best_idx = int(np.argmin(fitness))
    best_h = population[best_idx].copy()
    best_fitness = fitness[best_idx]

    # ----------------------------------------------------------------------
    # Tournament selection (size 3)
    # ----------------------------------------------------------------------
    def tournament():
        idxs = np.random.choice(len(population), size=3, replace=False)
        f_vals = [fitness[i] for i in idxs]
        return population[idxs[np.argmin(f_vals)]]

    # ----------------------------------------------------------------------
    # Genetic algorithm loop
    # ----------------------------------------------------------------------
    MUT_RATE = 0.10          # fraction of entries mutated per child
    MUT_STD  = 0.20          # standard deviation of Gaussian mutation

    while time.time() - start < budget_s * 0.95:
        # produce POP_SIZE offspring
        offspring = []
        offspring_fit = []
        for _ in range(POP_SIZE):
            p1 = tournament()
            p2 = tournament()
            beta = np.random.rand()
            child = beta * p1 + (1.0 - beta) * p2   # convex combo preserves feasibility
            # mutation
            if MUT_RATE > 0:
                num_mut = max(1, int(MUT_RATE * n_points))
                idxs = np.random.choice(n_points, size=num_mut, replace=False)
                child[idxs] += np.random.randn(num_mut) * MUT_STD
                child = repair(child)
            offspring.append(child)
            offspring_fit.append(evaluate_vec(child))

        # combine and keep the best POP_SIZE individuals
        combined = population + offspring
        combined_fit = fitness + offspring_fit
        order = np.argsort(combined_fit)[:POP_SIZE]
        population = [combined[i] for i in order]
        fitness = [combined_fit[i] for i in order]

        # update global best
        if fitness[0] < best_fitness:
            best_fitness = fitness[0]
            best_h = population[0].copy()

        # early exit if we already beat the target
        if best_fitness <= 0.3808:
            break

    # ----------------------------------------------------------------------
    # Final local refinement (simple hill‑climbing)
    # ----------------------------------------------------------------------
    while time.time() - start < budget_s * 0.99:
        cand = best_h.copy()
        num_mut = max(1, int(0.05 * n_points))
        idxs = np.random.choice(n_points, size=num_mut, replace=False)
        cand[idxs] += np.random.randn(num_mut) * 0.1
        cand = repair(cand)
        cand_fit = evaluate_vec(cand)
        if cand_fit < best_fitness - 1e-9:
            best_h = cand
            best_fitness = cand_fit
            if best_fitness <= 0.3808:
                break

    # Ensure final feasibility
    best_h = repair(best_h)
    best_fitness = evaluate_vec(best_h)

    return (best_h, float(best_fitness), n_points)
