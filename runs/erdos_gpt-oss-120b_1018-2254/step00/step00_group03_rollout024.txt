import numpy as np
import time

def max_c5(h):
    """Maximum of the overlap integral for a binary/continuous step function h."""
    n = len(h)
    conv = np.correlate(h, 1 - h, mode='full')
    return np.max(conv) * (2.0 / n)

def binary_initial(n, N, use_initial=False):
    """Create a binary start vector of length n with exactly N ones."""
    if use_initial:
        # Use a deterministic binary version of the provided initial_h_values
        init = np.array(initial_h_values, dtype=float)
        if len(init) == n:
            idx = np.argsort(-init)[:N]
            h = np.zeros(n, dtype=float)
            h[idx] = 1.0
            return h
    # random binary vector
    h = np.concatenate([np.ones(N), np.zeros(N)])
    np.random.shuffle(h)
    return h.astype(float)

def hill_climb_binary(h0, time_budget, start_time, target=0.3808):
    """Greedy hill‑climbing with targeted swaps to lower max_c5."""
    h = h0.copy()
    n = len(h)
    best_h = h.copy()
    best_val = max_c5(h)
    cur_val = best_val

    while time.time() - start_time < time_budget:
        # compute current max lag
        conv = np.correlate(h, 1 - h, mode='full')
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx] * (2.0 / n)

        if max_val < best_val:
            best_val, best_h = max_val, h.copy()
            if best_val <= target:
                break

        shift = max_idx - (n - 1)
        improved = False

        if shift != 0:
            if shift > 0:
                h_shift = np.concatenate([h[shift:], np.zeros(shift, dtype=h.dtype)])
            else:
                k = -shift
                h_shift = np.concatenate([np.zeros(k, dtype=h.dtype), h[:n - k]])

            A = np.where((h == 1) & (h_shift == 0))[0]
            B = np.where((h == 0) & (h_shift == 1))[0]

            if A.size and B.size:
                i = np.random.choice(A)
                j = np.random.choice(B)
                h[i], h[j] = 0.0, 1.0
                new_val = max_c5(h)
                if new_val <= max_val:
                    cur_val = new_val
                    improved = True
                    if new_val < best_val:
                        best_val, best_h = new_val, h.copy()
                        if best_val <= target:
                            break
                else:
                    h[i], h[j] = 1.0, 0.0

        if not improved:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            h[i], h[j] = 0.0, 1.0
            new_val = max_c5(h)
            if new_val <= max_val:
                cur_val = new_val
                if new_val < best_val:
                    best_val, best_h = new_val, h.copy()
                    if best_val <= target:
                        break
            else:
                h[i], h[j] = 1.0, 0.0

    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] with values in [0,1] and
    integral 1 that minimizes the Erdos overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start = time.time()
    target_bound = 0.3808

    # Global best
    best_h = None
    best_val = float('inf')
    best_n = None

    # Use provided initial_h_values if any
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
        val_init = max_c5(h_init)
        best_h, best_val, best_n = h_init, val_init, len(h_init)

    # Candidate grid sizes (even numbers only)
    n_candidates = [70, 60, 80, 50, 90, 100]
    # ensure uniqueness and positivity
    n_candidates = [n for n in n_candidates if n % 2 == 0 and n > 0]

    # Main search loop
    for n in n_candidates:
        if time.time() - start >= budget_s * 0.95:
            break
        N = n // 2
        # Prepare several starting vectors for this n
        starts = []
        if best_n == n and best_h is not None:
            # also try the current global best as a start (binary version)
            topsort = np.argsort(-best_h)[:N]
            h_bin = np.zeros(n, dtype=float)
            h_bin[topsort] = 1.0
            starts.append(h_bin)
        # use binary version of initial_h_values when sizes match
        if 'initial_h_values' in globals() and initial_h_values is not None:
            if len(initial_h_values) == n:
                starts.append(binary_initial(n, N, use_initial=True))
        # add a few random starts
        for _ in range(3):
            starts.append(binary_initial(n, N))

        # Allocate a modest time slice for each n
        time_per_n = (budget_s - (time.time() - start)) / max(1, len(n_candidates) - n_candidates.index(n))
        for h0 in starts:
            if time.time() - start >= budget_s * 0.95:

            # Hill climbing on this start
            h_opt, val_opt = hill_climb_binary(h0, time_per_n, start, target=target_bound)
            if val_opt < best_val:

                if best_val <= target_bound:

                    return best_h, best_val, best_n

    # Final check: recompute bound using the official evaluator
    if best_h is not None:

            best_val = evaluate_erdos_solution(best_h)
        except Exception:
            best_val = max_c5(best_h)

    # Fallback: ensure we always return something
    if best_h is None:
        # simple fallback: uniform 0.5 (satisfies integral)
        n = 70
        best_h = np.full(n, 0.5)
        best_val = max_c5(best_h)
        best_n = n

    return best_h, best_val, best_n
