import numpy as np
import time
import random

# helper: repair vector to satisfy box constraints and exact sum
def _repair(vec, target_sum):
    vec = np.clip(vec, 0.0, 1.0)
    for _ in range(10):
        s = vec.sum()
        if abs(s - target_sum) < 1e-12:
            break
        scale = target_sum / s
        vec = np.clip(vec * scale, 0.0, 1.0)
    s = vec.sum()
    diff = target_sum - s
    if abs(diff) > 1e-12:
        if diff > 0:
            free = 1.0 - vec
            total_free = free.sum()
            if total_free > 0:
                vec += diff * (free / total_free)
        else:
            pos = vec
            total_pos = pos.sum()
            if total_pos > 0:
                vec += diff * (pos / total_pos)   # diff is negative
        vec = np.clip(vec, 0.0, 1.0)
    return vec

# helper: fast C5 evaluation for a discrete vector
def _quick_c5(vec, dx):
    corr = np.correlate(vec, 1 - vec, mode='full')
    return np.max(corr) * dx

# helper: generate a sinusoidal family shape h(x) = sin^p with exact integral 1
def _generate_shape(p, n):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    base = np.sin(np.pi * x / 2.0) ** p
    target = n / 2.0
    h = base * (target / base.sum())
    h = _repair(h, target)
    return h

# helper: resample an existing vector to a new resolution
def _resample(vec, target_n):
    n = len(vec)
    if n == target_n:
        return vec.copy()
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
    return np.interp(x_new, x_old, vec)
