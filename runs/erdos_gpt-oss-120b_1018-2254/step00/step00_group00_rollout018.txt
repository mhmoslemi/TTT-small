        for i in range(len(h)):

            if 0.0 <= new_val <= 1.0:
                h[i] = new_val
                break
    return h

import numpy as np
import time

def compute_c5(h):
    """Maximum overlap integral for a candidate h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.shape[0])

def _project_capped_simplex(v, target):
    """Project v onto the intersection of [0,1]^n and sum == target."""
    lo, hi = -1.0, 1.0
    for _ in range(30):
        lam = (lo + hi) / 2.0
        h_tmp = np.clip(v + lam, 0.0, 1.0)
        s = h_tmp.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

def repair(h, target):
    """Clamp to [0,1] and enforce the exact integral constraint."""
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target) <= 1e-12:
        return h
    h = _project_capped_simplex(h, target)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # distribute the remaining difference to free coordinates
        caps = (1.0 - h) if diff > 0 else (-h)
        order = np.argsort(-caps)  # largest capacity first
        for idx in order:
            cap = caps[idx]
            if cap <= 0:
                continue
            add = min(cap, diff) if diff > 0 else max(-cap, diff)
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

        new[j] -= delta
        score = compute_c5(new)

        c5_bound = float(evaluate_erdos_solution(best_h))
    except NameError:
        c5_bound = float(compute_c5(best_h))

    return (best_h, c5_bound, n)
