    def thue_morse(N):
        seq = np.zeros(N, dtype=int)
        for i in range(N):
            seq[i] = bin(i).count('1') & 1  # parity
        return seq

    h_current = thue_morse(n).astype(float)  # initially binary

        return np.max(conv) * dx

    best_h = h_current.copy()
    best_c5 = compute_c5(best_h)

    # Simulated annealing: set temperature schedule
    T0 = 0.1
    T_end = 1e-5
    max_iters = 2000000  # potentially large

    # Determine schedule based on time.
    iteration = 0
    while True:
        if time.time() - start > budget_s - 0.5:
            break
        iteration += 1
        # choose random pair to swap
        # select indices: i one from ones, j from zeros
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        # For efficiency we could maintain list, but re-select each iteration is heavy; but n small.
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # propose new h by swapping.
        h_new = best_h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        c5_new = compute_c5(h_new)

        # acceptance probability: if lower, accept. else accept with exp(-(c5_new - best_c5)/T)
        # compute temperature schedule
        # linear schedule based on iteration count, adapt to time?
        # Calculate fraction: ratio = iteration / max_iters; if >1, ratio=1
        # But we don't know total iterations; we can define schedule decaying with time: T = T0 * (1 - elapsed / budget) + T_end

def compute_c5(h):
    n = len(h)

    N = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, N)

import time, math
import numpy as np

def thue_morse_seq(N: int) -> np.ndarray:
    """Binary Thue–Morse sequence of length N (0/1 values)."""
    seq = np.empty(N, dtype=float)
    for i in range(N):
        # parity of popcount gives Thue–Morse bit
        seq[i] = (bin(i).count("1") & 1)
    return seq

def max_overlap(h: np.ndarray) -> float:
    """
    Compute C₅ = max_k ∫ h(x)(1−h(x+k))dx using FFT‑based linear correlation.
    Returns the scaled value (including dx = 2/n).
    """
    n = h.size
    g = 1.0 - h
    # length for linear convolution/correlation
    N = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(g[::-1], N)          # reversed g for correlation
    conv = np.fft.irfft(H * G, N)[: 2 * n - 1]
    return np.max(conv) * (2.0 / n)

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a step function on [0,2] with integral 1 that lowers the Erdős C₅."""
    rng = np.random.default_rng(seed)
    start = time.time()

    # problem size – a power of two aids the FFT
    n = 256
    half = n // 2

    # start from Thue–Morse (balanced binary pattern)
    current_h = thue_morse_seq(n)
    current_c5 = max_overlap(current_h)

    best_h = current_h.copy()
    best_c5 = current_c5

    # SA parameters
    T_start = 0.08
    T_end = 1e-5
    no_improve = 0
    restart_limit = 3000

    # pre‑compute indices of ones and zeros for fast swaps
    while True:
        # time guard
        if time.time() - start > budget_s - 0.2:
            break

        # pick a random 1 and a random 0 to swap
        ones = np.flatnonzero(current_h > 0.5)
        zeros = np.flatnonzero(current_h < 0.5)
        i_one = rng.choice(ones)
        i_zero = rng.choice(zeros)

        # propose swap
        new_h = current_h.copy()
        new_h[i_one] = 0.0
        new_h[i_zero] = 1.0
        new_c5 = max_overlap(new_h)

        # temperature schedule (linear decay)
        elapsed = time.time() - start
        frac = elapsed / budget_s
        T = max(T_end, T_start * (1.0 - frac))

        # Metropolis acceptance
        if new_c5 < current_c5 or rng.random() < math.exp(-(new_c5 - current_c5) / T):
            current_h = new_h
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # restart if stuck
        if no_improve >= restart_limit:
            perm = rng.permutation(n)
            restart_h = np.zeros(n, dtype=float)
            restart_h[perm[:half]] = 1.0
            current_h = restart_h
            current_c5 = max_overlap(current_h)
            no_improve = 0

        # good enough?
        if best_c5 < 0.3808:
            break

    return best_h, best_c5, n
