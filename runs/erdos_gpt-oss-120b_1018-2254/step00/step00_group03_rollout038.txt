import time
import math
import numpy as np

def _compute_c5(h):
    """Return the C5 value for a binary step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # cross‑correlation of h with (1‑h) using full convolution
    corr = np.convolve(h, (1.0 - h)[::-1], mode='full')
    return np.max(corr) * dx


def _solve_milp(n, time_limit):
    """
    Solve the binary formulation for a given number of points using MILP.
    Returns (h, t) where h is a binary ndarray and t is the integer
    maximum overlap count (before scaling by dx). If the solver does not
    finish within the time limit, returns (None, None).
    """
    import cvxpy as cp

    start = time.time()
    S = n // 2                     # required half‑sum
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()              # real (will be integer at optimum)

    constraints = [cp.sum(h) == S, t >= 0]

    # create auxiliary variables y_ij = h_i * h_j for all overlapping pairs
    for d in range(-(n - 1), n):
        i_start = max(0, -d)
        i_end = min(n, n - d)
        length = i_end - i_start
        if length <= 0:
            continue
        y = cp.Variable(length, boolean=True)
        hi = h[i_start:i_end]
        hj = h[i_start + d:i_end + d]
        constraints += [
            y <= hi,
            y <= hj,
            y >= hi + hj - 1,
            cp.sum(y) >= S - t,            # enforce max‑overlap bound for this shift
        ]

    prob = cp.Problem(cp.Minimize(t), constraints)

    # try a few solvers, pick the first that works
    for solver in (cp.GLPK_MI, cp.ECOS_BB, cp.CBC, cp.SCIP):
        try:
            prob.solve(solver=solver, verbose=False, max_iters=500000)
            break
        except Exception:
            continue

    # time out handling – if solving took too long we abort
    if time.time() - start > time_limit:
        return None, None

    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None, None

    h_val = np.rint(np.array(h.value)).astype(float)   # force binary 0/1
    t_val = float(t.value)
    return h_val, t_val


def _local_search(h0, max_iters, time_left):
    """
    Simple simulated‑annealing hill climbing on binary vectors.
    Returns the best (h, c5) pair found.
    """
    n = h0.shape[0]
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _compute_c5(h)
    cur_c5 = best_c5

    rng = np.random.default_rng()

    # linear cooling schedule
    T0 = 0.03
    for it in range(1, max_iters + 1):
        if time_left() <= 0.0:
            break
        # temperature
        T = T0 * (1 - it / max_iters)
        # propose a swap of a 1 and a 0
        a = rng.integers(n)
        b = rng.integers(n - 1)
        if b >= a:
            b += 1
        if h[a] == h[b]:
            continue
        # swap
        h[a], h[b] = h[b], h[a]
        new_c5 = _compute_c5(h)
        delta = new_c5 - cur_c5
        accept = False
        if delta <= 0:
            accept = True
        else:
            if T > 0 and rng.random() < math.exp(-delta / T):
                accept = True
        if accept:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert swap
            h[a], h[b] = h[b], h[a]
    return best_h, best_c5

        # start from the current best or from a fresh random vector
        if remaining > 30.0 and rng.random() < 0.3:
            # occasional random restart
            h0 = np.zeros(n, dtype=float)
            h0[:n // 2] = 1.0
            rng.shuffle(h0)
        else:
            h0 = best_h.copy()
        h_candidate, c5_candidate = _local_search(
            h0, max_iters, time_left
        )
        if c5_candidate < best_c5:
            best_h, best_c5 = h_candidate, c5_candidate
            if best_c5 <= TARGET:
                break

    return (best_h, best_c5, n)
