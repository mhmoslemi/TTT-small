import time
import numpy as np

def compute_c5(h):
    """Return the C5 overlap value for a step vector h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / len(h))

def project_feasible(x, target_sum, tol=1e-9, max_iter=20):
    """Clip to [0,1] and adjust to satisfy sum(x)=target_sum (water‑filling)."""
    x = np.clip(x, 0.0, 1.0)
    for _ in range(max_iter):
        total = x.sum()
        diff = target_sum - total
        if abs(diff) <= tol:
            break
        # indices that can move in the needed direction
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        free = np.where(mask)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
    # final tiny correction (distribute to any interior element)
    residual = target_sum - x.sum()
    if abs(residual) > tol:
        interior = np.where((x > 0.0) & (x < 1.0))[0]
        if interior.size == 0:
            interior = np.arange(len(x))
        i = np.random.choice(interior)
        x[i] = np.clip(x[i] + residual, 0.0, 1.0)
    return x

def tournament_select(pop, scores, size=3):
    """Return a copy of a parent chosen by tournament."""
    idx = np.random.choice(len(pop), size, replace=False)
    best = idx[np.argmin(scores[idx])]
    return pop[best].copy()

def uniform_crossover(p1, p2):
    """Uniform crossover between two parents."""
    mask = np.random.rand(len(p1)) < 0.5
    child = np.where(mask, p1, p2)
    return child

def mutate(x, rate=0.2, sigma=0.05):
    """Gaussian mutation applied per coordinate with given probability."""
    mask = np.random.rand(len(x)) < rate
    noise = np.random.randn(len(x)) * sigma
    x = x + mask * noise
    return x

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    # Determine problem size
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)
        n = len(init)
    else:
        n = 45
        init = None
    target_sum = n / 2.0
    pop_size = 50
    # Initialise population
    population = []
    if init is not None:
        population.append(project_feasible(init, target_sum))
    while len(population) < pop_size:
        cand = np.random.rand(n)
        cand = project_feasible(cand, target_sum)
        population.append(cand)
    best_h = None
    best_score = np.inf
    start_time = time.time()
    generations_without_improve = 0
    # Main GA loop
    while time.time() - start_time < budget_s - 0.5:
        scores = np.array([compute_c5(ind) for ind in population])
        cur_best_idx = np.argmin(scores)
        cur_best_score = scores[cur_best_idx]
        if cur_best_score < best_score - 1e-12:
            best_score = cur_best_score
            best_h = population[cur_best_idx].copy()
            generations_without_improve = 0
        else:
            generations_without_improve += 1
        # Elitism: keep the best individual
        new_pop = [best_h.copy()]
        # Fill rest of the population
        while len(new_pop) < pop_size:
            p1 = tournament_select(population, scores, size=3)
            p2 = tournament_select(population, scores, size=3)
            child = uniform_crossover(p1, p2)
            child = mutate(child, rate=0.2, sigma=0.05)
            child = project_feasible(child, target_sum)
            new_pop.append(child)
        population = new_pop
        # optional early stop if stagnated long enough
        if generations_without_improve >= 500 and time.time() - start_time > budget_s * 0.3:
            break
    # Ensure final evaluation
    final_score = compute_c5(best_h)
    return best_h, float(final_score), int(n)
