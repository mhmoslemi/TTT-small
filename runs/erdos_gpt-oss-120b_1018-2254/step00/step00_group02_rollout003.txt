import numpy as np
import time

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * corr.max()

def project_box_simplex(v, s, tol=1e-9, max_iter=60):
    # binary search lambda
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        sum_w = w.sum()
        if sum_w > s:
            lo = lam
        else:
            hi = lam
    w = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # adjust small residual
    residual = s - w.sum()
    if abs(residual) > 1e-12:
        # find index with room
        # if residual>0 increase first non-1, else decrease first non-0
        if residual > 0:
            idx = np.where(w < 1 - 1e-12)[0]
            if idx.size > 0:
                w[idx[0]] += residual
        else:
            idx = np.where(w > 1e-12)[0]
            if idx.size > 0:
                w[idx[0]] += residual
        w = np.clip(w, 0.0, 1.0)
    return w

def target_move(h, step):
    n = h.shape[0]
    # compute correlation and find shift with maximum correlation
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx_max = np.argmax(corr)
    shift = idx_max - (n - 1)
    # overlapping region indices
    if shift >= 0:
        i_start = 0
        i_end = n - shift
        # i range 0 <= i < n-shift
        # corresponding partner j = i + shift
    else:
        i_start = -shift
        i_end = n
        # i range -shift .. n-1, partner j = i + shift (i.e., i -|shift|)
    if i_end - i_start <= 0:
        return h  # no overlap
    # Build list of eligible i where h[i] > eps and h[i+shift] < 1-eps
    eps = 1e-9
    i_vals = np.arange(i_start, i_end)
    j_vals = i_vals + shift
    # eligible indices:
    mask = (h[i_vals] > eps) & (h[j_vals] < 1 - eps)
    if not np.any(mask):
        return h
    eligible_i = i_vals[mask]
    # pick random one
    idx = np.random.choice(eligible_i)
    j = idx + shift
    delta = min(step, h[idx], 1.0 - h[j])
    # ensure delta positive
    if delta <= 0:
        return h
    h_new = h.copy()
    h_new[idx] -= delta
    h_new[j] += delta
    return h_new

def random_move(h, step):
    n = h.shape[0]
    # Gaussian perturb
    perturb = step * np.random.randn(n)
    v = h + perturb
    # project onto constraints
    s = n / 2.0
    return project_box_simplex(v, s)

import time
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return the overlap constant for a given step function."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * float(corr.max())

def _project_box_simplex(v: np.ndarray, target_sum: float, max_iter: int = 60) -> np.ndarray:
    """
    Project v onto the set {x | 0<=x_i<=1, sum(x)=target_sum}.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo = np.min(v) - 1.0      # ensures all clips to 1
    hi = np.max(v)            # ensures all clips to 0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    w = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # tiny adjustment for numerical drift
    residual = target_sum - w.sum()
    if abs(residual) > 1e-12:
        # try to distribute residual to a non‑tight component
        if residual > 0:
            idx = np.where(w < 1.0 - 1e-12)[0]
        else:
            idx = np.where(w > 1e-12)[0]
        if idx.size:
            w[idx[0]] = np.clip(w[idx[0]] + residual, 0.0, 1.0)
    return w

def _target_move(h: np.ndarray, step: float) -> np.ndarray:
    """
    Perform a single mass‑transfer move aimed at the shift that currently
    attains the maximal overlap.
    The move conserves the total sum and respects the box constraints.
    """
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx_max = int(np.argmax(corr))
    shift = idx_max - (n - 1)   # integer shift that maximizes overlap

    # overlapping index interval
    if shift >= 0:
        i_start, i_end = 0, n - shift
    else:
        i_start, i_end = -shift, n
    if i_end <= i_start:
        return h

    i_vals = np.arange(i_start, i_end)
    j_vals = i_vals + shift

    eps = 1e-12
    mask = (h[i_vals] > eps) & (h[j_vals] < 1.0 - eps)
    if not np.any(mask):
        return h

    eligible_i = i_vals[mask]
    i = np.random.choice(eligible_i)
    j = i + shift
    delta = min(step, h[i], 1.0 - h[j])
    if delta <= 0.0:
        return h

    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new

def _random_move(h: np.ndarray, step: float) -> np.ndarray:
    """Add Gaussian noise and project back onto the feasible set."""
    n = h.shape[0]
    pert = step * np.random.randn(n)
    v = h + pert
    return _project_box_simplex(v, n / 2.0)

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    np.random.seed(seed)

    # ----------------------------------------------------------------------
    # 1. Initialise
    # ----------------------------------------------------------------------
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n0 = init.shape[0]
    else:
        # fallback: 45‑point alternating pattern (balanced)
        n0 = 45
        init = np.concatenate([np.ones(n0 // 2), np.zeros(n0 - n0 // 2)])
        np.random.shuffle(init)

    # Choose a modest resolution (between 45 and 180 points)
    target_n = max(45, min(180, n0 * 2))
    if target_n != n0:
        x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
        x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
        h = np.interp(x_new, x_old, init)
    else:
        h = init.copy()

    # Ensure feasibility
    n = h.shape[0]
    h = _project_box_simplex(h, n / 2.0)

    best_h = h.copy()
    best_c5 = _compute_c5(best_h)

    # ----------------------------------------------------------------------
    # 2. Stochastic local search
    # ----------------------------------------------------------------------
    step = 0.12                     # initial magnitude of moves
    it = 0
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break
        it += 1

        # decaying step size
        if it % 3000 == 0 and step > 0.005:
            step *= 0.75

        curr_c5 = _compute_c5(h)

        if np.random.rand() < 0.5:
            cand = _target_move(h, step)
        else:
            cand = _random_move(h, step)

        cand_c5 = _compute_c5(cand)

        # Accept improvement for the current state
        if cand_c5 < curr_c5:
            h = cand
            curr_c5 = cand_c5

        # Keep global best
        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = cand.copy()

        # Occasionally inject a fresh random feasible point to escape plateaus
        if np.random.rand() < 0.0008:
            v = np.random.rand(n)
            h = _project_box_simplex(v, n / 2.0)

    # ----------------------------------------------------------------------
    # 3. Final projection and evaluation
    # ----------------------------------------------------------------------
    best_h = _project_box_simplex(best_h, n / 2.0)
    final_c5 = _compute_c5(best_h)

    return best_h, float(final_c5), int(n)
