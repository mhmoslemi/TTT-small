def hillclimb(h, dx, max_iter, time_limit):
    best_h = h.copy()
    best_score = compute_c5(h, dx)
    n = len(h)
    ones = np.where(h==1)[0]
    zeros = np.where(h==0)[0]
    start = time.time()
    i=0
    while i < max_iter and time.time()-start < time_limit:
        # pick random swap
        i1 = np.random.choice(ones)
        i0 = np.random.choice(zeros)
        # swap
        h[i1] = 0
        h[i0] = 1
        new_score = compute_c5(h, dx)
        if new_score <= best_score:
            # accept
            best_score = new_score
            best_h = h.copy()
            # update sets
            ones = np.where(h==1)[0]
            zeros = np.where(h==0)[0]
        else:
            # revert with probability exp(-(new_score-best_score)/temp)
            # simple simple: revert
            h[i1] = 1
            h[i0] = 0
        i+=1

    return best_h, best_score
```

def project_onto_box_simplex(v, target_sum):
    lo = -np.max(v) - 1  # ensure all values after adding lo are <=0
    hi = 1 - np.min(v) + 1  # ensure all values after adding hi >=1
    for _ in range(50):
        mid = (lo+hi)/2
        proj = np.clip(v + mid, 0, 1)
        s = proj.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo+hi)/2, 0, 1)
```

import numpy as np
import time

def _project_onto_box_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto {0 <= x <= 1, sum(x)=target_sum}."""
    # Bisection on lambda such that sum(clip(v+lam,0,1)) == target_sum
    lo = -np.max(v) - 2.0
    hi = 1.0 - np.min(v) + 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s < target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def _compute_O(h: np.ndarray) -> np.ndarray:
    """Autocorrelation O[s] = sum_i h[i] * h[i+s] (zero‑padded)."""
    return np.convolve(h, h[::-1], mode="full")

def _compute_c5_from_O(O: np.ndarray, n: int) -> float:
    """C5 = max_k ∫ h(1−h_shift) = (n/2 - min_O) * (2/n)."""
    target_sum = n * 0.5
    min_O = O.min()
    return (target_sum - min_O) * (2.0 / n)

def _gradient_for_worst_shifts(h: np.ndarray, min_indices: np.ndarray, offset: int) -> np.ndarray:
    """Gradient of min_O w.r.t. h (sum of shifts achieving the minimum)."""
    n = h.shape[0]
    g = np.zeros_like(h)
    for idx in min_indices:
        s = idx - offset
        if s == 0:
            continue
        if s > 0:
            g[: n - s] += h[s:]
            g[s:]       += h[: n - s]
        else:  # s < 0
            s_abs = -s
            g[: n - s_abs] += h[s_abs:]
            g[s_abs:]       += h[: n - s_abs]
    return g

def _initial_vector(n: int, seed: int = 0) -> np.ndarray:
    """Create a feasible random start (values in [0,1], sum = n/2)."""
    rng = np.random.RandomState(seed)
    v = rng.rand(n)
    return _project_onto_box_simplex(v, n * 0.5)

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] (discretized) to minimise
    C5 = max_k ∫ h(x)(1−h(x+k))dx, respecting 0≤h≤1 and ∫h=1.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()

    # Determine problem size
    if "initial_h_values" in globals():
        h = np.asarray(initial_h_values, dtype=np.float64).copy()
        n = h.size
        # ensure feasibility
        h = np.clip(h, 0.0, 1.0)
        h = _project_onto_box_simplex(h, n * 0.5)
    else:
        n = 200               # default modest size
        h = _initial_vector(n, seed)

    target_sum = n * 0.5
    offset = n - 1

    # initial evaluation
    O = _compute_O(h)
    best_c5 = _compute_c5_from_O(O, n)
    best_h = h.copy()

    # parameters for the simple gradient ascent
    max_no_improve = 3000            # restart if stagnation
    no_improve_cnt = 0
    iteration = 0

    while time.time() - start_time < budget_s * 0.95:
        # recompute O and current C5
        O = _compute_O(h)
        current_c5 = _compute_c5_from_O(O, n)

        # keep the best observed
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improve_cnt = 0
            # stop early if we have already beaten the target bound
            if best_c5 <= 0.3808:
                break
        else:
            no_improve_cnt += 1

        # gradient step towards increasing the smallest overlap (reducing max C5)
        min_O = O.min()
        min_idxs = np.where(O == min_O)[0]

        g = _gradient_for_worst_shifts(h, min_idxs, offset)
        norm_g = np.linalg.norm(g)
        if norm_g < 1e-12:
            # random perturb to escape flat region
            h += 0.02 * (np.random.rand(n) - 0.5)
            h = np.clip(h, 0.0, 1.0)
            h = _project_onto_box_simplex(h, target_sum)
            continue

        # step size decays with time
        elapsed = time.time() - start_time
        remaining = max(budget_s - elapsed, 0.0)
        eta = 0.7 * (remaining / budget_s)   # between 0 and 0.7
        step = eta * g / (norm_g + 1e-12)

        h_new = h + step
        h_new = _project_onto_box_simplex(h_new, target_sum)

        h = h_new
        iteration += 1

        # occasional random restart if stagnating too long
        if no_improve_cnt >= max_no_improve:
            h = _initial_vector(n, seed + iteration)
            no_improve_cnt = 0

    # final evaluation of the best vector found
    final_c5 = best_c5
    return best_h, float(final_c5), n
