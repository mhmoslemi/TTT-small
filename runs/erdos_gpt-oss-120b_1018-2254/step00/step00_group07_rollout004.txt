import time, numpy as np, random

def compute_c5(h, dx):
    corr = np.correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def random_binary(n):
    h = np.zeros(n)
    ones_idx = np.random.choice(n, n//2, replace=False)
    h[ones_idx]=1.0
    return h

def hill_climb(initial_h, dx, time_limit):
    h = initial_h.copy()
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    n = len(h)
    start = time.time()
    iter = 0
    # compute lists of ones and zeros indices to speed
    while time.time() - start < time_limit:
        # choose random swap
        # get indices
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if len(ones)==0 or len(zeros)==0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # swap
        h[i], h[j] = h[j], h[i]  # swap values
        cur_c5 = compute_c5(h, dx)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
            # accept as base
        else:
            # reject revert
            h[i], h[j] = h[j], h[i]  # revert swap
        # maybe occasionally random restart
        iter += 1
        if iter % 5000 == 0:
            # random restart with best and some random perturbation
            if np.random.rand() < 0.5:
                # random new h
                h = random_binary(n)
            else:
                # start from best with small random swaps
                h = best_h.copy()
                for _ in range(10):
                    # random swap

def project_onto_box_and_sum(v, lower=0.0, upper=1.0, target_sum):
    # Find λ such that sum_i np.clip(v - λ, lower, upper) = target_sum
    # Use binary search because the sum is decreasing in λ.
    lo = -max(v) - 1  # enough low
    hi = max(v) + 1   # enough high
    for _ in range(60):
        mid = (lo+hi)/2.0
        proj = np.clip(v - mid, lower, upper)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2.0, lower, upper)

Proof: function median of v - λ clipped is monotonic decreasing in λ.

import numpy as np
import time
import random

def project_onto_box_and_sum(v, target_sum, lower=0.0, upper=1.0):
    """Project vector v onto [lower,upper] with exact sum target_sum."""
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, lower, upper)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, lower, upper)


def compute_c5(arr, dx):
    """Return C5 for array arr (fallback if evaluate_erdos_solution missing)."""
    try:
        return evaluate_erdos_solution(arr)
    except Exception:
        corr = np.correlate(arr, 1.0 - arr, mode="full")
        return np.max(corr) * dx


def smooth_max_weights(self_corr, beta):
    """Compute softmax weights for smooth max of -self_corr."""
    v = -self_corr
    scaled = beta * v
    m = np.max(scaled)
    exp_vals = np.exp(scaled - m)
    return exp_vals / exp_vals.sum()


def gradient_self_corr_weights(h, weights):
    """Gradient of sum_i w_i * (-self_corr_i) w.r.t. h."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    offset = n - 1
    for idx, w in enumerate(weights):
        lag = idx - offset
        if lag >= 0:
            i_end = n - lag
            if i_end <= 0:
                continue
            i_slice = slice(0, i_end)
            j_slice = slice(lag, n)
        else:
            lag_abs = -lag
            i_start = lag_abs
            i_end = n
            if i_start >= n:
                continue
            i_slice = slice(i_start, i_end)
            j_slice = slice(0, n - lag_abs)
        hi = h[i_slice]
        hj = h[j_slice]
        grad[i_slice] += -w * hj
        grad[j_slice] += -w * hi
    return grad


def optimize_h(initial_h, target_sum, dx, budget_s, start_time):
    """Gradient-descent with projection to improve C5."""
    n = initial_h.shape[0]
    h = project_onto_box_and_sum(initial_h, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

    beta = 20.0          # smoothness of max approximation
    step = 0.5           # initial step size
    iteration = 0

    while time.time() - start_time < budget_s - 0.5:
        # compute self-correlation
        self_corr = np.correlate(h, h, mode="full")
        # softmax weights for smooth max of -self_corr
        weights = smooth_max_weights(self_corr, beta)
        # gradient of smooth objective
        grad = gradient_self_corr_weights(h, weights)
        # gradient step and projection
        h_new = project_onto_box_and_sum(h - step * grad, target_sum)
        cur_c5 = compute_c5(h_new, dx)

        if cur_c5 < best_c5 - 1e-12:
            best_c5 = cur_c5
            best_h = h_new.copy()
            h = h_new
            step = max(step * 0.9, 1e-4)
        else:
            # simple simulated-annealing acceptance
            remaining = max(budget_s - (time.time() - start_time), 1e-6)
            T = remaining / budget_s
            if np.random.rand() < np.exp(-(cur_c5 - best_c5) / (T + 1e-12)):
                h = h_new
            step = min(step * 1.01, 1.0)

        iteration += 1
        if iteration % 2000 == 0:
            beta = min(beta * 1.5, 200.0)
        if best_c5 <= 0.3808:
            break

    return best_h, best_c5
