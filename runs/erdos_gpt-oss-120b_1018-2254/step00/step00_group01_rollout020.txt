    total = v.sum()

            # cannot adjust further

        v = np.clip(v, lower, upper)
        total = v.sum()

        else:
            # just adjust the largest element if possible
            idx = np.argmax(v)
            v[idx] += target_sum - total
            v = np.clip(v, lower, upper)
    return v

import time
import math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return the C5 value for a given discretized step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)


def _project_onto_box_simplex(v: np.ndarray,
                              target_sum: float,
                              lower: float = 0.0,
                              upper: float = 1.0,
                              tol: float = 1e-9,
                              max_iter: int = 100) -> np.ndarray:
    """
    Project vector v onto the set {x | lower <= x_i <= upper, sum(x)=target_sum}.
    Simple water-filling style iteration.
    """
    v = np.clip(v, lower, upper)
    total = v.sum()
    if abs(total - target_sum) <= tol:
        return v

    for _ in range(max_iter):
        excess = total - target_sum
        if abs(excess) <= tol:
            break
        # indices that can still move without hitting bounds
        free = (v > lower + tol) & (v < upper - tol)
        n_free = free.sum()
        if n_free == 0:
            break
        delta = excess / n_free
        v[free] -= delta
        v = np.clip(v, lower, upper)
        total = v.sum()

    # final tiny correction if needed
    if abs(v.sum() - target_sum) > tol:
        free = (v > lower + tol) & (v < upper - tol)
        if free.any():
            scale = (target_sum - v[~free].sum()) / v[free].sum()
            v[free] *= scale
            v = np.clip(v, lower, upper)
        else:
            # adjust the largest element as a last resort
            idx = np.argmax(v)
            v[idx] += target_sum - v.sum()
            v = np.clip(v, lower, upper)
    return v


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # initialise h
    # ----------------------------------------------------------------------
    init = globals().get("initial_h_values")
    if init is not None:
        h0 = np.asarray(init, dtype=np.float64)
        n = h0.shape[0]
    else:
        n = 200                     # a modest size if no parent is supplied
        h0 = rng.random(n)
        h0 = _project_onto_box_simplex(h0, n / 2.0)

    target_sum = n / 2.0
    h = _project_onto_box_simplex(h0, target_sum)

    best_h = h.copy()
    best_val = _compute_c5(best_h)

    start_time = time.time()
    deadline = start_time + budget_s * 0.95  # small safety margin

    # ----------------------------------------------------------------------
    # optimisation loop (sub‑gradient + occasional stochastic moves)
    # ----------------------------------------------------------------------
    iteration = 0
    step0 = 0.5

    while time.time() < deadline:
        iteration += 1

        # correlation to find the worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = corr.max()
        shift = np.argmax(corr) - (n - 1)          # shift index (negative allowed)

        # ------------------- sub‑gradient for the current worst shift ----------
        left = np.zeros(n)
        right = np.zeros(n)
        if shift >= 0:
            if shift == 0:
                left = h.copy()
                right = h.copy()
            else:
                left[shift:] = h[:-shift]
                right[:n - shift] = h[shift:]
        else:
            s = -shift
            left[:n - s] = h[s:]
            right[s:] = h[:n - s]

        g = 1.0 - left - right                       # gradient of C_max
        step = step0 / math.sqrt(iteration)
        h_candidate = h - step * g
        h_candidate = _project_onto_box_simplex(h_candidate, target_sum)

        val_candidate = _compute_c5(h_candidate)
        cur_val = max_corr * (2.0 / n)

        # accept or reject (simple simulated‑annealing style)
        if val_candidate <= cur_val:
            h = h_candidate
            cur_val = val_candidate
        else:
            T = step0 / math.sqrt(iteration)
            if rng.random() < math.exp(-(val_candidate - cur_val) / max(T, 1e-12)):
                h = h_candidate
                cur_val = val_candidate

        # keep the best seen so far
        if val_candidate < best_val:
            best_val = val_candidate
            best_h = h_candidate.copy()

        # ------------------- occasional random pair swap -----------------------
        if iteration % 200 == 0:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                hi, hj = h[i], h[j]
                d_low = max(-hi, hj - 1.0)
                d_high = min(1.0 - hi, hj)
                if d_low < d_high:
                    d = rng.uniform(d_low, d_high)
                    h2 = h.copy()
                    h2[i] = hi + d
                    h2[j] = hj - d
                    val2 = _compute_c5(h2)
                    if val2 < best_val:
                        best_val = val2
                        best_h = h2.copy()
                    if val2 < cur_val:
                        h = h2
                        cur_val = val2

        # ------------------- occasional Gaussian jitter -----------------------
        if iteration % 500 == 0:
            pert = rng.normal(scale=0.05, size=n)
            h_pert = h + pert
            h_pert = _project_onto_box_simplex(h_pert, target_sum)
            val_pert = _compute_c5(h_pert)
            if val_pert < best_val:

                best_h = h_pert.copy()
            if val_pert < cur_val:
                h = h_pert
                cur_val = val_pert

        # early exit if we have already reached the target bound
        if best_val <= 0.3808:
            break

    # final evaluation (precision)
    best_val = _compute_c5(best_h)
    return best_h, best_val, n
