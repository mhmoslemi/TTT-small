    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Retrieve initial vector
    try:
        init = initial_h_values
        n = len(init)
        x0 = np.array(init, dtype=np.float64)
    except NameError:
        n = 80
        x0 = np.zeros(n, dtype=np.float64)
        # generate random feasible vector
        rand = rng.random(n)
        x0 = project_box_sum(rand, n/2.0)
    # Ensure feasibility
    target_sum = n / 2.0
    x0 = project_box_sum(x0, target_sum)
    best_x = x0
    best_val = compute_c5(best_x)
    # set minimal desired bound
    target_bound = 0.3808

    # Main loop
    while time.time() - start_time < budget_s:
        # decide restart size based on remaining time
        remaining = budget_s - (time.time() - start_time)
        iter_per_restart = int(min(20000, max(1000, remaining * 10))) # rough scaling
        # Generate new start by jitter
        jitter = rng.normal(scale=0.1, size=n)
        start = project_box_sum(best_x + jitter, target_sum)
        # Occasionally use random start
        if rng.random() < 0.05:
            start = project_box_sum(rng.random(n), target_sum)
        # Hill climb
        x_local, val_local = hill_climb(start, iter_per_restart, rng, start_time, budget_s)
        if val_local < best_val - 1e-9:
            best_x = x_local
            best_val = val_local
            # early exit if better than target
            if best_val <= target_bound:
                break
    return best_x, best_val, n

def hill_climb(x_start, max_iter, rng, start_time, budget_s):
    best = x_start.copy()
    best_val = compute_c5(best)
    n = len(best)
    for it in range(max_iter):
        # early stop
        if time.time() - start_time > budget_s - 0.5:
            break
        i, j = rng.choice(n, size=2, replace=False)
        if rng.random() < 0.5:
            from_i, to_i = i, j
        else:
            from_i, to_i = j, i
        max_transfer = min(best[from_i], 1.0 - best[to_i])
        if max_transfer <= 0:
            continue
        delta = rng.uniform(0.0, max_transfer)
        if delta == 0:
            continue
        cand = best.copy()
        cand[from_i] -= delta
        cand[to_i] += delta
        val = compute_c5(cand)
        if val < best_val - 1e-12:
            best = cand
            best_val = val
    return best, best_val
```

    low = -np.max(v) - 1.0
    high = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        mid = (low + high) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = np.sum(x)
        if s < S:
            low = mid
        else:
            high = mid

import numpy as np

def compute_c5(h):
    """Return the Erdős C5 value for a step function sampled in h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_box_sum(v, target_sum, eps=1e-12):
    """
    Project vector v onto the box [0,1]^n with the additional linear constraint sum(x)=target_sum.
    Uses a simple bisection on the additive offset tau such that x = clip(v+tau,0,1) satisfies the sum.
    """
    # Initial bounds guarantee the sum at low <= target <= sum at high
    low = -np.max(v) - 1.0
    high = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = np.sum(x)
        if s < target_sum:
            low = mid
        else:
            high = mid
    tau = (low + high) * 0.5
    return np.clip(v + tau, 0.0, 1.0)

def hill_climb(start, max_iter, rng, time_limit):
    """Simple mass‑transfer hill climbing that respects box and sum constraints."""
    n = start.shape[0]
    best = start.copy()
    best_val = compute_c5(best)
    target_sum = n / 2.0
    start_time = time.time()
    for _ in range(max_iter):
        # stop early if we are close to the overall budget
        if time.time() - start_time > time_limit:
            break
        i, j = rng.choice(n, size=2, replace=False)
        # random direction of transfer
        if rng.random() < 0.5:
            frm, to = i, j
        else:
            frm, to = j, i
        # maximum admissible transfer from frm to to
        max_transfer = min(best[frm], 1.0 - best[to])
        if max_transfer <= 0.0:
            continue
        delta = rng.uniform(0.0, max_transfer)
        if delta == 0.0:
            continue
        cand = best.copy()
        cand[frm] -= delta
        cand[to] += delta
        # no need to re‑project – construction respects bounds and sum
        cand_val = compute_c5(cand)
        if cand_val < best_val - 1e-12:
            best = cand
            best_val = cand_val
    return best, best_val
