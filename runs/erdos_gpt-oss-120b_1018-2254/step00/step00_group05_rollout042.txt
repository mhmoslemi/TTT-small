def propose_move(h, max_step):
    n = len(h)
    i, j = np.random.choice(n, 2, replace=False)
    # compute allowable epsilon range
    lower = max(-max_step, -h[i], h[j] - 1)  # because h_i + eps >=0 and h_j - eps <=1
    upper = min(max_step, 1 - h[i], h[j])   # because h_i + eps <=1 and h_j - eps >=0
    if lower >= upper:
        return None
    eps = np.random.uniform(lower, upper)
    h_new = h.copy()
    h_new[i] += eps
    h_new[j] -= eps
    return h_new, eps, i, j
```

def compute_c5(h):

import numpy as np
import time
import math

def _project_onto_box_sum(x, target):
    """
    Euclidean projection of a vector onto the intersection of the box [0,1]^n
    and the affine hyperplane sum(x) = target.
    Uses a simple bisection on the shift lambda such that
    sum(clamp(x + lambda, 0, 1)) = target.
    """
    lo = -np.max(x)            # all entries shifted ≤0 → sum=0
    hi = 1.0 - np.min(x)       # all entries shifted ≥1 → sum=n
    for _ in range(40):
        mid = (lo + hi) * 0.5
        proj = np.clip(x + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)


def _c5_of_h(h):
    """
    Compute C5 for a discrete step function h on [0,2] using the
    prescribed correlation formula.
    """
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def _min_overlap(h):
    """
    Returns the vector of overlaps J(k)=∑ h_i h_{i+k} for k=0..n-1.
    """
    n = h.shape[0]
    corr = np.correlate(h, h, mode='full')
    # corr indices: -(n-1) .. (n-1), shift k corresponds to index n-1+k
    return corr[n - 1:]


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimises a step function h to obtain a low value of the Erdős
    overlap constant C5.

    Returns:
        h_values : ndarray of shape (n_points,)
        c5_bound : float  (the achieved C5)
        n_points : int    (the number of samples)
    """
    np.random.seed(seed)
    # ------------------------------------------------------------------
    # Initialise h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # a modest default size
        n = 70
        h = np.full(n, 0.5, dtype=float)

    n = h.shape[0]
    target_sum = n / 2.0

    # enforce feasibility
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur != 0.0:
        h = h * (target_sum / cur)
    h = _project_onto_box_sum(h, target_sum)

    best_h = h.copy()
    best_c5 = _c5_of_h(best_h)

    start = time.time()
    # iterative ascent on the minimal overlap J(k)
    max_iter = 200000
    it = 0
    while it < max_iter:
        if time.time() - start > budget_s * 0.95:
            break

        # compute overlaps for all admissible shifts
        J = _min_overlap(h)            # length n, J[0] corresponds to k=0
        # we are interested in the *minimum* overlap (k=0 is usually maximal)
        # ignore k=0 because it cannot be the minimum for reasonable h
        k_min = np.argmin(J[1:]) + 1   # shift giving smallest J
        k = k_min

        # gradient of J(k) w.r.t. h
        if k == 0:
            g = 2.0 * h
        else:
            g = np.zeros_like(h)
            g[:n - k] += h[k:]          # contribution from h_i * h_{i+k}
            g[k:] += h[:n - k]          # contribution from the symmetric term

        # step size decays linearly
        frac = it / max_iter
        eta = 0.05 * (1.0 - frac) + 0.001

        # propose move
        h_new = h + eta * g
        h_new = _project_onto_box_sum(h_new, target_sum)

        c5_new = _c5_of_h(h_new)

        # keep the best ever seen
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

        # accept if improvement, otherwise occasionally accept worse move
        cur_c5 = _c5_of_h(h)
        if c5_new <= cur_c5:
            h = h_new
        else:
            # simple simulated‑annealing acceptance
            T = max(0.001, 1.0 - frac)
            if np.random.rand() < math.exp(-(c5_new - cur_c5) / T):
                h = h_new

        # occasional random perturbation around the current best
        if (it + 1) % 5000 == 0:
            h = best_h + 0.02 * np.random.randn(n)
            h = np.clip(h, 0.0, 1.0)
            h = _project_onto_box_sum(h, target_sum)

        it += 1

    # final evaluation
    final_c5 = _c5_of_h(best_h)
    # If a helper evaluator is present, prefer its value
    try:
        if callable(evaluate_erdos_solution):
            final_c5 = evaluate_erdos_solution(best_h)
    except Exception:
        pass

    return best_h, float(final_c5), n
