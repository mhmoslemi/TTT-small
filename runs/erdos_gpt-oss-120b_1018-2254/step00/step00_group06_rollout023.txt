import numpy as np
import time
import random

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return float(corr.max()), corr

def project_box_sum(v, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    random.seed(seed)
    np.random.seed(seed)

    # setup initial h
    try:
        init_h = initial_h_values
        n = int(init_h.shape[0])
        h = init_h.astype(np.float64).copy()
    except NameError:
        n = 95
        h = np.full(n, 0.5, dtype=np.float64)
    target_sum = n / 2.0
    # Ensure feasible
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        # project to satisfy sum
        h = project_box_sum(h, target_sum)

    dx = 2.0 / n

    # Evaluate initial
    best_c5, _ = compute_c5(h)
    best_h = h.copy()

    # parameters
    max_iters = 10**9  # placeholder, loop will break on time
    # schedule parameters
    eta_start = 0.5
    eta_end = 0.005
    alpha_start = 5.0
    alpha_end = 30.0

    iter_count = 0
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.95:
            break
        frac = elapsed / (budget_s * 0.95)

        eta = eta_start * (1 - frac) + eta_end * frac
        alpha = alpha_start * (1 - frac) + alpha_end * frac

        # Compute correlation vector and current C5
        corr = np.correlate(h, 1 - h, mode='full') * dx
        cur_c5 = float(corr.max())

        # update best
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()

        # compute smooth max weights
        # subtract max to avoid overflow
        exp_weights = np.exp(alpha * (corr - cur_c5))
        W = exp_weights.sum()

        # combine with reverse for symmetric kernel
        k = exp_weights + exp_weights[::-1]

        # convolution with h
        conv = np.convolve(k, h, mode='full')
        # extract slice for gradient
        grad = dx * (W - conv[n-1:2*n-1])

        # gradient descent step
        h_new = h - eta * grad

        # project onto box and sum constraint
        h = project_box_sum(h_new, target_sum)

        iter_count += 1

    # final random local search around best_h
    h = best_h.copy()
    cur_c5 = best_c5
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.98:
            break
        # random pair perturbation
        i, j = random.sample(range(n), 2)
        # feasible delta bounds
        lower = max(-h[i], -(1 - h[j]))
        upper = min(1 - h[i], h[j])
        if lower >= upper:

        # restrict to small delta
        max_delta = 0.02
        lower = max(lower, -max_delta)
        upper = min(upper, max_delta)
        if lower >= upper:

        delta = random.uniform(lower, upper)

        h_candidate[i] += delta

        cand_c5, _ = compute_c5(h_candidate)

import numpy as np
import time
import random

def _compute_c5(h):
    """Return C5 value and full correlation array."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return float(corr.max()), corr

def _project_box_sum(v, target):
    """
    Project vector v onto the hyperplane sum(v)=target while keeping 0<=v<=1.
    Finds λ such that sum(clip(v+λ,0,1)) = target by binary search.
    """
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    random.seed(seed)
    np.random.seed(seed)

    # --- initialise h -------------------------------------------------
    try:
        init_h = initial_h_values  # provided globally if exists
        n = int(init_h.shape[0])
        h = np.asarray(init_h, dtype=np.float64).copy()
    except Exception:
        n = 95
        h = np.full(n, 0.5, dtype=np.float64)

    target_sum = n / 2.0
    # Ensure feasibility (sum constraint)
    if abs(h.sum() - target_sum) > 1e-12:
        h = _project_box_sum(h, target_sum)

    dx = 2.0 / n

    # --- keep best solution -------------------------------------------
    best_c5, _ = _compute_c5(h)
    best_h = h.copy()

    # --- optimisation parameters ---------------------------------------
    eta_start, eta_end = 0.5, 0.005          # step size schedule
    alpha_start, alpha_end = 5.0, 30.0       # smooth‑max sharpness

    # --- main gradient descent loop ------------------------------------
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.95:
            break
        frac = elapsed / (budget_s * 0.95)

        eta = eta_start * (1 - frac) + eta_end * frac
        alpha = alpha_start * (1 - frac) + alpha_end * frac

        # correlation and current C5
        corr = np.correlate(h, 1 - h, mode='full') * dx
        cur_c5 = float(corr.max())

        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()

        # smooth‑max weights (stable by subtracting max)
        w = np.exp(alpha * (corr - cur_c5))
        W = w.sum()

        # symmetric kernel for gradient
        k = w + w[::-1]                     # length 2n‑1

        # convolution to obtain Σ (w_L + w_-L) h_{i+L}
        conv = np.convolve(k, h, mode='full')
        grad = dx * (W - conv[n-1:2*n-1])   # gradient of the smooth objective

        # descent step + projection onto feasible set
        h = _project_box_sum(h - eta * grad, target_sum)

    # --- fine‑grained random local search -----------------------------
    h = best_h.copy()
    cur_c5 = best_c5
    max_delta = 0.02   # magnitude of pair adjustments
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.98:
            break

        i, j = random.sample(range(n), 2)

        # feasible interval for delta that keeps both entries in [0,1]
        lower = max(-h[i], -(1 - h[j]))
        upper = min(1 - h[i], h[j])
        if lower >= upper:
            continue
        # restrict to a small step size
        lower = max(lower, -max_delta)
        upper = min(upper, max_delta)
        if lower >= upper:
            continue

        delta = random.uniform(lower, upper)
        cand = h.copy()
        cand[i] += delta
        cand[j] -= delta

        cand_c5, _ = _compute_c5(cand)
        if cand_c5 < cur_c5:
            h = cand
            cur_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = cand.copy()

    return best_h, best_c5, n
