import numpy as np
import time
import math

def _c5_of(h):
    """
    Compute the C5 overlap bound for a candidate step function h.
    Tries the provided evaluator, falls back to a direct correlation.
    """
    try:
        return evaluate_erdos_solution(h, len(h))[0]
    except Exception:
        # direct discrete approximation
        return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

    """
    Stochastic search for a low C5 bound.
    Returns (h_values, c5_bound, n_points).
    """

        n = 47

        rem = target - full
        if rem > 0 and full < n:
            h0[full] = rem

    diff = target_sum - h0.sum()

                break
