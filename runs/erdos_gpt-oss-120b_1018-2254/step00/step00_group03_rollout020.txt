def propose_move(h, rng):
    n = len(h)
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    hi, hj = h[i], h[j]
    pos = min(1 - hi, hj)
    neg = -min(hi, 1 - hj)
    if pos == 0 and neg == 0:
        return None
    delta = rng.uniform(neg, pos)
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new, delta, i, j

def project_to_sum_bounds(v, target):
    # v currently clipped to [0,1], may have sum not equal target.
    # Use water-filling algorithm.
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    # iterate while |diff| > 1e-12
    max_iter = 100
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        # free indices where v not at bound
        free = np.where((v > 0) & (v < 1))[0]
        if free.size == 0:
            # cannot adjust further
            break
        delta = diff / free.size
        # add delta, clip again
        v[free] += delta
        # clip to 0/1
        v = np.clip(v, 0.0, 1.0)
        diff = target - v.sum()
    # if still diff not zero, adjust arbitrarily within bounds
    if abs(diff) > 1e-8:
        # distribute over all entries with room
        for i in range(len(v)):
            if diff > 0 and v[i] < 1.0:
                add = min(diff, 1.0 - v[i])
                v[i] += add
                diff -= add
            elif diff < 0 and v[i] > 0.0:
                sub = min(-diff, v[i])
                v[i] -= sub
                diff += sub
            if abs(diff) < 1e-8:
                break
    return v

def _project_to_sum(v, target):

            break
        free = np.where((v > 0) & (v < 1))[0]
        if not free.size:
            break
