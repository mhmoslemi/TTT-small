def repair(h):
    h = np.clip(h,0,1)
    total = h.sum()
    target = n_points/2
    # If total != target, we adjust.
    # Compute difference.
    diff = target - total
    # We can add diff proportionally to entries not at bounds.
    # Identify free indices where (h>0 and h<1).
    free = (h>0) & (h<1)
    if np.any(free):
        # distribute diff equally among free entries
        delta = diff / free.sum()
        h[free] += delta
    else:
        # All at bounds; we need to move mass with small adjustments.
        # just uniformly add diff/n_points and clip.
        h += diff / n_points

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(10):
        total = h.sum()
        diff = target - total
        if abs(diff) < 1e-9:
            break
        # distribute diff equally among non-clamped positions
        # find indices not at bounds
        free = (h > 0) & (h < 1)
        if free.any():
            h[free] += diff / free.sum()
        else:
            # all at bounds, adjust all equally (some may exceed)
            h += diff / n
        h = np.clip(h,0.0,1.0)
    # final force rounding errors
    # Slightly adjust to match target exactly
    total = h.sum()
    if abs(total - target) > 1e-8:
        # adjust small residual by adding to a random free index
        free = np.where((h > 0) & (h < 1))[0]
        if free.size > 0:
            idx = free[0]
            h[idx] += target - total
            h = np.clip(h,0.0,1.0)
    return h
```

def mutate_h(h, n, step_std=0.2):
    h_new = h.copy()
    # perturb each coordinate with probability p mut_rate
    p = 0.2  # proportion for mutation
    mask = np.random.rand(n) < p
    perturb = np.random.normal(0.0, step_std, size=n)
    h_new[mask] += perturb[mask]
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = repair_h(h_new, n)
    return h_new

import numpy as np

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # iterative projection
    for _ in range(15):
        total = h.sum()
        diff = target - total
        if abs(diff) < 1e-12:
            break
        # find indices not at bounds
        free = (h > 0) & (h < 1)
        if free.any():
            h[free] += diff / free.sum()
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    # final adjustment of residual
    total = h.sum()
    diff = target - total
    if abs(diff) > 1e-12:
        # adjust a free position if any, else adjust first entry and clip
        free_idx = np.where((h > 0) & (h < 1))[0]
        if free_idx.size > 0:
            i = free_idx[0]
            h[i] += diff
        else:
            h[0] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def eval_h(h):
    # returns C5; if any constraint violation returns inf
    if np.isnan(h).any():
        return float('inf')
    if (h < -1e-12).any() or (h > 1+1e-12).any():
        return float('inf')
    # check integral
    n = len(h)
    dx = 2.0 / n
    if abs(h.sum() * dx - 1.0) > 1e-8:
        return float('inf')
    return compute_c5(h)

def mutate_h(h, n, step_std=0.2):
    h_new = h.copy()
    # probability of mutating each coordinate
    p = 0.2
    mask = np.random.rand(n) < p
    perturb = np.random.normal(0.0, step_std, size=n)
    h_new[mask] += perturb[mask]
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = repair_h(h_new, n)
    return h_new

def crossover_h(a, b, n):
    mask = np.random.rand(n) < 0.5
    child = np.where(mask, a, b)
    child = repair_h(child, n)
    return child

def local_search_h(h, n, max_iter=100):
    best_h = h.copy()
    best_val = eval_h(best_h)
    for _ in range(max_iter):
        cand = mutate_h(best_h, n, step_std=0.05)
        cand_val = eval_h(cand)
        if cand_val < best_val:
            best_h, best_val = cand, cand_val
    return best_h, best_val

        population.append(repair_h(h_rand, n_points))
    # Evaluate initial population

    best_idx = np.argmin(scores)

        if elapsed > budget_s - 2.0:

        # Selection and offspring creation
        new_pop = []
        # Keep elite
        elite_idx = np.argmin(scores)
        new_pop.append(population[elite_idx].copy())

            ids = np.random.choice(pop_size, 5, replace=False)
            i2 = ids[np.argmin(scores[ids])]
            parent_a = population[i1]
            parent_b = population[i2]
            child = crossover_h(parent_a, parent_b, n_points)
            if np.random.rand() < 0.9:
                child = mutate_h(child, n_points, step_std=0.15)
            new_pop.append(child)
        population = new_pop
        scores = np.array([eval_h(ind) for ind in population])
        # Update best
        idx = np.argmin(scores)
        if scores[idx] < best_val:
            best_val = scores[idx]
            best_h = population[idx].copy()
        # occasional local refinement
        if generation % 15 == 0:

            if refined_val < best_val:
                best_val, best_h = refined_val, refined_h.copy()
        generation += 1

    remaining = budget_s - (time.time() - start)
    if remaining > 1.0:
        iters = int(min(200, max(10, remaining * 30)))
        best_h, best_val = local_search_h(best_h, n_points, max_iter=iters)
    return best_h, best_val, n_points

    # Main evolutionary loop.
    while True:
        if time.time() - start_time > budget_s - 2.0:
            break

        # Keep the elite.
        elite_idx = scores.argmin()
        new_pop = [population[elite_idx].copy()]

        # Fill the rest of the new generation.
        while len(new_pop) < pop_size:
            # Tournament selection (size 5).
            ids = np.random.choice(pop_size, 5, replace=False)
            i1 = ids[np.argmin(scores[ids])]
            ids = np.random.choice(pop_size, 5, replace=False)
            i2 = ids[np.argmin(scores[ids])]

            parent_a = population[i1]
            parent_b = population[i2]

            child = crossover_h(parent_a, parent_b, n_points)
            if np.random.rand() < 0.9:
                child = mutate_h(child, n_points, step_std=0.15)
            new_pop.append(child)

        population = new_pop
        scores = np.array([eval_h(ind) for ind in population])

        # Update global best.
        idx = scores.argmin()
        if scores[idx] < best_val:
            best_val = scores[idx]
            best_h = population[idx].copy()

        # Periodic local refinement.
        if generation % 15 == 0:
            refined_h, refined_val = local_search_h(best_h, n_points, max_iter=30)
            if refined_val < best_val:
                best_val, best_h = refined_val, refined_h.copy()

        generation += 1

    # Final polishing with remaining time.
    remaining = budget_s - (time.time() - start_time)
    if remaining > 1.0:
        iters = int(min(200, max(10, remaining * 30)))
        best_h, best_val = local_search_h(best_h, n_points, max_iter=iters)

    return best_h, best_val, n_points
