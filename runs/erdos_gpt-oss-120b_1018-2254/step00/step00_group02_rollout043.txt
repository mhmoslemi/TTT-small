import numpy as np, time, math

def evaluate_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def project_to_sum(h, target_sum):
    # binary search lambda
    lo = -1.0
    hi = 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

import numpy as np
import time

def evaluate_c5(h):
    """Compute C5 = max_k ∫ h(x)*(1-h(x+k)) dx using discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def project_to_sum(h, target):
    """Project vector h onto the intersection of [0,1]^n and sum(h)=target."""
    lo, hi = -1.0, 1.0
    for _ in range(30):
        lam = (lo + hi) * 0.5
        proj = np.clip(h - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def make_feasible(arr, target):
    """Clip to [0,1] and enforce the sum constraint."""
    arr = np.clip(arr, 0.0, 1.0)
    return project_to_sum(arr, target)
