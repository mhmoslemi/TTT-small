def gradient_for_shift(h, shift):
    n = len(h)
    g = np.zeros_like(h)
    # term from first factor
    idx1 = np.arange(n)
    idx2 = idx1 + shift
    mask = (idx2 >= 0) & (idx2 < n)
    g[idx1[mask]] = 1.0 - h[idx2[mask]]
    # term from second factor
    idx1b = np.arange(n)
    idx2b = idx1b - shift
    maskb = (idx2b >= 0) & (idx2b < n)
    # subtract h[idx2b]
    g[idx1b[maskb]] -= h[idx2b[maskb]]
    return g

def project_onto_box_sum(y, target):
    # clip inside [0,1] after shifting by lambda.
    # Use bisection.
    lo = np.min(y) - 1.0  # ensure all clamped to 1 at lo
    hi = np.max(y)        # ensure all clamped to 0 at hi
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(y - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

import numpy as np
import random
import time
import math

def evaluate_c5(h):
    """Return C5 value for a step function represented by array h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_onto_box_sum(y, target):
    """Project vector y onto the set {0≤x≤1, sum(x)=target}."""
    lo = np.min(y) - 1.0          # guarantees sum = n (all clipped to 1)
    hi = np.max(y)                # guarantees sum = 0 (all clipped to 0)
    for _ in range(60):           # enough for double precision
        mid = (lo + hi) / 2.0
        proj = np.clip(y - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(y - lam, 0.0, 1.0)

def gradient_for_shift(h, shift):
    """Sub‑gradient of the maximal correlation w.r.t. h for a given integer shift."""
    n = h.shape[0]
    g = np.zeros_like(h)
    idx = np.arange(n)

    idx2 = idx + shift
    mask = (idx2 >= 0) & (idx2 < n)
    g[idx[mask]] = 1.0 - h[idx2[mask]]

    idx2b = idx - shift
    maskb = (idx2b >= 0) & (idx2b < n)
    g[idx[maskb]] -= h[idx2b[maskb]]

    return g

def random_feasible(n, target):
    """Generate a random vector in [0,1]^n with prescribed sum."""
    y = np.random.rand(n)
    return project_onto_box_sum(y, target)
