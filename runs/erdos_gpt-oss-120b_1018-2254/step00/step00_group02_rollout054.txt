import numpy as np
import time
import math
import random

def compute_c5(arr):
    """Return the C5 value for a step function given by `arr`."""
    n = arr.shape[0]
    dx = 2.0 / n
    b = 1.0 - arr
    corr = np.correlate(arr, b, mode='full')
    return np.max(corr) * dx

def random_feasible(n, target):
    """Generate a random vector with entries in [0,1] summing exactly to target."""
    h = np.zeros(n, dtype=float)
    remaining = target
    # simple stochastic fill
    while remaining > 1e-12:
        i = np.random.randint(0, n)
        inc = min(1.0 - h[i], remaining)
        if inc <= 0:
            continue
        delta = np.random.rand() * inc
        h[i] += delta
        remaining -= delta
    return h

def project_to_feasible(h, target):
    """Project vector onto box [0,1] while preserving the sum `target` (approx)."""
    # clip first
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if abs(cur - target) < 1e-12:
        return h
    diff = target - cur
    # distribute the difference uniformly over non‑saturated entries
    mask = (h > 0.0) & (h < 1.0)
    if mask.any():
        h[mask] += diff / mask.sum()
    else:
        # fallback: adjust the first element within bounds
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    # final clip to guarantee bounds
    return np.clip(h, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):

    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()

    # Try to use a parent construction if it exists

    # Candidate resolutions (larger sizes give more flexibility)

    best_h = None
    best_c5 = float('inf')
    best_n = None

    # Helper to possibly stop early
    def time_left():
        return time.time() - start_time < budget_s * 0.96

    for n in n_candidates:
        if not time_left():
            break

        # Build an initial feasible vector for this resolution
        if init_vec is not None:
            x_old = np.linspace(0.0, 2.0, init_vec.size)
            x_new = np.linspace(0.0, 2.0, n)
            h = np.interp(x_new, x_old, init_vec)
        else:
            # sin^2 shape: average 0.5 and respects bounds
            x = np.linspace(0.0, 2.0, n, endpoint=False) + (1.0 / n) * 0.5
            h = np.sin(np.pi * x / 2.0) ** 2

        # Rescale to satisfy the integral constraint exactly
        target_sum = n / 2.0
        h = h * (target_sum / h.sum())
        h = project_to_feasible(h, target_sum)

        # Record this initialization

        if cur_c5 < best_c5:

            best_h = h.copy()
            best_n = n

        # Simulated annealing / hill‑climbing
        max_iters = int(3000 * 200 / n) + 200   # modest number, scaled with size
        temperature = 0.5
