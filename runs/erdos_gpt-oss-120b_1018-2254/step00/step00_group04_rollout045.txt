import time, math, numpy as np

def _repair_h(h: np.ndarray, target_sum: float) -> np.ndarray:
    """Project h onto the box [0,1] and enforce sum(h)=target_sum."""
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    # iteratively distribute the surplus/deficit respecting bounds
    for _ in range(100):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            cap = 1.0 - h[mask]
        else:
            mask = h > 0.0
            cap = h[mask]
        if not np.any(mask):
            break
        total_cap = cap.sum()
        if total_cap == 0.0:
            break
        # proportionally distribute the needed adjustment
        delta = diff * (cap / total_cap)
        if diff > 0:
            h[mask] += delta
        else:
            h[mask] -= delta
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - h.sum()
    # final uniform tweak for any tiny residual
    residual = target_sum - h.sum()
    if abs(residual) > 1e-12:
        h += residual / h.size
        h = np.clip(h, 0.0, 1.0)
    return h

def _c5_of(h: np.ndarray) -> float:
    """Return the overlap constant for a candidate step function."""
    try:
        val = evaluate_erdos_solution(h)
        if isinstance(val, (list, tuple, np.ndarray)):
            return float(val[0])
        return float(val)
    except Exception:
        dx = 2.0 / h.size
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        return float(np.max(corr))

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a step function h minimizing C₅ under the given constraints.

    Returns
    -------
    (h_values, c5_bound, n_points)
        h_values : np.ndarray   – the discovered function values on a uniform grid.
        c5_bound : float        – the achieved maximal overlap.
        n_points : int          – number of discretisation points.
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # --------------------------------------------------------------
    # Initialise problem size and starting vector
    # --------------------------------------------------------------
    if "initial_h_values" in globals():
        n_points = int(len(initial_h_values))
        cur = np.array(initial_h_values, dtype=float)
    else:
        n_points = 100
        cur = rng.random(n_points)

    target_sum = n_points / 2.0          # because ∫ h = 1 ⇒ sum(h) = n/2
    cur = _repair_h(cur, target_sum)

    best_h = cur.copy()
    best_score = _c5_of(best_h)

    current_h = best_h.copy()
    current_score = best_score

    # --------------------------------------------------------------
    # Simulated‑annealing style search
    # --------------------------------------------------------------
    temp_start = 5e-2
    temp_end = 1e-6

    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.98:
            break

        # linear temperature schedule
        frac = elapsed / (budget_s * 0.98)
        temp = temp_start * (1.0 - frac) + temp_end * frac

        # propose a neighbour of the current solution
        cand = current_h.copy()
        n_mut = rng.integers(1, 5)                     # 1–4 mutations
        idx = rng.integers(0, n_points, size=n_mut)

        if rng.random() < 0.5:
            # fresh random values at the chosen positions
            cand[idx] = rng.random(n_mut)
        else:
            # small Gaussian perturbations
            cand[idx] += rng.normal(scale=0.1, size=n_mut)

        cand = _repair_h(cand, target_sum)
        cand_score = _c5_of(cand)

        # accept according to Metropolis rule
        if cand_score < current_score or rng.random() < math.exp(
            -(cand_score - current_score) / max(temp, 1e-12)
        ):
            current_h = cand
            current_score = cand_score

        # keep the best ever seen
        if cand_score < best_score:
            best_h = cand
            best_score = cand_score

        # occasional random restart to escape stagnation
        if rng.random() < 0.001:
            rand = rng.random(n_points)
            rand = _repair_h(rand, target_sum)
            rand_score = _c5_of(rand)
            if rand_score < best_score:
                best_h = rand
                best_score = rand_score
                current_h = rand
                current_score = rand_score
            else:
                current_h = rand
                current_score = rand_score

    # final safety repair (tiny numerical drifts)
    best_h = _repair_h(best_h, target_sum)
    best_score = _c5_of(best_h)

    return best_h, float(best_score), n_points
