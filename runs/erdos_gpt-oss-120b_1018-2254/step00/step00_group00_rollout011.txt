import time, math
import numpy as np

def _c5_value(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _repair_sum(h, target):
    """Clip to [0,1] and adjust the sum to match target (small residual)."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # Adjust the first element that can absorb the residual
        for idx in range(h.size):
            if diff > 0:
                inc = min(diff, 1.0 - h[idx])
                h[idx] += inc
                diff -= inc
            else:
                dec = min(-diff, h[idx])
                h[idx] -= dec
                diff += dec
            if abs(diff) < 1e-12:
                break
        h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) with improved upper bound for C5."""
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----------- initialise h -------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        N = h.size
        target_sum = N / 2.0
        h = _repair_sum(h, target_sum)
    else:
        N = 45
        target_sum = N / 2.0
        # start from a simple feasible binary+half pattern
        h = np.zeros(N, dtype=float)
        ones = int(math.floor(target_sum))
        h[:ones] = 1.0
        h[ones] = target_sum - ones   # this is 0.5 when N is odd
        rng.shuffle(h)

    # keep a copy of the best solution found
    best_h = h.copy()
    best_val = _c5_value(best_h)
    cur_h = h.copy()
    cur_val = best_val

    # ---- optimisation parameters --------------------------------------------
    max_no_improve = 5000          # not used directly, we rely on time limit
    prob_direct = 0.7              # probability to perform a directed move
    min_delta = 1e-8               # ignore negligible transfers

    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s - 0.2:
            break

        # Temperature for simulated‑annealing style acceptance (linear cooling)
        T = 0.02 * max(0.0, (budget_s - elapsed) / budget_s)

        if rng.random() < prob_direct:
            # ---- directed move aimed at the current worst shift -------------
            corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
            max_idx = int(np.argmax(corr))
            shift = max_idx - (N - 1)      # integer shift corresponding to max overlap

            # Determine index pairs (i, j=i+shift) that actually contribute
            if shift >= 0:
                i_vals = np.arange(N - shift)
                j_vals = i_vals + shift
            else:
                i_vals = np.arange(-shift, N)
                j_vals = i_vals + shift

            # Filter to pairs where there is room to move mass
            mask = (cur_h[i_vals] > min_delta) & (cur_h[j_vals] < 1 - min_delta)
            if not np.any(mask):
                # fallback to a random move
                prob_direct = 0.0
                continue
            candidates = np.where(mask)[0]
            idx = rng.choice(candidates)
            i = int(i_vals[idx])
            j = int(j_vals[idx])

            # Transfer as much as possible (or a random fraction thereof)
            delta_max = min(cur_h[i], 1.0 - cur_h[j])
            if delta_max <= min_delta:
                continue
            delta = rng.random() * delta_max   # uniform in [0, delta_max)
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        else:
            # ---- simple random transfer between two distinct positions -------
            i, j = rng.integers(N, size=2)
            if i == j:
                continue
            delta_max = min(cur_h[i], 1.0 - cur_h[j])
            if delta_max <= min_delta:
                continue
            delta = rng.random() * delta_max
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta

        # Evaluate the new candidate
        new_val = _c5_value(new_h)

        # Acceptance test
        if new_val < cur_val:
            accept = True
        else:
            # Metropolis criterion
            accept = rng.random() < math.exp(-(new_val - cur_val) / max(T, 1e-12))

        if accept:
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val - 1e-12:
                best_h = new_h.copy()
                best_val = new_val
                # early exit if we beat the target record
                if best_val <= 0.3808:
                    break

    # Final repair (tiny numerical drift may have accumulated)
    best_h = _repair_sum(best_h, N / 2.0)

    final_c5 = _c5_value(best_h)
    return best_h, float(final_c5), N
