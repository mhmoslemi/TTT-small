def _project_feasible(v: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and adjust to meet sum == target."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(100):
        diff = target - v.sum()
        if abs(diff) < 1e-12:
            break
        interior = (v > 1e-12) & (v < 1 - 1e-12)
        if not np.any(interior):
            break
        delta = diff / interior.sum()
        v[interior] += delta
        v = np.clip(v, 0.0, 1.0)
    # final tiny adjustment by scaling if needed
    cur = v.sum()
    if abs(cur - target) > 1e-9:
        scale = target / cur if cur != 0 else 1.0
        v *= scale
        v = np.clip(v, 0.0, 1.0)
    return v

def _random_feasible(n: int, target: float) -> np.ndarray:
    """Generate a random vector with entries in [0,1] and given sum."""
    int_part = int(np.floor(target))
    frac = target - int_part
    h = np.zeros(n, dtype=np.float64)

        if remaining.size:
            idx = np.random.choice(remaining)
            h[idx] = frac
    # shuffle to avoid structure bias
    np.random.shuffle(h)
    return h

    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

        cand_c5 = _c5_value(cand, dx)

        # simple annealing acceptance
        T = max(0.001, (1.0 - t_frac) * 0.08)
        accept = False
        if cand_c5 < current_c5:
            accept = True
        else:
            if np.exp((current_c5 - cand_c5) / T) > np.random.rand():
                accept = True

        if accept:
            current_h = cand
            current_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # restart if stuck too long
        if no_improve >= max_no_improve:
            current_h = _random_feasible(n, target_sum)
            current_c5 = _c5_value(current_h, dx)
            no_improve = 0

    # ------------------------------------------------------------------
    # Final projection to guarantee feasibility
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_feasible(best_h, target_sum)
    final_c5 = _c5_value(best_h, dx)

    return best_h, final_c5, n
