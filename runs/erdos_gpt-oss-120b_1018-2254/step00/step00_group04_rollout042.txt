def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_onto_bounds_and_sum(v, target):
    # Clip
    v = np.clip(v, 0.0, 1.0)
    # Use water-filling iteration
    tol = 1e-9
    max_iter = 1000
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = v > 0
        else:
            free = v < 1
        n = np.count_nonzero(free)
        if n == 0:
            break
        delta = diff / n
        v[free] -= delta
        v = np.clip(v, 0.0, 1.0)
    # final scaling if needed
    s = v.sum()
    if abs(s - target) > 1e-8:
        v = v * (target / s)
        v = np.clip(v, 0.0, 1.0)
    return v

def compute_c5(h):
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_onto_bounds_and_sum(v, target):
    v = np.clip(v, 0.0, 1.0)
    tol = 1e-9
    for _ in range(1000):
        s = v.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:

        else:

    if abs(s - target) > 1e-8:

        v = np.clip(v, 0.0, 1.0)

import numpy as np
import time
import math

def compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx approximated by correlation."""
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_onto_bounds_and_sum(v, target):
    """Project v onto the box [0,1]^n while enforcing sum(v)=target."""
    v = np.clip(v, 0.0, 1.0)
    tol = 1e-9
    for _ in range(1000):
        s = v.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = v > 0
        else:
            free = v < 1
        n = np.count_nonzero(free)
        if n == 0:
            break
        delta = diff / n
        v[free] -= delta
        v = np.clip(v, 0.0, 1.0)
    # final scaling if needed (tiny residual)
    s = v.sum()
    if abs(s - target) > 1e-8 and s > 0:
        v = v * (target / s)
        v = np.clip(v, 0.0, 1.0)
    return v
