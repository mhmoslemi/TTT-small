def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    # correlation of h and 1-h
    c = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(c)  # yields sum before scaling
    return max_val * dx

import time, math
import numpy as np

def enforce_sum_and_bounds(h, target_sum):
    """Clip to [0,1] and adjust to satisfy the sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    # iterative simple water-filling
    for _ in range(10):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx for discrete h."""
    N = h.shape[0]
    dx = 2.0 / N
    c = np.correlate(h, 1.0 - h, mode='full')
    return c.max() * dx

def subgradient_descent(h0, time_budget, rng):
    """Projected sub‑gradient descent on the worst‑shift overlap."""
    h = h0.copy()
    N = len(h)
    target_sum = N / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    it = 0
    while time.time() - start < time_budget:
        # correlation to locate worst shift
        c = np.correlate(h, 1.0 - h, mode='full')
        idx = int(c.argmax())
        shift = idx - (N - 1)           # integer offset
        # gradient of the worst‑shift overlap
        g = np.zeros_like(h)
        if shift >= 0:
            k = shift
            end = N - k
            if end > 0:
                g[:end] += (1.0 - h[k:])      # (1 - h[i+shift])
                g[k:]  += -h[:end]            # -h[i]
        else:
            d = -shift
            end = N - d
            if end > 0:
                g[d:] += (1.0 - h[:end])      # (1 - h[i-shift])
                g[:end] += -h[d:]             # -h[i+shift]
        # step size diminishing like 1/sqrt(iter)
        step = 0.5 / math.sqrt(it + 1)
        h_new = h - step * g
        h_new = enforce_sum_and_bounds(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_c5

def simulated_annealing_binary(N, time_budget, rng):
    """Binary SA: keep exactly N/2 ones, swap a 1 and a 0 each move."""
    # start from a random balanced binary vector
    h = np.zeros(N, dtype=float)
    h[rng.choice(N, N // 2, replace=False)] = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    start = time.time()
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = max(0.001, 0.5 * (1.0 - frac))   # linear cooling
        ones = np.flatnonzero(h > 0.5)
        zeros = np.flatnonzero(h < 0.5)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # propose swap
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < math.exp(-delta / T):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c5
