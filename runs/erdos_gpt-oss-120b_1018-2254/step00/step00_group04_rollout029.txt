    best_score = compute_c5(best_h)
    improved = True

        improved = False
        # find all positions where h == 1 and h == 0
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        # Shuffle order: consider random subset maybe for speed, break early if improvement found.
        # We can compute best neighbor improvement:
        for i in ones:

                # propose swap

                new_h[j] = 1
                new_score = compute_c5(new_h)

                break

def simulated_annealing(h, score, time_limit, start_time):
    n = len(h)
    current_h = h.copy()
    current_score = score
    best_h = current_h.copy()
    best_score = current_score
    T = 0.01
    iter = 0
    while time.time() - start_time < time_limit:
        iter += 1
        # gradually cool
        T = max(T * 0.9999, 1e-6)
        i, j = np.random.randint(n), np.random.randint(n)
        if i == j:
            continue
        # Choose direction: move mass from i to j if possible else reverse
        if current_h[i] > 1e-9 and current_h[j] < 1 - 1e-9:
            max_delta = min(current_h[i], 1 - current_h[j])
            delta = max_delta * random.random()

            # move from j to i
            max_delta = min(current_h[j], 1 - current_h[i])

import numpy as np
import random
import time
import math

def _compute_c5(h):
    """Return max overlap C5 for a step function h on [0,2]."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def _hill_climb_binary(h):
    """Greedy hill‑climb on binary vectors (0/1) preserving sum = n/2."""
    n = h.shape[0]
    best_h = h.copy()
    best_score = _compute_c5(best_h)
    improved = True
    while improved:
        improved = False
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        # shuffle to avoid deterministic cycles
        np.random.shuffle(ones)
        np.random.shuffle(zeros)
        # search best improving swap
        best_swap = None
        best_swap_score = best_score
        for i in ones:
            for j in zeros:
                cand = best_h.copy()
                cand[i] = 0.0
                cand[j] = 1.0
                sc = _compute_c5(cand)
                if sc < best_swap_score - 1e-12:
                    best_swap_score = sc
                    best_swap = (i, j, cand)
        if best_swap is not None:
            _, _, best_h = best_swap
            best_score = best_swap_score
            improved = True
    return best_h, best_score

def _simulated_annealing(h, score, time_budget, start_time):
    """Simple SA that moves a small amount of mass between two points."""
    n = h.shape[0]
    cur_h = h.copy()
    cur_score = score
    best_h = cur_h.copy()
    best_score = cur_score
    T = 0.02  # initial temperature
    while True:
        if time.time() - start_time > time_budget:
            break
        # cool down slowly
        T = max(T * 0.9999, 1e-6)
        i, j = np.random.randint(n), np.random.randint(n)
        if i == j:
            continue
        # decide direction of transfer
        if cur_h[i] > 1e-12 and cur_h[j] < 1.0 - 1e-12:
            max_delta = min(cur_h[i], 1.0 - cur_h[j])
        elif cur_h[j] > 1e-12 and cur_h[i] < 1.0 - 1e-12:
            # swap roles
            i, j = j, i
            max_delta = min(cur_h[i], 1.0 - cur_h[j])
        else:
            continue
        if max_delta <= 0:
            continue
        delta = max_delta * random.random()
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_score = _compute_c5(new_h)
        if new_score < cur_score - 1e-12 or random.random() < math.exp(-(new_score - cur_score) / T):
            cur_h = new_h
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_h = new_h
                best_score = new_score
                if best_score <= 0.3808:
                    break
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) for the Erdős C5 problem."""
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    # Determine problem size and initial vector
    if 'initial_h_values' in globals():
        init_h = np.asarray(initial_h_values, dtype=float)
    else:
        # fallback: use an even length of 42 with uniform half‑mass
        init_h = np.full(42, 0.5, dtype=float)
    n = init_h.shape[0]

    # Build a feasible binary start from init_h (keep half the mass as 1)
    binary_h = np.zeros(n, dtype=float)
    # indices of largest values become 1 (ensures sum = n/2)
    order = np.argsort(-init_h)
    half = n // 2
    binary_h[order[:half]] = 1.0

    # Initial hill‑climb
    best_h, best_score = _hill_climb_binary(binary_h)

    # Time‑aware loop of random restarts + hill‑climb
    while time.time() - start < budget_s * 0.75:
        # generate a fresh random binary vector
        perm = np.random.permutation(n)
        h0 = np.zeros(n, dtype=float)
        h0[perm[:half]] = 1.0
        h0, sc0 = _hill_climb_binary(h0)
        if sc0 < best_score - 1e-12:
            best_h, best_score = h0, sc0
        if best_score <= 0.3808:
            break

    # Fine‑tune with simulated annealing using remaining budget
    remaining = start + budget_s - time.time()
    if remaining > 0.5:
        best_h, best_score = _simulated_annealing(best_h, best_score, start + budget_s, start)

    # Final bound using the provided evaluator when available
    if 'evaluate_erdos_solution' in globals():
        try:
            bound, _ = evaluate_erdos_solution(best_h)
        except Exception:
            bound = _compute_c5(best_h)
    else:
        bound = _compute_c5(best_h)

    return best_h, bound, n
