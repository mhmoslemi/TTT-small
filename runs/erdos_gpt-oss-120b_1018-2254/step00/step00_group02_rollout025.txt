    h = np.clip(h, 0.0, 1.0)
    # Adjust sum
    current = np.sum(h)

import time
import numpy as np


def compute_c5(h: np.ndarray) -> float:
    """Return the overlap constant C5 for a step function defined by h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_to_constraints(h: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1] and adjust so that sum(h) == target_sum."""
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if cur == 0.0:
        # Uniform fill as a safe fallback
        return np.full_like(h, target_sum / h.size)

    # Scale to hit the target sum, then clip again
    h = h * (target_sum / cur)
    h = np.clip(h, 0.0, 1.0)

    # Fix any remaining discrepancy by distributing it on free entries
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size > 0:
            h[free] += diff / free.size
            h = np.clip(h, 0.0, 1.0)
        else:
            # No free entries – adjust a single element conservatively
            idx = np.argmax(h) if diff < 0 else np.argmin(h)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)

    # Final tiny correction, if needed
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        idxs = np.where((h > 0.0) & (h < 1.0))[0]
        if idxs.size:
            h[idxs[0]] += diff
            h = np.clip(h, 0.0, 1.0)

    return h
