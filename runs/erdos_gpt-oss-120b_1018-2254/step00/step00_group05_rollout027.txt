def propose_move(h, rng):
    N = len(h)
    i, j = rng.choice(N, size=2, replace=False)
    # direction random: moving mass from i to j or j to i, whichever feasible.
    if rng.random() < 0.5:
        src, dst = i, j
    else:
        src, dst = j, i
    # Determine max delta that keeps bounds.
    max_delta = min(h[src], 1 - h[dst])
    if max_delta <= 1e-12:
        return None, None
    delta = rng.random() * max_delta  # uniform between 0 and max_delta
    # Optionally limit delta to some max step to avoid too large jumps.
    # but it's fine.
    new_h = h.copy()
    new_h[src] -= delta
    new_h[dst] += delta
    return new_h, (src, dst, delta)

import time
import math
import numpy as np

def _c5_score(vec, dx):
    """Maximum overlap integral for a given step vector."""
    # correlation of h with (1-h)
    corr = np.correlate(vec, 1.0 - vec, mode="full")
    return float(corr.max() * dx)


def _random_swap_move(h, rng, max_step):
    """Propose a feasible move that swaps a small amount between two coordinates."""
    N = h.shape[0]
    i, j = rng.integers(N, size=2)
    if i == j:
        return None
    # Random direction
    if rng.random() < 0.5:
        src, dst = i, j
    else:
        src, dst = j, i
    # Feasible transfer amount
    max_eps = min(h[src], 1.0 - h[dst])
    if max_eps <= 1e-12:
        return None
    eps = rng.random() * min(max_eps, max_step)
    new_h = h.copy()
    new_h[src] -= eps
    new_h[dst] += eps
    return new_h


def _directed_move(h, worst_lag, rng, max_step):
    """Move mass from a low‑neighbor to a high‑neighbor index based on the worst lag."""
    N = h.shape[0]
    neighbor_sum = np.zeros(N, dtype=float)

    if worst_lag >= 0:
        lag = worst_lag
        neighbor_sum[: N - lag] += h[lag:]          # forward neighbor
        neighbor_sum[lag:] += h[: N - lag]          # backward neighbor
    else:
        lag = -worst_lag
        neighbor_sum[lag:] += h[: N - lag]          # forward neighbor (since lag negative)
        neighbor_sum[: N - lag] += h[lag:]          # backward neighbor

    mask_src = h > 1e-12
    mask_dst = h < 1.0 - 1e-12
    if not (mask_src.any() and mask_dst.any()):
        return None

    src_candidates = np.where(mask_src)[0]
    dst_candidates = np.where(mask_dst)[0]

    src = src_candidates[np.argmin(neighbor_sum[src_candidates])]
    dst = dst_candidates[np.argmax(neighbor_sum[dst_candidates])]

    max_eps = min(h[src], 1.0 - h[dst])
    if max_eps <= 1e-12:
        return None
    eps = rng.random() * min(max_eps, max_step)
    new_h = h.copy()
    new_h[src] -= eps
    new_h[dst] += eps
    return new_h


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to lower the Erdős C5 constant.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ---- initialise -------------------------------------------------
    init = globals().get("initial_h_values", None)
    if init is not None:
        h = np.array(init, dtype=float).copy()
        N = h.size
        h = np.clip(h, 0.0, 1.0)
    else:
        N = 70
        h = np.full(N, 0.5, dtype=float)

    # enforce exact integral constraint sum(h) = N/2
    target_sum = N / 2.0
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        diff = target_sum - cur_sum
        # simple redistribution loop (few passes)
        for _ in range(5):
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                # need to add mass
                idx = rng.integers(N)
                inc = min(diff, 1.0 - h[idx])
                h[idx] += inc
                diff -= inc
            else:
                # need to remove mass
                idx = rng.integers(N)
                dec = min(-diff, h[idx])
                h[idx] -= dec
                diff += dec
        # final clip (should be very close)
        h = np.clip(h, 0.0, 1.0)

    dx = 2.0 / N
    best_h = h.copy()
    best_score = _c5_score(best_h, dx)

    # Target bound – stop early if reached
    target_bound = 0.3808

    # ---- optimisation loop -----------------------------------------
    max_step = 0.05          # largest mass transfer per move
    T0 = 0.005               # initial temperature for SA
    it = 0

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 0.1:
            break

        # compute current score and worst lag
        corr = np.correlate(h, 1.0 - h, mode="full")
        cur_score = float(corr.max() * dx)
        max_idx = int(corr.argmax())
        worst_lag = max_idx - (N - 1)   # shift index relative to zero

        # with small probability take a completely random swap move
        if rng.random() < 0.07:
            new_h = _random_swap_move(h, rng, max_step)
        else:
            new_h = _directed_move(h, worst_lag, rng, max_step)

        if new_h is None:
            continue

        new_score = _c5_score(new_h, dx)

        # Simulated‑annealing acceptance
        dE = new_score - cur_score
        T = T0 * max(0.0, 1.0 - elapsed / budget_s)
        accept = dE < 0.0 or rng.random() < math.exp(-dE / max(T, 1e-12))

        if accept:
            h = new_h
            cur_score = new_score
            # update global best if improved
            if new_score < best_score:
                best_score = new_score
                best_h = new_h.copy()
                if best_score <= target_bound:
                    break

        it += 1

    # ---- final repair ------------------------------------------------
    # Ensure exact sum constraint (tiny rounding errors fixed)
    cur_sum = best_h.sum()
    diff = target_sum - cur_sum
    if abs(diff) > 1e-10:
        # try to distribute the remaining diff across feasible indices
        for idx in rng.permutation(N):
            if diff > 0:
                inc = min(diff, 1.0 - best_h[idx])
                best_h[idx] += inc
                diff -= inc
            else:
                dec = min(-diff, best_h[idx])
                best_h[idx] -= dec
                diff += dec
            if abs(diff) < 1e-12:
                break
        best_h = np.clip(best_h, 0.0, 1.0)

    final_score = _c5_score(best_h, dx)

    return best_h.tolist(), final_score, N
