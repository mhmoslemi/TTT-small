    # ---- initial construction -------------------------------------------------
    if "initial_h_values" in globals():
        base = np.array(initial_h_values, dtype=np.float64)
        n0 = base.shape[0]
        # upscale modestly to get a finer grid (keep it ≤300 for speed)
        new_n = min(300, max(95, 2 * n0 - 1))
        if new_n != n0:
            x0 = np.linspace(0.0, 2.0, n0)
            xn = np.linspace(0.0, 2.0, new_n)
            h = np.interp(xn, x0, base)
        else:
            h = base.copy()
    else:
        # no parent available – start from a uniform shape
        new_n = 151
        h = np.full(new_n, 0.5, dtype=np.float64)

    n = h.shape[0]
    target_sum = n / 2.0
    h = _project_onto_feasible(h, target_sum)

    best_h = h.copy()
    best_c5 = _c5_of(best_h)

    # ---- optimisation parameters ---------------------------------------------
    init_step = 0.5
    min_step = 0.001
    iter_cnt = 0

    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s:
            break

        # correlation and worst shift
        corr = np.correlate(h, 1 - h, mode="full")
        max_idx = int(np.argmax(corr))
        d = max_idx - (n - 1)          # integer shift producing the max
        # gradient for that shift
        grad = np.zeros_like(h)
        i_start = max(0, -d)
        i_end = min(n, n - d)
        if i_start < i_end:
            i = np.arange(i_start, i_end)
            grad[i] += (1 - h[i + d])
            grad[i + d] -= h[i]

        # step size (linear decay)
        step = init_step * (1 - elapsed / budget_s) + min_step * (elapsed / budget_s)

        # take a projected gradient step
        h_new = _project_onto_feasible(h - step * grad, target_sum)

        c5_new = _c5_of(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new

        # occasional random injection to escape plateaus
        if (iter_cnt & 2047) == 0:
            cand = _project_onto_feasible(np.random.rand(n), target_sum)
            c5_cand = _c5_of(cand)
            if c5_cand < best_c5:
                best_c5 = c5_cand
                best_h = cand.copy()
                h = cand

        iter_cnt += 1

    # final safety projection and evaluation
    best_h = _project_onto_feasible(best_h, target_sum)
    best_c5 = _c5_of(best_h)

    return best_h, float(best_c5), int(n)
