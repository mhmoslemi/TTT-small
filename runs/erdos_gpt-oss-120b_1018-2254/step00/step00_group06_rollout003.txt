import numpy as np, time

def compute_c5(h, dx):
    # correlation cross-correlation
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_capped_simplex(v, target, lower=0.0, upper=1.0, tol=1e-9, max_iter=100):
    # Clip on lower bound, then adjust by lambda to meet sum constraint
    lo = -np.max(v) - 10.0
    hi = upper - np.min(v) + 10.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        w = np.clip(v + lam, lower, upper)
        s = w.sum()
        if abs(s - target) <= tol:
            return w
        if s > target:
            hi = lam
        else:
            lo = lam
    # Final
    w = np.clip(v + (lo + hi) / 2.0, lower, upper)
    # Force sum correction if still off by tiny epsilon
    diff = target - w.sum()
    if abs(diff) > 1e-8:
        # distribute diff uniformly across non-saturated entries
        mask = (w > lower + 1e-12) & (w < upper - 1e-12)
        cnt = mask.sum()
        if cnt > 0:
            w[mask] += diff / cnt
            w = np.clip(w, lower, upper)
    return w

import numpy as np
import time

def _c5_from_h(h, dx):
    """Return C5 = max correlation * dx for a given h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx, corr

def _project_capped_simplex(v, target, lower=0.0, upper=1.0, tol=1e-9, max_iter=100):
    """
    Euclidean projection of v onto the set {x | lower <= x_i <= upper, sum(x)=target}.
    Uses a binary search on the additive shift λ.
    """
    n = v.shape[0]
    lo = -np.max(v) - 10.0   # ensures all v+λ <= lower
    hi = upper - np.min(v) + 10.0   # ensures all v+λ >= upper
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        w = np.clip(v + lam, lower, upper)
        s = w.sum()
        if abs(s - target) <= tol:
            return w
        if s > target:
            hi = lam
        else:
            lo = lam
    # final fallback
    w = np.clip(v + (lo + hi) * 0.5, lower, upper)
    # tiny correction if needed
    diff = target - w.sum()
    if abs(diff) > 1e-8:
        mask = (w > lower + 1e-12) & (w < upper - 1e-12)
        cnt = mask.sum()
        if cnt > 0:
            w[mask] += diff / cnt
            w = np.clip(w, lower, upper)
    return w

def _grad_at_lag(h, lag):
    """
    Gradient of the correlation at a given lag (maximising lag).
    For each i, grad_i = 1 - (h_{i+lag} + h_{i-lag}), with out‑of‑bounds treated as 0.
    """
    n = h.shape[0]
    plus = np.zeros_like(h)
    minus = np.zeros_like(h)
    if lag >= 0:
        if lag < n:
            plus[: n - lag] = h[lag:]
            minus[lag:] = h[: n - lag]
    else:
        la = -lag
        if la < n:
            plus[la:] = h[: n - la]
            minus[: n - la] = h[la:]
    return 1.0 - (plus + minus)

        cand_c5, _ = _c5_from_h(cand, dx)

        # simulated‑annealing like acceptance
        if cand_c5 < cur_c5:
            h = cand
        else:
            temperature = 0.02 * (1.0 - progress) + 1e-4
            if np.random.rand() < np.exp(-(cand_c5 - cur_c5) / temperature):
                h = cand

        # occasional random perturbation to escape plateaux
        if np.random.rand() < 0.1:
            sigma = max(sigma_start * (1.0 - progress) + min_sigma, min_sigma)
            rnd = h + sigma * np.random.randn(n)
            rnd = _project_capped_simplex(rnd, target_sum)
            rnd_c5, _ = _c5_from_h(rnd, dx)
            if rnd_c5 < best_c5:
                best_h = rnd.copy()
                best_c5 = rnd_c5
                h = rnd
                no_improve = 0

        # hard restart if stuck for a while
        if no_improve >= restart_after:
            rand_vec = np.random.rand(n)
            h = _project_capped_simplex(rand_vec, target_sum)
            no_improve = 0

    # final projection & evaluation
    best_h = _project_capped_simplex(best_h, target_sum)
    final_c5, _ = _c5_from_h(best_h, dx)
    return best_h, final_c5, n
