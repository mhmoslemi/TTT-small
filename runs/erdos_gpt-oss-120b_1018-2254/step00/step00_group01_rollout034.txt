def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')

def project_feasible(v, target):
    # binary search for lambda
    low = -1e6
    high = np.max(v) + 1e6
    for _ in range(60):
        mid = (low + high) * 0.5
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            low = mid
        else:
            high = mid
    x = np.clip(v - low, 0.0, 1.0)
    # fix sum if small error
    diff = target - np.sum(x)
    if abs(diff) > 1e-12:

        free = (x > 1e-10) & (x < 1 - 1e-10)
        n_free = np.count_nonzero(free)

            x[free] += adjust
            np.clip(x, 0.0, 1.0, out=x)
        else:
            # fallback: adjust first element
            x[0] = np.clip(x[0] + diff, 0.0, 1.0)

import numpy as np, time, math

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(v, target):
    # v: np.ndarray, target sum
    low = np.min(v) - 1.0
    high = np.max(v)
    for _ in range(60):
        mid = (low + high) * 0.5
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            low = mid
        else:
            high = mid
    x = np.clip(v - low, 0.0, 1.0)
    # correct residual sum error
    diff = target - np.sum(x)
    if abs(diff) > 1e-12:
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        n_free = np.count_nonzero(free)
        if n_free > 0:
            adjust = diff / n_free
            x[free] += adjust
            np.clip(x, 0.0, 1.0, out=x)
        else:
            # adjust first element if possible
            x[0] = np.clip(x[0] + diff, 0.0, 1.0)
    return x

def construct_shape(n, r):
    dx = 2.0 / n
    a = (1.0 - r) / 2.0
    L = 1.0 - r
    p1 = a
    p2 = a + r
    p3 = a + r + L
    p4 = a + 2.0 * r + L
    idx = np.arange(n)
    x = (idx + 0.5) * dx
    h = np.empty(n, dtype=np.float64)
    if r > 1e-12:
        h.fill(0.0)
        mask = (x >= p2) & (x < p3)
        h[mask] = 1.0
        mask = (x >= p1) & (x < p2)
        h[mask] = (x[mask] - p1) / r
        mask = (x >= p3) & (x < p4)

    else:
        # indicator of plateau length L = 1
        h.fill(0.0)
        mask = (x >= a) & (x < a + L)
        h[mask] = 1.0
    return h

    return np.max(corr) * dx

def project_feasible(v, target):
    low = np.min(v) - 1.0
    high = np.max(v)
    for _ in range(60):
        mid = (low + high) * 0.5
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            low = mid
        else:
            high = mid
    x = np.clip(v - low, 0.0, 1.0)
    diff = target - np.sum(x)
    if abs(diff) > 1e-12:
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        n_free = np.count_nonzero(free)
        if n_free > 0:
            adj = diff / n_free
            x[free] += adj
            np.clip(x, 0.0, 1.0, out=x)
        else:
            x[0] = np.clip(x[0] + diff, 0.0, 1.0)
    return x

def construct_shape(n, r):
    dx = 2.0 / n
    a = (1.0 - r) / 2.0
    L = 1.0 - r
    p1 = a
    p2 = a + r
    p3 = a + r + L
    p4 = a + 2.0 * r + L
    idx = np.arange(n)
    x = (idx + 0.5) * dx
    h = np.empty(n, dtype=np.float64)
    if r > 1e-12:
        h.fill(0.0)
        mask = (x >= p2) & (x < p3)
        h[mask] = 1.0
        mask = (x >= p1) & (x < p2)
        h[mask] = (x[mask] - p1) / r
        mask = (x >= p3) & (x < p4)
        h[mask] = (p4 - x[mask]) / r
    else:
        h.fill(0.0)
        mask = (x >= a) & (x < a + L)
        h[mask] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    n = 400                               # discretisation size
    target = n / 2.0
    best_h = None
    best_val = np.inf

    # start from the provided parent if it exists
    if 'initial_h_values' in globals():
        init_raw = np.array(initial_h_values, dtype=np.float64)
        init = project_feasible(init_raw, target)
        val = compute_c5(init)
        best_h = init
        best_val = val

    # grid search over the ramp length r ∈ [0,1]
    for r in np.linspace(0.0, 1.0, 501):
        h_shape = construct_shape(n, r)
        h_proj = project_feasible(h_shape, target)
        val = compute_c5(h_proj)
        if val < best_val:
            best_h = h_proj
            best_val = val

    # local refinement via mass‑transfer moves and simple annealing
    current_h = best_h.copy()
    current_val = best_val
    start = time.time()
    sigma_initial = 0.2
    sigma_min = 1e-5
    stagnation = 0
    stagnation_limit = 20000

    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s - 0.2:
            break
        progress = elapsed / budget_s
        step = max(sigma_min, sigma_initial * (1.0 - progress))

        # occasional larger random perturbation
        if rng.random() < 0.05:
            cand = current_h + rng.normal(scale=step * 5.0, size=n)
            cand = project_feasible(cand, target)
        else:
            # mass transfer between two coordinates
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            max_delta = min(current_h[i], 1.0 - current_h[j])
            if max_delta <= 0:
                max_delta = min(current_h[j], 1.0 - current_h[i])
                if max_delta <= 0:
                    cand = current_h + rng.normal(scale=step, size=n)
                    cand = project_feasible(cand, target)
                else:
                    delta = rng.random() * step * max_delta
                    cand = current_h.copy()
                    cand[j] -= delta
                    cand[i] += delta
            else:
                delta = rng.random() * step * max_delta
                cand = current_h.copy()
                cand[i] -= delta
                cand[j] += delta

        cand_val = compute_c5(cand)

        if cand_val < current_val:
            current_h = cand
            current_val = cand_val
            stagnation = 0
            if cand_val < best_val:
                best_h = cand
                best_val = cand_val
        else:
            stagnation += 1
            T = step
            if rng.random() < math.exp(-(cand_val - current_val) / (T + 1e-12)):
                current_h = cand
                current_val = cand_val

        if stagnation >= stagnation_limit:
            cand = best_h + rng.normal(scale=step, size=n)
            cand = project_feasible(cand, target)
            current_h = cand
            current_val = compute_c5(cand)
            stagnation = 0

    best_h = project_feasible(best_h, target)
    best_val = compute_c5(best_h)
    return best_h, best_val, n
