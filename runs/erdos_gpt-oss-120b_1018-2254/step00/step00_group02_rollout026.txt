3. If sum(z) > S: need reduction of mass. Set of indices where z > 0 (or > epsilon). Reduce uniformly via water-filling: reduce amount diff = sum(z) - S from entries >0 until either diff satisfied or some entries hit zero.

def project_onto_bounds_and_sum(y, S):
    x = np.clip(y, 0.0, 1.0)
    eps = 1e-12
    # compute sum diff
    total = x.sum()
    diff = total - S
    if abs(diff) < 1e-12:
        return x
    if diff > 0:
        # need to reduce total by diff
        # indices where x > eps
        while diff > 1e-12:
            free = x > eps
            cnt = free.sum()
            if cnt == 0:
                break
            reduction = diff / cnt
            # ensure not reducing below zero
            under = reduction > x[free]
            if under.any():
                # reduce those to zero fully
                reduction_amount = x[free][under].sum()
                x[free][under] = 0.0
                diff -= reduction_amount
            else:
                x[free] -= reduction
                diff = 0.0
    else:
        # diff < 0: need to increase total
        diff = -diff
        while diff > 1e-12:
            free = x < 1 - eps
            cnt = free.sum()
            if cnt == 0:
                break
            increase = diff / cnt
            over = increase > (1 - x[free])
            if over.any():
                increase_amount = (1 - x[free])[over].sum()
                x[free][over] = 1.0
                diff -= increase_amount
            else:
                x[free] += increase
                diff = 0.0
    # final correction
    # adjust tiny leftover due to numeric
    return x

import time
import numpy as np

def _compute_c5(h):
    """Return the C₅ bound for a step function represented by h."""
    n = h.shape[0]
    # correlation without scaling, then apply dx = 2/n
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / n)


def _project_feasible(y, target_sum):
    """Project y onto the box [0,1] and the affine set sum(x)=target_sum."""
    # clip to box first
    x = np.clip(y, 0.0, 1.0)
    eps = 1e-12
    diff = x.sum() - target_sum

    if abs(diff) < 1e-12:
        return x

    # Reduce excess mass
    if diff > 0:
        while diff > 1e-12:
            free = np.where(x > eps)[0]
            cnt = free.size
            if cnt == 0:
                break
            dec = diff / cnt
            cap = x[free]                     # how much each can be reduced
            over = cap <= dec
            if over.any():
                reduction = cap[over].sum()
                x[free[over]] = 0.0
                diff -= reduction
            else:
                x[free] -= dec
                diff = 0.0
    else:
        # Increase deficit
        diff = -diff
        while diff > 1e-12:
            free = np.where(x < 1 - eps)[0]
            cnt = free.size
            if cnt == 0:
                break
            inc = diff / cnt
            cap = 1 - x[free]                 # how much each can be increased
            over = cap <= inc
            if over.any():
                increase = cap[over].sum()
                x[free[over]] = 1.0
                diff -= increase
            else:
                x[free] += inc
                diff = 0.0

    # Tiny correction if needed
    residual = target_sum - x.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.where(x < 1 - eps)[0]
            if idx.size:
                i = idx[0]
                x[i] = min(1.0, x[i] + residual)
        else:
            idx = np.where(x > eps)[0]
            if idx.size:
                i = idx[0]
                x[i] = max(0.0, x[i] + residual)
    return x


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős C₅ bound.
    Returns (h_values, c5_bound, n_points).
    """
    start = time.time()
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Initial construction
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n0 = h0.size
        # Upsample to give the optimizer more freedom (at least 90 points)
        n_target = max(2 * n0, 90)
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h = np.interp(xt, x0, h0)
    else:
        n_target = 90
        h = np.full(n_target, 0.5, dtype=float)

    n = h.shape[0]
    target_sum = n / 2.0
    # Ensure feasibility
    h = _project_feasible(h, target_sum)

    best_h = h.copy()
    best_score = _compute_c5(h)

    # ------------------------------------------------------------------
    # Subgradient descent on the worst shift
    # ------------------------------------------------------------------
    step0 = 0.5
    it = 0
    max_no_improve = 5000
    no_improve = 0

    while True:
        # time guard
        elapsed = time.time() - start
        if elapsed >= budget_s - 0.2:
            break

        # correlation and worst shift
        corr = np.correlate(h, 1 - h, mode="full")
        s = int(np.argmax(corr))
        d = s - (n - 1)               # integer shift (negative allowed)

        # gradient for this shift
        pad = np.zeros(n - 1, dtype=float)
        hpad = np.concatenate([pad, h, pad])
        idx1 = np.arange(n) + (n - 1) - d   # corresponds to h[t-d]
        idx2 = np.arange(n) + (n - 1) + d   # corresponds to h[t+d]
        term1 = 1.0 - hpad[idx1]            # 1 - h[t-d] (zero outside)
        term2 = hpad[idx2]                  # h[t+d] (zero outside)
        g = term1 - term2                   # subgradient of the worst term

        # step size (sqrt schedule)
        step = step0 / np.sqrt(it + 1.0)

        # take step and project back onto feasible set
        h_new = h - step * g
        h = _project_feasible(h_new, target_sum)

        # evaluate
        cur_score = _compute_c5(h)

        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best_h = h.copy()
            no_improve = 0
            # early exit if target achieved
            if best_score <= 0.3808:
                break
        else:
            no_improve += 1

        # occasional random perturbation to escape flat regions
        if no_improve >= max_no_improve:
            # add small Gaussian noise and re‑project
            noise = np.random.normal(scale=0.02, size=n)
            h_candidate = _project_feasible(h + noise, target_sum)
            cand_score = _compute_c5(h_candidate)
            if cand_score < best_score:
                best_score = cand_score
                best_h = h_candidate.copy()
                h = h_candidate
                no_improve = 0
            else:
                # keep current h but reset counter
                no_improve = 0

        it += 1

    # ------------------------------------------------------------------
    # Final refinement: simple pairwise mass transfer
    # ------------------------------------------------------------------
    remaining = budget_s - (time.time() - start)
    if remaining > 0.5:
        attempts = int(remaining * 300)   # ~300 proposals per second
        for _ in range(attempts):
            i, j = np.random.choice(n, 2, replace=False)
            # direction: from i to j
            max_delta = min(best_h[i], 1.0 - best_h[j])
            if max_delta <= 0:
                continue
            delta = np.random.rand() * max_delta * 0.1
            h_tmp = best_h.copy()
            h_tmp[i] -= delta
            h_tmp[j] += delta
            h_tmp = _project_feasible(h_tmp, target_sum)
            sc = _compute_c5(h_tmp)
            if sc < best_score:
                best_score = sc
                best_h = h_tmp

    return best_h, best_score, n
