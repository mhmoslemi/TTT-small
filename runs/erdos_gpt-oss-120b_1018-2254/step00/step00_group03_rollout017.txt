def _project_sum_to_target(x, target):
    """Adjust vector x (inside [0,1]) so that sum(x) == target by a simple
    redistribution loop. Assumes target is feasible (target ≤ len(x))."""
    residual = target - x.sum()
    n = x.size

    attempts = 0
    while abs(residual) > 1e-12 and attempts < 1000:

            delta = min(cap, residual)

            cap = x[idx]
            delta = -min(cap, -residual)
        if delta != 0.0:
            x[idx] += delta
            residual -= delta
        attempts += 1
    # Final clip for safety
    np.clip(x, 0.0, 1.0, out=x)
    return x

                new_h[i] += delta
            else:
                # both at bounds, fall back to swap
                new_h = cur_h.copy()
                new_h[i], new_h[j] = cur_h[j], cur_h[i]

        # evaluate candidate
        new_val = compute_c5(new_h)

        # Metropolis acceptance
        if new_val < cur_val or np.random.rand() < math.exp(-(new_val - cur_val) / max(temp, 1e-12)):
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val - 1e-9:
                best_h = new_h.copy()
                best_val = new_val
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # occasional restart if stuck
        if no_improve >= max_no_improve:
            # perturb current solution around the best one
            cur_h = best_h.copy()
            # add small Gaussian noise then project back
            noise = np.random.randn(n) * 0.02
            cur_h += noise
            np.clip(cur_h, 0.0, 1.0, out=cur_h)
            cur_h = _project_sum_to_target(cur_h, target_sum)
            cur_val = compute_c5(cur_h)
            no_improve = 0

        # early exit if already very good
        if best_val < 0.38075:
            break

    # final feasibility clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_sum_to_target(best_h, target_sum)
    # compute final bound (use helper if available)
    eval_fun = globals().get('evaluate_erdos_solution')
    final_val = eval_fun(best_h) if callable(eval_fun) else compute_c5(best_h)

    return best_h, float(final_val), int(n)
