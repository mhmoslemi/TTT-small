def propose_move(h):
    N = len(h)
    a, b = np.random.choice(N, 2, replace=False)
    # Determine allowable delta range:
    delta_low = max(-h[a], -(1 - h[b]))
    delta_high = min(1 - h[a], h[b])
    # Actually if we increase a and decrease b, we need delta such that h[a] + delta <= 1 and h[b] - delta >= 0.
    # So delta in [-h[a], 1 - h[a]] for a, and delta in [h[b] - 1, h[b]] for b? Wait.
    # Since we propose new_h[a] = h[a] + delta, new_h[b] = h[b] - delta.
    # Conditions: 0 <= h[a] + delta <= 1 => delta >= -h[a] and delta <= 1 - h[a]
    # 0 <= h[b] - delta <= 1 => delta <= h[b] and delta >= h[b] - 1
    # Combine: delta_lower = max(-h[a], h[b] - 1)
    #          delta_upper = min(1 - h[a], h[b])
    lower = max(-h[a], h[b] - 1)
    upper = min(1 - h[a], h[b])
    if lower >= upper:
        # can't move? choose other indices
        # We'll just return copy unchanged.
        return None, None
    delta = np.random.uniform(lower, upper)
    new_h = h.copy()
    new_h[a] += delta
    new_h[b] -= delta
    return new_h, (a, b, delta)

def repair_h(h, N):
    # project onto box [0,1] and sum constraint sum(h)=N/2
    h = np.clip(h, 0.0, 1.0)
    target = N / 2.0
    # simple iterative proportional correction
    for _ in range(100):
        cur = h.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to increase some entries
            free = np.where(h < 1)[0]
            if free.size == 0:
                break
            # distribute equally up to capacity
            inc = diff / free.size
            # ensure not exceed capacity
            cap = 1 - h[free]
            inc_adj = np.minimum(inc, cap)
            h[free] += inc_adj
        else:
            # need to decrease
            free = np.where(h > 0)[0]
            if free.size == 0:
                break
            dec = (-diff) / free.size
            cap = h[free]
            dec_adj = np.minimum(dec, cap)
            h[free] -= dec_adj
    # final tiny adjustment
    cur = h.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # adjust one free entry
        if diff > 0:
            idx = np.where(h < 1)[0]
        else:
            idx = np.where(h > 0)[0]
        if idx.size > 0:
            i = int(np.random.choice(idx))
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def propose_move(h):
    N = h.shape[0]
    a, b = np.random.choice(N, 2, replace=False)

    lower = max(-h[a], h[b] - 1)

    delta = np.random.uniform(lower, upper)
    new_h = h.copy()
    new_h[a] += delta
    new_h[b] -= delta
    return new_h
