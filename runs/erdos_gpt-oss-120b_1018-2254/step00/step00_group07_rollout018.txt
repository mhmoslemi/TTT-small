import time
import numpy as np
from scipy.optimize import minimize

def _adjust_sum_exact(arr, target):
    """Adjust a single entry to enforce exact sum, respecting [0,1] bounds."""
    diff = arr.sum() - target
    if abs(diff) < 1e-12:
        return arr
    # try to fix by adjusting one coordinate
    for i in range(len(arr)):
        new_val = arr[i] - diff
        if 0.0 <= new_val <= 1.0:
            arr[i] = new_val
            break
    return arr

def hill_climb_binary(h_start, deadline):
    """Binary hill‑climb / simulated annealing on swaps, respecting the deadline."""
    n = len(h_start)
    best_h = h_start.copy()
    best_val = evaluate_erdos_solution(best_h)
    h = h_start.copy()
    cur_val = best_val
    temp = 0.01  # initial temperature for occasional uphill moves
    while time.time() < deadline:
        i1, i2 = np.random.choice(n, 2, replace=False)
        if h[i1] == h[i2]:  # swapping identical values does nothing
            continue
        h_new = h.copy()
        h_new[i1], h_new[i2] = h[i2], h[i1]  # swap a 1 and a 0
        val_new = evaluate_erdos_solution(h_new)
        if val_new < best_val:
            best_val, best_h = val_new, h_new.copy()
        # Metropolis acceptance
        if val_new < cur_val or np.random.rand() < np.exp(-(val_new - cur_val) / max(temp, 1e-12)):
            h, cur_val = h_new, val_new
        temp *= 0.9999
        if best_val < 0.3805:  # early stop if already very good
            break
    return best_h, best_val

def local_optimize_continuous(h_start, deadline):
    """Try a short SLSQP local optimisation, respecting the deadline."""
    n = len(h_start)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    eq_cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}
    # Give SLSQP a modest iteration budget; it will stop early if time expires.
    try:
        res = minimize(evaluate_erdos_solution, h_start, method='SLSQP',
                       bounds=bounds, constraints=[eq_cons],
                       options={'maxiter': 300, 'ftol': 1e-9})
        if res.success:
            h_opt = _adjust_sum_exact(np.clip(res.x, 0.0, 1.0), target)
            val_opt = evaluate_erdos_solution(h_opt)
            return h_opt, val_opt
    except Exception:
        pass
    # fallback – return the starting point
    return h_start, evaluate_erdos_solution(h_start)

    start_time = time.time()

        bin_arr[:n // 2] = 1.0
        np.random.shuffle(bin_arr)

    best_h = None
    best_val = float('inf')

    # Process each start point

        # Choose method based on whether the vector is binary

            break

    # If we still have time, refine the best solution with another binary hill‑climb
    if best_h is not None and time.time() < overall_deadline:
        # Ensure we have a binary version to start from
        if not np.all(np.logical_or(np.isclose(best_h, 0.0), np.isclose(best_h, 1.0))):
            # Take the top half entries as ones
            bin_candidate = np.zeros_like(best_h)
            idx = np.argsort(-best_h)[:n // 2]
            bin_candidate[idx] = 1.0
        else:
            bin_candidate = best_h.copy()
        h_refined, val_refined = hill_climb_binary(bin_candidate, overall_deadline)
        if val_refined < best_val:
            best_val, best_h = val_refined, h_refined.copy()

    # Final safety: enforce constraints exactly
    if best_h is None:
        # Fallback: a simple balanced binary construction
        best_h = np.zeros(n, dtype=float)
        best_h[:n // 2] = 1.0
        np.random.shuffle(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _adjust_sum_exact(best_h, target_sum)

    final_val = evaluate_erdos_solution(best_h)

    return best_h, final_val, n
