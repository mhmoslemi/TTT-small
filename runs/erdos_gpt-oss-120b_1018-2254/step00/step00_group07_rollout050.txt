def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    start = time.time()
    rng = np.random.default_rng(seed)
    # determine n_points
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 200
        h0 = None
    if h0 is None:
        # generate binary balanced vector
        h = np.zeros(n, dtype=float)
        idx = rng.choice(n, size=n//2, replace=False)
        h[idx] = 1.0
    else:
        # convert to binary if not exactly balanced
        # compute target sum:
        target = n // 2
        # we can threshold to 0.5 initially
        h = (h0 > 0.5).astype(float)
        s = int(np.sum(h))
        if s != target:
            # adjust: if too many ones, set some to 0 randomly; if too few, set some zeros to 1.
            diff = target - s
            if diff > 0:
                zero_idx = np.where(h == 0)[0]
                choose = rng.choice(zero_idx, size=diff, replace=False)
                h[choose] = 1.0
            elif diff < 0:
                one_idx = np.where(h == 1)[0]
                choose = rng.choice(one_idx, size=-diff, replace=False)
                h[choose] = 0.0
    # ensure sum exactly target
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    current_h = h.copy()
    current_c5 = best_c5
    # track last improvement iteration
    last_improve = 0
    iteration = 0
    # loops
    while time.time() - start < budget_s * 0.95:
        iteration += 1
        # propose swap
        # choose random one and zero indices
        one_idx = np.where(current_h == 1)[0]
        zero_idx = np.where(current_h == 0)[0]
        if len(one_idx)==0 or len(zero_idx)==0:
            break
        i = rng.choice(one_idx)
        j = rng.choice(zero_idx)
        new_h = current_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_c5 = compute_c5(new_h, dx)
        # accept if better
        if new_c5 <= current_c5:
            current_h = new_h
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
                last_improve = iteration
        # else maybe accept with small prob (simulated annealing)
        else:
            # temperature schedule: decreasing with iteration/time
            elapsed = time.time() - start
            remaining = budget_s - elapsed
            # set a simple temp proportional to remaining
            temp = remaining / budget_s # between 0 and 1
            # probability to accept if new_c5 not much bigger
            prob = np.exp(-(new_c5 - current_c5) / (0.001 + 0.01*temp)) # scale
            if rng.random() < prob:
                current_h = new_h
                current_c5 = new_c5
        # restart if no improvement in many iterations
        if iteration - last_improve > 5000:
            # random perturbation of best solution
            # choose a few random swaps (e.g., up to 10)
            perturb = rng.integers(1, 6)  # number of swaps
            pert_h = best_h.copy()
            for _ in range(perturb):
                one_idx = np.where(pert_h == 1)[0]
                zero_idx = np.where(pert_h == 0)[0]
                i = rng.choice(one_idx)
                j = rng.choice(zero_idx)
                pert_h[i] = 0.0
                pert_h[j] = 1.0
            current_h = pert_h
            current_c5 = compute_c5(current_h, dx)
            last_improve = iteration
    # final evaluate using evaluate_erdos_solution if exists

        final_c5 = evaluate_erdos_solution(best_h) if 'evaluate_erdos_solution' in globals() else best_c5
    except Exception:
        final_c5 = best_c5
    return (best_h, float(final_c5), n)

def _c5_value(h, dx):
    """Maximum overlap integral for a given step function h."""
    # Correlate h with (1-h); mode='full' gives all shifts.
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta

        new_c5 = _c5_value(new_h, dx)

        # Acceptance test
        if new_c5 <= cur_c5:
            accept = True
        else:
            t = T0 * (1.0 - elapsed / budget_s)
            if t <= 0:
                accept = False
            else:
                prob = np.exp(-(new_c5 - cur_c5) / t)
                accept = rng.random() < prob

        if accept:
            cur_h = new_h
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-12:
                best_h = new_h.copy()
                best_c5 = new_c5
                last_improve = it

        # If stagnation lasts too long, re‑seed around the best known point
        if it - last_improve > 50000:
            pert = best_h.copy()
            for _ in range(rng.integers(1, 6)):
                recv = np.where(pert < 1.0 - 1e-12)[0]
                donor = np.where(pert > 1e-12)[0]
                if recv.size == 0 or donor.size == 0:
                    break
                i = rng.choice(recv)
                j = rng.choice(donor)
                max_transfer = min(1.0 - pert[i], pert[j])
                if max_transfer <= 0:
                    continue
                delta = rng.random() * max_transfer * 0.5
                pert[i] += delta
                pert[j] -= delta
            cur_h = pert
            cur_c5 = _c5_value(cur_h, dx)
            last_improve = it

    # ----------------------------------------------------------------------
    # Final evaluation (use the helper if it exists for a possibly more
    # accurate assessment) and a final tiny projection onto the exact simplex.
    # ----------------------------------------------------------------------
    try:
        final_c5 = evaluate_erdos_solution(best_h)
    except Exception:
        final_c5 = best_c5

    # Enforce the integral exactly (tiny numerical correction)
    current_sum = best_h.sum()
    if abs(current_sum - target_sum) > 1e-8:
        correction = (target_sum - current_sum) / n
        best_h = np.clip(best_h + correction, 0.0, 1.0)

    return best_h, float(final_c5), n
