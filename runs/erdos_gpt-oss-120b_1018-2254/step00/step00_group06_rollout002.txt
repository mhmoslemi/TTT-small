    return h

def compute_c5(h):
    """Return max overlap integral for step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def tournament_select(fitness, rng, tour_size=3):
    """Return index of best individual in a random tournament."""
    idx = rng.integers(0, fitness.size, size=tour_size)
    best = idx[np.argmin(fitness[idx])]
    return int(best)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # Determine problem size and initial solution
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n_points = h0.size
        target_sum = n_points / 2.0
        h0 = repair(h0, target_sum)
    else:
        n_points = 95
        target_sum = n_points / 2.0
        h0 = None

    # GA hyper‑parameters
    pop_size = 200
    crossover_rate = 0.7
    mut_prob_gene = 0.02
    mut_std = 0.1
    max_gens = 5000  # safety cap; loop ends earlier on time budget

    # Initialise population
    pop = np.empty((pop_size, n_points), dtype=float)
    fitness = np.empty(pop_size, dtype=float)
    start = time.time()

    # first individual: use provided construction if any
    idx = 0
    if h0 is not None:
        pop[0] = h0
        idx = 1
    while idx < pop_size:
        cand = rng.random(n_points)
        cand = repair(cand, target_sum)
        pop[idx] = cand
        idx += 1

    # evaluate initial fitness
    for i in range(pop_size):
        fitness[i] = compute_c5(pop[i])

    best_idx = int(np.argmin(fitness))
    best_h = pop[best_idx].copy()
    best_score = fitness[best_idx]

    # Main GA loop
    gen = 0
    while True:
        elapsed = time.time() - start
        if elapsed > budget_s * 0.98:
            break
        if gen >= max_gens:
            break
        if best_score <= 0.3808:
            # Target reached; stop early
            break
        gen += 1

        # adaptive mutation std (cooling)
        cur_mut_std = mut_std * (1.0 - gen / max_gens) + 0.01

        new_pop = np.empty_like(pop)
        new_fit = np.empty_like(fitness)

        # elitism
        new_pop[0] = best_h
        new_fit[0] = best_score

        for i in range(1, pop_size):
            # tournament selection
            p1 = tournament_select(fitness, rng)
            p2 = tournament_select(fitness, rng)
            parent1 = pop[p1]
            parent2 = pop[p2]

            # crossover
            if rng.random() < crossover_rate:
                cut = rng.integers(1, n_points)
                child = np.concatenate([parent1[:cut], parent2[cut:]])
            else:
                child = parent1.copy()

            # mutation
            mask = rng.random(n_points) < mut_prob_gene
            if mask.any():
                child[mask] += rng.normal(0.0, cur_mut_std, size=mask.sum())
            child = repair(child, target_sum)

            # fitness
            f = compute_c5(child)
            new_pop[i] = child
            new_fit[i] = f

        pop = new_pop
        fitness = new_fit

        # update best
        cur_best = int(np.argmin(fitness))
        if fitness[cur_best] < best_score:
            best_score = fitness[cur_best]
            best_h = pop[cur_best].copy()
            best_idx = cur_best

    # Simple hill‑climbing refinement on best solution
    while True:
        elapsed = time.time() - start
        if elapsed > budget_s * 0.98:
            break
        if best_score <= 0.3808:
            break
        # propose a small random tweak
        cand = best_h.copy()
        mask = rng.random(n_points) < mut_prob_gene
        if mask.any():
            cand[mask] += rng.normal(0.0, mut_std / 3.0, size=mask.sum())
        cand = repair(cand, target_sum)
        cand_score = compute_c5(cand)
        if cand_score < best_score:
            best_h = cand
            best_score = cand_score

    return best_h.tolist(), float(best_score), int(n_points)
