import time
import numpy as np

# ---------- Helper Functions ----------

def compute_c5_corr(h):
    """Return C5 = max_k ∫ h(x)*(1-h(x+k)) dx for discretised h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def compute_c5_and_shift(h):
    """Return (C5 value, integer shift) where shift gives the maximal correlation."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    c5_val = corr[idx] * (2.0 / n)
    shift = idx - (n - 1)        # shift ∈ [-(n-1), …, n-1]
    return c5_val, shift


def shift_array(arr, offset):
    """Shift arr by `offset` positions, zero‑padding outside the interval."""
    n = arr.shape[0]
    out = np.zeros_like(arr)
    if offset >= 0:
        out[offset:] = arr[: n - offset]
    else:
        out[: n + offset] = arr[-offset:]
    return out


def gradient_worst_shift(h, shift):
    """Gradient of C5 for the worst shift (ignoring the common dx factor)."""
    h_fwd = shift_array(h, shift)      # h[i - shift] (zero‑padded)
    h_bwd = shift_array(h, -shift)     # h[i + shift]
    return (1.0 - h_fwd) - h_bwd


def project_sum_and_bounds(x, target):
    """
    Project vector x onto the intersection of:
        0 ≤ x ≤ 1  and  sum(x) = target
    Using an iterative water‑filling scheme (fast for small n).
    """
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x

    for _ in range(100):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            caps = 1.0 - x
            free = caps > 1e-12
            cnt = np.count_nonzero(free)
            if cnt == 0:
                break
            delta = diff / cnt
            inc = np.minimum(delta, caps[free])
            x[free] += inc
        else:
            caps = x
            free = caps > 1e-12
            cnt = np.count_nonzero(free)
            if cnt == 0:
                break
            delta = diff / cnt               # negative
            dec = np.maximum(delta, -caps[free])
            x[free] += dec
        diff = target - x.sum()
    # Final safeguard
    x = np.clip(x, 0.0, 1.0)
    return x


def binary_like_initial(n, rng):
    """Generate a binary (+ one fractional) vector with sum = n/2."""
    target = n / 2.0
    whole = int(np.floor(target))
    frac = target - whole
    h = np.zeros(n, dtype=float)
    perm = rng.permutation(n)
    h[perm[:whole]] = 1.0
    if frac > 1e-12:
        h[perm[whole]] = frac
    return h

    """
