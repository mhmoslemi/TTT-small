import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # use correlation for full range
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def project_to_feasible(v, target_sum):
    # Clip to [0,1] then adjust sum using a free index if possible
    v = np.clip(v, 0.0, 1.0)
    diff = target_sum - np.sum(v)
    if abs(diff) > 1e-12:
        free = np.where((v > 0.0) & (v < 1.0))[0]
        if free.size > 0:
            i = free[0]
            v[i] = np.clip(v[i] + diff, 0.0, 1.0)
        else:
            # all at bounds, distribute uniformly (may violate bounds slightly)
            v += diff / len(v)
            v = np.clip(v, 0.0, 1.0)
    # final correction for any tiny residual
    residual = target_sum - np.sum(v)
    if abs(residual) > 1e-12:
        i = np.argmin(v) if residual > 0 else np.argmax(v)
        v[i] = np.clip(v[i] + residual, 0.0, 1.0)
    return v

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Starting construction
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        # fall back to a simple feasible vector (half ones, half zeros)
        n0 = 45
        h0 = np.full(n0, 0.5)   # sum = n0/2
    n0 = h0.shape[0]
    target0 = n0 / 2.0
    # ensure exact feasibility
    h0 = project_to_feasible(h0, target0)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    best_n = n0

    # resolution list, trying finer discretizations

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Return the overlap bound C5 for a discrete step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(dx * np.max(corr))

def project_to_feasible(v, target_sum):
    """
    Clip to [0,1] and adjust a single free entry so that the total sum equals target_sum.
    """
    v = np.clip(v, 0.0, 1.0)
    diff = target_sum - np.sum(v)
    if abs(diff) > 1e-12:
        free = np.where((v > 0.0) & (v < 1.0))[0]
        if free.size:
            i = free[0]
            v[i] = np.clip(v[i] + diff, 0.0, 1.0)
        else:
            # all at bounds – distribute uniformly (may slightly violate bounds, fix later)
            v += diff / len(v)
            v = np.clip(v, 0.0, 1.0)
    # tiny residual correction
    residual = target_sum - np.sum(v)
    if abs(residual) > 1e-12:
        i = np.argmin(v) if residual > 0 else np.argmax(v)
        v[i] = np.clip(v[i] + residual, 0.0, 1.0)
    return v
