def propose_move(h, rng):
    n = h.shape[0]
    # pick two distinct indices
    i, j = rng.choice(n, size=2, replace=False)
    hi = h[i]
    hj = h[j]
    # feasible delta range
    delta_max = min(1 - hi, hj)
    delta_min = -min(hi, 1 - hj)
    if delta_max <= delta_min:
        # cannot move, return original copy
        return h.copy()
    # choose delta uniformly in [delta_min, delta_max] but not zero
    delta = rng.uniform(delta_min, delta_max)
    # if very small delta relative to tolerance, set to zero; but okay
    h_new = h.copy()
    h_new[i] = hi + delta
    h_new[j] = hj - delta
    return h_new

import numpy as np
import time

def compute_c5(h):
    """Return the overlap value C5 for a discretized step function h."""
    n = h.shape[0]
    # full cross-correlation of h with (1 - h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def propose_move(h, rng, max_tries=10):
    """Return a new h after moving a feasible amount of mass between two indices."""
    n = h.shape[0]
    for _ in range(max_tries):
        i, j = rng.choice(n, size=2, replace=False)
        hi, hj = h[i], h[j]
        # feasible delta so that both stay inside [0, 1]
        delta_max = min(1.0 - hi, hj)
        delta_min = -min(hi, 1.0 - hj)
        if delta_max > delta_min:
            delta = rng.uniform(delta_min, delta_max)
            if abs(delta) < 1e-12:
                continue
            h_new = h.copy()
            h_new[i] = hi + delta
            h_new[j] = hj - delta
            return h_new
    return h.copy()  # no feasible move found

def generate_random_binary(rng, n):
    """Generate a random vector respecting the sum constraint:
       - if n even: exactly n/2 ones, rest zeros
       - if n odd : floor(n/2) ones, one entry 0.5, rest zeros
    """
    target = n / 2.0
    if n % 2 == 0:
        w = int(target)
        ones = rng.choice(n, size=w, replace=False)
        h = np.zeros(n, dtype=np.float64)
        h[ones] = 1.0
        return h
    else:
        w = int(np.floor(target))
        ones = rng.choice(n, size=w, replace=False)
        remaining = np.setdiff1d(np.arange(n), ones, assume_unique=True)
        half_idx = rng.choice(remaining)
        h = np.zeros(n, dtype=np.float64)
        h[ones] = 1.0
        h[half_idx] = 0.5
        return h

def local_search(h_start, rng, start_time, budget_s):
    """Hill‑climbing / simulated‑annealing style improvement of h."""
    h = h_start.copy()
    c5_cur = compute_c5(h)

        if elapsed >= budget_s - 0.1:

            h, c5_cur = cand, c5_cand
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= 5000:
            break
    return h, c5_cur

        h_rand = generate_random_binary(rng, n)
        imp_h, imp_c5 = local_search(h_rand, rng, start, budget_s)
        if imp_c5 < best_c5:
            best_h, best_c5 = imp_h, imp_c5
            if best_c5 <= 0.3808:
                break

    # Final safety projection onto the exact constraints
    np.clip(best_h, 0.0, 1.0, out=best_h)
    target_sum = n / 2.0
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-8:
        for _ in range(100):
            idx = rng.integers(n)
            new_val = best_h[idx] + diff
            if 0.0 <= new_val <= 1.0:
                best_h[idx] = new_val
                break

    final_c5 = compute_c5(best_h)
    return best_h, float(final_c5), n
