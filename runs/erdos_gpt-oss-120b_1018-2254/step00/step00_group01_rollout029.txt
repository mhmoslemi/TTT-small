def propose_move(h, target_sum, rng):
    n = h.shape[0]
    # choose donor index where h>0
    donors = np.where(h > 1e-12)[0]
    receivers = np.where(h < 1 - 1e-12)[0]

    i = rng.choice(donors)

def binary_pattern(n, target_sum, rng):
    # target_sum may be .5 fraction
    whole = int(math.floor(target_sum))
    frac = target_sum - whole
    h = np.zeros(n, dtype=float)
    # choose indices for ones
    one_idx = rng.choice(n, size=whole, replace=False)
    h[one_idx] = 1.0
    if frac > 1e-12:
        # choose an extra index not in one_idx
        remaining = np.setdiff1d(np.arange(n), one_idx, assume_unique=True)
        idx = rng.choice(remaining)
        h[idx] = frac
    return h

    best_h = project_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h, dx)

    return (best_h, float(best_c5), int(n_points))
