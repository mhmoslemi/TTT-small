def compute_c5(h):
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def propose_move(h, step, rng):
    N = h.shape[0]
    i, j = rng.choice(N, size=2, replace=False)
    # Determine bounds for delta such that 0 <= new values <= 1
    max_delta = min(step, 1.0 - h[i], h[j])
    min_delta = -min(step, h[i], 1.0 - h[j])
    if max_delta < min_delta:
        # no feasible delta, return same h
        return None
    delta = rng.uniform(min_delta, max_delta)
    if abs(delta) < 1e-12:
        return None
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    return new_h

import time, math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)*(1-h(x+k)) dx using discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _proj_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Projection of v onto the simplex {x>=0, sum(x)=z} (Duchi et al.)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = 0
        theta = 0.0
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


def _feasible_init(rng: np.random.Generator, n: int) -> np.ndarray:
    """Create a random feasible vector of length n (bounds [0,1], sum=n/2)."""
    # start from uniform 0.5 and add perturbations, then project
    h = np.full(n, 0.5, dtype=np.float64)
    perturb = rng.uniform(-0.4, 0.4, size=n)
    h = np.clip(h + perturb, 0.0, 1.0)
    # project onto bounded simplex
    target = n / 2.0
    for _ in range(5):
        h = _proj_simplex(h, target)
        h = np.clip(h, 0.0, 1.0)
    # tiny uniform fix if needed
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        inc = diff / n
        h = np.clip(h + inc, 0.0, 1.0)
    return h


def _propose_move(h: np.ndarray, step: float, rng: np.random.Generator):
    """Return a new vector obtained by moving mass delta between two coordinates."""
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    # allowable interval for delta so that both stay in [0,1]
    lo = max(-step, -h[i], h[j] - 1.0)
    hi = min(step, 1.0 - h[i], h[j])
    if lo >= hi:
        return None
    delta = rng.uniform(lo, hi)
    if abs(delta) < 1e-14:
        return None
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    return new_h


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] for the Erdős C5 problem.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # Initialise h ---------------------------------------------------------
    # ----------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64).copy()
        n = h.shape[0]
        target_sum = n / 2.0
        # project onto feasible region if needed
        h = np.clip(h, 0.0, 1.0)
        for _ in range(5):
            h = _proj_simplex(h, target_sum)
            h = np.clip(h, 0.0, 1.0)
    else:
        n = 70
        h = _feasible_init(rng, n)

    # ----------------------------------------------------------------------
    # Baseline evaluation --------------------------------------------------
    # ----------------------------------------------------------------------
    curr_score = _compute_c5(h)
    best_h = h.copy()
    best_score = curr_score

    start = time.time()
    T0 = 0.01          # initial temperature
    step0 = 0.12       # initial max delta
    target_sum = n / 2.0
    it = 0

    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s:
            break

        # linear cooling and step decay
        T = T0 * (1.0 - elapsed / budget_s)
        step = step0 * (1.0 - elapsed / budget_s) + 0.001

        cand = _propose_move(h, step, rng)
        if cand is None:
            continue

        cand_score = _compute_c5(cand)

        # record best ever
        if cand_score < best_score:
            best_score = cand_score
            best_h = cand.copy()

        # Metropolis acceptance
        if cand_score <= curr_score:
            accept = True
        else:
            accept = rng.random() < math.exp(-(cand_score - curr_score) / (T + 1e-12))

        if accept:
            h = cand
            curr_score = cand_score

        # occasional re‑projection to keep the sum exact (every 1000 iters)
        it += 1
        if it % 1000 == 0:
            s = h.sum()
            if abs(s - target_sum) > 1e-9:
                diff = target_sum - s
                inc = diff / n
                h = np.clip(h + inc, 0.0, 1.0)
                # re‑project onto simplex to be safe
                for _ in range(3):
                    h = _proj_simplex(h, target_sum)
                    h = np.clip(h, 0.0, 1.0)
                curr_score = _compute_c5(h)

    final_score = _compute_c5(best_h)

    return best_h, final_score, n
