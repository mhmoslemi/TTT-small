def repair_h(v, target_sum):
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target_sum - total
    # If diff zero, done
    if abs(diff) < 1e-12:
        return v
    # indices not at bounds.
    # Determine if diff positive need to add mass; diff negative need to remove.
    # We'll distribute diff across free indices without exceeding bounds.
    while abs(diff) > 1e-10:
        free = np.where(((diff > 0) & (v < 1.0)) | ((diff < 0) & (v > 0.0)))[0]
        if free.size == 0:
            # cannot adjust further
            break
        # compute max possible change per free entry to not exceed bounds:
        # For each free idx, max increase = 1 - v[i], max decrease = -v[i]
        # We'll compute delta = diff / len(free) limited by these.
        delta = diff / free.size
        # limit delta
        # For each free idx, actual change = min(max(delta, -v[i]), 1-v[i])
        # But simplest: compute allowed = vector of limits per entry.
        # We'll compute sign.
        # Use np.clip
        max_inc = 1.0 - v[free]
        max_dec = -v[free]
        # allowed delta:
        allowed = np.where(delta > 0, max_inc, max_dec)
        # If allowed magnitude smaller than abs(delta), adjust only allowed.
        actual_delta = np.where(np.abs(delta) <= np.abs(allowed), delta, allowed)
        # update
        v[free] += actual_delta
        # reclip
        v = np.clip(v, 0.0, 1.0)
        total = v.sum()
        diff = target_sum - total
        # If all updates saturate, break to avoid infinite loop.
        if np.all(np.abs(actual_delta) < 1e-12):
            break
    # final ensure sum close:
    v = np.clip(v, 0.0, 1.0)
    # If still off due to rounding, adjust smallest free.
    diff = target_sum - v.sum()
    if abs(diff) > 1e-9:
        # adjust first free entry by diff
        free = np.where((v > 0) & (v < 1))[0]
        if free.size > 0:
            v[free[0]] += diff
            v = np.clip(v, 0.0, 1.0)
        else:
            # adjust all entries uniformly within bounds
            v += diff / len(v)
            v = np.clip(v, 0.0, 1.0)
    return v

def tournament_select(pop, fitnesses, rng, k=3):
    # choose k random individuals, return the one with lowest fitness.
    idx = rng.choice(len(pop), size=k, replace=False)
    best_idx = idx[0]
    best_fit = fitnesses[best_idx]
    for i in idx[1:]:
        if fitnesses[i] < best_fit:
            best_fit = fitnesses[i]
            best_idx = i
    return pop[best_idx]
