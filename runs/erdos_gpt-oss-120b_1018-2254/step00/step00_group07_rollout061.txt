import numpy as np, scipy.optimize as opt, time, math

# Global placeholders
n_points = None
dx = None
target_sum = None

def compute_c5_value(h):
    # Compute max overlap integral (objective)
    # raw correlation using np.correlate
    raw = np.correlate(h, 1.0 - h, mode='full')
    return raw.max() * dx

def compute_raw_correlation(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_c5(h):
    # gradient of C5 (max correlation) wrt h
    raw = compute_raw_correlation(h)
    idx = raw.argmax()
    k = idx - (n_points - 1)
    grad = np.zeros_like(h)
    if k >= 0:
        i = np.arange(0, n_points - k)
        j = i + k
    else:
        k_abs = -k
        i = np.arange(k_abs, n_points)
        j = i - k_abs
    grad[i] += 1.0 - h[j]
    grad[j] += -h[i]
    grad *= dx
    return grad

def eq_constraint_fun(h):
    return np.sum(h) - target_sum

def eq_constraint_jac(h):
    return np.ones_like(h)

def project_feasible(h):
    # Clip to bounds
    h = np.clip(h, 0.0, 1.0)
    # Adjust sum to target using interior redistribution
    for _ in range(50):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 0.0) & (h < 1.0)
        if free.any():
            adjust = diff / free.sum()
            h[free] += adjust
            h = np.clip(h, 0.0, 1.0)
        else:
            # No interior points; adjust a random element within bounds
            idx = np.random.randint(len(h))
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

import time
import numpy as np
import scipy.optimize as opt

# globals that will be set inside run()
n_points = None   # number of discretisation points
dx = None         # step size 2/n_points
target_sum = None  # required sum of h (n_points/2)

def compute_c5_value(h):
    """Maximum overlap integral C5 for a given h (uses global dx)."""
    raw = np.correlate(h, 1.0 - h, mode='full')
    return raw.max() * dx

def _raw_corr(h):
    """Raw correlation array used by gradient and other helpers."""
    return np.correlate(h, 1.0 - h, mode='full')

def grad_c5(h):
    """Subgradient of C5 (max correlation) w.r.t. h."""
    raw = _raw_corr(h)
    idx = raw.argmax()
    k = idx - (n_points - 1)            # shift corresponding to the max
    grad = np.zeros_like(h)

    if k >= 0:
        i = np.arange(0, n_points - k)
        j = i + k
    else:
        k_abs = -k
        i = np.arange(k_abs, n_points)
        j = i - k_abs

    # contribution from sum_i h_i term
    grad[i] += 1.0 - h[j]
    # contribution from -h_i*h_{i+k} term
    grad[j] += -h[i]
    grad *= dx
    return grad

def eq_constraint_fun(h):
    """Equality constraint: sum(h) = target_sum."""
    return np.sum(h) - target_sum

def eq_constraint_jac(h):
    """Jacobian of the equality constraint (vector of ones)."""
    return np.ones_like(h)

def project_feasible(h):
    """Clip to [0,1] and adjust sum to target_sum by redistributing mass."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 0.0) & (h < 1.0)
        if free.any():
            adjust = diff / free.sum()
            h[free] += adjust
            h = np.clip(h, 0.0, 1.0)
        else:
            # no interior points – tweak a random entry
            idx = np.random.randint(len(h))
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Optimise a step function h on [0,2] to minimise the Erdős C₅ bound."""
    global n_points, dx, target_sum

    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Determine problem size and initialise h
    # ------------------------------------------------------------------
    try:
        init = np.array(initial_h_values, dtype=float)
        n_points = int(init.shape[0])
    except Exception:
        n_points = 80
        init = None

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    if init is not None:
        h0 = np.clip(init, 0.0, 1.0)
        h0 = project_feasible(h0)
    else:
        # fallback – binary vector with exactly target_sum ones
        h0 = np.zeros(n_points)
        t = int(round(target_sum))
        idx = np.random.permutation(n_points)[:t]
        h0[idx] = 1.0

    best_h = h0.copy()
    best_c5 = compute_c5_value(best_h)

    # ------------------------------------------------------------------
    # Simple binary hill‑climbing (useful when the start is binary)
    # ------------------------------------------------------------------
    if np.all((best_h == 0) | (best_h == 1)):
        cur_h = best_h.copy()
        cur_c5 = best_c5
        while time.time() - start_time < budget_s * 0.2:
            ones = np.where(cur_h > 0.5)[0]
            zeros = np.where(cur_h < 0.5)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            new_h = cur_h.copy()
            new_h[i], new_h[j] = 0.0, 1.0
            new_c5 = compute_c5_value(new_h)
            if new_c5 < cur_c5 - 1e-12:
                cur_h, cur_c5 = new_h, new_c5
                if new_c5 < best_c5:
                    best_h, best_c5 = new_h.copy(), new_c5

    # ------------------------------------------------------------------
    # Main optimisation loop – SLSQP with many random restarts
    # ------------------------------------------------------------------
    bounds = [(0.0, 1.0)] * n_points
    constraints = [{'type': 'eq',
                    'fun': eq_constraint_fun,
                    'jac': eq_constraint_jac}]
    max_iter = 200

    while time.time() - start_time < budget_s * 0.9:
        # pick a starting point: perturb current best or random binary
        if np.random.rand() < 0.5:
            start_vec = best_h + 0.2 * np.random.randn(n_points)
        else:
            start_vec = np.zeros(n_points)
            t = int(round(target_sum))
            idx = np.random.permutation(n_points)[:t]
            start_vec[idx] = 1.0
        start_vec = project_feasible(start_vec)

        try:
            res = opt.minimize(compute_c5_value,
                               start_vec,
                               method='SLSQP',
                               bounds=bounds,
                               constraints=constraints,
                               jac=grad_c5,
                               options={'maxiter': max_iter, 'ftol': 1e-9, 'disp': False})
        except Exception:
            continue

        h_opt = np.clip(res.x, 0.0, 1.0)
        h_opt = project_feasible(h_opt)   # ensure exact feasibility

        c5_val = compute_c5_value(h_opt)
        if c5_val < best_c5:
            best_h, best_c5 = h_opt.copy(), c5_val

        if time.time() - start_time > budget_s - 5:
            break

    # final safety projection and evaluation
    best_h = project_feasible(best_h)
    best_c5 = compute_c5_value(best_h)

    return (best_h, float(best_c5), n_points)
