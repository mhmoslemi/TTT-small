def make_feasible(v):
    # v: any vector
    # clip to [0,1] and project onto sum=n/2
    v_clipped = np.clip(v, 0.0, 1.0)
    target = len(v) / 2.0
    # Use projection onto bounded simplex described.
    # solve lambda via bisection
    low = -1.0
    high = np.max(v_clipped) + 1.0 # ensure positive range
    for _ in range(50):
        lam = (low + high) / 2.0
        x = np.clip(v_clipped - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            low = lam
        else:
            high = lam
    x = np.clip(v_clipped - (low+high)/2.0, 0.0, 1.0)
    # to correct rounding errors:
    diff = target - x.sum()
    if abs(diff) > 1e-8:
        # distribute diff uniformly to entries not at bounds
        free = (x > 0) & (x < 1)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

import numpy as np

def _project_onto_box_simplex(v, target):
    """Project vector v onto {0<=x<=1, sum(x)=target}."""
    v = np.asarray(v, dtype=float)
    low, high = -1.0, np.max(v) + 1.0
    for _ in range(60):
        lam = (low + high) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            low = lam
        else:
            high = lam
    x = np.clip(v - (low + high) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 0) & (x < 1)
        if mask.any():
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def _c5_of(h):
    """Return the discrete C5 value for a step function h on [0,2]."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)

    best_c5 = _c5_of(best_h)

    # ------------------------------------------------------------------
    # Simulated annealing / directed‑move loop
    # ------------------------------------------------------------------
    start_time = time.time()
    temperature = 1.0
    decay = 0.9995
    min_temp = 1e-4
    step_base = 0.2                     # maximal transfer amount
    max_no_improve = 5000               # used only for occasional restarts
    no_improve_counter = 0
    target_c5 = 0.3808                  # early‑stop threshold

    while time.time() - start_time < budget_s:
        # Compute the shift that currently dominates the max overlap
        full_corr = np.correlate(h_curr, 1.0 - h_curr, mode='full')
        idx_max = int(np.argmax(full_corr))
        shift = idx_max - (n - 1)       # negative ⇔ left shift, positive ⇔ right shift

        cand = None

        # ------------------------------------------------------------------
        # Directed move along the worst shift, if possible
        # ------------------------------------------------------------------
        if shift != 0:
            idx = np.arange(n)
            dest = idx + shift
            if shift > 0:
                mask = (dest < n)
                src_ok = (h_curr > 1e-12) & mask
                dst_ok = (h_curr[dest] < 1.0 - 1e-12) & mask
                feasible = np.where(src_ok & dst_ok)[0]
            else:
                mask = (dest >= 0)
                src_ok = (h_curr > 1e-12) & mask
                dst_ok = (h_curr[dest] < 1.0 - 1e-12) & mask
                feasible = np.where(src_ok & dst_ok)[0]

            if feasible.size:
                i = np.random.choice(feasible)
                j = i + shift
                step = step_base * temperature
                max_transfer = min(step, h_curr[i], 1.0 - h_curr[j])
                if max_transfer > 0:
                    delta = max_transfer * np.random.rand()
                    cand = h_curr.copy()
                    cand[i] -= delta
                    cand[j] += delta

        # ------------------------------------------------------------------
        # If directed move impossible, fall back to a random feasible transfer
        # ------------------------------------------------------------------
        if cand is None:
            i, j = np.random.choice(n, size=2, replace=False)
            step = step_base * temperature
            max_transfer = min(step, h_curr[i], 1.0 - h_curr[j])
            if max_transfer <= 0:
                continue
            delta = max_transfer * np.random.rand()
            cand = h_curr.copy()
            cand[i] -= delta
            cand[j] += delta

        # ------------------------------------------------------------------
        # Evaluate candidate
        # ------------------------------------------------------------------
        cand_c5 = _c5_of(cand)

        # ------------------------------------------------------------------
        # Acceptance criterion (simulated annealing style)
        # ------------------------------------------------------------------
        accept = False
        if cand_c5 < _c5_of(h_curr):
            accept = True
        else:
            prob = math.exp(-(cand_c5 - _c5_of(h_curr)) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            h_curr = cand
            no_improve_counter = 0
        else:
            no_improve_counter += 1

        # ------------------------------------------------------------------
        # Update best solution if improved
        # ------------------------------------------------------------------
        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = cand.copy()
            temperature = 1.0                 # reset temperature after improvement
            no_improve_counter = 0
            if best_c5 <= target_c5:
                break

        # ------------------------------------------------------------------
        # Temperature schedule and occasional global perturbation
        # ------------------------------------------------------------------
        temperature = max(temperature * decay, min_temp)

        if no_improve_counter >= max_no_improve:
            # Inject a modest global noise around the best solution
            noise = (np.random.rand(n) - 0.5) * 2.0 * step_base * 0.3
            h_curr = _project_onto_box_simplex(best_h + noise, target_sum)
            no_improve_counter = 0

    return (best_h, best_c5, n)
