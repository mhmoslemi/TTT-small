def project_to_box_sum(v, desired_sum):
    lo = -1.0
    hi = 1.0
    # Expand bounds to be safe
    # Actually set lo = -np.max(v) - 1, hi = np.max(v) + 1
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0, 1)
        s = x.sum()
        if s < desired_sum:
            hi = lam
        else:
            lo = lam
    x = np.clip(v - lo, 0, 1)
    return x

def to_binary(v):
    n = len(v)
    half = n // 2
    binary = np.zeros(n, dtype=float)
    idx = np.argsort(-v)
    binary[idx[:half]] = 1.0
    if n % 2 == 1:
        binary[idx[half]] = 0.5
    # Project to ensure sum exactly (in case odd)
    desired = n / 2.0
    binary = project_to_box_sum(binary, desired)
    return binary

import time
import random
import numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, desired):
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s < desired:
            hi = lam
        else:
            lo = lam
    return np.clip(v - lo, 0.0, 1.0)

def random_binary(n):
    h = np.zeros(n, dtype=float)
    half = n // 2
    ones = np.random.choice(n, half, replace=False)
    h[ones] = 1.0
    if n % 2:
        not_one = np.setdiff1d(np.arange(n), ones)
        h[np.random.choice(not_one)] = 0.5
    return h

def to_binary(v):
    n = v.shape[0]
    half = n // 2
    idx = np.argsort(-v)
    h = np.zeros(n, dtype=float)
    h[idx[:half]] = 1.0
    if n % 2:
        h[idx[half]] = 0.5
    return project_to_box_sum(h, n / 2.0)

def gradient_ascent_opt(h0, deadline):
    h = h0.copy()
    n = h.shape[0]
    desired = n / 2.0
    best_h = h.copy()
    best_val = compute_c5(h)
    step = 0.5
    while time.time() < deadline:
        corr = np.correlate(h, h, mode='full')
        min_idx = int(np.argmin(corr))
        shift = min_idx - (n - 1)
        if shift == 0:
            grad = 2.0 * h
        else:
            grad = np.zeros_like(h)
            if shift > 0:
                grad[:n - shift] += h[shift:]
                grad[shift:] += h[:n - shift]
            else:
                s = -shift
                grad[s:] += h[:n - s]
                grad[:n - s] += h[s:]
        h_new = h + step * grad
        h_new = project_to_box_sum(h_new, desired)
        val = compute_c5(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        h = h_new
        step *= 0.995
        if step < 1e-6:
            step = 0.05
    return best_h, best_val
