def neighbor_move(h, step):
   n = len(h)
   i,j = np.random.choice(n,2,replace=False)
   lb = max(-h[i], h[j] - 1.0)
   ub = min(1.0 - h[i], h[j])
   if lb > ub:
       return h # cannot move between these indices
   # draw delta uniformly from [-step,step] intersect [lb,ub]
   lo = max(-step, lb)
   hi = min(step, ub)
   if lo > hi:
       return h
   delta = np.random.uniform(lo, hi)
   h_new = h.copy()
   h_new[i] += delta
   h_new[j] -= delta
   return h_new

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, h, mode='full')
    min_corr = np.min(conv)
    return 1.0 - min_corr * dx

def neighbor_move(h, max_step):
    n = len(h)
    i, j = np.random.choice(n, 2, replace=False)
    # compute feasible delta range
    # delta added to i, sub from j
    lb = max(-h[i], h[j] - 1.0)
    ub = min(1.0 - h[i], h[j])
    # constrain to max_step range
    lb = max(lb, -max_step)
    ub = min(ub, max_step)
    if lb >= ub:
        return None  # no feasible move
    delta = np.random.uniform(lb, ub)
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

import numpy as np
import time
import math

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, h, mode='full')
    min_corr = np.min(conv)
    return 1.0 - min_corr * dx

def neighbor_move_delta(h, max_step):
    n = len(h)
    i, j = np.random.choice(n, 2, replace=False)
    # feasible delta range
    lb = max(-h[i], h[j] - 1.0)
    ub = min(1.0 - h[i], h[j])
    # limit by max_step
    if max_step < 0:
        return None
    if lb > ub:
        return None
    lo = max(lb, -max_step)
    hi = min(ub, max_step)
    if lo > hi:
        return None
    delta = np.random.uniform(lo, hi)
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

def neighbor_move_swap(h):
    n = len(h)
    i, j = np.random.choice(n, 2, replace=False)
    h_new = h.copy()
    h_new[i], h_new[j] = h_new[j], h_new[i]
    return h_new

def ensure_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # water-filling iterative projection onto sum constraint
    for _ in range(100):
        current_sum = np.sum(h)
        diff = target - current_sum
        if abs(diff) < 1e-12:
            break
        # free indices not at bounds
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if len(free) == 0:
            # cannot adjust further
            break
        delta = diff / len(free)
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final small correction if needed
    diff = target - np.sum(h)
    if abs(diff) > 1e-9:
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if len(free) > 0:
            h[free] += diff / len(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    # Determine number of points
    init = globals().get('initial_h_values', None)
    if init is not None:
        h0 = np.array(init, dtype=np.float64)
        n = h0.shape[0]
        h0 = ensure_feasible(h0)
    else:
        n = 47
        h0 = np.full(n, 0.5, dtype=np.float64)
    # Initialize
    current_h = h0.copy()
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    current_c5 = best_c5
    # Parameters
    max_step_initial = 0.4
    max_step_min = 0.01
    T_initial = 0.02
    T_final = 0.0005
    swap_prob = 0.2
    # Main loop
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.99:
            break
        frac = elapsed / budget_s
        # Linear schedules
        max_step = max_step_initial * (1 - frac) + max_step_min * frac
        T = T_initial * (1 - frac) + T_final * frac
        # Generate neighbor
        if np.random.rand() < swap_prob:
            neighbor = neighbor_move_swap(current_h)
        else:
            neighbor = neighbor_move_delta(current_h, max_step)
            if neighbor is None:
                continue
        # Evaluate
        neighbor_c5 = compute_c5(neighbor)
        # Acceptance (simulated annealing)
        if neighbor_c5 < current_c5:
            current_h = neighbor
            current_c5 = neighbor_c5
        else:
            # Probabilistic acceptance of worse move
            if T > 0 and np.random.rand() < math.exp(-(neighbor_c5 - current_c5) / T):
                current_h = neighbor
                current_c5 = neighbor_c5
        # Update global best
        if neighbor_c5 < best_c5:
            best_c5 = neighbor_c5
            best_h = neighbor.copy()
    # Return result
    return (best_h.tolist(), float(best_c5), n)

import numpy as np
import time
import math

def compute_c5(h: np.ndarray) -> float:
    """Return the C5 value for a step function represented by samples h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def ensure_feasible(h: np.ndarray) -> np.ndarray:
    """Project h onto the box [0,1]^n and the hyperplane sum(h)=n/2."""
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

    # Simple water‑filling iteration
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)

    # One tiny correction pass if needed
    diff = target - np.sum(h)
    if abs(diff) > 1e-9:
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size:
            h[free] += diff / free.size
            h = np.clip(h, 0.0, 1.0)
    return h


def neighbor_move_delta(h: np.ndarray, max_step: float) -> np.ndarray | None:
    """Two‑index delta move preserving the sum."""
    n = h.size
    i, j = np.random.choice(n, 2, replace=False)

    # feasible delta interval
    lower = max(-h[i], h[j] - 1.0)
    upper = min(1.0 - h[i], h[j])
    if lower > upper:
        return None
    lo = max(lower, -max_step)
    hi = min(upper, max_step)
    if lo > hi:
        return None
    delta = np.random.uniform(lo, hi)
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new


def neighbor_move_swap(h: np.ndarray) -> np.ndarray:
    """Swap two entries (trivially keeps sum and bounds)."""
    n = h.size
    i, j = np.random.choice(n, 2, replace=False)
    h_new = h.copy()
    h_new[i], h_new[j] = h_new[j], h_new[i]
    return h_new


def neighbor_move_target(h: np.ndarray, max_step: float) -> np.ndarray | None:
    """
    Targeted move that tries to reduce the current maximal overlap.
    It selects a pair of indices participating in the maximal correlation
    and transfers a small amount of mass between them.
    """
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    imax = int(np.argmax(corr))
    shift = imax - (n - 1)          # shift = k such that corr corresponds to h[i] with (1-h)[i+shift]
    if shift >= 0:
        i_start, i_end = 0, n - shift - 1
    else:
        i_start, i_end = -shift, n - 1
    if i_start > i_end:
        return None
    i = np.random.randint(i_start, i_end + 1)
    j = i + shift  # partner index

    # Random direction: from i to j or opposite
    if np.random.rand() < 0.5:
        # move mass from i -> j
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            return None
        eps = np.random.uniform(0.0, min(max_eps, max_step))
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
    else:
        # move mass from j -> i
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0.0:
            return None
        eps = np.random.uniform(0.0, min(max_eps, max_step))
        h_new = h.copy()
        h_new[j] -= eps
        h_new[i] += eps
    return h_new


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a low‑C₅ step function.

    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start = time.time()

    # --------------------------------------------------------------
    # Initialise h
    init = globals().get('initial_h_values')
    if init is not None:
        h0 = np.asarray(init, dtype=np.float64)
        n = h0.size
        h0 = ensure_feasible(h0)
    else:
        n = 47
        h0 = np.full(n, 0.5, dtype=np.float64)   # uniform feasible start

    current_h = h0.copy()
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    current_c5 = best_c5

    # --------------------------------------------------------------
    # Search parameters
    max_step_initial = 0.4      # maximal delta in early stage
    max_step_min = 0.01         # minimal delta near the end
    temp_initial = 0.02
    temp_final = 0.0005
    p_swap = 0.15               # probability of performing a pure swap
    p_target = 0.20             # probability of a targeted reduction move

    # --------------------------------------------------------------
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.99:
            break

        frac = elapsed / budget_s
        max_step = max_step_initial * (1 - frac) + max_step_min * frac
        temperature = temp_initial * (1 - frac) + temp_final * frac

        # Choose move type
        r = np.random.rand()
        if r < p_target:
            neighbor = neighbor_move_target(current_h, max_step)
            if neighbor is None:
                continue
        elif r < p_target + p_swap:
            neighbor = neighbor_move_swap(current_h)
        else:
            neighbor = neighbor_move_delta(current_h, max_step)
            if neighbor is None:
                continue

        # Evaluation
        neighbor_c5 = compute_c5(neighbor)

        # Acceptance (simulated annealing)
        if neighbor_c5 < current_c5:
            current_h = neighbor
            current_c5 = neighbor_c5
        else:
            if temperature > 0 and np.random.rand() < math.exp(-(neighbor_c5 - current_c5) / temperature):
                current_h = neighbor
                current_c5 = neighbor_c5

        # Record best
        if neighbor_c5 < best_c5:
            best_c5 = neighbor_c5
            best_h = neighbor.copy()

    # --------------------------------------------------------------
    return (best_h.tolist(), float(best_c5), n)
