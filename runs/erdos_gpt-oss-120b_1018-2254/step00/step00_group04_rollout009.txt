    # Ensure sum exactly half n

        diff = target_sum - cur_sum
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)
        # if still off, distribute

        if abs(cur_sum - target_sum) > 1e-12:

            scale = target_sum / cur_sum

        corr = np.correlate(hv, 1.0 - hv, mode='full')
        return np.max(corr) * dx

    cur_h = h.copy()

    # parameters

    while True:

        if elapsed > budget_s * 0.96:

        # get correlation and worst shift
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        max_idx = np.argmax(corr)
        cur_val = corr[max_idx] * dx
        if cur_val < best_val:
            best_val = cur_val
            best_h = cur_h.copy()
        # compute shift s relative to zero
        s = max_idx - (n - 1)  # signed shift
        # compute gradient of current worst correlation
        grad = np.zeros(n, dtype=np.float64)
        if s >= 0:

            idx_i = np.arange(n - s)
            grad[idx_i] += 1.0 - cur_h[idx_i + s]
            grad[idx_i + s] -= cur_h[idx_i]

            grad[idx_i] += 1.0 - cur_h[idx_i - t]
            grad[idx_i - t] -= cur_h[idx_i]

            max_step = 0.2  # maximum mass transferred

import time
import numpy as np

def _compute_c5(h):
    """Return C5 = max_k ∫ h(x)*(1-h(x+k)) dx approximated on the grid."""
    n = h.size
    dx = 2.0 / n
    # full cross‑correlation (h with 1‑h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _project_constraints(h, target_sum):
    """Clip to [0,1] and adjust to achieve exactly target_sum."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return h
    # try to add diff to a single feasible entry
    for i in range(h.size):
        if 0.0 <= h[i] + diff <= 1.0:
            h[i] += diff
            return h
    # fall back: scale and clip
    scale = target_sum / cur if cur > 0 else 0.0
    h = np.clip(h * scale, 0.0, 1.0)
    cur = h.sum()
    diff = target_sum - cur
    for i in range(h.size):
        if 0.0 <= h[i] + diff <= 1.0:
            h[i] += diff
            break
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Find a step function h on [0,2] with ∫h=1 that minimizes C5."""
    np.random.seed(seed)
    start_time = time.time()
    # ------------------------------------------------------------------
    # initialise h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n = h.size
    else:
        n = 200                     # fallback size
        h = np.full(n, 0.5, dtype=np.float64)
    target_sum = n / 2.0
    # enforce the integral constraint exactly
    h = _project_constraints(h, target_sum)

    # best solution seen so far
    best_h = h.copy()
    best_val = _compute_c5(best_h)

    # ------------------------------------------------------------------
    # optimisation loop (gradient‑guided moves + random perturbations)
    max_step_initial = 0.2          # max mass transferred on a move
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.96:
            break

        # current correlation and worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = int(np.argmax(corr))
        cur_val = corr[max_idx] * (2.0 / n)
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()

        # signed shift giving the maximal overlap
        shift = max_idx - (n - 1)

        # gradient of that particular overlap w.r.t. h
        grad = np.zeros(n, dtype=np.float64)
        if shift >= 0:
            idx = np.arange(n - shift)
            grad[idx] += 1.0 - h[idx + shift]
            grad[idx + shift] -= h[idx]
        else:
            t = -shift
            idx = np.arange(t, n)
            grad[idx] += 1.0 - h[idx - t]
            grad[idx - t] -= h[idx]

        # select a source (positive grad) and a sink (negative grad)
        pos = np.where(grad > 1e-12)[0]
        neg = np.where(grad < -1e-12)[0]
        if pos.size and neg.size:
            # probabilistic choice weighted by magnitude
            p_weights = grad[pos]
            n_weights = -grad[neg]
            src = np.random.choice(pos, p=p_weights / p_weights.sum())
            dst = np.random.choice(neg, p=n_weights / n_weights.sum())
            # decaying step size
            step = max_step_initial * (1.0 - elapsed / (budget_s * 0.96))
            # feasible transfer respecting bounds
            delta = min(step, h[src], 1.0 - h[dst])
            if delta > 1e-12:
                h[src] -= delta
                h[dst] += delta
                continue

    final_c5 = _compute_c5(best_h)

    return (best_h.tolist(), float(final_c5), int(n))
