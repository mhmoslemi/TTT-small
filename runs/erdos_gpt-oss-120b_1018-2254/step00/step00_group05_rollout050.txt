def project_to_feasible(v, target_sum):
    # Clip v to [0,1] then adjust using water-filling to achieve sum target
    # Using binary search for lambda.
    lo = -1.0
    hi = 2.0
    for _ in range(50):
        lam = (lo+hi)/2.0
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    w = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    # After binary search may still be slightly off; adjust by adding small to some positions where not at bounds.
    diff = target_sum - w.sum()
    if abs(diff) > 1e-8:
        # Distribute diff equally to non-bound positions
        free = (w>0) & (w<1)
        num_free = free.sum()
        if num_free > 0:
            w[free] += diff/num_free
            w = np.clip(w,0.0,1.0)
        # If still off, adjust minimal digits
    return w

def simulated_annealing(h0, time_budget, time_start):
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    curr_h = h0.copy()
    curr_c5 = best_c5
    n = len(h0)
    T0 = 0.01
    while time.time()-time_start < time_budget:
        elapsed = time.time()-time_start
        progress = elapsed / time_budget
        T = T0 * (1 - progress) # linear cooling
        i, j = np.random.randint(n, size=2)
        if i == j: continue
        if curr_h[i] <= 1e-12 and curr_h[j] >= 1 - 1e-12: continue # can't move
        # choose direction: move from i to j
        max_eps = min(curr_h[i], 1.0 - curr_h[j])
        if max_eps <= 1e-12: continue
        eps = max_eps * np.random.rand() * 0.5  # up to half of allowable.
        # new h
        curr_h[i] -= eps
        curr_h[j] += eps
        new_c5 = compute_c5(curr_h)
        delta = new_c5 - curr_c5
        if delta < 0 or np.exp(-delta/T) > np.random.rand():
            # accept
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = curr_h.copy()
        else:
            # revert
            curr_h[i] += eps
            curr_h[j] -= eps
    return best_h, best_c5

def project_to_feasible(v, target):
    # Clip to [0,1] then adjust via water-filling.
    # Use binary search on lambda.
    # Ensure monotonic sum with lambda.
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix small sum error by adjusting a random free element if needed
    diff = target - w.sum()
    if abs(diff) > 1e-10:
        # find indices not at bounds
        free = np.where((w > 1e-12) & (w < 1 - 1e-12))[0]
        if free.size > 0:
            # distribute diff uniformly
            w[free] += diff / free.size
            w = np.clip(w, 0.0, 1.0)
        else:
            # adjust arbitrary element
            idx = np.argmin(np.abs(w - 0.5))  # any index
            w[idx] = np.clip(w[idx] + diff, 0.0, 1.0)
    return w

def compute_c5_fast(h):
    n = h.shape[0]
    dx = 2.0 / n
    m = 1 << ((2*n-1)-1).bit_length()  # next power of two >=2n-1
    H = np.fft.rfft(h, n=m)
    G = np.fft.rfft(1.0 - h, n=m)
    conv = np.fft.irfft(H * G, n=m)[:2*n-1]
    return conv.max() * dx

import time
import numpy as np

def _project_to_feasible(v: np.ndarray, target: float) -> np.ndarray:
    """Project vector v onto [0,1]^n with sum == target."""
    lo, hi = -1.0, 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        free = np.where((w > 1e-12) & (w < 1 - 1e-12))[0]
        if free.size:
            w[free] += diff / free.size
            w = np.clip(w, 0.0, 1.0)
        else:
            idx = np.argmin(np.abs(w - 0.5))
            w[idx] = np.clip(w[idx] + diff, 0.0, 1.0)
    return w

def _compute_c5_fast(h: np.ndarray) -> float:
    """Return the overlap integral max_k ∫ h(x)(1‑h(x+k))dx for discrete h."""
    n = h.shape[0]
    dx = 2.0 / n
    m = 1 << ((2 * n - 1) - 1).bit_length()  # next power of two ≥ 2n‑1
    H = np.fft.rfft(h, n=m)
    G = np.fft.rfft(1.0 - h, n=m)
    conv = np.fft.irfft(H * G, n=m)[: 2 * n - 1]
    return float(conv.max() * dx)

def _simulated_annealing(h0: np.ndarray, budget_s: float, start_time: float) -> tuple:
    """Simple annealing that preserves sum(h) and bounds 0‑1."""
    h = h0.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_c5 = _compute_c5_fast(h)
    cur_c5 = best_c5
    T0 = 0.015  # initial temperature
    while time.time() - start_time < budget_s:
        elapsed = time.time() - start_time
        frac = elapsed / budget_s
        T = T0 * (1.0 - frac) + 1e-9
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # choose direction of transfer
        if np.random.rand() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_eps = min(h[src], 1.0 - h[dst])
        if max_eps <= 1e-12:
            continue
        eps = max_eps * np.random.rand() * 0.5
        h[src] -= eps
        h[dst] += eps
        new_c5 = _compute_c5_fast(h)
        delta = new_c5 - cur_c5
        if delta < 0 or np.exp(-delta / T) > np.random.rand():
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert move
            h[src] += eps
            h[dst] -= eps
    return best_h, best_c5

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return (h_values, c5_bound, n_points) after a short optimisation."""
    np.random.seed(seed)
    start_all = time.time()
    best_h = None
    best_c5 = float("inf")
    # ----- use parent construction if available -----

        except Exception:
            pass

    # Candidate sample counts (prefer modest sizes for speed)
    candidates = [70, 100, 150, 200, 250, 300]

    # Main loop: keep trying different sizes while time permits
    while time.time() - start_all < budget_s * 0.95:
        remaining = budget_s - (time.time() - start_all)
        if remaining < 5.0:
            break

        v = np.random.rand(N)
        h0 = _project_to_feasible(v, N / 2.0)
        # allocate a slice of the remaining budget
        sub_budget = min(remaining * 0.45, 30.0)
        h_opt, c5_opt = _simulated_annealing(h0, sub_budget, time.time())
        if c5_opt < best_c5:

    if best_h is None:

        best_h[: N // 2] = 1.0
        best_h = _project_to_feasible(best_h, N / 2.0)
    else:
        best_h = _project_to_feasible(best_h, best_h.shape[0] / 2.0)
