def run(seed=42, budget_s=1100, **kwargs):
    import time
    import numpy as np

    rng = np.random.default_rng(seed)

    # Determine problem size
    try:
        init_vals = initial_h_values
        n = int(init_vals.shape[0] if hasattr(init_vals, "shape") else len(init_vals))
    except NameError:
        init_vals = None
        n = 70  # default size if no parent provided

    # Ensure even length for integer target sum; if odd, we will allow one fractional entry later
    if n % 2 != 0:
        # make n even by adding a dummy zero at the end (does not affect C5 much)
        n += 1

    target_sum = n // 2  # number of ones for binary representation
    dx = 2.0 / n

    def compute_c5(h):
        """Return the overlap constant C5 for a given 1‑D array h."""
        # np.correlate already uses zero padding for mode='full'
        corr = np.correlate(h, 1.0 - h, mode="full")
        return float(np.max(corr) * dx)

    # ----------------------------------------------------------------------
    # Greedy construction (deterministic)
    # ----------------------------------------------------------------------
    def greedy_build():
        """Build a binary sequence by adding ones one by one,
        always choosing the position that gives the smallest current C5."""
        h = np.zeros(n, dtype=np.float64)
        for _ in range(target_sum):
            best_c5 = np.inf
            best_idx = -1
            zero_positions = np.where(h == 0)[0]
            for idx in zero_positions:
                h[idx] = 1.0
                c5_val = compute_c5(h)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_idx = idx
                h[idx] = 0.0
            # place the chosen one
            h[best_idx] = 1.0
        return h

    # ----------------------------------------------------------------------
    # Helper to turn any initial vector into a feasible binary start
    # ----------------------------------------------------------------------
    def binary_from_initial(arr):
        """Create a binary pattern by selecting the target_sum largest entries."""
        flat = np.asarray(arr, dtype=np.float64).ravel()
        if flat.size != n:
            # If dimensions differ, interpolate/resize linearly

            x_new = np.linspace(0, 1, n, endpoint=False)
            flat = np.interp(x_new, x_old, flat)
