import numpy as np, time

def evaluate_h(h):
    N = h.shape[0]
    dx = 2.0 / N
    # compute correlation
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_sum(h):
    N = h.shape[0]
    target = N * 0.5
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # find an index with capacity to adjust within [0,1]
        # attempt to add diff to a random index with room
        idxs = np.where((h + diff >= 0) & (h + diff <= 1))[0]
        if idxs.size > 0:
            h[idxs[0]] += diff
        else:
            # adjust two indices proportionally
            # find idx where h can be increased
            inc = np.where(h < 1)[0]
            dec = np.where(h > 0)[0]
            if inc.size > 0 and diff > 0:
                # increase first inc index by min(diff, remaining capacity)
                i = inc[0]
                add = min(diff, 1 - h[i])
                h[i] += add
                diff -= add
            if dec.size > 0 and diff < 0:
                i = dec[0]
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
            # If still diff non-zero (tiny), distribute to first element clipped
            if abs(diff) > 1e-12:
                h[0] = np.clip(h[0] + diff, 0, 1)
    return h

    seq = np.zeros(p, dtype=float)
    for i in range(p):
        if i == 0:
            seq[i] = 1.0
        else:
            ls = pow(i, (p-1)//2, p)

        idx = np.where(seq==1)[0][0]

import time
import math
import numpy as np

def _evaluate_h(h: np.ndarray) -> float:
    """Return the C5 overlap bound for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with its complement (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _repair_sum(h: np.ndarray) -> np.ndarray:
    """Project h onto the affine subspace sum(h)=n/2 while keeping 0≤h≤1."""
    n = h.shape[0]
    target = n * 0.5
    diff = target - float(h.sum())
    if abs(diff) < 1e-12:
        return h
    # try to fix in a single entry that can accommodate the diff
    idxs = np.where((h + diff >= 0.0) & (h + diff <= 1.0))[0]
    if idxs.size:
        h[idxs[0]] += diff
        return h
    # otherwise distribute diff a little bit
    inc = np.where(h < 1.0)[0]
    dec = np.where(h > 0.0)[0]
    if diff > 0 and inc.size:
        i = inc[0]
        add = min(diff, 1.0 - h[i])
        h[i] += add
        diff -= add
    elif diff < 0 and dec.size:
        i = dec[0]
        sub = min(-diff, h[i])
        h[i] -= sub
        diff += sub
    if abs(diff) > 1e-12:
        # any residual is tiny; adjust the first entry safely
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h


def _random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector with entries in [0,1] and sum=n/2."""
    h = np.zeros(n, dtype=float)
    half = n // 2
    perm = rng.permutation(n)
    h[perm[:half]] = 1.0
    if n % 2:
        h[perm[half]] = 0.5
    return h


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős overlap constant C5.

    Returns (h_values, c5_bound, n_points)
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # --------------------------------------------------------------------
    # 1. Initialise the problem size and starting vector.
    # --------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float).copy()
        n = h.shape[0]
    else:
        n = 45                      # a modest odd size
        h = _random_feasible(n, rng)

    # Enforce feasibility (clip + exact sum)
    h = np.clip(h, 0.0, 1.0)
    h = _repair_sum(h)

    best_h = h.copy()
    best_val = _evaluate_h(best_h)

    cur_h = best_h.copy()
    cur_val = best_val

    # --------------------------------------------------------------------
    # 2. Simulated annealing parameters.
    # --------------------------------------------------------------------
    init_temp = 1.0
    min_temp = 1e-4
    temp = init_temp
    cooling = 0.9995
    max_delta_initial = 0.5     # maximal absolute change in a single move

    # counters for stagnation handling
    no_improve = 0
    iter_cnt = 0

    # --------------------------------------------------------------------
    # 3. Main optimisation loop.
    # --------------------------------------------------------------------
    while True:
        # time guard
        if time.time() - start >= budget_s * 0.95:
            break

        iter_cnt += 1

        # adaptive step size from temperature
        max_delta = max_delta_initial * (temp / init_temp)

        # propose a pairwise exchange that preserves the sum
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        hi, hj = cur_h[i], cur_h[j]

        # feasible interval for the delta
        delta_min = max(-hi, hj - 1.0)
        delta_max = min(1.0 - hi, hj)
        # intersect with the step-size window
        lo = max(delta_min, -max_delta)
        hi_ = min(delta_max, max_delta)
        if lo > hi_:
            continue
        delta = rng.uniform(lo, hi_)

        new_h = cur_h.copy()
        new_h[i] = hi + delta
        new_h[j] = hj - delta

        new_val = _evaluate_h(new_h)

        if new_val < cur_val:
            accept = True
        else:
            prob = math.exp(-(new_val - cur_val) / (temp + 1e-12))
            accept = rng.random() < prob

            cur_val = new_val
            if cur_val < best_val - 1e-12:
                best_h = cur_h.copy()
                best_val = cur_val
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

    return (best_h, final_c5, n)
