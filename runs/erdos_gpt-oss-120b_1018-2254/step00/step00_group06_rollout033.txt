import time
import numpy as np

def project_onto_simplex(v, target):
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = np.sum(x)
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def initialize_population(popsize, n, target, rng):
    pop = []
    # include initial_h_values if present
    if 'initial_h_values' in globals():
        try:
            iv = np.array(initial_h_values, dtype=float)
            if iv.shape[0] == n:
                iv = project_onto_simplex(iv, target)
                pop.append(iv)
        except Exception:
            pass
    while len(pop) < popsize:
        v = rng.random(n)
        v = project_onto_simplex(v, target)
        pop.append(v)
    return np.stack(pop, axis=0)

def tournament_select(pop, scores, tournament_size, rng):
    idx = rng.integers(0, len(pop), size=tournament_size)
    best_idx = idx[np.argmin(scores[idx])]
    return pop[best_idx]

def crossover(p1, p2, p_cross, rng):
    if rng.random() < p_cross:
        point = rng.integers(1, len(p1))
        c1 = np.concatenate([p1[:point], p2[point:]])
        c2 = np.concatenate([p2[:point], p1[point:]])
        return c1, c2
    else:
        return p1.copy(), p2.copy()

def mutate(child, mut_rate, mut_std, target, rng):
    mask = rng.random(child.shape[0]) < mut_rate
    n_mut = np.count_nonzero(mask)
    if n_mut:
        child[mask] += rng.normal(scale=mut_std, size=n_mut)
    child = project_onto_simplex(child, target)
    return child

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    # Determine problem size
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
    else:
        n = 95  # fallback, but should not happen
    target = n / 2.0
    popsize = 300
    p_cross = 0.7
    mut_rate = 0.01
    mut_std = 0.1
    tournament_size = 3
    start = time.time()
    # Initialize population
    pop = initialize_population(popsize, n, target, rng)
    scores = np.array([compute_c5(ind) for ind in pop])
    best_idx = int(np.argmin(scores))
    best = pop[best_idx].copy()
    best_score = float(scores[best_idx])
    # Main GA loop
    generation = 0
    while True:
        # Time check
        if time.time() - start > budget_s * 0.95:
            break
        # Elitism: keep best
        new_pop = [best]
        # Generate offspring
        while len(new_pop) < popsize:
            parent1 = tournament_select(pop, scores, tournament_size, rng)
            parent2 = tournament_select(pop, scores, tournament_size, rng)
            child1, child2 = crossover(parent1, parent2, p_cross, rng)
            child1 = mutate(child1, mut_rate, mut_std, target, rng)
            child2 = mutate(child2, mut_rate, mut_std, target, rng)
            new_pop.append(child1)
            if len(new_pop) < popsize:
                new_pop.append(child2)
        pop = np.stack(new_pop, axis=0)
        scores = np.array([compute_c5(ind) for ind in pop])
        # Update best
        cur_best_idx = int(np.argmin(scores))
        cur_best = pop[cur_best_idx]
        cur_score = float(scores[cur_best_idx])
        if cur_score < best_score:
            best = cur_best.copy()
            best_score = cur_score
        generation += 1
        # Early stop if we already beat target bound below 0.3808
        if best_score <= 0.3808:
            break
    # Ensure final solution respects constraints (project one last time)
    best = project_onto_simplex(best, target)
    best_score = compute_c5(best)
    return (best, best_score, n)

import time
import numpy as np

def project_onto_simplex(v, target):
    """Project vector v onto the set {0<=x<=1, sum(x)=target}."""
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = np.sum(x)
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    """Return the overlap integral C₅ for a candidate vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def initialize_population(popsize, n, target, rng):
    """Create an initial feasible population."""
    pop = []
    if 'initial_h_values' in globals():
        try:
            iv = np.array(initial_h_values, dtype=float)
            if iv.shape[0] == n:
                iv = project_onto_simplex(iv, target)
                pop.append(iv)
        except Exception:
            pass
    while len(pop) < popsize:
        v = rng.random(n)
        v = project_onto_simplex(v, target)
        pop.append(v)
    return np.stack(pop, axis=0)

def tournament_select(pop, scores, size, rng):
    idx = rng.integers(0, len(pop), size=size)
    return pop[idx[np.argmin(scores[idx])]]

def crossover(p1, p2, prob, rng):
    if rng.random() < prob:
        point = rng.integers(1, len(p1))
        c1 = np.concatenate([p1[:point], p2[point:]])
        c2 = np.concatenate([p2[:point], p1[point:]])
        return c1, c2
    return p1.copy(), p2.copy()

def mutate(vec, rate, std, target, rng):
    mask = rng.random(vec.shape[0]) < rate
    if np.any(mask):
        vec[mask] += rng.normal(scale=std, size=mask.sum())
    return project_onto_simplex(vec, target)

def to_binary_half(h, target):
    """Round h to a vector with floor(target) ones, one fractional entry (if needed)."""
    n = len(h)
    base = int(np.floor(target))
    frac = target - base
    idx_desc = np.argsort(-h)  # descending order
    out = np.zeros(n, dtype=float)
    out[idx_desc[:base]] = 1.0
    if frac > 1e-12:
        out[idx_desc[base]] = frac
    return out

def binary_half_local_search(h, start, total_budget, rng):
    """Hill‑climb on vectors that consist of 0/1 entries and at most one fractional entry."""
    n = len(h)
    best_h = h.copy()
    best_score = compute_c5(best_h)
    current_h = best_h.copy()
    while time.time() - start < total_budget * 0.95:
        # locate fractional index (if any)
        frac_idxs = np.where((current_h > 1e-12) & (current_h < 1 - 1e-12))[0]
        frac_idx = frac_idxs[0] if frac_idxs.size else None

        new_h = current_h.copy()
        if frac_idx is not None and rng.random() < 0.5:
            # move the fractional mass to a zero position
            zero_idxs = np.where(new_h == 0)[0]
            if zero_idxs.size == 0:
                continue
            dst = rng.choice(zero_idxs)
            new_h[dst] = new_h[frac_idx]
            new_h[frac_idx] = 0.0
        else:
            # swap a 1 and a 0
            one_idxs = np.where(new_h == 1)[0]
            zero_idxs = np.where(new_h == 0)[0]
            if one_idxs.size == 0 or zero_idxs.size == 0:
                continue
            i = rng.choice(one_idxs)
            j = rng.choice(zero_idxs)
            new_h[i] = 0.0
            new_h[j] = 1.0

        new_score = compute_c5(new_h)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_h = new_h.copy()
            current_h = new_h
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # problem size
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
    else:
        n = 95                     # fallback (should never be used)
    target = n / 2.0

    # GA hyper‑parameters
    popsize = 300
    p_cross = 0.7
    mut_rate = 0.01
    mut_std = 0.1
    tourn_size = 3

    start = time.time()
    # population
    pop = initialize_population(popsize, n, target, rng)
    scores = np.array([compute_c5(ind) for ind in pop])
    best_idx = int(np.argmin(scores))
    best = pop[best_idx].copy()
    best_score = float(scores[best_idx])

    # ----- Genetic Algorithm -----
    while True:
        if time.time() - start > budget_s * 0.95:
            break
        # elitism
        new_pop = [best]
        while len(new_pop) < popsize:
            p1 = tournament_select(pop, scores, tourn_size, rng)
            p2 = tournament_select(pop, scores, tourn_size, rng)
            c1, c2 = crossover(p1, p2, p_cross, rng)
            c1 = mutate(c1, mut_rate, mut_std, target, rng)
            c2 = mutate(c2, mut_rate, mut_std, target, rng)
            new_pop.append(c1)
            if len(new_pop) < popsize:
                new_pop.append(c2)
        pop = np.stack(new_pop, axis=0)
        scores = np.array([compute_c5(ind) for ind in pop])
        cur_best_idx = int(np.argmin(scores))
        cur_best = pop[cur_best_idx]
        cur_score = float(scores[cur_best_idx])
        if cur_score < best_score:
            best = cur_best.copy()
            best_score = cur_score
        if best_score <= 0.3808:   # early exit if record beaten
            break

    # ----- Rounding & Local Refinement -----
    remaining = budget_s * 0.95 - (time.time() - start)
    if remaining > 0.5:
        # Round to a 0/1 vector with at most one fractional entry
        rounded = to_binary_half(best, target)
        rnd_score = compute_c5(rounded)
        if rnd_score < best_score:
            best = rounded
            best_score = rnd_score

        # Hill‑climb on this structured representation
        best, best_score = binary_half_local_search(best, start, budget_s, rng)

    # Final projection (insurance) and score recomputation
    best = project_onto_simplex(best, target)
    best_score = compute_c5(best)

    return (best, best_score, n)
