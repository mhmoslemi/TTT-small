import numpy as np
import time

def compute_c5_fast(h):
    # correlation of h and (1 - h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.shape[0])

def propose_move(h, max_step):
    n = h.shape[0]
    # Randomly select two distinct indices
    i, j = np.random.choice(n, size=2, replace=False)
    # Compute feasible bounds for delta
    lo = max(-h[i], -(1 - h[j]), -max_step)
    hi = min(1 - h[i], h[j], max_step)
    if lo > hi:
        return None, None, None
    delta = np.random.uniform(lo, hi)
    return i, j, delta

def random_feasible(n):
    target = n / 2.0
    int_part = int(np.floor(target))
    frac = target - int_part
    h = np.zeros(n, dtype=float)
    if int_part > 0:
        h[:int_part] = 1.0
    if frac > 0:
        h[int_part] = frac
    np.random.shuffle(h)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    # Determine initial problem size and initial solution
    if 'initial_h_values' in globals() and isinstance(initial_h_values, (list, np.ndarray)):

                # Distribute remaining difference uniformly
                diff = target_sum - h0.sum()
                if abs(diff) > 1e-12:

                    idx = np.random.choice(n_points)
                    h0[idx] = np.clip(h0[idx] + diff, 0.0, 1.0)
    else:
        n_points = 200

    # Initial current and best

    current_c5 = compute_c5_fast(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5

    # Simulated annealing parameters

    step_decay = 0.9999
    min_step = 1e-3

        if delta_obj <= 0 or np.random.rand() < np.exp(-delta_obj / max(T, 1e-12)):

            current_c5 = cand_c5

def compute_c5_fast(h):
    """Fast computation of C5 for a discretized step function."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / h.shape[0])

def propose_move(h, max_step):
    """Propose a feasible two‑index move that preserves the total sum."""
    n = h.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)
    # delta added to i, subtracted from j; keep both entries in [0,1]
    lo = max(-h[i], -(1.0 - h[j]), -max_step)
    hi = min(1.0 - h[i], h[j], max_step)
    if lo > hi:
        return None, None, None
    delta = np.random.uniform(lo, hi)
    return i, j, delta

def random_feasible(n):
    """Generate a feasible vector with entries in [0,1] and sum = n/2."""
    target = n / 2.0
    whole = int(np.floor(target))
    frac = target - whole
    h = np.zeros(n, dtype=float)
    if whole > 0:
        h[:whole] = 1.0
    if frac > 0:
        h[whole] = frac
    np.random.shuffle(h)
    return h
