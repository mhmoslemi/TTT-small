import time
import math
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] that minimizes the Erdős C5 overlap.

    Returns
    -------
    h_values : np.ndarray
        Array of length n_points with values in [0,1] and sum = n_points/2.
    c5_bound : float
        The achieved C5 bound (max overlap) for this h.
    n_points : int
        Number of discretization points (len(h_values)).
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----------------------------------------------------------------------
    # Determine problem size and initial vector
    # ----------------------------------------------------------------------
    if "initial_h_values" in globals():
        init_h = np.array(initial_h_values, dtype=float)
        n_points = init_h.size
    else:
        n_points = 70
        init_h = np.full(n_points, 0.5, dtype=float)

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # ----------------------------------------------------------------------
    # Helper functions
    # ----------------------------------------------------------------------
    def evaluate_c5(h):
        """Return C5 for a given h (already discretized)."""
        # correlation of h with (1-h) over full range
        val = np.max(np.correlate(h, 1.0 - h, mode="full"))
        return val * dx

    def project_feasible(h):
        """Clip to [0,1] and enforce sum = target_sum."""
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if s == 0:
            h[:] = target_sum / n_points
            return h
        # Scale to hit the target sum, then clip again
        h *= (target_sum / s)
        h = np.clip(h, 0.0, 1.0)
        # Tiny residual correction
        diff = target_sum - h.sum()
        if abs(diff) > 1e-12:
            # Distribute diff over entries that are not at a bound
            free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
            if free.size:
                h[free] += diff / free.size
            else:
                h += diff / n_points
            h = np.clip(h, 0.0, 1.0)
        return h

    # Ensure feasibility of the starting point
    init_h = project_feasible(init_h)

    # ----------------------------------------------------------------------
    # Simulated annealing / hill climbing
    # ----------------------------------------------------------------------
    current_h = init_h.copy()
    current_c5 = evaluate_c5(current_h)

    best_h = current_h.copy()
    best_c5 = current_c5

    # Parameters
    max_delta = 0.1          # initial max mass moved per step
    T0 = 5e-3                # initial temperature
    T_end = 1e-5             # final temperature
    prob_target_move = 0.2   # fraction of iterations using a targeted move
    iter_without_improve = 0
    last_improve_iter = 0
    iter_count = 0

    # Main loop
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.97:   # safety margin
            break

        # Linear cooling schedule
        frac = elapsed / (budget_s * 0.97)
        T = T0 * (1 - frac) + T_end * frac

        # Decide move type
        if rng.random() < prob_target_move:
            # Targeted move: focus on the lag with maximal overlap
            corr_full = np.correlate(current_h, 1.0 - current_h, mode="full")
            max_idx = int(np.argmax(corr_full))
            shift = max_idx - (n_points - 1)  # integer lag
            # Compute contributions for this lag
            if shift >= 0:
                idx = np.arange(0, n_points - shift)
                prod = current_h[idx] * (1.0 - current_h[idx + shift])
            else:
                idx = np.arange(-shift, n_points)
                prod = current_h[idx] * (1.0 - current_h[idx + shift])
            total = prod.sum()
            if total > 0:
                probs = prod / total
                i = rng.choice(idx, p=probs)
            else:
                i = rng.integers(0, n_points)
            # Choose a distinct receiver
            j = rng.integers(0, n_points - 1)
            if j >= i:
                j += 1
        else:
            # Random move
            i, j = rng.choice(n_points, size=2, replace=False)

        # Propose a transfer amount epsilon
        eps_raw = (rng.random() - 0.5) * 2.0 * max_delta
        if eps_raw > 0:
            eps = min(eps_raw, current_h[i], 1.0 - current_h[j])
        else:
            eps = -min(-eps_raw, current_h[j], 1.0 - current_h[i])
        if abs(eps) < 1e-12:
            continue

        # Apply move
        new_h = current_h.copy()
        new_h[i] -= eps
        new_h[j] += eps

        # No need to project; move respects bounds by construction
        new_c5 = evaluate_c5(new_h)

        # Acceptance criterion
        if new_c5 < current_c5:
            # Accept improvement
            current_h = new_h
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
                last_improve_iter = iter_count
        else:
            # Metropolis acceptance for uphill moves
            delta = new_c5 - current_c5
            if T > 0.0 and rng.random() < math.exp(-delta / T):
                current_h = new_h
                current_c5 = new_c5

        iter_count += 1

        # Adaptive step size logic
        if iter_count % 5000 == 0:

    best_c5 = evaluate_c5(best_h)
