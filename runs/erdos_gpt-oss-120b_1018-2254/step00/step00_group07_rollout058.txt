def project_onto_feasible(x, target, lower=0.0, upper=1.0, max_iter=1000):
    h = np.array(x, dtype=float)
    for _ in range(max_iter):
        h = np.clip(h, lower, upper)
        sum_h = h.sum()
        if abs(sum_h - target) < 1e-12:
            break
        # Need to adjust all non-saturated values?
        # Use water-filling: compute mask of interior (0< h < 1)
        mask = (h > lower + 1e-12) & (h < upper - 1e-12)
        if np.any(mask):
            # adjust only interior values
            interior_n = mask.sum()
            delta = (target - sum_h) / interior_n
            h[mask] += delta
        else:
            # cannot adjust because all are at bounds, break
            break
    # final clip again
    return np.clip(h, lower, upper)

def project_feasible(x, target, lower=0.0, upper=1.0, max_iter=1000):
    h = np.clip(x, lower, upper)
    for _ in range(max_iter):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # Which entries can be moved?
        # For diff>0 need to increase some entries that are not at upper bound
        if diff > 0:
            mask = h < upper - 1e-12
        else:
            mask = h > lower + 1e-12
        if not np.any(mask):
            break
        # distribute diff equally among mask entries
        delta = diff / mask.sum()
        h[mask] += delta
        # clip again
        h = np.clip(h, lower, upper)
    # Final clipping
    h = np.clip(h, lower, upper)
    # Adjust any tiny sum error
    s = h.sum()
    if abs(s - target) > 1e-9:
        # distribute residual to first element(s) within bounds
        for i in range(len(h)):
            if h[i] < upper and s < target:

                h[i] += add
                s += add
            elif h[i] > lower and s > target:
                sub = min(h[i] - lower, s - target)
                h[i] -= sub
                s -= sub
            if abs(s - target) < 1e-12:
                break
    return h

import time
import scipy.optimize as opt

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(x, target, lower=0.0, upper=1.0, max_iter=500):
    h = np.clip(x, lower, upper)
    for _ in range(max_iter):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < upper - 1e-12
        else:
            mask = h > lower + 1e-12
        if not np.any(mask):
            break
        delta = diff / mask.sum()
        h[mask] += delta
        h = np.clip(h, lower, upper)
    # final repair if needed
    s = h.sum()
    if abs(s - target) > 1e-9:
        # distribute residual greedily
        for i in range(len(h)):
            if s == target:
                break
            if s < target and h[i] < upper:
                add = min(upper - h[i], target - s)
                h[i] += add
                s += add
            elif s > target and h[i] > lower:
                sub = min(h[i] - lower, s - target)
                h[i] -= sub
                s -= sub
    return h

                       bounds=bounds, constraints=[cons],
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})

    dx = 2.0 / n

    return np.max(corr) * dx

def project_feasible(x, target, lower=0.0, upper=1.0, max_iter=500):
    h = np.clip(x, lower, upper)
    for _ in range(max_iter):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < upper - 1e-12
        else:
            mask = h > lower + 1e-12
        if not np.any(mask):
            break
        delta = diff / mask.sum()
        h[mask] += delta
        h = np.clip(h, lower, upper)
    # final adjustment
    s = h.sum()
    if abs(s - target) > 1e-9:
        for i in range(h.size):
            if s == target:
                break
            if s < target and h[i] < upper:
                add = min(upper - h[i], target - s)
                h[i] += add
                s += add
            elif s > target and h[i] > lower:
                sub = min(h[i] - lower, s - target)
                h[i] -= sub
                s -= sub
