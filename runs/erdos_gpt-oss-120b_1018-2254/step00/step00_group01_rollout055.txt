def project_sum_bounds(x, target):
    # Clip to [0,1]
    x = np.clip(x, 0.0, 1.0)
    # Simple iterative approach:
    # While sum != target within tolerance:
    # compute diff = target - sum(x)
    # If diff > 0: we need to increase some values that are not at 1. Distribute diff uniformly across those.
    # If diff < 0: we need to decrease some values not at 0.
    # This may overshoot if some values hit bound, then repeat.
    max_iter = 100
    for _ in range(max_iter):
        s = x.sum()
        diff = target - s
        if abs(diff) < 1e-9:
            break
        # Determine which indices can be adjusted (not at bound)
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        if not np.any(mask):
            break
        # Distribute diff equally among adjustable entries
        adjust = diff / np.count_nonzero(mask)
        x[mask] += adjust
        # Clip again
        x = np.clip(x, 0.0, 1.0)
    return x
```

def project_sum_bounds(v, target):
    lo = -1.0  # start low enough
    hi = np.max(v) + 1.0  # high enough
    for _ in range(50):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # adjust any small error
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        # distribute diff uniformly to non-extreme entries
        free = (x > 0) & (x < 1)
        if free.any():
            adjust = diff / free.sum()
            x[free] += adjust
            x = np.clip(x, 0.0, 1.0)
        else:
            # if all at extremes, adjust random small
            x[0] = np.clip(x[0] + diff, 0.0, 1.0)
    return x

import time
import numpy as np

def compute_c5(h, dx):
    """Return C5 = max_k ∫ h(x)*(1-h(x+k))dx approximated by discrete correlation."""
    y = 1.0 - h
    corr = np.correlate(h, y, mode='full')
    return np.max(corr) * dx

def project_sum_bounds(v, target):
    """Project v onto the box [0,1] with sum(v)=target (L2 minimum projection)."""
    lo = -1.0
    hi = np.max(v) + 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        free = (x > 0.0) & (x < 1.0)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            x[0] = np.clip(x[0] + diff, 0.0, 1.0)
    return x

def random_feasible(n, target):
    """Generate a random feasible vector of length n with sum(target)."""
    v = np.random.rand(n)
    return project_sum_bounds(v, target)
