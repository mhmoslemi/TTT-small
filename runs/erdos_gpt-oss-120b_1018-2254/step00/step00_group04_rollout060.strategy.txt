def repair_h(h, n_points):
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    h_clip = np.clip(h.copy(), 0.0, 1.0)  # Ensure boundaries
    total = h_clip.sum() * dx  # Evaluate discrete integral considering dx

    # Adjust sum while respecting boundaries
    while np.abs(total - 1.0) > 1e-6:
        if total < 1.0:
            scale_factor = (1.0 / total)**(1.0/len(h))
            h_clip *= scale_factor
        else:
            scale_factor = (1.0 / total)**(1.0/len(h))
            h_clip *= scale_factor
        total = (h_clip.copy()).sum() * dx  # Recheck

    # Further refinement to meet exact sum
    target = 1.0 / dx  # Exact grid sum (discrete)
    diff = (target - np.sum(h_clip)) / (n_points * dx)
    h_clip += diff * dx  # Apply small adjustment
    h_clip = np.clip(h_clip, 0.0, 1.0)
    return h_clip
