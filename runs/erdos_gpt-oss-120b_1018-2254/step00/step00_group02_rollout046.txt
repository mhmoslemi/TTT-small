    # enforce sum= n/2; ensure values in [0,1]
    n = len(h)
    target = n / 2.0
    # Clip
    h = np.clip(h, 0.0, 1.0)
    # Compute sum diff
    s = h.sum()
    if s == target:
        return h
    # Adjust uniformly within bounds...
    # To avoid exceeding bounds, we compute delta = target - s.
    # We'll distribute delta proportionally among entries not at bounds.
    # Simpler: add delta/n to each entry, then clip again, repeat until close.
    # Or we can just set first element to target - (s - h[0]) ensuring within bounds.
    # Since delta may be small if only slight deviation, we can modify first element within bounds.
    # Let's adjust h[0] by delta, ensuring after adjustment within [0,1].
    # If out of bounds, fallback to scaling whole vector.
    delta = target - s
    # Try adjusting h[0]
    new_val = h[0] + delta
    if 0 <= new_val <= 1:
        h[0] = new_val
        return h
    # else scale whole vector
    factor = target / s
    h = h * factor
    h = np.clip(h, 0, 1)
    # If still off, rescale again
    s2 = h.sum()
    if s2 != target:
        factor2 = target / s2
        h = h * factor2
        h = np.clip(h, 0, 1)
    return h

    bounds = [(0.0,1.0)]*n

def normalize(h):
    n = len(h)
    target = n/2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    # find index that can be adjusted without violating bounds
    for idx in range(n):
        # Try to add diff to h[idx]
        new_val = h[idx] + diff
        if 0 <= new_val <= 1:
            h[idx] = new_val
            return h
    # else scale whole vector
    factor = target / s if s != 0 else 0
    h = h * factor
    h = np.clip(h, 0.0, 1.0)
    # after scaling, adjust again if needed
    s2 = h.sum()
    diff2 = target - s2
    if abs(diff2) > 1e-12:
        # adjust first element if possible
        new_val = h[0] + diff2

import time
import numpy as np
from scipy.optimize import differential_evolution, minimize

# ---------- Helper Functions (top-level) ----------

def compute_c5(h):
    """Return the overlap constant C5 for a given step function array."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


def normalize(h):
    """
    Clip to [0,1] and enforce the exact sum constraint sum(h) = n/2.
    Modifies and returns the array.
    """
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    # Try to adjust a single coordinate without breaking bounds
    for idx in range(n):
        new_val = h[idx] + diff
        if 0.0 <= new_val <= 1.0:
            h[idx] = new_val
            return h
    # Fallback: scale whole vector then clip & fix residual
    if s > 0:
        factor = target / s
        h = h * factor
    h = np.clip(h, 0.0, 1.0)
    s2 = h.sum()
    diff2 = target - s2
    if abs(diff2) > 1e-12:
        # adjust first element if possible
        new_val = h[0] + diff2
        if 0.0 <= new_val <= 1.0:
            h[0] = new_val
        else:
            # final uniform scaling (should be very close already)
            if s2 > 0:
                h = h * (target / s2)
                h = np.clip(h, 0.0, 1.0)
    return h


def upscale(h, new_n):
    """Linear interpolation of a step function to a finer grid."""
    old_n = h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


def random_feasible(n):
    """Generate a random feasible vector of length n."""
    h = np.random.rand(n)
    target = n / 2.0
    if h.sum() == 0:
        h = np.full(n, 0.5)
    else:
        h = h / h.sum() * target
    h = np.clip(h, 0.0, 1.0)
    h = normalize(h)
    return h


class SumConstraint:
    """Equality constraint enforcing sum(x) == target."""
    def __init__(self, target):
        self.target = target

    def __call__(self, x):
        return np.sum(x) - self.target


def objective_de(x, n):
    """
    Objective for differential evolution: C5 with a penalty for violating sum constraint.
    """
    # penalty for sum deviation
    diff = np.sum(x) - n / 2.0
    penalty = 0.0
    if abs(diff) > 1e-4:
        penalty = 100.0 * abs(diff)
    # C5 term
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    c5 = dx * np.max(corr)
    return c5 + penalty


def hill_climb(initial_h, time_deadline, max_iters=2000, step_size=0.05):
    """
    Simple hill-climbing with random mass transfers that preserve the total sum.
    Returns the best vector found and its C5.
    """
    n = initial_h.shape[0]
    best_h = initial_h.copy()
    best_val = compute_c5(best_h)
    cur_h = best_h.copy()
    no_improve = 0

    while time.time() < time_deadline and max_iters > 0:
        max_iters -= 1
        i, j = np.random.choice(n, 2, replace=False)
        if np.random.rand() < 0.5:
            max_move = min(step_size, cur_h[i], 1.0 - cur_h[j])
            if max_move <= 0.0:
                continue
            delta = np.random.rand() * max_move
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        else:
            max_move = min(step_size, cur_h[j], 1.0 - cur_h[i])
            if max_move <= 0.0:
                continue
            delta = np.random.rand() * max_move
            new_h = cur_h.copy()
            new_h[j] -= delta
            new_h[i] += delta

        val = compute_c5(new_h)
        if val + 1e-12 < best_val:
            best_val = val
            best_h = new_h.copy()
            cur_h = new_h
            no_improve = 0
        else:
            no_improve += 1
            # occasional random restart to escape stagnation
            if no_improve >= 5000:
                cur_h = random_feasible(n)
                no_improve = 0
    return best_h, best_val


# ---------- Main Entry Point ----------

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.98  # slight safety margin

    # ----- Initial solution -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.shape[0]
        h0 = normalize(h0)
    else:
        n0 = 45
        h0 = random_feasible(n0)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # List of target resolutions (start, then refined)
    n_list = [n0, n0 * 2, n0 * 4]

    for n in n_list:
        if time.time() >= deadline:
            break

        # ----- Prepare initial vector for this resolution -----
        if n == best_h.shape[0]:
            init_h = best_h.copy()
        else:
            init_h = upscale(best_h, n)
            init_h = normalize(init_h)

        # ----- Global search via Differential Evolution -----
        bounds = [(0.0, 1.0)] * n
        try:
            de_res = differential_evolution(
                func=objective_de,
                bounds=bounds,
                args=(n,),
                strategy="best1bin",
                maxiter=20,
                popsize=10,
                tol=1e-3,
                mutation=(0.5, 1.0),
                recombination=0.7,
                seed=seed,
                init="latinhypercube",
                polishing=False,
                updating="deferred",
            )
            de_candidate = normalize(de_res.x)
            de_val = compute_c5(de_candidate)
            if de_val < best_c5:
                best_c5 = de_val
                best_h = de_candidate.copy()
        except Exception:
            de_candidate = None  # fall back to init_h later

        # ----- Local refinement with SLSQP -----
        target_sum = n / 2.0
        sum_cons = SumConstraint(target_sum)
        refine_start = de_candidate if de_candidate is not None else init_h
        try:
            slsqp_res = minimize(
                fun=compute_c5,
                x0=refine_start,
                method="SLSQP",
                bounds=bounds,
                constraints=[{"type": "eq", "fun": sum_cons}],
                options={"maxiter": 200, "ftol": 1e-9},
            )
            if slsqp_res.success:
                refined = normalize(slsqp_res.x)
                refined_val = compute_c5(refined)
                if refined_val < best_c5:
                    best_c5 = refined_val
                    best_h = refined.copy()
        except Exception:
            pass

        # ----- Additional hill climbing phase -----
        # allocate remaining time proportionally
        remaining = deadline - time.time()
        # aim for at most 0.6 * remaining seconds in hill climbing
        hc_deadline = time.time() + min(remaining * 0.6, 30.0)  # cap to 30 s per level
        hc_h, hc_val = hill_climb(best_h, hc_deadline, max_iters=5000, step_size=0.05)
        if hc_val < best_c5:
            best_c5 = hc_val
            best_h = hc_h.copy()

    # Final normalization and evaluation
    best_h = normalize(best_h)
    best_c5 = compute_c5(best_h)
    return best_h.tolist(), float(best_c5), best_h.shape[0]
