def hill_climb(h, best_c5, time_limit):
    n = len(h)
    target = n/2.0
    start = time.time()
    # iterate while time left.
    while time.time() - start < time_limit:
        # propose random pair swap: choose i,j distinct
        i,j = np.random.choice(n,2,replace=False)
        # propose delta up to max allowable
        max_up = min(1 - h[i], h[j]) # amount we can transfer from j to i
        max_down = min(h[i], 1 - h[j]) # amount we can transfer from i to j (but we pick direction randomly)
        # choose direction randomly
        if np.random.rand() < 0.5:
            max_delta = max_up
        else:
            max_delta = -max_down
        # If max_delta is zero continue
        if max_delta == 0:
            continue
        # Choose delta uniformly between 0 and max_delta (if positive) or max_delta to 0 (if negative)
        if max_delta > 0:
            delta = np.random.rand() * max_delta
        else:
            delta = -np.random.rand() * (-max_delta) # delta negative
        # apply
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
    return h,best_c5

Implementation of project_onto_constraints:

def project_to_constraints(x, target):
    import cvxpy as cp
    n = x.shape[0]
    var = cp.Variable(n)
    obj = cp.Minimize(cp.sum_squares(var - x))
    constraints = [cp.sum(var) == target, var >= 0, var <= 1]
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.ECOS, warm_start=False, verbose=False)
    if var.value is None:
        # fallback to simple scaling and clipping
        y = np.clip(x, 0, 1)
        s = y.sum()
        if s == 0:
            y = np.full_like(x, target / n)
        else:
            y *= target / s
            y = np.clip(y, 0, 1)
        return y
    return np.array(var.value).flatten()

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    # The correlate returns array of length 2*n-1 where index shift = -(n-1) .. (n-1)
    # Max correlation (largest overlap) is max of corr.
    return (2.0 / n) * np.max(corr)

import time
import numpy as np
import cvxpy as cp

def compute_c5(h):
    """Return the overlap constant C5 for a step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return (2.0 / n) * np.max(corr)


def project_to_constraints(x, target):
    """Project vector x onto {0≤y≤1, sum(y)=target} using a tiny QP."""
    n = x.shape[0]
    y = cp.Variable(n)
    obj = cp.Minimize(cp.sum_squares(y - x))
    cons = [cp.sum(y) == target, y >= 0, y <= 1]
    prob = cp.Problem(obj, cons)
    try:
        prob.solve(solver=cp.ECOS, warm_start=False, verbose=False)
        if y.value is not None:
            return np.array(y.value, dtype=float)
    except Exception:
        pass
    # fallback: simple scaling and clipping
    y0 = np.clip(x, 0.0, 1.0)
    s = y0.sum()
    if s == 0:
        y0[:] = target / n
    else:
        y0 *= target / s
    y0 = np.clip(y0, 0.0, 1.0)
    return y0


def interpolate_h(src, n_target):
    """Linearly interpolate a source h to a new number of points."""
    src = np.asarray(src, dtype=float)
    n_src = src.shape[0]
    if n_src == n_target:
        return src.copy()
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_tgt = np.linspace(0.0, 2.0, n_target, endpoint=False)
    return np.interp(x_tgt, x_src, src)


def random_dirichlet_candidate(n, target, alpha):
    """Generate a random feasible vector using a Dirichlet distribution."""
    vec = np.random.dirichlet(np.full(n, alpha))
    cand = vec * target
    # Small chance of slight violation due to rounding; clip and renormalise
    if np.any(cand > 1.0):
        cand = np.clip(cand, 0.0, 1.0)
        s = cand.sum()
        if s != 0:
            cand *= target / s
        cand = np.clip(cand, 0.0, 1.0)
    return cand


def hill_climb(best_h, best_c5, time_allowed):
    """Simple pairwise transfer hill climbing on the feasible set."""
    n = best_h.shape[0]
    target = n / 2.0
    h = best_h.copy()
    c5 = best_c5
    end_time = time.time() + time_allowed
    while time.time() < end_time:
        i, j = np.random.choice(n, 2, replace=False)
        # amount we may move from j to i (positive delta)
        max_up = min(1.0 - h[i], h[j])
        # amount we may move from i to j (negative delta)
        max_down = min(h[i], 1.0 - h[j])
        # choose direction
        if np.random.rand() < 0.5:
            max_delta = max_up
        else:
            max_delta = -max_down
        if max_delta == 0.0:
            continue
        if max_delta > 0.0:
            delta = np.random.rand() * max_delta
        else:
            delta = -np.random.rand() * (-max_delta)
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
    return h, c5
