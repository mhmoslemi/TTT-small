import time
import math
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return the overlap integral C5 for a given step function."""
    n = h.shape[0]
    dx = 2.0 / n
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _project_to_feasible(z: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project vector z onto the set {x | 0 <= x_i <= 1, sum x_i = target_sum}.
    Uses a bisection on the Lagrange multiplier lambda.
    """
    lo = np.min(z) - 1.0          # ensures all entries become 1 after subtraction
    hi = np.max(z)                # ensures all entries become 0 after subtraction
    for _ in range(60):           # enough for double precision
        mid = (lo + hi) * 0.5
        clipped = np.clip(z - mid, 0.0, 1.0)
        s = clipped.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(z - (lo + hi) * 0.5, 0.0, 1.0)

def random_feasible_h(n: int, target_sum: float) -> np.ndarray:
    """Generate a random feasible h-vector (box bounds and fixed sum)."""
    z = np.random.rand(n)
    return _project_to_feasible(z, target_sum)

    else:

        if abs(best_h.sum() - target_sum) > 1e-12:

    best_c5 = compute_c5(best_h)

    cur_h = best_h.copy()
    cur_c5 = best_c5
