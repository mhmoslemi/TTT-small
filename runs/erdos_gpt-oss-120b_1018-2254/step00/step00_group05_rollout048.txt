def project_sum(h, target):
    # Simple scaling then clip then adjust diff
    # Ensure sum exactly target, maybe with small epsilon.
    total = np.sum(h)
    if total == 0:
        # distribute uniformly
        h[:] = target / len(h)
        return h
    h = h * (target / total)
    # Clip and fix sum
    np.clip(h, 0, 1, out=h)
    diff = target - np.sum(h)
    # Add diff to first element not at bounds perhaps
    # If diff >0, add to first <1 element; if diff<0 subtract from first >0.
    for idx in range(len(h)):
        if diff > 1e-12 and h[idx] < 1:
            add = min(diff, 1 - h[idx])
            h[idx] += add
            diff -= add

            sub = min(-diff, h[idx])
            h[idx] -= sub
            diff += sub
        if abs(diff) < 1e-12:
            break
    return h

import time, numpy as np, math

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # np.correlate uses convolution with reversed second argument; for our purpose it's as required.
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(h, target):
    # Scale to target sum then clip and adjust diff.
    total = np.sum(h)
    if total != 0:
        h = h * (target / total)
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # distribute diff
        for idx in range(h.size):
            if diff > 0 and h[idx] < 1.0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            elif diff < 0 and h[idx] > 0.0:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _c5_value(h: np.ndarray) -> float:
    """Compute C5 for a given discrete h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project_feasible(h: np.ndarray, target: float) -> np.ndarray:
    """Scale/clip h so that 0≤h≤1 and sum(h)=target (exactly)."""
    total = np.sum(h)
    if total != 0.0:
        h = h * (target / total)
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # distribute the remaining mass greedily
        for i in range(h.size):
            if diff > 0.0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0.0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h
