import numpy as np
import time
import math

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def simulated_anneal(h_init, budget):
    import numpy as np, time, math
    start_local = time.time()
    n = len(h_init)
    cur = h_init.copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    T0 = 0.5
    rng = np.random.default_rng()
    while True:
        elapsed = time.time() - start_local
        if elapsed >= budget:
            break
        # linear cooling
        T = T0 * (1.0 - elapsed / budget)
        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # direction random
        if rng.random() < 0.5:
            delta_max = min(cur[i], 1.0 - cur[j])
            if delta_max <= 0:
                continue
            delta = rng.random() * delta_max * (T / T0 + 0.01)
        else:
            delta_max = min(cur[j], 1.0 - cur[i])
            if delta_max <= 0:
                continue
            delta = rng.random() * delta_max * (T / T0 + 0.01)
            i, j = j, i
        if delta <= 1e-15:
            continue
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_c5 = compute_c5(cand)
        # Acceptance criterion
        if cand_c5 < cur_c5 or rng.random() < math.exp(-(cand_c5 - cur_c5) / (T + 1e-12)):
            cur = cand
            cur_c5 = cand_c5
            if cand_c5 < best_c5:
                best = cand.copy()
                best_c5 = cand_c5
    return best, best_c5

    rng = np.random.default_rng(seed)
    # Get initial vector if available

    target_sum = n / 2.0
    # Ensure feasibility: clip and adjust sum
    h0 = np.clip(h0, 0.0, 1.0)
    cur_sum = h0.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        # Simple fix: create binary vector with exact sum
        h0 = np.zeros(n, dtype=float)
        half = int(round(target_sum))
        h0[:half] = 1.0
        rng.shuffle(h0)
    # First annealing
    elapsed = time.time() - start
    remaining = max(0.0, budget_s - elapsed)
    if remaining > 0.1:
        best_h, best_c5 = simulated_anneal(h0, remaining * 0.7)  # allocate 70% time
    else:
        best_h = h0
        best_c5 = compute_c5(best_h)
    # Optional second run with random start if time left

    remaining = max(0.0, budget_s - elapsed)

import numpy as np
import time
import math

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx using discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def simulated_anneal(h_init, budget):
    """Simple SA that preserves sum and box constraints; returns best vector."""
    start_local = time.time()
    n = len(h_init)
    cur = h_init.copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    T0 = 0.5                       # initial temperature
    rng = np.random.default_rng()
    while True:
        elapsed = time.time() - start_local
        if elapsed >= budget:
            break
        # linear cooling
        T = T0 * (1.0 - elapsed / budget)
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # decide direction of transfer
        if rng.random() < 0.5:
            delta_max = min(cur[i], 1.0 - cur[j])
            if delta_max <= 0:
                continue
            delta = rng.random() * delta_max * (T / T0 + 0.01)
        else:
            delta_max = min(cur[j], 1.0 - cur[i])
            if delta_max <= 0:
                continue
            delta = rng.random() * delta_max * (T / T0 + 0.01)
            i, j = j, i
        if delta <= 1e-15:
            continue
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_c5 = compute_c5(cand)
        # Metropolis acceptance
        if cand_c5 < cur_c5 or rng.random() < math.exp(-(cand_c5 - cur_c5) / (T + 1e-12)):
            cur = cand
            cur_c5 = cand_c5
            if cand_c5 < best_c5:
                best = cand.copy()
                best_c5 = cand_c5
    return best, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) respecting all constraints."""
    start = time.time()
    rng = np.random.default_rng(seed)

    # Obtain initial construction if provided
    if 'initial_h_values' in globals():

    else:
        n_default = 70
        h0 = np.full(n_default, 0.5, dtype=float)

    n = len(h0)
    target_sum = n / 2.0
    h0 = np.clip(h0, 0.0, 1.0)

    # Ensure exact sum if needed (fallback to binary vector)
    if abs(h0.sum() - target_sum) > 1e-9:
        h0 = np.zeros(n, dtype=float)
        half = int(round(target_sum))
        h0[:half] = 1.0
        rng.shuffle(h0)

    # First annealing phase (≈70% of remaining budget)
    elapsed = time.time() - start
    remaining = max(0.0, budget_s - elapsed)
    if remaining > 0.5:
        phase_budget = remaining * 0.7
        best_h, best_c5 = simulated_anneal(h0, phase_budget)
    else:
        best_h, best_c5 = h0, compute_c5(h0)

    # Optional second phase with a random binary start if time left
    elapsed = time.time() - start
    remaining = max(0.0, budget_s - elapsed)
    if remaining > 0.3:
        h_rand = np.zeros(n, dtype=float)
        half = int(round(target_sum))
        h_rand[:half] = 1.0
        rng.shuffle(h_rand)

        if cand_c5 < best_c5:
