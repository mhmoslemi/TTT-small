import time
import numpy as np

def _compute_c5_and_corr(h):
    """Return (C5 value, full correlation array) for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = np.max(corr) * dx
    return c5, corr

def _gradient_for_shift(h, shift):
    """Subgradient of max_k C(k) w.r.t. h for the given integer shift."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if shift >= 0:
        m = n - shift
        # contribution from (1 - h[i+shift])
        grad[:m] = 1.0 - h[shift:]
        # contribution from -h[i-shift]
        grad[shift:] -= h[:m]
    else:
        s = -shift
        m = n - s
        grad[s:] = 1.0 - h[:m]
        grad[:m] -= h[s:]
    return grad

def _project_box_and_sum(v, target):
    """
    Euclidean projection of v onto the box [0,1]^n with the linear equality sum=v_target.
    Uses a bisection on the Lagrange multiplier.
    """
    low, high = -2.0, 2.0
    for _ in range(50):
        mid = (low + high) * 0.5
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] (discretised) to minimise the
    Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Initialise h
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        # fallback: uniform half‑mass
        n_default = 200
        h = np.full(n_default, 0.5, dtype=np.float64)

    n = h.shape[0]
    target_sum = n / 2.0
    h = _project_box_and_sum(h, target_sum)

    # keep the best seen solution
    best_h = h.copy()
    best_c5, _ = _compute_c5_and_corr(best_h)

    # ------------------------------------------------------------------
    # Gradient‑descent phase (subgradient on the active shift)
    # ------------------------------------------------------------------
    gd_time = min(0.3 * budget_s, 300.0)   # at most 30% of the budget or 5 min
    gd_deadline = start_time + gd_time
    step = 0.5

    while time.time() < gd_deadline:
        c5, corr = _compute_c5_and_corr(h)
        # record best
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        shift = int(np.argmax(corr) - (n - 1))
        grad = _gradient_for_shift(h, shift) * (2.0 / n)   # include Δx

        improved = False
        for factor in (1.0, 0.5, 0.2):
            cand = _project_box_and_sum(h - step * factor * grad, target_sum)
            cand_c5, _ = _compute_c5_and_corr(cand)
            if cand_c5 < c5:
                h = cand
                step *= (1.0 + 0.1 * factor)   # modest step increase
                improved = True
                break
        if not improved:
            step *= 0.5
            if step < 1e-6:
                break

    # final check after GD
    c5, _ = _compute_c5_and_corr(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # ------------------------------------------------------------------
    # Simulated‑annealing / random pairwise moves to fine‑tune
    # ------------------------------------------------------------------
    cur_h = best_h.copy()
    cur_c5, _ = _compute_c5_and_corr(cur_h)

    temperature = 0.02
    max_transfer = 0.1          # max mass moved between two points per trial

    while True:
        now = time.time()
        if now - start_time > budget_s - 0.2:
            break

        # propose a mass transfer between two distinct coordinates
        i, j = np.random.choice(n, 2, replace=False)

        # two possible directions: i receives from j, or i gives to j
        if np.random.rand() < 0.5:
            # transfer from j -> i
            delta_max = min(1.0 - cur_h[i], cur_h[j], max_transfer)
            if delta_max <= 0:
                continue
            delta = np.random.rand() * delta_max
        else:
            # transfer from i -> j
            delta_max = min(cur_h[i], 1.0 - cur_h[j], max_transfer)
            if delta_max <= 0:
                continue
            delta = -np.random.rand() * delta_max

        cand = cur_h.copy()
        cand[i] += delta
        cand[j] -= delta   # sum unchanged, bounds respected by construction

        cand_c5, _ = _compute_c5_and_corr(cand)
        diff = cand_c5 - cur_c5

        if diff < 0 or np.random.rand() < np.exp(-diff / max(temperature, 1e-12)):
            cur_h = cand
            cur_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = cand.copy()

        # annealing schedule
        temperature *= 0.9995
        max_transfer *= 0.9999
        if max_transfer < 1e-5:
            max_transfer = 1e-5

    # ------------------------------------------------------------------
    # Return the best discovered construction
    # ------------------------------------------------------------------
    h_values = best_h.tolist()
    c5_bound = float(best_c5)
    return (h_values, c5_bound, n)
