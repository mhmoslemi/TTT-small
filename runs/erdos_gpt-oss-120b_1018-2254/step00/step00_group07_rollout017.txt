def project_simple(x, target, lower=0.0, upper=1.0, max_iter=10):
    for _ in range(max_iter):
        x = np.clip(x, lower, upper)
        s = x.sum()
        if np.isclose(s, target, atol=1e-8):
            break
        # Avoid division by zero
        if s == 0:
            x += target / len(x)
        else:
            x *= target / s
    # final clip
    x = np.clip(x, lower, upper)
    # final adjust maybe small residual by adding to first non-bound entry
    s = x.sum()
    diff = target - s
    if abs(diff) > 1e-8:
        # adjust small amount equally among interior entries (if any)
        interior = (x > lower) & (x < upper)
        cnt = interior.sum()
        if cnt > 0:
            x[interior] += diff / cnt
        else:
            # fallback: assign diff to first entry within bounds if possible
            for i in range(len(x)):
                if (x[i] + diff) >= lower and (x[i] + diff) <= upper:
                    x[i] += diff
                    break
    return x

    np.random.seed(seed)
    start_time = time.time()

    # Acquire initial h if available
    try:
        init_h = np.array(initial_h_values, dtype=float)
        n = init_h.size
    except NameError:
        # default size
        n = 80
        init_h = np.full(n, 0.5)  # sum = n/2
    target_sum = n/2.0
    dx = 2.0 / n

    # Ensure init_h feasible
    if init_h.sum() != target_sum:
        init_h = project_simple(init_h, target_sum)

    best_h = init_h.copy()
    best_c5 = compute_c5(best_h, dx)

    def obj(x):
        return compute_c5(x, dx)

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},)

def project_simple(x, target, lower=0.0, upper=1.0, max_iter=10):
    x = np.array(x, dtype=float)
    for _ in range(max_iter):
        np.clip(x, lower, upper, out=x)
        s = x.sum()
        if np.isclose(s, target, atol=1e-10):
            break
        if s == 0:
            # distribute equally
            x += target / len(x)
        else:
            x *= target / s
    # final clip
    np.clip(x, lower, upper, out=x)
    # fix small residual difference
    s = x.sum()
    diff = target - s
    if abs(diff) > 1e-8:
        interior = (x > lower) & (x < upper)
        cnt = interior.sum()
        if cnt > 0:
            x[interior] += diff / cnt
            # clip again
            np.clip(x, lower, upper, out=x)

            # try adjusting any element within bounds.

                new_val = x[i] + diff
                if lower <= new_val <= upper:
                    x[i] = new_val
                    break
    return x

def hill_climb(h_initial, best_val, target, dx, time_limit, start_time):

    h = h_initial.copy()
    val = best_val
    n = len(h)
    # parameters
    eps = 0.2  # magnitude of delta

import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h, dx):
    """Return the max overlap integral for step function values h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_simple(x, target, lower=0.0, upper=1.0, max_iter=10):
    """Project vector x onto the box [lower,upper] and the hyperplane sum(x)=target."""
    x = np.asarray(x, dtype=float)
    for _ in range(max_iter):
        np.clip(x, lower, upper, out=x)
        s = x.sum()
        if np.isclose(s, target, atol=1e-12):
            break
        if s == 0:
            # distribute evenly if zero
            x += target / x.size
        else:
            x *= target / s
    # final clip and tiny correction
    np.clip(x, lower, upper, out=x)
    s = x.sum()
    diff = target - s
    if abs(diff) > 1e-10:
        interior = (x > lower) & (x < upper)
        cnt = interior.sum()
        if cnt > 0:
            x[interior] += diff / cnt
        else:
            # fallback: adjust first feasible entry
            for i in range(x.size):
                new = x[i] + diff
                if lower <= new <= upper:
                    x[i] = new
                    break
        np.clip(x, lower, upper, out=x)
    return x

def hill_climb(start_h, start_val, target, dx, end_time, start_time):
    """Simple hill‑climbing with pairwise mass transfers."""
    h = start_h.copy()
    val = start_val
    n = h.size
    eps = 0.2  # maximal transfer magnitude
    while True:
        if time.time() - start_time >= end_time - start_time:
            break
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        delta = (np.random.rand() - 0.5) * 2.0 * eps
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        h_new = project_simple(h_new, target)
        new_val = compute_c5(h_new, dx)
        if new_val < val:
            h, val = h_new, new_val
    return h, val
