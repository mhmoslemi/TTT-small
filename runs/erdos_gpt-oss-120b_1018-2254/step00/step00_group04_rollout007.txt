from scipy.optimize import minimize

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def gradient_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1 - h, mode='full')
    idx_max = np.argmax(corr)
    shift = idx_max - (N - 1)
    grad = np.zeros_like(h)
    if shift >= 0:
        grad[:N - shift] += (1 - h[shift:])
        grad[shift:] -= h[:N - shift]
    else:
        m = -shift
        grad[m:] += (1 - h[:N - m])
        grad[:N - m] -= h[m:]
    return grad * dx, shift, np.max(corr) * dx

def project_to_constraints(v, target):
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = np.sum(x)
        if s < target:
            lo = mid
        else:

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def gradient_c5(h):
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1 - h, mode='full')
    idx_max = np.argmax(corr)
    shift = idx_max - (N - 1)
    grad = np.zeros_like(h)
    if shift >= 0:
        # part where i+shift within bounds
        grad[:N - shift] += (1 - h[shift:])
        # part where i-shift within bounds
        grad[shift:] -= h[:N - shift]
    else:
        m = -shift
        grad[m:] += (1 - h[:N - m])
        grad[:N - m] -= h[m:]
    grad *= dx
    c5_val = corr[idx_max] * dx
    return grad, shift, c5_val

def project_to_constraints(v, target_sum):
    # binary search for lambda to satisfy sum constraint after clipping
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = np.sum(x)
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    return x

def obj_fun(h, target_sum, lam):
    grad, shift, c5_val = gradient_c5(h)
    sum_h = np.sum(h)
    penalty = lam * (sum_h - target_sum) ** 2
    return c5_val + penalty

def obj_jac(h, target_sum, lam):
    grad, shift, c5_val = gradient_c5(h)
    sum_h = np.sum(h)
    grad_pen = 2.0 * lam * (sum_h - target_sum) * np.ones_like(h)
    return grad + grad_pen

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on a uniform grid."""
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def gradient_c5(h):
    """Gradient of C5 (without any penalty) w.r.t. h for the shift attaining the max."""
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx_max = np.argmax(corr)
    shift = idx_max - (N - 1)        # lag corresponding to the max
    grad = np.zeros_like(h)

    if shift >= 0:
        # i+shift term exists for i=0..N-shift-1
        grad[:N - shift] += (1.0 - h[shift:])
        # i-shift term exists for i=shift..N-1
        grad[shift:] -= h[:N - shift]
    else:
        m = -shift
        # i+shift = i-m term exists for i=m..N-1
        grad[m:] += (1.0 - h[:N - m])
        # i-shift = i+m term exists for i=0..N-m-1
        grad[:N - m] -= h[m:]

    grad *= dx
    c5_val = corr[idx_max] * dx
    return grad, shift, c5_val

def project_to_constraints(v, target):
    """Project vector v onto the intersection {0≤x≤1, sum(x)=target}."""
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = np.sum(x)
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

def obj_fun(h, target, lam):
    """Scalar objective: C5 + penalty for violating the sum constraint."""
    _, _, c5_val = gradient_c5(h)
    penalty = lam * (np.sum(h) - target) ** 2
    return c5_val + penalty

def obj_jac(h, target, lam):
    """Gradient of the scalar objective."""
    grad, _, _ = gradient_c5(h)
    sum_h = np.sum(h)
    grad_pen = 2.0 * lam * (sum_h - target) * np.ones_like(h)
    return grad + grad_pen

def run(seed=42, budget_s=1100, **kwargs):
    """Optimize a step function for the Erdős C₅ constant."""
    np.random.seed(seed)
    start = time.time()

    # number of samples (even, to keep the integral exactly 1 after projection)
    N = 200
    target_sum = N / 2.0

    # ------------------------------------------------------------------
    # initialise h
    init = globals().get('initial_h_values')
    if init is not None:
        h0_raw = np.array(init, dtype=np.float64)
        if h0_raw.size != N:
            x_old = np.linspace(0.0, 2.0, h0_raw.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, N, endpoint=False)
            h0_raw = np.interp(x_new, x_old, h0_raw)
        h0 = project_to_constraints(h0_raw, target_sum)
    else:
        h0 = project_to_constraints(np.random.rand(N), target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # ------------------------------------------------------------------
    # gradient‑based refinement (L‑BFGS‑B with a quadratic sum penalty)
    lam = 1e4                      # penalty weight
    bounds = [(0.0, 1.0)] * N
    max_restarts = 10

    for r in range(max_restarts):
        if time.time() - start > 0.9 * budget_s:
            break
        x0 = best_h.copy() if r == 0 else project_to_constraints(np.random.rand(N), target_sum)

        res = minimize(fun=obj_fun,

        h_opt = project_to_constraints(res.x, target_sum)
