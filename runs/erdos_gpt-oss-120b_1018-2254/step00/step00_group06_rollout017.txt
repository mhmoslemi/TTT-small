def project_to_feasible(v, target_sum):
    # v is real vector
    # need to find lambda such that sum(clip(v+lambda, 0, 1)) = target_sum
    # because sum is monotonic in lambda.
    # Use binary search.
    low = -np.max(v)  # ensure all v+low <=0 => all clipped to 0
    high = 1 - np.min(v)  # ensure all v+high >=1 => all clipped to 1
    # maybe broaden bounds a bit to guarantee solution
    for _ in range(60):
        mid = (low + high) / 2.0
        w = v + mid
        np.clip(w, 0.0, 1.0, out=w)
        s = w.sum()
        if s < target_sum:
            low = mid
        else:
            high = mid
    w = v + (low+high)/2.0

    # due to numeric tolerance, some small difference may remain; fix by adjusting a few entries

import numpy as np, time, random

def project_to_feasible(v, target):
    # binary search for lambda s.t. sum(clip(v+lambda,0,1)) = target
    low = -np.max(v)               # all clipped to 0
    high = 1.0 - np.min(v)         # all clipped to 1
    for _ in range(50):
        mid = (low + high) / 2.0
        w = v + mid
        np.clip(w, 0.0, 1.0, out=w)
        s = w.sum()
        if s < target:
            low = mid
        else:
            high = mid
    w = v + (low + high) / 2.0
    np.clip(w, 0.0, 1.0, out=w)
    # small correction
    diff = target - w.sum()
    if abs(diff) > 1e-9:
        # distribute diff to non-bound entries
        free = (w > 0) & (w < 1)
        if free.any():
            add = diff / free.sum()
            w[free] += add
            np.clip(w, 0.0, 1.0, out=w)
    return w

def compute_fitness(h, n_points):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n_points)

def tournament_select(pop, fit, k):
    idx = np.random.choice(len(pop), k, replace=False)
    best = idx[0]
    best_fit = fit[best]
    for i in idx[1:]:
        if fit[i] < best_fit:
            best = i
            best_fit = fit[i]
    return pop[best]

def mutate(ind, mut_prob, mut_std):
    mask = np.random.rand(ind.size) < mut_prob
    if mask.any():
        ind = ind.copy()
        ind[mask] += np.random.randn(mask.sum()) * mut_std
    return ind

def random_feasible(n_points, target):
    v = np.random.rand(n_points)
    return project_to_feasible(v, target)

        target = n / 2.0
        init = random_feasible(n, target)
    # GA parameters

    max_gen = 10000  # effectively limited by time
    tournament_k = 3
    crossover_p = 0.7
    mut_prob = 0.02
    mut_std = 0.1
    # initialize population
    pop = [init]
    for _ in range(pop_size - 1):
        pop.append(random_feasible(n, target))
    fitness = [compute_fitness(ind, n) for ind in pop]
    best_idx = int(np.argmin(fitness))
    best_h = pop[best_idx].copy()
    best_fit = fitness[best_idx]
    start = time.time()
    gen = 0
    while time.time() - start < budget_s and gen < max_gen:
        gen += 1
        # build next generation
        new_pop = []
        # elitism: keep best
        elite = best_h.copy()
        # produce children

            p1 = tournament_select(pop, fitness, tournament_k)

import numpy as np
import time
import random

def _project_to_feasible(v, target):
    """Project vector v onto the box [0,1]^n with sum(v)=target."""
    low = -np.max(v)          # all entries would become 0
    high = 1.0 - np.min(v)    # all entries would become 1
    for _ in range(50):
        mid = (low + high) / 2.0
        w = v + mid
        np.clip(w, 0.0, 1.0, out=w)
        s = w.sum()
        if s < target:
            low = mid
        else:
            high = mid
    w = v + (low + high) / 2.0
    np.clip(w, 0.0, 1.0, out=w)
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        free = np.where((w > 0) & (w < 1))[0]
        if free.size == 0:
            free = np.arange(w.size)
        i = free[0]
        w[i] += diff
        w[i] = max(0.0, min(1.0, w[i]))
    return w

def _fitness(h, n_points):
    """Return C5 bound for step function h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n_points)

def _tournament_select(pop, fits, k):
    idx = np.random.choice(len(pop), k, replace=False)
    best = idx[0]
    best_fit = fits[best]
    for i in idx[1:]:
        if fits[i] < best_fit:
            best = i
            best_fit = fits[i]
    return pop[best]

def _mutate(ind, prob, std):
    mask = np.random.rand(ind.size) < prob
    if mask.any():
        ind = ind.copy()
        ind[mask] += np.random.randn(mask.sum()) * std
    return ind

def _random_feasible(n, target):
    v = np.random.rand(n)
    return _project_to_feasible(v, target)
