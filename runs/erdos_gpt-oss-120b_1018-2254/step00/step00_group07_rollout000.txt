def compute_c5(h):
    """Return the overlap bound C5 for a step function sample h."""
    n = h.shape[0]

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

        np.clip(a, 0.0, 1.0, out=a)

    return a

            max_inc = 1.0 - current_h[i]
            max_dec = current_h[j]
            max_delta = min(max_inc, max_dec, delta_max)
            if max_delta <= 0.0:
                continue
            delta = random.random() * max_delta
            new_h = current_h.copy()
            new_h[i] += delta
            new_h[j] -= delta

        new_c5 = compute_c5(new_h)

        # Accept if better, otherwise probabilistically according to temperature
        accept = False
        if new_c5 < current_c5:
            accept = True
        else:
            prob = np.exp((current_c5 - new_c5) / max(temp, 1e-12))
            if random.random() < prob:
                accept = True

        if accept:
            current_h = new_h
            current_c5 = new_c5
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = new_h.copy()
                iters_without_improve = 0
            else:
                iters_without_improve += 1
        else:
            iters_without_improve += 1

        total_iters += 1

        # If stuck for many iterations, restart from the best or a fresh binary
        if iters_without_improve >= 8000:
            if random.random() < 0.75:
                current_h = best_h.copy()
                current_c5 = best_c5
            else:
                current_h = random_binary_h(n)
                current_c5 = compute_c5(current_h)
            iters_without_improve = 0

    # ----------------------------------------------------------------------
    # Return the best solution discovered
    # ----------------------------------------------------------------------
    return (best_h, float(best_c5), int(n))
```
