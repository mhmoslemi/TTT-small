        diff = s - target
        if abs(diff) < 1e-9:

            # need to subtract diff

            if len(idx) == 0:
                break

            if total_possible <= diff + 1e-12:
                # set all to zero
                h[idx] = 0.0
                # continue loop
                continue

            # reduce proportionally
            h[idx] -= reduce_frac * max_reduce

import time
import numpy as np

def _evaluate(h, dx):
    """Compute the overlap C5 for a given step function h."""
    # correlation without wrap-around, as specified
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def _project_feasible(vec, target_sum):
    """
    Project a vector onto the intersection of the box [0,1]^n and the hyperplane sum = target_sum.
    Uses a simple water‑filling scheme.
    """
    v = np.clip(vec, 0.0, 1.0)
    # Iteratively adjust the sum while respecting bounds
    for _ in range(500):
        s = v.sum()
        diff = s - target_sum
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Need to decrease the sum
            idx = np.where(v > 0.0)[0]
            if idx.size == 0:
                break
            max_reduce = v[idx]
            total = max_reduce.sum()
            if total <= diff:
                v[idx] = 0.0
                continue
            factor = diff / total
            v[idx] -= factor * max_reduce
        else:
            # Need to increase the sum
            need = -diff
            idx = np.where(v < 1.0)[0]
            if idx.size == 0:
                break
            max_add = 1.0 - v[idx]
            total = max_add.sum()
            if total <= need:
                v[idx] = 1.0
                continue
            factor = need / total
            v[idx] += factor * max_add
        v = np.clip(v, 0.0, 1.0)
    return v

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős C5 overlap.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Determine the problem size
    if "initial_h_values" in globals():
        init_h = np.asarray(initial_h_values, dtype=float)
        n = init_h.shape[0]
        init_h = _project_feasible(init_h, n / 2.0)
    else:
        n = 70
        init_h = None

    dx = 2.0 / n
    target_sum = n / 2.0

    # GA parameters
    pop_size = 30
    tournament_size = 3
    crossover_prob = 0.7
    mutation_prob = 0.9
    mutation_rate = 0.2      # per coordinate
    mutation_std = 0.1

    # ----- Population initialization -----
    population = []
    if init_h is not None:
        population.append(init_h.copy())
    while len(population) < pop_size:
        cand = rng.random(n)
        cand = _project_feasible(cand, target_sum)
        population.append(cand)

    # Evaluate initial fitness
    fitness = np.array([_evaluate(ind, dx) for ind in population])
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]

    stagnation = 0
    last_improvement = best_c5

    generation = 0
    # Main optimisation loop
    while True:
        # Time check (leave a small safety margin)
        if time.time() - start_time > budget_s * 0.95:
            break

        new_pop = [best_h.copy()]  # elitism: keep the best

        while len(new_pop) < pop_size:
            # ----- Tournament selection -----
            parents = []
            for _ in range(2):
                candidates = rng.integers(pop_size, size=tournament_size)
                cand_fitness = fitness[candidates]
                winner_idx = candidates[np.argmin(cand_fitness)]
                parents.append(population[winner_idx])

            # ----- Crossover -----
            if rng.random() < crossover_prob:
                point = rng.integers(1, n)  # cut point in [1, n-1]
                child = np.concatenate([parents[0][:point], parents[1][point:]])
            else:
                child = parents[0].copy()

            # ----- Mutation -----
            if rng.random() < mutation_prob:
                mask = rng.random(n) < mutation_rate
                if mask.any():
                    child[mask] += rng.normal(0.0, mutation_std, size=mask.sum())

            # ----- Projection onto feasible set -----
            child = _project_feasible(child, target_sum)

            new_pop.append(child)

        # Evaluate new population
        new_fitness = np.array([_evaluate(ind, dx) for ind in new_pop])

        # Update best solution
        cur_best_idx = np.argmin(new_fitness)
        cur_best_c5 = new_fitness[cur_best_idx]
        if cur_best_c5 + 1e-12 < best_c5:
            best_c5 = cur_best_c5
            best_h = new_pop[cur_best_idx].copy()
            stagnation = 0
            last_improvement = best_c5
        else:
            stagnation += 1

        # Adapt mutation strength if stuck
        if stagnation >= 50:
            mutation_std = min(0.5, mutation_std * 1.5)
            stagnation = 0

        population = new_pop
        fitness = new_fitness
        generation += 1

    # Final bound (use the provided evaluator if available)
    bound = None
    eval_func = globals().get("evaluate_erdos_solution", None)
    if callable(eval_func):
        try:
            bound = eval_func(best_h)
        except Exception:
            try:
                bound = eval_func(best_h, n)
            except Exception:
                bound = None
    if bound is None:
        bound = _evaluate(best_h, dx)

    return (best_h.tolist(), float(bound), n)
