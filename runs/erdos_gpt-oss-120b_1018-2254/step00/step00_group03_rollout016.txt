def random_feasible_h(n, target):
    # Use Dirichlet to get random proportions that sum to target
    # Dirichlet parameters all ones yields uniform distribution over simplex
    # uses np.random.dirichlet
    # Then clip entries >1 and reallocate excess.
    h = np.random.dirichlet(np.ones(n)) * target
    # now if any > 1, we need to adjust.
    # While there are entries > 1, clip them to 1 and redistribute excess among those <1.
    excess_mask = h > 1.0
    while np.any(excess_mask):
        excess = h[excess_mask] - 1.0
        total_excess = excess.sum()
        h[excess_mask] = 1.0
        # redistributable indices
        under_mask = h < 1.0 - 1e-12
        if not np.any(under_mask):
            # cannot redistribute; break
            break
        # distribute excess uniformly (or proportional to capacity)
        capacity = 1.0 - h[under_mask]
        # proportionally to capacity
        total_capacity = capacity.sum()
        if total_capacity <= 0:
            break
        # allocate
        add = np.minimum(capacity, total_excess * capacity / total_capacity)
        h[under_mask] += add
        # recompute excess mask
        excess_mask = h > 1.0
    # final clip to bounds and adjust sum
    h = np.clip(h, 0.0, 1.0)
    cur_sum = h.sum()
    diff = target - cur_sum
    # adjust diff by distributing to random indices with slack
    if abs(diff) > 1e-9:
        # gather indices that can receive increase if diff > 0, else decrease if diff < 0
        if diff > 0:
            # increase
            idxs = np.where(h < 1.0)[0]
            while diff > 1e-12 and len(idxs) > 0:
                i = np.random.choice(idxs)
                inc = min(diff, 1.0 - h[i])
                # add small random fraction of inc?
                add = inc * np.random.rand()
                if add < 1e-12:
                    add = inc  # fallback
                h[i] += add
                diff -= add
                if h[i] >= 1.0 - 1e-12:
                    idxs = idxs[idxs != i]
        else:
            # decrease

                dec = min(-diff, h[i])

                    sub = dec
                h[i] -= sub
                diff += sub
                if h[i] <= 1e-12:
                    idxs = idxs[idxs != i]
    # final clip
    h = np.clip(h, 0.0, 1.0)
    return h

import time

def compute_c5(h, dx):
    """Return the max overlap integral C5 for a discretised step function."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def random_feasible_h(n, target):
    """Generate a random feasible vector with entries in [0,1] and sum = target."""
    # start from a binary vector with exactly target ones (target is integer for even n)
    h = np.zeros(n, dtype=float)
    ones = np.random.choice(n, size=int(target), replace=False)
    h[ones] = 1.0
    # perform a few random mass‑redistribution moves to create fractional entries
    for _ in range(n):
        pos = np.where(h > 1e-12)[0]
        neg = np.where(h < 1.0 - 1e-12)[0]
        if len(pos) == 0 or len(neg) == 0:
            break
        i = np.random.choice(pos)
        j = np.random.choice(neg)
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand()
        h[i] -= delta
        h[j] += delta
