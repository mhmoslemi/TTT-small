    lag = lag_val - (n - 1)
    g = np.zeros_like(h)
    if lag >= 0:
        # term1: 1 - h[j - lag] for j >= lag
        idx1 = np.arange(lag, n)

        g[idx1] = 1 - h[idx1 + l]
        idx2 = np.arange(l, n)

def project_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-9):

import numpy as np
import time

def evaluate_h(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def compute_subgradient(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    lag = idx - (n - 1)
    g = np.ones(n)
    if lag == 0:
        g -= 2.0 * h
    elif lag > 0:
        # subtract h[j-lag] for j >= lag
        g[lag:] -= h[:-lag]
        # subtract h[j+lag] for j <= n-1-lag
        g[:n - lag] -= h[lag:]
    else:  # lag < 0
        l = -lag
        # subtract h[j - lag] = h[j + l] for j < n-l
        g[:n - l] -= h[l:]
        # subtract h[j + lag] = h[j - l] for j >= l
        g[l:] -= h[:n - l]
    dx = 2.0 / n
    return g * dx

def project_box_sum(v, target, lower=0.0, upper=1.0):
    lo = lower - v.max() - 1.0
    hi = upper - v.min() + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, lower, upper)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        free = (x > lower + 1e-12) & (x < upper - 1e-12)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, lower, upper)
        else:
            # adjust a single element
            idx = np.argmax(x) if diff < 0 else np.argmin(x)
            x[idx] += diff
            x = np.clip(x, lower, upper)
    return x

    start_time = time.time()

    else:

        h = rnd.rand(n)
        h = project_box_sum(h, target_sum)
    best_h = h.copy()
    best_c = evaluate_h(best_h)
    no_improve = 0
    # parameters
