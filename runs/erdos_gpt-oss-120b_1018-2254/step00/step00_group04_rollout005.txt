import time
import numpy as np

def _compute_c5(h):
    """Return the overlap bound C5 for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1 - h) gives the unscaled integral
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _project_onto_capped_simplex(v, target_sum, max_iter=50, tol=1e-9):
    """
    Project vector v onto the set {x | 0 <= x_i <= 1, sum(x) = target_sum}.
    Uses a binary search on the Lagrange multiplier tau.
    """
    n = v.shape[0]
    # Bounds for tau: at low tau all components are at upper bound 1,
    # at high tau all components are at lower bound 0.
    lo = np.min(v - 1.0)   # ensures v - lo >= 1  => x_i = 1
    hi = np.max(v)         # ensures v - hi <= 0 => x_i = 0
    for _ in range(max_iter):
        tau = (lo + hi) * 0.5
        x = np.clip(v - tau, 0.0, 1.0)
        s = x.sum()
        if abs(s - target_sum) <= tol:
            return x
        # sum decreases when tau increases
        if s > target_sum:
            lo = tau
        else:
            hi = tau
    # Final projection (may be slightly off target, fix tiny residual)
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    residual = target_sum - x.sum()
    if abs(residual) > 1e-12:
        # distribute residual uniformly over non‑boundary entries
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        free_idx = np.where(free)[0]
        if free_idx.size > 0:
            add = residual / free_idx.size
            x[free_idx] += add
            # clip again in case of slight overflow
            x = np.clip(x, 0.0, 1.0)
        else:
            # fallback: adjust the whole vector (should be rare)
            x += residual / n
            x = np.clip(x, 0.0, 1.0)
    return x

    else:
        n = 100                     # a modest default size
        h = np.full(n, 0.5, dtype=float)

    target = n / 2.0                # because ∫_0^2 h = 1 => sum(h) = n/2
    h = _project_onto_capped_simplex(h, target)

    best_h = h.copy()
    best_c5 = _compute_c5(best_h)

    start_time = time.time()
    iteration = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.98:
        # 1) find the shift with the worst (largest) overlap
        corr_full = np.correlate(best_h, 1.0 - best_h, mode="full")
        idx_max = np.argmax(corr_full)
        shift = idx_max - (n - 1)          # integer shift (may be negative)

        # 2) build a descent direction that increases the autocorrelation
        #    at the identified shift (i.e. reduces the overlap)
        abs_m = abs(shift)
        direction = np.zeros_like(best_h)
        if abs_m == 0:
            # gradient for zero shift is -2*h, so ascent direction is +2*h
            direction = 2.0 * best_h
        else:
            # contributions from neighbours at distance abs_m
            direction[: n - abs_m] += best_h[abs_m:]      # h_{j+abs_m}
            direction[abs_m :] += best_h[: n - abs_m]     # h_{j-abs_m}

        # 3) step size schedule (large early, small later)
        elapsed = time.time() - start_time
        decay = 1.0 - elapsed / budget_s
        step_size = 0.5 * decay + 0.001   # between ~0.5 and ~0.001

        # add a modest random perturbation to maintain exploration
        noise = np.random.randn(n) * step_size * 0.1
        cand = _project_onto_capped_simplex(best_h + step_size * direction + noise,
                                            target)

        cand_c5 = _compute_c5(cand)

        # 4) accept if improvement
        if cand_c5 < best_c5:
            best_h = cand
            best_c5 = cand_c5

        # occasional random restart to escape plateaus
        if iteration % 10000 == 0 and iteration > 0:
            rnd = np.random.rand(n)
            rnd = _project_onto_capped_simplex(rnd, target)
            rnd_c5 = _compute_c5(rnd)
            if rnd_c5 < best_c5:
                best_h = rnd
                best_c5 = rnd_c5

        iteration += 1

    return best_h, float(best_c5), n
