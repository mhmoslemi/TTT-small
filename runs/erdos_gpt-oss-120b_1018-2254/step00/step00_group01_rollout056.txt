def project_to_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-9, max_iter=100):
    # find lambda such that sum_clipped = target
    # Use bisection.
    lo = -1e6
    hi = 1e6
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        w = np.clip(v + lam, lower, upper)
        s = w.sum()
        if abs(s - target) < tol:
            return w
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo+hi)/2, lower, upper)

import numpy as np
import time

def compute_c5(h, dx):
    """Return max_k ∫ h(x)*(1-h(x+k)) dx approximated by discrete correlation."""
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-9, max_iter=100):
    """
    Project vector v onto the box [lower,upper]^n while enforcing sum(v)=target.
    Solves for λ in w_i = clip(v_i+λ, lower, upper) with Σ w_i = target.
    """
    lo = lower - np.max(v) - 1.0   # guarantee all clipped to lower
    hi = upper - np.min(v) + 1.0   # guarantee all clipped to upper
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        w = np.clip(v + lam, lower, upper)
        s = w.sum()
        if abs(s - target) <= tol:
            return w
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, lower, upper)

def gradient_refine(h, target, dx, max_iter=200, alpha=0.02):
    """
    Perform a few projected gradient steps on the shift that currently gives
    the maximal overlap.  This often reduces the objective further.
    """
    n = len(h)
    for _ in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = int(np.argmax(corr))
        shift = max_idx - (n - 1)        # shift corresponding to the max correlation
        # term1 = 1 - h[i+shift] where defined, else 1
        term1 = np.ones(n)
        if shift >= 0:
            term1[: n - shift] = 1.0 - h[shift:]
        else:
            k = -shift
            term1[k:] = 1.0 - h[: n - k]
        # term2 = h[i-shift] where defined, else 0
        term2 = np.zeros(n)
        if shift >= 0:
            term2[shift:] = h[: n - shift]
        else:
            k = -shift
            term2[: n - k] = h[k:]
        grad = (term1 - term2) * dx
        v = h - alpha * grad
        h = project_to_box_sum(v, target)
        # optional early stop if no improvement (omitted for speed)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] with values in [0,1],
    integral 1, minimizing the Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)

    # ----- initialise h -------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=np.float64)   # may be absent
    except Exception:
        n = int(kwargs.get('n_points', 47))
        h0 = np.full(n, 0.5, dtype=np.float64)
    n_points = len(h0)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h0 = np.clip(h0, 0.0, 1.0)
    if abs(h0.sum() - target_sum) > 1e-12:
        h0 = project_to_box_sum(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)
    cur_h = h0.copy()
    cur_c5 = best_c5

    # ----- simulated annealing -----------------------------------------
    start = time.time()
    T0 = 0.05                     # initial temperature scale
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed > budget_s * 0.95:
            break
        # exponential cooling based on elapsed proportion
        temp = T0 * math.exp(-5.0 * elapsed / budget_s)

        # pick two distinct indices and reshuffle their mass
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points - 1)
        if j >= i:
            j += 1
        total = cur_h[i] + cur_h[j]
        low = max(0.0, total - 1.0)
        high = min(1.0, total)
        if low >= high:
            continue
        new_a = np.random.uniform(low, high)
        new_b = total - new_a
        new_h = cur_h.copy()
        new_h[i] = new_a
        new_h[j] = new_b

        new_c5 = compute_c5(new_h, dx)
        delta = new_c5 - cur_c5
        if delta < 0.0 or np.random.rand() < math.exp(-delta / max(temp, 1e-12)):
            cur_h, cur_c5 = new_h, new_c5
            if new_c5 < best_c5:
                best_h, best_c5 = new_h.copy(), new_c5

        # periodic re‑projection to keep sum accurate (~once per second)
        if int(elapsed * 1000) % 1000 == 0:
            if abs(best_h.sum() - target_sum) > 1e-9:
                best_h = project_to_box_sum(best_h, target_sum)
                best_c5 = compute_c5(best_h, dx)
            if abs(cur_h.sum() - target_sum) > 1e-9:
                cur_h = project_to_box_sum(cur_h, target_sum)
                cur_c5 = compute_c5(cur_h, dx)

    # ----- gradient refinement -----------------------------------------
    best_h = gradient_refine(best_h, target_sum, dx, max_iter=200, alpha=0.02)
    best_c5 = compute_c5(best_h, dx)

    # final projection (safety)
    best_h = project_to_box_sum(best_h, target_sum)

    # obtain official bound if helper is available
    try:
        val = evaluate_erdos_solution(best_h, n_points)
        if isinstance(val, (list, tuple)):
            final_c5 = float(val[0])
        else:
            final_c5 = float(val)
    except Exception:
        final_c5 = compute_c5(best_h, dx)

    return (best_h, float(final_c5), n_points)
