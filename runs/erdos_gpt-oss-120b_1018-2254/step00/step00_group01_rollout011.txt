def propose_move(h, step_max):
    n = len(h)
    i, j = np.random.choice(n, size=2, replace=False)
    hi = h[i]; hj = h[j]
    eps_low = max(-hi, hj - 1, -step_max)
    eps_high = min(1 - hi, hj, step_max)
    if eps_low >= eps_high:
        return None, None
    epsilon = np.random.uniform(eps_low, eps_high)
    new_h = h.copy()
    new_h[i] = hi + epsilon
    new_h[j] = hj - epsilon
    return new_h, epsilon
```

import numpy as np
import time

def _c5_of(h: np.ndarray) -> float:
    """Return the overlap bound C5 for a discrete step function h."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h); np.correlate implements sum_i h[i] * (1-h)[i+lag]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _propose_swap(h: np.ndarray, step_max: float):
    """Return a new feasible h after exchanging epsilon between two indices.
    The move keeps the total sum unchanged and respects the [0,1] bounds.
    If no feasible ε exists, returns (None, None)."""
    n = h.size
    i, j = np.random.choice(n, size=2, replace=False)
    hi, hj = h[i], h[j]

    eps_low = max(-hi, hj - 1.0, -step_max)
    eps_high = min(1.0 - hi, hj, step_max)
    if eps_low >= eps_high:
        return None, None

    eps = np.random.uniform(eps_low, eps_high)
    new_h = h.copy()
    new_h[i] = hi + eps
    new_h[j] = hj - eps
    return new_h, eps


def _random_feasible(n: int, target_sum: float) -> np.ndarray:
    """Generate a random vector in [0,1]^n whose entries sum to target_sum."""
    h = np.zeros(n, dtype=float)
    remaining = target_sum
    order = np.random.permutation(n)
    for idx in order[:-1]:
        max_val = min(1.0, remaining)
        val = np.random.rand() * max_val
        h[idx] = val
        remaining -= val
    h[order[-1]] = remaining  # the remainder fits because remaining ≤ 1
    return h

    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Initialise problem size and starting vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float, copy=True)
        n = h.size
    else:
        n = 47
        target = n / 2.0
        h = _random_feasible(n, target)

    # Ensure feasibility (bounds & exact integral)
    h = np.clip(h, 0.0, 1.0)
    target_sum = n / 2.0
    cur_sum = h.sum()
    if not np.isclose(cur_sum, target_sum):
        diff = target_sum - cur_sum
        # adjust a single interior entry if possible
        interior = np.where((h > 0.0) & (h < 1.0))[0]
        if interior.size == 0:
            i = np.random.randint(n)
        else:
            i = np.random.choice(interior)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)

    # ------------------------------------------------------------------
    # Main optimisation loop (simulated‑annealing style)
    # ------------------------------------------------------------------
    cur_c5 = _c5_of(h)
    best_h = h.copy()
    best_c5 = cur_c5

    step_max = 0.5               # maximal exchange magnitude
    T0 = 0.02                    # initial temperature
    it = 0
    no_improve = 0

    while time.time() - start < budget_s * 0.97:
        new_h, _ = _propose_swap(h, step_max)
        if new_h is None:
            continue

        new_c5 = _c5_of(new_h)
        elapsed = time.time() - start
        # linear temperature decay
        T = T0 * (1.0 - elapsed / budget_s)

        accept = False
        if new_c5 < cur_c5:
            accept = True
        elif T > 1e-12:
            # uphill acceptance probability
            prob = np.exp(-(new_c5 - cur_c5) / T)
            if np.random.rand() < prob:
                accept = True

        if accept:
            h = new_h
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # adapt step size slowly
        if it % 2000 == 0 and it > 0:
            step_max = max(0.001, step_max * 0.95)

        # if stuck for a while, perturb around the current best
        if no_improve > 20000:
            h = best_h.copy()
            for _ in range(100):
                pert, _ = _propose_swap(h, step_max=0.5)
                if pert is not None:
                    h = pert
            cur_c5 = _c5_of(h)
            no_improve = 0

        it += 1

    # ------------------------------------------------------------------
    # Final safety projection (bounds + exact sum)
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-9:
        interior = np.where((best_h > 0.0) & (best_h < 1.0))[0]
        if interior.size == 0:
            i = np.random.randint(n)
        else:
            i = np.random.choice(interior)
        best_h[i] = np.clip(best_h[i] + diff, 0.0, 1.0)

    return (best_h, best_c5, n)
