def project_to_feasible(v, target_sum):
    # v is a numpy array
    # ensure 0<=v<=1 and sum(v) = target_sum (target_sum = n/2)
    w = np.clip(v, 0.0, 1.0)
    # Use scaling iterative
    for _ in range(10):
        current_sum = w.sum()
        if abs(current_sum - target_sum) < 1e-12:
            break
        diff = target_sum - current_sum
        # identify free indices (not at bound)
        if diff > 0:
            free = (w < 1 - 1e-12)
        else:
            free = (w > 1e-12)
        nfree = free.sum()
        if nfree == 0:
            # cannot adjust further
            break
        # distribute diff equally across free indices
        w[free] += diff / nfree
        # clip again
        w = np.clip(w, 0.0, 1.0)
    # minor correction scaling if still off
    # final adjust by uniform scaling if still off in interior region (all non bounds)
    # but essentially it should be close.
    # check if sum is close
    if w.sum() != target_sum:
        # final adjustment via scaling of all components within bounds using simple approach:
        # compute factor = target_sum / w.sum(); w *= factor, then clip.
        w = np.clip(w * (target_sum / w.sum()), 0.0, 1.0)
        # final renorm by distributing residual
        diff = target_sum - w.sum()
        if abs(diff) > 1e-9:
            # distribute over not-at-bounds
            free = (w > 1e-12) & (w < 1-1e-12)
            if free.sum() > 0:
                w[free] += diff / free.sum()
                w = np.clip(w, 0.0, 1.0)
    return w

def propose_move(h, rng, step):
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    # compute allowable delta interval
    low = max(-h[i], h[j] - 1.0)
    high = min(1.0 - h[i], h[j])
    # limit to [-step, step]
    low = max(low, -step)
    high = min(high, step)

        return None  # no feasible move
    delta = rng.uniform(low, high)
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta

    n = len(h)
    # compute correlation all shifts, find worst shift (max C5)
    corr = np.correlate(h, 1-h, mode='full')
    idx = np.argmax(corr)

    # Determine candidate indices i where 0 <= i < n and 0 <= i+lag < n

        i_vals = np.arange(0, n - lag)
    else:
        i_vals = np.arange(-lag, n)

import numpy as np, time, math

def compute_c5(h, dx):
    """Maximum overlap integral for discretized h."""
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_to_feasible(v, target):
    """Project v onto the box [0,1]^n while enforcing sum(v)=target."""
    w = np.clip(v, 0.0, 1.0)
    for _ in range(10):
        s = w.sum()
        if abs(s - target) < 1e-12:
            break
        diff = target - s
        if diff > 0:
            free = (w < 1.0 - 1e-12)
        else:
            free = (w > 1e-12)
        nfree = free.sum()
        if nfree == 0:
            break
        w[free] += diff / nfree
        w = np.clip(w, 0.0, 1.0)
    # tiny correction if needed
    if abs(w.sum() - target) > 1e-8:
        scale = target / w.sum()
        w = np.clip(w * scale, 0.0, 1.0)
        diff = target - w.sum()
        if abs(diff) > 1e-9:
            free = (w > 1e-12) & (w < 1.0 - 1e-12)
            if free.any():
                w[free] += diff / free.sum()
                w = np.clip(w, 0.0, 1.0)
    return w

def propose_move(h, rng, step):
    """Two‑coordinate move that keeps the sum exactly unchanged."""
    n = h.size
    i, j = rng.choice(n, size=2, replace=False)
    # allowable delta for i (increase) and j (decrease)
    low = max(-h[i], h[j] - 1.0)          # min delta so i stays >=0 and j stays <=1 after subtracting delta
    high = min(1.0 - h[i], h[j])         # max delta so i stays <=1 and j stays >=0 after subtracting delta
    low = max(low, -step)
    high = min(high, step)
    if low >= high:
        return None
    delta = rng.uniform(low, high)
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    return new_h

def targeted_move(h, rng, step):
    """Move that tries to improve the worst shift."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    lag = idx - (n - 1)               # shift index (positive or negative)
    # indices where both i and i+lag lie in range
    if lag >= 0:
        i_vals = np.arange(0, n - lag)
    else:
        i_vals = np.arange(-lag, n)
    if i_vals.size == 0:
        return None
    products = h[i_vals] * h[i_vals + lag]
    i_min = i_vals[int(np.argmin(products))]
    i2 = i_min + lag
    # pick a donor index j distinct from i_min and i2
    donor_candidates = np.where(h > 1e-12)[0]
    donor_candidates = donor_candidates[(donor_candidates != i_min) & (donor_candidates != i2)]
    if donor_candidates.size == 0:
        return None
    j = rng.choice(donor_candidates)
    # allowed increase alpha for both i_min and i2, taken from j (2*alpha)
    max_inc_i = 1.0 - h[i_min]
    max_inc_i2 = 1.0 - h[i2]
    max_dec_j = h[j] / 2.0
    max_alpha = min(max_inc_i, max_inc_i2, max_dec_j, step)
    if max_alpha <= 0:
        return None
    alpha = rng.uniform(0.0, max_alpha)
    new_h = h.copy()
    new_h[i_min] += alpha
    new_h[i2] += alpha
    new_h[j] -= 2.0 * alpha
    return new_h

def random_feasible(n, target, rng):
    """Generate a random feasible vector."""
    v = rng.random(n)
    return project_to_feasible(v, target)
