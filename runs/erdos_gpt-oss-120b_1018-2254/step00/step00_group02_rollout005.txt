def project_onto_hyperplane_and_bounds(v, target):
    w = np.clip(v, 0, 1)
    # compute difference
    diff = w.sum() - target
    if abs(diff) < 1e-8:
        return w
    # We'll adjust in a loop
    # Use while with max iterations
    for _ in range(100):
        if diff > 0:
            # need to reduce sum
            idx = w > 0
            if not np.any(idx):
                break
            reduction = diff / np.count_nonzero(idx)
            w[idx] -= reduction
            w = np.maximum(w, 0)
        else:
            # need to increase sum
            idx = w < 1
            if not np.any(idx):
                break
            increase = -diff / np.count_nonzero(idx)
            w[idx] += increase
            w = np.minimum(w, 1)
        diff = w.sum() - target
        if abs(diff) < 1e-8:
            break
    # After loop, due to rounding may still off, adjust last index.
    # Use simple linear correction: compute remaining diff and adjust first feasible element
    if abs(diff) > 1e-6:
        # Find index where adjusting by diff stays within [0,1]
        for i in range(len(w)):
            if 0 <= w[i] - diff <= 1:
                w[i] -= diff
                break
    return w

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # Using np.dot to compute correlation? Use np.correlate
    # np.correlate may be slightly slower but fine.

def project_onto_constraints(v, target):
    # Clip to [0,1]
    w = np.clip(v, 0.0, 1.0)
    # adjust sum to target
    # Use the algorithm described
    # compute diff
    diff = w.sum() - target
    # Tolerance
    tol = 1e-9
    if abs(diff) < tol:
        return w
    # iterate a few times
    max_iter = 100
    for _ in range(max_iter):
        if abs(diff) < tol:
            break
        if diff > 0:
            # need to reduce sum
            idx = w > 0
            if not np.any(idx):
                break
            reduction = diff / np.count_nonzero(idx)
            w[idx] -= reduction
            w = np.maximum(w, 0.0)
        else:
            # need to increase sum
            idx = w < 1
            if not np.any(idx):
                break
            increase = -diff / np.count_nonzero(idx)
            w[idx] += increase
            w = np.minimum(w, 1.0)
        diff = w.sum() - target
    # final adjust if still off by tiny amount
    if abs(diff) > 1e-8:
        # adjust single element if possible
        for i in range(len(w)):
            if 0 <= w[i] - diff <= 1:
                w[i] -= diff
                diff = 0.0
                break
    return w

def optimize_one_dimension(n, init_h, target, time_budget):

    start = time.time()

    best_c5 = compute_c5(best_h)
    # Random search loop
    while time.time() < time_limit:
        # generate random candidate
        v = np.random.rand(n)
        cand = project_onto_constraints(v, target)
        c5 = compute_c5(cand)
        if c5 < best_c5:
            best_c5 = c5
            best_h = cand
        # also maybe try small perturbations of best_h
        # We'll add a few perturbation attempts each loop

        # but ensure keep in bounds and sum
        # for simplicity omitted

        # break if time close
        if time.time() > time_limit - 0.1:
            break

    # local refinement if time permits
    remaining = time_limit - time.time()
    if remaining > 0.2:
        bounds = [(0.0, 1.0)] * n
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}

    dx = 2.0 / n
    # correlate returns float array
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_to_constraints(v, target):
    w = np.clip(v, 0.0, 1.0)
    diff = w.sum() - target
    tol = 1e-9
    if abs(diff) < tol:
        return w
    for _ in range(100):
        if abs(diff) < tol:
            break
        if diff > 0:
            idx = w > 0
            cnt = np.count_nonzero(idx)
            if cnt == 0:
                break
            red = diff / cnt
            w[idx] -= red
            w = np.maximum(w, 0.0)
        else:
            idx = w < 1
            cnt = np.count_nonzero(idx)
            if cnt == 0:
                break
            inc = -diff / cnt
            w[idx] += inc
            w = np.minimum(w, 1.0)

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h: np.ndarray) -> float:
    """Return the discrete C5 value for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * float(np.max(corr))

def project_to_constraints(v: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and adjust to satisfy sum(v) == target."""
    w = np.clip(v, 0.0, 1.0)
    diff = w.sum() - target
    if abs(diff) < 1e-12:
        return w
    for _ in range(100):
        if abs(diff) < 1e-12:
            break
        if diff > 0:  # need to reduce sum
            idx = w > 0
            cnt = np.count_nonzero(idx)
            if cnt == 0:
                break
            red = diff / cnt
            w[idx] -= red
            w = np.maximum(w, 0.0)
        else:          # need to increase sum
            idx = w < 1
            cnt = np.count_nonzero(idx)
            if cnt == 0:
                break
            inc = -diff / cnt
            w[idx] += inc
            w = np.minimum(w, 1.0)
        diff = w.sum() - target
    # final tiny correction
    diff = w.sum() - target
    if abs(diff) > 1e-8:
        for i in range(len(w)):
            if 0.0 <= w[i] - diff <= 1.0:
                w[i] -= diff
                break
    return w

def hill_climb(start_h: np.ndarray, target: float, time_limit: float):
    """Simple hill‑climbing with mass transfers between two coordinates."""
    best = start_h.copy()
    best_val = compute_c5(best)
    n = best.size
    start = time.time()
    max_step = 0.2  # maximal transferred mass
    while True:
        now = time.time()
        if now >= time_limit:
            break
        elapsed = now - start
        frac = elapsed / (time_limit - start)
        step = max(0.001, max_step * (1.0 - frac))

        # occasional full random restart
        if np.random.rand() < 0.005:
            cand = project_to_constraints(np.random.rand(n), target)
        else:
            i, j = np.random.choice(n, 2, replace=False)
            # choose donor (higher value) and receiver (lower value)
            if best[i] > best[j]:
                donor, receiver = i, j
            else:
                donor, receiver = j, i
            max_transfer = min(best[donor], 1.0 - best[receiver], step)
            if max_transfer <= 0:
                continue
            d = np.random.rand() * max_transfer
            cand = best.copy()
            cand[donor] -= d
            cand[receiver] += d

        cand_val = compute_c5(cand)
        if cand_val < best_val:
            best = cand
            best_val = cand_val
    return best, best_val

def optimize_one_dimension(n: int, init_h: np.ndarray, target: float, budget: float):
    """Run hill‑climbing and a short SLSQP refinement within a time budget."""
    end_time = time.time() + budget
    best_h, best_val = hill_climb(init_h, target, end_time)

    remaining = end_time - time.time()
    if remaining > 0.5:
        bounds = [(0.0, 1.0)] * n
        cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
        try:
            res = minimize(lambda x: compute_c5(x), best_h,
                           method="SLSQP", bounds=bounds,
                           constraints=cons,
                           options={"maxiter": 2000, "ftol": 1e-12, "disp": False})
            if res.success:
                refined = np.clip(res.x, 0.0, 1.0)
                refined = project_to_constraints(refined, target)
                refined_val = compute_c5(refined)
                if refined_val < best_val:
                    best_h, best_val = refined, refined_val
        except Exception:
            pass
    return best_h, best_val

    if remaining > 20.0:
