import time, math, random
import numpy as np
from scipy.optimize import minimize

def _project_to_feasible(x, target):
    # Clip to [0,1] then adjust via binary search for lambda
    x = np.clip(x, 0.0, 1.0)
    # binary search lambda
    low, high = -1.0, 1.0
    for _ in range(50):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low+high)/2.0, 0.0, 1.0)
    # fix possible remaining small diff
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        # distribute diff to a non-saturated coordinate
        for i in range(len(y)):
            if (diff > 0 and y[i] < 1.0) or (diff < 0 and y[i] > 0.0):
                adjust = min(diff, 1.0 - y[i]) if diff > 0 else max(diff, -y[i])
                y[i] += adjust
                break
    return y

def _c5_value(h, dx):
    # correlation full mode
    # numpy correlate returns array, compute max
    # using float64 array
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def _anneal(start_h, dx, budget_s, start_time):
    n = len(start_h)
    # copy start
    cur_h = start_h.copy()
    cur_c5 = _c5_value(cur_h, dx)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng()
    # Simple binary annealing (swap 1<->0)
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 1.0:
            break
        # temperature schedule: exponential decrease from 1 to 1e-4
        frac = elapsed / (budget_s - 1.0)
        T = 1.0 * ((1e-4) ** frac)  # but this yields 1 * (1e-4)^frac? Actually need linear? Use pow
        # Use exponent: T = T0 * (Tf/T0) ** frac
        T = 1.0 * ((1e-4 / 1.0) ** frac)
        # propose swap
        ones_idx = np.where(cur_h > 0.5)[0]
        zeros_idx = np.where(cur_h <= 0.5)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i1 = rng.choice(ones_idx)
        i2 = rng.choice(zeros_idx)
        new_h = cur_h.copy()
        new_h[i1], new_h[i2] = new_h[i2], new_h[i1]
        new_c5 = _c5_value(new_h, dx)
        if new_c5 < cur_c5 or rng.random() < math.exp(-(new_c5 - cur_c5) / (T + 1e-12)):
            cur_h = new_h
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_h = cur_h.copy()
                best_c5 = cur_c5
    return best_h, best_c5

    # Continuous refinement with SLSQP from best_h

            x0 = _project_to_feasible(cur_h, target_sum)

                            constraints=cons,
                            options={'maxiter': 150, 'ftol': 1e-9, 'disp': False})

        except Exception:
            pass
    # Final projection to ensure feasibility
    best_h = _project_to_feasible(best_h, target_sum)
    best_c5 = _c5_value(best_h, dx)
    # Return as list for easy serialization

import time
import math
import random
import numpy as np
from scipy.optimize import minimize

def _project_to_feasible(x, target):
    """Clip to [0,1] and adjust the sum to `target` by a simple water‑filling."""
    x = np.clip(x, 0.0, 1.0)
    low, high = -1.0, 1.0
    for _ in range(50):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, 0.0, 1.0)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        for i in range(len(y)):
            if (diff > 0 and y[i] < 1.0) or (diff < 0 and y[i] > 0.0):
                adjust = min(diff, 1.0 - y[i]) if diff > 0 else max(diff, -y[i])
                y[i] += adjust
                break
    return y

def _c5_value(h, dx):
    """Maximum overlap integral for a discrete step function."""
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def _anneal(start_h, dx, budget_s, start_time, rng):
    """Simple binary annealing that swaps a 1 with a 0."""
    n = len(start_h)
    cur_h = start_h.copy()
    cur_c5 = _c5_value(cur_h, dx)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 1.0:
            break
        frac = elapsed / (budget_s - 1.0)
        T = (1e-4) ** frac  # exponential cooling from 1 to 1e-4
        ones = np.where(cur_h > 0.5)[0]
        zeros = np.where(cur_h <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i1 = rng.choice(ones)
        i2 = rng.choice(zeros)
        new_h = cur_h.copy()
        new_h[i1], new_h[i2] = new_h[i2], new_h[i1]
        new_c5 = _c5_value(new_h, dx)
        if new_c5 < cur_c5 or rng.random() < math.exp(-(new_c5 - cur_c5) / (T + 1e-12)):
            cur_h = new_h
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_h = cur_h.copy()
                best_c5 = cur_c5
    return best_h, best_c5
