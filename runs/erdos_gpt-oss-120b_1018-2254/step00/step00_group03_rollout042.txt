def project(h, target):
    h = np.clip(h, 0.0, 1.0)
    # Compute total sum
    total = h.sum()
    # iterative adjustment
    # We'll repeat a few times (max 10) to converge, using water-filling.
    # Compute indices where h is not at bounds.
    for _ in range(10):
        if abs(total - target) < 1e-9:
            break
        # Those indices that are not at bounds (0<h<1)
        free = np.where((h > 0) & (h < 1))[0]
        if free.size == 0:
            # all at bounds, cannot adjust; break
            break
        # compute amount to add/subtract uniformly across free indices
        delta = (target - total) / free.size
        h[free] += delta
        # after adding, clip again
        h = np.clip(h, 0.0, 1.0)
        total = h.sum()
    # After loop, we may still be off by a tiny amount; adjust by scaling if needed:
    # if sum diff still > 1e-6, we can adjust one free element.
    if abs(total - target) > 1e-8:
        # distribute difference to a single free index if possible
        diff = target - total
        # find any free index (or bound index that can be shifted)
        idx = np.argmax(h)  # placeholder
        # clamp addition:
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    # final clip and ensure sum close to target
    return h
```

import time, numpy as np

def project_onto_box_simplex(v, target, max_iter=100):
    x = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        s = x.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # free indices
        free = np.where((x > 0) & (x < 1))[0]
        if free.size == 0:
            # adjust with bound variables if possible
            # if diff > 0 try to increase some at 0
            if diff > 0:
                zero_idx = np.where(x == 0)[0]
                if zero_idx.size:
                    idx = zero_idx[0]
                    inc = min(diff, 1.0)
                    x[idx] = inc
                    continue
            elif diff < 0:
                one_idx = np.where(x == 1)[0]
                if one_idx.size:
                    idx = one_idx[0]
                    dec = min(-diff, 1.0)
                    x[idx] = 1.0 - dec
                    continue
            break
        delta = diff / free.size
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
    # Final adjustment if still off by tiny epsilon
    s = x.sum()
    if abs(s - target) > 1e-8:
        diff = target - s
        # adjust first free element
        free = np.where((x > 0) & (x < 1))[0]
        if free.size:
            idx = free[0]
            x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
        else:
            # adjust any element within bounds
            idx = np.argmin(np.abs(x - target/len(v)))  # just pick one
            x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
    return x

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the intersection of the box [0,1]^n and the hyperplane sum = target
def _project_box_sum(v, target, max_iter=100):
    """Project vector v onto {0<=x<=1, sum(x)=target}."""
    x = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        s = x.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        free = np.where((x > 0) & (x < 1))[0]
        if free.size == 0:
            # No interior variables – adjust a boundary variable if possible
            if diff > 0:
                zero_idx = np.where(x == 0)[0]
                if zero_idx.size:
                    idx = zero_idx[0]
                    inc = min(diff, 1.0)
                    x[idx] = inc
                    continue
            else:
                one_idx = np.where(x == 1)[0]
                if one_idx.size:
                    idx = one_idx[0]
                    dec = min(-diff, 1.0)
                    x[idx] = 1.0 - dec
                    continue
            break
        delta = diff / free.size
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
    # final tiny adjustment if needed
    s = x.sum()
    if abs(s - target) > 1e-8:
        diff = target - s
        idx = np.argmin(np.abs(x - 0.5))
        x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
    return x


# ----------------------------------------------------------------------
# Compute the maximal overlap C5 for a given step vector h
def _c5_value(h):
    n = h.shape[0]
    dx = 2.0 / n
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


# ----------------------------------------------------------------------
# Gradient of a single shift s (s >= 0) for the non‑scaled C_s = total - f_s
def _grad_for_shift(h, s):
    n = h.shape[0]
    if s == 0:
        return 1.0 - 2.0 * h
    grad = np.ones_like(h)
    # forward part (h_{j+s})
    if s < n:
        grad[: n - s] -= h[s:]
        # backward part (h_{j-s})
        grad[s:] -= h[: n - s]
    return grad

            eta = min(eta * 1.05, 1.0)   # accelerate a bit
        else:
            eta *= 0.5                  # backtrack
            if eta < 1e-6:              # too small – random restart
                h = _project_box_sum(rng.random(n), target_sum)
                eta = 0.1

        # occasional tiny random swaps to escape flat regions
        if rng.random() < 0.01:
            i, j = rng.choice(n, size=2, replace=False)
            delta = min(h[i], 1.0 - h[j], 0.05)
            if delta > 1e-12:
                h_tmp = h.copy()
                h_tmp[i] -= delta
                h_tmp[j] += delta
                h_tmp = _project_box_sum(h_tmp, target_sum)
                if _c5_value(h_tmp) < cur_max:
                    h = h_tmp

    # final evaluation (use the provided helper if available)
    try:
        final_score = float(evaluate_erdos_solution(best_h))  # pylint: disable=undefined-variable
    except Exception:
        final_score = _c5_value(best_h)

    return (best_h, final_score, n)
