import time, numpy as np

def compute_c5(h):
    N = h.shape[0]
    dx = 2.0 / N
    # np.correlate returns array of shape 2N-1
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_sum_to_target(h):
    # Ensure sum exactly N/2 while staying in [0,1].
    N = h.shape[0]
    target = N / 2.0
    # Clip first
    h = np.clip(h, 0.0, 1.0)
    # Compute diff
    diff = target - h.sum()
    # If diff is negligible, return
    if abs(diff) < 1e-12:
        return h
    # Distribute diff uniformly among free variables
    # Determine indices not at bound in direction we need
    if diff > 0:
        free = np.where(h < 1.0)[0]
        if free.size == 0:
            return h
        # add diff equally
        add = diff / free.size
        # but ensure not exceed bound
        # We'll iteratively add to each free index until diff done.
        # Simpler: use water-filling
        while diff > 1e-12 and free.size > 0:
            add = diff / free.size
            # saturate
            inc = np.minimum(add, 1.0 - h[free])
            h[free] += inc
            diff = target - h.sum()
            free = np.where(h < 1.0 - 1e-12)[0]
    else:
        free = np.where(h > 0.0)[0]
        while diff < -1e-12 and free.size > 0:
            sub = diff / free.size  # negative
            dec = np.maximum(sub, -h[free])
            h[free] += dec
            diff = target - h.sum()
            free = np.where(h > 1e-12)[0]
    # final clip
    h = np.clip(h, 0.0, 1.0)
    return h

def anneal_h(initial_h, budget_s, start_time):

    cur_h = initial_h.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    rng = np.random.default_rng(42)  # seed - can incorporate seed param
    T0 = 0.02
    Tf = 1e-5
    # set iteration limit as high as possible
    while True:
        now = time.time()
        if now - start_time >= budget_s:
            break
        elapsed = now - start_time
        frac = elapsed / budget_s
        T = T0 * (1 - frac) + Tf * frac
        # propose move
        i = rng.integers(N)
        j = rng.integers(N-1)
        if j >= i:
            j += 1
        h_i = cur_h[i]
        h_j = cur_h[j]
        # determine max transferable from i to j or vice versa
        # Choose direction randomly
        if rng.random() < 0.5:
            max_transfer = min(h_i, 1.0 - h_j)
            if max_transfer <= 0:
                continue
            delta = rng.random() * max_transfer
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        else:
            max_transfer = min(h_j, 1.0 - h_i)
            if max_transfer <= 0:
                continue
            delta = rng.random() * max_transfer
            new_h = cur_h.copy()
            new_h[i] += delta
            new_h[j] -= delta
        new_score = compute_c5(new_h)
        delta_score = new_score - cur_score
        if delta_score < 0 or rng.random() < np.exp(-delta_score / (T + 1e-12)):
            cur_h = new_h
            cur_score = new_score
            if cur_score < best_score:
                best_h = cur_h.copy()
                best_score = cur_score
    # return best found
    return best_h, best_score

import time
import numpy as np

def _c5_score(h: np.ndarray) -> float:
    """Return the C5 value for a discrete step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _project_to_target(h: np.ndarray) -> np.ndarray:
    """
    Adjust array h in-place so that:
      * 0 ≤ h[i] ≤ 1
      * sum(h) = len(h)/2   (exactly the integral constraint)
    Simple water‑filling is sufficient for the small sizes we use.
    """
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        cur = h.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = np.where(h < 1.0)[0]
            if free.size == 0:
                break
            inc = diff / free.size
            delta = np.minimum(inc, 1.0 - h[free])
            h[free] += delta
        else:
            free = np.where(h > 0.0)[0]
            if free.size == 0:
                break
            dec = diff / free.size
            delta = np.maximum(dec, -h[free])
            h[free] += delta
    return np.clip(h, 0.0, 1.0)


def _random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector in [0,1]^n with sum = n/2."""
    x = rng.random(n)
    x *= (n / 2.0) / x.sum()
    x = np.clip(x, 0.0, 1.0)
    return _project_to_target(x)


def _anneal(initial_h: np.ndarray, allocated_s: float, rng: np.random.Generator):
    """
    Simple simulated‑annealing on the continuous space.
    Keeps the sum constraint exactly by transferring a random amount
    between two coordinates.
    """
    n = initial_h.shape[0]
    cur_h = initial_h.copy()
    cur_score = _c5_score(cur_h)

    best_h = cur_h.copy()
    best_score = cur_score

    start = time.time()
    T0 = 0.02
    Tf = 1e-5

    while time.time() - start < allocated_s:
        elapsed = time.time() - start
        frac = elapsed / allocated_s
        T = T0 * (1.0 - frac) + Tf * frac

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        if rng.random() < 0.5:
            max_tr = min(cur_h[i], 1.0 - cur_h[j])
            if max_tr <= 0.0:
                continue
            delta = rng.random() * max_tr
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        else:
            max_tr = min(cur_h[j], 1.0 - cur_h[i])
            if max_tr <= 0.0:
                continue
            delta = rng.random() * max_tr
            new_h = cur_h.copy()
            new_h[i] += delta
            new_h[j] -= delta

        new_score = _c5_score(new_h)
        dscore = new_score - cur_score
        if dscore < 0.0 or rng.random() < np.exp(-dscore / (T + 1e-12)):
            cur_h = new_h
            cur_score = new_score
            if cur_score < best_score:
                best_h = cur_h.copy()
                best_score = cur_score

    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):

    Optimize a step function h on [0,2] to minimise the Erdős C5 constant.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------
    # Initialise h
    # ------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
        n = h0.shape[0]
        h0 = np.clip(h0, 0.0, 1.0)
        # enforce the exact integral constraint
        h0 = _project_to_target(h0)
    else:
        n = 80
        h0 = np.full(n, 0.5, dtype=np.float64)  # already satisfies sum=n/2

    best_h = h0.copy()
    best_score = _c5_score(best_h)

    # ------------------------------------------------------------
    # Main optimisation loop (restarts with annealing)
    # ------------------------------------------------------------
    while True:
        elapsed_total = time.time() - start_time
        remaining = budget_s - elapsed_total
        if remaining < 5.0:
            break

        # allocate a chunk (max 30 s) for a single annealing run
        chunk = min(30.0, remaining * 0.9)

        # occasional random restart to avoid local traps
        if rng.random() < 0.3:
            start_h = _random_feasible(n, rng)
        else:
            start_h = best_h.copy()

        cand_h, cand_score = _anneal(start_h, chunk, rng)

        if cand_score < best_score:
            best_h = cand_h
            best_score = cand_score

    # final safety projection and score recomputation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_to_target(best_h)
    final_score = _c5_score(best_h)

    return best_h, final_score, best_h.shape[0]
