def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    from scipy.optimize import minimize

    start_time = time.time()
    np.random.seed(seed)

    # Determine n_points
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 80
        h0 = np.full(n, 0.5)  # uniform

    dx = 2.0 / n
    target_sum = n / 2.0

    # Ensure initial feasible
    h0 = np.clip(h0, 0, 1)
    s = h0.sum()
    # Adjust to target by scaling difference equally while staying within bounds
    if abs(s - target_sum) > 1e-12:
        # Simple scaling: adjust rest of slack proportionally
        # We'll project onto simplex with box [0,1] and sum=target.
        # Use iterative algorithm for projection onto capped simplex.
        h0 = project_onto_box_simplex(h0, target_sum, 0.0, 1.0)

    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h)

    # Define objective wrapper
    def objective(x):
        return evaluate_c5(x)

    # Equality constraint: sum(x) - target_sum = 0
    eq_cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0.0, 1.0)] * n

    # Helper: local optimizer
    def local_optimize(x0):
        res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                       constraints=[eq_cons],
                       options={'ftol':1e-12, 'maxiter':500, 'disp': False})
        if not res.success:
            # try again with increased maxiter
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=[eq_cons],
                           options={'ftol':1e-12, 'maxiter':1500, 'disp': False})
        return res

    # projection function
    def project_onto_box_simplex(v, target, lo, hi):
        # Project onto [lo,hi] box and enforce sum=target via simple algorithm:
        # Clip, then adjust by adding/subtracting uniformly to non-bound entries.
        v = np.clip(v, lo, hi)
        # Use iterative water-filling: adjust residual difference.
        # Implementation adapted from algorithm for projection onto simplex with bounds.
        # We'll do simple approach: while remaining diff > tolerance:
        tol = 1e-12
        max_iter = 1000
        for _ in range(max_iter):
            s = v.sum()
            diff = target - s
            if abs(diff) <= tol:
                break
            # indices not at bounds
            free = (v > lo + tol) & (v < hi - tol)
            nfree = np.count_nonzero(free)
            if nfree == 0:
                # cannot adjust further, break
                break
            adjustment = diff / nfree
            # apply adjustment, but stay within bounds
            v[free] += adjustment
            # clip again to bounds
            v = np.clip(v, lo, hi)
        # final clipping
        v = np.clip(v, lo, hi)
        return v

    # Main loop
    iteration = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 5:  # leave 5 seconds safety
            break

        # generate random perturbation around best or initial
        if iteration == 0:
            x0 = best_h.copy()
        else:
            # random perturbation of best or initial
            base = best_h if np.random.rand() < 0.7 else h0
            sigma = 0.2 * (0.5 ** (iteration // 20))  # decay over iterations
            x0 = base + sigma * np.random.randn(n)
        # Project onto feasible region
        x0 = project_onto_box_simplex(x0, target_sum, 0.0, 1.0)

        # Run local optimization
        res = local_optimize(x0)

        if res.success:
            h_candidate = np.clip(res.x, 0.0, 1.0)  # just ensure bounds
            # enforce sum exactly via small correction
            h_candidate = project_onto_box_simplex(h_candidate, target_sum, 0.0, 1.0)
            c5_val = evaluate_c5(h_candidate)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_candidate.copy()
        iteration += 1

    # final rounding/casting
    return best_h, best_c5, n

import time
import numpy as np
from scipy.optimize import minimize

def evaluate_c5(h):
    """Return max_k ∫ h(x)(1−h(x+k))dx approximated on a grid."""
    dx = 2.0 / h.size
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def project_onto_box_simplex(v, target, lo=0.0, hi=1.0, tol=1e-12, max_iter=100):
    """
    Euclidean projection of v onto the set {x | lo ≤ x_i ≤ hi, sum x_i = target}.
    Uses bisection on the dual variable λ for the box-constrained simplex.
    """
    n = v.size
    if target < n * lo - 1e-9 or target > n * hi + 1e-9:
        raise ValueError("Target sum infeasible")
    # Bounds for λ large enough to saturate all at lo / hi
    lam_low = -np.max(v) - (hi - lo) - 1.0
    lam_high = (hi - lo) - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (lam_low + lam_high) / 2.0
        x = np.clip(v + lam, lo, hi)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s < target:
            lam_low = lam
        else:
            lam_high = lam
    # Return final approximation
    return np.clip(v + lam, lo, hi)

def sum_constraint_fun(x, target):
    """Equality constraint: sum(x) = target."""
    return np.sum(x) - target

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Determine problem size and initial vector
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
        n = h0.size
    else:
        n = 80
        h0 = np.full(n, 0.5, dtype=np.float64)

    target_sum = n / 2.0
    h0 = project_onto_box_simplex(h0, target_sum, 0.0, 1.0)

    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h)

    # Optimizer set‑up
    eq_cons = {'type': 'eq', 'fun': sum_constraint_fun, 'args': (target_sum,)}
    bounds = [(0.0, 1.0)] * n

    it = 0
    while True:
        if time.time() - start_time > budget_s - 5.0:
            break

        # Generate a feasible starting point
        if it == 0:
            x0 = best_h.copy()
        else:
            base = best_h if np.random.rand() < 0.7 else np.full(n, 0.5)
            sigma = 0.3 * (0.9 ** (it // 20))
            x0 = base + sigma * np.random.randn(n)
        x0 = project_onto_box_simplex(x0, target_sum, 0.0, 1.0)

        # Local refinement with SLSQP
        res = minimize(evaluate_c5, x0, method='SLSQP', bounds=bounds,
                       constraints=[eq_cons],
                       options={'ftol': 1e-12, 'maxiter': 600, 'disp': False})

            cand_c5 = evaluate_c5(cand)
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = cand.copy()
        it += 1

    return best_h, best_c5, n
