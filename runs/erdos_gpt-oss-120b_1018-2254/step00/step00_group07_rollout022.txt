def _max_overlap(h, dx):
    """Return max_{lag} ∫ h(x)*(1-h(x+lag)) dx approximated on the grid."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * dx

    best_val = cur_val

    max_no_improve = 2000

    best_val = _max_overlap(best_h, dx)

    start = time.time()
    end = start + budget_s * 0.95  # safety margin

    # Main optimisation loop: alternate refinement and random restarts
    while time.time() < end:
        # 30 % chance to refine the current best, otherwise start from fresh random binary
        if random.random() < 0.3:
            start_vec = best_h.copy()
        else:
            start_vec = np.zeros(n, dtype=float)
            idx = np.random.choice(n, n // 2, replace=False)
            start_vec[idx] = 1.0

        cand_h, cand_val = _local_optimize(start_vec, dx, end)

        if cand_val < best_val - 1e-12:
            best_h, best_val = cand_h, cand_val
            if best_val < 0.3808:
                break

    return (best_h.tolist(), float(best_val), n)
