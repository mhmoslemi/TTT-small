import time

def repair_h(v: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and enforce sum(v) == target by a water‑filling scheme."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(10):
        total = v.sum()
        diff = target - total
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v < 1 - 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            inc = diff / cnt
            # increase each allowed entry without exceeding 1
            incs = np.minimum(inc, 1 - v[mask])
            v[mask] += incs
        else:
            mask = v > 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            dec = (-diff) / cnt
            decs = np.minimum(dec, v[mask])
            v[mask] -= decs
    # final tiny adjustment if rounding left a residual
    total = v.sum()
    diff = target - total
    if abs(diff) > 1e-9:
        # try to adjust a non‑boundary entry
        interior = np.where((v > 0) & (v < 1))[0]
        if interior.size:
            i = interior[0]
            v[i] = max(0.0, min(1.0, v[i] + diff))
        else:
            # fallback: adjust first entry (may violate bound slightly, but later clipping fixes)
            v[0] = max(0.0, min(1.0, v[0] + diff))
    return v


def evaluate_candidate(h: np.ndarray, target: float):
    """Repair h and compute its C₅ value using the pre‑imported evaluator."""
    h_rep = repair_h(h, target)
    c_val = evaluate_erdos_solution(h_rep)
    return h_rep, c_val


def mutate_targeted(h: np.ndarray, shift: int, max_delta: float = 0.2) -> np.ndarray:
    """Transfer a small amount from a position i to i+shift to lower the worst overlap."""
    n = h.shape[0]
    cand = h.copy()
    if shift >= 0:
        i_start, i_end = 0, n - shift
    else:
        i_start, i_end = -shift, n
    if i_end > i_start:
        i = random.randint(i_start, i_end - 1)
        j = i + shift
        # amount that can be moved while staying in bounds
        move = min(cand[i], 1.0 - cand[j], max_delta)
        if move > 1e-12:
            delta = random.random() * move
            cand[i] -= delta
            cand[j] += delta
    return cand


def mutate_noise(h: np.ndarray, sigma: float = 0.05) -> np.ndarray:
    """Add Gaussian noise."""
    return h + sigma * np.random.randn(h.shape[0])


def mutate_swap(h: np.ndarray) -> np.ndarray:
    """Swap two entries (preserves the total sum for binary vectors)."""
    cand = h.copy()
    n = cand.shape[0]
    i, j = random.sample(range(n), 2)
    cand[i], cand[j] = cand[j], cand[i]
    return cand
