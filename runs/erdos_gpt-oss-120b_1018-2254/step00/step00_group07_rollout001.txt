def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    from scipy.optimize import minimize

    np.random.seed(seed)
    start_time = time.time()
    # Determine n_points
    try:
        n_points = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        n_points = 80
        h0 = np.full(n_points, 0.5, dtype=float)  # uniform meets sum
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h):
        # expecting numpy array
        return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

    # ensure initial feasible:
    h0 = project_onto_box_simplex(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # random search
    # Determine loop count based on time
    max_random_iters = 2000  # just a cap.
    i = 0
    while i < max_random_iters:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.6:  # use 60% time for random search
            break
        # generate random feasible point
        v = np.random.rand(n_points)
        h_candidate = project_onto_box_simplex(v, target_sum)
        c5_val = compute_c5(h_candidate)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_candidate.copy()
        i += 1

    # local refinement using SLSQP from best_h, maybe multiple restarts near best
    # Define constraint
    eq_cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}
    bounds = [(0.0, 1.0)] * n_points

    # Define objective wrapper
    def obj(h):
        return compute_c5(h)

    # Use SLSQP with maxiter dependent on remaining time
    remaining = budget_s - (time.time() - start_time)
    # We'll divide remaining time among a few refinements
    num_refine = 5
    for r in range(num_refine):
        if remaining <= 0:
            break
        # start from best_h with perturbation
        start = best_h + np.random.randn(n_points) * 0.05

def _project_onto_box_simplex(v, target, lo=0.0, hi=1.0, max_iter=200):
    """Project v onto the intersection of the box [lo,hi]^n and the hyperplane sum=v_target."""
    y = np.clip(v, lo, hi).astype(float)
    for _ in range(max_iter):
        total = y.sum()
        diff = target - total
        if abs(diff) < 1e-12:
            break
        free = (y > lo + 1e-12) & (y < hi - 1e-12)
        if not np.any(free):
            break
        delta = diff / np.count_nonzero(free)
        y[free] += delta
        y = np.clip(y, lo, hi)
    # final tiny correction if needed
    diff = target - y.sum()
    if abs(diff) > 1e-9:
        free_idx = np.where((y > lo + 1e-12) & (y < hi - 1e-12))[0]
        if free_idx.size:
            i = free_idx[0]
            y[i] = min(max(y[i] + diff, lo), hi)
        else:
            # adjust any coordinate that stays within bounds
            for i in range(len(y)):
                if lo < y[i] < hi:
                    y[i] = min(max(y[i] + diff, lo), hi)
                    break
    return y

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Determine problem size and initial point
    try:
        n_points = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        n_points = 80
        h0 = np.full(n_points, 0.5, dtype=float)  # uniform satisfies sum constraint

    target_sum = n_points / 2.0          # integral constraint (dx * sum = 1)
    dx = 2.0 / n_points                  # grid spacing

    def _c5_of_h(h):
        """Compute C5 for a given step vector."""
        return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

    # Make sure initial vector is feasible
    h0 = _project_onto_box_simplex(h0, target_sum)
    best_h = h0.copy()
    best_c5 = _c5_of_h(best_h)

    # ----------------------------------------------------------------------
    # Phase 1: Random exploration (≈60% of allocated time)
    # ----------------------------------------------------------------------
    max_random_iters = 3000
    i = 0
    while i < max_random_iters:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.6:
            break
        # generate a feasible random point
        rnd = np.random.rand(n_points)
        cand = _project_onto_box_simplex(rnd, target_sum)
        c5_val = _c5_of_h(cand)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = cand.copy()
        i += 1

    # ----------------------------------------------------------------------
    # Phase 2: Local refinement with SLSQP
    # ----------------------------------------------------------------------
    bounds = [(0.0, 1.0)] * n_points
    eq_con = {'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}
    def obj(h):
        return _c5_of_h(h)

    # Allocate remaining budget among a few refinements
    remaining = budget_s - (time.time() - start_time)
    refinements = 5
    for r in range(refinements):
        if remaining <= 0:
            break
        # start from the current best perturbed a little
        start = best_h + np.random.randn(n_points) * 0.05
        start = np.clip(start, 0.0, 1.0)
        start = _project_onto_box_simplex(start, target_sum)

        # heuristic number of iterations based on remaining time
        maxiter = max(30, int(min(400, remaining * 5)))
        res = minimize(obj, start, method='SLSQP', bounds=bounds,
                       constraints=eq_con,
                       options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
        h_opt = _project_onto_box_simplex(res.x, target_sum)
        c5_opt = _c5_of_h(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        remaining = budget_s - (time.time() - start_time)

    # Final safeguard: enforce feasibility exactly and recompute C5
    best_h = _project_onto_box_simplex(best_h, target_sum)
    best_c5 = _c5_of_h(best_h)

    return (best_h.tolist(), float(best_c5), n_points)
