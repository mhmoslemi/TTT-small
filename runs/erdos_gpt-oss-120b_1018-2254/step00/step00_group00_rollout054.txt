def repair(v, target):
    v = np.clip(v, 0.0, 1.0)
    # Compute sum
    cur = v.sum()
    # while diff > tolerance
    for _ in range(10):
        diff = target - cur
        if abs(diff) < 1e-9:
            break
        # indices that can be moved in direction of diff
        if diff > 0:
            mask = v < 1.0 - 1e-12
        else:
            mask = v > 1e-12
        n = np.count_nonzero(mask)
        if n == 0:
            break
        # delta per element
        delta = diff / n
        # Apply limited by bounds
        v[mask] += delta
        # Clip again
        np.clip(v, 0.0, 1.0, out=v)
        cur = v.sum()
    # Finally ensure sum exactly target within tolerance by distributing any remaining small diff
    # Might need final adjustment to one element (avoid bound)
    if abs(v.sum() - target) > 1e-6:
        diff = target - v.sum()
        # find any index that can accept diff
        idx = np.where((diff > 0) & (v < 1.0 - 1e-12) | (diff < 0) & (v > 1e-12))[0]
        if len(idx) > 0:
            i = idx[0]
            v[i] += diff
            v[i] = np.clip(v[i], 0.0, 1.0)
    return v
```
