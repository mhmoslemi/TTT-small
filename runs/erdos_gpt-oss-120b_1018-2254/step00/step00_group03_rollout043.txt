import numpy as np
import time, math

def max_corr_sum(h):
    # h is numpy array of 0/1 ints
    corr = np.correlate(h, 1 - h, mode='full')
    return int(corr.max())

def compute_bound(corr_sum, n):
    return corr_sum * (2.0 / n)

def hill_climb_binary(rng, n, max_iters, time_deadline):
    N = n // 2
    # random init
    ones = rng.choice(n, size=N, replace=False)
    h = np.zeros(n, dtype=np.int8)
    h[ones] = 1
    # maintain index lists
    ones_idx = list(ones)
    zeros_idx = [i for i in range(n) if h[i]==0]
    cur_c = max_corr_sum(h)
    best_h_local = h.copy()
    best_c_local = cur_c
    no_improve = 0
    # for annealing we can define temperature schedule if needed
    while True:
        if time.time() >= time_deadline:
            break
        if no_improve >= max_iters:
            break
        # random swap
        i_idx = rng.integers(len(ones_idx))
        i = ones_idx[i_idx]
        j_idx = rng.integers(len(zeros_idx))
        j = zeros_idx[j_idx]
        # swap
        h[i] = 0
        h[j] = 1
        # update index lists
        ones_idx[i_idx] = j
        zeros_idx[j_idx] = i
        new_c = max_corr_sum(h)
        if new_c <= cur_c:
            cur_c = new_c
            no_improve = 0
            if new_c < best_c_local:
                best_c_local = new_c
                best_h_local = h.copy()
        else:
            # revert
            h[i] = 1
            h[j] = 0
            ones_idx[i_idx] = i
            zeros_idx[j_idx] = j
            no_improve += 1
    return best_h_local, best_c_local

import numpy as np
import time
import math

def max_corr_sum(h):
    """Maximum overlap sum for binary vector h."""
    # h assumed 0/1 integer array
    corr = np.correlate(h, 1 - h, mode="full")
    return int(corr.max())

def hill_climb_binary(rng, n, N, h_init, start_time, deadline,
                      max_attempts=30000, max_no_improve=10000,
                      max_T=5.0, min_T=0.01):
    """
    Simple hill‑climbing / simulated‑annealing on a binary vector.
    Returns best (h, max_corr) found before deadline.
    """
    h = h_init.copy()
    ones_idx = list(np.flatnonzero(h))
    zeros_idx = [i for i in range(n) if h[i] == 0]

    cur_c = max_corr_sum(h)
    best_h = h.copy()
    best_c = cur_c

    attempts = 0
    no_improve = 0
    while attempts < max_attempts and time.time() < deadline:
        attempts += 1

        # pick a random 1 and a random 0 to swap
        i_idx = rng.integers(len(ones_idx))
        j_idx = rng.integers(len(zeros_idx))
        i = ones_idx[i_idx]
        j = zeros_idx[j_idx]

        # perform swap
        h[i] = 0
        h[j] = 1
        ones_idx[i_idx] = j
        zeros_idx[j_idx] = i

        new_c = max_corr_sum(h)
        delta = new_c - cur_c
        accept = False
        if delta <= 0:
            accept = True
        else:
            # temperature based on overall progress
            elapsed = time.time() - start_time
            total = max(deadline - start_time, 1e-9)
            progress = min(elapsed / total, 1.0)
            T = max_T * (1 - progress) + min_T * progress
            if rng.random() < math.exp(-delta / T):
                accept = True

        if accept:
            cur_c = new_c
            no_improve = 0
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1
            h[j] = 0
            ones_idx[i_idx] = i
            zeros_idx[j_idx] = j
            no_improve += 1

        if no_improve >= max_no_improve:
            break

    return best_h, best_c
