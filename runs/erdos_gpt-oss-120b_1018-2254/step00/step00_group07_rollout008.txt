    rng = np.random.default_rng(seed)
    start = time.time()
    # Determine initial vector and n_points
    init = globals().get('initial_h_values', None)
    if init is not None:
        h0 = np.array(init, dtype=float)

    best_h = h0.copy()

        best_score = float(_objective(best_h))
    except Exception:
        best_score = float('inf')

        try:
            return float(_objective(vec))
        except Exception:
            return float('inf')
    # Build start set

    for _ in range(5):

    best_score = _safe_evaluate(best_h)

    # ------------------------------------------------------------------
    # Build a list of starting points
    # ------------------------------------------------------------------
    starts = [best_h]

    # perturbations of the parent solution
    for _ in range(5):
        noise = rng.normal(scale=0.2, size=n)
        cand = np.clip(best_h + noise, 0.0, 1.0)
        starts.append(cand)

    # random Dirichlet-like starts
    for _ in range(5):
        rand = rng.random(n)
        cand = rand / rand.sum() * target_sum
        # enforce box constraints while approximately keeping the sum
        over = cand > 1.0
        if np.any(over):
            excess = cand[over] - 1.0
            cand[over] = 1.0
            under = ~over
            if np.any(under):
                cand[under] -= excess.sum() / under.sum()
        cand = np.clip(cand, 0.0, 1.0)
        cand = cand * (target_sum / cand.sum())
        starts.append(cand)

    # ------------------------------------------------------------------
    # Local optimization from each start
    # ------------------------------------------------------------------
    for x0 in starts:
        if time.time() - start_time > budget_s - 5:
            break

        res = minimize(
            _objective,
            x0,
            method='SLSQP',
            bounds=bounds,
            constraints=[constraints],
            options={'maxiter': 200, 'ftol': 1e-9, 'disp': False}
        )
        h_opt = np.clip(res.x, 0.0, 1.0)

        # small correction of the sum constraint (if needed)
        sum_diff = h_opt.sum() - target_sum
        if abs(sum_diff) > 1e-8:
            h_opt -= sum_diff / n
            h_opt = np.clip(h_opt, 0.0, 1.0)

        score = _safe_evaluate(h_opt)
        if score < best_score:
            best_score = score
            best_h = h_opt.copy()

        if best_score < 0.3808:
            break

    # ------------------------------------------------------------------
    # Light random local search while time permits
    # ------------------------------------------------------------------
    while time.time() - start_time < budget_s - 1:
        cand = np.clip(best_h + rng.normal(scale=0.05, size=n), 0.0, 1.0)
        cand = cand * (target_sum / cand.sum())
        cand = np.clip(cand, 0.0, 1.0)
        score = _safe_evaluate(cand)
        if score < best_score:
            best_score = score
            best_h = cand.copy()
        if best_score < 0.3808:
            break

    return (best_h.tolist(), float(best_score), n)
