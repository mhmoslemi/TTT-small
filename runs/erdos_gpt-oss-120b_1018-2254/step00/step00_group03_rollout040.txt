def _evaluate_h(h):
    try:
        return evaluate_erdos_solution(h)
    except TypeError:

            return evaluate_erdos_solution(h, len(h))

def _evaluate_h(h):
    """Wrapper for the pre‑imported evaluator."""
    try:
        return evaluate_erdos_solution(h)

            return evaluate_erdos_solution(h, len(h))
        except Exception:
            h_arr = np.asarray(h, dtype=float)
            n = h_arr.shape[0]
            corr = np.correlate(h_arr, 1 - h_arr, mode='full')
            return np.max(corr) * (2.0 / n)

def _local_improve(h, N, time_limit, start_time):
    """Simple hill‑climbing / simulated‑annealing style improvement."""
    n = h.shape[0]
    best_h = h.copy()
    best_val = _evaluate_h(best_h)
    # Main loop respects the allocated time budget.
    while True:
        if time.time() - start_time >= time_limit:
            break
        # correlation and dominant shift
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = int(np.argmax(corr))
        shift = max_idx - (n - 1)
        # try a targeted move that reduces the dominant shift
        moved = False
        if shift != 0:
            i = np.random.randint(0, n)
            j = i + shift
            if 0 <= j < n:
                # move mass from i to j (or opposite) while staying in [0,1]
                delta = min(best_h[i], 1.0 - best_h[j])

                    cand = best_h.copy()
                    cand[i] -= d
                    cand[j] += d
                    moved = True
