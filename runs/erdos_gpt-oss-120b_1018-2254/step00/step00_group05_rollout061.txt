def _max_corr(vec: np.ndarray) -> float:
    """Maximum correlation value max_k ∫ h(x)(1−h(x+k))dx approximated on a grid."""
    return float(np.max(np.correlate(vec, 1.0 - vec, mode="full")))


def _triangular_h(N: int) -> np.ndarray:
    """Symmetric triangular step function on [0,2] (height 1, base 2) scaled to sum N/2."""
    dx = 2.0 / N
    # sample at mid‑points
    x = (np.arange(N) + 0.5) * dx
    h = np.maximum(0.0, 1.0 - np.abs(x - 1.0))

        return np.full(N, 0.5)

    # enforce exact bounds and sum
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s != target:
        h *= target / s
    return h
