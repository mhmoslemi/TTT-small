def compute_c5(h):

def gradient_max_shift(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    k = np.argmax(corr)  # index for max correlation
    shift = k - (n - 1)   # integer shift
    g = np.full(n, dx)  # derivative of sum_i h_i
    if shift >= 0:
        s = shift
        # subtract h[i+s] from g[i] for i< n-s
        g[:n - s] -= dx * h[s:]
        # subtract h[i-s] from g[i] for i>=s
        g[s:] -= dx * h[:n - s]
    else:

        # for i >= s: subtract h[i - s]
        g[s:] -= dx * h[:n - s]
        # for i <= n-1-s: subtract h[i + s]
        g[:n - s] -= dx * h[s:]
    return g, shift

def project(x, target):
    x = np.clip(x, 0.0, 1.0)
    # water-filling iterative
    for _ in range(20):
        current_sum = x.sum()
        diff = target - current_sum
        if abs(diff) < 1e-12:
            break
        free = (x > 1e-12) & (x < 1 - 1e-12)
        nfree = free.sum()
        if nfree == 0:
            # no free variables, adjust by distributing to all (will breach)
            break
        delta = diff / nfree
        # add delta, clip
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
    # any residual diff? distribute to a variable not at bound
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        # try to adjust first variable within bounds
        for i in range(len(x)):
            lo = 0.0 - x[i]
            hi = 1.0 - x[i]
            if lo <= diff <= hi:
                x[i] += diff
                break
    # Ensure bounds again
    x = np.clip(x, 0.0, 1.0)
    return x

    # raw correlation without scaling for speed

    g = np.full(n, dx)
    if shift >= 0:

        # i from 0 to n-s-1: subtract h[i+s] from g[i]
        g[:n - s] -= dx * h[s:]

        g[s:] -= dx * h[:n - s]

        g[s:] -= dx * h[:n - s]
        # i from 0 to n-s-1: subtract h[i+s]

def _project_vec(v, target):
    """Clip v to [0,1] and adjust to have sum == target."""
    x = np.clip(v, 0.0, 1.0)
    for _ in range(20):
        diff = target - x.sum()
        if abs(diff) < 1e-12:
            break
        free = (x > 1e-12) & (x < 1 - 1e-12)
        nfree = free.sum()
        if nfree == 0:
            break
        delta = diff / nfree
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
    # tiny residual fix
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        for i in range(len(x)):
            lo, hi = -x[i], 1 - x[i]
            if lo <= diff <= hi:
                x[i] += diff
                break
    return np.clip(x, 0.0, 1.0)
