        dx = 2.0 / len(arr)
        corr = np.correlate(arr, 1.0 - arr, mode='full')
        return np.max(corr) * dx

        n = len(best_h)

            i_end = n

        region = np.arange(i_start, i_end)

import time
import numpy as np

def compute_c5(arr):
    dx = 2.0 / arr.shape[0]
    corr = np.correlate(arr, 1.0 - arr, mode='full')
    return np.max(corr) * dx

def compute_c5_and_worst(arr):
    dx = 2.0 / arr.shape[0]
    corr = np.correlate(arr, 1.0 - arr, mode='full')
    idx = np.argmax(corr)
    return corr[idx] * dx, idx

def random_binary_vec(n, rng):
    arr = np.zeros(n, dtype=int)
    arr[:n//2] = 1
    rng.shuffle(arr)
    return arr

def make_feasible_from_initial(initial):
    n = len(initial)
    # Ensure binary:
    # take top n//2 indices as 1
    idx_sorted = np.argsort(-initial)  # descending order
    arr = np.zeros(n, dtype=int)
    target = n // 2
    arr[idx_sorted[:target]] = 1
    return arr

def propose_candidate(arr, d, rng):
    n = arr.shape[0]
    if d >= 0:
        i_start = 0
        i_end = n - d
    else:
        i_start = -d
        i_end = n
    region = np.arange(i_start, i_end)
    # mismatches
    region_vals = arr[region]
    shifted_vals = arr[region + d]
    # type A: 1 -> region, 0 in shifted
    mask_A = (region_vals == 1) & (shifted_vals == 0)
    mism_A = region[mask_A]
    # type B: 0 in region, 1 in shifted
    mask_B = (region_vals == 0) & (shifted_vals == 1)
    mism_B = region[mask_B]

        i = rng.choice(ones)
        j = rng.choice(zeros)

    cand[i], cand[j] = cand[j], cand[i]

import time
import numpy as np

def compute_c5(arr: np.ndarray) -> float:
    """Return C5 value for a step function given by arr."""
    n = arr.shape[0]
    dx = 2.0 / n
    corr = np.correlate(arr, 1.0 - arr, mode="full")
    return float(np.max(corr) * dx)


def compute_c5_and_worst(arr: np.ndarray):
    """Return (C5, worst_index) where worst_index is the index in the full
    correlation that attains the maximum."""
    n = arr.shape[0]
    dx = 2.0 / n
    corr = np.correlate(arr, 1.0 - arr, mode="full")
    worst_idx = int(np.argmax(corr))
    return float(corr[worst_idx] * dx), worst_idx


def random_binary_vec(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a binary vector of length n with exactly n//2 ones."""
    h = np.zeros(n, dtype=int)
    h[: n // 2] = 1
    rng.shuffle(h)
    return h


def make_feasible_from_initial(arr: np.ndarray) -> np.ndarray:
    """Turn an arbitrary initial vector into a feasible binary one
    (sum = n//2, values in {0,1})."""
    n = arr.shape[0]
    target = n // 2
    idx = np.argsort(-arr)  # descending
    h = np.zeros(n, dtype=int)
    h[idx[:target]] = 1
    return h


def propose_candidate(arr: np.ndarray, d: int, rng: np.random.Generator) -> np.ndarray:
    """Propose a new binary vector by swapping two entries.
    If possible, swap a mismatched pair for the worst shift d."""
    n = arr.shape[0]
    # overlapping region for shift d
    if d >= 0:
        i_start, i_end = 0, n - d
    else:
        i_start, i_end = -d, n
    region = np.arange(i_start, i_end)

    # locate mismatches for the worst shift
    region_vals = arr[region]
    shifted_vals = arr[region + d]

    mask_A = (region_vals == 1) & (shifted_vals == 0)
    mask_B = (region_vals == 0) & (shifted_vals == 1)
    mism_A = region[mask_A]
    mism_B = region[mask_B]

    if mism_A.size > 0 and mism_B.size > 0:
        i = rng.choice(mism_A)
        j = rng.choice(mism_B)
    else:
        # generic swap of any 1 and 0
        ones = np.where(arr == 1)[0]
        zeros = np.where(arr == 0)[0]
        i = rng.choice(ones)
        j = rng.choice(zeros)

    cand = arr.copy()
    cand[i], cand[j] = cand[j], cand[i]
    return cand


def continuous_move(arr: np.ndarray, rng: np.random.Generator, max_step: float = 0.5) -> np.ndarray:
    """Move a small amount of mass from one coordinate to another while preserving
    the total sum and staying in [0,1]."""
    n = arr.shape[0]
    # choose a source with positive mass and a sink with room
    src = np.where(arr > 1e-12)[0]
    dst = np.where(arr < 1.0 - 1e-12)[0]
    if src.size == 0 or dst.size == 0:
        return arr.copy()
    i = rng.choice(src)
    j = rng.choice(dst)
    max_delta = min(arr[i], 1.0 - arr[j])
    if max_delta <= 0.0:
        return arr.copy()
    delta = rng.random() * max_delta * max_step
    cand = arr.copy()
    cand[i] -= delta
    cand[j] += delta
    return cand


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a (h_values, c5_bound, n_points) tuple with an improved
    upper bound for C5."""
    rng = np.random.default_rng(seed)
    start_time = time.time()
    max_time = budget_s * 0.96  # leave a small safety margin

    # ------------------------------------------------------------------
    # Initial vector
    try:
        init_arr = np.array(initial_h_values, dtype=float)
        n = init_arr.shape[0]
        cur_h = make_feasible_from_initial(init_arr).astype(int)
    except Exception:
        n = 70
        cur_h = random_binary_vec(n, rng).astype(int)

    # Ensure basic feasibility
    cur_h = cur_h.astype(float)  # float works for both phases
    cur_c5, _ = compute_c5_and_worst(cur_h)

    best_h = cur_h.copy()
    best_c5 = cur_c5

    # ------------------------------------------------------------------
    # Parameters
    max_no_improve = 5000
    no_improve = 0
    phase_boundary = start_time + max_time * 0.75  # binary phase then refinement

    while True:
        now = time.time()
        if now - start_time >= max_time:
            break

        if now < phase_boundary:
            # ----------- Binary hill climbing phase --------------------
            cur_c5, worst_idx = compute_c5_and_worst(cur_h)
            d = worst_idx - (n - 1)

            cand_h = propose_candidate(cur_h, d, rng)
            cand_c5, _ = compute_c5_and_worst(cand_h)

            if cand_c5 < cur_c5 - 1e-12:
                cur_h = cand_h
                cur_c5 = cand_c5
                no_improve = 0
                if cand_c5 < best_c5 - 1e-12:
                    best_h = cand_h.copy()
                    best_c5 = cand_c5
            else:
                no_improve += 1

            if no_improve >= max_no_improve:
                # restart with a fresh random binary vector
                cur_h = random_binary_vec(n, rng).astype(float)
                cur_c5, _ = compute_c5_and_worst(cur_h)
                no_improve = 0

        else:
            # -------------- Continuous refinement phase -----------------
            cand_h = continuous_move(cur_h, rng, max_step=0.5)
            cand_c5, _ = compute_c5_and_worst(cand_h)

            if cand_c5 < cur_c5 - 1e-12:
                cur_h = cand_h
                cur_c5 = cand_c5
                if cand_c5 < best_c5 - 1e-12:
                    best_h = cand_h.copy()
                    best_c5 = cand_c5
            # No restart logic in this phase; simply keep searching.

    # Final safety recomputation
    best_c5 = compute_c5(best_h.astype(float))

    return (best_h.astype(float).tolist(), float(best_c5), n)
