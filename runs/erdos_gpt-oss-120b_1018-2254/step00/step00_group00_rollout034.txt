import numpy as np, time, random, math

def compute_c5(h):
    # h: 1D np.ndarray of dtype float or int 0-1
    N = h.size
    # Convert to float for np.correlate
    # Using float ensures cast but produce float results, but product of 0/1 is integer.
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = corr.max()
    return float(max_corr) * (2.0 / N)

def max_mismatch_count(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return int(corr.max())

def optimize(N, rng, time_limit, start_h=None):
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    # random sampling
    n_samples = 2000 if N >= 180 else (4000 if N >= 90 else 8000)
    for _ in range(n_samples):
        if time.time() - start_time > time_limit * 0.5:
            break
        # generate random balanced binary
        arr = np.zeros(N, dtype=np.float64)
        arr[:N//2] = 1.0
        rng.shuffle(arr)
        c5 = compute_c5(arr)
        if c5 < best_c5:
            best_c5 = c5
            best_h = arr.copy()
    # If have start_h and it is better, set as best.
    if start_h is not None:
        c5 = compute_c5(start_h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = start_h.copy()
    # hill climbing:
    # compute list of ones indices and zeros.
    # Continue until time or no improvement after many tries.
    max_hc_iters = 20000
    ones = list(np.where(best_h == 1)[0])
    zeros = list(np.where(best_h == 0)[0])
    for iter_ in range(max_hc_iters):
        if time.time() - start_time > time_limit:
            break
        # random swap:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # perform swap
        best_h[i] = 0
        best_h[j] = 1
        # compute new c5
        new_c5 = compute_c5(best_h)
        if new_c5 < best_c5 - 1e-12:
            # accept
            best_c5 = new_c5
            # update lists:
            ones.remove(i); zeros.append(i)
            zeros.remove(j); ones.append(j)
        else:
            # revert
            best_h[i] = 1
            best_h[j] = 0
        # if best_c5 already <= target, break
        if best_c5 <= 0.3808:
            break
    return best_h, best_c5
```

def hill_climb_targeted(h, rng, time_limit):
    N = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start = time.time()
    while time.time() - start < time_limit:
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        max_corr = corr.max()
        # compute current c5
        cur_c5 = max_corr * 2.0 / N
        # if already <=target break maybe
        # find indices where corr achieves max (first)
        idx = int(np.argmax(corr))  # index in full array
        s = idx - (N - 1)  # shift value
        # find mismatched positions for shift s
        if s >= 0:
            mism1 = np.where((best_h[:N-s] == 1) & (best_h[s:] == 0))[0]
            mism0 = np.where((best_h[:N-s] == 0) & (best_h[s:] == 1))[0]
        else:
            t = -s
            mism1 = np.where((best_h[t:] == 1) & (best_h[:N-t] == 0))[0] + t
            mism0 = np.where((best_h[t:] == 0) & (best_h[:N-t] == 1))[0] + t
        if mism1.size == 0 or mism0.size == 0:
            # cannot improve? break
            break
        i = rng.choice(mism1)
        j = rng.choice(mism0)
        # swap
        best_h[i], best_h[j] = 0, 1
        new_c5 = compute_c5(best_h)
        if new_c5 <= best_c5:

def optimize(N, rng, time_limit, start_h=None):
    best_h = None; best_c5 = float('inf')
    start_time = time.time()
    # allocate time budget for random sampling vs hill climbing
    # let's use 20% for random sampling, 80% for hill climbing.
    sampling_time = time_limit * 0.2
    hill_time = time_limit - sampling_time
    # random sampling
    while time.time() - start_time < sampling_time:
        # generate random balanced
        arr = np.zeros(N, dtype=np.float64)
        arr[:N//2] = 1
        rng.shuffle(arr)
        c5 = compute_c5(arr)
        if c5 < best_c5:
            best_c5 = c5
            best_h = arr.copy()
    # include start_h if better
    if start_h is not None:
        c5 = compute_c5(start_h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = start_h.copy()
    # hill climbing on best_h
    if best_h is not None:
        # given remaining time for hill climbing
        h_opt, c5_opt = hill_climb_targeted(best_h, rng, hill_time)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
    return best_h, best_c5

def compute_c5(h):
    N = h.shape[0]
    max_corr = np.max(np.correlate(h, 1.0 - h, mode='full'))
    return float(max_corr) * (2.0 / N)

def random_binary(N, rng):
    arr = np.zeros(N, dtype=np.float64)
    arr[:N//2] = 1.0
    rng.shuffle(arr)
    return arr

def hill_climb_targeted(h, rng, time_limit):
    N = h.shape[0]
    best = h.copy()
    best_c5 = compute_c5(best)
    start = time.time()
    # Precompute target integer mismatch threshold? We'll just use c5.
    while time.time() - start < time_limit:
        # compute correlation and find shift of maximal mismatch
        corr = np.correlate(best, 1.0 - best, mode='full')
        max_corr = corr.max()
        cur_c5 = max_corr * (2.0 / N)
        if cur_c5 <= TARGET:
            break
        idx = int(np.argmax(corr))
        shift = idx - (N - 1)  # shift value (can be negative)
        if shift >= 0:
            # positions i where best[i]=1 and best[i+shift]=0
            mism1 = np.where((best[:N-shift] == 1) & (best[shift:] == 0))[0]
            mism0 = np.where((best[:N-shift] == 0) & (best[shift:] == 1))[0]
        else:
            t = -shift
            mism1 = np.where((best[t:] == 1) & (best[:N-t] == 0))[0] + t
            mism0 = np.where((best[t:] == 0) & (best[:N-t] == 1))[0] + t
        if mism1.size == 0 or mism0.size == 0:
            break
        i = mism1[rng.integers(mism1.size)]
        j = mism0[rng.integers(mism0.size)]

import numpy as np
import time

_TARGET = 0.3808  # target C5 bound


def _c5_of(h: np.ndarray) -> float:
    """Return C5 value for binary step function h."""
    n = h.shape[0]
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return float(max_corr) * (2.0 / n)


def _random_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a binary array of length n with exactly n/2 ones."""
    arr = np.zeros(n, dtype=np.float64)
    arr[: n // 2] = 1.0
    rng.shuffle(arr)
    return arr


def _hill_climb_targeted(start_h: np.ndarray, rng: np.random.Generator, time_limit: float):
    """Greedy hill‑climb focusing on the current worst shift."""
    n = start_h.shape[0]
    h = start_h.copy()
    best_c5 = _c5_of(h)
    start = time.time()

    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = corr.max()
        cur_c5 = max_corr * (2.0 / n)
        if cur_c5 <= _TARGET:
            break

        idx = int(np.argmax(corr))
        shift = idx - (n - 1)  # shift value (negative allowed)

        if shift >= 0:
            mism1 = np.where((h[: n - shift] == 1) & (h[shift:] == 0))[0]
            mism0 = np.where((h[: n - shift] == 0) & (h[shift:] == 1))[0]
        else:
            t = -shift
            mism1 = np.where((h[t:] == 1) & (h[: n - t] == 0))[0] + t
            mism0 = np.where((h[t:] == 0) & (h[: n - t] == 1))[0] + t

        if mism1.size == 0 or mism0.size == 0:
            break

        i = mism1[rng.integers(mism1.size)]
        j = mism0[rng.integers(mism0.size)]

        # perform swap
        h[i], h[j] = 0.0, 1.0
        new_c5 = _c5_of(h)

        if new_c5 < best_c5:
            best_c5 = new_c5
        else:
            # revert
            h[i], h[j] = 1.0, 0.0

    return h, best_c5


def _optimize_n(n: int, rng: np.random.Generator, budget: float):
    """Search for a good binary vector of length n within the given time."""
    start = time.time()
    best_h = None
    best_c5 = float("inf")

    # ---- random sampling (≈20% of budget) ----
    sample_budget = budget * 0.2
    max_samples = 3000
    for _ in range(max_samples):
        if time.time() - start > sample_budget:
            break
        cand = _random_binary(n, rng)
        c5 = _c5_of(cand)
        if c5 < best_c5:
            best_c5, best_h = c5, cand.copy()
            if best_c5 <= _TARGET:
                break

    # ---- hill climbing on the best sample (remaining budget) ----
    remaining = budget - (time.time() - start)
    if best_h is not None and remaining > 0:
        h_opt, c5_opt = _hill_climb_targeted(best_h, rng, remaining)
        if c5_opt < best_c5:
            best_c5, best_h = c5_opt, h_opt.copy()

    return best_h, best_c5


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    best_h, best_c5, best_n = None, float("inf"), None

    # ---- consider the provided initial construction if present ----
    init = globals().get("initial_h_values", None)
    if init is not None:
        try:
            init_arr = np.asarray(init, dtype=np.float64)
            c5_init = _c5_of(init_arr)
            if c5_init < best_c5:
                best_h, best_c5, best_n = init_arr.copy(), c5_init, init_arr.shape[0]
        except Exception:
            pass

    # ---- try several problem sizes, larger first ----
    for n in (180, 90, 45):
        if time.time() - t0 >= budget_s:
            break
        if best_c5 <= _TARGET:
            break

        remaining = budget_s - (time.time() - t0)
        # allocate a portion of remaining time for this size
        alloc = min(remaining, budget_s * 0.33)

        h_candidate, c5_candidate = _optimize_n(n, rng, alloc)
        if h_candidate is not None and c5_candidate < best_c5:
            best_h, best_c5, best_n = h_candidate.copy(), c5_candidate, n

    # ---- fallback (should rarely happen) ----
    if best_h is None:
        n = 45
        best_h = _random_binary(n, rng)
        best_c5 = _c5_of(best_h)
        best_n = n

    return best_h, best_c5, best_n
