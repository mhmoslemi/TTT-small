def project_to_bounds_and_sum(x, target):
    # Bisection to find lambda so sum(clip(x+lam,0,1)) = target
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo+hi)/2.0
        proj = np.clip(x + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(x + (lo+hi)/2.0, 0.0, 1.0)
    # adjust small residual
    residual = target - proj.sum()
    if abs(residual) > 1e-10:
        # add residual to some non-saturated entry
        # find indices where 0 < proj < 1
        idx = np.where((proj > 1e-12) & (proj < 1-1e-12))[0]
        if idx.size > 0:
            adjust_idx = idx[0]
            proj[adjust_idx] = np.clip(proj[adjust_idx] + residual, 0.0, 1.0)
        else:
            # all saturated; adjust first entry
            proj[0] = np.clip(proj[0] + residual, 0.0, 1.0)
    return proj
```

def propose_swap(h, delta):
    n = len(h)
    i = np.random.randint(n)
    j = np.random.randint(n)
    if i==j: return h.copy()
    # attempt to move delta from i to j
    new = h.copy()
    max_take = min(delta, new[i])
    max_give = min(delta, 1 - new[j])
    move = min(max_take, max_give)
    new[i] -= move
    new[j] += move
    # no need to project
    return new
```

import numpy as np
import time

def compute_c5(h, dx):
    # correlation
    # Use np.correlate - note returns length 2*n -1
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_bounds_and_sum(x, target):
    # bisection
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(x + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(x + (lo+hi)/2.0, 0.0, 1.0)
    residual = target - proj.sum()
    if abs(residual) > 1e-12:
        idx = np.where((proj > 1e-12) & (proj < 1-1e-12))[0]
        if idx.size > 0:
            i = idx[0]
            proj[i] = np.clip(proj[i] + residual, 0.0, 1.0)
        else:
            # saturate first element
            proj[0] = np.clip(proj[0] + residual, 0.0, 1.0)
    return proj

def run(seed=42, budget_s=1100, **kwargs):

import time

def _compute_c5(h, dx):
    """Return C5 = max correlation of h with (1-h) times dx."""
    # Full correlation (length 2*n-1)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project_bounds_sum(x, target):
    """Project vector x onto [0,1] box with exact sum = target using bisection."""
    lo, hi = -2.0, 2.0
    # 30 iterations give sufficient precision for our problem size
    for _ in range(30):
        mid = (lo + hi) * 0.5
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)
    # Fix any tiny residual to hit the target exactly
    residual = target - y.sum()
    if abs(residual) > 1e-12:
        free = np.where((y > 1e-12) & (y < 1.0 - 1e-12))[0]
        if free.size:
