import numpy as np
import time
import math

def _project_to_constraints(x: np.ndarray, target_sum: float) -> np.ndarray:
    """Project x onto the box [0,1]^n with sum equal to target_sum."""
    low, high = -1.0, 1.0  # bounds for the shift lambda
    for _ in range(60):   # enough for double precision
        mid = (low + high) * 0.5
        clipped = np.clip(x - mid, 0.0, 1.0)
        s = clipped.sum()
        if s > target_sum:
            low = mid
        else:
            high = mid
    return np.clip(x - (low + high) * 0.5, 0.0, 1.0)


def _c5_value(h: np.ndarray) -> float:
    """Compute C₅ = max_k ∫ h(x)(1−h(x+k))dx for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


def _gradient_worst_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """
    Gradient of the overlap term corresponding to a given integer shift.
    For O(shift) = Σ_i h[i]·(1−h[i+shift]), the gradient is:
        ∂O/∂h[i] = (1−h[i+shift])  if i+shift in range
        ∂O/∂h[i] = −h[i−shift]    if i-shift in range
    """
    n = h.shape[0]
    grad = np.zeros_like(h)
    if shift >= 0:
        m = n - shift
        grad[:m] += 1.0 - h[shift:]
        grad[shift:] += -h[:m]
    else:
        s = -shift
        m = n - s
        grad[s:] += 1.0 - h[:m]
        grad[:m] += -h[s:]
    return grad


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a step function h, its C₅ bound, and number of sample points."""
    np.random.seed(seed)
    start_time = time.time()

    # ----------------------------------------------------------------------
    # Acquire starting vector (if provided) and determine dimension.
    # ----------------------------------------------------------------------
    init = globals().get("initial_h_values")
    if init is not None:
        h0 = np.array(init, dtype=float)
        n_pts = h0.shape[0]
    else:
        n_pts = 47                     # a compact size that works well
        h0 = np.random.rand(n_pts)

    target_sum = n_pts / 2.0           # because ∫_0^2 h = 1 and dx = 2/n

    # ----------------------------------------------------------------------
    # Feasibility projection and initial evaluation.
    # ----------------------------------------------------------------------
    cur_h = _project_to_constraints(h0, target_sum)
    best_h = cur_h.copy()
    best_c5 = _c5_value(best_h)

    cur_c5 = best_c5

    # ----------------------------------------------------------------------
    # Simulated‑annealing style search parameters.
    # ----------------------------------------------------------------------
    T = 0.2               # magnitude of random perturbations
    Tmin = 1e-5
    decay = 0.9995        # exponential cooling each iteration
    steps_without_improve = 0
    restart_limit = 5000  # after this many steps without improvement we restart

    # ----------------------------------------------------------------------
    # Main optimisation loop.
    # ----------------------------------------------------------------------
    while time.time() - start_time < budget_s:
        # Identify the shift that currently attains the maximum overlap.
        full_corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        idx_max = int(np.argmax(full_corr))
        shift = idx_max - (n_pts - 1)

        # Gradient pointing **up** the worst overlap; we step opposite.
        grad = _gradient_worst_shift(cur_h, shift)

        # Random perturbation whose scale follows the temperature.
        noise = (np.random.rand(n_pts) * 2.0 - 1.0) * T
        beta = 0.5 * T                     # relative weight of gradient step
        cand = cur_h - beta * grad + noise

        # Enforce constraints.
        cand = _project_to_constraints(cand, target_sum)

        # Evaluate candidate.
        cand_c5 = _c5_value(cand)

        # Acceptance rule (Metropolis).
        if cand_c5 < cur_c5 or np.random.rand() < math.exp(
            -(cand_c5 - cur_c5) / max(T, 1e-12)
        ):
            cur_h, cur_c5 = cand, cand_c5
            if cand_c5 < best_c5:
                best_h, best_c5 = cand.copy(), cand_c5
                steps_without_improve = 0
            else:
                steps_without_improve += 1
        else:
            steps_without_improve += 1

        # Cooling.
        T = max(Tmin, T * decay)

        # Occasional restart to escape stagnation.
        if steps_without_improve >= restart_limit:
            cur_h = best_h + (np.random.rand(n_pts) * 2.0 - 1.0) * 0.1
            cur_h = _project_to_constraints(cur_h, target_sum)
            cur_c5 = best_c5
            T = 0.2
            steps_without_improve = 0

    # Final safety projection and exact bound recomputation.
    best_h = _project_to_constraints(best_h, target_sum)
    final_c5 = _c5_value(best_h)

    return (best_h, float(final_c5), n_pts)
