import time
import numpy as np

# ------------------- Helper Functions ------------------- #

def compute_c5(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)*(1-h(x+k)) dx for a discretized h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * float(np.max(corr))


def interpolate_to_n(src: np.ndarray, n_target: int) -> np.ndarray:
    """Linear interpolation of src (samples on [0,2)) onto n_target points."""
    n_src = src.size
    x_src = np.arange(n_src, dtype=np.float64) * (2.0 / n_src)
    x_tgt = np.arange(n_target, dtype=np.float64) * (2.0 / n_target)
    return np.interp(x_tgt, x_src, src)


def binary_from_continuous(vals: np.ndarray, target_sum: int) -> np.ndarray:
    """Create a binary vector (0/1) with exactly target_sum ones, picking the largest entries."""
    idx = np.argpartition(-vals, target_sum - 1)[:target_sum]
    bin_vec = np.zeros_like(vals, dtype=np.float64)
    bin_vec[idx] = 1.0
    return bin_vec


def random_binary_vector(n: int, target_sum: int, rng: np.random.Generator) -> np.ndarray:
    """Random binary vector of length n with exactly target_sum ones."""
    idx = rng.permutation(n)[:target_sum]
    vec = np.zeros(n, dtype=np.float64)
    vec[idx] = 1.0
    return vec


def hill_climb_binary(start_h: np.ndarray, rng: np.random.Generator, time_limit: float):
    """
    Simple hill‑climbing on binary vectors.
    Swaps a random 1 with a random 0; accepts only improving moves.
    Returns the best vector found (within the time limit) and its C5.
    """
    h = start_h.astype(np.float64).copy()
    n = h.size
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    while time.time() < time_limit:
        i = rng.integers(n)
        j = rng.integers(n)
        if h[i] == h[j]:
            continue
        # perform swap
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5(h)
        if new_c5 < cur_c5 - 1e-14:
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]
    return best_h, best_c5


def continuous_refine(start_h: np.ndarray, rng: np.random.Generator,
                     time_limit: float, current_best: float):
    """
    Small random pairwise adjustments preserving the total sum.
    Accepts only improving moves (no temperature schedule).
    """
    h = start_h.astype(np.float64).copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_val = current_best
    step = 0.2
    iters = 0

    while time.time() < time_limit:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue

        # Determine feasible delta respecting bounds [0,1]
        delta_low = -min(step, h[i], 1.0 - h[j])
        delta_high = min(step, 1.0 - h[i], h[j])
        if delta_low >= delta_high:
            continue
        delta = rng.uniform(delta_low, delta_high)
        if delta == 0.0:
            continue

        h[i] += delta
        h[j] -= delta
        new_c5 = compute_c5(h)
        if new_c5 < best_val - 1e-14:
            best_val = new_c5
            best_h = h.copy()
        else:
            # revert
            h[i] -= delta
            h[j] += delta

        iters += 1
        if iters % 2000 == 0:
            step *= 0.9
            if step < 1e-5:
                break
    return best_h, best_val


def enforce_sum_constraint(h: np.ndarray, target_sum: float):
    """Adjust the first element to make the sum exact (tiny correction)."""
    total = float(np.sum(h))
    diff = target_sum - total
    if abs(diff) > 1e-12:
        h[0] += diff
        # Clip in case of slight overflow
        if h[0] < 0.0:
            h[0] = 0.0
        elif h[0] > 1.0:
            h[0] = 1.0
    return h

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return (h_values, c5_bound, n_points) for an improved upper bound on C5."""
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Try to load an initial construction if the parent provided one.
    try:
        init_h = np.asarray(initial_h_values, dtype=np.float64)
    except NameError:
        init_h = None

    # Global best record
    best_h = None
    best_c5 = float('inf')
    best_n = None

    # Evaluate the provided initial construction (if any) – may already be good.
    if init_h is not None:
        try:
            c5_init = compute_c5(init_h)
            if c5_init < best_c5:
                best_c5 = c5_init
                best_h = init_h.copy()
                best_n = init_h.size
        except Exception:
            pass  # safety – ignore if something unexpected

    # Candidate sample sizes (must be even for exact half‑mass binary vectors)
    n_candidates = [60, 80, 100, 120, 140]

    # Main stochastic search loop
    while True:
        now = time.time()
        elapsed = now - start_time
        remaining = budget_s - elapsed
        if remaining <= 0.01:
            break

        # Allocate a small time slice for each hill‑climb (0.5 s or remaining)
        slice_time = min(0.5, remaining * 0.9)  # leave some margin for final refinement
        time_limit = now + slice_time

        # Randomly pick a resolution
        n = int(rng.choice(n_candidates))
        if n % 2 != 0:
            n += 1  # ensure even

        target_sum = n // 2

        # ----- Start from a random binary vector ----- #
        start_bin = random_binary_vector(n, target_sum, rng)
        h_opt, c5_opt = hill_climb_binary(start_bin, rng, time_limit)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n

        # ----- Occasionally also try a start derived from the parent construction ----- #
        if init_h is not None and (rng.random() < 0.2):
            # Interpolate parent to current size and binarize
            interp = interpolate_to_n(init_h, n)
            start_from_parent = binary_from_continuous(interp, target_sum)
            # Use remaining part of the slice for this extra climb (if time allows)
            now2 = time.time()
            if now2 < time_limit:
                h_opt2, c5_opt2 = hill_climb_binary(start_from_parent, rng, time_limit)
                if c5_opt2 < best_c5:
                    best_c5 = c5_opt2
                    best_h = h_opt2.copy()
                    best_n = n

        # Loop continues until budget runs out
        # (Note: if the slice finishes early, the loop will immediately start a new slice.)

    # ----- Continuous refinement on the best binary solution found ----- #
    if best_h is not None:
        now = time.time()
        remaining = budget_s - (now - start_time) - 0.05  # small safety margin
        if remaining > 0.0:
            refine_limit = now + remaining
            best_h, best_c5 = continuous_refine(best_h, rng, refine_limit, best_c5)

        # Ensure the sum constraint is exactly satisfied (tiny fixing)
        best_h = enforce_sum_constraint(best_h, best_n / 2.0)

    # Fallback: if nothing succeeded, use a uniform half‑mass construction.
    if best_h is None:
        best_n = 60
        best_h = np.full(best_n, 0.5, dtype=np.float64)
        best_c5 = compute_c5(best_h)

    # Final check (sanity)
    best_c5 = compute_c5(best_h)

    # Return as list for compatibility with the harness
    return (best_h.tolist(), float(best_c5), int(best_n))
