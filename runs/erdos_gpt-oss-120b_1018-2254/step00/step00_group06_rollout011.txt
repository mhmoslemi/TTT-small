def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, random
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    # Determine n_points
    try:
        n = len(initial_h_values)
    except NameError:
        n = 95
    dx = 2.0 / n
    target_sum = n / 2.0

    def compute_c5(h):
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx

    # Projection function as defined
    def project(v):
        # project onto [0,1] and sum = target_sum
        # binary search lambda
        low = -1.0
        high = 2.0
        for _ in range(40):
            mid = (low + high) / 2.0
            x = np.clip(v - mid, 0, 1)
            s = x.sum()
            if s > target_sum:
                low = mid
            else:
                high = mid
        return np.clip(v - (low + high) / 2.0, 0, 1)

    # Initialize population
    popsize = 200
    pop = np.empty((popsize, n), dtype=float)
    # first individual: initial_h_values if exists
    try:
        h0 = np.array(initial_h_values, dtype=float)
        # ensure feasible
        h0 = project(h0)
        pop[0] = h0
    except NameError:
        # generate random feasible
        v = np.random.rand(n)
        pop[0] = project(v)
    # fill rest random
    for i in range(1, popsize):
        v = np.random.rand(n)
        pop[i] = project(v)

    # Evaluate
    scores = np.empty(popsize)
    best_idx = None
    best_score = np.inf
    for i in range(popsize):
        scores[i] = compute_c5(pop[i])
        if scores[i] < best_score:
            best_score = scores[i]
            best_idx = i

    gen = 0
    max_gen = 1000
    tournament_size = 3
    mutation_sigma = 0.1
    gene_mut_prob = 0.05

    while time.time() - start < budget_s and gen < max_gen:
        # produce next population
        new_pop = np.empty_like(pop)
        # elitism: keep best
        new_pop[0] = pop[best_idx]
        for i in range(1, popsize):
            # tournament selection for parents
            def select():
                idxs = np.random.choice(popsize, tournament_size, replace=False)
                return idxs[np.argmin(scores[idxs])]
            p1 = select()
            p2 = select()
            # crossover blend
            alpha = np.random.rand()
            child = alpha * pop[p1] + (1 - alpha) * pop[p2]
            # mutation
            mask = np.random.rand(n) < gene_mut_prob
            if mask.any():
                child[mask] += np.random.randn(mask.sum()) * mutation_sigma
            # repair
            child = project(child)
            new_pop[i] = child

        pop = new_pop
        # evaluate new population
        for i in range(popsize):
            scores[i] = compute_c5(pop[i])
        # update best
        cur_best_idx = np.argmin(scores)
        cur_best_score = scores[cur_best_idx]
        if cur_best_score < best_score:
            best_score = cur_best_score
            best_idx = cur_best_idx

        gen += 1

    h_best = pop[best_idx]
    # final C5 bound
    c5_best = compute_c5(h_best)
    return (h_best.tolist(), float(c5_best), n)
```

import time
import numpy as np

def compute_c5(h, dx):
    """Return the overlap integral C5 for a discrete step function h."""
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx

def project_onto_capped_simplex(v, target_sum, max_iter=40, rtol=1e-8):
    """
    Project vector v onto the set {x | 0 <= x_i <= 1, sum x_i = target_sum}
    using a simple binary search on the Lagrange multiplier.
    """
    low, high = -1.0, 2.0  # bounds that guarantee sum(high) <= target <= sum(low)
    for _ in range(max_iter):
        mid = (low + high) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            low = mid
        else:
            high = mid
        if abs(s - target_sum) <= rtol * target_sum:
            break
    x = np.clip(v - (low + high) * 0.5, 0.0, 1.0)
    # Minor correction for any residual error
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        interior = (x > 0.0) & (x < 1.0)
        cnt = interior.sum()
        if cnt > 0:
            x[interior] += diff / cnt
            np.clip(x, 0.0, 1.0, out=x)
        else:
            # fallback: distribute uniformly and clip again
            x += diff / len(x)
            np.clip(x, 0.0, 1.0, out=x)
    return x

def tournament_select(scores, k):
    """Return index of the best individual among k random tournament participants."""
    participants = np.random.choice(len(scores), k, replace=False)
    return participants[np.argmin(scores[participants])]

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Determine problem size
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
    else:
        n = 95  # default if no parent provided
    dx = 2.0 / n
    target_sum = n / 2.0

    # Prepare initial population
    popsize = 200
    pop = np.empty((popsize, n), dtype=float)

    # First individual: the provided parent (projected to be feasible)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        pop[0] = project_onto_capped_simplex(h0, target_sum)
    else:
        pop[0] = project_onto_capped_simplex(np.random.rand(n), target_sum)

    # Remaining individuals: random feasible vectors
    for i in range(1, popsize):
        pop[i] = project_onto_capped_simplex(np.random.rand(n), target_sum)

    # Evaluate initial population
    scores = np.empty(popsize, dtype=float)
    for i in range(popsize):
        scores[i] = compute_c5(pop[i], dx)
    best_idx = int(np.argmin(scores))
    best_score = float(scores[best_idx])

    # GA parameters
    max_gen = 1000
    tourn_size = 3
    mutation_sigma = 0.1
    gene_mut_prob = 0.05

    generation = 0
    while time.time() - start_time < budget_s and generation < max_gen:
        new_pop = np.empty_like(pop)
        # Elitism: keep current best
        new_pop[0] = pop[best_idx]

        for i in range(1, popsize):
            # Select parents via tournament
            p1 = tournament_select(scores, tourn_size)
            p2 = tournament_select(scores, tourn_size)

            # Blend crossover (preserves sum)
            alpha = np.random.rand()
            child = alpha * pop[p1] + (1.0 - alpha) * pop[p2]

            # Gaussian mutation per gene
            mask = np.random.rand(n) < gene_mut_prob
            if mask.any():
                child[mask] += np.random.randn(mask.sum()) * mutation_sigma

            # Repair to satisfy bounds and sum
            child = project_onto_capped_simplex(child, target_sum)
            new_pop[i] = child

        pop = new_pop

        # Re‑evaluate
        for i in range(popsize):
            scores[i] = compute_c5(pop[i], dx)

        cur_best = int(np.argmin(scores))
        cur_score = float(scores[cur_best])
        if cur_score < best_score:
            best_score = cur_score
            best_idx = cur_best

        generation += 1

        # Early exit if time is almost exhausted
        if time.time() - start_time > budget_s * 0.99:
            break

    # Local refinement: simple stochastic hill‑climbing from the current best
    h_cur = pop[best_idx].copy()
    cur_score = best_score
    while time.time() - start_time < budget_s:
        # Small random perturbation and projection
        proposal = h_cur + np.random.randn(n) * 0.02
        proposal = project_onto_capped_simplex(proposal, target_sum)
        prop_score = compute_c5(proposal, dx)
        if prop_score < cur_score:
            h_cur = proposal
            cur_score = prop_score

    # Final result
    final_h = h_cur
    final_c5 = compute_c5(final_h, dx)
    return (final_h.tolist(), float(final_c5), n)
