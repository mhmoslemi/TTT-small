def project_onto_box_and_sum(v, target):
    # Clip to [0,1]
    w = np.clip(v, 0, 1)
    sum_w = w.sum()
    if np.isclose(sum_w, target):
        return w
    # Determine if need to increase or decrease.
    if sum_w > target:
        # need to decrease by excess = sum_w - target
        excess = sum_w - target
        # consider indices where w>0
        idx = np.where(w > 0)[0]
        # sort w[idx] descending so we remove from largest first (makes sense)
        # Doing water-filling:
        # We'll subtract equally across those indices until some reach zero.
        # Loop:
        while excess > 1e-12:
            # get non-zero indices
            idx = np.where(w > 0)[0]
            if len(idx) == 0:
                break
            # amount to subtract per index if equal reduction
            dec = excess / len(idx)
            # subtract dec but not below 0
            w[idx] -= dec
            # clip any negatives to 0, adjust excess accordingly
            # any negative values now become zero and excess increased accordingly
            neg = np.where(w < 0)[0]
            if len(neg) > 0:
                # amount we subtracted too much on these indices
                excess_incorrect = -w[neg].sum()
                # reset those entries to 0
                w[neg] = 0
                # restore the overshoot: we need to re-distribute this amount
                excess = excess_incorrect
            else:
                excess = 0
        return w
    else:
        # sum_w < target
        deficit = target - sum_w
        # need to increase values that are <1
        while deficit > 1e-12:
            idx = np.where(w < 1)[0]
            if len(idx) == 0:
                break
            inc = deficit / len(idx)
            w[idx] += inc
            over = np.where(w > 1)[0]
            if len(over) > 0:
                # overshoot, adjust deficit
                excess_inc = w[over].sum() - len(over) # each >1 overshoot amount
                w[over] = 1
                deficit = excess_inc
            else:
                deficit = 0
        return w

import numpy as np, time

def _proj_capped_simplex(v, target, tol=1e-9, max_iter=60):
    n = v.shape[0]
    if target <= 0:
        return np.zeros_like(v)
    if target >= n:
        return np.ones_like(v)
    lo = np.min(v) - 1.0 - 1.0
    hi = np.max(v) + 1.0 + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _c5_of_h(h):
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return conv.max() * dx

def _worst_shift_gradient(h):
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode='full')
    i_max = np.argmax(conv)
    shift = i_max - (n - 1)
    g = np.empty_like(h)
    for k in range(n):
        val = 0.0
        idx1 = k + shift
        if 0 <= idx1 < n:
            val += (1.0 - h[idx1])
        idx2 = k - shift
        if 0 <= idx2 < n:
            val -= h[idx2]
        g[k] = val
    return g, shift

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, random
    np.random.seed(seed)
    random.seed(seed)
    # initialize h
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h = h0.copy()
    else:
        n = 47
        # random feasible point
        v = np.random.rand(n)
        target = n / 2.0
        h = _proj_capped_simplex(v, target)
    target = n / 2.0
    # ensure feasibility
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target) > 1e-12:
        h = _proj_capped_simplex(h, target)
    cur_h = h.copy()
    cur_score = _c5_of_h(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    start = time.time()
    step = 0.2
    min_step = 0.005
    # main loop
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s:
            break
        # choose move type
        if np.random.rand() < 0.35:
            # gradient move
            g, _ = _worst_shift_gradient(cur_h)
            h_new = cur_h - step * g
        else:
            # random transfer
            i, j = np.random.choice(n, 2, replace=False)
            delta = (np.random.rand() - 0.5) * 2.0 * step
            h_new = cur_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
        # project
        h_new = _proj_capped_simplex(h_new, target)
        new_score = _c5_of_h(h_new)
        # acceptance
        if new_score < cur_score:
            cur_h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score
        else:
            # simulated annealing acceptance
            temp = max(0.0, (budget_s - elapsed) / budget_s) * 0.02
            if np.random.rand() < np.exp(-(new_score - cur_score) / (temp + 1e-12)):
                cur_h = h_new
                cur_score = new_score
        # occasional perturbation of best solution
        if np.random.rand() < 0.001:
            cur_h = best_h + (np.random.rand(n) - 0.5) * step * 0.5
            cur_h = _proj_capped_simplex(cur_h, target)
            cur_score = _c5_of_h(cur_h)
        # decay step size
        step = max(min_step, step * 0.9995)
    # final evaluation
    final_score = _c5_of_h(best_h)
    return best_h.tolist(), float(final_score), n

import numpy as np
import time
import random

def _proj_capped_simplex(v: np.ndarray, target: float, tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    """
    Project vector v onto {x | sum(x)=target, 0<=x<=1}.
    Uses a simple bisection on the Lagrange multiplier.
    """
    n = v.shape[0]
    if target <= 0:
        return np.zeros_like(v)
    if target >= n:
        return np.ones_like(v)

    lo = np.min(v) - 2.0   # sufficiently low -> sum > target
    hi = np.max(v) + 2.0   # sufficiently high -> sum < target

    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def _c5_of_h(h: np.ndarray) -> float:
    """Compute the C5 value for a discretised step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(conv.max() * dx)


def _worst_shift_gradient(h: np.ndarray):
    """
    Return the sub‑gradient of the worst (maximal) overlap term.
    Also returns the shift producing the worst overlap.
    """
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode='full')
    i_max = int(np.argmax(conv))
    shift = i_max - (n - 1)

    g = np.empty_like(h)
    for k in range(n):
        val = 0.0
        idx1 = k + shift
        if 0 <= idx1 < n:
            val += (1.0 - h[idx1])
        idx2 = k - shift
        if 0 <= idx2 < n:
            val -= h[idx2]
        g[k] = val
    return g, shift
