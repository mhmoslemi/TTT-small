def project_to_simplex_bounds(y, target, low=0.0, high=1.0):
    left = np.min(y) - high
    right = np.max(y) - low
    for _ in range(50):
        mid = (left + right) / 2.0
        x = np.clip(y - mid, low, high)
        s = np.sum(x)
        if s > target:
            left = mid
        else:
            right = mid
    # After loop; final x as clipped y-mid
    x = np.clip(y - (left+right)/2.0, low, high)
    # Might be off by epsilon; adjust
    # Use simple scaling of remaining slack
    # But fine.
    return x

def neighbor_move(h, max_delta=0.1):
    n = len(h)
    i, j = np.random.choice(n, size=2, replace=False)
    delta = (np.random.rand() - 0.5) * 2 * max_delta  # uniform in [-max_delta, max_delta]
    # ensure i - delta stays within [0,1] and j + delta stays within [0,1]
    # Actually we will subtract delta from i and add to j to keep sum constant
    # So h_i_new = h_i - delta, h_j_new = h_j + delta.
    # Ensure both within [0,1].
    hi = h[i]
    hj = h[j]
    # adjust delta to bounds
    lower_delta = max(-hj, hi - 1)   # min allowed delta (cannot make h_j <0, cannot make h_i >1)
    upper_delta = min(1 - hj, hi)   # max allowed delta (cannot make h_j >1, h_i <0)
    # Actually hi - delta must be >=0 => delta <= hi; hi - delta <=1 => delta >= hi-1.
    # Similarly, hj + delta must be >=0 => delta >= -hj; <=1 => delta <= 1-hj.
    lo = max(-hj, hi - 1)   # lower bound for delta
    hi_bound = min(hi, 1 - hj)   # upper bound
    # random delta within [lo, hi_bound] intersect [-max_delta,max_delta]
    low = max(lo, -max_delta)
    high = min(hi_bound, max_delta)
    if high <= low:
        return h  # no move possible; return unchanged
    delta = np.random.uniform(low, high)
    new_h = h.copy()
    new_h[i] = hi - delta
    new_h[j] = hj + delta
    return new_h

def neighbor_move(h, max_delta=0.2):
    n = len(h)
    i, j = np.random.choice(n, size=2, replace=False)
    hi = h[i]
    hj = h[j]
    # compute allowed delta range
    low = max(-hj, hi - 1)  # delta lower bound: can't make hj negative, hi > 1
    high = min(hi, 1.0 - hj)  # delta upper bound: can't make hi negative, hj >1
    # Then intersect with [-max_delta, max_delta]
    low = max(low, -max_delta)
    high = min(high, max_delta)
    if low >= high:
        # cannot move; return unchanged
        return h
    delta = np.random.uniform(low, high)
    new_h = h.copy()
    new_h[i] = hi - delta
    new_h[j] = hj + delta
    return new_h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    # Ensure type double
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    return float(dx * max_corr)

        init = np.array(initial_h_values, dtype=float)

    # Ensure constraints
    target = n / 2.0

    if abs(sum_init - target) > 1e-9 or np.any(init < -1e-9) or np.any(init > 1+1e-9):

    best_h = init.copy()
    best_c5 = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_c5 = best_c5
    # Annealing parameters
    temperature = 0.05  # start temperature
    cooling = 0.99995
    max_delta = 0.2
    iter = 0

    while time.time() - start_time < budget_s - 0.2:

        cand_h = neighbor_move(cur_h, max_delta)
        cand_c5 = compute_c5(cand_h)
        delta_obj = cand_c5 - cur_c5

        # cooling schedule
        temperature *= cooling
        max_delta *= cooling
        # periodic restart
        if iter % 50000 == 0:
            cur_h = random_feasible(n)
            cur_c5 = compute_c5(cur_h)
            # optionally keep best unchanged
        # occasional reduce step if temperature too low
        if temperature < 1e-6:

            max_delta = 0.2
    # Return
    return best_h.tolist(), float(best_c5), int(n)

import time
import math
import numpy as np

def compute_c5(h):
    """Return the overlap integral C5 for a step function h sampled on [0,2]."""
    h = np.asarray(h, dtype=float)
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def project_to_bounds_simplex(y, target, low=0.0, high=1.0, iters=60):
    """
    Project vector y onto the intersection of the box [low,high]^n and the hyperplane sum(x)=target.
    The projection has the form x_i = clip(y_i - λ, low, high) for some λ.
    """
    y = np.asarray(y, dtype=float)
    left = np.min(y) - high
    right = np.max(y) - low
    for _ in range(iters):
        mid = (left + right) * 0.5
        x = np.clip(y - mid, low, high)
        s = np.sum(x)
        if s > target:
            left = mid
        else:
            right = mid
    λ = (left + right) * 0.5
    return np.clip(y - λ, low, high)

def random_feasible(n):
    """Generate a random feasible h with n samples (0≤h_i≤1 and sum = n/2)."""
    y = np.random.rand(n)
    return project_to_bounds_simplex(y, target=n / 2.0)

def neighbor_move(h, max_delta=0.2):
    """
    Produce a new feasible vector by moving a random amount δ from one entry to another.
    The move preserves the total sum and respects box constraints.
    """
    n = h.size
    i, j = np.random.choice(n, size=2, replace=False)
    hi, hj = h[i], h[j]

    # feasible interval for δ (hi -> hi-δ, hj -> hj+δ)
    low = max(-hj, hi - 1.0)   # δ >= low
    high = min(hi, 1.0 - hj)   # δ <= high
    # intersect with the prescribed step size limit
    low = max(low, -max_delta)
    high = min(high, max_delta)
    if low >= high:
        return h  # no feasible move
    delta = np.random.uniform(low, high)
    new_h = h.copy()
    new_h[i] = hi - delta
    new_h[j] = hj + delta
    return new_h
