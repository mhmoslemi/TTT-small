def _compute_c5(h):
    """Return C5 = max_k ∫ h(x)*(1-h(x+k)) dx for discrete h."""
    N = h.size
    dx = 2.0 / N
    # np.correlate gives sum_i a[i] * b[i+k] for shift k (full mode)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _best_pair_for_shift(h, shift):

            overflow = np.maximum(h - 1.0, 0.0).sum()
            deficit = np.maximum(-h, 0.0).sum()
            if overflow > 0.0:
                h = np.minimum(h, 1.0)
                free = (h < 1.0).sum()
                if free:
                    h[h < 1.0] += overflow / free
            if deficit > 0.0:
                h = np.maximum(h, 0.0)
                free = (h > 0.0).sum()
                if free:
                    h[h > 0.0] -= deficit / free
            # Final scaling to exact sum
            h = np.clip(h, 0.0, 1.0)
            h *= target_sum / h.sum()
    else:
        N = 70
        x = np.linspace(0.0, 2.0, N)
        raw = np.sin(np.pi * x / 2.0) ** 2          # sin^2 shape, integral≈1
        target_sum = N / 2.0
        h = raw * (target_sum / raw.sum())
        h = np.clip(h, 0.0, 1.0)

    # Ensure exact sum by a tiny uniform correction
    diff = (N / 2.0) - h.sum()
    if abs(diff) > 1e-12:
        h += diff / N
        h = np.clip(h, 0.0, 1.0)
        h *= (N / 2.0) / h.sum()

    # ---- optimisation -------------------------------------------------------
    curr_h = h.copy()
    curr_c = _compute_c5(curr_h)
    best_h = curr_h.copy()
    best_c = curr_c

    # parameters controlling move size and annealing
    max_step_frac = 0.3   # proportion of the admissible delta at the start
    min_step_frac = 0.005 # minimal fraction at the end
    temp_initial = 0.005  # temperature scale for uphill moves

    it_without_improve = 0
    patience = 5000

    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.99:
            break

        # identify shift with maximal overlap
        N = curr_h.size
        corr = np.correlate(curr_h, 1.0 - curr_h, mode="full")
        k_idx = int(np.argmax(corr))
        shift = k_idx - (N - 1)

        # pick the pair (i,j) that contributes most to this shift
        i, j = _best_pair_for_shift(curr_h, shift)

        # admissible delta preserving bounds
        max_delta = min(curr_h[i], 1.0 - curr_h[j])
        if max_delta <= 0.0:
            # fallback: random pair
            i, j = random.sample(range(N), 2)
            max_delta = min(curr_h[i], 1.0 - curr_h[j])

        # schedule step size
        frac = max(min_step_frac,
                   max_step_frac * (1.0 - elapsed / budget_s))
        delta = frac * max_delta
        if delta <= 0.0:
            continue

        # propose move i -> j
        new_h = curr_h.copy()
        new_h[i] -= delta
        new_h[j] += delta

        new_c = _compute_c5(new_h)

        # simulated‑annealing acceptance
        temp = temp_initial * (1.0 - elapsed / budget_s) + 1e-8
        accept = False
        if new_c < curr_c:
            accept = True
        else:
            prob = math.exp(-(new_c - curr_c) / temp)
            if random.random() < prob:
                accept = True

        if accept:
            curr_h = new_h
            curr_c = new_c
            it_without_improve = 0
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()
        else:
            it_without_improve += 1

        # occasional random jitter to escape stagnation
        if it_without_improve >= patience:
            # re‑initialize from a smooth shape while keeping sum constraint
            x = np.linspace(0.0, 2.0, N)
            raw = np.sin(np.pi * x / 2.0) ** 2
            cand = raw * ((N / 2.0) / raw.sum())
            # add small uniform noise then re‑project
            cand += (np.random.rand(N) - 0.5) * 0.02
            cand = np.clip(cand, 0.0, 1.0)
            cand *= (N / 2.0) / cand.sum()
            curr_h = cand
            curr_c = _compute_c5(curr_h)
            it_without_improve = 0

    # ---- post‑process -------------------------------------------------------
    # final projection to exact sum & bounds
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h *= (N / 2.0) / best_h.sum()
    final_c = _compute_c5(best_h)

    return best_h, float(final_c), int(N)
