import time
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return the Erdős C5 value for a step function h."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h); np.correlate uses zero padding.
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _random_binary_half(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a vector of length n with 22 ones, one 0.5 and remaining 0s (n must be 45)."""
    vec = np.zeros(n, dtype=float)
    # number of full ones for odd n (n/2 = 22.5)
    ones = rng.choice(n, size=n // 2, replace=False)  # 22 ones when n=45
    vec[ones] = 1.0
    # place the half
    remaining = np.setdiff1d(np.arange(n), ones, assume_unique=True)
    half_idx = rng.choice(remaining)
    vec[half_idx] = 0.5
    return vec


def _make_initial_vector(rng: np.random.Generator):
    """Return a starting vector respecting the constraints."""
    # Try to use the supplied initial_h_values if it exists.
    try:
        init = np.asarray(initial_h_values, dtype=float).ravel()
        n = init.size
        target = n / 2.0
        # Clip to [0,1] and repair the sum if needed (tiny numerical issue only)
        init = np.clip(init, 0.0, 1.0)
        cur_sum = init.sum()
        if abs(cur_sum - target) > 1e-9:
            # Simple proportional scaling then a tiny adjustment
            if cur_sum > 0:
                init *= target / cur_sum
            init = np.clip(init, 0.0, 1.0)
            diff = target - init.sum()
            # adjust a random index within its slack
            for idx in rng.permutation(n):
                slack_low = -init[idx]
                slack_high = 1.0 - init[idx]
                if slack_low <= diff <= slack_high:
                    init[idx] += diff
                    break
        return init
    except NameError:
        # No supplied construction – fall back to a random binary+half vector (n=45)
        n = 45
        return _random_binary_half(n, rng)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function h that minimizes the Erdős C5 bound.

    Returns
    -------
    (h_values, c5_bound, n_points)
        h_values : list of floats, length n_points, sum = n_points/2, each in [0,1]
        c5_bound : float, the achieved C5 value
        n_points : int, the number of discretisation points used
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # small safety margin

    # initialise
    h_cur = _make_initial_vector(rng).copy()
    n = h_cur.size
    target_sum = n / 2.0
    best_h = h_cur.copy()
    best_c5 = _compute_c5(best_h)

    # sanity: ensure constraints
    h_cur = np.clip(h_cur, 0.0, 1.0)
    if abs(h_cur.sum() - target_sum) > 1e-8:
        # correct the sum very gently
        diff = target_sum - h_cur.sum()
        idx = rng.integers(n)
        slack_low = -h_cur[idx]
        slack_high = 1.0 - h_cur[idx]
        adjust = np.clip(diff, slack_low, slack_high)
        h_cur[idx] += adjust

    cur_c5 = _compute_c5(h_cur)

    # ---------- main stochastic optimisation loop ----------
    # Parameters
    swap_prob = 0.3          # probability to perform a pure swap
    restart_prob = 0.01      # chance each iteration to jump back to the global best
    random_restart_prob = 0.01  # chance to start from a fresh random vector
    target_bound = 0.3808    # we stop early if we beat this

    while time.time() < deadline:
        # occasional resets
        if rng.random() < restart_prob:
            h_cur = best_h.copy()
            cur_c5 = best_c5
            continue
        if rng.random() < random_restart_prob:
            # generate a new random binary+half candidate
            h_cur = _random_binary_half(n, rng)
            cur_c5 = _compute_c5(h_cur)
            continue

        # propose a move
        if rng.random() < swap_prob:
            i, j = rng.choice(n, size=2, replace=False)
            h_new = h_cur.copy()
            h_new[i], h_new[j] = h_new[j], h_new[i]
        else:
            i, j = rng.choice(n, size=2, replace=False)
            # Move a small amount delta from j to i while preserving the sum
            # h_new[i] = h_cur[i] + delta
            # h_new[j] = h_cur[j] - delta
            delta_low = -min(h_cur[i], 1.0 - h_cur[j])   # negative delta (move from i to j)
            delta_high = min(1.0 - h_cur[i], h_cur[j])   # positive delta (move from j to i)
            if delta_low >= delta_high:
                continue  # no feasible move
            # restrict step size to keep changes modest (helps stability)
            delta = rng.uniform(delta_low, delta_high) * 0.5
            h_new = h_cur.copy()
            h_new[i] += delta
            h_new[j] -= delta

        # evaluate
        new_c5 = _compute_c5(h_new)

        # accept if improvement (hill‑climbing)
        if new_c5 < cur_c5:
            h_cur = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5
                if best_c5 <= target_bound:
                    break  # reached desired target

    # final sanity check & repair (tiny floating drift possible)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-8:
        # distribute the tiny difference to a random coordinate that can absorb it
        for idx in rng.permutation(n):
            low = -best_h[idx]
            high = 1.0 - best_h[idx]
            if low <= diff <= high:
                best_h[idx] += diff
                break
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = _compute_c5(best_h)

    return (best_h.tolist(), final_c5, n)
