def random_move(vec, rng, step_max):
    n = vec.size
    i, j = rng.integers(n, size=2)
    if i == j:
        return None, None, None
    max_inc = min(step_max, 1.0 - vec[i])
    max_dec = min(step_max, vec[j])
    if max_inc <= 0 and max_dec <= 0:
        return None
    delta = rng.uniform(-max_dec, max_inc)
    new_vec = vec.copy()
    new_vec[i] += delta
    new_vec[j] -= delta
    return new_vec, i, j

def compute_c5(h):
    dx = 2.0 / h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx

def run(seed=42, budget_s=1100, **kwargs):

    # Determine n_points and initial h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.size
    else:
        n = 200
        h = np.full(n, 0.5)
    target_sum = n / 2.0
    # Ensure feasible

    current_sum = h.sum()
    if abs(current_sum - target_sum) > 1e-12:
        # Simple scaling then clip
        # Use projection via water-filling
        def project(v):
            lo, hi = -1.0, 1.0
            for _ in range(40):
                mid = (lo + hi) / 2.0
                proj = np.clip(v + mid, 0.0, 1.0)
                s = proj.sum()
                if s < target_sum:
                    lo = mid
                else:
                    hi = mid
            return np.clip(v + (lo+hi)/2.0, 0.0, 1.0)
        h = project(h)
    curr_h = h.copy()
    curr_c5 = compute_c5(curr_h)
    best_h = curr_h.copy()
    best_c5 = curr_c5
    start = time.time()
    # Parameters
    temperature = 1.0
    cooling = 0.9995
    step_max = 0.2
    while time.time() - start < budget_s * 0.9:
        # propose move
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        max_inc = min(step_max, 1.0 - curr_h[i])
        max_dec = min(step_max, curr_h[j])
        if max_inc <= 0 and max_dec <= 0:
            continue
        delta = rng.uniform(-max_dec, max_inc)
        new_h = curr_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        # compute new c5
        new_c5 = compute_c5(new_h)
        # Acceptance
        if new_c5 < curr_c5 or rng.random() < math.exp(-(new_c5 - curr_c5) / temperature):

            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
        # update temperature and step size

        step_max *= cooling  # gradually reduce step

import time, math
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral using the prescribed correlation."""
    dx = 2.0 / h.shape[0]
    # correlation of h and (1-h) with zero padding
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_to_constraints(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project v onto the box [0,1]^n while enforcing sum(v) == target_sum.
    The projection is performed by finding a scalar λ such that
    sum(clip(v + λ, 0, 1)) == target_sum (water‑filling).
    """
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = (lo + hi) / 2.0
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)


def shift_gradient(h: np.ndarray, shift: int) -> np.ndarray:
    """Sub‑gradient of the overlap term corresponding to a given shift."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        s = shift
        if n - s > 0:
            g[: n - s] += 1.0 - h[s:]          # derivative of Σ h_i
            g[s:] += -h[: n - s]              # derivative of -Σ h_i h_{i+s}
    else:
        s = -shift
        if n - s > 0:
            g[s:] += 1.0 - h[: n - s]          # derivative of Σ h_{i+s}
            g[: n - s] += -h[s:]               # derivative of -Σ h_{i+s} h_i
    return g

            cand = np.clip(cand, 0.0, 1.0)
            cand = project_to_constraints(cand, target_sum)
            cand_c5 = compute_c5(cand)
            if cand_c5 < cur_c5:
                cur_h, cur_c5 = cand, cand_c5
                if cand_c5 < best_c5:
                    best_h, best_c5 = cand.copy(), cand_c5

    # final re‑projection (tiny numerical drift)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-10:
        best_h[0] = np.clip(best_h[0] + diff, 0.0, 1.0)

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n)
