def _project_to_constraints(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project vector v onto the set {0 <= x_i <= 1, sum(x) = target_sum}
    using a simple water‑filling (bisection) method.
    """
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -2.0, 2.0  # wide enough bounds for the shift
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)


def _c5_value(h: np.ndarray) -> float:
    """Return the C5 value for a step function represented by h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    # positive shifts correspond to the second half of the full correlation
    max_corr = np.max(corr[n - 1 :])
    return max_corr * (2.0 / n)

        n = h.shape[0]
    except Exception:
        # fallback: 70 points, uniform start (average 0.5)
        n = 70
        h = np.full(n, 0.5, dtype=np.float64)

    # enforce feasibility (box + sum)
    target_sum = n / 2.0
    h = _project_to_constraints(h, target_sum)

    cur_h = h.copy()
    cur_c5 = _c5_value(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    # ------------------------------------------------------------------
    # Parameters
    # ------------------------------------------------------------------
    T0 = 0.02                     # initial temperature for SA
    target_c5 = 0.3808            # desired bound (stop early if reached)
    max_noimp = 5000              # steps after which a small random restart is tried
    step_scale = 0.5              # typical move size as fraction of the feasible transfer
    small_scale = 0.1             # size for occasional tiny perturbations

    it = 0
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.95:
            break

        # --------------------------------------------------------------
        # Evaluate current worst shift and contributions
        # --------------------------------------------------------------
        n = cur_h.shape[0]
        full_corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        corr_vec = full_corr[n - 1 :]               # shifts 0 … n-1
        k = int(np.argmax(corr_vec))                # shift giving the current max

        candidates = []

        # ---- random donor–receiver move (generic) ----
        i_rand = np.random.randint(n)
        j_rand = np.random.randint(n - 1)
        if j_rand >= i_rand:
            j_rand += 1
        max_trans = min(cur_h[i_rand], 1.0 - cur_h[j_rand])
        if max_trans > 1e-12:
            eps = np.random.rand() * max_trans * step_scale
            h_new = cur_h.copy()
            h_new[i_rand] -= eps
            h_new[j_rand] += eps
            candidates.append(h_new)

        # ---- targeted move for the worst shift ----
        if k == 0:
            # shift 0: contribution = h*(1-h); move mass from a large term to a small one
            contrib = cur_h * (1.0 - cur_h)
            i_max = int(np.argmax(contrib))
            i_min = int(np.argmin(contrib))
            if i_max != i_min:
                max_trans2 = min(cur_h[i_max], 1.0 - cur_h[i_min])
                if max_trans2 > 1e-12:
                    eps2 = max_trans2 * step_scale
                    h_new2 = cur_h.copy()
                    h_new2[i_max] -= eps2
                    h_new2[i_min] += eps2
                    candidates.append(h_new2)
        else:
            # contribution for shift k
            contrib = cur_h[: n - k] * (1.0 - cur_h[k:])
            i_max = int(np.argmax(contrib))
            donor = i_max
            receiver = i_max + k
            # donor → receiver
            max_trans2 = min(cur_h[donor], 1.0 - cur_h[receiver])
            if max_trans2 > 1e-12:
                eps2 = max_trans2 * step_scale
                h_new2 = cur_h.copy()
                h_new2[donor] -= eps2
                h_new2[receiver] += eps2
                candidates.append(h_new2)
            # opposite direction
            max_trans3 = min(cur_h[receiver], 1.0 - cur_h[donor])
            if max_trans3 > 1e-12:
                eps3 = max_trans3 * step_scale
                h_new3 = cur_h.copy()
                h_new3[receiver] -= eps3
                h_new3[donor] += eps3
                candidates.append(h_new3)

        # ---- tiny random perturbation (helps escape plateaus) ----
        i2 = np.random.randint(n)
        j2 = np.random.randint(n - 1)
        if j2 >= i2:
            j2 += 1
        max_trans4 = min(cur_h[i2], 1.0 - cur_h[j2])
        if max_trans4 > 1e-12:
            eps4 = np.random.rand() * max_trans4 * small_scale
            h_new4 = cur_h.copy()
            h_new4[i2] -= eps4
            h_new4[j2] += eps4
            candidates.append(h_new4)

        # --------------------------------------------------------------
        # Accept the best candidate (greedy + occasional SA)
        # --------------------------------------------------------------
        improved = False
        T = T0 * max(0.0, 1.0 - elapsed / budget_s)  # linear cooling
        T = max(T, 1e-12)

        for cand in candidates:
            cand_c5 = _c5_value(cand)
            if cand_c5 < cur_c5 - 1e-12:
                cur_h = cand
                cur_c5 = cand_c5
                improved = True
                if cand_c5 < best_c5:
                    best_c5 = cand_c5
                    best_h = cand.copy()
                break
            else:
                # simulated‑annealing acceptance
                prob = math.exp(-(cand_c5 - cur_c5) / T)
                if np.random.rand() < prob:
                    cur_h = cand
                    cur_c5 = cand_c5
                    # not a new best, just a move
                    break

        # --------------------------------------------------------------
        # Stagnation handling: occasional restart from the best solution
        # --------------------------------------------------------------
        it += 1
        if not improved and it % max_noimp == 0:
            tmp = best_h.copy()
            for _ in range(5):
                a = np.random.randint(n)
                b = np.random.randint(n - 1)
                if b >= a:
                    b += 1
                max_t = min(tmp[a], 1.0 - tmp[b])
                if max_t > 0:
                    eps = np.random.rand() * max_t
                    tmp[a] -= eps
                    tmp[b] += eps
            cur_h = tmp
            cur_c5 = _c5_value(cur_h)

        # Early exit if we already beat the target
        if best_c5 <= target_c5:
            break

    # final projection (clean up any tiny numerical drift)
    best_h = _project_to_constraints(best_h, target_sum)
    best_c5 = _c5_value(best_h)

    return best_h, best_c5, n
