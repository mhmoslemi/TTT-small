    n = len(h)
    C = np.empty(2*n-1)
    grads = np.zeros((2*n-1, n))
    for s in range(-(n-1), n):
        idx = s + n - 1
        # compute product sum term
        # Over i where 0<=i<n and 0<=i+s<n
        # Use slicing:
        if s >= 0:
            h_i = h[:n-s]
            h_ip = h[s:]
        else:
            h_i = h[-s:]
            h_ip = h[:n+s]  # because s negative
        # C_s = sum_i h_i * (1 - h_ip) * dx = (np.sum(h_i) - np.sum(h_i * h_ip)) * dx
        sum_h_i = np.sum(h_i)
        sum_prod = np.sum(h_i * h_ip)
        C[idx] = (sum_h_i - sum_prod) * dx

        # gradient: for each j in 0..n-1 compute:
        # if j in range of i (i.e., i where i=j):
        # term1: dx
        # term2: -dx * h_{j+shift} if j+shift in range else 0
        # term3: -dx * h_{j-shift} if j-shift in range else 0
        # So gradient for shift s:
        # g = np.full(n, dx)
        # if shift >=0: j+shift < n => subtract h[j+shift] for j < n-shift (i.e., j from 0 .. n-shift-1)
        #                      j-shift >=0 => subtract h[j-shift] for j >= shift (j from shift .. n-1)
        # else similar.

        g = np.full(n, dx)
        if s >= 0:
            # subtract h_{j+shift} for j in 0..n-s-1
            g[:n-s] -= dx * h[s:]
            # subtract h_{j-shift} for j in s..n-1 (i.e., indexed to subtract h[j-s] for j >= s)
            g[s:] -= dx * h[:n-s]
        else:
            s_abs = -s
            # shift negative: j+shift = j - s_abs
            # subtract h_{j - s_abs} for j >= s_abs
            g[s_abs:] -= dx * h[:n-s_abs]
            # subtract h_{j - (-s)}??? Actually j - shift = j + s_abs
            # subtract h_{j+ s_abs} for j in 0..n-s_abs-1
            g[:n-s_abs] -= dx * h[s_abs:]
        grads[idx] = g

    # compute weighted gradient for smooth max:
    weights = np.exp(gamma * C)
    weights /= np.sum(weights)

    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
