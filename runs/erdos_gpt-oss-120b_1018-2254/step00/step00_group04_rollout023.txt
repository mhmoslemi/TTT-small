def make_binary(h):
    # h: array float
    # round >0.5 to 1
    arr = (h > 0.5).astype(float)
    n = arr.shape[0]
    target = n // 2
    current = int(np.sum(arr))
    # adjust
    if current > target:
        # need to flip some 1's to 0
        ones_idx = np.where(arr == 1)[0]
        # randomly select current - target of them
        to_flip = np.random.choice(ones_idx, size=current-target, replace=False)
        arr[to_flip] = 0.0
    elif current < target:
        zeros_idx = np.where(arr == 0)[0]
        to_flip = np.random.choice(zeros_idx, size=target-current, replace=False)
        arr[to_flip] = 1.0
    return arr

    arr = np.zeros(n, dtype=float)

            cand_i.append(i)

    cand_j = []
    for j in ones:
        i = j - k
        if not (0 <= i < n and h[i] == 1):
            cand_j.append(j)

    if cand_i and cand_j:
        i = _rng.choice(cand_i)
        j = _rng.choice(cand_j)
    else:
        # fallback to a generic swap if the targeted move is not possible
        return random_swap(h)

    new_h = h.copy()
    new_h[i] = 1.0
    new_h[j] = 0.0
    return new_h

def generate_neighbor(h: np.ndarray, prob_target: float = 0.85) -> np.ndarray:
    """Return a neighboring solution (swap) using either targeted or random move."""
    if _rng.random() < prob_target:
        return targeted_swap(h)
    else:
        return random_swap(h)

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a step function h on [0,2] that minimizes the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    # initialise RNGs
    set_rng(seed)
    random.seed(seed)

    start = time.time()
    max_time = budget_s * 0.95  # safety margin

    # obtain an initial feasible vector
    try:
        init = initial_h_values   # provided by the harness when available
        n = len(init)
        h0 = make_binary(np.array(init, dtype=float))
    except Exception:
        n = 100
        if n % 2 != 0:
            n += 1
        h0 = random_binary(n)

    # ensure exact sum constraint (n/2 ones)
    target_sum = n // 2
    current_sum = int(np.sum(h0))
    if current_sum != target_sum:
        if current_sum > target_sum:
            excess = current_sum - target_sum
            ones_idx = np.where(h0 == 1)[0]
            flip = _rng.choice(ones_idx, size=excess, replace=False)
            h0[flip] = 0.0
        else:
            deficit = target_sum - current_sum
            zeros_idx = np.where(h0 == 0)[0]
            flip = _rng.choice(zeros_idx, size=deficit, replace=False)
            h0[flip] = 1.0

    # initialise search state
    current_h = h0.copy()
    current_score = compute_c5(current_h)
    best_h = current_h.copy()
    best_score = current_score

    # simulated‑annealing schedule
    T0 = 0.02                     # initial temperature
    it = 0
    last_improve = 0

    while time.time() - start < max_time:
        # linear cooling
        elapsed = time.time() - start
        frac = elapsed / max_time
        T = T0 * (1 - frac)

        # propose neighbour
        neigh = generate_neighbor(current_h, prob_target=0.9)
        neigh_score = compute_c5(neigh)

        # Metropolis acceptance
        if neigh_score < current_score:
            accept = True
        else:
            if T > 0:
                accept = _rng.random() < math.exp(-(neigh_score - current_score) / T)
            else:
                accept = False

        if accept:
            current_h = neigh
            current_score = neigh_score
            if current_score < best_score:
                best_h = current_h.copy()
                best_score = current_score
                last_improve = it

        it += 1

        # stagnation handling – restart periodically
        if it - last_improve > 40000:
            if _rng.random() < 0.5:
                current_h = best_h.copy()
                current_score = best_score
            else:
                current_h = random_binary(n)
                current_score = compute_c5(current_h)
            last_improve = it

    return (best_h, float(best_score), n)
