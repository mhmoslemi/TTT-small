def project_to_constraints(h):
    # ensure [0,1] bounds
    np.clip(h,0,1,inplace=True)  # wont change sum but careful if clipped out of bounds leads to sum deviation.
    # adjust sum to target
    target = len(h) / 2.0
    current = h.sum()
    # compute diff
    diff = target - current
    # distribute diff across entries that not at bounds.
    # Find indices where h is not at bound (0 or 1) depending on sign of diff.
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # need to increase sum: increase some entries that are <1
        idxs = np.where(h < 1)[0]
        if len(idxs)==0:
            return h
        # spread diff equally among idxs (or just add to first)
        inc = diff / len(idxs)
        h[idxs] += inc
        # after adding may exceed 1; clip again and recompute diff iterative
        np.clip(h,0,1,inplace=True)
    else:
        # need to decrease sum: decrease entries >0
        idxs = np.where(h > 0)[0]
        dec = -diff / len(idxs)
        h[idxs] -= dec
        np.clip(h,0,1,inplace=True)
    # recursion to ensure sum close
    # but might not converge exactly due to bound saturation. So loop a few times.
    for _ in range(5):
        current = h.sum()
        diff = target - current
        if abs(diff) < 1e-12:
            break
        if diff > 1e-12:
            idxs = np.where(h < 1)[0]
            if len(idxs)==0:
                break
            inc = diff / len(idxs)
            h[idxs] += inc
            np.clip(h,0,1,inplace=True)
        elif diff < -1e-12:
            idxs = np.where(h > 0)[0]
            if len(idxs)==0:
                break
            dec = -diff / len(idxs)
            h[idxs] -= dec
            np.clip(h,0,1,inplace=True)
    # at end, we may have slight floating error; we can adjust small amount on first non bound index.
    current = h.sum()
    diff = target - current
    if abs(diff) > 1e-12:
        # adjust first index that can accommodate.
        for i in range(len(h)):
            if diff > 0 and h[i] < 1:
                add = min(diff, 1 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

        iter += 1

import time
import numpy as np

def compute_c5(h):
    """Return the overlap integral C5 for a discretized step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)


def project_to_constraints(h):
    """Clip h to [0,1] and adjust to satisfy sum(h) = n/2 exactly."""
    np.clip(h, 0.0, 1.0, out=h)
    target = h.shape[0] / 2.0
    # Iteratively spread the discrepancy over free entries.
    for _ in range(10):
        cur = h.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if idx.size == 0:
            break
        step = diff / idx.size
        h[idx] += step
        np.clip(h, 0.0, 1.0, out=h)
    # Small residual fix on a single coordinate if needed.
    cur = h.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        for i in range(h.size):
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h
