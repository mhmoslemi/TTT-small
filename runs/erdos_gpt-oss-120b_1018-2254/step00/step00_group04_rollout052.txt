import numpy as np
import random
import time
import math

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a given step function h."""
    n = h.size
    dx = 2.0 / n
    # full cross‑correlation of h and (1‑h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a step function h that minimizes C5.

    Returns
    -------
    (h_values, c5_bound, n_points) : tuple
        h_values – np.ndarray of length n_points with values in [0,1] and sum = n_points/2
        c5_bound – the achieved upper bound for C5
        n_points – number of discretisation points used
    """
    random.seed(seed)
    np.random.seed(seed)

    # ---- initialise ---------------------------------------------------------
    try:
        base = np.array(initial_h_values, dtype=np.float64)  # pre‑imported if present
        have_initial = True
    except NameError:
        have_initial = False

    if have_initial:
        # Aim for a modest grid (~250 points) while preserving the pattern.
        target_n = 256
        repeat = max(1, target_n // base.size)
        n_points = base.size * repeat
        h = np.repeat(base, repeat)
    else:
        n_points = 200
        h = np.full(n_points, 0.5, dtype=np.float64)

    # Ensure even length (required for exact half‑sum)
    if n_points % 2 != 0:
        h = h[:-1]
        n_points -= 1

    # Exact integral constraint: sum(h) == n_points/2
    target_sum = n_points / 2.0
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        h = h * (target_sum / cur_sum)
        h = np.clip(h, 0.0, 1.0)

    # ---- optimisation -------------------------------------------------------
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5

    start_time = time.time()
    max_time = budget_s * 0.95          # safety margin

    # Simple simulated‑annealing schedule
    T_start = 0.02
    T_end = 0.0005

    # restart control
    stagnation = 0
    RESTART_AFTER = 20000

    while True:
        if time.time() - start_time > max_time:
            break

        # temperature decreasing linearly with elapsed time
        elapsed = time.time() - start_time
        frac = elapsed / max_time
        T = T_start * (1 - frac) + T_end * frac

        # propose a mass transfer between two positions
        i, j = random.sample(range(n_points), 2)
        if random.random() < 0.5:

        new_h = cur_h.copy()
        new_h[tgt] += delta
        new_h[src] -= delta

        new_c5 = compute_c5(new_h)

        # Metropolis acceptance
        accept = False
        if new_c5 < cur_c5:
            accept = True
        else:
            diff = new_c5 - cur_c5
            if T > 0.0:
                prob = math.exp(-diff / T)
