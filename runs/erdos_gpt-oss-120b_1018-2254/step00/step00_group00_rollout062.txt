def evaluate(h, dx):
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def project_to_sum_bounds(h, target):
    # binary search lambda
    lo = -2.0
    hi = 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(h + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def repair(h, target):
    # For safety: first clip, then project to sum.
    np.clip(h, 0.0, 1.0, out=h)
    return project_to_sum_bounds(h, target)

def random_vector(n, target):
    # generate uniform random in [0,1], then repair
    h = np.random.rand(n)
    return repair(h, target)

def random_centered_vector(n, target):
    # generate a peaked shape
    xs = np.linspace(0, 2, n, endpoint=False) + 1e-12  # ensure not EXACT 0
    # compute distance from center
    d = np.abs(xs - 1.0)
    sigma = np.random.uniform(0.1, 0.8)   # width
    h = np.exp(- (d**2) / (2 * sigma**2))
    # maybe add small random noise
    h += 0.05 * np.random.rand(n)
    h = np.clip(h, 0.0, 1.0)
    return repair(h, target)

def tournament(pop, fitness, size=3):
    best_idx = None
    best_fit = float('inf')
    n = len(pop)
    for _ in range(size):
        idx = np.random.randint(n)
        if fitness[idx] < best_fit:
            best_fit = fitness[idx]
            best_idx = idx
    return pop[best_idx]

def ga_optimize(initial_h, n, target_sum, budget):
    # parameters
    POP_SIZE = 30
    MUT_PROB = 0.2
    MUT_SIGMA = 0.2
    # initialization
    pop = []
    fitness = []
    # include initial solution if provided
    if initial_h is not None:
        h0 = repair(np.array(initial_h, dtype=float), target_sum)
        pop.append(h0)
        fitness.append(evaluate(h0, 2.0 / n))
    # fill rest
    while len(pop) < POP_SIZE:
        # generate random peaked or uniform choice
        if np.random.rand() < 0.5:
            h = random_centered_vector(n, target_sum)
        else:
            h = random_vector(n, target_sum)
        pop.append(h)
        fitness.append(evaluate(h, 2.0 / n))
    # maintain best
    best_idx = int(np.argmin(fitness))
    best_h = pop[best_idx].copy()
    best_f = fitness[best_idx]
    start = time.time()
    # main loop
    it = 0
    while time.time() - start < budget * 0.95:
        it += 1
        # select parents
        p1 = tournament(pop, fitness, size=3)
        p2 = tournament(pop, fitness, size=3)
        # crossover
        mask = np.random.rand(n) < 0.5
        child = np.where(mask, p1, p2)  # choose from p2 where mask True else p1? Actually this yields child entries = p2 where mask True else p1.
        # mutation
        mut_mask = np.random.rand(n) < MUT_PROB
        child[mut_mask] += np.random.randn(mut_mask.sum()) * MUT_SIGMA
        # repair
        child = repair(child, target_sum)
        # evaluate
        child_f = evaluate(child, 2.0 / n)
        # replacement: replace worst if better
        worst_idx = int(np.argmax(fitness))
        if child_f < fitness[worst_idx]:
            pop[worst_idx] = child
            fitness[worst_idx] = child_f
            # update best
            if child_f < best_f:
                best_f = child_f
                best_h = child.copy()
        # occasional random injection: replace random individual with new random peaked vector
        if it % 1000 == 0:
            # replace random worse or random index with new random vector
            idx = np.random.randint(POP_SIZE)
            new_h = random_centered_vector(n, target_sum)
            new_f = evaluate(new_h, 2.0 / n)
            pop[idx] = new_h
            fitness[idx] = new_f
            if new_f < best_f:
                best_f = new_f
                best_h = new_h.copy()
    # final best
    return best_h, best_f

import numpy as np
import time

def _evaluate(h, dx):
    """Return C5 for a given discretized function h."""
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project_to_sum_bounds(x, target):
    """Project x onto the set {0<=y_i<=1, sum(y)=target} using bisection."""
    lo, hi = -2.0, 2.0  # wide enough to bracket the solution
    for _ in range(30):
        mid = (lo + hi) * 0.5
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)

def _repair(h, target):
    """Clamp to [0,1] and enforce the sum constraint."""
    np.clip(h, 0.0, 1.0, out=h)
    return _project_to_sum_bounds(h, target)

def _random_centered_vector(n, target):
    """Generate a smooth, centrally‑peaked vector and repair it."""
    xs = np.linspace(0.0, 2.0, n, endpoint=False)
    d = np.abs(xs - 1.0)                      # distance from centre
    sigma = np.random.uniform(0.1, 0.8)       # width of the peak
    h = np.exp(-0.5 * (d / sigma) ** 2)       # Gaussian bump
    h += 0.05 * np.random.rand(n)            # tiny noise for diversity
    return _repair(h, target)

def _random_vector(n, target):
    """Uniform random vector repaired to meet constraints."""
    h = np.random.rand(n)
    return _repair(h, target)

def _tournament(pop, fits, size=3):
    """Return a parent chosen by tournament selection."""
    best_idx = None
    best_fit = np.inf
    for _ in range(size):
        idx = np.random.randint(len(pop))
        if fits[idx] < best_fit:
            best_fit = fits[idx]
            best_idx = idx
    return pop[best_idx]

def _ga_optimize(init_h, n, target_sum, budget):
    """Simple GA that respects the box & sum constraints."""
    POP_SIZE = 30
    MUT_PROB = 0.2
    MUT_SIGMA = 0.2
    THRESHOLD = 0.3808

    dx = 2.0 / n

    # ----- initialise population ------------------------------------------------
    pop = []
    fits = []

    if init_h is not None:
        h0 = _repair(np.array(init_h, dtype=float), target_sum)
        pop.append(h0)
        fits.append(_evaluate(h0, dx))

    while len(pop) < POP_SIZE:
        if np.random.rand() < 0.5:
            h = _random_centered_vector(n, target_sum)
        else:
            h = _random_vector(n, target_sum)
        pop.append(h)
        fits.append(_evaluate(h, dx))

    best_idx = int(np.argmin(fits))
    best_h = pop[best_idx].copy()
    best_f = fits[best_idx]

    start = time.time()
    it = 0
    while time.time() - start < budget * 0.95:
        it += 1

        # ----- selection -------------------------------------------------------
        p1 = _tournament(pop, fits)
        p2 = _tournament(pop, fits)

        # ----- crossover (uniform) --------------------------------------------
        mask = np.random.rand(n) < 0.5
        child = np.where(mask, p2, p1)

        # ----- mutation --------------------------------------------------------
        mut_mask = np.random.rand(n) < MUT_PROB
        n_mut = mut_mask.sum()
        if n_mut:
            child[mut_mask] += np.random.randn(n_mut) * MUT_SIGMA

        # ----- repair & evaluate -----------------------------------------------
        child = _repair(child, target_sum)
        child_f = _evaluate(child, dx)

        # ----- replacement ------------------------------------------------------
        worst_idx = int(np.argmax(fits))
        if child_f < fits[worst_idx]:
            pop[worst_idx] = child
            fits[worst_idx] = child_f
            if child_f < best_f:
                best_f = child_f
                best_h = child.copy()

        # ----- periodic random injection ----------------------------------------
        if it % 1000 == 0:
            idx = np.random.randint(POP_SIZE)
            new_h = _random_centered_vector(n, target_sum)
            new_f = _evaluate(new_h, dx)
            pop[idx] = new_h
            fits[idx] = new_f
            if new_f < best_f:
                best_f = new_f
                best_h = new_h.copy()

        # ----- early exit if we already beat the target -------------------------
        if best_f <= THRESHOLD:
            break

    return best_h, best_f

def _fine_tune(best_h, target_sum, time_limit):
    """Local Gaussian perturbation search around the current best."""
    n = len(best_h)
    dx = 2.0 / n
    start = time.time()
    best_f = _evaluate(best_h, dx)

    while time.time() - start < time_limit:
        cand = best_h + np.random.randn(n) * 0.05
        cand = _repair(cand, target_sum)
        cand_f = _evaluate(cand, dx)
        if cand_f < best_f:
            best_f = cand_f
            best_h = cand

    return best_h, best_f

def run(seed=42, budget_s=1100, **kwargs):
    """Run the optimiser and return (h_values, c5_bound, n_points)."""
    np.random.seed(seed)

    # Determine problem size
    if "initial_h_values" in globals() and initial_h_values is not None:
        n_points = len(initial_h_values)
        init_vec = np.array(initial_h_values, dtype=float)
    else:
        n_points = 50  # reasonable fallback
        init_vec = None

    target_sum = n_points / 2.0

    # Main GA optimisation
    best_h, best_val = _ga_optimize(init_vec, n_points, target_sum, budget_s)

    # Optional fine‑tuning with remaining time budget
    remaining = budget_s - (time.time() - (time.time() - budget_s * 0.95))
    if remaining > 2.0:  # a couple of seconds are useful
        best_h, best_val = _fine_tune(best_h, target_sum, remaining * 0.9)

    return best_h.tolist(), float(best_val), n_points
