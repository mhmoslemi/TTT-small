def project_to_feasible(v, target):
    # Project onto [0,1] box and sum constraint.
    v = np.clip(v, 0.0, 1.0)
    # Now adjust sum by shifting uniformly and clipping repeatedly.
    for _ in range(20):
        s = np.sum(v)
        if abs(s - target) < 1e-10:
            break
        delta = (target - s) / np.sum((v > 0) & (v < 1)) if np.any((v > 0) & (v < 1)) else (target - s) / len(v)
        # Add delta only to free entries
        mask = (v > 0) & (v < 1)
        if not np.any(mask):
            v += delta
        else:
            v[mask] += delta
        v = np.clip(v, 0.0, 1.0)
    # final correction if needed
    s = np.sum(v)
    diff = target - s
    if abs(diff) > 1e-9:
        # adjust free entries
        mask = (v > 0) & (v < 1)
        free = np.where(mask)[0]
        if free.size > 0:
            v[free] += diff / free.size
            v = np.clip(v, 0.0, 1.0)
        else:
            # adjust uniformly - may violate bounds but diff small
            v += diff / len(v)
            v = np.clip(v, 0.0, 1.0)
    return v

def project_to_feasible(x, target, max_iter=100):
    x = np.clip(x, 0.0, 1.0)
    for _ in range(max_iter):
        s = np.sum(x)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # Find indices where adjustments possible
        if diff > 0:
            # need to increase sum: can increase where x < 1
            free = x < 1 - 1e-12
        else:
            free = x > 1e-12
        n_free = np.count_nonzero(free)
        if n_free == 0:
            break
        # amount to add/subtract per free entry
        delta = diff / n_free
        # Update x for free entries, but ensure staying within bounds
        x[free] += delta
        # Clip again (some may go out-of-bounds)
        x = np.clip(x, 0.0, 1.0)
    # final adjustment to exact sum if still off due to rounding
    s = np.sum(x)
    if abs(s - target) > 1e-9:
        # distribute residual to entries not at bounds
        free = (x > 0) & (x < 1)
        n_free = np.count_nonzero(free)
        if n_free > 0:
            x[free] += (target - s) / n_free
            x = np.clip(x, 0.0, 1.0)
        else:
            # can't adjust; distribute across all
            x += (target - s) / len(x)
            x = np.clip(x, 0.0, 1.0)
    return x
```

def interpolate_h(source_h, target_n):
    source_n = source_h.size
    if source_n == target_n:
        return source_h.copy()
    x_old = np.linspace(0, 2, source_n)

    h_new = project_to_feasible(h_new, target)

import time, numpy as np
from scipy.optimize import minimize, LinearConstraint

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def project_to_feasible(v, target, max_iter=20):
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        s = np.sum(v)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        free = (v > 0) & (v < 1)
        n_free = np.count_nonzero(free)
        if n_free == 0:
            # adjust all uniformly
            v += diff / v.size
        else:
            v[free] += diff / n_free
        v = np.clip(v, 0.0, 1.0)
    # final small correction
    s = np.sum(v)
    diff = target - s
    if abs(diff) > 1e-9:
        free = (v > 0) & (v < 1)
        n_free = np.count_nonzero(free)
        if n_free > 0:
            v[free] += diff / n_free
        else:
            v += diff / v.size
        v = np.clip(v, 0.0, 1.0)
    return v

def random_feasible(n, target, rng):
    # start from uniform 0.5
    v = np.full(n, 0.5)
    # add noise
    v += (rng.random(n) - 0.5) * rng.random() * 0.5
    return project_to_feasible(v, target)

def interpolate_h(source_h, target_n):
    source_n = source_h.size
    if source_n == target_n:
        return source_h.copy()
    x_old = np.linspace(0.0, 2.0, source_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
    h_new = np.interp(x_new, x_old, source_h)
    # ensure feasibility
    target = target_n / 2.0
    h_new = project_to_feasible(h_new, target)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    from scipy.optimize import minimize, LinearConstraint
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Initial construction
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values, dtype=float)
    else:
        base_h = np.full(45, 0.5, dtype=float)
    base_n = base_h.size
    # Ensure base_h feasible
    base_h = project_to_feasible(base_h, base_n / 2.0)
    best_h = base_h.copy()
    best_c5 = compute_c5(best_h)
    best_n = base_n
    # Candidate sample sizes
    n_list = [45, 60, 90, 120, 180]
    # Main search
    for n in n_list:
        now = time.time()
        if now - start_time >= budget_s * 0.95:
            break
        target = n / 2.0
        # initial guess from interpolation or baseline
        if n == best_n:
            init_guess = best_h.copy()
        else:
            init_guess = interpolate_h(base_h, n)
        # Linear equality constraint for sum

import time
import numpy as np
from scipy.optimize import minimize, LinearConstraint

def compute_c5(h):
    """Return C₅ for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def project_to_feasible(v, target, max_iter=20):
    """Clip v to [0,1] and adjust so that its sum equals target."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        s = np.sum(v)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        free = (v > 0) & (v < 1)
        n_free = np.count_nonzero(free)
        if n_free == 0:
            v += diff / v.size
        else:
            v[free] += diff / n_free
        v = np.clip(v, 0.0, 1.0)
    # final tiny correction
    s = np.sum(v)
    diff = target - s
    if abs(diff) > 1e-9:
        free = (v > 0) & (v < 1)
        n_free = np.count_nonzero(free)
        if n_free > 0:
            v[free] += diff / n_free
        else:
            v += diff / v.size
        v = np.clip(v, 0.0, 1.0)
    return v

def random_feasible(n, target, rng):
    """Create a random feasible vector of length n with sum = target."""
    v = np.full(n, 0.5)
    v += (rng.random(n) - 0.5) * 0.5  # uniform noise in [-0.25,0.25]
    return project_to_feasible(v, target)

def interpolate_h(src_h, target_n):
    """Linear interpolation of src_h to target_n points and project to feasibility."""

    x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
    h_new = np.interp(x_new, x_old, src_h)
    return project_to_feasible(h_new, target_n / 2.0)
