def project_simplex_box(v, target, lo=0.0, hi=1.0):
    # Clip to bounds first
    x = np.clip(v, lo, hi)
    # iterative fix for sum discrepancy
    # Use while loop until sum approx target within tolerance
    # we adjust equally among free indices
    # This may converge in few steps
    tol=1e-9
    max_iter=100
    for _ in range(max_iter):
        current_sum = x.sum()
        diff = target - current_sum
        if abs(diff) <= tol:
            break
        # Identify free indices where x not at bound
        free = (x > lo + tol) & (x < hi - tol)
        n_free = np.count_nonzero(free)
        if n_free == 0:
            # cannot adjust without violating bounds; just break
            break
        # Adjust equally
        delta = diff / n_free
        # Update x with clipping again
        x[free] += delta
        x = np.clip(x, lo, hi)
    # final rounding or ensure sum matches target within tolerance: optionally adjust small amount with one variable
    # To correct residual diff: find index i that is not at bound and adjust
    residual = target - x.sum()
    if abs(residual) > tol:
        # find any index that can handle residual
        for i in range(len(x)):
            if lo <= x[i] + residual <= hi:
                x[i] += residual
                break
    return x

def run(seed=42, budget_s=1100, **kwargs):

        init = np.random.rand(n_points)
    # Ensure proper shape and constraints
    target_sum = n_points / 2.0
    # Clip and project init
    init = np.clip(init, 0.0, 1.0)
    init = project_to_sum_and_bounds(init, target_sum)
    best_h = init
    dx = 2.0 / n_points
    def evaluate(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_score = evaluate(best_h)
    # if best_score already good enough, we can return early
    start = time.time()
    time_limit = budget_s * 0.95
    # Keep a list of top candidates maybe
    top_h = best_h.copy()
    top_score = best_score
    # main loop
    iteration = 0
    # parameters
    sigma_initial = 0.1
    sigma_final = 0.001
    # We will schedule sigma decreasing linearly over time
    while time.time() - start < time_limit:
        # compute sigma based on time
        elapsed = time.time() - start
        frac = elapsed / time_limit
        sigma = sigma_initial * (1 - frac) + sigma_final * frac
        # choose move type
        if random.random() < 0.5:
            # Gaussian perturbation of best
            cand = perturb_gaussian(best_h, sigma, target_sum)
        else:
            # pairwise transfer
            cand = best_h.copy()
            i, j = random.sample(range(n_points), 2)
            # max amount we can shift from j to i while keeping bounds
            max_up = 1.0 - cand[i]
            max_down = cand[j]
            max_delta = min(max_up, max_down)
            if max_delta > 0:
                delta = random.random() * max_delta
                cand[i] += delta
                cand[j] -= delta
            else:

                max_up = cand[i]
                max_down = 1.0 - cand[j]

                    cand[j] += delta

        cand_score = evaluate(cand)
        # Accept if better
        if cand_score < best_score:
            best_h = cand
            best_score = cand_score
            # also update top
            top_h = cand.copy()
            top_score = cand_score

    best_h = project_to_sum_and_bounds(best_h, target_sum)

import numpy as np

def _project_to_sum_and_bounds(v, target):
    """Project vector v onto the box [0,1] with sum(v)=target."""
    lo, hi = 0.0, 1.0
    x = np.clip(v, lo, hi)
    diff = target - x.sum()
    tol = 1e-12
    max_iter = 1000
    for _ in range(max_iter):
        if abs(diff) <= tol:
            break
        free = (x > lo + tol) & (x < hi - tol)
        n_free = int(free.sum())
        if n_free == 0:
            break
        delta = diff / n_free
        x[free] += delta
        x = np.clip(x, lo, hi)
        diff = target - x.sum()
    # residual fix
    residual = target - x.sum()
    if abs(residual) > tol:
        for i in range(len(x)):
            if lo <= x[i] + residual <= hi:
                x[i] += residual
                break
    return x

def _shift_array(arr, shift):
    """Return arr shifted by `shift` positions, zero‑filled outside."""
    n = len(arr)
    if shift > 0:
        out = np.empty_like(arr)
        out[:n - shift] = arr[shift:]
        out[n - shift:] = 0.0
        return out
    if shift < 0:
        k = -shift
        out = np.empty_like(arr)
        out[k:] = arr[:n - k]
        out[:k] = 0.0
        return out
    return arr.copy()

def _evaluate(h, dx):
    """C₅ value for discretised h."""
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def _gradient_step(h, lr, target_sum, dx):
    """Single descent step against the worst shift."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)          # integer shift corresponding to max
    up = _shift_array(h, shift)   # h[i+shift] (zero where out of range)
    down = _shift_array(h, -shift)  # h[i-shift]
    grad = (1.0 - up) - down       # ∇ C_s
    new = h - lr * grad
    new = np.clip(new, 0.0, 1.0)
    new = _project_to_sum_and_bounds(new, target_sum)
    new_val = np.max(np.correlate(new, 1.0 - new, mode='full')) * dx
    return new, new_val

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] with ∫h=1 and h∈[0,1] minimizing C₅."""
    random.seed(seed)
    np.random.seed(seed)

    # initialise
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 45
        h0 = np.random.rand(n)

    target_sum = n / 2.0            # because ∫h = 1 ⇒ sum(h)=n/2
    dx = 2.0 / n

    h0 = np.clip(h0, 0.0, 1.0)
    h0 = _project_to_sum_and_bounds(h0, target_sum)

    best_h = h0
    best_val = _evaluate(best_h, dx)

    start = time.time()
    time_limit = budget_s * 0.95   # leave a safety margin

    # ---- Gradient‑descent stage (focus on the worst shift) ----
    max_g_iter = 5000
    lr_start, lr_end = 0.5, 0.001
    for it in range(max_g_iter):
        if time.time() - start >= time_limit:
            break
        frac = it / max_g_iter
        lr = lr_start * (1 - frac) + lr_end * frac
        cand_h, cand_val = _gradient_step(best_h, lr, target_sum, dx)
        if cand_val < best_val - 1e-12:
            best_h, best_val = cand_h, cand_val
            if best_val <= 0.3808:
                break

    # ---- Stochastic refinement stage ----
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        tfrac = elapsed / time_limit
        sigma = 0.1 * (1 - tfrac) + 0.001 * tfrac

        if random.random() < 0.5:                     # Gaussian perturbation
            cand = best_h + np.random.randn(n) * sigma
            cand = np.clip(cand, 0.0, 1.0)
            cand = _project_to_sum_and_bounds(cand, target_sum)
        else:                                         # Pairwise mass transfer
            cand = best_h.copy()
            i, j = random.sample(range(n), 2)
            # move from j to i
            up = 1.0 - cand[i]
            down = cand[j]
            delta = min(up, down) * random.random()
            if delta > 0:
                cand[i] += delta
                cand[j] -= delta
            else:
                # opposite direction if previous impossible
                up = 1.0 - cand[j]
                down = cand[i]
                delta = min(up, down) * random.random()
                if delta > 0:
                    cand[j] += delta
                    cand[i] -= delta

        cand_val = _evaluate(cand, dx)

        if cand_val < best_val:
            best_h, best_val = cand, cand_val
            if best_val <= 0.3808:
                break
        else:
            # occasional uphill acceptance (simulated annealing)
            T = max(1e-4, 0.02 * (1 - tfrac))
            if np.exp(-(cand_val - best_val) / T) > random.random():
                best_h, best_val = cand, cand_val

    # ---- Fine‑grained pairwise adjustments (very cheap) ----
    while time.time() - start < time_limit:
        i, j = random.sample(range(n), 2)
        up = 1.0 - best_h[i]
        down = best_h[j]
        delta = min(up, down) * random.random()
        if delta <= 0:
            continue
        cand = best_h.copy()
        cand[i] += delta
        cand[j] -= delta
        cand_val = _evaluate(cand, dx)
        if cand_val < best_val:
            best_h, best_val = cand, cand_val
            if best_val <= 0.3808:
                break

    # final projection (just in case) and evaluation
    best_h = _project_to_sum_and_bounds(best_h, target_sum)
    final_val = _evaluate(best_h, dx)

    return (best_h, final_val, n)
