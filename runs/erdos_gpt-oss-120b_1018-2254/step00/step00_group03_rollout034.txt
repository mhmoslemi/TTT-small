def enforce_bounds_and_sum(h, target_sum, rng):
    # Ensure 0 <= h <= 1.
    h = np.clip(h, 0.0, 1.0)
    # Adjust sum exactly:
    current_sum = h.sum()
    diff = target_sum - current_sum
    if abs(diff) > 1e-12:
        # find indices where we can adjust (+/-) without violating bounds
        # If diff > 0, we need to increase some component up to 1.
        # If diff < 0, decrease some component down to 0.
        # Choose an index at random that can be adjusted.
        if diff > 0:
            # find indices where h[i] < 1
            idxs = np.where(h < 1)[0]
            if len(idxs) > 0:
                i = rng.choice(idxs)
                h[i] = min(1.0, h[i] + diff)
            else:
                # no room to increase; ignore
                pass
        else:
            idxs = np.where(h > 0)[0]
            if len(idxs) > 0:
                i = rng.choice(idxs)
                h[i] = max(0.0, h[i] + diff) # diff is negative
    return h

def neighbor_sum_for_shift(h, shift):
    n = h.size
    ns = np.zeros(n, dtype=np.float64)
    if shift > 0:
        # i + shift within range for i = 0..n-1-shift
        ns[:n-shift] += h[shift:]
        # i - shift within range for i = shift..n-1
        ns[shift:] += h[:n-shift]
    elif shift < 0:
        s = -shift
        # i + shift = i - s, valid for i >= s
        ns[s:] += h[:n-s]
        # i - shift = i + s, valid for i <= n-1-s
        ns[:n-s] += h[s:]
    else:
        # shift zero, neighbor sum is 2*h maybe? But not needed.
        ns = 2 * h
    return ns

import numpy as np, time, math

def compute_c5(h, dx):
    """C5 = max_k ∫ h(x)(1−h(x+k))dx approximated on the grid."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def propose_swap(h, rng):

    i = rng.integers(n)
    j = rng.integers(n - 1)
    if j >= i:
        j += 1
    nh = h.copy()
    nh[i], nh[j] = nh[j], nh[i]
    return nh

def propose_random_transfer(h, rng):
    inc = np.where(h < 1.0)[0]
    dec = np.where(h > 0.0)[0]
    if inc.size == 0 or dec.size == 0:
        return None
    i = rng.choice(inc)
    j = rng.choice(dec)

            return None

        return None

    nh[j] -= eps
    return nh
