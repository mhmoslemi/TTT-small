    n = len(h)
    dx = 2.0 / n
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    for i in range(steps):
        # Compute temperature
        frac = i / steps
        T = T_start * (T_end / T_start) ** frac
        # Choose move: pair random or gradient
        # We'll simple random pair move: pick i, j
        src, tgt = random.sample(range(n), 2)
        max_delta = min(h[src], 1 - h[tgt])
        if max_delta <= 0:

        delta = max_delta * random.random() * min(1.0, T*10)  # scale
        if delta <= 0:
            continue
        h_new = h.copy()
        h_new[src] -= delta
        h_new[tgt] += delta
        new_c5 = compute_c5(h_new)
