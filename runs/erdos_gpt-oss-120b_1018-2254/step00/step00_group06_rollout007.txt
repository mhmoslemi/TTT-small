import numpy as np, time, math

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr * dx)

def project_to_constraints(x, target):
    # Returns x projected to [0,1] with sum == target.
    # Use binary search on lambda.
    # Clip after adding lambda.
    # Note: x can be any float array.
    # Compute lam bounds.
    lam_low = -np.max(x)  # ensures x+lam_low <= 0 => all zeros after clip
    lam_high = 1.0 - np.min(x)  # ensures x+lam_high >= 1 => all ones after clip
    # Binary search for lam.
    for _ in range(50):
        lam_mid = (lam_low + lam_high) / 2.0
        shifted = x + lam_mid
        # use np.clip to compute sum without modifying x
        s = np.sum(np.clip(shifted, 0.0, 1.0))
        if s > target:
            lam_high = lam_mid
        else:
            lam_low = lam_mid
    lam = (lam_low + lam_high) / 2.0
    y = np.clip(x + lam, 0.0, 1.0)
    # To avoid tiny floating errors re-adjust sum; simple final scaling:
    # compute diff and adjust uniformly if needed
    # But lam binary search already ensures sum approx target.
    return y

def gradient_for_shift(h, shift):
    n = h.shape[0]
    g = np.empty(n, dtype=np.float64)
    for i in range(n):
        term1 = (1.0 - h[i + shift]) if 0 <= i + shift < n else 0.0
        term2 = h[i - shift] if 0 <= i - shift < n else 0.0
        g[i] = term1 - term2
    return g

import numpy as np
import time

def compute_c5(h: np.ndarray) -> float:
    """Return max_{k} ∫ h(x)(1‑h(x+k)) dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def project_to_constraints(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto the set {0≤y≤1, sum(y)=target}."""
    # Find λ such that sum(clamp(x+λ,0,1)) = target via binary search.
    lam_low = -np.max(x)          # all entries ≤0 → sum=0 after clipping
    lam_high = 1.0 - np.min(x)    # all entries ≥1 → sum=n after clipping
    for _ in range(50):
        lam_mid = (lam_low + lam_high) / 2.0
        s = np.sum(np.clip(x + lam_mid, 0.0, 1.0))
        if s > target:
            lam_high = lam_mid
        else:
            lam_low = lam_mid
    lam = (lam_low + lam_high) / 2.0
    return np.clip(x + lam, 0.0, 1.0)

def gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of C(shift)=∑ h[i](1‑h[i+shift]) w.r.t. h."""
    n = h.shape[0]
    g = np.empty(n, dtype=np.float64)
    for i in range(n):
        term1 = (1.0 - h[i + shift]) if 0 <= i + shift < n else 0.0
        term2 = h[i - shift] if 0 <= i - shift < n else 0.0
        g[i] = term1 - term2
    return g

    n = h.shape[0]
    target_sum = n / 2.0
    h = project_to_constraints(h, target_sum)

    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    # step sizes for gradient descent (large → small)
    alphas = [
