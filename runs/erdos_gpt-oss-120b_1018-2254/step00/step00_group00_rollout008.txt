import time
import math
import random
import numpy as np

def _compute_score(vec):
    """Return C5 bound for a discrete h."""
    n = vec.size
    dx = 2.0 / n
    corr = np.correlate(vec, 1.0 - vec, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function values, its C5 bound, and number of points.

    The function tries to improve the overlap constant C5 using a simple
    simulated‑annealing / targeted‑swap heuristic.  It works with the optional
    global `initial_h_values` if present; otherwise it starts from a uniform
    distribution.
    """
    # ----------------------------------------------------------------------
    # Setup
    # ----------------------------------------------------------------------
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()

    # Determine number of points and initialise h
    try:
        # initial_h_values may be a list, tuple or np.ndarray
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = h.size
    except Exception:
        # fallback configuration
        n_points = 200                     # modest size, well below 1k
        h = np.full(n_points, 0.5, dtype=np.float64)

    target_sum = n_points / 2.0           # because ∫h = 1  and dx = 2/n
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        # Simple projection: adjust a non‑boundary element to fix the sum
        diff = target_sum - cur_sum
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            # all entries are at a bound – force first entry
            h[0] = np.clip(h[0] + diff, 0.0, 1.0)
        else:
            idx = free[0]
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)

    # Guarantee feasibility
    h = np.clip(h, 0.0, 1.0)

    # ----------------------------------------------------------------------
    # Optimisation state
    # ----------------------------------------------------------------------
    best_h = h.copy()
    best_score = _compute_score(best_h)

    cur_h = best_h.copy()
    cur_score = best_score

    # Simulated‑annealing schedule
    T0 = 0.12               # initial temperature
    final_T = 1e-6
    T = T0

    max_step_initial = 0.45   # maximal amount transferable in one swap
    min_step = 1e-4

    # ----------------------------------------------------------------------
    # Main loop
    # ----------------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.95:
        # Compute full correlation once per iteration
        full_corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        idx_max = int(np.argmax(full_corr))
        lag = idx_max - (n_points - 1)   # integer shift where overlap peaks

        # ------------------------------------------------------------------
        # Propose a move that reduces the contribution at the worst lag
        # ------------------------------------------------------------------
        if lag != 0:
            if lag > 0:
                i_range = np.arange(n_points - lag)
                j_range = i_range + lag
                contrib = cur_h[i_range] * (1.0 - cur_h[j_range])
            else:  # lag < 0
                L = -lag
                i_range = np.arange(L, n_points)
                j_range = i_range - L
                contrib = cur_h[i_range] * (1.0 - cur_h[j_range])

            # feasible positions: we can move mass from i to j (i -> j)
            feasible = (cur_h[i_range] > 1e-12) & (cur_h[j_range] < 1.0 - 1e-12)
            if feasible.any():
                # pick the pair with largest contribution
                idx = np.argmax(contrib[feasible])
                i = i_range[feasible][idx]
                j = j_range[feasible][idx]
            else:
                # fall back to random pair
                i, j = random.sample(range(n_points), 2)
        else:
            # lag zero gives no directional info – pick random distinct pair
            i, j = random.sample(range(n_points), 2)

        # Determine maximal feasible delta (preserving bounds)
        delta_max = min(cur_h[i], 1.0 - cur_h[j])
        if delta_max <= 0.0:
            continue

        # Scale step size with temperature (cooling)
        alpha = max(T / T0, min_step / max_step_initial)
        step_limit = max_step_initial * alpha
        delta = np.random.uniform(0.0, min(delta_max, step_limit))

        # --------------------------------------------------------------
        # Apply swap
        # --------------------------------------------------------------
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta

        # Minor numerical guard
        if new_h[i] < -1e-12 or new_h[j] > 1.0 + 1e-12:
            continue

        new_score = _compute_score(new_h)

        # --------------------------------------------------------------
        # Acceptance test (simulated annealing)
        # --------------------------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / (T + 1e-12))
            if random.random() < prob:
                accept = True

        if accept:
            cur_h = new_h
            cur_score = new_score
            if new_score < best_score:
                best_h = new_h.copy()
                best_score = new_score
                # early exit if we already beat the target
                if best_score <= 0.3808:
                    break

        # --------------------------------------------------------------
        # Temperature update (linear cooling toward final_T)
        # --------------------------------------------------------------
        elapsed = time.time() - start_time
        frac = elapsed / (budget_s * 0.95)
        T = max(final_T, T0 * (1.0 - frac))

    # ----------------------------------------------------------------------
    # Return result
    # ----------------------------------------------------------------------
    return (best_h.tolist(), float(best_score), int(n_points))
