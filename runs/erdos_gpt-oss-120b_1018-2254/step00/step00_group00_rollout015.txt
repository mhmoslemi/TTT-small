def project_to_feasible(h):
    target = n_points / 2.
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        sum_h = h.sum()
        diff = target - sum_h
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to increase; eligible are those <1
            eligible = h < 1 - 1e-12
        else:
            eligible = h > 1e-12
        if not np.any(eligible):
            # cannot adjust, break
            break
        k = eligible.sum()
        # distribute diff equally (or scaled)
        delta = diff / k
        # Apply delta but limit to bounds
        # For each eligible index, apply min(max(delta, -h_i) for decreasing)
        # Actually we can compute tentative new values
        new_vals = h[eligible] + delta
        # Clip to 0-1, compute actual change applied
        new_vals = np.clip(new_vals, 0.0, 1.0)
        actual_change = new_vals - h[eligible]
        h[eligible] = new_vals
        # Continue loop
    return h

def random_move(h):
    n = len(h)
    i, j = np.random.choice(n, size=2, replace=False)
    # Determine max delta we can move from i to j without violation.
    max_transfer = min(h[i], 1 - h[j])
    if max_transfer <= 0:
        # try opposite direction or other pair
        # we can return h unchanged
        return h
    # choose delta in [0, max_transfer]
    delta = np.random.rand() * max_transfer
    new_h = h.copy()
