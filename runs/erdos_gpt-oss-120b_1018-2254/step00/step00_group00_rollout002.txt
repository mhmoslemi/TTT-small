def tournament(pop, k, rng):
    idx = rng.integers(0, len(pop), size=k)
    best = None
    best_score = None
    for i in idx:
        cand = pop[i]
        if best is None or cand[1] < best_score:
            best = cand
            best_score = cand[1]
    return best

import time
import math
import numpy as np

def _compute_c5(h):
    """Return max overlap integral for a given discretized h."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with 1-h
    vals = np.correlate(h, 1.0 - h, mode="full")
    return np.max(vals) * dx

def _project_bounds_sum(x, target, tol=1e-9, max_iter=100):
    """
    Project vector x onto the set {0<=y_i<=1, sum(y)=target}
    via bisection on the shift lambda such that y_i = clip(x_i - lambda,0,1).
    """
    n = x.size
    # generous bounds for lambda
    lo = -np.max(x) - 10.0
    hi = np.max(x) + 10.0
    # ensure bounds bracket the solution
    # sum at lo >= target ?
    sum_lo = np.clip(x - lo, 0.0, 1.0).sum()
    if sum_lo < target:
        lo = -np.max(x) - 1e3
    sum_hi = np.clip(x - hi, 0.0, 1.0).sum()
    if sum_hi > target:
        hi = np.max(x) + 1e3
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if abs(s - target) <= tol:
            return y
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def _tournament(pop, k, rng):
    """Select the best individual among k random draws."""
    idx = rng.integers(0, len(pop), size=k)
    best = pop[idx[0]]
    for i in idx[1:]:
        cand = pop[i]
        if cand[1] < best[1]:
            best = cand
    return best

def _crossover(p1, p2, rng):
    """Uniform crossover."""
    mask = rng.random(p1.shape) < 0.5
    return np.where(mask, p1, p2)

def _mutate(child, rng, rate=0.15, sigma=0.2):
    """Gaussian mutation on a fraction of genes."""
    mask = rng.random(child.shape) < rate
    noise = rng.normal(0.0, sigma, size=child.shape)
    child = child + mask * noise
    return child

def _init_population(n, rng, init_from_best=None):
    """Create initial population respecting constraints."""
    target_sum = n / 2.0
    pop = []
    pop_size = max(30, n // 6)

    # optional seed from previous stage
    if init_from_best is not None:
        h = _project_bounds_sum(init_from_best, target_sum)
        pop.append((h, _compute_c5(h)))

    # optional seed from the supplied initial_h_values (if same size)
    if "initial_h_values" in globals():
        base = np.asarray(initial_h_values, dtype=float)
        if base.size == n:
            base = _project_bounds_sum(base, target_sum)
            pop.append((base, _compute_c5(base)))

    # fill the rest randomly
    while len(pop) < pop_size:
        cand = rng.random(n)
        cand = _project_bounds_sum(cand, target_sum)
        pop.append((cand, _compute_c5(cand)))
    return pop

def _ga_optimize(pop, n, rng, time_limit):
    """Simple GA that respects the time budget."""
    start = time.time()
    target_sum = n / 2.0
    best = min(pop, key=lambda t: t[1])
    elite_sz = max(1, len(pop) // 10)

    while time.time() - start < time_limit:
        new = []
        # generate offspring until population filled (except elites)
        while len(new) < len(pop) - elite_sz:
            p1 = _tournament(pop, 3, rng)
            p2 = _tournament(pop, 3, rng)
            child = _crossover(p1[0], p2[0], rng)
            child = _mutate(child, rng, rate=0.2, sigma=0.15)
            child = _project_bounds_sum(child, target_sum)
            c5 = _compute_c5(child)
            new.append((child, c5))
            if c5 < best[1]:
                best = (child, c5)
                if best[1] <= 0.3808:
                    return best[0], best[1]

        # elitism
        elites = sorted(pop, key=lambda t: t[1])[:elite_sz]
        pop = elites + new

        # occasional early exit check
        if best[1] <= 0.3808:
            break
    return best[0], best[1]

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_all = time.time()

    # Determine base resolution
    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
    else:
        n0 = 45

    # Build a small hierarchy of resolutions (up to 180 points)
    max_n = 180
    n_list = [n0]
    while n_list[-1] < max_n:
        nxt = min(n_list[-1] * 2, max_n)
        if nxt == n_list[-1]:
            break
        n_list.append(nxt)

    # Allocate equal time per stage (simple split)
    stage_budget = budget_s / len(n_list)

    best_h = None
    best_c5 = math.inf
    best_n = n0

    for stage_idx, n in enumerate(n_list):
        # remaining global budget check
        elapsed = time.time() - start_all
        if elapsed + 0.5 >= budget_s:
            break
        # prepare starting vector for this stage
        if best_h is not None and best_h.size != n:
            # linear interpolation to new resolution
            old_x = np.linspace(0.0, 2.0, best_h.size, endpoint=False)
            new_x = np.linspace(0.0, 2.0, n, endpoint=False)
            init_vec = np.interp(new_x, old_x, best_h)
        else:
            init_vec = best_h

        # initialise population
        pop = _init_population(n, rng, init_from_best=init_vec)

        # run GA for the allocated stage time (adjusted for time already spent)
        remaining_stage = max(0.0, stage_budget - (time.time() - start_all - elapsed))
        h_stage, c5_stage = _ga_optimize(pop, n, rng, remaining_stage)

        if c5_stage < best_c5:
            best_h, best_c5, best_n = h_stage, c5_stage, n

        if best_c5 <= 0.3808:
            break

    # Final recompute for safety
    best_c5 = _compute_c5(best_h)
    return best_h, best_c5, best_n
