import time
import numpy as np
from scipy.optimize import differential_evolution

def _c5_of(h, dx):
    """Return the overlap value C5 for a candidate vector h."""
    return np.max(np.correlate(h, 1 - h, mode="full") * dx)

def _adjust_sum(vec, target):
    """Project vec onto the hyperplane sum(vec)=target while staying in [0,1]."""
    v = np.clip(vec, 0.0, 1.0)
    diff = target - v.sum()
    # greedy redistribution
    for _ in range(len(v) * 5):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            slack = 1.0 - v
            idx = np.argmax(slack)
            delta = min(diff, slack[idx])
            if delta <= 0:
                break
            v[idx] += delta
        else:
            slack = v
            idx = np.argmax(slack)          # largest entry, can be reduced
            delta = min(-diff, slack[idx])
            if delta <= 0:
                break
            v[idx] -= delta
        diff = target - v.sum()
    return np.clip(v, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound, and the number of sample points."""
    np.random.seed(seed)
    start = time.time()

    # -------------------- problem size --------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        N = len(h0)
    else:
        N = 70
        h0 = np.zeros(N, dtype=float)
        h0[: N // 2] = 1.0
        np.random.shuffle(h0)

    target_sum = N / 2.0          # ensures ∫_0^2 h = 1
    dx = 2.0 / N

    # make sure the starter respects the constraints
    h0 = _adjust_sum(h0, target_sum)

    # -------------------- global search --------------------
    penalty = 10.0

    def objective(x):
        # penalty for violating the integral constraint
        s = x.sum()
        pen = penalty * abs(s - target_sum)
        val = _c5_of(x, dx)
        return val + pen

    bounds = [(0.0, 1.0)] * N
    # modest settings keep the run fast and well within the time budget
    result = differential_evolution(
        objective,
        bounds,
        seed=seed,
        maxiter=150,
        popsize=5,
        polish=False,
        updating="deferred",
        strategy="best1bin",
    )
    h_best = np.clip(result.x, 0.0, 1.0)
    h_best = _adjust_sum(h_best, target_sum)

    best_c5 = _c5_of(h_best, dx)

    # -------------------- local refinement --------------------
    # simple pairwise delta moves with a tiny simulated‑annealing schedule
    while time.time() - start < 0.9 * budget_s:
        i, j = np.random.randint(N, size=2)
        if i == j:
            continue
        # amount that can be transferred from j to i while staying feasible
        max_inc = 1.0 - h_best[i]
        max_dec = h_best[j]
        max_delta = min(max_inc, max_dec)
        if max_delta <= 0.0:
            continue
        delta = max_delta * np.random.rand()
        cand = h_best.copy()
        cand[i] += delta
        cand[j] -= delta
        cand_c5 = _c5_of(cand, dx)

        if cand_c5 < best_c5:
            h_best, best_c5 = cand, cand_c5
        else:
            # occasional uphill moves (temperature falls linearly)
            elapsed = time.time() - start
            T = max(0.0, (budget_s - elapsed) / budget_s) * 0.1

    h_best = _adjust_sum(np.clip(h_best, 0.0, 1.0), target_sum)
