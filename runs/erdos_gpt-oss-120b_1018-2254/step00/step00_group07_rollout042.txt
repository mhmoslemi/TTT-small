def random_feasible(n, target_sum):
    # generate random numbers uniform on [0,1], then scale to meet sum while clipping
    h = np.random.rand(n)
    # scale to have sum = target_sum
    h = h / h.sum() * target_sum
    # clip values to [0,1]
    h = np.clip(h, 0, 1)
    # After clipping, we may need to adjust sum again, iterate:
    for _ in range(10):
        s = h.sum()
        if abs(s - target_sum) < 1e-10:
            break
        diff = target_sum - s
        # distribute diff uniformly to non-saturated entries
        mask = (h > 0) & (h < 1)
        if mask.any():
            # add diff equally
            h[mask] += diff / mask.sum()
            # clip again
            h = np.clip(h, 0, 1)
        else:
            break
    return h

Now project to feasible:

```
def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    # correct sum by adding/subtracting uniform offset iteratively
    for _ in range(10):
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        # distribute diff equally among all entries that can move
        mask = (h > 0) & (h < 1)
        if mask.any():
            # limit adjustment per entry to avoid overflow
            adjustment = diff / mask.sum()
            # clip per entry
            h[mask] += adjustment
            h = np.clip(h, 0.0, 1.0)
        else:
            # if all entries at bounds, just distribute to all and clip
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    return h
```

def evaluate_params(alpha, a, n):
    target_sum = n / 2
    if alpha < 0 or alpha > 1:
        return None, None
    # compute b
    if alpha >= 1.0 - 1e-12:
        # uniform case
        b = 0.5
    else:
        denom = 1 - alpha
        b = (0.5 - a * alpha) / denom
    if b < 0 or b > 1:
        return None, None
    ne = int(round(alpha * n / 2))
    # ensure not exceed
    ne = max(0, min(ne, n//2))
    center_len = n - 2*ne
    if center_len < 0:
        center_len = 0
        ne = n // 2
    # Build h
    h = np.empty(n, dtype=np.float64)
    if ne > 0:
        h[:ne] = a
        h[-ne:] = a
    if center_len > 0:
        h[ne:n-ne] = b
    # Project to ensure sum
    h = project_feasible(h, target_sum)
    c5 = compute_c5(h)
    return h, c5

import numpy as np
import time
import random

def _compute_c5(h):
    # full correlation of h with (1-h), scaled by dx = 2/n
    n = h.shape[0]
    dx = 2.0 / n
    # np.correlate uses zero padding outside the array
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return corr.max()

def _project_feasible(h, target_sum, max_iter=10):
    """Clip to [0,1] and adjust the sum to target_sum by uniform redistribution."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        # distribute diff uniformly over entries that are not at a bound
        mask = (h > 0.0) & (h < 1.0)
        if mask.any():
            adjustment = diff / mask.sum()
            h[mask] += adjustment
        else:
            # all entries at a bound; distribute over all and re‑clip
            h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős C5 value.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    random.seed(seed)

    # discretisation
    n = 200                     # number of sample points (adjustable)
    target = n / 2.0            # integral constraint: sum(h) = n/2
    dx = 2.0 / n

    # initialise h
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size != n:
            # resample to required size by linear interpolation
            x_old = np.linspace(0.0, 2.0, h0.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(x_new, x_old, h0)
        h = _project_feasible(h0, target)
    else:
        h = np.full(n, 0.5, dtype=float)   # uniform feasible start
        h = _project_feasible(h, target)

    best_h = h.copy()
    best_c5 = _compute_c5(best_h)

    start_time = time.time()
    step_size = 0.5   # initial gradient step
    iteration = 0

    # main optimisation loop (gradient descent on the current worst lag)
    while time.time() - start_time < budget_s * 0.94:
        # compute all overlap values
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        max_idx = int(np.argmax(corr))
