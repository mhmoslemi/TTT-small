def project_sum(h, target):
    # clip bounds first
    h = np.clip(h, 0.0, 1.0)
    # adjust to meet target sum via simple water-filling algorithm.
    # If sum exceeds target, reduce excess from entries >0 gradually.
    # If sum below target, increase from entries <1.
    # We'll use a while loop with tolerance.
    tolerance = 1e-9
    max_iter = 50
    for _ in range(max_iter):
        sum_h = h.sum()
        diff = target - sum_h
        if abs(diff) < tolerance:
            break
        # Determine adjustable entries
        if diff > 0:
            # need to increase; only entries <1
            mask = h < 1.0 - 1e-12
        else:
            # need to decrease; only entries >0
            mask = h > 1e-12
        n_mask = mask.sum()
        if n_mask == 0:
            break
        # amount to add to each adjustable entry
        delta = diff / n_mask
        # apply and clip
        h[mask] += delta
        h = np.clip(h, 0.0, 1.0)
    # final correction: adjust by scaling if still off
    sum_h = h.sum()
    diff = target - sum_h
    if abs(diff) > 1e-6:
        # adjust uniformly across all entries (may break bounds but diff small)
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

import time
import numpy as np
from scipy.optimize import minimize

def _compute_c5(h):
    """Exact C5 value for a discretized h."""
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h), mode='full' includes zero padding
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def _project_to_sum(h, target):
    """Project vector h onto [0,1] box and enforce sum == target (water-filling)."""
    h = np.clip(h, 0.0, 1.0)
    tol = 1e-12
    max_iter = 50
    for _ in range(max_iter):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # Select entries that can move in the needed direction
        if diff > 0:
            mask = h < 1.0 - tol
        else:
            mask = h > tol
        cnt = mask.sum()
        if cnt == 0:
            break
        delta = diff / cnt
        h[mask] += delta
        h = np.clip(h, 0.0, 1.0)
    # Final tiny correction (should be negligible)
    s = h.sum()
    if abs(s - target) > 1e-8:
        h += (target - s) / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Return a step function h (as a NumPy array), its C5 bound, and n_points.
    The function tries to lower the C5 value using constrained optimization
    and simple local search within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Determine problem size and initial vector
    # ------------------------------------------------------------------
    try:
        # `initial_h_values` is provided by the harness if a parent solution exists
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
        target_sum = n / 2.0
        h_best = _project_to_sum(h0, target_sum)
    except Exception:
        # No parent solution – fall back to a uniform feasible start
        n = 80
        target_sum = n / 2.0
        h_best = np.full(n, 0.5, dtype=float)

    # ------------------------------------------------------------------
    # Helper: smooth approximation of the maximum correlation
    # ------------------------------------------------------------------
    alpha = 50.0  # larger -> tighter approximation of max
    dx = 2.0 / n

    def _smooth_max_corr(vec):
        """Smooth max using log-sum-exp (stable)."""
        corr = np.correlate(vec, 1.0 - vec, mode='full') * dx
        vmax = corr.max()
        # stable log-sum-exp
        exps = np.exp(alpha * (corr - vmax))
        return float(vmax + np.log(exps.sum()) / alpha)

    # ------------------------------------------------------------------
    # Objective for the continuous optimizer
    # ------------------------------------------------------------------
    def objective(x):
        return _smooth_max_corr(x)

    # Equality constraint: sum(h) = target_sum
    def eq_sum(x):
        return np.sum(x) - target_sum

    constraints = ({'type': 'eq', 'fun': eq_sum},)
    bounds = [(0.0, 1.0)] * n

    # ------------------------------------------------------------------
    # Initial evaluation
    # ------------------------------------------------------------------
    best_c5 = _compute_c5(h_best)

    # ------------------------------------------------------------------
    # Main optimization loop – several restarts with SLSQP
    # ------------------------------------------------------------------
    max_restarts = 5
    for restart in range(max_restarts):
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.85:
            break  # leave margin for final local search

        if restart == 0:
            x0 = h_best.copy()
        else:
            # Perturb the current best solution
            noise = np.random.randn(n) * 0.06
            x0 = h_best + noise
            x0 = _project_to_sum(x0, target_sum)

        # SLSQP works with bounds + equality constraint
        res = minimize(
            objective,
            x0,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 1500, 'ftol': 1e-9, 'disp': False}
        )
        if not res.success:
            continue

        cand = _project_to_sum(res.x, target_sum)
        cand_c5 = _compute_c5(cand)

        if cand_c5 < best_c5:
            best_c5 = cand_c5
            h_best = cand

        # Early stop if we already beat the target bound
        if best_c5 <= 0.3808:
            break

    # ------------------------------------------------------------------
    # Simple stochastic hill‑climbing refinement (uses remaining budget)
    # ------------------------------------------------------------------
    def _local_search(h, time_limit):
        best_h = h.copy()
        best_val = _compute_c5(best_h)
        n_local = len(best_h)
        while time.time() < time_limit:
            # pick two distinct indices
            i, j = np.random.choice(n_local, size=2, replace=False)
            # move a random amount from i to j (preserve sum)
            max_move = min(best_h[i], 1.0 - best_h[j])
            if max_move <= 0:
                continue
            eps = np.random.rand() * 0.1  # step size up to 0.1
            move = min(eps, max_move)
            new_h = best_h.copy()
            new_h[i] -= move
            new_h[j] += move
            val = _compute_c5(new_h)
            if val + 1e-12 < best_val:
                best_h = new_h
                best_val = val
        return best_h, best_val

    remaining = start_time + budget_s - time.time()
    if remaining > 5.0:  # need some time to be worthwhile
        h_best, best_c5 = _local_search(h_best, start_time + budget_s - 0.5)

    # ------------------------------------------------------------------
    # Final result (ensure constraints are still satisfied)
    # ------------------------------------------------------------------
    h_best = _project_to_sum(h_best, target_sum)
    final_c5 = _compute_c5(h_best)

    return (h_best, final_c5, n)
