def project_box_hyperplane(y, target, lower=0.0, upper=1.0, tol=1e-9, max_iter=100):
    # Find lambda s.t. sum(min(upper, max(lower, y_i - lambda))) = target.
    # lambda in [low_lambda, high_lambda] where low_lambda = min(y_i - lower) (=> all at upper?), high_lambda = max(y_i - upper) ??? Let's find bounds.
    # Actually as lambda -> -inf, y_i - lambda -> +inf, min(upper, ...) = upper, sum = n*upper.
    # As lambda -> +inf, y_i - lambda -> -inf, max(lower, ...) = lower, sum = n*lower.
    # So we can binary search between lambda_low = -1e6 and lambda_high = 1e6 or compute based on data.
    # simple enough.
    lo = -1e6
    hi = 1e6
    for _ in range(max_iter):
        lam = (lo + hi) / 2
        x = np.clip(y - lam, lower, upper)
        s = np.sum(x)
        if s > target:
            lo = lam
        else:
            hi = lam
        if hi - lo < tol:
            break
    return np.clip(y - lam, lower, upper)

def project_h(y):

    # Solve for lambda.
    lower, upper = 0.0, 1.0
    target = n/2
    lo = -1e6

def anneal(initial_h, time_budget):
    import time, numpy as np
    best_h = initial_h.copy()
    best_val = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_val = best_val
    start = time.time()
    # schedule parameters
    init_temp = 0.1
    end_temp = 1e-4
    init_step = 0.2
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        temp = init_temp * (1 - frac) + end_temp * frac
        step = init_step * (1 - frac) + 1e-4 * frac
        # propose
        proposal = cur_h + step * np.random.randn(len(cur_h))
        proposal = project_h(proposal)
        val = compute_c5(proposal)
        # acceptance
        if val < cur_val or np.random.rand() < np.exp((cur_val - val) / (temp + 1e-12)):
            cur_h = proposal
            cur_val = val
            if val < best_val:
                best_h = proposal
                best_val = val
    return best_h, best_val

import time
import numpy as np

def _project_to_constraints(y, lower=0.0, upper=1.0, target_sum=None):
    """Project y onto the box [lower,upper]^n and the affine hyperplane sum=h."""
    n = y.shape[0]
    if target_sum is None:
        target_sum = n / 2.0
    lo, hi = -1e6, 1e6
    # binary search for Lagrange multiplier λ such that sum(clamp(y-λ)) = target_sum
    for _ in range(30):
        lam = (lo + hi) * 0.5
        x = np.clip(y - lam, lower, upper)
        s = x.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    x = np.clip(y - lam, lower, upper)
    # tiny residual correction (rare)
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > lower + 1e-12) & (x < upper - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt:
            x[mask] += diff / cnt
            np.clip(x, lower, upper, out=x)
    return x

def _c5_value(h):
    """Compute C₅ = max_k ∫ h(x)(1‑h(x+k)) dx using discrete sampling."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] with ∫h=1 that minimizes C₅."""
    np.random.seed(seed)
    start_time = time.time()
    # total time allocated (a bit less than the hard limit)
    time_limit = budget_s * 0.95

    # target discretisation
    n = 90                     # 90 points gives a good trade‑off
    target_sum = n / 2.0

    # initialise from the provided parent if available, otherwise use a flat start
    parent = globals().get('initial_h_values', None)
    if parent is not None:
        parent = np.asarray(parent, dtype=float)
        if parent.size != n:
            # linear interpolation to the desired size
            x_src = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            x_dst = np.linspace(0.0, 2.0, n, endpoint=False)
            parent = np.interp(x_dst, x_src, parent)
    else:
        parent = np.full(n, 0.5, dtype=float)

    cur_h = _project_to_constraints(parent, target_sum=target_sum)
    cur_val = _c5_value(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val

    # annealing schedule parameters
    init_temp, final_temp = 1.0, 1e-4
    init_step, final_step = 0.5, 1e-4

    # main simulated‑annealing loop
    while True:
        elapsed = time.time() - start_time
        if elapsed >= time_limit:
            break
        frac = elapsed / time_limit
        # linear interpolation of temperature and step size
        temp = init_temp * (1.0 - frac) + final_temp * frac
        step = init_step * (1.0 - frac) + final_step * frac

        # propose a new candidate
        if np.random.rand() < 0.02:
            # occasional full random restart
            proposal_raw = np.random.rand(n)
        else:
            proposal_raw = cur_h + step * np.random.randn(n)

        proposal = _project_to_constraints(proposal_raw, target_sum=target_sum)
        val = _c5_value(proposal)

        # Metropolis acceptance
        if val < cur_val or np.random.rand() < np.exp((cur_val - val) / (temp + 1e-16)):
            cur_h, cur_val = proposal, val
            if val < best_val:
                best_h, best_val = proposal.copy(), val
                # break early if we already beat the target bound
                if best_val <= 0.3808:
                    break

        # occasional reset to the best found point
        if np.random.rand() < 0.001:
            cur_h, cur_val = best_h.copy(), best_val

    # ensure final solution obeys constraints exactly (tiny numerical drift)
    best_h = _project_to_constraints(best_h, target_sum=target_sum)
    best_val = float(_c5_value(best_h))

    return best_h.tolist(), best_val, n
