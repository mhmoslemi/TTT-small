import time
import math
import numpy as np

def compute_c5(h):
    """Return the Erdős overlap bound for a given step vector."""
    n = h.shape[0]
    # full correlation of h and (1-h), then scale by dx = 2/n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return conv.max() * (2.0 / n)


def project_onto_bounds_sum(v, target, lower=0.0, upper=1.0, max_iter=60):
    """
    Project a vector onto the set {x | lower <= x_i <= upper, sum(x)=target}.
    Uses a simple bisection on the Lagrange multiplier.
    """
    # initial bounds for lambda
    lam_low = -np.max(v - lower) - 1.0
    lam_high = np.max(upper - v) + 1.0
    for _ in range(max_iter):
        lam = (lam_low + lam_high) / 2.0
        clipped = np.clip(v + lam, lower, upper)
        s = clipped.sum()
        if s > target:
            lam_high = lam
        else:
            lam_low = lam
    return np.clip(v + lam, lower, upper)


def run(seed=42, budget_s=1100, **kwargs):
    """
    Stochastic search for a step function h on [0,2] that minimizes
    the Erdős overlap constant C5.

    Returns:
        h_values : ndarray of shape (n_points,) with entries in [0,1]
        c5_bound : float, the achieved upper bound C5
        n_points : int, number of discretization points
    """
    np.random.seed(seed)
    # starting vector
    if "initial_h_values" in globals():
        h = np.asarray(initial_h_values, dtype=np.float64).copy()
    else:
        n = 80
        h = np.full(n, 0.5, dtype=np.float64)
    n = h.shape[0]
    target_sum = n / 2.0
    # ensure feasibility of the initial guess
    h = project_onto_bounds_sum(h, target_sum, 0.0, 1.0)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    cur_h = h.copy()
    cur_c5 = best_c5

    start_time = time.time()
    # annealing schedule parameters
    init_temp = 0.05          # start temperature
    final_temp = 1e-6         # final temperature
    max_step = 0.2            # maximal amount transferred between two indices

    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.96:
            break

        # linear temperature decay
        t = init_temp * (1.0 - elapsed / budget_s) + final_temp

        # propose a move: transfer delta from one index to another
        i, j = np.random.choice(n, 2, replace=False)
        if np.random.rand() < 0.5:
            # increase i, decrease j
            max_inc = min(max_step, 1.0 - cur_h[i])
            max_dec = min(max_step, cur_h[j])
            if max_inc <= 0.0 or max_dec <= 0.0:
                continue
            delta = np.random.rand() * min(max_inc, max_dec)
            new_h = cur_h.copy()
            new_h[i] += delta
            new_h[j] -= delta
        else:
            # increase j, decrease i
            max_inc = min(max_step, 1.0 - cur_h[j])
            max_dec = min(max_step, cur_h[i])
            if max_inc <= 0.0 or max_dec <= 0.0:
                continue
            delta = np.random.rand() * min(max_inc, max_dec)
            new_h = cur_h.copy()
            new_h[j] += delta
            new_h[i] -= delta

        # no need to re‑project: the move preserves the sum and respects bounds
        new_c5 = compute_c5(new_h)

        # acceptance test (Metropolis)
        if new_c5 < cur_c5 or np.random.rand() < math.exp(-(new_c5 - cur_c5) / (t + 1e-12)):
            cur_h = new_h

            if new_c5 < best_c5:

            cur_h = trial
            cur_c5 = compute_c5(cur_h)

    # final evaluation to ensure the bound matches the returned vector
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n
