def project_onto_box_and_sum(x, target):
    # Clip to bounds initially
    x = np.clip(x, 0.0, 1.0)
    for _ in range(max_iter):
        total = x.sum()
        diff = total - target
        if abs(diff) <= tol:
            break
        # Determine direction
        if diff > 0:
            # Need to decrease sum
            # indices where x > lower
            idx = np.where(x > 0)[0]
            if len(idx) == 0:
                break
            # Compute decrease amount per variable (equal share)
            dec = diff / len(idx)
            # Apply dec, ensuring not below lower
            x[idx] = np.maximum(0.0, x[idx] - dec)
        else:
            # Need to increase sum
            idx = np.where(x < 1)[0]
            if len(idx) == 0:
                break
            inc = -diff / len(idx)
            x[idx] = np.minimum(1.0, x[idx] + inc)
    return x
```

def project_box_sum(y, target, lower=0.0, upper=1.0, tol=1e-9, max_iter=50):
    # need to find lambda such that sum(clip(y - lambda, lower, upper)) = target
    # lambda can be in [min(y)-upper, max(y)-lower]
    lo = min(y) - upper
    hi = max(y) - lower
    # but can also bound by -max - lower etc. We'll do bisection.
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(y - lam, lower, upper)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            lo = lam
        else:
            hi = lam
    return x
```

import time
import numpy as np

def project_box_sum(y, target, lower=0.0, upper=1.0, tol=1e-9, max_iter=100):
    """
    Projects vector y onto the intersection of the box [lower,upper]^n
    and the hyperplane sum(x) = target using a bisection on the Lagrange multiplier.
    """
    lo = y.min() - upper      # at lo sum = n*upper
    hi = y.max() - lower      # at hi sum = 0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(y - lam, lower, upper)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            lo = lam
        else:
            hi = lam
    return x  # best effort after max_iter
