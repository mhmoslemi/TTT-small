import time

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, random, math, time
    random.seed(seed)
    np.random.seed(seed)

    # Determine n_points
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        n_points = len(init)
        # create binary from top half
        idx = np.argsort(-init)  # descending
        h = np.zeros(n_points, dtype=float)
        half = n_points // 2
        h[idx[:half]] = 1.0
    else:
        n_points = 200
        # start with random binary half ones
        h = np.zeros(n_points, dtype=float)
        half = n_points // 2
        ones = np.random.choice(n_points, half, replace=False)
        h[ones] = 1.0

    def compute_c(arr):
        corr = np.correlate(arr, 1 - arr, mode='full')
        return np.max(corr) * (2.0 / len(arr))

    cur_h = h.copy()
    cur_c = compute_c(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c

    start_time = time.time()
    # parameters
    max_time = budget_s * 0.95  # leave margin
    # temperature schedule: starting epsilon high
    # let's use linear cooling from initial_temp to final_temp
    initial_temp = 0.01  # adjust? For C difference scale.
    final_temp = 0.0001

    while True:
        elapsed = time.time() - start_time
        if elapsed >= max_time:
            break
        # temperature
        t = initial_temp + (final_temp - initial_temp) * (elapsed / max_time)
        # propose swap
        i = random.randrange(n_points)
        j = random.randrange(n_points)
        if cur_h[i] == cur_h[j]:
            continue
        # swap
        new_h = cur_h.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]
        new_c = compute_c(new_h)
        delta = new_c - cur_c
        if delta < 0 or random.random() < math.exp(-delta / (t + 1e-12)):
            cur_h = new_h
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()
    # final best_h may be float (0,1). Ensure sum constraint as integer
    # compute actual bound using evaluator if available
    if 'evaluate_erdos_solution' in globals():
        # evaluator maybe returns bound directly.
        try:
            bound = evaluate_erdos_solution(best_h)
        except Exception:
            bound = compute_c(best_h)
    else:
        bound = compute_c(best_h)

    # convert best_h to list for safe serialization
    h_values = best_h.tolist()
    return (h_values, bound, n_points)

def gradient_step(arr, lr):
    N = len(arr)
    # compute correlation vector
    corr = np.correlate(arr, 1 - arr, mode='full')
    # find max index
    idx_max = int(np.argmax(corr))
    s = idx_max - (N - 1)  # shift
    # compute gradient
    grad = np.zeros_like(arr)
    # compute contributions
    # term (1 - h_{i+s}) applied for i where i+s in range
    if s >= 0:
        grad[:N-s] += (1 - arr[s:])    # (1 - h_{i+s})
        grad[s:] -= arr[:N-s]          # - h_{i-s}
    else:
        s_abs = -s
        grad[s_abs:] += (1 - arr[:N-s_abs])  # because i+s >=0 => shift negative
        grad[:N-s_abs] -= arr[s_abs:]        # - h_{i-s}
    # gradient descent update
    new = arr - lr * grad
    # clip
    new = np.clip(new, 0.0, 1.0)
    # renormalize to sum target
    target = N / 2.0
    total = np.sum(new)
    if total != target:
        new = new * (target / total)
        # ensure within [0,1] after scaling
        new = np.clip(new, 0.0, 1.0)
        # re-scale iteratively maybe:
        # adjust again to meet sum exactly:
        total = np.sum(new)
        if total != 0:
            new = new * (target / total)
    return new

def project_onto_capped_simplex(v, target, cap=1.0):
    # v: np array
    # target: desired sum
    # cap: maximum bound (>= target)
    # Find tau such that sum(np.clip(v - tau, 0, cap)) = target
    # Use binary search on tau.
    lo = -np.max(v)  # lower bound maybe negative large?
    hi = np.max(v)   # upper bound
    for _ in range(50):
        tau = (lo + hi) / 2.0
        y = np.clip(v - tau, 0.0, cap)
        s = np.sum(y)
        if s > target:
            lo = tau
        else:
            hi = tau
    y = np.clip(v - (lo+hi)/2.0, 0.0, cap)
    # due to numeric, adjust sum difference:
    diff = np.sum(y) - target
    if abs(diff) > 1e-8:
        # adjust by adding/subtracting uniformly among non-boundary elements
        mask = (y > 0.0) & (y < cap)
        if np.any(mask):
            y[mask] -= diff / np.sum(mask)
            y = np.clip(y, 0.0, cap)
    return y

    if target < 0:
        target = 0.0
    # Clip also just in case target > N*cap
    N = v.size
    max_sum = N * cap
    target = min(target, max_sum)
    lo = -np.max(v) - 10.0
    hi = np.max(v) + 10.0
    for _ in range(60):

        proj = np.clip(v - tau, 0.0, cap)
        s = proj.sum()
        if s > target:
            lo = tau
        else:
            hi = tau

def gradient_step(arr, lr):
    N = arr.shape[0]
    shift, _, _ = compute_worst_shift(arr)
    grad = np.zeros_like(arr)
    if shift >= 0:
        # positive shift
        # term (1 - h_{i+shift}) for i where i+shift < N
        grad[:N-shift] += (1 - arr[shift:])
        # term -h_{i-shift} for i where i-shift >=0
        grad[shift:] -= arr[:N-shift]
    else:
        s = -shift
        # negative shift: shift = -s
        # term (1 - h_{i-s}) for i where i-s >=0 => i>=s
        grad[s:] += (1 - arr[:N-s])
        # term -h_{i+s} for i where i+s < N => i < N - s
        grad[:N-s] -= arr[s:]
    # gradient descent

    target = N / 2.0
    # clip bounds
    new = np.clip(new, 0.0, 1.0)
    # projection onto capped simplex
    new = project_onto_capped_simplex(new, target, cap=1.0)

import time
import random
import math
import numpy as np

# ---------- helper functions ----------
def compute_c(arr):
    """Return the C5 value for a step function given by arr."""
    N = arr.shape[0]
    corr = np.correlate(arr, 1.0 - arr, mode='full')
    return np.max(corr) * (2.0 / N)


def project_onto_capped_simplex(v, target, cap=1.0):
    """Euclidean projection of v onto {0<=x_i<=cap, sum x_i = target}."""
    v = np.asarray(v, dtype=np.float64)
    N = v.size
    # ensure feasible target
    target = max(0.0, min(target, N * cap))
    lo = -np.max(v) - 10.0
    hi = np.max(v) + 10.0
    for _ in range(60):
        tau = (lo + hi) * 0.5
        proj = np.clip(v - tau, 0.0, cap)
        s = proj.sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    proj = np.clip(v - (lo + hi) * 0.5, 0.0, cap)
    # tiny correction if needed
    diff = proj.sum() - target
    if abs(diff) > 1e-8:
        mask = (proj > 0.0) & (proj < cap)
        if mask.any():
            proj[mask] -= diff / mask.sum()
            proj = np.clip(proj, 0.0, cap)
    return proj


def propose_move(arr):
    """Swap a small amount of mass between two coordinates while preserving sum."""
    N = arr.shape[0]
    i, j = np.random.choice(N, 2, replace=False)
    # delta added to i, subtracted from j
    lo = max(-arr[i], arr[j] - 1.0)        # keep i >=0 and j <=1
    hi = min(1.0 - arr[i], arr[j])        # keep i <=1 and j >=0
    if lo >= hi:
        return arr
    delta = np.random.uniform(lo, hi)
    new = arr.copy()
    new[i] += delta
    new[j] -= delta
    return new


def compute_worst_shift(arr):
    """Return the shift (lag) that yields the maximal correlation."""
    N = arr.shape[0]
    corr = np.correlate(arr, 1.0 - arr, mode='full')
    idx = int(np.argmax(corr))
    shift = idx - (N - 1)   # lag value
    return shift, corr, idx


def gradient_step(arr, lr):
    """Sub‑gradient step for the active worst shift, followed by projection."""
    N = arr.shape[0]
    shift, _, _ = compute_worst_shift(arr)
    grad = np.zeros_like(arr)

    if shift >= 0:
        # (1 - h_{i+shift}) part
        grad[:N - shift] += (1.0 - arr[shift:])
        # -h_{i-shift} part
        grad[shift:] -= arr[:N - shift]
    else:
        s = -shift
        # (1 - h_{i-s}) part
        grad[s:] += (1.0 - arr[:N - s])
        # -h_{i+s} part
        grad[:N - s] -= arr[s:]

    new = arr - lr * grad
    # project back onto feasible set (box + sum constraint)
    new = np.clip(new, 0.0, 1.0)
    target = N / 2.0
    new = project_onto_capped_simplex(new, target, cap=1.0)
    return new

def run(seed=42, budget_s=1100, **kwargs):
    """Optimize a step function for the Erdős minimum overlap problem."""
    random.seed(seed)
    np.random.seed(seed)

    # ----- initialise h -----
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=np.float64)
        N = init.shape[0]
        h = np.clip(init, 0.0, 1.0)
        target = N / 2.0
        if np.sum(h) == 0:
            h[:] = target / N
        else:
            h = h * (target / np.sum(h))
        h = np.clip(h, 0.0, 1.0)
        if abs(np.sum(h) - target) > 1e-9:
            h = project_onto_capped_simplex(h, target, cap=1.0)
    else:
        N = 200
        if N % 2:
            N += 1
        h = np.random.rand(N)
        h = np.clip(h, 0.0, 1.0)
        target = N / 2.0
        h = h * (target / np.sum(h))
        h = np.clip(h, 0.0, 1.0)
        if abs(np.sum(h) - target) > 1e-9:
            h = project_onto_capped_simplex(h, target, cap=1.0)

    # ----- optimisation loop -----
    best_h = h.copy()
    best_c = compute_c(best_h)
    cur_h = h.copy()
    cur_c = best_c

    start_time = time.time()
    end_time = start_time + budget_s * 0.95  # leave a safety margin

    while time.time() < end_time:
        # linear cooling for temperature and learning rate
        remain_frac = (end_time - time.time()) / (budget_s * 0.95)
        temp = 0.04 * remain_frac + 1e-9          # temperature for SA
        lr = 0.12 * remain_frac                  # learning rate for gradient steps

        if random.random() < 0.22:
            # sub‑gradient descent step
            new_h = gradient_step(cur_h, lr)
        else:
            # simple mass‑swap move
            new_h = propose_move(cur_h)

        new_c = compute_c(new_h)
        delta = new_c - cur_c

        if delta < 0 or random.random() < math.exp(-delta / temp):
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()

    # final bound (use provided evaluator if available)
    try:
        # evaluate_erdos_solution may be pre‑imported
        bound = evaluate_erdos_solution(best_h)
    except Exception:
        bound = best_c

    return (best_h.tolist(), float(bound), N)
