def mutate(h, step):
    new_h = h.copy()
    num_swaps = np.random.randint(1, 4) # 1-3 swaps
    n = len(h)
    for _ in range(num_swaps):
        i, j = np.random.choice(n, size=2, replace=False)
        # max delta to keep bounds
        max_inc = min(step, 1 - new_h[i], new_h[j]) # increase i, decrease j
        max_dec = min(step, new_h[i], 1 - new_h[j]) # decrease i, increase j
        # choose direction
        if max_inc > 0 and max_dec > 0:
            # randomly choose sign
            if np.random.rand() < 0.5:
                delta = np.random.uniform(0, max_inc)
                new_h[i] += delta
                new_h[j] -= delta
            else:
                delta = np.random.uniform(0, max_dec)
                new_h[i] -= delta
                new_h[j] += delta
        elif max_inc > 0:
            delta = np.random.uniform(0, max_inc)
            new_h[i] += delta
            new_h[j] -= delta
        elif max_dec > 0:
            delta = np.random.uniform(0, max_dec)
            new_h[i] -= delta
            new_h[j] += delta
        else:
            # cannot adjust due to bounds; skip
            pass
    return new_h

def compute_c5(h):
   n = h.shape[0]
   dx = 2.0 / n
   corr = np.correlate(h, 1 - h, mode='full')
   return float(np.max(corr) * dx)

def mutate_random(h, step):
    n = h.shape[0]
    new_h = h.copy()
    # perform a few (1-3) balanced swaps
    swaps = random.randint(1, 3)
    for _ in range(swaps):
        i, j = np.random.choice(n, size=2, replace=False)
        hi, hj = new_h[i], new_h[j]
        # compute max delta increasing i decreasing j
        max_inc = min(step, 1.0 - hi, hj)
        max_dec = min(step, hi, 1.0 - hj)
        # choose direction
        if max_inc <= 0 and max_dec <= 0:
            continue
        if max_inc > 0 and max_dec > 0:
            if random.random() < 0.5:
                delta = random.uniform(0, max_inc)
                new_h[i] += delta
                new_h[j] -= delta
            else:
                delta = random.uniform(0, max_dec)
                new_h[i] -= delta
                new_h[j] += delta
        elif max_inc > 0:
            delta = random.uniform(0, max_inc)
            new_h[i] += delta
            new_h[j] -= delta
        else:  # max_dec > 0
            delta = random.uniform(0, max_dec)
            new_h[i] -= delta
            new_h[j] += delta
    return new_h

def mutate_targeted(h, step):
    n = h.shape[0]
    # compute correlation array
    corr = np.correlate(h, 1 - h, mode='full')
    # find shift with maximal correlation
    idx = int(np.argmax(corr))
    k = idx - (n - 1)
    new_h = h.copy()
    if k >= 0:
        max_i = n - k
        # candidate indices where reduction possible
        mask = (new_h[:max_i] > 0) & (new_h[k:] < 1)
        candidates = np.flatnonzero(mask)
        if candidates.size == 0:
            return new_h
        i = int(np.random.choice(candidates))
        max_delta = min(step, new_h[i], 1.0 - new_h[i + k])
        if max_delta <= 0:
            return new_h
        delta = random.uniform(0, max_delta)
        new_h[i] -= delta
        new_h[i + k] += delta
    else:
        pos = -k
        # i runs from pos to n-1
        mask = (new_h[pos:] > 0) & (new_h[:n - pos] < 1)
        candidates = np.flatnonzero(mask) + pos

import numpy as np
import random
import math
import time

def _compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1-h(x+k)) dx approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _mutate_random(h: np.ndarray, step: float) -> np.ndarray:
    """Balanced random move: pick 1‑3 index pairs and shift mass while preserving sum."""
    n = h.shape[0]
    new_h = h.copy()
    swaps = random.randint(1, 3)
    for _ in range(swaps):
        i, j = np.random.choice(n, size=2, replace=False)
        hi, hj = new_h[i], new_h[j]

        # amount we could add to i (remove from j) and vice‑versa
        max_inc = min(step, 1.0 - hi, hj)      # increase i, decrease j
        max_dec = min(step, hi, 1.0 - hj)      # decrease i, increase j

        if max_inc <= 0 and max_dec <= 0:
            continue
        if max_inc > 0 and max_dec > 0:
            if random.random() < 0.5:
                delta = random.uniform(0.0, max_inc)
                new_h[i] += delta
                new_h[j] -= delta
            else:
                delta = random.uniform(0.0, max_dec)
                new_h[i] -= delta
                new_h[j] += delta
        elif max_inc > 0:
            delta = random.uniform(0.0, max_inc)
            new_h[i] += delta
            new_h[j] -= delta
        else:  # max_dec > 0
            delta = random.uniform(0.0, max_dec)
            new_h[i] -= delta
            new_h[j] += delta
    return new_h


def _mutate_targeted(h: np.ndarray, step: float) -> np.ndarray:
    """Move mass from a 1‑position to a 0‑position that reduces the worst shift."""
    n = h.shape[0]
    # correlation to locate the shift with maximal overlap
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    k = idx - (n - 1)                     # signed shift
    new_h = h.copy()

    if k >= 0:
        max_i = n - k
        mask = (new_h[:max_i] > 0.0) & (new_h[k:] < 1.0)
        cand = np.flatnonzero(mask)
        if cand.size == 0:
            return new_h
        i = int(np.random.choice(cand))
        max_delta = min(step, new_h[i], 1.0 - new_h[i + k])
        if max_delta <= 0.0:
            return new_h
        delta = random.uniform(0.0, max_delta)
        new_h[i] -= delta
        new_h[i + k] += delta
    else:
        pos = -k
        mask = (new_h[pos:] > 0.0) & (new_h[:n - pos] < 1.0)
        cand = np.flatnonzero(mask) + pos
        if cand.size == 0:
            return new_h
        i = int(np.random.choice(cand))
        max_delta = min(step, new_h[i], 1.0 - new_h[i - pos])
        if max_delta <= 0.0:
            return new_h
        delta = random.uniform(0.0, max_delta)
        new_h[i] -= delta
        new_h[i - pos] += delta
    return new_h
