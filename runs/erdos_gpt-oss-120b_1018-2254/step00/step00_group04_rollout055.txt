def project_to_simplex_bounds(x, target):
    y = np.clip(x, 0, 1)
    # iterative water-filling
    for _ in range(10):  # enough
        sum_y = y.sum()
        if abs(sum_y - target) < 1e-9:
            break
        if sum_y < target:
            free = (y < 1 - 1e-12)
            if not free.any():
                # cannot increase, failure, return as is
                break
            delta = (target - sum_y) / free.sum()
            y[free] = np.minimum(y[free] + delta, 1)
        else:  # sum_y > target
            free = (y > 1e-12)
            if not free.any():
                break
            delta = (sum_y - target) / free.sum()
            y[free] = np.maximum(y[free] - delta, 0)
    # final clip
    return np.clip(y, 0, 1)

def repair_h(h, target_sum):
    # target_sum = n_points/2
    # Clip to [0,1] and project to sum.
    y = np.clip(h, 0.0, 1.0)
    # water-filling iterative
    for _ in range(10):
        s = y.sum()
        if abs(s - target_sum) < 1e-12:
            break
        if s < target_sum:
            free = y < 1.0
            if not free.any():
                break
            delta = (target_sum - s) / free.sum()
            y[free] = np.minimum(y[free] + delta, 1.0)
        else:
            free = y > 0.0
            if not free.any():
                break
            delta = (s - target_sum) / free.sum()
            y[free] = np.maximum(y[free] - delta, 0.0)
    # final clipping
    y = np.clip(y, 0.0, 1.0)
    # Slight adjust numeric errors:
    # Ensure sum is close: if small error, distribute among free.
    diff = target_sum - y.sum()

import numpy as np, random, time, math

def repair_h(h, target_sum):
    y = np.clip(h, 0.0, 1.0)
    # iterative water-filling
    for _ in range(20):
        s = y.sum()
        if abs(s - target_sum) < 1e-12:
            break
        if s < target_sum:
            # need to increase some below 1
            free = y < 1.0 - 1e-12
            if not free.any():
                break
            delta = (target_sum - s) / free.sum()
            y[free] = np.minimum(y[free] + delta, 1.0)
        else:
            # need to decrease some above 0
            free = y > 1e-12
            if not free.any():
                break
            delta = (s - target_sum) / free.sum()
            y[free] = np.maximum(y[free] - delta, 0.0)
    # final clip
    y = np.clip(y, 0.0, 1.0)
    # ensure sum exactly target via tiny adjustment on free entries
    diff = target_sum - y.sum()
    if abs(diff) > 1e-12:
        free = (y > 1e-12) & (y < 1.0 - 1e-12)
        if free.any():
            y[free] += diff / free.sum()
            y = np.clip(y, 0.0, 1.0)
    return y

def propose_transfer(h):
    n = h.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    # compute max transfer in both directions
    max_up_i = 1.0 - h[i]
    max_down_j = h[j]
    max_transfer_ij = min(max_up_i, max_down_j)
    max_up_j = 1.0 - h[j]
    max_down_i = h[i]
    max_transfer_ji = min(max_up_j, max_down_i)
    if max_transfer_ij <= 0 and max_transfer_ji <= 0:
        return h.copy()
    # choose direction randomly weighted
    if max_transfer_ij > 0 and max_transfer_ji > 0:
        if random.random() < 0.5:
            delta = random.random() * max_transfer_ij
            new = h.copy()
            new[i] += delta
            new[j] -= delta
            return new
        else:
            delta = random.random() * max_transfer_ji
            new = h.copy()
            new[i] -= delta
            new[j] += delta
            return new
    elif max_transfer_ij > 0:
        delta = random.random() * max_transfer_ij
        new = h.copy()
        new[i] += delta
        new[j] -= delta
        return new
    else:
        delta = random.random() * max_transfer_ji
        new = h.copy()
        new[i] -= delta
        new[j] += delta
        return new

def compute_c5(h):
    # evaluate using pre-imported helper if available
    try:
        # evaluate_erdos_solution expected to return (c5, other)
        return float(evaluate_erdos_solution(h)[0])
    except Exception:
        # fallback to manual correlation
        n = h.shape[0]
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        return float(np.max(corr))

    # Evaluate initial best

    best_score = compute_c5(best_h)
    # Also consider a random start
    # But we will use random restarts within loop
    # Control parameters
    non_improve = 0

    while True:
        now = time.time()
        if now - start > budget_s * 0.97:
            break
        # Propose neighbor from best
        cand = propose_transfer(best_h)
        # Ensure constraints (project)
        cand = repair_h(cand, target_sum)
        cand_score = compute_c5(cand)
        if cand_score < best_score - 1e-12:
            best_h = cand
            best_score = cand_score
            non_improve = 0
            continue
        else:

        # occasional acceptance of worse move to escape local minima

            best_score = cand_score

        # random restart if stuck too long

            else:

import numpy as np
import random

def _project_to_sum_and_bounds(x, target_sum):
    """Project vector x onto the hypercube [0,1]^n with given L1 sum."""
    y = np.clip(x, 0.0, 1.0)
    # Water-filling style iterative correction
    for _ in range(20):
        s = y.sum()
        if abs(s - target_sum) < 1e-12:
            break
        if s < target_sum:
            free = y < 1.0 - 1e-12
            if not free.any():
                break
            delta = (target_sum - s) / free.sum()
            y[free] = np.minimum(y[free] + delta, 1.0)
        else:
            free = y > 1e-12
            if not free.any():
                break
            delta = (s - target_sum) / free.sum()
            y[free] = np.maximum(y[free] - delta, 0.0)
    y = np.clip(y, 0.0, 1.0)
    diff = target_sum - y.sum()
    if abs(diff) > 1e-12:
        free = (y > 1e-12) & (y < 1.0 - 1e-12)
        if free.any():
            y[free] += diff / free.sum()
            y = np.clip(y, 0.0, 1.0)
    return y

def repair_h(h, target_sum):
    """Clamp h to [0,1] and enforce the integral constraint."""
    return _project_to_sum_and_bounds(h, target_sum)

def propose_transfer(h):
    """Generate a nearby feasible vector by moving mass between two entries."""
    n = h.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    # direction i -> j
    max_up_i = 1.0 - h[i]
    max_down_j = h[j]
    max_ij = min(max_up_i, max_down_j)
    # direction j -> i
    max_up_j = 1.0 - h[j]
    max_down_i = h[i]
    max_ji = min(max_up_j, max_down_i)

    if max_ij <= 0 and max_ji <= 0:
        return h.copy()

    if max_ij > 0 and max_ji > 0:
        if random.random() < 0.5:
            delta = random.random() * max_ij
            new = h.copy()
            new[i] += delta
            new[j] -= delta
            return new
        else:
            delta = random.random() * max_ji
            new = h.copy()
            new[i] -= delta
            new[j] += delta
            return new
    elif max_ij > 0:
        delta = random.random() * max_ij
        new = h.copy()
        new[i] += delta
        new[j] -= delta
        return new
    else:
        delta = random.random() * max_ji
        new = h.copy()
        new[i] -= delta
        new[j] += delta
        return new

def _fallback_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))
