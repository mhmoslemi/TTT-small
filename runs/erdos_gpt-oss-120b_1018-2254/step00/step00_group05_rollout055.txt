    N = len(h)
    dx = 2.0 / N
    M = 1 << ((2*N - 1).bit_length())  # next power of two

    conv = conv[:2*N-1]

    return max_val * dx

def _compute_c5(h):
    """Return the overlap constant C5 for a given h."""
    try:
        # helper provided by the harness
        return evaluate_erdos_solution(h)
    except Exception:
        N = h.shape[0]
        dx = 2.0 / N
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr) * dx

def _smooth(h, sigma):
    """Gaussian smoothing (same mode, keep length)."""
    if sigma <= 0:
        return h.copy()
    radius = int(math.ceil(3 * sigma))
    x = np.arange(-radius, radius + 1, dtype=float)
    kernel = np.exp(-0.5 * (x / sigma) ** 2)
    kernel /= kernel.sum()
    return np.convolve(h, kernel, mode="same")

def _project_bounds(v, target):
    """
    Euclidean projection of vector v onto the polytope:
        0 <= x_i <= 1, sum(x) = target
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(v - lam, 0.0, 1.0)

    # tiny correction to hit the target sum exactly
    diff = target - proj.sum()
    if abs(diff) > 1e-10:
        free = np.where((proj > 1e-12) & (proj < 1 - 1e-12))[0]
        if free.size > 0:
            proj[free] += diff / free.size

            proj[idx] = np.clip(proj[idx] + diff, 0.0, 1.0)
        proj = np.clip(proj, 0.0, 1.0)

    return proj

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # --- problem size -------------------------------------------------
    N = 200                     # number of samples (dx = 2/N)
    target_sum = N / 2.0        # required integral constraint

    # --- initial construction -----------------------------------------
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=float)
        x_old = np.linspace(0.0, 2.0, init.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, N, endpoint=False)
        h = np.interp(x_new, x_old, init)
    else:
        # start from a simple 0‑1 pattern (exactly half ones)
        h = np.zeros(N, dtype=float)
        h[:N // 2] = 1.0
        rng.shuffle(h)

    h = np.clip(h, 0.0, 1.0)
    h = _project_bounds(h, target_sum)

    best_h = h.copy()
    best_val = _compute_c5(best_h)

    # --- deterministic smoothing sweep --------------------------------
    sigma_list = [0.5, 1.0, 1.5, 2.0, 2.5, 3.0]
    for sigma in sigma_list:
        if time.time() - start_time > budget_s * 0.7:
            break
        h_s = _smooth(best_h, sigma)
        h_s = _project_bounds(h_s, target_sum)
        val = _compute_c5(h_s)
        if val < best_val:
            best_val = val
            best_h = h_s.copy()

    # --- stochastic refinement ----------------------------------------
    cur_h = best_h.copy()
    cur_val = best_val
    no_improve = 0
    iteration = 0
    max_iter = 10_000_000  # safety upper bound

    while time.time() - start_time < budget_s * 0.95 and iteration < max_iter:
        iteration += 1
        # propose a small mass exchange between two coordinates
        i, j = rng.integers(N, size=2)
        if i == j:
            continue
        delta = (rng.random() - 0.5) * 0.2  # move at most ±0.1
        hi, hj = cur_h[i], cur_h[j]
        new_hi = hi + delta
        new_hj = hj - delta
        if not (0.0 <= new_hi <= 1.0 and 0.0 <= new_hj <= 1.0):
            continue
        cand = cur_h.copy()
        cand[i] = new_hi
        cand[j] = new_hj
        # sum is preserved automatically
        cand = _project_bounds(cand, target_sum)  # correct possible rounding drift
        cand_val = _compute_c5(cand)

        if cand_val < cur_val:
            cur_h, cur_val = cand, cand_val
            if cand_val < best_val:
                best_val, best_h = cand_val, cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # occasional uphill move (simulated annealing)
            elapsed = time.time() - start_time
            remaining = max(budget_s - elapsed, 1e-6)
            temp = remaining / budget_s  # linearly decreasing temperature
            if rng.random() < math.exp(-(cand_val - cur_val) / (temp + 1e-12)):
                cur_h, cur_val = cand, cand_val
                no_improve += 1
            else:
                no_improve += 1

        # restart if stagnating
        if no_improve > 5000:
            # perturb the current best a little
            pert = _smooth(best_h, 0.8)
            cur_h = _project_bounds(pert, target_sum)
            cur_val = _compute_c5(cur_h)
            no_improve = 0

    # final safety projection and evaluation
    best_h = _project_bounds(best_h, target_sum)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_val = _compute_c5(best_h)

    return best_h, final_val, N
