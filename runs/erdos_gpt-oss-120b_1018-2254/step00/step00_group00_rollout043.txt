def project_box_sum(x, target):
    # binary search for lambda that makes sum (clip(x + lam)) = target
    lo, hi = -2.0, 2.0
    for _ in range(40):
        lam = (lo + hi) / 2.0
        y = np.clip(x + lam, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    y = np.clip(x + (lo+hi)/2.0, 0.0, 1.0)
    # ensure numerical precision
    correction = target - y.sum()
    # distribute small correction equally among non-saturated entries
    # but correction should be within few eps; just adjust first element within bounds
    if abs(correction) > 1e-12:
        # find index not at bound
        free = (y > 1e-12) & (y < 1-1e-12)
        if free.any():
            idx = np.flatnonzero(free)[0]
            y[idx] = np.clip(y[idx] + correction, 0.0, 1.0)
    return y

def random_swap(h, eps, target):
    n = h.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = (np.random.rand()*2-1)*eps
    # attempt to apply delta: h_i += delta, h_j -= delta
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    # clip and project
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_box_sum(h_new, target)
    val, _ = max_overlap(h_new)
    return h_new, val
