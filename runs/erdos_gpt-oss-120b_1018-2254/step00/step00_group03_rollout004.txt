def project_onto_box_and_sum(v, target_sum):
    lo = 0.0; hi = 1.0
    # binary search for lambda
    lam_low = -1.0
    lam_high = 1.0
    # ensure sum(clip(v+lam_low)) <= target; sum(clip(v+lam_high)) >= target
    def sum_clip(lam):
        return np.sum(np.clip(v + lam, lo, hi))
    # expand bounds if needed
    while sum_clip(lam_low) > target_sum:
        lam_low -= 1.0
    while sum_clip(lam_high) < target_sum:
        lam_high += 1.0
    for _ in range(50):
        lam_mid = (lam_low + lam_high) / 2.0
        s = sum_clip(lam_mid)
        if s < target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v + lam_mid, lo, hi)
```

def gradient_of_corr(h, lag, dx):
    n = h.size
    grad = np.zeros_like(h)
    if lag >= 0:
        # term1: (1 - h[i+lag]) for i in [0, n-lag)
        grad[:n-lag] += (1 - h[lag:])
        # term2: -h[i-lag] for i in [lag, n)
        grad[lag:] -= h[:n-lag]
    else:
        # lag negative; let l = -lag
        l = -lag
        # term1: (1 - h[i+lag]) means i+lag = i - l >=0; i >= l
        grad[l:] += (1 - h[:n-l])
        # term2: -h[i - lag] = -h[i + l] for i in [0, n-l)
        grad[:n-l] -= h[l:]
    return grad * dx

def project_onto_box_and_sum(v, target_sum, lo=0.0, hi=1.0, tol=1e-10, max_iter=100):
    # v: ndarray
    # Returns x = clip(v + lam, lo, hi) sum = target_sum
    # ensure feasible: target_sum within [lo*n, hi*n]
    n = v.size
    # initial bounds for lambda
    lam_low = lo - np.max(v) - 1.0
    lam_high = hi - np.min(v) + 1.0
    def sum_clip(lam):
        return np.sum(np.clip(v + lam, lo, hi))
    # ensure bounds bracket target
    while sum_clip(lam_low) > target_sum:
        lam_low -= 1.0
    while sum_clip(lam_high) < target_sum:
        lam_high += 1.0
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) / 2.0
        s = sum_clip(lam_mid)
        if s < target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
        if lam_high - lam_low < tol:
            break
    return np.clip(v + lam_mid, lo, hi)
```

import numpy as np
import time


def project_onto_box_and_sum(v, target_sum, lo=0.0, hi=1.0, tol=1e-10, max_iter=100):
    """Project vector v onto the box [lo,hi]^n intersected with the hyperplane sum(v)=target_sum."""
    n = v.size
    # initial bounds for the Lagrange multiplier
    lam_low = lo - np.max(v) - 1.0
    lam_high = hi - np.min(v) + 1.0

    def sum_clip(lam):
        return np.sum(np.clip(v + lam, lo, hi))

    # enlarge bounds until they bracket the target
    while sum_clip(lam_low) > target_sum:
        lam_low -= 1.0
    while sum_clip(lam_high) < target_sum:
        lam_high += 1.0

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) / 2.0
        s = sum_clip(lam_mid)
        if s < target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
        if lam_high - lam_low < tol:
            break
    return np.clip(v + lam_mid, lo, hi)


def calc_c5(h, dx):
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx for the discrete vector h."""
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx


def gradient_of_corr(h, lag, dx):
    """Gradient of the correlation term at a given lag (including the dx factor)."""
    n = h.size
    grad = np.zeros_like(h)
    if lag >= 0:
        # (1 - h[i+lag]) contribution
        grad[: n - lag] += (1 - h[lag:])
        # -h[i-lag] contribution
        grad[lag:] -= h[: n - lag]
    else:
        l = -lag
        grad[l:] += (1 - h[: n - l])
        grad[: n - l] -= h[l:]
    return grad * dx

        init_h = project_onto_box_and_sum(init_h, n / 2.0)

    dx = 2.0 / n
    target_sum = n / 2.0

    # Ensure feasibility
    init_h = np.clip(init_h, 0.0, 1.0)
    init_h = project_onto_box_and_sum(init_h, target_sum)

    best_h = init_h.copy()
    best_c5 = calc_c5(best_h, dx)

    # Current working point (may differ from global best)
    cur_h = best_h.copy()
    cur_c5 = best_c5

    # ----------------------------------------------------------------------
    # Optimisation parameters
    # ----------------------------------------------------------------------
    step = 0.5          # step size for gradient moves
    epsilon = 0.05      # magnitude for random perturbations
    no_improve = 0      # counter for stagnation detection

    # ----------------------------------------------------------------------
    # Main search loop
    # ----------------------------------------------------------------------
    while time.time() - start < budget_s * 0.95:
        if np.random.rand() < 0.3:

            corr_full = np.correlate(cur_h, 1 - cur_h, mode='full')
