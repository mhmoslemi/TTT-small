import numpy as np
import cvxpy as cp
import time

def solve_milp(n, time_limit):
    h = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    constraints = [cp.sum(h) == n//2]  # n is even
    # For each shift
    for k in range(-n+1, n):
        if k == 0:
            continue
        L = n - abs(k)
        # indices i from max(0,-k) to min(n-1,n-1-k)
        i_start = max(0, -k)
        i_end = min(n, n - k)  # exclusive
        # compute list of i positions
        # w vector for this shift: length L
        w = cp.Variable(L, boolean=True)
        # constraints linking w to h
        # for each j in 0..L-1, mapping to i = i_start + j, i+k
        # w_j <= h[i]
        # w_j <= 1 - h[i+k]
        # w_j >= h[i] - h[i+k]
        constraints += [w <= h[i_start:i_end]]
        constraints += [w <= 1 - h[i_start + k : i_end + k]]
        constraints += [w >= h[i_start:i_end] - h[i_start + k : i_end + k]]
        # sum w <= t

def evaluate_corr(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def random_balanced(n, rng):
    h = np.zeros(n, dtype=int)
    ones = rng.choice(n, n//2, replace=False)
    h[ones] = 1
    return h

def hill_climb(n, initial, budget_s, start_time):
    dx = 2.0 / n
    h = initial.copy()
    best = h.copy()
    best_score = evaluate_corr(best.astype(float))
    # determine temperature schedule
    T0 = 0.1
    while time.time() - start_time < budget_s:
        # compute temperature based on elapsed fraction
        elapsed = time.time() - start_time
        frac = elapsed / budget_s
        T = T0 * (1 - frac)
        # pick random swap
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)
        # swap
        h[i], h[j] = 0, 1
        cur_score = evaluate_corr(h.astype(float))
        delta = cur_score - best_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-12)):
            # accept move
            if cur_score < best_score:
                best_score = cur_score
                best = h.copy()
        else:
            # revert
            h[i], h[j] = 1, 0
    return best.astype(float), best_score

import time
import numpy as np
import cvxpy as cp

def _max_overlap(h):
    """Return max_{k} integral h(x)*(1-h(x+k)) dx for discrete h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return float(np.max(corr))

def _random_balanced(n, rng):
    """Return a random binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=int)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1
    return h

def _hill_climb(n, init_h, budget, start_time):
    """Simple simulated‑annealing hill‑climb for binary h."""
    dx = 2.0 / n
    h = init_h.astype(int).copy()
    best = h.copy()
    best_score = _max_overlap(best.astype(float))
    # temperature schedule (linear decay)
    T0 = 0.1
    while time.time() - start_time < budget:
        elapsed = time.time() - start_time
        frac = elapsed / budget
        T = max(1e-9, T0 * (1.0 - frac))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h[i], h[j] = 0, 1
        cur_score = _max_overlap(h.astype(float))
        delta = cur_score - best_score
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            if cur_score < best_score:
                best_score = cur_score
                best = h.copy()
        else:
            h[i], h[j] = 1, 0
    return best.astype(float), best_score

        cons = [cp.sum(h) == n // 2, t >= 0]

            if k == 0:

            i0 = max(0, -k)

            L = i1 - i0

            hi = h[i0:i1]

        prob = cp.Problem(cp.Minimize(t), cons)
