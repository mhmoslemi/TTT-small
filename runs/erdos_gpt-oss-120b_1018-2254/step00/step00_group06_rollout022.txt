        new_h[donor] -= delta
        new_h[receiver] += delta

import numpy as np
import time
import math

def _c5_and_lag(h, dx):
    """Return C5 = max correlation * dx and the lag at which it occurs."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(corr))
    max_corr = float(corr[max_idx])
    lag = max_idx - (len(h) - 1)
    return max_corr * dx, lag

def _gradient(h, lag):
    """Gradient of the worst‑lag correlation w.r.t. h."""
    n = len(h)
    g = np.zeros(n, dtype=float)
    for i in range(n):
        j = i + lag
        if 0 <= j < n:
            g[i] += 1.0 - h[j]
        j = i - lag
        if 0 <= j < n:
            g[i] -= h[j]
    return g

def _project(h):
    """Clip to [0,1] and enforce sum(h)=n/2 exactly."""
    n = len(h)
    target = n * 0.5
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        free = (h > 0.0) & (h < 1.0)
        cnt = np.count_nonzero(free)
        if cnt == 0:
            # All at bounds – distribute uniformly then clip again.
            h += diff / n
            np.clip(h, 0.0, 1.0, out=h)
        else:
            adj = diff / cnt
            h[free] += adj
            np.clip(h, 0.0, 1.0, out=h)
