def repair_h(h):
    N = len(h)
    target = N / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) > 1e-9:
        # simple scaling
        # To preserve bounds, perform iterative water-filling:
        # Use cvxpy for guaranteed feasible projection for simplicity.
        try:
            import cvxpy as cp
            var = cp.Variable(N)
            objective = cp.Minimize(cp.sum_squares(var - h))
            constraints = [var >= 0, var <= 1, cp.sum(var) == target]
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)
            if var.value is not None:
                h = np.array(var.value).flatten()
                # Clip just in case numerical errors
                h = np.clip(h, 0.0, 1.0)
                # Ensure sum exactly target by adjusting small epsilon
                diff = target - h.sum()
                if abs(diff) > 1e-8:
                    # add diff to first non-bound
                    for i in range(N):
                        if 0 <= h[i] + diff <= 1:
                            h[i] += diff
                            break
        except Exception:
            # fallback simple scaling (may violate bounds)
            scale = target / s
            h = np.clip(h * scale, 0.0, 1.0)
            # adjust slight difference:
            diff = target - h.sum()
            if abs(diff) > 1e-8:
                # adjust first entry within bounds
                for i in range(N):
                    if 0 <= h[i] + diff <= 1:
                        h[i] += diff
                        break
    return h

def project_box_sum(v, lower=0.0, upper=1.0, target_sum=None):
    N = len(v)
    if target_sum is None:
        target_sum = N / 2.0
    # lambda bounds
    lam_low = -np.max(v) - upper  # ensure v+lam <= lower for all? Actually we want v+lam < lower; lower bound may be 0.
    lam_low = -np.max(v) - 1.0  # safe.
    lam_high = upper - np.min(v) + 1.0  # safe.
    for _ in range(60):
        lam = (lam_low + lam_high) / 2.0
        h = np.clip(v + lam, lower, upper)
        s = h.sum()
        if s < target_sum:
            lam_low = lam
        else:
            lam_high = lam
    h = np.clip(v + (lam_low + lam_high) / 2.0, lower, upper)
    # adjust residual
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        # find index to adjust without violating bounds
        # concentrate on capacity to absorb diff
        for i in range(N):
            if lower < h[i] + diff < upper:
                h[i] += diff
                break
        else:
            # fallback: distribute diff evenly and clip again
            h += diff / N
            h = np.clip(h, lower, upper)
            # final adjustment if still off
            residual = target_sum - h.sum()
            if abs(residual) > 1e-10:
                # adjust one element if possible
                for i in range(N):
                    if lower <= h[i] + residual <= upper:
                        h[i] += residual
                        break
    return h

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    np.random.seed(seed)
    start = time.time()
    # Determine size
    if 'initial_h_values' in globals():
        N = len(initial_h_values)
    else:
        N = 100  # choose even number
        if N % 2 != 0:
            N += 1
    # ensure even
    if N % 2 != 0:
        N += 1
    # initial candidate
    best_h = get_initial_h(N)
    best_score = compute_c5(best_h)
    # current candidate (heat)
    cur_h = best_h.copy()
    cur_score = best_score
    # parameters
    sigma = 0.2  # step size for Gaussian perturbation, relative to [0,1]
    T = 0.1  # initial temperature
    # schedule decay
    decay_sigma = 0.9995
    decay_T = 0.9998
    iter_count = 0
    # main loop
    while time.time() - start < budget_s:
        # propose
        cand = cur_h + np.random.randn(N) * sigma
        # project to constraints
        cand = project_box_sum(cand, 0.0, 1.0, N/2.0)
        cand_score = compute_c5(cand)
        delta = cand_score - cur_score
        if delta < 0 or np.exp(-delta / max(T, 1e-12)) > np.random.rand():
            # accept as current
            cur_h = cand
            cur_score = cand_score
            # update best
            if cand_score < best_score:
                best_h = cand.copy()
                best_score = cand_score
        # Update schedules
        sigma *= decay_sigma
        T *= decay_T
        iter_count += 1
        # occasional random restart to avoid stagnation
        if iter_count % 5000 == 0:
            # if stagnation, restart
            cur_h = get_initial_h(N)
            cur_score = compute_c5(cur_h)
            # optionally keep sigma at moderate value
            sigma = 0.2
            T = 0.1
    # final correction for sum and bounds
    best_h = project_box_sum(best_h, 0.0, 1.0, N/2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, N

import numpy as np
import time

def project_box_sum(v, lower=0.0, upper=1.0, target_sum=None, max_iter=60):
    """Projection of v onto {lower <= x <= upper, sum(x)=target_sum}
    using the simple shift-and-clip method (binary search for lambda)."""
    n = v.shape[0]
    if target_sum is None:
        target_sum = n / 2.0
    # binary search for lambda
    lam_low = -np.max(v) - 1.0          # ensures all v+lam < lower
    lam_high = upper - np.min(v) + 1.0  # ensures all v+lam > upper
    for _ in range(max_iter):
        lam = (lam_low + lam_high) * 0.5
        h = np.clip(v + lam, lower, upper)
        s = h.sum()
        if s < target_sum:
            lam_low = lam
        else:
            lam_high = lam
    h = np.clip(v + (lam_low + lam_high) * 0.5, lower, upper)
    # fix any tiny residual
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        for i in range(n):
            if lower <= h[i] + diff <= upper:
                h[i] += diff
                break
        else:
            # fall back: distribute evenly and re‑clip
            h += diff / n
            h = np.clip(h, lower, upper)
    return h

def compute_c5(h):
    """Maximum overlap integral C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx

def compute_gradient(h, shift):
    """Gradient of C_k (the k giving the current maximum) w.r.t h."""
    n = h.shape[0]
    if shift >= 0:
        # term from h_i * (1 - h_{i+shift})
        term1 = np.empty(n)
        term1[: n - shift] = 1.0 - h[shift:]
        term1[n - shift:] = 1.0
        # term from h_{i-shift} * (1 - h_i)
        term2 = np.empty(n)
        term2[:shift] = 0.0
        term2[shift:] = -h[: n - shift]
    else:
        k = -shift
        term1 = np.empty(n)
        term1[:k] = 1.0
        term1[k:] = 1.0 - h[: n - k]
        term2 = np.empty(n)
        term2[: n - k] = -h[k:]
        term2[n - k:] = 0.0
    return term1 + term2

def get_initial_h(n_points):
    """Create an initial feasible vector, optionally up‑sampling a parent."""
    if 'initial_h_values' in globals():
        parent = np.asarray(initial_h_values, dtype=float)
        m = parent.shape[0]
        # simple linear interpolation onto the new grid
        x_old = np.linspace(0.0, 2.0, m, endpoint=False) + 1.0 / m
        x_new = np.linspace(0.0, 2.0, n_points, endpoint=False) + 1.0 / n_points
        h0 = np.interp(x_new, x_old, parent)
    else:
        h0 = np.random.rand(n_points)
    return project_box_sum(h0, 0.0, 1.0, n_points / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    """Optimise a step function h on [0,2] to minimise the Erdős C₅ constant."""
    np.random.seed(seed)
    start_time = time.time()

    # problem size (larger than the parent to give more freedom)
    N = 200
    if N % 2:
        N += 1

    # initialise current and best solutions
    cur_h = get_initial_h(N)
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score

    # simulated‑annealing / gradient‑descent parameters
    sigma = 0.2          # scale of random perturbations
    temp = 0.5           # initial temperature for SA
    decay_sigma = 0.9995
    decay_temp = 0.9998
    min_temp = 1e-6
    min_sigma = 5e-5
    iteration = 0
    target_bound = 0.3808

    while time.time() - start_time < budget_s:
        iteration += 1

        # decide move type
        if np.random.rand() < 0.65:
            # gradient move aimed at the worst shift
            corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
            max_idx = int(np.argmax(corr))
            shift = max_idx - (N - 1)
            grad = compute_gradient(cur_h, shift)
            cand = cur_h - sigma * grad
        else:
            # simple Gaussian perturbation
            cand = cur_h + sigma * np.random.randn(N)

        # enforce constraints
        cand = project_box_sum(cand, 0.0, 1.0, N / 2.0)

        cand_score = compute_c5(cand)
        delta = cand_score - cur_score

        # accept/reject (simulated annealing)
        if delta < 0 or np.exp(-delta / max(temp, 1e-12)) > np.random.rand():
            cur_h = cand
            cur_score = cand_score

        if cand_score < best_score:
            best_h = cand.copy()
            best_score = cand_score
            # stop early if we beat the target bound
            if best_score <= target_bound:
                break

        # schedule updates
        sigma = max(sigma * decay_sigma, min_sigma)
        temp = max(temp * decay_temp, min_temp)

        # occasional restart to escape possible stagnation
        if iteration % 8000 == 0:
            cur_h = get_initial_h(N)
            cur_score = compute_c5(cur_h)

    # final projection for safety and recompute exact bound
    best_h = project_box_sum(best_h, 0.0, 1.0, N / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, N
