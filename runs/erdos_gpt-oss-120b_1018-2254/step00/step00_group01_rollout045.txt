import time
import numpy as np

def _c5_from_h(h, dx):
    """Return max overlap integral for a given h (discrete version)."""
    # correlate h with 1-h, full mode gives all integer shifts
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)

    # ----- initialization -------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64).copy()
    else:
        # fallback: uniform 0.5 (automatically satisfies integral)
        n_points = 47
        h = np.full(n_points, 0.5, dtype=np.float64)

    n_points = int(h.shape[0])
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Ensure exact sum (tiny adjustments if needed)
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:

        h *= target_sum / cur_sum
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - h.sum()
        if abs(diff) > 1e-12:
            # add diff to a random index within capacity
            idx = rng.integers(n_points)
            cap_low = -h[idx]
            cap_up = 1.0 - h[idx]
            adj = np.clip(diff, cap_low, cap_up)
            h[idx] += adj

    # ----- bookkeeping ----------------------------------------------------
    best_h = h.copy()
