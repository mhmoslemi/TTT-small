import time
import numpy as np

# ---------- helper functions (top level) ----------

def _project_to_feasible(v, target_sum):
    """
    Euclidean projection of vector v onto the box [0,1]^n
    intersected with the hyperplane sum(v) = target_sum.
    """
    lo, hi = -1e6, 1e6
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def _corr_nonoverlap(h):
    """C₅ = max_k ∫ h(x)(1−h(x+k))dx  (discrete version)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max()), corr


def _gradient_for_lag(h, lag, dx):
    """
    Sub‑gradient of the correlation for a specific lag.
    Returns gradient of  g_lag(h) = Σ h[i]*(1−h[i+lag]).
    """
    n = h.shape[0]
    g = np.zeros_like(h)
    if lag >= 0:
        k = lag
        if k < n:
            g[: n - k] += (1.0 - h[k:]) * dx
            g[k:] -= h[: n - k] * dx
    else:
        k = -lag
        if k < n:
            g[: n - k] -= h[k:] * dx
            g[k:] += (1.0 - h[: n - k]) * dx
    return g


def _upsample(arr, new_n):
    """Linear interpolation from len(arr) to new_n points on [0,2)."""
    old_n = arr.shape[0]
    xs_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(xs_new, xs_old, arr)


# ---------- main entry point ----------

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] (discretized) to
    minimise  C₅ = max_k ∫ h(x)(1−h(x+k))dx .
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # start resolution
    if "initial_h_values" in globals():
        base_h = np.asarray(initial_h_values, dtype=float)
        n0 = base_h.size
    else:
        n0 = 70
        base_h = None

    # list of resolutions (doubling each step, limited by time)
    resolutions = [n0]
    while resolutions[-1] < 560:
        resolutions.append(resolutions[-1] * 2)

    best_h = None
    best_c5 = float("inf")

    for n in resolutions:
        # time guard – leave a safety margin
        if time.time() - start_time > budget_s * 0.94:
            break

        # initialise h for this resolution
        if n == n0 and base_h is not None:
            h = base_h.copy()
        elif n == n0:
            h = rng.random(n).astype(float)
        else:
            # upscale previously best solution
            if best_h is not None:
                h = _upsample(best_h, n)
            else:
                h = rng.random(n).astype(float)

        # enforce feasibility
        target = n / 2.0
        h = _project_to_feasible(h, target)

        # optimisation parameters
        max_iters = 2500
        lr = 0.7
        no_improve_cnt = 0

        for it in range(max_iters):
            # overall time check
            if time.time() - start_time > budget_s * 0.99:
                break

            cur_c5, corr_vec = _corr_nonoverlap(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                no_improve_cnt = 0
                if best_c5 <= 0.3808:               # target reached – exit early
                    break
            else:
                no_improve_cnt += 1

            # stagnation handling
            if no_improve_cnt >= 200:
                # perturb a little to escape flat region
                h = h + 0.02 * (rng.random(n) - 0.5)
                h = _project_to_feasible(h, target)
                no_improve_cnt = 0

            # lag giving the current maximum
            idx_max = int(np.argmax(corr_vec))
            lag = idx_max - (n - 1)   # shift index (can be negative)

            # gradient step for this lag
            grad = _gradient_for_lag(h, lag, 2.0 / n)
            y = h - lr * grad
            h = _project_to_feasible(y, target)

            # gradual decay of learning rate
            lr *= 0.9995

        # if target already achieved, stop upgrading resolution
        if best_c5 <= 0.3808:
            break

    if best_h is None:                     # fallback – should never happen
        n = n0
        best_h = np.full(n, 0.5, dtype=float)
        best_h = _project_to_feasible(best_h, n / 2.0)

    # final bound (use the provided evaluator if present)
    try:
        c5_bound = evaluate_erdos_solution(best_h)
    except Exception:
        c5_bound, _ = _corr_nonoverlap(best_h)
