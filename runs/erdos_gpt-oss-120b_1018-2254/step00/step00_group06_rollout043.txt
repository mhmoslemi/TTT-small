        lo = np.min(y) - 1.0

            if s > target:

            else:

                x = np.clip(x, 0.0, 1.0)

import time
import numpy as np
from math import inf

def _project_bounds_sum(y, target):
    """Project vector y onto the box [0,1]^n while enforcing sum(y)=target."""
    lo = np.min(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(y - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    err = target - x.sum()
    if abs(err) > 1e-12:
        free = (x > 0.0) & (x < 1.0)
        cnt = free.sum()
        if cnt:
            x[free] += err / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _c5_value(vec, dx):
    """Return the C5 value for a given vector."""
    return np.max(np.correlate(vec, 1 - vec, mode="full")) * dx


def run(seed=42, budget_s=1100, **kwargs):

    # Determine problem size

    dx = 2.0 / n

    # --- objective ---------------------------------------------------------
    def compute_c5(v):
        return _c5_value(v, dx)

    # --- start from available constructions ---------------------------------
    best_h = None
    best_c = inf

    if init_vec is not None:
        c0 = compute_c5(init_vec)
        best_h = init_vec.copy()
        best_c = c0

    # Uniform baseline (always feasible)
    uniform = np.full(n, 0.5)
    uniform = _project_bounds_sum(uniform, target_sum)
    c_uni = compute_c5(uniform)
    if c_uni < best_c:
        best_h = uniform.copy()
        best_c = c_uni

    start_time = time.time()

    # --- parametric families (power‑law V shapes) ---------------------------
    idx = np.arange(n)
    mid = (n - 1) / 2.0
    v = np.abs(idx - mid) / mid                     # distance ratio in [0,1]

    p_vals = np.concatenate((np.linspace(0.2, 1.0, 9), np.linspace(1.0, 3.0, 9)))
    for p in p_vals:
        vp = v ** p
        sum_vp = vp.sum()
        # feasibility from a>=0
        b_max1 = target_sum / sum_vp if sum_vp > 0 else 0.0
        # feasibility from a+b<=1
        avg_vp = sum_vp / n
        if 1.0 - avg_vp > 0:
            b_max2 = 0.5 / (1.0 - avg_vp)
            b_max = min(b_max1, b_max2)
        else:
            b_max = b_max1
        if b_max <= 0:
            continue
        nb = 60
        for b in np.linspace(0.0, b_max, nb)[1:]:   # skip b=0 (uniform)
            a = (target_sum - b * sum_vp) / n
            h = a + b * vp
            h = np.clip(h, 0.0, 1.0)
            h = _project_bounds_sum(h, target_sum)
            c = compute_c5(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        if time.time() - start_time > budget_s * 0.5:

    # --- simple hill‑climbing refinement -----------------------------------
    if best_h is None:
        best_h = uniform.copy()
        best_c = compute_c5(best_h)

    cur_c = best_c
    rng = np.random.default_rng(seed)

        if elapsed > budget_s * 0.9:

            step_scale = 0.05

        delta = (rng.random() - 0.5) * 2.0 * step_scale
        delta = np.clip(delta, delta_low, delta_high)
        if abs(delta) < 1e-12:
            continue

        new = cur.copy()
        new[i] += delta
        new[j] -= delta
        new_c = compute_c5(new)
        if new_c < cur_c:
            cur, cur_c = new, new_c
            if new_c < best_c:
                best_c, best_h = new_c, new.copy()

    # --- optional local optimiser (SLSQP) -----------------------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 5.0:

            bounds = [(0.0, 1.0)] * n
            cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},)
            maxit = max(10, int(200 * remaining / budget_s))
            res = minimize(compute_c5, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons,

                refined_c = compute_c5(refined)
                if refined_c < best_c:
                    best_c, best_h = refined_c, refined.copy()
        except Exception:
            pass

    return (best_h, float(best_c), n)
