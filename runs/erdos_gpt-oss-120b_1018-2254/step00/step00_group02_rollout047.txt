        start = random_feasible(n, target_sum, rng)
        # Optionally mix with best:

        # constraints and bounds:
        bounds = opt.Bounds(0, 1)
        constraints = {'type':'eq', 'fun': lambda x: np.sum(x) - target_sum}
        # run minimize
        res = opt.minimize(compute_c5, start, method='SLSQP', bounds=bounds,
                            constraints=constraints,

    v = v * scale
    # Clip and project to sum target using simple iterative method
    # This algorithm ensures values within [0,1] after projection.
    # We'll perform a loop of few iterations.
    for _ in range(10):
        # Clip to bounds
        np.clip(v, 0.0, 1.0, out=v)

            # cannot adjust further; break
            break
        # distribute diff equally among free entries
        v[free] += diff / free_count

    np.clip(v, 0.0, 1.0, out=v)

import time
import numpy as np
import scipy.optimize as opt

_TARGET_SUM = None  # will be set before each optimization


def compute_c5(h):
    """Return the overlap integral C5 for a step function represented by h."""
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return np.inf
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


def eq_sum(x):
    """Equality constraint: sum(x) == _TARGET_SUM."""
    return np.sum(x) - _TARGET_SUM


def random_feasible(n, target_sum, rng):
    """Generate a random vector in [0,1]^n with sum equal to target_sum."""
    # start from uniform random numbers
    v = rng.random(n)
    # scale to correct total mass
    v *= target_sum / v.sum()
    # iterative projection onto box and hyperplane
    for _ in range(10):
        np.clip(v, 0.0, 1.0, out=v)
        cur = v.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        free = (v > 0.0) & (v < 1.0)
        cnt = free.sum()
        if cnt == 0:
            break
        v[free] += diff / cnt
    # final safeguard for tiny residual
    cur = v.sum()
    diff = target_sum - cur
    if abs(diff) > 1e-9:
        for i in range(n):
            if diff > 0 and v[i] < 1.0:
                step = min(diff, 1.0 - v[i])
                v[i] += step
                diff -= step
            elif diff < 0 and v[i] > 0.0:
                step = min(-diff, v[i])
                v[i] -= step
                diff += step
            if abs(diff) < 1e-12:
                break
    np.clip(v, 0.0, 1.0, out=v)
    return v


def upsample_h(h, n_new):
    """Linear interpolation of h to a new grid with n_new points."""
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.arange(n_new) * (2.0 / n_new)
    return np.interp(x_new, x_old, h)

    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5  # small safety margin

    # Acquire initial construction if supplied
    if "initial_h_values" in globals():
        h_initial = np.asarray(initial_h_values, dtype=float).copy()
    else:
        # fall back to a uniform half‑density function
        n_default = 45
        h_initial = np.full(n_default, 0.5, dtype=float)

    n0 = h_initial.size
    target0 = n0 / 2.0
    # Ensure feasibility of the starting vector
    if abs(h_initial.sum() - target0) > 1e-12:
        h_initial = random_feasible(n0, target0, rng)

    best_h = h_initial
    best_val = compute_c5(best_h)

    # Resolution schedule (keep the current size and try higher resolutions)
    resolutions = sorted({n0, 90, 180})
    for n in resolutions:
        if time.time() >= deadline:
            break
        # Prepare starting point for this resolution
        if n == best_h.size:
            start_vec = best_h.copy()
        else:
            start_vec = upsample_h(best_h, n)

        # Allocate a slice of remaining time (capped at 30 s)
        remaining = deadline - time.time()
        sub_deadline = time.time() + min(remaining * 0.5, 30.0)
        if sub_deadline > deadline:
            sub_deadline = deadline

        # Run local optimisation
        opt_h, opt_val = local_optimize(start_vec, n / 2.0, sub_deadline, rng)
        if opt_val < best_val:
            best_h = opt_h
            best_val = opt_val

    # Final post‑processing: enforce bounds and exact mass
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_h.size / 2.0
    cur_sum = best_h.sum()
    diff = target - cur_sum
    if abs(diff) > 1e-10:
        free = (best_h > 0.0) & (best_h < 1.0)
        if free.any():
            best_h[free] += diff / free.sum()
            best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), float(final_c5), best_h.size)
