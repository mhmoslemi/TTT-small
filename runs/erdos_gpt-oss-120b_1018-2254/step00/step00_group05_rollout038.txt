import time, math
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    # Determine initial h
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
        h = h0.copy()
    else:
        n = 70
        h = np.full(n, 0.5, dtype=float)
    target_sum = n / 2.0
    # Normalize to satisfy integral constraint
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        h = h * (target_sum / cur_sum)
    # Clip to [0,1] and fix sum if clipping changes it
    h = np.clip(h, 0.0, 1.0)
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        # adjust by adding the small difference to a non‑border element
        diff = target_sum - cur_sum
        # find an index not at a bound (or any)
        idx = np.argmax(np.abs(h - 0.5))  # best candidate (maybe far from bounds)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    dx = 2.0 / n

    def compute_c5(arr):
        conv = np.correlate(arr, 1.0 - arr, mode='full')
        return np.max(conv) * dx

    best_h = h.copy()
    best_score = compute_c5(best_h)
    current_h = h.copy()
    current_score = best_score

    # Simulated‑annealing parameters
    init_temp = 0.02          # initial temperature
    init_step = 0.2           # initial maximum mass transfer
    min_step = 1e-4           # final step size
    # Main optimization loop
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= budget_s:
            break
        # Linear cooling & step schedule
        frac = elapsed / budget_s
        temp = init_temp * (1.0 - frac)
        step = max(min_step, init_step * (1.0 - frac))

        # Propose a transfer between two distinct indices
        i = np.random.randint(0, n)
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1  # ensure i != j

        # Determine feasible delta respecting bounds
        max_up = min(step, 1.0 - current_h[i], current_h[j])
        max_down = min(step, current_h[i], 1.0 - current_h[j])
        if max_up + max_down == 0.0:
            # nothing to move, skip this iteration
            continue
        delta = np.random.uniform(-max_down, max_up)

        # Apply the transfer
        new_h = current_h.copy()
        new_h[i] += delta
        new_h[j] -= delta

        # Compute new objective
        new_score = compute_c5(new_h)
        dE = new_score - current_score

        if dE < 0.0:
            accept = True
        elif temp > 0.0:
            # Metropolis criterion
            accept = math.exp(-dE / temp) > np.random.rand()
        else:
            accept = False

        if accept:
            current_h = new_h
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = new_h.copy()

    # Post‑process best_h to strictly satisfy constraints
    best_h = np.clip(best_h, 0.0, 1.0)
    total = best_h.sum()
    diff = target_sum - total
    if abs(diff) > 1e-12:
        # distribute the small difference to a non‑boundary entry
        mask = (best_h > 0.0) & (best_h < 1.0)
        indices = np.where(mask)[0]
        if indices.size == 0:
            # fallback: rescale all entries
            best_h = best_h * (target_sum / total)
            best_h = np.clip(best_h, 0.0, 1.0)
        else:
            # spread uniformly among free entries
            add = diff / indices.size
            best_h[indices] += add
            best_h = np.clip(best_h, 0.0, 1.0)
            # final minor fix if needed
            total = best_h.sum()
            if abs(target_sum - total) > 1e-9:
                best_h = best_h * (target_sum / total)
                best_h = np.clip(best_h, 0.0, 1.0)

    final_score = compute_c5(best_h)
    return best_h, final_score, n

def run(seed=42, budget_s=1100, **kwargs):
