import time
import numpy as np

def _c5_of(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    return dx * np.max(np.correlate(h, 1.0 - h, mode="full"))

def _project_to_bounds(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the set
        {x : 0 <= x_i <= 1, sum(x) = target }.
    Uses a bisection on the Lagrange multiplier.
    """
    low, high = -1.0, 1.0          # guarantees sum(x)=n at low, 0 at high
    for _ in range(50):
        lam = (low + high) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            low = lam
        else:
            high = lam
    x = np.clip(v - (low + high) / 2.0, 0.0, 1.0)
    # Tiny correction if rounding left a small error
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 0.0) & (x < 1.0)
        cnt = free.sum()
        if cnt > 0:
            x[free] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _tournament_select(pop, fits, rng, size=3):
    """Return a single individual chosen by tournament selection."""
    idx = rng.integers(len(pop), size=size)
    best = idx[0]
    best_fit = fits[best]
    for i in idx[1:]:
        if fits[i] < best_fit:
            best = i
            best_fit = fits[i]
    return pop[best]

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] with improved C₅ bound."""
    start = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------
    # Determine problem size and optional starter vector
    # ------------------------------------------------------------

        init_vec = np.array(initial_h_values, dtype=float)
        n = init_vec.shape[0]
    except NameError:
        n = 95               # fallback size when no parent is given
        init_vec = None

    target_sum = n / 2.0     # ∑ h_i = n/2  (since dx = 2/n and ∫h = 1)

    # ------------------------------------------------------------
    # Build initial population
    # ------------------------------------------------------------
    pop_size = 30
    population = []
    if init_vec is not None:
        population.append(_project_to_bounds(init_vec, target_sum))
    while len(population) < pop_size:
        rnd = rng.random(n)
        population.append(_project_to_bounds(rnd, target_sum))

    # ------------------------------------------------------------
    # Initial fitness evaluation
    # ------------------------------------------------------------
    fitness = np.array([_c5_of(ind) for ind in population])
    best_idx = int(np.argmin(fitness))
    best_h = population[best_idx].copy()
    best_c5 = float(fitness[best_idx])

    # ------------------------------------------------------------
    # Genetic algorithm parameters
    # ------------------------------------------------------------
    crossover_prob = 0.7
    mutation_rate = 0.05
    sigma_start = 0.25
    sigma_end = 0.005
    generation = 0

    # ------------------------------------------------------------
    # Main optimisation loop (runs until budget exhausted)
    # ------------------------------------------------------------
    while time.time() - start < budget_s:
        generation += 1
        # linear schedule for mutation strength
        prog = min(generation / 2000.0, 1.0)
        sigma = sigma_start * (1.0 - prog) + sigma_end * prog

        # elitism: keep the best individual
        new_pop = [best_h.copy()]

        while len(new_pop) < pop_size:
            p1 = _tournament_select(population, fitness, rng, size=3)
            p2 = _tournament_select(population, fitness, rng, size=3)

            # crossover
            if rng.random() < crossover_prob:
                mask = rng.random(n) < 0.5
                child = np.where(mask, p1, p2)
            else:
                child = p1.copy()

            # mutation
            mut_mask = rng.random(n) < mutation_rate
            if mut_mask.any():
                child[mut_mask] += rng.normal(0.0, sigma, size=mut_mask.sum())

            # repair to feasibility
            child = _project_to_bounds(child, target_sum)
            new_pop.append(child)

        population = new_pop

    # Local refinement (small random perturbations)

        refine_iters -= 1

    # Final exact evaluation
    best_c5 = _c5_of(best_h)

    return (best_h.tolist(), float(best_c5), int(n))
