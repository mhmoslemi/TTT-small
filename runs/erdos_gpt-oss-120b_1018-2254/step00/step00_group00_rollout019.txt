def gradient_of_shift(h, k):
    n = h.shape[0]
    ones = np.ones(n)
    if k >= 0:
        term1 = np.concatenate([h[k:], np.zeros(k, dtype=h.dtype)])
        term2 = np.concatenate([np.zeros(k, dtype=h.dtype), h[:n-k]])
    else:

        term2 = np.concatenate([h[kk:], np.zeros(kk, dtype=h.dtype)])

def random_move(h):
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h, False
    # let delta = min(h[i], 1 - h[j])
    max_move = min(h[i], 1 - h[j])
    if max_move <= 0:
        return h, False
    eps = rng.random() * max_move
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h, True

import numpy as np
import time

def compute_c5(h: np.ndarray) -> float:
    """Return the scaled maximal overlap C5 for a step function h."""
    n = h.shape[0]
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return float(max_corr * (2.0 / n))


def compute_gradient(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the worst‑shift correlation term f_k(h)=∑h_i(1-h_{i+k})."""
    n = h.shape[0]
    if shift >= 0:
        term1 = np.concatenate([h[shift:], np.zeros(shift, dtype=h.dtype)])
        term2 = np.concatenate([np.zeros(shift, dtype=h.dtype), h[: n - shift]])
    else:
        k = -shift
        term1 = np.concatenate([np.zeros(k, dtype=h.dtype), h[: n - k]])
        term2 = np.concatenate([h[k:], np.zeros(k, dtype=h.dtype)])
    return np.ones(n, dtype=h.dtype) - term1 - term2


def project_to_feasible(v: np.ndarray, target_sum: float, iters: int = 60) -> np.ndarray:
    """
    Project v onto the set {x∈[0,1]^n : sum(x)=target_sum}.
    Uses a bisection on the Lagrange multiplier λ.
    """
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(iters):
        lam = (lo + hi) / 2.0
        u = np.clip(v - lam, 0.0, 1.0)
        s = u.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) / 2.0
    u = np.clip(v - lam, 0.0, 1.0)

    # tiny correction if floating‑point sum is off
    diff = target_sum - u.sum()
    if abs(diff) > 1e-12:
        free = (u > 1e-12) & (u < 1 - 1e-12)
        cnt = free.sum()
        if cnt:
            u[free] += diff / cnt
            u = np.clip(u, 0.0, 1.0)
        else:
            u[0] = np.clip(u[0] + diff, 0.0, 1.0)
    return u

    # ------------------------------------------------------------------
    # Main optimisation loop (gradient steps + occasional random transfers)
    # ------------------------------------------------------------------
    start = time.time()
    end_time = start + budget_s * 0.95   # reserve a thin margin
    it = 0
    alpha0 = 0.5

    while time.time() < end_time:
        # worst shift
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        shift = int(np.argmax(corr)) - (n - 1)

        # gradient of that shift
        grad = compute_gradient(best_h, shift)

        # step size diminishing with √iteration
        step = alpha0 / np.sqrt(it + 1)
        v = best_h - step * grad
        new_h = project_to_feasible(v, target_sum)

        new_c5 = compute_c5(new_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h

        # occasional random transfer move (exploration)
        if it % 20 == 0:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                move_max = min(best_h[i], 1.0 - best_h[j])
                if move_max > 1e-9:
                    eps = rng.random() * move_max
                    h_move = best_h.copy()
                    h_move[i] -= eps
                    h_move[j] += eps
                    c5_move = compute_c5(h_move)
                    if c5_move < best_c5:
                        best_c5 = c5_move
                        best_h = h_move

        it += 1

    # ------------------------------------------------------------------
    # Final clean‑up projection and bound computation
    # ------------------------------------------------------------------
    best_h = project_to_feasible(best_h, target_sum)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
