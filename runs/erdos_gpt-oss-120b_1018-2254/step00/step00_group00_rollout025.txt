def random_mutation(h, step=0.2):
    h_new = h.copy()
    n = len(h)
    for _ in range(3):
        i, j = np.random.choice(n, size=2, replace=False)
        delta = (np.random.rand() - 0.5) * step  # between -step/2 and step/2
        # apply delta to i, -delta to j
        h_new[i] += delta
        h_new[j] -= delta
        # clip both - but clip after all adjustments to ensure sum maybe off.
    h_new = np.clip(h_new, 0, 1)
    # Now repair sum:
    h_new = repair_h(h_new, target_sum = len(h)/2.0)
    return h_new

def compute_gradient(h):
    n = len(h)
    # compute correlation to identify max shift
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n - 1)   # shift = positive => h[i] * (1 - h[i+shift]) product
    # gradient: dx * (1 - (h[i+shift] if exists) - (h[i-shift] if exists))
    dx = 2.0 / n
    grad = np.empty(n, dtype=float)
    for i in range(n):
        term = 1.0
        i_plus = i + shift
        if 0 <= i_plus < n:
            term -= h[i_plus]
        i_minus = i - shift
        if 0 <= i_minus < n:
            term -= h[i_minus]
        grad[i] = dx * term
    return grad

    tol = 1e-10

import numpy as np
import time

def _evaluate_overlap(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def _repair(h, target_sum):
    # clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    tol = 1e-12
    max_iter = 500
    for _ in range(max_iter):
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h < 1.0)[0]
            if free.size == 0:
                break
            inc = diff / free.size
            possible = 1.0 - h[free]
            delta = np.minimum(inc, possible)
            h[free] += delta
        else:
            free = np.where(h > 0.0)[0]
            if free.size == 0:
                break
            dec = (-diff) / free.size
            possible = h[free]
            delta = np.minimum(dec, possible)
            h[free] -= delta
        diff = target_sum - np.sum(h)
    # final fine correction (if still off)
    diff = target_sum - np.sum(h)
    if abs(diff) > tol:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
            if idx.size > 0:
                i = idx[0]
                h[i] = min(1.0, h[i] + diff)
        else:
            idx = np.where(h > 0.0)[0]
            if idx.size > 0:
                i = idx[0]
                h[i] = max(0.0, h[i] + diff)
    return h

def _gradient(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n - 1)
    dx = 2.0 / n
    grad = np.empty_like(h)
    for i in range(n):
        term = 1.0
        ip = i + shift
        if 0 <= ip < n:
            term -= h[ip]
        im = i - shift
        if 0 <= im < n:
            term -= h[im]
        grad[i] = dx * term
    return grad

def _mutate(h, step):
    n = h.shape[0]
    h_new = h.copy()
    # apply three pairwise adjustments
    for _ in range(3):
        i, j = np.random.choice(n, size=2, replace=False)
        delta = (np.random.rand() - 0.5) * step * 2.0  # uniform in [-step, step]
        h_new[i] += delta
        h_new[j] -= delta
    h_new = np.clip(h_new, 0.0, 1.0)
    target = n / 2.0
    h_new = _repair(h_new, target)
    return h_new

def _local_optimize(h):
    try:
        from scipy.optimize import minimize
    except Exception:
        return h
    n = h.shape[0]
    target = n / 2.0
    bounds = [(0.0, 1.0) for _ in range(n)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    # SLSQP may need Jacob? let it approximate.
    res = minimize(_evaluate_overlap, h, method='SLSQP', bounds=bounds,
                   constraints=constraints, options={'maxiter': 200, 'ftol': 1e-9})
    if res.success:
        return res.x
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    # Initialization
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 45
        h0 = np.random.rand(n)
    target_sum = n / 2.0
    h0 = _repair(np.clip(h0, 0.0, 1.0), target_sum)
    best_h = h0.copy()
    best_c5 = _evaluate_overlap(best_h)

    step = 0.2
    grad_step = 0.02
    no_improve = 0
    max_no_improve = 3000
    iter_cnt = 0

    # main stochastic search
    while True:
        now = time.time()
        if now - start > budget_s * 0.92:
            break
        cand = _mutate(best_h, step)
        cand_c5 = _evaluate_overlap(cand)
        iter_cnt += 1
        if cand_c5 + 1e-12 < best_c5:
            best_h = cand
            best_c5 = cand_c5
            no_improve = 0
            step *= 0.97
        else:
            no_improve += 1
        # periodic gradient refinement
        if iter_cnt % 80 == 0:
            grad_h = best_h - grad_step * _gradient(best_h)
            grad_h = _repair(np.clip(grad_h, 0.0, 1.0), target_sum)
            grad_c5 = _evaluate_overlap(grad_h)
            if grad_c5 + 1e-12 < best_c5:
                best_h = grad_h
                best_c5 = grad_c5
        # random restart if stagnating
        if no_improve > max_no_improve:
            rand_h = np.random.rand(n)
            rand_h = _repair(rand_h, target_sum)
            rand_c5 = _evaluate_overlap(rand_h)
            if rand_c5 < best_c5:
                best_h = rand_h
                best_c5 = rand_c5
                step = 0.2
                grad_step = 0.02
                no_improve = 0
            else:
                no_improve = 0

    # final local optimization under remaining budget
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0:
        best_h = _local_optimize(best_h)
        best_c5 = _evaluate_overlap(best_h)

    # final repair for safety
    best_h = _repair(np.clip(best_h, 0.0, 1.0), target_sum)
    best_c5 = _evaluate_overlap(best_h)
    return (best_h, float(best_c5), n)

import numpy as np
import time

def _evaluate_overlap(h):
    """Return C5 for a discrete step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def _repair(h, target_sum):
    """Project h onto [0,1]^n while forcing sum(h)=target_sum."""
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    tol = 1e-12
    for _ in range(500):
        if abs(diff) <= tol:
            break
        if diff > 0:          # need to increase some components
            free = np.where(h < 1.0)[0]
            if free.size == 0:
                break
            inc = diff / free.size
            possible = 1.0 - h[free]
            delta = np.minimum(inc, possible)
            h[free] += delta
        else:                  # need to decrease some components
            free = np.where(h > 0.0)[0]
            if free.size == 0:
                break
            dec = (-diff) / free.size
            possible = h[free]
            delta = np.minimum(dec, possible)
            h[free] -= delta
        diff = target_sum - np.sum(h)

    # final tiny correction if needed
    diff = target_sum - np.sum(h)
    if abs(diff) > tol:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
            if idx.size:
                i = idx[0]
                h[i] = min(1.0, h[i] + diff)
        else:
            idx = np.where(h > 0.0)[0]
            if idx.size:
                i = idx[0]
                h[i] = max(0.0, h[i] + diff)
    return h


def _gradient(h):
    """Gradient of the objective for the shift attaining the maximum overlap."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    shift = idx - (n - 1)                # k such that corr[idx] = sum_i h[i]*(1-h[i+shift])
    dx = 2.0 / n
    grad = np.empty_like(h)
    for i in range(n):
        term = 1.0
        ip = i + shift
        if 0 <= ip < n:
            term -= h[ip]
        im = i - shift
        if 0 <= im < n:
            term -= h[im]
        grad[i] = dx * term
    return grad


def _mutate(h, step):
    """Random pairwise perturbation that keeps the sum unchanged (up to repair)."""
    n = h.shape[0]
    h_new = h.copy()
    for _ in range(3):
        i, j = np.random.choice(n, size=2, replace=False)
        delta = (np.random.rand() - 0.5) * 2.0 * step   # uniform in [-step, step]
        h_new[i] += delta
        h_new[j] -= delta
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = _repair(h_new, n / 2.0)
    return h_new


def _local_optimize(h):
    """Refine a solution with SLSQP (if SciPy is available)."""
    try:
        from scipy.optimize import minimize
    except Exception:
        return h
    n = h.shape[0]
    target = n / 2.0
    bounds = [(0.0, 1.0) for _ in range(n)]
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    res = minimize(_evaluate_overlap, h, method="SLSQP", bounds=bounds,
                   constraints=cons, options={"maxiter": 200, "ftol": 1e-9})
    if res.success:
        return np.clip(res.x, 0.0, 1.0)
    return h


def _upscale_vector(v, new_n):
    """Linearly interpolate a smaller vector v to length new_n."""
    old_n = v.shape[0]
    if old_n == new_n:
        return v.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, v)
