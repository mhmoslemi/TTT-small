def evaluate(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_onto_bounds_sum(v, target):
    # binary search lambda for projection onto [0,1] with sum constraint
    lo = -max(v) - 1.0
    hi = 1 - min(v) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo+hi)/2.0, 0.0, 1.0)

def _evaluate(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_sum(v, target):
    # binary search for lambda such that sum(clip(v+λ,0,1)) = target
    lo = -np.max(v) - 5.0
    hi = 1.0 - np.min(v) + 5.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        mask = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        if mask.any():
            idx = np.where(mask)[0][0]
            proj[idx] = np.clip(proj[idx] + diff, 0.0, 1.0)
        else:
            proj[0] = np.clip(proj[0] + diff, 0.0, 1.0)
    return proj

    # Simulated annealing parameters
    T0 = 0.02
    target_sum = n / 2.0
    max_no_imp = 0

    while time.time() - start < budget_s * 0.95:
        # Propose a move that preserves the sum exactly
        i, j = np.random.choice(n, 2, replace=False)
        hi, hj = cur_h[i], cur_h[j]
        low = max(-hj, hi - 1.0)
        high = min(1.0 - hj, hi)
        if low >= high:
            continue
        delta = np.random.uniform(low, high)
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_val = _evaluate(new_h)

        # Acceptance
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / max(T0, 1e-12)):
            cur_h, cur_val = new_h, new_val
            if cur_val < best_val:
                best_h, best_val = cur_h.copy(), cur_val
                max_no_imp = 0
            else:
                max_no_imp += 1
        else:
            max_no_imp += 1

        # Temperature schedule (linear decay)
        elapsed = time.time() - start
        T0 = 0.02 * (1.0 - elapsed / budget_s)

        # Early stop if target achieved
        if best_val <= 0.3808:
            break

        # Occasionally re‑randomise a little if stuck
        if max_no_imp >= 2000:
            # Small perturbation of current best
            cur_h = best_h.copy()
            for _ in range(3):
                a, b = np.random.choice(n, 2, replace=False)
                ha, hb = cur_h[a], cur_h[b]
                lo = max(-hb, ha - 1.0)
                hi = min(1.0 - hb, ha)
                if lo < hi:
                    d = np.random.uniform(lo, hi) * 0.5
                    cur_h[a] -= d
                    cur_h[b] += d
            cur_val = _evaluate(cur_h)
            max_no_imp = 0

    # Final evaluation using the provided helper if available
    if 'evaluate_erdos_solution' in globals():
        final_c5 = evaluate_erdos_solution(best_h)
    else:
        final_c5 = _evaluate(best_h)

    return (best_h, float(final_c5), n)
