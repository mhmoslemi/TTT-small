def project_onto_feasible(h, target_sum):
    # clip to [0,1]
    x = np.clip(h, 0.0, 1.0)
    # iterative adjustment
    eps = 1e-12
    max_iter = 100
    for _ in range(max_iter):
        s = x.sum()
        diff = target_sum - s
        if abs(diff) <= eps:
            break
        # indices not at bound:
        free = (x > 0.0 + eps) & (x < 1.0 - eps)
        n_free = np.count_nonzero(free)
        if n_free == 0:
            # cannot adjust, break
            # distribute diff proportionally? but just break
            break
        delta = diff / n_free
        # add delta to free entries
        x[free] += delta
        # clip again
        np.clip(x, 0.0, 1.0, out=x)
    # after loop ensure sum close to target (tiny diff)
    return x

def smooth_max_and_grad(h, alpha):
    n = h.shape[0]
    C = np.sum(h)
    autocorr = np.correlate(h, h, mode='full')
    c = C - autocorr
    exp_c = np.exp(alpha * c)
    sum_exp = np.exp(alpha * c).sum()
    smooth = np.log(sum_exp) / alpha
    w = exp_c / sum_exp
    # gradient code using w
    g = np.zeros_like(h)
    # loop over shift indices
    for idx, weight in enumerate(w):
        s = idx - (n - 1)
        if s >= 0:
            end = n - s
            if end > 0:
                g[:end] -= weight * h[s:]
                g[s:] -= weight * h[:end]
        else:
            s_abs = -s
            end = n - s_abs
            if end > 0:
                g[s_abs:] -= weight * h[:end]
                g[:end] -= weight * h[s_abs:]
    # compute c5 from c: max(c) * (2.0/n)
    dx = 2.0 / n
    c5 = np.max(c) * dx
    return smooth, g, c5

def simulated_annealing_binary(h_init, time_budget):
    # h_init must be binary with sum = n/2
    n = h_init.shape[0]
    target_sum = n/2
    h = h_init.copy()

        c = np.correlate(v, 1 - v, mode='full')
        return np.max(c) * (2.0 / n)
    current_c5 = compute_c5_vec(h)
    best_h = h.copy()
    best_c5 = current_c5
    start = time.time()
    # schedule parameters
    T0 = 0.5
    T_end = 0.001
    # set approx number of steps e.g., 200000 or based on time
    # We'll loop until time budget
    iteration = 0

    # The move: select random i where h[i]==1, random j where h[j]==0, swap.
    while time.time() - start < time_budget:
        iteration += 1
        # pick i,j
        # find random i with h=1
        # generate random index for i until h[i]==1
        i = np.random.randint(n)
        while h[i] == 0:
            i = np.random.randint(n)
        j = np.random.randint(n)
        while h[j] == 1:
            j = np.random.randint(n)
        # propose swap
        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        new_c5 = compute_c5_vec(h_new)
        delta = new_c5 - current_c5
        # temperature schedule: linear or exponential based on iteration vs max_iters
        # We can approximate total steps as iteration count max, but we don't know final iteration count.
        # Let's compute temperature factor decaying as (time elapsed)/time_budget.
        elapsed = time.time() - start
        frac = elapsed / time_budget

                best_h = h.copy()

        if best_c5 <= 0.3808:

def project_onto_constraints(x, target):
    # Clip to [0,1]
    x = np.clip(x, 0.0, 1.0)
    # iterative water-filling
    for _ in range(10):
        s = x.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # indices not at bounds
        free = (x > 0) & (x < 1)
        n_free = int(free.sum())
        if n_free == 0:
            # distribute diff uniformly then clip
            x += diff / len(x)
            x = np.clip(x, 0.0, 1.0)
            break
        delta = diff / n_free
        x[free] += delta
        # clip again
        np.clip(x, 0.0, 1.0, out=x)
    # final clip
    np.clip(x, 0.0, 1.0, out=x)
    return x

import numpy as np
import time

def _project_onto_constraints(x, target):
    """Project x onto the feasible set 0<=x<=1 and sum(x)=target."""
    # Clip to box
    x = np.clip(x, 0.0, 1.0)
    # Water-filling style adjustment
    for _ in range(10):
        s = x.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        free = (x > 0.0) & (x < 1.0)
        n_free = int(free.sum())
        if n_free == 0:
            # Distribute uniformly and re‑clip
            x += diff / len(x)
            x = np.clip(x, 0.0, 1.0)
            break
        delta = diff / n_free
        x[free] += delta
        np.clip(x, 0.0, 1.0, out=x)
    np.clip(x, 0.0, 1.0, out=x)
    return x

def _smooth_max_and_grad(h, alpha):
    """Return smooth approximation of max(C‑autocorr), its gradient, and C5."""
    n = h.shape[0]
    C = np.sum(h)
    autocorr = np.correlate(h, h, mode='full')
    c = C - autocorr                     # c_k = C - autocorr_k
    M = np.max(c)                        # exact max (used for C5)
    # stable softmax
    exp_shifted = np.exp(alpha * (c - M))
    sum_exp = np.sum(exp_shifted)
    smooth = M + np.log(sum_exp) / alpha
    w = exp_shifted / sum_exp            # normalized weights (sum to 1)

    # gradient of smooth max w.r.t. h
    g = np.zeros_like(h)
    for idx, weight in enumerate(w):
        s = idx - (n - 1)                # shift value
        if s >= 0:
            end = n - s

            if end > 0:
                g[s_abs:] -= weight * h[:end]
                g[:end]   -= weight * h[s_abs:]
    dx = 2.0 / n
    c5 = M * dx
    return smooth, g, c5
