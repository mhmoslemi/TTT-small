import numpy as np, time, math

def _compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1‑h(x+k))dx using the discretisation rule."""
    n = h.shape[0]
    dx = 2.0 / n
    # full correlation of h with (1‑h); the overlapping part is handled automatically
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def _project_to_feasible(h: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Clip to [0,1] and adjust to obtain the exact target sum
    (sum(h) == target_sum). The adjustment is done by moving mass
    between entries that are not at a bound.
    """
    np.clip(h, 0.0, 1.0, out=h)
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return h

    # Simple water‑filling style correction
    if diff > 0:                     # need to add mass
        while diff > 1e-12:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = idx[0]
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:                            # need to remove mass
        diff = -diff
        while diff > 1e-12:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = idx[0]
            sub = min(diff, h[i])
            h[i] -= sub
            diff -= sub

    # Tiny leftover correction (should be negligible)
    residual = target_sum - h.sum()
    if abs(residual) > 1e-12:
        h[0] = np.clip(h[0] + residual, 0.0, 1.0)
    return h

def _pair_move(h: np.ndarray, shift: int, max_step: float) -> np.ndarray:
    """
    Produce a candidate vector by moving a small amount of mass between a
    pair of overlapping indices (i, i+shift) so that they become more similar.
    """
    n = h.shape[0]
    if shift >= 0:
        i_min, i_max = 0, n - shift - 1
    else:
        i_min, i_max = -shift, n - 1
    if i_max < i_min:
        return h.copy()
    i = np.random.randint(i_min, i_max + 1)
    j = i + shift
    diff = h[i] - h[j]                     # >0 means i larger than j
    # amount we would like to transfer to make them equal
    wanted = 0.5 * abs(diff)
    # respect bounds
    if diff > 0:
        allowed = min(h[i], 1.0 - h[j])
    else:
        allowed = min(h[j], 1.0 - h[i])
    delta = min(wanted, allowed) * (0.5 + 0.5 * np.random.rand())
    delta = min(delta, max_step)
    if delta <= 1e-12:
        return h.copy()
    cand = h.copy()
    if diff > 0:
        cand[i] -= delta
        cand[j] += delta
    else:
        cand[i] += delta
        cand[j] -= delta
    return cand

    n = h0.shape[0]
    target_sum = n / 2.0                     # because ∫ h = 1 and dx = 2/n
    h0 = _project_to_feasible(h0, target_sum)

    best_h = h0.copy()
    best_c5 = _compute_c5(best_h)

    cur_h = best_h.copy()
    cur_c5 = best_c5

    # ------------------------------------------------------------------
    # Simulated‑annealing / hill‑climbing loop
    # ------------------------------------------------------------------
    start = time.time()
    max_step = 0.25                         # maximal transferred mass per move
    T0 = max(0.02, best_c5 * 0.06)          # initial temperature
    last_improve = 0
    ITER_CHECK = 2000                      # how often we test for early stop

    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s - 0.5:
            break

        # worst shift for the current vector
        n = cur_h.shape[0]
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        idx_max = int(np.argmax(corr))
        shift = idx_max - (n - 1)                     # integer shift –(n‑1)…(n‑1)

        # propose a small pair‑wise move that equalises the two entries
        cand = _pair_move(cur_h, shift, max_step)
        cand = _project_to_feasible(cand, target_sum)

        cand_c5 = _compute_c5(cand)

        # accept if better, otherwise metropolised acceptance
        if cand_c5 < cur_c5:
            cur_h, cur_c5 = cand, cand_c5
            if cand_c5 < best_c5:
                best_h, best_c5 = cand.copy(), cand_c5
                last_improve = 0
        else:
            T = T0 * max(0.0, 1.0 - elapsed / budget_s)   # linear cooling
            if T > 0.0 and np.random.rand() < math.exp(-(cand_c5 - cur_c5) / T):
                cur_h, cur_c5 = cand, cand_c5

        # occasional stochastic restart to escape deep plateaus
        last_improve += 1
        if last_improve > 300000:
            # perturb the best solution a little and continue from there
            cur_h = best_h + 0.02 * (np.random.rand(n) - 0.5)
            cur_h = _project_to_feasible(cur_h, target_sum)
            cur_c5 = _compute_c5(cur_h)
            last_improve = 0

        # optional early termination if we already beat the target
        if best_c5 <= 0.3808:
            break

        # modest housekeeping
        if last_improve % ITER_CHECK == 0 and elapsed > budget_s * 0.9:
            # as time runs out we shrink the step size to fine‑tune
            max_step = max(0.02, max_step * 0.99)

    # final projection to be perfectly feasible
    best_h = _project_to_feasible(best_h, target_sum)
    best_c5 = _compute_c5(best_h)

    return (best_h, best_c5, n)
