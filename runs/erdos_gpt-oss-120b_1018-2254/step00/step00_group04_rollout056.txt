def project_bounds_sum(v, target):
    # v: any real vector
    # returns projection onto {0 <= x <= 1, sum x = target}
    # Start by clipping
    x = np.clip(v, 0., 1.)
    # iterative waterfilling
    for _ in range(10):
        s = x.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # Identify free indices where adjusting possible
        if diff > 0:
            mask = x < 1 - 1e-12
        else:
            mask = x > 1e-12
        if not np.any(mask):
            break
        # distribute diff equally
        delta = diff / np.sum(mask)
        x[mask] += delta
        # clip again
        x = np.clip(x, 0., 1.)
    # After loop, maybe small residual outcome possible; we can adjust a single free coordinate
    residual = target - x.sum()
    if abs(residual) > 1e-9:
        # adjust a coordinate not at bound if possible
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if np.any(mask):
            idx = np.where(mask)[0][0]
            x[idx] = np.clip(x[idx] + residual, 0., 1.)
        else:
            # fallback scale
            x = x * (target / x.sum())
    return x

    total = x.sum()
    diff = target - total

        while diff > 1e-12:
            free = x < 1 - 1e-12

import numpy as np
import time

def _repair_h(vec, target):
    """Project vec onto [0,1] with sum == target (capped simplex)."""
    x = np.clip(vec, 0.0, 1.0)
    diff = target - x.sum()
    # Increase if sum too low
    if diff > 1e-12:
        while diff > 1e-12:
            free = x < (1.0 - 1e-12)
            if not np.any(free):
                break
            cap = np.sum(1.0 - x[free])
            if cap <= diff + 1e-12:
                x[free] = 1.0
                diff -= cap
                continue
            inc = diff / np.count_nonzero(free)
            x[free] += inc
            break
    # Decrease if sum too high
    elif diff < -1e-12:
        while diff < -1e-12:
            free = x > 1e-12
            if not np.any(free):
                break
            cap = np.sum(x[free])
            if cap <= -diff + 1e-12:
                x[free] = 0.0
                diff += cap
                continue
            dec = diff / np.count_nonzero(free)   # diff negative
            x[free] += dec
            break
    # Tiny residual fix
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        idx = np.random.randint(len(x))
        x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
    return x

def _c5_value(h):
    """C5 for a discretised step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def _perturb(h, step):
    """Move a small amount of mass between two coordinates."""
    n = h.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    if np.random.rand() < 0.5:
        max_delta = min(step, h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            return h
        delta = np.random.uniform(0.0, max_delta)
        h2 = h.copy()
        h2[i] -= delta
        h2[j] += delta
    else:
        max_delta = min(step, h[j], 1.0 - h[i])
        if max_delta <= 0.0:
            return h
        delta = np.random.uniform(0.0, max_delta)
        h2 = h.copy()
        h2[j] -= delta
        h2[i] += delta
    return h2
