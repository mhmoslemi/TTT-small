def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, random
    rng = np.random.default_rng(seed)
    start = time.time()
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
        # ensure binary: threshold at 0.5
        h = (h0 > 0.5).astype(float)
    else:
        n = 70
        h = np.zeros(n)
        # randomly assign half ones
        indices = rng.choice(n, size=n//2, replace=False)
        h[indices] = 1.0
    # adjust if not exactly half ones
    ones = int(np.sum(h))
    target = n//2
    if ones != target:
        # adjust
        idx_one = np.where(h == 1)[0]
        idx_zero = np.where(h == 0)[0]
        if ones > target:
            # turn some ones to zero
            to_flip = rng.choice(idx_one, size=ones-target, replace=False)
            h[to_flip] = 0.0
        else:
            to_flip = rng.choice(idx_zero, size=target-ones, replace=False)
            h[to_flip] = 1.0
    # compute initial C5
    best_h = h.copy()
    best_c = evaluate_erdos_solution(best_h) if 'evaluate_erdos_solution' in globals() else evaluate_erdos(best_h)
    current_h = best_h.copy()
    current_c = best_c
    # neighbor generation
    n_iters = 0
    while time.time() - start < budget_s * 0.95:
        # propose swap
        idx_one = np.where(current_h == 1)[0]
        idx_zero = np.where(current_h == 0)[0]
        i = rng.choice(idx_one)
        j = rng.choice(idx_zero)
        # swap
        new_h = current_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        # evaluate
        new_c = evaluate_erdos_solution(new_h) if 'evaluate_erdos_solution' in globals() else evaluate_erdos(new_h)
        if new_c < current_c:
            current_h = new_h
            current_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()
        else:
            # optional probabilistic acceptance (annealing)
            # compute temperature schedule:
            t = max(0.01, (budget_s - (time.time() - start)) / budget_s) # linear decaying
            # accept with probability exp(-(new_c - current_c)/t)
            if rng.random() < np.exp(-(new_c - current_c) / t):
                current_h = new_h
                current_c = new_c
        n_iters += 1
    # return
    return best_h, float(best_c), n

import time, math
import numpy as np

def _c5_value(h):
    """Compute C5 for a given vector h."""
    # Use the pre‑imported evaluator if available, otherwise fall back.
    ev = globals().get("evaluate_erdos_solution", None)
    if ev is not None:
        return float(ev(h))
    # Manual computation: correlation of h with (1-h)
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _repair_binary(h):
    """Force a binary 0/1 vector to have exactly n/2 ones."""
    n = len(h)
    target = n // 2
    # round to nearest binary
    h_bin = (h > 0.5).astype(float)
    cur = int(h_bin.sum())
    if cur == target:
        return h_bin
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    rng = np.random.default_rng()
    if cur > target:
        # too many ones – turn some to zero
        to_zero = rng.choice(ones, size=cur - target, replace=False)
        h_bin[to_zero] = 0.0
    else:
        # too few ones – turn some zeros to one
        to_one = rng.choice(zeros, size=target - cur, replace=False)
        h_bin[to_one] = 1.0
    return h_bin

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) within the given time budget."""
    rng = np.random.default_rng(seed)
    start = time.time()
    target_c5 = 0.3808

    # ----------------------------------------------------------------------

    if "initial_h_values" in globals():

        n = h0.shape[0]

        cur_c5 = _c5_value(cur_h)
        best_h = cur_h.copy()
        best_c5 = cur_c5
        if best_c5 <= target_c5:
            return best_h, best_c5, n
    else:
        # No parent – start from a random binary balance
        n = 70
        cur_h = np.zeros(n, dtype=float)
        cur_h[rng.choice(n, size=n // 2, replace=False)] = 1.0
        cur_c5 = _c5_value(cur_h)
        best_h = cur_h.copy()
        best_c5 = cur_c5

    # ----------------------------------------------------------------------

    cur_h = _repair_binary(cur_h)

    if cur_c5 < best_c5:
