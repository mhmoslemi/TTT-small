import numpy as np
import time
import math

# ---------- Helper functions ----------

def project_onto_bounds_and_sum(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project vector `v` onto the box [0,1]^n intersected with the hyperplane sum = target.
    Uses a binary search on the offset λ such that sum(clip(v+λ,0,1)) = target.
    """
    lo, hi = -100.0, 100.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h = np.clip(v + mid, 0.0, 1.0)
        s = h.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    # Minor correction for tiny numerical drift
    diff = target - h.sum()
    if abs(diff) > 1e-8:
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size:
            i = free[0]
            h[i] = min(1.0, max(0.0, h[i] + diff))
        else:
            h += diff / h.size
            h = np.clip(h, 0.0, 1.0)
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector in [0,1]^n with sum exactly n/2."""
    v = rng.random(n)
    return project_onto_bounds_and_sum(v, n / 2.0)


def compute_c5(h: np.ndarray, dx: float) -> float:
    """Compute the discretized C5 value for a candidate vector h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float((corr * dx).max())


def propose(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Create a new candidate from h by a small random perturbation and re‑projection."""
    n = h.shape[0]
    y = h.copy()
    if rng.random() < 0.7:
        # Gaussian perturbation on a few coordinates
        k = rng.integers(1, max(2, n // 10))
        idx = rng.choice(n, size=k, replace=False)
        y[idx] += rng.normal(scale=0.2, size=k)
    else:
        # Simple swap of two entries
        i, j = rng.choice(n, size=2, replace=False)
        y[i], y[j] = y[j], y[i]
    # Project back to feasibility
    y = project_onto_bounds_and_sum(y, n / 2.0)
    return y


# ---------- Main entry point ----------

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a step function h on [0,2] (represented by samples) that yields a small C5.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Initialise from parent construction if available
    try:
        init = initial_h_values  # provided by the harness when present
        n = len(init)
        h_current = np.array(init, dtype=float)
        h_current = np.clip(h_current, 0.0, 1.0)
        h_current = project_onto_bounds_and_sum(h_current, n / 2.0)
    except NameError:
        # No parent construction – start from scratch
        n = 70
        h_current = random_feasible_vector(n, rng)

    dx = 2.0 / n
    best_h = h_current.copy()
    best_c5 = compute_c5(best_h, dx)

    # Simulated‑annealing style search
    T0 = 0.5            # initial temperature
    Tmin = 1e-4         # final temperature
    no_improve = 0

    while True:
        # Time check
        if time.time() - start_time > budget_s:
            break

        # Temperature schedule (linear cooling)
        elapsed = time.time() - start_time
        temp = max(Tmin, T0 * (1.0 - elapsed / budget_s))

        # Generate a neighbour
        cand = propose(best_h, rng)
        cand_c5 = compute_c5(cand, dx)

        # Accept if better, otherwise probabilistically
        if cand_c5 < best_c5 or rng.random() < math.exp(-(cand_c5 - best_c5) / temp):
            best_h = cand
            best_c5 = cand_c5
            no_improve = 0
        else:
            no_improve += 1

        # Occasionally inject a fresh random candidate to escape stagnation
        if no_improve >= 5000:
            cand = random_feasible_vector(n, rng)
            cand_c5 = compute_c5(cand, dx)
            if cand_c5 < best_c5:
                best_h = cand
                best_c5 = cand_c5
            no_improve = 0

    return best_h, float(best_c5), n
