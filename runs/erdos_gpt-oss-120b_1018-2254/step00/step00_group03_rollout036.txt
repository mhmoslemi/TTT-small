import time
import math
import random
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)*(1-h(x+k))dx approximated on the grid."""
    n = h.size
    dx = 2.0 / n
    # np.correlate implements sum_i a[i] * b[i+lag] (zero‑padded outside)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def make_two_step_vector(n: int, n_end: int, a: float) -> np.ndarray:
    """
    Build a symmetric two‑step vector of length n.
    First and last n_end entries are set to a (0 ≤ a ≤ 1).
    The middle part receives value b that satisfies the integral constraint.
    """
    target_sum = n / 2.0
    mid_len = n - 2 * n_end
    if mid_len <= 0:
        # degenerate case: whole vector constant a (should be a = 0.5)
        return np.full(n, target_sum / n, dtype=float)
    # solve for b from a*2*n_end + b*mid_len = target_sum
    b = (target_sum - 2.0 * a * n_end) / mid_len
    # clip to keep inside bounds, then fix any sum deviation later
    h = np.empty(n, dtype=float)
    h[:n_end] = a
    h[-n_end:] = a
    h[n_end:n_end + mid_len] = b
    # tiny correction to hit the sum exactly without breaking bounds
    s = h.sum()
    diff = target_sum - s
    if abs(diff) > 1e-12 and mid_len > 0:
        inc = diff / mid_len
        # ensure we stay inside [0,1]
        if np.all(h[n_end:n_end + mid_len] + inc <= 1.0 + 1e-12) and np.all(
            h[n_end:n_end + mid_len] + inc >= -1e-12
        ):
            h[n_end:n_end + mid_len] += inc
        else:
            # fallback: simple redistribution to a single free slot
            for i in range(n):
                new_val = h[i] + diff
                if 0.0 <= new_val <= 1.0:
                    h[i] = new_val
                    break
    return h

def adjust_sum_to_target(h: np.ndarray, target: float) -> np.ndarray:
    """
    Return a copy of h that satisfies the exact sum constraint while staying in [0,1].
    The adjustment is tiny (order of 1e-12) and therefore safe.
    """
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    # try to fix by adding diff to a non‑saturated entry
    for i in range(h.size):
        new_val = h[i] + diff
        if 0.0 <= new_val <= 1.0:
            h[i] = new_val
            return h
    # as a last resort, scale uniformly (should be very small)
    scale = target / s if s > 0 else 0.0
    h = np.clip(h * scale, 0.0, 1.0)
    return h

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretised) that lowers the
    Erdős overlap constant C5.  Returns (h_values, c5_bound, n_points).
    """
    random.seed(seed)
    np.random.seed(seed)

    # Preferred number of points – use the parent if available, otherwise 200
    init = globals().get("initial_h_values")
    if init is not None:
        h_parent = np.asarray(init, dtype=float)
        n = h_parent.size
    else:
        n = 200
        h_parent = None

    target_sum = n / 2.0
    best_h = None
    best_c5 = math.inf

    # ------------------------------------------------------------------
    # 1) Evaluate the supplied parent (if any)
    if h_parent is not None:
        h = np.clip(h_parent, 0.0, 1.0)
        h = adjust_sum_to_target(h, target_sum)
        c5 = compute_c5(h)
        best_h, best_c5 = h, c5

    # ------------------------------------------------------------------
    # 2) Grid search over symmetric two‑step shapes
    start_time = time.time()
    a_grid = np.linspace(0.5, 1.0, 26)  # step ≈0.02
    # n_end runs from 1 up to n//2 - 1 (leave at least one middle point)
    for n_end in range(1, n // 2):
        if time.time() - start_time > budget_s * 0.85:
            break
        for a in a_grid:
            h = make_two_step_vector(n, n_end, a)
            # Discard if b fell out of bounds (handled inside)
            if np.any(h < 0.0) or np.any(h > 1.0):
                continue
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    if best_h is None:

        best_h = np.zeros(n, dtype=float)
        ones = np.random.choice(n, n // 2, replace=False)
        best_h[ones] = 1.0
        best_c5 = compute_c5(best_h)

    current_c5 = best_c5
    elapsed = lambda: time.time() - start_time

    final_c5 = compute_c5(best_h)

    return best_h, float(final_c5), n
