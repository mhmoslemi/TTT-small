def repair_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # iterative distribution
    max_iter = 1000
    for _ in range(max_iter):
        cur = h.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        # Determine indices that can be moved in direction of diff
        if diff > 0:
            idx = np.where(h < 1)[0]
            if idx.size == 0:
                break
            # max allowed increase per idx
            max_inc = np.min(1 - h[idx])
            # distribute equally, but ensure not exceed max_inc per var
            inc = min(diff / idx.size, max_inc)
            h[idx] += inc
        else:
            idx = np.where(h > 0)[0]
            if idx.size == 0:
                break
            max_dec = np.min(h[idx])
            dec = max(diff / idx.size, -max_dec)  # diff negative, so dec negative
            h[idx] += dec  # dec negative
    # After loop, final clip (should be satisfied)
    h = np.clip(h, 0.0, 1.0)
    # final adjust for small residual using scaling on all entries (to guarantee sum exactly)
    cur = h.sum()
    if abs(cur - target) > 1e-8:
        # adjust smallest difference by scaling factor
        # simple approach: adjust all entries proportionally (then clip again)
        factor = target / cur if cur != 0 else 0
        h = np.clip(h * factor, 0.0, 1.0)
        # final iteration to meet sum perhaps still off; adjust by small correction on first entry that does not violate bound
        cur = h.sum()
        diff = target - cur
        if abs(diff) > 1e-9:
            # add diff to first non-bound entry
            for i in range(n):
                if (diff > 0 and h[i] < 1) or (diff < 0 and h[i] > 0):
                    h[i] += diff
                    break
    # final clip
    h = np.clip(h, 0.0, 1.0)
    return h

import time
import random
import math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """
    Compute C5 = max_k ∫ h(x) (1 - h(x+k)) dx
    using discrete correlation on the vector h.
    """
    # correlation of h with (1 - h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / h.size
    return float(corr.max() * dx)

def _repair_h(h: np.ndarray, n: int) -> np.ndarray:
    """
    Project h onto the feasible set:
        0 <= h_i <= 1
        sum(h_i) = n/2
    Simple iterative water-filling approach.
    """
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

    # Distribute excess/deficit greedily
    eps = 1e-12
    for _ in range(1000):
        cur = h.sum()
        diff = target - cur
        if abs(diff) < eps:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
            if idx.size == 0:
                break
            # add uniformly but respect upper bound
            add = min(diff / idx.size, np.min(1.0 - h[idx]))
            h[idx] += add
        else:
            idx = np.where(h > 0.0)[0]
            if idx.size == 0:
                break
            sub = min(-diff / idx.size, np.min(h[idx]))
            h[idx] -= sub

    # Final tiny correction if needed
    cur = h.sum()
    diff = target - cur
    if abs(diff) > 1e-9:
        # scale uniformly (still within bounds after clipping)
        if cur > 0:
            factor = target / cur
            h = np.clip(h * factor, 0.0, 1.0)
        cur = h.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            # adjust a single free entry
            for i in range(n):
                if diff > 0 and h[i] < 1.0:
                    inc = min(diff, 1.0 - h[i])
                    h[i] += inc
                    diff -= inc
                elif diff < 0 and h[i] > 0.0:
                    dec = min(-diff, h[i])
                    h[i] -= dec
                    diff += dec
                if abs(diff) < 1e-12:
                    break
    return np.clip(h, 0.0, 1.0)

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] (discretized) to minimize C5.
    Returns (h_values, c5_bound, n_points).
    """
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()

    # Determine problem size
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        h_curr = np.array(initial_h_values, dtype=float)
    else:
        n = 42
        h_curr = np.full(n, 0.5, dtype=float)  # uniform feasible start

    # Ensure feasibility
    h_curr = _repair_h(h_curr, n)

    cur_score = _compute_c5(h_curr)
    best_h = h_curr.copy()
    best_score = cur_score

    # Simulated‑annealing parameters
    init_temp = 0.05  # initial temperature (allows occasional uphill moves)
    # Main optimisation loop
    while time.time() - start_time < budget_s * 0.95:
        # Occasionally inject a global perturbation around the best solution
        if random.random() < 0.01:
            perturb = np.random.uniform(-0.2, 0.2, size=n)
            cand = best_h + perturb
            cand = _repair_h(cand, n)
            cand_score = _compute_c5(cand)
            if cand_score < best_score:
                best_h = cand.copy()
                best_score = cand_score
                h_curr = cand.copy()
                cur_score = cand_score
            continue

        # Pairwise elementary move that keeps sum exactly constant
        i, j = random.sample(range(n), 2)
        low = max(-h_curr[i], h_curr[j] - 1.0)
        high = min(1.0 - h_curr[i], h_curr[j])
        if low >= high:
            continue
        delta = random.uniform(low, high)
        cand = h_curr.copy()
        cand[i] += delta
        cand[j] -= delta
        cand_score = _compute_c5(cand)

        # Temperature decreasing linearly with elapsed time
        elapsed = time.time() - start_time
        frac = elapsed / budget_s
        T = init_temp * (1.0 - frac)

        # Acceptance rule
        if cand_score < cur_score or random.random() < math.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            h_curr = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_h = cand.copy()
                best_score = cand_score

    # Final projection to guarantee feasibility
    best_h = _repair_h(best_h, n)
    final_score = _compute_c5(best_h)

    return best_h.tolist(), float(final_score), int(n)
