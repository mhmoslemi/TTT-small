import time, numpy as np

def _project_to_box_simplex(v, target):
    # binary search for lambda such that sum_i clip(v_i - lam,0,1) = target
    lo = -np.max(v) - 2.0
    hi = np.max(v) + 2.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _c5_circular(h):
    N = h.shape[0]
    corr = np.fft.ifft(np.fft.fft(h) * np.conj(np.fft.fft(1.0 - h))).real
    return np.max(corr) * (2.0 / N)

def _objective(h):
    try:
        return evaluate_erdos_solution(h)  # pre‑imported helper
    except Exception:
        return _c5_circular(h)

def _perturb_pair(h, i, j, rng):
    hi, hj = h[i], h[j]
    # delta will be added to hi and subtracted from hj
    d_min = max(-hi, hj - 1.0)
    d_max = min(1.0 - hi, hj)
    if d_min >= d_max:
        return None
    delta = d_min + rng.random() * (d_max - d_min) * rng.random()
    # the second rng.random() scales variation, giving small moves
    h_new = h.copy()
    h_new[i] = hi + delta
    h_new[j] = hj - delta
    return h_new

import time
import numpy as np

def _project_to_box_simplex(v, target):
    """Project v onto {0<=x_i<=1, sum x_i = target}."""
    lo = -np.max(v) - 2.0
    hi = np.max(v) + 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _c5_circular(h):
    """Circular overlap bound: max_k ∑ h_i (1‑h_{i+k}) · (2/N)."""
    N = h.shape[0]
    corr = np.fft.ifft(np.fft.fft(h) * np.conj(np.fft.fft(1.0 - h))).real
    return np.max(corr) * (2.0 / N)


def _objective(h):
    """Use the provided evaluator if present, else fall back."""
    try:
        return evaluate_erdos_solution(h)   # pre‑imported helper
    except Exception:
        return _c5_circular(h)


def _perturb_pair(h, i, j, rng):
    """Move a small amount from index j to i, preserving the sum."""
    hi, hj = h[i], h[j]
    # delta added to hi, subtracted from hj
    d_min = max(-hi, hj - 1.0)
    d_max = min(1.0 - hi, hj)
    if d_min >= d_max:
        return None
    # bias towards small moves
    delta = d_min + rng.random() * (d_max - d_min) * rng.random()
    h_new = h.copy()
    h_new[i] = hi + delta
    h_new[j] = hj - delta
    return h_new


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- initialise -------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.asarray(initial_h_values, dtype=np.float64)
        N = h.size
    else:
        N = 120                     # modest size, faster optimization
        h = rng.random(N)

    target_sum = N * 0.5
    h = _project_to_box_simplex(h, target_sum)

    cur_val = _objective(h)
    best_h = h.copy()
    best_val = cur_val

    # ----- stochastic improvement --------------------------------------
    while time.time() - start < budget_s * 0.96:
        i, j = rng.choice(N, size=2, replace=False)
        h_new = _perturb_pair(h, i, j, rng)
        if h_new is None:
            continue
        new_val = _objective(h_new)

        if new_val < best_val:
            best_val = new_val
            best_h = h_new.copy()

        if new_val < cur_val:
            h, cur_val = h_new, new_val
        else:
            # simple simulated‑annealing acceptance
            t = max(0.0, 1.0 - (time.time() - start) / budget_s)

            if rng.random() < prob:

        # occasional random restart to escape stagnation
        if rng.random() < 0.001:
