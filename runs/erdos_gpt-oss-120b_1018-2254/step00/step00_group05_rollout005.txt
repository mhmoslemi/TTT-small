def propose_move(h, rng, step):
    n = h.shape[0]
    # pick i where h[i] < 1
    inc_candidates = np.where(h < 1 - 1e-12)[0]
    dec_candidates = np.where(h > 1e-12)[0]
    if inc_candidates.size == 0 or dec_candidates.size == 0:
        return None, None, None
    i = rng.choice(inc_candidates)
    j = rng.choice(dec_candidates)
    max_inc = 1 - h[i]
    max_dec = h[j]
    max_delta = min(max_inc, max_dec, step)
    if max_delta <= 0:
        return None, None, None
    delta = rng.random() * max_delta
    return i, j, delta
```

    total_needed = n/2
    cur_sum = h.sum()

        # Scale to bring sum to needed while staying within [0,1]
        # We'll solve by scaling the whole vector and then clipping.
        # Simple approach: compute diff = total_needed - cur_sum; then add equally to all entries within bounds.
        diff = total_needed - cur_sum
        # distribute diff proportionally to entries not at bounds
        # Use simple iteration: while diff not near zero:
        for _ in range(1000):
            # find indices wnon = where (diff>0 and h<1) or (diff<0 and h>0)
            if diff > 0:
                mask = h < 1 - 1e-12
            else:

            if not np.any(mask):
                break
            adjust = diff / np.count_nonzero(mask)

            adjust = np.clip(adjust, -np.min(h[mask]), 1 - np.max(h[mask]))

            cur_sum = h.sum()
            diff = total_needed - cur_sum

        h = np.clip(h, 0, 1)
        # final renormalize if small error
        cur_sum = h.sum()

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time
    import math

    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # Initialize h
    # ----------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        # fallback: uniform 0.5 on a modest grid
        n = 140
        h = np.full(n, 0.5, dtype=np.float64)

    n = h.shape[0]
    target_sum = n / 2.0

    # Project onto the feasible set {0≤h≤1, sum=h_target}
    def project_to_feasible(v):
        v = np.clip(v, 0.0, 1.0)
        cur = v.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            return v
        # distribute diff over non‑boundary entries
        for _ in range(100):
            mask = (v > 1e-12) & (v < 1 - 1e-12)
            if not mask.any():
                break
            adjust = diff / mask.sum()
            adjust = np.clip(adjust,
                             -np.min(v[mask]),
                             1 - np.max(v[mask]))
            v[mask] += adjust
            cur = v.sum()
            diff = target_sum - cur
            if abs(diff) < 1e-12:
                break
        # final tiny tweak (may slightly violate bounds, then re‑clip)
        if abs(diff) > 1e-9:
            v += diff / n
            v = np.clip(v, 0.0, 1.0)
        return v

    h = project_to_feasible(h)

    # ----------------------------------------------------------------------
    # Core evaluation: C5 = max_x ∫ h(x)(1‑h(x+dx·k)) dx
    # ----------------------------------------------------------------------
    def compute_c5(arr):
        n_local = arr.shape[0]
        dx = 2.0 / n_local
        size = 1 << ((2 * n_local - 1).bit_length())   # next power of two
        H = np.fft.rfft(arr, n=size)
        G = np.fft.rfft(1.0 - arr, n=size)
        conv = np.fft.irfft(H * np.conj(G), n=size)[: 2 * n_local - 1]
        return np.max(conv.real) * dx

    # ----------------------------------------------------------------------
    # Simulated‑annealing / hill‑climbing loop
    # ----------------------------------------------------------------------
    best_h = h.copy()
    best_score = compute_c5(best_h)

    curr_h = h.copy()
    curr_score = best_score

    start = time.time()
    end_time = start + budget_s * 0.95   # leave a safety margin

    while time.time() < end_time:
        # schedule for step size and temperature
        elapsed = time.time() - start
        prog = elapsed / (budget_s * 0.95)

        max_step = 0.1 * (1.0 - prog) + 0.001  # maximum delta to move
        temperature = 0.01 * (1.0 - prog) + 1e-5

        # choose increasing and decreasing indices
        inc_idx = np.where(curr_h < 1.0 - 1e-12)[0]
        dec_idx = np.where(curr_h > 1e-12)[0]
        if inc_idx.size == 0 or dec_idx.size == 0:
            break

        i = rng.choice(inc_idx)
        j = rng.choice(dec_idx)

        # how far can we shift while staying feasible?
        max_inc = 1.0 - curr_h[i]
        max_dec = curr_h[j]
        delta_max = min(max_inc, max_dec, max_step)
        if delta_max <= 0.0:
            continue
        delta = rng.random() * delta_max

        # propose move
        curr_h[i] += delta
        curr_h[j] -= delta

        new_score = compute_c5(curr_h)

        # Metropolis acceptance
        if new_score < curr_score or rng.random() < math.exp(-(new_score - curr_score) / temperature):
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = curr_h.copy()
        else:
            # revert
            curr_h[i] -= delta
            curr_h[j] += delta

    # ----------------------------------------------------------------------
    # Final projection & scoring
    # ----------------------------------------------------------------------
    best_h = project_to_feasible(best_h)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, best_h.shape[0]
