import numpy as np
import time
import math

def _project_onto_simplex_box(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto the set {x : 0<=x_i<=1, sum(x)=target_sum}."""
    # Clip to box first to ensure bounds.
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    # Binary search for a shift delta such that sum(clip(v+delta,0,1)) == target_sum.
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Optimize a step function for the Erdős C₅ constant.

    Returns:
        (list of h values, best C₅ bound found, number of sample points)
    """
    rng = np.random.default_rng(seed)

    # Determine number of points and initial candidate.
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        n = init.size
        current = _project_onto_simplex_box(init, n / 2.0)
    else:
        n = 200
        current = _project_onto_simplex_box(rng.random(n), n / 2.0)

    # Helper to evaluate C₅ for a candidate.
    def _evaluate(h_vec: np.ndarray) -> float:
        try:
            # evaluate_erdos_solution is provided in the execution environment.
            return float(evaluate_erdos_solution(h_vec)[0])
        except Exception:
            # Fallback: direct computation.
            dx = 2.0 / n
            corr = np.correlate(h_vec, 1.0 - h_vec, mode="full") * dx
            return float(corr.max())

    current_score = _evaluate(current)
    best = current.copy()
    best_score = current_score

    start_time = time.time()
    deadline = start_time + budget_s
    iteration = 0

    # Simulated annealing parameters.
    T0 = 0.02          # initial temperature
    tau = 200000.0    # decay constant

    while time.time() < deadline - 0.1:
        iteration += 1
        # Propose a new candidate.
        new = current.copy()
        if rng.random() < 0.05:
            # Occasionally replace with a fresh random feasible vector.
            new = _project_onto_simplex_box(rng.random(n), n / 2.0)
        else:
            # Mass‑transfer move that respects bounds and keeps the total sum.
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1  # ensure i != j
            # Transfer from i to j if possible, otherwise try the opposite direction.
            max_transfer = min(new[i], 1.0 - new[j])
            if max_transfer > 0:
                delta = rng.random() * max_transfer
                new[i] -= delta
                new[j] += delta
            else:
                # Try transfer from j to i.
                max_transfer = min(new[j], 1.0 - new[i])
                if max_transfer > 0:
                    delta = rng.random() * max_transfer
                    new[j] -= delta
                    new[i] += delta
                # If neither direction works, keep new identical to current.

        new_score = _evaluate(new)

        # Track the best solution seen.
        if new_score < best_score:
            best = new.copy()
            best_score = new_score

        # Acceptance criterion (simulated annealing).
        if new_score <= current_score:
            accept = True
        else:
