def propose_neighbor(h, step_max):
    N = len(h)
    i,j = np.random.choice(N, 2, replace=False)
    # ensure i will increase, j will decrease
    # compute max increase, max decrease
    max_inc = min(step_max, 1.0 - h[i])
    max_dec = min(step_max, h[j])
    max_delta = min(max_inc, max_dec)
    if max_delta <= 0:
        return h
    delta = np.random.uniform(0, max_delta)
    # randomly choose direction: perhaps also allow decreasing i and increasing j
    # to keep variety
    if np.random.rand() < 0.5:
        # increase i, decrease j
        new_h = h.copy()
        new_h[i] += delta
        new_h[j] -= delta
    else:
        # opposite direction
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta

def project_to_constraints(v, target):
    v = np.clip(v, 0.0, 1.0)
    # iterative water-filling
    for _ in range(1000):
        cur = v.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to increase sum: add equally to elements <1
            idx = v < 1 - 1e-12
            if not np.any(idx):
                break
            add = diff / np.sum(idx)
            v[idx] += add
            v = np.clip(v, 0.0, 1.0)
        else:
            # need to decrease sum: subtract equally from elements >0
            idx = v > 1e-12
            if not np.any(idx):
                break
            sub = diff / np.sum(idx)  # diff negative
            v[idx] += sub  # subtract because diff negative
            v = np.clip(v, 0.0, 1.0)
    # final clamp
    v = np.clip(v, 0.0, 1.0)
    # adjust possible small remaining error by adding to first index if possible
    cur = v.sum()
    if abs(cur - target) > 1e-8:
        # adjust one element within bounds
        diff = target - cur
        for i in range(len(v)):
            if (diff > 0 and v[i] < 1) or (diff < 0 and v[i] > 0):
                change = min(abs(diff), 1 - v[i]) if diff > 0 else min(abs(diff), v[i])
                v[i] += np.sign(diff) * change
                break
    return v
```

def propose_neighbor(h, step_max):
    n = len(h)
    i, j = np.random.choice(n, 2, replace=False)
    # random direction sign
    # Compute max allowed increase at i and max allowed decrease at j
    max_inc_i = 1.0 - h[i]
    max_dec_i = h[i]
    max_inc_j = 1.0 - h[j]
    max_dec_j = h[j]
    # Choose direction randomly: increase i, decrease j or opposite.
    if np.random.rand() < 0.5:
        # increase i, decrease j
        delta_max = min(step_max, max_inc_i, max_dec_j)
        if delta_max <= 0:
            return h
        delta = np.random.uniform(0, delta_max)
        new_h = h.copy()
        new_h[i] += delta
        new_h[j] -= delta
    else:
        # decrease i, increase j
        delta_max = min(step_max, max_dec_i, max_inc_j)
        if delta_max <= 0:
            return h
        delta = np.random.uniform(0, delta_max)
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
    # No need to adjust sum; delta added/subtracted cancel.
    return new_h

import numpy as np, time, math

def _project_to_constraints(v, target_sum):
    """Project vector v onto the box [0,1]^n with sum = target_sum."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(500):
        cur = v.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = v < 1.0
            if not np.any(idx):
                break
            add = diff / np.count_nonzero(idx)
            v[idx] += add
        else:
            idx = v > 0.0
            if not np.any(idx):
                break
            sub = diff / np.count_nonzero(idx)   # diff is negative
            v[idx] += sub
        v = np.clip(v, 0.0, 1.0)
    # final tiny fix if needed
    cur = v.sum()
    if abs(cur - target_sum) > 1e-9:
        diff = target_sum - cur
        for i in range(len(v)):
            if diff > 0 and v[i] < 1.0:
                step = min(diff, 1.0 - v[i])
                v[i] += step
                break
            if diff < 0 and v[i] > 0.0:
                step = min(-diff, v[i])
                v[i] -= step
                break
    return v

def _eval_solution(h):

    res = evaluate_erdos_solution(h)
    try:

        return float(res[0])

def _propose_neighbor(h, step_max):
    """Create a neighbor by moving a small amount from one index to another,
    preserving the total sum."""
    n = h.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    # random direction
    if np.random.rand() < 0.5:
        # increase i, decrease j
        max_inc = min(step_max, 1.0 - h[i])
        max_dec = min(step_max, h[j])
        delta_max = min(max_inc, max_dec)
        if delta_max <= 0:
            return h
        delta = np.random.uniform(0.0, delta_max)
        new_h = h.copy()
        new_h[i] += delta
        new_h[j] -= delta
    else:
        # decrease i, increase j
        max_dec = min(step_max, h[i])
        max_inc = min(step_max, 1.0 - h[j])
        delta_max = min(max_dec, max_inc)
        if delta_max <= 0:
            return h
        delta = np.random.uniform(0.0, delta_max)
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
    return new_h
