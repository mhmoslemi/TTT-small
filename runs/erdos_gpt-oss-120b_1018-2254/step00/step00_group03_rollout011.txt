import time
import numpy as np
import random
import math

def _compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def _gradient_for_lag(h, lag):
    n = h.shape[0]
    g = np.ones(n)
    if lag >= 0:
        L = lag
        g[:n-L] -= h[L:]
        g[L:]   -= h[:n-L]
    else:
        L = -lag
        g[L:]   -= h[:n-L]
        g[:n-L] -= h[L:]
    return g

def _project_onto_sum_box(v, target):
    # find λ such that sum(clip(v+λ,0,1)) = target
    lo = -np.max(v)
    hi = 1.0 - np.min(v)
    for _ in range(30):
        mid = (lo + hi) * 0.5
        z = np.clip(v + mid, 0.0, 1.0)
        s = z.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    z = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - z.sum()
    if abs(diff) > 1e-12:
        free = np.where((z > 1e-12) & (z < 1.0 - 1e-12))[0]
        if free.size:
            inc = diff / free.size
            z[free] += inc
        else:
            idx = np.argmax(z) if diff > 0 else np.argmin(z)
            z[idx] = np.clip(z[idx] + diff, 0.0, 1.0)
        z = np.clip(z, 0.0, 1.0)
    return z

import time
import numpy as np
import random
import math

def _compute_c5(h):
    """Return max_k ∫ h(x)*(1-h(x+k)) dx for discrete h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _gradient_for_lag(h, lag):
    """Gradient ∂/∂h of the overlap term at a given integer lag."""
    n = h.shape[0]
    g = np.ones(n)
    if lag >= 0:
        L = lag
        g[:n - L] -= h[L:]
        g[L:] -= h[:n - L]
    else:
        L = -lag
        g[L:] -= h[:n - L]
        g[:n - L] -= h[L:]
    return g

def _project_onto_sum_box(v, target):
    """
    Project vector v onto the hyperplane sum(z)=target with box constraints 0≤z≤1.
    Uses a simple bisection on the common offset λ: z = clip(v+λ,0,1).
    """
    lo = -np.max(v)                     # makes v+lo ≤0 → sum=0
    hi = 1.0 - np.min(v)                # makes v+hi ≥1 → sum=n
    for _ in range(30):
        mid = (lo + hi) * 0.5
        z = np.clip(v + mid, 0.0, 1.0)
        s = z.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    z = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - z.sum()
    if abs(diff) > 1e-12:
        free = np.where((z > 1e-12) & (z < 1.0 - 1e-12))[0]
        if free.size:
            inc = diff / free.size
            z[free] += inc
        else:
            idx = np.argmax(z) if diff > 0 else np.argmin(z)
            z[idx] = np.clip(z[idx] + diff, 0.0, 1.0)
        z = np.clip(z, 0.0, 1.0)
    return z
