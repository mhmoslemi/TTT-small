def projection(v, S):
    low = -np.max(v) - 2.0
    high = 1.0 - np.min(v) + 2.0
    for _ in range(50):
        lam = (low + high) / 2.0
        h_proj = np.clip(v + lam, 0.0, 1.0)
        current_sum = h_proj.sum()
        if current_sum < S:
            low = lam
        else:
            high = lam
    # final projection
    h_proj = np.clip(v + (low+high)/2.0, 0.0, 1.0)
    return h_proj

def max_overlap(h, dx):
    n = len(h)
    rev = (1 - h)[::-1]
    L = 1 << ((2*n - 1).bit_length())
    # compute using np.fft.rfft
    H = np.fft.rfft(h, n=L)
    R = np.fft.rfft(rev, n=L)
    corr = np.fft.irfft(H * R, n=L)[:2*n-1]
    max_corr = np.max(corr)
    return max_corr * dx, max_corr, int(np.argmax(corr) - (n - 1))

def project_onto_box_and_sum(v, target_sum):
    # use binary search for lambda
    low = -np.max(v) - 5.0
    high = 1.0 - np.min(v) + 5.0
    for _ in range(50):
        lam = (low + high) / 2.0
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s < target_sum:
            low = lam
        else:
            high = lam
    proj = np.clip(v + (low + high) / 2.0, 0.0, 1.0)
    return proj

def propose_move(h, rng):
    n = len(h)
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    # Compute allowed delta
    max_inc_i = 1.0 - h[i]
    max_dec_j = h[j]
    max_transfer = min(max_inc_i, max_dec_j)
    max_dec_i = h[i]
    max_inc_j = 1.0 - h[j]
    min_transfer = -min(max_dec_i, max_inc_j)  # negative amount means transfer from i to j
    if max_transfer == 0 and min_transfer == 0:
        return None
    d = rng.uniform(min_transfer, max_transfer)

def max_corr(h):
    n = len(h)
    rev = (1 - h)[::-1]
    L = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, n=L)
    R = np.fft.rfft(rev, n=L)
    corr = np.fft.irfft(H * R, n=L)[:2*n-1]
    idx = np.argmax(corr)
    max_val = corr[idx]
    shift = idx - (n - 1)
    return max_val, shift

import time
import math
import numpy as np

def _project_box_sum(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto the box [0,1]^n with sum(v)=target_sum."""
    # binary search for lambda such that sum(clip(v+λ,0,1)) = target_sum
    lo = -np.max(v) - 5.0
    hi = 1.0 - np.min(v) + 5.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s < target_sum:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v + lam, 0.0, 1.0)


def _max_corr_and_shift(h: np.ndarray):
    """Return (max_correlation, shift) for correlation of h with (1-h)."""
    n = h.size
    rev = (1.0 - h)[::-1]
    L = 1 << ((2 * n - 1).bit_length())  # next power of two
    H = np.fft.rfft(h, n=L)
    R = np.fft.rfft(rev, n=L)
    corr = np.fft.irfft(H * R, n=L)[: 2 * n - 1]
    idx = int(np.argmax(corr))
    max_val = float(corr[idx])
    shift = idx - (n - 1)
    return max_val, shift


def _c5_value(h: np.ndarray, dx: float) -> float:
    """Compute C5 for a given h."""
    max_corr, _ = _max_corr_and_shift(h)
    return max_corr * dx


def _propose_move(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a new feasible h by transferring mass between two indices."""
    n = h.size
    # try a few times to find a pair with non‑zero transfer range
    for _ in range(10):
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        # allowed increase of i and decrease of j
        inc_i = 1.0 - h[i]
        dec_j = h[j]
        max_pos = min(inc_i, dec_j)
        # allowed decrease of i and increase of j
        dec_i = h[i]
        inc_j = 1.0 - h[j]
        max_neg = -min(dec_i, inc_j)
        if max_pos > 0.0 or max_neg < 0.0:
            d = rng.uniform(max_neg, max_pos)
            h_new = h.copy()
            h_new[i] += d
            h_new[j] -= d
            # safeguard against tiny numerical drift
            h_new[i] = min(max(h_new[i], 0.0), 1.0)
            h_new[j] = min(max(h_new[j], 0.0), 1.0)
            return h_new
    # fallback: return unchanged copy (very unlikely)
    return h.copy()

    """
    Search for a step function h on [0,2] with values in [0,1] and integral 1
    that minimizes the Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)

    # problem size (feel free to change; must stay < 1000)
    n = 400
    target_sum = n / 2.0
    dx = 2.0 / n

    # initialise h
    if "initial_h_values" in globals():

    best_h = h.copy()
