def project_onto_bounds_and_sum(x, target_sum):
    # Clip to bounds 0,1
    x = np.clip(x, 0, 1)
    # Compute diff
    diff = np.sum(x) - target_sum
    # loop until diff small
    for _ in range(10):
        if abs(diff) < 1e-8:
            break
        # find free indices (not at bounds)
        mask = (x > 0) & (x < 1)
        n_free = np.count_nonzero(mask)
        if n_free == 0:
            # can't adjust further, break
            break
        adjustment = diff / n_free
        # adjust free vars
        x[mask] -= adjustment
        # clip again (some may go out of bounds)
        x = np.clip(x, 0, 1)
        diff = np.sum(x) - target_sum
    return x

def compute_score(h):

def compute_score(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1-h(x+k)) dx approximated on the grid."""
    N = len(h)
    # use float for correlation

    """Return the shift k (relative index) that attains the maximal correlation."""
    N = len(h)
    corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode="full")
    idx = int(np.argmax(corr))
    k = idx - (N - 1)

    """
    For a given shift k, return two arrays:
    - indices i where h[i]==1 and h[i+k]==0  (type‑1 mismatches)
    - indices i where h[i]==0 and h[i+k]==1  (type‑2 mismatches)
    Only indices where both i and i+k are inside the array are considered.
    """
    N = len(h)
    i_start = max(0, -k)
    i_end = min(N, N - k)
    if i_start >= i_end:
        return np.empty(0, dtype=int), np.empty(0, dtype=int)
    i_vals = np.arange(i_start, i_end, dtype=int)
    partner = i_vals + k
    mism1 = i_vals[(h[i_vals] == 1) & (h[partner] == 0)]
    mism2 = i_vals[(h[i_vals] == 0) & (h[partner] == 1)]
    return mism1, mism2

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Return a tuple (h_values, c5_bound, n_points) where
    - h_values is a NumPy array with values in [0,1] and sum = n_points/2,
    - c5_bound is the C5 value of the returned h,
    - n_points is the length of the array.
    The algorithm uses a simple simulated‑annealing / hill‑climbing
    search to reduce the maximal overlap integral.
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # initialise a binary candidate
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        # duplicate each entry to obtain an even length and keep the same total mass
        parent = np.repeat(parent, 2)
        N = len(parent)
        # turn the duplicated vector into a binary one with exactly N/2 ones
        order = np.argsort(-parent)               # decreasing order
        h = np.zeros(N, dtype=int)
        h[order[: N // 2]] = 1
    else:
        N = 140                     # a modest size that still fits the budget
        h = np.zeros(N, dtype=int)
        ones = np.random.choice(N, N // 2, replace=False)
        h[ones] = 1

    # current and best states
    cur_score = compute_score(h)
    best_h = h.copy()
    best_score = cur_score

    start_time = time.time()
    # simple temperature schedule (linear cooling)
    T0 = 0.02

    while time.time() - start_time < budget_s * 0.95:
        # locate the worst shift for the current configuration
        k, _ = worst_shift(h)

        # try to fix a type‑1 / type‑2 mismatch pair for that shift
        mism1, mism2 = mismatch_indices(h, k)
        if mism1.size and mism2.size:
            i = int(np.random.choice(mism1))
            j = int(np.random.choice(mism2))
        else:
            # fallback: any random 1↔0 swap
            ones_idx = np.where(h == 1)[0]
            zeros_idx = np.where(h == 0)[0]
            i = int(np.random.choice(ones_idx))
            j = int(np.random.choice(zeros_idx))

        # propose the swap
        h_new = h.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]

        new_score = compute_score(h_new)

        # deterministic improvement
        if new_score < cur_score - 1e-12:
            h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            # occasional uphill move governed by a decreasing temperature
            elapsed = time.time() - start_time
            temp = T0 * max(0.0, 1.0 - elapsed / (budget_s * 0.9))
            if temp > 0.0:
                prob = math.exp(-(new_score - cur_score) / temp)
                if random.random() < prob:
                    h = h_new
                    cur_score = new_score

        # early exit if we already beat the target bound
        if best_score <= 0.3808:
            break

    # final preparation of the result
    best_h_float = best_h.astype(float)          # required format [0,1]
    final_score = compute_score(best_h_float)
    return best_h_float, final_score, len(best_h_float)
