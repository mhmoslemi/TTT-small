def compute_c5(h):
    conv = np.correlate(h, 1 - h, mode='full')
    return np.max(conv) * dx

def random_swap(h):
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones)==0 or len(zeros)==0:
        return h.copy()
    i = np.random.choice(ones)
    j = np.random.choice(zeros)
    h_new = h.copy()
    h_new[i] = 0
    h_new[j] = 1
    return h_new, i, j

def anneal(h, max_iter, time_limit):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_h = h.copy()
    current_c5 = best_c5
    start = time.time()
    T0 = 0.1
    for it in range(max_iter):
        if time.time() - start > time_limit: break
        # temperature schedule: T = T0 * (1 - it/max_iter)
        T = T0 * (1 - it / max_iter)
        # generate neighbor
        h_new, i, j = random_swap(current_h)
        new_c5 = compute_c5(h_new)
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / (T+1e-12)):
            current_h = h_new
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5
    return best_h, best_c5
```

def project_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-9, max_iter=100):
    lo = -1.0
    hi = 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, lower, upper)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
        if abs(s - target) <= tol:
            return proj
    return np.clip(v - (lo + hi) / 2.0, lower, upper)
```

import numpy as np
import time

def project_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-9, max_iter=100):
    """Project v onto the hyperplane sum=h_target intersected with box [lower,upper]."""
    lo, hi = -1.0, 1.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, lower, upper)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, lower, upper)

def compute_c5(h, dx):
    """Return C5 for a discrete step function h."""
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def local_search(h_start, target, n, dx, time_limit, rng):
    """Simple projected swap annealing within a time budget."""
    h = h_start.copy()
    best_h = h.copy()
    best_val = compute_c5(h, dx)
    start = time.time()
    max_iter = 20000
    it = 0
    while time.time() - start < time_limit and it < max_iter:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        delta_max = min(h[i], 1.0 - h[j])
        if delta_max <= 0.0:
            continue
        T = max(0.01, 1.0 - it / max_iter)
        delta = rng.random() * delta_max * T
        if delta == 0.0:
            continue
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        # numerical safety
        h_new = np.clip(h_new, 0.0, 1.0)
        s = h_new.sum()
        if abs(s - target) > 1e-8:
            h_new = project_box_sum(h_new, target)
        val = compute_c5(h_new, dx)
        if val < best_val or rng.random() < np.exp((best_val - val) / (0.001 + T)):
            h = h_new
            if val < best_val:
                best_val = val
                best_h = h_new.copy()
        it += 1
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_total = time.time()

    # ----- initial construction -----
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
        target = n / 2.0
        h0 = np.clip(h0, 0.0, 1.0)
        if abs(h0.sum() - target) > 1e-9:
            h0 = project_box_sum(h0, target)
    else:
        n = 120
        target = n / 2.0
        h0 = rng.random(n)
        h0 = project_box_sum(h0, target)

    dx = 2.0 / n
    best_h = h0.copy()
    best_val = compute_c5(best_h, dx)

    # ----- iterative improvement -----
    while True:
        elapsed = time.time() - start_total
        remaining = budget_s - elapsed
        if remaining < 5.0:
            break
        # allocate a slice of remaining time for this restart
        time_limit = min(30.0, remaining * 0.5)

        # generate a starting point: perturb current best or random fresh
        if rng.random() < 0.6:

            h_start = np.clip(h_start, 0.0, 1.0)
            if abs(h_start.sum() - target) > 1e-8:
                h_start = project_box_sum(h_start, target)
