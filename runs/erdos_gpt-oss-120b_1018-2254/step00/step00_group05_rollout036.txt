import numpy as np, time, random, math

def project_to_constraints(v, target):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo+hi)/2.0
        x = np.clip(v-mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v-hi, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-8:
        free = (x > 1e-12) & (x < 1-1e-12)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def evaluate_h(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return corr.max() * dx

def gradient_maxlag(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    lag = idx - (n - 1)
    # gradient
    g = np.ones(n)
    if lag == 0:
        g = 1 - 2*h
    else:
        indices = np.arange(n)
        idx1 = indices + lag
        idx2 = indices - lag
        term1 = np.zeros(n)
        term2 = np.zeros(n)
        valid1 = (idx1 >= 0) & (idx1 < n)
        term1[valid1] = h[idx1[valid1]]
        valid2 = (idx2 >= 0) & (idx2 < n)
        term2[valid2] = h[idx2[valid2]]
        g -= (term1 + term2)
    return g, lag

def sample_five_interval_binary(n, rng):
    H = n // 2
    Z = n - H
    # sample Dirichlet for ones (5) and zeros gaps (6)
    a = rng.dirichlet([1.0]*5)
    b = rng.dirichlet([1.0]*6)
    # allocate integer counts to match sums
    a_counts = np.floor(a * H).astype(int)
    remainder = H - a_counts.sum()
    if remainder > 0:
        frac = a * H - a_counts
        idx = np.argsort(-frac)[:remainder]
        a_counts[idx] += 1
    b_counts = np.floor(b * Z).astype(int)
    remainder = Z - b_counts.sum()
    if remainder > 0:
        frac = b * Z - b_counts
        idx = np.argsort(-frac)[:remainder]
        b_counts[idx] += 1
    # construct vector
    vec = []
    for i in range(5):
        vec.extend([0]*b_counts[i])
        vec.extend([1]*a_counts[i])
    vec.extend([0]*b_counts[5])
    h = np.array(vec, dtype=float)
    if h.size < n:
        h = np.concatenate([h, np.zeros(n - h.size)])
    elif h.size > n:
        h = h[:n]
    return h

def hill_climb_binary(h, score, end_time):

    target = n / 2.0

        zero_idx = np.where(best_h < 0.5)[0]
        if ones_idx.size == 0 or zero_idx.size == 0:
            break
        i = np.random.choice(ones_idx)
        j = np.random.choice(zero_idx)
        new_h = best_h.copy()
        new_h[i], new_h[j] = 0.0, 1.0
        new_score = evaluate_h(new_h)
        if new_score < best_score:
            best_h = new_h
            best_score = new_score
            if best_score < 0.3808:

    return best_h, best_score

def gradient_descent(h, target_sum, end_time):
    n = h.size

            best_score = new_score

import numpy as np, time, random, math

def _project_to_constraints(v: np.ndarray, target: float) -> np.ndarray:
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - hi, 0.0, 1.0)
    return x

def _evaluate_h(h: np.ndarray) -> float:
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx

def _gradient_maxlag(h: np.ndarray):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    lag = idx - (n - 1)
    g = np.ones(n)
    if lag == 0:
        g = 1.0 - 2.0 * h
    else:
        indices = np.arange(n)
        i1 = indices + lag
        i2 = indices - lag
        term1 = np.zeros(n)
        term2 = np.zeros(n)
        ok1 = (i1 >= 0) & (i1 < n)
        ok2 = (i2 >= 0) & (i2 < n)
        term1[ok1] = h[i1[ok1]]
        term2[ok2] = h[i2[ok2]]
        g -= term1 + term2
    return g, lag

def _rand_dirichlet(k: int, rng: np.random.Generator) -> np.ndarray:
    # Dirichlet(alpha=1) via Gamma(1,1)
    g = rng.gamma(1.0, 1.0, size=k)
    return g / g.sum()

def _sample_five_interval_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    H = n // 2          # number of ones
    Z = n - H           # number of zeros
    ones_frac = _rand_dirichlet(5, rng)
    gaps_frac = _rand_dirichlet(6, rng)

    ones_cnt = np.floor(ones_frac * H).astype(int)
    rest = H - ones_cnt.sum()
    if rest > 0:
        frac = ones_frac * H - ones_cnt
        ones_cnt[np.argsort(-frac)[:rest]] += 1

    gaps_cnt = np.floor(gaps_frac * Z).astype(int)
    rest = Z - gaps_cnt.sum()
    if rest > 0:
        frac = gaps_frac * Z - gaps_cnt
        gaps_cnt[np.argsort(-frac)[:rest]] += 1

    seq = []
    for i in range(5):
        seq.extend([0] * gaps_cnt[i])
        seq.extend([1] * ones_cnt[i])
    seq.extend([0] * gaps_cnt[5])
    h = np.array(seq, dtype=float)
    if h.size < n:
        h = np.concatenate([h, np.zeros(n - h.size)])
    elif h.size > n:
        h = h[:n]
    return h

def _hill_climb_binary(h: np.ndarray, score: float, deadline: float):
    n = h.size
    best_h = h.copy()
    best_score = score
    while time.time() < deadline:
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        cand = best_h.copy()
        cand[i], cand[j] = 0.0, 1.0
        cand_score = _evaluate_h(cand)
        if cand_score < best_score:
            best_h, best_score = cand, cand_score
            if best_score < 0.3808:
                break
    return best_h, best_score

def _gradient_descent(h: np.ndarray, target: float, deadline: float):
    step = 0.2
    best_h = h.copy()
    best_score = _evaluate_h(best_h)
    while time.time() < deadline:
        grad, _ = _gradient_maxlag(best_h)
        cand = best_h - step * grad
        cand = _project_to_constraints(cand, target)
        cand_score = _evaluate_h(cand)
        if cand_score < best_score:
            best_h, best_score = cand, cand_score
            step = min(step * 1.05, 1.0)
            if best_score < 0.3808:
                break
        else:
            step *= 0.5
            if step < 1e-5:
                # random perturbation to escape
                perturb = np.random.uniform(-0.2, 0.2, size=h.size)
                cand = best_h + perturb
                cand = np.clip(cand, 0.0, 1.0)
                cand = _project_to_constraints(cand, target)
                cand_score = _evaluate_h(cand)
                if cand_score < best_score:
                    best_h, best_score = cand, cand_score
                step = 0.2
    return best_h, best_score

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start = time.time()

    init = globals().get("initial_h_values")
    if init is not None:
        init_arr = np.asarray(init, dtype=float)
    else:
        init_arr = None

    # target resolution (at least 200 points for decent flexibility)
    n_target = max(200, init_arr.shape[0] if init_arr is not None else 70)
    target_sum = n_target / 2.0

    # Build a starting vector
    if init_arr is not None:
        if init_arr.size != n_target:
            old_x = np.linspace(0.0, 2.0, init_arr.size, endpoint=False)
            new_x = np.linspace(0.0, 2.0, n_target, endpoint=False)
            h0 = np.interp(new_x, old_x, init_arr)
        else:
            h0 = init_arr.copy()
    else:
        rng0 = np.random.default_rng(seed)
        h0 = rng0.random(n_target)

    h0 = _project_to_constraints(h0, target_sum)
    best_h = h0.copy()
    best_score = _evaluate_h(best_h)

    # Include the (projected) initial vector if it improves
    if init_arr is not None:
        init_score = _evaluate_h(h0)
        if init_score < best_score:
            best_h, best_score = h0.copy(), init_score

    # Phase 1: random five‑interval binary constructions
    now = time.time()
    remaining = budget_s - (now - start)
    if remaining > 12.0:
        rng = np.random.default_rng(seed + 1)
        phase1_deadline = start + remaining * 0.5
        while time.time() < phase1_deadline:
            cand = _sample_five_interval_binary(n_target, rng)
            cand_score = _evaluate_h(cand)
            if cand_score < best_score:
                best_h, best_score = cand, cand_score
                if best_score < 0.3808:
                    break
        now = time.time()
        remaining = budget_s - (now - start)

    # Phase 2: projected gradient descent from the best so far
    if remaining > 6.0:
        phase2_deadline = start + budget_s - 0.6
        gd_h, gd_score = _gradient_descent(best_h, target_sum, phase2_deadline)
        if gd_score < best_score:
            best_h, best_score = gd_h, gd_score

    # Phase 3: binary hill climbing on a rounded version
    now = time.time()
    remaining = budget_s - (now - start)
    if remaining > 5.0:
        k = int(round(target_sum))
        idx = np.argsort(-best_h)
        bin_h = np.zeros(n_target, dtype=float)
        bin_h[idx[:k]] = 1.0
        bin_score = _evaluate_h(bin_h)
        hc_deadline = start + budget_s - 0.2
        bin_h2, bin_score2 = _hill_climb_binary(bin_h, bin_score, hc_deadline)
        if bin_score2 < best_score:
            best_h, best_score = bin_h2, bin_score2

    final_score = _evaluate_h(best_h)
    return (best_h.tolist(), float(final_score), int(n_target))
