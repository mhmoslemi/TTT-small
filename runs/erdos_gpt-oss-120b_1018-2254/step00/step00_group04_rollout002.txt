def repair_h(h):
    n = len(h)
    target = n/2
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    # If total == target within tolerance, done.
    # If diff > 0: need to increase sum => add to entries < 1.
    # If diff < 0: need to decrease sum => subtract from entries > 0.

    # Compute indices where we can adjust.
    if abs(total - target) < 1e-9:
        return h
    diff = target - total
    # tolerance to avoid infinite loops
    max_iter = 10
    for _ in range(max_iter):
        if abs(diff) < 1e-9:
            break
        if diff > 0:
            # increase sum
            incable = np.where(h < 1 - 1e-12)[0]
            if incable.size == 0:
                break
            # amount each can take at most
            slack = 1 - h[incable]
            # distribute diff equally into incable, but not exceed slack
            add = diff / incable.size
            # limit add to slack
            add = np.minimum(add, slack)
            h[incable] += add
            total = h.sum()
            diff = target - total
        else:
            # diff < 0, need to decrease sum
            decable = np.where(h > 1e-12)[0]
            if decable.size == 0:
                break
            slack = h[decable]  # can subtract up to this amount
            sub = (-diff) / decable.size
            sub = np.minimum(sub, slack)
            h[decable] -= sub
            total = h.sum()
            diff = target - total
    # final clip again
    return np.clip(h, 0.0, 1.0)

def _repair_h(h):
    """Project h onto [0,1]^n with sum(h) = n/2."""
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    eps = 1e-12
    # iterative scaling + clipping
    for _ in range(10):
        if abs(total - target) <= eps:
            break
        if total == 0:
            break
        factor = target / total
        h = np.clip(h * factor, 0.0, 1.0)
        total = h.sum()
    # small residual correction
    diff = target - total
    if abs(diff) > eps:
        free = np.where((h > 0) & (h < 1))[0]
        if free.size:

        else:
            # pick any index and adjust (will stay within bounds because diff is tiny)
            i = np.argmin(h)
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

        # accept if better, otherwise probabilistically
        if cand_score < best_score:
            best_h, best_score = cand, cand_score
        else:
            # Metropolis criterion (lower is better)
            prob = np.exp((best_score - cand_score) / max(temp, 1e-12))
            if random.random() < prob:
                best_h = cand  # keep as current state (but not as best)

        # cooling
        temp = max(min_temp, temp * cooling)

        # early exit if target reached
        if best_score <= goal:
            break

        # occasional random restart to escape plateaus
        if random.random() < 0.001:
            rnd = np.random.rand(n)
            rnd = _repair_h(rnd)
            try:
                rnd_score = evaluate_erdos_solution(rnd)
            except Exception:
                rnd_score = float("inf")
            if rnd_score < best_score:
                best_h, best_score = rnd, rnd_score

    # final repair before returning
    best_h = _repair_h(best_h)
    return best_h, float(best_score), n
```
