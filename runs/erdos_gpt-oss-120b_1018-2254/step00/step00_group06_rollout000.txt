def project_to_bounds_and_sum(y, target, tol=1e-9, max_iter=60):
    low = np.min(y) - 1.0  # ensure sum > target
    high = np.max(y)       # ensure sum < target
    for _ in range(max_iter):
        lam = (low + high) / 2.0
        x = np.clip(y - lam, 0, 1)
        s = np.sum(x)
        if abs(s - target) <= tol:
            return x
        if s > target:
            low = lam
        else:
            high = lam
    # Return final x
    return np.clip(y - ((low + high) / 2.0), 0, 1)

import time
import numpy as np

def _compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project_to_sum_bounds(y, target, tol=1e-12, max_iter=80):
    """Project y onto the box [0,1] with sum equal to target using bisection on lambda."""
    low = np.min(y) - 1.0  # ensures y - low >= 1 ⇒ clipped to 1
    high = np.max(y)       # ensures y - high <= 0 ⇒ clipped to 0
    for _ in range(max_iter):
        lam = (low + high) * 0.5
        x = np.clip(y - lam, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            low = lam   # need larger lambda to reduce sum
        else:
            high = lam
    return np.clip(y - ((low + high) * 0.5), 0.0, 1.0)

def _random_feasible(n, target):
    y = np.random.rand(n)
    return _project_to_sum_bounds(y, target)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    # ------------------------------------------------------------------
    # initialise h
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = h0.size
    except Exception:
        n = 95
        h0 = np.full(n, 0.5, dtype=float)

    target_sum = n * 0.5
    h0 = np.clip(h0, 0.0, 1.0)
    if abs(h0.sum() - target_sum) > 1e-12:
        h0 = _project_to_sum_bounds(h0, target_sum)

    # best solution so far
    best_h = h0.copy()
    best_score = _compute_c5(best_h)

    # current state for stochastic search
    cur_h = best_h.copy()
    cur_score = best_score

    # annealing parameters
    max_temp = 0.02
    restart_prob = 0.001  # probability to restart from a fresh random point

    # main loop
    while time.time() - start_time < budget_s * 0.95:
        # occasional random restart to escape stagnation
        if np.random.rand() < restart_prob:
            cur_h = _random_feasible(n, target_sum)
            cur_score = _compute_c5(cur_h)
            continue

        # propose a move: transfer δ from i to j
        i, j = np.random.choice(n, 2, replace=False)
        delta_max = min(cur_h[i], 1.0 - cur_h[j])
        if delta_max <= 0.0:
            continue
        delta = delta_max * np.random.rand()  # uniform in [0, delta_max)

        y = cur_h.copy()
        y[i] -= delta
        y[j] += delta

        new_score = _compute_c5(y)

        # record improvement
        if new_score < best_score:
            best_score = new_score
            best_h = y.copy()

        # acceptance rule (simulated annealing)
        diff = new_score - cur_score
        elapsed = time.time() - start_time
        temp = max_temp * max(0.0, 1.0 - elapsed / (budget_s * 0.95))
        if diff < 0.0 or np.random.rand() < np.exp(-diff / (temp + 1e-14)):
            cur_h = y
            cur_score = new_score

    return best_h, best_score, n
