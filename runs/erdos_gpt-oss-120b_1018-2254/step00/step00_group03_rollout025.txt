import numpy as np
import time

def _project_to_bounds_and_sum(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project vector v onto the box [0,1]^n while enforcing sum(v)=target_sum."""
    lo, hi = -1.0, 2.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _max_overlap_and_shift(h: np.ndarray, dx: float):
    """Return (max_overlap, shift_index, full_correlation_array)."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    idx = int(np.argmax(corr))
    shift = idx - (len(h) - 1)          # integer shift corresponding to the max
    return float(corr[idx]), int(shift), corr


def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the worst overlap term w.r.t. h."""
    n = len(h)
    k = abs(shift)
    if k == 0:
        return 1.0 - 2.0 * h
    # left shift by k (h[i+k] → position i)
    left = np.concatenate([h[k:], np.zeros(k, dtype=h.dtype)])
    # right shift by k (h[i-k] → position i)
    right = np.concatenate([np.zeros(k, dtype=h.dtype), h[: n - k]])
    return 1.0 - (left + right)


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] for the Erdős minimum overlap constant C5.

    Returns
    -------
    h_values : np.ndarray
        Optimized values of h on a uniform grid.
    c5_bound : float
        The achieved upper bound C5 = max_k ∫ h(x)(1−h(x+k)) dx.
    n_points : int
        Number of discretization points (length of h_values).
    """
    np.random.seed(seed)
    start_time = time.time()

    # ----------------------------------------------------------------------
    # Initialise problem size and starting vector
    # ----------------------------------------------------------------------
    init_global = globals().get("initial_h_values")
    if init_global is not None:
        n = int(len(init_global))
        h_start = np.array(init_global, dtype=np.float64)
    else:
        n = 70
        h_start = np.full(n, 0.5, dtype=np.float64)

    target_sum = n / 2.0               # because dx = 2/n and ∫h = 1
    dx = 2.0 / n

    # Ensure feasibility of the initial vector
    h_start = _project_to_bounds_and_sum(h_start, target_sum)

    # Record the best solution seen so far
    best_h = h_start.copy()
    best_c, _, _ = _max_overlap_and_shift(best_h, dx)

    # ----------------------------------------------------------------------
    # Hyper‑parameters (light weight, all deterministic apart from RNG)
    # ----------------------------------------------------------------------
    total_iters = 0
    max_total_iters = 30000
    restarts = 8
    inner_iters_per_restart = 2000
    step0 = 0.5                         # initial step size for gradient descent
    time_cutoff = budget_s * 0.95       # leave a safety margin

    # ----------------------------------------------------------------------
    # Main optimisation loop
    # ----------------------------------------------------------------------
    for r in range(restarts):
        if time.time() - start_time > time_cutoff:
            break

        # ------------------------------------------------------------------
        # Starting point for this restart
        # ------------------------------------------------------------------
        if r == 0 and init_global is not None:
            h = h_start.copy()
        else:
            h = _project_to_bounds_and_sum(np.random.rand(n), target_sum)

        # current evaluation
        cur_c, cur_shift, _ = _max_overlap_and_shift(h, dx)
        if cur_c < best_c:
            best_c, best_h = cur_c, h.copy()

        step = step0

        # ------------------------------------------------------------------
        # Local optimisation iterations
        # ------------------------------------------------------------------
        for _ in range(inner_iters_per_restart):
            total_iters += 1
            if total_iters >= max_total_iters or time.time() - start_time > time_cutoff:
                break

            # recompute worst shift
            cur_c, cur_shift, _ = _max_overlap_and_shift(h, dx)

            # gradient of the worst term
            grad = _gradient_for_shift(h, cur_shift)

            # --------------------------------------------------------------
            # Backtracking line search on the projected gradient step
            # --------------------------------------------------------------
            improved = False
            step_try = step
            for _ in range(10):
                h_candidate = _project_to_bounds_and_sum(h - step_try * grad, target_sum)
                cand_c, cand_shift, _ = _max_overlap_and_shift(h_candidate, dx)
                if cand_c < cur_c - 1e-12:          # strict improvement
                    h = h_candidate
                    cur_c = cand_c
                    cur_shift = cand_shift
                    improved = True
                    break
                step_try *= 0.5

            # --------------------------------------------------------------
            # If gradient step made no progress, apply a small random swap
            # --------------------------------------------------------------
            if not improved:
                i, j = np.random.choice(n, 2, replace=False)
                delta = (np.random.rand() - 0.5) * 0.2    # [-0.1,0.1]
                hi = h[i] + delta
                hj = h[j] - delta
                if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                    h_tmp = h.copy()
                    h_tmp[i] = hi
                    h_tmp[j] = hj
                    cand_c, cand_shift, _ = _max_overlap_and_shift(h_tmp, dx)
                    if cand_c < cur_c:
                        h = h_tmp
                        cur_c = cand_c
                        cur_shift = cand_shift

            # keep global best
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()

        # gentle decay of step size across restarts
        step0 *= 0.9

    # Final safety projection and evaluation
    best_h = _project_to_bounds_and_sum(best_h, target_sum)
    best_c, _, _ = _max_overlap_and_shift(best_h, dx)

    return best_h, float(best_c), int(n)
