def project_to_box_sum(x, lower, upper, target):
    # Clip
    x = np.clip(x, lower, upper)
    # Compute sum diff
    diff = target - np.sum(x)
    # tolerance
    tol = 1e-9
    # iterate
    for _ in range(100):
        if abs(diff) < tol:
            break
        if diff > 0:
            # need to increase
            # indices with room to increase (< upper)
            idx = x < upper - 1e-12
            capacity = upper - x[idx]
            total_cap = np.sum(capacity)
            if total_cap == 0:
                break
            delta = diff * capacity / total_cap
            x[idx] += delta
        else:
            # need to decrease
            idx = x > lower + 1e-12
            capacity = x[idx] - lower
            total_cap = np.sum(capacity)
            if total_cap == 0:
                break
            delta = diff * capacity / total_cap
            x[idx] += delta
        # Clip again ensuring no overshoot
        x = np.clip(x, lower, upper)
        diff = target - np.sum(x)
    return x

def subgradient_optimize(h, max_iter, start_time, total_time):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for it in range(max_iter):
        # check time
        if time.time() - start_time > total_time:
            break
        # compute correlation
        conv = np.correlate(h, 1 - h, mode='full')
        l_max = int(np.argmax(conv))
        k = l_max - (n - 1)
        # gradient
        if k >= 0:
            fwd = np.concatenate((h[k:], np.zeros(k)))
            bwd = np.concatenate((np.zeros(k), h[:n - k]))
        else:
            k_abs = -k
            fwd = np.concatenate((np.zeros(k_abs), h[:n - k_abs]))
            bwd = np.concatenate((h[k_abs:], np.zeros(k_abs)))
        g = np.ones_like(h) - (fwd + bwd)
        # step size
        alpha = 0.5 * (1 - it / max_iter)  # start 0.5 decreasing
        # update
        h_new = h - alpha * g
        # project
        h_new = project_to_box_sum(h_new, 0.0, 1.0, n/2)
        # evaluate
        c5_new = compute_c5(h_new)
        # accept if improved
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            # maybe accept with probability
            # temperature decays
            temp = 0.01 * (1 - it / max_iter)
            if temp > 0 and np.random.rand() < np.exp(-(c5_new - best_c5) / temp):
                h = h_new
        # optional: occasionally random perturb to escape plateau:
        if it % 500 == 0 and np.random.rand() < 0.1:
            # small random noise
            noise = np.random.normal(scale=0.02, size=n)
            h = project_to_box_sum(h + noise, 0.0, 1.0, n/2)
            # maybe update best if improved
            c5_noi = compute_c5(h)
            if c5_noi < best_c5:
                best_c5 = c5_noi
                best_h = h.copy()
    return best_h, best_c5

import time
import numpy as np

def compute_c5(h):
    """Return the overlap constant C5 for a given step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_to_box_sum(x, low, high, target, max_iter=100):
    """
    Euclidean projection of x onto the set {y : low <= y <= high, sum(y)=target}.
    """
    x = np.clip(x, low, high)
    for _ in range(max_iter):
        diff = target - x.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = x < high - 1e-12
            if not np.any(mask):
                break
            cap = high - x[mask]
            total = cap.sum()
            if total == 0:
                break
            delta = diff * cap / total
            x[mask] += delta
        else:
            mask = x > low + 1e-12
            if not np.any(mask):
                break
            cap = x[mask] - low
            total = cap.sum()
            if total == 0:
                break
            delta = diff * cap / total
            x[mask] += delta
        x = np.clip(x, low, high)
    return x

def subgradient_optimize(h, max_iter, rng, start_time, end_time):
    """Sub‑gradient descent on the max‑overlap objective."""
    n = h.shape[0]
    best_h = h.copy()
    best_val = compute_c5(best_h)

    for it in range(max_iter):
        if time.time() >= end_time:
            break

        # correlation and the most offending shift
        conv = np.correlate(h, 1.0 - h, mode='full')
        lmax = int(np.argmax(conv))
        k = lmax - (n - 1)          # shift corresponding to the max

        # build shifted copies (zero padding outside bounds)
        if k >= 0:
            fwd = np.concatenate((h[k:], np.zeros(k)))
            bwd = np.concatenate((np.zeros(k), h[:n - k]))
        else:
            ka = -k
            fwd = np.concatenate((np.zeros(ka), h[:n - ka]))
            bwd = np.concatenate((h[ka:], np.zeros(ka)))

        # sub‑gradient of the active C_k
        g = 1.0 - (fwd + bwd)

        # simple diminishing step size
        alpha = 0.5 * (1.0 - it / max_iter)

        h_new = h - alpha * g
        h_new = project_to_box_sum(h_new, 0.0, 1.0, n / 2.0)

        val_new = compute_c5(h_new)

        # accept if improvement, otherwise Metropolis‑like acceptance
        if val_new < best_val - 1e-12:
            best_val = val_new
            best_h = h_new.copy()
            h = h_new
        else:
            temp = 0.01 * (1.0 - it / max_iter)
            if temp > 0 and rng.random() < np.exp(-(val_new - best_val) / temp):
                h = h_new

        # occasional random jitter to escape plateaus
        if it % 500 == 0 and rng.random() < 0.05:
            h = np.clip(h + rng.normal(scale=0.02, size=n), 0.0, 1.0)
            h = project_to_box_sum(h, 0.0, 1.0, n / 2.0)

    return best_h, best_val

def refine_h(h, current_best, iters, rng):
    """Simple pairwise mass‑transfer hill climbing."""
    n = h.shape[0]
    best_h = h.copy()
    best_val = current_best

    for _ in range(iters):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # move mass from i to j (respect bounds)
        max_dec = best_h[i] - 0.0
        max_inc = 1.0 - best_h[j]
        max_move = min(max_dec, max_inc)
        if max_move <= 0:
            continue
        delta = rng.random() * max_move
        if delta == 0:
            continue

        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta

        val_new = compute_c5(h_new)
        if val_new < best_val - 1e-12:
            best_val = val_new
            best_h = h_new

    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to lower the Erdős C5 constant.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise h
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n = 70
        h0 = rng.random(n)

    h0 = project_to_box_sum(h0, 0.0, 1.0, n / 2.0)
    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # ------------------------------------------------------------------
    # time budget handling (leave a small safety margin)
    total_deadline = start_time + budget_s * 0.95

    # a single sub‑gradient run (allow many iterations, internal time cut‑off)
    sub_max_iter = 20000
    h_opt, val_opt = subgradient_optimize(best_h, sub_max_iter, rng,
                                          start_time, total_deadline)
    if val_opt < best_val:
        best_h, best_val = h_opt, val_opt

    # ------------------------------------------------------------------
    # final refinement with random pairwise moves
    while time.time() < total_deadline:
        time_left = total_deadline - time.time()
        if time_left < 0.2:
            break
        # roughly 2000 moves per second; adapt to remaining time
        iters = int(min(30000, max(1000, time_left * 2000)))
        h_ref, val_ref = refine_h(best_h, best_val, iters, rng)
        if val_ref < best_val:
            best_h, best_val = h_ref, val_ref
            if best_val <= 0.3808:   # achieved target
                break
        else:
            break   # no improvement, exit early

    # final safety projection
    best_h = project_to_box_sum(best_h, 0.0, 1.0, n / 2.0)

    return best_h, best_val, n
