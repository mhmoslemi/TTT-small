    # choose n

        h_j = h[s:s+length]

            w <= h_j,

    prob.solve(solver=cp.ECOS, max_iters=5000, abstol=1e-8, reltol=1e-8, feastol=1e-8, verbose=False, warm_start=False)

import time
import numpy as np

def _compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def _project_onto_capped_simplex(v, target):
    """Euclidean projection of v onto {0<=x<=1, sum(x)=target}."""
    # Bisection on lambda for x_i = clip(v_i - lam, 0, 1)
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # final tiny correction to hit target exactly (within tolerance)
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        # distribute the small residue on free variables
        free = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        if np.any(free):
            proj[free] += diff / free.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def _smooth_min(s, beta):
    """Smooth approximation of min(s) using log-sum-exp."""
    exp_term = np.exp(-beta * s)
    sm = -np.log(np.sum(exp_term)) / beta
    prob = exp_term / exp_term.sum()
    return sm, prob

def _gradient_step(h, beta, step, target):
    """One gradient-descent step on -smooth_min of autocorrelations."""
    n = h.shape[0]
    # autocorrelation (zero-padded)
    s = np.correlate(h, h, mode='full')[n - 1:]  # length n, s[k] = sum_i h_i h_{i+k}
    _, prob = _smooth_min(s, beta)  # prob[k] = softmin weight
    grad = np.zeros_like(h)

    # k = 0 term (derivative 2*h_i)
    grad += 2.0 * prob[0] * h

    # other shifts
    for k in range(1, n):
        pk = prob[k]
        if pk == 0.0:
            continue
        # forward contribution: i from 0..n-k-1, term h[i+k]
        grad[: n - k] += pk * h[k:]
        # backward contribution: i from k..n-1, term h[i-k]
        grad[k:] += pk * h[: n - k]

    # gradient ascent on min(s) => descent on loss
    v = h - step * grad
    return _project_onto_capped_simplex(v, target)

def _initialize_h(n, target):
    """Create a feasible initial h (triangular shape, then projected)."""
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    # triangular tent: peaks at 1 in the middle
    h = np.where(x <= 1.0, x, 2.0 - x)
    h = np.clip(h, 0.0, 1.0)
    # scale to satisfy integral constraint
    h = h * (target / h.sum())
    return _project_onto_capped_simplex(h, target)

def _solve_linear_relaxation(n, target):
    """Linear program with McCormick envelopes to boost the lower bound."""
    try:
        import cvxpy as cp
    except Exception:
        return None, None

    h = cp.Variable(n, name="h")
    tmin = cp.Variable(name="tmin")
    constraints = [h >= 0, h <= 1, cp.sum(h) == target]

    # add auxiliary variables for each shift
    for shift in range(n):
        length = n - shift
        if length == 0:
            continue
        w = cp.Variable(length, name=f"w_{shift}")
        hi = h[:length]
        hj = h[shift : shift + length]
        constraints += [
            w >= 0,
            w <= hi,
            w <= hj,
            w >= hi + hj - 1,
            cp.sum(w) >= tmin,
        ]

    prob = cp.Problem(cp.Maximize(tmin), constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iters=5000, abstol=1e-9, reltol=1e-9, feastol=1e-9, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=5000, verbose=False)
        except Exception:
            return None, None

    if h.value is None:
        return None, None
    h_opt = np.array(h.value).flatten()
    h_opt = np.clip(h_opt, 0.0, 1.0)
    h_opt = _project_onto_capped_simplex(h_opt, target)
    c5_val = _compute_c5(h_opt)
    return h_opt, c5_val

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # pick a modest grid size (even, >= 100)
    n = 200
    target_sum = n / 2.0  # because ∑h_i = n/2  <=> integral =1

    # 1) try the linear-program relaxation (fast, gives a strong start)
    best_h, best_c5 = _solve_linear_relaxation(n, target_sum)
    if best_h is None:
        # fallback simple initialization
        best_h = _initialize_h(n, target_sum)
        best_c5 = _compute_c5(best_h)

    # keep a copy for possible re‑starts later
    current_h = best_h.copy()
    current_c5 = best_c5

    # 2) smooth‑min gradient ascent on the true product (min‑overlap) 
    beta = 5.0          # initial smoothness
    step = 0.25         # initial step size
    iter_count = 0
    max_iters = 5000

    while time.time() - start_time < budget_s and iter_count < max_iters:
        # gradient step
        current_h = _gradient_step(current_h, beta, step, target_sum)
        # evaluate
        cur_c5 = _compute_c5(current_h)
        if cur_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = cur_c5
            # early exit if target already reached
            if best_c5 <= 0.3808 + 1e-7:
                break
        # schedule adjustments
        if (iter_count + 1) % 500 == 0:
            beta = min(beta * 2.0, 200.0)      # sharpen the min approximation
            step = step * 0.6                  # finer steps
        iter_count += 1

    # 3) simple random pairwise perturbation refinement (greedy hill‑climb)
    #    to squeeze out any remaining improvement
    while time.time() - start_time < budget_s:

        if i == j:

        hi, hj = best_h[i], best_h[j]
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)

        delta = np.random.uniform(low, high)
        cand = best_h.copy()
        cand[i] += delta
        cand[j] -= delta
        cand = np.clip(cand, 0.0, 1.0)   # safety
        cand = _project_onto_capped_simplex(cand, target_sum)
        cand_c5 = _compute_c5(cand)

            if best_c5 <= 0.3808 + 1e-7:

    # final projection to guarantee feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_onto_capped_simplex(best_h, target_sum)

    final_c5 = _compute_c5(best_h)
    return best_h, final_c5, n
