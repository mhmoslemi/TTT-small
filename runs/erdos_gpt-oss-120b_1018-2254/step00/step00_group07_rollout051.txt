def evaluate_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    # Project onto {x: lo <= x <= hi, sum x = target_sum}
    # Use bisection to find tau such that sum(clamp(v - tau, lo, hi)) = target_sum
    lo_bound = np.min(v) - hi
    hi_bound = np.max(v) - lo
    for _ in range(max_iter):
        tau = (lo_bound + hi_bound) / 2.0
        x = np.clip(v - tau, lo, hi)
        s = x.sum()
        if s > target_sum:
            lo_bound = tau
        else:
            hi_bound = tau
    return np.clip(v - (lo_bound + hi_bound) / 2.0, lo, hi)

def random_move(h, rng, max_step=0.05):
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    # allowable move
    max_decrease = min(max_step, h[i])
    max_increase = min(max_step, 1.0 - h[j])
    # if either zero, try swapping roles
    if max_decrease == 0 or max_increase == 0:

def evaluate_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def compute_c5_and_shift(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(corr)
    idx = np.argmax(corr)
    shift = idx - (len(h) - 1)
    return max_val * dx, shift

def project_to_simplex_box(v, target, lo=0.0, hi=1.0, max_iter=60):
    lo_bound = np.min(v) - hi
    hi_bound = np.max(v) - lo
    for _ in range(max_iter):
        tau = (lo_bound + hi_bound) / 2.0
        x = np.clip(v - tau, lo, hi)
        s = x.sum()
        if s > target:
            lo_bound = tau
        else:
            hi_bound = tau
    return np.clip(v - (lo_bound + hi_bound) / 2.0, lo, hi)

def random_move(h, rng, max_step):
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    dec = min(max_step, h[i])
    inc = min(max_step, 1.0 - h[j])
    if dec <= 0 or inc <= 0:
        # try up to 5 more attempts
        for _ in range(5):
            i, j = rng.choice(n, size=2, replace=False)
            dec = min(max_step, h[i])
            inc = min(max_step, 1.0 - h[j])
            if dec > 0 and inc > 0:
                break
        else:
            return h.copy()
    delta = rng.random() * min(dec, inc)
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new

def gradient_step(h, shift, lr, target):
    n = h.shape[0]
    idx = np.arange(n)
    left = np.where(idx - shift >= 0, h[idx - shift], 0.0)
    right = np.where(idx + shift < n, h[idx + shift], 0.0)
    grad = 1.0 - left - right
    h_new = h - lr * grad
    # clip then project to ensure sum
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_to_simplex_box(h_new, target)
    return h_new

    rng = np.random.default_rng(seed)
    start = time.time()
    try:
        init = np.array(initial_h_values, dtype=float)

import time
import numpy as np

def evaluate_c5(h, dx):
    """Return max_k ∫ h(x)(1−h(x+k))dx approximated by discrete correlation."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def compute_c5_and_shift(h, dx):
    """Return (max value, integer shift) for the correlation."""
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(corr)
    idx = np.argmax(corr)
    shift = idx - (len(h) - 1)   # shift in samples, can be negative
    return max_val * dx, shift

def project_to_box_simplex(v, target, lo=0.0, hi=1.0, iters=60):
    """Project v onto {x : lo ≤ x_i ≤ hi, sum x_i = target}."""
    low = np.min(v) - hi
    high = np.max(v) - lo
    for _ in range(iters):
        tau = (low + high) / 2.0
        x = np.clip(v - tau, lo, hi)
        s = x.sum()
        if s > target:
            low = tau
        else:
            high = tau
    return np.clip(v - (low + high) / 2.0, lo, hi)

def random_move(h, rng, max_step):
    """Move a small amount of mass from one index to another, preserving sum."""
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    dec = min(max_step, h[i])
    inc = min(max_step, 1.0 - h[j])
    if dec <= 0 or inc <= 0:
        for _ in range(5):
            i, j = rng.choice(n, size=2, replace=False)
            dec = min(max_step, h[i])
            inc = min(max_step, 1.0 - h[j])
            if dec > 0 and inc > 0:
                break
        else:
            return h.copy()
    delta = rng.random() * min(dec, inc)
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new

def gradient_step(h, shift, lr, target):
    """One gradient descent step on the active shift."""
    n = h.shape[0]
    idx = np.arange(n)
    left = np.where(idx - shift >= 0, h[idx - shift], 0.0)
    right = np.where(idx + shift < n, h[idx + shift], 0.0)
    grad = 1.0 - left - right               # ∂/∂h of the active correlation
    h_new = h - lr * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_to_box_simplex(h_new, target)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # initialise h
    try:
        init = np.array(initial_h_values, dtype=float)
        n = init.size
        h = init.copy()
    except NameError:
        n = 200                     # fallback size
        h = np.full(n, 0.5, dtype=float)

    dx = 2.0 / n
    target_sum = n / 2.0
    h = project_to_box_simplex(h, target_sum)

    best_h = h.copy()
    best_val, best_shift = compute_c5_and_shift(best_h, dx)
    best_c5 = best_val

    max_step = 0.05          # max mass moved in a random step
    lr = 0.02                # learning rate for gradient step
    it = 0

    while True:
        if time.time() - start > budget_s - 0.5:
            break

        if rng.random() < 0.6:
            # gradient descent move
            cand = gradient_step(best_h, best_shift, lr, target_sum)
        else:
            # random local move
            cand = random_move(best_h, rng, max_step)
            if rng.random() < 0.02:      # occasional full restart
                cand = rng.random(n)
                cand = np.clip(cand, 0.0, 1.0)
                cand = project_to_box_simplex(cand, target_sum)

        cand_val, cand_shift = compute_c5_and_shift(cand, dx)
        it += 1

        if cand_val < best_c5:
            best_h = cand
            best_c5 = cand_val
            best_shift = cand_shift
            # tighten step sizes as we improve
            max_step = max(0.005, max_step * 0.95)
            lr = max(0.001, lr * 0.95)
        else:
            if it % 300 == 0:
                max_step = min(0.2, max_step * 1.05)
                lr = min(0.1, lr * 1.05)

    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), float(final_c5), n)
