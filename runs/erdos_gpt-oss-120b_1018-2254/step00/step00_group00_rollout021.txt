def repair_sum(h, target):
    # Clip first
    np.clip(h, 0.0, 1.0, out=h)
    cur = h.sum()
    diff = target - cur
    # Slight tolerance
    eps = 1e-10
    max_iter = 100
    it = 0
    while abs(diff) > 1e-8 and it < max_iter:
        if diff > 0:
            free = np.where(h < 1.0)[0]
            if free.size == 0:
                break
            inc = np.minimum(diff / free.size, 1.0 - h[free])
            # sum_inc may be less than diff if some saturate at 1.
            h[free] += inc
        else:
            free = np.where(h > 0.0)[0]
            if free.size == 0:
                break
            dec = np.maximum(diff / free.size, -h[free])
            h[free] += dec
        cur = h.sum()
        diff = target - cur
        it += 1
    # In case still diff remains, adjust one element directly
    if abs(diff) > 1e-8:
        # find index that can adjust without breaching bounds

            idx = np.argmax(1.0 - h)  # largest room

    cur = h.sum()

        # scale all values uniformly:

            h[:] = h * (target / cur)

    cur = h.sum()

    eps = 1e-12

import numpy as np, time

def evaluate(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def repair_sum(vec, target):
    h = np.clip(vec, 0.0, 1.0)
    cur = h.sum()
    diff = target - cur
    eps = 1e-12
    max_iter = 1000
    it = 0
    while abs(diff) > eps and it < max_iter:
        if diff > 0:
            free = np.where(h < 1.0)[0]
            if free.size == 0:
                break
            add = diff / free.size
            cap = 1.0 - h[free]
            inc = np.minimum(add, cap)
            h[free] += inc
        else:
            free = np.where(h > 0.0)[0]
            if free.size == 0:
                break
            sub = diff / free.size
            cap = -h[free]
            dec = np.maximum(sub, cap)
            h[free] += dec
        cur = h.sum()
        diff = target - cur
        it += 1
    if abs(diff) > eps:
        if diff > 0:
            idx = np.argmax(1.0 - h)
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
        else:
            idx = np.argmin(h)
            sub = max(diff, -h[idx])
            h[idx] += sub
    np.clip(h, 0.0, 1.0, out=h)
    return h

def tournament(pop, scores, size):
    idx = np.random.choice(len(pop), size, replace=False)
    best = idx[0]
    for i in idx[1:]:
        if scores[i] < scores[best]:
            best = i
    return pop[best]

def uniform_crossover(p1, p2, rate):
    mask = np.random.rand(p1.size) < rate
    child = np.where(mask, p2, p1)
    return child

def gaussian_mutation(vec, prob, sigma, target):
    mutated = vec.copy()
    mask = np.random.rand(mutated.size) < prob
    noise = np.random.randn(mutated.size) * sigma
    mutated[mask] += noise[mask]
    np.clip(mutated, 0.0, 1.0, out=mutated)
    mutated = repair_sum(mutated, target)
    return mutated

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    # Determine dimension
    if 'initial_h_values' in globals() and initial_h_values is not None:
        init = np.array(initial_h_values, dtype=float)
        n = init.size
    else:
        n = 45
        init = np.full(n, 0.5)
    target = n / 2.0
    init = repair_sum(init, target)
    # GA parameters
    POP_SIZE = 30
    MUT_RATE = 0.25
    MUT_STD = 0.07
    CROSS_RATE = 0.5
    TOURN_SIZE = 3
    # Initialize population
    pop = [init]
    for _ in range(POP_SIZE - 1):
        r = np.random.rand(n)
        pop.append(repair_sum(r, target))
    scores = np.array([evaluate(ind) for ind in pop])
    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]
    # main loop
    while time.time() - start < budget_s * 0.95:
        parent1 = tournament(pop, scores, TOURN_SIZE)
        parent2 = tournament(pop, scores, TOURN_SIZE)
        child = uniform_crossover(parent1, parent2, CROSS_RATE)
        child = repair_sum(child, target)
        child = gaussian_mutation(child, MUT_RATE, MUT_STD, target)
        child_score = evaluate(child)
        worst = int(np.argmax(scores))
        if child_score < scores[worst]:
            pop[worst] = child
            scores[worst] = child_score
            if child_score < best_score:
                best_score = child_score
                best_h = child.copy()
        if best_score <= 0.3808:
            break
    return best_h, best_score, n

import numpy as np, time

def evaluate(h):
    """Return C5 for a given step function sampled in h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)


def repair_sum(vec, target):
    """Project vec onto {0≤x≤1, sum(x)=target}."""
    h = np.clip(vec, 0.0, 1.0)
    cur = h.sum()
    diff = target - cur
    eps = 1e-12
    max_iter = 1000
    it = 0
    while abs(diff) > eps and it < max_iter:
        if diff > 0:
            free = np.where(h < 1.0)[0]
            if free.size == 0:
                break
            add = diff / free.size
            cap = 1.0 - h[free]
            inc = np.minimum(add, cap)
            h[free] += inc
        else:
            free = np.where(h > 0.0)[0]
            if free.size == 0:
                break
            sub = diff / free.size  # negative
            cap = -h[free]
            dec = np.maximum(sub, cap)   # most we can subtract without crossing 0
            h[free] += dec
        cur = h.sum()
        diff = target - cur
        it += 1
    if abs(diff) > eps:
        if diff > 0:
            idx = np.argmax(1.0 - h)
            h[idx] += min(diff, 1.0 - h[idx])
        else:
            idx = np.argmin(h)
            h[idx] += max(diff, -h[idx])
    np.clip(h, 0.0, 1.0, out=h)
    return h


def tournament(pop, scores, size):
    """Select one individual by tournament."""
    idx = np.random.choice(len(pop), size, replace=False)
    best = idx[0]
    for i in idx[1:]:
        if scores[i] < scores[best]:
            best = i
    return pop[best]


def uniform_crossover(p1, p2, rate):
    """Uniform crossover with probability `rate` of taking gene from p2."""
    mask = np.random.rand(p1.size) < rate
    return np.where(mask, p2, p1)


def gaussian_mutation(vec, prob, sigma, target):
    """Gaussian mutation on a fraction of genes, then repair."""
    mutated = vec.copy()
    mask = np.random.rand(mutated.size) < prob
    noise = np.random.randn(mutated.size) * sigma
    mutated[mask] += noise[mask]
    np.clip(mutated, 0.0, 1.0, out=mutated)
    return repair_sum(mutated, target)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    # Determine problem size and starting vector
    if 'initial_h_values' in globals() and initial_h_values is not None:
        init = np.array(initial_h_values, dtype=float)
        n = init.size
    else:
        n = 45
        init = np.full(n, 0.5)          # already satisfies the integral

    target = n / 2.0                     # sum must equal n/2
    init = repair_sum(init, target)

    # GA hyper‑parameters
    POP_SIZE = 30
    MUT_RATE = 0.25      # probability each gene mutates
    MUT_STD = 0.07       # Gaussian std. dev.
    CROSS_RATE = 0.5
    TOURN_SIZE = 3

    # Initial population
    pop = [init]
    for _ in range(POP_SIZE - 1):
        rnd = np.random.rand(n)
        pop.append(repair_sum(rnd, target))

    scores = np.array([evaluate(ind) for ind in pop])
    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]

    # Evolutionary loop
    while time.time() - start < budget_s * 0.95:
        p1 = tournament(pop, scores, TOURN_SIZE)
        p2 = tournament(pop, scores, TOURN_SIZE)

        child = uniform_crossover(p1, p2, CROSS_RATE)
        child = repair_sum(child, target)
        child = gaussian_mutation(child, MUT_RATE, MUT_STD, target)

        child_score = evaluate(child)

        worst = int(np.argmax(scores))
        if child_score < scores[worst]:
            pop[worst] = child
            scores[worst] = child_score
            if child_score < best_score:
                best_score = child_score
                best_h = child.copy()

        if best_score <= 0.3808:
            break

    return best_h, best_score, n
