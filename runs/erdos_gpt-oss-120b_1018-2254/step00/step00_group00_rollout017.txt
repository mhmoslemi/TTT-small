import numpy as np
import time

def _compute_c5(h: np.ndarray) -> float:
    """Compute C5 = max_k ∫ h(x)*(1-h(x+k)) dx using discrete correlation."""
    n = h.size
    dx = 2.0 / n
    # np.correlate with mode='full' gives sums over overlapping indices
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _make_linear_ramp(n: int) -> np.ndarray:
    """Create a feasible linear ramp h(i)=i/(n-1) with sum = n/2."""
    # np.linspace yields values from 0 to 1 inclusive
    h = np.linspace(0.0, 1.0, n, dtype=np.float64)
    # Adjust for floating‑point drift so that sum(h) == n/2 exactly
    desired = n * 0.5
    cur = h.sum()
    if abs(cur - desired) > 1e-12:
        h += (desired - cur) / n
        # Clip to stay in [0,1] (tiny adjustments will not violate bounds)
        np.clip(h, 0.0, 1.0, out=h)
    return h
