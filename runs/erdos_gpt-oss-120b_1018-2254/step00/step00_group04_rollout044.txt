    h = np.clip(h, 0.0, 1.0)

    diff = target_sum - current_sum

    it = 0

import numpy as np
import random
import time

def _repair_h(vec, target_sum):
    """Project vec onto [0,1] with sum constraint = target_sum."""
    v = np.clip(vec, 0.0, 1.0)
    cur_sum = v.sum()
    diff = target_sum - cur_sum
    # iterative water-filling style adjustment
    for _ in range(100):
        if abs(diff) < 1e-12:
            break
        free = np.where((v > 0) & (v < 1))[0]
        if free.size == 0:
            # all at bounds, distribute uniformly (will be re‑clipped)
            v += diff / v.size
            v = np.clip(v, 0.0, 1.0)
        else:
            delta = diff / free.size
            v[free] += delta
            v = np.clip(v, 0.0, 1.0)
        cur_sum = v.sum()
        diff = target_sum - cur_sum
    # final tiny correction if needed
    if abs(diff) > 1e-12:
        v += diff / v.size
        v = np.clip(v, 0.0, 1.0)
    return v

def _c5_of(h):
    """Return the C5 bound for a candidate h using the pre‑imported evaluator."""
    # evaluate_erdos_solution is provided by the harness
    return evaluate_erdos_solution(h)[0]

def _uniform_crossover(a, b):
    mask = np.random.rand(a.shape[0]) < 0.5
    return np.where(mask, a, b)

def _mutate(ind, mut_prob=0.2, sigma=0.1):
    """Add Gaussian noise to a fraction of coordinates."""
    ind = ind.copy()
    mask = np.random.rand(ind.shape[0]) < mut_prob
    noise = np.random.normal(scale=sigma, size=ind.shape[0])
    ind[mask] += noise[mask]
    return ind

def _tournament_select(pop, scores, k=5):
    """Select one individual by k‑way tournament."""
    candidates = random.sample(range(len(pop)), k)
    best = min(candidates, key=lambda i: scores[i])
    return pop[best]

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize the step function for the Erdős C5 problem.

    Returns
    -------
    h_values : np.ndarray
        Discrete values of h on the grid.
    c5_bound : float
        C5 bound achieved by h_values.
    n_points : int
        Number of grid points used.
    """
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Determine problem size and initial feasible vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        # fallback: 42 points (same as typical record)
        n_points = 42
        h0 = np.random.rand(n_points).astype(float)

    n_points = h0.shape[0]
    target_sum = n_points / 2.0          # because integral = sum * 2/n = 1

    h0 = _repair_h(h0, target_sum)

    # ------------------------------------------------------------------
    # GA parameters
    # ------------------------------------------------------------------
    pop_size = 30
    max_gens = 5000   # generous upper bound; time limit will stop earlier
    mutation_rate = 0.7
    tournament_k = 5

    # initialise population
    population = [h0]
    while len(population) < pop_size:
        cand = np.random.rand(n_points)
        cand = _repair_h(cand, target_sum)
        population.append(cand)

    best_h = None
    best_score = float('inf')

    start_time = time.time()
    generation = 0

    while generation < max_gens and (time.time() - start_time) < budget_s:
        # evaluate fitness
        scores = [_c5_of(ind) for ind in population]

        # update best
        for ind, sc in zip(population, scores):
            if sc < best_score:
                best_score = sc
                best_h = ind.copy()

        # keep elite (best individual)
        elite_idx = int(np.argmin(scores))
        elite = population[elite_idx].copy()

        # create next generation
        new_pop = [elite]  # elitism
        while len(new_pop) < pop_size:
            p1 = _tournament_select(population, scores, k=tournament_k)
            p2 = _tournament_select(population, scores, k=tournament_k)

            child = _uniform_crossover(p1, p2)

            if random.random() < mutation_rate:
                child = _mutate(child, mut_prob=0.3, sigma=0.08)

            child = _repair_h(child, target_sum)
            new_pop.append(child)

        population = new_pop
        generation += 1

    # ------------------------------------------------------------------
    # Local refinement (simple hill climbing) on the best found solution
    # ------------------------------------------------------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 0.5 and best_h is not None:
        attempts = 0
        max_attempts = 20000
        # use a modest number of attempts or run out of time
        while attempts < max_attempts and (time.time() - start_time) < budget_s:
            cand = best_h.copy()
            # perturb a random subset (≈10% of entries)
            n_mut = max(1, int(0.1 * n_points))
            idx = np.random.choice(n_points, size=n_mut, replace=False)
            cand[idx] += np.random.normal(scale=0.05, size=n_mut)
            cand = _repair_h(cand, target_sum)
            cand_score = _c5_of(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_h = cand
            attempts += 1

    # Final projection to ensure feasibility and compute final bound
    best_h = _repair_h(best_h, target_sum)
    final_c5 = _c5_of(best_h)

    return best_h, final_c5, n_points
