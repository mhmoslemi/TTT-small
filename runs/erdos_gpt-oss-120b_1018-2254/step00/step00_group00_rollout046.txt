def repair_sum_bounds(h, target):
    # Clip to [0,1]
    np.clip(h, 0.0, 1.0, out=h)
    sum_h = h.sum()
    diff = target - sum_h
    # tolerance
    eps = 1e-12
    if abs(diff) < eps:
        return h
    # Use simple greedy adjust
    if diff > 0:
        # Need to add mass
        while diff > 1e-12:
            # indices that can increase
            free = np.where(h < 1 - 1e-12)[0]
            if free.size == 0:
                break
            # compute capacities
            capacities = 1.0 - h[free]
            total_cap = capacities.sum()
            if total_cap <= diff + 1e-12:
                # fill all to 1
                h[free] = 1.0
                diff -= total_cap
            else:
                # distribute proportionally
                # Could add diff proportionally to capacities
                # add = diff * capacities / total_cap
                # but ensure not exceed 1
                add = diff * capacities / total_cap
                h[free] += add
                diff = 0.0
    else:
        # diff < 0 : need to reduce sum
        diff = -diff
        while diff > 1e-12:
            free = np.where(h > 1e-12)[0]
            if free.size == 0:
                break
            capacities = h[free]  # amount we can remove
            total_cap = capacities.sum()
            if total_cap <= diff + 1e-12:
                # reduce all to 0
                h[free] = 0.0
                diff -= total_cap
            else:
                rem = diff * capacities / total_cap
                h[free] -= rem
                diff = 0.0
    # final clipping to handle tiny negative or >1
    np.clip(h, 0.0, 1.0, out=h)
    # due to floating offsets adjust small diff by adding to a random index not at bounds
    sum_h = h.sum()
    diff = target - sum_h
    if abs(diff) > 1e-9:
        # adjust a random index that can accept
        # find any index not at bound in direction of diff
        if diff > 0:
            idx = np.where(h < 1)[0]
            if idx.size > 0:
                i = idx[0]
                h[i] = min(1.0, h[i] + diff)
        else:
            idx = np.where(h > 0)[0]
            if idx.size > 0:
                i = idx[0]
                h[i] = max(0.0, h[i] + diff)
    return h

def hill_climb_best(best_h, rng, iterations, sigma, target_sum, dx):
    best_fitness = evaluate_h(best_h, dx)
    for i in range(iterations):
        candidate = best_h.copy()
        # mutate a few entries
        idx = rng.choice(len(candidate), size=max(1, int(0.1*len(candidate))), replace=False)
        candidate[idx] += rng.normal(0, sigma, size=idx.shape[0])
        np.clip(candidate, 0, 1, out=candidate)
        candidate = repair_sum_bounds(candidate, target_sum)
        fitness = evaluate_h(candidate, dx)
        if fitness < best_fitness:
            best_fitness = fitness
            best_h = candidate

def repair_sum_bounds(h, target):
    """Clip to [0,1] and adjust to meet sum == target."""
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    eps = 1e-12
    if abs(diff) < eps:
        return h
    # Add or subtract mass greedily with proportional distribution
    if diff > 0.0:
        # need to increase total sum
        while diff > eps:
            free = np.where(h < 1.0 - eps)[0]
            if free.size == 0:
                break
            caps = 1.0 - h[free]
            cap_sum = caps.sum()
            if cap_sum <= diff + eps:
                h[free] = 1.0
                diff -= cap_sum
            else:
                add = diff * caps / cap_sum
                h[free] += add
                diff = 0.0
    else:
        # need to decrease total sum
        diff = -diff
        while diff > eps:
            free = np.where(h > eps)[0]
            if free.size == 0:
                break
            caps = h[free]
            cap_sum = caps.sum()
            if cap_sum <= diff + eps:

                diff -= cap_sum
            else:
                sub = diff * caps / cap_sum
                h[free] -= sub
                diff = 0.0

    np.clip(h, 0.0, 1.0, out=h)

def evaluate_h(h, dx):
    """Return C5 bound for h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def mutate_h(h, rng, mut_prob, sigma, target):
    """Gaussian mutation with repair."""
    child = h.copy()
    mask = rng.rand(child.size) < mut_prob
    if mask.any():
        child[mask] += rng.normal(0.0, sigma, size=mask.sum())
    np.clip(child, 0.0, 1.0, out=child)
    repair_sum_bounds(child, target)
    return child

def crossover_h(p1, p2, rng, target):

def tournament_select(pop, fitness, rng, k):
    """Select one individual by tournament."""

    best_fit = fitness[best]
