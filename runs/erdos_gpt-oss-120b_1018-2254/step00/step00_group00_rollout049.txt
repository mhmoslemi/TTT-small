def tournament(pop, scores, k=3):
    # pick random k indices
    idx = np.random.choice(len(pop), k, replace=False)
    # return the index with best (lowest) score
    best_idx = idx[0]
    best_score = scores[best_idx]
    for i in idx[1:]:
        if scores[i] < best_score:
            best_score = scores[i]
            best_idx = i
    return pop[best_idx]

    s = np.sum(h)
    diff = target_sum - s

        return h

    return h

def repair_sum(v, target_sum):
    v = np.clip(v, 0.0, 1.0)
    # iterative approach
    # We'll loop up to maybe 100 times to converge
    for _ in range(10):
        s = np.sum(v)
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            slack = 1.0 - v
            slack_sum = np.sum(slack)
            if slack_sum < 1e-12:
                break
            # distribute proportionally
            v += diff * (slack / slack_sum)
        else:
            slack = v
            slack_sum = np.sum(slack)
            if slack_sum < 1e-12:
                break
            v -= (-diff) * (slack / slack_sum)
        v = np.clip(v, 0.0, 1.0)
    # final adjust for any residual diff
    s = np.sum(v)
    diff = target_sum - s
    if abs(diff) > 1e-10:
        # adjust first element within bounds
        indices = np.where((diff > 0) & (v < 1) | (diff < 0) & (v > 0))[0]
        if len(indices) > 0:
            i = indices[0]
            v[i] = np.clip(v[i] + diff, 0.0, 1.0)
    # final clip again
    v = np.clip(v, 0.0, 1.0)
    return v

import numpy as np
import time

def evaluate_vector(h, dx):
    """Return C5 bound for a candidate h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def repair_sum(v, target):
    """Project v onto [0,1] with sum == target."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(20):
        s = np.sum(v)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            slack = 1.0 - v
            slack_sum = np.sum(slack)
            if slack_sum < 1e-12:
                break
            v += diff * (slack / slack_sum)
        else:
            slack = v
            slack_sum = np.sum(slack)
            if slack_sum < 1e-12:
                break
            v -= (-diff) * (slack / slack_sum)
        v = np.clip(v, 0.0, 1.0)
    # final tiny adjustment if needed
    s = np.sum(v)
    diff = target - s
    if abs(diff) > 1e-10:
        if diff > 0:

    best = h.copy()
    best_score = evaluate_vector(best, dx)
