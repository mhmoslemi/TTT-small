import numpy as np, time, math

def _project_capped_simplex(v, S):
    """Project onto {x | 0<=x_i<=1, sum(x)=S}."""
    N = v.shape[0]
    lam_low = np.min(v) - 1.0          # forces all entries to 1
    lam_high = np.max(v)               # forces all entries to 0
    for _ in range(60):                # enough for double precision
        lam = (lam_low + lam_high) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > S:
            lam_low = lam
        else:
            lam_high = lam
    x = np.clip(v - lam, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_c5(h):
    N = h.shape[0]
    dx = 2.0 / N
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def _pgd_optimize(h0, max_iter=200, alpha=0.1, S=None):
    h = h0.copy()
    N = h.shape[0]
    if S is None:
        S = N / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        idx = int(np.argmax(corr))
        shift = idx - (N - 1)
        g = np.zeros_like(h)
        if shift >= 0:
            g[:N - shift] = 1 - h[shift:]
        else:
            g[-shift:] = 1 - h[:N + shift]
        y = h - alpha * g
        h = _project_capped_simplex(y, S)
        alpha *= 0.99
    return h, _compute_c5(h)

def _random_feasible(N):
    v = np.random.rand(N)
    return _project_capped_simplex(v, N / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    init_h = globals().get('initial_h_values')
    if init_h is not None:
        N = int(init_h.shape[0])
        h0 = _project_capped_simplex(init_h.astype(float).copy(), N / 2.0)
    else:
        N = 70
        h0 = _random_feasible(N)

    best_h, best_score = _pgd_optimize(h0, max_iter=300, alpha=0.12, S=N / 2.0)

    # Simulated annealing on top of the best found so far
    cur_h = best_h.copy()
    cur_score = best_score

    while True:
