import time
import numpy as np

def compute_c5(h: np.ndarray, dx: float) -> float:
    """Return max_{k} integral h(x)(1-h(x+k)) dx for discretised h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def random_binary_pattern(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a random 0/1/0.5 vector with sum exactly n/2."""
    half_needed = n % 2  # 1 if n is odd, else 0
    ones_needed = n // 2
    size = ones_needed + half_needed
    sel = rng.choice(n, size=size, replace=False)
    h = np.zeros(n, dtype=np.float64)
    if half_needed:
        half_idx = rng.choice(sel)
        h[half_idx] = 0.5
        ones_idx = sel[sel != half_idx]
        h[ones_idx] = 1.0
    else:
        h[sel] = 1.0
    return h

def is_prime(p: int) -> bool:
    if p < 2:
        return False
    if p % 2 == 0:
        return p == 2
    r = 3
    while r * r <= p:
        if p % r == 0:
            return False
        r += 2
    return True

def hill_climb(start_h: np.ndarray, start_val: float, dx: float,
               rng: np.random.Generator, time_limit: float):
    """Simple downhill hill‑climbing with random pairwise mass transfers."""
    n = start_h.shape[0]
    cur_h = start_h.copy()
    cur_val = start_val
    best_h = cur_h.copy()
    best_val = cur_val

    while time.time() < time_limit:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        max_inc = min(1.0 - cur_h[i], cur_h[j])   # increase i, decrease j
        max_dec = min(cur_h[i], 1.0 - cur_h[j])   # decrease i, increase j

        if max_inc <= 0 and max_dec <= 0:
            continue

        if max_inc > 0 and max_dec > 0:
            # choose direction uniformly
            if rng.random() < 0.5:
                delta = rng.random() * max_inc
            else:

        elif max_inc > 0:
            delta = rng.random() * max_inc
        else:
            delta = -rng.random() * max_dec

        cand_h = cur_h.copy()
        cand_h[i] += delta
        cand_h[j] -= delta

            cur_val = cand_val
            if cur_val < best_val:
