def project_onto_box_simplex(v, target_sum, tol=1e-9, max_iter=100):
    # ensures 0 <= x_i <= 1, sum(x) = target_sum.
    # Use bisection on lambda.
    # lower bound for lambda: maybe min(v_i - 1), because if lambda small negative large then clamped to 1.
    # upper bound: max(v_i) because if lambda big positive then clamped to 0.
    # Actually we need to find lambda such that sum_i clamp(v_i - lambda, 0, 1) = target_sum.
    # Since clamp is decreasing in lambda, we can do bisection.
    low = np.min(v - 1) - 1  # ensure low enough; set low = -np.max(v) maybe
    high = np.max(v) + 1  # ensure high enough
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        x = np.clip(v - mid, 0, 1)
        s = np.sum(x)
        if abs(s - target_sum) <= tol:
            return x
        if s > target_sum:
            low = mid
        else:
            high = mid
    # Return final
    return np.clip(v - (low+high)/2.0, 0, 1)

import time
import numpy as np
import math

def _compute_c5(h: np.ndarray) -> float:
    """Return max overlap integral C5 for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _project_onto_box_simplex(v: np.ndarray, target_sum: float, tol: float = 1e-9,
                              max_iter: int = 100) -> np.ndarray:
    """
    Project vector v onto the feasible set:
        0 <= x_i <= 1  and  sum(x) = target_sum.
    Uses a bisection on the Lagrange multiplier lambda such that
    x_i = clip(v_i - lambda, 0, 1).
    """
    low = np.min(v - 1.0) - 1.0
    high = np.max(v) + 1.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = np.sum(x)
        if abs(s - target_sum) <= tol:
            return x
        if s > target_sum:
            low = mid
        else:
            high = mid
    # Final clipping
    return np.clip(v - (low + high) / 2.0, 0.0, 1.0)


def _worst_shift_info(h: np.ndarray):
    """Return (lag, i) where the overlap contribution h[i]*(1-h[i+lag]) is maximal."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    i_max = int(np.argmax(corr))
    lag = i_max - (n - 1)

    if lag >= 0:
        i_start = 0
        i_end = n - lag
        idx = np.arange(i_start, i_end, dtype=int)
        contrib = h[idx] * (1.0 - h[idx + lag])
    else:
        i_start = -lag
        i_end = n
        idx = np.arange(i_start, i_end, dtype=int)
        contrib = h[idx] * (1.0 - h[idx + lag])

    local_max = int(np.argmax(contrib))
    i_best = int(idx[local_max])
    return int(lag), i_best


def _propose_targeted_move(h: np.ndarray, max_delta: float, rng: np.random.Generator):
    """Try to move mass from the high‑h side of the worst lag to the low‑h side."""
    lag, i = _worst_shift_info(h)
    j = i + lag
    # Ensure indices are distinct and within bounds
    if i == j or not (0 <= j < h.shape[0]):
        return None

    # Decide direction: from i (high) to j (low)
    delta_allowed = min(max_delta, h[i], 1.0 - h[j])
    if delta_allowed <= 0.0:
        # opposite direction might be possible
        delta_allowed = min(max_delta, h[j], 1.0 - h[i])
        if delta_allowed <= 0.0:
            return None
        i, j = j, i  # swap direction

    delta = rng.uniform(0.0, delta_allowed)
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h


def _propose_random_move(h: np.ndarray, max_delta: float, rng: np.random.Generator):
    """Randomly transfer a small amount of mass between two distinct coordinates."""
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)

    # Randomly decide which way to transfer
    if rng.random() < 0.5:
        delta_allowed = min(max_delta, h[i], 1.0 - h[j])
        if delta_allowed <= 0.0:
            delta_allowed = min(max_delta, h[j], 1.0 - h[i])
            if delta_allowed <= 0.0:
                return None
            i, j = j, i
    else:
        delta_allowed = min(max_delta, h[j], 1.0 - h[i])
        if delta_allowed <= 0.0:
            delta_allowed = min(max_delta, h[i], 1.0 - h[j])
            if delta_allowed <= 0.0:
                return None
            i, j = j, i

    delta = rng.uniform(0.0, delta_allowed)
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h


def _local_search(start_h: np.ndarray, time_budget: float, rng: np.random.Generator):
    """Hill‑climbing / simulated‑annealing on a feasible h."""
    n = start_h.shape[0]
    target_sum = n / 2.0
    if abs(np.sum(start_h) - target_sum) > 1e-9:
        start_h = _project_onto_box_simplex(start_h, target_sum)

    cur_h = start_h.copy()
    cur_c5 = _compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    max_delta = 0.2          # initial magnitude of a move
    temperature = 0.001      # annealing temperature
    decay = 0.9995           # exponential decay per iteration

    start = time.time()
    it = 0
    while time.time() - start < time_budget:
        # adapt step size gradually
        if it % 1000 == 0 and max_delta > 0.01:
            max_delta *= 0.9

        # generate a candidate
        if rng.random() < 0.65:
            cand = _propose_targeted_move(cur_h, max_delta, rng)
            if cand is None:
                cand = _propose_random_move(cur_h, max_delta, rng)
        else:
            cand = _propose_random_move(cur_h, max_delta, rng)

        if cand is None:
            continue

        cand_c5 = _compute_c5(cand)

        # accept if better or with annealing probability
        if cand_c5 <= cur_c5 or rng.random() < math.exp(-(cand_c5 - cur_c5) / max(temperature, 1e-12)):
            cur_h = cand
            cur_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = cand.copy()

        temperature = max(temperature * decay, 1e-9)
        it += 1

    return best_h, best_c5
