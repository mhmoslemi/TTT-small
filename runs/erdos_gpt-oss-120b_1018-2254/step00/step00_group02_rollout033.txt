def project_onto_box_sum(x, S, tol=1e-9, max_iter=60):
    lo = -1.0
    hi = 2.0
    # Expand bounds if needed
    # ensure lo low enough to make sum > S, hi high enough to make sum < S
    while np.clip(x - lo, 0, 1).sum() > S:
        lo *= 2
    while np.clip(x - hi, 0, 1).sum() < S:
        hi *= 2
    for _ in range(max_iter):

        proj = np.clip(x - lam, 0, 1)
        s = proj.sum()
        if s > S:
            lo = lam
        else:
            hi = lam
    return proj
