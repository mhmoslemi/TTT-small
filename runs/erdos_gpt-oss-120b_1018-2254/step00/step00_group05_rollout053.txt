def hill_climb(N, max_time, seed):
    start = time.time()
    # initialize binary vector balanced
    h = np.zeros(N, dtype=np.int8)
    indices = np.arange(N)
    ones = np.random.choice(N, N//2, replace=False)
    h[ones] = 1
    # maintain lists
    ones_list = list(ones)
    zeros_list = list(set(range(N)) - set(ones))
    best_h = h.copy()
    best_c5 = compute_c5_vec(h)
    current_c5 = best_c5
    # parameters
    iter = 0
    while time.time() - start < max_time:
        # random swap
        i_idx = random.randrange(N//2)  # len(ones_list) = N/2
        j_idx = random.randrange(N//2)  # length zerosList also N/2
        i = ones_list[i_idx]
        j = zeros_list[j_idx]
        # swap
        h[i] = 0
        h[j] = 1
        # update lists temporarily
        ones_list[i_idx] = j
        zeros_list[j_idx] = i
        new_c5 = compute_c5_vec(h)
        if new_c5 <= current_c5:
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert
            h[i] = 1
            h[j] = 0
            ones_list[i_idx] = i
            zeros_list[j_idx] = j
        iter += 1
        # optional restart if no improvement for many iterations
        if iter % 10000 == 0:
            # if current_c5 none better for a while may restart
            # We can random restart with probability
            if random.random() < 0.1:
                # random new
                h[:] = 0
                new_ones = np.random.choice(N, N//2, replace=False)
                h[new_ones] = 1
                ones_list = list(new_ones)
                zeros_list = [i for i in range(N) if i not in ones_list]
                current_c5 = compute_c5_vec(h)
    return best_h, best_c5

import numpy as np
import random
import time
import math

def compute_c5(vec: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1−h(x+k))dx for the discrete vector."""
    N = vec.shape[0]
    corr = np.correlate(vec, 1 - vec, mode="full")
    return float(np.max(corr) * (2.0 / N))


def hill_climb_binary(N: int, init_vec: np.ndarray, time_limit: float):
    """Binary hill‑climbing with occasional random restarts."""
    # ensure exactly N/2 ones
    h = init_vec.astype(np.int8).copy()
    target_ones = N // 2
    cur = int(h.sum())
    if cur != target_ones:
        if cur > target_ones:
            excess = cur - target_ones
            idx = np.where(h == 1)[0]
            turn = np.random.choice(idx, excess, replace=False)
            h[turn] = 0
        else:
            deficit = target_ones - cur
            idx = np.where(h == 0)[0]
            turn = np.random.choice(idx, deficit, replace=False)
            h[turn] = 1

    ones_list = list(np.flatnonzero(h))
    zeros_list = list(np.flatnonzero(1 - h))

    best_h = h.copy()
    best_score = compute_c5(h)
    current_score = best_score

    start = time.time()
    accept_worse = 0.005   # probability to accept a worse move
    restart_prob = 0.0003 # probability to restart from scratch each step
    target_c5 = 0.3808

    while time.time() - start < time_limit:
        # propose swapping a random 1 with a random 0
        i_idx = random.randrange(target_ones)
        j_idx = random.randrange(target_ones)
        i = ones_list[i_idx]
        j = zeros_list[j_idx]

        # perform swap
        h[i] = 0
        h[j] = 1
        ones_list[i_idx] = j
        zeros_list[j_idx] = i

        new_score = compute_c5(h)

        # accept if not worse or with small probability
        if new_score <= current_score or random.random() < accept_worse:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1
            h[j] = 0
            ones_list[i_idx] = i
            zeros_list[j_idx] = j

        # occasional full random restart
        if random.random() < restart_prob:
            h[:] = 0
            new_ones = np.random.choice(N, target_ones, replace=False)
            h[new_ones] = 1
            ones_list = list(new_ones)
            zeros_set = set(range(N)) - set(new_ones)
            zeros_list = list(zeros_set)
            current_score = compute_c5(h)

        # early exit if target reached
        if best_score <= target_c5:
            break

    return best_h.astype(float), best_score


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return (h_values, c5_bound, n_points) within the given time budget."""
    random.seed(seed)
    np.random.seed(seed)

    # Choose initial vector and size
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, copy=False)
        N = init.shape[0]
        # start from a binary version of the parent construction
        init_bin = (init >= 0.5).astype(np.int8)
    else:
        N = 100
        if N % 2:
            N += 1
        init_bin = np.zeros(N, dtype=np.int8)
        init_bin[np.random.choice(N, N // 2, replace=False)] = 1

    # allocate remaining time for the optimizer
    start = time.time()
    remaining = max(0.0, budget_s - (time.time() - start))

    best_h, best_c5 = hill_climb_binary(N, init_bin, remaining)
