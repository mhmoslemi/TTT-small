def evaluate_erdos(h):
    """Discrete evaluation of C5 for a step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)


def enforce_sum_and_bounds(vec, target_sum):

    v = np.clip(vec, 0.0, 1.0)

            i = idx[0]
            dec = min(-diff, v[i])
            v[i] -= dec
        cur = np.sum(v)
        diff = target_sum - cur
    return v
