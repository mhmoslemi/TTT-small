def hill_climb(init_h, max_iters=10000):
    h = init_h.copy()
    current_c5 = compute_c5(h)
    # generate one indices etc.
    one_idx = np.where(h > 0.5)[0]
    zero_idx = np.where(h < 0.5)[0]
    no_improve = 0
    for it in range(max_iters):
        # propose random swap of one and zero
        i = np.random.choice(one_idx)
        j = np.random.choice(zero_idx)
        # swap using direct modifications on array (no copy) and recompute?

        conv = np.convolve(arr, (1 - arr)[::-1])
        return np.max(conv) * (2.0 / len(arr))
    # initialize best
    best_h = np.full(n_points, 0.5, dtype=float)  # placeholder
    best_c5 = float('inf')
    # optionally evaluate whether initial_h_values could be used
    # Not using it.

    max_no_improve = 500

        # initialize random binary vector
        perm = np.arange(n_points)
        np.random.shuffle(perm)
        h = np.zeros(n_points, dtype=float)
        h[perm[:target_sum]] = 1.0
        current_c5 = compute_c5(h)
        # track local best improvement counter
        no_improve = 0
        # local search loop
        while time.time() - start < budget_s * 0.95 and no_improve < max_no_improve:
            # propose swap

            j = np.random.randint(n_points)

                j = np.random.randint(n_points)
            # swap
            h[i] = 0.0
            h[j] = 1.0

            # compute temperature

import numpy as np
import time
import math

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] with values in [0,1],
    integral 1, that minimizes the overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    # Choose problem size (even for binary representation)
    n_points = 48                     # can be changed, must be <= 1000
    target_sum = n_points // 2        # exactly half of the entries are 1

    # Objective: max overlap integral C5
    def compute_c5(arr):
        # convolution of h with reversed (1-h) gives the full correlation
        conv = np.convolve(arr, (1.0 - arr)[::-1])
        return np.max(conv) * (2.0 / len(arr))

    # Global best record
    best_h = np.full(n_points, 0.5, dtype=float)
    best_c5 = float('inf')
    target_c5 = 0.3808                # stop early if we beat this bound
    max_no_improve = 500             # terminate local search after this many rejects
    found_target = False

    # Main search loop (multiple random restarts)
    while not found_target and (time.time() - start_time) < budget_s * 0.95:
        # Random binary start with exactly target_sum ones
        perm = np.arange(n_points)
        np.random.shuffle(perm)
        h = np.zeros(n_points, dtype=float)
        h[perm[:target_sum]] = 1.0
        cur_c5 = compute_c5(h)

        # Update global best if this start is already good
        if cur_c5 < best_c5:
            best_c5, best_h = cur_c5, h.copy()
            if best_c5 < target_c5:
                found_target = True
                break

        no_improve = 0
        # Local stochastic search (simulated annealing style)
        while not found_target and no_improve < max_no_improve:
            # Pick a random 1-index and a random 0-index to swap
            i = np.random.randint(n_points)
            while h[i] < 0.5:
                i = np.random.randint(n_points)
            j = np.random.randint(n_points)
            while h[j] > 0.5:
                j = np.random.randint(n_points)

            # Perform the swap
            h[i] = 0.0
            h[j] = 1.0

            new_c5 = compute_c5(h)

            # Linear cooling temperature (starts at ~0.02, ends near 0)
            elapsed = time.time() - start_time
            remaining = max(0.0, budget_s - elapsed)
            T = 0.02 * (remaining / budget_s)

            # Acceptance test
            accept = False
            if new_c5 < cur_c5:
                accept = True
            elif T > 1e-12:
                prob = math.exp(-(new_c5 - cur_c5) / T)
                if np.random.rand() < prob:
                    accept = True

            if accept:
                cur_c5 = new_c5
                no_improve = 0
                if cur_c5 < best_c5:
                    best_c5, best_h = cur_c5, h.copy()
                    if best_c5 < target_c5:
                        found_target = True
                        break
            else:
                # revert the swap
                h[i] = 1.0
                h[j] = 0.0
                no_improve += 1

            # Safety check for time budget
            if (time.time() - start_time) >= budget_s * 0.95:
                break

    # Ensure feasibility of the best solution (clip & repair sum if needed)
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_h = best_h.sum()
    if abs(sum_h - target_sum) > 1e-8:
        diff = target_sum - sum_h
        # Adjust the first non‑boundary entry
        for idx in range(n_points):
            if (diff > 0 and best_h[idx] < 1.0) or (diff < 0 and best_h[idx] > 0.0):
                best_h[idx] += diff
                break
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, n_points)
