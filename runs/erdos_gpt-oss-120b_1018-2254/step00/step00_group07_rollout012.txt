import time
import random
from scipy.optimize import minimize

def _project_to_simplex_bound(v, target, lower=0.0, upper=1.0, tol=1e-9, max_iter=1000):
    w = np.clip(v, lower, upper)
    total = np.sum(w)
    for _ in range(max_iter):
        diff = total - target
        if abs(diff) <= tol:
            break
        free = np.where((w > lower + 1e-12) & (w < upper - 1e-12))[0]
        if free.size == 0:
            # if no free variable, cannot adjust; break
            break
        delta = diff / free.size
        w[free] -= delta
        w = np.clip(w, lower, upper)
        total = np.sum(w)
    return w

def run(seed=42, budget_s=1100, **kwargs):

        n = init_h.size
    else:
        n = 80
        init_h = None
    target_sum = n / 2.0
    dx = 2.0 / n

    def obj(h):

        h = np.asarray(h, dtype=float)

        return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

    # constraints for optimizer

    bounds = [(0.0, 1.0)] * n

    best_score = np.inf

        score = obj(candidate)
        if score < best_score:
            best_score = score
            best_h = candidate.copy()

def _project_to_simplex_bound(v, target, lower=0.0, upper=1.0, tol=1e-9, max_iter=1000):
    """Project vector v onto the set {x: lower<=x_i<=upper, sum(x)=target}."""
    w = np.clip(v, lower, upper)
    total = np.sum(w)
    for _ in range(max_iter):
        diff = total - target
        if abs(diff) <= tol:
            break
        free = np.where((w > lower + 1e-12) & (w < upper - 1e-12))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        w[free] -= delta
        w = np.clip(w, lower, upper)
        total = np.sum(w)
    return w
