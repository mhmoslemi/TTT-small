def targeted_move(h, shift, rng, step_frac=0.5):
    n = len(h)
    start = max(0, -shift)
    end = min(n, n - shift)
    if start >= end:
        return h
    idx_range = np.arange(start, end)
    # contributions
    i_candidates = idx_range
    j_candidates = i_candidates + shift
    contributions = h[i_candidates] * (1.0 - h[j_candidates])
    i_max = i_candidates[np.argmax(contributions)]
    j = i_max + shift
    # propose delta
    max_delta = min(h[i_max], 1.0 - h[j])
    if max_delta <= 0:
        # try opposite direction if possible
        max_delta2 = min(h[j], 1.0 - h[i_max])
        if max_delta2 <= 0:
            return h
        delta = rng.random() * max_delta2
        new_h = h.copy()
        new_h[i_max] += delta
        new_h[j] -= delta
    else:
        # Choose delta = step_frac * max_delta (or random fraction)
        delta = step_frac * max_delta * rng.random()  # random between 0 and step_frac*max_delta
        if delta <= 0:
            return h
        new_h = h.copy()
        new_h[i_max] -= delta
        new_h[j] += delta
    return new_h

def project_to_sum_and_bounds(v, target):
    # Input: v as np.ndarray dtype float
    # Output: x satisfying 0<=x<=1 and sum(x)=target (within tolerance)
    lo = -2.0
    hi = 2.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def random_move(h, rng):
    n = len(h)
    # Choose i, j distinct
    i = rng.integers(0, n)
    j = rng.integers(0, n-1)
    if j >= i:
        j += 1
    # Determine direction: move from i to j or j to i
    if rng.random() < 0.5:
        # move from i to j
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            return h
        delta = rng.random() * max_delta
        new = h.copy()
        new[i] -= delta
        new[j] += delta

        if max_delta <= 0:
            return h
        delta = rng.random() * max_delta
        new = h.copy()
        new[i] += delta
        new[j] -= delta

def targeted_move(h, shift, rng, frac=0.5):
    n = len(h)
    start = max(0, -shift)
    end = min(n, n - shift)
    if start >= end:
        return h
    i_range = np.arange(start, end)
    j_range = i_range + shift
    contributions = h[i_range] * (1.0 - h[j_range])
    max_idx = np.argmax(contributions)
    i = i_range[max_idx]
    j = j_range[max_idx]
    # Move from i to j (reduce contribution)
    max_delta = min(h[i], 1.0 - h[j])
    if max_delta <= 0:
        # try opposite direction
        max_delta2 = min(h[j], 1.0 - h[i])
        if max_delta2 <= 0:
            return h
        delta = rng.random() * max_delta2 * frac
        new = h.copy()
        new[i] += delta
        new[j] -= delta
        return new
    delta = rng.random() * max_delta * frac
    new = h.copy()
    new[i] -= delta
    new[j] += delta
    return new

import time
import numpy as np

def _project_to_sum_and_bounds(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the hypercube [0,1]^n with sum(v)=target using bisection."""
    lo, hi = -2.0, 2.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _compute_c5(h: np.ndarray, dx: float) -> float:
    """Return max_k ∫ h(x)(1‑h(x+k))dx approximated by discrete correlation."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _random_move(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Transfer a random amount of mass between two distinct indices."""
    n = h.shape[0]
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if j >= i:
        j += 1
    # decide direction
    if rng.random() < 0.5:
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            return h
        delta = rng.random() * max_delta
        new = h.copy()
        new[i] -= delta
        new[j] += delta
    else:
        max_delta = min(h[j], 1.0 - h[i])
        if max_delta <= 0.0:
            return h
        delta = rng.random() * max_delta
        new = h.copy()
        new[i] += delta
        new[j] -= delta
    return new


def _targeted_move(h: np.ndarray, shift: int, rng: np.random.Generator, frac: float = 0.5) -> np.ndarray:
    """Move mass from the index contributing most to the worst shift."""
    n = h.shape[0]
    start = max(0, -shift)
    end = min(n, n - shift)
    if start >= end:
        return h
    i_range = np.arange(start, end, dtype=int)
    j_range = i_range + shift
    contrib = h[i_range] * (1.0 - h[j_range])
    idx = np.argmax(contrib)
    i = i_range[idx]
    j = j_range[idx]

    max_delta = min(h[i], 1.0 - h[j])
    if max_delta > 0.0:
        delta = rng.random() * max_delta * frac
        new = h.copy()
        new[i] -= delta
        new[j] += delta
        return new

    # try the opposite direction
    max_delta = min(h[j], 1.0 - h[i])
    if max_delta > 0.0:
        delta = rng.random() * max_delta * frac
        new = h.copy()
        new[i] += delta
        new[j] -= delta
        return new
    return h


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a step function h on [0,2] with ∫h=1 minimizing C₅."""
    rng = np.random.default_rng(seed)

    # Determine problem size
    try:
        init_arr = np.asarray(initial_h_values, dtype=float)
        n = init_arr.shape[0]
    except NameError:
        n = 80
        init_arr = None

    target_sum = n / 2.0
    dx = 2.0 / n

    # Starting point
    if init_arr is not None:
        h0 = np.clip(init_arr, 0.0, 1.0)
        h0 = _project_to_sum_and_bounds(h0, target_sum)
    else:
        h0 = _project_to_sum_and_bounds(rng.random(n), target_sum)

    best_h = h0.copy()
    best_c5 = _compute_c5(best_h, dx)

    cur_h = best_h.copy()
    cur_c5 = best_c5
    no_improve = 0

    start_time = time.time()
    time_limit = budget_s * 0.95

    while True:
        now = time.time()
        if now - start_time >= time_limit:
            break

        # pick move type
        if rng.random() < 0.85:
            # compute current worst shift
            corr_full = np.correlate(cur_h, 1.0 - cur_h, mode="full")
            max_idx = int(np.argmax(corr_full))
            shift = max_idx - (n - 1)
            if shift != 0:
                new_h = _targeted_move(cur_h, shift, rng, frac=0.6)
            else:
                new_h = _random_move(cur_h, rng)
        else:
            new_h = _random_move(cur_h, rng)

        new_c5 = _compute_c5(new_h, dx)

        if new_c5 < cur_c5:
            cur_h = new_h
            cur_c5 = new_c5
            no_improve = 0
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
        else:
            no_improve += 1
            # occasional uphill move (tiny probability)
            if rng.random() < 0.001:
                cur_h = new_h
                cur_c5 = new_c5

        if no_improve >= 5000:
            # restart from a fresh random feasible point
            cur_h = _project_to_sum_and_bounds(rng.random(n), target_sum)
            cur_c5 = _compute_c5(cur_h, dx)
            no_improve = 0

    return best_h, float(best_c5), int(n)
