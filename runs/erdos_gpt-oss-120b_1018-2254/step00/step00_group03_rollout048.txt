def _compute_c5(arr: np.ndarray) -> float:
    """Return the max overlap integral for a given discretized h."""
    conv = np.correlate(arr, 1.0 - arr, mode="full")
    return float(np.max(conv) * (2.0 / arr.size))

def _random_binary_feasible(rng: np.random.Generator, n: int) -> np.ndarray:
    """Return a binary feasible vector (half ones, half zeros)."""
    h = np.zeros(n, dtype=float)
    half = n // 2
    idx = rng.choice(n, size=half, replace=False)
    h[idx] = 1.0
    # If n is odd, give the remaining mass 0.5 to keep the integral 1.
    if n % 2:
        rest = (n / 2.0) - half
        if rest > 0:
            remaining = np.setdiff1d(np.arange(n), idx)
            h[rng.choice(remaining)] = rest
    return h
