def generate_random_feasible_vector(n, rng):
    target = n / 2.0
    h = np.zeros(n)
    remaining = target
    # For each iteration, assign random increments up to capacity 1
    # We'll iterate over indices randomly.
    # While remaining > 0:
    while remaining > 1e-12:
        # random index where h < 1
        free_idx = np.where(h < 1 - 1e-12)[0]
        if free_idx.size == 0:
            break
        i = rng.choice(free_idx)
        max_add = min(1 - h[i], remaining)
        # add random portion
        add = rng.random() * max_add
        h[i] += add
        remaining -= add
    # after loop, due to rounding may have left small remainder; assign to random free index
    if remaining > 1e-12:
        # try assign to any index with room
        free_idx = np.where(h < 1 - 1e-12)[0]
        if free_idx.size > 0:
            i = rng.choice(free_idx)
            h[i] = np.clip(h[i] + remaining, 0, 1)
        else:
            # fallback: adjust first element
            h[0] = np.clip(h[0] + remaining, 0, 1)
    # Ensure final sum within tolerance
    h = project_to_feasible(h) # to clean any rounding.
    return h

import numpy as np
import time

def compute_c5(h, dx):
    """Maximum overlap integral for a discrete step function."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_feasible(v):
    """Project vector onto box [0,1] and sum(v)=n/2."""
    n = v.shape[0]
    target = n / 2.0
    v = np.clip(v, 0.0, 1.0)
    for _ in range(100):
        s = v.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(v < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            inc = diff / idx.size
            v[idx] += inc
            v = np.clip(v, 0.0, 1.0)
        else:
            idx = np.where(v > 1e-12)[0]
            if idx.size == 0:
                break
            dec = diff / idx.size
            v[idx] += dec
            v = np.clip(v, 0.0, 1.0)
    # final tweak for residual error
    diff = target - v.sum()
    if abs(diff) > 1e-12:
        free = np.where((v > 1e-12) & (v < 1.0 - 1e-12))[0]
        if free.size > 0:
            v[free[0]] = np.clip(v[free[0]] + diff, 0.0, 1.0)
        else:
            v[0] = np.clip(v[0] + diff, 0.0, 1.0)
    return v

def generate_random_feasible_vector(n, rng):
    """Random feasible vector using Dirichlet and projection."""
    target = n / 2.0
    v = rng.dirichlet(np.ones(n)) * target
    return project_to_feasible(v)

def mutate_vector(h, rng):
    """Gaussian mutation with projection."""
    n = h.shape[0]
    prob = 0.20        # per-coordinate mutation probability
    sigma = 0.07       # mutation scale
    mask = rng.random(n) < prob
    if mask.any():
        h = h.copy()
        h[mask] += rng.normal(0.0, sigma, size=mask.sum())
    return project_to_feasible(h)
