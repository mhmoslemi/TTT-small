import time
import numpy as np

def _compute_c5(h):
    """Return the Erdős overlap constant for a step function h."""
    n = h.size
    dx = 2.0 / n
    # correlation h with (1-h); mode='full' gives all integer shifts
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def _binary_initial(h_source, n):
    """Create a binary vector of length n with exactly n/2 ones.
    Uses the largest entries of h_source (if provided) as a heuristic."""
    target = n // 2
    if h_source is not None and h_source.size == n:
        # choose the entries with largest values
        idx = np.argsort(-h_source)
        bin_h = np.zeros(n, dtype=float)
        bin_h[idx[:target]] = 1.0
        return bin_h
    # otherwise random binary initialization
    bin_h = np.zeros(n, dtype=float)
    bin_h[:target] = 1.0
    np.random.shuffle(bin_h)
    return bin_h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()

    # Determine problem size
    try:
        init_vals = initial_h_values  # provided by harness when available
        n_points = int(init_vals.shape[0])
    except NameError:
        init_vals = None
        n_points = 200  # default size if no parent construction
    # Ensure even length for exact half‑sum constraint
    if n_points % 2:
        n_points += 1

    # Build an initial binary vector respecting the sum constraint
    h_current = _binary_initial(init_vals, n_points).copy()
    c5_current = _compute_c5(h_current)

    # Keep the best seen solution
    best_h = h_current.copy()
    best_c5 = c5_current

    # Simple simulated‑annealing parameters
    T0 = 0.02                     # initial temperature
    min_temp = 1e-6               # floor temperature
    target_ones = n_points // 2

    # Main optimisation loop
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:   # leave a safety margin
            break

        # Temperature schedule (linear decay)
        T = max(min_temp, T0 * (1.0 - elapsed / budget_s))

        # Select a random 1‑position and a random 0‑position to swap
        ones = np.where(h_current > 0.5)[0]
        zeros = np.where(h_current < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break  # degenerate case, should not happen

        i = np.random.choice(ones)
        j = np.random.choice(zeros)

        # Propose a new candidate by swapping the bits
        h_candidate = h_current.copy()
        h_candidate[i] = 0.0
        h_candidate[j] = 1.0
        c5_candidate = _compute_c5(h_candidate)

        # Acceptance rule (Metropolis)
        if c5_candidate < c5_current:
            accept = True
        else:
            # probability to accept a worsening move
            delta = c5_candidate - c5_current
            accept = np.random.rand() < np.exp(-delta / T)

        if accept:
            h_current = h_candidate
            c5_current = c5_candidate
            # Update the global best if improved
            if c5_current < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_current

    # Final evaluation – use the supplied helper if present
    try:
        final_c5 = evaluate_erdos_solution(best_h)
    except NameError:
        final_c5 = _compute_c5(best_h)

    return (best_h, final_c5, n_points)
