def anneal_binary(h, total_time, start_time, best_h, best_c5):
    n = len(h)
    dx = 2.0 / n
    curr_h = h.copy()
    curr_c5 = compute_c5(curr_h)
    if curr_c5 < best_c5:
        best_c5, best_h = curr_c5, curr_h.copy()
    # Precompute indices of ones and zeros to sample efficiently
    ones = np.flatnonzero(curr_h == 1)
    zeros = np.flatnonzero(curr_h == 0)
    while time.time() - start_time < total_time:
        # Choose a random one and zero to swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        curr_h[i], curr_h[j] = 0, 1
        # compute new c5
        new_c5 = compute_c5(curr_h)
        # acceptance
        delta = new_c5 - curr_c5
        # schedule T
        elapsed = time.time() - start_time
        remaining = total_time - elapsed
        # simple linear cooling
        T = T0 * (remaining / total_time)  # start high, go low
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            # accept move
            curr_c5 = new_c5
            # update ones, zeros
            ones = np.where(curr_h == 1)[0]
            zeros = np.where(curr_h == 0)[0]
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = curr_h.copy()
        else:
            # revert swap
            curr_h[i], curr_h[j] = 1, 0
            # ones and zeros unchanged
        # iterate
    return best_h, best_c5
```

import time
import math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1−h(x+k))dx ≈ dx * max correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_total = time.time()
    best_h = None
    best_c5 = float('inf')

    # ------------------------------------------------------------------
    # Helper to obtain an initial feasible vector for a given size n.
    # ------------------------------------------------------------------

            if init.shape[0] == n:

                h = init.copy()

                target = n / 2.0
                cur = h.sum()
                if abs(cur - target) > 1e-12:
                    diff = target - cur
                    h[0] = min(1.0, max(0.0, h[0] + diff))
                return h
            if init.shape[0] == n // 2 and n % init.shape[0] == 0:
                # repeat pattern (e.g. 45 → 90)
                factor = n // init.shape[0]
                h = np.tile(init, factor)

                return h
            # otherwise fall back to interpolation

        # No parent construction: make a simple feasible vector.
