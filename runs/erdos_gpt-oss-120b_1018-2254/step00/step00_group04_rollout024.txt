import time, random, math
import numpy as np

def _project_solution(h, target):
    """Clip to [0,1] and adjust sum to target exactly."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Adjust entries in random order until sum matches
    idxs = np.arange(len(h))
    np.random.shuffle(idxs)
    for i in idxs:
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            cap = 1.0 - h[i]
            inc = min(cap, diff)
            h[i] += inc
            diff -= inc
        else:
            cap = h[i]
            dec = min(cap, -diff)
            h[i] -= dec
            diff += dec
    # Very small residual correction
    if abs(np.sum(h) - target) > 1e-9:
        h[0] = np.clip(h[0] + (target - np.sum(h)), 0.0, 1.0)
    return h

def _c5_and_shift(h, dx):
    """Return (C5 value, integer shift index) for array h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    return float(corr[idx] * dx), idx - (len(h) - 1)
