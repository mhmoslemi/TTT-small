import numpy as np
import time
import math

def _project_onto_bounds_and_sum(y, target_sum):
    """Project y onto [0,1] elementwise and enforce sum == target_sum."""
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)

def _compute_c5_and_shift(h):
    """Return (C5 value, shift index) for the current h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)  # lag corresponding to the max correlation
    return corr.max() * dx, shift

def _subgradient_for_shift(h, shift):
    """Subgradient of C5 w.r.t. h for a given shift (the maximising one)."""
    n = h.shape[0]
    dx = 2.0 / n
    g = np.full(n, dx)          # start from the constant term (dx)
    if shift >= 0:
        if shift < n:
            i = np.arange(n - shift)
            g[i] -= dx * h[i + shift]
            i2 = np.arange(shift, n)
            g[i2] -= dx * h[i2 - shift]
    else:
        s = -shift
        if s < n:
            i = np.arange(s, n)
            g[i] -= dx * h[i - s]
            i2 = np.arange(n - s)
            g[i2] -= dx * h[i2 + s]
    return g

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős C5 overlap bound.
    Returns a tuple (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start = time.time()
    # initialise h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_default = 95
        h = np.zeros(n_default, dtype=float)
        half = n_default // 2
        h[:half] = 1.0
        np.random.shuffle(h)
    n = h.shape[0]
    target_sum = n * 0.5
    h = _project_onto_bounds_and_sum(h, target_sum)

    # initial best
    best_h = h.copy()
    best_c5, _ = _compute_c5_and_shift(best_h)

    # optimisation parameters
    alpha0 = 0.5               # initial step size factor
    max_iters = 50000
    no_improve = 0

    for it in range(1, max_iters + 1):
        # time check
        if time.time() - start > budget_s * 0.99:
            break

        # current objective and subgradient direction
        c5, shift = _compute_c5_and_shift(h)
        g = _subgradient_for_shift(h, shift)

        # step size diminishing like 1/sqrt(it)
        step = alpha0 / math.sqrt(it)
        h_new = h - step * g

        # occasional stochastic perturbation to escape plateaus
        if np.random.rand() < 0.1:
            eps = 0.02 / math.sqrt(it)   # decreasing noise
            h_new += eps * np.random.randn(n)

        # enforce feasibility
        h_new = _project_onto_bounds_and_sum(h_new, target_sum)

        # evaluate candidate
        c5_new, _ = _compute_c5_and_shift(h_new)

        # accept if improvement (or with tiny probability to allow moves)
        if c5_new < c5 or np.random.rand() < 0.001:
            h = h_new

        # update global best
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # stagnation handling: re‑inject noise around the best solution
        if no_improve >= 2000:
            h = best_h + 0.05 * np.random.randn(n)
            h = _project_onto_bounds_and_sum(h, target_sum)
            no_improve = 0

    # final evaluation (ensure consistency)
    final_c5, _ = _compute_c5_and_shift(best_h)
    return (best_h, final_c5, n)
