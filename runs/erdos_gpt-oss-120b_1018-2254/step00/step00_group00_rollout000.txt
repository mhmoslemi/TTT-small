def evaluate_h(h, dx):
    # compute correlation and max
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_sum(v, target):
    # ensure within [0,1] and sum = target
    v = np.clip(v, 0.0, 1.0)
    current = v.sum()
    diff = target - current
    # tolerance
    max_iter = 1000
    rng = np.random.default_rng()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to increase
            idx = np.where(v < 1.0)[0]
            if len(idx)==0:
                break
            i = rng.choice(idx)
            avail = 1.0 - v[i]
            delta = min(avail, diff)
        else:
            idx = np.where(v > 0.0)[0]
            if len(idx) == 0:
                break
            i = rng.choice(idx)
            avail = v[i]
            delta = -min(avail, -diff)
        v[i] += delta
        diff = target - v.sum()
    return v

def tournament(pop, fitness, size, rng):
    # randomly select size individuals and return best (lowest fitness)
    idx = rng.choice(len(pop), size, replace=False)
    best_idx = idx[0]
    best_fit = fitness[best_idx]
    for i in idx[1:]:
        if fitness[i] < best_fit:
            best_fit = fitness[i]
            best_idx = i
    return pop[best_idx]

def crossover(p1, p2, rng):
    # arithmetic uniform crossover
    alpha = rng.random()
    child = alpha * p1 + (1 - alpha) * p2
    # ensure bounds
    child = np.clip(child, 0.0, 1.0)
    return child

def mutate(vec, sigma, prob, rng):
    child = vec.copy()
    mask = rng.random(vec.shape) < prob
    # Gaussian noise
    child[mask] += rng.normal(0.0, sigma, size=mask.sum())
    # clip to [0,1]
    child = np.clip(child, 0.0, 1.0)
    return child

def evaluate_h(h, dx):
    """Return max overlap integral C5 for step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_sum(v, target, rng):
    """Clip to [0,1] and adjust to hit the exact sum target."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(v < 1.0)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            delta = min(1.0 - v[i], diff)
        else:
            idx = np.where(v > 0.0)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            delta = -min(v[i], -diff)
        v[i] += delta
        diff = target - v.sum()
    return v

def tournament(pop, fit, size, rng):
    """Select the best individual among a random subset."""
    idx = rng.choice(len(pop), size, replace=False)
    best = idx[0]
    best_val = fit[best]
    for i in idx[1:]:
        if fit[i] < best_val:
            best = i
            best_val = fit[i]
    return pop[best]

    SIGMA = 0.07
    MUT_PROB = 0.2

    # Initialise population
    pop = []
    if init_h is not None:
        init_h = repair_sum(init_h, target_sum, rng)
        pop.append(init_h.copy())
    while len(pop) < POP_SIZE:
        v = rng.random(n_points)
        v = repair_sum(v, target_sum, rng)
        pop.append(v)
    pop = np.array(pop)

    # Evaluate initial fitness
    fitness = np.array([evaluate_h(ind, dx) for ind in pop])
    best_idx = int(np.argmin(fitness))
    best_h = pop[best_idx].copy()
    best_fit = float(fitness[best_idx])

    start_time = time.time()
    # Main evolutionary loop
    while time.time() - start_time < budget_s * 0.95:
        # Produce offspring
        children = []
        child_fit = []
        while len(children) < POP_SIZE:
            p1 = tournament(pop, fitness, TOURNAMENT_SIZE, rng)
            p2 = tournament(pop, fitness, TOURNAMENT_SIZE, rng)
            child = crossover(p1, p2, rng)
            child = mutate(child, SIGMA, MUT_PROB, rng)
            child = repair_sum(child, target_sum, rng)
            children.append(child)
            child_fit.append(evaluate_h(child, dx))
        children = np.array(children)
        child_fit = np.array(child_fit)

        # Combine and keep the best POP_SIZE individuals
        combined = np.concatenate([pop, children], axis=0)
        combined_fit = np.concatenate([fitness, child_fit], axis=0)
        keep = np.argsort(combined_fit)[:POP_SIZE]
        pop = combined[keep]
        fitness = combined_fit[keep]

        # Update global best
        cur_best = int(np.argmin(fitness))
        if fitness[cur_best] < best_fit:
            best_fit = float(fitness[cur_best])
            best_h = pop[cur_best].copy()

        # Early exit if we already beat the target bound
        if best_fit < 0.3808:
            break

        # Local refinement on the current best solution
        for _ in range(5):
            cand = best_h.copy()
            num = max(1, int(0.1 * n_points))
            idxs = rng.choice(n_points, size=num, replace=False)
            cand[idxs] += rng.normal(0.0, SIGMA, size=num)
            cand = np.clip(cand, 0.0, 1.0)
            cand = repair_sum(cand, target_sum, rng)
            cand_fit = evaluate_h(cand, dx)
            if cand_fit < best_fit:
                best_fit = cand_fit
                best_h = cand.copy()
                # Replace worst individual in the population
                worst = int(np.argmax(fitness))
                pop[worst] = cand
                fitness[worst] = cand_fit

    # Final safety repair and evaluation
    best_h = repair_sum(np.clip(best_h, 0.0, 1.0), target_sum, rng)
    best_fit = evaluate_h(best_h, dx)
    return (best_h.tolist(), float(best_fit), n_points)
