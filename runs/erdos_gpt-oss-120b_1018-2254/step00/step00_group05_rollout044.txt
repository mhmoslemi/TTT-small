import time
import numpy as np
from scipy.optimize import minimize

def _compute_c5(h):
    """Compute the Erdős C5 bound for a discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_sum_box(x, target):
    """
    Project vector x onto the feasible set:
        0 <= y_i <= 1  and  sum(y) == target.
    Simple alternating projection; sufficient for small dimensions.
    """
    y = np.clip(x, 0.0, 1.0)
    for _ in range(15):
        s = np.sum(y)
        diff = s - target
        if abs(diff) < 1e-12:
            break
        free = (y > 0.0) & (y < 1.0)
        nfree = np.count_nonzero(free)
        if nfree == 0:
            # fallback: distribute uniformly
            y -= diff / len(y)
        else:
            delta = diff / nfree
            y[free] -= delta
        y = np.clip(y, 0.0, 1.0)
    # final correction for any tiny residual
    s = np.sum(y)
    if abs(s - target) > 1e-9:
        # adjust uniformly within bounds
        adjust = (target - s) / len(y)
        y = np.clip(y + adjust, 0.0, 1.0)
    return y

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] (discretised) to lower the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start = time.time()

    # -------------------------------------------------
    # 1. Initialise h
    # -------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_init = 70
        h = np.random.rand(n_init)
        h = _project_to_sum_box(h, n_init / 2)

    n = h.shape[0]
    target_sum = n / 2
    h = _project_to_sum_box(h, target_sum)

    best_h = h.copy()
    best_score = _compute_c5(best_h)

    # -------------------------------------------------
    # 2. Local optimisation via SLSQP (continuous)
    # -------------------------------------------------
    eq_cons = {"type": "eq", "fun": lambda x: np.sum(x) - target_sum}
    bounds = [(0.0, 1.0)] * n

    # Time budget for the deterministic optimiser
    opt_time = budget_s * 0.4
    opt_options = {"maxiter": 2000, "ftol": 1e-9, "disp": False}

    try:
        res = minimize(
            _compute_c5,
            best_h,
            method="SLSQP",
            bounds=bounds,
            constraints=[eq_cons],
            options=opt_options,
        )
        if res.success:
            cand = _project_to_sum_box(res.x, target_sum)
            cand_score = _compute_c5(cand)
            if cand_score < best_score:
                best_h, best_score = cand, cand_score
    except Exception:
        pass

    # -------------------------------------------------
    # 3. Stochastic refinement (pairwise mass transfers)
    # -------------------------------------------------
    def _random_transfer(x):
        """Perform a small transfer of mass between two random indices."""
        i, j = np.random.choice(len(x), size=2, replace=False)
        max_add = min(0.05, 1.0 - x[i])
        max_sub = min(0.05, x[j])
        delta = np.random.uniform(0.0, min(max_add, max_sub))
        if delta <= 0.0:
            return x
        y = x.copy()
        y[i] += delta
        y[j] -= delta
        return y

    while time.time() - start < budget_s * 0.9:
        # generate a neighbour by one transfer
        neigh = _random_transfer(best_h)
        sco = _compute_c5(neigh)
        if sco < best_score - 1e-12:
            best_h, best_score = neigh, sco
            continue

        # occasional larger random restart
        if np.random.rand() < 0.005:
            trial = np.random.rand(n)
            trial = _project_to_sum_box(trial, target_sum)
            sco = _compute_c5(trial)
            if sco < best_score:
                best_h, best_score = trial, sco

    # -------------------------------------------------
    # 4. Final feasibility enforcement
    # -------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_to_sum_box(best_h, target_sum)

    # -------------------------------------------------
    # 5. Compute final bound (use provided evaluator if present)
    # -------------------------------------------------
    try:
        # evaluate_erdos_solution is supplied by the harness
        c5_bound = float(evaluate_erdos_solution(best_h))
    except Exception:
        c5_bound = _compute_c5(best_h)

    return best_h, c5_bound, n
