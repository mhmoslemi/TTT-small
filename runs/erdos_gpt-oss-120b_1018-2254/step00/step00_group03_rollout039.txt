def binary_optimize(n, target_m, time_limit):
    start = time.time()
    N = n // 2
    best_h = None
    best_val = np.inf
    while time.time() < time_limit:
        # random vector
        h = random_binary_vector(n, N)
        val = max_mismatch(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
            if best_val <= target_m:
                break
        # hill climb: targeted swaps for few steps
        # we can iterate limited times (e.g., up to 1000 steps)
        for _ in range(200):
            # find worst shift
            mis = np.correlate(h, 1 - h, mode='full')
            max_idx = np.argmax(mis)
            shift = max_idx - (n - 1)
            # get contributors
            if shift >= 0:
                contrib = np.where((h[:n - shift] == 1) & (h[shift:] == 0))[0]
            else:
                s = -shift
                contrib = np.where((h[s:] == 1) & (h[:n - s] == 0))[0] + s
            if contrib.size == 0:
                break
            i = np.random.choice(contrib)
            zeros = np.where(h == 0)[0]
            j = np.random.choice(zeros)
            # swap
            h[i] = 0; h[j] = 1
            new_val = max_mismatch(h)
            if new_val < val:
                val = new_val
                if val < best_val:
                    best_val = val
                    best_h = h.copy()
                    if best_val <= target_m:
                        return best_h, best_val
            else:
                # revert
                h[i] = 1; h[j] = 0
        # check time
        if time.time() >= time_limit:
            break
    return best_h, best_val

        else:
            s = -shift
            grad[s:] += -dx * h[:n - s]

        # step size decreasing

            best_val = cur_val
            best_h = h_new.copy()
            if best_val <= target_c5:
                break
        h = h_new
        iteration += 1
    return best_h, best_val

import time
import numpy as np

def max_mismatch(h):
    """Maximum of ∫ h(x)(1‑h(x+k)) dx approximated by discrete correlation."""
    return int(np.max(np.correlate(h, 1 - h, mode="full")))

def project_box_sum(v, target_sum, max_iter=60, tol=1e-9):
    """Euclidean projection of v onto {0≤x≤1, sum(x)=target_sum}."""
    lo, hi = -1e6, 1e6
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if abs(s - target_sum) <= tol:
            return w
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return w

def random_binary_vector(n, k):
    """Random 0/1 vector of length n with exactly k ones."""
    h = np.zeros(n, dtype=int)
    idx = np.random.choice(n, size=k, replace=False)
    h[idx] = 1
    return h

def continuous_optimize(h, n, target_c5, deadline):
    """Gradient descent on the worst‑shift objective."""
    dx = 2.0 / n
    target = target_c5
    N = n / 2.0
    best_h = h.copy()
    best_val = max_mismatch(h) * dx
    it = 0
    while time.time() < deadline:
        mis = np.correlate(h, 1 - h, mode="full")
        idx = int(np.argmax(mis))
        shift = idx - (n - 1)
        grad = np.zeros_like(h, dtype=float)
        if shift >= 0:
            grad[: n - shift] += -dx * h[shift:]
            grad[shift:] += -dx * h[: n - shift]
        else:
            s = -shift
            grad[s:] += -dx * h[: n - s]
            grad[: n - s] += -dx * h[s:]
        step = 0.5 / (it + 1.0)
        h_new = h - step * grad
        h = project_box_sum(h_new, N)
        cur = max_mismatch(h) * dx
        if cur < best_val:
            best_val, best_h = cur, h.copy()
            if best_val <= target:
                break
        it += 1
    return best_h, best_val

    N = n // 2
    best_h = None

        if val < best_val:

            else:
                s = -shift

            if contrib.size == 0:

            i = np.random.choice(contrib)
            zeros = np.where(h == 0)[0]
