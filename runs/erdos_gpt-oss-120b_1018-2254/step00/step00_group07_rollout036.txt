def project_simplex_bounds(v, target, max_iter=1000, tol=1e-9):
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        s = v.sum()
        delta = s - target
        if abs(delta) <= tol:
            break
        # Determine indices that can be moved (if need to decrease, those >0; if increase, those <1)
        if delta > 0:
            # reduce
            idx = v > 0
        else:
            idx = v < 1
        if not np.any(idx):
            # cannot adjust further due to bounds, break
            break
        # compute adjustment per element
        adj = delta / np.count_nonzero(idx)
        # apply
        v[idx] -= adj
        # clip again after adjustment
        v = np.clip(v, 0.0, 1.0)
    # final clip
    v = np.clip(v, 0.0, 1.0)
    # after loop maybe sum not exactly target due to bound restrictions; adjust small leftover using method of distributing onto free indices (<1 & >0)
    # Using iterative approach again may converge.
    return v

def project_feasible(v, target):
    # iterative projection onto box [0,1] and sum target using simple waterfilling
    # Using algorithm from Duchi et al. for simplex with upper bound.
    # Equivalent to solving: min ||x - v||^2 s.t. 0 <= x <= 1, sum x = target.
    # We can use generic algorithm: first clip to [0,1], then adjust with projection onto hyperplane ignoring bounds, then clip again, iterate.
    # However for simplicity implement naive iterative approach.
    x = np.clip(v, 0.0, 1.0)

        s = x.sum()
        diff = s - target
        if abs(diff) <= 1e-12:
            break
        # Determine adjustable indices
        if diff > 0:
            # need to decrease sum

        else:

def _c5_value(h: np.ndarray) -> float:
    """Return the C₅ value of a step function h."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h)
    r = np.correlate(h, 1.0 - h, mode='full')
    return np.max(r) * dx

def _project_onto_feasible(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project vector v onto the set {x : 0 ≤ x_i ≤ 1, sum x_i = target}.
    Simple water‑filling iteration; sufficient for the modest dimension (≤1000).
    """
    x = np.clip(v, 0.0, 1.0)
    for _ in range(500):
        s = x.sum()
        diff = s - target
        if abs(diff) <= 1e-12:
            break
        if diff > 0:
            adjustable = x > 0.0
        else:
            adjustable = x < 1.0
        cnt = np.count_nonzero(adjustable)
        if cnt == 0:
            break
        adj = diff / cnt
        x[adjustable] -= adj
        x = np.clip(x, 0.0, 1.0)
    # tiny residual correction (should be negligible)
    diff = target - x.sum()
    if abs(diff) > 1e-12:

            if (diff > 0 and x[i] < 1.0) or (diff < 0 and x[i] > 0.0):
                x[i] += diff
                x[i] = min(max(x[i], 0.0), 1.0)
                break
    return x

def _local_refine(h: np.ndarray, target: float, rng: np.random.Generator,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple pairwise hill‑climbing: move a small amount of mass from one point to another
    while keeping the total integral fixed.
    """
    best = h.copy()
    best_val = _c5_value(best)
    step = 0.12                     # initial perturbation magnitude
    n = best.size
    while time.time() < deadline - 0.3:
        i, j = rng.choice(n, size=2, replace=False)
        # permissible delta for i (respect bounds)
        lo = -min(step, best[i])
        hi = min(step, 1.0 - best[i])
        if lo == hi:
            continue
        delta = rng.uniform(lo, hi)
        if not (0.0 <= best[j] - delta <= 1.0):
            continue
        cand = best.copy()
        cand[i] += delta
        cand[j] -= delta
        val = _c5_value(cand)
        if val < best_val:
            best, best_val = cand, val
            step = max(step * 0.97, 0.004)   # slowly shrink step
    return best, best_val

def _optimize(initial: np.ndarray, n: int,
              rng: np.random.Generator, deadline: float) -> tuple[np.ndarray, float]:
    """
    Multiple random restarts with SLSQP local search, followed by a light
    hill‑climbing phase.
    """
    target = n / 2.0
    dx = 2.0 / n

    # ensure the starter respects the constraints
    cur = _project_onto_feasible(initial, target)
    best_h = cur.copy()
    best_val = _c5_value(best_h)

    # objective in the form required by scipy
    def obj(x):
        return np.max(np.correlate(x, 1.0 - x, mode='full')) * dx

    eq_cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}
    bounds = [(0.0, 1.0)] * n

    restart = 0
    while time.time() < deadline - 2.5 and restart < 2000:
        restart += 1
        # jitter around the current best
        sigma = 0.22
        x0 = best_h + rng.normal(scale=sigma, size=n)
        x0 = _project_onto_feasible(x0, target)

        res = opt.minimize(obj, x0, method='SLSQP',
                           bounds=bounds,
                           constraints=[eq_cons],
                           options={'maxiter': 180, 'ftol': 1e-9, 'disp': False})

        cand = _project_onto_feasible(res.x, target)
        cand_val = _c5_value(cand)

        if cand_val < best_val:
            best_h, best_val = cand, cand_val

        # occasional pure‑random restart to escape stagnation
        if restart % 23 == 0:
            rand = rng.random(n)
            rand = _project_onto_feasible(rand, target)
            rand_val = _c5_value(rand)
            if rand_val < best_val:
                best_h, best_val = rand, rand_val

    # final hill‑climbing polish
    best_h, best_val = _local_refine(best_h, target, rng, deadline)
    return best_h, best_val

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Entry point required by the harness.
    Returns (h_values, c5_bound, n_points).
    """
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # Determine dimension and starting vector
    try:
        init = np.asarray(initial_h_values, dtype=float)
        n = init.size
        init_h = init.copy()
    except NameError:
        n = 80
        # start from a random binary pattern (exactly n/2 ones)
        target_int = n // 2
        init_h = np.zeros(n, dtype=float)
        idx = rng.choice(n, size=target_int, replace=False)
        init_h[idx] = 1.0

    # Bring the starter into the feasible set (exact sum)
    target_sum = n / 2.0
    if abs(init_h.sum() - target_sum) > 1e-12:
        init_h = _project_onto_feasible(init_h, target_sum)

    # Main optimization
    best_h, best_c5 = _optimize(init_h, n, rng, deadline)

    # Final safety projection (exact sum, bounds)
    best_h = _project_onto_feasible(best_h, target_sum)
    best_c5 = _c5_value(best_h)

    return (best_h, best_c5, n)
