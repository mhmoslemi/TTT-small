def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, cvxpy as cp
    from scipy.optimize import differential_evolution, minimize, LinearConstraint

    start = time.time()
    rng = np.random.default_rng(seed)

    # Determine initial h and dimension
    try:
        initial = initial_h_values
        if not isinstance(initial, np.ndarray):
            initial = np.array(initial, dtype=float)
    except NameError:
        # No initial provided, create uniform 0.5
        n = 45
        initial = np.full(n, 0.5)
    else:
        n = initial.shape[0]

    # compute objective function
    def c5_of_h(h):
        # ensure float array
        h = np.asarray(h, dtype=float)
        dx = 2.0 / h.shape[0]
        corr = np.correlate(h, 1 - h, mode='full')
        return dx * np.max(corr)

    best_h = initial.copy()
    best_c5 = c5_of_h(best_h)

    # define constraints
    sum_target = n / 2.0
    linear_constraint = LinearConstraint(np.ones((1, n)), [sum_target], [sum_target])
    bounds = [(0.0, 1.0)] * n

    # callback for time limit
    def de_callback(xk, convergence):
        if time.time() - start > budget_s * 0.9:
            return True
        return False

    # run DE
    de_options = dict(strategy='best1bin',
                      maxiter=500,
                      popsize=15,
                      tol=1e-6,
                      mutation=(0.5, 1),
                      recombination=0.7,
                      seed=seed,
                      polish=False,
                      callback=de_callback,
                      disp=False,
                      init='latinhypercube')
    try:
        result = differential_evolution(c5_of_h, bounds,
                                         constraints=(linear_constraint,),
                                         **de_options)
        if result.success:
            cand_h = result.x
            cand_c5 = c5_of_h(cand_h)
            if cand_c5 < best_c5:
                best_h = cand_h
                best_c5 = cand_c5
    except Exception as e:
        # ignore if DE fails
        pass

    # check remaining time for local refinement
    remaining = budget_s - (time.time() - start)
    if remaining > 10.0:
        # local SLSQP refinement
        # define constraints for minimize
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - sum_target},)
        # bounds already defined
        # use minimize with method SLSQP
        try:
            res_local = minimize(c5_of_h, best_h,
                                 method='SLSQP',
                                 bounds=bounds,
                                 constraints=cons,
                                 options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False})
            if res_local.success:
                cand_h = res_local.x
                cand_c5 = c5_of_h(cand_h)
                if cand_c5 < best_c5:
                    best_h = cand_h
                    best_c5 = cand_c5
        except Exception as e:
            pass

    # final projection onto feasibility using cvxpy
    try:
        x = cp.Variable(n)
        objective = cp.Minimize(cp.sum_squares(x - best_h))
        constraints = [x >= 0, x <= 1, cp.sum(x) == sum_target]
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
        if x.value is not None:
            proj_h = np.clip(x.value, 0.0, 1.0)
            proj_c5 = c5_of_h(proj_h)
            if proj_c5 <= best_c5 + 1e-12:
                best_h = proj_h
                best_c5 = proj_c5
    except Exception as e:
        # fallback to best_h as is
        pass

    # ensure final sum close to target; if needed slight renormalize
    # Might get minor floating error; we can adjust a small amount at a non-bound position
    # But we trust cvxpy.

    # final compute
    final_c5 = c5_of_h(best_h)

    # Return list or array:
    return best_h, float(final_c5), n

            best_c5 = new_c5

    # ----------------------------------------------------------------------
    # Project onto the exact feasible set using cvxpy (if available)
    # ----------------------------------------------------------------------
    try:
        import cvxpy as cp

        x = cp.Variable(n)
        prob = cp.Problem(
            cp.Minimize(cp.sum_squares(x - best_h)),
            [x >= 0, x <= 1, cp.sum(x) == target_sum],
        )
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
        if x.value is not None:
            proj = np.clip(x.value, 0.0, 1.0)
            proj_c5 = compute_c5(proj)
            if proj_c5 <= best_c5 + 1e-12:
                best_h = proj
                best_c5 = proj_c5
    except Exception:
        pass

    # ----------------------------------------------------------------------
    # Tiny post‑processing to correct any residual sum error
    # ----------------------------------------------------------------------
    sum_err = np.sum(best_h) - target_sum
    if abs(sum_err) > 1e-9:
        free_idx = np.where((best_h > 1e-8) & (best_h < 1 - 1e-8))[0]
        if free_idx.size > 0:
            correction = sum_err / free_idx.size
            best_h[free_idx] -= correction
            best_h = np.clip(best_h, 0.0, 1.0)
            # final scaling if still off
            sum_err2 = np.sum(best_h) - target_sum
            if abs(sum_err2) > 1e-9:
                best_h *= target_sum / np.sum(best_h)

    final_c5 = compute_c5(best_h)

    return best_h, float(final_c5), n
