def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, random, math
    # Setup
    np.random.seed(seed)
    random.seed(seed)
    # Determine starting h
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n_points = len(h0)
    except NameError:

        # generate random feasible vector

        # scale to meet sum constraint

        # Clip and adjust
        # Simple approach: scale to sum target, then clip and re-normalize iteratively
        h0 = np.clip(h0, 0, 1)
        current_sum = np.sum(h0)
        # If sum too low or high, uniformly add some slack

        # But simpler: use Dirichlet: draw random positives from exponential 1

        if current_sum != 0:

        # iterative correction:
        for _ in range(5):
            # Clip
            h0 = np.clip(h0, 0, 1)
            s = np.sum(h0)
            if abs(s - target) < 1e-8:
                break
            diff = target - s
            # Distribute diff uniformly over indices not at bounds
            free = (h0 > 0) & (h0 < 1)
            # If no free, break
            if not np.any(free):
                break
            # add diff / count to free slots
            h0[free] += diff / np.count_nonzero(free)
        # final clip
        h0 = np.clip(h0, 0, 1)
    # Ensure constraints:
    n_points = len(h0)
    target_sum = n_points / 2.0
    # Force sum exactly via scaling then clipping maybe:
    # We'll define a repair function.

    def repair(vec):
        # Clip to [0,1]
        vec = np.clip(vec, 0, 1)
        # Ensure sum exactly target_sum via simple scaling of free entries
        s = np.sum(vec)
        if abs(s - target_sum) < 1e-9:
            return vec
        # Compute difference
        diff = target_sum - s
        # Iterate adjusting free entries:
        # We'll use a simple method: if diff >0, add to random free entries up to 1; if diff <0, subtract from random free entries down to 0.
        # Avoid infinite loops: limit iterations.
        max_iter = 1000
        for _ in range(max_iter):
            if abs(diff) < 1e-9:
                break
            # Determine candidates that can be increased/decreased
            if diff > 0:
                candidates = np.where(vec < 1 - 1e-12)[0]
                if len(candidates) == 0:
                    break
                # amount to add per candidate = min(diff/len(candidates), remaining capacity)
                add = diff / len(candidates)
                # limit add to capacity
                capacity = 1 - vec[candidates]
                # actual add limited
                actual = np.minimum(add, capacity)
                vec[candidates] += actual
                diff = target_sum - np.sum(vec)
            else:
                candidates = np.where(vec > 1e-12)[0]
                if len(candidates) == 0:
                    break
                sub = (-diff) / len(candidates)
                capacity = vec[candidates]
                actual = np.minimum(sub, capacity)
                vec[candidates] -= actual
                diff = target_sum - np.sum(vec)
        # final clip
        vec = np.clip(vec, 0, 1)
        return vec

    # Repair h0 to ensure exact sum
    h0 = repair(h0)

    # Predefine objective
    def c5_of_h(h):
        # correlation full gives length 2*n-1, centered at lag 0 at index n-1
        n = len(h)
        dx = 2.0 / n
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return float(np.max(corr))

    best_score = c5_of_h(best_h)
    # current solution for annealing
    cur_h = np.copy(best_h)
    cur_score = best_score

    max_iter = 2000000  # large, will break by time limit

    # track best and potentially restart after certain time if stuck
    # but simple approach.

    for it in range(max_iter):
        if elapsed() > total_budget:
            break
        # Adjust temperature
        fraction = elapsed() / total_budget
        temp = init_temp * (1 - fraction) + final_temp * fraction
        # Propose a move: pick i, j
        i, j = np.random.randint(0, n_points, size=2)
        while i == j:
            j = np.random.randint(0, n_points)
        # compute max allowable delta due to bounds
        # current hi, hj
        hi = cur_h[i]
        hj = cur_h[j]
        # delta change: hi -> hi+δ, hj->hj-δ
        # δ max up to maintain hi+δ <=1 and hj-δ >=0
        # δ_max_pos = min(1 - hi, hj)
        # δ_min_neg = -min(hi, 1 - hj)
        # Actually δ can be positive or negative.
        max_pos = min(1 - hi, hj)  # max positive delta
        max_neg = -min(hi, 1 - hj)  # max negative delta (negative value)
        if max_pos <= 0 and max_neg >= 0:
            continue  # cannot move
        # sample delta uniformly within allowed range
        delta = (max_pos - max_neg) * np.random.rand() + max_neg
        # Optional small set of bigger moves? We'll also add occasional larger jump
        # Perform move
        new_h = cur_h.copy()
        new_h[i] = hi + delta
        new_h[j] = hj - delta
        # Evaluate
        new_score = c5_of_h(new_h)
        # Acceptance
        if new_score < cur_score or np.random.rand() < math.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_h = new_h
            cur_score = new_score
            # Update best if improved
            if new_score < best_score:
                best_h = new_h.copy()
                best_score = new_score
        # Optionally occasionally perform random restart: if not improved for many steps
        # we can skip for simplicity.

    # At end: compute final C5 bound
    final_score = c5_of_h(best_h)
    # Return
    return (best_h.tolist(), final_score, n_points)

def _c5_of_h(h):
    """Compute C5 for a given step function sampled as array h."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1 - h) over all integer shifts
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def _repair(vec):
    """Project a vector onto the feasible set: 0<=v<=1 and sum(v)=n/2."""
    n = vec.size
    target = n / 2.0
    v = np.clip(vec, 0.0, 1.0)
    # Simple iterative redistribution to hit the target sum
    for _ in range(10):
        s = v.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to add mass – spread over entries that are <1
            free = np.where(v < 1.0)[0]
            if free.size == 0:
                break
            add = diff / free.size
            capacity = 1.0 - v[free]
            inc = np.minimum(add, capacity)
            v[free] += inc
        else:
            # need to remove mass – spread over entries that are >0
            free = np.where(v > 0.0)[0]
            if free.size == 0:
                break
            sub = (-diff) / free.size
            capacity = v[free]
            dec = np.minimum(sub, capacity)
            v[free] -= dec
    return np.clip(v, 0.0, 1.0)

def _random_feasible(n):
    """Create a random feasible vector of length n."""
    target = n / 2.0
    int_part = int(math.floor(target))
    frac = target - int_part
    h = np.zeros(n, dtype=float)
    if int_part > 0:
        idx_one = np.random.choice(n, int_part, replace=False)
        h[idx_one] = 1.0
    if frac > 1e-12:
        free = np.where(h == 0.0)[0]
        idx_frac = np.random.choice(free)
        h[idx_frac] = frac
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound, and the number of samples."""
    np.random.seed(seed)
    random.seed(seed)

    # Obtain initial vector if provided, otherwise start from a random feasible one
    try:
        h0 = np.array(initial_h_values, dtype=float)  # provided by harness
    except NameError:
        # fallback to a fresh random construction of moderate size
        n_points = 96  # an even size makes pure binary feasible
        h0 = _random_feasible(n_points)
    else:
        n_points = h0.size

    target_sum = n_points / 2.0
    h0 = _repair(h0)                     # ensure exact feasibility
    best_h = h0.copy()
    best_score = _c5_of_h(best_h)

    # current state for simulated annealing
    cur_h = best_h.copy()
    cur_score = best_score

    start_time = time.time()
    total_budget = budget_s * 0.95  # leave a safety margin
    restart_interval = 30.0         # seconds without improvement before a random restart
    last_improve = start_time

    # Annealing schedule parameters
    init_temp = 0.02
    final_temp = 1e-6

    while True:
        elapsed = time.time() - start_time
        if elapsed >= total_budget:
            break

        # Temperature decreasing linearly with elapsed fraction
        frac = elapsed / total_budget
        temp = init_temp * (1.0 - frac) + final_temp * frac

        # Propose a feasible swap/move between two indices
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        while i == j:
            j = np.random.randint(0, n_points)

        hi = cur_h[i]
        hj = cur_h[j]

        # Determine allowable delta so that both entries stay in [0,1]
        max_pos = min(1.0 - hi, hj)          # largest positive delta
        max_neg = -min(hi, 1.0 - hj)         # most negative delta (<=0)
        if max_pos <= 0.0 and max_neg >= 0.0:
            continue  # no feasible move between these two indices

        delta = (max_pos - max_neg) * np.random.rand() + max_neg

        new_h = cur_h.copy()
        new_h[i] = hi + delta
        new_h[j] = hj - delta

        new_score = _c5_of_h(new_h)

        # Acceptance rule: better always, worse with probability depending on temperature
        if new_score < cur_score:
            accept = True
        else:
            # avoid division by zero
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = np.random.rand() < prob

        if accept:
            cur_h = new_h
            cur_score = new_score
            if new_score < best_score:
                best_h = new_h.copy()
                best_score = new_score
                last_improve = time.time()

        # Random restart if stagnation lasts too long
        if (time.time() - last_improve) > restart_interval:
            # restart from a fresh random feasible point
            cur_h = _random_feasible(n_points)
            cur_score = _c5_of_h(cur_h)
            last_improve = time.time()  # reset timer

    # Final evaluation (ensuring exact feasibility)
    best_h = _repair(best_h)          # final safety repair
    final_score = _c5_of_h(best_h)

    return (best_h.tolist(), final_score, n_points)
