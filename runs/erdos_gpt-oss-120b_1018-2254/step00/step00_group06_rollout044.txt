import time, numpy as np

def project_onto_capped_simplex(v, target, lower=0.0, upper=1.0, max_iter=60):
    lo = -1e5
    hi = 1e5
    for _ in range(max_iter):
        mid = (lo+hi)/2.0
        w = np.clip(v - mid, lower, upper)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return w

def evaluate_c5(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def initialize_population(rng, n, pop_size, target):
    pop = []
    for _ in range(pop_size):
        v = rng.random(n)  # uniform [0,1)
        h = project_onto_capped_simplex(v, target)
        pop.append(h)
    return pop

def tournament_select(pop, fitness, rng, tournament_size=3):
    idxs = rng.integers(0, len(pop), size=tournament_size)
    best = idxs[0]
    best_f = fitness[best]
    for idx in idxs[1:]:
        if fitness[idx] < best_f:
            best = idx
            best_f = fitness[idx]
    return pop[best]

def crossover(parent1, parent2, rng, crossover_rate=0.7):
    n = len(parent1)
    if rng.random() < crossover_rate:
        point = rng.integers(1, n)  # crossover point at least 1 and <n
        child = np.empty(n)
        child[:point] = parent1[:point]
        child[point:] = parent2[point:]
    else:
        child = parent1.copy()
    return child

def mutate(child, rng, mutation_rate=0.02, sigma=0.1):
    n = len(child)
    # generate mask of mutations
    mask = rng.random(n) < mutation_rate
    if np.any(mask):
        noise = rng.normal(0.0, sigma, size=n)
        child = child + mask * noise
    return child

import time
import numpy as np

def _project_onto_capped_simplex(v, target, lower=0.0, upper=1.0, max_iter=60):
    """Euclidean projection of v onto the set {x | lower <= x_i <= upper, sum(x)=target}."""
    lo, hi = -1e5, 1e5
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, lower, upper)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return w

def _c5_score(h, dx):
    """Compute the overlap integral C5 for a discretized h."""
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def _initialize_population(rng, n, pop_size, target):
    pop = []
    for _ in range(pop_size):
        v = rng.random(n)
        h = _project_onto_capped_simplex(v, target)
        pop.append(h)
    return pop

def _tournament_select(pop, fitness, rng, size=3):
    idx = rng.integers(0, len(pop), size=size)
    best = idx[0]
    best_f = fitness[best]
    for i in idx[1:]:
        if fitness[i] < best_f:
            best = i
            best_f = fitness[i]
    return pop[best]

def _crossover(p1, p2, rng, prob=0.7):
    n = len(p1)
    if rng.random() < prob:
        pt = rng.integers(1, n)
        child = np.empty(n, dtype=float)
        child[:pt] = p1[:pt]
        child[pt:] = p2[pt:]
        return child
    return p1.copy()

def _mutate(child, rng, rate=0.02, sigma=0.15):
    mask = rng.random(child.shape) < rate
    if np.any(mask):
        child = child + mask * rng.normal(0.0, sigma, size=child.shape)
    return child
