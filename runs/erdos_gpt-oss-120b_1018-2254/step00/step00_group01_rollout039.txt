def _project_onto_bounds_and_sum(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto the set {0<=x<=1, sum(x)=target_sum}."""
    lo = -np.max(v)               # makes v+lam <= 0 -> sum 0
    hi = 1.0 - np.min(v)          # makes v+lam >= 1 -> sum = len(v)

    return proj

def _compute_f_vals(h: np.ndarray) -> np.ndarray:
    """Return the vector f_L = sum_i h_i (1-h_{i+L}) for all integer shifts L."""
    n = h.shape[0]
    m = 2 * n - 1
    f = np.empty(m, dtype=np.float64)
    idx = 0
    for L in range(-(n - 1), n):
        if L >= 0:
            seg = n - L
            hi = h[:seg]
            hj = h[L:L + seg]
        else:
            K = -L
            seg = n - K
            hi = h[K:K + seg]
            hj = h[:seg]
        f[idx] = hi @ (1.0 - hj)
        idx += 1
    return f

def _smooth_gradient(h: np.ndarray, beta: float, f_vals: np.ndarray) -> np.ndarray:
    """Gradient of (1/beta)*logsumexp(beta*f_vals) w.r.t. h."""
    n = h.shape[0]
    max_f = f_vals.max()
    exp_terms = np.exp(beta * (f_vals - max_f))
    weights = exp_terms / exp_terms.sum()
    grad = np.zeros_like(h)
    idx = 0
    for L in range(-(n - 1), n):
        w = weights[idx]
        if w != 0.0:
            if L >= 0:
                seg = n - L
                grad[:seg] += w * (1.0 - h[L:L + seg])
                grad[L:L + seg] -= w * h[:seg]
            else:
                K = -L
                seg = n - K
                grad[K:K + seg] += w * (1.0 - h[:seg])
                grad[:seg] -= w * h[K:K + seg]
        idx += 1
    return grad

def _c5_from_h(h: np.ndarray) -> float:
    """Exact C5 value for a given h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a good step function for the Erdős C5 problem.

    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()

    # ---- initial vector ----------------------------------------------------
    init = globals().get("initial_h_values", None)
    if init is not None:
        h = np.asarray(init, dtype=np.float64).copy()
    else:
        # fallback: 48 points (even) with uniform value 0.5
        h = np.full(48, 0.5, dtype=np.float64)

    n = h.shape[0]
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    h = _project_onto_bounds_and_sum(h, target_sum)

    # ---- bookkeeping -------------------------------------------------------
    best_h = h.copy()
    best_val = _c5_from_h(best_h)

    # optimisation parameters
    beta = 15.0                     # smoothing for max
    alpha0 = 0.5                    # initial step size multiplier
    decay = 0.9995                  # step-size decay per iteration
    max_no_improve = 800
    no_improve = 0
    iteration = 0

    # ---- main optimisation loop -------------------------------------------
    while time.time() - start_time < budget_s * 0.95:
        # evaluate current objective and its smooth gradient
        f_vals = _compute_f_vals(h)
        cur_val = f_vals.max() * (2.0 / n)

        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # stochastic perturbation every 120 iterations
        if iteration % 120 == 0 and iteration > 0:
            pert = h + np.random.normal(scale=0.04, size=n)
            pert = np.clip(pert, 0.0, 1.0)
            pert = _project_onto_bounds_and_sum(pert, target_sum)
            pert_val = _c5_from_h(pert)
            if pert_val < best_val:
                best_val = pert_val
                best_h = pert.copy()
                h = pert
                continue

        # smooth gradient step
        grad = _smooth_gradient(h, beta, f_vals)
        alpha = alpha0 * (decay ** iteration)
        h_new = h - alpha * grad
        h_new = _project_onto_bounds_and_sum(h_new, target_sum)

        # accept if it improves (or with tiny probability to escape)
        new_val = _c5_from_h(h_new)
        if new_val <= cur_val:
            h = h_new
        else:
            # simple simulated‑annealing acceptance
            temperature = max(1e-4, (budget_s - (time.time() - start_time)) / budget_s)
            prob = np.exp(-(new_val - cur_val) / (temperature * 0.01 + 1e-12))
            if np.random.rand() < prob:
                h = h_new

        # random restart if stuck for a while
        if no_improve >= max_no_improve:
            rand_vec = np.random.rand(n)
            h = _project_onto_bounds_and_sum(rand_vec, target_sum)
            no_improve = 0

        # early exit if we already beat the target bound
        if best_val <= 0.3808:
            break

        iteration += 1

    final_c5 = _c5_from_h(best_h)
    return best_h, float(final_c5), n
