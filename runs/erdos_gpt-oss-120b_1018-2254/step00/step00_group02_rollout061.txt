def improve_binary_h(h_initial, rng, time_budget, start_time):
    n = len(h_initial)
    cur_h = h_initial.copy()
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    # early exit if already good
    if best_c5 <= 0.3808:
        return best_h, best_c5
    while True:
        elapsed = time.time() - start_time
        if elapsed >= time_budget:
            break
        # compute correlation
        corr = np.correlate(cur_h, 1 - cur_h, mode='full')
        max_idx = int(np.argmax(corr))
        shift = max_idx - (n - 1)
        # find indices where h[i]==1 and h[i+shift]==0 (bad ones)
        bad_i = None
        if shift >= 0:
            i_vals = np.arange(0, n - shift)
            mask = (cur_h[i_vals] == 1) & (cur_h[i_vals + shift] == 0)
            bad_i = i_vals[mask]
        else:
            s = -shift
            i_vals = np.arange(s, n)
            mask = (cur_h[i_vals] == 1) & (cur_h[i_vals - s] == 0)
            bad_i = i_vals[mask]
        # ones and zeros indices
        ones_idx = np.where(cur_h == 1)[0]
        zeros_idx = np.where(cur_h == 0)[0]
        # select i and j for swap
        if bad_i.size > 0:
            i = rng.choice(bad_i)
            # choose j among zeros that won't create new bad pair at this shift
            cand_js = []
            for j in zeros_idx:
                if shift >= 0:
                    jp = j + shift
                    if jp < 0 or jp >= n or cur_h[jp] == 1:
                        cand_js.append(j)
                else:
                    s = -shift
                    jm = j - s
                    if jm < 0 or jm >= n or cur_h[jm] == 1:
                        cand_js.append(j)
            if len(cand_js) > 0:
                j = rng.choice(cand_js)
            else:
                j = rng.choice(zeros_idx)
        else:
            # random swap
            if len(ones_idx) == 0 or len(zeros_idx) == 0:
                break
            i = rng.choice(ones_idx)

    x_old = np.linspace(0, 2, len(old_h))
    x_new = np.linspace(0, 2, n_new)
    new_h = np.interp(x_new, x_old, old_h)

    binary[idx] = 1.0
    return binary

import numpy as np
import time

def compute_c5(h: np.ndarray) -> float:
    """Return C5 = dx * max correlation of h with (1-h)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return dx * float(np.max(corr))


def interpolate_h(old_h: np.ndarray, n_new: int) -> np.ndarray:
    """Interpolate a previous h to a new resolution."""
    x_old = np.linspace(0.0, 2.0, old_h.shape[0])
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, old_h)


def improve_binary_h(h_initial: np.ndarray, rng: np.random.Generator,
                     time_budget: float, start_time: float) -> tuple:
    """
    Hill‑climbing on a binary vector, swapping 1↔0 while keeping the sum fixed.
    Stops when the allocated time expires or a target C5 is reached.
    Returns the best found vector and its C5.
    """
    n = h_initial.shape[0]
    cur_h = h_initial.copy()
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    target = 0.3808

    while True:
        # time check
        if time.time() - start_time >= time_budget:
            break

        # correlation and shift that realizes the maximum
        corr = np.correlate(cur_h, 1 - cur_h, mode="full")
        max_idx = int(np.argmax(corr))
        shift = max_idx - (n - 1)

        # indices where a 1 is paired with a 0 at this shift (bad contributions)
        if shift >= 0:
            i_vals = np.arange(0, n - shift)
            bad_mask = (cur_h[i_vals] == 1) & (cur_h[i_vals + shift] == 0)
            bad_i = i_vals[bad_mask]
        else:
            s = -shift
            i_vals = np.arange(s, n)
            bad_mask = (cur_h[i_vals] == 1) & (cur_h[i_vals - s] == 0)
            bad_i = i_vals[bad_mask]

        ones_idx = np.where(cur_h == 1)[0]
        zeros_idx = np.where(cur_h == 0)[0]

        # choose a swap
        if bad_i.size > 0 and zeros_idx.size > 0:
            i = rng.choice(bad_i)
            # try to pick a zero that does NOT create a new bad pair at this shift
            cand_js = []
            if shift >= 0:
                for j in zeros_idx:
                    jp = j + shift
                    if jp >= n or cur_h[jp] == 1:
                        cand_js.append(j)
            else:
                s = -shift
                for j in zeros_idx:
                    jm = j - s
                    if jm < 0 or cur_h[jm] == 1:
                        cand_js.append(j)
            if cand_js:
                j = rng.choice(cand_js)
            else:
                j = rng.choice(zeros_idx)
        else:
            # random 1↔0 swap
            if ones_idx.size == 0 or zeros_idx.size == 0:
                break
            i = rng.choice(ones_idx)
            j = rng.choice(zeros_idx)

        # propose the swap
        new_h = cur_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_c5 = compute_c5(new_h)

        # accept if improvement
        if new_c5 < cur_c5:
            cur_h = new_h
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_h = cur_h.copy()
                best_c5 = cur_c5
                if best_c5 <= target:
                    break

    return best_h, best_c5


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a step function h on [0,2] with ∫h = 1 minimizing C5.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    best_h = None
    best_c5 = np.inf
    best_n = None
    target = 0.3808

    # candidate sample sizes, larger first (prefer even for binary constructions)
    candidate_ns = [200, 180, 150, 120, 90, 45]

    for n in candidate_ns:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining <= 5.0:
            break

        # --------------------------------------------------------------------
        # even n → binary search (exact sum)
        # --------------------------------------------------------------------
        if n % 2 == 0:
            # Try to initialise from a known construction if available
            try:
                base_h = initial_h_values  # provided by the harness
                interp = interpolate_h(base_h, n)
                # Threshold to exactly half ones
                idx = np.argsort(interp)[-n // 2 :]
                h0 = np.zeros(n, dtype=float)
                h0[idx] = 1.0
            except Exception:
                # fallback: random half‑ones / half‑zeros
                h0 = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)])
                rng.shuffle(h0)

            # Allocate a large fraction of the remaining time to this size
            alloc = remaining * 0.85
            time_budget = elapsed + alloc
            # improve and possibly update the global best
            h_opt, c5_opt = improve_binary_h(h0, rng, time_budget, start_time)
            if c5_opt < best_c5:
                best_h, best_c5, best_n = h_opt, c5_opt, n
            if best_c5 <= target:
                break
        else:
            # ----------------------------------------------------------------
            # odd n → use the provided initial vector if it matches this size
            # ----------------------------------------------------------------
            try:
                hv = initial_h_values
                if hv.shape[0] == n:
                    c5 = compute_c5(hv)
                    if c5 < best_c5:
                        best_h, best_c5, best_n = hv.copy(), c5, n
                    if best_c5 <= target:
                        break
            except NameError:
                pass

    # ------------------------------------------------------------------------
    # If nothing was found (unlikely), fall back to the raw initial vector.
    # ------------------------------------------------------------------------
    if best_h is None:
        try:
            best_h = initial_h_values.copy()
            best_n = best_h.shape[0]
            best_c5 = compute_c5(best_h)
        except NameError:
            # final desperate fallback: uniform 0.5 on a tiny grid
            best_n = 45
            best_h = np.full(best_n, 0.5, dtype=float)
            best_c5 = compute_c5(best_h)

    # ------------------------------------------------------------------------
    # Minor post‑processing: enforce the integral constraint exactly.
    # ------------------------------------------------------------------------
    target_sum = best_n / 2.0
    cur_sum = float(best_h.sum())
    if abs(cur_sum - target_sum) > 1e-9:
        # distribute the discrepancy uniformly and clip
        delta = target_sum - cur_sum
        best_h = best_h + delta / best_n
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5, best_n
