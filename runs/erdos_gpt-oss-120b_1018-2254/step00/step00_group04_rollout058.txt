def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    L = 1 << ((2*n - 1).bit_length())  # next power of two >=2n-1
    H = np.fft.rfft(np.pad(h, (0, L - n), mode='constant'))
    G = np.fft.rfft(np.pad(1 - h, (0, L - n), mode='constant'))
    corr = np.fft.irfft(H * np.conj(G), n=L)
    # correlation length = 2n-1 where valid indices from 0..2n-2
    # The result of ifft is length L, but only first 2n-1 entries are the linear correlation.
    corr = corr[:2*n-1]
    return np.max(corr) * dx
