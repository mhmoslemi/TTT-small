import numpy as np, time, math

def evaluate_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(x, target_sum):
    # Clip
    x = np.clip(x, 0.0, 1.0)
    # iterative adjust
    for _ in range(10):
        s = np.sum(x)
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to increase
            eligible = np.where(x < 1.0)[0]
            if eligible.size == 0:
                break
            # compute total capacity
            capacity = np.sum(1.0 - x[eligible])
            if capacity <= diff + 1e-12:
                x[eligible] = 1.0
                continue
            delta = diff / eligible.size
            # ensure not overshoot
            x[eligible] += delta
        else:
            eligible = np.where(x > 0.0)[0]
            if eligible.size ==0:
                break
            capacity = np.sum(x[eligible])
            if capacity <= -diff + 1e-12:
                x[eligible] = 0.0
                continue
            delta = diff / eligible.size
            x[eligible] += delta
        x = np.clip(x, 0.0, 1.0)
    # final correction maybe
    # To ensure sum exactly target within tolerance, adjust a random index
    s = np.sum(x)
    diff = target_sum - s
    if abs(diff) > 1e-9:
        # adjust a random eligible index
        indices = np.arange(len(x))
        # Find indices where adjusting by diff doesn't violate bounds
        if diff > 0:
            candidate = indices[x < 1.0]
        else:
            candidate = indices[x > 0.0]
        if candidate.size > 0:
            i = np.random.choice(candidate)
            x[i] += diff
            x[i] = np.clip(x[i], 0.0, 1.0)
    return x

import time
import numpy as np


def _evaluate_c5(h):
    """Return the overlap integral C5 for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _project_to_constraints(x, target_sum, max_iter=10):
    """Project x onto the feasible set: 0≤x≤1 and sum(x)=target_sum."""
    # clip to box
    x = np.clip(x, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(x)
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to increase some components
            eligible = np.where(x < 1.0)[0]
            if eligible.size == 0:
                break
            capacity = np.sum(1.0 - x[eligible])
            if capacity <= diff + 1e-12:
                x[eligible] = 1.0
                continue
            delta = diff / eligible.size
            x[eligible] += delta
        else:
            # need to decrease some components
            eligible = np.where(x > 0.0)[0]
            if eligible.size == 0:
                break
            capacity = np.sum(x[eligible])
            if capacity <= -diff + 1e-12:
                x[eligible] = 0.0
                continue
            delta = diff / eligible.size
            x[eligible] += delta
        x = np.clip(x, 0.0, 1.0)

    # final tiny fix (deterministic)
    cur = np.sum(x)
    diff = target_sum - cur
    if abs(diff) > 1e-9:
        if diff > 0:
            cand = np.where(x < 1.0)[0]
            if cand.size:
                i = cand[np.argmin(x[cand])]
                inc = min(diff, 1.0 - x[i])
                x[i] += inc
        else:
            cand = np.where(x > 0.0)[0]
            if cand.size:
                i = cand[np.argmax(x[cand])]
                dec = min(-diff, x[i])
                x[i] -= dec
    return x
