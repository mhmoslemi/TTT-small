def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    try:
        import cvxpy as cp
    except Exception:
        cp = None
    start = time.time()
    rng = np.random.default_rng(seed)
    try:
        n = int(len(initial_h_values))
    except NameError:
        n = 80
    total_sum = n / 2
    dx = 2.0 / n

    best_h = None
    best_val = np.inf

    # Try MILP if possible and n even and not too large
    if cp is not None and n % 2 == 0 and n <= 200:
        # Build MILP
        h = cp.Variable(n, boolean=True, name='h')
        t = cp.Variable(name='t')
        constraints = [cp.sum(h) == total_sum,
                       t >= 0]
        # Build w variables and constraints
        for k in range(1, n):
            w_k = cp.Variable(n - k, boolean=True, name=f'w_{k}')
            constraints += [w_k <= h[:n - k],
                            w_k <= h[k:],
                            w_k >= h[:n - k] + h[k:] - 1]
            ip_k = cp.sum(w_k)
            constraints += [ip_k >= total_sum - t]
        prob = cp.Problem(cp.Minimize(t), constraints)
        try:
            prob.solve(solver=cp.GLPK_MI, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_opt = np.rint(np.array(h.value)).astype(float)
                # ensure sum
                # If rounding changed sum, adjust by fixing some entries.
                cur_sum = h_opt.sum()
                if cur_sum != total_sum:
                    # fix by adjusting random entries
                    diff = int(total_sum - cur_sum)
                    if diff > 0:
                        zeros = np.where(h_opt == 0)[0]
                        idx = rng.choice(zeros, size=diff, replace=False)
                        h_opt[idx] = 1.0
                    elif diff < 0:
                        ones = np.where(h_opt == 1)[0]
                        idx = rng.choice(ones, size=-diff, replace=False)
                        h_opt[idx] = 0.0
                best_h = h_opt
                best_val = float(t.value) * dx
        except Exception:
            pass

    # If MILP not solved, fallback to heuristic
    if best_h is None:
        # optional: try initial_h_values as start
        # generate legendre candidate
        def legendre_candidate(n, rng):
            # find largest prime less than n
            def is_prime(p):
                if p < 2:
                    return False
                if p % 2 == 0:
                    return p == 2
                r = int(np.sqrt(p)) + 1
                for d in range(3, r, 2):
                    if p % d == 0:
                        return False
                return True
            for p in range(n - 1, 2, -1):
                if is_prime(p):
                    seq = np.zeros(p, dtype=float)
                    for i in range(1, p):
                        if pow(i, (p - 1) // 2, p) == 1:
                            seq[i] = 1.0
                    h = np.zeros(n, dtype=float)
                    h[:p] = seq
                    target = n // 2
                    cur = int(h.sum())
                    if cur < target:
                        zeros = np.where(h == 0)[0]
                        add = target - cur
                        if len(zeros) >= add:
                            idx = rng.choice(zeros, size=add, replace=False)
                            h[idx] = 1.0
                    elif cur > target:
                        ones = np.where(h == 1)[0]
                        rem = cur - target
                        idx = rng.choice(ones, size=rem, replace=False)
                        h[idx] = 0.0
                    return h
            # fallback random
            h = np.zeros(n, dtype=float)
            h[rng.choice(n, size=n // 2, replace=False)] = 1.0
            return h

        # start candidate
        cur_h = legendre_candidate(n, rng)
        def compute_c5(h):
            return np.max(np.correlate(h, 1 - h, mode='full')) * dx

        cur_val = compute_c5(cur_h)
        best_h = cur_h.copy()
        best_val = cur_val

        stagnation = 0
        max_stagnation = 5000
        initial_temp = 0.2
        while True:
            elapsed = time.time() - start
            if elapsed >= budget_s - 0.2:
                break
            # temperature schedule
            frac = elapsed / budget_s
            temp = max(1e-4, initial_temp * (1 - frac))
            # get indices
            ones_idx = np.where(cur_h == 1)[0]
            zeros_idx = np.where(cur_h == 0)[0]
            if len(ones_idx) == 0 or len(zeros_idx) == 0:
                break
            i1 = rng.choice(ones_idx)
            i0 = rng.choice(zeros_idx)
            # swap
            cur_h[i1] = 0.0
            cur_h[i0] = 1.0
            new_val = compute_c5(cur_h)
            delta = new_val - cur_val
            if delta <= 0 or rng.random() < np.exp(-delta / temp):
                cur_val = new_val
                if new_val < best_val - 1e-12:
                    best_val = new_val
                    best_h = cur_h.copy()
                    stagnation = 0
                else:
                    stagnation += 1
            else:
                # revert swap
                cur_h[i0] = 0.0
                cur_h[i1] = 1.0
                stagnation += 1
            if stagnation >= max_stagnation:
                # restart with random binary
                cur_h = np.zeros(n, dtype=float)
                cur_h[rng.choice(n, size=n // 2, replace=False)] = 1.0
                cur_val = compute_c5(cur_h)
                stagnation = 0
    # Return
    return (best_h, float(best_val), n)

import time
import numpy as np

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns a step function h on [0,2] sampled at n points,
    together with its C5 bound and n.
    The function tries a MILP formulation (if cvxpy is available)
    and falls back to a simulated‑annealing search.
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine number of samples
    try:
        n = int(len(initial_h_values))  # use parent if present
    except Exception:
        n = 80  # default size
    dx = 2.0 / n
    total_sum = n / 2  # required sum of h

    # -----------------------------------------------------------------
    # Helper: compute C5 from a vector h (dx already applied)
    def compute_c5(h_vec):
        return np.max(np.correlate(h_vec, 1.0 - h_vec, mode="full")) * dx

    # -----------------------------------------------------------------
    # Try MILP with cvxpy (binary formulation)
    best_h = None
    best_val = np.inf

    try:
        import cvxpy as cp

        # Only attempt MILP for moderate sizes (up to ~200 points)
        if n % 2 == 0 and n <= 200:
            # Decision variables
            h = cp.Variable(n, boolean=True, name="h")
            t = cp.Variable(name="t")          # maximal overlap count (not scaled)

            constraints = [cp.sum(h) == total_sum,
                           t >= 0]

            # For each shift k (both forward and backward)
            for k in range(1, n):
                # binary products w_i,k = h_i * h_{i+k}
                w = cp.Variable(n - k, boolean=True, name=f"w_{k}")
                constraints += [
                    w <= h[: n - k],
                    w <= h[k:],
                    w >= h[: n - k] + h[k:] - 1
                ]
                ip_k = cp.sum(w)                     # inner product for shift k
                fwd_sum = cp.sum(h[: n - k])         # Σ_{i=0}^{n-1-k} h_i
                rev_sum = cp.sum(h[k:])              # Σ_{i=k}^{n-1}   h_i
                constraints += [
                    fwd_sum - ip_k <= t,              # forward shift overlap
                    rev_sum - ip_k <= t               # backward shift overlap
                ]

            prob = cp.Problem(cp.Minimize(t), constraints)
            # Attempt solve; many solvers accept a time limit but we omit it for simplicity
            prob.solve(solver=getattr(cp, "CBC", cp.GLPK_MI), verbose=False)

            if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
                # Retrieve and round solution
                h_val = np.rint(np.array(h.value)).astype(float)
                # Fix possible rounding drift in the sum
                cur_sum = h_val.sum()
                if cur_sum != total_sum:
                    diff = int(total_sum - cur_sum)
                    if diff > 0:
                        zeros = np.where(h_val == 0)[0]
                        if len(zeros) >= diff:
                            idx = rng.choice(zeros, size=diff, replace=False)
                            h_val[idx] = 1.0
                    elif diff < 0:
                        ones = np.where(h_val == 1)[0]
                        if len(ones) >= -diff:
                            idx = rng.choice(ones, size=-diff, replace=False)
                            h_val[idx] = 0.0
                t_val = float(t.value) if t.value is not None else None
                if t_val is not None:
                    t_int = int(np.ceil(t_val - 1e-9))  # ensure feasibility
                    best_h = h_val
                    best_val = t_int * dx
    except Exception:
        # cvxpy not available or MILP failed; simply continue to heuristic
        pass

    # -----------------------------------------------------------------
    # Heuristic fallback: simulated annealing on binary vectors
    if best_h is None:
        # Produce an initial candidate (Legendre‑type if possible)
        def legendre_candidate(p, rng):
            # find largest prime < p
            def is_prime(q):
                if q < 2:
                    return False
                if q % 2 == 0:
                    return q == 2
                r = int(np.sqrt(q)) + 1
                for d in range(3, r, 2):
                    if q % d == 0:
                        return False
                return True

            for prime in range(p - 1, 2, -1):
                if is_prime(prime):
                    seq = np.zeros(prime, dtype=float)
                    for i in range(1, prime):
                        if pow(i, (prime - 1) // 2, prime) == 1:
                            seq[i] = 1.0
                    h0 = np.zeros(p, dtype=float)
                    h0[:prime] = seq
                    target = p // 2
                    cur = int(h0.sum())
                    if cur < target:
                        zeros = np.where(h0 == 0)[0]
                        add = target - cur
                        idx = rng.choice(zeros, size=add, replace=False)
                        h0[idx] = 1.0
                    elif cur > target:
                        ones = np.where(h0 == 1)[0]
                        rem = cur - target
                        idx = rng.choice(ones, size=rem, replace=False)
                        h0[idx] = 0.0
                    return h0
            # fallback random
            h0 = np.zeros(p, dtype=float)
            h0[rng.choice(p, size=p // 2, replace=False)] = 1.0
            return h0

        # start with Legendre‑type or random if not possible
        cur_h = legendre_candidate(n, rng)
        cur_val = compute_c5(cur_h)

        best_h = cur_h.copy()
        best_val = cur_val

        stagnation = 0
        max_stagnation = 5000
        initial_temp = 0.2

        # Simulated annealing loop
        while True:
            elapsed = time.time() - start_time
            if elapsed >= budget_s - 0.2:
                break
            # temperature schedule (linear cooling)
            frac = elapsed / budget_s
            temp = max(1e-6, initial_temp * (1 - frac))

            ones_idx = np.where(cur_h == 1)[0]
            zeros_idx = np.where(cur_h == 0)[0]
            if len(ones_idx) == 0 or len(zeros_idx) == 0:
                break

            i1 = rng.choice(ones_idx)
            i0 = rng.choice(zeros_idx)

            # propose swap
            cur_h[i1] = 0.0
            cur_h[i0] = 1.0
            new_val = compute_c5(cur_h)
            delta = new_val - cur_val

            if delta <= 0 or rng.random() < np.exp(-delta / temp):
                cur_val = new_val
                if new_val < best_val - 1e-12:
                    best_val = new_val
                    best_h = cur_h.copy()
                    stagnation = 0
                else:
                    stagnation += 1
            else:
                # revert swap
                cur_h[i0] = 0.0
                cur_h[i1] = 1.0
                stagnation += 1

            if stagnation >= max_stagnation:
                # restart from a fresh random configuration
                cur_h = np.zeros(n, dtype=float)
                cur_h[rng.choice(n, size=n // 2, replace=False)] = 1.0
                cur_val = compute_c5(cur_h)
                stagnation = 0

    # -----------------------------------------------------------------
    # Return result
    return (best_h, float(best_val), n)
