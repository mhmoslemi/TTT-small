import numpy as np, time, math

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    init = globals().get('initial_h_values', None)
    if init is not None:
        h0 = np.array(init, dtype=np.float64).copy()
        n = h0.shape[0]
    else:
        n = 45
        h0 = np.zeros(n, dtype=np.float64)
        ones = rng.choice(n, size=n//2, replace=False)
        h0[ones] = 1.0

    target_sum = n / 2.0
    # project to feasible box and sum
    h0 = np.clip(h0, 0.0, 1.0)
    cur_sum = h0.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        diff = target_sum - cur_sum
        # simple repair by moving mass between random indices
        for _ in range(1000):
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                candidates = np.where(h0 < 1 - 1e-12)[0]
                if candidates.size == 0:
                    break
                idx = rng.choice(candidates)
                inc = min(diff, 1.0 - h0[idx])
                h0[idx] += inc
                diff -= inc
            else:
                candidates = np.where(h0 > 1e-12)[0]
                if candidates.size == 0:
                    break
                idx = rng.choice(candidates)
                dec = min(-diff, h0[idx])
                h0[idx] -= dec
                diff += dec
        h0 = np.clip(h0, 0.0, 1.0)
        # final adjust a single element if needed
        if abs(h0.sum() - target_sum) > 1e-9:
            discrepancy = target_sum - h0.sum()
            for i in range(n):
                new_val = h0[i] + discrepancy
                if 0.0 <= new_val <= 1.0:
                    h0[i] = new_val
                    break

    dx = 2.0 / n

    def evaluate(arr):
        # Compute max overlap integral
        corr = np.correlate(arr, 1.0 - arr, mode='full')
        return np.max(corr) * dx

    current_h = h0.copy()
    current_val = evaluate(current_h)
    best_h = current_h.copy()
    best_val = current_val

    # SA parameters
    start_time = time.time()
    T_initial = 0.02
    T_final = 1e-6
    max_move_initial = 0.2

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
        # temperature schedule
        T = T_initial * (T_final / T_initial) ** (elapsed / (budget_s * 0.95))
        max_move = max_move_initial * (T / T_initial)  # scales with temperature

        if rng.random() < 0.5:
            # targeted move based on worst shift
            auto_corr = np.correlate(current_h, current_h, mode='full')
            min_idx = int(np.argmin(auto_corr))
            r_star = min_idx - (n - 1)  # shift value
            # list indices i where i and i+r_star are valid
            if r_star >= 0:
                i_vals = np.arange(0, n - r_star, dtype=int)
            else:
                i_vals = np.arange(-r_star, n, dtype=int)
            # mask where we can increase h[i]
            mask_inc = current_h[i_vals] < 1.0 - 1e-12
            if np.any(mask_inc):
                counterpart = current_h[i_vals + r_star]
                inc_candidates = i_vals[mask_inc]
                inc_counter = counterpart[mask_inc]
                # choose i with largest counterpart (most benefit)
                target_i = inc_candidates[np.argmax(inc_counter)]
                # donor selection (decrease somewhere)
                donor_candidates = np.where(current_h > 1e-12)[0]
                donor_candidates = donor_candidates[donor_candidates != target_i]
                if donor_candidates.size > 0:
                    donor = rng.choice(donor_candidates)
                    # feasible delta
                    delta_max = min(1.0 - current_h[target_i], current_h[donor], max_move)
                    if delta_max > 1e-12:
                        delta = rng.random() * delta_max
                        new_h = current_h.copy()
                        new_h[target_i] += delta
                        new_h[donor] -= delta
                        new_val = evaluate(new_h)
                        if new_val < current_val or rng.random() < math.exp(-(new_val - current_val) / max(T, 1e-12)):
                            current_h = new_h
                            current_val = new_val
                            if new_val < best_val:
                                best_h = new_h.copy()
                                best_val = new_val
                        continue  # move to next iteration
            # fallback to random move if targeting fails
        # random pairwise move
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)
        hi, hj = current_h[i], current_h[j]
        lower_raw = max(-hi, hj - 1.0)
        upper_raw = min(1.0 - hi, hj)
        # enforce max_move limit
        lower = max(lower_raw, -max_move)
        upper = min(upper_raw, max_move)
        if lower >= upper:
            continue
        delta = rng.uniform(lower, upper)
        new_h = current_h.copy()
        new_h[i] = hi + delta
        new_h[j] = hj - delta
        # Clip just in case due to numeric noise
        new_h[i] = min(1.0, max(0.0, new_h[i]))
        new_h[j] = min(1.0, max(0.0, new_h[j]))
        new_val = evaluate(new_h)
        if new_val < current_val or rng.random() < math.exp(-(new_val - current_val) / max(T, 1e-12)):
            current_h = new_h
            current_val = new_val
            if new_val < best_val:
                best_h = new_h.copy()
                best_val = new_val

    # final correction of feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_diff = target_sum - best_h.sum()
    if abs(sum_diff) > 1e-8:
        # try to correct by adjusting a suitable index
        idxs = np.arange(n)
        rng.shuffle(idxs)
        for idx in idxs:
            new_val = best_h[idx] + sum_diff
            if 0.0 <= new_val <= 1.0:
                best_h[idx] = new_val
                break
    # final bound using official evaluator if available
    eval_func = globals().get('evaluate_erdos_solution', None)
    if eval_func is not None:
        final_val = float(eval_func(best_h))
    else:
        final_val = float(evaluate(best_h))

    return (best_h, final_val, n)
