import time
import numpy as np

def _repair_sum(vec, target):
    """Project vec onto the simplex sum(vec)=target with 0<=vec<=1."""
    # Clip to bounds first
    vec = np.clip(vec, 0.0, 1.0)
    # Iteratively distribute the difference proportionally to available capacity
    for _ in range(10):
        diff = target - vec.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            cap = 1.0 - vec
        else:
            cap = vec
        idx = np.where(cap > 1e-12)[0]
        if idx.size == 0:
            break
        total_cap = cap[idx].sum()
        # If diff exceeds total capacity due to rounding, cap it
        factor = diff / total_cap if total_cap != 0 else 0.0
        # Update and clip
        vec[idx] += cap[idx] * factor
        vec = np.clip(vec, 0.0, 1.0)
    return vec

def _evaluate(vec):
    """Return C5 value for a given h vector."""
    N = vec.size
    dx = 2.0 / N
    corr = np.correlate(vec, 1.0 - vec, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Determine problem size and initial vector
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        n_points = init.shape[0]
        target_sum = n_points / 2.0
        h0 = _repair_sum(np.clip(init, 0.0, 1.0), target_sum)
    else:
        n_points = 45
        target_sum = n_points / 2.0
        rnd = np.random.rand(n_points)
        h0 = _repair_sum(rnd, target_sum)

    # Parameters
    POP_SIZE = 30
    MUT_PROB = 0.2      # per gene mutation probability
    SIGMA = 0.12        # mutation stddev
    TOURNAMENT = 3
    TARGET_BOUND = 0.3808

    # Initialize population
    population = [h0]
    while len(population) < POP_SIZE:
        rnd = np.random.rand(n_points)
        vec = _repair_sum(rnd, target_sum)
        population.append(vec)

    fitness = [_evaluate(ind) for ind in population]
    best_idx = int(np.argmin(fitness))
    best_h = population[best_idx].copy()
    best_val = fitness[best_idx]

    # Main evolutionary loop
    while time.time() - start_time < budget_s - 0.05:
        # Tournament selection for a parent
        candidates = np.random.choice(POP_SIZE, size=TOURNAMENT, replace=False)
        parent_idx = candidates[int(np.argmin([fitness[i] for i in candidates]))]
        parent = population[parent_idx]

        # Produce offspring by mutation
        child = parent.copy()
        mask = np.random.rand(n_points) < MUT_PROB
        if mask.any():
            child[mask] += np.random.randn(mask.sum()) * SIGMA
        child = np.clip(child, 0.0, 1.0)
        child = _repair_sum(child, target_sum)

        # Evaluate child
        child_val = _evaluate(child)

        # Replace worst if improvement
        worst_idx = int(np.argmax(fitness))
        if child_val < fitness[worst_idx]:
            population[worst_idx] = child
            fitness[worst_idx] = child_val

        # Track global best
        if child_val < best_val:
            best_val = child_val
            best_h = child.copy()
            # Early exit if we reached target

            vec = _repair_sum(rnd, target_sum)
