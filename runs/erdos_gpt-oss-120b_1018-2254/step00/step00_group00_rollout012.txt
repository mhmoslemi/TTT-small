    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    s = idx - (n - 1)

    i_shift2 = i - s

import numpy as np
import time
import math

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] with ∫h = 1 to minimize
    C5 = max_k ∫ h(x) (1 - h(x+k)) dx.
    Returns (h_values, c5_bound, n_points).
    """

    # ---------- Helper utilities ----------
    def evaluate(h):
        """Return C5 for a given h (numpy array)."""
        n = h.shape[0]
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr) * dx

    def project_to_bounds_and_sum(v, target):
        """
        Euclidean projection of vector v onto the box [0,1]^n
        with the linear equality sum(v)=target.
        Uses a simple binary search on the Lagrange multiplier.
        """
        # Search interval for lambda
        lo = -np.max(v) - 1.0
        hi = np.max(v) + 1.0
        for _ in range(60):
            lam = (lo + hi) / 2.0
            x = np.clip(v - lam, 0.0, 1.0)
            s = x.sum()
            if s > target:
                lo = lam
            else:
                hi = lam
        x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

        # Tiny correction to hit the target sum more accurately
        diff = target - x.sum()
        if abs(diff) > 1e-10:
            free = (x > 1e-12) & (x < 1.0 - 1e-12)
            if free.any():
                cnt = free.sum()
                x[free] += diff / cnt
                x = np.clip(x, 0.0, 1.0)
            else:
                # Fall back to uniform adjustment (very rare)
                x += diff / len(x)
                x = np.clip(x, 0.0, 1.0)
        return x

    # ---------- Initialization ----------
    np.random.seed(seed)
    start_time = time.time()
    max_elapsed = budget_s * 0.95  # keep a safety margin

    # Determine problem size and starting vector
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = kwargs.get("n_points", 45)
        h0 = np.random.rand(n).astype(float)

    target_sum = n / 2.0
    h0 = project_to_bounds_and_sum(h0, target_sum)

    best_h = h0.copy()
    best_score = evaluate(best_h)
    cur_h = best_h.copy()
    cur_score = best_score

    # ---------- Optimization loop ----------
    # Parameters (will decay over time)
    sigma = 0.3          # scale of random perturbations
    step_scale = 0.2    # step size for gradient moves

    while True:
        elapsed = time.time() - start_time
        if elapsed >= max_elapsed:
            break

        # Choose move type
        if np.random.rand() < 0.7:
            # Random Gaussian perturbation
            cand = cur_h + sigma * np.random.randn(n)
        else:
            # Gradient step on the worst shift
            corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
            idx = np.argmax(corr)
            shift = idx - (n - 1)

            grad = np.zeros_like(cur_h)

            i = np.arange(n)

            # Contribution where h[i] is the first factor
            ip = i + shift
            mask1 = (ip >= 0) & (ip < n)
            grad[mask1] += 1.0 - cur_h[ip[mask1]]

            # Contribution where h[i] appears in the complement term
            im = i - shift
            mask2 = (im >= 0) & (im < n)
            grad[mask2] += -cur_h[im[mask2]]

            cand = cur_h - step_scale * grad

        # Project back to feasible region
        cand = project_to_bounds_and_sum(cand, target_sum)
        cand_score = evaluate(cand)

        # Acceptance rule (simulated annealing style)
        if cand_score < cur_score:
            cur_h, cur_score = cand, cand_score
            if cand_score < best_score:
                best_h, best_score = cand.copy(), cand_score
        else:
            temperature = max(sigma, 1e-4)
            if np.random.rand() < math.exp(-(cand_score - cur_score) / temperature):
                cur_h, cur_score = cand, cand_score

    try:

    except Exception:
        pass

    return (best_h.tolist(), float(final_score), n)
