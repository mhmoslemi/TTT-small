def run(seed=42, budget_s=1100, **kwargs):

            non_improve_counter += 1

import numpy as np
import time
import math

def compute_c5(h: np.ndarray) -> float:
    """Return max overlap C5 for a given step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_constraints(h: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project vector h onto the box [0,1]^n while enforcing sum(h) = target_sum.
    Simple water-filling iteration.
    """
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        total = h.sum()
        diff = target_sum - total
        if abs(diff) <= 1e-12:
            break
        # Determine which coordinates can still move in the needed direction
        if diff > 0:
            free = np.where(h < 1.0)[0]      # can increase
        else:
            free = np.where(h > 0.0)[0]      # can decrease
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        # Re‑clip after adjustment
        h = np.clip(h, 0.0, 1.0)
    # Minor residual fixing (distribute any remaining diff to first free slot)
    residual = target_sum - h.sum()
    if abs(residual) > 1e-12:
        # Find first index that can absorb residual without breaking bounds
        for idx in range(h.size):
            lo = 0.0 - h[idx]
            hi = 1.0 - h[idx]
            if lo <= residual <= hi:
                h[idx] += residual
                break
    return h

def propose_move(h: np.ndarray, step: float) -> np.ndarray:
    """Return a new h by moving a small amount of mass between two positions."""
    n = h.size
    i, j = np.random.choice(n, 2, replace=False)
    if np.random.rand() < 0.5:
        # move from i to j
        max_delta = min(step, h[i], 1.0 - h[j])
        if max_delta <= 0:
            return h
        delta = np.random.rand() * max_delta
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
    else:
        # move from j to i
        max_delta = min(step, h[j], 1.0 - h[i])
        if max_delta <= 0:
            return h
        delta = np.random.rand() * max_delta
        new_h = h.copy()
        new_h[j] -= delta
        new_h[i] += delta
    return new_h
