import time
import math
import numpy as np

def compute_c5(h):
    """
    Compute C5 = max_k ∫ h(x) (1 - h(x+k)) dx
    using discrete correlation with zero padding.
    """
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def _adjust_sum_to_target(h, target):
    """Adjust the last element to enforce the exact sum, staying within [0,1]."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # try to fix by tweaking the element with most slack
    idx = np.argmin(h) if diff > 0 else np.argmax(h)
    new_val = np.clip(h[idx] + diff, 0.0, 1.0)
    h[idx] = new_val
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] with integral 1 that minimizes
    C5 = max_k ∫ h(x)(1-h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)

    # --- initialise h -------------------------------------------------
    try:
        init_h = np.asarray(initial_h_values, dtype=np.float64)
        N = init_h.shape[0]
        h_cur = init_h.copy()
    except NameError:
        N = 80
        h_cur = np.zeros(N, dtype=np.float64)
        ones_idx = rng.choice(N, size=N // 2, replace=False)
        h_cur[ones_idx] = 1.0

    target_sum = N / 2.0
    h_cur = _adjust_sum_to_target(h_cur, target_sum)

    best_h = h_cur.copy()
    best_c5 = compute_c5(best_h)
    cur_c5 = best_c5

    start_time = time.time()
    end_time = start_time + max(0.0, budget_s - 0.5)  # leave a small margin

    # --- simulated annealing parameters -------------------------------
    step_max = 0.2          # maximum transferred mass in a single move
    T0 = 0.1                # initial temperature
    Tmin = 1e-5             # final temperature
    max_iter = 10_000_000   # safety upper bound
    it = 0

    while time.time() < end_time and it < max_iter:
        it += 1

        # temperature schedule (linear decline)
        T = max(Tmin, T0 * (1.0 - it / 5_000_000))

        # choose source (h>0) and target (h<1) indices
        src_candidates = np.where(h_cur > 1e-12)[0]
        tgt_candidates = np.where(h_cur < 1.0 - 1e-12)[0]
        if src_candidates.size == 0 or tgt_candidates.size == 0:
            # degenerate situation – restart from a fresh binary vector
            h_cur = np.zeros(N, dtype=np.float64)
            ones_idx = rng.choice(N, size=N // 2, replace=False)
            h_cur[ones_idx] = 1.0
            cur_c5 = compute_c5(h_cur)
            continue

        i = rng.choice(src_candidates)
        j = rng.choice(tgt_candidates)

        max_delta = min(step_max, h_cur[i], 1.0 - h_cur[j])
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # apply move (mass transfer from i to j)
        h_cur[i] -= delta
        h_cur[j] += delta

        new_c5 = compute_c5(h_cur)

        # acceptance rule
        if new_c5 < cur_c5 or rng.random() < math.exp(-(new_c5 - cur_c5) / T):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_cur.copy()
        else:
            # revert
            h_cur[i] += delta
            h_cur[j] -= delta

        # slowly tighten step size
        if it % 50_000 == 0:
            step_max = max(0.01, step_max * 0.9)

    # final correction of the sum (tiny numerical drift)
    best_h = _adjust_sum_to_target(best_h, target_sum)

    return best_h, float(best_c5), N
