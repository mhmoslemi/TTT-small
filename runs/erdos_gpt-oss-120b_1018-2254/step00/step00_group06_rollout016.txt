def project_onto_box_sum(y, target):
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(y + mid, 0, 1)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(y + (lo+hi)/2.0, 0, 1)

def project_box_sum(y, target):
    # y: np.ndarray
    # target: scalar
    # returns projected array x
    lo = -np.max(y) - 1.0  # lower bound for lambda
    hi = 1.0 - np.min(y) + 1.0  # upper bound
    # Actually we can set lo, hi more tightly:
    lo = -np.max(y)  # making all <=0
    hi = 1.0 - np.min(y)  # making all >=1
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(y + mid, 0, 1)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y + (lo + hi) / 2.0, 0, 1)
    # Due to numeric rounding, adjust any small residual to match target exact
    # Could distribute diff across non‑boundary entries
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # adjust in direction of gradient (or simply add small to free vars)
        free = (x > 0 + 1e-12) & (x < 1 - 1e-12)
        if free.any():
            # distribute equally
            x[free] += diff / free.sum()
            # clip again
            x = np.clip(x, 0, 1)
    return x

import time
import math
import numpy as np

def _project_box_sum(y: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of y onto the set {x | 0 <= x <= 1, sum(x) = target}.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo = -np.max(y)          # ensures y+lo <= 0 -> all zeros after clipping
    hi = 1.0 - np.min(y)     # ensures y+hi >= 1 -> all ones after clipping
    for _ in range(50):
        mid = (lo + hi) * 0.5
        x = np.clip(y + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)

    # Minor correction for floating‑point drift
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = np.where((x > 1e-12) & (x < 1.0 - 1e-12))[0]
        if free.size:
            adj = diff / free.size
            x[free] += adj
            x = np.clip(x, 0.0, 1.0)
    return x

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] to lower the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.RandomState(seed)

    # ------------------------------------------------------------------
    # Initialise h
    # ------------------------------------------------------------------
    try:
        # initial_h_values is provided by the harness (list or array)
        init_h = np.asarray(initial_h_values, dtype=float)
        n = init_h.shape[0]
        target_sum = n * 0.5          # because integral = 1 and dx = 2/n
        if not math.isclose(init_h.sum(), target_sum, rel_tol=1e-9, abs_tol=1e-9):
            init_h = _project_box_sum(init_h, target_sum)
    except NameError:
        # Fallback: generate a random feasible start
        n = 200                     # a modest size well below 1k
        target_sum = n * 0.5
        init_h = rng.rand(n)
        init_h = _project_box_sum(init_h, target_sum)

    dx = 2.0 / n

    # ------------------------------------------------------------------
    # Helper utilities (closed over n, dx)
    # ------------------------------------------------------------------
    def _c5_of(h_vec: np.ndarray) -> float:
        """C5 = max_k ∫ h(x)(1‑h(x+k))dx = 1 - min shift overlap."""
        # min convolution of h with itself (dx already factored later)
        min_corr = np.min(np.correlate(h_vec, h_vec, mode='full')) * dx
        return 1.0 - min_corr

    def _worst_shift(h_vec: np.ndarray) -> int:
        """Return the integer shift (relative to centre) attaining minimal overlap."""
        corr = np.correlate(h_vec, h_vec, mode='full')
        idx = np.argmin(corr)
        return idx - (n - 1)          # map index to shift in [-n+1, n-1]

    def _grad_overlap(h_vec: np.ndarray, shift: int) -> np.ndarray:
        """
        Gradient of the overlap ∫ h(x)h(x+shift)dx w.r.t. h.
        Implements the formula:
            ∂/∂h[i] = h[i+shift] + h[i-shift]  (zero where index out of range)
        multiplied by dx.
        """
        g = np.zeros_like(h_vec)
        if shift >= 0:
            s = shift
            # contribution from h[i+s]
            g[: n - s] += h_vec[s:]
            # contribution from h[i-s]
            g[s:] += h_vec[: n - s]
        else:
            s = -shift
            # i+shift = i-s
            g[s:] += h_vec[: n - s]        # h[i-s] term
            # i-shift = i+s
            g[: n - s] += h_vec[s:]        # h[i+s] term
        g *= dx
        return g

    # ------------------------------------------------------------------
    # Optimisation loop
    # ------------------------------------------------------------------
    best_h = init_h.copy()
    best_c5 = _c5_of(best_h)

    start_time = time.time()
    time_limit = budget_s - 0.5                # small safety margin

    max_restarts = 5
    for restart in range(max_restarts):
        if time.time() - start_time > time_limit:
            break

        # Initialise current candidate
        if restart == 0:
            h = best_h.copy()
        else:
            h = rng.rand(n)
            h = _project_box_sum(h, target_sum)

        step = 0.1                               # initial step size
        it = 0
        while True:
            if time.time() - start_time > time_limit:
                break

            cur_c5 = _c5_of(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                # early exit if we already beat the target record
                if best_c5 < 0.3808:
                    # keep searching a little longer for robustness
                    pass

            # Identify the shift where overlap is minimal
            s = _worst_shift(h)

            # Gradient ascent on the minimal overlap (which reduces C5)
            g = _grad_overlap(h, s)
            trial = _project_box_sum(h + step * g, target_sum)
            trial_c5 = _c5_of(trial)

            if trial_c5 < cur_c5:
                h = trial
                step = min(step * 1.02, 1.0)     # modest step growth
            else:
                step *= 0.5
                if step < 1e-8:
                    break                       # stagnated

            it += 1
            if it >= 5000:
                break

    # ------------------------------------------------------------------
    # Final projection and evaluation
    # ------------------------------------------------------------------
    best_h = _project_box_sum(best_h, target_sum)
    final_c5 = _c5_of(best_h)

    # Use the provided evaluator if available (more precise)

        final_c5 = float(evaluate_erdos_solution(best_h))
    except NameError:
        pass

    return (best_h, float(final_c5), int(n))
