def optimize_h(n, budget, initial_h=None):
    best_h = np.full(n, 0.5)  # fallback
    if initial_h is not None:
        best_h = initial_h.copy()
    # ensure projection to constraints
    target_sum = n/2
    best_h = project(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    start = time.time()
    elapsed = 0.0
    # parameters
    iteration = 0
    while elapsed < budget:
        # annealing temperature based on iteration count
        temp = max(0.01, 1.0 - elapsed / budget)  # linearly decreasing from 1 to 0.01
        # propose candidate
        # step scale = temp * 0.5 maybe.
        step_scale = temp * 0.5
        perturb = np.random.randn(n) * step_scale
        cand = best_h + perturb
        cand = project(cand, target_sum)
        cand_c5 = compute_c5(cand)
        # accept if better
        if cand_c5 < best_c5:
            best_h = cand
            best_c5 = cand_c5
        # else maybe accept with prob to escape local minima
        else:
            # Metropolis acceptance
            prob = np.exp(-(cand_c5 - best_c5) / (temp * 0.01 + 1e-9))
            if np.random.rand() < prob:
                best_h = cand
                best_c5 = cand_c5
        iteration += 1
        elapsed = time.time() - start
    return best_h, best_c5

    lo = -1.0

import numpy as np
import random
import time

def compute_c5(h):
    """Return the overlap constant C5 for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)


def project_onto_box_and_sum(v, target):
    """
    Euclidean projection of a vector onto the box [0,1]^n with an
    exact sum constraint sum(x)=target using a simple bisection on the
    Lagrange multiplier.
    """
    lo, hi = -1.0, 2.0                     # bounds guarantee sum(lo) > target > sum(hi)
    for _ in range(60):
        tau = (lo + hi) * 0.5
        proj = np.clip(v - tau, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    tau = (lo + hi) * 0.5
    return np.clip(v - tau, 0.0, 1.0)


def run(seed=42, budget_s=1100, **kwargs):
    """Find a step function h on [0,2] minimizing the max overlap integral."""
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Initialise problem size and a feasible starting point
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        target = n / 2.0
        h0 = project_onto_box_and_sum(h0, target)
    else:
        n = 42                     # fallback size
        target = n / 2.0
        h0 = np.zeros(n, dtype=float)
        h0[:int(target)] = 1.0
        np.random.shuffle(h0)

    best_h = h0.copy()
    best_score = compute_c5(best_h)
    best_solution = best_h.copy()

    start = time.time()

    # ------------------------------------------------------------------
    # Simple simulated-annealing / stochastic search
    # ------------------------------------------------------------------
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s:

        # linear cooling schedule: temperature in [0.01,1.0]
        temp = max(0.01, 1.0 - elapsed / budget_s)
