def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math

    np.random.seed(seed)

    # ----- helper functions -----
    def project(vec, target_sum):
        """Clip to [0,1] and adjust to hit the exact target sum."""
        vec = np.clip(vec, 0.0, 1.0)
        diff = target_sum - vec.sum()
        for _ in range(10):
            if abs(diff) < 1e-12:
                break
            mask = (vec > 0) & (vec < 1)
            cnt = mask.sum()
            if cnt == 0:
                add = diff / vec.size
                vec += add
            else:
                add = diff / cnt
                vec[mask] += add
            vec = np.clip(vec, 0.0, 1.0)
            diff = target_sum - vec.sum()
        return vec

    def compute_c5_and_shift(v):
        """Return (max_product, lag) using linear correlation (zero padding)."""
        n = v.shape[0]
        size = 2 * n
        # linear correlation via FFT with zero padding
        H = np.fft.fft(v, size)
        G = np.fft.fft(1.0 - v, size)
        corr = np.fft.ifft(H * np.conj(G)).real[: 2 * n - 1]
        idx = int(np.argmax(corr))
        max_val = float(corr[idx] * (2.0 / n))
        lag = idx - (n - 1)          # lag range: -(n-1) … (n-1)
        return max_val, lag

    # ----- initialization -----
    # use provided construction if available, otherwise a cosine bump
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=np.float64)
        n0 = init.shape[0]
        n = max(200, n0)                 # aim for at least 200 points
        if n % 2:
            n += 1
        if n != n0:
            # linear interpolation to new length
            x_old = np.linspace(0, 2, n0, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h = np.interp(x_new, x_old, init)
        else:
            h = init.copy()
    else:
        n = 200
        if n % 2:
            n += 1
        dx = 2.0 / n
        x = np.linspace(0, 2, n, endpoint=False) + dx * 0.5
        h = (np.cos(np.pi * (x - 1)) + 1.0) * 0.5   # zero at edges, 1 at centre

    target_sum = n / 2.0
    h = project(h, target_sum)

    # ----- main optimization loop -----
    best_h = h.copy()
    best_score, _ = compute_c5_and_shift(best_h)

    start = time.time()
    step = 0.05                     # initial transfer size
    while True:
        if time.time() - start > budget_s * 0.95:
            break

        cur_score, cur_lag = compute_c5_and_shift(h)

        # choose a source index that can transfer mass in the direction of cur_lag
        if cur_lag >= 0:
            src_max = n - cur_lag
            if src_max <= 0:
                continue
            src = np.random.randint(0, src_max)
            tgt = src + cur_lag
        else:
            src_min = -cur_lag
            src = np.random.randint(src_min, n)
            tgt = src + cur_lag   # cur_lag is negative here

        # amount we are allowed to move while respecting bounds
        max_delta = min(step, h[src], 1.0 - h[tgt])
        if max_delta <= 0.0:
            continue
        delta = np.random.rand() * max_delta

        h_new = h.copy()
        h_new[src] -= delta
        h_new[tgt] += delta

        new_score, _ = compute_c5_and_shift(h_new)

        # keep the best solution found
        if new_score < best_score:
            best_score = new_score
            best_h = h_new.copy()

        # accept improvement or occasional uphill move
        if new_score < cur_score:
            h = h_new
        else:

            temp = 0.01 * (1.0 - (time.time() - start) / budget_s)

    final_score, _ = compute_c5_and_shift(best_h)
