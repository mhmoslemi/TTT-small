def compute_c5(h):
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def compute_c5_and_shift(h):
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    max_val = corr[max_idx] * dx
    shift = max_idx - (N - 1)
    return max_val, shift

def gradient(h, shift):
    N = len(h)
    G = np.ones_like(h)
    s = shift
    if s >= 0:
        # i + s in bound: i = 0..N-1-s
        G[:N - s] -= h[s:]
        # i - s in bound: i = s..N-1
        G[s:] -= h[:N - s]
    else:
        s_abs = -s
        # i + s = i - s_abs
        G[s_abs:] -= h[:N - s_abs]
        # i - s = i + s_abs
        G[:N - s_abs] -= h[s_abs:]
    return G

def _project_to_sum_bounds(x: np.ndarray, target_sum: float) -> np.ndarray:
    """Clamp x to [0,1] and adjust uniformly so that sum(x) == target_sum."""
    x = np.clip(x, 0.0, 1.0)
    # binary search for the additive shift lambda
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        y = np.clip(x + lam, 0.0, 1.0)
        s = y.sum()
        if s < target_sum:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    y = np.clip(x + lam, 0.0, 1.0)
    # tiny correction if needed
    diff = target_sum - y.sum()
    if abs(diff) > 1e-12:
        # add/subtract diff from a non‑boundary element
        for idx in range(len(y)):
            if 0.0 < y[idx] < 1.0:
                y[idx] += diff
                y[idx] = min(1.0, max(0.0, y[idx]))
                break
    return y


def _c5_and_shift(h: np.ndarray) -> tuple[float, int]:
    """Return (C5 value, integer shift that attains the maximum)."""
    N = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    c5 = corr[idx] * (2.0 / N)
    shift = idx - (N - 1)          # integer offset corresponding to the correlation index
    return c5, shift


def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the overlap for a given integer shift."""
    N = h.shape[0]
    g = np.ones_like(h)
    s = shift
    if s >= 0:
        # term -h[i+s]
        if s < N:
            g[:N - s] -= h[s:]
        # term -h[i-s]
        if s > 0:
            g[s:] -= h[:N - s]
    else:
        s_abs = -s
        # term -h[i+s]  → i+s = i-s_abs
        if s_abs < N:
            g[s_abs:] -= h[:N - s_abs]
        # term -h[i-s]  → i-s = i+s_abs
        if s_abs > 0:
            g[:N - s_abs] -= h[s_abs:]
    return g

        # compute descent direction for the worst shift
        grad = _gradient_for_shift(h, cur_shift)
        donors = np.where(grad > 1e-9)[0]       # decreasing these reduces the worst overlap
        receivers = np.where(grad < -1e-9)[0]   # increasing these reduces the worst overlap

        if donors.size == 0 or receivers.size == 0:
            # fallback random feasible transfer
            i = rng.integers(0, n_points)
            j = rng.integers(0, n_points - 1)
            if j >= i:
                j += 1
            max_delta = min(max_step, h[i], 1.0 - h[j])
            if max_delta > 0:
                delta = rng.random() * max_delta
                h[i] -= delta
                h[j] += delta
            continue

        i = rng.choice(donors)
        j = rng.choice(receivers)

        # adaptive step size (shrinks as time runs out)
        elapsed = time.time() - start_time
        step = max_step * (1.0 - elapsed / budget_s)
        step = max(step, 0.001)

        max_delta = min(step, h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # propose move
        h[i] -= delta
        h[j] += delta

        # accept only if the worst overlap does not increase
        new_c5, _ = _c5_and_shift(h)
        if new_c5 > cur_c5 + 1e-12:
            # revert
            h[i] += delta
            h[j] -= delta

    # ------------------------------------------------------------------
    # final projection and evaluation
    best_h = _project_to_sum_bounds(best_h, target_sum)
    final_c5, _ = _c5_and_shift(best_h)

    return best_h.tolist(), float(final_c5), n_points
