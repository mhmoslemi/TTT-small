def local_optimize(h0, n):
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n/2}
    res = opt.minimize(lambda x: compute_c5(x), h0, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': 200, 'ftol': 1e-9, 'disp': False})
    h_opt = res.x
    # Clip very small out-of-bound errors:
    h_opt = np.clip(h_opt, 0, 1)
    # Possibly enforce sum exactly:
    cur_sum = np.sum(h_opt)
    if not np.isclose(cur_sum, n/2, atol=1e-6):
        # adjust by adding uniform correction to meet sum while staying within bounds.
        diff = n/2 - cur_sum
        # distribute diff proportionally to capacity to move forward/backwards:
        # compute capacities for addition and subtraction
        add_cap = 1 - h_opt
        sub_cap = h_opt
        # Determine sign of diff

            if total_cap > 0:
                h_opt += diff * (-sub_cap) / total_cap

            if time.time() - start > budget_s * 0.96:
                break
            h_rand = random_feasible_h(n, rng)
            try:
                h_opt, c_opt = local_optimize(h_rand, n)
                if c_opt < best_c5:
                    best_c5, best_h = c_opt, h_opt
            except Exception:
                pass

    # Fallback in the very unlikely event nothing was found
    if best_h is None:
        n = 45
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)

    # Final sanity enforcement of the constraints
    n_out = len(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    target_sum = n_out / 2.0
    cur_sum = np.sum(best_h)
    if not np.isclose(cur_sum, target_sum, atol=1e-5):
        best_h += (target_sum - cur_sum) / n_out
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n_out
