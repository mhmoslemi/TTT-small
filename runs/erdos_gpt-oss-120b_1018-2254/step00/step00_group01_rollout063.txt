def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time

    start = time.time()
    np.random.seed(seed)

    n = 200
    target = n // 2
    dx = 2.0 / n

    # function to compute C5
    def compute_c5(h):
        # compute overlap via correlation with self
        conv = np.correlate(h, h, mode='full')
        # exclude center
        # find min overlap for all non-zero shifts
        # but for binary h, conv values are ints
        # compute min among all indices except center
        # The center is conv[n-1]
        # we can use np.min on array where mask true
        mask = np.arange(2*n-1) != n-1
        min_overlap = conv[mask].min()
        # compute C5
        return dx * (target - min_overlap)

    # initial solution: random binary vector with target ones
    h = np.zeros(n, dtype=int)
    h[:target] = 1
    np.random.shuffle(h)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # optionally also start from initial_h_values if exists:
    if 'initial_h_values' in globals():
        init_h = initial_h_values
        # if length matches, use it
        if init_h.shape[0] == n:
            # Round to integer? Possibly not binary; we can project to binary but ignore.

            # Attempt to use as is (clip to 0-1, scale to sum target using water-filling)
            init_h = np.clip(init_h, 0, 1)
            # Force sum to target by scaling? Actually we need binary; skip.
        elif init_h.shape[0] < n:
            # upsample: maybe repeat to fill
            repeats = n // init_h.shape[0] + 1
            hh = np.tile(init_h, repeats)[:n]
            # adjust sum
            # convert to binary:
            idx = np.argsort(-hh)[:target]
            h2 = np.zeros(n, dtype=int)
            h2[idx] = 1
            # evaluate
            c2 = compute_c5(h2)
            if c2 < best_c5:
                best_h = h2.copy()
                best_c5 = c2
                h = h2.copy()
    # else, maybe use given

    # primary search loop
    iteration = 0
    while time.time() - start < budget_s * 0.98:
        # compute overlap array and min shift
        conv = np.correlate(h, h, mode='full')
        # compute minimal overlap and shift list
        # exclude center
        mask = np.arange(2*n-1) != n-1
        min_overlap = conv[mask].min()
        # get indices where min overlap occurs
        min_idxs = np.where(conv == min_overlap)[0]
        # ignore center index if in it
        min_idxs = min_idxs[min_idxs != n-1]
        # pick a shift index random
        idx = np.random.choice(min_idxs)
        # convert idx to shift d (positive or negative)
        d = idx - (n-1)  # can be negative
        # Ensure d != 0.
        if d == 0:
            # choose random shift maybe
            d = np.random.randint(1, n)
        # Determine candidate indices to swap
        # we will consider only forward case for d>0; for d<0 we can mirror
        if d > 0:
            # indices i in [0,n-d-1] where h[i]==1 and h[i+d]==0
            i_candidates = np.where((h[:n-d] == 1) & (h[d:] == 0))[0]
            # indices j where h[j]==0 and h[j+d]==1
            j_candidates = np.where((h[:n-d] == 0) & (h[d:] == 1))[0]
        else:
            d_abs = -d
            # For negative shift, we consider i where h[i]==1 and h[i-d]==0 => same as checking forward shift on reversed? Simpler: we can treat as positive by swapping roles.
            # Equivalent to forward shift d_abs but with indices offset.
            i_candidates = np.where((h[d_abs:] == 1) & (h[:n-d_abs] == 0))[0]  # i where i >= d_abs
            # Actually for negative shift, we need i such that h[i]==1 and h[i+d]==0 (where d negative => i+d < i)
            # So i in [ -d .. n-1 ] (i+d >=0). Equivalent to i' = i+d: for clarity we can mirror.
            # Simpler: use symmetry: treat shift D = -d positive, then condition becomes h[i] == 1 and h[i - D] == 0 for i>=D.
            # So i_candidates = indices i where i>=D and h[i]==1 and h[i-D]==0.
            i_candidates = np.where((h[d_abs:] == 1) & (h[:n-d_abs] == 0))[0] + d_abs
            # j candidates: h[j]==0 and h[j-D]==1 for j>=D
            j_candidates = np.where((h[d_abs:] == 0) & (h[:n-d_abs] == 1))[0] + d_abs

        # If both candidate sets non-empty, choose one each
        if i_candidates.size > 0 and j_candidates.size > 0:
            i = np.random.choice(i_candidates)
            j = np.random.choice(j_candidates)
            # swap
            h_candidate = h.copy()
            h_candidate[i] = 0
            h_candidate[j] = 1
        else:
            # fallback random swap
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            h_candidate = h.copy()
            h_candidate[i] = 0
            h_candidate[j] = 1

        # evaluate
        new_c5 = compute_c5(h_candidate)
        iteration += 1

        # Accept if better
        if new_c5 < best_c5 - 1e-12:
            best_c5 = new_c5
            best_h = h_candidate.copy()
        # Accept move if improves current h (not necessarily global) else maybe revert.
        # let's keep h as candidate if new_c5 <= current c5
        cur_c5 = compute_c5(h)  # maybe we can store but compute earlier
        if new_c5 <= cur_c5:
            h = h_candidate
        else:
            # maybe accept with small probability
            # Use temperature schedule
            # compute temperature decays over iterations
            elapsed = time.time() - start
            frac = elapsed / (budget_s * 0.98)
            temp = max(0.001, 0.05 * (1 - frac))
            prob = np.exp(-(new_c5 - cur_c5) / temp) if temp>0 else 0.0
            if np.random.rand() < prob:
                h = h_candidate
        # optional early breaking if best_c5 less than target 0.3808 (some threshold).
        if best_c5 <= 0.3808 + 1e-6:
            break

    # Return solution
    return (best_h.astype(float), float(best_c5), n)

def project_onto_bounds_sum(y, w):
    lo = -np.max(y) - 2.0
    hi = np.max(y) + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2
        x = np.clip(y - mid, 0, 1)
        s = np.sum(x)
        if s > w:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo+hi)/2, 0, 1)
    # after binary may have slight error due to rounding; adjust by a small correction
    # compute diff = w - sum(x)
    diff = w - np.sum(x)
    if abs(diff) > 1e-9:
        # distribute diff to free variables not at bounds
        # find indices where 0 < x < 1
        free = (x > 1e-8) & (x < 1 - 1e-8)
        cnt = np.sum(free)
        if cnt > 0:
            x[free] += diff / cnt
            # clip again just in case
            x = np.clip(x, 0, 1)
    return x

import numpy as np
import time
import math

def _project_onto_sum_and_bounds(v, target):
    """Project vector v onto the box [0,1]^n while preserving the sum = target."""
    # binary search for lambda such that sum(clip(v - lam,0,1)) = target
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction for numerical drift
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        free = (x > 1e-8) & (x < 1.0 - 1e-8)
        cnt = int(free.sum())
        if cnt > 0:
            x[free] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
        else:
            # adjust a random coordinate if nothing is free
            i = np.random.randint(len(x))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x


def _c5_of_h(h):
    """Compute C5 using the definition from the statement."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx
