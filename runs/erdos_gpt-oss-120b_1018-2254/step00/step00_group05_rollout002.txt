def target_move(h):
    N = len(h)
    dx = 2.0/N
    corr = np.correlate(h, 1-h, mode='full') * dx
    k = np.argmax(corr) - (N - 1) # because np.correlate full index shift = like lag = i - j? Actually np.correlate returns length 2N-1 with lag = -(N-1) to (N-1). At index idx = lag + (N-1). So lag = idx - (N-1).
    lag = k  # could be negative.
    # compute gradient
    grad = np.zeros(N)
    if lag >= 0:
        # for i from 0 to N-lag-1: term (1 - h[i+lag])
        grad[:N-lag] += 1 - h[lag:]  # (1 - h[i+lag])
        # for i >= lag: term -h[i - lag] (negative)
        grad[lag:] -= h[:N-lag]
    else:  # lag negative => we can treat symmetrical
        # define L = -lag >0; then c_lag = sum_i h_i * (1 - h_{i-lag})? Actually h_{i+lag} where lag negative means i+lag < i.
        # Use identity: c_{-L} = sum_i h_i * (1 - h_{i - L})
        L = -lag
        # for i from L to N-1: have (1 - h[i - L])
        grad[L:] += 1 - h[:N-L]
        # for i < N-L: term -h[i + L]
        grad[:N-L] -= h[L:]
    # We now have gradient of c_max w.r.t h.
    # Determine direction to reduce c_max: move opposite of grad.
    # So indices where grad>0: decreasing h reduces c.
    # indices where grad<0: increasing h reduces c.
    up_indices = np.where(grad < -1e-8)[0]  # we can increase
    down_indices = np.where(grad > 1e-8)[0]  # we can decrease
    if up_indices.size ==0 or down_indices.size ==0:

    i_up = up_indices[np.random.randint(up_indices.size)]
    i_down = down_indices[np.random.randint(down_indices.size)]
    # Determine allowed delta
    delta_up = 1 - h[i_up]
    delta_down = h[i_down]
    max_delta = min(delta_up, delta_down, 0.05)  # limited step
    if max_delta <= 1e-12:

    h_new = h.copy()

import time
import numpy as np
from math import exp

# -------------------- Helper Functions --------------------

def compute_c5(h):
    """Return the overlap constant C5 for a discrete h."""
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


def project_onto_bounds_and_sum(y, target_sum):
    """
    Project vector y onto the feasible set:
        0 <= h_i <= 1  and  sum(h) = target_sum
    using a bisection on the Lagrange multiplier.
    """
    lo = -1e6
    hi = 1e6
    for _ in range(80):
        lam = (lo + hi) * 0.5
        h = np.clip(y - lam, 0.0, 1.0)
        s = h.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)


def propose_swap_move(h, max_step):
    """Randomly transfer mass between two coordinates while preserving the sum."""
    N = h.shape[0]
    i, j = np.random.randint(N, size=2)
    if i == j:
        return None
    # possible transfer i -> j
    max_up_i = 1.0 - h[i]
    max_down_j = h[j]
    transfer_ij = min(max_up_i, max_down_j, max_step)
    # possible transfer j -> i
    max_up_j = 1.0 - h[j]
    max_down_i = h[i]
    transfer_ji = min(max_up_j, max_down_i, max_step)

    options = []
    if transfer_ij > 1e-12:
        options.append((i, j, transfer_ij))
    if transfer_ji > 1e-12:
        options.append((j, i, transfer_ji))
    if not options:
        return None
    i_up, i_down, max_transfer = options[np.random.randint(len(options))]
    delta = np.random.rand() * max_transfer
    h_new = h.copy()
    h_new[i_up] += delta
    h_new[i_down] -= delta
    return h_new


def targeted_move(h, max_step):
    """
    Move mass in a direction that directly reduces the maximal correlation.
    Uses the lag that attains the current C5.
    """
    N = h.shape[0]
    # correlation and lag of the worst shift
    corr_full = np.correlate(h, 1.0 - h, mode="full")
    idx_max = int(np.argmax(corr_full))
    lag = idx_max - (N - 1)   # lag can be negative

    grad = np.zeros(N)
    if lag >= 0:
        # (1 - h[i+lag]) contribution
        grad[: N - lag] += 1.0 - h[lag:]
        # -h[i-lag] contribution
        grad[lag:] -= h[: N - lag]
    else:
        L = -lag
        grad[L:] += 1.0 - h[: N - L]
        grad[: N - L] -= h[L:]

    up_idx = np.where(grad < -1e-12)[0]
    down_idx = np.where(grad > 1e-12)[0]
    if up_idx.size == 0 or down_idx.size == 0:
        return None

    i_up = up_idx[np.random.randint(up_idx.size)]
    i_down = down_idx[np.random.randint(down_idx.size)]

    max_up = 1.0 - h[i_up]
    max_down = h[i_down]
    max_transfer = min(max_up, max_down, max_step)
    if max_transfer <= 1e-12:
        return None

    delta = np.random.rand() * max_transfer
    h_new = h.copy()
    h_new[i_up] += delta
    h_new[i_down] -= delta
    return h_new


# -------------------- Main Optimization --------------------

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to lower the Erdős C5 constant.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start = time.time()

    # Initialise h
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        N = h0.shape[0]
    else:
        N = 100
        h0 = np.random.rand(N)

    target_sum = N / 2.0
    h0 = project_onto_bounds_and_sum(h0, target_sum)

    cur_h = h0.copy()
    cur_score = compute_c5(cur_h)

    best_h = cur_h.copy()
    best_score = cur_score

    # Parameters for the stochastic search
    max_step = 0.08          # maximum amount transferred in a single move
    prob_target = 0.6        # chance to use targeted move
    temp_initial = 0.05      # initial temperature scale for SA acceptance

    # Main loop respects the supplied time budget
    while time.time() - start < budget_s * 0.99:
        # Choose move type
        if np.random.rand() < prob_target:
            h_new = targeted_move(cur_h, max_step)
        else:
            h_new = propose_swap_move(cur_h, max_step)

        if h_new is None:
            continue

        new_score = compute_c5(h_new)

        # Accept if better; otherwise use a simple annealing probability
        if new_score < cur_score:
            cur_h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score
        else:

            # Linear cooling to zero

            if np.random.rand() < exp(-(new_score - cur_score) / temp):
                cur_h = h_new
                cur_score = new_score

        # Optional occasional random restart if stuck (simple heuristic)

            y = np.random.rand(N)
            cur_h = project_onto_bounds_and_sum(y, target_sum)
            cur_score = compute_c5(cur_h)

    # Ensure feasibility of the final answer
    best_h = project_onto_bounds_and_sum(best_h, target_sum)

    # Compute final C5 using the provided evaluator if available
    if 'evaluate_erdos_solution' in globals():
        final_c5 = evaluate_erdos_solution(best_h)
    else:
        final_c5 = compute_c5(best_h)

    return best_h, final_c5, N
