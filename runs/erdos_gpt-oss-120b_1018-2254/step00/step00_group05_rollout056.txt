import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # discretisation parameters (can be adjusted)
    K = 20               # number of segments (must be even for binary start)
    m = 10               # points per segment
    n = K * m            # total sample points
    dx = 2.0 / n
    target_sum = n / 2.0

    # expand segment values to full vector
    def expand(params):
        return np.repeat(params, m)

    # raw max correlation (without dx)
    def max_corr(params):
        vec = expand(params)
        corr = np.correlate(vec, 1 - vec, mode='full')
        return np.max(corr)

    # smooth approximation of max for the optimiser
    beta = 20.0
    def smooth_obj(params):
        vec = expand(params)
        corr = np.correlate(vec, 1 - vec, mode='full')
        mx = np.max(corr)
        # log‑sum‑exp stabilised
        exp_terms = np.exp(beta * (corr - mx))
        return mx + np.log(np.sum(exp_terms)) / beta

    # equality constraint: sum(params) = K/2
    eq_cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - K / 2.0}
    bounds = [(0.0, 1.0)] * K

    # construct an initial guess
    if 'initial_h_values' in globals():
        init_full = np.array(initial_h_values, dtype=float)
        # rescale to satisfy total integral if needed
        cur_sum = np.sum(init_full)
