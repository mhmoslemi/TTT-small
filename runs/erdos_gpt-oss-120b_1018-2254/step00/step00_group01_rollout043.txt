import time
import numpy as np

def _project_to_sum_and_bounds(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project vector v onto the hyperplane sum(x)=target_sum with bounds 0<=x<=1."""
    # Binary search for Lagrange multiplier lambda such that sum(clip(v-lambda,0,1)) = target_sum
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = np.sum(x)
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

    # Tiny residual correction
    diff = target_sum - np.sum(x)
    if abs(diff) > 1e-12:
        free = (x > 0.0) & (x < 1.0)
        cnt = np.count_nonzero(free)
        if cnt == 0:
            # adjust a single entry if everything is at a bound
            idx = np.argmin(x) if diff > 0 else np.argmax(x)
            x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
        else:
            inc = diff / cnt
            x[free] += inc
            x = np.clip(x, 0.0, 1.0)
    return x

def _c5_value(h: np.ndarray) -> float:
    """Compute C₅ for a given step function h sampled on [0,2]."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)
