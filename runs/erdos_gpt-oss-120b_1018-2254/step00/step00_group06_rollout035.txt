def project_onto_simplex_box(v, target):
    # v: initial vector possibly outside [0,1] and sum not target
    # Works by clipping to [0,1] and then adjusting the remainder.
    # iterative method:
    x = np.clip(v, 0, 1)
    # compute current sum
    current = x.sum()
    # while not close:
    for _ in range(100):
        diff = target - current
        if abs(diff) <= 1e-10:
            break
        # Determine free indices that are not at bounds
        if diff > 0:
            free = np.where(x < 1 - 1e-12)[0]
        else:
            free = np.where(x > 1e-12)[0]
        if free.size == 0:
            # cannot adjust further; break
            break
        # distribute diff uniformly over free indices
        delta = diff / free.size
        # Add delta to free indices
        x[free] += delta
        # Clip again
        np.clip(x, 0, 1, out=x)
        current = x.sum()
    # final adjust small diff if any by adding to a random free index
    diff = target - current
    if abs(diff) > 1e-12:
        # try to add diff to an index that can accommodate
        if diff > 0:
            candidates = np.where(x < 1 - 1e-12)[0]
        else:
            candidates = np.where(x > 1e-12)[0]
        if candidates.size > 0:
            idx = candidates[0]
            x[idx] = np.clip(x[idx] + diff, 0, 1)
        else:
            # As fallback, scale all values slightly to match sum exactly (tiny)
            scale = target / x.sum()
            x *= scale
            # Clip again
            np.clip(x, 0, 1, out=x)
    return x

import time
import numpy as np

def compute_c5(h):
    """Return the C5 value for a given step function vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_onto_simplex_box(v, target):
    """Project vector v onto the box [0,1]^n while forcing sum(v)=target."""
    x = np.clip(v, 0.0, 1.0)
    cur = x.sum()
    # Iteratively distribute the difference uniformly over non‑bound entries.
    for _ in range(100):
        diff = target - cur
        if abs(diff) <= 1e-12:
            break
        if diff > 0:
            free = np.where(x < 1.0 - 1e-12)[0]
        else:
            free = np.where(x > 1e-12)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        x[free] += delta
        np.clip(x, 0.0, 1.0, out=x)
        cur = x.sum()
    # Final tiny correction if needed.
    diff = target - cur
    if abs(diff) > 1e-12:
        if diff > 0:
            cand = np.where(x < 1.0 - 1e-12)[0]
        else:
            cand = np.where(x > 1e-12)[0]
        if cand.size:
            idx = cand[0]
            x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
        else:
            # fallback: scale everything (should be rare)
            scale = target / x.sum()
            x *= scale
            np.clip(x, 0.0, 1.0, out=x)
    return x

def random_feasible(n, target):
    """Generate a random feasible vector of length n with given target sum."""
    v = np.random.exponential(1.0, size=n)
    h = target * v / v.sum()
    return project_onto_simplex_box(h, target)

def tournament_select(pop, scores, size):
    """Return a copy of the winner of a tournament of given size."""
    idx = np.random.choice(len(pop), size=size, replace=False)
    best = idx[0]
    best_score = scores[best]
    for i in idx[1:]:
        if scores[i] < best_score:
            best = i
            best_score = scores[i]
    return pop[best].copy()

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    np.random.seed(seed)

    # Determine number of points
    if "initial_h_values" in globals():
        n = int(len(initial_h_values))
    else:
        n = 100  # fallback size (even for easy binary constructions)

    target_sum = n / 2.0
    pop_size = 200
    elite_size = 1
    crossover_rate = 0.7
    mutation_rate = 0.02
    mutation_sigma = 0.1
    tournament_k = 3

    # Build initial population
    pop = []
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
        if h0.shape[0] != n:
            # If length mismatches, generate a fresh feasible vector
            h0 = random_feasible(n, target_sum)
        else:
            h0 = project_onto_simplex_box(h0, target_sum)
        pop.append(h0)
    while len(pop) < pop_size:
        pop.append(random_feasible(n, target_sum))

    # Initial evaluation
    scores = np.array([compute_c5(ind) for ind in pop])
    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_score = float(scores[best_idx])

    generation = 0
    # Main GA loop
    while time.time() - start < budget_s - 0.5:
        generation += 1
        new_pop = []

        # Elitism: keep the best individual
        elite = [best_h.copy()]

        # Generate offspring
        while len(new_pop) + elite_size < pop_size:
            parent1 = tournament_select(pop, scores, tournament_k)
            parent2 = tournament_select(pop, scores, tournament_k)

            child = parent1.copy()
            if np.random.rand() < crossover_rate:
                cut = np.random.randint(1, n)  # split point
                child = np.concatenate([parent1[:cut], parent2[cut:]])

            # Mutation
            mutate_mask = np.random.rand(n) < mutation_rate
            if mutate_mask.any():
                noise = np.random.randn(n) * mutation_sigma
                child = child + mutate_mask * noise

            # Repair to feasible region
            child = project_onto_simplex_box(child, target_sum)
            new_pop.append(child)

        # Form next generation
        pop = elite + new_pop
        scores = np.array([compute_c5(ind) for ind in pop])

        # Update best
        cur_best_idx = int(np.argmin(scores))
        cur_best_score = float(scores[cur_best_idx])
        if cur_best_score < best_score:
            best_score = cur_best_score
            best_h = pop[cur_best_idx].copy()
            best_idx = cur_best_idx

        # Early stop if target reached
        if best_score <= 0.3808:
            break

    # Local refinement (mass transfer between two positions)
    while time.time() - start < budget_s - 0.2:
        h = best_h.copy()
        i, j = np.random.choice(n, size=2, replace=False)
        # Amount we can move from i to j while staying in [0,1]
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        delta = np.random.rand() * max_transfer * 0.5  # up to half of the possible transfer
        h[i] -= delta
        h[j] += delta
        cost = compute_c5(h)
        if cost < best_score:
            best_score = cost
            best_h = h

    return (best_h, best_score, n)
