    # get indices sorted descending

def repair_binary(h, target_ones):
    # h is binary 0/1 float array but may have sum diff; adjust by flipping bits
    current = int(round(np.sum(h)))
    diff = target_ones - current
    if diff > 0:
        # need to turn diff zeros to ones
        zeros = np.where(h < 0.5)[0]
        if len(zeros) < diff:
            # just set all zeros
            h[zeros] = 1.0
        else:
            chosen = np.random.choice(zeros, diff, replace=False)
            h[chosen] = 1.0
    elif diff < 0:
        # need to turn -diff ones to zeros
        ones = np.where(h > 0.5)[0]
        if len(ones) < -diff:
            h[ones] = 0.0
        else:

    max_val = np.max(conv[n-1:])  # non-negative shifts
    return max_val * (2.0 / n)

def crossover(parent_a, parent_b):
    # Convert to 1D arrays
    n = len(parent_a)
    point = np.random.randint(1, n)  # not at extreme
    child = np.concatenate([parent_a[:point], parent_b[point:]])
    # binary may have values not 0/1 due to different composition: just keep as 0/1
    # Already parent_a and parent_b are binary, so child is binary.
    return child

def mutate_swap(ind):
    # swap values at two indices where they differ
    n = len(ind)
    # find indices where 0 or 1
    ones = np.where(ind > 0.5)[0]
    zeros = np.where(ind <= 0.5)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return ind  # nothing to swap
    i = np.random.choice(ones)
    j = np.random.choice(zeros)
    # swap
    ind[i], ind[j] = ind[j], ind[i]
    return ind

def select_tournament(scores, population, k=5):
    # lower score is better
    participants = np.random.choice(len(population), k, replace=False)
    best_idx = participants[np.argmin([scores[i] for i in participants])]
    # return copy
    return population[best_idx].copy()

def binary_from_initial(arr, target_ones):
    """Convert a continuous initial array to a binary vector with exact sum."""
    idx = np.argsort(-arr)  # descending order
    h = np.zeros_like(arr, dtype=np.float64)
    h[idx[:target_ones]] = 1.0
    return h

def repair_binary(h, target_ones):
    """Flip bits to enforce the exact number of ones."""
    cur = int(round(h.sum()))
    diff = target_ones - cur
    if diff > 0:
        zeros = np.where(h < 0.5)[0]
        if len(zeros) >= diff:
            chosen = np.random.choice(zeros, diff, replace=False)
            h[chosen] = 1.0
    elif diff < 0:
        ones = np.where(h > 0.5)[0]
        if len(ones) >= -diff:
            chosen = np.random.choice(ones, -diff, replace=False)
            h[chosen] = 0.0
    # ensure strict 0/1
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def compute_c5(h):
    """C5 value for a given (binary or continuous) step function."""
    n = len(h)
    conv = np.correlate(h, 1.0 - h, mode='full')
    # consider only non‑negative shifts (index n‑1 onward)
    max_val = np.max(conv[n - 1 :])
    return max_val * (2.0 / n)

def fitness(h):
    """Fitness to minimise – the C5 bound."""
    return compute_c5(h)

def crossover(a, b):
    """One‑point crossover producing a binary offspring (may violate sum)."""
    n = len(a)
    pt = np.random.randint(1, n)  # avoid trivial cuts
    child = np.concatenate([a[:pt], b[pt:]])
    return child

def mutate_swap(ind):
    """Swap a 1 and a 0 – preserves the total number of ones."""
    ones = np.where(ind > 0.5)[0]
    zeros = np.where(ind <= 0.5)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return ind
    i = np.random.choice(ones)
    j = np.random.choice(zeros)
    ind[i], ind[j] = ind[j], ind[i]
    return ind

def select_tournament(scores, pop, k=5):
    """Tournament selection – lower score is better."""
    participants = np.random.choice(len(pop), k, replace=False)
    best = participants[0]
    best_score = scores[best]
    for idx in participants[1:]:
        if scores[idx] < best_score:
            best = idx
            best_score = scores[idx]
    return pop[best].copy()

# ---------------- Main entry point ----------------

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)

    # Determine problem size
    try:
        n_points = len(initial_h_values)  # provided globally if available
    except NameError:
        n_points = 84                     # fall back to a modest even size
    if n_points % 2 != 0:
        n_points += 1
    target_ones = n_points // 2

    # Initialise population
    if 'initial_h_values' in globals():
        init_arr = np.asarray(initial_h_values, dtype=np.float64)
        base = binary_from_initial(init_arr, target_ones)
    else:
        base = np.zeros(n_points, dtype=np.float64)
        base[:target_ones] = 1.0
        np.random.shuffle(base)

    pop_size = 30
    population = [base.copy()]
    for _ in range(pop_size - 1):
        ind = np.zeros(n_points, dtype=np.float64)
        ind[:target_ones] = 1.0
        np.random.shuffle(ind)
        population.append(ind)
    population = [repair_binary(p, target_ones) for p in population]

    best_h = None
    best_score = float('inf')
    start = time.time()
    target_c5 = 0.3808  # early‑stop threshold if achieved

    # Evolutionary loop
    while time.time() - start < budget_s - 5.0:   # keep a small safety margin
        scores = [fitness(ind) for ind in population]

        # Update global best
        for ind, sc in zip(population, scores):
            if sc < best_score:
                best_score = sc
                best_h = ind.copy()
                if best_score <= target_c5:
                    break
        if best_score <= target_c5:
            break

        # Elitism: keep two best individuals
        elite_idxs = np.argsort(scores)[:2]
        new_pop = [population[i].copy() for i in elite_idxs]

        # Fill the rest of the population
        while len(new_pop) < pop_size:
            p1 = select_tournament(scores, population, k=5)
            p2 = select_tournament(scores, population, k=5)
            child = crossover(p1, p2)
            if random.random() < 0.7:
                child = mutate_swap(child)
            child = repair_binary(child, target_ones)
            new_pop.append(child)

        population = new_pop

    # Final evaluation (using the provided helper if available)
    try:
        c5_bound, _ = evaluate_erdos_solution(best_h)
    except Exception:
        c5_bound = compute_c5(best_h)

    return (best_h.astype(float), float(c5_bound), int(n_points))
