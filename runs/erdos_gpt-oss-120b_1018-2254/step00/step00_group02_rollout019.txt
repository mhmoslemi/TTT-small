def _project_to_sum(y, target):
    # Project y onto [0,1]^n intersect hyperplane sum == target using bisection on shift.
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    # ensure proper bounds
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(y + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)

def _c5_value(h):
    # compute C5 using correlation definition
    c = np.correlate(h, 1 - h, mode='full')
    return np.max(c) * (2.0 / h.size)

import numpy as np
import time

def _project_to_sum(y, target):
    """Project vector y onto the box [0,1]^n while enforcing sum(y)=target."""
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(y + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)

def _c5_value(h):
