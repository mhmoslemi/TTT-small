    # lower <= x_i <= upper, sum = desired_sum
    # Use algorithm: clip to bounds, then adjust by scaling residual.
    if desired_sum is None:
        desired_sum = np.sum(x)
    x = np.clip(x, lower, upper)
    total = np.sum(x)
    # Adjust via scaling if not far from bounds:
    for _ in range(10):
        if abs(total - desired_sum) < 1e-9:
            break
        # compute scaling factor for interior entries:
        # compute mask: not at bounds
        mask = (x > lower + 1e-12) & (x < upper - 1e-12)
        if np.any(mask):
            # scaling factor to move remaining sum
            factor = (desired_sum - np.sum(x[~mask])) / np.sum(x[mask])
            # multiply interior entries:
            x[mask] = np.clip(x[mask] * factor, lower, upper)
        else:
            # all at bounds but totals off: adjust uniformly (but will exceed)
            # fallback: distribute diff equally to all (clipping later)
            x = np.clip(x + (desired_sum - total)/len(x), lower, upper)

    x = np.clip(x, lower, upper)

        x = np.clip(x, lower, upper)
    return x

def subgradient_step(h, desired_sum, step):
    n = len(h)
    # compute correlation vector
    corr = np.correlate(h, 1 - h, mode='full')
    # identify index of max
    idx = np.argmax(corr)
    lag = idx - (n - 1)  # integer shift
    # compute gradient
    grad = np.zeros_like(h)
    if lag >= 0:
        k = lag
        if k < n:
            grad[:n - k] += 1 - h[k:]
            grad[k:] -= h[:n - k]
    else:
        kk = -lag
        # grad[kk:] plus
        grad[kk:] += 1 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    # update
    h_new = h - step * grad
    # project back
    h_new = project(h_new, desired_sum)
    return h_new

    n = len(h)
    # find lag with maximal correlation
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    lag = idx - (n - 1)                # integer shift k
    grad = np.zeros_like(h)

    if lag >= 0:
        k = lag
        if k < n:
            grad[: n - k] += 1.0 - h[k:]
            grad[k:]      -= h[: n - k]
    else:
        kk = -lag
        grad[kk:]        += 1.0 - h[: n - kk]
        grad[: n - kk]   -= h[kk:]

    h_new = h - step * grad
    return project_vec(h_new, desired_sum)


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] with ∫h = 1 to minimize
    C5 = max_k ∫ h(x)(1 - h(x+k)) dx.

    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise ---------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.shape[0]
    else:
        n = 70
        h = np.full(n, 0.5, dtype=float)

    desired_sum = n / 2.0
    h = project_vec(h, desired_sum)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # ------------------------------------------------------------------
    # subgradient descent phase (≈40% of time) -------------------------
    start_time = time.time()
    sub_end = start_time + budget_s * 0.4
    sub_iter = 0
    cur_c5 = best_c5

    while time.time() < sub_end:
        sub_iter += 1
        step = 0.20 / np.sqrt(sub_iter + 1.0)    # decreasing step size

        h_new = subgradient_step_vec(h, desired_sum, step)
        new_c5 = compute_c5(h_new)

        # store improvement
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            if best_c5 <= 0.3808:
                break

        # accept if improvement, otherwise occasional uphill move
        if new_c5 < cur_c5:
            h = h_new
            cur_c5 = new_c5
        else:
            diff = new_c5 - cur_c5
            # tiny probability to accept uphill move
            if rng.random() < np.exp(-diff / (0.01 + 0.001 * sub_iter)):
                h = h_new
                cur_c5 = new_c5

    # ------------------------------------------------------------------
    # hill‑climbing / stochastic refinement phase (≈55% of time) -------
    hill_end = start_time + budget_s * 0.95
    iteration = 0
    last_improve = 0
    restart_interval = 12000

    while time.time() < hill_end:
        iteration += 1
        elapsed = time.time() - start_time
        frac = elapsed / (budget_s * 0.95)
        temp = max(0.001, 0.01 * (1.0 - frac))   # annealing temperature

        # propose a mass transfer between two points
        delta = rng.uniform(0.001, 0.03)
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue

        allowed = min(delta, h[i], 1.0 - h[j])
        if allowed <= 0.0:
            continue

        h_new = h.copy()
        h_new[i] -= allowed
        h_new[j] += allowed

        new_c5 = compute_c5(h_new)
        cur_c5 = compute_c5(h)

        if new_c5 < cur_c5:
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
                last_improve = iteration
                if best_c5 <= 0.3808:
                    break
        else:
            diff = new_c5 - cur_c5
            if temp > 0.0 and rng.random() < np.exp(-diff / temp):
                h = h_new

        # restart if stagnant for a while
        if iteration - last_improve > restart_interval:
            h = rng.random(n)
            h = project_vec(h, desired_sum)
            last_improve = iteration

    # ------------------------------------------------------------------
    # final projection and result ----------------------------------------
    best_h = project_vec(best_h, desired_sum)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
