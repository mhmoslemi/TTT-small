    t = cp.Variable(integer=True)
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == N/2,
        t >= 0,
    ]

            w >= 0,
            w <= 1,
        ]

def trapezoid_h(N, r):
    dx = 2.0 / N
    x = (np.arange(N) + 0.5) * dx
    if r <= 0:
        a = 0.5
        h = np.full(N, a)
    else:
        a = 1.0 / (2.0 - r)
        w = 2.0 - 2.0*r
        # plateau region: (r <= x <= r + w) = (r <= x <= 2 - r)
        h = np.empty(N)
        left = x < r
        plateau = (x >= r) & (x <= 2 - r)
        right = x > 2 - r
        # left ramp
        h[left] = a * (x[left] / r)
        # plateau
        h[plateau] = a
        # right ramp
        h[right] = a * ((2 - x[right]) / r)
    # correct sum (scaling)
    target_sum = N / 2.0
    current_sum = h.sum()
    # scaling factor = target_sum / current_sum
    scale = target_sum / current_sum
    h *= scale
    # clip to [0,1] after scaling (unlikely to violate)
    np.clip(h,0,1,inplace=True)
    # ensure sum after clipping maybe small error; but we assume OK
    return h

import numpy as np, time

def compute_c5(h):
    N = h.shape[0]
    dx = 2.0 / N
    conv = np.correlate(h, 1 - h, mode='full')
    return np.max(conv) * dx

def trapezoid_h(N, r):
    dx = 2.0 / N
    x = (np.arange(N) + 0.5) * dx
    if r <= 0:
        a = 0.5
        h = np.full(N, a)
    else:
        a = 1.0 / (2.0 - r)
        left = x < r
        plateau = (x >= r) & (x <= 2 - r)
        right = x > 2 - r
        h = np.empty(N)
        # avoid divide by zero
        # left ramp
        h[left] = a * (x[left] / r)
        h[plateau] = a
        h[right] = a * ((2 - x[right]) / r)
    # scale to have sum N/2 (integral 1)
    target_sum = N / 2.0
    cur_sum = h.sum()
    scale = target_sum / cur_sum
    h = h * scale
    # enforce bounds
    np.clip(h, 0.0, 1.0, out=h)
    return h

def find_best_h(N, budget_s):
    start = time.time()
    best_c5 = np.inf
    best_h = None
    best_r = None
    # coarse grid
    n_grid = 51

        c5 = compute_c5(h)

            best_h = h.copy()

    # refine around best_r
    # define refine range +/- step
    step = 1.0 / (n_grid - 1)
    lo = max(0.0, best_r - step)
    hi = min(1.0, best_r + step)
    # golden-section iterations
    phi = (np.sqrt(5) - 1) / 2  # 0.618...
    tol = 1e-4
    while hi - lo > tol and (time.time() - start) < (budget_s - 0.5):
        r1 = lo + (1 - phi) * (hi - lo)
        r2 = lo + phi * (hi - lo)
        h1 = trapezoid_h(N, r1)

import numpy as np
import time
from math import sqrt

def _compute_c5(h: np.ndarray) -> float:
    """Return the C5 value for a given discretised function h."""
    n = h.size
    dx = 2.0 / n
    # linear correlation with zero padding (full mode)
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def _trapezoid_h(n: int, r: float) -> np.ndarray:
    """
    Build a trapezoidal step function on n samples.
    r : length of each ramp (0 <= r <= 1). 0 gives the constant 0.5 function,
        1 gives the pure triangular (tent) function.
    The function is symmetric around x=1 and has total integral 1.
    """
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx  # mid‑point sampling

    if r <= 0.0:                     # constant 0.5
        h = np.full(n, 0.5, dtype=float)
    else:
        a = 1.0 / (2.0 - r)          # plateau height
        left = x < r
        plateau = (x >= r) & (x <= 2.0 - r)
        right = x > 2.0 - r
        h = np.empty(n, dtype=float)
        # left ramp 0 -> a
        h[left] = a * (x[left] / r)
        # flat top at height a
        h[plateau] = a
        # right ramp a -> 0
        h[right] = a * ((2.0 - x[right]) / r)

    # enforce the integral constraint exactly (sum * dx == 1)
    target_sum = n / 2.0
    cur_sum = h.sum()
    scale = target_sum / cur_sum
    h *= scale
    np.clip(h, 0.0, 1.0, out=h)      # safety
    return h


def _best_trapezoid(n: int, budget_s: float) -> tuple:
    """Search the best ramp length r for the given n within the time budget."""
    start = time.time()
    best_c5 = np.inf
    best_h = None
    best_r = None

    # coarse scan
    coarse_steps = 51
    for r in np.linspace(0.0, 1.0, coarse_steps):
        h = _trapezoid_h(n, r)
        c5 = _compute_c5(h)
        if c5 < best_c5:
            best_c5, best_h, best_r = c5, h, r

    # local refinement with golden‑section search
    phi = (sqrt(5.0) - 1.0) / 2.0          # 0.618...
    lo = max(0.0, best_r - 1.0 / (coarse_steps - 1))
    hi = min(1.0, best_r + 1.0 / (coarse_steps - 1))
    tol = 1e-4
    while hi - lo > tol and (time.time() - start) < (budget_s - 0.5):
        r1 = lo + (1.0 - phi) * (hi - lo)
        r2 = lo + phi * (hi - lo)

        h1 = _trapezoid_h(n, r1)
        c5_1 = _compute_c5(h1)
        h2 = _trapezoid_h(n, r2)
        c5_2 = _compute_c5(h2)

        if c5_1 < c5_2:
            hi = r2
            if c5_1 < best_c5:
                best_c5, best_h, best_r = c5_1, h1, r1
        else:
            lo = r1
            if c5_2 < best_c5:
                best_c5, best_h, best_r = c5_2, h2, r2

    return best_h, best_c5, n


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Construct a step function h on [0,2] that yields a small C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)   # deterministic, even if not explicitly used

    # Choose a modest number of samples (less than 1000, >0, even)
    n_points = 200

    # If a parent construction is present, we ignore its length;
    # the new family of trapezoidal shapes is self‑contained.
    h_opt, c5_opt, n_opt = _best_trapezoid(n_points, float(budget_s))

    # Final safety check: enforce exact integral and bounds
    target_sum = n_opt / 2.0
    cur_sum = h_opt.sum()
    if not np.isclose(cur_sum, target_sum, atol=1e-9):
        h_opt *= target_sum / cur_sum
    np.clip(h_opt, 0.0, 1.0, out=h_opt)

    # Return the result
    return h_opt, float(c5_opt), n_opt
