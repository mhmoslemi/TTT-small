import numpy as np, time, random, math
from scipy.optimize import minimize

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def init_binary(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    ones_idx = rng.choice(n, n//2, replace=False)   # n even
    h[ones_idx] = 1.0
    return h

def hill_climb(h, time_limit, start_time):
    n = len(h)
    best_h = h.copy()
    best_val = compute_c5(best_h)
    # Simulated annealing schedule
    iter = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed > time_limit:
            break
        # Choose random i,j (i!=j)
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # Determine maximum transfer from i to j
        max_transfer = min(best_h[i], 1 - best_h[j])
        if max_transfer <= 1e-12:
            continue
        # Transfer amount: random fraction
        eps = max_transfer * np.random.rand()
        # Create candidate
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        val = compute_c5(cand)
        # Accept if better, else maybe with probability
        if val < best_val:
            best_h = cand
            best_val = val
        else:
            # Simulated annealing: prob = exp(-(val - best_val) / T)
            # Temperature schedule: decrease linearly with time
            T = max(0.01, (time_limit - elapsed) / time_limit) * 0.1  # maybe smaller
            if np.exp(-(val - best_val) / T) > np.random.rand():
                best_h = cand
                best_val = val
        iter += 1
    return best_h, best_val
```

def enforce_sum_bounds(h, target_sum, rng):
    # ensures sum==target_sum while staying within [0,1]
    # use iterative adjustment
    current_sum = h.sum()
    diff = target_sum - current_sum
    # while diff not zero within tolerance:
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        # choose random index where we can adjust in direction of diff
        if diff > 0:
            # need increase sum, pick index with h[i] < 1
            candidates = np.where(h < 1 - 1e-12)[0]
        else:
            # need decrease sum, pick index with h[i] > 0
            candidates = np.where(h > 1e-12)[0]
        if len(candidates) == 0:
            break
        i = rng.choice(candidates)
        max_adj = (1 - h[i]) if diff > 0 else h[i]
        adj = min(abs(diff), max_adj)
        h[i] += np.sign(diff) * adj
        diff = target_sum - h.sum()
    return h

import numpy as np
import time
from scipy.optimize import minimize

# Global target sum for equality constraint (set inside run)
TARGET_SUM = None

def compute_c5(arr):
    """Return C5 = dx * max correlation of h with (1-h)."""
    n = arr.shape[0]
    dx = 2.0 / n
    # correlation with zero padding at both ends
    corr = np.correlate(arr, 1 - arr, mode='full')
    return dx * np.max(corr)

def enforce_sum_bounds(vec, target, rng, max_iter=1000):
    """
    Adjust entries of vec (in-place) so that sum(vec) == target
    while keeping 0 <= vec[i] <= 1.
    """
    for _ in range(max_iter):
        diff = target - vec.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            cand = np.where(vec < 1 - 1e-12)[0]
        else:
            cand = np.where(vec > 1e-12)[0]
        if cand.size == 0:
            break
        i = rng.choice(cand)
        max_adj = (1 - vec[i]) if diff > 0 else vec[i]
        adj = min(abs(diff), max_adj)
        vec[i] += np.sign(diff) * adj
    return vec

def random_binary_vector(n, rng):
    """Generate a binary vector of length n with exactly n/2 ones."""
    v = np.zeros(n, dtype=float)
    ones = rng.choice(n, n // 2, replace=False)
    v[ones] = 1.0
    return v

def eq_sum_constraint(x):
    """Equality constraint: sum(x) must equal TARGET_SUM."""
    return np.sum(x) - TARGET_SUM

def run(seed=42, budget_s=1100, **kwargs):
