def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    random.seed(seed)

    # Obtain initial h
    try:
        init = np.array(initial_h_values, dtype=np.float64)
        n = init.size
        h = init.copy()
    except NameError:
        n = 60
        h = rng.random(n)
        h = np.clip(h, 0, 1)
        # Adjust sum to n/2
        target = n/2.0
        # Simple scaling then clip refinement
        s = target / h.sum()
        h *= s
        h = np.clip(h, 0, 1)
        # fix sum by redistribution
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # distribute diff uniformly among non‑boundary entries
            for _ in range(10):
                mask = (h > 0) & (h < 1)
                cnt = np.count_nonzero(mask)
                if cnt == 0:
                    break
                inc = diff / cnt
                h[mask] += inc
                h = np.clip(h, 0, 1)
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break

    # Ensure feasibility
    h = np.clip(h, 0.0, 1.0)
    target_sum = n/2.0
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # adjust first non‑boundary entry
        for idx in range(n):
            if (diff > 0 and h[idx] < 1) or (diff < 0 and h[idx] > 0):
                adj = diff
                if diff > 0:
                    adj = min(adj, 1 - h[idx])
                else:
                    adj = max(adj, -h[idx])
                h[idx] += adj
                diff -= adj
                if abs(diff) < 1e-12:
                    break

    # Initial evaluation
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    curr_h = h.copy()
    curr_c5 = best_c5

    start = time.time()
    # Parameters
    prob_targeted = 0.7
    step_factor = 1.0          # fraction of allowed transfer taken
    target_factor = 1.0
    T0 = 0.05
    T_final = 1e-5
    no_improve = 0
    restart_threshold = 300000  # iterations without improvement before a mild restart

    # Main loop
    iter_count = 0
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.95:
            break
        # temperature schedule (linear)
        frac = elapsed / (budget_s * 0.95)
        T = T0 * (1 - frac) + T_final * frac

        # decide move type
        n = curr_h.size
        i = j = None
        direction = 1   # 1 means transfer from i to j (i decreases, j increases)
        delta = 0.0

        if rng.random() < prob_targeted:
            # targeted move based on current worst shift
            corr = np.correlate(curr_h, 1 - curr_h, mode='full')
            idx_max = int(np.argmax(corr))
            shift = idx_max - (n - 1)
            # Determine range of i where i and i+shift both inside [0,n)
            if shift >= 0:
                i_min = 0
                i_max = n - shift - 1
            else:
                i_min = -shift
                i_max = n - 1
            if i_max >= i_min:
                i = rng.integers(i_min, i_max + 1)
                j = i + shift
                # Transfer from i to j
                max_transfer = min(curr_h[i], 1.0 - curr_h[j])
                if max_transfer <= 0:
                    # fallback to random move
                    i = None
                else:
                    direction = 1
                    delta = rng.random() * max_transfer * target_factor
                    curr_h[i] -= delta
                    curr_h[j] += delta
        if i is None:

            # pick two distinct indices
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            # choose direction randomly
            if rng.random() < 0.5:

                max_transfer = min(curr_h[i], 1.0 - curr_h[j])
                direction = 1
            else:

                max_transfer = min(curr_h[j], 1.0 - curr_h[i])
                direction = -1
            if max_transfer <= 0:

            delta = rng.random() * max_transfer * step_factor
            if direction == 1:
                curr_h[i] -= delta
                curr_h[j] += delta
            else:
                curr_h[i] += delta
                curr_h[j] -= delta

        # evaluate new state
        new_c5 = compute_c5(curr_h)

            if new_c5 < best_c5:

                curr_h[i] -= delta
                curr_h[j] += delta

    # Re‑compute bound for safety
    final_c5 = compute_c5(best_h)

    return best_h, float(final_c5), n

import numpy as np
import time
import random
import math

def _c5_of(h):
    """Return C₅ for a discretized step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)
