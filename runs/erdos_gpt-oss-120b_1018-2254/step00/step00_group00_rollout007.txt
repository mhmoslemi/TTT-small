def hill_climb_step(h, target_sum, dx, time_left):
    # attempt to adjust two random positions
    n = len(h)
    i, j = np.random.choice(n, size=2, replace=False)
    # compute max allowed delta such that 0 <= h_i - delta <= 1 and 0 <= h_j + delta <= 1.
    max_decrease = h[i]  # can remove up to current value
    max_increase = 1.0 - h[j]
    max_delta = min(max_decrease, max_increase)
    if max_delta < 1e-6:
        return h, False
    delta = np.random.rand() * max_delta
    # optionally try both directions: also decreasing j and increasing i.
    # We'll try both later.
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    # evaluate
    c5_new = evaluate(h_new)
    c5_old = evaluate(h) # but we can pass old value to avoid recompute.
    if c5_new < c5_old:
        return h_new, True
    else:

        # Let's try decreasing j and increasing i

        max_increase_i = 1 - h[i]
