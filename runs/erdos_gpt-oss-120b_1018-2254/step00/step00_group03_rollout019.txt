import time
import random
import math
import numpy as np

def compute_c5(h, dx):
    """Maximum overlap integral for a given step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def random_binary_vector(n, ones):
    """Binary vector of length n with exactly `ones` entries equal to 1."""
    h = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, int(ones), replace=False)
    h[idx] = 1.0
    return h


def transfer_move(h):
    """Return a new vector obtained by moving a positive amount from i to j,
    preserving the sum and keeping values in [0,1]. May return None if no
    feasible transfer."""
    n = h.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    # Randomly decide direction of transfer
    if random.random() < 0.5:
        give = min(h[i], 1.0 - h[j])
        if give <= 0.0:
            return None
        delta = np.random.uniform(0.0, give)
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        return new_h
    else:
        give = min(h[j], 1.0 - h[i])
        if give <= 0.0:
            return None
        delta = np.random.uniform(0.0, give)
        new_h = h.copy()
        new_h[j] -= delta
        new_h[i] += delta
        return new_h


def random_swap_move(h):
    """Swap a 1 and a 0 in a binary vector."""
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    if ones.size == 0 or zeros.size == 0:
        return h.copy()
    i = np.random.choice(ones)
    j = np.random.choice(zeros)
    new_h = h.copy()
    new_h[i] = 0.0
    new_h[j] = 1.0
    return new_h


def simulated_annealing(start_h, dx, budget_s, start_time):
    """Simple SA that respects the sum constraint via transfer moves."""
    best_h = start_h.copy()
    best_score = compute_c5(best_h, dx)
    cur_h = start_h.copy()
    cur_score = best_score
    n = len(start_h)

    # tuned parameters
    max_no_improve = 20000
    no_improve = 0

    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break

        # Temperature decays linearly to 0
        T = max(1e-4, (budget_s - elapsed) / budget_s)

        # Propose a move
        cand = transfer_move(cur_h)
        if cand is None:
            # fall back to a simple binary swap (useful if vector became binary)
            cand = random_swap_move(cur_h)

        cand_score = compute_c5(cand, dx)

        # Metropolis acceptance
        if cand_score < cur_score or random.random() < math.exp((cur_score - cand_score) / T):
            cur_h = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_h = cand.copy()
                best_score = cand_score
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # If stagnation, restart from best (adds diversity)
        if no_improve >= max_no_improve:
            cur_h = best_h.copy()
            cur_score = best_score
            no_improve = 0

    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimizes a step function h on [0,2] to minimize C5.
    Returns (h_values, c5_bound, n_points).
    """
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------

    if abs(diff) > 1e-9:
