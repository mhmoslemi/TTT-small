def project_onto_box_sum(v, target):
    # v is array, target sum, bounds [0,1].
    low = np.min(v) - 1.0
    high = np.max(v)
    for _ in range(50):
        mid = (low + high) / 2
        proj = np.clip(v - mid, 0, 1)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    # final projection
    proj = np.clip(v - (low+high)/2, 0, 1)
    # To make sum exactly target within tolerance, we may need slight adjustment.
    cur = proj.sum()

    if abs(diff) > 1e-8:
        # adjust on free variables (not at bounds) proportionally
        free = (proj > 0) & (proj < 1)

            proj = np.clip(proj, 0, 1)
        else:
            # fallback: distribute to all
            proj += diff / len(v)
            proj = np.clip(proj, 0, 1)

def compute_gradient(h):
    n = h.shape[0]
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(c)
    shift = idx - (n - 1)
    g = np.ones(n)
    if shift >= 0:
        g[:n-shift] -= h[shift:]
        g[shift:] -= h[:n-shift]
    else:
        s = -shift
        g[s:] -= h[:n-s]
        g[:n-s] -= h[s:]
    return dx * g
```

def project_box_sum(v, target):
    # binary search lambda to meet sum constraint
    low = np.min(v) - 1.0
    high = np.max(v)
    for _ in range(50):
        mid = (low + high) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) / 2.0
    proj = np.clip(v - lam, 0.0, 1.0)
    # Minor adjustment to ensure sum = target within tolerance
    diff = target - proj.sum()
    if abs(diff) > 1e-9:
        # adjust using free variable
        free = (proj > 0) & (proj < 1)
        if free.any():
            proj[free] += diff / free.sum()
            proj = np.clip(proj, 0.0, 1.0)
        else:
            # fallback uniform distribution
            proj += diff / len(v)
            proj = np.clip(proj, 0.0, 1.0)
    return proj
```

import time
import numpy as np

def _project_box_sum(v, target):
    """Project vector v onto the hyperplane sum(x)=target with bounds [0,1]."""
    low = np.min(v) - 1.0
    high = np.max(v)
    for _ in range(50):
        mid = (low + high) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    proj = np.clip(v - lam, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-10:
        free = (proj > 0.0) & (proj < 1.0)
        if free.any():
            proj[free] += diff / free.sum()
            proj = np.clip(proj, 0.0, 1.0)
        else:
            proj += diff / len(v)
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def _correlation_and_obj(h):
    """Return full correlation h * (1-h) and the scaled max value."""
    n = h.shape[0]
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    max_idx = int(np.argmax(c))
    return c, dx * c[max_idx], max_idx

def _gradient(h, shift):
    """Gradient of the max-overlap objective for a given shift."""
    n = h.shape[0]
    dx = 2.0 / n
    g = np.ones(n)
    if shift >= 0:
        g[: n - shift] -= h[shift:]
        g[shift:] -= h[: n - shift]
    else:
        s = -shift
        g[s:] -= h[: n - s]
        g[: n - s] -= h[s:]
    return dx * g

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    # ------------------------------------------------------------------
    # initialise h
    try:
        h0 = np.array(initial_h_values, dtype=float)  # type: ignore
    except Exception:
        # fallback: 95 points, uniform mass
        n = 95
        h0 = np.full(n, 0.5, dtype=float)
    else:
        n = h0.size
    target_sum = n / 2.0
    h_curr = _project_box_sum(h0.copy(), target_sum)

    # ------------------------------------------------------------------
    # helpers
    def obj_val(h):
        _, val, _ = _correlation_and_obj(h)
        return val

    best_h = h_curr.copy()
    best_obj = obj_val(best_h)
    curr_obj = best_obj

    # algorithm parameters
    max_iter = 200000
    step0 = 0.5
    iter_no_improve = 0

    for it in range(max_iter):
        # time guard
        if time.time() - start > budget_s - 0.05:
            break

        # Decide move type
        if np.random.rand() < 0.3:
            # gradient-based move
            c, _, idx = _correlation_and_obj(h_curr)
            shift = idx - (n - 1)
            grad = _gradient(h_curr, shift)
            grad -= np.mean(grad)                # keep sum unchanged
            norm = np.linalg.norm(grad)
            if norm == 0.0:
                step = 0.0
            else:
                step = step0 / np.sqrt(it + 1) * (target_sum / norm)
            cand = _project_box_sum(h_curr - step * grad, target_sum)
        else:
            # random perturbation move
            # scale of perturbation slowly shrinks
            frac = it / max_iter
            sigma = (0.05 * (1.0 - frac) + 0.001) * (1.0 + np.random.rand())
            cand = _project_box_sum(h_curr + np.random.randn(n) * sigma, target_sum)

        cand_obj = obj_val(cand)

        # accept improvement
        if cand_obj < curr_obj:
            h_curr, curr_obj = cand, cand_obj
            iter_no_improve = 0
        else:
            iter_no_improve += 1
            # occasional uphill move (simulated annealing style)
            if np.random.rand() < 0.001:
                h_curr, curr_obj = cand, cand_obj
                iter_no_improve = 0

        # update global best
        if cand_obj < best_obj:
            best_obj, best_h = cand_obj, cand.copy()
            iter_no_improve = 0

        # random restart if stuck
        if iter_no_improve > 5000:
            rand = np.random.rand(n)
            h_curr = _project_box_sum(rand, target_sum)
            curr_obj = obj_val(h_curr)
            iter_no_improve = 0

    # final sanity projection (ensures constraints)
    best_h = _project_box_sum(best_h, target_sum)
    best_obj = obj_val(best_h)

    return best_h.tolist(), best_obj, n
