def project_to_box_sum(v, target):
    # v: np.ndarray, any shape, target sum between 0 and len(v)
    # Find lambda s.t. sum(clip(v + lambda, 0, 1)) = target
    # Use binary search on lambda.
    n = v.size
    # lower bound: lambda_low = -np.max(v) - 1 # ensures all v+lam <=0
    # but we can set low = -1 - np.max(v) , high = 1 - np.min(v)
    lam_low = -1.0 - np.max(v)
    lam_high = 1.0 - np.min(v)
    for _ in range(60):
        lam_mid = (lam_low + lam_high) / 2.0
        w = np.clip(v + lam_mid, 0.0, 1.0)
        s = np.sum(w)
        if s < target:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    w = np.clip(v + (lam_low + lam_high) / 2.0, 0.0, 1.0)
    # Due to rounding, sum may be slightly off; adjust by distributing diff across middle elements.
    diff = target - np.sum(w)
    if abs(diff) > 1e-12:
        # adjust by adding diff to some free elements (not at bounds)
        free = np.where((w > 0) & (w < 1))[0]
        if free.size > 0:
            # distribute equally
            w[free] += diff / free.size
            # clip again to ensure bounds
            w = np.clip(w, 0.0, 1.0)
    return w

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Helper: evaluate the C5 value for a candidate step function.
# Uses the pre‑imported evaluate_erdos_solution if available,
# otherwise falls back to the direct correlation formula.
def _c5_value(h):
    try:
        # evaluate_erdos_solution may return a scalar or a sequence
        val = evaluate_erdos_solution(h, len(h))
        if isinstance(val, (list, tuple, np.ndarray)):
            return float(val[0])
        return float(val)
    except Exception:
        # fallback: max correlation of h with (1-h)
        n = len(h)
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full")
        return float(np.max(corr) * dx)


# ----------------------------------------------------------------------
# Projection of a vector onto the box [0,1]^n intersecting the hyperplane
# sum(v) = target. Uses a binary search for the Lagrange multiplier.
def _project_box_sum(v, target):
    # lambda search interval
    lo = -1.0 - np.max(v)          # makes v+lo <= 0 -> all zeros
    hi = 1.0 - np.min(v)           # makes v+hi >= 1 -> all ones
    # binary search
    for _ in range(60):
        lam = (lo + hi) * 0.5
        w = np.clip(v + lam, 0.0, 1.0)
        s = w.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction if rounding error persists
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        free = np.where((w > 0.0) & (w < 1.0))[0]
        if free.size:
            w[free] += diff / free.size
            w = np.clip(w, 0.0, 1.0)
    return w


# ----------------------------------------------------------------------
# Generate a random feasible vector (0 <= x_i <= 1, sum = target)
def _random_feasible(n, target, rng):
    v = rng.random(n)
    return _project_box_sum(v, target)


# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    # Determine number of samples
    if 'initial_h_values' in globals():
        n = int(len(initial_h_values))
    else:
        n = 47                     # default size if no parent is supplied

    target_sum = n / 2.0           # because ∫_0^2 h = 1  ->  sum_i h_i = n/2
    dx = 2.0 / n

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise current candidate
    if 'initial_h_values' in globals():
        h_curr = np.array(initial_h_values, dtype=float).copy()
        # enforce exact constraints (safety)
        h_curr = _project_box_sum(h_curr, target_sum)
    else:
        h_curr = _random_feasible(n, target_sum, rng)

    cur_c5 = _c5_value(h_curr)

    # Best solution seen so far
    best_h = h_curr.copy()
    best_c5 = cur_c5

    # ------------------------------------------------------------------
    # Simulated‑annealing parameters
    start = time.time()
    T0 = 0.02                     # initial temperature
    T_min = 1e-6                  # floor temperature
    max_no_improve = 200000      # steps before a mild restart
    no_improve = 0
    iteration = 0

    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.94:
            break

        # Linear cooling schedule
        progress = elapsed / budget_s
        T = max(T_min, T0 * (1.0 - progress))

        # ---- propose a two‑coordinate move that preserves the sum ----------
        i, j = rng.integers(0, n, size=2)
        while j == i:
            j = rng.integers(0, n)

        hi, hj = h_curr[i], h_curr[j]

        # admissible delta: new_h[i]=hi+δ, new_h[j]=hj-δ must stay in [0,1]
        delta_low = max(-hi, hj - 1.0)
        delta_high = min(1.0 - hi, hj)
        if delta_low >= delta_high:
            continue  # cannot move these two coordinates

        delta = delta_low + rng.random() * (delta_high - delta_low)

        h_new = h_curr.copy()
        h_new[i] = hi + delta
        h_new[j] = hj - delta

        new_c5 = _c5_value(h_new)

        # ---- acceptance test ----
        accept = False
        if new_c5 < cur_c5:
            accept = True
        else:
            # Metropolis criterion
            prob = math.exp(-(new_c5 - cur_c5) / (T + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            h_curr = h_new
            cur_c5 = new_c5
            no_improve = 0
        else:
            no_improve += 1

        # ---- record best ----
        if new_c5 < best_c5 - 1e-9:
            best_h = h_new.copy()
            best_c5 = new_c5
            no_improve = 0

        # ---- occasional mild restart to escape flat regions ----
        if no_improve > max_no_improve:
            # perturb the current best a little and resume from there
            h_curr = best_h.copy()
            # perturb ~10% of indices with small random tweaks
            n_perturb = max(1, n // 10)
            idxs = rng.choice(n, size=n_perturb, replace=False)
            for idx in idxs:
                max_up = 1.0 - h_curr[idx]
                max_dn = h_curr[idx] - 0.0
                # random change in [-0.1,0.1] limited by bounds
                delta2 = (rng.random() - 0.5) * 0.2
                delta2 = np.clip(delta2, -max_dn, max_up)
                h_curr[idx] += delta2
            # re‑project to keep the sum exact
            h_curr = _project_box_sum(h_curr, target_sum)
            cur_c5 = _c5_value(h_curr)
            no_improve = 0

        iteration += 1

    # ------------------------------------------------------------------
    # Final projection of the best candidate (ensures feasibility)
    best_h = _project_box_sum(best_h, target_sum)
    final_c5 = _c5_value(best_h)

    return (best_h, final_c5, n)
