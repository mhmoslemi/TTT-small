import numpy as np
import time
import math

def _compute_c5(h, dx):
    """Return max_k ∫ h(x)*(1-h(x+k)) dx for discrete h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _random_binary(N, rng):
    """Return a binary feasible vector with sum N/2."""
    h = np.zeros(N, dtype=float)
    ones = rng.choice(N, size=N // 2, replace=False)
    h[ones] = 1.0
    return h

def _make_feasible(h):
    """Clip to [0,1] and adjust to satisfy sum(h) == N/2 exactly."""
    N = h.shape[0]
    target = N / 2.0
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    # adjust using simple greedy fill / removal

            caps = 1.0 - h

            sub = min(-diff, vals[idx])
