import numpy as np
import time

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def worst_lag_and_gradient(h):
    n = h.shape[0]
    autocorr = np.correlate(h, h, mode='full')
    idx_min = np.argmin(autocorr)
    lag = idx_min - (n - 1)
    # gradient
    grad = np.zeros_like(h)
    if lag >= 0:
        # indices where both sides are in range
        # part1: for i in [0, n-lag-1] -> h[i] * h[i+lag]
        # contribution to grad[i] from its pairing with i+lag
        grad[:n - lag] += h[lag:]
        # contribution to grad[i+lag] from pairing with i
        grad[lag:] += h[:n - lag]
    else:
        pos = -lag
        grad[pos:] += h[:n - pos]
        grad[:n - pos] += h[pos:]
    return lag, grad

def generate_random_feasible(n):
    target = n / 2.0
    k = int(np.floor(target))
    h = np.zeros(n, dtype=float)
    if k > 0:
        idx = np.random.choice(n, k, replace=False)
        h[idx] = 1.0
    remaining = target - k
    if remaining > 1e-12:
        zeros = np.where(h == 0)[0]
        i = np.random.choice(zeros)
        h[i] = remaining
    return h

def local_search(h0, time_budget):
    n = h0.shape[0]
    target_sum = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    start = time.time()
    # parameters
    step = 0.5   # max amount transferred in one move
    min_step = 1e-5
    cooling = 0.9995
    no_improve = 0
    # temperature for simulated annealing
    T0 = best_c5 * 0.1 if best_c5 > 0 else 0.01
    it = 0
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        # identify worst lag and gradient
        lag, grad = worst_lag_and_gradient(h)
        # feasible indices for increase/decrease
        inc_candidates = np.where(h < 1 - 1e-12)[0]
        dec_candidates = np.where(h > 1e-12)[0]
        if inc_candidates.size == 0 or dec_candidates.size == 0:
            # no move possible, break
            break
        # Choose inc index with largest gradient, dec with smallest gradient
        i_inc = inc_candidates[np.argmax(grad[inc_candidates])]
        i_dec = dec_candidates[np.argmin(grad[dec_candidates])]
        if i_inc == i_dec:
            # fallback random pair
            i_inc, i_dec = np.random.choice(n, 2, replace=False)
        # allowable epsilon
        max_inc = min(step, 1.0 - h[i_inc])
        max_dec = min(step, h[i_dec])
        eps = min(max_inc, max_dec)
        if eps <= 0:
            # cannot move, reduce step and continue
            step = max(step * cooling, min_step)
            continue
        # random fraction of eps
        eps = eps * np.random.rand()
        # propose new h
        h_new = h.copy()
        h_new[i_inc] += eps
        h_new[i_dec] -= eps
        # compute new objective
        new_c5 = compute_c5(h_new)
        # Simulated annealing acceptance
        if new_c5 < current_c5:
            accept = True
        else:
            # temperature schedule
            T = max(T0 * (1 - it / 200000.0), 1e-9)
            prob = np.exp(-(new_c5 - current_c5) / T)
            accept = np.random.rand() < prob
        if accept:
            h = h_new
            current_c5 = new_c5
            # update best if better
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
                no_improve = 0
        else:
            no_improve += 1
        # cool step size
        step = max(step * cooling, min_step)
        # occasional random restart if stuck
        if no_improve >= 20000:
            # restart from best with a random perturbation
            h = best_h.copy()
            # make a random feasible small move

import numpy as np
import time

def compute_c5(h: np.ndarray) -> float:
    """Return the Erdős overlap constant for a given step vector."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def worst_lag_and_gradient(h: np.ndarray):
    """
    Find the lag with the smallest autocorrelation (worst overlap)
    and the gradient of that autocorrelation with respect to h.
    Returns (lag, grad) where lag is an integer (negative allowed)
    and grad is a vector of the same length as h.
    """
    n = h.shape[0]
    autocorr = np.correlate(h, h, mode="full")
    idx_min = np.argmin(autocorr)
    lag = idx_min - (n - 1)

    grad = np.zeros_like(h)

    if lag >= 0:
        # pair (i, i+lag)
        grad[: n - lag] += h[lag:]        # derivative from h[i] * h[i+lag]
        grad[lag:] += h[: n - lag]        # derivative from h[i] * h[i+lag] (other side)
    else:
        p = -lag
        grad[p:] += h[: n - p]
        grad[: n - p] += h[p:]

    return lag, grad


def generate_random_feasible(n: int) -> np.ndarray:
    """Create a random feasible vector with entries in [0,1] and sum = n/2."""
    target = n / 2.0
    k = int(np.floor(target))
    h = np.zeros(n, dtype=float)
    if k > 0:
        idx = np.random.choice(n, k, replace=False)
        h[idx] = 1.0
    remain = target - k
    if remain > 1e-12:
        zero_idxs = np.where(h == 0)[0]
        i = np.random.choice(zero_idxs)
        h[i] = remain
    return h


def local_search(h0: np.ndarray, time_budget: float):
    """Heuristic local search using gradient information and simulated annealing."""
    n = h0.shape[0]
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5

    start = time.time()
    step = 0.5               # maximal transfer amount per move
    min_step = 1e-5
    cooling = 0.9995
    no_improve = 0
    T0 = max(best_c5 * 0.1, 1e-3)
    it = 0

    while True:
        if time.time() - start >= time_budget:
            break

        lag, grad = worst_lag_and_gradient(h)

        inc_cands = np.where(h < 1.0 - 1e-12)[0]
        dec_cands = np.where(h > 1e-12)[0]
        if inc_cands.size == 0 or dec_cands.size == 0:
            break

        i_inc = inc_cands[np.argmax(grad[inc_cands])]
        i_dec = dec_cands[np.argmin(grad[dec_cands])]
        if i_inc == i_dec:
            i_inc, i_dec = np.random.choice(n, 2, replace=False)

        max_inc = min(step, 1.0 - h[i_inc])
        max_dec = min(step, h[i_dec])
        eps = min(max_inc, max_dec)
        if eps <= 0:
            step = max(step * cooling, min_step)
            continue
        eps = eps * np.random.rand()

        h_new = h.copy()
        h_new[i_inc] += eps
        h_new[i_dec] -= eps

        new_c5 = compute_c5(h_new)

        if new_c5 < cur_c5:
            accept = True
        else:
            T = max(T0 * (1.0 - it / 200000.0), 1e-9)
            prob = np.exp(-(new_c5 - cur_c5) / T)
            accept = np.random.rand() < prob

        if accept:
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
                no_improve = 0
        else:
            no_improve += 1

        step = max(step * cooling, min_step)

        if no_improve >= 20000:
            # escape stagnation by re‑starting from the current best
            h = best_h.copy()
            # make a tiny random transfer to keep the search moving
            i_a, i_b = np.random.choice(n, 2, replace=False)
            max_inc = min(step, 1.0 - h[i_a])
            max_dec = min(step, h[i_b])
            if max_inc > 0 and max_dec > 0:
                eps = min(max_inc, max_dec) * np.random.rand()
                h[i_a] += eps
                h[i_b] -= eps
            cur_c5 = compute_c5(h)
            no_improve = 0

        it += 1

    return best_h, best_c5

    """
    Return a tuple (h_values, c5_bound, n_points).
    The algorithm starts from the supplied parent (if any) and performs

    best_h, best_c5 = local_search(h0, remaining)

    # final recomputation (ensures consistency with the evaluation metric)
    final_c5 = compute_c5(best_h)
