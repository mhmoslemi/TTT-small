def project_to_feasible(y, target_sum, tol=1e-9, max_iter=100):
    # Solve x_i = min(max(y_i - lam, 0), 1) ensuring sum x = target_sum.
    # lam lower bound: y_i - 1 (if y_i>1), upper bound: y_i (if y_i<0) etc. 
    lo = -1.0
    hi = 1.0
    # Actually lam can be any real number. We'll find interval such that sum > target_sum at lo and sum < target_sum at hi.
    # Because as lam -> -inf, x_i -> 1 (clipped) because y_i - lam huge positive, min(...,1) => 1, sum -> n.
    # As lam -> +inf, x_i -> 0, sum -> 0.
    # So we can start lo = -max(y) - 2, hi = max(y) + 2.
    lo = -max(y) - 1
    hi = max(y) + 1
    for _ in range(max_iter):
        lam = (lo+hi)/2
        x = np.clip(y - lam, 0, 1)
        s = x.sum()
        if np.abs(s-target_sum) < tol:
            return x
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(y - lam, 0, 1)
```

    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()

            return x
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - lam, 0.0, 1.0)

import time
import numpy as np

def _project_to_feasible(v: np.ndarray, target_sum: float, tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    """Euclidean projection of v onto the box [0,1]^n intersect {sum = target_sum}."""
    lo = -np.max(v) - 2.0   # ensures sum(v - lo) > target_sum (all become 1)
    hi = np.max(v) + 2.0    # ensures sum(v - hi) < target_sum (all become 0)
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if abs(s - target_sum) <= tol:
            return x
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    # final clamp
    return np.clip(v - lam, 0.0, 1.0)

def _c5_of_h(h: np.ndarray) -> float:
    """Compute C5 = max_k ∫ h(x)(1-h(x+k))dx via discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()

    # ------------------------------------------------------------------

    else:

    # enforce minimal length and limit

    best_h = None
    best_c5 = np.inf
    best_n = None

    # ------------------------------------------------------------------
    # Main optimization loop over selected sizes
    # ------------------------------------------------------------------
    for n in sizes:
        target_sum = n / 2.0

        # --------------------------------------------------------------
        # Initial feasible vector for this resolution
        # --------------------------------------------------------------
        if have_initial:
            if base_h.shape[0] == n:
                h0 = base_h.copy()
                if abs(h0.sum() - target_sum) > 1e-8:
                    h0 = _project_to_feasible(h0, target_sum)
            else:
