def project_bounds_sum(v, target):
    w = np.clip(v, 0.0, 1.0)
    s = w.sum()
    if np.isclose(s, target):
        return w
    if s < target:
        # need to add
        slack = target - s
        # iterative water-filling
        while slack > 1e-12:
            active = (w < 1 - 1e-12)
            if not np.any(active):
                # cannot add: break
                break
            add = slack / np.count_nonzero(active)
            w[active] += add
            # clip overshoot
            exceed = w > 1
            if np.any(exceed):
                excess = w[exceed] - 1.0
                slack = slack - np.sum(add * np.count_nonzero(active)) + np.sum(excess)
                w[exceed] = 1.0
                # recalc slack as target - sum(w)
                slack = target - w.sum()
            else:
                # no exceed
                slack = target - w.sum()
        return w
    else:
        # need to reduce
        slack = s - target
        while slack > 1e-12:
            active = (w > 1e-12)
            if not np.any(active):
                break
            sub = slack / np.count_nonzero(active)
            w[active] -= sub
            # clip undershoot
            under = w < 0
            if np.any(under):
                deficit = -w[under]
                slack = slack - np.sum(sub * np.count_nonzero(active)) + np.sum(deficit)
                w[under] = 0.0
                slack = w.sum() - target
            else:
                slack = w.sum() - target
        return w

def project_to_sum_and_bounds(v, target, tol=1e-9, max_iter=100):
    low = -1.0
    high = 1.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            low = mid
        else:
            high = mid
        if abs(s - target) <= tol:
            break
    # final w
    w = np.clip(v - (low+high)/2.0, 0.0, 1.0)
    return w

def evaluate_c5(h):

def dominant_lag(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    lag = idx - (n - 1)
    return lag, corr[idx] * dx

def gradient_for_lag(h, lag):
    n = len(h)
    grad = np.zeros_like(h)
    if lag >= 0:
        grad[:n - lag] += 1 - h[lag:]
        grad[lag:] -= h[:n - lag]
    else:
        k = -lag
        grad[k:] += 1 - h[:n - k]
        grad[:n - k] -= h[k:]
    # gradient of C (unscaled)
    return grad

def project(v, target, tol=1e-9, max_iter=100):

    high = 2.0

        mid = (low + high) / 2.0

        s = w.sum()
        if s > target:
            low = mid
        else:
            high = mid
        if abs(s - target) < tol:
            break
    return np.clip(v - (low + high) / 2.0, 0.0, 1.0)

def gradient_for_lag(h, lag):
    n = h.shape[0]
    grad = np.empty_like(h)
    grad.fill(0.0)
    if lag >= 0:
        # first part: indices 0..n-lag-1
        grad[:n - lag] += 1 - h[lag:]
        # second part: indices lag..n-1
        grad[lag:] -= h[:n - lag]
    else:
        k = -lag
        grad[k:] += 1 - h[:n - k]
        grad[:n - k] -= h[k:]
    return grad

import time
import numpy as np

# ---------- Helper functions ----------

def compute_c5(h: np.ndarray) -> float:
    """Return the C5 value for a given vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full").max() * dx


def dominant_lag(h: np.ndarray):
    """Return the lag (int) achieving the maximal correlation and its raw (unscaled) value."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    lag = idx - (n - 1)  # convert index to lag
    return lag, corr[idx]


def gradient_for_lag(h: np.ndarray, lag: int) -> np.ndarray:
    """Subgradient of the correlation at the given lag."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if lag >= 0:
        # indices where i+lag is valid
        grad[: n - lag] += 1.0 - h[lag:]
        # indices where i-lag is valid
        grad[lag:] -= h[: n - lag]
    else:
        k = -lag
        grad[k:] += 1.0 - h[: n - k]
        grad[: n - k] -= h[k:]
    return grad


def project_to_bounds_and_sum(v: np.ndarray, target_sum: float, tol: float = 1e-12, max_iter: int = 60) -> np.ndarray:
    """
    Euclidean projection of v onto the set {x | 0 <= x_i <= 1, sum x_i = target_sum}.
    Uses a simple bisection on the dual variable lambda.
    """
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target_sum:
            low = mid
        else:
            high = mid
        if abs(s - target_sum) <= tol:
            break
    return np.clip(v - (low + high) * 0.5, 0.0, 1.0)


# ---------- Main entry point ----------

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize C5.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    # Determine dimension and initial vector
    try:
        # initial_h_values may be provided by the harness
        n = len(initial_h_values)  # type: ignore[name-defined]
        h = np.array(initial_h_values, dtype=np.float64)  # type: ignore[name-defined]
    except Exception:
        n = 95
        h = np.full(n, 0.5, dtype=np.float64)

    target_sum = n / 2.0
    h = project_to_bounds_and_sum(h, target_sum)

    # Record the best solution seen so far
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # ---------- Phase 1: Random exploration ----------
    random_phase = 0.30 * budget_s
    while time.time() - start_time < random_phase:
        # Sample a random point and project it onto the feasible set
        v = np.random.rand(n)
        v = project_to_bounds_and_sum(v, target_sum)
        c5_val = compute_c5(v)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = v.copy()
        # Early exit if we already beat the target
        if best_c5 <= 0.3808:
            break

    # Use the best found vector as starting point for gradient descent
    h = best_h.copy()

    # ---------- Phase 2: Subgradient descent ----------
    step0 = 0.5  # initial step magnitude
    iteration = 0
    while True:
        now = time.time()
        if now - start_time > budget_s - 0.2:
            break
        iteration += 1

        # Compute the lag giving the current maximal overlap
        lag, _ = dominant_lag(h)
        grad = gradient_for_lag(h, lag)

        # Base step size (diminishing)
        base_step = step0 / np.sqrt(iteration)

        # Try a shortlist of step scalings and keep the best improvement
        factors = (1.0, 0.5, 0.2, 0.1)
        best_local_h = h
        best_local_c5 = compute_c5(h)

        for f in factors:
            cand = h - base_step * f * grad
            cand = project_to_bounds_and_sum(cand, target_sum)
            cand_c5 = compute_c5(cand)
            if cand_c5 < best_local_c5:
                best_local_c5 = cand_c5
                best_local_h = cand

        # Accept the best local move (guaranteed not worsening)
        h = best_local_h

        # Update global best if improved
        if best_local_c5 < best_c5:
            best_c5 = best_local_c5
            best_h = h.copy()
            if best_c5 <= 0.3808:
                break

        # Periodic small random perturbation to escape flat regions
        if iteration % 200 == 0:
            noise = (np.random.rand(n) - 0.5) * 0.02
            h = project_to_bounds_and_sum(h + noise, target_sum)

    # Final projection to ensure feasibility
    best_h = project_to_bounds_and_sum(best_h, target_sum)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
