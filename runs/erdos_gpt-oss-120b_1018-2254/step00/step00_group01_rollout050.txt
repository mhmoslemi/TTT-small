import numpy as np
import time

def enforce_sum_and_bounds(h):
    """Clip to [0,1] and adjust to satisfy sum(h)=len(h)/2 exactly."""
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = target - cur
    if abs(diff) > 1e-12:
        # try to distribute the needed amount over free entries
        for idx in range(n):
            if diff == 0.0:
                break
            if diff > 0.0:
                inc = min(diff, 1.0 - h[idx])
                h[idx] += inc
                diff -= inc
            else:
                dec = min(-diff, h[idx])
                h[idx] -= dec
                diff += dec
        # any tiny remainder is pushed onto the first entry
        if abs(diff) > 1e-12:
            adj = np.clip(h[0] + diff, 0.0, 1.0) - h[0]
            h[0] += adj
    return np.clip(h, 0.0, 1.0)

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def propose_adjust(h, rng):
    """Propose a feasible two‑coordinate adjustment preserving the sum."""
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    hi, hj = h[i], h[j]
    lo = max(-hi, hj - 1.0)        # lower bound for delta
    hi_bound = min(1.0 - hi, hj)   # upper bound for delta
    if hi_bound <= lo:
        return None
    d = rng.uniform(lo, hi_bound)
    if abs(d) < 1e-14:
        return None
    new_h = h.copy()
    new_h[i] = hi + d
    new_h[j] = hj - d
    return new_h

def random_initial_h(n, rng):
    """Create a binary‑plus‑fraction vector with sum = n/2."""
    target = n / 2.0
    ones = int(np.floor(target))
    frac = target - ones
    h = np.zeros(n)
    idx_one = rng.choice(n, size=ones, replace=False)
    h[idx_one] = 1.0
    if frac > 1e-12:
        remaining = np.setdiff1d(np.arange(n), idx_one)
        if remaining.size:
            idx_frac = rng.choice(remaining, size=1)[0]
            h[idx_frac] = frac
    return enforce_sum_and_bounds(h)

def random_perturb(h, rng, scale=0.2):
    """Add uniform noise then project back to the feasible set."""
    cand = h + (rng.rand(h.size) - 0.5) * 2.0 * scale
    cand = np.clip(cand, 0.0, 1.0)
    return enforce_sum_and_bounds(cand)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)

    # Initialise h
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        n_points = 47
        h0 = random_initial_h(n_points, rng)
    n = h0.shape[0]
    h0 = enforce_sum_and_bounds(h0)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    current_h = best_h.copy()
    current_c5 = best_c5

    start_time = time.time()
    max_time = budget_s * 0.95   # reserve a safety margin

    # Temperature schedule (linear annealing)
    temp_start = max(best_c5, 1e-3)
    temp_end = temp_start * 0.01 + 1e-8

    stagnation = 0
    STAGNATE_LIMIT = 200_000

    while True:
        elapsed = time.time() - start_time
        if elapsed >= max_time:
            break

        # propose a feasible adjustment
        new_h = propose_adjust(current_h, rng)
        if new_h is None:
            continue

        new_c5 = compute_c5(new_h)
        delta = new_c5 - current_c5

        # linear cooling
        progress = elapsed / max_time
        temp = temp_start * (1.0 - progress) + temp_end * progress

        if delta < 0.0 or rng.rand() < np.exp(-delta / temp):
            current_h = new_h
            current_c5 = new_c5
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = new_h.copy()
                stagnation = 0
            else:
                stagnation += 1
        else:
            stagnation += 1

        # restart from a perturbed best if stuck
        if stagnation >= STAGNATE_LIMIT:
            current_h = random_perturb(best_h, rng, scale=0.2)
            current_c5 = compute_c5(current_h)
            stagnation = 0

    # final safety projection
    best_h = enforce_sum_and_bounds(best_h)
    best_c5 = compute_c5(best_h)

    return (best_h, float(best_c5), n)
