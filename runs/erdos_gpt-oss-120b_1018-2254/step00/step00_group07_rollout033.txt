import time, math, random
import numpy as np

# Helper: compute the overlap bound C5 for a given discretisation
def _compute_c5(h, dx):
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


# Propose a feasible move that preserves the total integral.
# It picks two distinct indices i, j and transfers a small amount δ from j to i.
def _propose_move(h, max_step):
    n = h.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    hi, hj = h[i], h[j]

    # feasible interval for δ so that 0 ≤ hi+δ ≤ 1 and 0 ≤ hj-δ ≤ 1
    lower = max(-hi, -(1.0 - hj))
    upper = min(1.0 - hi, hj)
    if lower >= upper:
        return h, 0.0  # no feasible move (should be rare)

    # draw δ uniformly from the feasible interval, limited by max_step
    delta = np.random.uniform(lower, upper)
    # optionally limit the step size
    if abs(delta) > max_step:
        delta = math.copysign(max_step, delta)

    h_new = h.copy()
    h_new[i] = hi + delta
    h_new[j] = hj - delta
    return h_new, delta


# Adjust a vector so that its sum equals target while staying inside [0,1].
def _force_sum(v, target, max_iter=10):
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        cur = v.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        # distribute the remaining difference uniformly over non‑boundary entries
        free = np.where((v > 0.0) & (v < 1.0))[0]
        if free.size == 0:
            # fall back to any index
            free = np.arange(v.size)
        add = diff / free.size
        v[free] = np.clip(v[free] + add, 0.0, 1.0)
    return v
