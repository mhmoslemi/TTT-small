def project_onto_bounds_and_sum(x, target):
    # Clip to [0,1]
    x = np.clip(x, 0.0, 1.0)
    # Compute residual needed to achieve target sum
    # Use algorithm: Initialize free set = all indices where 0 < x < 1 (i.e., not at bounds)
    # Perform water-filling: adjust by adding delta uniformly to free variables: delta = (target - sum(x)) / len(free)
    # Then clip again and repeat until convergence.
    # Use loop of at most n times.
    # Assumes target between min possible sum (0) and max possible sum (n). Should be satisfied if feasible.
    eps = 1e-12
    n = len(x)
    while True:
        current_sum = x.sum()
        if abs(current_sum - target) <= eps:
            break
        # Determine free indices: those not at bounds (within eps)
        free = (x > eps) & (x < 1 - eps)
        if not np.any(free):
            # no free variables; need to adjust by moving bound values? fallback: scale entire vector
            x = x + (target - current_sum) / n
            x = np.clip(x, 0.0, 1.0)
            continue
        delta = (target - current_sum) / free.sum()
        x[free] += delta
        # Clip again
        x = np.clip(x, 0.0, 1.0)
    return x

def simulated_annealing(h0, max_iters, time_limit):
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_c5 = best_c5
    n = len(h0)
    start = time.time()
    for i in range(max_iters):
        if time.time() - start > time_limit:
            break
        # temperature schedule: exponential decay
        t = max(0.0, 1.0 - i / max_iters)  # simple linear schedule
        # propose new h: copy cur but add Gaussian noise to a random subset
        # we could modify randomly 5% of entries
        new_h = cur_h.copy()
        # select k indices
        k = max(1, int(0.05 * n))
        idx = np.random.choice(n, size=k, replace=False)
        # add noise
        new_h[idx] += np.random.normal(scale=0.1 * t + 0.01, size=k)
        # project to feasible set
        new_h = project_onto_bounds_and_sum(new_h, n/2)
        new_c5 = compute_c5(new_h)

        if new_c5 < cur_c5 or np.random.rand() < np.exp((cur_c5 - new_c5) / max(t, 1e-8)):

            cur_c5 = new_c5
            if cur_c5 < best_c5:

                best_c5 = cur_c5

def gradient_of_c5(h):
    # compute full correlation
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_idx = np.argmax(corr)  # index in full correlation array (0..2n-2)
    # shift index relative to zero shift: shift = k_idx - (n-1)
    n = len(h)
    shift = k_idx - (n - 1)
    # compute derivative: grad[i] = (1 - h[i+shift]) - (h[i-shift]) where indexes within bounds else 0.
    grad = np.zeros_like(h)
    # vectorized:
    # term from (1 - h_{i+shift})
    idx_i = np.arange(n)
    idx_j = idx_i + shift
    mask = (idx_j >= 0) & (idx_j < n)
    grad[mask] += (1.0 - h[idx_j[mask]])
    # term from -h_{i-shift}
    idx_j2 = idx_i - shift
    mask2 = (idx_j2 >= 0) & (idx_j2 < n)
    # The contribution is -h[i - shift], i.e., minus h[idx_j2] for each i where idx_j2 in range.
    grad[mask2] -= h[idx_j2[mask2]]

import time
import math
import random
import numpy as np

def _project_onto_bounds_and_sum(x, target):
    """Clip x to [0,1] and adjust to meet sum(target) using water-filling."""
    x = np.clip(x, 0.0, 1.0)
    eps = 1e-12
    for _ in range(100):
        cur_sum = x.sum()
        diff = target - cur_sum
        if abs(diff) <= eps:
            break
        free = (x > eps) & (x < 1 - eps)
        n_free = free.sum()
        if n_free == 0:
            # distribute uniformly and re‑clip
            delta = diff / x.size
            x += delta
            x = np.clip(x, 0.0, 1.0)
            continue
        delta = diff / n_free
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
    return np.clip(x, 0.0, 1.0)

def _compute_c5(h):
    """Return the discrete C5 value for step vector h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)

def _gradient_of_c5(h):
    """Gradient of the maximal overlap term (the shift attaining the max)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)                     # signed shift producing max
    grad = np.zeros_like(h)

    i = np.arange(n)
    j = i + shift
    mask = (j >= 0) & (j < n)
    grad[mask] += (1.0 - h[j[mask]])        # derivative from h_i * (1-h_{i+shift})

    j2 = i - shift
    mask2 = (j2 >= 0) & (j2 < n)
    grad[mask2] -= h[j2[mask2]]             # derivative from h_{i-shift} * (1-h_i)

    c5 = corr[idx] * (2.0 / n)
    return grad, shift, c5

def _random_feasible(n, rng):
    """Generate a random feasible vector (values in [0,1], sum = n/2)."""
    # start from uniform random then project – simple and fast
    x = rng.random(n)
    target = n / 2.0
    return _project_onto_bounds_and_sum(x, target)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős C5 overlap.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    random.seed(seed)

    # ----------------------------------- start vector -----------------------------------
    if 'initial_h_values' in globals():
        base = np.array(initial_h_values, dtype=float)
        # repeat to increase resolution (factor 3 keeps the integral exact)
        repeat_factor = 3
        h0 = np.repeat(base, repeat_factor)
    else:
        # fallback: simple half‑ones half‑zeros pattern
        n_points = 80
        h0 = np.zeros(n_points, dtype=float)
        h0[: n_points // 2] = 1.0
        rng.shuffle(h0)

    n_points = h0.size
    target_sum = n_points / 2.0
    h0 = _project_onto_bounds_and_sum(h0, target_sum)

    # ------------------------------------ optimization -----------------------------------
    best_h = h0.copy()
    best_c5 = _compute_c5(best_h)

    cur_h = best_h.copy()
    cur_c5 = best_c5

    start_time = time.time()
    max_iters = 2_000_000          # generous upper bound; loop exits on time
    alpha_start = 0.5
    alpha_min = 0.001
    stagnation_limit = 30_000      # after this many non‑improving steps restart
