import time
import numpy as np

def _c5_of(arr, dx):
    """Return the C5 value for a given step function array."""
    # correlation of h with itself (full)
    conv = np.correlate(arr, arr, mode='full')
    min_prod = conv.min()
    return 1.0 - min_prod * dx

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős C5 constant.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- initialization -------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.asarray(initial_h_values, dtype=float)
        n = h.size
    else:
        n = 70
        h = np.zeros(n, dtype=float)
        idx = rng.choice(n, size=n // 2, replace=False)
        h[idx] = 1.0

    target_sum = n / 2.0
    # enforce exact sum while respecting bounds
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        scale = target_sum / cur_sum if cur_sum != 0 else 0.0
        h = np.clip(h * scale, 0.0, 1.0)
        diff = target_sum - h.sum()
        for i in range(n):
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break

    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = _c5_of(best_h, dx)

    # --------------------------------------------------------------------
    # main optimization loop
    while True:

        conv = np.correlate(h, h, mode='full')

        s = abs(shift)

        if s == 0 or rng.random() < 0.2:

            inc = rng.integers(n)

            if inc_cands.size == 0 or dec_cands.size == 0:
