def repair_binary_sum(arr, target_ones):
    # arr may have values between 0 and 1 (float), need to adjust to binary? But we will keep arr binary.
    # Ensure sum close to target_ones, adjust.
    current = int(round(arr.sum()))
    if current == target_ones:
        # ensure binary values 0/1
        arr[:] = np.where(arr >= 0.5, 1.0, 0.0)
        return arr
    # If too many ones:
    if current > target_ones:
        # need to turn (current - target) ones to zeros randomly chosen among indices where arr > 0.5
        ones_idx = np.where(arr >= 0.5)[0]
        # random select to zero
        to_zero = np.random.choice(ones_idx, current - target_ones, replace=False)
        arr[to_zero] = 0.0
    else: # too few ones
        zeros_idx = np.where(arr < 0.5)[0]
        to_one = np.random.choice(zeros_idx, target_ones - current, replace=False)
        arr[to_one] = 1.0
    # set any residual floats to 0 or 1 based on threshold
    arr[:] = np.where(arr >= 0.5, 1.0, 0.0)
    return arr

def repair_h(h, target_sum):
    """Clip to [0,1] and adjust entries so that sum(h) == target_sum."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    # Iteratively distribute the deficit/excess proportionally.
    for _ in range(10):
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            cap = 1.0 - h
            total_cap = cap.sum()
            if total_cap == 0:
                break
            delta = diff * (cap / total_cap)
            h += delta
        else:
            cap = h
            total_cap = cap.sum()
            if total_cap == 0:
                break
            delta = diff * (cap / total_cap)   # diff is negative
            h += delta
        h = np.clip(h, 0.0, 1.0)
        cur = h.sum()
    # Final tiny adjustment if needed
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size:
                i = np.random.choice(idx)
                inc = min(diff, 1.0 - h[i])
                h[i] += inc
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size:
                i = np.random.choice(idx)
                dec = min(-diff, h[i])
                h[i] -= dec
    return np.clip(h, 0.0, 1.0)

    if np.random.rand() < 0.5:
        i, j = np.random.choice(n, 2, replace=False)
        child[i], child[j] = child[j], child[i]
