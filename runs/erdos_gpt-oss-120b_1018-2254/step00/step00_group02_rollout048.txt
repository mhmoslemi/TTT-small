    x = np.clip(x, 0.0, 1.0)
    # offset to match sum
    # compute delta = (target - sum(x))/n
    # add delta to each component, but may violate bounds, so iterate.
    for _ in range(10):
        sum_x = x.sum()
        diff = target - sum_x
        if abs(diff) < 1e-9:
            break
        # Distribute diff equally to unfixed variables (that are not at bound)
        mask = (x > 0) & (x < 1)
        n_free = mask.sum()
        if n_free == 0:
            # all at bounds: distribute arbitrarily? break
            break
        delta = diff / n_free
        # update only free variables
        x[mask] += delta
        # clip again
        x = np.clip(x, 0.0, 1.0)
    return x

def project_to_constraints(x, target, max_iter=10):
    x = np.clip(x, 0.0, 1.0)
    for _ in range(max_iter):
        diff = target - x.sum()
        if abs(diff) < 1e-12:
            break
        mask = (x > 0.0) & (x < 1.0)
        free = np.where(mask)[0]
        n_free = len(free)
        if n_free == 0:
            # adjust all by diff/n, then clip again
            delta = diff / x.size
            x += delta
            x = np.clip(x, 0.0, 1.0)
            continue
        delta = diff / n_free
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
    return x

import time
import numpy as np
from scipy.optimize import differential_evolution, minimize

# Global placeholders for differential evolution penalty
_de_target = None
_de_penalty = None


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)*(1-h(x+k)) dx approximated on a grid."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)


def project_to_constraints(x, target, max_iter=15):
    """Project x onto [0,1]^n intersect {sum x = target}."""
    x = np.clip(x, 0.0, 1.0)
    for _ in range(max_iter):
        diff = target - x.sum()
        if abs(diff) < 1e-12:
            break
        free_mask = (x > 0.0) & (x < 1.0)
        free_idx = np.where(free_mask)[0]
        n_free = free_idx.size
        if n_free == 0:
            # all entries at a bound; distribute uniformly and re‑clip
            delta = diff / x.size
            x += delta
            x = np.clip(x, 0.0, 1.0)
            continue
        delta = diff / n_free
        x[free_idx] += delta
        x = np.clip(x, 0.0, 1.0)
    return x


class SumConstraint:
    """Callable equality constraint enforcing sum(x) = target."""
    def __init__(self, target):
        self.target = target

    def __call__(self, x):
        return np.sum(x) - self.target


def de_objective(x):
    """Objective for DE: C5 plus penalty for deviating from the sum constraint."""
    global _de_target, _de_penalty
    return compute_c5(x) + _de_penalty * abs(np.sum(x) - _de_target)


def local_refine(h, target, max_iter=200):
    """SLSQP refinement with exact equality constraint."""
    n = h.shape[0]
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": SumConstraint(target)}
    # SLSQP may not converge; we ignore its status and project afterwards.
    res = minimize(compute_c5, h, method="SLSQP", bounds=bounds,
                   constraints=[cons],
                   options={"maxiter": max_iter, "ftol": 1e-9})
    refined = project_to_constraints(res.x if hasattr(res, "x") else h, target)
    return refined, compute_c5(refined)


def interpolate_h(h, n_target):
    """Linear interpolation of h onto a new grid of size n_target."""
    n = h.shape[0]
    if n == n_target:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
    return np.interp(x_new, x_old, h)
