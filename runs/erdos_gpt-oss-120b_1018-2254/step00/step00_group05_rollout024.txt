def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # correlation using np.correlate (full convolution with zero padding)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_sum_and_bounds(h):
    """Clip h into [0,1] and adjust to exact sum n/2."""
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    diff = target - total
    # apply diff to first element that can absorb it without breaking bounds
    if abs(diff) > 1e-12:
        # find index where adding diff stays within bounds
        for i in range(n):
            new_val = h[i] + diff
            if 0.0 <= new_val <= 1.0:
                h[i] = new_val
                break
        else:
            # fallback: distribute diff uniformly and re‑clip
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
    # final clip to be safe
    return np.clip(h, 0.0, 1.0)

    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # Distribute the tiny residual to a single coordinate that can absorb it.
    if abs(diff) > 1e-12:
        for i in range(n):
            cand = h[i] + diff
            if 0.0 <= cand <= 1.0:
                h[i] = cand
                break
        else:
            # fallback uniform distribution (should rarely happen)
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
    return h

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step‑function h, its C₅ bound, and number of points."""
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # Initialise h
    if "initial_h_values" in globals():
        # Upsample the parent construction to a modest size for more flexibility.
        parent = np.asarray(initial_h_values, dtype=np.float64)
        n_parent = parent.size
        n = 200                       # chosen grid size (< 1 k as required)
        x_parent = np.linspace(0.0, 2.0, n_parent)
        x_grid = np.linspace(0.0, 2.0, n)
        # linear interpolation (no external data needed)
        from scipy.interpolate import interp1d
        interp = interp1d(x_parent, parent, kind="linear", fill_value="extrapolate")
        h = interp(x_grid).astype(np.float64)
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum_and_bounds(h)
    else:
        n = 200
        h = np.full(n, 0.5, dtype=np.float64)

    # make sure we start on the feasible set
    h = project_to_sum_and_bounds(h)

    best_h = h.copy()
    best_score = compute_c5(best_h)
    current_score = best_score

    start_time = time.time()
    total_budget = budget_s * 0.97          # keep a small safety margin
    max_temp = 0.02                         # SA temperature start
    step_factor = 0.5                       # max fraction of a feasible transfer

    # ------------------------------------------------------------------
    while True:
        now = time.time()
        if now - start_time > total_budget:
            break

        # ---- propose a mass transfer between two coordinates -------------
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        if random.random() < 0.5:
            donor, receiver = i, j
        else:
            donor, receiver = j, i

        max_transfer = min(h[donor], 1.0 - h[receiver])
        if max_transfer <= 1e-12:
            continue
        delta = random.random() * max_transfer * step_factor
        if delta <= 0.0:
            continue

        # apply the move
        h[donor] -= delta
        h[receiver] += delta

        new_score = compute_c5(h)

        # ---- Simulated‑annealing acceptance -----------------------------
        elapsed = now - start_time
        temperature = max_temp * (1.0 - elapsed / total_budget)
        temperature = max(temperature, 1e-6)

        accept = False
        if new_score < current_score:
            accept = True
        else:
            prob = math.exp((current_score - new_score) / temperature)
            if random.random() < prob:
                accept = True

        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                if best_score <= 0.3808:   # target reached – stop early
                    break
        else:
            # revert the move
            h[receiver] -= delta
            h[donor] += delta

        # occasional tiny perturbation to shake out of plateaus
        if random.random() < 0.001:
            k = random.randint(0, n - 1)
            l = random.randint(0, n - 1)
            while l == k:
                l = random.randint(0, n - 1)
            delta2 = (random.random() - 0.5) * 0.1
            # clamp delta2 to keep both entries in [0,1]
            delta2 = max(-h[k], min(1.0 - h[k], delta2))
            if delta2 > 0:
                transfer = min(h[l], delta2)
                h[l] -= transfer
                h[k] += transfer
            else:
                transfer = min(1.0 - h[l], -delta2)
                h[l] += transfer
                h[k] -= transfer

    # ------------------------------------------------------------------
    best_h = project_to_sum_and_bounds(best_h)
    final_score = compute_c5(best_h)

    return best_h, final_score, n
