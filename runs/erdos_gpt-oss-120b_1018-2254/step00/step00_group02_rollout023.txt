import time
import numpy as np


def _shift_array(arr: np.ndarray, k: int) -> np.ndarray:
    """Shift arr by k with zero padding (positive k shifts left)."""
    n = arr.shape[0]
    if k == 0:
        return arr
    if k > 0:
        return np.concatenate((arr[k:], np.zeros(k, dtype=arr.dtype)))
    else:
        k = -k
        return np.concatenate((np.zeros(k, dtype=arr.dtype), arr[: n - k]))


def _gradient(h: np.ndarray, shift: int) -> np.ndarray:
    """Sub‑gradient of the worst‑case overlap for a given shift."""
    fwd = _shift_array(h, shift)
    bwd = _shift_array(h, -shift)
    return 1.0 - (fwd + bwd)


def _project_box_sum(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto the feasible set {0≤x≤1, sum(x)=target_sum}."""
    # binary search for λ such that sum(clamp(v+λ,0,1)) = target_sum
    low = -v.max() - 2.0
    high = 1.0 - v.min() + 2.0
    for _ in range(50):
        lam = (low + high) * 0.5
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            high = lam
        else:
            low = lam
    lam = (low + high) * 0.5
    x = np.clip(v + lam, 0.0, 1.0)
    # tiny correction to hit the sum exactly (helps the evaluator)
    err = target_sum - x.sum()
    if abs(err) > 1e-12:
        free = np.where((x > 1e-12) & (x < 1.0 - 1e-12))[0]
        if free.size:
            i = free[0]
            x[i] = np.clip(x[i] + err, 0.0, 1.0)
        else:
            # fall back: adjust the first element within bounds
            i = np.argmin(x)
            x[i] = np.clip(x[i] + err, 0.0, 1.0)
    return x


def _c5_value(h: np.ndarray):
    """Return C₅ and the full correlation array for a given h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx, corr


def run(seed=42, budget_s=1100, **kwargs):
    """Optimise a step function for the Erdős C₅ problem."""
    rng = np.random.default_rng(seed)

    # ----- initialise -------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)
        n = int(init_h.size)
        h = init_h.copy()
    except Exception:  # initial_h_values not provided
        n = 120  # even size gives enough freedom
        h = np.full(n, 0.5, dtype=float)

    # Ensure feasibility (sum = n/2, 0≤h≤1)
    h = _project_box_sum(h, n / 2.0)

    best_h = h.copy()
    best_c5, _ = _c5_value(best_h)

    start = time.time()
    it = 0

    # ----- optimisation loop -----------------------------------------
    while time.time() - start < budget_s:
        c5, corr = _c5_value(h)

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # worst‑case shift (index of maximum overlap)
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)

        # sub‑gradient step
        grad = _gradient(h, shift)
        step = 0.5 / np.sqrt(it + 1.0)
        h = _project_box_sum(h - step * grad, n / 2.0)

        # occasional small random perturbation to escape plateaus
        if it % 500 == 0 and it > 0:

            delta = (rng.random() - 0.5) * 0.1

        # occasional full random restart

            h = rng.random(n)
            h = _project_box_sum(h, n / 2.0)

        it += 1

    # ----- finalise ---------------------------------------------------
    best_h = _project_box_sum(best_h, n / 2.0)
    best_c5, _ = _c5_value(best_h)
