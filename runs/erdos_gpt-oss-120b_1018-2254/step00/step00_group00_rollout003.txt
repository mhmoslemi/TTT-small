def project_to_simplex_bounded(v, target_sum):
    # v is numpy array
    # Goal: x in [0,1] and sum(x)=target_sum. Use iterative water-filling.
    x = np.clip(v, 0, 1)
    # compute sum
    current_sum = np.sum(x)
    # tolerance
    tol = 1e-8
    # if sum matches target, done
    if abs(current_sum - target_sum) <= tol:
        return x
    # if sum > target, need to reduce
    # We'll reduce from entries that are >0
    # Approach: while difference > tol
    iter_count = 0
    while abs(current_sum - target_sum) > tol and iter_count < 1000:
        if current_sum > target_sum:
            # need to reduce sum
            # active indices are those >0
            active = x > 0
            if np.sum(active) == 0:
                # cannot reduce further, break
                break
            reduction = (current_sum - target_sum) / np.sum(active)
            # subtract reduction but ensure not below 0
            x[active] -= reduction
            x = np.clip(x, 0, 1)
        else:
            # need to increase sum
            active = x < 1
            if np.sum(active) == 0:
                break
            increase = (target_sum - current_sum) / np.sum(active)
            x[active] += increase
            x = np.clip(x, 0, 1)
        current_sum = np.sum(x)
        iter_count += 1
    return x

def project_to_feasible(v, target):
    # Clip
    x = np.clip(v, 0, 1)
    # iterative waterfilling
    max_iter = 1000
    tol = 1e-10
    for _ in range(max_iter):
        s = np.sum(x)
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            # need to reduce sum: subtract equally from indices >0
            active = x > 0
            if not np.any(active):
                break
            delta = diff / np.sum(active)
            x[active] -= delta
        else:
            # need to increase sum
            active = x < 1
            if not np.any(active):
                break
            delta = -diff / np.sum(active)
            x[active] += delta
        # clip
        x = np.clip(x, 0, 1)
    # final tiny adjustments
    diff = target - np.sum(x)
    if abs(diff) > 1e-10:
        for i in range(len(x)):
            if diff > 0 and x[i] < 1:
                add = min(diff, 1 - x[i])
                x[i] += add
                diff -= add
            elif diff < 0 and x[i] > 0:
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
            if abs(diff) <= 1e-12:
                break
    return x

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def tournament_select(pop, fitness, tsize):
    # choose random indices
    idx = np.random.choice(len(pop), tsize, replace=False)
    # best is min fitness
    best_idx = idx[np.argmin([fitness[i] for i in idx])]
    return pop[best_idx]

import time
import random
import numpy as np

def _project_to_feasible(v, target, tol=1e-10, max_iter=1000):
    """Project vector v onto [0,1]^n with sum == target."""
    x = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        s = np.sum(x)
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            active = x > 0.0
            if not np.any(active):
                break
            delta = diff / np.count_nonzero(active)
            x[active] -= delta
        else:
            active = x < 1.0
            if not np.any(active):
                break
            delta = -diff / np.count_nonzero(active)
            x[active] += delta
        x = np.clip(x, 0.0, 1.0)
    # final tiny adjustment
    diff = target - np.sum(x)
    if abs(diff) > tol:
        for i in range(len(x)):
            if diff > 0 and x[i] < 1.0:
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
            elif diff < 0 and x[i] > 0.0:
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
            if abs(diff) <= tol:
                break
    return x

def _evaluate_overlap(h):
    """Return C5 bound for a given step function h (numpy array)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _tournament_select(pop, fitness, tsize):
    idx = np.random.choice(len(pop), tsize, replace=False)
    best = min(idx, key=lambda i: fitness[i])
    return pop[best]

def _uniform_crossover(p1, p2):
    mask = np.random.rand(len(p1)) < 0.5
    return np.where(mask, p1, p2)

def _mutate(ind, rate, std):
    mask = np.random.rand(len(ind)) < rate
    noise = np.random.randn(len(ind)) * std
    child = ind.copy()
    child[mask] += noise[mask]
    return child
