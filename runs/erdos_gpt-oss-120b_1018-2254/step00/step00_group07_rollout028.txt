def compute_overlap(h, dx):
    # returns full correlation vector length 2n-1 multiplied by dx
    return np.correlate(h, 1 - h, mode='full') * dx

def max_overlap(h, dx):
    return np.max(compute_overlap(h, dx))

def smoothed_max(c, p):
    # p >= 1
    # Weighted mean of c^p, then (mean)^(1/p)
    # To avoid overflow or underflow, maybe use safe.
    return (np.mean(c**p))**(1/p)

def gradient_smoothed_max(h, dx, p):
    c = compute_overlap(h, dx) # shape L
    # compute weights
    c_pow = c**(p-1)
    total = np.sum(c_pow)  # sum of c**(p-1)
    if total == 0:
        # c all zero? then gradient zero (not possible)

    else:

    # compute gradient

def gradient_smoothed_max(h, dx, p):

    c = np.correlate(h, 1 - h, mode='full') * dx

    weights = np.nan_to_num(weights, nan=0.0, posinf=0.0, neginf=0.0)

import time
import numpy as np

def project_to_capped_simplex(v, target_sum, cap=1.0):
    """Project v onto {x: sum x = target_sum, 0 <= x_i <= cap}."""
    v = np.clip(v, 0.0, cap)
    s = v.sum()
    if abs(s - target_sum) < 1e-12:
        return v
    n = v.shape[0]
    # binary search for lambda
    lower = -np.max(v) - cap - 1.0   # ensures sum = n*cap > target
    upper = np.max(v) + 1.0          # ensures sum = 0 < target
    for _ in range(60):
        lam = (lower + upper) * 0.5
        x = np.clip(v - lam, 0.0, cap)
        s = x.sum()
        if s > target_sum:
            lower = lam
        else:
            upper = lam
    lam = (lower + upper) * 0.5
    return np.clip(v - lam, 0.0, cap)

def random_feasible(n, target_sum):
    v = np.random.rand(n)
    return project_to_capped_simplex(v, target_sum, cap=1.0)

def compute_overlap(h, dx):
    """Full correlation h with (1-h) multiplied by dx."""
    return np.correlate(h, 1.0 - h, mode='full') * dx

def max_overlap_val(h, dx):
    """Exact C5 value for a given h."""
    return float(np.max(compute_overlap(h, dx)))

def gradient_smoothed_max(h, dx, p):
    """Gradient of the smoothed max (L_p) of the overlap values."""
    n = h.shape[0]
    c = compute_overlap(h, dx)                # length 2n-1
    L = c.shape[0]
    # smoothed max
    mean_cp = np.mean(c ** p)
    if mean_cp <= 0:
        return np.zeros_like(h)
    f = mean_cp ** (1.0 / p)
    # weights = c_i^{p-1} / (L * f^{p-1})
    with np.errstate(divide='ignore', invalid='ignore'):
        weights = c ** (p - 1) / (L * (f ** (p - 1)))
    weights = np.nan_to_num(weights, nan=0.0, posinf=0.0, neginf=0.0)

    grad = np.zeros_like(h)
    for idx, w in enumerate(weights):
        if w == 0.0:
            continue
        shift = idx - (n - 1)          # - (n-1) .. (n-1)
        if shift >= 0:
            h_plus = np.concatenate((np.zeros(shift), h[:n - shift]))
            h_minus = np.concatenate((h[shift:], np.zeros(shift)))
        else:
            s = -shift
            h_plus = np.concatenate((h[s:], np.zeros(s)))
            h_minus = np.concatenate((np.zeros(s), h[:n - s]))
        grad += w * (dx * (1.0 - h_plus - h_minus))
    return grad

def binary_round(h, target_sum):
    """Return a binary vector (0/1) with exactly target_sum ones."""
    n = h.shape[0]
    k = int(round(target_sum))
    idx = np.argpartition(h, -k)[-k:]
    bh = np.zeros(n, dtype=float)
    bh[idx] = 1.0
    return bh
