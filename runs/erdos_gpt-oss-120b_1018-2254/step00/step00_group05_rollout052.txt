    np.random.seed(seed)

    n = 200

    # initial h
    if 'initial_h_values' in globals():
        init_len = len(initial_h_values)

        xs_target = np.linspace(0, 2, n)

    else:

def gradient_weighted_max(h, alpha):

    # correlation v (raw, not multiplied by dx)
    v = np.correlate(h, 1 - h, mode='full')
    dx = 2.0 / n

    expo = np.exp(alpha * (vdx - maxv))

    for shift in range(-n + 1, n):

        if w_shift == 0:

        if shift >= 0:

            G[:n-shift] += w_shift * (1 - h[shift:])
            # term2
            G[shift:] -= w_shift * h[:n-shift]
        else:
            s = -shift
            G[s:] += w_shift * (1 - h[:n-s])
            G[:n-s] -= w_shift * h[s:]
    return G

import numpy as np

def _project_bounds_sum(v, target, lower=0.0, upper=1.0, iters=80):
    lo, hi = -1e5, 1e5
    for _ in range(iters):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, lower, upper)
        if x.sum() > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, lower, upper)
    s = x.sum()
    diff = target - s
    if abs(diff) > 1e-12:
        free = (x > lower + 1e-12) & (x < upper - 1e-12)
        nf = free.sum()
        if nf:
            x[free] += diff / nf
        x = np.clip(x, lower, upper)
    return x

def _evaluate_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def _grad_weighted_max(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    dx = 2.0 / n
    vals = corr * dx
    m = vals.max()
    w = np.exp(alpha * (vals - m))
    w /= w.sum()
    G = np.zeros_like(h)

    for shift in range(-n + 1, n):
        wsh = w[shift + n - 1]
        if wsh == 0.0:
            continue
        if shift >= 0:
            G[:n - shift] += wsh * (1 - h[shift:])
            G[shift:]      -= wsh * h[:n - shift]
        else:
            s = -shift
            G[s:]          += wsh * (1 - h[:n - s])
            G[:n - s]      -= wsh * h[s:]
    return G
