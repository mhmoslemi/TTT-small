        new_c5, new_max = compute_c5(h)
        if new_max < cur_max:
            cur_c5, cur_max = new_c5, new_max
            if cur_max <= 13:   # target already achieved
                break
        else:
            h[i], h[j] = h[j], h[i]  # revert
    return h, cur_c5, cur_max

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    # Determine dimension and optional starter
    try:
        init = initial_h_values
    except NameError:
        init = None
    if init is not None:
        n = len(init)
        base = (np.array(init, dtype=float) > 0.5).astype(float)
        target = n // 2
        ones = int(base.sum())
        if ones != target:
            diff = target - ones
            if diff > 0:
                zero_idx = np.where(base == 0)[0]
                np.random.shuffle(zero_idx)
                base[zero_idx[:diff]] = 1.0
            else:
                one_idx = np.where(base == 1)[0]
                np.random.shuffle(one_idx)
                base[one_idx[: -diff]] = 0.0
    else:
        n = 70
        base = None

    best_h = None
    best_max = math.inf
    best_c5 = math.inf

    total_budget = budget_s
    # Main stochastic search
    while time.time() - start < total_budget * 0.8:
        method = np.random.choice(["beatty", "random", "init"])
        if method == "beatty":
            alpha = (math.sqrt(5) - 1) / 2.0
            offset = np.random.rand()
            cand = generate_beatty(n, alpha, offset)
        elif method == "random":
            cand = random_binary(n)
        else:  # init
            if base is None:
                continue
            cand = base.copy()
            # introduce a few random swaps to diversify
            for _ in range(3):
                ones_idx = np.where(cand == 1)[0]
                zeros_idx = np.where(cand == 0)[0]
                if len(ones_idx) == 0 or len(zeros_idx) == 0:
                    break
                i = np.random.choice(ones_idx)
                j = np.random.choice(zeros_idx)
                cand[i], cand[j] = cand[j], cand[i]
        c5_val, max_corr = compute_c5(cand)
        if max_corr < best_max:
            best_max = max_corr
            best_c5 = c5_val
            best_h = cand.copy()
        if best_max <= 13:
            break
        # short local refinement
        remain = total_budget - (time.time() - start)
        if remain < total_budget * 0.1:
            continue
        refine_time = min(remain * 0.2, 5.0)
        refined_h, refined_c5, refined_max = hill_climb(cand.copy(),
                                                        refine_time,
                                                        time.time())
        if refined_max < best_max:
            best_max = refined_max
            best_c5 = refined_c5
            best_h = refined_h.copy()
        if best_max <= 13:
            break

    # Final refinement on the current best solution
    remaining = total_budget - (time.time() - start)
    if best_h is not None and remaining > 0.2:
        best_h, best_c5, best_max = hill_climb(best_h, remaining, time.time())

    # Fallback if nothing found
    if best_h is None:
        best_h = random_binary(n)
        best_c5, best_max = compute_c5(best_h)

    # Ensure exact feasibility of the integral constraint
    target_sum = n / 2.0
    cur_sum = best_h.sum()
    diff = target_sum - cur_sum
    if abs(diff) > 1e-8:
        free = np.where((best_h > 0) & (best_h < 1))[0]
        if len(free) > 0:
            adjust = diff / len(free)
            best_h[free] += adjust
            best_h = np.clip(best_h, 0.0, 1.0)
        cur_sum = best_h.sum()
        if abs(cur_sum - target_sum) > 1e-6:
            need = int(round(target_sum - cur_sum))
            if need > 0:
                zeros = np.where(best_h < 0.5)[0]
                if len(zeros) >= need:
                    best_h[zeros[:need]] = 1.0
            elif need < 0:
                ones = np.where(best_h > 0.5)[0]
                if len(ones) >= -need:
                    best_h[ones[: -need]] = 0.0
    best_c5 = compute_c5(best_h)[0]

    return best_h.tolist(), best_c5, n
