def project_to_box_and_sum(v, target):
    # Use simple heuristic: iterative adjust lambda via bisection.
    # v may be any real numbers.
    # We'll find lambda such that sum(clamp(v+lambda, 0,1)) = target.
    # The function sum_i clamp(v_i + lambda) is monotonic in lambda.
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        x = np.clip(v + mid, 0, 1)
        s = x.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(v + lo, 0, 1)
    # Ensure sum approx target: adjust residual via distributing small epsilon or scaling
    diff = target - x.sum()
    if abs(diff) > 1e-8:
        # Distribute diff among interior points not at bounds
        mask = (x > 1e-9) & (x < 1 - 1e-9)
        if np.any(mask):
            # distribute equally
            x[mask] += diff / mask.sum()
            # clip again just in case
            x = np.clip(x, 0, 1)
        else:
            # fallback to scaling
            x = x * (target / x.sum())
    return x

import numpy as np, time, math, random

def project_to_box_and_sum(v, target):
    # Clip to [0,1] after shifting by lambda via bisection.
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(v + lo, 0.0, 1.0)
    # Fix residual
    diff = target - x.sum()
    if abs(diff) > 1e-8:
        # distribute proportionally among non-bound entries
        mask = (x > 1e-9) & (x < 1 - 1e-9)
        if mask.any():
            adjustment = diff / mask.sum()
            x[mask] += adjustment
            x = np.clip(x, 0.0, 1.0)
        else:
            # all at bounds, just scale
            x = x * (target / x.sum())
    return x

def compute_c5(h):
    n = h.shape[0]
    # convolution
    corr = np.convolve(h, (1 - h)[::-1])
    # length 2n-1
    dx = 2.0 / n
    return np.max(corr) * dx

def compute_gradient(h, shift):
    n = h.shape[0]
    grad = np.zeros_like(h)
    if shift >= 0:
        end = n - shift
        if end > 0:
            grad[:end] += 1.0 - h[shift:]
            grad[shift:] -= h[:end]
    else:
        s = -shift
        end = n - s
        if end > 0:
            grad[s:] += 1.0 - h[:end]
            grad[:end] -= h[s:]
    return grad

def project_to_box_and_sum(v, target):
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(v + lo, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        mask = (x > 1e-9) & (x < 1.0 - 1e-9)
        if mask.any():
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            # fallback uniform scaling
            total = x.sum()
            if total > 0:

    return np.max(corr) * dx

def compute_gradient(h, shift):
    n = h.shape[0]
    grad = np.zeros_like(h)
    if shift >= 0:
        end = n - shift
        if end > 0:
            grad[:end] += 1.0 - h[shift:]
            grad[shift:] -= h[:end]
    else:
        s = -shift
        end = n - s
        if end > 0:
            grad[s:] += 1.0 - h[:end]
            grad[:end] -= h[s:]
    return grad

def projected_gradient_descent(initial_h, time_limit):
    n = initial_h.shape[0]
    target = n / 2.0
    h = project_to_box_and_sum(initial_h, target)
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    lr = 0.02
    while time.time() - start < time_limit:
        corr = np.convolve(h, (1.0 - h)[::-1])
        s_idx = int(np.argmax(corr))
        shift = s_idx - (n - 1)
        grad = compute_gradient(h, shift)
        h_new = h - lr * grad
        h_new = project_to_box_and_sum(h_new, target)
        new_score = compute_c5(h_new)
        if new_score < cur_score:
            h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score
        else:
            elapsed = time.time() - start
            frac = elapsed / time_limit
            temp = max(0.001 * (1.0 - frac), 1e-6)
            if np.random.rand() < math.exp((cur_score - new_score) / temp):
                h = h_new
                cur_score = new_score
        # occasional random jitter
        if np.random.rand() < 0.01:
            jitter = (np.random.rand(n) - 0.5) * 0.02
            h = project_to_box_and_sum(h + jitter, target)
            cur_score = compute_c5(h)
        # decay learning rate
        lr = max(lr * 0.9999, 5e-5)
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    total_start = time.time()
    n_points = 200   # moderate resolution

    def make_initial():
        if 'initial_h_values' in globals():
            init = initial_h_values
            N0 = len(init)
            x_old = np.linspace(0.0, 2.0, N0, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
            h0 = np.interp(x_new, x_old, init)
        else:
            h0 = np.random.rand(n_points)
        return project_to_box_and_sum(h0, n_points / 2.0)

    best_h = None
    best_score = float('inf')
    restart = 0
    while True:
        elapsed = time.time() - total_start
        if elapsed >= budget_s * 0.95:
            break
        time_left = budget_s - elapsed
        # each restart gets up to 30 s (or less if remaining time is low)
        restart_time = min(30.0, time_left * 0.8)
        if restart_time < 2.0:
            break
        if restart == 0:
            h0 = make_initial()
        else:
            h0 = project_to_box_and_sum(np.random.rand(n_points), n_points / 2.0)
        h_opt, score_opt = projected_gradient_descent(h0, restart_time)
        if score_opt < best_score:
            best_score = score_opt
            best_h = h_opt.copy()
        restart += 1

    if best_h is None:
        best_h = make_initial()
        best_score = compute_c5(best_h)

    best_h = project_to_box_and_sum(best_h, n_points / 2.0)
    final_score = compute_c5(best_h)
    return best_h, final_score, n_points

import numpy as np
import time
import math
import random

def _project_to_box_and_sum(v, target):
    """
    Project vector v onto the box [0,1]^n while fixing the sum to target.
    Uses a simple bisection on a Lagrange multiplier.
    """
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(v + lo, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        mask = (x > 1e-9) & (x < 1.0 - 1e-9)
        if mask.any():
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            total = x.sum()
            if total > 0:
                x *= target / total
    return x


def _compute_c5(h):
    """Return the Erdős C5 value for a discrete step function h."""
    n = h.shape[0]
    corr = np.convolve(h, (1.0 - h)[::-1])
    dx = 2.0 / n
    return np.max(corr) * dx


def _gradient(h, shift):
    """Gradient of the worst‑shift correlation w.r.t. h (for a fixed shift)."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        end = n - shift
        if end > 0:
            g[:end] += 1.0 - h[shift:]
            g[shift:] -= h[:end]
    else:
        s = -shift
        end = n - s
        if end > 0:
            g[s:] += 1.0 - h[:end]
            g[:end] -= h[s:]
    return g


def _pgd(initial_h, limit):
    """Projected gradient descent with a tiny simulated‑annealing twist."""
    n = initial_h.shape[0]
    target = n / 2.0
    h = _project_to_box_and_sum(initial_h, target)
    cur = _compute_c5(h)
    best_h, best = h.copy(), cur
    start = time.time()
    lr = 0.02

    while time.time() - start < limit:
        # worst‑shift identification
        corr = np.convolve(h, (1.0 - h)[::-1])
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)

        # gradient step
        grad = _gradient(h, shift)
        h_new = _project_to_box_and_sum(h - lr * grad, target)
        new = _compute_c5(h_new)

        if new < cur:
            h, cur = h_new, new
            if new < best:
                best_h, best = h_new.copy(), new
        else:
            # occasional uphill move (simulated annealing)
            elapsed = time.time() - start
            frac = elapsed / limit
            temp = max(0.001 * (1.0 - frac), 1e-6)
            if np.random.rand() < math.exp((cur - new) / temp):
                h, cur = h_new, new

        # small random jitter to escape plateaus
        if np.random.rand() < 0.01:
            jitter = (np.random.rand(n) - 0.5) * 0.02
            h = _project_to_box_and_sum(h + jitter, target)
            cur = _compute_c5(h)

        lr = max(lr * 0.9999, 5e-5)

    return best_h, best


def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h with a low C₅ bound."""
    np.random.seed(seed)
    random.seed(seed)

    total_start = time.time()
    n_points = 200  # a modest resolution balancing quality and speed

    def _make_initial():
        if 'initial_h_values' in globals():
            init = initial_h_values
            N0 = len(init)
            x_old = np.linspace(0.0, 2.0, N0, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
            h0 = np.interp(x_new, x_old, init)
        else:
            h0 = np.random.rand(n_points)
        return _project_to_box_and_sum(h0, n_points / 2.0)

    best_h, best_score = None, float('inf')
    restart = 0

    while True:
        elapsed = time.time() - total_start
        if elapsed >= budget_s * 0.95:
            break
        remaining = budget_s - elapsed
        # allocate up to 30 s per restart, shrinking when time is low
        t_restart = min(30.0, remaining * 0.8)
        if t_restart < 2.0:
            break

        if restart == 0:
            h0 = _make_initial()
        else:
            h0 = _project_to_box_and_sum(np.random.rand(n_points), n_points / 2.0)

        h_opt, sc_opt = _pgd(h0, t_restart)
        if sc_opt < best_score:
            best_h, best_score = h_opt.copy(), sc_opt
        restart += 1

    if best_h is None:
        best_h = _make_initial()
        best_score = _compute_c5(best_h)

    best_h = _project_to_box_and_sum(best_h, n_points / 2.0)
    final_score = _compute_c5(best_h)

    return best_h, final_score, n_points
