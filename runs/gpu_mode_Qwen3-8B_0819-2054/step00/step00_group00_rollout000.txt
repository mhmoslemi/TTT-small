import triton
import triton.language as tl
import torch
import math

def custom_kernel(data):
    """
    Implements the "outgoing" version of the TriMul operator used in AlphaFold3 and similar models.
    This kernel optimizes the forward pass using Triton and leverages efficient memory access patterns.
    The output is a [bs, seq_len, seq_len, dim] tensor with float32 precision.

    Args:
        data: Tuple of (input: torch.Tensor, mask: torch.Tensor, weights: Dict, config: Dict)
            - input: Input tensor of shape [bs, seq_len, seq_len, dim]
            - mask: Mask tensor of shape [bs, seq_len, seq_len]
            - weights: Dictionary of weights for model components
            - config: Dictionary containing configuration parameters

    Returns:
        output: Processed tensor of shape [bs, seq_len, seq_len, dim] with float32 precision
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, n, _, _ = input_tensor.shape

    # Access weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Compute norms for input
    input_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-6
    )

    # Compute projections
    # Left projection
    left_proj = torch.nn.functional.linear(
        input_norm,
        left_proj_weight,
        None
    )
    # Right projection
    right_proj = torch.nn.functional.linear(
        input_norm,
        right_proj_weight,
        None
    )

    # Apply mask
    mask = mask.unsqueeze(-1)  # [bs, seq_len, seq_len, 1]
    left_proj = left_proj * mask
    right_proj = right_proj * mask

    # Compute gates
    left_gate = torch.nn.functional.linear(
        input_norm,
        left_gate_weight,
        None
    ).sigmoid()
    right_gate = torch.nn.functional.linear(
        input_norm,
        right_gate_weight,
        None
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        input_norm,
        out_gate_weight,
        None
    ).sigmoid()

    # Apply gates
    left_proj = left_proj * left_gate
    right_proj = right_proj * right_gate

    # Perform the TriMul operation using Triton
    # Output shape [bs, seq_len, seq_len, hidden_dim]
    output = torch.zeros(
        [bs, n, n, hidden_dim],
        device=input_tensor.device,
        dtype=input_tensor.dtype
    )

    # Triton kernel for TriMul forward pass
    @triton.jit
    def tri_mul_kernel(
        left_ptr, right_ptr, out_ptr,
        n, dim,
        BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
        STRIDE_LN: tl.constexpr, STRIDE_LK: tl.constexpr, STRIDE_LD: tl.constexpr,
        STRIDE_RN: tl.constexpr, STRIDE_RK: tl.constexpr, STRIDE_RD: tl.constexpr,
        STRIDE_ON: tl.constexpr, STRIDE_OL: tl.constexpr, STRIDE_OD: tl.constexpr,
        mask_ptr: tl.constexpr,
    ):
        # Compute the position in the batch
        batch_idx = tl.program_id(0)
        # Compute the index in the sequence
        seq_idx_i = tl.program_id(1)
        seq_idx_j = tl.program_id(2)

        # Create offsets for the current batch, i, j
        offset_n = tl.arange(0, BLOCK_N)
        offset_k = tl.arange(0, BLOCK_K)
        offset_d = tl.arange(0, BLOCK_D)

        # Compute the offset for the current batch, i
        offset_l = (seq_idx_i * n + offset_n) * dim + offset_d
        offset_r = (seq_idx_j * n + offset_n) * dim + offset_d
        offset_o = (seq_idx_i * n + offset_n) * n * dim + (seq_idx_j * dim + offset_d)

        # Compute the offsets
        offs_l = (batch_idx * STRIDE_LN + offset_l[:, None, None] * STRIDE_LD + offset_k[None, :, None] * STRIDE_LK)
        offs_r = (batch_idx * STRIDE_RN + offset_r[:, None, None] * STRIDE_RD + offset_k[None, :, None] * STRIDE_RK)
        offs_o = (batch_idx * STRIDE_ON + offset_o[:, None, None] * STRIDE_OD + offset_d[None, None, :])

        # Load left and right data
        a = tl.load(offs_l, mask=offset_k < n)
        b = tl.load(offs_r, mask=offset_k < n)

        # Compute the matrix multiplication
        out = tl.dot(a, b)

        # Apply the mask
        if mask_ptr is not None:
            mask = tl.load(mask_ptr + batch_idx * STRIDE_LN + seq_idx_i * n + offset_n)
            out = out * mask

        # Store the result
        tl.store(offs_o, out)

    # Launch Triton kernel
    # Shape parameters
    n_val = n
    dim_val = dim
    bs_val = bs

    # Kernel parameters
    num_blocks_n = triton.cdiv(n_val, BLOCK_N)
    num_blocks_k = triton.cdiv(n_val, BLOCK_K)
    num_blocks_d = triton.cdiv(dim_val, BLOCK_D)

    # Grid dimensions
    grid = (bs_val, num_blocks_n, num_blocks_n)

    # Launch the kernel
    tri_mul_kernel[
        grid,
        (BLOCK_N, BLOCK_K, BLOCK_D),
        (n_val, dim_val),
        (n_val, n_val, dim_val),
        (n_val, n_val, dim_val),
        (n_val, n_val, dim_val),
        (n_val, n_val, dim_val),
        (mask is not None)
    ](
        left_proj,
        right_proj,
        output,
        n_val,
        dim_val,
        BLOCK_N=128,
        BLOCK_K=128,
        BLOCK_D=128,
        STRIDE_LN=left_proj.stride(0),
        STRIDE_LK=left_proj.stride(2),
        STRIDE_LD=left_proj.stride(3),
        STRIDE_RN=right_proj.stride(0),
        STRIDE_RK=right_proj.stride(2),
        STRIDE_RD=right_proj.stride(3),
        STRIDE_ON=output.stride(0),
        STRIDE_OL=output.stride(2),
        STRIDE_OD=output.stride(3),
        mask_ptr=mask if mask is not None else None,
    )

    # Apply the output norm
    output = torch.nn.functional.layer_norm(
        output,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-6
    )

    # Apply the output gate
    output = output * out_gate

    # Final projection
    output = torch.nn.functional.linear(
        output,
        to_out_weight,
        None
    )

    return output
