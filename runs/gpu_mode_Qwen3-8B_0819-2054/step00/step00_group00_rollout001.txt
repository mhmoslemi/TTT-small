import torch
import triton
import triton.language as tl

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    B, N, N, C = input_tensor.shape
    dim, hidden_dim = config["dim"], config["hidden_dim"]

    # Access weights from the dictionary
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Output shape
    output_shape = (B, N, N, dim)
    output = torch.zeros(output_shape, device=input_tensor.device, dtype=input_tensor.dtype)

    # Function to apply LayerNorm in Triton
    @triton.jit
    def apply_layer_norm(x_ptr, y_ptr, mean_ptr, var_ptr, weight_ptr, bias_ptr, M, N, C, eps=1e-5):
        pid = tl.program_id(axis=0)
        num_pid_m = tl.cdiv(M, BLOCK_M)
        pid_m = pid // num_pid_m
        pid_n = pid % num_pid_m
        offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
        x = tl.load(x_ptr + (offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn), mask=offs_m[:, None] * stride_xm < M * stride_xm)
        mean = tl.load(mean_ptr + pid_n * BLOCK_N + tl.arange(0, BLOCK_N))
        var = tl.load(var_ptr + pid_n * BLOCK_N + tl.arange(0, BLOCK_N))
        weight = tl.load(weight_ptr + tl.arange(0, C))
        bias = tl.load(bias_ptr + tl.arange(0, C))
        x = (x - mean) * tl.rsqrt(var + eps)
        x = x * weight + bias
        tl.store(y_ptr + (offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn), x)

    # Function to apply activation (sigmoid)
    @triton.jit
    def apply_sigmoid(x_ptr, y_ptr, M, N, C):
        pid = tl.program_id(axis=0)
        num_pid_m = tl.cdiv(M, BLOCK_M)
        pid_m = pid // num_pid_m
        pid_n = pid % num_pid_m
        offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
        x = tl.load(x_ptr + (offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn), mask=offs_m[:, None] * stride_xm < M * stride_xm)
        y = x * (1 - x)
        tl.store(y_ptr + (offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn), y)

    # Function to apply linear transformation in Triton
    @triton.jit
    def apply_linear(x_ptr, y_ptr, weight_ptr, bias_ptr, M, N, C, K):
        pid = tl.program_id(axis=0)
        num_pid_m = tl.cdiv(M, BLOCK_M)
        num_pid_n = tl.cdiv(N, BLOCK_N)
        pid_m = pid // num_pid_n
        pid_n = pid % num_pid_n
        offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
        x = tl.load(x_ptr + (offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn), mask=offs_m[:, None] * stride_xm < M * stride_xm)
        weight = tl.load(weight_ptr + tl.arange(0, K)[:, None] * stride_wk + tl.arange(0, C)[None, :], mask=tl.arange(0, K)[:, None] * stride_wk < K * stride_wk)
        y = tl.dot(x, weight)
        if bias_ptr is not None:
            bias = tl.load(bias_ptr + tl.arange(0, C))
            y += bias
        tl.store(y_ptr + (offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn), y)

    # Function to perform the TriMul operation
    @triton.jit
    def tri_mul_kernel(x_ptr, y_ptr, out_ptr, mask_ptr, M, N, C, K):
        pid = tl.program_id(axis=0)
        num_pid_m = tl.cdiv(M, BLOCK_M)
        num_pid_n = tl.cdiv(N, BLOCK_N)
        pid_m = pid // num_pid_n
        pid_n = pid % num_pid_n
        offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
        x = tl.load(x_ptr + (offs_m[:, None] * stride_xm + tl.arange(0, K)[None, :] * stride_xk + offs_n[None, :] * stride_xn), mask=offs_m[:, None] * stride_xm < M * stride_xm)
        mask = tl.load(mask_ptr + (offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn), mask=offs_m[:, None] * stride_xm < M * stride_xm)
        mask = mask.to(tl.float32)
        x = x * mask
        y = tl.load(y_ptr + (tl.arange(0, K)[:, None] * stride_yk + offs_n[None, :] * stride_yn), mask=tl.arange(0, K)[:, None] * stride_yk < K * stride_yk)
        out = tl.dot(x, y)
        tl.store(out_ptr + (offs_m[:, None] * stride_om + offs_n[None, :] * stride_on), out)

    # Constants for Triton kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 128

    # LayerNorm on input tensor
    mean, var = torch.empty((B, N, N), device=input_tensor.device), torch.empty((B, N, N), device=input_tensor.device)
    input_tensor = input_tensor.view(B, N, N, C)
    tl.layer_norm(input_tensor, mean, var, norm_weight, norm_bias, eps=1e-6)

    # Left_proj
    left = input_tensor @ left_proj_weight.t()
    left = left.reshape(B, N, N, hidden_dim)
    left = left * mask.unsqueeze(-1)

    # Right_proj
    right = input_tensor @ right_proj_weight.t()
    right = right.reshape(B, N, N, hidden_dim)
    right = right * mask.unsqueeze(-1)

    # Left gate
    left_gate = input_tensor @ left_gate_weight.t()
    left_gate = left_gate.reshape(B, N, N, hidden_dim)
    left_gate = torch.sigmoid(left_gate)

    # Right gate
    right_gate = input_tensor @ right_gate_weight.t()
    right_gate = right_gate.reshape(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(right_gate)

    # Out gate
    out_gate = input_tensor @ out_gate_weight.t()
    out_gate = out_gate.reshape(B, N, N, hidden_dim)
    out_gate = torch.sigmoid(out_gate)

    # Scale left and right with gates
    left = left * left_gate
    right = right * right_gate

    # Compute out = einsum('...i k d, ...j k d -> ...i j d', left, right)
    out = torch.einsum('...i k d, ...j k d -> ...i j d', left, right)

    # Apply to_out_norm
    mean_to_out, var_to_out = torch.empty((B, N, N), device=input_tensor.device), torch.empty((B, N, N), device=input_tensor.device)
    out = out.view(B, N, N, hidden_dim)
    out = out * out_gate
    out = out.view(B, N, N, hidden_dim)

    # Apply to_out_norm and to_out
    out = out.view(B, N, N, hidden_dim)
    tl.layer_norm(out, mean_to_out, var_to_out, to_out_norm_weight, to_out_norm_bias, eps=1e-6)
    out = out @ to_out_weight.t()

    # Final output
    return out
