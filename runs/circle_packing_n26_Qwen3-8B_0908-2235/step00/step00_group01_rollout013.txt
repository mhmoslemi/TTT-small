import numpy as np
import scipy.optimize as opt
import matplotlib.pyplot as plt

def run_packing():
    # Initial positions in hexagonal grid
    n = 26
    row_count = int(np.ceil(n / 3))
    col_count = 3
    x_centers = []
    y_centers = []
    for i in range(row_count):
        for j in range(col_count):
            if i % 2 == 0:
                x = j
            else:
                x = j + 0.5
            y = i
            if i * col_count + j < n:
                x_centers.append(x / 2)
                y_centers.append(y / 2)

    centers = np.array([x_centers, y_centers]).T
    radii = np.full(n, 0.1)  # Start with small radii

    # Define the objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum

    # Define constraints
    def constraint_boundary(radii, centers):
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12 or
                y - r < -1e-12 or y + r > 1 + 1e-12):
                return -1
        return 0

    def constraint_overlap(radii, centers):
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return -1
        return 0

    # Create bounds for radii (non-negative)
    bounds = [(0, 1) for _ in range(n)]

    # Optimization using Sequential Least Squares Programming (SLSQP)
    result = opt.minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': lambda r: -constraint_boundary(r, centers)},
            {'type': 'ineq', 'fun': lambda r: -constraint_overlap(r, centers)}
        ],
        method='SLSQP'
    )

    # Retrieve results
    radii_opt = result.x
    centers_opt = centers
    sum_radii = np.sum(radii_opt)

    return centers_opt, radii_opt, sum_radii
