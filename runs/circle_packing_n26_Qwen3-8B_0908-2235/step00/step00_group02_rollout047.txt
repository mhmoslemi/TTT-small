import scipy.optimize as opt

def run_packing():

    cols = int(np.ceil(n / rows))

    radii = np.zeros(n)

        centers[i, 1] = y

        return -np.sum(radii_vec)  # Negative for minimization

    constraints = []

        x, y = centers[i]
        r = radii_vec[i]
        return x - r >= 0 and x + r <= 1 and y - r >= 0 and y + r <= 1

    def non_overlap_constraint(radii_vec, centers, i, j):
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        dist = np.sqrt(dx * dx + dy * dy)
        return dist >= radii_vec[i] + radii_vec[j] - 1e-6

            'type': 'ineq',

                'args': (centers, i, j)
            })

def validate_packing(centers, radii):
