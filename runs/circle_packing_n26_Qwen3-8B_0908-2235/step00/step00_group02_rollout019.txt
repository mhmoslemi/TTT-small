import scipy.optimize as optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    cols = 5
    spacing = 0.25  # space between centers
    centers = np.zeros((rows * cols, 2))
    radii = np.zeros(rows * cols)

    for i in range(rows):

            x = j * spacing + 0.05 * (i % 2)

            centers[i * cols + j] = [x, y]
            radii[i * cols + j] = 0.05  # initial radius

    radii_bounds = [(0.0, 0.1) for _ in range(26)]

    def objective(radii, centers):
        return -np.sum(radii)  # Negative for maximization

    # Constraints: no overlap and within square
    def constraints_func(x):
        centers = x[:52].reshape((26, 2))
        radii = x[52:]
        # Check if all circles are within the unit square
        for i in range(26):

            if (x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):

            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
