import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds

def run_packing():
    # Initial placement: grid layout for 26 circles
    num_circles = 26
    rows = int(np.ceil(np.sqrt(num_circles)))
    cols = num_circles // rows + (1 if num_circles % rows != 0 else 0)
    spacing = 1.0 / (cols + 1)  # Spacing to avoid edges
    initial_centers = np.zeros((num_circles, 2))
    idx = 0
    for r in range(rows):
        for c in range(cols):
            if idx >= num_circles:
                break
            x = spacing * (c + 1)
            y = spacing * (r + 1)
            initial_centers[idx] = [x, y]
            idx += 1

    # Initial guess for radii (approximate)
    initial_radii = np.full(num_circles, 0.03)

    # Objective function: maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Negative for minimization

    # Constraints: Non-overlap and boundary conditions
    def constraint_non_overlap(radii, centers):
        n = len(radii)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist - (radii[i] + radii[j]) < -1e-8:
                    return dist - (radii[i] + radii[j])
        return 0

    def constraint_boundary(radii, centers):
        for i in range(len(radii)):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8:
                return -1
        return 0

    # Optimization problem setup
    bounds = Bounds(0, np.inf, keep_feasible=True)
    cons = (
        {'type': 'ineq', 'fun': lambda x: constraint_non_overlap(x, initial_centers)},
        {'type': 'ineq', 'fun': lambda x: constraint_boundary(x, initial_centers)}
    )

    # Run the optimization
    result = opt.minimize(
        fun=objective,
        x0=initial_radii,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-9, 'disp': False}
    )

    # Extract the result
    radii = result.x
    centers = initial_centers.copy()

    # Check if the optimization was successful
    if not result.success:
        raise RuntimeError("Optimization failed: " + result.message)

    # Validate the packing
    success, message = validate_packing(centers, radii)
    if not success:
        raise RuntimeError("Packing validation failed: " + message)

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
