import numpy as np
import scipy.optimize as opt

def run_packing():

    # We will use a hexagonal grid to place the 26 circles
    # Start with rows and columns to evenly spread the circles inside the unit square.

    # Number of circles
    n = 26

    # Start with a hexagonal-like grid
    rows = int(np.sqrt(n)) + 1
    cols = n // rows + 1

    # Create a 2D grid of circle centers
    centers = np.zeros((n, 2))
    idx = 0
    for r in range(rows):

            if idx >= n:
                break
            x = c / (cols - 1) if cols > 1 else 0.5
            y = r / (rows - 1) if rows > 1 else 0.5
            centers[idx] = np.array([x, y])
            idx += 1

    radii = np.full(n, 0.1)

    # Objective function: maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative of sum (maximize sum)

    # Constraints

        for i in range(n):

                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                constraints.append(dist >= radii[i] + radii[j] - 1e-12)
        return constraints

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
