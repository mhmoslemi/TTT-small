import scipy.optimize as optimize
from scipy import spatial
import math

def run_packing():

    n = 26

    # Start with a hexagonal packing
    # Row 0, 1, ..., k
    # Each row has 1 + 2i circles (for i-th row from the top)
    # We aim to place as many rows as possible in the unit square

    # We will use a compact initial arrangement
    # We'll set up a hexagonal packing and adjust it to fit in the unit square

    # Place circles in hexagonal packing
    r = 0.05  # Initial radius (will be adjusted)
    centers = []
    for row in range(6):
        for col in range(11):
            x = col * (2 * r) + r
            y = row * (r * math.sqrt(3)) + r
            if x >= 1 - 1e-6 or y >= 1 - 1e-6:
                break
            centers.append([x, y])
        if len(centers) >= n:
            break

    # Adjust the radius to fit the unit square
    max_radius = 0.15  # Initial guess
    # Convert to numpy arrays
    centers = np.array(centers)
    radii = np.full(n, max_radius)

    def objective(radii):
        # Objective is to maximize the sum of radii
        return -np.sum(radii)

    def constraints(x):
        # x is a flattened array with [r_0, r_1, ..., r_{n-1}]
        radii = x.reshape((n, 1))

        # Non-negativity constraint
        if np.any(radii < 0):
            return np.inf

        # Boundary constraint
        for i in range(n):

        for i in range(n):
