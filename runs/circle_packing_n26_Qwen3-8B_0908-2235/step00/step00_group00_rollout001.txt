import scipy.optimize as opt
from scipy.spatial.distance import pdist, cdist
import warnings

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Constants
    num_circles = 26
    square_size = 1.0
    tol = 1e-6  # Tolerance for overlap and boundary
    max_iterations = 50
    max_time = 600  # Seconds

    # Helper function to compute pairwise distance between centers
    def pairwise_distance_matrix(centers):
        return cdist(centers, centers)

    # Objective function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # We use negative to minimize it

    # Constraint: Circles do not overlap
    def constraint_overlap(centers, radii):
        dists = pairwise_distance_matrix(centers)
        radii_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
        return dists - radii_sum + tol

    # Constraint: Circles do not exceed square boundary
    def constraint_boundary(centers, radii):
        x, y = centers[:, 0], centers[:, 1]
        x_min, x_max = x - radii, x + radii
        y_min, y_max = y - radii, y + radii
        return np.hstack((x_min - 0 + tol, x_max - 1 + tol, y_min - 0 + tol, y_max - 1 + tol))

    # Sequential packing approach
    centers = np.random.rand(num_circles, 2)  # Initial random positions

    # Optimizer setup

    bounds += [(0.0, square_size - 1e-6) for _ in range(num_circles)]  # For radii

    # Initial optimization

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
