import numpy as np
import scipy.optimize as opt
from scipy.spatial.distance import pdist, squareform

def run_packing():
    # Initial hexagonal grid packing

    # Maximum number of circles
    n = 26

    # Generate initial hexagonal grid
    # Each row has floor((n + 1)/2) circles
    # Starting from the bottom
    centers = np.zeros((n, 2))
    r = 0.12  # initial guess for radius

    # Generate the initial hexagonal grid
    row = 0
    col = 0
    for i in range(n):
        # Horizontal spacing between centers: 2*r
        centers[i, 0] = col * 2 * r
        # Vertical spacing between rows: sqrt(3) * r
        centers[i, 1] = row * np.sqrt(3) * r
        col += 1
        if col == 2:
            col = 0
            row += 1

    # Optimizing positions and radii
    def objective(x):
        # x contains the radii in the order of the centers
        sum_r = np.sum(x)
        return -sum_r  # Minimize -sum_r to maximize sum_r

    def constraint_overlap(x, centers):
        # Distance between circles must be >= r_i + r_j
        dist = squareform(pdist(centers))
        for i in range(len(x)):
            for j in range(i + 1, len(x)):
                if dist[i, j] < x[i] + x[j] - 1e-12:
                    return dist[i, j] - x[i] - x[j] + 1e-12
        return np.inf

    def constraint_boundary(x, centers):
        # Check if all circles are within [0,1]^2
        for i in range(len(x)):
            x_i, y_i = centers[i]
            r_i = x[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12:
                return x_i - r_i - (-1e-12)
            if y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return y_i - r_i - (-1e-12)
        return np.inf

    # Initial guess for radii (same for all)
    x0 = np.full(n, r)

    # Optimization bounds
    bounds = [(0, 0.5) for _ in range(n)]

    # Optimization constraints
    cons = [
        {'type': 'ineq', 'fun': lambda x: constraint_overlap(x, centers)},
        {'type': 'ineq', 'fun': lambda x: constraint_boundary(x, centers)}
    ]

    # Optimization
    result = opt.minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        tol=1e-8
    )

    radii = result.x
    centers = np.array(centers)

    return centers, radii, np.sum(radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
