import scipy.optimize as optimize

    # Start with a grid of 5 rows with 6, 5, 6, 5, 6 circles respectively
    # Then adjust the positions to fit within the square and avoid overlaps

    initial_centers = []
    for row in range(5):
        if row % 2 == 0:
            # Even rows: start with 6 circles
            for col in range(6):
                x = col + 0.5
                y = row + 0.5
                initial_centers.append([x, y])

    initial_radii = np.ones(n_circles) * 0.15

        return -np.sum(radii)  # Minimize negative of sum to maximize sum

    def constraint_no_overlap(i, j):

            # Extract center and radius for circle i
            x_i, y_i = x[2*i], x[2*i+1]
            r_i = radii[i]

            r_j = radii[j]

            dy = y_i - y_j
