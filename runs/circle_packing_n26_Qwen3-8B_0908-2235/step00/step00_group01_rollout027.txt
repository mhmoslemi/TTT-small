from scipy.optimize import minimize
from scipy.optimize._differentiable_functions import ScalarFunction

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial guess: hexagonal packing with equal radii
    num_circles = 26
    initial_radius = 0.05  # Initial guess for radius
    initial_centers = np.zeros((num_circles, 2))

    # Hexagonal packing pattern
    rows = int(np.sqrt(num_circles))
    cols = num_circles // rows + 1
    radius = initial_radius

        row = i // cols
        col = i % cols
        x = col * (2 * radius) + radius

        initial_centers[i] = [x, y]

    max_y = max(initial_centers[:, 1])

    initial_radii = np.full(num_circles, initial_radius * scaling)

    # Objective function: maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Constraints: non-overlapping and within boundaries
    def constraint_overlap(radii, centers):
        n = len(radii)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    return dist - (radii[i] + radii[j] - 1e-8)
        return 0

    def constraint_boundary(radii, centers):

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
