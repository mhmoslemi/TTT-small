import numpy as np
import scipy.optimize as opt

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def create_initial_guess(num_circles):
    # Generate an initial hexagonal grid with some circles placed in corners
    rows = int(np.ceil(np.sqrt(num_circles)))
    cols = int(np.ceil(num_circles / rows))

    centers = np.zeros((num_circles, 2))
    for i in range(num_circles):
        row = i // cols
        col = i % cols
        x = col + 0.5 * (row % 2)
        y = row * np.sqrt(3) / 2
        centers[i] = [x, y]

    # Normalize to [0, 1] square
    centers /= max(np.max(centers), 1)
    return centers

def objective_function(params, num_circles):
    centers = params[:2 * num_circles].reshape(num_circles, 2)
    radii = params[2 * num_circles:]
    return -np.sum(radii)  # Maximize sum of radii

def constraints_function(params, num_circles):

    radii = params[2 * num_circles:]
    
    # Constraint 1: Boundary constraints
    constraints = []
    for i in range(num_circles):
        x, y = centers[i]
        r = radii[i]
        cons = {
            'type': 'ineq',
            'fun': lambda p, i=i, x=x, y=y, r=r: x - r,
            'jac': lambda p, i=i, x=x, y=y, r=r: [1, 0]
        }
        constraints.append(cons)
        cons = {
            'type': 'ineq',
            'fun': lambda p, i=i, x=x, y=y, r=r: 1 - x - r,
            'jac': lambda p, i=i, x=x, y=y, r=r: [-1, 0]
        }
        constraints.append(cons)
        cons = {
            'type': 'ineq',
            'fun': lambda p, i=i, x=x, y=y, r=r: y - r,
            'jac': lambda p, i=i, x=x, y=y, r=r: [0, 1]
        }
        constraints.append(cons)
        cons = {
            'type': 'ineq',
            'fun': lambda p, i=i, x=x, y=y, r=r: 1 - y - r,
            'jac': lambda p, i=i, x=x, y=y, r=r: [0, -1]
        }
        constraints.append(cons)

    # Constraint 2: Non-overlap constraints
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            def dist_func(p):
                x1, y1 = centers[i]
                x2, y2 = centers[j]
                dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
                return dist - (radii[i] + radii[j])
            cons = {
                'type': 'ineq',
                'fun': dist_func,
                'jac': lambda p, i=i, j=j: [0, 0]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26
    initial_centers = create_initial_guess(num_circles)
    initial_radii = np.ones(num_circles) * 0.05

    # Flatten centers and radii to a single vector of parameters
    params = np.concatenate((initial_centers.flatten(), initial_radii))

    # Define constraints
    constraints = constraints_function(params, num_circles)

    # Define bounds: centers must be in [0, 1], radii must be >= 0
    bounds = []
    for _ in range(num_circles):
        bounds.append((0, 1))  # x
        bounds.append((0, 1))  # y
    for _ in range(num_circles):
        bounds.append((0, np.inf))  # radius

    result = opt.minimize(
        objective_function,
        params,
        args=(num_circles,),
        method='SLSQP',
        constraints=constraints,
        bounds=bounds,
        options={'disp': False, 'maxiter': 200}

        raise ValueError("Optimization failed: " + str(result.message))

    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise ValueError(f"Packaging validation failed: {msg}")

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
