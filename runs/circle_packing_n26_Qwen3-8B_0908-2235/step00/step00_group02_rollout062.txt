import numpy as np
import scipy.optimize as optimize
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Define initial approximation for circle positions (hexagonal lattice)
    n = 26
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))
    
    # Initial guess for centers
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // cols
        col = i % cols
        x = col + 0.5 * (row % 2)
        y = row * np.sqrt(3) / 2
        centers[i] = [x, y]
    
    # Normalize so the hexagonal lattice fits in [0, 1] x [0, 1]
    max_x = np.max(centers[:, 0])
    max_y = np.max(centers[:, 1])
    centers /= np.array([max_x, max_y])
    
    # Initial guess for radii
    radii = np.full(n, 1.0 / np.sqrt(n))
    
    # Define optimization variables
    variables = np.hstack((centers.reshape(-1), radii))
    
    # Define the optimization problem
    def objective(variables):
        centers = variables[:2*n].reshape(n, 2)
        radii = variables[2*n:]
        return -np.sum(radii)  # Maximize sum of radii (minimize negative sum)
    
    def constraint_center_inside(variables):
        centers = variables[:2*n].reshape(n, 2)
        radii = variables[2*n:]
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return 1.0  # Violation
        return 0.0
    
    def constraint_non_overlap(variables):
        centers = variables[:2*n].reshape(n, 2)
        radii = variables[2*n:]
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    return 1.0  # Violation
        return 0.0
    
    # Set up constraints
    constraints = [
        {'type': 'ineq', 'fun': constraint_center_inside},
        {'type': 'ineq', 'fun': constraint_non_overlap},
    ]

    bounds = []
    for _ in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0)])
    # Bounds for radii (non-negative)
    for _ in range(n):
        bounds.extend([(0.0, np.inf)])
    
    # Run optimization
    result = optimize.minimize(
        objective,
        variables,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=constraints,
        tol=1e-8,
        options={'maxiter': 1000}
    )

    if not valid:
        raise ValueError(message)
    
    # Return the result
    sum_radii = np.sum(optimized_radii)
    return optimized_centers, optimized_radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
