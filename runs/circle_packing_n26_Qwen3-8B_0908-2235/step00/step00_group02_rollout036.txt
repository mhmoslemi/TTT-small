import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from scipy.optimize import NonlinearConstraint

def run_packing():
    # Number of circles
    n = 26

    # Initialize circle positions in hexagonal-like grid
    def initial_centers():
        # We will use a hexagonal grid with some repeated rows
        # and adjust positions to fit in the unit square
        # Start with 5 rows (top to bottom), 6 circles per row
        centers = []
        for row in range(5):
            for col in range(6):
                # Add slight offset for even rows
                if row % 2 == 0:
                    x = col / 5.5
                else:
                    x = (col + 0.5) / 5.5
                y = row / 5
                centers.append([x, y])
        # Add diagonal circles to fill in gaps
        for row in range(3):
            for col in range(4):
                centers.append([(col + 1) / 4, (row + 0.5) / 4])
        centers.sort()
        return np.array(centers)

    centers = initial_centers()
    radii = np.full(n, 0.05)  # Initial small radii

    # Objective function to minimize (negative of sum of radii)
    def objective(radii):
        return -np.sum(radii)

    # Constraints: non-overlapping and within square
    def constraint_overlap(i, j):
        def cons(radii):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            return dist - (radii[i] + radii[j]) + 1e-8
        return cons

    def constraint_boundary(i):
        def cons(radii):
            x, y = centers[i]
            r = radii[i]
            return np.array([x - r, x + r, y - r, y + r]) - np.array([0, 1, 0, 1]) + 1e-8
        return cons

    # Create nonlinear constraint for all pairwise overlaps
    nonlinear_constraints = []
    for i in range(n):
        for j in range(i+1, n):
            nonlinear_constraints.append(NonlinearConstraint(constraint_overlap(i, j), 0, 0))
    for i in range(n):
        nonlinear_constraints.append(NonlinearConstraint(constraint_boundary(i), 0, 0))

    # Bounds: radii >= 0
    bounds = Bounds(0, np.inf)

    # Run the optimization
    result = minimize(
        fun=objective,
        x0=radii,
        bounds=bounds,
        constraints=nonlinear_constraints,
        method='SLSQP',
        tol=1e-6
    )

    if not result.success:
        raise RuntimeError(f"Optimization failed: {result.message}")

    centers_final = centers
    radii_final = result.x

    if not valid:
        raise RuntimeError(f"Validation failed: {msg}")

    sum_radii = np.sum(radii_final)
    return centers_final, radii_final, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
