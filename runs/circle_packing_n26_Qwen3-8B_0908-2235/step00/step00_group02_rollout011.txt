import scipy.optimize as opt
from scipy.spatial.distance import cdist

def run_packing():

    num_circles = 26
    grid_size = np.ceil(np.sqrt(num_circles))
    spacing = 1.0 / (grid_size + 1)  # to leave space near the boundaries

    centers = np.zeros((num_circles, 2))
    radius_initial = 0.05

    # Hexagonal grid: each row has one more circle than the previous
    for row in range(grid_size):

            x = col * spacing * np.sqrt(3)

    centers /= np.max(centers)

    radii = np.full(num_circles, radius_initial)

    variables = np.hstack((centers.flatten(), radii))

    def objective(x):
        return -np.sum(x[-num_circles:])

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):

            if dist < radii[i] + radii[j] - 1e-12:

    return True, "ok"
