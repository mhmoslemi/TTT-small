import numpy as np
import scipy.optimize as optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial guess for centers and radii
    num_circles = 26
    centers_initial = np.random.rand(num_circles, 2)
    radii_initial = np.random.rand(num_circles) * 0.1

    # Define the optimization variables
    x = np.concatenate([centers_initial.flatten(), radii_initial])

    # Objective function: maximize sum of radii (minimize negative sum)
    def objective(x):
        centers = x[:num_circles * 2].reshape(num_circles, 2)
        radii = x[num_circles * 2:]
        return -np.sum(radii)

    # Constraints: 1. Each circle is inside the unit square
    def boundary_constraints(x):
        centers = x[:num_circles * 2].reshape(num_circles, 2)
        radii = x[num_circles * 2:]
        constraints = []
        for i in range(num_circles):
            x_i, y_i = centers[i]
            r_i = radii[i]
            constraints.append((x_i - r_i >= -1e-12) * 1.0)
            constraints.append((x_i + r_i <= 1 + 1e-12) * 1.0)
            constraints.append((y_i - r_i >= -1e-12) * 1.0)
            constraints.append((y_i + r_i <= 1 + 1e-12) * 1.0)
        return np.array(constraints)

    # Constraints: 2. No circle overlaps with any other
    def no_overlap_constraints(x):
        centers = x[:num_circles * 2].reshape(num_circles, 2)
        radii = x[num_circles * 2:]
        constraints = []
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                constraints.append((dist >= radii[i] + radii[j] - 1e-12) * 1.0)
        return np.array(constraints)

    # Constraints: 3. Non-negative radii
    def non_negative_radii(x):
        radii = x[num_circles * 2:]
        return np.array(radii >= -1e-12)

    # Set up the optimization problem
    bounds = []
    for _ in range(num_circles * 2):
        bounds.append((0.0, 1.0))  # center coordinates between 0 and 1
    for _ in range(num_circles):
        bounds.append((0.0, 1.0))  # radii between 0 and 1

    constraints = [
        {'type': 'ineq', 'fun': boundary_constraints},
        {'type': 'ineq', 'fun': no_overlap_constraints},
        {'type': 'ineq', 'fun': non_negative_radii}
    ]

    # Run the optimization
    result = optimize.minimize(
        objective,
        x,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        tol=1e-8
    )

    # Extract the results
    centers = result.x[:num_circles * 2].reshape(num_circles, 2)
    radii = result.x[num_circles * 2:]
    sum_radii = np.sum(radii)

    # Validate the packing
    valid, message = validate_packing(centers, radii)
    if not valid:
        raise ValueError(f"Packing validation failed: {message}")

    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
