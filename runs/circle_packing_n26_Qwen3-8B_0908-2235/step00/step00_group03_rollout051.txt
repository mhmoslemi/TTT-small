import numpy as np
import scipy.optimize as optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial configuration: Hexagonal packing in a grid-like structure
    def initial_centers(n_circles):
        # Arrange in 5 rows with 5, 5, 5, 5, 6 circles
        rows = [5, 5, 5, 5, 6]
        centers = []
        for i, row in enumerate(rows):
            for j in range(row):
                x = j * (1 + np.sqrt(3)/5) / (np.sqrt(3) + 1)
                y = i * (1 + np.sqrt(3)/5) / (np.sqrt(3) + 1)
                centers.append((x, y))
        return np.array(centers)
    
    # Objective function to maximize the total sum of radii
    def objective(x, n_circles):
        # x is a flattend vector: [x1, y1, r1, x2, y2, r2, ...]
        x_vec = x[:n_circles*2]
        r_vec = x[n_circles*2:]
        return -np.sum(r_vec)  # Minimize negative sum to maximize the sum

    # Constraints
    def constraints(x, n_circles):
        x_vec = x[:n_circles*2]
        r_vec = x[n_circles*2:]
        centers = x_vec.reshape(n_circles, 2)
        radii = r_vec

        # Constraint 1: Each circle must be within the unit square
        def in_square(c, r):
            return np.all((c[0] - r >= 0) & (c[0] + r <= 1) & (c[1] - r >= 0) & (c[1] + r <= 1))
        for i in range(n_circles):
            if not in_square(centers[i], radii[i]):
                return np.inf, f"Circle {i} outside the unit square"

        # Constraint 2: No two circles can overlap
        def no_overlap(c1, r1, c2, r2):
            dx = c1[0] - c2[0]
            dy = c1[1] - c2[1]
            dist_sq = dx*dx + dy*dy
            return dist_sq >= (r1 + r2)**2
        for i in range(n_circles):
            for j in range(i+1, n_circles):
                if not no_overlap(centers[i], radii[i], centers[j], radii[j]):
                    return np.inf, "Circles overlap"

        return 0, "Constraints satisfied"

    # Run optimization with an initial guess based on hexagonal packing
    initial_centers_arr = initial_centers(26)
    initial_radii = np.ones(26) * 0.1  # Initial guess for radii
    initial_guess = np.concatenate((initial_centers_arr.flatten(), initial_radii))

    # Set up the optimization problem
    bounds = []
    for _ in range(26):
        bounds.extend([(0, 1), (0, 1), (0, 0.5)])  # x, y, r for each circle

    result = optimize.minimize(
        fun=lambda x: objective(x, 26),
        x0=initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[lambda x: constraints(x, 26)],
        options={'ftol': 1e-9, 'maxiter': 500}
    )

    if not result.success:
        raise RuntimeError(f"Optimization failed with message: {result.message}")

    centers = result.x[:26*2].reshape(26, 2)
    radii = result.x[26*2:]
    sum_radii = np.sum(radii)

    # Validate the packing
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise RuntimeError(msg)

    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
