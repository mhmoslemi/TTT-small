import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial estimate of the number of rows and columns in the hexagonal grid
    n_circles = 26
    num_rows = int(np.ceil(np.sqrt(n_circles)))
    num_cols = int(np.ceil(n_circles / num_rows))

    # Calculate the diameter and spacing for the hexagonal grid
    circle_diameter = 1.0
    spacing = circle_diameter / np.sqrt(3)  # Hexagonal packing spacing

    # Generate initial positions for the hexagonal grid
    centers = np.zeros((n_circles, 2), dtype=np.float64)
    radius = np.ones(n_circles, dtype=np.float64)

    for i in range(n_circles):
        row = i // num_cols
        col = i % num_cols
        x = col * (1.5 * spacing) + 0.5 * spacing  # Offset for hexagonal grid
        y = row * spacing
        centers[i] = [x, y]

    # Define the objective function to maximize the sum of radii
    def objective(x):
        x = x.reshape((n_circles, 2))
        return -np.sum(x[:, 2])  # Maximize sum of radii by minimizing its negative

    # Define constraints
    def box_constraint(x):
        return np.array([x[i, 0] - radius[i] >= 0 for i in range(n_circles)] +
                        [1.0 - x[i, 0] - radius[i] >= 0 for i in range(n_circles)] +
                        [x[i, 1] - radius[i] >= 0 for i in range(n_circles)] +
                        [1.0 - x[i, 1] - radius[i] >= 0 for i in range(n_circles)])

    # Combine constraints
    cons = [{'type': 'ineq', 'fun': box_constraint}, {'type': 'ineq', 'fun': non_overlap_constraint}]

    result = minimize(objective, x0, method='SLSQP', constraints=cons, tol=1e-10)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
