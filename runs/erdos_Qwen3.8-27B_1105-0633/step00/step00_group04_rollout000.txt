def smooth_max_grad(h, alpha):

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    w = np.exp((corr - m) / alpha)
    w /= np.sum(w)
    
    obj = (m + alpha * np.log(np.sum(w))) * dx
    
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        for k in range(len(corr)):

            if 0 <= i1 < n:
                g += w[k] * (1.0 - h[i1])
            i2 = j + k
            if 0 <= i2 < n:
                g -= w[k] * h[i2]
        grad[j] = dx * g
    
    return obj, grad

        h_new, obj_new = slsqp_solve(best_h, t_limit, alphas)
        try_update(h_new)
    
    # Phase 5: Another round of pair moves + SLSQP
    if remaining() > 30:
        t_limit = min(remaining() * 0.2, 60.0)
        h_new, obj_new = pair_move_optim(best_h, n_passes=5000, time_limit=t_limit)
        try_update(h_new)
    
    if remaining() > 20:
        t_limit = min(remaining() * 0.3, 60.0)
        alphas = [0.001, 0.0005, 0.0003]
        h_new, obj_new = slsqp_solve(best_h, t_limit, alphas)
        try_update(h_new)
    
    # Phase 6: Final local polish
    if remaining() > 15:
        t_limit = min(remaining() * 0.4, 60.0)
        h_new, obj_new = pair_move_optim(best_h, n_passes=5000, time_limit=t_limit)
        try_update(h_new)
    
    # Try a few more diverse starts if time remains
    if remaining() > 40:
        for s in starts[4:]:
            if remaining() < 30:
                break
            if compute_c5_exact(s) < best_c5 + 0.02:
                h_new, obj_new = slsqp_solve(s, min(30.0, remaining() * 0.3), [0.02, 0.01, 0.005])
                try_update(h_new)
    
    # Final projection
    best_h = project_feasible(best_h, target_sum)
    final_c5 = compute_c5_exact(best_h)
    
    return (best_h, final_c5, n)
