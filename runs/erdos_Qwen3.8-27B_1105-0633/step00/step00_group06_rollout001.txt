    """Project x onto {h: 0 <= h[i] <= 1, sum(h) = target_sum}"""
    n = len(x)
    # Binary search on lambda
    lo, hi = -np.max(np.abs(x)) - 2, np.max(np.abs(x)) + 2
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x + mid, 0, 1)
        s = np.sum(h)
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo+hi)/2, 0, 1)

    x_max = np.max(x)
    return x_max + np.log(np.sum(np.exp(x - x_max))) / t

def project_feasible(x, target_sum):
    """Project x onto {h: 0<=h[i]<=1, sum(h)=target_sum} using water-filling"""
    n = len(x)
    lo = x.min() - target_sum - 1
    hi = x.max() - target_sum + 1
    for _ in range(200):
        mid = (lo + hi) / 2
        h = np.clip(x + mid, 0, 1)
        s = np.sum(h)
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2, 0, 1)
    # Fix exact sum
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        free = (h > 1e-12) & (h < 1 - 1e-12)
        nfree = np.sum(free)
        if nfree > 0:
            adj = diff / nfree
            if 1e-12 <= h[free].min() + adj and h[free].max() + adj <= 1 - 1e-12:
                h[free] += adj
            else:
                # More careful adjustment
                h = np.clip(h + diff / n * np.ones(n), 0, 1)
    return h

def subgradient_step(h, step_size):
    """One step of projected subgradient method"""
    n = len(h)
    c = np.correlate(h, 1-h, mode='full')
    j_star = np.argmax(c)
    lag = j_star - (n - 1)
    # Gradient of c[j_star] with respect to h
    grad = np.zeros(n)
    # From h[n+lag] term: grad[m] += (1-h[m-lag]) if 0 <= m-lag < n
    if lag >= 0:
        # h[n+lag] is defined for n = 0..n-1-lag
        # m = n+lag, so m ranges from lag to n-1
        # grad[m] += (1-h[m-lag])
        grad[lag:] += (1 - h[:n-lag])
        # h[n] term: grad[m] -= h[m+lag] if 0 <= m+lag < n
        # m ranges from 0 to n-1-lag
        grad[:n-lag] -= h[lag:]
    else:
        # lag < 0, let m = -lag > 0
        # h[n+lag] = h[n-m] is defined for n = m..n-1
        # m_idx = n+lag = n-m, ranges from 0 to n-1-m
        # grad[n-m] += (1-h[n]) for n = m..n-1
        # So grad[0:n-m] += (1-h[m:])
        grad[:n+m] += (1 - h[-m:])  # h[m:] has length n-m
        # Wait, let me redo this.
        # c = sum_n h[n+lag] * (1-h[n]) where lag < 0
        # Let lag = -m, m > 0
        # c = sum_n h[n-m] * (1-h[n]) for n = m..n-1 (so n-m >= 0)
        # grad w.r.t. h[k]:
        #   From h[n-m] = h[k] when n = k+m: contributes (1-h[k+m])
        #   From (1-h[n]) = (1-h[k]) when n = k: contributes -h[k-m]
        # So grad[k] = (1-h[k+m]) * [k+m < n] - h[k-m] * [k-m >= 0]

        grad[m:] -= h[:n-m]  # h[k-m] for k = m..n-1
    
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n / 2.0)
