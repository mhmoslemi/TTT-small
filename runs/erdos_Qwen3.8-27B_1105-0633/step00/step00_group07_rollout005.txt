```python
def eval_c_t(f, g, t, dx):
    """Evaluate C(t) = dx * max_k (f[k] + t*g[k])"""
    return dx * np.max(f + t * g)

def ternary_search(f, g, t_lo, t_hi, dx, iters=80):
    """Minimize C(t) over [t_lo, t_hi] using ternary search."""
    for _ in range(iters):
        m1 = t_lo + (t_hi - t_lo) / 3.0
        m2 = t_hi - (t_hi - t_lo) / 3.0
        f1 = eval_c_t(f, g, m1, dx)
        f2 = eval_c_t(f, g, m2, dx)
        if f1 < f2:
            t_hi = m2
        else:
            t_lo = m1
    return (t_lo + t_hi) / 2.0
```
