    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new

    h_new = np.zeros(n_new)

        x_hi = (j + 1) * dx_new

        i_hi = int(np.ceil(x_hi / dx_old))

def project_polytope(y, target):
    n = len(y)
    def f(t):
        return np.clip(y + t, 0.0, 1.0).sum() - target
    lo = -y.max() - 0.5
    hi = 1.0 - y.min() + 0.5
    if f(lo) >= 0:
        h = np.clip(y + lo, 0.0, 1.0)
    elif f(hi) <= 0:
        h = np.clip(y + hi, 0.0, 1.0)
    else:
        t = brentq(f, lo, hi, xtol=1e-13)
        h = np.clip(y + t, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-11:
        if residual > 0:
            order = np.argsort(-(1.0 - h))
        else:
            order = np.argsort(-h)
        for i in order:
            if abs(residual) < 1e-12:
                break
            if residual > 0:
                add = min(residual, 1.0 - h[i])
                h[i] += add
                residual -= add
            else:
                rem = min(-residual, h[i])
                h[i] -= rem
                residual += rem
    return h

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def lse_grad(h, beta):
    n = len(h)
    lags = np.correlate(h, 1.0 - h, mode='full')
    lmax = lags.max()
    expv = np.exp(beta * (lags - lmax))
    w = expv / expv.sum()
    
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        d = i - (n - 1)
        wi = w[i]
        if wi < 1e-12:
            continue
        lo = max(0, -d)
        hi = n - max(0, d)
        # Term 1: h[n+d] factor -> grad at j=n+d gets (1-h[n])
        a1 = max(lo + d, 0)
        b1 = min(hi + d, n)
        if a1 < b1:
            grad[a1:b1] += wi * (1.0 - h[a1-d:b1-d])
        # Term 2: (1-h[n]) factor -> grad at j=n gets -h[n+d]
        a2 = max(lo, 0)
        b2 = min(hi, n)
        if a2 < b2:
            grad[a2:b2] -= wi * h[a2+d:b2+d]
    return grad, lmax + np.log(expv.sum()) / beta

def fw_step(h, grad, target):
    """Frank-Wolfe: find vertex z minimizing grad^T z over P_n, then line search."""
    n = len(h)
    order = np.argsort(grad)
    z = np.zeros(n)
    rem = target
    for i in order:
        if rem <= 0:
            break
        z[i] = min(1.0, rem)
        rem -= z[i]
    return z

def reshape_h(h, n_new):
    n = len(h)
    dx_o = 2.0 / n
    dx_n = 2.0 / n_new
    # Build using interpolation - for each new point, find contributing old cells
    x_new = (np.arange(n_new) + 0.5) * dx_n
    # For each new x, find which old cell it falls in and interpolate
    old_idx = (x_new / dx_o).astype(int)
    old_idx = np.clip(old_idx, 0, n - 1)
    # Simple nearest-neighbor would lose area; use trapezoidal
    # Better: compute area overlap
    h_new = np.zeros(n_new)
    for j in range(n_new):
        x_lo = j * dx_n
        x_hi = (j + 1) * dx_n
        i_start = max(0, int(x_lo / dx_o))
        i_end = min(n, int(np.ceil(x_hi / dx_o)))
        for i in range(i_start, i_end):
            o_lo = i * dx_o
            o_hi = o_lo + dx_o
            overlap = min(x_hi, o_hi) - max(x_lo, o_lo)
            if overlap > 0:
                h_new[j] += h[i] * overlap
        h_new[j] /= dx_n
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_polytope(h_new, n_new / 2.0)
