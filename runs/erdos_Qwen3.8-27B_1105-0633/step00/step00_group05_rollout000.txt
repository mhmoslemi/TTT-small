import numpy as np
import math
from scipy.optimize import minimize


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full') * dx))


def project_box_affine(h, target_sum):
    x = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        s = float(np.sum(x))
        diff = target_sum - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            free = x < 1.0 - 1e-14
        else:
            free = x > 1e-14
        if not np.any(free):
            break
        x[free] += diff / float(np.sum(free))
        np.clip(x, 0.0, 1.0, out=x)
    # Final fine-tune
    s = float(np.sum(x))
    diff = target_sum - s
    if abs(diff) > 1e-15:
        free = np.abs(x - 0.5) > 1e-6
        if np.any(free):
            x[free] += diff / float(np.sum(free))
            np.clip(x, 0.0, 1.0, out=x)
    return x


class SmoothObj:
    def __init__(self, n, alpha):
        self.n = n
        self.dx = 2.0 / n
        self.alpha = alpha
        self.shifts = np.arange(-(n - 1), n)

    def compute_F(self, h):
        n = self.n
        dx = self.dx
        F = np.zeros(2 * n - 1)
        for k in range(2 * n - 1):
            s = int(self.shifts[k])
            i_lo = max(0, -s)
            i_hi = min(n, n - s)
            if i_lo >= i_hi:
                continue
            idx1 = np.arange(i_lo, i_hi)
            idx2 = idx1 + s
            F[k] = dx * float(np.dot(h[idx1], 1.0 - h[idx2]))
        return F

    def __call__(self, h):
        F = self.compute_F(h)
        c = float(np.max(F))
        z = self.alpha * (F - c)
        max_z = float(np.max(z))
        exp_z = np.exp(z - max_z)
        s = float(np.sum(exp_z))
        logsumexp = max_z + math.log(s)
        phi = c + logsumexp / self.alpha
        # Gradient
        n = self.n
        dx = self.dx
        weights = exp_z / s
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            w = weights[k]
            if w < 1e-14:
                continue
            s_shift = int(self.shifts[k])
            i_lo = max(0, -s_shift)
            i_hi = min(n, n - s_shift)
            if i_lo >= i_hi:
                continue
            idx1 = np.arange(i_lo, i_hi)
            idx2 = idx1 + s_shift
            grad[idx1] += w * dx * (1.0 - h[idx2])
            grad[idx2] += w * dx * (-h[idx1])
        return phi, grad


class SumCon:
    def __init__(self, target):
        self.target = target

    def __call__(self, h):
        return float(np.sum(h)) - self.target


class SumJac:
    def __init__(self, n):
        self.ones = np.ones(n)

    def __call__(self, h):
        return self.ones


def interpolate_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    # Treat h as cell-centered values on [0,2]
    x_src = (np.arange(n_src) + 0.5) / n_src * 2.0
    x_dst = (np.arange(n_dst) + 0.5) / n_dst * 2.0
    h_dst = np.interp(x_dst, x_src, h_src, left=h_src[0], right=h_src[-1])
    return h_dst
