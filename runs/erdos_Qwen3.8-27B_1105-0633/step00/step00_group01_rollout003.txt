def exact_c5(h):
    """Compute exact C5 for a feasible h vector."""
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    terms = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(terms)), terms

def repair_feasibility(h, target_mass):
    """Repair h to exactly satisfy 0<=h<=1 and sum(h)=target_mass."""
    h = np.clip(h, 0.0, 1.0)
    h = np.where(np.isfinite(h), h, 0.5)
    n = len(h)
    target_mass = n / 2.0
    diff = target_mass - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Try uniform shift with clipping
    # If diff > 0, we need to add mass; if diff < 0, remove
    lo = -max(h) if diff > 0 else -(1.0 - h).min()
    hi = (1.0 - h).min() if diff > 0 else h.max()
    # Binary search for lambda such that sum(clip(h + lambda, 0, 1)) = target_mass
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        trial = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(trial)
        if s < target_mass:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + 0.5*(lo+hi), 0.0, 1.0)
    # Fix residual exactly
    residual = target_mass - np.sum(h)
    if abs(residual) > 0:
        # Find an index where we can adjust
        if residual > 0:
            # Need to add mass, find index with h < 1
            for i in range(n):
                room = 1.0 - h[i]
                if room > 1e-15:
                    adj = min(residual, room)
                    h[i] += adj
                    residual -= adj
                    if abs(residual) < 1e-14:
                        break
        else:
            for i in range(n):
                room = h[i]
                if room > 1e-15:
                    adj = min(-residual, room)
                    h[i] -= adj
                    residual += adj
                    if abs(residual) < 1e-14:
                        break
    return h

import numpy as np
from scipy.optimize import minimize
import time

_clock_start = 0.0
_clock_budget = 1100.0

def _elapsed():
    return time.time() - _clock_start

def _time_left():
    return _clock_budget - _elapsed()

def _exact_c5(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def _exact_terms(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _repair(h):
    h = np.array(h, dtype=np.float64, copy=True)
    h = np.where(np.isfinite(h), h, 0.5)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        lo, hi = 0.0, (1.0 - h).min() + 1e-9
    else:
        lo, hi = -h.max() - 1e-9, 0.0
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        idx = np.arange(n)
        rng = np.random.default_rng(42)
        rng.shuffle(idx)
        for i in idx:
            if residual > 0:
                room = 1.0 - h[i]
                if room > 1e-15:
                    adj = min(residual, room)
                    h[i] += adj
                    residual -= adj
                    if abs(residual) < 1e-15:
                        break
            else:
                if h[i] > 1e-15:
                    adj = min(-residual, h[i])
                    h[i] -= adj
                    residual += adj
                    if abs(residual) < 1e-15:
                        break
    return h

def _epi_objective(x):
    return x[-1]

def _epi_obj_grad(x):
    g = np.zeros(len(x))
    g[-1] = 1.0
    return g

def _epi_ineq_vals(x):
    h = x[:-1]
    t = x[-1]
    return t - _exact_terms(h)

def _epi_ineq_grad(x):
    h = x[:-1]
    n = len(h)
    dx = 2.0 / n
    m = 2 * n - 1
    jac = np.zeros((m, n + 1))
    jac[:, -1] = 1.0
    for q in range(n):
        for p in range(n):
            k = (p - q) + (n - 1)
            if 0 <= k < m:
                jac[k, p] -= dx * (1.0 - h[q])
                jac[k, q] -= dx * h[p]
    return jac

def _epi_eq(x):
    n = len(x) - 1
    return np.sum(x[:n]) - n / 2.0

def _epi_eq_grad(x):
    g = np.zeros(len(x))
    g[:len(x)-1] = 1.0
    return g

def _run_epigraph_slsqp(h0, max_time=60.0):
    h0 = _repair(h0.copy())
    n = len(h0)
    c0 = _exact_c5(h0)
    x0 = np.concatenate([h0, [c0 + 1e-6]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [
        {'type': 'ineq', 'fun': _epi_ineq_vals, 'jac': _epi_ineq_grad},
        {'type': 'eq', 'fun': _epi_eq, 'jac': _epi_eq_grad}
    ]
    try:
        res = minimize(_epi_objective, x0, method='SLSQP', jac=_epi_obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 500, 'ftol': 1e-10})
    except Exception:
        return h0, c0
    if _time_left() < 5:
        return h0, c0
    h_new = _repair(res.x[:n])
    c_new = _exact_c5(h_new)
    if c_new < c0:
        return h_new, c_new
    return h0, c0

def _smooth_max_objective(x, beta):
    h = x
    terms = _exact_terms(h)
    m = np.max(terms)
    shifted = beta * (terms - m)
    shifted = np.clip(shifted, -500, 500)
    log_sum = m + np.log(np.sum(np.exp(shifted))) / beta
    penalty = 0.0
    if np.any(h < 0) or np.any(h > 1):
        penalty = 100.0 * (np.sum(np.clip(-h, 0, None)**2) + np.sum(np.clip(h - 1, 0, None)**2))
    return float(log_sum + penalty)

def _smooth_max_grad(x, beta):
    h = x
    n = len(h)
    dx = 2.0 / n
    terms = _exact_terms(h)
    m = np.max(terms)
    shifted = beta * (terms - m)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        d = k - (n - 1)
        w = weights[k]
        if d >= 0:
            for q in range(n - d):
                p = q + d
                grad[p] += w * dx * (1.0 - h[q])
                grad[q] += w * dx * (-h[p])
        else:
            for q in range(-d, n):
                p = q + d
                grad[p] += w * dx * (1.0 - h[q])
                grad[q] += w * dx * (-h[p])
    if np.any(h < 0):
        grad[h < 0] += 200.0 * (-h[h < 0])
    if np.any(h > 1):
        grad[h > 1] += 200.0 * (h[h > 1] - 1.0)
    return grad

def _run_smooth_slsqp(h0, beta=50.0, max_time=30.0):
    h0 = _repair(h0.copy())
    n = len(h0)
    c0 = _exact_c5(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_smooth_max_objective, h0, args=(beta,),
                       method='SLSQP', jac=lambda x: _smooth_max_grad(x, beta),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 300, 'ftol': 1e-9})
    except Exception:
        return h0, c0
    if _time_left() < 5:
        return h0, c0
    h_new = _repair(res.x)
    c_new = _exact_c5(h_new)
    if c_new < c0:
        return h_new, c_new
    return h0, c0

def _pair_swap_optimize(h, max_pairs=5000, seed=42):
    h = _repair(h.copy())
    n = len(h)
    best_c = _exact_c5(h)
    rng = np.random.default_rng(seed)
    for _ in range(max_pairs):
        if _time_left() < 10:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        if h[i] <= 1e-10 or h[j] >= 1.0 - 1e-10:
            continue
        max_d = min(h[i], 1.0 - h[j])
        for frac in (0.3, 0.5, 0.7):
            delta = frac * max_d
            h2 = h.copy()
            h2[i] -= delta
            h2[j] += delta
            c2 = _exact_c5(h2)
            if c2 < best_c - 1e-10:
                h = h2
                best_c = c2
                break
    return h, best_c

def _resample_to_n(h, target_n):
    n = len(h)
    if n == target_n:
        return _repair(h.copy())
    x_old = np.linspace(0, 2, n, endpoint=False) + 1.0/n
    x_new = np.linspace(0, 2, target_n, endpoint=False) + 1.0/target_n
    h_new = np.interp(x_new, x_old, h, period=None)
    h_new = np.clip(h_new, 0.0, 1.0)
    return _repair(h_new)

def _make_binary_start(n, seed=0):
    rng = np.random.default_rng(seed)
    h = np.zeros(n)
    h[rng.choice(n, n//2, replace=False)] = 1.0

def _make_periodic_start(n, period, seed=0):

    for i in range(n):
