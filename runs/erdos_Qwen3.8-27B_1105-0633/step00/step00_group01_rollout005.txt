    exp_terms = np.exp(beta * (terms - log_max))
    weights = exp_terms / np.sum(exp_terms)
    
    grad = np.zeros(n)

        if w < 1e-15:

        # Positive part: for t=d to n-1, add (1-h[t-d]) = b[0] to b[n-1-d]
        grad[d:] += w * b[:n-d]
        # Negative part: for t=0 to n-1-d, subtract h[t+d] = h[d] to h[n-1]
        grad[:n-d] -= w * h[d:]
    
    return grad, terms, weights

    lo = -1.0
    hi = 2.0  # ensure f(lo) > target_sum > f(hi)
    # Actually: f(lo) = sum(clamp(v-lo, 0, 1)) = sum(1) if v-lo > 0 for all, = n
    #           f(hi) = sum(clamp(v-hi, 0, 1)) = sum(0) if v-hi < 0 for all, = 0
    # Need: v - lo > 1 for all (so clamp = 1): lo < min(v) - 1
    #       v - hi < 0 for all (so clamp = 0): hi > max(v)
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    
    for _ in range(100):
        mid = (lo + hi) / 2
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-12:
            break
    
    lambda_val = (lo + hi) / 2
    return np.clip(v - lambda_val, 0.0, 1.0)
