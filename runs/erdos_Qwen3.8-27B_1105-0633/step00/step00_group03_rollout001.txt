```python
def smooth_obj_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx  # length 2n-1
    m = np.max(corr)
    tc = tau * (corr - m)
    ev = np.exp(tc)
    W = ev.sum()
    F = m + np.log(W) / tau
    w = ev / W  # length 2n-1
    
    # shifts: output index i corresponds to shift k = i - (n-1)
    k_vals = np.arange(-(n-1), n)  # length 2n-1
    p_vals = np.arange(n)
    
    # First term: w[k] * (1-h[p+k]) for 0 <= p+k < n
    pk = p_vals[:, None] + k_vals[None, :]  # shape (n, 2n-1)
    valid1 = (pk >= 0) & (pk < n)
    
    # Second term: w[k] * h[p-k] for 0 <= p-k < n
    pk2 = p_vals[:, None] - k_vals[None, :]  # shape (n, 2n-1)
    valid2 = (pk2 >= 0) & (pk2 < n)
    
    # Clip indices for safe access
    pk_clip = np.clip(pk, 0, n-1)
    pk2_clip = np.clip(pk2, 0, n-1)
    
    first = np.where(valid1, w[None, :] * (1.0 - h[pk_clip]), 0.0)
    second = np.where(valid2, w[None, :] * h[pk2_clip], 0.0)
    
    grad = dx * (first.sum(axis=1) - second.sum(axis=1))
    return F, grad
```
