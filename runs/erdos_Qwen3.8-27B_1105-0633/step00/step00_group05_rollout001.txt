def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(x, target_sum):
    n = len(x)
    target_sum = float(target_sum)
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(x - lam, 0.0, 1.0)
    diff = target_sum - np.sum(result)
    if abs(diff) > 1e-10:
        if diff > 0:
            candidates = np.where(result < 1.0 - 1e-12)[0]
        else:
            candidates = np.where(result > 1e-12)[0]
        for idx in candidates:
            adj = min(diff, 1.0 - result[idx]) if diff > 0 else max(diff, -result[idx])
            result[idx] += adj
            diff -= adj
            if abs(diff) < 1e-12:
                break
    return result

def lse_value(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    F = corr * dx
    c = np.max(F)
    shifted = alpha * (F - c)
    m = np.max(shifted)
    if m < -700:
        return c
    return c + (m + np.log(np.sum(np.exp(shifted - m)))) / alpha

def lse_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    F = corr * dx
    c = np.max(F)
    shifted = alpha * (F - c)
    m = np.max(shifted)
    exp_s = np.exp(shifted - m)
    w = exp_s / np.sum(exp_s)
    grad = np.zeros(n)
    for mrow in range(len(F)):
        if w[mrow] < 1e-15:
            continue
        ell = (n - 1) - mrow
        lo = max(0, -ell)
        hi = min(n, n - ell)
        if lo >= hi:
            continue
        idx1 = np.arange(lo, hi)
        idx2 = idx1 + ell
        grad[idx1] += w[mrow] * dx * (1.0 - h[idx2])
        grad[idx2] += w[mrow] * dx * (-h[idx1])
    return grad

class LSEObj:
    def __init__(self, alpha):
        self.alpha = alpha
    def f(self, x):
        return lse_value(x, self.alpha)
    def g(self, x):
        return lse_gradient(x, self.alpha)

class SumCon:
    def __init__(self, n):
        self.target = n / 2.0
        self.n = n
    def f(self, x):
        return np.sum(x) - self.target
    def j(self, x):
        return np.ones(self.n)

def binary_exhaustive_swap(h_bin, n):
    """Try all single swaps, return best."""
    best_c5 = compute_c5(h_bin)
    improved = True
    while improved:
        improved = False
        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
        best_swap = None
        best_val = best_c5
        for i in ones:
            for j in zeros:
                if i == j:
                    continue
                h_bin[i] = 0.0
                h_bin[j] = 1.0
                c = compute_c5(h_bin)
                if c < best_val - 1e-15:
                    best_val = c
                    best_swap = (i, j)
                else:
                    h_bin[i] = 1.0
                    h_bin[j] = 0.0
        if best_swap is not None:
            i, j = best_swap
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            best_c5 = best_val
            improved = True
    return best_c5
