    else:
        order = np.argsort(-h)
        remaining = -diff
        for i in order:
            sub = min(h[i], remaining)
            h[i] -= sub
            remaining -= sub
            if remaining <= 1e-13:
                break
    diff2 = target - np.sum(h)
    if abs(diff2) > 1e-14:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(interior) > 0:
            h[interior] += diff2 / np.sum(interior)
            h = np.clip(h, 0.0, 1.0)
    return h

def optimize_pair(h, p, q, dx):
    s = h[p] + h[q]
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-9:
        return False
    hp_orig = h[p]
    
    def obj(x):
        h[p] = x
        h[q] = s - x
        c = np.correlate(h, 1.0 - h, mode="full")
        return float(np.max(c) * dx)
    
    orig_val = obj(hp_orig)
    best_x = hp_orig
    best_val
