import numpy as np
from scipy.optimize import minimize_scalar
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h):
    h = np.array(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = 0.5 * n
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    mask = (h > 0.001) & (h < 0.999)
    if mask.sum() > 0:
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        for i in range(n):
            if diff > 0:
                step = min(diff, 1.0 - h[i])
            else:
                step = max(diff, -h[i])
            h[i] += step
            diff -= step
            if abs(diff) < 1e-14:
                break
    h = np.clip(h, 0.0, 1.0)
    return h

    """Optimize h along the pair (p, q) direction. Returns (new_h, new_score)."""
    n = len(h)
    s = h[p] + h[q]
    L = max(0.0, s - 1.0)
    U = min(1.0, s)
    if U - L < 1e-12:
        return h, exact_score(h)
    
    orig_p = h[p]
    orig_q = h[q]
    current_score = exact_score(h)
    
    def obj(t):
        h[p] = t
        h[q] = s - t
        return exact_score(h)

    best_t = orig_p
    best_val = current_score
    t_vals = [L, U, orig_p]
    for t in t_vals:
        val = obj(t)
        if val < best_val:

            best_t = t

            best_t = res.x

        pass
