    h = np.clip(np.asarray(x, dtype=np.float64).copy(), 0.0, 1.0)

        return h
    if diff > 0:

            rng.shuffle(pairs)
            pairs = pairs[:num_pairs]
        elif pass_count % 3 == 0:
            # All pairs with some random
            pairs = []
            for i in range(n - 1):
                pairs.append((i, i + 1))
            for i in range(n - 2):
                pairs.append((i, i + 2))
            for i in range(n - 3):
                pairs.append((i, i + 3))
            for _ in range(n):
                i, j = int(rng.integers(0, n)), int(rng.integers(0, n))
                if i != j:
                    pairs.append((min(i, j), max(i, j)))
            rng.shuffle(pairs)
        else:
            # Adjacent + random
            pairs = []
            for i in range(n - 1):
                pairs.append((i, i + 1))
            for i in range(n - 2):
                pairs.append((i, i + 2))
            for _ in range(n // 3):
                i, j = int(rng.integers(0, n)), int(rng.integers(0, n))
                if i != j:
                    pairs.append((min(i, j), max(i, j)))

        for (p, q) in pairs:
            if time.time() > deadline:
                break

            d = compute_d(h, g, p, q, n)

            x = h[p]
            y = h[q]
            lo = max(-x, y - 1.0)
            hi = min(1.0 - x, y)
            if hi - lo < 1e-14:
                continue

            best_t, best_v = pair_minimize(c, d, lo, hi)

            if best_v < current_score - 1e-11:
                h[p] = x + best_t
                h[q] = y - best_t
                g[p] = 1.0 - h[p]
                g[q] = 1.0 - h[q]
                c = c + best_t * d
                current_score = best_v
                pass_improved = True
                if current_score < best_score:
                    best_score = current_score
                    best_h = h.copy()

        if pass_improved:
            no_improve = 0
            perturbation_scale *= 0.95
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                if time.time() > deadline - 30:
                    break
                # Perturbation restart
                pert = (rng.random(n) - 0.5) * 2.0 * perturbation_scale
                h_pert = project_feasible(h + pert, n)
                c_pert = compute_correlation(h_pert, n)
                s_pert = float(np.max(c_pert))
                if s_pert < current_score + 0.005:
                    h = h_pert
                    c = c_pert
                    g = 1.0 - h
                    current_score = s_pert
                else:
                    # Try smaller perturbation
                    pert_small = (rng.random(n) - 0.5) * 2.0 * perturbation_scale * 0.3
                    h_pert = project_feasible(h + pert_small, n)
                    c_pert = compute_correlation(h_pert, n)
                    s_pert = float(np.max(c_pert))
                    if s_pert < current_score + 0.002:
                        h = h_pert
                        c = c_pert
                        g = 1.0 - h
                        current_score = s_pert
                        pass_improved = True
                no_improve = 0
                if perturbation_scale > 0.005:
                    perturbation_scale *= 0.7
                else:
                    perturbation_scale = 0.03

    # Final: recompute score exactly from best_h
    best_h = project_feasible(best_h, n)
    final_score = exact_score(best_h, n)

    # Try a few other lengths if time permits
    if time.time() < deadline - 20:
        for n_try in [n - 1, n + 1, n - 2, n + 2]:
            if n_try < 10 or n_try > 80:
                continue
            if time.time() > deadline - 10:
                break
            h_try = resample_h(best_h, n, n_try)
            s_try = exact_score(h_try, n_try)
            if s_try < best_score:
                best_score = s_try
                best_h = h_try
                n = n_try
                # Quick local optimization
                c = compute_correlation(best_h, n)
                current_score = s_try
                for _ in range(30):
                    if time.time() > deadline - 10:
                        break
                    improved = False
                    for i in range(n - 1):
                        p, q = i, i + 1
                        d = compute_d(best_h, 1.0 - best_h, p, q, n)
                        x, y = best_h[p], best_h[q]
                        lo = max(-x, y - 1.0)
                        hi = min(1.0 - x, y)
                        if hi - lo < 1e-14:
                            continue
                        best_t, best_v = pair_minimize(c, d, lo, hi)
                        if best_v < current_score - 1e-11:
                            best_h[p] = x + best_t
                            best_h[q] = y - best_t
                            c = c + best_t * d
                            current_score = best_v
                            improved = True
                    if not improved:
                        break
                if current_score < best_score:
                    best_score = current_score
                    best_h = best_h.copy()

    best_h = project_feasible(best_h, n)
    final_score = exact_score(best_h, n)

    return (best_h, final_score, n)
