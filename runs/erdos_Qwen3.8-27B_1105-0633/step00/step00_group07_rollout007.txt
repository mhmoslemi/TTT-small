def _project(y, target):

    hi = 1.0 - y.min() + 0.5

    if abs(diff) > 1e-13:

def _command_local_search(h, best_c, deadline, max_rounds=10):
    """Pairwise local search: try all pairs, move mass to reduce objective"""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    current_c = best_c
    
    for round_idx in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        # Try pairs in a deterministic order
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                # Transfer from j to i
                alpha_max = min(1.0 - h[i], h[j])
                if alpha_max < 1e-12:
                    continue
                # Coarse search
                best_a = 0.0
                best_v = current_c
                n_coarse = 10
                for a in np.linspace(0, alpha_max, n_coarse)[1:]:
                    h[i] += a
                    h[j] -= a
                    v = _evaluate(h)

                        best_a = a
                    h[i] -= a
                    h[j] += a

                        h[j] += b

                        h[j] -= b

def _command_fw(h, best_c, deadline, iters=100):
    """Frank-Wolfe method"""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    current_c = best_c
    
    for it in range(iters):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = corr.max()
        
        # Use high beta for gradient (approximate max gradient)
        beta = 50.0
        scaled = beta * (corr - m)
        max_s = scaled.max()
        exps = np.exp(scaled - max_s)
        weights = exps / exps.sum()
        
        grad = np.zeros(n)
        for k in range(len(corr)):
            w = weights[k]
            if w < 1e-12:
                continue
            d = k - (n - 1)
            if d >= 0:
                grad[:n - d] += w * (1.0 - h[d:])
                grad[d:] -= w * h[:n - d]
            else:
                dd = -d
                grad[dd:] += w * (1.0 - h[:n - dd])
                grad[:n - dd] -= w * h[dd:]
        
        # FW vertex
        order = np.argsort(grad)
        v = np.zeros(n)
        remaining = target
        for i in order:
            if remaining <= 1e-12:
                break
            v[i] = min(1.0, remaining)
            remaining -= v[i]
        
        # Line search: h_new = (1-t)*h + t*v
        best_t = 0.0
        best_c_fw = current_c
        for t in np.linspace(0, 1, 20):
            h_trial = (1 - t) * h + t * v
            h_trial = _project(h_trial, target)
            c = _evaluate(h_trial)
            if c < best_c_fw:
                best_c_fw = c
                best_t = t
        
        if best_t > 0 and best_c_fw < current_c - 1e-12:
            h = (1 - best_t) * h + best_t * v
            h = _project(h, target)
            current_c = best_c_fw
        else:
            break
    
    return h, current_c

def _command_upsample(h, n_new):
    """Upsample h to n_new points preserving area"""
    n_old = len(h)
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    target_new = n_new / 2.0
    
    # For each new sample position, compute the area of the old function over that cell
    h_new_raw = np.zeros(n_new)
    for j in range(n_new):
        x_lo = j * dx_new
        x_hi = (j + 1) * dx_new
        area = 0.0
        for i in range(n_old):
            old_lo = i * dx_old
            old_hi = (i + 1) * dx_old
            overlap = min(x_hi, old_hi) - max(x_lo, old_lo)
            if overlap > 0:
                area += h[i] * overlap
        h_new_raw[j] = area / dx_new
    
    return _project(h_new_raw, target_new)

def _command_downsample(h, n_new):

def _command_random_restart(h, seed, n_trials=5):
