import time

def exact_score(h):
    """Compute C5 = max of full correlation of h with 1-h, scaled by dx."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def project_box_sum(x, n):
    """Project x onto {0 <= x_i <= 1, sum(x) = n/2} using water-filling."""
    target = n / 2.0
    # Bisection on the threshold
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix rounding: adjust last element slightly
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        # Find an element that can absorb the difference
        if diff > 0:
            idx = np.where(y < 1.0)[0]
        else:
            idx = np.where(y > 0.0)[0]
        if len(idx) > 0:
            y[idx[-1]] += diff
            y[idx[-1]] = np.clip(y[idx[-1]], 0.0, 1.0)
    return y

def smooth_obj_grad(h, tau):
    """Compute smooth log-sum-exp objective and its gradient."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlation values
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Log-sum-exp
    scaled = tau * corr
    m = np.max(scaled)
    exp_shifted = np.exp(scaled - m)
    W = np.sum(exp_shifted)
    F = (m + np.log(W)) / tau
    w = exp_shifted / W  # normalized weights
    
    # Compute gradient
    grad = np.zeros(n)
    # corr[j] corresponds to shift s = j - (n-1)
    for j, s in enumerate(range(-(n-1), n)):
        if w[j] < 1e-15:
            continue
        # For this shift s, valid range of n: max(0,-s) <= n <= min(n-1, n-1-s)
        n_lo = max(0, -s)
        n_hi = min(n-1, n-1-s)
        if n_lo > n_hi:
            continue
        # Part 1: h[i] appears as h[n+s], so i = n+s, contribution (1-h[n])
        # i ranges from n_lo+s to n_hi+s
        i_lo = n_lo + s
        i_hi = n_hi + s
        grad[i_lo:i_hi+1] += w[j] * dx * (1.0 - h[n_lo:n_hi+1])
        # Part 2: h[i] appears in (1-h[n]), so i = n, contribution -h[n+s]
        # i ranges from n_lo to n_hi
        grad[n_lo:n_hi+1] += w[j] * dx * (-h[n_lo+s:n_hi+s+1])
    
    return F, grad

def local_search(h, n, rng, max_iters=5000, step_scale=0.01):
    """Direct pair-move local search to improve the exact objective."""
    dx = 2.0 / n
    h = h.copy()
    current_score = exact_score(h)
    
    for it in range(max_iters):
        # Try a random pair move
        for attempt in range(20):
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            s = h[i] + h[j]
            # Feasible range for h[i] new value
            lo = max(0.0, s - 1.0)
            hi = min(1.0, s)
            step = rng.uniform(-step_scale, step_scale)
            h_new_i = h[i] + step
            h_new_i = np.clip(h_new_i, lo, hi)
            h_new_j = s - h_new_i
            if h_new_j < -1e-10 or h_new_j > 1.0 + 1e-10:
                continue
            h_work = h.copy()
            h_work[i] = h_new_i
            h_work[j] = h_new_j
            new_score = exact_score(h_work)
            if new_score < current_score - 1e-10:
                h = h_work
                current_score = new_score
                break
    return h, current_score

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_box_sum(x, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        y = np.clip(x - mid, 0.0, 1.0)
        if y.sum() > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - y.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.where(y < 1.0)[0]
        else:
            idx = np.where(y > 0.0)[0]
        if len(idx) > 0:
            y[idx[0]] += diff
            y[idx[0]] = min(max(y[idx[0]], 0.0), 1.0)
    return y

def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    scaled = tau * corr
    m = np.max(scaled)
    exp_shifted = np.exp(scaled - m)
    W = np.sum(exp_shifted)
    F = (m + np.log(W)) / tau
    w = exp_shifted / W
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        if w[j] < 1e-14:
            continue
        s = j - (n - 1)
        n_lo = max(0, -s)
        n_hi = min(n - 1, n - 1 - s)
        if n_lo > n_hi:
            continue
        i_lo = n_lo + s
        i_hi = n_hi + s
        grad[i_lo:i_hi + 1] += w[j] * dx * (1.0 - h[n_lo:n_hi + 1])
        grad[n_lo:n_hi + 1] += w[j] * dx * (-h[n_lo + s:n_hi + s + 1])
    return F, grad

        h_work = h.__array__()
        h_work[i] = h_new_i

            current_score = new_score

        else:

            if no_improve % 400 == 0 and scale_idx < len(scales) - 1:
