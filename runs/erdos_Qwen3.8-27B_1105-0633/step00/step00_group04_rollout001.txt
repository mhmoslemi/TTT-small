    """Project onto {x: 0<=x_i<=1, sum(x)=target_sum}"""
    n = len(v)
    lo = float(v.max()) + 1.0
    hi = float(v.min()) - 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
