def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    terms = np.correlate(h, 1.0 - h, mode="full") * dx

def smooth_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    h = x
    terms = np.correlate(h, 1.0 - h, mode="full") * dx
    m0 = np.max(terms)
    shifted = beta * (terms - m0)
    # Handle case where all shifted are very negative
    max_shifted = np.max(shifted)
    log_sum = max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))
    return m0 + log_sum / beta

def smooth_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    h = x
    terms = np.correlate(h, 1.0 - h, mode="full") * dx
    m0 = np.max(terms)
    shifted = beta * (terms - m0)

    # Phase 2: Try other n values
    for n_try in n_list:
        if n_try == n_parent:
            continue
        if time_left() < 30:
            break
        if best_c <= target:
            break
        
        # Interpolate parent to new size
        x_old = np.linspace(0, 2, n_parent, endpoint=False) + 1.0 / n_parent
        x_new = np.linspace(0, 2, n_try, endpoint=False) + 1.0 / n_try
        # Wrap-around interpolation
        x_ext = np.concatenate([x_old - 2.0, x_old, x_old + 2.0])
        h_ext = np.concatenate([parent, parent, parent])
        h_interp = np.interp(x_new, x_ext, h_ext)
        h_interp = repair_h(h_interp, n_try)
        
        starts2 = [h_interp]
        starts2.extend(binary_starts(n_try, rng, count=2))
        
        for h0 in starts2:
            for beta in [50.0, 100.0]:
                if time_left() < 10:
                    break
                res = run_slsqp(h0, beta, n_try, maxiter=200, ftol=1e-10)
                if res is not None and res.x is not None:
                    h_new = repair_h(res.x, n_try)
                    c_new = exact_c5(h_new)
                    if c_new < best_c - 1e-10:
                        best_c = c_new
                        best_h = h_new.copy()
    
    # Phase 3: Local polish on best
    if time_left() > 15:
        n_best = len(best_h)
        h_pol, c_pol = local_polish(best_h, n_best, time_limit=min(40, time_left() - 10))
        if c_pol < best_c - 1e-10:
            best_c = c_pol
            best_h = h_pol.copy()
    
    # Phase 4: Another round of SLSQP on the polished best with high beta
    if time_left() > 15:
        n_best = len(best_h)
        for beta in [150.0, 300.0, 500.0]:
            if time_left() < 8:
                break
            res = run_slsqp(best_h, beta, n_best, maxiter=200, ftol=1e-11)
            if res is not None and res.x is not None:
                h_new = repair_h(res.x, n_best)
                c_new = exact_c5(h_new)
                if c_new < best_c - 1e-10:
                    best_c = c_new
                    best_h = h_new.copy()
    
    # Phase 5: Final local polish
    if time_left() > 10:
        n_best = len(best_h)
        h_pol, c_pol = local_polish(best_h, n_best, time_limit=min(30, time_left() - 5))
        if c_pol < best_c - 1e-10:
            best_c = c_pol
            best_h = h_pol.copy()
    
    # Final validation
    n_final = len(best_h)
    best_h = repair_h(best_h, n_final)
    final_c = exact_c5(best_h)
    if final_c < best_c:
        best_c = final_c
    
    # Try the pre-imported evaluator
    try:
        ev = evaluate_erdos_solution(best_h)
        if ev is not None and np.isfinite(ev) and ev < best_c:
            best_c = float(ev)
    except (NameError, Exception):
        pass
    
    # Ensure we return a valid finite result
    if not np.all(np.isfinite(best_h)):
        best_h = np.ones(n_parent) * 0.5
        best_c = exact_c5(best_h)
    
    return (best_h, best_c, n_final)
