            best_t = t

        return None, best_val

                    C_base = np.correlate(h, 1.0 - h, mode="full") * dx
                    current_score = float(np.max(C_base))
                    improved = True
        if not improved:
            converged = True
            break
    h = repair_feasible(h, n)
    if h is not None:
        current_score = exact_score(h, n)
    return h, current_score, converged

def random_binary(n, rng):
    idx = rng.choice(n, size=n // 2, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def perturb_repair(h, n, rng, mag=0.15):
    h_new = h + rng.uniform(-mag, mag, n)
    return repair_feasible(h_new, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float)
        n_init = len(h_init)
        h_init = repair_feasible(h_init, n_init)
        if h_init is None:
            h_init = random_binary(n_init, rng)
    except (NameError, Exception):
        n_init = 42
        h_init = random_binary(n_init, rng)
    best_h = h_init.copy()
    best_n = len(best_h)
    best_score = exact_score(best_h, best_n)
    # Phase 1: intensive optimization of initial solution
    remaining = budget_s - (time.time() - t0)
    h_opt, s_opt, conv = optimize_pair_swaps(best_h, best_n, max_rounds=500, time_limit=remaining * 0.4, t0=t0, seed=seed)
    if h_opt is not None and s_opt < best_score:
        best_h = h_opt.copy()
        best_score = s_opt
    # If not converged, continue with more rounds
    if not conv and (time.time() - t0) < budget_s * 0.55:
        remaining = budget_s*0.55 - (time.time() - t0)
        h_opt, s_opt, conv = optimize_pair_swaps(best_h, best_n, max_rounds=500, time_limit=remaining, t0=time.time(), seed=seed+1)
        if h_opt is not None and s_opt < best_score:
            best_h = h_opt.copy()
            best_score = s_opt
    # Phase 2: perturbation + re-optimize cycles
    time_left = budget_s - (time.time() - t0)
    n_perturb = max(1, int(time_left / 8.0))
    for r in range(n_perturb):
        if time.time() - t0 > budget_s * 0.82:
            break
        h_p = perturb_repair(best_h, best_n, rng, mag=0.1 + 0.1 * rng.random())
        if h_p is None:
            continue
        h_p, s_p, _ = optimize_pair_swaps(h_p, best_n, max_rounds=40, time_limit=budget_s * 0.03, t0=time.time(), seed=seed + 100 + r)
        if h_p is not None and s_p < best_score:
            best_h = h_p.copy()
            best_score = s_p
    # Phase 3: random restarts
    time_left = budget_s - (time.time() - t0)
    n_restarts = max(1, int(time_left / 6.0))
    for r in range(n_restarts):
        if time.time() - t0 > budget_s * 0.90:
            break
        h_r = random_binary(best_n, rng)
        h_r, s_r, _ = optimize_pair_swaps(h_r, best_n, max_rounds=60, time_limit=budget_s * 0.04, t0=time.time(), seed=seed + 500 + r)
        if h_r is not None and s_r < best_score:
            best_h = h_r.copy()
            best_score = s_r
    # Phase 4: try different resolutions
    time_left = budget_s - (time.time() - t0)
    for n_try in [best_n * 2, best_n + 14, max(28, best_n - 14), best_n + 28]:
        if time.time() - t0 > budget_s * 0.95:
            break
        if n_try < 14 or n_try > 168 or n_try % 2 != 0:
            continue
        h_try = random_binary(n_try, rng)
        h_try, s_try, _ = optimize_pair_swaps(h_try, n_try, max_rounds=80, time_limit=budget_s * 0.04, t0=time.time(), seed=seed + 900 + n_try)
        if h_try is not None and s_try < best_score:
            best_h = h_try.copy()
            best_score = s_try
            best_n = n_try
    # Final: one more optimization pass if time remains
    if (time.time() - t0) < budget_s * 0.97:
        h_f, s_f, _ = optimize_pair_swaps(best_h, best_n, max_rounds=100, time_limit=budget_s*0.02, t0=time.time(), seed=seed+2000)
        if h_f is not None and s_f < best_score:
            best_h = h_f.copy()
            best_score = s_f
    # Final repair and exact score
    best_h = repair_feasible(best_h, best_n)
    if best_h is None:
        best_h = random_binary(best_n, rng)
    best_score = exact_score(best_h, best_n)
    return (best_h, best_score, best_n)
