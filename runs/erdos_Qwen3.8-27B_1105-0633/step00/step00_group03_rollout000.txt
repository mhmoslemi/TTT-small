def waterfill(x, target_sum, n):
    """Project x onto {0 <= y <= 1, sum(y) = target_sum}."""
    x = np.asarray(x, dtype=float).copy()
    # Binary search on the threshold
    lo = x.min() - target_sum  # will make all positive
    hi = x.max() + target_sum  # will make all negative
    # Actually: y(t) = clip(x - t, 0, 1), find t such that sum(y(t)) = target_sum
    # sum(y(t)) is decreasing in t
    # At t = -inf: all y = 1, sum = n
    # At t = +inf: all y = 0, sum = 0
    lo = -10.0  # all will be clipped to 1
    hi = 10.0   # all will be clipped to 0

        mid = (lo + hi) / 2.0
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any numerical issues
    diff = target_sum - result.sum()
    if abs(diff) > 1e-12:
        # Distribute the diff among elements not at bounds
        mask = (result > 1e-10) & (result < 1.0 - 1e-10)
        if mask.sum() > 0:
            result[mask] += diff / mask.sum()
            result = np.clip(result, 0.0, 1.0)
    return result

            sc = exact_evaluate(h_init)
            if sc < best_score:
                best_score = sc
                best_h = h_init.copy()
            
            # Run smooth gradient optimization with tau annealing
            tau_seq = [10.0, 30.0, 100.0, 300.0]
            h_opt, sc_opt = smooth_optimize(h_init, target_sum, tau_seq, 50, 0.5)
            sc_final = exact_evaluate(h_opt)
            if sc_final < best_score:
                best_score = sc_final
                best_h = h_opt.copy()
            
            # Local pair search on the best
            h_ls, sc_ls = pair_local_search(h_opt, target_sum, 500, rng)
            sc_ls = exact_evaluate(h_ls)
            if sc_ls < best_score:
                best_score = sc_ls
                best_h = h_ls.copy()
            
            # Try SLSQP with high tau for a few starts
            if si < 3 and remaining() > 10:
                try:
                    h_sl, sc_sl = slsqp_optimize(h_opt, target_sum, tau=200, max_iter=100)
                    sc_sl = exact_evaluate(h_sl)
                    if sc_sl < best_score:
                        best_score = sc_sl
                        best_h = h_sl.copy()
                        # Further local search
                        h_ls2, sc_ls2 = pair_local_search(h_sl, target_sum, 300, rng)
                        sc_ls2 = exact_evaluate(h_ls2)
                        if sc_ls2 < best_score:
                            best_score = sc_ls2
                            best_h = h_ls2.copy()
                except Exception:
                    pass
        
        # Final polishing on best at this n if time permits
        if best_h is not None and len(best_h) == n and remaining() > 5:
            h_pol, sc_pol = pair_local_search(best_h, target_sum, 1000, rng)
            sc_pol = exact_evaluate(h_pol)
            if sc_pol < best_score:
                best_score = sc_pol
                best_h = h_pol.copy()
    
    # Final safety: ensure best_h is valid
    if best_h is None:
        # Fallback: constant 0.5
        n = 45
        best_h = np.full(n, 0.5)
        best_score = exact_evaluate(best_h)
    
    # Final validation
    best_h = waterfill(best_h, len(best_h) / 2.0)
    final_score = exact_evaluate(best_h)
    if final_score != best_score:
        best_score = final_score
    
    # One more polishing pass if time
    if remaining() > 5:
        h_fin, sc_fin = pair_local_search(best_h, len(best_h) / 2.0, min(500, int(remaining() * 100)), rng)
        sc_fin = exact_evaluate(h_fin)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin
    
    best_h = waterfill(best_h, len(best_h) / 2.0)
    best_score = exact_evaluate(best_h)
    
    return (best_h, float(best_score), int(len(best_h)))
