One option: use a class to hold the state, and define the objective/constraint functions as methods.

Another option: use `functools.partial` - but that might count as a closure.

class _SLSQPSolver:
    def __init__(self, h_init):
        self.n = len(h_init)
        self.target = self.n / 2.0
        self.dx = 2.0 / self.n
        self.m = 2 * self.n - 1
        self.h0 = h_init.copy()
        self.c0 = _eval_c5(h0)
        d_vals = np.arange(self.m) - (self.n - 1)
        self.i_mins = np.maximum(0, -d_vals).astype(int)
        self.i_maxs = np.minimum(self.n - 1, self.n - 1 - d_vals).astype(int)
        self.d_vals = d_vals
    
    def obj(self, x):
        return x[-1]
    
    def obj_jac(self, x):
        g = np.zeros(self.n + 1)
        g[-1] = 1.0
        return g
    
    def eq(self, x):
        return np.sum(x[:self.n]) - self.target
    
    def eq_jac(self, x):
        g = np.zeros(self.n + 1)
        g[:self.n] = 1.0
        return g
    
    def ineq(self, x):
        hx = x[:self.n]
        tx = x[-1]
        c = np.correlate(hx, 1.0 - hx, mode='full') * self.dx
        return tx - c
    
    def ineq_jac(self, x):
        hx = x[:self.n]
        jac = np.zeros((self.m, self.n + 1))
        jac[:, -1] = 1.0

            im = self.i_mins[j]

            jac[j, i_arr] += (-hx[i_arr + d])
