            if n_try < 40 or remaining() < 35:
                break
            # Interpolate best to new dimension
            x_old = np.linspace(0.0, 2.0, n_init, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
            h_try = np.interp(x_new, x_old, best_h)
            h_try = project_feasible(h_try)
            # Optimize
            h_try, c5_try = slsqp_optimize(h_try, [100, 200, 400], max_iter=50)
            h_try, c5_try = active_set_polish(h_try, max_iters=40)
            # Accept if not worse (prefer smaller n on tie)
            if c5_try <= best_c5 + 1e-7:
                best_c5 = c5_try
                best_h = h_try

    # Final verification
    best_h = project_feasible(best_h)
    best_c5 = evaluate_c5(best_h)

    # Ensure exact feasibility
    n = len(best_h)
    diff = n / 2.0 - np.sum(best_h)
    if abs(diff) > 1e-12:
        free = (best_h > 1e-8) & (best_h < 1.0 - 1e-8)
        if np.any(free):
            best_h[free] += diff / np.sum(free)
            best_h = np.clip(best_h, 0.0, 1.0)

    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
