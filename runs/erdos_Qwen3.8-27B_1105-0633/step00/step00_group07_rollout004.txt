def project_onto_feasible(h, n):
    """Project onto {x: 0<=xi<=1, sum(xi)=n/2} via 1D bisection."""
    target = 0.5 * n

    hi = 1.0 - h.min() + 0.5

        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid

def lse_gradient(h, beta):
    """Gradient of (1/beta)*log(sum exp(beta*R_d)) where R_d are lag values."""
    n = len(h)
