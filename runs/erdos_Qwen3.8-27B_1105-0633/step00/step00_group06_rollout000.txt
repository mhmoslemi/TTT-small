∂f_m/∂h[i]:

So ∂f_m/∂h[i] = (1-h[i-m]) * [0 ≤ i-m < n] - h[i+m] * [0 ≤ i+m < n]

import numpy as np
from scipy.optimize import minimize
import time


def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_box_sum(h, target_sum):
    n = len(h)
    if target_sum <= 0:
        return np.zeros(n)
    if target_sum >= n:
        return np.ones(n)
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)


def smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    spread = alpha * (m - np.min(corr))
    if spread > 400:
        return m
    shifted = corr - m
    return m + np.log(np.mean(np.exp(alpha * shifted))) / alpha


def subgradient_max(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_val = np.max(corr)
    active = np.where(corr >= max_val - 1e-9)[0]
    grad = np.zeros(n)
    for m_idx in active:
        m = m_idx - (n - 1)
        for i in range(n):
            if 0 <= i - m < n:
                grad[i] += (1.0 - h[i - m])
            if 0 <= i + m < n:
                grad[i] -= h[i + m]
    grad /= len(active)
    return grad


def try_slsqp(h_best, n, target_sum, alpha, time_limit, start_time, budget_s):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]

    def obj(x):
        return smooth_obj(x, alpha)

    try:
        res = minimize(obj, h_best, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': 200, 'ftol': 1e-11})
    except Exception:
        return h_best, evaluate_c5(h_best)

    if res.x is not None and np.all(np.isfinite(res.x)):
        h_new = project_box_sum(res.x, target_sum)
        c5_new = evaluate_c5(h_new)
        if c5_new < evaluate_c5(h_best):
            return h_new, c5_new
    return h_best, evaluate_c5(h_best)


def subgradient_phase(h_best, n, target_sum, start_time, budget_s, rng):
    best_h = h_best.copy()
    best_c5 = evaluate_c5(best_h)
    lr = 0.02
    stall = 0

    while budget_s - (time.time() - start_time) > 15:
        grad = subgradient_max(best_h, n)
        h_new = project_box_sum(best_h - lr * grad, target_sum)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_h = h_new
            best_c5 = c5_new
            stall = 0
            lr = min(lr * 1.05, 0.1)
        else:
            stall += 1
            if stall % 50 == 0:
                lr *= 0.7
            if stall > 500:
                # Restart with perturbation
                pert = best_h + rng.randn(n) * 0.08
                pert = project_box_sum(pert, target_sum)
                c5_pert = evaluate_c5(pert)
                if c5_pert < best_c5 + 0.005:
                    best_h = pert
                    best_c5 = c5_pert
                lr = 0.02
                stall = 0
        if lr < 1e-9:
            break

    return best_h, best_c5


def coordinate_descent(h_best, n, target_sum, start_time, budget_s, rng):
    best_h = h_best.copy()
    best_c5 = evaluate_c5(best_h)
    no_improve = 0

    while budget_s - (time.time() - start_time) > 5:
        improved = False
        for _ in range(n * 4):
            if budget_s - (time.time() - start_time) < 5:
                break
            i = rng.randint(n)
            j = (i + rng.randint(1, n)) % n
            for delta in [0.04, 0.02, 0.01, 0.005, -0.04, -0.02, -0.01, -0.005]:
                h_try = best_h.copy()
                h_try[i] += delta
                h_try[j] -= delta
                if 0.0 <= h_try[i] <= 1.0 and 0.0 <= h_try[j] <= 1.0:
                    if abs(np.sum(h_try) - target_sum) > 1e-9:
                        continue
                    c5_try = evaluate_c5(h_try)
                    if c5_try < best_c5 - 1e-12:
                        best_h = h_try
                        best_c5 = c5_try
                        improved = True
                        break
        if not improved:
            no_improve += 1
            if no_improve > 3:
                break
        else:
            no_improve = 0

    return best_h, best_c5


def random_restart_phase(h_best, n, target_sum, start_time, budget_s, rng, best_c5):
    local_best_h = h_best.copy()
    local_best_c5 = best_c5
    attempts = 0

    while budget_s - (time.time() - start_time) > 30 and attempts < 20:
        attempts += 1
        scale = rng.uniform(0.03, 0.15)
        h_pert = local_best_h + rng.randn(n) * scale
        h_pert = project_box_sum(h_pert, target_sum)
        c5_pert = evaluate_c5(h_pert)
        if c5_pert < local_best_c5:
            local_best_h = h_pert
            local_best_c5 = c5_pert

        # Quick SLSQP polish on the perturbation
        if budget_s - (time.time() - start_time) > 40:
            try:
                bounds = [(0.0, 1.0)] * n
                cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                         'jac': lambda x: np.ones(n)}]
                res = minimize(lambda x: smooth_obj(x, 200), h_pert,
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 100, 'ftol': 1e-10})
                if res.x is not None:
                    h_pol = project_box_sum(res.x, target_sum)
                    c5_pol = evaluate_c5(h_pol)
                    if c5_pol < local_best_c5:
                        local_best_h = h_pol
                        local_best_c5 = c5_pol
            except Exception:
                pass

    return local_best_h, local_best_c5


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    # Get initial values
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.full(70, 0.5)

    n = len(h_init)
    target_sum = n / 2.0
    best_h = project_box_sum(h_init.copy(), target_sum)
    best_c5 = evaluate_c5(best_h)

    def remaining():
        return budget_s - 3.0 - (time.time() - start_time)

    # Phase 1: SLSQP with increasing alpha
    alphas = [20, 50, 100, 200, 500, 1000]
    for alpha in alphas:
        if remaining() < 40:
            break
        h_new, c5_new = try_slsqp(best_h, n, target_sum, alpha,
                                   30, start_time, budget_s)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 2: Subgradient descent with exact max
    if remaining() > 40:
        best_h, best_c5 = subgradient_phase(best_h, n, target_sum,
                                            start_time, budget_s, rng)

    # Phase 3: Random restarts with SLSQP polish
    if remaining() > 60:
        h_rr, c5_rr = random_restart_phase(best_h, n, target_sum,
                                           start_time, budget_s, rng, best_c5)
        if c5_rr < best_c5:
            best_h = h_rr
            best_c5 = c5_rr

    # Phase 4: Another SLSQP pass with high alpha on best found
    if remaining() > 30:
        h_new, c5_new = try_slsqp(best_h, n, target_sum, 2000,
                                   20, start_time, budget_s)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 5: Coordinate descent polishing
    if remaining() > 10:
        best_h, best_c5 = coordinate_descent(best_h, n, target_sum,
                                             start_time, budget_s, rng)

    # Final verification
    final_c5 = evaluate_c5(best_h)
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, target_sum)
    final_c5 = evaluate_c5(best_h)

    return (best_h, final_c5, n)
