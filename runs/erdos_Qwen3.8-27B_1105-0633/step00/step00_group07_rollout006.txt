    """Project x onto {0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0
    lo = x.min() - 2.0
    hi = x.max() + 2.0

        s = np.clip(x - mid, 0.0, 1.0).sum()

    """Gradient of correlation at lag k (actual lag, signed) w.r.t. h."""
    dx = 2.0 / n
    g = np.zeros(n)

    else:
        s = -k

            g[:n - s] += (1.0 - h[s:])
            g[s:] -= h[:n - s]
    return g * dx

        h_new, c5_new = _lp_descent(best_h, best_n, t, rng)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            best_n = best_n
    
    # Phase 7: Final gradient polish
    if _time_left() > 10:
        t = min(20, _time_left() * 0.5)
        h_new, c5_new = _gradient_descent(best_h, best_n, t, rng)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            best_n = best_n
    
    # Phase 8: Single lag refinement
    if _time_left() > 5:
        t = min(10, _time_left() * 0.6)
        h_new, c5_new = _refine_single_lag(best_h, best_n, t)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            best_n = best_n
    
    # Final validation
    best_h = _project_feasible(best_h, best_n)
    final_c5 = _eval_c5(best_h)
    s = best_h.sum()
    target = best_n / 2.0
    if abs(s - target) > 1e-8:
        # Final repair
        diff = target - s
        # Distribute among elements with room
        if diff > 0:
            room = 1.0 - best_h
            total_room = room.sum()
            if total_room > 0:
                best_h += diff * room / total_room
        else:
            total_mass = best_h.sum()
            if total_mass > 0:
                best_h += diff * best_h / total_mass
        best_h = _project_feasible(best_h, best_n)
        final_c5 = _eval_c5(best_h)
    
    return (best_h, final_c5, best_n)
