import numpy as np

def _exact_score(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))


def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-12
    else:
        mask = h > 1e-12
    if not np.any(mask):
        return h
    h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def _interval(h, d):
    L = 0.0
    U = 0.0
    nz = np.where(np.abs(d) > 1e-14)[0]
    if len(nz) == 0:
        return 0.0, 0.0
    for i in nz:
        if d[i] > 0:
            L = max(L, -h[i] / d[i])
            U = min(U, (1.0 - h[i]) / d[i])
        else:
            L = max(L, (1.0 - h[i]) / d[i])
            U = min(U, -h[i] / d[i])
    if U < L:
        return 0.0, 0.0
    return L, U


def _line_search(C, B, A, L, U):
    if U - L < 1e-14:
        return 0.0
    ng = 40
    ts = np.linspace(L, U, ng)
    ts2 = ts * ts
    vals = np.empty(ng)

        vals[j] = np.max(C + ts[j] * B + ts2[j] * A)
    bi = np.argmin(vals)
    if bi == 0:
        a, b = L, ts[1] if ng > 1 else U
    elif bi == ng - 1:
        a, b = ts[ng - 2] if ng > 1 else L, U
    else:
        a, b = ts[bi - 1], ts[bi + 1]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(60):
        if b - a < 1e-14:

        c = b - gr * (b - a)
        d = a + gr * (b - a)
        fc = np.max(C + c * B + c * c * A)
        fd = np.max(C + d * B + d * d * A)
        if fc < fd:
            b = d
        else:
            a = c
    return (a + b) * 0.5

def _cosine_feasible(n, rng):

    h = 0.5 + amp * np.cos(2 * np.pi * freq * x / 2.0 + phase)
    return _repair(h, n)
