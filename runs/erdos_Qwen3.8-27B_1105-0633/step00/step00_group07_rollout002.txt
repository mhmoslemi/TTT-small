def project_to_polytope(y, target_sum):
    n = len(y)
    if n == 0:
        return y
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y + mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    z = np.clip(y + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(z)
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.where(z < 0.9999999)[0]
        else:
            idx = np.where(z > 1e-9)[0]
        for i in idx:
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                step = min(diff, 1.0 - z[i])
                z[i] += step
                diff -= step
            else:
                step = min(-diff, z[i])
                z[i] -= step
                diff += step
    return z

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def lse_gradient(h, beta):
    n = len(h)
    corr = compute_corr(h)
    L = len(corr)
    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    shifted = np.clip(shifted, -400.0, 400.0)
    weights = np.exp(shifted)
    sum_w = np.sum(weights)
    if sum_w < 1e-300:
        weights = np.ones(L) / L
    else:
        weights /= sum_w
    grad = np.zeros(n)
    for k in range(L):
        if weights[k] < 1e-14:
            continue
        d = k - (n - 1)
        w = weights[k]
        if d >= 0:
            grad[d:] += w * (1.0 - h[:n - d])
            grad[:n - d] -= w * h[d:]
        else:
            dd = -d
            grad[:n - dd] += w * (1.0 - h[dd:])
            grad[dd:] -= w * h[:n - dd]
    return grad


def fw_vertex(h, grad):
    n = len(h)

        if remaining <= 1e-14:
            break
        if remaining >= 1.0:
            z[i] = 1.0
            remaining -= 1.0
        else:
            z[i] = remaining
            remaining = 0.0
    return z
