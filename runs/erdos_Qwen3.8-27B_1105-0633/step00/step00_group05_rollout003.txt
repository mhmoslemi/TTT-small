                h_new, sc = pair_line_search(h, best_n, a, b)
                if sc < curr - 1e-13:
                    h = h_new
                    curr = sc
            if curr < best_score - 1e-14:
                best_h = h
                best_score = curr

    # Final validation
    best_h = repair_feasible(np.asarray(best_h, dtype=float), best_n)
    final_score = exact_score(best_h, best_n)
    return best_h, final_score, best_n
