def slsqp_optimize(h0, max_iter=100, sigma=0.01):
    """SLSQP optimization with smooth max objective"""
    n = len(h0)
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        max_c = np.max(corr)
        if max_c < 1e-15:
            return 0.0
        return (max_c + sigma * np.log(np.sum(np.exp((corr - max_c) / sigma)))) * (2.0 / n)
    
    def obj_grad(x):
        # Numerical gradient
        eps = 1e-6
        grad = np.zeros(n)
        for j in range(n):
            xm = x.copy()
            xm[j] -= eps
            xp = x.copy()
            xp[j] += eps
            grad[j] = (obj(xp) - obj(xm)) / (2 * eps)
        return grad
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': 1e-10})
        h_res = project_feasible(res.x, n)
        c_res = exact_c5(h_res)
        h0_c = exact_c5(h0)
        if c_res < h0_c:
            return h_res, c_res
    except Exception:
        pass
    return h0, exact_c5(h0)
