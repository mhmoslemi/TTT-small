    """Log-sum-exp smooth maximum"""
    m = np.max(terms)
    if m > 500 / beta:

    return m + np.log(np.sum(np.exp(shifted))) / beta

    w /= np.sum(w)

    for f in freqs:
        amp = rng.uniform(0.1, 0.3)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.sin(np.pi * f * x + phase)
    return _repair(h)

# ---------- main ----------

def run(seed=42, budget_s=1100, **kwargs):
    t0 = _time.time()
    np.random.seed(seed)

    def remaining():
        return budget_s - (_time.time() - t0)

    # Get parent
    best_h = None
    best_c = 1.0
    try:
        if 'initial_h_values' in globals():
            p = np.asarray(initial_h_values, dtype=float)
            if p.ndim == 1 and len(p) >= 10:
                best_h = _repair(p)
                best_c = _c5(best_h)
    except Exception:
        pass

    if best_h is None:
        best_h = _repair(np.full(63, 0.5))
        best_c = _c5(best_h)

    n_parent = len(best_h)
    rng = np.random.default_rng(seed)

    # Phase 1: SLSQP smooth max on parent, increasing beta
    h_curr = best_h.copy()
    betas = [15.0, 40.0, 100.0, 300.0, 800.0]
    for beta in betas:
        if remaining() < 15:
            break
        h_new = _slsqp_smooth(h_curr, beta, max_iter=400)
        if h_new is not None:
            c_new = _c5(h_new)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new.copy()
            h_curr = h_new.copy()

    # Phase 2: Epigraph refinement on best
    if remaining() > 60:
        h_new = _epigraph_solve(best_h, max_iter=300)
        if h_new is not None:
            c_new = _c5(h_new)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new.copy()

    # Phase 3: Multiple starts (perturbed + structured)
    if remaining() > 150:
        max_starts = min(12, int((remaining() - 10) / 20))
        for s in range(max_starts):
            if remaining() < 25:
                break
            if s == 0:
                h_start = _repair(best_h + rng.normal(0, 0.08, n_parent))
            elif s == 1:
                h_start = _repair(best_h + rng.normal(0, 0.15, n_parent))
            elif s == 2:
                h_start = _make_binary_start(n_parent, rng)
            elif s == 3:
                h_start = _make_smooth_start(n_parent, rng)
            elif s == 4:
                # Alternate pattern
                h_alt = np.where(np.arange(n_parent) % 2 == 0, 1.0, 0.0)
                h_start = _repair(h_alt + rng.normal(0, 0.1, n_parent))
            else:
                h_start = _repair(best_h + rng.normal(0, 0.1, n_parent))

            for beta in [80.0, 300.0]:
                if remaining() < 20:
                    break
                h_new = _slsqp_smooth(h_start, beta, max_iter=300)
                if h_new is not None:
                    c_new = _c5(h_new)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        best_h = h_new.copy()
                        h_start = h_new.copy()

    # Phase 4: Try different n values
    if remaining() > 200:
        for n_try in [n_parent - 3, n_parent + 1, n_parent + 3, 60, 66, 70]:
            if remaining() < 50:
                break
            if n_try <= 0 or n_try == n_parent:
                continue
            h_interp = _interpolate_to(best_h, n_try)
            h_curr_n = h_interp.copy()
            for beta in [80.0, 300.0]:
                if remaining() < 30:
                    break
                h_new = _slsqp_smooth(h_curr_n, beta, max_iter=300)
                if h_new is not None:
                    c_new = _c5(h_new)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        best_h = h_new.copy()
                    h_curr_n = h_new.copy()

    # Phase 5: Local search (coordinate descent)
    if remaining() > 20:
        t_rem = min(remaining() - 5, 60)
        h_ls, c_ls = _local_search(best_h.copy(), max_iter=300, time_limit=t_rem)
        if c_ls < best_c - 1e-14:
            best_c = c_ls
            best_h = h_ls.copy()

    # Phase 6: Final gradient polish
    if remaining() > 10 and best_h is not None:
        h_cur = best_h.copy()
        c_cur = best_c
        for beta in [500.0, 2000.0]:
            for _ in range(50):
                if remaining() < 5:
                    break
                h_next = _gradient_descent_step(h_cur, beta, lr=0.005)
                c_next = _c5(h_next)
                if c_next < c_cur - 1e-14:
                    h_cur = h_next
                    c_cur = c_next
                else:
                    break
        if c_cur < best_c - 1e-14:
            best_c = c_cur
            best_h = h_cur.copy()

    # Final repair and evaluation
    best_h = _repair(best_h)
    best_c = _c5(best_h)

    # Safety: ensure finite
    if not np.all(np.isfinite(best_h)):
        best_h = np.full(n_parent, 0.5)
        best_h = _repair(best_h)
        best_c = _c5(best_h)

    return (best_h, best_c, len(best_h))
