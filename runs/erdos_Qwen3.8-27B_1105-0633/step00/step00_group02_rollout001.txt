            x_old = np.arange(n_best, dtype=np.float64) * (2.0 / n_best) + 1.0 / n_best
            x_new = np.arange(n_try, dtype=np.float64) * (2.0 / n_try) + 1.0 / n_try
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp)
            c5_interp = exact_c5(h_interp)
            if c5_interp < best_c5 + 0.02:
                # Quick SA on new dimension
                if remaining > 20:
                    sb = min(remaining * 0.2, 40)
                    u_t, c5_t = run_sa(h_interp, 14, c5_interp * 0.1,
                                       c5_interp * 0.002, 0.999,
                                       200000, rng, sb, t0)
                    h_t = u_t.astype(np.float64) / (1 << 14)
                    h_t = project_feasible(h_t)
                    c5_t = exact_c5(h_t)
                    if c5_t < best_c5:
                        best_c5 = c5_t
                        best_h = h_t.copy()
                        n_best = n_try
                elif c5_interp < best_c5:
                    best_c5 = c5_interp
                    best_h = h_interp.copy()
                    n_best = n_try

    # Final: one more polish
    elapsed = time.time() - t0
    remaining = budget_s - elapsed
    if remaining > 10:
        h_f, c5_f = continuous_optimize(best_h, [0.001, 0.0005, 0.0002], 100, t0, t0 + min(remaining * 0.7, 60))
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            n_best = len(h_f)

    # Ensure feasibility
    best_h = project_feasible(best_h)
    best_c5 = exact_c5(best_h)

    # Fallback
    if not validate_h(best_h) or not np.isfinite(best_c5):
        best_h = np.full(n_par, 0.5)
        best_c5 = exact_c5(best_h)
        n_best = len(best_h)

    return (best_h, best_c5, int(n_best))
