import numpy as np
from scipy.optimize import linprog
import time

def evaluate(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_terms(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_lag(h, k):
    """Gradient of correlation term at output index k (0 to 2n-2)."""
    n = len(h)
    dx = 2.0 / n
    d = k - (n - 1)
    grad = np.zeros(n)
    if d >= 0:
        # valid i: 0 to n-1-d
        indices = np.arange(n - d)
        # First factor contribution: h[i] -> (1 - h[i+d])
        for i in indices:
            grad[i] += dx * (1.0 - h[i + d])
            grad[i + d] -= dx * h[i]
    else:
        p = -d
        indices = np.arange(p, n)
        for i in indices:
            grad[i] += dx * (1.0 - h[i - p])
            grad[i - p] -= dx * h[i]
    return grad

def grad_lag_vectorized(h, k):
    """Vectorized gradient of correlation term at output index k."""
    n = len(h)
    dx = 2.0 / n
    d = k - (n - 1)
    grad = np.zeros(n)
    if d >= 0:
        m = n - d  # number of valid pairs
        # h[i] for i in 0..m-1, h[i+d] for i in 0..m-1
        grad[:m] += dx * (1.0 - h[d:d+m])
        grad[d:d+m] -= dx * h[:m]
    else:
        p = -d
        m = n - p  # number of valid pairs
        grad[p:p+m] += dx * (1.0 - h[:m])
        grad[:m] -= dx * h[p:p+m]
    return grad

    """One gradient descent step."""
    n = len(h)
    C = np.max(terms)
    
    # Find active set
    active = np.where(terms >= C - active_threshold)[0]
    if len(active) == 0:

    # Use exponential weighting based on how close to max
    weights = np.exp((terms[active] - C) / 0.01)

import numpy as np
import time

def evaluate(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_terms(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_for_lag(h, d):
    """Gradient of the correlation term at lag d (negative or positive).
    d is the actual lag (offset), not the output index.
    Output index k = d + (n-1).
    """
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if d >= 0:
        m = n - d
        if m <= 0:
            return grad
        # h[i] * (1-h[i+d]) for i = 0..m-1
        grad[:m] += dx * (1.0 - h[d:d+m])
        grad[d:d+m] -= dx * h[:m]
    else:
        p = -d
        m = n - p
        if m <= 0:
            return grad
        # h[i] * (1-h[i-p]) for i = p..n-1
        grad[p:p+m] += dx * (1.0 - h[:m])
        grad[:m] -= dx * h[p:p+m]
    return grad

def soft_max_gradient(h, tau=20.0):
    n = len(h)
    terms = compute_terms(h)
    max_t = np.max(terms)
    exp_t = np.exp(tau * (terms - max_t))
    weights = exp_t / np.sum(exp_t)
    
    grad = np.zeros(n)
    for k in range(2*n-1):
        if weights[k] > 1e-8:
            d = k - (n-1)
            grad += weights[k] * grad_for_lag(h, d)
    
    return grad, terms
