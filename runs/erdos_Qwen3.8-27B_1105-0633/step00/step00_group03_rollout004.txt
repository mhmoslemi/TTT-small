def project_feasible(h, n):
    """Project h to feasible set: 0<=h[i]<=1, sum(h)=n/2"""
    h = np.clip(h, 0.0, 1.0).copy()

    # Distribute diff to interior points

import numpy as np
from scipy.optimize import minimize_scalar
import time

def project_feasible(h, n):
    h = np.asarray(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        n_int = np.sum(interior)
        if n_int > 0:
            h[interior] += diff / n_int
            h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        interior = np.where((h > 1e-14) & (h < 1.0 - 1e-14))[0]
        if len(interior) > 0:
            step = diff / len(interior)
            h[interior] += step
            h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def exact_score(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def score_with_pair(h, p, q, t, n):
    s = h[p] + h[q]
    h2 = h.copy()
    h2[p] = t
    h2[q] = s - t
    dx = 2.0 / n
    corr = np.correlate(h2, 1.0 - h2, mode='full') * dx
    return float(np.max(corr))

def pair_descent(h, n, deadline, max_passes=300):
    h = h.copy()
    current_score = exact_score(h, n)
    no_improve = 0
    rng = np.random.default_rng(12345)
    
    for pass_num in range(max_passes):
        if time.time() > deadline:
            break
        pass_improved = False
        
        pair_list = [(p, q) for p in range(n) for q in range(p + 1, n)]
        rng.shuffle(pair_list)
        
        for p, q in pair_list:
            if time.time() > deadline:
                return h, current_score
            s = h[p] + h[q]
            lo = max(0.0, s - 1.0)
            hi = min(1.0, s)
            if hi - lo < 1e-12:
                continue
            
            n_grid = 25
            grid = np.linspace(lo, hi, n_grid)
            vals = np.array([score_with_pair(h, p, q, float(grid[i]), n) for i in range(n_grid)])
            best_idx = int(np.argmin(vals))
            best_t = float(grid[best_idx])
            best_val = float(vals[best_idx])
            
            i_lo = max(0, best_idx - 1)
            i_hi = min(n_grid - 1, best_idx + 1)
            t_lo = float(grid[i_lo])
            t_hi = float(grid[i_hi])
            
            if t_hi - t_lo > 1e-10:
                def obj_1d(tv):
                    return score_with_pair(h, p, q, float(tv), n)
                try:
                    res = minimize_scalar(obj_1d, bounds=(t_lo, t_hi), method='bounded',
                                         options={'maxiter': 25, 'xatol': 1e-10})
                    if res.fun < best_val - 1e-15:
                        best_val = float(res.fun)
                        best_t = float(res.x)
                except Exception:
                    pass
            
            if best_val < current_score - 1e-13:
                h[p] = best_t
                h[q] = s - best_t
                current_score = best_val
                pass_improved = True
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

def resample_h(h, n_old, n_new):

        return h.copy()
    x_old = np.linspace(0.0, 2.0, n_old + 2)[1:-1]
    x_new = np.linspace(0.0, 2.0, n_new + 2)[1:-1]
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)
