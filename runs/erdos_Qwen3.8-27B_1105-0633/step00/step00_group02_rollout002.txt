    if hard_deadline - t1 > 30:
        dur1 = min(budget_s * 0.40, hard_deadline - t1 - 30)
        h1, c1 = sa_optimize(h0, n, target_sum, rng,
                           T0=0.004, T_min=1e-10, duration=dur1, t_start=t1)
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1

    # Phase 2: SLSQP polish (10% budget)
    t2 = time.monotonic()
    if hard_deadline - t2 > 25:
        h2, c2 = slsqp_polish(best_h, n, n / 2.0, rng, max_iter=120, beta=60.0)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2

    # Phase 3: SA on polished (20% budget)
    t3 = time.monotonic()
    if hard_deadline - t3 > 20:
        dur3 = min(budget_s * 0.20, hard_deadline - t3 - 15)
        h3, c3 = sa_optimize(best_h, len(best_h), len(best_h) / 2.0, rng,
                           T0=0.002, T_min=1e-11, duration=dur3, t_start=t3)
        if c3 < best_c5:
            best_c5 = c3
            best_h = h3

    # Phase 4: Try nearby dimensions (15% budget)
    t4 = time.monotonic()
    n_cur = len(best_h)
    candidates = []
    for dn in [-2, 2, 4, 8, -4, 6]:
        nc = n_cur + dn
        if 8 <= nc <= 200:
            candidates.append(nc)
    candidates = sorted(set(candidates))

    for n_trial in candidates:
        if time.monotonic() > hard_deadline - 30:
            break
        h_t = resample_to_n(best_h, n_trial, rng)
        c_t = exact_c5(h_t)
        ts = time.monotonic()
        h_t2, c_t2 = sa_optimize(h_t, n_trial, n_trial / 2.0, rng,
                               T0=0.003, T_min=1e-10, duration=25, t_start=ts)
        if c_t2 < best_c5:
            best_c5 = c_t2
            best_h = h_t2

    # Phase 5: Final SLSQP polish
    t5 = time.monotonic()
    if hard_deadline - t5 > 20:
        nf = len(best_h)
        h5, c5 = slsqp_polish(best_h, nf, nf / 2.0, rng, max_iter=80, beta=100.0)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h5

    # Phase 6: Final short SA
    t6 = time.monotonic()
    if hard_deadline - t6 > 10:
        nf = len(best_h)
        dur6 = min(15.0, hard_deadline - t6 - 3)
        h6, c6 = sa_optimize(best_h, nf, nf / 2.0, rng,
                           T0=0.001, T_min=1e-12, duration=dur6, t_start=t6)
        if c6 < best_c5:
            best_c5 = c6
            best_h = h6

    # Final verification
    nf = len(best_h)
    final_h = make_feasible(best_h, nf / 2.0, rng)
    final_c5 = exact_c5(final_h)

    return final_h, final_c5, nf
