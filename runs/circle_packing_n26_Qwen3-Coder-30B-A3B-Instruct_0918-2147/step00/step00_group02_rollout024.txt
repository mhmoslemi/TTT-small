import numpy as np
from scipy.optimize import minimize
from ortools.linear_solver import pywraplp
from itertools import combinations

def generate_candidates(g, radius_levels):
    candidates = []
    grid_points = np.arange(0, 1 + g/2, g)
    for x in grid_points:
        for y in grid_points:
            for r in radius_levels:
                if x - r >= 0 and x + r <= 1 and y - r >= 0 and y + r <= 1:
                    candidates.append((x, y, r))
    return candidates

def build_milp(candidates):
    solver = pywraplp.Solver.CreateSolver('SCIP')
    if not solver:
        raise RuntimeError("SCIP solver not available")
    
    n = len(candidates)
    
    # Binary variables for selecting candidates
    z = [solver.BoolVar(f'z_{i}') for i in range(n)]
    
    # Grid point membership (map grid point to list of candidates)
    grid_map = {}
    for i, (x, y, r) in enumerate(candidates):
        grid_key = (round(x/g), round(y/g))
        if grid_key not in grid_map:
            grid_map[grid_key] = []
        grid_map[grid_key].append(i)
    
    # At most one candidate per grid point
    for grid_key, indices in grid_map.items():
        solver.Add(sum(z[i] for i in indices) <= 1)
    
    # Exactly 26 circles
    solver.Add(sum(z) == 26)
    
    # Non-overlap constraints using L-infinity approximation
    # For each pair of candidates
    for i, j in combinations(range(n), 2):
        x1, y1, r1 = candidates[i]
        x2, y2, r2 = candidates[j]
        
        # L-infinity constraints: 
        # |x1 - x2| + |y1 - y2| >= r1 + r2
        # |x1 - x2| - |y1 - y2| >= r1 + r2 (after rotation)
        
        # Add auxiliary variables for absolute values
        dx_ij = solver.NumVar(0, 1, f'dx_{i}_{j}')
        dy_ij = solver.NumVar(0, 1, f'dy_{i}_{j}')
        
        solver.Add(dx_ij >= x1 - x2)
        solver.Add(dx_ij >= x2 - x1)
        solver.Add(dy_ij >= y1 - y2)
        solver.Add(dy_ij >= y2 - y1)
        
        # Linear approx of |x1 - x2| + |y1 - y2| >= r1 + r2
        solver.Add(dx_ij + dy_ij >= r1 + r2)
        
        # add the second set of linearized constraints
        solver.Add(dx_ij - dy_ij >= r1 + r2)
        solver.Add(dy_ij - dx_ij >= r1 + r2)
        
        # big-M linking: dx_ij only active if both selected
        solver.Add(dx_ij <= z[i] + z[j])
        solver.Add(dy_ij <= z[i] + z[j])
    
    # Objective: maximize sum of radii
    solver.Maximize(sum(candidates[i][2] * z[i] for i in range(n)))
    
    return solver, z

def solve_milp(solver, z):
    status = solver.Solve()
    if status != pywraplp.Solver.OPTIMAL:
        raise RuntimeError("MILP solver did not find optimal solution")
    
    selected_indices = [i for i in range(len(z)) if z[i].solution_value() > 0.5]
    return selected_indices

def extract_initial_solution(selected_indices, candidates):
    centers = np.array([[candidates[i][0], candidates[i][1]] for i in selected_indices])
    radii = np.array([candidates[i][2] for i in selected_indices])
    return centers, radii

def shrink_initial(radii, delta=1e-5):
    return radii - delta

def pack_variables(centers, radii):
    n = centers.shape[0]
    z = np.zeros(3 * n)
    z[0::3] = centers[:, 0]  # x coordinates
    z[1::3] = centers[:, 1]  # y coordinates
    z[2::3] = radii           # radii
    return z

def unpack_solution(z):
    n = z.shape[0] // 3
    centers = np.zeros((n, 2))
    centers[:, 0] = z[0::3]  # x coordinates
    centers[:, 1] = z[1::3]  # y coordinates
    radii = z[2::3]          # radii
    return centers, radii

def build_constraint_list(centers0, radii0):
    n = centers0.shape[0]
    constraints = []
    
    # Boundary constraints for each circle (4 constraints per circle)
    for i in range(n):
        def boundary_constraint_func(z):
            centers, radii = unpack_solution(z)
            x, y = centers[i, 0], centers[i, 1]
            r = radii[i]
            return [r - x, r - (1-x), r - y, r - (1-y)]
        
        def boundary_constraint_jac(z):
            J = np.zeros((4, 3*n))
            centers, radii = unpack_solution(z)
            x, y = centers[i, 0], centers[i, 1]
            r = radii[i]
            J[0, 3*i+0] = -1  # d/dr - (x) = -1
            J[1, 3*i+0] = 1   # d/dr - (1-x) = 1
            J[2, 3*i+1] = -1  # d/dr - (y) = -1
            J[3, 3*i+1] = 1   # d/dr - (1-y) = 1
            return J
            
        constraints.append({
            'fun': boundary_constraint_func,
            'jac': boundary_constraint_jac,
            'type': 'ineq'
        })
    
    # Non-overlap constraints for each pair of circles
    for i, j in combinations(range(n), 2):
        def overlap_constraint_func(z):
            centers, radii = unpack_solution(z)
            x1, y1 = centers[i, 0], centers[i, 1]
            x2, y2 = centers[j, 0], centers[j, 1]
            r1 = radii[i]
            r2 = radii[j]
            dx = x1 - x2
            dy = y1 - y2
            dist = np.sqrt(dx*dx + dy*dy)
            # check that we're using the constraint g_i_j = (r1+r2) - dist <= 0
            return [(r1 + r2) - dist]
        
        def overlap_constraint_jac(z):
            centers, radii = unpack_solution(z)
            x1, y1 = centers[i, 0], centers[i, 1]
            x2, y2 = centers[j, 0], centers[j, 1]
            r1 = radii[i]
            r2 = radii[j]
            dx = x1 - x2
            dy = y1 - y2
            dist = np.sqrt(dx*dx + dy*dy)
            dist = max(dist, 1e-12)  # prevent singularity
            
            J = np.zeros((1, 3*n))
            J[0, 3*i+0] = dx/dist  # d/dr1 - (dist) = dx/dist
            J[0, 3*i+1] = dy/dist
            J[0, 3*i+2] = 1        # d/dr1 - (dist) = 1
            J[0, 3*j+0] = -dx/dist # d/dr2 - (dist) = -dx/dist
            J[0, 3*j+1] = -dy/dist
            J[0, 3*j+2] = 1        # d/dr2 - (dist) = 1
            return J
            
        constraints.append({
            'fun': overlap_constraint_func,
            'jac': overlap_constraint_jac,
            'type': 'ineq'
        })
    return constraints

def objective_func(z):
    return -np.sum(z[2::3])  # minus for maximization (minimize negative)

def jacobian_func(z):
    n = z.shape[0] // 3
    grad = np.zeros_like(z)
    grad[2::3] = -1         # derivative wrt. radii is -1
    return grad

def run_packing():
    radius_levels = np.arange(0.02, 0.26, 0.02)
    g = 0.04
    candidates = generate_candidates(g, radius_levels)
    
    solver, z_vars = build_milp(candidates)
    selected_indices = solve_milp(solver, z_vars)
    centers0, radii0 = extract_initial_solution(selected_indices, candidates)
    
    # shrink provided radii to avoid initial constraint violations
    radii0 = shrink_initial(radii0, 1e-5)
    
    # flatten for optimizer
    z0 = pack_variables(centers0, radii0)
    
    # build list of constraints
    constraints = build_constraint_list(centers0, radii0)
    
    bounds = [(0., 1.) for _ in range(3*len(centers0))]   # (0,1) for x,y; (0,0.5) for r
    for i in range(len(centers0)):
        bounds[3*i+2] = (0., 0.5)  # radius bounds
    
    # minimize the objective using trust-constr
    res = minimize(objective_func, z0, method='trust-constr', jac=jacobian_func,
                   bounds=bounds, constraints=constraints, options={'maxiter': 1000})
    
    # unpack result
    centers, radii = unpack_solution(res.x)
    
    # apply minor repair to ensure constraints met strictly
    # 1. Make sure boundaries are maintained (project rad to square)
    for i in range(len(radii)):
        radii[i] = min(radii[i], centers[i, 0], 1-centers[i, 0], 
                        centers[i, 1], 1-centers[i, 1])
    
    # 2. resolve overlaps if any
    for _ in range(5):
        changed = False
        for i, j in combinations(range(len(radii)), 2):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            r_sum = radii[i] + radii[j]
            if dist < r_sum:
                overlap = r_sum - dist
                # reduce both by half the overlap amount (but don't make negative)
                rad_diff = radii[i] - radii[j]
                rad_sum = radii[i] + radii[j]
                if rad_sum > 1e-12:
                    r1adj = overlap * radii[i] / rad_sum
                    r2adj = overlap * radii[j] / rad_sum
                    if r1adj > radii[i]:

                        r2adj = 0
                    if r2adj > radii[j]:
                        r2adj = radii[j]
                        r1adj = 0
                    radii[i] -= r1adj
                    radii[j] -= r2adj
