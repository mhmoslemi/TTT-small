def generate_lattice(s):
    """Generate a hexagonal lattice inside [0,1]x[0,1] with spacing s."""
    sqrt3 = math.sqrt(3)
    rows = int(1.0 / (s * sqrt3 / 2)) + 1
    cols = int(1.0 / s) + 1
    points = []
    for r in range(rows):
        y = r * (s * sqrt3 / 2)
        if y + s/2 > 1:
            break
        for c in range(cols):
            x = c * s
            if x + s/2 > 1:
                break
            if r % 2 == 0:
                x += 0.0
            else:
                x += s / 2
            if x - s/2 >= 0 and x + s/2 <= 1 and y - s/2 >= 0 and y + s/2 <= 1:
                points.append([x, y])
    return np.array(points)

def build_milp(P):
    """Build the MILP model as a function to return a callable for solving LP relaxations."""
    K = len(P)
    # Generate all pairwise distances
    D = np.zeros((K, K))
    for i in range(K):
        for j in range(i+1, K):
            D[i, j] = D[j, i] = np.linalg.norm(P[i] - P[j])
    
    # precompute wall distances
    wL = P[:, 0]
    wR = 1 - P[:, 0]
    wB = P[:, 1]
    wT = 1 - P[:, 1]
    w = np.minimum.reduce([wL, wR, wB, wT])
    
    # Define variables
    z = Variable(K, boolean=True)
    r = Variable(K, nonneg=True)

    # objective
    objective = Maximize(cvx_sum(r))

    # Build constraints
    constraints = [
        cvx_sum(z) == 26,
        r <= w * z,
    ]
    M = 1
    for i in range(K):
        for j in range(i+1, K):
            # p1 (i) and p2 (j) are both selected
            # d_ij >= r_i + r_j - M * (2 - z_i - z_j)
            constraints.append(D[i, j] >= r[i] + r[j] - M * (2 - z[i] - z[j]))
    
    def solve_relaxation(z_fixed=None):
        # Fix some z values if provided
        cons = constraints[:]
        if z_fixed is not None:
            for k in range(K):
                if z_fixed[k] is not None:
                    cons.append(z[k] == z_fixed[k])
        
        # Solve LP Relaxation
        prob = Problem(objective, cons)
        try:
            prob.solve(solver=ECOS, verbose=False, feastol=1e-9, abstol=1e-9)
        except:
            return None, float('-inf')

        if prob.status == Optimal:
            return np.array([r_k.value for r_k in r]), prob.value / 1.0
        else:
            return None, float('-inf')

    return solve_relaxation

def refine_solution(centers, radii):
    """Refine the solution with SLSQP-based continuous optimization"""
    # Define objective and constraints
    n = len(centers)
    def obj(radii):
        return -np.sum(radii)
    
    def constraint_op(i, j):
        def c(x):
            r = radii[i]
            c = (x[2*i] - x[2*j])**2 + (x[2*i+1] - x[2*j+1])**2 - (r + radii[j])**2
            return c
        return c
    def constraint_wall(i):
        def c(x):
            r = radii[i]
            # x[2*i], x[2*i+1] are c_i = (x_i, y_i)
            out = [
                x[2*i] - r,                    # left
                1 - x[2*i] - r,                # right
                x[2*i+1] - r,                  # bottom
                1 - x[2*i+1] - r               # top
            ]
            return np.array(out)
        return c

    bounds = []
    for i in range(n):
        bounds.extend([(1e-9, 1), (1e-9, 1)])  # (x, y) for each circle
        bounds.extend([(1e-9, 1)] * n) # radii - they are nonnegative and also limited by distance to boundary
    bounds = bounds[:-n]  # remove extra radii bounds (needed for closure)

    x0 = []
    for i in range(n):
        x0.append(centers[i][0])
        x0.append(centers[i][1])
        x0.append(radii[i])
        
    cons = []
    # Non-overlapping
    for i in range(n):
        for j in range(i+1, n):
            cons.append({'type': 'ineq', 'fun': constraint_op(i, j)})
    # Walls
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': constraint_wall(i)})

    result = minimize(obj, x0, method='SLSQP', constraints=cons, bounds=bounds, options={'ftol':1e-9, 'maxiter':2000})
    if result.success:
        x_sol = result.x
        centers_new = np.array([[x_sol[2*i], x_sol[2*i+1]] for i in range(n)])
        radii_new = np.array([x_sol[2*n+i] for i in range(n)])
        return centers_new, radii_new, float(np.sum(radii_new))
    else:
        return centers, radii, float(np.sum(radii))
