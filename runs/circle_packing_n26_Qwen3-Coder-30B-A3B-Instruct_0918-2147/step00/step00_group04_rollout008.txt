import numpy as np
import cvxpy as cp
from scipy.spatial.distance import cdist

def build_leaf_problem(cell, k):
    # cell: (x_min, x_max, y_min, y_max)
    x_min, x_max, y_min, y_max = cell
    # Variables for k circles
    centers = cp.Variable((k, 2))
    radii = cp.Variable(k, nonneg=True)
    
    # Constraints: center in cell and radius >= 0
    constraints = []
    for i in range(k):
        constraints.append(centers[i, 0] - radii[i] >= x_min)
        constraints.append(centers[i, 0] + radii[i] <= x_max)
        constraints.append(centers[i, 1] - radii[i] >= y_min)
        constraints.append(centers[i, 1] + radii[i] <= y_max)
    
    # Non-overlap constraints (SOC)
    for i in range(k):
        for j in range(i+1, k):
            d = cp.sqrt(cp.sum((centers[i] - centers[j])**2))
            constraints.append(d >= radii[i] + radii[j])
    
    # Objective
    objective = cp.Maximize(cp.sum(radii))
    
    prob = cp.Problem(objective, constraints)
    return prob, centers, radii

def solve_leaf(cell, k):
    prob, centers, radii = build_leaf_problem(cell, k)
    # Use ECOS for reliability
    prob.solve(solver=cp.ECOS, abstol=1e-10, reltol=1e-10)
    
    if prob.status != cp.OPTIMAL:
        # This should not happen in our structured approach
        raise RuntimeError(f"Leaf optimization failed: {prob.status}")
    
    return centers.value, radii.value

def merge_clusters(cluster_a, cluster_b, shared_edge):
    # Returns new merged cluster with all properties and updated centers/radii
    # Simply rewrite the merged problem from scratch here
    # This ensures it's always exact and respects contacts
    all_centers = np.vstack([cluster_a['centers'], cluster_b['centers']])
    all_radii = np.hstack([cluster_a['radii'], cluster_b['radii']])
    n_a = len(cluster_a['centers'])
    n_b = len(cluster_b['centers'])
    
    # Create merged problem
    centers = cp.Variable((n_a + n_b, 2))
    radii = cp.Variable(n_a + n_b, nonneg=True)
    
    constraints = []
    
    # Boundary constraints from all cells
    for i in range(n_a + n_b):
        x, y = centers[i]
        r = radii[i]
        # Pull in original constraints
        if i < n_a:
            # Already in cluster_a
            x_min, x_max, y_min, y_max = cluster_a['bounds']
        else:
            # Already in cluster_b
            x_min, x_max, y_min, y_max = cluster_b['bounds']
        
        constraints.append(x - r >= x_min)
        constraints.append(x + r <= x_max)
        constraints.append(y - r >= y_min)
        constraints.append(y + r <= y_max)
    
    # Non-overlap
    for i in range(n_a + n_b):
        for j in range(i+1, n_a + n_b):
            d = cp.sqrt(cp.sum((centers[i] - centers[j])**2))
            constraints.append(d >= radii[i] + radii[j])
    
    # Solve
    objective = cp.Maximize(cp.sum(radii))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, abstol=1e-10, reltol=1e-10)
    
    if prob.status != cp.OPTIMAL:
        raise RuntimeError(f"Merging failed: {prob.status}")
    
    merged_centers = centers.value
    merged_radii = radii.value
    
    return {
        'centers': merged_centers,
        'radii': merged_radii,
        'bounds': (0, 1, 0, 1)
    }

def get_all_cells():
    # 9 cells: 4 corners, 4 edges, 1 center
    # Each cell is a rectangle
    a = 0.25
    
    cells = [

        (0, a, 1-a, 1),  # Top-left
        (1-a, 1, 1-a, 1),  # Top-right
        (0, a, 0, a),  # Bottom-left
        (1-a, 1, 0, a),  # Bottom-right
        
        # Edge cells
        (a, 1-a, 1-a, 1),  # Top middle
        (a, 1-a, 0, a),   # Bottom middle
        (0, a, a, 1-a),   # Left middle
        (1-a, 1, a, 1-a), # Right middle
        
        # Center cell
        (a, 1-a, a, 1-a)
    ]
    
    # Circle assignments for 26 circles
    assignments = [5, 5, 5, 5, 1, 1, 1, 1, 2]
    return cells, assignments

def run_packing():
    # Get cells and initial circle count allocations
    cells, assignments = get_all_cells()
    assert len(cells) == len(assignments)
    
    # Solve each cell independently
    clusters = []
    for idx, (cell, num_circles) in enumerate(zip(cells, assignments)):
        if num_circles == 0:
            clusters.append({'centers': np.empty((0, 2)), 'radii': np.array([]), 'bounds': cell})
        else:
            centers, radii = solve_leaf(cell, num_circles)
            clusters.append({
                'centers': centers,
                'radii': radii,
                'bounds': cell
            })
    
    # Merge clusters in a binary-tree fashion (simplified here as sequential merging)
    merged_cluster = clusters[0]
    for i in range(1, len(clusters)):
        merged_cluster = merge_clusters(merged_cluster, clusters[i], None)

    # Now do a final polishing via a simple barrier gradient-like approach
    centers = merged_cluster['centers']
    radii = merged_cluster['radii']
    num_circles = len(centers)
    
    # Apply max sum radii, jiggering
    # Inline simple gradient
    def gradient_step(centers, radii, lr=0.01):
        # Compute forces: if too close, push apart
        # This is very approximate...
        n = len(centers)
        forces = np.zeros((n, 2))

        # Compute pairwise distances and forces
        c = centers.copy()
        r = radii
        dist = cdist(c, c)
        for i in range(n):
            for j in range(i+1, n):
                diff = c[i] - c[j]
                d = np.linalg.norm(diff)
                if d > 0 and d < r[i] + r[j]:
                    # Repel
                    force = (diff / d) * (r[i] + r[j] - d)
                    forces[i] += force
                    forces[j] -= force
                elif d == 0:
                    # Hack for identical centers
                    forces[i] += np.array([0.01, 0.01]) + np.random.randn(2)*0.001
                    forces[j] -= np.array([0.01, 0.01]) + np.random.randn(2)*0.001

        # Add wall forces
        for i in range(n):
            x, y = c[i]
            rad = r[i]
            if x - rad < 0:
                forces[i][0] += 0.1 * (rad - x)
            if x + rad > 1:
                forces[i][0] -= 0.1 * (x + rad - 1)
            if y - rad < 0:
                forces[i][1] += 0.1 * (rad - y)
            if y + rad > 1:
                forces[i][1] -= 0.1 * (y + rad - 1)

        # Normal update
        new_centers = c + lr * forces
        # Polishing: redraw radii to maximize total sum, with Google-style fixed
        # Take after all, new too-small circles might trigger vanishing-step issues.

        # Just do projection + average update to satisfy constraints
        for i in range(n):
            new_c = new_centers[i]
            new_c[0] = np.clip(new_c[0], r[i], 1 - r[i])
            new_c[1] = np.clip(new_c[1], r[i], 1 - r[i])
        
        return new_centers
    
    # Do 5 small optimization steps to improve total
    for _ in range(3):
        centers = gradient_step(centers, radii)
    # Final validation + large step-limited high precision refine
    sy = np.sum(radii)
    for _ in range(10):
        prob, c, r = build_leaf((0, 1, 0, 1), num_circles)
        # Force high precision small prec
        prob.solve(solver=cp.ECOS, abstol=1e-12, reltol=1e-12)
        if prob.status == cp.OPTIMAL:
            radii = r.value
            centers = c.value
            # If we can do better, keep going
            if np.sum(radii) > sy:
                sy = np.sum(radii)
            else:
                break
        else:
            break
    
    # Ensure final validation passes 
    pass_test, msg = validate_packing(centers, radii)
    if not pass_test:
        raise RuntimeError(f"Validation failed: {msg}")

    return centers, radii, np.sum(radii)
