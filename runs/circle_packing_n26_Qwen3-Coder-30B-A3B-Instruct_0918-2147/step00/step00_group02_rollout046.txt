import numpy as np
from scipy.optimize import minimize
from ortools.linear_solver import pywraplp
import math

def generate_candidates(g, radius_levels):
    candidates = []
    for i in range(int(1/g) + 1):
        for j in range(int(1/g) + 1):
            x = i * g
            y = j * g
            for r in radius_levels:
                if (x - r >= 0 and x + r <= 1 and y - r >= 0 and y + r <= 1):
                    candidates.append((x, y, r))
    return candidates

def build_milp(candidates):
    solver = pywraplp.Solver.CreateSolver('SCIP')
    if not solver:
        raise Exception('Could not create solver')
    
    n = len(candidates)
    z = [solver.IntVar(0, 1, f'z_{i}') for i in range(n)]
    
    # Single circle per grid point
    grid_points = {}
    for i, (x, y, r) in enumerate(candidates):
        key = (round(x/g), round(y/g))
        if key not in grid_points:
            grid_points[key] = []
        grid_points[key].append(i)
    
    for points in grid_points.values():
        solver.Add(sum(z[i] for i in points) <= 1)
    
    # Exactly 26 circles
    solver.Add(sum(z) == 26)
    
    # Non-overlap constraints using L-infinity approximations
    max_dist = 1.0
    for i in range(n):
        for j in range(i+1, n):
            x1, y1, r1 = candidates[i]
            x2, y2, r2 = candidates[j]
            dx = solver.NumVar(0, max_dist, f'dx_{i}_{j}')
            dy = solver.NumVar(0, max_dist, f'dy_{i}_{j}')
            
            solver.Add(dx >= x1 - x2)
            solver.Add(dx >= x2 - x1)
            solver.Add(dy >= y1 - y2)
            solver.Add(dy >= y2 - y1)
            solver.Add(dx + dy >= r1 + r2)
            solver.Add(dx - dy >= r1 + r2 - 2*max_dist)
            solver.Add(dx - dy <= 2*max_dist - r1 - r2)
    
    # Objective
    solver.Maximize(sum(candidates[i][2] * z[i] for i in range(n)))
    return solver, z

def solve_milp(solver, z):
    status = solver.Solve()
    if status != pywraplp.Solver.OPTIMAL:
        raise Exception(f'MILP solver failed with status {status}')
    
    selected = []
    for i, var in enumerate(z):
        if var.solution_value() > 0.5:
            selected.append(i)
    return selected

def extract_initial_solution(selected_indices, candidates):
    centers = np.array([[candidates[i][0], candidates[i][1]] for i in selected_indices])
    radii = np.array([candidates[i][2] for i in selected_indices])
    return centers, radii

def shrink_initial(radii, delta):
    return radii - delta

def pack_variables(centers0, radii0):
    n = len(centers0)
    z0 = np.zeros(3*n)
    for i in range(n):
        z0[3*i] = centers0[i][0]
        z0[3*i+1] = centers0[i][1]
        z0[3*i+2] = radii0[i]
    return z0

def unpack_solution(z):
    n = len(z) // 3
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i][0] = z[3*i]
        centers[i][1] = z[3*i+1]
        radii[i] = z[3*i+2]
    return centers, radii

def define_objective(z):
    return -sum(z[3*i+2] for i in range(len(z)//3))

def define_jacobian(z):
    n = len(z) // 3
    grad = np.zeros_like(z)
    for i in range(n):
        grad[3*i+2] = -1.0
    return grad

def define_constraints(z, centers, radii):
    n = len(centers)
    constraints = []
    # Boundary constraints
    for i in range(n):
        x, y, r = z[3*i], z[3*i+1], z[3*i+2]
        constraints.append({
            'type': 'ineq',
            'fun': lambda z, i=i: x - r,
            'jac': lambda z, i=i: np.array([1, 0, -1] + [0]*(3*n-3))
        })
        constraints.append({
            'type': 'ineq',
            'fun': lambda z, i=i: (1 - x) - r,
            'jac': lambda z, i=i: np.array([-1, 0, -1] + [0]*(3*n-3))
        })
        constraints.append({
            'type': 'ineq',
            'fun': lambda z, i=i: y - r,
            'jac': lambda z, i=i: np.array([0, 1, -1] + [0]*(3*n-3))
        })
        constraints.append({
            'type': 'ineq',
            'fun': lambda z, i=i: (1 - y) - r,
            'jac': lambda z, i=i: np.array([0, -1, -1] + [0]*(3*n-3))
        })
        
    # Pairwise distance constraints
    for i in range(n):
        for j in range(i+1, n):
            def fun(z):
                x1, y1, r1 = z[3*i], z[3*i+1], z[3*i+2]
                x2, y2, r2 = z[3*j], z[3*j+1], z[3*j+2]
                dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
                return (r1 + r2) - dist
            constraints.append({
                'type': 'ineq',
                'fun': fun,
                'jac': lambda z: np.array([
                    (x1 - x2) / (dist + 1e-12) if i == k else
                    -(x1 - x2) / (dist + 1e-12) if j == k else
                    0
                    for k in range(3*n)
                ])
            })
    return constraints

def repair_packing(centers, radii):
    n = len(centers)
    # Enforce boundary constraints
    for i in range(n):
        r = radii[i]
        x, y = centers[i]
        radii[i] = min(r, x, 1 - x, y, 1 - y)
    
    # Resolve overlaps
    for it in range(5):
        any_changed = False
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx*dx + dy*dy)
                r_sum = radii[i] + radii[j]
                if dist < r_sum:
                    overlap = r_sum - dist
                    old_r_i, old_r_j = radii[i], radii[j]
                    radii[i] = max(0, radii[i] - overlap/2)
                    radii[j] = max(0, radii[j] - overlap/2)
                    if old_r_i != radii[i] or old_r_j != radii[j]:
                        any_changed = True
        if not any_changed:
            break
    return centers, radii

def run_packing():
    g = 0.03
    radius_levels = [r*0.02 for r in range(1, 13)]  # [0.02, ..., 0.24]
    candidates = generate_candidates(g, radius_levels)
    
    solver, z = build_milp(candidates)
    selected = solve_milp(solver, z)
    
    centers0, radii0 = extract_initial_solution(selected, candidates)
    radii0 = shrink_initial(radii0, 1e-5)
    
    z0 = pack_variables(centers0, radii0)
    
    def fun(z):
        return define_objective(z)
    
    def jac(z):
        return define_jacobian(z)
    
    bounds = [(0, 1) for _ in range(len(z0))]
    
    res = minimize(fun, z0, method='trust-constr', jac=jac, bounds=bounds)
    
    centers, radii = unpack_solution(res.x)
    centers, radii = repair_packing(centers, radii)
    
    # Make sure offsets are positive
    eps = 1e-10
    for i in range(len(centers)):
        r = radii[i]
        x, y = centers[i]
        radii[i] = min(r, x - eps, 1 - x - eps, y - eps, 1 - y - eps)
    
    sum_radii = np.sum(radii)
    
    # Validate final packing
    success, msg = validate_packing(centers, radii)
    if not success:
        raise AssertionError(f"Validation failed: {msg}")
    
    return centers, radii, sum_radii
