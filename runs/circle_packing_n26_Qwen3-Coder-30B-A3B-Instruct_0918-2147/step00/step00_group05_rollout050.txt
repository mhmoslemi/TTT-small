import numpy as np

def hex_lattice(d):
    """Generate hexagonal lattice points within unit square."""
    rows = []
    y = 0
    while y <= 1:
        x = 0
        row = []
        offset = (y % (d * np.sqrt(3) / 2)) == 0
        if offset:
            x = 0
        else:
            x = d / 2
        while x <= 1:
            row.append([x, y])
            x += d
        rows.extend(row)
        y += d * np.sqrt(3) / 2
    return np.array(rows)

def add_border_circles(points, d):
    """Add six border-enhanced circles."""
    r = d / 4
    border = [
        [r, r],  # bottom-left
        [1 - r, r],  # bottom-right
        [r, 1 - r],  # top-left
        [1 - r, 1 - r],  # top-right
        [0.5, r],  # middle-bottom
        [0.5, 1 - r]  # middle-top
    ]
    return np.vstack([points, border])

def initial_radii(points):
    """Compute a safe initial radius for all points."""
    n = points.shape[0]
    min_wall = np.minimum(
        points[:, 0], 1 - points[:, 0],
        points[:, 1], 1 - points[:, 1]
    )
    # Compute distance matrix once
    diff = points[:, np.newaxis, :] - points[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dist, np.inf)
    min_dist = np.min(dist, axis=1)
    r0 = np.min([min_wall, min_dist / 2], axis=0)
    return np.min(r0)

def inflate_radii(centers, radii):
    """Inflate radii to their maximum allowable values."""
    n = centers.shape[0]
    # Compute distance matrix once
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dist, np.inf)
    new_radii = radii.copy()
    changed = True
    while changed:
        changed = False
        for i in range(n):
            # Wall constraints
            wall_dist = np.array([
                centers[i, 0],
                1 - centers[i, 0],
                centers[i, 1],
                1 - centers[i, 1]
            ])
            
            # Neighbor constraints
            neighbor_dist = dist[i, :] - radii[:]
            
            # Max radius is the minimum of all upper bounds
            max_r = np.min(np.hstack([wall_dist, neighbor_dist[neighbor_dist > 0]]))
            if max_r - new_radii[i] > 1e-7:
                new_radii[i] = max_r
                changed = True
    return new_radii

def relax_centers(centers, radii, steps=10):
    """Relax center positions using repulsive forces."""
    n = centers.shape[0]
    # Constants
    kappa = 0.02
    lambda_force = 0.01
    alpha = 0.5
    beta = 0.02
    gamma = 0.02
    updated_centers = centers.copy()
    
    for _ in range(steps):
        forces = np.zeros_like(updated_centers)
        # Compute distance matrix once
        diff = updated_centers[:, np.newaxis, :] - updated_centers[np.newaxis, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dist, np.inf)
        
        # Repulsive forces
        for i in range(n):
            for j in range(n):
                if i == j:
                    continue
                d = dist[i, j]
                r_sum = radii[i] + radii[j]
                if d < r_sum + beta:
                    force_mag = kappa * max(0, beta - d)
                    direction = diff[i, j] / (d + 1e-12)
                    forces[i] += force_mag * direction
        
        # Wall forces
        for i in range(n):
            fx = 0
            fy = 0
            x, y = updated_centers[i]
            r = radii[i]
            # Left wall
            if x - r < gamma:
                fx += -lambda_force * max(0, gamma + r - x)
            # Right wall
            if 1 - x - r < gamma:
                fx += lambda_force * max(0, gamma + r - (1 - x))
            # Bottom wall
            if y - r < gamma:
                fy += -lambda_force * max(0, gamma + r - y)
            # Top wall
            if 1 - y - r < gamma:
                fy += lambda_force * max(0, gamma + r - (1 - y))
            forces[i] += [fx, fy]
        
        # Update centers
        updated_centers += alpha * forces
        
        # Keep centers within bounds
        updated_centers[:, 0] = np.clip(updated_centers[:, 0], radii, 1 - radii)
        updated_centers[:, 1] = np.clip(updated_centers[:, 1], radii, 1 - radii)
    
    return updated_centers

def run_packing():
    d = 0.18  # Hexagonal spacing
    core_points = hex_lattice(d)
    # Keep only first 20 (or fewer) core points
    core_points = core_points[:20]
    # Add border circles
    extended_points = add_border_circles(core_points, d)
    # Compute initial uniform radius
    initial_r = initial_radii(extended_points)
    initial_radii_arr = np.full(extended_points.shape[0], initial_r)
    centers = extended_points.copy()
    radii = initial_radii_arr.copy()
    
    # Iterate to increase sum
    MAX_OUTER = 30
    previous_sum = np.sum(radii)
    for outer in range(MAX_OUTER):
        radii = inflate_radii(centers, radii)
        centers = relax_centers(centers, radii)
        current_sum = np.sum(radii)
        if current_sum - previous_sum < 1e-6:
            break
        previous_sum = current_sum
    
    # Ensure all radii are non-negative
    radii = np.maximum(radii, 0)
    
    # Validate
    is_valid, message = validate_packing(centers, radii)
    if not is_valid:
        raise AssertionError(f"Final packing invalid: {message}")
    
    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
