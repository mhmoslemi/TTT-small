import numpy as np
from scipy.optimize import linprog
from scipy.spatial.distance import cdist

def generate_hex_lattice(h):
    sqrt3 = np.sqrt(3)
    points = []
    y = 0
    while y <= 1 + h:
        x = 0
        row_offset = (y % (h * sqrt3)) * 0.5
        while x <= 1 + h:
            points.append([x, y])
            x += h
        y += h * sqrt3
    points = np.array(points)
    # Keep only those in [0,1] x [0,1]
    mask = (points[:, 0] >= 0) & (points[:, 0] <= 1) & (points[:, 1] >= 0) & (points[:, 1] <= 1)
    points = points[mask]
    # Compute wall radii
    wall_radii = np.minimum(np.minimum(points[:, 0], 1 - points[:, 0]),
                            np.minimum(points[:, 1], 1 - points[:, 1]))
    return points, wall_radii

def build_neighbour_list(centers, max_dist=0.4):
    distances = cdist(centers, centers)
    indices = np.where((distances > 0) & (distances < max_dist))
    pairs = list(zip(indices[0], indices[1], distances[indices]))
    return pairs

def solve_radius_lp(selected_indices, R_wall, neighbour_list):
    n_selected = len(selected_indices)
    if n_selected == 0:
        return np.array([]), True
    # Set up LP: maximize sum(r_i)
    c = -np.ones(n_selected)
    # Inequality constraints: A_ub @ x <= b_ub
    A_ub = []
    b_ub = []
    # Wall constraints
    for i in range(n_selected):
        A_ub.append([0] * n_selected)
        A_ub[-1][i] = -1  # -r_i <= -R_wall[i] --> r_i >= R_wall[i]
        b_ub.append(-R_wall[selected_indices[i]])
    # Separation constraints
    for i, j, d_ij in neighbour_list:
        if i in selected_indices and j in selected_indices:
            idx_i = selected_indices.index(i)
            idx_j = selected_indices.index(j)
            A_ub.append([0] * n_selected)
            A_ub[-1][idx_i] = 1
            A_ub[-1][idx_j] = 1
            b_ub.append(d_ij)
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    if res.success:
        radii = res.x
        if np.any(radii < -1e-10):
            return np.array([]), False
        return radii, True
    return np.array([]), False

def upper_bound(partial_set, remaining_indices, R_wall):
    ub = 0
    for i in partial_set:
        ub += R_wall[i]
    # Take largest wall radius from remaining candidates
    if len(remaining_indices) > 0:
        max_r = max(R_wall[i] for i in remaining_indices)
        slots = 26 - len(partial_set)
        ub += max_r * slots
    return ub

def branch_and_bound(centers, R_wall, neighbour_list):
    best_total = -np.inf
    best_indices = None
    best_radii = None

    def search(partial_set, remaining_indices, lb):
        nonlocal best_total, best_indices, best_radii
        if len(partial_set) == 26:
            # Final check: solve exact LP
            radii, feasible = solve_radius_lp(partial_set, R_wall, neighbour_list)
            if not feasible:
                return lb
            total = np.sum(radii)
            if total > best_total:
                best_total = total
                best_indices = partial_set.copy()
                best_radii = radii.copy()
            return max(lb, total)
        elif len(partial_set) > 26:
            return lb
        # Compute upper bound
        ub = upper_bound(partial_set, remaining_indices, R_wall)
        if ub <= lb:
            return lb
        # Select next candidate with max wall radius
        next_i = max(remaining_indices, key=lambda i: R_wall[i])
        # Branch: include next_i
        partial_set.append(next_i)
        new_remaining = [i for i in remaining_indices if i != next_i]
        new_lb = search(partial_set, new_remaining, lb)
        partial_set.pop()
        # Branch: exclude next_i
        new_lb = search(partial_set, new_remaining, new_lb)
        return new_lb

    initial_remaining = list(range(len(centers)))
    initial_partial = []
    initial_lb = -np.inf
    final_lb = search(initial_partial, initial_remaining, initial_lb)
    if best_indices is None:
        raise RuntimeError("Branch and bound failed to find a solution")
    return best_indices, best_radii

def local_continuous_refinement(centers, radii):
    from scipy.optimize import minimize

    # Store original centers for bound updates
    orig_centers = centers.copy()
    n = len(centers)
    # Initial parameters: centers + radii = 26 * 2 + 26 = 78 variables
    x0 = np.concatenate([centers.ravel(), radii])
    # Constraint handling
    def constraint_func(x):
        # Reconstruct centers and radii
        centers_flat = x[:2*n].reshape((n, 2))
        radii_vec = x[2*n:]
        # Make sure radius >= 0
        constraints = []
        for i in range(n):
            constraints.append(radii_vec[i])
        # Make sure circle is inside square
        for i in range(n):
            x_i, y_i = centers_flat[i]
            r_i = radii_vec[i]
            constraints.append(x_i - r_i) # x-r >= 0
            constraints.append(1 - x_i - r_i) # 1-x-r >= 0
            constraints.append(y_i - r_i) # y-r >= 0
            constraints.append(1 - y_i - r_i) # 1-y-r >= 0
        # Separation constraint
        for i in range(n):
            for j in range(i+1, n):
                dx = centers_flat[i][0] - centers_flat[j][0]
                dy = centers_flat[i][1] - centers_flat[j][1]
                dist_sq = dx*dx + dy*dy
                r_i = radii_vec[i]
                r_j = radii_vec[j]
                # Only add constraint if packable in at least one direction
                if dist_sq < (r_i + r_j) ** 2:
                    constraints.append((r_i + r_j)**2 - dist_sq)
        return np.array(constraints)
    
    # Objective: maximize sum of radii (minimize minus)
    def obj_func(x):
        radii = x[2*n:]
        return -np.sum(radii)

    # Create bounds: normal interval constraints
    bounds = [(0,1)] * (2*n) + [(0, None)] * n  # For radii we allow positive
    # Note: scipy minimizes, so we invert sign to maximize sum of radii.

    # Final refine:
    # Minimize the negative sum of radii
    result = minimize(obj_func, x0, method='trust-constr', bounds=bounds, 
                      constraints={'type': 'ineq', 'fun': constraint_func}, 
                      options={'maxiter': 200, 'gtol': 1e-6})
    if result.success:
        refined_centers = result.x[:2*n].reshape((n, 2))
        refined_radii = result.x[2*n:]
        return refined_centers, refined_radii
    else:
        # Fallback: return original
        return orig_centers, radii

def run_packing():
    # Parameters
    h = 0.075
    # Generate hexagonal lattice
    centers, wall_radii = generate_hex_lattice(h)
    # Build neighbor list
    neigh = build_neighbour_list(centers, 0.4)
    # Perform search
    selected, radii_opt = branch_and_bound(centers, wall_radii, neigh)
    # Extract centers and radii
    centers_selected = centers[selected]
    # Refine using local optimization
    final_centers, final_radii = local_continuous_refinement(centers_selected, radii_opt)

    # Ensure no violation and validate
    validated, msg = validate_packing(final_centers, final_radii)
    if not validated:
        # Shrink slightly
        shrink_rate = 5e-8
        for _ in range(5):
            final_radii = np.maximum(0, final_radii - shrink_rate)
            validated, msg = validate_packing(final_centers, final_radii)
            if validated:
                break
        if not validated:
            raise RuntimeError("Could not produce valid packing: %s" % msg)
    sum_radii = float(np.sum(final_radii))
    return final_centers, final_radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
