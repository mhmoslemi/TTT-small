import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def initial_configuration(n=26):
    # Use Sobol-like low discrepancy points
    phi = (1 + np.sqrt(5)) / 2
    indices = np.arange(n)
    golden_section = (indices * phi) % 1
    spiral_angle = 2 * np.pi * indices / (phi + 1)
    r = np.sqrt(indices + 0.5) / np.sqrt(n - 0.5)
    x = r * np.cos(spiral_angle)
    y = r * np.sin(spiral_angle)
    # Scale to [0.1, 0.9] avoiding borders
    min_val = 0.1
    max_val = 0.9
    x = min_val + (x + 1) * (max_val - min_val) / 2
    y = min_val + (y + 1) * (max_val - min_val) / 2
    centers = np.column_stack((x, y))
    
    # Compute initial radii using distance to nearest neighbors
    distances = cdist(centers, centers)
    np.fill_diagonal(distances, np.inf)
    min_distances = np.min(distances, axis=1)
    radii = np.minimum(
        centers[:, 0],          # left
        1 - centers[:, 0],      # right
        centers[:, 1],          # bottom
        1 - centers[:, 1],      # top
        min_distances / 2       # half nearest neighbor distance
    )
    # Ensure they're positive
    radii = np.maximum(radii, 1e-6)
    return centers, radii

def unpack(z):
    n = 26
    centers = z[:2*n].reshape((n, 2))
    radii = z[2*n:]
    return centers, radii

def pack(centers, radii):
    z = np.concatenate([centers.flatten(), radii])
    return z

def energy_and_grad(z):
    n = 26
    centers, radii = unpack(z)
    # Wall penalties squared
    wall_x = np.maximum(0, radii - centers[:, 0])
    wall_y = np.maximum(0, radii - centers[:, 1])
    wall_r = np.maximum(0, radii - (1 - centers[:, 0]))
    wall_t = np.maximum(0, radii - (1 - centers[:, 1]))
    
    # Compute pairwise distances but do not enforce zero elements at diagonal
    # let each r contribute to the gradient for all others
    # (much more efficient than element-wise logic)
    D = cdist(centers, centers)
    # We compute overlap term for all pairs i,j as a vectorized sum
    # We scale the matrix for overlaps as r_i + r_j - d_ij
    r_sum = radii[:, None] + radii[None, :]
    diffs = r_sum - D
    # Truncate negative values (non-overlapping)
    overlap = np.maximum(diffs, 0)
    # Total overlap energy
    # multiply by 0.5 so we don't double-count pairs (i<j only counts once)
    E_pair = 0.5 * np.sum(overlap**2)
    
    # Wall energy
    E_wall = np.sum(wall_x**2 + wall_y**2 + wall_r**2 + wall_t**2)
    
    # Budget term
    # We will minimize E = E_wall + E_pair + lambda * (B - sum_r)**2
    # So we will have B = sum_r (at equilibrium) up to small fluctuations
    # Coefficient for lambda is usually of order 1.
    lambda_budget = 1000.0
    sum_r = np.sum(radii)
    E_budget = lambda_budget * (sum_r - 3.0)**2  # dummy B=3.0 fixed here
    
    # Now compute the gradient wrt both centers and radii:
    n = 26
    grad_c = np.zeros((n, 2))
    grad_r = np.zeros(n)

    # Wall adjustments for r:
    grad_r += 2 * (wall_x + wall_y + wall_r + wall_t)

    # Overlap energy: for pair (i,j) inside overlap
    # Gradient w.r.t. center of i:
    #   ∂E/∂c_i = -2 * (r_i + r_j - d_ij) * (c_i - c_j) / d_ij
    # where d_ij = ||c_i - c_j||
    # if d_ij = 0 then use tiny value to avoid division by zero
    
    # Compute directions and norms:
    # fix indexing for i over j in this loop
    grad_terms = np.zeros_like(centers)
    # avoid double counting:
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    A = np.zeros((n, n))
    A[mask] = np.triu(overlap, k=1).flatten()
    A = A + A.T
    V = np.zeros((n,n,2))
    # Direction vectors between circle centers
    diff = centers[:, None, :] - centers[None, :, :]
    d = np.linalg.norm(diff, axis=2)
    d = np.where(d > 1e-12, d, 1e-12)  # protect from nan
    dr = diff / d[:,:,None]
    # compute gradient of pair k terms
    # these are based on the relationship between Q = (r_i + r_j - d_ij) and c_i
    #   dE/dx_i = -2 * Q * (c_i - c_j)/d_ij
    dyn_grads = -2 * np.triu(overlap, k=1)[:,:, None] * dr
    # now sum over second index
    # This is already computed correctly when we use positve indices with all pairs
    # To make it symmetric: we must convert to 2D matrix properly
    tmp = dyn_grads.reshape((n*n,2))
    # unpack into i, j slice structure
    # build per-circle sum:
    for i in range(n):
        for j in range(n):
            if i != j:
                grad_terms[i] -= (overlap[i,j] * dr[i,j])
    # equivalent to summing over all i,j such that i != j and overlap > 0
    # leading to:
    grad_c = np.sum(grad_terms, axis=0)
    
    # Gradient for radii based on overlaps:
    # Each circle returns a contribution from:
    # dE/d[r_i] = 2 * sum_over_j[max(0, r_i + r_j - d_ij)] 
    # e.g. each overlap contributes 2 * (r_i + r_j - d_ij)
    for i in range(n):
        for j in range(n):
            if i != j and overlap[i,j] > 0:
                grad_r[i] += 2 * overlap[i,j]
                
    # Gradient from budget:
    grad_r += -2 * lambda_budget * (sum_r - 3.0)
    
    grad = np.concatenate([grad_c.flatten(), grad_r])
    E_total = E_wall + E_pair + E_budget
    return E_total, grad

def inner_optimize(z0, B, lambda_budget):
    # Use L-BFGS-B
    res = minimize(energy_and_grad, z0,
                   method='L-BFGS-B',
                   bounds=[(0, 1)]*52 + [(0, 0.5)]*26,
                   jac=True,
                   options={'maxiter': 300, 'ftol': 1e-12})

    z_opt = res.x
    return z_opt

def max_feasible_budget(z_start):
    B = 3.0
    lambda_budget = 1000.0
    delta = 0.0005
    max_iter = 40
    min_delta = 1e-6
    z_best = z_start
    B_best = 0.0
    
    for iter_count in range(max_iter):
        z_opt = inner_optimize(z_start.copy(), B, lambda_budget)
        centers, radii = unpack(z_opt)
        
        # Check feasibility somewhat tighter than validation function's 1e-12 tolerance
        max_violation = 0
        max_violation = max(max_violation, np.max(centers[:, 0] - radii)) # left
        max_violation = max(max_violation, np.max(1 - centers[:, 0] - radii)) # right
        max_violation = max(max_violation, np.max(centers[:, 1] - radii)) # bottom
        max_violation = max(max_violation, np.max(1 - centers[:, 1] - radii)) # top
        
        # compute overlap violations
        D = cdist(centers, centers)
        np.fill_diagonal(D, np.inf)
        diff = radii[:, None] + radii[None, :] - D
        overlap_max = np.max(np.maximum(diff, 0))
        max_violation = max(max_violation, overlap_max)
        
        feasible = max_violation <= 1e-13
        
        if feasible:
            B_best = B
            z_best = z_opt
            B += delta
        else:
            B -= delta
        
        if delta < min_delta:
            break
        if B <= 0:
            break
            
    return z_best

def postprocess(z):
    centers, radii = unpack(z)
    eps = 5e-10
    
    # Ensure radii are not too large
    radii = np.minimum(radii, 0.5)
    radii = np.maximum(radii - eps, 0)
    
    # Clip centers to ensure they are inside [eps, 1 - eps]
    centers[:, 0] = np.clip(centers[:, 0], eps, 1 - eps)
    centers[:, 1] = np.clip(centers[:, 1], eps, 1 - eps)
    
    # One final repair pass:
    D = cdist(centers, centers)
    np.fill_diagonal(D, np.inf)
    r_sum = radii[:, None] + radii[None, :]
    over = np.maximum(0, r_sum - D)
    
    for i in range(len(centers)):
        for j in range(i):
            if over[i, j] > 0:
                # If we are w/in epsilon from touching threshold, leave alone
                if over[i, j] < 1e-10:
                    continue
                    
                # Halve the overlap and push radii apart
                amount = over[i, j] / 2.0 + eps
                radii[i] -= amount
                radii[j] -= amount
    
    radii = np.maximum(radii, 0)
    sum_r = np.sum(radii)
    return centers, radii, sum_r

def run_packing():
    n = 26
    centers, radii = initial_configuration(n)  # low discrepancy relaxation
    z = pack(centers, radii)
    z_opt = max_feasible_budget(z)
    centers, radii, sum_r = postprocess(z_opt)
    return centers, radii, sum_r
