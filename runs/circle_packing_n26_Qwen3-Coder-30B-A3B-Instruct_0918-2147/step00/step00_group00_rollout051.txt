import numpy as np
import random

def make_hex_seed(n, margin):
    # Create a hexagonal lattice inside the unit square
    # Adjust spacing to fit within the square with margin
    rows = int(np.sqrt(n) * 1.2)
    cols = int(n / rows) + 1
    
    # Overestimate to ensure coverage
    spacing = (1 - 2 * margin) / max(rows, cols)
    if spacing <= 0:
        spacing = 0.05
    
    centers = []
    radii = []
    
    for i in range(rows):
        for j in range(cols):
            x = margin + (j + 0.5 * (i % 2)) * spacing
            y = margin + i * spacing * np.sqrt(3) / 2
            if x <= 1 - margin and y <= 1 - margin:
                centers.append([x, y])
                # Initial radius based on distance to nearest wall
                r = min(x, 1 - x, y, 1 - y) * 0.5
                radii.append(max(0.015, r))
                if len(centers) >= n:
                    break
        if len(centers) >= n:
            break
    
    # Fill up to n if needed
    while len(centers) < n:
        # Add random positions near the center
        x = 0.5 + random.uniform(-0.2, 0.2)
        y = 0.5 + random.uniform(-0.2, 0.2)
        if 0 <= x <= 1 and 0 <= y <= 1:
            centers.append([x, y])
            r = min(x, 1 - x, y, 1 - y) * 0.3
            radii.append(max(0.015, r))
    
    # Truncate to n
    centers = centers[:n]
    radii = radii[:n]
    
    return np.array(centers), np.array(radii)

def max_feasible_radius(i, centers, radii):
    c_i = centers[i]
    r_i = radii[i]
    
    # Distance to walls
    d_L = c_i[0]
    d_R = 1.0 - c_i[0]
    d_B = c_i[1]
    d_T = 1.0 - c_i[1]
    
    min_dist = min(d_L, d_R, d_B, d_T)
    
    # Distance to other circles
    for j in range(centers.shape[0]):
        if i != j:
            dist = np.sqrt(np.sum((c_i - centers[j]) ** 2))
            min_dist = min(min_dist, dist - radii[j])
    
    return min(min_dist, r_i)

def inflate_radii(centers, radii, alpha):
    mean_r = np.mean(radii)
    delta = alpha * mean_r
    
    new_radii = radii + delta
    total_increase = 0.0
    
    for i in range(len(centers)):
        max_feasible = max_feasible_radius(i, centers, radii)
        old_r = radii[i]
        new_radii[i] = min(new_radii[i], max_feasible)
        total_increase += new_radii[i] - old_r
    
    return new_radii, total_increase

def anneal_centers(centers, radii, temp0, gamma, sigma0, steps):
    # Copy to prevent modification of original during annealing
    centers = centers.copy()
    radii = radii.copy()
    
    best_sum = np.sum(radii)
    best_centers = centers.copy()
    best_radii = radii.copy()
    
    temp = temp0
    sigma = sigma0
    
    spatial_shift = np.array([0.0, 0.0])
    shift_period = 50
    shift_magnitude = 0.003
    
    for step in range(steps):
        # Occasionally shift all centers globally
        if step % shift_period == 0:
            spatial_shift = np.random.uniform(-shift_magnitude, shift_magnitude, 2)
            centers += spatial_shift
        
        # Pick a random circle to move
        i = np.random.randint(0, centers.shape[0])
        old_pos = centers[i].copy()
        
        # Propose new position
        delta = np.random.normal(0, sigma, 2)
        centers[i] += delta
        
        # Ensure new position is valid
        centers[i][0] = np.clip(centers[i][0], 0, 1)
        centers[i][1] = np.clip(centers[i][1], 0, 1)
        
        # Compute new feasible radius
        new_feasible = max_feasible_radius(i, centers, radii)
        old_radius = radii[i]
        
        # Accept move if achievable radius increases
        if new_feasible > old_radius:
            radii[i] = new_feasible
            new_sum = np.sum(radii)
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = centers.copy()
                best_radii = radii.copy()
        else:
            # Accept with probability
            delta_r = new_feasible - old_radius
            prob = np.exp(delta_r / temp)
            if np.random.random() < prob:
                radii[i] = new_feasible
                new_sum = np.sum(radii)
                if new_sum > best_sum:
                    best_sum = new_sum
                    best_centers = centers.copy()
                    best_radii = radii.copy()
            else:
                # Restore previous position and radius
                centers[i] = old_pos
                radii[i] = old_radius
        
        # Update temperature and noise
        temp = max(1e-6, temp * gamma)
        sigma = max(1e-5, sigma * gamma)
    
    return best_centers, best_radii, best_sum

def mixed_cycle(centers, radii, params):
    alpha = params.get('alpha', 0.04)
    inflation_tolerance = params.get('inflation_tolerance', 1e-6)
    stagnation_threshold = params.get('stagnation_threshold', 8)
    sa_steps = params.get('sa_steps', 200)
    max_cycles = params.get('max_cycles', 5)
    temp0 = params.get('temp0', 0.02)
    gamma = params.get('gamma', 0.97)
    sigma0 = params.get('sigma0', 0.03)
    
    sum_radii = np.sum(radii)
    last_increase = 1.0
    stagnation_count = 0
    cycles = 0
    
    while cycles < max_cycles:
        # Perform inflation
        iteration = 0
        while iteration < 100 and stagnation_count < stagnation_threshold:
            old_sum = np.sum(radii)
            radii, increase = inflate_radii(centers, radii, alpha)
            new_sum = np.sum(radii)
            iteration += 1
            
            if increase < inflation_tolerance:
                stagnation_count += 1
            else:
                stagnation_count = 0
                last_increase = increase
                sum_radii = new_sum
        
        # Proceed to SA stage
        centers, radii, new_sum = anneal_centers(
            centers, radii, temp0, gamma, sigma0, sa_steps
        )
        if new_sum > sum_radii:
            sum_radii = new_sum
        
        # Try to reinflate after SA
        radii, increase = inflate_radii(centers, radii, alpha)
        sum_radii = np.sum(radii)
        
        if increase < inflation_tolerance:
            stagnation_count += 1
        else:
            stagnation_count = 0
            last_increase = increase
        
        cycles += 1
        
        if stagnation_count >= stagnation_threshold:
            break
    
    return centers, radii, sum_radii

def polish_radii(centers, radii, beta, passes):
    # Iteratively inflate all circles
    for _ in range(passes):
        updated = True
        iterations = 0
        while updated and iterations < 10:
            updated = False
            for i in range(len(centers)):
                g_wall = min(centers[i][0], 1 - centers[i][0], 
                             centers[i][1], 1 - centers[i][1])
                g_nei = float('inf')
                
                for j in range(len(centers)):
                    if i != j:
                        dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                        gap = dist - (radii[i] + radii[j])
                        if gap < g_nei:
                            g_nei = gap
                
                minimal = min(g_wall, g_nei)
                if minimal > 0 and minimal < np.inf:
                    increase = beta * minimal
                    if radii[i] + increase < max_feasible_radius(i, centers, radii):
                        radii[i] += increase
                        updated = True
            iterations += 1
        
        # If overlaps still exist, resolve them
        new_radii = radii.copy()
        max_iter = 3
        for _ in range(max_iter):
            any_overlap = False
            for i in range(len(centers)):
                for j in range(i+1, len(centers)):
                    dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                    total_r = radii[i] + radii[j]
                    if dist < total_r:
                        overlap = total_r - dist
                        new_radii[i] -= overlap / 2
                        new_radii[j] -= overlap / 2
                        any_overlap = True
            radii = new_radii.copy()
            if not any_overlap:
                break
    
    return centers, radii
