import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import math

def generate_lattice(s):
    """Generate a hexagonal lattice of candidate center points."""
    sqrt3 = math.sqrt(3)
    y_spacing = s * sqrt3 / 2
    x_spacing = s
    points = []
    y = s / 2
    row = 0
    while y < 1 - s / 2:
        x_start = s / 2 if row % 2 == 0 else s
        x = x_start
        while x < 1 - s / 2:
            points.append([x, y])
            x += x_spacing
        y += y_spacing
        row += 1
    return np.array(points)

def build_milp(P):
    """Build a MILP model for circle packing using cvxpy."""
    K = len(P)
    # Create cvxpy variables
    z = cp.Variable(K, boolean=True)
    r = cp.Variable(K, nonneg=True)
    
    # Precompute wall distances for each point
    wL = P[:, 0]
    wR = 1 - P[:, 0]
    wB = P[:, 1]
    wT = 1 - P[:, 1]
    w = np.minimum(np.minimum(wL, wR), np.minimum(wB, wT))
    
    # Precompute distances between all pairs
    d_kl = np.zeros((K, K))
    for i in range(K):
        for j in range(i + 1, K):
            d = np.linalg.norm(P[i] - P[j])
            d_kl[i, j] = d
            d_kl[j, i] = d
    
    # Constraints
    constrs = []
    # Exactly 26 circles
    constrs.append(cp.sum(z) == 26)
    
    # Radii must not exceed wall distances
    for i in range(K):
        constrs.append(r[i] <= w[i] * z[i])
    
    # Non-overlap constraints using big-M method
    M = 1
    for i in range(K):
        for j in range(i + 1, K):
            # If z_i and z_j are both 1, then d > r_i + r_j
            # This is equivalent to: d >= r_i + r_j - M*(2 - z_i - z_j)
            constrs.append(d_kl[i, j] >= r[i] + r[j] - M * (2 - z[i] - z[j]))
    
    # Objective
    obj = cp.Maximize(cp.sum(r))
    
    # Return a function that solves the LP relaxation given z_fixed
    def solve(z_fixed):
        # Fix z variables
        prob = cp.Problem(obj, constrs)
        for i in range(K):
            if z_fixed[i] is not None:
                z[i].value = z_fixed[i]
            else:
                z[i].value = None
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                sol_r = r.value
                sol_z = z.value
                bound = prob.value
                return sol_r, sol_z, bound
            else:
                return None, None, float('-inf')
        except:
            return None, None, float('-inf')
    
    return len(P), z, r, w, d_kl, solve

def branch_and_bound(milp_builder, node_limit):
    """Custom branch-and-bound solver."""
    K, z_vars, r_vars, w, d_kl, solve_fn = milp_builder
    
    # Priority queue of nodes: (bound, z_fixed, depth)
    import heapq
    queue = []
    
    # A simple check whether solution is valid
    def is_feasible(z_fixed):
        nonzeros = sum(1 for val in z_fixed if val is not None and val > 0.5)
        return nonzeros == 26
    
    # Initialize
    heapq.heappush(queue, (float('inf'), [None]*K, 0))
    best_obj = float('-inf')
    best_z = None
    best_r = None
    nodes_processed = 0
    
    while queue and nodes_processed < node_limit:
        _, z_fixed, depth = heapq.heappop(queue)
        nodes_processed += 1
        sol_r, sol_z, bound = solve_fn(z_fixed)
        if sol_z is None:
            continue
        if bound <= best_obj:
            continue
        if is_feasible(z_fixed):
            if bound > best_obj:
                best_obj = bound
                best_z = sol_z
                best_r = sol_r
        else:
            # Find the first unfixed variable that is fractional
            unfixed = []
            for i in range(K):
                if z_fixed[i] is None:
                    unfixed.append(i)
            if unfixed:
                # Branch on the one closest to 0.5 (for best exploration of the search space)
                best_index = max(unfixed, key=lambda i: abs(z_vars[i].value - 0.5) if z_vars[i].value is not None else 0.5)
                # Branch with 0
                z_n0 = list(z_fixed)
                z_n0[best_index] = 0
                heapq.heappush(queue, (bound, z_n0, depth + 1))
                # Branch with 1
                z_n1 = list(z_fixed)
                z_n1[best_index] = 1
                heapq.heappush(queue, (bound, z_n1, depth + 1))
    
    return best_z, best_r

def refine_solution(centers, radii):
    """Perform local refinement using SLSQP."""
    K = len(centers)
    
    # Define the objective and constraints for refinement
    def obj(x):
        # Reconstruct centers and radii
        r_vals = x[K*2:K*2+K]
        return -np.sum(r_vals)  # Negative because we minimize
    
    def con_eq(x):
        return np.sum(x[:K*2:2]) - 1  # This constraint is artificial; fix later
    
    def con_ineq(x):
        c_vals = x[:K*2]
        r_vals = x[K*2:K*2+K]
        # Wall constraints
        out = []
        for i in range(K):
            y = c_vals[2*i+1]
            x_c = c_vals[2*i]
            r = r_vals[i]
            out.append(r - x_c)  # x <= 1 - r -> r + x - 1 <= 0
            out.append(r - (1 - x_c))
            out.append(r - y)      # y <= 1 - r -> r + y - 1 <= 0
            out.append(r - (1 - y))
    
        # Non-overlap constraints
        for i in range(K):
            for j in range(i + 1, K):
                x_i = c_vals[2*i]
                y_i = c_vals[2*i+1]
                x_j = c_vals[2*j]
                y_j = c_vals[2*j+1]
                r_i = r_vals[i]
                r_j = r_vals[j]
                # (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
                dist_sq = (x_i - x_j)**2 + (y_i - y_j)**2
                diff = (r_i + r_j)**2 - dist_sq
                out.append(diff)
        return np.array(out)
    
    # Initialize
    x0 = np.zeros(K * 2 + K)
    x0[:K*2:2] = centers[:, 0]  # x coords
    x0[1:K*2:2] = centers[:, 1]  # y coords
    x0[K*2:] = radii
    
    # Minimize with boundary
    bounds = []
    for i in range(K):
        bounds.extend([(0, 1), (0, 1)])  # x, y in [0,1]
    for i in range(K):
        bounds.append((0, 0.5))  # radii bounded by 0.5
    
    # Solve
    res = minimize(obj, x0, method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': con_ineq},
                   options={'ftol': 1e-9, 'maxiter': 2000})
    
    if not res.success:
        # Possibly try again with jittered starting point or restart strategy
        pass
    
    final_centers = np.vstack((res.x[:K*2:2], res.x[1:K*2:2])).T
    final_radii = res.x[K*2:]
    final_sum = -res.fun
    
    return final_centers, final_radii, final_sum
