import numpy as np
from scipy.spatial.distance import cdist

def penalty(z, alpha=10.0, beta=10.0):
    n = 26
    centers = z[:2*n].reshape(n, 2)
    radii = z[2*n:]
    
    # Wall penalties
    walls = np.array([centers[:, 0], 1 - centers[:, 0], centers[:, 1], 1 - centers[:, 1]]).T
    min_wall_dist = np.min(walls, axis=1)
    wall_penalty = np.sum(np.maximum(0, radii - min_wall_dist))
    
    # Overlap penalties
    dist_matrix = cdist(centers, centers)
    ri_rj = radii[:, None] + radii[None, :]
    overlap_matrix = ri_rj - dist_matrix
    overlap_penalty = np.sum(np.maximum(0, -overlap_matrix * (1 - np.eye(n))))
    
    return alpha * wall_penalty + beta * overlap_penalty

def energy(z, lam, alpha=10.0, beta=10.0):
    sum_radii = np.sum(z[2*26:])
    return -sum_radii + lam * penalty(z, alpha, beta)

def propose_move(z, sigma_c, sigma_r, T=1.0):
    z_new = z.copy()
    move_type = np.random.choice(3)
    
    if move_type == 0:  # Center jitter
        i = np.random.randint(0, 26)
        z_new[2*i] += np.random.normal(0, sigma_c * np.sqrt(T))
        z_new[2*i+1] += np.random.normal(0, sigma_c * np.sqrt(T))
        z_new[2*i] = np.clip(z_new[2*i], 0, 1)
        z_new[2*i+1] = np.clip(z_new[2*i+1], 0, 1)
        
    elif move_type == 1:  # Radius perturb
        i = np.random.randint(0, 26)
        z_new[2*26 + i] += np.random.normal(0, sigma_r * np.sqrt(T))
        z_new[2*26 + i] = np.maximum(0, z_new[2*26 + i])
        
    else:  # Swap pair
        i, j = np.random.choice(26, 2, replace=False)
        r_i = z_new[2*26 + i]
        z_new[2*26 + i] = z_new[2*26 + j]
        z_new[2*26 + j] = r_i
    
    return z_new

def anneal():
    n = 26
    # Initial configuration on hexagonal grid
    grid_spacing = 0.18
    x = np.arange(0.05, 1.0, grid_spacing)
    y = np.arange(0.05, 1.0, grid_spacing)
    centers = np.array([[xi, yi] for yi in y for xi in x if (xi**2 + yi**2) < 0.99 and (xi**2 + (1-yi)**2) < 0.99 and ((1-xi)**2 + yi**2) < 0.99 and ((1-xi)**2 + (1-yi)**2) < 0.99][:26])
    if len(centers) < 26:
        # Fill up using random points
        for _ in range(26 - len(centers)):
            x = np.random.uniform(0.05, 0.95)
            y = np.random.uniform(0.05, 0.95)
            centers = np.vstack([centers, [x, y]])
    centers = centers[:26]
    
    # Assign initial radii
    radii = np.array([np.min([c[0], 1-c[0], c[1], 1-c[1]]) for c in centers])
    
    z = np.hstack([centers.flatten(), radii])
    
    # Annealing parameters
    T = 1.0
    T_min = 1e-5
    c_T = 0.995
    lam = 0.1
    c_lam = 1.03
    max_iter = 10000
    sigma_c = 0.03
    sigma_r = 0.02
    
    best_feasible = z.copy()
    best_feasible_energy = float('inf')
    best_energy = float('inf')
    
    for iteration in range(max_iter):
        # Make a move
        z_new = propose_move(z, sigma_c, sigma_r, T)
        delta_E = energy(z_new, lam) - energy(z, lam)
        
        if delta_E <= 0 or np.random.rand() < np.exp(-delta_E / T):
            z = z_new
        
        # Update best feasible
        p = penalty(z)
        if p < 1e-8:
            if energy(z, lam) < best_feasible_energy:
                best_feasible = z.copy()
                best_feasible_energy = energy(z, lam)
        
        # Update best overall
        if energy(z, lam) < best_energy:
            best_energy = energy(z, lam)
        
        # Update temperature and penalty weight
        T *= c_T
        if T < T_min:
            break
        lam *= c_lam
    
    return best_feasible[:2*n].reshape(n, 2), best_feasible[2*n:]

def project_and_ascent(centers, radii, max_iter=500, eta=5e-3, decay=0.95):
    n = len(centers)
    z = np.hstack([centers.flatten(), radii])
    
    for i in range(max_iter):
        # Gradient: partial derivative of sum of radii
        grad = np.zeros_like(z)
        grad[2*n:] = 1.0
        
        # Gradient step
        z_old = z.copy()
        z += eta * grad
        
        # Project to feasible domain
        centers = z[:2*n].reshape(n, 2)
        radii = z[2*n:]
        
        # Wall clamping
        for j in range(n):
            max_r_wall = min(centers[j, 0], 1 - centers[j, 0], centers[j, 1], 1 - centers[j, 1])
            radii[j] = min(radii[j], max_r_wall)
        
        # Overlap resolution
        dist_matrix = cdist(centers, centers)
        ri_rj = radii[:, None] + radii[None, :]
        overlap_matrix = ri_rj - dist_matrix
        overlaps = np.triu(overlap_matrix, k=+1)
        overlap_indices = np.where(overlaps > 1e-12)
        
        for k in range(len(overlap_indices[0])):
            i_idx, j_idx = overlap_indices[0][k], overlap_indices[1][k]
            overlap = overlaps[i_idx, j_idx]
            if overlap > 0:
                # Reduce both radii proportionally
                radii[i_idx] -= overlap / 2
                radii[j_idx] -= overlap / 2
                # Ensure non-negative
                radii[i_idx] = max(0, radii[i_idx])
                radii[j_idx] = max(0, radii[j_idx])
        
        # Re-clamp to walls
        for j in range(n):
            max_r_wall = min(centers[j, 0], 1 - centers[j, 0], centers[j, 1], 1 - centers[j, 1])
            radii[j] = min(radii[j], max_r_wall)
        
        z = np.hstack([centers.flatten(), radii])
        
        # Check convergence
        diff = np.sum(np.abs(z - z_old))
        if diff < 1e-7:
            break
        eta *= decay
    
    return z[:2*n].reshape(n, 2), z[2*n:]
