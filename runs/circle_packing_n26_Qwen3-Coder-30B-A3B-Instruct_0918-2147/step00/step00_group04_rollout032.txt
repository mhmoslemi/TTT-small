```python
import numpy as np
from scipy.optimize import minimize

NUM_CIRCLES = 26
NUM_RESTARTS = 5
LATTICESpacing = 0.25
INITIAL_RADIUS = 0.02
BOUNDARY_MARGIN = 1e-9

def hex_lattice_points(n_rows, spacing):
    points = []
    for i in range(n_rows):
        y = i * spacing * np.sqrt(3) / 2
        if y > 1:
            break
        offset = (i % 2) * spacing / 2
        n_cols = int(np.ceil(1.0 / spacing)) + 1
        for j in range(n_cols):
            x = offset + j * spacing
            if x > 1:
                break
            points.append([x, y])
    return np.array(points)

def initial_guess(seed):
    np.random.seed(seed)
    points = hex_lattice_points(10, LATTICESpacing)
    if points.shape[0] < NUM_CIRCLES:
        raise ValueError("Not enough lattice points for 26 circles")
    selected_points = points[:NUM_CIRCLES]
    # Add jitter
    jitter = np.random.uniform(-0.02, 0.02, (NUM_CIRCLES, 2))
    centers = selected_points + jitter
    # Ensure centers within bounds
    centers = np.clip(centers, 0, 1)
    radii = np.full(NUM_CIRCLES, INITIAL_RADIUS)
    vars0 = np.empty(3 * NUM_CIRCLES)
    vars0[0::3] = centers[:, 0]
    vars0[1::3] = centers[:, 1]
    vars0[2::3] = radii
    return vars0

def objective(vars):
    return -np.sum(vars[2::3])

def constraint_boundary(vars):
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    # boundary constraints: x - r, 1 - x - r, y - r, 1 - y - r
    boundaries = np.column_stack([
        x - r,
        1 - x - r,
        y - r,
        1 - y - r
    ])
    return boundaries.reshape(-1)

def constraint_nonoverlap(vars):
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    n = len(x)
    nonoverlap = []
    for i in range(n):
        for j in range(i + 1, n):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dist_sq = dx * dx + dy * dy
            r_sum = r[i] + r[j]
            nonoverlap.append(np.sqrt(dist_sq) - r_sum)
    return np.array(nonoverlap)

def postprocess(vars):
    x = vars[0::3].copy()
    y = vars[1::3].copy()
    r = vars[2::3].copy()
    
    # Update centers to ensure they're inside the square
    for i in range(len(x)):
        x[i] = np.clip(x[i], r[i], 1 - r[i])
        y[i] = np.clip(y[i], r[i], 1 - r[i])

    # Handle overlap
    # For simplicity, use a greedy fix
    # Go through all pairs and adjust if needed
    # We don't need to adjust too much since numerical errors are small
    # Let's just check all pairs once more and apply small corrections
    for i in range(len(x)):
        for j in range(i + 1, len(x)):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dist = np.sqrt(dx * dx + dy * dy)
            r_sum = r[i] + r[j]
            if dist < r_sum:
                # Move the circles apart a bit
                overlap = r_sum - dist
                # Move center of larger circle toward the smaller one
                if r[i] >= r[j]:
                    # Move i's center a bit
                    move_dist = overlap * 0.5
                    dir_x = -dx / max(1e-8, dist)
                    dir_y = -dy / max(1e-8, dist)
                    x[i] += move_dist * dir_x
                    y[i] += move_dist * dir_y
                else:
                    # Move j's center
                    move_dist = overlap * 0.5
                    dir_x = dx / max(1e-8, dist)
                    dir_y = dy / max(1e-8, dist)
                    x[j] += move_dist * dir_x
                    y[j] += move_dist * dir_y
                # Re-adjust so we don't go out of bounds
                x[i] = np.clip(x[i], r[i], 1 - r[i])
                y[i] = np.clip(y[i], r[i], 1 - r[i])
                x[j] = np.clip(x[j], r[j], 1 - r[j])
                y[j] = np.clip(y[j], r[j], 1 - r[j])
    
    # Re-update vars
    new_vars = vars.copy()
    new_vars[0::3] = x
    new_vars[1::3] = y
    new_vars[2::3] = r
    
    return new_vars

def run_packing():
    best_sum = -np.inf
    best_vars = None

    # Define constraints
    global_constraints = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_nonoverlap}
    ]

    bounds = []
    for i in range(NUM_CIRCLES):
        # x, y, r
        bounds.append((0, 1))
        bounds.append((0, 1))
        bounds.append((1e-6, 0.5))  # r should be positive and reasonable


    for seed in range(NUM_RESTARTS):
        vars0 = initial_guess(seed)
        res = minimize(objective, vars0,
                       method='SLSQP',
                       bounds=bounds,
                       constraints=global_constraints,
                       options={'maxiter': 3000, 'ftol': 1e-7})
        
        if not res.success:
            continue

        vars_corr = postprocess(res.x)
        s = np.sum(vars_corr[2::3])
        if s > best_sum:
            best_sum = s
            best_vars = vars_corr

    if best_vars is None:
        raise RuntimeError("Optimization failed to find a feasible solution.")

    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii = best_vars[2::3]
    return centers, radii, float(best_sum)
```
