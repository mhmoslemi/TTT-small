import numpy as np
from scipy.optimize import linprog
from math import sqrt

def initial_hex_centers(k, spacing):
    """Generate k centers from a hexagonal lattice scaled to unit square"""
    sqrt3 = sqrt(3)
    rows = []
    i0 = 0
    while len(rows) < k:
        for j in range(int(1.0 / spacing) + 2):
            x = i0 * spacing + (j % 2) * spacing / 2
            y = j * spacing * sqrt3 / 2
            if 0 <= x <= 1 and 0 <= y <= 1:
                rows.append((x, y))
                if len(rows) == k:
                    break
        i0 += 1
    return np.array(rows[:k])

def initial_radii(k, r0):
    """Return k radii, all set to r0"""
    return np.full(k, r0)

def build_lp_constants(centers, radii):
    """Compute wall distances and pairwise distances"""
    n = centers.shape[0]
    d_left = centers[:, 0]
    d_right = 1.0 - centers[:, 0]
    d_bottom = centers[:, 1]
    d_top = 1.0 - centers[:, 1]
    wall_distances = np.stack([d_left, d_right, d_bottom, d_top], axis=1)
    
    # Pairwise distances
    D = np.zeros((n, n))
    for i in range(n):
        for j in range(i+1, n):
            diff = centers[i] - centers[j]
            dist = sqrt(np.sum(diff ** 2))
            D[i, j] = dist
            D[j, i] = dist
    return wall_distances, D

def solve_radius_lp(consts, radii, graph, alpha):
    """Solve linear program to allocate radii increments"""
    # Build matrix constraints for linear programming
    n = len(radii)
    wall_dists, D = consts
    
    # Variables Δr_i
    # Objective: maximize Σ Δr_i
    c = np.ones(n)
    
    # Constraints:
    # 1. Wall bounds: Δr_i ≤ d_i_wall - r_i
    A_ub = []
    b_ub = []
    
    for i in range(n):
        # Add one constraint per wall
        for w in range(4):  # left, right, bottom, top
            d = wall_dists[i, w] - radii[i]
            a_row = np.zeros(n)
            a_row[i] = -1.0  # -Δr_i ≤ d - r_i
            A_ub.append(a_row)
            b_ub.append(-d)
    
    # 2. Pairwise bounds: Δr_i + Δr_j ≤ D_ij - (r_i + r_j)
    for i in range(n):
        for j in range(i+1, n):
            a_row = np.zeros(n)
            a_row[i] = 1.0
            a_row[j] = 1.0
            A_ub.append(a_row)
            b_ub.append(D[i, j] - (radii[i] + radii[j]))
    
    # If graph is given and using α, add extra capacity constraints
    # Optional: for better balance, limit how much a circle can receive from neighbors
    # This doesn't significantly alter the structure.
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'presolve': True})
    if res.success:
        delta_r = res.x
    else:
        # Return zero increments if infeasible
        delta_r = np.zeros(n)
    return delta_r

def apply_increment(radii, delta_r):
    """Add Δr to current radii"""
    return radii + delta_r

def push_feasibility(centers, radii):
    """Apply hard geometric projection to restore feasibility"""
    # Make a copy to not mutate originals
    new_centers = centers.copy()
    new_radii = radii.copy()
    n = len(radii)
    
    # Wall check and correction
    for i in range(n):
        x, y = new_centers[i]
        r = new_radii[i]
        # Clamps
        if x - r < 0:
            new_radii[i] = x
        if x + r > 1:
            new_radii[i] = 1 - x
        if y - r < 0:
            new_radii[i] = y
        if y + r > 1:
            new_radii[i] = 1 - y
    
    # Pairwise overlaps
    updated_centers = new_centers.copy()
    updated_radii = new_radii.copy()
    
    changed = True
    while changed:
        changed = False
        for i in range(n):
            for j in range(i+1, n):
                diff = updated_centers[i] - updated_centers[j]
                dist = sqrt(np.sum(diff ** 2))
                r_sum = updated_radii[i] + updated_radii[j]
                if dist < r_sum:
                    delta = r_sum - dist
                    unit_vector = diff / dist
                    delta_vec = (delta / 2) * unit_vector
                    updated_centers[i] -= delta_vec
                    updated_centers[j] += delta_vec
                    changed = True
    
    for i in range(n):
        # Handle boundary constraints after movement
        x, y = updated_centers[i]
        r = updated_radii[i]
        if x - r < 0:
            updated_radii[i] = x
        if x + r > 1:
            updated_radii[i] = 1 - x
        if y - r < 0:
            updated_radii[i] = y
        if y + r > 1:
            updated_radii[i] = 1 - y
    
    return updated_centers, updated_radii

def run_iterations(centers, radii, max_iter=200, ε=1e-5, α=0.5):
    """Run the main optimization loop"""
    graph = {}
    for i in range(len(centers)):
        graph[i] = set()
    
    for iteration in range(max_iter):
        # Build LP constants
        wall_dists, D = build_lp_constants(centers, radii)
        
        # Solve LP
        delta_r = solve_radius_lp((wall_dists, D), radii, graph, α)
        
        # Add to radii
        new_radii = apply_increment(radii, delta_r)
        
        # Calculate sum delta
        delta_sum = np.sum(delta_r)
        if delta_sum < ε:
            break
        
        # Update radii
        radii = new_radii
        
        # Modify centers slightly to enforce non-overlap
        centers, radii = push_feasibility(centers, radii)
    
    return centers, radii

def final_refinement(centers, radii, ε=1e-12):
    """Do one final high-precision LP and clipping to pass strict validator"""
    wall_dists, D = build_lp_constants(centers, radii)
    delta_r = solve_radius_lp((wall_dists, D), radii, {}, 0.5)
    final_radii = apply_increment(radii, delta_r)
    
    # Clipping
    n = len(final_radii)
    for i in range(n):
        r = final_radii[i]
        x, y = centers[i]
        # Box constraints
        r = min(r, x, 1 - x, y, 1 - y)
        final_radii[i] = max(0, r)
    
    # Pairwise clipping
    for i in range(n):
        for j in range(i+1, n):
            diff = centers[i] - centers[j]
            dist = sqrt(np.sum(diff ** 2))
            r_sum = final_radii[i] + final_radii[j]
            if dist < r_sum:
                delta = r_sum - dist
                final_radii[i] -= delta / 2
                final_radii[j] -= delta / 2
    
    return centers, final_radii

def run_packing():
    # Seed configuration
    centers = initial_hex_centers(26, 0.12)
    radii = initial_radii(26, 0.005)
    
    # Run iteration loop
    centers, radii = run_iterations(centers, radii, max_iter=100, ε=1e-5, α=0.5)
    
    # Final refinement for validator compliance
    centers, radii = final_refinement(centers, radii)
    
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
