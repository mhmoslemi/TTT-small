import numpy as np
from scipy.optimize import minimize

def make_hex_lattice(n, margin=0.05):
    """Generate an approximate hexagonal lattice of n points within a square with margin."""
    # Estimate spacing for about n circles in unit square
    sqrt_n = np.ceil(np.sqrt(n))
    dx = (1 - 2 * margin) / (sqrt_n + 0.5)
    dy = dx * np.sqrt(3) / 2
    points = []
    for j in range(int(np.ceil(n / (sqrt_n + 0.5)))):
        for i in range(int(np.ceil(sqrt_n + 0.5))):
            x = margin + (i + (j % 2) * 0.5) * dx
            y = margin + j * dy
            if len(points) >= n:
                break
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append([x, y])
        if len(points) >= n:
            break
    points = np.array(points[:n])
    # Center the pattern
    center = np.mean(points, axis=0)
    offset = 0.5 - center
    points += offset
    return points

def objective(z):
    """Objective function: negative sum of radii"""
    n = len(z) // 3
    radii = z[2::3]
    return -np.sum(radii)

def jacobian_objective(z):
    """Jacobian of objective function"""
    n = len(z) // 3
    grad = np.zeros_like(z)
    grad[2::3] = -1.0
    return grad

def boundary_constraints(z):
    """Return array of boundary constraint values (should be >= 0)"""
    n = len(z) // 3
    constraints = []
    for i in range(n):
        x, y, r = z[3*i], z[3*i+1], z[3*i+2]
        constraints.extend([
            x - r,                     # x - r >= 0
            1 - x - r,                 # 1 - x - r >= 0
            y - r,                     # y - r >= 0
            1 - y - r                  # 1 - y - r >= 0
        ])
    return np.array(constraints)

def jacobian_boundary(z):
    """Jacobian of boundary constraints (shape (4*n, 3*n))"""
    n = len(z) // 3
    jac = np.zeros((4*n, 3*n))
    for i in range(n):
        jac[4*i  , 3*i  ] = 1.0  # d(x-r)/dx = 1
        jac[4*i  , 3*i+2] = -1.0 # d(x-r)/dr = -1
        jac[4*i+1, 3*i  ] = -1.0 # d(1-x-r)/dx = -1
        jac[4*i+1, 3*i+2] = -1.0 # d(1-x-r)/dr = -1
        jac[4*i+2, 3*i+1] = 1.0  # d(y-r)/dy = 1
        jac[4*i+2, 3*i+2] = -1.0 # d(y-r)/dr = -1
        jac[4*i+3, 3*i+1] = -1.0 # d(1-y-r)/dy = -1
        jac[4*i+3, 3*i+2] = -1.0 # d(1-y-r)/dr = -1
    return jac

def pairwise_constraints(z):
    """Return array of pairwise separation constraint values (should be >= 0)"""
    n = len(z) // 3
    constraints = []
    for i in range(n):
        for j in range(i+1, n):
            x1, y1, r1 = z[3*i], z[3*i+1], z[3*i+2]
            x2, y2, r2 = z[3*j], z[3*j+1], z[3*j+2]
            dist_sq = (x1 - x2)**2 + (y1 - y2)**2
            sum_radii_sq = (r1 + r2)**2
            constraints.append(dist_sq - sum_radii_sq)
    return np.array(constraints)

def jacobian_pairwise(z):
    """Jacobian of pairwise constraints (shape (n*(n-1)/2, 3*n))"""
    n = len(z) // 3
    jac = np.zeros((n*(n-1)//2, 3*n))
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            x1, y1, r1 = z[3*i], z[3*i+1], z[3*i+2]
            x2, y2, r2 = z[3*j], z[3*j+1], z[3*j+2]
            # i part
            jac[idx, 3*i  ] = 2 * (x1 - x2)   # d(dist_sq)/dx1
            jac[idx, 3*i+1] = 2 * (y1 - y2)   # d(dist_sq)/dy1
            jac[idx, 3*i+2] = 2 * (r1 + r2)   # d(sum_radii_sq)/dr1
            # j part
            jac[idx, 3*j  ] = -2 * (x1 - x2)  # d(dist_sq)/dx2
            jac[idx, 3*j+1] = -2 * (y1 - y2)  # d(dist_sq)/dy2
            jac[idx, 3*j+2] = 2 * (r1 + r2)   # d(sum_radii_sq)/dr2
            idx += 1
    return jac

def solve_nlp(initial_z):
    """Solve the NLP using SLSQP"""
    n = len(initial_z) // 3

    # Define bounds
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0))   # x
        bounds.append((0.0, 1.0))   # y
        bounds.append((0.0, 0.5))   # r

    # Constraints
    constraints = [
        {'type': 'ineq', 'fun': boundary_constraints, 'jac': jacobian_boundary},
        {'type': 'ineq', 'fun': pairwise_constraints, 'jac': jacobian_pairwise}
    ]

    # Optimize
    result = minimize(
        objective,
        initial_z,
        method='SLSQP',
        jac=jacobian_objective,
        bounds=bounds,
        constraints=constraints,
        options={'ftol': 1e-9, 'maxiter': 1000},
    )
    if not result.success:
        raise RuntimeError("Optimization failed: " + result.message)
    return result.x

def nudge_centers(centers, radii):
    """Nudge circles that are tight against a wall away slightly"""
    epsilon = 0.001
    new_centers = centers.copy()
    for i in range(len(centers)):
        x, y = centers[i]
        r = radii[i]
        # Check if circle is near a wall
        if x < r + 1e-6:
            new_centers[i, 0] += epsilon
        elif x > 1 - r - 1e-6:
            new_centers[i, 0] -= epsilon
        if y < r + 1e-6:
            new_centers[i, 1] += epsilon
        elif y > 1 - r - 1e-6:
            new_centers[i, 1] -= epsilon
    return new_centers

    centers = make_hex_lattice(26, margin=0.05)

    z = np.concatenate([centers.flatten(), radii])

        new_radii = z[2::3]
        new_sum = np.sum(new_radii)
        if abs(new_sum - prev_sum) < 1e-5:
            break
        prev_sum = new_sum
        # Nudge potential tight circles
        new_centers = nudge_centers(new_centers, new_radii)
        z[0::3] = new_centers.flatten()
        z[1::3] = new_centers[:, 1]
        z[2::3] = new_radii

    # Final valid check
