import numpy as np
from scipy.optimize import linprog
from scipy.spatial.distance import cdist

def generate_hex_lattice(h):
    sqrt3 = np.sqrt(3)
    rows = int(np.ceil(1 / (h * sqrt3 / 2))) + 1
    cols = int(np.ceil(1 / h)) + 1
    centers = []
    for j in range(rows):
        y = j * h * sqrt3 / 2
        if y > 1:
            break
        for i in range(cols):
            x = i * h + (j % 2) * (h / 2)
            if x > 1:
                break
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers.append([x, y])
    centers = np.array(centers)
    R_wall = np.minimum(centers[:, 0], 1 - centers[:, 0])
    R_wall = np.minimum(R_wall, centers[:, 1])
    R_wall = np.minimum(R_wall, 1 - centers[:, 1])
    return centers, R_wall

def build_neighbour_list(centers, max_dist=0.4):
    distances = cdist(centers, centers)
    edges = []
    n = centers.shape[0]
    for i in range(n):
        for j in range(i + 1, n):
            d = distances[i, j]
            if d < max_dist:
                edges.append((i, j, d))
    return edges

def solve_radius_lp(selected_indices, R_wall, neighbour_list):
    n_selected = len(selected_indices)
    if n_selected == 0:
        return np.array([]), True

    # Build LP constraints
    A_ub = []
    b_ub = []
    c = np.ones(n_selected)

    # Wall constraints: r_i <= R_wall[i]
    for i in range(n_selected):
        A_ub.append([0] * i + [-1] + [0] * (n_selected - i - 1))
        b_ub.append(-R_wall[selected_indices[i]])

    # Neighbour separation constraints: r_i + r_j <= d_ij
    for i, j, d_ij in neighbour_list:
        if i in selected_indices and j in selected_indices:
            idx_i = selected_indices.index(i)
            idx_j = selected_indices.index(j)
            row = [0] * n_selected
            row[idx_i] = 1
            row[idx_j] = 1
            A_ub.append(row)
            b_ub.append(d_ij)

    # Solve
    try:
        res = linprog(-c, A_ub=A_ub, b_ub=b_ub, method='highs')
        if res.success:
            return res.x, True
        else:
            return np.array([]), False
    except:
        return np.array([]), False

def upper_bound(partial_set, remaining_indices, R_wall):
    ub = 0
    for i in partial_set:
        ub += R_wall[i]
    if len(remaining_indices) > 0:
        for i in remaining_indices:
            ub += R_wall[i]
    return ub

def branch_and_bound(centers, R_wall, neighbour_list, n_circles=26):
    best_sum = -np.inf
    best_solution = None

    def search(S_partial, U, lb):
        nonlocal best_sum, best_solution
        if len(S_partial) == n_circles:
            radii, feasible = solve_radius_lp(S_partial, R_wall, neighbour_list)
            if feasible and len(radii) > 0:
                total = np.sum(radii)
                if total > best_sum:
                    best_sum = total
                    best_solution = (S_partial.copy(), radii.copy())
            return

        if len(S_partial) + len(U) < n_circles:
            return

        # Upper bound
        ub = upper_bound(S_partial, U, R_wall)
        if ub <= lb:
            return

        # Select next candidate with largest wall radius
        if not U:
            return
        max_r_idx = max(U, key=lambda x: R_wall[x])
        U_new = set(U)
        U_new.remove(max_r_idx)
        S_new_partial = S_partial + [max_r_idx]

        # Include
        search(S_new_partial, U_new, lb)
        # Exclude
        search(S_partial, U_new, lb)

    # Start
    U_all = set(range(len(centers)))
    search([], U_all, -np.inf)
    if best_solution is not None:
        selected_indices, radii = best_solution
        return selected_indices, radii
    else:
        return [], np.array([])

def local_continuous_refinement(centers, radii):
    # Refine using a small NLP
    from scipy.optimize import minimize
    n = centers.shape[0]
    # Variables: [center_x0, center_y0, ..., center_x25, center_y25, r0, ..., r25]
    # The problem is only to prevent overlap and wall violations.
    # Plasticine Method: Using a small trust-constr optimization
    # Optimize directly using approximated gradient/linesearch

    # We use SLSQP for a light refinement of the discrete solution
    # Objective: maximize sum of radii
    def obj_minus_sum_radii(x):
        # x = [cx0, cy0, ..., cx25, cy25, r0, ..., r25]
        centers_flat = x[:2*n].reshape((n,2))
        radii_vec = x[2*n:]
        return -np.sum(radii_vec)

    # Constraints
    def boundary_constraint(x):
        centers_flat = x[:2*n].reshape((n,2))
        radii_vec = x[2*n:]
        viol = []
        for i in range(n):
            cx, cy = centers_flat[i]
            r = radii_vec[i]
            if cx - r < 0 or cx + r > 1 or cy - r < 0 or cy + r > 1:
                # Return large value for infeasible (positive)
                viol.append(1000)
            else:
                viol.append(0.0)
        return np.array(viol)

    def overlap_constraint(x):
        centers_flat = x[:2*n].reshape((n,2))
        radii_vec = x[2*n:]
        viol = []
        for i in range(n):
            for j in range(i+1, n):
                dist = np.sqrt(np.sum((centers_flat[i] - centers_flat[j])**2))
                r_sum = radii_vec[i] + radii_vec[j]
                viol.append(max(0.0, r_sum - dist))
        return np.array(viol)

    # Initial guess from discrete centers (slightly jittered) and radii
    cons = []
    cons.append({'type': 'ineq', 'fun': boundary_constraint})
    cons.append({'type': 'ineq', 'fun': lambda x: -overlap_constraint(x)})  # cares about margin
    bounds = []
    # Bounds for centers: [0, 1]
    for _ in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0)])
    # Bounds for radii: [0, 1]
    for _ in range(n):
        bounds.append((1e-6, 1.0))

    # Start with fine parameters
    x_start = np.hstack((centers.flatten(), radii))
    x_start += np.random.uniform(-0.002, 0.002, x_start.shape)
    # Ensure radii are still feasible and centered properly
    for i in range(n):
        r = radii[i]
        cx, cy = centers[i]
        x_start[2*n+i] = r

    # Optimize
    res = minimize(obj_minus_sum_radii, x_start, method='trust-constr', bounds=bounds,
                   constraints=cons, options={'maxiter': 200})
    if res.success:
        refined_centers = res.x[:2*n].reshape((n,2))
        refined_radii = res.x[2*n:]
        return refined_centers, refined_radii

    # default fallback to input
    return centers, radii

def run_packing():
    h = 0.07
    centers, R_wall = generate_hex_lattice(h)
    neighbour_list = build_neighbour_list(centers)

    selected_indices, radii = branch_and_bound(centers, R_wall, neighbour_list, 26)
    if len(selected_indices) < 26:
        raise RuntimeError("Branch and bound failed to find meaningful solution")

    final_centers = centers[selected_indices]
    final_radii = radii

    # Apply local refinement
    final_centers, final_radii = local_continuous_refinement(final_centers, final_radii)

    # Validate
    valid, msg = validate_packing(final_centers, final_radii)
    if not valid:
        raise RuntimeError("Final packing invalid: " + msg)

    # Try to slightly shrink if needed
    for _ in range(5):
        valid, msg = validate_packing(final_centers, final_radii)
        if valid:
            break
        # Shrink all slightly
        final_radii *= (1.0 - 1e-7)
        # Ensure no negative
        final_radii = np.maximum(final_radii, 1e-12)

    sum_radii = float(np.sum(final_radii))
    return final_centers, final_radii, sum_radii
