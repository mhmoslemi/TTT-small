import numpy as np
from scipy.optimize import minimize

def make_hex_lattice(n, margin=0.05):
    # Generate hexagonal lattice points in unit square
    # spacing based on hexagonal packing
    dx = 1.0 / (np.ceil(np.sqrt(n)) + 0.5)
    dy = dx * np.sqrt(3) / 2.0
    points = []
    for j in range(int(np.ceil(n / 0.9)), -1, -1):
        for i in range(int(np.ceil(n / 0.9))):
            x = i * dx + (j % 2) * dx / 2.0
            y = j * dy
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append([x, y])
            if len(points) >= n:
                break
        if len(points) >= n:
            break
    # Trim to exactly n points
    points = points[:n]
    # Center and add margin
    points = np.array(points)
    cx = np.mean(points[:, 0])
    cy = np.mean(points[:, 1])
    points[:, 0] = (points[:, 0] - cx) * 0.9 + 0.5
    points[:, 1] = (points[:, 1] - cy) * 0.9 + 0.5
    # Apply margin to ensure we have space for radii
    points[:, 0] = np.clip(points[:, 0], margin, 1 - margin)
    points[:, 1] = np.clip(points[:, 1], margin, 1 - margin)
    return points

def objective(z):
    # z is [x0, y0, r0, x1, y1, r1, ...]
    n = len(z) // 3
    radii = z[2::3]
    return -np.sum(radii)

def boundary_constraints(z):
    # Return array of 4n values for each circle:
    # x - r >= 0, x + r <= 1, y - r >= 0, y + r <= 1
    n = len(z) // 3
    constraints = []
    for i in range(n):
        x = z[3*i]
        y = z[3*i + 1]
        r = z[3*i + 2]
        constraints.extend([
            x - r,              # x - r >= 0
            1 - x - r,          # x + r <= 1
            y - r,              # y - r >= 0
            1 - y - r           # y + r <= 1
        ])
    return np.array(constraints)

def pairwise_constraints(z):
    # Return array of pairwise constraints
    n = len(z) // 3
    constraints = []
    for i in range(n):
        for j in range(i+1, n):
            xi, yi, ri = z[3*i], z[3*i+1], z[3*i+2]
            xj, yj, rj = z[3*j], z[3*j+1], z[3*j+2]
            dist_sq = (xi - xj)**2 + (yi - yj)**2
            sum_r_sq = (ri + rj)**2
            constraints.append(dist_sq - sum_r_sq)
    return np.array(constraints)

def solve_nlp(initial_z):
    n = len(initial_z) // 3
    bounds = []
    for i in range(n):
        bounds.extend([(0, 1), (0, 1), (0, 0.5)])
    # Constraints dictionary
    cons = []
    # Boundary constraints
    cons.append({'type': 'ineq', 'fun': boundary_constraints})
    # Pairwise constraints
    cons.append({'type': 'ineq', 'fun': pairwise_constraints})
    res = minimize(
        objective, initial_z,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 1000, 'ftol': 1e-9}
    )
    return res

def run_packing():
    n = 26
    margin = 0.05
    centers = make_hex_lattice(n, margin)
    radii = np.array([0.02] * n)
    # Initial guess z: [x0, y0, r0, x1, y1, r1, ...]
    initial_z = np.zeros(3 * n)
    for i in range(n):
        initial_z[3*i] = centers[i, 0]   # x
        initial_z[3*i + 1] = centers[i, 1] # y
        initial_z[3*i + 2] = radii[i]    # r

    # Optimization loop
    for _ in range(5):
        res = solve_nlp(initial_z)
        if not res.success:
            break
        # Extract new centers and radii
        new_centers = np.zeros((n, 2))
        new_radii = np.zeros(n)
        for i in range(n):
            new_centers[i, 0] = res.x[3*i]
            new_centers[i, 1] = res.x[3*i + 1]
            new_radii[i] = res.x[3*i + 2]
        # Check if improvement is too small
        if np.sum(new_radii) - np.sum(radii) < 1e-5:
            break
        # Nudge points tight to walls to escape local minima
        for i in range(n):
            x, y, r = new_centers[i, 0], new_centers[i, 1], new_radii[i]
            # Check if it's compact to a wall
            if abs(x - r) < 1e-6 or abs(x + r - 1) < 1e-6 \
               or abs(y - r) < 1e-6 or abs(y + r - 1) < 1e-6:
                # Compute offset normal vector pointing away from wall
                nx = 0
                ny = 0
                if abs(x - r) < 1e-6:
                    nx = 1
                elif abs(x + r - 1) < 1e-6:
                    nx = -1
                elif abs(y - r) < 1e-6:
                    ny = 1
                elif abs(y + r - 1) < 1e-6:
                    ny = -1
                # To avoid biasing toward one direction, we rotate the normal
                # slightly. Ultimately, it's just an implicit tightening.
                # Do move in normal direction
                step = 0.001
                new_centers[i, 0] += step * nx
                new_centers[i, 1] += step * ny
                # Clamp to valid range
                new_centers[i, 0] = np.clip(new_centers[i, 0], r, 1 - r)
                new_centers[i, 1] = np.clip(new_centers[i, 1], r, 1 - r)
                # Adjust the corresponding variable in z
                initial_z[3*i] = new_centers[i, 0]
                initial_z[3*i + 1] = new_centers[i, 1]
        # Update for next iteration
        centers = new_centers
        radii = new_radii
        # Reconstruct z
        for i in range(n):
            initial_z[3*i] = centers[i, 0]
            initial_z[3*i+1] = centers[i, 1]
            initial_z[3*i+2] = radii[i]
