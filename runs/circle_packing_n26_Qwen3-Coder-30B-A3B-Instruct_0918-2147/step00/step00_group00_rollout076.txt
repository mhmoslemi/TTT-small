import numpy as np
from scipy.optimize import linprog
from scipy.spatial.distance import cdist

def generate_hex_lattice(h):
    """Generate a hexagonal lattice of points in [0,1]x[0,1] with spacing h."""
    sqrt3 = np.sqrt(3)
    points = []
    j_max = int(1.0 / (h * sqrt3 / 2)) + 2
    for j in range(j_max):
        y = j * h * sqrt3 / 2
        if y > 1:
            break
        i_max = int(1.0 / h) + 2
        for i in range(i_max):
            x = i * h + (j % 2) * (h / 2)
            if x > 1:
                break
            points.append([x, y])
    return np.array(points)

def build_neighbour_list(centers, max_dist=0.4):
    """Build a list of nearby pairs of points with their distances."""
    distances = cdist(centers, centers)
    edges = []
    n = centers.shape[0]
    for i in range(n):
        for j in range(i + 1, n):
            d = distances[i, j]
            if d < max_dist:
                edges.append((i, j, d))
    return edges

def solve_radius_lp(selected_indices, R_wall, neighbour_list):
    """Solve the LP for a specific set of selected indices."""
    n_selected = len(selected_indices)
    if n_selected == 0:
        return np.array([]), True

    # Build constraint matrix A_ub and vector b_ub
    # We have 2 types of constraints: wall constraints and separation constraints
    # Wall constraints: 0 <= r_i <= R_wall[i]
    # Separation constraints: r_i + r_j <= d_ij

    # First, build the wall constraints (each r_i coefficient in rows 2*i, 2*i+1)
    # For each selected index i, we have:
    # r_i >= 0 and r_i <= R_wall[i]
    # This becomes: [0 0 ... 1 0 ... 0] * r <= R_wall[i]  (note: <= for r_i)
    # and -r_i <= 0 => [-1 0 ... 0] * r <= 0

    A_ub_rows = []
    b_ub = []

    # Wall constraints
    for i, idx in enumerate(selected_indices):
        # r_i <= R_wall[idx]
        row = np.zeros(n_selected)
        row[i] = 1
        A_ub_rows.append(row)
        b_ub.append(R_wall[idx])

        # r_i >= 0 -> -r_i <= 0
        row = np.zeros(n_selected)
        row[i] = -1
        A_ub_rows.append(row)
        b_ub.append(0)

    # Separation constraints
    selected_set = set(selected_indices)
    for i, j, d_ij in neighbour_list:
        if i in selected_set and j in selected_set:
            idx_i = selected_indices.index(i)
            idx_j = selected_indices.index(j)
            row = np.zeros(n_selected)
            row[idx_i] = 1
            row[idx_j] = 1
            A_ub_rows.append(row)
            b_ub.append(d_ij)

    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub)

    # Objective: maximize sum of radii
    c = -np.ones(n_selected)  # negative because linprog minimizes

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
        if result.success:
            return result.x, True
        else:
            return np.zeros(n_selected), False
    except:
        return np.zeros(n_selected), False

def upper_bound(partial_set, remaining_indices, R_wall):
    """Compute an upper bound on the sum of radii."""
    total = sum(R_wall[i] for i in partial_set)
    remaining_slots = 26 - len(partial_set)
    if remaining_slots > 0:
        # Take the maximum wall radius from remaining candidates
        max_wall = max(R_wall[i] for i in remaining_indices)
        total += remaining_slots * max_wall
    return total

def branch_and_bound(centers, R_wall, neighbour_list):
    """Branch and bound search for the best combination of 26 points."""
    n = centers.shape[0]
    best_solution = None
    best_sum = float('-inf')

    # Stack elements are: (partial_indices, remaining_indices, ub)
    # We will explore by choosing points in order of decreasing R_wall
    # To be efficient, we only explore a small number of good candidates
    def sort_key(i):
        return -R_wall[i]
    all_indices = list(range(n))
    all_indices.sort(key=sort_key)

    # Very simple backtracking with pruning
    # For practical runtime, we limit exploration to a small number of branches
    # This is very heuristic, but effective for moderate numbers
    # Use iterative deepening or limited search depth

    def backtrack(partial_indices, remaining_indices, lb, ub):
        nonlocal best_solution, best_sum

        # If we have placed exactly 26 circles
        if len(partial_indices) == 26:
            radii, feasible = solve_radius_lp(partial_indices, R_wall, neighbour_list)
            if feasible and np.all(radii >= 0):
                s = np.sum(radii)
                if s > best_sum:
                    best_sum = s
                    best_solution = (partial_indices.copy(), radii.copy())
            return

        if ub < best_sum:
            return  # Prune

        # Recurse on 10 candidates that give best potential
        to_consider = remaining_indices[:10] if len(remaining_indices) > 10 else remaining_indices
        for i, candidate in enumerate(to_consider):
            new_partial = partial_indices + [candidate]
            new_remaining = [j for j in remaining_indices if j != candidate]
            new_ub = upper_bound(new_partial, new_remaining, R_wall)
            if new_ub > best_sum:
                backtrack(new_partial, new_remaining, lb, new_ub)
            # Pruning for early infeasibility of the current partial solution
            if len(partial_indices) >= 2:
                radii, feasible = solve_radius_lp(partial_indices, R_wall, neighbour_list)
                if not feasible:
                    return
        return

    initial_partial = all_indices[:26]  # Take first 26 (largest wall radii) to get a good default

    initial_partial = sorted(initial_partial, key=sort_key, reverse=True)

            tmp_remaining = []
            for idx in initial_partial:
                if idx not in initial:
                    tmp_remaining.append(idx)
            ub = upper_bound(initial, tmp_remaining, R_wall)
            q.put((ub, initial, tmp_remaining))

        while not q.empty():
            current_ub, current_partial, current_remaining = q.get()
            if current_ub < best_sum:
                continue
            if len(current_partial) == 26:
                radii, feasible = solve_radius_lp(current_partial, R_wall, neighbour_list)
                if feasible and np.all(radii >= 0):
                    s = np.sum(radii)
                    if s > best_sum:
                        best_sum = s
                        best_solution = (current_partial.copy(), radii.copy())
            elif len(current_partial) < 26:
                # Select next candidate to add (look for good ones)
                candidates_to_add = [idx for idx in current_remaining if R_wall[idx] > 0]
                candidates_to_add.sort(key=lambda x: R_wall[x], reverse=True)
                for i, cand in enumerate(candidates_to_add[:3]):  # Just the top 3
                    new_partial = current_partial + [cand]
                    new_remaining = [idx for idx in current_remaining if idx != cand]
                    ub = upper_bound(new_partial, new_remaining, R_wall)
                    if ub > best_sum:
                        q.put((ub, new_partial, new_remaining))
    except Exception as e:
        print("Exception in branch_and_bound:", str(e))

    if best_solution is not None:
        return best_solution
    else:
        # Fallback: just try one that works
        print("Falling back to dense selection")
        selected_indices = all_indices[:26]
        radii, feasible = solve_radius_lp(selected_indices, R_wall, neighbour_list)

def local_continuous_refinement(initial_centers, initial_radii, centers, R_wall, neighbour_list):
    """Perform refinement using SLSQP to get exact solution."""
    # Reconstruct the indices
    n = centers.shape[0]
    selected_indices = [i for i in range(n) if np.any(np.all(np.abs(centers[i:i+1] - initial_centers) < 1e-10, axis=1))]
    if len(selected_indices) < 26:
        return initial_centers, initial_radii

    def obj_func(radii_and_centers):
        m = len(selected_indices)
        radii = radii_and_centers[:m]
        centers_flat = radii_and_centers[m:]
        return -np.sum(radii)

    def compute_constraints(x):
        m = len(selected_indices)
        radii = x[:m]
        centers_2d = x[m:].reshape(-1, 2)
        # containment
        c1 = radii - R_wall[selected_indices]  # <= 0
        c2 = -radii  # <= 0
        # separation
        const = []
        for i in range(m):
            for j in range(i+1, m):
                dist = np.sqrt(np.sum((centers_2d[i] - centers_2d[j])**2))
                const.append(dist - (radii[i] + radii[j]))  # <= 0
        return list(c1) + list(c2) + const

    # Start refinement from input x = [r0, r1, ..., c0x, c0y, ...]
    x0 = np.concatenate([initial_radii, initial_centers.flatten()])
    bounds = []
    for i in range(len(selected_indices)):
        bounds.append((0.0, R_wall[selected_indices[i]]))  # radii bounds
    for i in range(len(selected_indices)):
        bounds.append((0.0, 1.0))  # x
        bounds.append((0.0, 1.0))  # y

    try:
        from scipy.optimize import minimize
        result = minimize(obj_func, x0, method='SLSQP', bounds=bounds, constraints=[], options={'disp': False})
        if result.success:
            final_radii = result.x[:len(selected_indices)]
            final_centers_flat = result.x[len(selected_indices):]
            final_centers = final_centers_flat.reshape(-1, 2)
            return final_centers, final_radii
        else:
            return initial_centers, initial_radii
    except Exception:
        return initial_centers, initial_radii

    # Compute wall radii
    R_wall = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),

    neighbour_list = build_neighbour_list(centers)

    final_centers = centers[selected_indices]
