import numpy as np

def hex_lattice(d):
    """Generate a hexagonal lattice of centres within [0,1]x[0,1]"""
    sqrt3o2 = np.sqrt(3) / 2
    cols = int(np.ceil(1.0 / d)) + 1
    rows = int(np.ceil(1.0 / (sqrt3o2 * d))) + 1
    centers = []
    for i in range(rows):
        y = i * sqrt3o2 * d
        if y > 1.0:
            continue
        offset = (i % 2) * d / 2
        for j in range(cols):
            x = j * d + offset
            if x > 1.0:
                continue
            centers.append([x, y])
    return np.array(centers)

def add_border_circles(points, d):
    """Add 6 border circles: 4 corners and 2 edges"""
    r = d / 4
    border = [
        [r, r], [1 - r, r], [r, 1 - r], [1 - r, 1 - r],
        [0.5, r], [0.5, 1 - r]
    ]
    return np.vstack([points, border])

def initial_radii(points):
    """Compute a safe initial uniform radius"""
    n = points.shape[0]
    # Distance to walls
    wall_dist = np.minimum(points[:, 0], 1 - points[:, 0])
    wall_dist = np.minimum(wall_dist, points[:, 1])
    wall_dist = np.minimum(wall_dist, 1 - points[:, 1])
    
    # Distance to nearest other point
    min_dist = np.full(n, np.inf)
    for i in range(n):
        diffs = points[i] - points
        dists = np.sqrt(np.sum(diffs ** 2, axis=1))
        dists[i] = np.inf
        min_dist[i] = np.min(dists)
    
    radii = np.minimum(wall_dist, min_dist / 2)
    return np.min(radii)

def inflate_radii(centers, radii):
    """Inflate radii to their maximum feasible values"""
    n = centers.shape[0]
    new_radii = radii.copy()
    epsilon = 1e-7
    max_change = 1.0
    iteration = 0
    max_iterations = 200

    while max_change > epsilon and iteration < max_iterations:
        max_change = 0.0
        new_radii_new = np.zeros_like(radii)
        dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
        
        for i in range(n):
            wall_limits = [
                centers[i, 0], 1 - centers[i, 0],
                centers[i, 1], 1 - centers[i, 1]
            ]
            dist_limits = []
            for j in range(n):
                if i != j:
                    dist_limits.append(dists[i, j] - radii[j])
            limits = wall_limits + dist_limits
            new_radius = min(limits)
            if new_radius > radii[i] + epsilon:
                max_change = max(max_change, new_radius - radii[i])
            new_radii_new[i] = new_radius
        
        radii = new_radii_new
        iteration += 1
    
    return radii

def relax_centers(centers, radii, steps=10):
    """Apply force-based relaxation to improve spacing"""
    n = centers.shape[0]
    kappa = 0.02
    lambda_wall = 0.01
    gamma = 0.02
    beta = 0.02
    alpha = 0.5
    epsilon = 1e-8

    for _ in range(steps):
        old_centers = centers.copy() 
        forces = np.zeros_like(centers)

        # Compute repulsive forces
        dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
        for i in range(n):
            for j in range(n):
                if i == j:
                    continue
                dist = dists[i, j]
                if dist < radii[i] + radii[j] + beta:
                    f = kappa * max(0, beta - dist) * (centers[i] - centers[j]) / (dist + 1e-12)
                    forces[i] += f

        # Compute wall forces
        for i in range(n):
            # x direction
            force = 0
            force += -lambda_wall * max(0, gamma - (centers[i, 0] - radii[i]))
            force += lambda_wall * max(0, gamma - (1 - centers[i, 0] - radii[i]))
            forces[i, 0] += force

            # y direction
            force = 0
            force += -lambda_wall * max(0, gamma - (centers[i, 1] - radii[i]))
            force += lambda_wall * max(0, gamma - (1 - centers[i, 1] - radii[i]))
            forces[i, 1] += force

        # Update positions
        centers += alpha * forces

        # Clamp to valid region
        centers[:, 0] = np.clip(centers[:, 0], radii, 1 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1 - radii)

    return centers

def run_packing():
    # Generate a hexagonal core
    d = 0.18  # roughly 20 circles in a hex grid within [0,1]^2
    core = hex_lattice(d)
    core = core[:20]  # Only keep first 20 points
    
    # Add border-enhanced circles
    all_centers = add_border_circles(core, d)
    
    # Set initial uniform radius
    init_radius = initial_radii(all_centers)
    init_radii = np.full(26, init_radius)
    
    # Perform iterative optimization
    centers = all_centers.copy()
    radii = init_radii.copy()
    prev_sum = np.sum(radii)
    max_outer = 30

    for _ in range(max_outer):
        radii = inflate_radii(centers, radii)
        centers = relax_centers(centers, radii)
        current_sum = np.sum(radii)
        if current_sum - prev_sum < 1e-6:
            break
        prev_sum = current_sum
