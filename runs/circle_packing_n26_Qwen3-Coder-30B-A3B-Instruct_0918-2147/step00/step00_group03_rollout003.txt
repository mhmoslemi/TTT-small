import numpy as np
from scipy.optimize import minimize
from math import sqrt

def run_packing():
    # Seed for reproducibility
    np.random.seed(0)
    
    # Constants
    N = 26
    dt = 0.02
    k_w = 10.0
    k_p = 5.0
    gamma = 1.0
    alpha = 1e-3
    tol_force = 1e-6
    tol_radius = 1e-9
    max_iter_sim = 5000
    max_iter_refine = 2000
    lambda_barrier = 1e-4
    eps = 1e-12
    
    def init_grid():
        s = 0.07
        coords = []
        for k in range(10):
            y = s/2 + k * s * sqrt(3)/2
            if y > 1 - s:
                break
            for m in range(20):
                if k % 2 == 0:
                    x = s/2 + m * s
                else:
                    x = s/2 + m * s + s/2
                if x > 1 - s:
                    break
                if x >= s and x <= 1-s and y >= s and y <= 1-s:
                    coords.append((x, y))
        coords = coords[:N]
        centers = np.array(coords) + np.random.uniform(-0.001, 0.001, size=(N, 2))
        radii = np.full(N, 0.001)
        return centers, radii

    def compute_forces(centers, radii):
        n = centers.shape[0]
        forces = np.zeros_like(centers)
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            # Wall forces
            dL = x - r
            dR = (1 - x) - r
            dB = y - r
            dT = (1 - y) - r
            fl = k_w * max(-dL, 0) * np.array([-1, 0])
            fr = k_w * max(-dR, 0) * np.array([1, 0])
            fb = k_w * max(-dB, 0) * np.array([0, -1])
            ft = k_w * max(-dT, 0) * np.array([0, 1])
            forces[i] += fl + fr + fb + ft
            
            # Pairwise repulsion
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = sqrt(dx*dx + dy*dy + eps)
                gap = dist - (radii[i] + radii[j])
                if gap < 0:
                    f = k_p * (-gap) * np.array([dx, dy]) / dist
                    forces[i] += f
                    forces[j] -= f
        return forces

    def grow_radii(centers, radii, forces):
        n = len(radii)
        new_radii = radii.copy()
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            dL = x - r
            dR = (1 - x) - r
            dB = y - r
            dT = (1 - y) - r
            # Only grow if not touching wall or other circle
            can_grow = all(d > 0 for d in [dL, dR, dB, dT])
            if can_grow:
                # Collect interactions
                overlap = False
                for j in range(n):
                    if i == j:
                        continue
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = sqrt(dx*dx + dy*dy + eps)
                    gap = dist - (radii[i] + radii[j])
                    if gap < 0:
                        overlap = True
                        break
                if not overlap:
                    new_radii[i] += alpha * dt
        return new_radii

    def max_radii_fixed_centers(centers, radii):
        n = len(radii)
        converged = False
        max_iter = 30
        for iteration in range(max_iter):
            new_radii = np.zeros_like(radii)
            for i in range(n):
                r = radii[i]
                x, y = centers[i]
                min_dist = min(x, 1 - x, y, 1 - y)
                for j in range(n):
                    if i != j:
                        dx = centers[i, 0] - centers[j, 0]
                        dy = centers[i, 1] - centers[j, 1]
                        dist = sqrt(dx*dx + dy*dy + eps)
                        gap = dist - radii[j]
                        min_dist = min(min_dist, gap)
                new_radii[i] = max(0, min_dist)
            diff = np.sum(np.abs(new_radii - radii))
            if diff < tol_radius:
                converged = True
                break
            radii = new_radii
        return new_radii

    def objective_sum_radii(centers_flat):
        # Reshape flat to centers array
        n = 26
        centers = centers_flat.reshape(n, 2)
        # Compute radii for current centers
        radii = max_radii_fixed_centers(centers, np.full(26, 0.001))
        return -np.sum(radii)

    def constraint_bound(x_flat):
        # Constraints for L-BFGS-B
        # x and y go from 0 to 1
        bounds = []
        for i in range(0, len(x_flat), 2):
            bounds.extend([0, 1])  # x between 0 and 1
            bounds.extend([0, 1])  # y between 0 and 1
        return bounds

    # 1. Initial placement
    centers, radii = init_grid()

    # 2. Physics simulation
    velocities = np.zeros_like(centers)
    for step in range(max_iter_sim):
        forces = compute_forces(centers, radii)
        velocities += dt * (forces - gamma * velocities)
        centers += dt * velocities
        # Clip to [0,1]
        centers = np.clip(centers, 0, 1)
        # Update radius
        radii = grow_radii(centers, radii, forces)
        # Check termination
        if all(abs(r) < 1e-9 for r in [np.linalg.norm(f) for f in forces]) and \
           all(r < 1e-9 for r in radii):
            break

    # 3. Exact max radii for fixed centres
    radii = max_radii_fixed_centers(centers, radii)

    # 4. L-BFGS-B refinement
    centers_flat = centers.flatten()
    result = minimize(objective_sum_radii, centers_flat, method='L-BFGS-B', 
                      bounds=[(0,1) for _ in range(52)], 
                      options={'maxiter': max_iter_refine, 'gtol': 1e-9})

    final_centers = result.x.reshape(26, 2)
    
    # 5. Final max radius pass
    final_radii = max_radii_fixed_centers(final_centers, np.full(26, 0.001))
    
    # 6. Sum of radii
    sum_radii = np.sum(final_radii)

    # 7. Validate
    valid, msg = validate_packing(final_centers, final_radii)
    if not valid:
        raise ValueError(f"Validation failed: {msg}")

    return final_centers, final_radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
