import numpy as np

from scipy.spatial.distance import cdist

def init_candidate_grid(radii_list, grid_spacing_factor=1.0):
    candidates = []
    for r in radii_list:
        spacing = r * np.sqrt(2) * grid_spacing_factor
        x_range = np.arange(r, 1 - r + 1e-10, spacing)
        y_range = np.arange(r, 1 - r + 1e-10, spacing)
        for x in x_range:
            for y in y_range:
                candidates.append((x, y, r))
    return candidates

def build_overlap_matrix(candidates):
    n = len(candidates)
    overlap = np.zeros((n, n), dtype=bool)
    coords = np.array([[c[0], c[1]] for c in candidates])
    radii = np.array([c[2] for c in candidates])
    dists = cdist(coords, coords)
    for i in range(n):
        for j in range(i+1, n):
            if dists[i, j] < radii[i] + radii[j]:
                overlap[i, j] = overlap[j, i] = True
    return overlap

def max_radii_fixed_centres(centers, radii_initial):
    n = len(centers)
    radii = radii_initial.copy()
    
    # Update each circle's radius to be as large as allowed without violating constraints
    for i in range(n):
        # Constraint: center must be inside the square with assigned radius
        x, y = centers[i]
        r_i = radii[i]
        
        # Minimum allowed radius w.r.t. walls
        wall_radius = min(x, 1 - x, y, 1 - y)
        
        # Check overlaps with other circles
        max_r = wall_radius
        for j in range(n):
            if i != j:
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                max_r = min(max_r, dist - radii[j])
        
        radii[i] = max_r
    return radii

def nlp_refine(centers, radii_initial):
    def obj_func(v):
        # v = [x1, y1, ..., x26, y26, r1, ..., r26]
        centers_v = v[:2*26].reshape((26, 2))
        radii_v = v[2*26:]
        return -np.sum(radii_v) # minimize negative since maximize

    def eq_constraints(v):
        centers_v = v[:2*26].reshape((26, 2))
        radii_v = v[2*26:]
        return np.array([]) # No linear equality constraints

    def ineq_constraints(v):
        centers_v = v[:2*26].reshape((26, 2))
        radii_v = v[2*26:]
        # Wall constraints
        cons = []
        for i in range(26):
            r = radii_v[i]
            x, y = centers_v[i]
            cons.extend([
                r - x,           # r >= x
                r + x - 1,       # r + x <= 1
                r - y,           # r >= y
                r + y - 1        # r + y <= 1
            ])
        
        # Non-overlap constraints
        for i in range(26):
            for j in range(i+1, 26):
                dx = centers_v[i, 0] - centers_v[j, 0]
                dy = centers_v[i, 1] - centers_v[j, 1]
                dist_sq = dx*dx + dy*dy
                dist = np.sqrt(dist_sq) if dist_sq > 1e-15 else 1e-15
                cons.append(radii_v[i] + radii_v[j] - dist)
        
        return np.array(cons)

    # Get Jacobian of each constraint
    def jacobian(v):
        centers_v = v[:2*26].reshape((26, 2))
        radii_v = v[2*26:]
        # Check whether to compute element-wise derivative
        rows, cols = [], []
        data = []
        
        # Wall constraints
        for i in range(26):
            r = radii_v[i]
            x, y = centers_v[i]
            
            # r >= x
            rows.extend([i*4, i*4])
            cols.extend([26*2 + i, 2*i])
            data.extend([1.0, -1.0])
            
            # r <= 1 - x
            rows.extend([i*4+1, i*4+1])
            cols.extend([26*2 + i, 2*i])
            data.extend([1.0, 1.0])
            
            # r >= y
            rows.extend([i*4+2, i*4+2])
            cols.extend([26*2 + i, 2*i+1])
            data.extend([1.0, -1.0])
            
            # r <= 1 - y
            rows.extend([i*4+3, i*4+3])
            cols.extend([26*2 + i, 2*i+1])
            data.extend([1.0, 1.0])
        
        # Non-overlap constraints
        k = 26*4  # start offset for overlap constraint index
        for i in range(26):
            for j in range(i+1, 26):
                dx = centers_v[i, 0] - centers_v[j, 0]
                dy = centers_v[i, 1] - centers_v[j, 1]
                dist_sq = dx*dx + dy*dy
                dist = np.sqrt(dist_sq) if dist_sq > 1e-15 else 1e-15
                
                # Derivative w.r.t. centers
                if dist > 1e-15:
                    d_dist_dx = dx / dist
                    d_dist_dy = dy / dist
                        
                    rows.extend([
                        k, k, k+1, k+1
                    ])
                    cols.extend([
                        2*i, 2*j, 2*i+1, 2*j+1
                    ])
                    data.extend([
                        -d_dist_dx, d_dist_dx,
                        -d_dist_dy, d_dist_dy
                    ])
                
                # Derivative w.r.t. radii
                rows.extend([
                    k, k+1
                ])
                cols.extend([
                    26*2+i,
                    26*2+j
                ])
                data.extend([
                    1.0, 1.0
                ])
                
                k += 2
        
        # Continuous alone...
        # Row size is total inequality constraints:
        num_cons = 26*4 + 26*25//2
        J = sp.coo_matrix((data, (rows, cols)), shape=(num_cons, len(v)))
        return J.tocsr()
    
    # Initialize with input
    v_init = np.hstack([centers.reshape(-1), radii_initial])
    v_init = v_init + np.random.normal(0, 1e-4, size=v_init.shape)
    
    # Optimization
    try:
        result = minimize(
            obj_func, 
            v_init, 
            method='trust-constr',
            constraints=[{'type': 'ineq', 'fun': ineq_constraints}],
            options={
                'maxiter': 300,
                'xtol': 1e-9,
                'gtol': 1e-9
            },
            # jac=jacobian  # We don't compute numeric jacobian for simplicity
        )
        
        if result.success:
            centers_opt = result.x[:2*26].reshape((26, 2))
            radii_opt = result.x[2*26:]
            return centers_opt, radii_opt
        else:
            # Fall back to a simple projection
            return centers, radii_initial
    except:
        return centers, radii_initial

def run_packing():
    # Parameters
    radii_list = np.arange(0.025, 0.12, 0.01)  # starts from 0.025 with step 0.01
    init_grid_spacing_factor = 1.0
    
    # Phase 1: Discrete optimization
    candidates = init_candidate_grid(radii_list, init_grid_spacing_factor)
    overlap = build_overlap_matrix(candidates)
    centers, radii = solve_discrete_bip(candidates, overlap, 2.636)
    
    # Final numerical refinement after exact assignment and continuous adjustment
    # First compute exact max radii constrain 
    radii = max_radii_fixed_centres(centers, radii)
    
    # Refine using trust-constr
    centers_refined, radii_refined = nlp_refine(centers, radii)
    
    # Final projection step
    radii_final = max_radii_fixed_centres(centers_refined, radii_refined)
    
    # computation final sum
    sum_radii = np.sum(radii_final)
    
    # Validation: enforce precision
    assert validate_packing(centers_refined, radii_final)[0], "Packing is not valid."
    return centers_refined, radii_final, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
