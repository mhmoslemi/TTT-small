```python
import numpy as np
from scipy.optimize import minimize

NUM_CIRCLES = 26
NUM_RESTARTS = 5
GLOBAL_BOUNDS = [(0, 1), (0, 1), (0, 0.5)] * NUM_CIRCLES

def hex_lattice_points(n_rows, spacing):
    points = []
    for i in range(n_rows):
        y = i * spacing * np.sqrt(3) / 2
        if y > 1:
            break
        offset = (i % 2) * spacing / 2
        n_cols = int((1 - offset) / spacing) + 1
        for j in range(n_cols):
            x = offset + j * spacing
            if x > 1:
                break
            points.append((x, y))
    return points

def initial_guess(seed):
    np.random.seed(seed)
    points = hex_lattice_points(6, 0.15)
    while len(points) < NUM_CIRCLES:
        points = hex_lattice_points(len(points) // 2 + 3, 0.10)
    points = points[:NUM_CIRCLES]
    jitter = np.random.uniform(-0.02, 0.02, (NUM_CIRCLES, 2))
    centers = np.array(points) + jitter
    radii = np.full(NUM_CIRCLES, 0.02)
    vars0 = np.empty(NUM_CIRCLES * 3)
    vars0[0::3] = centers[:, 0]
    vars0[1::3] = centers[:, 1]
    vars0[2::3] = radii
    return vars0

def objective(vars):
    return -np.sum(vars[2::3])

def constraint_boundary(vars):
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    boundaries = np.array([
        x - r,           # left
        1 - x - r,       # right
        y - r,           # bottom
        1 - y - r        # top
    ])
    return boundaries.T.flatten()

def constraint_nonoverlap(vars):
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    n = NUM_CIRCLES
    margins = []
    for i in range(n):
        for j in range(i+1, n):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dist = np.sqrt(dx*dx + dy*dy)
            overlap = dist - (r[i] + r[j])
            margins.append(overlap)
    return np.array(margins)

def postprocess(vars):
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]

    # Fix boundary violations
    for i in range(NUM_CIRCLES):
        if x[i] - r[i] < 0:
            x[i] = r[i]
        elif x[i] + r[i] > 1:
            x[i] = 1 - r[i]
        if y[i] - r[i] < 0:
            y[i] = r[i]
        elif y[i] + r[i] > 1:
            y[i] = 1 - r[i]

    # Refine non-overlaps
    for i in range(NUM_CIRCLES):
        for j in range(i+1, NUM_CIRCLES):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dist = np.sqrt(dx*dx + dy*dy)
            overlap = dist - (r[i] + r[j])
            if overlap < -1e-9:
                # Only impact correction if it's small numerical error
                adjustment = -overlap / 2.0
                if r[i] > r[j]:
                    r[i] -= adjustment
                    r[j] += adjustment
                else:
                    r[i] += adjustment
                    r[j] -= adjustment

    vars_new = vars.copy()
    vars_new[0::3] = x
    vars_new[1::3] = y
    vars_new[2::3] = r
    return vars_new

def run_packing():
    best_sum = -np.inf
    best_vars = None
    
    constraints = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_nonoverlap}
    ]

    for seed in range(NUM_RESTARTS):
        vars0 = initial_guess(seed)
        res = minimize(
            objective,
            vars0,
            method='SLSQP',
            bounds=GLOBAL_BOUNDS,
            constraints=constraints,
            options={'maxiter': 3000, 'ftol': 1e-7}
        )
        if not res.success:
            continue
        vars_corr = postprocess(res.x)
        s = np.sum(vars_corr[2::3])
        if s > best_sum:
            best_sum = s
            best_vars = vars_corr

    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii = best_vars[2::3]
    return centers, radii, float(best_sum)
```
