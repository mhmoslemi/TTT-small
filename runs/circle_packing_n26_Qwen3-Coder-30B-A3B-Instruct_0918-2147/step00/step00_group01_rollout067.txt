import numpy as np
from scipy.spatial.distance import cdist

def penalty(z, alpha=10.0, beta=10.0):
    # z has shape (78,) = 26 * (x,y,r)
    centers = z.reshape(-1, 3)[:, :2]
    radii = z.reshape(-1, 3)[:, 2]

    # Wall penalty
    wall_penalty = 0.0
    for i in range(26):
        r = radii[i]
        x, y = centers[i]
        dists = [x, 1 - x, y, 1 - y]
        for d in dists:
            if d < r:
                wall_penalty += alpha * (r - d)

    # Overlap penalty
    overlap_penalty = 0.0
    for i in range(26):
        for j in range(i + 1, 26):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            r_sum = radii[i] + radii[j]
            if dist < r_sum:
                overlap_penalty += beta * (r_sum - dist)

    return wall_penalty + overlap_penalty

def energy(z, lam, alpha=10.0, beta=10.0):
    total_radius = np.sum(z.reshape(-1, 3)[:, 2])
    return -total_radius + lam * penalty(z, alpha, beta)

def propose_move(z, sigma_c, sigma_r, rng):
    z_new = z.copy()
    move_type = rng.choice([0, 1, 2])
    centers = z_new.reshape(-1, 3)[:, :2]
    radii = z_new.reshape(-1, 3)[:, 2]
    
    if move_type == 0:
        # Center jitter
        idx = rng.randint(0, 26)
        noise = rng.normal(0, sigma_c, 2)
        centers[idx] += noise
        centers[idx] = np.clip(centers[idx], 0, 1)
    elif move_type == 1:
        # Radius perturb
        idx = rng.randint(0, 26)
        noise = rng.normal(0, sigma_r)
        radii[idx] += noise
        radii[idx] = np.maximum(radii[idx], 0)
    else:
        # Swap radii
        i, j = rng.choice(26, 2, replace=False)
        radii[i], radii[j] = radii[j], radii[i]
    
    return z_new

def anneal(rng):
    # Initial configuration: hexagonal grid
    base_grid_size = 0.18
    num_per_side = int(1.0 / base_grid_size) + 1
    grid_x = np.linspace(0.0, 1.0, num_per_side)
    grid_y = np.linspace(0.0, 1.0, num_per_side)
    grid_x = grid_x[1:-1] if len(grid_x) > 2 else grid_x
    grid_y = grid_y[1:-1] if len(grid_y) > 2 else grid_y
    xx, yy = np.meshgrid(grid_x, grid_y)
    centers = np.column_stack([xx.ravel(), yy.ravel()])
    centers = centers[:26]  # Allow only 26 centers
    if len(centers) < 26:
        # Fill with random centers
        missing = 26 - len(centers)
        extra_centers = rng.random((missing, 2))
        centers = np.vstack([centers, extra_centers])
    
    radii = np.array([min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1]) 
                      for i in range(26)])
    z = np.column_stack([centers, radii]).flatten()

    # Annealing parameters
    T = 1.0
    c_T = 0.995
    lam = 0.1
    c_lam = 1.03
    max_iter = 10000
    sigma_c = 0.03
    sigma_r = 0.02
    best_z = z.copy()
    best_sum = np.sum(radii)
    T_min = 1e-5
    lam_min = 1e15  # essentially no penalty, but don't go too far

    for it in range(max_iter):
        z_prop = propose_move(z, sigma_c * np.sqrt(T), sigma_r * np.sqrt(T), rng)
        E_old = energy(z, lam)
        E_new = energy(z_prop, lam)
        delta_E = E_new - E_old
        if delta_E <= 0 or rng.random() < np.exp(-delta_E / T):
            z = z_prop
        T *= c_T

        # Update penalty weight
        if it % 100 == 0:
            lam *= c_lam

        # Update best feasible solution
        rad = z.reshape(-1, 3)[:, 2]
        penalty_val = penalty(z)
        if penalty_val < 1e-9:  # Feasible
            rad_sum = np.sum(rad)
            if rad_sum > best_sum:
                best_sum = rad_sum
                best_z = z.copy()

        if T < T_min:
            break

    return best_z

def project_and_ascent(centers, radii):
    max_iter = 500
    eta = 5e-3
    tol = 1e-7
    for _ in range(max_iter):
        # Gradient ascent: modify radii to increase sum
        grad = np.zeros(26)
        grad[:] = 1.0
        step = eta * grad
        radii += step
        # Project to wall constraints
        for i in range(26):
            max_r = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
            radii[i] = max(0, min(radii[i], max_r))
        # Overlap resolve
        centers_full = centers
        radii_full = radii
        for _ in range(10):  # Max iterations for overlap resolution
            updated = False
            for i in range(26):
                for j in range(i + 1, 26):
                    dist = np.sqrt(np.sum((centers_full[i] - centers_full[j]) ** 2))
                    r_sum = radii_full[i] + radii_full[j]
                    if dist < r_sum:
                        overlap = r_sum - dist
                        # Reduce appropriately
                        radii_full[i] -= overlap / 2.0
                        radii_full[j] -= overlap / 2.0
                        updated = True
            # Renormalize
            rescale = 0.9
            radii_full = np.clip(radii_full, 0, np.inf)
            for i in range(26):
                max_r = min(centers_full[i, 0], 1 - centers_full[i, 0], 
                            centers_full[i, 1], 1 - centers_full[i, 1])
                radii_full[i] = min(radii_full[i], max_r)
            if not updated:
                break
        radii = radii_full.copy()
        eta *= 0.95
        if eta < 1e-8:
            break

    return centers, radii

def run_packing():

    # Run simulated annealing
    z_best = anneal(RNG)

    # Extract centers and radii from best solution
