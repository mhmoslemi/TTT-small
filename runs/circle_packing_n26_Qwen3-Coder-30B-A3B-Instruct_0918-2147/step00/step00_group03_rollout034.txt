import numpy as np

def build_lp_model(n_circles):
    # Build the constraint matrix for the LP
    n_vars = 2 * n_circles + n_circles  # x, y, r for each circle
    A_ub = []
    b_ub = []
    
    # Separation constraints (axis-wise)
    # For each pair (i, j), we have 4 constraints, add them all
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            # x_i + r_i <= x_j - r_j
            row = [0] * n_vars
            row[2*i] = 1.0      # x_i
            row[2*i+1] = 0.0    # y_i
            row[2*n_circles+i] = 1.0  # r_i
            row[2*j] = -1.0     # x_j
            row[2*j+1] = 0.0    # y_j
            row[2*n_circles+j] = 1.0  # r_j
            A_ub.append(row)
            b_ub.append(0.0)
            
            # x_j + r_j <= x_i - r_i
            row = [0] * n_vars
            row[2*i] = -1.0     # x_i

        row[2*n_circles+i] = 1.0  # r_i
        A_ub.append(row)
        b_ub.append(0.0)
        
    # r_i <= 1 - x_i => x_i + r_i <= 1
    for i in range(n_circles):
        row = [0] * n_vars
        row[2*i] = 1.0      # x_i
        row[2*n_circles+i] = 1.0  # r_i
        A_ub.append(row)
        b_ub.append(1.0)
        
    # r_i <= y_i => -y_i + r_i <= 0
    for i in range(n_circles):
        row = [0] * n_vars
        row[2*i+1] = -1.0   # y_i
        row[2*n_circles+i] = 1.0  # r_i
        A_ub.append(row)
        b_ub.append(0.0)
        
    # r_i <= 1 - y_i => y_i + r_i <= 1
    for i in range(n_circles):
        row = [0] * n_vars
        row[2*i+1] = 1.0    # y_i
        row[2*n_circles+i] = 1.0  # r_i
        A_ub.append(row)
        b_ub.append(1.0)
    
    # Objective: maximize sum of radii
    c = np.zeros(n_vars)
    c[2*n_circles:] = -1.0  # negative because linprog does minimization

    return A_ub, b_ub, c

def solve_lp(A_ub, b_ub, c):
    # Solve the LP using 'highs' method
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'presolve': True})
    if not res.success:
        raise RuntimeError("LP solver failed to find an optimal solution")
    
    n = len(c) // 3  # number of circles
    
    centers = res.x[:2*n].reshape(n, 2)
    radii = res.x[2*n:]
    
    return centers, radii

def detect_diagonal_conflicts(centers, radii, tol=1e-12):
    n = centers.shape[0]
    conflicts = []

            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - tol:
                conflicts.append((i, j))
    
    return conflicts
