import numpy as np
from scipy.optimize import minimize

def initial_lattice(n=26):
    # Generate hexagonal lattice points
    sqrt3 = np.sqrt(3)
    spacing = 1.0
    while True:
        # Compute number of points that fit
        nx = int(1.0 / spacing) + 1
        ny = int(1.0 / (spacing * sqrt3 / 2)) + 1
        total = 0
        points = []
        for i in range(nx):
            for j in range(ny):
                x = i * spacing
                y = j * spacing * sqrt3 / 2
                if j % 2 == 1:
                    x += spacing / 2
                if x <= 1 and y <= 1:
                    points.append([x, y])
                    total += 1
                    if total >= n:
                        break
            if total >= n:
                break
        if total >= n:
            break
        spacing *= 0.95

    points = np.array(points[:n])
    radii = np.full(n, 0.5)

    # For each point, compute the feasible radius
    for i in range(n):
        x, y = points[i]
        r_max = min(x, 1 - x, y, 1 - y)
        for j in range(n):
            if i != j:
                dx = points[i, 0] - points[j, 0]
                dy = points[i, 1] - points[j, 1]
                dist = np.hypot(dx, dy)
                r_limit = dist / 2
                r_max = min(r_max, r_limit)
        radii[i] = max(0, r_max)

    return points, radii

def pack(centers, radii):
    # Flatten centers and radii into a single array
    return np.hstack([centers.flatten(), radii])

def unpack(z):
    n = len(z) // 3
    centers = z[:2*n].reshape((n, 2))
    radii = z[2*n:]
    return centers, radii

def objective(z):
    _, radii = unpack(z)
    return -np.sum(radii)

def constraint_boundary(z):
    # returns array of values >= 0 for each boundary constraint
    centers, radii = unpack(z)
    n = len(radii)
    cons = []
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # left, right, bottom, top
        cons.extend([r - x, r + x - 1, r - y, r + y - 1])
    return np.array(cons)

def constraint_nonoverlap(z):
    # returns array of values >= 0 for each non-overlap constraint
    centers, radii = unpack(z)
    n = len(radii)
    cons = []
    for i in range(n):
        for j in range(i+1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.hypot(dx, dy) + 1e-9  # avoid singularities
            rsum = radii[i] + radii[j]
            cons.append(rsum - dist)
    return np.array(cons)

def repair(centers, radii):
    # repair any boundary or overlap violations
    n = len(radii)
    max_iter = 10
    for _ in range(max_iter):
        improved = False
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            # enforce boundary
            r_new = min(r, x, 1 - x, y, 1 - y)
            if r_new < r:
                radii[i] = r_new
                improved = True
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.hypot(dx, dy)
                rsum = radii[i] + radii[j]
                if dist < rsum:
                    overlap = rsum - dist
                    delta = overlap / 2
                    if radii[i] > delta:
                        radii[i] -= delta
                    if radii[j] > delta:
                        radii[j] -= delta
                    improved = True
        if not improved:
            break
    return centers, radii

def run_packing():
    centers, radii = initial_lattice(n=26)
    z0 = pack(centers, radii)

    # set bounds and constraints
    bounds = [(0, 1)] * 52 + [(0, 0.5)] * 26  # x, y in [0,1], r in [0,0.5]
    
    constraints = []
    constraints.append({'type': 'ineq', 'fun': constraint_boundary})
    constraints.append({'type': 'ineq', 'fun': constraint_nonoverlap})

    # optimize
    res = minimize(objective, z0, method='SLSQP', bounds=bounds, constraints=constraints,
                   options={'maxiter': 500, 'ftol': 1e-9, 'disp': False})

    # unpack result
    centers, radii = unpack(res.x)
    
    # post-process
    centers, radii = repair(centers, radii)

    # compute final sum
    sum_radii = np.sum(radii)
    # Validate result
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise RuntimeError(f"Validation failed after optimization: {msg}")

    # retarget
    if sum_radii < 2.636:
        raise RuntimeError(f"Sum of radii is {sum_radii}, below target 2.636")

    return centers, radii, sum_radii
