def build_hex_lattice(spacing):
    # Generate a hexagonal lattice inside the unit square [0,1]x[0,1]
    # with cell spacing slightly less than the basic hex spacing to allow packing
    lattice_points = []
    row = 0
    y = spacing * 0.5
    while y < 1 - spacing * 0.5:
        offset = (row % 2) * spacing * 0.5
        x = offset + spacing * 0.5
        while x < 1 - spacing * 0.5:
            lattice_points.append([x, y])
            x += spacing
        row += 1
        y += spacing * sqrt(3) * 0.5
    return np.array(lattice_points)

def local_optimize(centers0, radii):
    # Refine given initial guess using scipy.minimize (trust-constr)
    n = centers0.shape[0]

    # Form all variables as [x1, y1, r1, x2, y2, r2, ..., x26, y26, r26]
    def to_vars(centers, radii):
        vars_ = []
        for i in range(n):
            vars_.extend([centers[i,0], centers[i,1], radii[i]])
        return np.array(vars_)

    def from_vars(vars_):
        centers = np.zeros((n, 2))
        radii = np.zeros(n)
        for i in range(n):
            centers[i,0] = vars_[3*i]
            centers[i,1] = vars_[3*i + 1]
            radii[i] = vars_[3*i + 2]
        return centers, radii

    def objective(vars_):
        _, radii = from_vars(vars_)
        return -np.sum(radii)  # negative since we want maximum

    def jacobian(vars_):
        j = np.full_like(vars_, -1.0)
        for i in range(n):
            j[3*i + 2] = -1.0  # derivative w.r.t. radii is -1
        return j

    def constraints(vars_):
        centers, radii = from_vars(vars_)
        cons = []

        # Wall constraints as inequality = true if feasible
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            # x - r >= 0 (left)
            cons.append(x - r)
            # 1 - x - r >= 0 (right)
            cons.append(1 - x - r)
            # y - r >= 0 (bottom)
            cons.append(y - r)
            # 1 - y - r >= 0 (top)
            cons.append(1 - y - r)

        # Pairwise non-overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i,0] - centers[j,0]
                dy = centers[i,1] - centers[j,1]
                distsq = dx*dx + dy*dy
                sumrad = radii[i] + radii[j]
                # distance^2 - (sumrad)^2 >= 0
                cons.append(distsq - sumrad * sumrad)

        # Return negated values to satisfy scipy convention of g(x) <= 0
        return np.array(cons)

    def jacobian_constraints(vars_):
        centers, radii = from_vars(vars_)
        jac = []
        # Wall constraint gradients (n circles with 4 wall bounds each => 4n total constraints)
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            row1 = np.zeros(3*n)  # For left wall x - r >= 0 => 1, 0, -1
            row1[3*i] = 1.0
            row1[3*i + 2] = -1.0
            jac.append(row1)

            row2 = np.zeros(3*n)
            row2[3*i] = -1.0
            row2[3*i + 2] = -1.0
            jac.append(row2)

            row3 = np.zeros(3*n)
            row3[3*i + 1] = 1.0
            row3[3*i + 2] = -1.0
            jac.append(row3)

            row4 = np.zeros(3*n)
            row4[3*i + 1] = -1.0
            row4[3*i + 2] = -1.0
            jac.append(row4)

        # Non-overlap gradients (n*(n-1)/2 constraints)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i,0] - centers[j,0]
                dy = centers[i,1] - centers[j,1]
                sumrad = radii[i] + radii[j]
                # grad(w.r.t. center[i])
                grad_i = np.zeros(3*n)
                grad_i[3*i] = 2*dx
                grad_i[3*i + 1] = 2*dy
                grad_i[3*i + 2] = -2*sumrad

                # grad(w.r.t. center[j])
                grad_j = np.zeros(3*n)
                grad_j[3*j] = -2*dx
                grad_j[3*j + 1] = -2*dy
                grad_j[3*j + 2] = -2*sumrad

                grad = grad_i + grad_j
                jac.append(grad)

        return np.array(jac)

    # Flatten initial into vars array
    vars0 = to_vars(centers0, radii)

    # Define bounds
    bounds = []
    for i in range(n):
        bounds.append((1e-10, 1 - 1e-10))   # x: near [0,1]
        bounds.append((1e-10, 1 - 1e-10))   # y: .       .
        bounds.append((1e-10, 0.5))         # r: [1e-10, 0.5] max circle

    # Construct non-overlap inequalities: see below AFTER binding variables:
    def fun(vars_):
        return objective(vars_)

    def jac(vars_):
        return jacobian(vars_)

    # Expand constraint matrix carefully with variable bindings
    cons = constraints(vars0)
    from scipy.optimize import Bounds, LinearConstraints, NonlinearConstraint

    # Use SCIPY minimizer with trust-constr
    result = minimize(
        fun, vars0, method='trust-constr',
        jac=jac,
        bounds=bounds,
        constraints=NonlinearConstraint(constraints, 0, np.inf, jac=jacobian_constraints),
        options={'disp': False, 'maxiter': 200, 'gtol': 1e-12}
    )

    # Extract final result
    final_centers, final_radii = from_vars(result.x)
    return final_centers, final_radii
