import numpy as np
from scipy.optimize import linprog
from scipy.spatial.distance import cdist

def generate_hex_lattice(h):
    sqrt3 = np.sqrt(3)
    points = []
    for j in range(int(1.0 / (h * sqrt3 / 2)) + 2):
        for i in range(int(1.0 / h) + 2):
            x = i * h + (j % 2) * (h / 2)
            y = j * h * sqrt3 / 2
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append([x, y])
    return np.array(points)

def build_neighbour_list(centers, max_dist=0.4):
    n = centers.shape[0]
    distances = cdist(centers, centers)
    edges = []
    for i in range(n):
        for j in range(i+1, n):
            if distances[i, j] < max_dist:
                edges.append((i, j, distances[i, j]))
    return edges

def solve_radius_lp(selected_indices, R_wall, neighbour_list):
    n_selected = len(selected_indices)
    if n_selected == 0:
        return np.array([]), True

    # Set up LP:
    # Variables: radii r_i for i in selected_indices
    # Objective: maximize sum of radii
    c = -np.ones(n_selected)
    
    # Bounds for radii: 0 <= r_i <= R_wall[i] for all i in selected_indices
    bounds = [(0, R_wall[i]) for i in selected_indices]

    # Inequalities: r_i + r_j <= d_ij for (i,j) in neighbour_list where both i,j in selected_indices
    A_eq = []
    b_eq = []
    A_ub = []
    b_ub = []

    # Add wall constraints as upper bounds (already handled by bounds)
    # Also add separation constraints
    selected_set = set(selected_indices)
    for i, j, d_ij in neighbour_list:
        if i in selected_set and j in selected_set:
            idx_i = selected_indices.index(i)
            idx_j = selected_indices.index(j)
            # Add row: r_i + r_j <= d_ij  =>  -r_i - r_j >= -d_ij
            row = [0] * n_selected
            row[idx_i] = -1
            row[idx_j] = -1
            A_ub.append(row)
            b_ub.append(-d_ij)

    # Solve the LP
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, True
        else:
            return None, False
    except Exception:
        return None, False

def upper_bound(partial_set, remaining_indices, R_wall):
    ub = sum(R_wall[i] for i in partial_set)
    # Add optimistic upper bound on remaining circles
    max_remaining = max(R_wall[i] for i in remaining_indices) if remaining_indices else 0
    remaining_slots = 26 - len(partial_set)
    ub += max_remaining * remaining_slots
    return ub

def branch_and_bound(centers, R_wall, neighbour_list):
    best_total_radius = -np.inf
    best_indices = None
    best_radii = None

    # Initial bounds
    lb = -np.inf
    ub = np.sum(R_wall)  # Heuristic upper bound

    # BFS search queue, represented with tuples: (partial_indices, remaining_indices, ub_value)
    queue = [(set(), set(range(len(centers))), ub)]

    # Precompute indices for faster lookup
    max_lattice_size = len(centers)

    while queue:
        partial_indices, remaining_indices, upper_b = queue.pop(0)
        if len(partial_indices) == 26:
            selected_indices = list(sorted(partial_indices))
            # Solve exact LP for this subset
            total_radius, is_feasible = solve_radius_lp(selected_indices, R_wall, neighbour_list)
            if is_feasible and total_radius > best_total_radius:
                best_total_radius = total_radius
                best_indices = selected_indices
                # Compute actual radii
                radii, _ = solve_radius_lp(selected_indices, R_wall, neighbour_list)
                best_radii = radii
        else:
            # Try to prune
            if upper_b <= lb:
                continue
                
            # Pick the next index to add (greedy by wall radius)
            if remaining_indices:
                next_index = max(remaining_indices, key=lambda i: R_wall[i])
                # Branch: include this index
                new_partial = partial_indices | {next_index}
                new_remaining = remaining_indices - {next_index}
                current_ub = upper_bound(list(partial_indices) + [next_index], new_remaining, R_wall)
                if current_ub > lb:
                    queue.append((new_partial, new_remaining, current_ub))
                # Branch: exclude this index
                new_remaining = remaining_indices - {next_index}
                current_ub = upper_bound(list(partial_indices), new_remaining, R_wall)
                if current_ub > lb:
                    queue.append((partial_indices, new_remaining, current_ub))
    return best_indices, best_radii

def local_continuous_refinement(initial_centers, initial_radii):
    from scipy.optimize import minimize
    n_circles = len(initial_centers)
    
    # Prepare initial guess: we will vary the centers slightly
    x_guess = np.hstack([
        initial_centers.flatten(),
        initial_radii
    ])
    
    def obj(x):
        # x = [x1, y1, x2, y2, ..., xn, yn, r1, r2, ..., rn]
        centers = x[:2*n_circles].reshape(-1, 2)
        radii = x[2*n_circles:]
        return -np.sum(radii)  # Negative because we minimize
    
    # Inequality constraints for non-overlapping
    def constraint_nonoverlap(i, j):
        def fun(x):
            centers = x[:2*n_circles].reshape(-1, 2)
            radii = x[2*n_circles:]
            c1 = centers[i]
            c2 = centers[j]
            r1 = radii[i]
            r2 = radii[j]
            dist = np.sqrt(np.sum((c1 - c2)**2))
            # Overlap if dist < r1 + r2
            # We want to penalize it
            return dist - (r1 + r2)
        return fun
    
    # Bounds for centers
    bounds = [(0, 1)] * (2*n_circles) + [(0, 1)] * n_circles  # x, y, r for each circle
    
    # Feasibility check using current solution for sphere packing
    constraints = []
    
    # For the current problem, we simulate inner-function for constraints
    def make_constraint(i, j):
        def fun(x):
            centers = x[:2*n_circles].reshape(-1, 2)
            radii = x[2*n_circles:]
            c1 = centers[i]
            c2 = centers[j]
            r1 = radii[i]
            r2 = radii[j]
            dist = np.sqrt(np.sum((c1 - c2)**2))
            return dist - (r1 + r2)
        return {'type': 'ineq', 'fun': fun}

    for i in range(n_circles):
        for j in range(i+1, n_circles):
            constraints.append(make_constraint(i, j))

    # Bound constraints for center and radius
    for i in range(n_circles):
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[2*i] - 0})   # x >= 0
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[2*i]})   # x <= 1
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[2*i+1] - 0}) # y >= 0
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[2*i+1]}) # y <= 1
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[2*n_circles+i] - 0})  # r >= 0

    # Run optimization
    # Very small inner network trying to improve
    res = minimize(obj, x_guess, constraints=constraints, bounds=bounds,
                   method='trust-constr', options={'maxiter': 100, 'disp': False})

    if res.success:
        final_centers = res.x[:2*n_circles].reshape(-1, 2)
        final_radii = res.x[2*n_circles:]
    else:
        final_centers = initial_centers.copy()
        final_radii = initial_radii.copy()
    return final_centers, final_radii

def run_packing():
    h = 0.07  # lattice spacing
    centers = generate_hex_lattice(h)
    R_wall = np.array([min(c[0], 1-c[0], c[1], 1-c[1]) for c in centers])
    edges = build_neighbour_list(centers, max_dist=0.4)
    
    # now use branch-and-bound approach
    selected_indices, radii_values = branch_and_bound(centers, R_wall, edges)
    if selected_indices is None:
        raise RuntimeError("No valid solution found in branch-and-bound")

    # Get final centers and radii from the discrete solution
    selected_centers = centers[selected_indices]
    selected_radii = radii_values

    # Now do continuous refinement
    refined_centers, refined_radii = local_continuous_refinement(selected_centers, selected_radii)
