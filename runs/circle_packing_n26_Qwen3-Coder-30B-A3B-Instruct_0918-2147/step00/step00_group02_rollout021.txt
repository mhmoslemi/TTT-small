import numpy as np
from scipy.optimize import minimize

def initial_lattice(n=26):
    # Generate hexagonal lattice points
    sqrt3 = np.sqrt(3)
    spacing = 1.0
    while True:
        rows = int(np.ceil(1.0 / (spacing * sqrt3 / 2))) + 1
        cols = int(np.ceil(1.0 / spacing)) + 1
        count = 0
        centers = []
        for i in range(rows):
            for j in range(cols):
                x = j * spacing
                y = i * spacing * sqrt3 / 2
                if i % 2 == 1:
                    x += spacing / 2
                if 0 <= x <= 1 and 0 <= y <= 1:
                    centers.append([x, y])
                    count += 1
                    if count >= n:
                        break
            if count >= n:
                break
        if count >= n:
            break
        spacing *= 0.95

    centers = np.array(centers[:n])
    radii = np.full(n, 0.5)

    # Compute admissible radii
    for i in range(n):
        x, y = centers[i]
        r = min(x, 1 - x, y, 1 - y)
        for j in range(n):
            if i != j:
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.hypot(dx, dy)
                r = min(r, dist / 2)
        radii[i] = max(0, r)

    return centers, radii

def pack(centers, radii):
    z = np.empty(2 * len(centers) + len(radii))
    z[0::3] = centers[:, 0]
    z[1::3] = centers[:, 1]
    z[2::3] = radii
    return z

def unpack(z):
    n = len(z) // 3
    centers = np.empty((n, 2))
    centers[:, 0] = z[0::3]
    centers[:, 1] = z[1::3]
    radii = z[2::3]
    return centers, radii

def objective(z):
    _, radii = unpack(z)
    return -np.sum(radii)

def constraint_boundary(z):
    centers, radii = unpack(z)
    constraints = []
    for i in range(len(centers)):
        x, y = centers[i]
        r = radii[i]
        # left boundary: r - x <= 0
        constraints.append(r - x)
        # right boundary: r + x - 1 <= 0
        constraints.append(r + x - 1)
        # bottom boundary: r - y <= 0
        constraints.append(r - y)
        # top boundary: r + y - 1 <= 0
        constraints.append(r + y - 1)
    return np.array(constraints)

def constraint_nonoverlap(z):
    centers, radii = unpack(z)
    constraints = []
    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.hypot(dx, dy) + 1e-9  # prevent division by zero
            r_sum = radii[i] + radii[j]
            constraints.append(r_sum - dist)
    return np.array(constraints)

def repair(centers, radii):
    # Simple iterative repair process
    max_iter = 10
    for _ in range(max_iter):
        improved = False
        # Repair boundary constraints
        for i in range(len(centers)):
            x, y = centers[i]
            r = radii[i]
            r_new = min(r, x, 1 - x, y, 1 - y)
            if abs(r_new - r) > 1e-9:
                radii[i] = r_new
                improved = True

        # Repair overlaps
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.hypot(dx, dy)
                r_sum = radii[i] + radii[j]
                if dist < r_sum:
                    overlap = r_sum - dist
                    radii[i] -= overlap / 2
                    radii[j] -= overlap / 2
                    improved = True
        if not improved:
            break
    # Ensure non-negativity
    radii = np.maximum(0, radii)
    return centers, radii

def run_packing():
    centers, radii = initial_lattice(26)
    z0 = pack(centers, radii)

    # Define constraints and bounds
    cons = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_nonoverlap}
    ]

    bounds = [(0, 1) for _ in range(26 * 2)] + [(0, 0.5) for _ in range(26)]

    # Run optimization
    res = minimize(objective, z0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': 500, 'ftol': 1e-9}, disp=False)

    if not res.success:
        raise RuntimeError("Optimization failed")

    centers, radii = unpack(res.x)
    # Post-process for robustness
    centers, radii = repair(centers, radii)

    # Recheck assumptions
    success, msg = validate_packing(centers, radii)
    if not success:
        raise RuntimeError("Validation failed: " + msg)

    sum_radii = np.sum(radii)
    if sum_radii < 2.636:
        raise RuntimeError("Sum of radii insufficient:" + str(sum_radii))

    return centers, radii, sum_radii
