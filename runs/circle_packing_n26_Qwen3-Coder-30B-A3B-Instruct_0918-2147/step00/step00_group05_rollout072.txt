import numpy as np

def hex_lattice(d):
    """Generate hexagonal lattice points within unit square."""
    rows = []
    y = 0.0
    dy = d * np.sqrt(3) / 2
    while y <= 1.0:
        row = []
        x = 0.0
        dx = d
        offset = 0.0 if len(rows) % 2 == 0 else dx / 2
        while x <= 1.0:
            row.append([x + offset, y])
            x += dx
        if row:
            rows.extend(row)
        y += dy
    return np.array(rows)

def add_border_circles(points, d):
    """Append 6 border circles to the core points."""
    r = d / 4
    border = [
        [r, r],
        [1 - r, r],
        [r, 1 - r],
        [1 - r, 1 - r],
        [0.5, r],
        [0.5, 1 - r]
    ]
    return np.vstack([points, border])

def initial_radii(points):
    """Compute a uniform conservative initial radius."""
    wall_distances = np.minimum(points[:, 0], 1 - points[:, 0])
    wall_distances = np.minimum(wall_distances, points[:, 1])
    wall_distances = np.minimum(wall_distances, 1 - points[:, 1])
    min_distance = np.inf
    for i in range(len(points)):
        for j in range(len(points)):
            if i != j:
                dist = np.linalg.norm(points[i] - points[j])
                if dist < min_distance:
                    min_distance = dist
    r0 = min(np.min(wall_distances), min_distance / 2)
    return r0

def inflate_radii(centers, radii):
    """Perform linear feasibility max radius update."""
    n = len(centers)
    D = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    new_radii = np.zeros_like(radii)
    for i in range(n):
        wall_caps = [
            centers[i][0],
            1 - centers[i][0],
            centers[i][1],
            1 - centers[i][1]
        ]
        dist_caps = []
        for j in range(n):
            if i != j:
                dist_caps.append(D[i, j] - radii[j])
        caps = wall_caps + dist_caps
        new_radii[i] = min(caps)
    return np.maximum(radii, new_radii)

def relax_centers(centers, radii, steps=10):
    """Apply force-based relaxation with wall repulsion."""
    κ = 0.02
    λ = 0.01
    α = 0.5
    β = 0.02
    γ = 0.02
    for _ in range(steps):
        D = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
        forces = np.zeros_like(centers)
        for i in range(len(centers)):
            for j in range(len(centers)):
                if i == j: continue
                dist = D[i, j]
                delta = radii[i] + radii[j] - dist
                if dist > 0:
                    f = κ * max(0, β - dist) * (centers[i] - centers[j]) / dist
                    forces[i] += f
            # Wall forces
            fx = 0
            fx += -λ * max(0, γ - (centers[i][0] - radii[i]))
            fx += λ * max(0, γ - (1 - centers[i][0] - radii[i]))
            fy = 0
            fy += -λ * max(0, γ - (centers[i][1] - radii[i]))
            fy += λ * max(0, γ - (1 - centers[i][1] - radii[i]))
            forces[i][0] += fx
            forces[i][1] += fy
        new_centers = centers + α * forces
        # Clamp to valid positions
        new_centers[:, 0] = np.clip(new_centers[:, 0], radii, 1 - radii)
        new_centers[:, 1] = np.clip(new_centers[:, 1], radii, 1 - radii)
        # Check for overlaps
        new_D = np.linalg.norm(new_centers[:, None, :] - new_centers[None, :, :], axis=2)
        overlap = False
        for i in range(len(new_centers)):
            for j in range(i+1, len(new_centers)):
                if new_D[i, j] < radii[i] + radii[j] - 1e-8:
                    overlap = True
                    break
            if overlap:
                break
        if overlap:
            break
        centers = new_centers
    return centers

def run_packing():
    # Step 1: Generate pattern
    d = 0.18
    core_points = hex_lattice(d)
    # Keep first 20 points
    core_points = core_points[:20]
    all_points = add_border_circles(core_points, d)
    r0 = initial_radii(all_points)
    radii = np.full(len(all_points), r0)
    centers = all_points
    # Step 2: Iterate growth and relaxation
    max_outer = 30
    previous_sum = np.sum(radii)
    for _ in range(max_outer):
        radii = inflate_radii(centers, radii)
        centers = relax_centers(centers, radii)
        current_sum = np.sum(radii)
        if current_sum - previous_sum < 1e-6:
            break
        previous_sum = current_sum
    # Final validation
    valid, msg = validate_packing(centers, radii)

    return centers, radii, np.sum(radii)
