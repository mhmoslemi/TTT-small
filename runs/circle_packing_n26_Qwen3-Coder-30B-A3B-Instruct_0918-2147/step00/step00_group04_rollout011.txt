import numpy as np
from scipy.optimize import linprog
from math import sqrt

def initial_hex_centers(k, spacing):
    """Generate initial centers using a hexagonal lattice scaled to [0,1]^2."""
    sqrt3 = sqrt(3)
    points = []
    i = 0
    j = 0
    while len(points) < k:
        x = i * spacing + (j % 2) * spacing / 2
        y = j * spacing * sqrt3 / 2
        if 0 <= x <= 1 and 0 <= y <= 1:
            points.append([x, y])
        if x > 1:
            j += 1
            i = 0
        else:
            i += 1
    return np.array(points[:k])

def initial_radii(k, r0):
    """Initialize all radii to the same small value."""
    return np.full(k, r0)

def build_lp_constants(centers, radii):
    """Build constants needed for the LP."""
    n = centers.shape[0]
    # Wall distances
    d_left = centers[:, 0]
    d_right = 1 - centers[:, 0]
    d_bottom = centers[:, 1]
    d_top = 1 - centers[:, 1]
    # Pairwise center distances
    D = np.zeros((n, n))
    for i in range(n):
        for j in range(i+1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = sqrt(dx*dx + dy*dy)
            D[i, j] = d
            D[j, i] = d
    return d_left, d_right, d_bottom, d_top, D

def solve_radius_lp(consts, radii, graph, alpha):
    """Solve the linear programming problem for radius increments."""
    d_left, d_right, d_bottom, d_top, D = consts
    n = len(radii)
    # Create A_ub and b_ub for LP problem
    # 1. Wall constraints: Δr_i <= min(d_i_left, d_i_right, d_i_bottom, d_i_top) - r_i
    A_ub = []
    b_ub = []
    for i in range(n):
        wall_capacity = min(d_left[i], d_right[i], d_bottom[i], d_top[i]) - radii[i]
        if wall_capacity < 0:
            wall_capacity = 0
        A_ub.append([-1.0])  # -Δr_i <= wall_capacity
        b_ub.append(-wall_capacity)
    # 2. Pairwise constraints: Δr_i + Δr_j <= D_ij - (r_i + r_j)
    for i in range(n):
        for j in range(i + 1, n):
            pair_capacity = D[i, j] - (radii[i] + radii[j])
            if pair_capacity < 0:
                pair_capacity = 0
            # Add constraint: Δr_i + Δr_j <= pair_capacity
            row = [0.0] * n
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(pair_capacity)
    # 3. Optional graph-based constraints (limited use)
    # They can introduce non-linearity but we keep it simple here.
    # The LP assumes no conflicts in this version,
    # so we don't add any such constraints
    # Objective: maximize sum(Δr_i) => minimize -sum(Δr_i)
    c = [-1.0] * n
    # Solve LP
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    if res.success:
        return res.x
    else:
        # Fallback in case of failure
        return np.zeros(n)

def apply_increment(radii, Δr):
    """Update radii with increments."""
    return radii + Δr

def push_feasibility(centers, radii):
    """Perform exact geometric push to maintain feasibility."""
    n = len(radii)
    # If overflows
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            radii[i] = x
        if x + r > 1:
            radii[i] = 1 - x
        if y - r < 0:
            radii[i] = y
        if y + r > 1:
            radii[i] = 1 - y

    # Pairwise overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = sqrt(dx*dx + dy*dy)
            sum_radii = radii[i] + radii[j]
            if dist < sum_radii:
                overlap = sum_radii - dist
                delta = overlap / 2
                # Compute normalized direction vector
                if dist > 1e-15:
                    scale = delta / dist
                    dxn = dx * scale
                    d yn = dy * scale
                else:
                    # Avoid division by zero
                    dxn = 0
                    dy = 0
                centers[i, 0] -= dxn
                centers[i, 1] -= d yn
                centers[j, 0] += dxn
                centers[j, 1] += d yn
                # Clamp to ensure they haven't exceeded bounds
                x, y = centers[i]
                r_i = radii[i]
                if x - r_i < 0:
                    radii[i] = x
                if x + r_i > 1:
                    radii[i] = 1 - x
                x, y = centers[j]
                r_j = radii[j]
                if x - r_j < 0:
                    radii[j] = x
                if x + r_j > 1:
                    radii[j] = 1 - x
    return centers, radii

def run_iterations(centers, radii, max_iter, ε, α):
    """Main iterative loop."""
    for _ in range(max_iter):
        consts = build_lp_constants(centers, radii)
        Δr = solve_radius_lp(consts, radii, None, α)
        old_sum = np.sum(radii)
        radii = apply_increment(radii, Δr)
        centers, radii = push_feasibility(centers, radii)
        new_sum = np.sum(radii)
        if new_sum - old_sum < ε:
            break
    return centers, radii

def final_refinement(centers, radii):
    """Do final high precision LP and clipping."""
    consts = build_lp_constants(centers, radii)
    Δr = solve_radius_lp(consts, radii, None, 0.5)
    radii = apply_increment(radii, Δr)
    # Very precise clipping
    for i in range(len(radii)):
        r = radii[i]
        x, y = centers[i]
        r = min(r, x, 1 - x, y, 1 - y)
        radii[i] = r
    # Double-check pairwise overlaps
    n = len(radii)
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = sqrt(dx*dx + dy*dy)
            sum_radii = radii[i] + radii[j]
            if dist < sum_radii:
                delta = sum_radii - dist
                radii[i] -= delta / 2 - 1e-13
                radii[j] -= delta / 2 - 1e-13
                if radii[i] < 1e-12:
                    radii[i] = 1e-12
                if radii[j] < 1e-12:
                    radii[j] = 1e-12
    return centers, radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Run the full packing algorithm."""
    # Initialize parameters
    n = 26
    spacing = 0.12
    r0 = 0.005
    max_iter = 200
    ε = 1e-5
    α = 0.5

    # Step 1: Generate initial centers
    centers = initial_hex_centers(n, spacing)

    # Step 2: Initialize radii
    radii = initial_radii(n, r0)

    # Step 3: Perform main iterations
    centers, radii = run_iterations(centers, radii, max_iter, ε, α)

    # Step 4: Final refinement
    centers, radii = final_refinement(centers, radii)

    # Return results
    total_sum = np.sum(radii)
    return centers, radii, total_sum
