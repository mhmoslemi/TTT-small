import numpy as np
from scipy.optimize import linprog, minimize
from math import sqrt, ceil, floor

def build_hex_lattice(d):
    """Generate a hexagonal lattice with at least d points inside the unit square."""
    # Hexagon packing with spacing s
    # For a hexagonal lattice, the spacing is s = 2 * r, where r is the radius
    # The number of rows and columns are calculated appropriately
    r = 1.0 / (2 * sqrt(3)) * 2  # approximate radius to get enough points in [0,1]^2
    s = 2 * r  # spacing between centers in the hexagonal lattice
    spacing = s
    
    # Center spacing in x and y directions
    x_spacing = spacing
    y_spacing = spacing * sqrt(3) / 2

    # Initial guess at how many rows and cols we need
    cols = int(ceil(1.0 / x_spacing))
    rows = int(ceil(1.0 / y_spacing))

    centers = []
    y = 0.0
    row = 0
    while y <= 1.0:
        x_start = (row % 2) * x_spacing / 2  # offset every other row
        x = x_start
        while x <= 1.0:
            if 0 <= x <= 1.0 and 0 <= y <= 1.0:
                centers.append([x, y])
            x += x_spacing
        y += y_spacing
        row += 1

    # If we don't have enough, expand slightly
    if len(centers) < d:
        # Generate a denser grid manually to get exactly d points
        step = 1.0 / max(1, int(ceil(sqrt(d))))
        centers = []
        for i in range(0, int(ceil(1.0 / step))+1):
            for j in range(0, int(ceil(1.0 / step))+1):
                x = i * step
                y = j * step
                if 0 <= x <= 1.0 and 0 <= y <= 1.0:
                    centers.append([x, y])
    # Take only first d points
    centers = centers[:d]
    return np.array(centers)

def lp_max_radii(centers):
    """
    Compute the maximum sum of radii for fixed centers using linear programming.
    Returns (radii, sum_radii)
    """
    n = centers.shape[0]
    # Wall constraints
    A_ub_wall = []
    b_ub_wall = []
    for i in range(n):
        x, y = centers[i]
        A_ub_wall.append([-1, 0])  # -r_i <= -x_i => r_i >= x_i
        b_ub_wall.append(-x)
        A_ub_wall.append([1, 0])   # r_i <= 1 - x_i => r_i - (1 - x_i) <= 0
        b_ub_wall.append(1 - x)
        A_ub_wall.append([0, -1])   # r_i >= y_i
        b_ub_wall.append(-y)
        A_ub_wall.append([0, 1])    # r_i <= 1 - y_i
        b_ub_wall.append(1 - y)

    # Pairwise overlap constraints
    A_ub_overlap = []
    b_ub_overlap = []
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            A_ub_overlap.append([1, 1])  # r_i + r_j <= dist
            b_ub_overlap.append(dist)
            A_ub_overlap.append([-1, -1]) # r_i + r_j >= 0 => -r_i - r_j <= 0
            b_ub_overlap.append(0)

    A_ub = np.vstack((A_ub_wall, A_ub_overlap))
    b_ub = np.hstack((b_ub_wall, b_ub_overlap))

    # Objective: maximize sum of radii
    c = -np.ones(n)  # -c because linprog minimizes

    # Solve LP
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'presolve': True, 'tolerance': 1e-9})
    if res.status != 0:
        # Fallback: return small radii to prevent crashing
        return np.zeros(n), 0.0

    radii = res.x
    sum_radii = np.sum(radii)
    return radii, sum_radii

def perturb_centers(centers, sigma):
    """Add Gaussian noise and clamp to [0,1]."""
    perturbed = centers + np.random.normal(0, sigma, centers.shape)
    return np.clip(perturbed, 0, 1)

def anneal_centers(initial_centers):
    """Anneal centre positions using simulated annealing."""
    # Annealing schedule
    # T, sigma, iterations
    steps = [
        (0.02, 0.05, 2000),
        (0.01, 0.03, 4000),
        (0.005, 0.015, 3000),
        (0.002, 0.008, 2000)
    ]
    current_centers = initial_centers.copy()
    current_radii, current_sum = lp_max_radii(current_centers)
    best_centers = current_centers.copy()
    best_radii = current_radii.copy()
    best_sum = current_sum
    np.random.seed(42)

    for T, sigma, num_iter in steps:
        for _ in range(num_iter):
            new_centers = perturb_centers(current_centers, sigma)
            new_radii, new_sum = lp_max_radii(new_centers)
            if new_sum >= current_sum:
                current_centers = new_centers
                current_radii = new_radii
                current_sum = new_sum
                if new_sum > best_sum:
                    best_centers = new_centers
                    best_radii = new_radii
                    best_sum = new_sum
            else:
                # Metropolis acceptance criterion
                if np.random.uniform() < np.exp((new_sum - current_sum) / T):
                    current_centers = new_centers
                    current_radii = new_radii
                    current_sum = new_sum

    return best_centers, best_radii, best_sum

def local_refine(centers, radii):
    """Refine the configuration with a local optimization using trust-constr."""
    n = len(centers)
    init_x = np.concatenate([centers.flatten(), radii])

    def objective(x):
        """Minimize negative sum of radii (we want to maximize it)."""
        return -np.sum(x[n*2:])

    def grad(x):
        """Gradient of the objective function."""
        g = np.zeros_like(x)
        g[n*2:] = -1.0  # gradient of sum(radii) wrt radii
        return g

    def constraint_func(x):
        """Test function that encodes all constraints."""
        centers_x = x[:n*2].reshape(-1, 2)
        radii_x = x[n*2:]
        constraints = []

        # Wall constraints
        for i in range(n):
            x_c = centers_x[i][0]
            y_c = centers_x[i][1]
            r = radii_x[i]
            # x - r >= 0
            constraints.append(x_c - r)
            # 1 - x - r >= 0
            constraints.append(1 - x_c - r)
            # y - r >= 0
            constraints.append(y_c - r)
            # 1 - y - r >= 0
            constraints.append(1 - y_c - r)

        # Non-overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_x[i][0] - centers_x[j][0]
                dy = centers_x[i][1] - centers_x[j][1]
                dist_sq = dx*dx + dy*dy
                r_sum = radii_x[i] + radii_x[j]
                # distance squared > (r_sum)^2
                constraints.append(r_sum*r_sum - dist_sq)

        return np.array(constraints)

    def jacobian(x):
        """Jacobian of the constraint vector."""
        centers_x = x[:n*2].reshape(-1, 2)
        radii_x = x[n*2:]
        J = np.zeros((len(constraint_func(x)), len(x)))

        # Wall Jacobian
        count = 0
        j = 0  # row of Jacobian
        for i in range(n):
            x_c = centers_x[i][0]
            y_c = centers_x[i][1]
            r = radii_x[i]
            
            # x - r >= 0
            J[count][j] = 1
            J[count][n*2 + i] = -1
            count += 1
            
            # 1 - x - r >= 0
            J[count][j] = -1
            J[count][n*2 + i] = -1
            count += 1
            
            # y - r >= 0
            J[count][j+1] = 1
            J[count][n*2 + i] = -1
            count += 1
            
            # 1 - y - r >= 0
            J[count][j+1] = -1
            J[count][n*2 + i] = -1
            count += 1
            
            j += 2

        # Non-overlap Jacobian
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_x[i][0] - centers_x[j][0]
                dy = centers_x[i][1] - centers_x[j][1]
                r_sum = radii_x[i] + radii_x[j]
                # Need to differentiate (r_sum)^2 - (dx^2 + dy^2)

                # For dimension i
                J[count][i*2] = 2*dx
                J[count][i*2 + 1] = 2*dy
                J[count][n*2 + i] = 2*r_sum

                # For dimension j
                J[count][j*2] = -2*dx
                J[count][j*2 + 1] = -2*dy
                J[count][n*2 + j] = 2*r_sum

                count += 1

        return J

    bounds = []
    for i in range(n):
        bounds.append((0, 1))  # x
        bounds.append((0, 1))  # y
    for i in range(n):
        bounds.append((0, None))  # radii

    # Back to the constrained problem
    def fun_constraints(x):
        return constraint_func(x)

    # See that r_i should not be smaller than the original
    # So we must impose lower bounds on r_i.
    lower_bounds_r = radii.copy()
    for i in range(n):
        bounds[n*2 + i] = (lower_bounds_r[i], None)

    # Use scipy.optimize.minimize with trust-constr
    res = minimize(objective, init_x, method='trust-constr', jac=grad,
                   constraints={'type': 'ineq', 'fun': fun_constraints, 'jac': jacobian},
                   bounds=bounds, options={'disp': False})

    if res.success:
        refined_centers = res.x[0:n*2].reshape(-1, 2)
        refined_radii = res.x[n*2:]
        return refined_centers, refined_radii
    else:
        return centers, radii

    centers_lattice = build_hex_lattice(26)
