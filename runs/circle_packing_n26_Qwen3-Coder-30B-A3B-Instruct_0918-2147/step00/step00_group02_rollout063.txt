import numpy as np
from scipy.optimize import minimize
from math import sqrt

def initial_lattice(n=26):
    # Generate a hexagonal lattice that fits in [0,1]x[0,1]
    # We'll start with a coarse grid and then refine
    spacing = 0.2
    rows = int(np.ceil(1.0 / spacing))
    cols = int(np.ceil(1.0 / (spacing * sqrt(3) / 2)))
    
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = j * spacing
            y = i * spacing * sqrt(3) / 2
            if i % 2 == 1:
                x += spacing / 2
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers.append([x, y])
    
    # If we don't have enough, let's expand the grid        
    while len(centers) < n:
        spacing *= 0.9
        centers = []
        rows = int(np.ceil(1.0 / spacing))
        cols = int(np.ceil(1.0 / (spacing * sqrt(3) / 2)))
        for i in range(rows):
            for j in range(cols):
                x = j * spacing
                y = i * spacing * sqrt(3) / 2
                if i % 2 == 1:
                    x += spacing / 2
                if 0 <= x <= 1 and 0 <= y <= 1:
                    centers.append([x, y])
    
    centers = np.array(centers[:n])
    
    # Compute admissible radii based on distance to sides and neighbors
    radii = []
    for i in range(len(centers)):
        x, y = centers[i]
        r = min(x, 1-x, y, 1-y)  # distance to sides
        
        for j in range(len(centers)):
            if i != j:
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = sqrt(dx*dx + dy*dy)
                r = min(r, dist / 2.0)
        radii.append(max(0, r))
    
    return centers, np.array(radii)

def unpack(z):
    n = len(z) // 3
    centers = z[0:2*n].reshape((n, 2))
    radii = z[2*n:3*n]
    return centers, radii

def pack(centers, radii):
    return np.concatenate([centers.flatten(), radii])

def objective(z):
    centers, radii = unpack(z)
    return -np.sum(radii)

def constraint_boundary(z):
    centers, radii = unpack(z)
    n = len(centers)
    constraints = []
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # left, right, bottom, top
        constraints.extend([
            r - x,  # x - r >= 0
            r + x - 1.0,  # x + r <= 1
            r - y,  # y - r >= 0
            r + y - 1.0   # y + r <= 1
        ])
    return np.array(constraints)

def constraint_nonoverlap(z):
    centers, radii = unpack(z)
    n = len(centers)
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist = sqrt(dx*dx + dy*dy)
            r_sum = radii[i] + radii[j]
            constraints.append(r_sum - dist)
    return np.array(constraints)

def repair(centers, radii):
    # Check boundaries and fix any violations
    for i in range(len(centers)):
        x, y = centers[i]
        r = radii[i]
        r = min(r, x, 1-x, y, 1-y)
        radii[i] = max(0, r)

    # Check overlaps and fix
    changed = True
    iterations = 0
    while changed and iterations < 10:
        changed = False
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.hypot(dx, dy)
                r_sum = radii[i] + radii[j]
                if dist < r_sum:
                    overlap = r_sum - dist
                    shrink = overlap / 2.0
                    if radii[i] > shrink and radii[j] > shrink:
                        radii[i] -= shrink
                        radii[j] -= shrink
                        changed = True
                    else:
                        # If one is too small, just shrink both
                        radii[i] = max(0, radii[i] - overlap)
                        radii[j] = max(0, radii[j] - overlap)
                        changed = True
        iterations += 1

    # Final boundary check
    for i in range(len(centers)):
        x, y = centers[i]
        r = radii[i]
        r = min(r, x, 1-x, y, 1-y)
        radii[i] = max(0, r)
        
    return centers, radii

def run_packing():
    # Initial
    centers, radii = initial_lattice(26)
    z0 = pack(centers, radii)
    
    # Prepare bounds
    bounds = [(0, 1), (0, 1), (0, 0.5)] * 26
    constraints = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_nonoverlap}
    ]

    # Solve with trust-constr
    result = minimize(objective, z0, method='trust-constr', 
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': 1000, 'ftol': 1e-9})

    # Extract final solution
    final_centers, final_radii = unpack(result.x)
    
    # Repair if needed
    final_centers, final_radii = repair(final_centers, final_radii)
    
    sum_radii = np.sum(final_radii)
    
    # Validate final result
    valid, msg = validate_packing(final_centers, final_radii)
    if not valid:
        raise RuntimeError(f"Validation failed after optimization: {msg}")
    
    if sum_radii < 2.636:
        # Try a second optimization with random perturbation
        centers_perturbed = final_centers + np.random.uniform(-0.01, 0.01, (26, 2))
        # Keep within bounds
        centers_perturbed = np.clip(centers_perturbed, 0, 1)
        z_pert = pack(centers_perturbed, final_radii)
        result2 = minimize(objective, z_pert, method='trust-constr', 
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': 500, 'ftol': 1e-9})
        final_centers2, final_radii2 = unpack(result2.x)
        final_centers2, final_radii2 = repair(final_centers2, final_radii2)
        sum_radii2 = np.sum(final_radii2)
        if sum_radii2 > sum_radii:
            final_centers, final_radii = final_centers2, final_radii2
            sum_radii = sum_radii2

    if np.sum(final_radii) < 2.636:
        raise RuntimeError("Failed to reach sum of radii target")

    return final_centers, final_radii, sum_radii
