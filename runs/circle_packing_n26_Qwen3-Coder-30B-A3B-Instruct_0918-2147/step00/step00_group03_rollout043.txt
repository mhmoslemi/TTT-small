import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
from itertools import combinations

def init_candidate_grid(R, Δ):
    """Generate a list of candidate (center, radius) pairs."""
    candidates = []
    for r in R:
        # Grid spacing to ensure no overlap for same radius
        s = r * np.sqrt(2)
        # Generate grid within [r, 1-r]^2
        x_vals = np.arange(r, 1 - r + 1e-10, s)
        y_vals = np.arange(r, 1 - r + 1e-10, s)
        for x in x_vals:
            for y in y_vals:
                candidates.append(((x, y), r))
    return candidates

def max_radii_fixed_centres(centers, radii_initial):
    """Maximize radii given fixed centers (projection step)."""
    n = len(centers)
    radii = radii_initial.copy()
    eps = 1e-12
    
    # Ensure centers are valid
    for i in range(n):
        cx, cy = centers[i]
        r = radii[i]
        radii[i] = min(r, cx, 1 - cx, cy, 1 - cy)
    
    # Iterative improvement
    for _ in range(20):  # Limit iterations
        changed = False
        for i in range(n):
            cx, cy = centers[i]
            # Minimum allowed by bounds
            r_min = min(cx, 1 - cx, cy, 1 - cy)
            r_old = radii[i]
            r_new = r_min
            # For other circles, find max radii subject to non-overlap
            for j in range(n):
                if i != j:
                    dist = np.sqrt((cx - centers[j][0]) ** 2 + (cy - centers[j][1]) ** 2)
                    r_limit = dist - radii[j] - eps
                    if r_limit > 0:
                        r_new = min(r_new, r_limit)
            # Also limit by current value
            r_new = min(r_new, r_old + 1e-4)
            if r_new > radii[i]:
                changed = True
                radii[i] = r_new
        if not changed:
            break
    return radii

def nlp_refine(centers, radii_initial):
    """Use trust-constr to refine centers and radii."""
    # Decision variables: [x1, y1, ..., x26, y26, r1, ..., r26]
    n = len(centers)
    x0 = np.concatenate([centers.flatten(), radii_initial])
    # Objective function to minimize negative sum of radii
    def objective(v):
        return -np.sum(v[2*n:])

    def jac(v):
        return np.concatenate([np.zeros(2*n), -np.ones(n)])

    # Constraints
    def wall_ineq(v):
        centers_v = v[:2*n].reshape((n, 2))
        radii_v = v[2*n:]
        res = []
        # r - x <= 0  => r - x <= 0
        res.extend(radii_v - centers_v[:, 0])
        # r + x - 1 <= 0  => r + x - 1 <= 0
        res.extend(radii_v + centers_v[:, 0] - 1)
        # r - y <= 0
        res.extend(radii_v - centers_v[:, 1])
        # r + y - 1 <= 0
        res.extend(radii_v + centers_v[:, 1] - 1)
        return np.array(res)

    def wall_jac(v):
        J = np.zeros((4*n, 2*n + n))
        for i in range(n):
            # r - x <= 0
            J[4*i, 2*i + 0] = -1
            J[4*i, 2*n + i] = 1
            # r + x - 1 <= 0
            J[4*i + 1, 2*i + 0] = 1
            J[4*i + 1, 2*n + i] = 1
            # r - y <= 0
            J[4*i + 2, 2*i + 1] = -1
            J[4*i + 2, 2*n + i] = 1
            # r + y - 1 <= 0
            J[4*i + 3, 2*i + 1] = 1
            J[4*i + 3, 2*n + i] = 1
        return J

    def overlap_ineq(v):
        centers_v = v[:2*n].reshape((n, 2))
        radii_v = v[2*n:]
        res = []
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers_v[i] - centers_v[j]) ** 2))
                # r_i + r_j - dist <= 0
                res.append(radii_v[i] + radii_v[j] - dist)
        return np.array(res)

    def overlap_jac(v):
        centers_v = v[:2*n].reshape((n, 2))
        radii_v = v[2*n:]
        J = np.zeros((n*(n-1)//2, 2*n + n))
        idx = 0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_v[i][0] - centers_v[j][0]
                dy = centers_v[i][1] - centers_v[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist > 1e-12:
                    dxd = dx / dist
                    dyd = dy / dist
                else:
                    dxd = 0
                    dyd = 0
                J[idx, 2*i + 0] = -dxd
                J[idx, 2*i + 1] = -dyd
                J[idx, 2*j + 0] = dxd
                J[idx, 2*j + 1] = dyd
                J[idx, 2*n + i] = 1
                J[idx, 2*n + j] = 1
                idx += 1
        return J

    # We'll pass identity for jacs so trust-constr infers, or approximate
    constraints = [
        {'type': 'ineq', 'fun': wall_ineq, 'jac': wall_jac},
        {'type': 'ineq', 'fun': overlap_ineq, 'jac': overlap_jac},
    ]

    bounds = []
    # Bounds for centers
    for _ in range(n):
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 1.0))
    # Bounds for radii
    for _ in range(n):
        bounds.append((0.0, 1.0))

    res = minimize(objective, x0, method='trust-constr',
        jac=jac, hess=None, constraints=constraints,
        bounds=bounds, options={'maxiter': 300, 'xtol': 1e-9, 'gtol': 1e-9})

    if not res.success:
        print("Warning: Trust-constr did not converge:", res.message)

    new_centers = res.x[:2*n].reshape((n, 2))
    new_radii = res.x[2*n:]
    
    # Enforce bounds for safety
    new_radii = np.clip(new_radii, 0, None)
    for i in range(n):
        new_centers[i] = np.clip(new_centers[i], 0, 1)
        new_radii[i] = min(new_radii[i], new_centers[i][0], 1 - new_centers[i][0],
                          new_centers[i][1], 1 - new_centers[i][1])
    
    return new_centers, new_radii
