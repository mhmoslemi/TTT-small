import numpy as np
import random

def make_hex_seed(n, margin):
    # Create hexagonal arrangement with different initial radii
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))
    centers = []
    radii = []
    
    # Hexagonal layout with centers
    y_spacing = 1.0 / (rows + 1)
    x_spacing = y_spacing * np.sqrt(3) / 2
    
    for i in range(rows):
        for j in range(cols):
            if len(centers) >= n:
                break
            x = (j + 0.5) * x_spacing
            y = (i + 0.5) * y_spacing
            if x < margin or x > 1 - margin or y < margin or y > 1 - margin:
                continue
            centers.append([x, y])
            # Set radius based on distance to nearest edge
            dist_to_edges = [x, 1-x, y, 1-y]
            min_edge_dist = min(dist_to_edges)
            radius = 0.015 + 0.03 * min_edge_dist  # Ensure minimal radius and vary per location
            radii.append(radius)
    
    # Pad if needed, truncate if over
    centers = centers[:n]
    radii = radii[:n]
    if len(centers) < n:
        # Add extra circles at random positions with minimal radius
        for _ in range(n - len(centers)):
            x, y = np.random.uniform(margin, 1 - margin, 2)
            centers.append([x, y])
            radii.append(0.015)
    
    return np.array(centers), np.array(radii)

def max_feasible_radius(i, centers, radii):
    # Compute exact maximal feasible radius for circle i with current configuration
    c_i = centers[i]
    r_i = radii[i]
    
    # Wall constraints
    d_L = c_i[0]
    d_R = 1.0 - c_i[0]
    d_B = c_i[1]
    d_T = 1.0 - c_i[1]
    
    # Neighbour constraints
    d_ij_min = float('inf')
    for j in range(len(centers)):
        if i == j:
            continue
        dist = np.linalg.norm(c_i - centers[j])
        d_ij = dist - radii[j]
        if d_ij < d_ij_min:
            d_ij_min = d_ij
    
    # Compute max feasible radius
    r_max = min(d_L, d_R, d_B, d_T)
    if d_ij_min < r_max:
        r_max = d_ij_min
    
    return r_max

def inflate_radii(centers, radii, alpha):
    old_sum = np.sum(radii)
    
    # Global inflation
    delta = alpha * np.mean(radii)
    new_radii = radii + delta
    
    # Clamp each radius to its maximum feasible value
    for i in range(len(radii)):
        max_feasible = max_feasible_radius(i, centers, new_radii)
        new_radii[i] = min(new_radii[i], max_feasible)
    
    new_sum = np.sum(new_radii)
    return new_radii, (new_sum - old_sum)

def anneal_centers(centers, radii, temp0, gamma, sigma0, steps):
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = np.sum(radii)
    
    temp = temp0
    sigma = sigma0
    
    for step in range(steps):
        # Occasionally global shift to avoid clustering
        if step % 50 == 0:
            shift = np.random.uniform(-0.001, 0.001, 2)
            centers += shift
        
        # Select random circle to modify
        i = random.randrange(len(centers))
        
        # Displace in random direction
        displacement = np.random.normal(0, sigma, 2)
        old_center = centers[i].copy()
        old_radius = radii[i]
        centers[i] += displacement
        
        # Clamp center to square
        centers[i] = np.clip(centers[i], 0, 1)
        
        # Recompute max feasible radius for this circle
        r_max = max_feasible_radius(i, centers, radii)
        new_radius = min(r_max, radii[i])  # don't go beyond new max
        
        # Accept or reject based on Boltzmann probability
        if new_radius > old_radius:
            radii[i] = new_radius
        else:
            # Compute acceptance probability
            delta = new_radius - old_radius
            if delta < -1e-10:  # Handle near-zero differences
                prob = np.exp(delta / temp)
                if random.random() < prob:
                    radii[i] = new_radius
                else:
                    centers[i] = old_center  # revert center
                    radii[i] = old_radius
        
        # Update best
        current_sum = np.sum(radii)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
        
        # Anneal temperature and sigma
        temp *= gamma
        sigma = sigma0 * temp / temp0
    
    return best_centers, best_radii, best_sum

def mixed_cycle(centers, radii, params):
    alpha = 0.04
    epsilon = 1e-6
    stagnation_count = 0
    max_stagnation = 8
    sa_steps = 200
    cycles_without_improvement = 0
    max_no_improvement = 5
    
    prev_sum = np.sum(radii)
    
    while stagnation_count < max_stagnation or cycles_without_improvement < max_no_improvement:
        # Phase B: inflation
        new_radii, delta_sum = inflate_radii(centers, radii, alpha)
        radii = new_radii
        
        if delta_sum < epsilon:
            stagnation_count += 1
        else:
            stagnation_count = 0
        prev_sum = np.sum(radii)
        
        # Reinitialize parameters if needed
        if stagnation_count >= max_stagnation:
            # Phase C: SA refinement
            centers, radii, sa_sum = anneal_centers(
                centers, radii,
                temp0=0.02,
                gamma=0.97,
                sigma0=0.03,
                steps=sa_steps
            )
            if sa_sum <= prev_sum:
                cycles_without_improvement += 1
            else:
                cycles_without_improvement = 0
            prev_sum = np.sum(radii)
            if alpha > 0.001:
                alpha *= 0.95  # decay inflation rate
            stagnation_count = 0  # reset
            continue
            
    return centers, radii

def polish_radii(centers, radii, beta, passes):
    # Run local polishing with gap filling and overlap resolution
    for _ in range(passes):
        old_sum = np.sum(radii)
        updated = True
        
        while updated:
            updated = False
            changed = [False] * len(centers)
            
            for i in range(len(centers)):
                # Get available gap
                c_i = centers[i]
                r_i = radii[i]
                
                # Compute wall gaps
                wall_gaps = [
                    c_i[0],                             # left
                    1.0 - c_i[0],                       # right
                    c_i[1],                             # bottom
                    1.0 - c_i[1]                        # top
                ]
                g_wall = min(wall_gaps)
                
                # Compute neighbour gaps
                g_nei = float('inf')
                for j in range(len(centers)):
                    if i == j:
                        continue
                    dist = np.linalg.norm(c_i - centers[j])
                    gap = dist - (r_i + radii[j])
                    if gap < g_nei:
                        g_nei = gap
                
                g = min(g_wall, g_nei)
                
                if g > 0:
                    new_r = r_i + beta * g
                    max_feasible = max_feasible_radius(i, centers, radii)
                    if new_r <= max_feasible:
                        radii[i] = new_r
                        changed[i] = True
                        updated = True
                    else:
                        # Clamp to maximum
                        radii[i] = max_feasible
                        changed[i] = True
                        updated = True
            
            # Detect overlaps and resolve
            resolve = True
            while resolve:
                resolve = False
                for i in range(len(centers)):
                    for j in range(i + 1, len(centers)):
                        dist = np.linalg.norm(centers[i] - centers[j])
                        overlap = (radii[i] + radii[j]) - dist
                        if overlap > 1e-10:
                            shrink = overlap / 2.0
                            radii[i] -= shrink
                            radii[j] -= shrink
                            resolve = True
                    
    # Doubles-check all constraints
    for i in range(len(centers)):
        max_feas = max_feasible_radius(i, centers, radii)
        if radii[i] > max_feas:
            radii[i] = max_feas
    
    return centers, radii
