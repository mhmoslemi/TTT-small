import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
from itertools import product

def init_candidate_grid(R, Δ):
    """Generate a list of candidate (center, radius) pairs."""
    candidates = []
    for r in R:
        # Grid spacing slightly larger than 2*r to allow for margin
        s = r * np.sqrt(2)
        x_coords = np.arange(r, 1 - r + s / 2, s)
        y_coords = np.arange(r, 1 - r + s / 2, s)
        for x, y in product(x_coords, y_coords):
            candidates.append(((x, y), r))
    return candidates

def build_overlap_matrix(candidates):
    """Build a boolean overlap matrix for candidate centers."""
    n = len(candidates)
    overlap = np.zeros((n, n), dtype=bool)
    centers = np.array([c[0] for c in candidates])
    radii = np.array([c[1] for c in candidates])
    dists = cdist(centers, centers)

    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j]:
                overlap[i, j] = True
                overlap[j, i] = True

    return overlap

def solve_discrete_bip(candidates, overlap_matrix, target_sum):
    """Solve the discrete BIP for a feasible configuration."""
    n = len(candidates)
    centers = np.array([c[0] for c in candidates])
    radii = np.array([c[1] for c in candidates])

    # Greedy selection to start
    selected = []
    used = set()
    total_radius = 0

    # Sort candidates by decreasing radius to prioritize large ones
    idxs = np.argsort(radii)[::-1]
    for idx in idxs:
        if len(selected) >= 26:
            break
        if idx not in used:
            # Check overlap with already selected
            valid = True
            for sel_idx in selected:
                if overlap_matrix[idx, sel_idx]:
                    valid = False
                    break
            if valid:
                selected.append(idx)
                used.add(idx)
                total_radius += radii[idx]

    # If not enough, try picking larger radius candidates with no overlaps
    remaining_idx = set(range(n)) - used
    if len(selected) < 26:
        # Two pass greedy to maximize sum before reaching 26
        for idx in idxs:
            if len(selected) >= 26:
                break
            if idx not in used:
                # Check overlap with already selected
                valid = True
                for sel_idx in selected:
                    if overlap_matrix[idx, sel_idx]:
                        valid = False
                        break
                if valid:
                    selected.append(idx)
                    used.add(idx)
                    total_radius += radii[idx]

    # Ensure we have 26
    if len(selected) < 26:
        needed = 26 - len(selected)
        for i in range(n):
            if i not in used and len(selected) < 26:
                selected.append(i)
                used.add(i)
                total_radius += radii[i]

    final_selected = [candidates[i] for i in selected]
    return final_selected

def max_radii_fixed_centres(centers, radii_initial):
    """Update radii to maximum allowable values given fixed centers."""
    new_radii = radii_initial.copy()
    for i in range(len(centers)):
        x, y = centers[i]
        r = new_radii[i]
        # Minimum margin to borders
        min_border = min(x, y, 1 - x, 1 - y)
        allowed_r = min_border
        # Check other circles for overlap
        for j in range(len(centers)):
            if i != j:
                xj, yj = centers[j]
                dist = np.sqrt((x - xj)**2 + (y - yj)**2)
                allowed_r = min(allowed_r, dist - new_radii[j])
        new_radii[i] = max(0, allowed_r)
    return new_radii

def nlp_refine(centers, radii_initial):
    """Refine using continuous optimization with trust-constr."""
    n = len(centers)
    x0 = np.concatenate((centers.flatten(), radii_initial))

    def func(x):
        return -np.sum(x[2*n:])

    def jac(x):
        grad = np.zeros_like(x)
        grad[2*n:] = -1.0
        return grad

    def cons(i, j):
        def c(x):
            xi, yi = x[i], x[i + n]
            xj, yj = x[j], x[j + n]
            ri, rj = x[i + 2*n], x[j + 2*n]
            dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
            return ri + rj - dist
        return c

    def jacobian(i, j):
        def jac_c(x):
            xi, yi = x[i], x[i + n]
            xj, yj = x[j], x[j + n]
            ri, rj = x[i + 2*n], x[j + 2*n]
            dx = (xi - xj) / np.sqrt((xi - xj)**2 + (yi - yj)**2)
            dy = (yi - yj) / np.sqrt((xi - xj)**2 + (yi - yj)**2)
            grad = np.zeros_like(x)
            grad[i] = -dx
            grad[i + n] = -dy
            grad[j] = dx
            grad[j + n] = dy
            grad[i + 2*n] = -1
            grad[j + 2*n] = -1
            return grad
        return jac_c

    constrs = []
    jacs = []

    # Wall constraints
    for i in range(n):
        x, y = centers[i]
        r = radii_initial[i]
        # r - x <= 0
        def wall1(x):
            return r - x[i]
        def wall1_grad(x):
            g = np.zeros_like(x)
            g[i] = -1
            return g
        constrs.append({'type': 'ineq', 'fun': wall1, 'jac': wall1_grad})

        # r + x - 1 <= 0
        def wall2(x):
            return r + x[i] - 1
        def wall2_grad(x):
            g = np.zeros_like(x)
            g[i] = 1
            return g
        constrs.append({'type': 'ineq', 'fun': wall2, 'jac': wall2_grad})

        # r - y <= 0
        def wall3(x):
            return r - x[i + n]
        def wall3_grad(x):
            g = np.zeros_like(x)
            g[i + n] = -1
            return g
        constrs.append({'type': 'ineq', 'fun': wall3, 'jac': wall3_grad})

        # r + y - 1 <= 0
        def wall4(x):
            return r + x[i + n] - 1
        def wall4_grad(x):
            g = np.zeros_like(x)
            g[i + n] = 1
            return g
        constrs.append({'type': 'ineq', 'fun': wall4, 'jac': wall4_grad})

    # Pairwise overlap constraints
    for i in range(n):
        for j in range(i+1, n):
            c = cons(i, j)
            j = jacobian(i, j)
            constrs.append({'type': 'ineq', 'fun': c, 'jac': j})

    # Solve
    res = minimize(func, x0, method='trust-constr', jac=jac, constraints=constrs, options={'maxiter': 300, 'gtol': 1e-9, 'xtol': 1e-9})

    if not res.success:
        # fallback to projecting to max radius
        centers_refine = res.x[:2*n].reshape(n, 2)
        radii_refine = res.x[2*n:]
        radii_refine = max_radii_fixed_centres(centers_refine, radii_refine)
        centers_refine = centers_refine / 1.0  # keep same once sorted out
        return centers_refine, radii_refine

    centers_refine = res.x[:2*n].reshape(n, 2)
    radii_refine = res.x[2*n:]

    # final projection step, must ensure no overlap
    radii_refine = max_radii_fixed_centres(centers_refine, radii_refine)
    return centers_refine, radii_refine

def run_packing():
    """Main function."""
    np.random.seed(42)
    R = np.arange(0.025, 0.12, 0.01)
    candidates = init_candidate_grid(R, 0.01)
    overlap = build_overlap_matrix(candidates)
    selected = solve_discrete_bip(candidates, overlap, 2.636)

    radii = np.array([c[1] for c in selected])

    # Initial projection to max radius
    radii = max_radii_fixed_centres(centers, radii)

    # Continuous refinement
    centers, radii = nlp_refine(centers, radii)

    # Final projection
