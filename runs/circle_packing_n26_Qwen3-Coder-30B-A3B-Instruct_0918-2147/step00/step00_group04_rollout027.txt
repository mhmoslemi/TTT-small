import numpy as np
import cvxpy as cp
from typing import Tuple, List, Optional, Any
import math

def build_leaf_problem(cell, k):
    x_min, y_min, x_max, y_max = cell
    # Add a small buffer for numerics
    x_min += 1e-8
    x_max -= 1e-8
    y_min += 1e-8
    y_max -= 1e-8
    
    # Variables for k circles
    centers = []
    radii = []
    for i in range(k):
        centers.append(cp.Variable(2))
        radii.append(cp.Variable(1, nonneg=True))
    
    # Constraints for all circles
    constraints = []
    
    # Boundary constraints
    for i in range(k):
        cx, cy = centers[i]
        r = radii[i]
        constraints.append(cx - r >= x_min)
        constraints.append(cx + r <= x_max)
        constraints.append(cy - r >= y_min)
        constraints.append(cy + r <= y_max)
    
    # Non-overlap constraints
    for i in range(k):
        for j in range(i+1, k):
            cx1, cy1 = centers[i]
            r1 = radii[i]
            cx2, cy2 = centers[j]
            r2 = radii[j]
            #SOC constraint
            dist = cp.sqrt((cx1-cx2)**2 + (cy1-cy2)**2)
            constraints.append(dist >= r1 + r2)
    
    # Maximize sum of radii
    objective = cp.Maximize(cp.sum(radii))
    
    prob = cp.Problem(objective, constraints)
    return prob, centers, radii


def solve_leaf(cell, k):
    prob, centers, radii = build_leaf_problem(cell, k)
    prob.solve(solver=cp.OSQP,verbose=False,eps_abs=1e-12,eps_rel=1e-12)
    if prob.status != cp.OPTIMAL:
        raise ValueError(f"Leaf SOCP solver failed with status: {prob.status}")
    
    sol_centers = []
    sol_radii = []
    for i in range(k):
        sol_centers.append(np.array([centers[i][0].value, centers[i][1].value]))
        sol_radii.append(float(radii[i].value))
    
    return np.array(sol_centers), np.array(sol_radii)

def hierarchical_merge(cells_info, adjacency_map):
    clusters = [(ci['centers'], ci['radii'], ci['cell']) for ci in cells_info]
    while len(clusters) > 1:
        # Just merge pairs greedily for simplicity
        new_clusters = []
        for i in range(0, len(clusters) - 1, 2):
            c1 = clusters[i]
            c2 = clusters[i+1]
            centers1, radii1, _ = c1
            centers2, radii2, _ = c2
            new_centers = np.vstack([centers1, centers2])
            new_radii = np.concatenate([radii1, radii2])
            new_clusters.append((new_centers, new_radii, None))
        if len(clusters) % 2 == 1:
            new_clusters.append(clusters[-1])
        clusters = new_clusters
    return clusters[0][0], clusters[0][1]


def global_polish(centers, radii):
    n = len(centers)
    # Set up barrier-based optimization with gradient ascent
    c = centers.copy()
    r = radii.copy()
    iteration = 0
    
    # Barrier function and gradient is analytically derivable but we use scipy
    # We will do simple gradient ascent with barriers
    
    step = 0.01
    mu = 1e-5
    while iteration < 3:
        counter = 0
        while counter < 50:
            counter += 1
            # Compute gradients
            i_grad = np.zeros_like(c)
            r_grad = np.zeros_like(r)
            for i in range(n):
                x_i, y_i = c[i]
                r_i = r[i]
                
                # Barrier term gradients
                i_grad[i, 0] -= mu * (1.0 / (x_i - r_i) - 1.0 / (1 - x_i - r_i))
                i_grad[i, 1] -= mu * (1.0 / (y_i - r_i) - 1.0 / (1 - y_i - r_i))
                r_grad[i] -= 1
                for j in range(n):
                    if i != j:
                        x_j, y_j = c[j]
                        r_j = r[j]
                        dx = x_j - x_i
                        dy = y_j - y_i
                        dist = np.sqrt(dx**2 + dy**2)
                        if dist > 0:
                            factor = mu / (dist - r_i - r_j)
                            i_grad[i, 0] += factor * dx / dist
                            i_grad[i, 1] += factor * dy / dist
                            r_grad[i] += factor
                    
            # Take step and project
            nc = c + step * i_grad
            nr = r + step * r_grad
            
            # Project
            for i in range(n):
                nc[i, 0] = np.clip(nc[i, 0], r[i] + 1e-10, 1 - r[i] - 1e-10)
                nc[i, 1] = np.clip(nc[i, 1], r[i] + 1e-10, 1 - r[i] - 1e-10)
                nr[i] = np.clip(nr[i], 1e-10, None)
                
            c = nc
            r = nr
        iteration += 1
    
    return c, r


def final_refine(centers, radii):
    # Re-solve big problem
    n = len(centers)
    x = cp.Variable((n, 2))
    r = cp.Variable(n, nonneg=True)
    constraints = []
    for i in range(n):
        x_i, y_i = x[i]
        r_i = r[i]
        # Apply box constraints
        constraints.append(x_i - r_i >= 1e-12)
        constraints.append(x_i + r_i <= 1 - 1e-12)
        constraints.append(y_i - r_i >= 1e-12)
        constraints.append(y_i + r_i <= 1 - 1e-12)
    
    # Pairwise constraints
    for i in range(n):
        for j in range(i+1, n):
            dist = cp.sqrt((x[i, 0] - x[j, 0])**2 + (x[i, 1] - x[j, 1])**2)
            constraints.append(dist >= r[i] + r[j] + 1e-12)
    
    objective = cp.Maximize(cp.sum(r))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.OSQP, verbose=False, eps_abs=1e-12, eps_rel=1e-12)
    
    if prob.status != cp.OPTIMAL:
        # Fallback if something went wrong
        return centers, radii
        
    new_centers = np.array([x[i].value for i in range(n)])
    new_radii = np.array([r[i].value for i in range(n)])
    
    # Ensure no overlaps
    for i in range(n):
        for j in range(i+1, n):
            dx = new_centers[i,0] - new_centers[j,0]
            dy = new_centers[i,1] - new_centers[j,1]
            dist = np.sqrt(dx*dx + dy*dy)
            required = new_radii[i] + new_radii[j]
            if dist < required - 1e-13:
                # For safety, reduce radius to make contact
                new_radii[i] -= (required - dist) * 0.5
                new_radii[j] -= (required - dist) * 0.5
                new_radii[i] = max(0, new_radii[i])
                new_radii[j] = max(0, new_radii[j])
    
    return new_centers, new_radii


def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Decompose into 9 rectangular cells
    
    # 4 corners: 0.25 x 0.25
    # 4 edges: 0.25 x 0.5  
    # 1 center: 0.5 x 0.5
    
    # Nominal allocation: 5 1 2
    # But we will instead simply all start with a grid-like tiling and optimize
    
    # 9 regions
    cells = []
    cell_info = []
    
    a = 0.25
    # Corners
    cells.append([0, 0, a, a])                # Bottom-left
    cells.append([1-a, 0, 1, a])              # Bottom-right  
    cells.append([0, 1-a, a, 1])              # Top-left
    cells.append([1-a, 1-a, 1, 1])            # Top-right
    
    # Edges
    cells.append([a, 0, 1-a, a])              # Bottom edge
    cells.append([a, 1-a, 1-a, 1])            # Top edge
    cells.append([0, a, a, 1-a])              # Left edge
    cells.append([1-a, a, 1, 1-a])            # Right edge
    
    # Center
    cells.append([a, a, 1-a, 1-a])
    
    # Assign circles
    # For 26 we use: 5 per corner (4x5=20), 1 per mid-edge (4x1=4), 2 in center
    circle_counts = [5, 5, 5, 5, 1, 1, 1, 1, 2]
    assert sum(circle_counts) == 26
    
    # Solve each cell independently
    all_centers = []
    all_radii = []
    
    for i, (cell, count) in enumerate(zip(cells, circle_counts)):
        if count > 0:
            c, r = solve_leaf(cell, count)
            all_centers.extend(c)
            all_radii.extend(r)
    
    all_centers = np.array(all_centers)
    all_radii = np.array(all_radii)
    
    # Final polishing
    all_centers, all_radii = global_polish(all_centers, all_radii)
    
    # Refine to ensure validity
    all_centers, all_radii = final_refine(all_centers, all_radii)
    
    return all_centers, all_radii, float(np.sum(all_radii))
