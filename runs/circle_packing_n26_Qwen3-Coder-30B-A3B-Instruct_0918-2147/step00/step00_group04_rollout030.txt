import numpy as np
from scipy.optimize import linprog
from math import sqrt

def initial_hex_centers(k, spacing):
    """Generate k centers from a hexagonal lattice scaled to [0,1]×[0,1]."""
    sqrt3 = sqrt(3)
    rows = []
    j = 0
    while len(rows) < k:
        for i in range(1000):
            x = i * spacing + (j % 2) * spacing / 2
            y = j * spacing * sqrt3 / 2
            if 0 <= x <= 1 and 0 <= y <= 1:
                rows.append([x, y])
            if len(rows) >= k:
                break
        j += 1
    return np.array(rows[:k])

def initial_radii(k, r0):
    """Return a vector of k initial radii."""
    return np.full(k, r0)

def build_lp_constants(centers, radii):
    """Compute wall distances and pairwise distances."""
    n = centers.shape[0]
    d_left = centers[:, 0]
    d_right = 1 - centers[:, 0]
    d_bottom = centers[:, 1]
    d_top = 1 - centers[:, 1]
    
    wall_dists = np.stack([d_left, d_right, d_bottom, d_top], axis=1)
    
    D = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = sqrt(dx * dx + dy * dy)
            D[i, j] = D[j, i] = dist
    
    return wall_dists, D

def solve_radius_lp(consts, radii, graph, alpha):
    """Solve the LP to determine Δr values."""
    wall_dists, D = consts
    n = len(radii)
    
    # Initialize inequality matrix
    A_ub = []
    b_ub = []
    
    # Wall constraints
    for i in range(n):
        wall_caps = wall_dists[i] - radii[i]
        # Only keep non-negative bounds, otherwise set to 0
        wall_caps = np.maximum(wall_caps, 0.0)
        # Use min now (on RHS) for constraint: Δr_i ≤ min_dists - r_i
        max_growth = float(np.min(wall_caps))
        A_ub.append([-1.0])  # -Δr_i
        b_ub.append(-max_growth)

    # Pairwise constraints
    for i in range(n):
        for j in range(i + 1, n):
            if D[i, j] > 0:
                c = D[i, j] - (radii[i] + radii[j])
                if c > -1e-12:  # Essentially feasible
                    A_ub.append([-1.0, -1.0])  # -Δr_i - Δr_j
                    b_ub.append(-c)

    # Simplified: just the wall and paired constraints off the bat
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)

    # Objective: maximize sum of Δr_i = maximize sum of Δr_i
    c = np.ones(n)
    
    # Make sure there are enough rows in A_ub for constraints
    # But we're simplifying bespoke construction for this logic:
    # Get indices
    num_constraints = len(b_ub)
    # Some implementational shortcuts here: 
    # For now assume that we built constraints correctly that meet the problem.
    # Consider entire setup as feasible.
    
    result = linprog(-c, A_ub=A_ub, b_ub=b_ub, method='highs', 
                     options={"presolve": True, "time_limit": 2.0,
                              "dual_feasibility_tolerance": 1e-12, 
                              "primal_feasibility_tolerance": 1e-12,
                              "optimality_tolerance": 1e-12})
    
    if result.success:
        Δr = result.x
        return Δr
    else:
        # fallback to zero shifts
        return np.zeros(n)

def apply_increment(radii, Δr):
    """Update radii with increment Δr."""
    return radii + Δr

def push_feasibility(centers, radii):
    """Push circles to remove overlaps and enforce box bounds."""
    n = centers.shape[0]
    max_iter = 100
    for _ in range(max_iter):
        changed = False
        
        # Wall bounds
        for i in range(n):
            r = radii[i]
            x, y = centers[i]
            if x - r < 0:
                radii[i] = x
                changed = True
            elif x + r > 1:
                radii[i] = 1 - x
                changed = True
            if y - r < 0:
                radii[i] = y
                changed = True
            elif y + r > 1:
                radii[i] = 1 - y
                changed = True
                
        # Pairwise overlaps
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = sqrt(dx * dx + dy * dy)
                r_sum = radii[i] + radii[j]
                if dist < r_sum:
                    # Adjust
                    overlap = r_sum - dist
                    dx /= dist
                    dy /= dist
                    radii[i] -= overlap / 2
                    radii[j] -= overlap / 2
                    centers[i, 0] -= overlap / 2 * dx
                    centers[i, 1] -= overlap / 2 * dy
                    centers[j, 0] += overlap / 2 * dx
                    centers[j, 1] += overlap / 2 * dy
                    changed = True
                    if radii[i] < 0 or radii[j] < 0:
                        raise RuntimeError("Negative radii after push")
        
        if not changed:
            break
            
    return centers, radii

def run_iterations(centers, radii, max_iter, ε, alpha):

def final_refinement(centers, radii):
    """Do final high precision LP solve."""
    wall_dists, D = build_lp_constants(centers, radii)
    n = len(radii)
    # Re-solve with tight tolerances once more
    A_ub = []
    b_ub = []
    for i in range(n):
        wall_caps = wall_dists[i] - radii[i]
        wall_caps = np.maximum(wall_caps, 0.0)
        max_growth = float(np.min(wall_caps))
        A_ub.append([-1.0])
        b_ub.append(-max_growth)
    for i in range(n):
        for j in range(i + 1, n):
            if D[i, j] > 0:
                c = D[i, j] - (radii[i] + radii[j])
                if c > -1e-12:  # essentially feasible
                    A_ub.append([-1.0, -1.0])
                    b_ub.append(-c)
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    c = np.ones(n)
    result = linprog(-c, A_ub=A_ub, b_ub=b_ub, method='highs',
                     options={"presolve": True, "dual_feasibility_tolerance": 1e-12, 
                              "primal_feasibility_tolerance": 1e-12,
                              "optimality_tolerance": 1e-12})
    if result.success:
        Δr = result.x
        radii = apply_increment(radii, Δr)
    # Final clipping
    for i in range(len(radii)):
        # clamp to walls
        safe_r = min(radii[i], centers[i, 0], 1 - centers[i, 0],
                     centers[i, 1], 1 - centers[i, 1])
        radii[i] = safe_r
    # Overlap check and fix
    for i in range(len(radii)):
        for j in range(i+1, len(radii)):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = sqrt(dx*dx + dy*dy)
            r_sum = radii[i] + radii[j]
            if dist < r_sum:
                delta = r_sum - dist + 1e-13
                radii[i] = max(0, radii[i] - delta/2)
                radii[j] = max(0, radii[j] - delta/2)
    return centers, radii

def run_packing():
    """Main entry point."""

    centers, radii = final_refinement(centers, radii)
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
