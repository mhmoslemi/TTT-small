import numpy as np
from scipy.optimize import minimize

def initial_layout():
    # Create a hexagonal grid pattern inside the unit square
    n_cols = 5
    n_rows = 5
    spacing = 0.18
    radius = spacing / 2 - 0.01
    
    centers = []
    for i in range(n_rows):
        for j in range(n_cols):
            x = (j + 0.5) * spacing
            y = (i + 0.5) * spacing
            # Adjust to keep within bounds
            x = max(radius, min(1 - radius, x))
            y = max(radius, min(1 - radius, y))
            centers.append([x, y])
    
    # Add one more circle at the center of the square
    centers.append([0.5, 0.5])
    
    centers = np.array(centers)
    radii = np.full(26, radius)
    return centers, radii

def pack_vector(centers, radii):
    # Pack centers and radii into a single vector
    z = np.empty(3 * 26)
    z[0::3] = centers[:, 0]  # x coordinates
    z[1::3] = centers[:, 1]  # y coordinates
    z[2::3] = radii          # radii
    return z

def unpack_vector(z):
    # Unpack the vector into centers and radii
    centers = np.empty((26, 2))
    centers[:, 0] = z[0::3]  # x
    centers[:, 1] = z[1::3]  # y
    radii = z[2::3]
    return centers, radii

def objective(z):
    centers, radii = unpack_vector(z)
    return -np.sum(radii)

def constraint_boundary(z):
    centers, radii = unpack_vector(z)
    constraints = []
    
    # For each circle, add four boundary constraints:
    # x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    for i in range(26):
        x, y = centers[i]
        r = radii[i]
        constraints.append(x - r)               # x - r >= 0
        constraints.append(1 - x - r)           # 1 - x - r >= 0
        constraints.append(y - r)               # y - r >= 0
        constraints.append(1 - y - r)           # 1 - y - r >= 0
    
    return np.array(constraints)

def constraint_nonoverlap(z):
    centers, radii = unpack_vector(z)
    constraints = []
    for i in range(26):
        for j in range(i + 1, 26):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            distance_squared = dx ** 2 + dy ** 2
            r_sum = radii[i] + radii[j]
            # Non-overlap constraint: (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
            constraints.append(distance_squared - r_sum ** 2)
    return np.array(constraints)
