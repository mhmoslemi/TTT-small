import numpy as np
from scipy.optimize import minimize

def make_hex_lattice(n, margin=0.05):
    # Generate hexagonal lattice points and trim to n points
    # We use a hexagonal grid with spacing that fits within the unit square
    dx = 1.0 / (np.ceil(np.sqrt(n)) + 0.5)
    dy = dx * np.sqrt(3) / 2.0

    points = []
    i = 0
    while len(points) < n:
        for j in range(int(np.ceil(n / 1.5)) + 1):
            # Even rows (j even): full spacing
            # Odd rows (j odd): offset by dx/2
            x = i * dx + (j % 2) * dx / 2.0
            y = j * dy
            if x < 1.0 - margin and y < 1.0 - margin:
                points.append((x, y))
            else:
                break
        i += 1
        if i > 1000:
            break

    # Trim to exactly n points
    points = points[:n]
    # Center the points to allow margin
    points = np.array(points)
    cx = np.mean(points[:, 0])
    cy = np.mean(points[:, 1])
    points[:, 0] -= cx - 0.5
    points[:, 1] -= cy - 0.5

    return points

def boundary_constraints(z):
    # z is array: [x0, y0, r0, x1, y1, r1, ...]
    n = len(z) // 3
    con = []
    for i in range(n):
        x, y, r = z[3*i], z[3*i+1], z[3*i+2]
        con.append(x - r)  # left
        con.append(1 - x - r)  # right
        con.append(y - r)  # bottom
        con.append(1 - y - r)  # top
    return np.array(con)

def pairwise_constraints(z):
    n = len(z) // 3
    con = []
    for i in range(n):
        for j in range(i + 1, n):
            xi, yi, ri = z[3*i], z[3*i+1], z[3*i+2]
            xj, yj, rj = z[3*j], z[3*j+1], z[3*j+2]
            dist_sq = (xi - xj)**2 + (yi - yj)**2
            sum_r = ri + rj
            con.append(dist_sq - sum_r**2)
    return np.array(con)

def objective(z):
    # Minimize negative sum of radii to maximize sum
    n = len(z) // 3
    return -np.sum(z[2::3])

def solve_nlp(initial_z):
    # Define constraints
    bnd = []
    for i in range(len(initial_z)):
        if i % 3 == 2:  # radius
            bnd.append((0, 0.5))  # radius non-negative and bounded
        else:
            bnd.append((0, 1))  # x,y coordinates

    cons = [
        {'type': 'ineq', 'fun': boundary_constraints},
        {'type': 'ineq', 'fun': pairwise_constraints}
    ]

    res = minimize(objective, initial_z, method='SLSQP',
                   bounds=bnd, constraints=cons, options={'ftol': 1e-9, 'maxiter': 1000})
    
    if not res.success:
        # If optimization fails, return the initial guess
        return initial_z
    return res.x

def nudge_centers(centers, radii):
    # Small nudging to avoid trivial fixed points
    center_t = centers.copy()
    for i in range(len(centers)):
        x, y = centers[i]
        r = radii[i]
        # If close to a boundary, nudge a little inwards in a tangential direction
        if x - r < 1e-4:
            center_t[i, 0] += 0.001
        if 1 - x - r < 1e-4:
            center_t[i, 0] -= 0.001
        if y - r < 1e-4:
            center_t[i, 1] += 0.001
        if 1 - y - r < 1e-4:
            center_t[i, 1] -= 0.001
    return center_t

def run_packing():

    centers = make_hex_lattice(26, margin=0.05)

        z_nudge = np.concatenate([new_centers.flatten(), radii])
        final_z = solve_nlp(z_nudge)

        centers_tmp = final_z[::3].reshape(-1, 2)
