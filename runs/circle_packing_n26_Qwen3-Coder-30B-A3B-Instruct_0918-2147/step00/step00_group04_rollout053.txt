```python
import numpy as np
from scipy.optimize import minimize

NUM_CIRCLES = 26
NUM_RESTARTS = 5
BOUNDARY_TOLERANCE = 1e-9
NONOVERLAP_TOLERANCE = 1e-9
INITIAL_RADIUS = 0.02

def hex_lattice_points(n_rows, spacing):
    points = []
    for i in range(n_rows):
        y = i * spacing * np.sqrt(3) / 2
        if y > 1:
            break
        offset = (i % 2) * spacing / 2
        n_cols = int((1 - offset) / spacing) + 1
        for j in range(n_cols):
            x = offset + j * spacing
            if x > 1:
                break
            points.append([x, y])
    return np.array(points)

def initial_guess(seed):
    np.random.seed(seed)
    lattice_points = hex_lattice_points(10, 0.15)
    n_points = min(NUM_CIRCLES, len(lattice_points))
    selected_points = lattice_points[:n_points]
    jitter = (np.random.rand(n_points, 2) - 0.5) * 0.04
    centers = np.clip(selected_points + jitter, 0, 1)
    radii = np.full(n_points, INITIAL_RADIUS)
    vars0 = np.empty(3 * n_points)
    vars0[0::3] = centers[:, 0]
    vars0[1::3] = centers[:, 1]
    vars0[2::3] = radii
    return vars0

def objective(vars):
    return -np.sum(vars[2::3])

def constraint_boundary(vars):
    n = len(vars) // 3
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    boundaries = np.empty(4 * n)
    boundaries[0*n:1*n] = x - r
    boundaries[1*n:2*n] = 1 - x - r
    boundaries[2*n:3*n] = y - r
    boundaries[3*n:4*n] = 1 - y - r
    return boundaries

def constraint_nonoverlap(vars):
    n = len(vars) // 3
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    result = []
    for i in range(n):
        for j in range(i + 1, n):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dist_sq = dx * dx + dy * dy
            sum_radii = r[i] + r[j]
            margin = np.sqrt(dist_sq) - sum_radii
            result.append(margin)
    return np.array(result)

def postprocess(vars):
    n = len(vars) // 3
    x = np.copy(vars[0::3])
    y = np.copy(vars[1::3])
    r = np.copy(vars[2::3])
    # Fix boundary issues
    for i in range(n):
        if x[i] - r[i] < -BOUNDARY_TOLERANCE:
            x[i] = r[i]
        if 1 - x[i] - r[i] < -BOUNDARY_TOLERANCE:
            x[i] = 1 - r[i]
        if y[i] - r[i] < -BOUNDARY_TOLERANCE:
            y[i] = r[i]
        if 1 - y[i] - r[i] < -BOUNDARY_TOLERANCE:
            y[i] = 1 - r[i]
    # Resolve overlaps
    done = False
    while not done:
        done = True
        for i in range(n):
            for j in range(i + 1, n):
                dx = x[i] - x[j]
                dy = y[i] - y[j]
                dist = np.sqrt(dx * dx + dy * dy)
                sum_radii = r[i] + r[j]
                overlap = sum_radii - dist
                if overlap < -NONOVERLAP_TOLERANCE:
                    continue
                if overlap > NONOVERLAP_TOLERANCE:
                    # Slight overlap; reduce larger radius
                    if r[i] > r[j]:
                        r[i] -= overlap
                        if r[i] < 0:
                            r[i] = 0
                    else:
                        r[j] -= overlap
                        if r[j] < 0:
                            r[j] = 0
                    done = False
                if r[i] < 0:
                    r[i] = 0
                if r[j] < 0:
                    r[j] = 0
    result = np.empty_like(vars)
    result[0::3] = x
    result[1::3] = y
    result[2::3] = r
    return result

def run_packing():
    best_sum = -np.inf
    best_vars = None
    
    for seed in range(NUM_RESTARTS):
        vars0 = initial_guess(seed)
        constraints = [
            {'type': 'ineq', 'fun': constraint_boundary},
            {'type': 'ineq', 'fun': constraint_nonoverlap}
        ]
        bounds = [(0, 1), (0, 1), (0, 0.5)] * NUM_CIRCLES
        res = minimize(
            objective,
            vars0,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 2000, 'ftol': 1e-6}
        )
        if not res.success:
            continue
        vars_corr = postprocess(res.x)
        s = np.sum(vars_corr[2::3])
        if s > best_sum:
            best_sum = s
            best_vars = vars_corr
    
    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii = best_vars[2::3]
    return centers, radii, float(best_sum)
```
