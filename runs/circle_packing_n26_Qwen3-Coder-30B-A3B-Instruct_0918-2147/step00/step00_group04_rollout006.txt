import numpy as np
import cvxpy as cp
from scipy.spatial.distance import cdist

def build_leaf_problem(cell, k):
    x_min, y_min, x_max, y_max = cell
    x = cp.Variable(k)
    y = cp.Variable(k)
    r = cp.Variable(k, nonneg=True)
    
    # Boundary constraints
    constraints = []
    for i in range(k):
        constraints.append(x[i] - r[i] >= x_min)
        constraints.append(x[i] + r[i] <= x_max)
        constraints.append(y[i] - r[i] >= y_min)
        constraints.append(y[i] + r[i] <= y_max)
        
    # Non-overlap constraints
    for i in range(k):
        for j in range(i+1, k):
            dist = cp.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
            constraints.append(dist >= r[i] + r[j])
    
    # Maximize sum of radii
    objective = cp.Maximize(cp.sum(r))
    prob = cp.Problem(objective, constraints)
    return prob, x, y, r

def solve_leaf(cell, k):
    prob, x, y, r = build_leaf_problem(cell, k)
    prob.solve(solver=cp.OSQP, verbose=False, eps_abs=1e-10, eps_rel=1e-10)
    assert prob.status == cp.OPTIMAL, "Leaf problem failed to solve optimally"
    return np.array([x.value, y.value]).T, r.value

def merge_clusters(cluster_a, cluster_b, shared_edge):

def global_polish(centers, radii, max_iter=3):
    n = len(centers)
    x = np.copy(centers[:, 0])
    y = np.copy(centers[:, 1])
    r = np.copy(radii)
    
    mu = 1e-4
    grad_step = 1e-3
    
    for _ in range(max_iter):
        # Compute gradients
        # Objective gradient: d/dr = 1
        grad_r = np.ones(n)
        
        # Constraints gradients (log barrier approx)
        for i in range(n):
            # Walls
            grad_r[i] -= mu / (x[i] - r[i] + 1e-12) if x[i] - r[i] > 0 else 0
            grad_r[i] -= mu / (1 - x[i] - r[i] + 1e-12) if 1 - x[i] - r[i] > 0 else 0
            grad_r[i] -= mu / (y[i] - r[i] + 1e-12) if y[i] - r[i] > 0 else 0
            grad_r[i] -= mu / (1 - y[i] - r[i] + 1e-12) if 1 - y[i] - r[i] > 0 else 0
            
            # Non-overlap penalties
            for j in range(n):
                if i != j:
                    dist = np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
                    if dist > 0:
                        if dist < r[i] + r[j] + 1e-12:
                            grad_r[i] -= mu / (dist - r[i] - r[j] + 1e-12)
        
        # Update radii
        r += grad_step * grad_r
        
        # Keep radii non-negative
        r = np.maximum(r, 0)
        
        # Apply other smoothed constraint (voluntary) with small strength
        # (This is simplified: just a small proxy effect)
        
    return centers, r

        # If final solve fails, keep original

    # Ensure no circle is out of bounds or overlaps
    # For safety, now do manual clipping and close ensure

def run_packing():

    # Now perform hierarchical merge from bottom up (binary tree)
    # Simplified for direct reproduction:

    # So we'll cluster directly:
    merged_centers = np.vstack([c for c, _ in solved_clusters])
    merged_radii = np.hstack([r for _, r in solved_clusters])
    
    # Lets refine just two rounds directly with full logs
    merged_centers, merged_radii = global_polish(merged_centers, merged_radii, max_iter=3)
    
    # Finally do high precision final refinement using the same optimization
    merged_centers, merged_radii = final_refine(merged_centers, merged_radii)
    
    total_sum = np.sum(merged_radii)
