import numpy as np
from scipy.optimize import minimize

def make_hex_lattice(n, margin=0.05):
    # Create a hexagonal lattice pattern that fits in [margin, 1-margin]^2
    # Try to fit at least n points
    dx = 1.0 / (np.ceil(np.sqrt(n)) + 0.5)
    dy = dx * np.sqrt(3) / 2.0
    centers = []
    i = 0
    j = 0
    while len(centers) < n:
        x = i * dx + (j % 2) * dx / 2.0
        y = j * dy
        if margin <= x <= 1 - margin and margin <= y <= 1 - margin:
            centers.append([x, y])
        if x > 1 - margin:
            j += 1
            i = 0
        else:
            i += 1
            if i > np.ceil(np.sqrt(n)) + 1:
                i = 0
                j += 1
                if j > np.ceil(np.sqrt(n)) + 1:
                    break
    return np.array(centers[:n])

def objective(z):
    # z is [x0,y0,r0,x1,y1,r1,...]
    n = len(z) // 3
    radii = z[2::3]
    return -np.sum(radii)

def boundary_constraints(z):
    # return [x-r, x+r-1, y-r, y+r-1] for each circle
    n = len(z) // 3
    result = []
    for i in range(n):
        x = z[3*i]
        y = z[3*i+1]
        r = z[3*i+2]
        result.extend([x - r, r - x + 1, y - r, r - y + 1])
    return np.array(result)

def pairwise_constraints(z):
    # return [dist_ij^2 - (r_i + r_j)^2] for all pairs i < j
    n = len(z) // 3
    result = []
    for i in range(n):
        for j in range(i + 1, n):
            x1, y1, r1 = z[3*i], z[3*i+1], z[3*i+2]
            x2, y2, r2 = z[3*j], z[3*j+1], z[3*j+2]
            dist_sq = (x1 - x2)**2 + (y1 - y2)**2
            sep_sq = (r1 + r2)**2
            result.append(dist_sq - sep_sq)
    return np.array(result)

def solve_nlp(initial_z):
    n = len(initial_z) // 3
    constraints = []
    # Boundary constraints
    constraints.append({'type': 'ineq', 'fun': boundary_constraints})
    # Pairwise separation constraints
    constraints.append({'type': 'ineq', 'fun': pairwise_constraints})
    # Bounds
    bounds = []
    for i in range(n):
        bounds.append((1e-12, 1))  # x
        bounds.append((1e-12, 1))  # y
        bounds.append((1e-12, 0.5))  # r
    result = minimize(objective, initial_z, method='SLSQP', constraints=constraints, bounds=bounds, tol=1e-9, options={'maxiter': 1000})
    if not result.success:
        # Try a less strict convergence
        result = minimize(objective, initial_z, method='SLSQP', constraints=constraints, bounds=bounds, tol=1e-6, options={'maxiter': 1000})
    return result.x

def nudge_centers(centers, radii):
    # For circles close to a wall, nudge them out slightly
    epsilon = 0.001
    new_centers = centers.copy()
    new_radii = radii.copy()
    for i in range(len(centers)):
        x, y = centers[i]
        r = radii[i]
        # Check constraints
        if abs(x - r) < 1e-6:
            new_centers[i, 0] += epsilon
        elif abs(x + r - 1) < 1e-6:
            new_centers[i, 0] -= epsilon
        if abs(y - r) < 1e-6:
            new_centers[i, 1] += epsilon
        elif abs(y + r - 1) < 1e-6:
            new_centers[i, 1] -= epsilon
    # Re-solve the optimization with nudged positions
    z0 = np.concatenate([new_centers.flatten(), new_radii])
    z_sol = solve_nlp(z0)
    return z_sol[::3].reshape(-1, 2), z_sol[2::3]
