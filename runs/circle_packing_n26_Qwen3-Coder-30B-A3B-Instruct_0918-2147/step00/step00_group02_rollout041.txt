import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def initial_configuration(n=26):
    # Generate Sobol sequence using a simple quasi-random approach
    # Here we use a simple pattern to spread points reasonably
    indices = np.arange(n)
    phi = (1 + np.sqrt(5)) / 2
    # Golden ratio based spiral pattern, scaled to [0,1]
    theta = indices * 2 * np.pi / phi
    r = np.sqrt(indices / (n - 1))  # Radial distribution to avoid clustering
    x = r * np.cos(theta)
    y = r * np.sin(theta)
    # Remap to [0,1] square
    x = (x - x.min()) / (x.max() - x.min())
    y = (y - y.min()) / (y.max() - y.min())
    centers = np.column_stack([x,y])
    # Start with a small radius
    radii = np.ones(n) * 0.01
    # Adjust radii based on wall distances and nearest neighbors
    # Distance to walls
    wall_distances = np.array([
        centers[:,0], 1 - centers[:,0],
        centers[:,1], 1 - centers[:,1]
    ])
    wall_min = np.min(wall_distances, axis=0)
    # Nearest neighbor distance estimation
    if n > 1:
        dist_matrix = cdist(centers, centers)
        np.fill_diagonal(dist_matrix, np.inf)
        nearest_distances = np.min(dist_matrix, axis=0)
        radii = np.minimum(wall_min, nearest_distances / 2)
    else:
        radii = wall_min
    radii = np.clip(radii, 0, 0.5)
    return centers, radii

def pack(centers, radii):
    return np.column_stack([centers.reshape(-1), radii])

def unpack(z):
    n = len(z) // 3
    centers = z[:2*n].reshape(n, 2)
    radii = z[2*n:]
    return centers, radii

def max_violation(centers, radii):
    n = len(radii)
    # wall violations
    wall_viol = np.maximum(0, radii - centers[:,0]) + \
                np.maximum(0, radii - (1 - centers[:,0])) + \
                np.maximum(0, radii - centers[:,1]) + \
                np.maximum(0, radii - (1 - centers[:,1]))
    # overlap violations
    dist_matrix = cdist(centers, centers)
    np.fill_diagonal(dist_matrix, np.inf)
    overlap_viol = np.maximum(0, radii.reshape(-1,1) + radii.reshape(1,-1) - dist_matrix)
    overlap_viol = np.max(overlap_viol[np.triu_indices(n, k=1)])
    return np.maximum(np.max(wall_viol), overlap_viol)

def energy_and_grad(z):
    centers, radii = unpack(z)
    assert len(radii) == len(centers)
    n = len(radii)
    # Compute distance matrix
    dist_matrix = cdist(centers, centers)
    # Wall energy and gradients
    wall_viol = np.maximum(0, radii - centers[:,0])
    wall_grad_x = -2 * wall_viol
    wall_viol = np.maximum(0, radii - (1 - centers[:,0]))
    wall_grad_x += 2 * wall_viol
    wall_viol = np.maximum(0, radii - centers[:,1])
    wall_grad_y = -2 * wall_viol
    wall_viol = np.maximum(0, radii - (1 - centers[:,1]))
    wall_grad_y += 2 * wall_viol
    wall_grad_r = 2 * (radii - np.minimum(centers[:,0], 1 - centers[:,0]))
    wall_grad_r += 2 * (radii - np.minimum(centers[:,1], 1 - centers[:,1]))
    # Centre gradient
    grad_x = np.zeros(n)
    grad_y = np.zeros(n)
    grad_r = np.zeros(n)
    # Pairwise energy
    for i in range(n):
        for j in range(i):
            d = dist_matrix[i,j]
            o = max(0, radii[i] + radii[j] - d)
            if o > 0:
                # Add contribution
                dx = centers[i,0] - centers[j,0]
                dy = centers[i,1] - centers[j,1]
                grad_x[i] -= 2 * o * dx / (d + 1e-12)
                grad_x[j] += 2 * o * dx / (d + 1e-12)
                grad_y[i] -= 2 * o * dy / (d + 1e-12)
                grad_y[j] += 2 * o * dy / (d + 1e-12)
                grad_r[i] += 2 * o
                grad_r[j] += 2 * o
    # Total gradient
    grad = np.hstack([grad_x, grad_y, grad_r])
    # Wall gradient
    grad[2*n:] += wall_grad_r
    # If outside bounds, gradients are zero
    grad_x = np.where(centers[:,0] <= radii, 0, grad_x)
    grad_x = np.where(centers[:,0] >= 1 - radii, 0, grad_x)
    grad_y = np.where(centers[:,1] <= radii, 0, grad_y)
    grad_y = np.where(centers[:,1] >= 1 - radii, 0, grad_y)
    grad[0:2*n:2] = grad_x
    grad[1:2*n:2] = grad_y
    return grad

def inner_optimize(z0, B, lambda_budget):
    # This function solves the optimisation step of the algorithm
    # Using L-BFGS-B optimizer
    obj = lambda z: energy(z, B, lambda_budget)
    grad = lambda z: energy_and_grad(z)
    res = minimize(
        obj, z0, method='L-BFGS-B',
        jac=grad,
        bounds=[(0, 1)] * (len(z0) // 3 * 2) + [(0, 0.5)] * (len(z0) // 3),
        options={'ftol': 1e-15, 'gtol': 1e-12, 'maxiter': 300}
    )
    return res.x

def energy(z, B, lambda_budget):
    # Compute total energy
    centers, radii = unpack(z)
    sum_r = np.sum(radii)
    # Wall energy
    wall_violation = np.maximum(0, radii - centers[:,0]) + \
                    np.maximum(0, radii - (1 - centers[:,0])) + \
                    np.maximum(0, radii - centers[:,1]) + \
                    np.maximum(0, radii - (1 - centers[:,1]))
    wall_energy = np.sum(wall_violation**2)
    # Pair energy
    dist_matrix = cdist(centers, centers)
    overlap_viol = np.maximum(0, radii.reshape(-1,1) + radii.reshape(1,-1) - dist_matrix)
    np.fill_diagonal(overlap_viol, 0)
    pair_energy = np.sum(overlap_viol**2)
    # Budget energy
    budget_energy = lambda_budget * (B - sum_r)**2
    return wall_energy + pair_energy + budget_energy

def max_feasible_budget(z_start, B0=3.0, max_iter=40):
    B = B0
    delta = 5e-4
    best_feasible_z = z_start
    best_feasible_B = 0.0
    failed_count = 0
    iter_count = 0
    for _ in range(max_iter):
        try:
            z_opt = inner_optimize(z_start, B, 1e4)
            c_opt, r_opt = unpack(z_opt)
            max_viol = max_violation(c_opt, r_opt)
            if max_viol <= 1e-12:
                best_feasible_z = z_opt
                best_feasible_B = B
                B += delta
                failed_count = 0
            else:
                B -= delta
                failed_count += 1
                if failed_count > 15:  # Avoid oscillation
                    break
        except:
            B -= delta
            failed_count += 1
            if failed_count > 15:
                break
        iter_count += 1
        if delta < 1e-6:
            break
        if iter_count > max_iter:
            break
    return best_feasible_z, best_feasible_B

def postprocess(z):
    centers, radii = unpack(z)
    eps = 5e-10
    # Post-process to ensure everything is strictly within bounds
    for i in range(len(centers)):
        # Clamp centers
        centers[i,0] = np.clip(centers[i,0], eps, 1 - eps)
        centers[i,1] = np.clip(centers[i,1], eps, 1 - eps)
        # Shrink radii
        radii[i] = max(0, radii[i] - eps)
    # Ensure non-overlapping by repair pass
    max_iter = 10
    for _ in range(max_iter):
        dist_matrix = cdist(centers, centers)
        overlap = np.maximum(0, radii.reshape(-1, 1) + radii.reshape(1, -1) - dist_matrix)
        np.fill_diagonal(overlap, 0)
        if np.max(overlap) < 1e-10:
            break
        overlap = overlap / 2
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                if overlap[i, j] > 0:
                    radii[i] -= overlap[i, j]
                    radii[j] -= overlap[i, j]
    sum_r = np.sum(radii)
    return centers, radii, sum_r

def run_packing():
    centers, radii = initial_configuration(26)
    z_start = pack(centers, radii).flatten()
    z_final, B_final = max_feasible_budget(z_start, B0=3.0)
    centers_final, radii_final, sum_r = postprocess(z_final)
    return centers_final, radii_final, sum_r
