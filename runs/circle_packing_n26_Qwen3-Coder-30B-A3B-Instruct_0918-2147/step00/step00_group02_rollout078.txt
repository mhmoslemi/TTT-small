import numpy as np
from scipy.optimize import minimize
from sobol_seq import i4_sobol_generate

def initial_configuration(n=26):
    # Generate Sobol sequence for initial centers
    seed_points = i4_sobol_generate(2, n)
    centers = seed_points

    # Initial radii
    radii = np.zeros(n)
    # For each point, set radius to the minimum of its distance to all walls
    # and half the distance to its nearest neighbor
    for i in range(n):
        x, y = centers[i]
        r_wall = min(x, 1 - x, y, 1 - y)
        min_dist = np.inf
        for j in range(n):
            if i != j:
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < min_dist:
                    min_dist = dist
        r_neighbor = 0.5 * min_dist if min_dist > 0 else 0.0
        radii[i] = min(r_wall, r_neighbor)

    return centers, radii

def pack(centers, radii):
    # Flatten vectors to 1D
    return np.concatenate([centers.flatten(), radii])

def unpack(z):
    n = len(z) // 3
    centers = z[:2*n].reshape((n, 2))
    radii = z[2*n:]
    return centers, radii

def energy_and_grad(z):
    n = len(z) // 3
    centers, radii = unpack(z)
    
    lambda_budget = 1000.0
    budget = 2.70  # will be updated in main method
    total_radius = np.sum(radii)
    
    # Wall energy
    E_wall = 0.0
    grad_wall = np.zeros_like(z)
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Wall penalties
        penalties = []
        if r > x: penalties.append((r - x)**2)
        if r > (1 - x): penalties.append((r - (1 - x))**2)
        if r > y: penalties.append((r - y)**2)
        if r > (1 - y): penalties.append((r - (1 - y))**2)
        
        E_wall += sum(penalties)
        
        # Gradients
        if r > x:
            grad_wall[2*i] -= 2 * (r - x) # dx
            grad_wall[2*n + i] += 2 * (r - x) # dr
        if r > (1 - x):
            grad_wall[2*i + 1] -= 2 * (r - (1 - x)) # dy
            grad_wall[2*n + i] += 2 * (r - (1 - x)) # dr
        if r > y:
            grad_wall[2*i] += 2 * (r - y) # dx (Actually sign should be minus)
            grad_wall[2*n + i] += 2 * (r - y) # dr
        if r > (1 - y):
            grad_wall[2*i + 1] += 2 * (r - (1 - y)) # dy 
            grad_wall[2*n + i] += 2 * (r - (1 - y)) # dr
    
    # Pairwise overlap
    E_overlap = 0.0
    grad_overlap = np.zeros_like(z)
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            d = np.sqrt(dx*dx + dy*dy)
            r_sum = radii[i] + radii[j]
            overlap = max(0, r_sum - d)
            
            E_overlap += overlap**2
            
            if overlap > 1e-12:
                # dz = (d - dx) / d if d != 0 else 0 (but we want (c_i - c_j) / d)
                scale = 2 * overlap / (d + 1e-12)  # To prevent div by zero
                grad_overlap[2*i] -= scale * dx
                grad_overlap[2*i + 1] -= scale * dy 
                grad_overlap[2*j] += scale * dx
                grad_overlap[2*j + 1] += scale * dy
                grad_overlap[2*n + i] += scale
                grad_overlap[2*n + j] += scale
                
    # Budget penalty
    E_budget = lambda_budget * (budget - total_radius)**2
    grad_budget = np.zeros_like(z)
    grad_budget[2*n:] = -2 * lambda_budget * (budget - total_radius)

    # Total energy and gradient
    E_total = E_wall + E_overlap + E_budget
    grad_total = grad_wall + grad_overlap + grad_budget

    return E_total, grad_total

def inner_optimize(z0, B, lambda_budget):
    def f_and_g(z):
        return energy_and_grad(z)
    
    res = minimize(
        f_and_g,
        z0,
        method='L-BFGS-B',
        bounds=[
            (0, 1), (0, 1),   # x, y
            (0, 0.5),         # r
            (0, 1), (0, 1),
            (0, 0.5),
            # ... for all centers and radii
        ] * 26,
        options={'disp': False},
    )
    return res.x

def max_feasible_budget(z_start):
    # Binary search with small step size
    delta = 0.0005
    B = 3.0  # Overestimate to start
    z_current = z_start
    prev_feasible = None
    last_feasible = None

    for iteration in range(100):
        # Inner optimization
        z_opt = inner_optimize(z_current, B, 1000.0)
        centers, radii = unpack(z_opt)

        # Test feasibility
        max_violation = 0.0
        for i in range(len(centers)):
            x, y = centers[i]
            r = radii[i]
            violations = [r - x, r - (1 - x), r - y, r - (1 - y)]
            for v in violations:
                if v > max_violation:
                    max_violation = v
            
            for j in range(i + 1, len(centers)):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                d = np.sqrt(dx*dx + dy*dy)
                r_sum = radii[i] + radii[j]
                overlap = r_sum - d
                if overlap > max_violation:
                    max_violation = overlap

        # Update budget
        if max_violation < 1e-13:
            last_feasible = z_opt.copy()
            prev_feasible = B
            B += delta
        else:
            B -= delta

        # Continue until small delta
        if delta < 1e-6:
            break

    return last_feasible if last_feasible is not None else z_current

def postprocess(z):
    # Unpack
    centers, radii = unpack(z)
    
    # Shrink radii slightly to ensure boundary condition
    eps = 5e-10
    radii = np.maximum(0, radii - eps)
    centers = np.clip(centers, eps, 1.0 - eps)
    
    # Repair if needed
    n = len(radii)
    for _ in range(5):
        any_overlap = False
        for i in range(n):
            x1, y1 = centers[i]
            r1 = radii[i]
            for j in range(i + 1, n):
                x2, y2 = centers[j]
                r2 = radii[j]
                dx = x1 - x2
                dy = y1 - y2
                d = np.sqrt(dx*dx + dy*dy)
                overlap = r1 + r2 - d
                if overlap > eps:
                    any_overlap = True
                    # Reduce both radii
                    reduction = overlap/2.0 + eps
                    radii[i] -= reduction
                    radii[j] -= reduction
        if not any_overlap:
            break

    # Ensure radii are positive
    radii = np.maximum(0, radii)
    
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii

def run_packing():
    n = 26
    centers, radii = initial_configuration(n)
    z_start = pack(centers, radii)
    z_opt = max_feasible_budget(z_start)
    centers, radii, sum_radii = postprocess(z_opt)
    return centers, radii, sum_radii
