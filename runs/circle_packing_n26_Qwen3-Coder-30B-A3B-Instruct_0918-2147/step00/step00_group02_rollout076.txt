import numpy as np
from scipy.stats.qmc import Sobol

def initialize_seeds(n):
    sobol = Sobol(d=2, seed=42)
    points = sobol.random(n)
    delta = 0.02
    return np.clip(delta + (1 - 2 * delta) * points, delta, 1 - delta)

def compute_feasible_radii(centers):
    n = centers.shape[0]
    # Compute pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_matrix = np.sqrt(np.sum(diff**2, axis=2))
    
    # Nearest neighbor distances (excluding self)
    np.fill_diagonal(dist_matrix, np.inf)
    nn_distances = np.min(dist_matrix, axis=1)
    
    # Wall distances
    wall_distances = np.min(np.stack([
        centers[:, 0],
        1 - centers[:, 0],
        centers[:, 1],
        1 - centers[:, 1]
    ]), axis=0)
    
    # Feasible radii
    radii = 0.5 * np.minimum(nn_distances, wall_distances)
    return radii

def compute_slack(radii, nn_distances):
    slack = 0.5 * nn_distances - radii
    return np.maximum(slack, 0.0)

def redistribute_radii(radii, slack, nn_distances, alpha):
    total_slack = np.sum(slack)
    if total_slack <= 1e-12:
        return radii.copy()
    
    # Create new radii
    new_radii = radii.copy()
    
    # Add α * slack to wall-limited circles
    wall_limited = slack > 1e-12
    if np.any(wall_limited):
        added = alpha * slack[wall_limited]
        new_radii[wall_limited] += added
    
    # Reduce from neighbour-limited circles, proportional to existing radii
    neighbour_limited = ~wall_limited
    if np.any(neighbour_limited):
        reduction_factor = added.sum() / (np.sum(radii[neighbour_limited]) + 1e-15)
        new_radii[neighbour_limited] *= (1 - reduction_factor)
    
    # Ensure non-negativity
    new_radii = np.maximum(new_radii, 0.0)
    return new_radii

def push_away(centers, radii, beta, eps):
    n = centers.shape[0]
    new_centers = centers.copy()
    diff = new_centers[:, np.newaxis, :] - new_centers[np.newaxis, :, :]
    dist_matrix = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(n):
            if i == j:
                continue
            d = dist_matrix[i, j]
            if d < radii[i] + radii[j] + eps:
                overlap = radii[i] + radii[j] + eps - d
                direction = diff[i, j] / (d + eps)
                new_centers[i] += beta * overlap * direction
    
    # Project back to valid range
    new_centers = np.clip(new_centers, eps, 1 - eps)
    return new_centers

def iterate(centers, radii, alpha, beta, max_iter):
    for _ in range(max_iter):
        # Compute nearest neighbor distances
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dist_matrix = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dist_matrix, np.inf)
        nn_distances = np.min(dist_matrix, axis=1)
        
        # Slack calculation
        slack = compute_slack(radii, nn_distances)
        
        # Redistribute radii
        new_radii = redistribute_radii(radii, slack, nn_distances, alpha)
        
        # Push away
        new_centers = push_away(centers, new_radii, beta, 1e-8)
        
        # Re-evaluate radii
        radii = compute_feasible_radii(new_centers)
        
        # Check convergence
        sum_radii = np.sum(radii)
        if sum_radii - np.sum(new_radii) < 1e-5:
            break
            
        centers = new_centers
        radii = new_radii
        
    return centers, radii

def repair(centers, radii, eps_final):
    # Shrink all radii
    radii = np.maximum(0, radii - eps_final)
    
    # Clamp centers
    centers = np.clip(centers, eps_final, 1 - eps_final)
    
    # Post-fix overlap
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_matrix = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(centers.shape[0]):
        for j in range(i+1, centers.shape[0]):
            if dist_matrix[i, j] < radii[i] + radii[j] - 1e-12:
                total_required = radii[i] + radii[j] - dist_matrix[i, j]
                reduction = total_required / 2 + eps_final
                radii[i] = max(0, radii[i] - reduction)
                radii[j] = max(0, radii[j] - reduction)
    
    return centers, radii

def run_packing():
    # Initial seeds
    centers = initialize_seeds(26)
    radii = compute_feasible_radii(centers)
    
    # Iterative improvement
    centers, radii = iterate(centers, radii, 0.4, 0.3, 200)
    
    # Final repair
    centers, radii = repair(centers, radii, 5e-10)
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
