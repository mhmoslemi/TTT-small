import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial.distance import cdist

def build_lp_model(n_circles):
    # We'll create: 26 * 2 variables for centers (x, y) and 26 for radii
    # The constraints are:
    # 1. Wall constraints (4 per circle): r <= x, r <= 1-x, r <= y, r <= 1-y
    # 2. Pairwise separation constraints (4 per pair):

    n_vars = 2 * n_circles + n_circles  # [x1, y1, x2, y2, ..., r1, r2, ...]
    n_pairs = n_circles * (n_circles - 1) // 2

    # Generate all pairwise indices
    pairs = [(i, j) for i in range(n_circles) for j in range(i+1, n_circles)]

    # For each pair, we have up to 4 inequality constraints to linearize:
    # x_i + r_i <= x_j - r_j  or  x_j + r_j <= x_i - r_i
    # y_i + r_i <= y_j - r_j  or  y_j + r_j <= y_i - r_i

    # Each constraint can be written in form: Ax <= b
    # But since we must avoid overlap, we should express as strict inequality
    # So for each pair, we need to add constraints of type:
    # - (x_i + r_i <= x_j - r_j)  or  (x_j + r_j <= x_i - r_i)
    # - (y_i + r_i <= y_j - r_j)  or  (y_j + r_j <= y_i - r_i)

    # To model this, we split into two types of constraints for both x,y:
    # For x axis: define 2 constraints per pair, for left/right relationship:
    # [x_i + r_i + 0*x_j + 0*y_j + 0*r_j <= x_j - r_j]
    # Which implies: [x_i - x_j + r_i + r_j <= 0]
    # Also the inverse:  [x_j - x_i + r_j + r_i <= 0]

    # Now build A matrix and b vector
    num_constraints = 4 * n_pairs + 4 * n_circles  # pairwise + walls

    A = np.zeros((num_constraints, n_vars))
    b = np.zeros(num_constraints)

    offset = 0

    # Separate for x_i + r_i <= x_j - r_j (i on left of j)
    for i, j in pairs:
        # First direction: x_i + r_i <= x_j - r_j
        A[offset][i] = 1    # x_i
        A[offset][n_circles + i] = 0  # y_i (not involved)
        A[offset][j] = -1   # x_j
        A[offset][n_circles + j] = 0  # y_j (not involved)
        A[offset][2*n_circles + i] = 1  # r_i
        A[offset][2*n_circles + j] = 1  # r_j
        b[offset] = 0
        offset += 1

        # Second direction: x_j + r_j <= x_i - r_i
        A[offset][i] = -1   # x_i
        A[offset][n_circles + i] = 0
        A[offset][j] = 1    # x_j
        A[offset][n_circles + j] = 0
        A[offset][2*n_circles + i] = 1  # r_i
        A[offset][2*n_circles + j] = 1  # r_j
        b[offset] = 0
        offset += 1

        # Third direction: y_i + r_i <= y_j - r_j
        A[offset][i] = 0    # x_i
        A[offset][n_circles + i] = 1  # y_i
        A[offset][j] = 0    # x_j
        A[offset][n_circles + j] = -1  # y_j
        A[offset][2*n_circles + i] = 1  # r_i
        A[offset][2*n_circles + j] = 1  # r_j
        b[offset] = 0
        offset += 1

        # Fourth direction: y_j + r_j <= y_i - r_i
        A[offset][i] = 0    # x_i
        A[offset][n_circles + i] = -1  # y_i
        A[offset][j] = 0    # x_j
        A[offset][n_circles + j] = 1   # y_j
        A[offset][2*n_circles + i] = 1  # r_i
        A[offset][2*n_circles + j] = 1  # r_j
        b[offset] = 0
        offset += 1

    # Wall constraints: r_i <= x_i, r_i <= 1 - x_i, r_i <= y_i, r_i <= 1 - y_i
    # y coords start at n_circles
    for i in range(n_circles):
        # r_i <= x_i  =>  -r_i + x_i >= 0
        A[offset][i] = 1    # x_i
        A[offset][2*n_circles + i] = -1  # r_i
        b[offset] = 0
        offset += 1

        # r_i <= 1 - x_i  =>  x_i + r_i <= 1
        A[offset][i] = 1    # x_i
        A[offset][2*n_circles + i] = 1  # r_i
        b[offset] = 1
        offset += 1

        # r_i <= y_i  =>  -r_i + y_i >= 0
        A[offset][n_circles + i] = 1  # y_i
        A[offset][2*n_circles + i] = -1  # r_i
        b[offset] = 0
        offset += 1

        # r_i <= 1 - y_i  =>  y_i + r_i <= 1
        A[offset][n_circles + i] = 1  # y_i
        A[offset][2*n_circles + i] = 1  # r_i
        b[offset] = 1
        offset += 1

    # Objective coefficient: maximize sum of radii
    c = np.zeros(n_vars)
    c[2*n_circles:] = -1

    return A, b, c


def solve_lp(A, b, c):
    # Solve the linear program using the "highs" method which is exact
    sol = linprog(c, A_ub=A, b_ub=b, method="highs", options={"presolve": True})

    # Extract only the solution values
    x = sol.x

    # Reshape to get centers and radii
    n_circles = (x.shape[0] - 26) // 2  # 26, 26, 26
    centers = x[:2*n_circles].reshape(n_circles, 2)
    radii = x[2*n_circles:]

    # All as NumPy arrays:
    assert centers.shape == (n_circles, 2)
    assert radii.shape == (n_circles,)

    return centers, radii


def detect_diagonal_conflicts(centers, radii):
    """Return list of pairs of indices that violate Euclidean non-overlap constraint."""
    pairs = []
    n = len(centers)
    for i in range(n):
        for j in range(i+1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            total_radius = radii[i] + radii[j]
            if dist < total_radius - 1e-12:
                pairs.append((i, j))
    return pairs


def resolve_conflicts(centers, radii, active_set, conflict_pairs):

        current_radii = ini_radii.copy()
        for idx, r in enumerate(involved):
            current_radii[r] = r_vals[idx]
        current_centers = ini_centers.copy()
        # All involved have x,y values from goal
        # Return the constraint violations: 0 if good

        # Materials are still only individual circles' non-overlap
        # Compute all pairwise distances,
        all_dist = cdist(ini_centers, ini_centers)
        boxes = [[0, 1], [0, 1]]

                if d < rad_sum - 1e-12:
                    # Add a constraint for this bond
                    violations.append(d - rad_sum)  # Negative if violated
        return np.array(violations)

    bounds = [(1e-9, 1 - 1e-9) for i in range(len(involved))]
    starting_radii = [radii[i] for i in involved]
    # Simple objective: negative sum (since we're minimizing)
    # but since the current solution is already very likely better, run it briefly
    # The conflict resolution is harder than accurate.
    # For now we mostly avoid solving small problems here.
    # Instead, proceed directly to more direct polishing.

def polish_global(centers, radii, active_set):
    """Final polishing using trust-constr to optimize over all variables."""
    # Active constraints: solve with constrained optimization maintaining them
    # We're going to make constraints:
    #   - Wall inequalities: r <= x, r <= 1-x, r <= y, r <= 1-y
    #   - Pairwise equality, often many or none, note this is tricky to extract
    #   - Non-overlap as regular inequalities except what's already defined

    # Since we've relaxed into LP, now go back to exact Euclidean constraints.
    # Our only mental constraint is that we fix no full equalities.

    n_circles = centers.shape[0]
    x0 = np.concatenate([centers.reshape(-1), radii])

    def obj(x):
        # Return -sum of radii
        r = x[-n_circles:]
        return -np.sum(r)

    def constraint_wall(x):
        # All wall inequalities must be maintained (or力を bind)
        ch = x[:2*n_circles].reshape(n_circles, 2)
        r = x[2*n_circles:]
        res = np.concatenate([
            r-ch[:,0], # r <= x
            1-r-ch[:,0], # r <= 1-x
            r-ch[:,1], # r <= y
            1-r-ch[:,1] # r <= 1-y
        ])
        return res  # easier as >= 0

    def constraint_overlap(x):
        # Ensures d(c_i, c_j) >= r_i + r_j
        # Create mathematically stronger version
        ch = x[:2*n_circles].reshape(n_circles, 2)
        r = x[2*n_circles:]

        # Invalid (just a nameless function)
        res = []
        for i in range(n_circles):
            for j in range(i+1, n_circles):
                d = np.sqrt(np.sum(np.square(ch[i]-ch[j])))
                dist = d - (r[i] + r[j])
                res.append(dist)
        return np.array(res)

    # Run a feasibility version only
    # Retain full constraint logic for that step.
    # Let's restructure
    # Double-check our constraint representation
    # Makes a real copper cold, separate market:
    # equations are:
    # 1. All points (x,y) pre-bound at edge.
    # 2. Always R_i <= x_i, 1-x_i, y_i, 1-y_i
    # 3. Each pair i,j gets Euclidean non-overlap.

    # Based on how hard it is to track active pairs, just cut the geometric by triangle way
    # Then we'll mostly leave final local balancing
    bounds = []
    for i in range(2*n_circles):
        bounds.append((1e-6, 1-1e-6))  # Safe
    for i in range(n_circles):
        bounds.append((1e-6, 0.5))     # Also safe only some IoT way

    def wrapper_con_wall(x):
        # Want all: r <= (or equivalent forms) the ratio
        ch = x[:2*n_circles].reshape(n_circles, 2)
        r = x[2*n_circles:]
        # Put into negative form: < 0 if violating
        return np.concatenate([
            r - ch[:,0],
            1 - r - ch[:,0],
            r - ch[:,1],
            1 - r - ch[:,1]
        ])

    def wrapper_con_over(x):
        ch = x[:2*n_circles].reshape(n_circles, 2)
        r = x[2*n_circles:]
        # d >= r_i + r_j
        out = []
        for i in range(n_circles):
            for j in range(i+1, n_circles):
                d = np.sqrt(np.sum(np.square(ch[i] - ch[j])))
                out.append(d - (r[i] + r[j]))
        return np.array(out)

    res = minimize(obj,
                   x0,
                   method='trust-constr',
                   constraints=[
                       {'type': 'ineq', 'fun': lambda x: wrapper_con_wall(x)},
                       {'type': 'ineq', 'fun': lambda x: wrapper_con_over(x)}
                   ],
                   bounds=bounds,
                   options={'maxiter': 200}
                   )
    res = res.x.copy()
    ch = res[:2*n_circles].reshape(n_circles, 2)
    r = res[2*n_circles:]

    # Ensure result is valid, immediately
    if np.any(r < 0) or np.isnan(r).any():
        raise ValueError("Invalid radii in polishing")
    return ch, r


def max_radius_projection(centers, radii):
    """Adjust centers and radii to be locally maximum in feasible domain."""
    n = len(centers)
    changed = True
    while changed:
        changed = False
        new_radii = []
        for i in range(n):
            px, py = centers[i]
            r_bound = min(px, 1 - px, py, 1 - py)
            min_d = float('inf')
            for j in range(n):
                if i == j:
                    continue
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                ds = dist - radii[j]  # distance to circle j
                min_d = min(min_d, ds)
            new_r = min(r_bound, min_d)
            if abs(new_r - radii[i]) > 1e-12:
                changed = True
            new_radii.append(new_r)
        # Clamp new radii to keep internal bounds valid
        for i in range(n):
            radii[i] = new_radii[i]

        # Update centers to fit maximal new radii as good as possible
        # but use the position information only as much as not interacting with others.
        # Since they've been given correct values from polishing, preserve positions
        # and just use this to validate internal properties vs actual best fits.
    return centers, radii
