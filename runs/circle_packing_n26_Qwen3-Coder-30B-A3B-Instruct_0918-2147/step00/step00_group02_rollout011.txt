import numpy as np
from scipy.optimize import minimize
from scipy.stats import qmc

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    # Initial configuration
    centers, radii = initial_configuration(n)
    z_start = pack(centers, radii)

    # Energy parameters
    B0 = 3.0
    lambda_budget = 1000.0
    delta = 5e-4
    max_iter = 40
    eps = 1e-12

    # Binary search on budget
    B = B0
    for _ in range(max_iter):
        z_opt = inner_optimize(z_start, B, lambda_budget)
        sum_r = np.sum(z_opt[n*2:n*2+n])
        if np.allclose(sum_r, B, atol=eps):
            B += delta
        else:
            B -= delta
        if abs(delta) < 1e-6:
            break
        z_start = z_opt

    # Final processing
    centers, radii = unpack(z_opt)
    centers = np.clip(centers, eps, 1 - eps)
    radii = np.maximum(0, radii - 5e-10)
    sum_r = np.sum(radii)
    
    # Repair pass to ensure valid configuration
    max_iter_repair = 5
    for _ in range(max_iter_repair):
        valid, _ = validate_packing(centers, radii)
        if valid:
            break
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                r_sum = radii[i] + radii[j]
                if dist < r_sum:
                    overlap = r_sum - dist
                    radii[i] -= overlap / 2 + eps
                    radii[j] -= overlap / 2 + eps
                    radii[i] = max(0, radii[i])
                    radii[j] = max(0, radii[j])

    sum_r = np.sum(radii)
    assert sum_r > 2.636, "Sum of radii is below required threshold"
    return centers, radii, sum_r

def initial_configuration(n):
    # Use Sobol sequence for better distribution
    sampler = qmc.Sobol(d=2, seed=42)
    points = sampler.random(n)
    centers = points.copy()
    radii = np.zeros(n)
    
    for i in range(n):
        x, y = centers[i]
        r_wall = min(x, 1-x, y, 1-y)
        # Compute min distance to other points
        min_dist = float('inf')
        for j in range(n):
            if i == j:
                continue
            dx = x - centers[j, 0]
            dy = y - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            if dist < min_dist:
                min_dist = dist
        r_neighbour = min_dist / 2 if min_dist < float('inf') else 0.5
        radii[i] = min(r_wall, r_neighbour)
    
    # Ensure initial is feasible
    radii = np.maximum(0.0, radii)
    
    return centers, radii

def pack(centers, radii):
    """Pack centers and radii into a flat vector"""
    z = np.empty((26*2 + 26,))
    z[:26*2] = centers.flatten()
    z[26*2:] = radii
    return z

def unpack(z):
    """Unpack flat vector into centers and radii"""
    n = 26
    centers = z[:26*2].reshape((n, 2))
    radii = z[26*2:]
    return centers, radii

def energy_and_grad(z):
    """Compute total energy and its gradient"""
    n = 26
    centers, radii = unpack(z)
    
    # Compute all pairwise distances
    centers_reshaped = centers[:, np.newaxis, :]
    diff = centers_reshaped - centers[np.newaxis, :, :]
    distances = np.sqrt(np.sum(diff**2, axis=2))
    
    # Pairwise overlap terms
    pairs = np.triu_indices(n, k=1)
    r_sum = radii[pairs[0]] + radii[pairs[1]]
    overlaps = np.maximum(0, r_sum - distances[pairs])
    
    energy_pair = np.sum(overlaps**2)
    
    # Wall forces
    wall_term = 0
    grad_wall_x = np.zeros(n)
    grad_wall_y = np.zeros(n)
    grad_wall_r = np.zeros(n)
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        dx = max(0, r - x)
        dy = max(0, r - (1-x))
        dz = max(0, r - y)
        dw = max(0, r - (1-y))
        wall_term += dx**2 + dy**2 + dz**2 + dw**2
        
        if dx > 0:
            grad_wall_x[i] -= 2*dx
            grad_wall_r[i] += 2*dx
        if dy > 0:
            grad_wall_x[i] -= 2*dy
            grad_wall_r[i] += 2*dy
        if dz > 0:
            grad_wall_y[i] -= 2*dz
            grad_wall_r[i] += 2*dz
        if dw > 0:
            grad_wall_y[i] -= 2*dw
            grad_wall_r[i] += 2*dw
    
    # Budget term
    lambda_budget = 1000.0
    sum_r = np.sum(radii)
    energy_budget = lambda_budget * (sum_r - 3.0)**2
    grad_budget_r = -2 * lambda_budget * (sum_r - 3.0)
    
    # Total energy
    total_energy = wall_term + energy_pair + energy_budget
    
    # Gradient computation
    grad_centers = np.zeros_like(centers)
    grad_radii = np.zeros(n)
    
    # Add wall forces to gradients
    grad_radii += grad_wall_r
    grad_centers[:, 0] += grad_wall_x
    grad_centers[:, 1] += grad_wall_y
    
    # Add overlap forces to gradients
    for i in range(len(pairs[0])):
        idx1, idx2 = pairs[0][i], pairs[1][i]
        if overlaps[i] > 0:
            dx = centers[idx1, 0] - centers[idx2, 0]
            dy = centers[idx1, 1] - centers[idx2, 1]
            dist = max(distances[idx1, idx2], 1e-12)
            force = 2 * overlaps[i] / dist
            grad_centers[idx1, 0] -= force * dx
            grad_centers[idx1, 1] -= force * dy
            grad_centers[idx2, 0] += force * dx
            grad_centers[idx2, 1] += force * dy
            
            grad_radii[idx1] += 2 * overlaps[i]
            grad_radii[idx2] += 2 * overlaps[i]
    
    # Add budget term to radii gradient
    grad_radii += grad_budget_r
    
    # Pack gradients
    grad = np.empty_like(z)
    grad[:26*2] = grad_centers.flatten()
    grad[26*2:] = grad_radii
    
    return total_energy, grad

def inner_optimize(z0, B, lambda_budget):
    # Optimization bounds
    bounds = []
    n = 26
    for _ in range(n):
        bounds.append((0, 1))  # x bounds
        bounds.append((0, 1))  # y bounds
        bounds.append((0, 0.5))  # r bounds
    
    # Adjust initial z to fit bounds
    z0 = np.minimum(z0, [1.0] * (n*2) + [0.5] * n)
    z0 = np.maximum(z0, [0.0] * (n*2) + [0.0] * n)
    
    # Final configuration redefined
    def wrapper_func(z):
        z_n = z.copy()
        z_n[n*2:] = np.minimum(z[n*2:], 0.5) 
        z_n[n*2:] = np.maximum(z[n*2:], 0.0)
        energy, grad = energy_and_grad(z_n)
        return energy, grad
    
    res = minimize(wrapper_func, z0, method="L-BFGS-B", bounds=bounds, jac=True, options={"maxiter": 200})
    if not res.success:
        raise RuntimeError("Optimization failed")
    return res.x

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
