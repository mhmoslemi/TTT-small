import numpy as np
from scipy.optimize import minimize

def make_hex_lattice(n, margin=0.05):
    # Generate points on a hexagonal lattice, centered in unit square
    # with margin from edges
    dx = 1.0 / (np.ceil(np.sqrt(n)) + 0.5)
    dy = dx * np.sqrt(3) / 2
    points = []
    i_start = int(np.ceil(margin / dx))
    j_start = int(np.ceil(margin / dy))
    i_end = int(np.ceil((1 - margin) / dx)) + 1
    j_end = int(np.ceil((1 - margin) / dy)) + 1

    for j in range(j_start, j_end):
        for i in range(i_start, i_end):
            x = i * dx + (j % 2) * dx / 2
            y = j * dy
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append([x, y])
                if len(points) >= n:
                    break
        if len(points) >= n:
            break

    points = np.array(points[:n])
    # Center the pattern
    center = np.mean(points, axis=0)
    offset = 0.5 - center
    points += offset
    # Ensure all points are within margin of the square
    points = np.clip(points, margin, 1 - margin)
    return points

def objective(z):
    n = len(z) // 3
    radii = z[2::3]
    return -np.sum(radii)

def boundary_constraints(z):
    n = len(z) // 3
    result = []
    for i in range(n):
        x = z[3*i]
        y = z[3*i+1]
        r = z[3*i+2]
        result.append(x - r)  # x - r >= 0
        result.append(1 - x - r)  # 1 - x - r >= 0
        result.append(y - r)  # y - r >= 0
        result.append(1 - y - r)  # 1 - y - r >= 0
    return np.array(result)

def pairwise_constraints(z):
    n = len(z) // 3
    result = []
    for i in range(n):
        for j in range(i + 1, n):
            x1, y1, r1 = z[3*i], z[3*i+1], z[3*i+2]
            x2, y2, r2 = z[3*j], z[3*j+1], z[3*j+2]
            dist_sq = (x1 - x2)**2 + (y1 - y2)**2
            sum_radii = r1 + r2
            result.append(dist_sq - sum_radii**2)
    return np.array(result)

def solve_nlp(initial_z):
    n = len(initial_z) // 3
    bounds = []
    for i in range(n):
        bounds.extend([(0, 1), (0, 1), (0, 0.5)])
    constraints = []
    constraints.append({
        'type': 'ineq',
        'fun': boundary_constraints
    })
    constraints.append({
        'type': 'ineq',
        'fun': pairwise_constraints
    })
    res = minimize(
        objective, initial_z,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 1000, 'ftol': 1e-9}
    )
    return res.x

def nudge_centers(centers, radii, z):
    n = len(centers)
    nudged_z = z.copy()
    for i in range(n):
        x, y, r = z[3*i], z[3*i+1], z[3*i+2]
        if abs(x - r) < 1e-6 or abs(1 - x - r) < 1e-6 or \
           abs(y - r) < 1e-6 or abs(1 - y - r) < 1e-6:
            dx, dy = 0.0, 0.0
            if abs(x - r) < 1e-6:
                dx = 0.001
            elif abs(1 - x - r) < 1e-6:
                dx = -0.001
            if abs(y - r) < 1e-6:
                dy = 0.001
            elif abs(1 - y - r) < 1e-6:
                dy = -0.001
            if abs(dx) > 0 or abs(dy) > 0:
                nudged_z[3*i] += dx
                nudged_z[3*i+1] += dy
    return nudged_z

def run_packing():

    initial_centers = make_hex_lattice(n, margin=0.05)
    r0 = 0.02
    initial_z = np.zeros(3 * n)

        initial_z[3*i+1] = initial_centers[i, 1]

    for it in range(5):

        centers = new_z[::3].reshape(n, 2)
        radii = new_z[2::3]
        if np.sum(radii) - np.sum(initial_z[2::3]) < 1e-5:

        initial_z = nudge_centers(centers, radii, new_z)
        # Restore centers from z
