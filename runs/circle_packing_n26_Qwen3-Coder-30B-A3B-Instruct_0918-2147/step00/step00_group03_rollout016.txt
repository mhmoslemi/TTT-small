import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
from itertools import combinations

def init_candidate_grid(R, Δ):
    candidates = []
    for r in R:
        s = r * np.sqrt(2)
        x_range = np.arange(r, 1 - r + 1e-10, s)
        y_range = np.arange(r, 1 - r + 1e-10, s)
        for x in x_range:
            for y in y_range:
                candidates.append((x, y, r))
    return candidates

def build_overlap_matrix(candidates):
    n = len(candidates)
    overlap = np.zeros((n, n), dtype=bool)
    coords = np.array([[c[0], c[1]] for c in candidates])
    radii = np.array([c[2] for c in candidates])
    distances = cdist(coords, coords)
    for i in range(n):
        for j in range(i + 1, n):
            if distances[i, j] < radii[i] + radii[j]:
                overlap[i, j] = True
                overlap[j, i] = True
    return overlap

def max_radii_fixed_centres(centers, radii_initial):
    # For each circle, compute the maximum allowable radius
    radii = radii_initial.copy()
    n = len(centers)
    for i in range(n):
        x, y = centers[i]
        r_min = min(x, y, 1 - x, 1 - y)
        # Compute non-overlapping radii
        for j in range(n):
            if i != j:
                dist = np.sqrt((centers[i][0] - centers[j][0])**2 + (centers[i][1] - centers[j][1])**2)
                r_max = dist - radii[j]
                if r_max > 0:
                    r_min = min(r_min, r_max)
        radii[i] = r_min
    return radii

def nlp_refine(centers_initial, radii_initial):
    n = len(centers_initial)
    bounds = []
    # x,y, r for each circle
    num_vars = 2 * n + n
    variables = np.zeros(num_vars)
    
    for i in range(n):
        variables[2*i] = centers_initial[i][0]
        variables[2*i+1] = centers_initial[i][1]
        variables[2*n+i] = radii_initial[i]
        
    # Bounds
    for i in range(n):
        bounds.append((1e-9, 1 - 1e-9))  # x
        bounds.append((1e-9, 1 - 1e-9))  # y
        bounds.append((1e-9, 1 - 1e-9))  # r
    
    def objective(vars):
        sum_radii = sum(vars[2*n+i] for i in range(n))
        return -sum_radii  # Minimize negation to maximize

    def constraints(vars):
        # Separation constraints
        # Wall constraints
        eqs = []
        for i in range(n):
            x, y, r = vars[2*i], vars[2*i+1], vars[2*n+i]
            # Wall constraints: r <= x, r <= y, r <= 1-x, r <= 1-y
            eqs.append(r - x)
            eqs.append(r - y)
            eqs.append(r - (1 - x))
            eqs.append(r - (1 - y))
        # Non-overlap constraints
        for i in range(n):
            for j in range(i+1, n):
                x1, y1, r1 = vars[2*i], vars[2*i+1], vars[2*n+i]
                x2, y2, r2 = vars[2*j], vars[2*j+1], vars[2*n+j]
                dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
                eqs.append(r1 + r2 - dist)
        return np.array(eqs)

    cons = {'type': 'ineq', 'fun': constraints}
    sol = minimize(objective, variables, method='trust-constr', bounds=bounds, constraints=cons, options={'maxiter': 300, 'xtol': 1e-9, 'gtol': 1e-9})
    
    if sol.success:
        refined_centers = np.array([[sol.x[2*i], sol.x[2*i+1]] for i in range(n)])
        refined_radii = np.array([sol.x[2*n+i] for i in range(n)])
        return refined_centers, refined_radii
    else:
        raise RuntimeError("NLP failed")
