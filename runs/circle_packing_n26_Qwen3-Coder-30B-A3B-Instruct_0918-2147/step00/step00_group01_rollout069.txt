import numpy as np
from scipy.optimize import minimize

def initial_layout():
    # Create a hexagonal grid pattern inside the unit square
    n_cols = 5
    n_rows = 5
    spacing = 0.18
    radius = spacing / 2 - 0.01
    
    centers = []
    for i in range(n_rows):
        for j in range(n_cols):
            x = (j + 0.5) * spacing
            y = (i + 0.5) * spacing
            # Shift every other row
            if i % 2 == 1:
                x += spacing / 2
            # Keep within bounds
            x = max(radius, min(1 - radius, x))
            y = max(radius, min(1 - radius, y))
            centers.append([x, y])
    
    # Add one more circle in the center
    centers.append([0.5, 0.5])
    
    # Make sure we have exactly 26 centers
    centers = centers[:26]
    
    radii = [radius] * 26
    
    return np.array(centers), np.array(radii)

def pack_vector(centers, radii):
    z = np.empty(3 * 26)
    for i in range(26):
        z[3*i] = centers[i, 0]
        z[3*i+1] = centers[i, 1]
        z[3*i+2] = radii[i]
    return z

def unpack_vector(z):
    centers = np.empty((26, 2))
    radii = np.empty(26)
    for i in range(26):
        centers[i, 0] = z[3*i]
        centers[i, 1] = z[3*i+1]
        radii[i] = z[3*i+2]
    return centers, radii

def objective(z):
    centers, radii = unpack_vector(z)
    return -np.sum(radii)

def constraint_boundary(z):
    centers, radii = unpack_vector(z)
    constraints = []
    for i in range(26):
        x, y = centers[i]
        r = radii[i]
        # x - r >= 0
        constraints.append(x - r)
        # 1 - x - r >= 0
        constraints.append(1 - x - r)
        # y - r >= 0
        constraints.append(y - r)
        # 1 - y - r >= 0
        constraints.append(1 - y - r)
    return np.array(constraints)

def constraint_nonoverlap(z):
    centers, radii = unpack_vector(z)
    constraints = []
    for i in range(26):
        for j in range(i+1, 26):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx*dx + dy*dy
            r_sum = radii[i] + radii[j]
            # (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
            constraint = dist_sq - r_sum * r_sum
            constraints.append(constraint)
    return np.array(constraints)

def run_packing():

    best_centers = None
    best_radii = None
    
    for restart in range(3):
        # Generate initial layout
        centers, radii = initial_layout()
        
        # Add a little noise to avoid worst-case starting points
        if restart > 0:
            centers += np.random.uniform(-0.01, 0.01, centers.shape)
            radii += np.random.uniform(-0.005, 0.005, radii.shape)
            # Ensure non-negative
            radii = np.maximum(radii, 0.001)
        
        # Pack into vector
        z0 = pack_vector(centers, radii)
        
        # Define bounds
        bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * 26
        
        # Define constraints
        cons = [
            {'type': 'ineq', 'fun': constraint_boundary},
            {'type': 'ineq', 'fun': constraint_nonoverlap}
        ]
        
        # Run optimization
        result = minimize(
            objective,
            z0,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 2000}
        )
        
        # Unpack result
