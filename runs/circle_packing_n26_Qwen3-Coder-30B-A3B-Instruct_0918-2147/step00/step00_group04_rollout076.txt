def build_leaf_problem(cell, num_circles):
    """Builds and solves a standalone SOCP for a cell with num_circles circles"""
    if num_circles == 0:
        return np.array([]), np.array([]), set()

    x_min, y_min, width, height = cell['x'], cell['y'], cell['width'], cell['height']
    x_max, y_max = x_min + width, y_min + height

    # Use cvxpy variables for centers and radii
    centers = cp.Variable((num_circles, 2))
    radii = cp.Variable(num_circles, nonneg=True)

    # constraints:
    constrs = []

    # Wall boundary constraints
    for i in range(num_circles):
        constrs.append(centers[i, 0] - radii[i] >= x_min)
        constrs.append(centers[i, 0] + radii[i] <= x_max)
        constrs.append(centers[i, 1] - radii[i] >= y_min)
        constrs.append(centers[i, 1] + radii[i] <= y_max)

    # Internal non-overlap constraints (quadratic, expressed as SOCPs)
    for i in range(num_circles):
        for j in range(i+1, num_circles):
            # for each pair, we try to make the distance exactly r_i + r_j
            # Here: norm(c_i - c_j) >= r_i + r_j
            # So: norm(c_i - c_j) - (r_i + r_j) >= 0
            dist_var = cp.Variable(nonneg=True)
            constrs.append(dist_var >= cp.norm(centers[i] - centers[j]))
            constrs.append(dist_var >= radii[i] + radii[j])

    # Objective is to maximize sum of radii
    obj = cp.Maximize(cp.sum(radii))
    prob = cp.Problem(obj, constrs)
    prob.solve(solver=cp.ECOS, abstol=1e-12, reltol=1e-12)

    if prob.status in ['optimal', 'optimal_inaccurate']:
        final_centers = centers.value
        final_radii = radii.value
        # return centers and radii, and "active set" of which constraints are tight
        return final_centers, final_radii, set()
    else:
        raise Exception(f"Failed to solve leaf SOCP for cell {cell['idx']}, status: {prob.status}")

def merge_clusters(cluster_a, cluster_b, shared_edge):
    # indices refer to the current cell indices
    # default case: merge by constraints
    num_circles = len(cluster_a) + len(cluster_b)
    if num_circles == 0:
        return np.array([]), np.array([])

    # Select indices for A, B circles
    ids_a = range(len(cluster_a))
    ids_b = range(len(cluster_a), len(cluster_a) + len(cluster_b))
    
    # merge sets of circles
    centers = cp.Variable((num_circles, 2))
    radii = cp.Variable(num_circles, nonneg=True)

    constrs = []
    
    # Add boundary constraints from both reports
    for i, (cid, c, r) in enumerate(cluster_a):
        if c[0] - r <= 0 or c[0] + r >= 1:
            constrs.append(centers[i, 0] - radii[i] >= 0)
            constrs.append(centers[i, 0] + radii[i] <= 1)
        if c[1] - r <= 0 or c[1] + r >= 1:
            constrs.append(centers[i, 1] - radii[i] >= 0)
            constrs.append(centers[i, 1] + radii[i] <= 1)
    
    for i, (cid, c, r) in enumerate(cluster_b):
        offset_i = len(cluster_a)
        if c[0] - r <= 0 or c[0] + r >= 1:
            constrs.append(centers[i + offset_i, 0] - radii[i + offset_i] >= 0)
            constrs.append(centers[i + offset_i, 0] + radii[i + offset_i] <= 1)
        if c[1] - r <= 0 or c[1] + r >= 1:
            constrs.append(centers[i + offset_i, 1] - radii[i + offset_i] >= 0)
            constrs.append(centers[i + offset_i, 1] + radii[i + offset_i] <= 1)

    # Internal non-overlap constraint between all pairs in merged cluster
    for i in range(num_circles):
        for j in range(i+1, num_circles):
            dist_var = cp.Variable(nonneg=True)
            constrs.append(dist_var >= cp.norm(centers[i] - centers[j]))
            constrs.append(dist_var >= radii[i] + radii[j])

    obj = cp.Maximize(cp.sum(radii))
    prob = cp.Problem(obj, constrs)
    prob.solve(solver=cp.ECOS, abstol=1e-12, reltol=1e-12)
    if prob.status in ['optimal', 'optimal_inaccurate']:
        final_centers = centers.value
        final_radii = radii.value
        return [(i, final_centers[i], final_radii[i]) for i in range(num_circles)]
    else:
        raise Exception(f"Failed to merge clusters {cluster_a} and {cluster_b}")
