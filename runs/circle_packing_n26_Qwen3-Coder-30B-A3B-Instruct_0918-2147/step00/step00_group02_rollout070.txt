import numpy as np
from scipy.optimize import minimize
from sobol import Sobol

def initial_configuration(n=26):
    # Generate low-discrepancy Sobol points
    sobol = Sobol(2)
    points = sobol.generate(n)
    # Clip to [0.01, 0.99] to avoid boundary issues in initial radii calculation
    points = np.clip(points, 0.01, 0.99)
    centers = points

    # Compute initial radii based on wall distances and neighbor distances
    radii = np.zeros(n)
    for i in range(n):
        x, y = centers[i]
        # Wall distances
        dwall = np.array([x, 1-x, y, 1-y])
        # Distance to nearest neighbor
        dists = np.sqrt(np.sum((centers - centers[i])**2, axis=1))
        dists[i] = np.inf # exclude self
        dneighbor = np.min(dists)
        # Initial radius is the smaller of wall limit and half neighbor distance
        radii[i] = min(np.min(dwall), dneighbor / 2.0)

    return centers, radii

def pack(centers, radii):
    """Convert to flattened vector"""
    n = centers.shape[0]
    z = np.zeros(3*n)
    z[0::3] = centers[:, 0]
    z[1::3] = centers[:, 1]
    z[2::3] = radii
    return z

def unpack(z):
    """Convert from flattened vector"""
    n = z.shape[0] // 3
    centers = np.zeros((n, 2))
    centers[:, 0] = z[0::3]
    centers[:, 1] = z[1::3]
    radii = z[2::3]
    return centers, radii

def max_violation(centers, radii):
    """Compute maximum constraint violation"""
    n = centers.shape[0]
    # Wall violations
    violations_wall = np.maximum(0, radii - centers[:, 0]) + \
                      np.maximum(0, radii - (1 - centers[:, 0])) + \
                      np.maximum(0, radii - centers[:, 1]) + \
                      np.maximum(0, radii - (1 - centers[:, 1]))
    # Pairwise overlaps
    violations_pair = 0
    for i in range(n):
        for j in range(i+1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            ov = np.maximum(0, radii[i] + radii[j] - dist)
            violations_pair = max(violations_pair, ov)
    return max(np.max(violations_wall), violations_pair)

def energy_and_grad(z):
    """Compute total energy and gradient"""
    centers, radii = unpack(z)
    n = centers.shape[0]

    # Hard-coded penalty parameters
    lambda_budget = 1000.0

    # Compute distance matrix D
    D = np.zeros((n, n))
    for i in range(n):
        for j in range(i+1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            D[i, j] = dist
            D[j, i] = dist

    # Wall penalties
    wall_energy = 0.0
    wall_grad = np.zeros_like(z)
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Wall distances
        dxl, dxr = x, 1.0 - x
        dyl, dyr = y, 1.0 - y
        dwall = np.array([dxl, dxr, dyl, dyr])

        wall_energy += np.sum(np.maximum(0, r - dwall)**2)

        # Derivatives w.r.t. center and radius
        d_wall_r = 2 * np.sum(np.maximum(0, r - dwall))
        d_wall_x = 0.0
        d_wall_y = 0.0
        for k in range(4):
            if r > dwall[k]:
                if k == 0: d_wall_x -= 2 * (r - dwall[k]) # x - r < 0
                if k == 1: d_wall_x += 2 * (r - dwall[k]) # (1-x) - r < 0
                if k == 2: d_wall_y -= 2 * (r - dwall[k]) # y - r < 0
                if k == 3: d_wall_y += 2 * (r - dwall[k]) # (1-y) - r < 0
        wall_grad[3*i + 2] += d_wall_r
        wall_grad[3*i] += d_wall_x
        wall_grad[3*i + 1] += d_wall_y

    # Pairwise overlap penalties
    pair_energy = 0.0
    pair_grad = np.zeros_like(z)

    for i in range(n):
        for j in range(i+1, n):
            d = D[i, j]
            r_sum = radii[i] + radii[j]
            if r_sum > d:
                overlap = r_sum - d
                pair_energy += overlap * overlap

                # Gradient w.r.t. centers and radii
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                norm = np.sqrt(dx*dx + dy*dy) + 1e-12
                d_overlap_d = -1.0 if norm > 1e-12 else 0.0
                d_overlap_dr = 2.0 * overlap

                # Gradient w.r.t. i
                pair_grad[3*i + 2] += d_overlap_dr  # for ri
                pair_grad[3*j + 2] += d_overlap_dr  # for rj
                if norm > 0.0:
                    pair_grad[3*i] -= d_overlap_d * dx / norm
                    pair_grad[3*i + 1] -= d_overlap_d * dy / norm
                # Gradient w.r.t. j
                if norm > 0.0:
                    pair_grad[3*j] += d_overlap_d * dx / norm
                    pair_grad[3*j + 1] += d_overlap_d * dy / norm

    # Budget penalty
    budget_energy = lambda_budget * (np.sum(radii) - B)**2
    budget_grad = np.zeros_like(z)
    for i in range(n):
        budget_grad[3*i + 2] -= 2 * lambda_budget * (np.sum(radii) - B)

    total_energy = wall_energy + pair_energy + budget_energy
    total_grad = wall_grad + pair_grad + budget_grad

    return total_energy, total_grad

def inner_optimize(z0, B0, lambda_budget):
    global B
    B = B0
    res = minimize(
        energy_and_grad,
        z0,
        method='L-BFGS-B',
        jac=True,
        bounds=[(0.0, 1.0)] * (3*26), # x, y bounded
        options={'maxiter': 500, 'ftol': 1e-9, 'gtol': 1e-9}
        )
    return res.x

def max_feasible_budget(z_start):
    B0 = 3.0
    delta = 5e-4
    B = B0
    lambda_budget = 1000.0
    z = z_start.copy()

    for _ in range(100):
        z_opt = inner_optimize(z, B, lambda_budget)
        centers, radii = unpack(z_opt)
        max_v = max_violation(centers, radii)
        feasi = max_v <= 1e-13
        if feasi:
            B += delta
            z = z_opt # accept this as the new base
        else:
            if delta < 1e-6:
                break
            B -= delta
            # no accept -- keep old z;

        if abs(delta) < 1e-6:
            break
        if B > 2.72 and delta > 1e-5:
            delta *= 0.5
    return z

def postprocess(z):
    centers, radii = unpack(z)
    eps = 5e-10

    # Shrink radii
    radii = np.maximum(0, radii - eps)
    # Clamp centers to keep away from boundaries
    centers = np.clip(centers, eps, 1 - eps)
    # Re-compute

    # Repair step if any overlap remains
    MAX_REPAIR = 5
    for _ in range(MAX_REPAIR):
        improved = False
        for i in range(len(centers)):
            for j in range(i+1, len(centers)):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                ov = radii[i] + radii[j] - dist
                if ov > 1e-10:
                    delta = ov / 2.0 + eps
                    radii[i] -= delta
                    radii[j] -= delta
                    improved = True
        if not improved:
            break

    # Final postprocessing
    radii = np.maximum(0, radii - eps) # final epsilon shrink
    sum_r = np.sum(radii)

    return centers, radii, sum_r

def run_packing():
    centers0, radii0 = initial_configuration(26)
    z0 = pack(centers0, radii0)
    z_final = max_feasible_budget(z0)
    centers, radii, sum_r = postprocess(z_final)
    return centers, radii, sum_r
