import numpy as np
from scipy.optimize import minimize

def initial_layout():
    # Create a hexagonal-like layout with 25 circles in a 5x5 grid
    # Then place one more in the center
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    spacing = 0.18
    margin = 0.01
    
    # Generate a 5x5 grid of centers
    grid_size = 5
    y_coords = np.linspace(spacing, 1 - spacing, grid_size)
    x_coords = np.linspace(spacing, 1 - spacing, grid_size)
    
    idx = 0
    for i in range(grid_size):
        for j in range(grid_size):
            centers[idx] = [x_coords[j], y_coords[i]]
            radii[idx] = spacing / 2 - margin
            idx += 1
    
    # Place the 26th circle in the center (larger, placed last to squeeze in)
    centers[idx] = [0.5, 0.5]
    radii[idx] = 0.15  # Start with 0.15 radius, will adjust during optimization
    
    return centers, radii

def pack_vector(centers, radii):
    z = np.zeros(3 * len(centers))
    for i in range(len(centers)):
        z[3*i] = centers[i, 0]
        z[3*i+1] = centers[i, 1]
        z[3*i+2] = radii[i]
    return z

def unpack_vector(z):
    n = len(z) // 3
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = z[3*i]
        centers[i, 1] = z[3*i+1]
        radii[i] = z[3*i+2]
    return centers, radii

def objective(z):
    _, radii = unpack_vector(z)
    return -np.sum(radii)

def constraint_boundary(z):
    centers, radii = unpack_vector(z)
    n = len(centers)
    
    constraints = []
    
    for i in range(n):
        x, y, r = centers[i, 0], centers[i, 1], radii[i]
        # x - r >= 0
        constraints.append(x - r)
        # 1 - x - r >= 0
        constraints.append(1 - x - r)
        # y - r >= 0
        constraints.append(y - r)
        # 1 - y - r >= 0
        constraints.append(1 - y - r)
    
    return np.array(constraints)

def constraint_nonoverlap(z):
    centers, radii = unpack_vector(z)
    n = len(centers)
    
    constraints = []
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx*dx + dy*dy
            r_sum = radii[i] + radii[j]
            # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
            constraints.append(dist_sq - r_sum * r_sum)
    
    return np.array(constraints)

            {'type': 'ineq', 'fun': constraint_nonoverlap}
        ]

        result = minimize(

            z0,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'ftol': 1e-9, 'maxiter': 2000}
        )
