import numpy as np
from scipy.optimize import minimize

def initial_layout():
    # Hexagonal grid layout for 25 circles, then one more in the center
    n = 26
    rows = 5
    cols = 5
    spacing = 0.18
    radius = spacing / 2.0 - 0.01
    centers = []
    
    # Generate hexagonal grid
    for i in range(rows):
        for j in range(cols):
            x = j * spacing + 0.05
            y = i * spacing * np.sqrt(3) / 2.0 + 0.05
            if i % 2 == 1:
                x += spacing / 2.0
            centers.append([x, y])
    
    # Ensure all are within bounds
    for i in range(len(centers)):
        x, y = centers[i]
        if x < radius:
            x = radius
        if x > 1 - radius:
            x = 1 - radius
        if y < radius:
            y = radius
        if y > 1 - radius:
            y = 1 - radius
        centers[i] = [x, y]
    
    # Add one more circle in the center
    centers.append([0.5, 0.5])
    
    # Make sure we have exactly 26 points
    # If not, append more in the center
    while len(centers) < 26:
        centers.append([0.5, 0.5])
    centers = centers[:26]
    
    # Initialize all radii to the same value
    radii = [radius] * 26
    return np.array(centers), np.array(radii)

def pack_vector(centers, radii):
    z = np.zeros(3 * len(centers))
    for i in range(len(centers)):
        z[3*i] = centers[i][0]
        z[3*i+1] = centers[i][1]
        z[3*i+2] = radii[i]
    return z

def unpack_vector(z):
    n = len(z) // 3
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i][0] = z[3*i]
        centers[i][1] = z[3*i+1]
        radii[i] = z[3*i+2]
    return centers, radii

def objective(z):
    centers, radii = unpack_vector(z)
    return -np.sum(radii)

def constraint_boundary(z):
    centers, radii = unpack_vector(z)
    constraints = []
    for i in range(len(centers)):
        x, y = centers[i]
        r = radii[i]
        constraints.append(x - r)  # x - r >= 0
        constraints.append(1 - x - r)  # 1 - x - r >= 0
        constraints.append(y - r)  # y - r >= 0
        constraints.append(1 - y - r)  # 1 - y - r >= 0
    return np.array(constraints)

def constraint_nonoverlap(z):
    centers, radii = unpack_vector(z)
    constraints = []
    n = len(centers)
    for i in range(n):
        for j in range(i+1, n):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist_sq = dx*dx + dy*dy
            r_sum = radii[i] + radii[j]
            # Need: (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
            constraints.append(dist_sq - r_sum * r_sum)
    return np.array(constraints)

def run_packing():
    best_sum = 0
    best_centers = None
    best_radii = None

    for _ in range(3):
        centers, radii = initial_layout()
        # Add small random jitter to break symmetry
        centers += np.random.uniform(-0.01, 0.01, centers.shape)
        centers = np.clip(centers, 0, 1)
        z0 = pack_vector(centers, radii)

        # Define constraints
        constraints = []
        constraints.append({'type': 'ineq', 'fun': constraint_boundary})
        constraints.append({'type': 'ineq', 'fun': constraint_nonoverlap})

        # Bounds for x, y, r
        bounds = [(0, 1), (0, 1), (0, 0.5)] * len(centers)
        
        # Optimization
        result = minimize(objective, z0, method='SLSQP', constraints=constraints, 
                          bounds=bounds, options={'ftol': 1e-9, 'maxiter': 2000})

        # Extract solution
        centers_new, radii_new = unpack_vector(result.x)
        sum_radii = np.sum(radii_new)

        success, msg = validate_packing(centers_new, radii_new)
        if success and sum_radii > best_sum:
            best_sum = sum_radii
            best_centers = centers_new.copy()
            best_radii = radii_new.copy()

    return best_centers, best_radii, best_sum
