import numpy as np
from scipy.optimize import minimize

def initial_layout():
    # Create a hexagonal grid pattern with 5 rows and 5 columns
    rows, cols = 5, 5
    spacing = 0.18
    radius = spacing / 2 - 0.01  # Leave a small margin for overlap

    centers = []
    for i in range(rows):
        for j in range(cols):
            x = (j + 0.5) * spacing
            y = (i + 0.5) * spacing
            # Adjust for the square boundary
            x = max(radius, min(1 - radius, x))
            y = max(radius, min(1 - radius, y))
            centers.append([x, y])

    # Add a 26th circle in the center
    centers.append([0.5, 0.5])
    radii = [radius] * 26

    return np.array(centers), np.array(radii)

def pack_vector(centers, radii):
    z = np.zeros(3 * centers.shape[0])
    z[0::3] = centers[:, 0]
    z[1::3] = centers[:, 1]
    z[2::3] = radii
    return z

def unpack_vector(z):
    n = z.shape[0] // 3
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    centers[:, 0] = z[0::3]
    centers[:, 1] = z[1::3]
    radii = z[2::3]
    return centers, radii

def objective(z):
    centers, radii = unpack_vector(z)
    return -np.sum(radii)

def constraint_boundary(z):
    centers, radii = unpack_vector(z)
    n = centers.shape[0]
    constraints = []
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # x - r >= 0
        constraints.append(x - r)
        # 1 - x - r >= 0
        constraints.append(1 - x - r)
        # y - r >= 0
        constraints.append(y - r)
        # 1 - y - r >= 0
        constraints.append(1 - y - r)
    return np.array(constraints)

def constraint_nonoverlap(z):
    centers, radii = unpack_vector(z)
    n = centers.shape[0]
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist2 = dx * dx + dy * dy
            r_sum = radii[i] + radii[j]
            # We want dist2 >= (r_sum)^2
            constraints.append(dist2 - r_sum * r_sum)
    return np.array(constraints)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_sum = -np.inf
    best_centers = None
    best_radii = None

    for _ in range(3):
        # Use a fixed seed for reproducibility (but still randomized)
        np.random.seed(42)

        # Create initial layout
        centers, radii = initial_layout()

        # Perturb slightly
        centers += np.random.uniform(-0.02, 0.02, centers.shape)
        radii += np.random.uniform(-0.005, 0.005, radii.shape)
        radii = np.maximum(radii, 0.001)

        # Pack into vector
        z0 = pack_vector(centers, radii)

        # Set bounds
        bounds = [(0, 1), (0, 1), (0, 0.5)] * centers.shape[0]

        # Build constraints
        constraints = []
        constraints.append({'type': 'ineq', 'fun': constraint_boundary})
        constraints.append({'type': 'ineq', 'fun': constraint_nonoverlap})

        # Optimize
        result = minimize(
            objective,
            z0,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'ftol': 1e-9, 'maxiter': 2000}
        )

        if not result.success:
            continue

        centers_opt, radii_opt = unpack_vector(result.x)
        sum_radii = np.sum(radii_opt)

        # Check if it's valid
        valid, msg = validate_packing(centers_opt, radii_opt)
        if not valid:
            continue

        if sum_radii > best_sum:
            best_sum = sum_radii
            best_centers = centers_opt.copy()
            best_radii = radii_opt.copy()

    # Final check
    assert best_centers is not None, "Failed to find a valid solution"
    return best_centers, best_radii, best_sum
