import numpy as np
from scipy.optimize import linprog
from itertools import combinations

def generate_hex_lattice(h):
    sqrt3 = np.sqrt(3)
    points = []
    for j in range(int(1.0 / (h * sqrt3 / 2)) + 2):
        y = j * h * sqrt3 / 2
        if y > 1:
            break
        for i in range(int(1.0 / h) + 2):
            x = i * h + (j % 2) * (h / 2)
            if x > 1:
                break
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append([x, y])
    return np.array(points)

def build_neighbour_list(centers, max_dist):
    edges = []
    n = centers.shape[0]
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < max_dist:
                edges.append((i, j, dist))
    return edges

def solve_radius_lp(selected_indices, R_wall, neighbour_list):
    n_selected = len(selected_indices)
    if n_selected == 0:
        return np.array([]), True

    # Build LP
    # Variables: r_0, r_1, ..., r_{n_selected-1}
    # Objective: maximize sum(r_i)
    c = -np.ones(n_selected)
    
    # Inequality matrix A_ub * x <= b_ub
    A_ub = []
    b_ub = []

    # Wall constraints
    for i in range(n_selected):
        row = np.zeros(n_selected)
        row[i] = -1
        A_ub.append(row)
        b_ub.append(-R_wall[selected_indices[i]])
        
    # Separation constraints
    for i, j, dist in neighbour_list:
        if i in selected_indices and j in selected_indices:
            idx_i = selected_indices.index(i)
            idx_j = selected_indices.index(j)
            row = np.zeros(n_selected)
            row[idx_i] = 1
            row[idx_j] = 1
            A_ub.append(row)
            b_ub.append(dist)
    
    # Solve LP
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, method="highs")
    if result.success:
        return result.x, True
    else:
        return np.array([]), False

def upper_bound(partial_set, remaining_indices, R_wall):
    ub = 0
    for i in partial_set:
        ub += R_wall[i]
    for i in remaining_indices:
        ub += R_wall[i]
        break
    return ub

def branch_and_bound(centers, R_wall, neighbour_list, num_circles=26):
    # Initial feasibility check
    best_sum = -np.inf
    best_indices = []

    # Start with small codebook
    heap = []
    initial_candidates = list(range(len(centers)))
    # Sort by wall radius (decreasing) for presort heuristic
    initial_candidates.sort(key=lambda i: R_wall[i], reverse=True)

    # Set of all candidates
    all_candidates = set(initial_candidates)

    def recurse(current_indices, available_indices, partial_sum):
        nonlocal best_sum, best_indices
        depth = len(current_indices)
        
        if depth == num_circles:
            # Check phase
            radii, feasible = solve_radius_lp(current_indices, R_wall, neighbour_list)
            if feasible and np.sum(radii) > best_sum:
                best_sum = np.sum(radii)
                best_indices = current_indices[:]
            return
            
        if depth > num_circles:
            return

        # Compute optimistic upper bound
        ub = partial_sum
        remaining_count = num_circles - depth
        # Add largest available wall radii
        sorted_walls = sorted([R_wall[i] for i in available_indices], reverse=True)
        ub += sum(sorted_walls[:remaining_count])
        if ub <= best_sum:
            return  # Pruned

        # Smart branching: choose largest wall radius
        for i in available_indices:
            new_indices = current_indices + [i]
            new_available = available_indices - {i}
            new_sum = partial_sum + R_wall[i]
            recurse(new_indices, new_available, new_sum)
            break # Only explore first for greedy branching
    
    # Initiate recursive search on the full set
    # Use a simple optimization to avoid O(2^N) enumeration by removing most low value
    # We exhaustively handle first 30 to maximize initial seeds, but dynamically truncate as needed
    candidates = initial_candidates[:100] if len(initial_candidates) > 100 else initial_candidates
    candidates.sort(key=lambda i: R_wall[i], reverse=True)
    for i0 in range(len(candidates)):
        i = candidates[i0]
        recurse([i], all_candidates - {i}, R_wall[i])
        
        if abs(best_sum - 2.636) > 1e-6:
            # Early stopping of algorithm if we are good
            break
    return best_indices
