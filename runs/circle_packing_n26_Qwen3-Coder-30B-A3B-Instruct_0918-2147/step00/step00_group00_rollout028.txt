import numpy as np
from scipy.optimize import linprog
from itertools import combinations

def generate_hex_lattice(h):
    sqrt3 = np.sqrt(3)
    points = []
    max_i = int(1.0 / h) + 2
    max_j = int(1.0 / (h * sqrt3 / 2)) + 2

    for i in range(max_i):
        for j in range(max_j):
            x = i * h + (j % 2) * (h / 2)
            y = j * h * sqrt3 / 2
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append([x, y])

    return np.array(points)

def build_neighbour_list(centers, max_dist):
    n = centers.shape[0]
    edges = []
    for i in range(n):
        for j in range(i+1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            if dist <= max_dist:
                edges.append((i, j, dist))
    return edges

def solve_radius_lp(selected_indices, R_wall, neighbour_list):
    n_selected = len(selected_indices)
    if n_selected == 0:
        return np.array([]), True

    # Build LP: maximize sum of radii
    c = -np.ones(n_selected)  # coefficients for objective (negate for maximization)

    # Wall constraints: 0 <= r_i <= R_wall[i]
    A_ub = np.zeros((2*n_selected, n_selected))
    b_ub = np.zeros(2*n_selected)
    for i in range(n_selected):
        A_ub[i, i] = -1  # r_i >= 0
        A_ub[i+n_selected, i] = 1  # r_i <= R_wall[selected_indices[i]]
        b_ub[i] = 0
        b_ub[i+n_selected] = R_wall[selected_indices[i]]

    # Separation constraints: r_i + r_j <= d_ij
    # For each pair in neighbour_list involving selected indices:
    A_ub_sep = []
    b_ub_sep = []
    for i, j, d_ij in neighbour_list:
        if i in selected_indices and j in selected_indices:
            idx_i = selected_indices.index(i)
            idx_j = selected_indices.index(j)
            A_row = np.zeros(n_selected)
            A_row[idx_i] = 1
            A_row[idx_j] = 1
            A_ub_sep.append(A_row)
            b_ub_sep.append(d_ij)
    if A_ub_sep:
        A_ub = np.vstack([A_ub, A_ub_sep])
        b_ub = np.hstack([b_ub, b_ub_sep])

    # Solve LP
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    if res.success:
        return res.x, True
    else:
        return np.array([]), False

def upper_bound(partial_set, remaining_indices, R_wall):
    ub = 0
    for idx in partial_set:
        ub += R_wall[idx]
    # For remaining slots, add max R_wall of remaining indices
    max_r = 0
    for i in remaining_indices:
        if R_wall[i] > max_r:
            max_r = R_wall[i]
    remaining_slots = 26 - len(partial_set)
    ub += max_r * remaining_slots
    return ub

def branch_and_bound(centers, R_wall, neighbour_list):
    best_sum = -1.0
    best_selection = None
    best_radii = None

    def recurse(selected_set, remaining_indices, lb):
        nonlocal best_sum, best_selection, best_radii

        if len(selected_set) == 26:
            # solve exact LP
            radii, feasible = solve_radius_lp(selected_set, R_wall, neighbour_list)
            if feasible:
                total_radius = np.sum(radii)
                if total_radius > best_sum:
                    best_sum = total_radius
                    best_selection = selected_set[:]
                    best_radii = radii.copy()
                return True
            return False

        # Heuristic: pick next index with largest wall radius
        max_r = 0
        index_to_pick = -1
        for i in remaining_indices:
            if R_wall[i] > max_r:
                max_r = R_wall[i]
                index_to_pick = i

        if index_to_pick == -1:
            return False

        # Try including it:
        new_selected = selected_set + [index_to_pick]
        new_remaining = [i for i in remaining_indices if i != index_to_pick]
        ub = upper_bound(new_selected, new_remaining, R_wall)
        if ub <= lb:
            pass # Pruned by upper bound
        else:
            # Ensure partial selection is feasible (rough check)
            if len(selected_set) > 0:
                # Check if recent inclusion is too close/wall-constrained
                # Simple wall check (may be insufficient, but bound helps)
                pass
            recurse(new_selected, new_remaining, lb)

        # Try excluding it:
        recurse(selected_set, new_remaining, lb)

    # Start recursion
    recurse([], list(range(len(centers))), 0.0)
    return best_selection, best_radii

def local_continuous_refinement(centers, radii):
    # Refine to get small improvements
    from scipy.optimize import minimize
    n = centers.shape[0]
    x0 = np.concatenate([centers.flatten(), radii])

    def objective(x):
        return -np.sum(x[n*2:])

    def constraint_func(x):
        # containment
        centers_x = x[:n*2].reshape((n, 2))
        radii_vals = x[n*2:]

        cons = []
        for i in range(n):
            cx, cy = centers_x[i]
            r = radii_vals[i]
            if cx - r < 0:
                cons.append(-(cx - r))
            if cx + r > 1:
                cons.append(-(1 - cx - r))
            if cy - r < 0:
                cons.append(-(cy - r))
            if cy + r > 1:
                cons.append(-(1 - cy - r))

        # non-overlap
        for i in range(n):
            ci = centers_x[i]
            ri = radii_vals[i]
            for j in range(i+1, n):
                cj = centers_x[j]
                rj = radii_vals[j]
                dist = np.sqrt(np.sum((ci - cj)**2))
                cons.append(rj + ri - dist)  # Should be >= 0

        return np.array(cons)

    # bounds on x: [0,1] x for centers, [0,1] for radii
    bounds = [(0, 1) for _ in range(n*2)] + [(0, 1) for _ in range(n)]
    cons = {'type': 'ineq', 'fun': constraint_func}

    res = minimize(objective, x0, method='trust-constr', constraints=cons, bounds=bounds, options={'maxiter': 200})
    if res.success:
        updated_centers = res.x[:n*2].reshape((n, 2))
        updated_radii = res.x[n*2:]
    else:
        # fall back to original if refinement fails
        updated_centers = centers.copy()
        updated_radii = radii.copy()
    return updated_centers, updated_radii

def run_packing():
    # Generate hexagonal lattice with spacing h to get enough candidates
    h = 0.07  # fine enough
    centers = generate_hex_lattice(h)
    R_wall = np.minimum(centers[:, 0], 1 - centers[:, 0])
    R_wall = np.minimum(R_wall, centers[:, 1])
    R_wall = np.minimum(R_wall, 1 - centers[:, 1])

    # Build neighbour list
    neighbour_list = build_neighbour_list(centers, 0.4)

    # Run branch-and-bound
    selected_indices, radii = branch_and_bound(centers, R_wall, neighbour_list)
    if selected_indices is None:
        raise RuntimeError("Branch and bound failed to find a solution")

    # Local refinement
    centers_final = centers[selected_indices]
    centers_final, radii_final = local_continuous_refinement(centers_final, radii)

    # Final validation and adjustment if needed
    valid, msg = validate_packing(centers_final, radii_final)
    if not valid:

    # Ensure final separation is admitted
    sum_radii = float(np.sum(radii_final))
