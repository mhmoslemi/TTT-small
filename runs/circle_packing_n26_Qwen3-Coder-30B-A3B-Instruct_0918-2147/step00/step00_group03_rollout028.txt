import numpy as np
from scipy.optimize import minimize
import scipy.sparse as sp

def init_hex_grid(n=26, r_init=0.04):
    # Generate hexagonal grid points
    h = np.sqrt(3) * r_init
    rows = []
    y = r_init
    row_idx = 0
    while len(rows) < n:
        # Alternate row shift
        x_start = r_init if row_idx % 2 == 0 else 2 * r_init
        x = x_start
        row = []
        while x <= 1 - r_init and len(row) < n - len(rows):
            row.append((x, y))
            x += 2 * r_init
        if row:
            rows.extend(row)
        y += h
        row_idx += 1
        if y > 1 - r_init:
            break

    # truncate to exact number n
    centers = np.array(rows[:n])
    radii = np.full(n, r_init)
    return centers, radii

def max_radii(centers, radii):
    # Iteratively compute maximal radii for fixed centers
    max_iter = 100
    eps = 1e-12
    for _ in range(max_iter):
        # Wall limits
        wall = np.minimum(centers, 1 - centers).min(axis=1)
        # Neighbour limits
        D = np.sqrt(((centers[:,None,:] - centers[None,:,:])**2).sum(-1))
        np.fill_diagonal(D, np.inf)
        neighbor = D - radii[:,None]
        # Set these to 0 for the diagonal, and find minimum
        neighbor = np.maximum(neighbor, 0)
        neighbor_min = neighbor.min(axis=1)
        new_radii = np.minimum(wall, neighbor_min)
        new_radii = np.clip(new_radii, 0, None)
        if np.max(np.abs(new_radii - radii)) < eps:
            break
        radii = new_radii
    return radii

def hill_climb(centers, radii, iterations=5000, step=0.01):
    # Stochastic hill climbing adjusting centers
    N = centers.shape[0]
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = float(best_radii.sum())
    for _ in range(iterations):
        # Randomly pick a circle
        idx = np.random.randint(N)
        old_center = centers[idx].copy()
        old_radius = radii[idx]
        # Propose a random displacement
        delta = np.random.uniform(-step, step, 2)
        new_center = old_center + delta
        # Check if it's still valid to move
        if (new_center[0] - old_radius < 0 or new_center[0] + old_radius > 1 or
            new_center[1] - old_radius < 0 or new_center[1] + old_radius > 1):
            continue
        # Update it temporarily
        centers[idx] = new_center
        # Run inner loop to get new radii
        old_radii = radii.copy()
        radii = max_radii(centers, radii)
        new_sum = float(radii.sum())
        if new_sum > best_sum:
            best_sum = new_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
        else:
            centers[idx] = old_center
            radii = old_radii
    return best_centers, best_radii

def refine_with_scipy(centers, radii):
    # Use a scipy optimizer to further improve the solution
    n, = radii.shape
    def obj(x):
        return -np.sum(x[n*2:])

    def wall_ineq(x):
        c = x[:n*2].reshape((n, 2))
        r = x[n*2:]
        res = []
        res.extend(r - c[:,0])  # r - x
        res.extend(r + c[:,0] - 1)  # r + x - 1
        res.extend(r - c[:,1])  # r - y
        res.extend(r + c[:,1] - 1)  # r + y - 1
        return np.array(res)

    def overlap_ineq(x):
        c = x[:n*2].reshape((n, 2))
        r = x[n*2:]
        res = []
        D = np.sqrt(((c[:,None,:] - c[None,:,:])**2).sum(-1))
        for i in range(n):
            for j in range(i+1, n):
                res.append(r[i] + r[j] - D[i,j])
        return np.array(res)

    # Build sparse Jacobian for wall constraints
    n_vars = 2*n + n
    n_wall = 4*n
    n_overlap = n*(n-1)//2

    rows = []
    cols = []
    data = []
    # Wall terms
    for i in range(n):
        # r - x_i
        rows.extend([i, i+n, i+n*2, i+n*3])
        cols.extend([n*2+i, i, n*2+i, n*2+i])
        data.extend([-1, 1, 0, 0])  # J[r - x_i] w.r.t [x_i, y_i, r_i] = [-1, 0, 1]

        # r + x_i - 1
        rows.extend([i, i+n, i+n*2, i+n*3])
        cols.extend([n*2+i, i, n*2+i, n*2+i])
        data.extend([-1, 0, 1, 0])

        # r - y_i
        rows.extend([i+n, i, i+n*2, i+n*3])
        cols.extend([n*2+i, i+1, n*2+i, n*2+i])
        data.extend([-1, 0, 1, 0])

        # r + y_i - 1
        rows.extend([i+n, i, i+n*2, i+n*3])
        cols.extend([n*2+i, i+1, n*2+i, n*2+i])
        data.extend([-1, 0, 1, 0])

    # Overlap terms
    start = 4*n
    for i in range(n):
        for j in range(i+1, n):
            di = (x[i*2] - x[j*2]) / (D[i,j] + 1e-12)
            dj = (x[j*2] - x[i*2]) / (D[i,j] + 1e-12)
            d_i = (x[i*2+1] - x[j*2+1]) / (D[i,j] + 1e-12)
            d_j = (x[j*2+1] - x[i*2+1]) / (D[i,j] + 1e-12)
            rows.append(start)
            cols.append(i)
            data.append(-di)
            rows.append(start)
            cols.append(j)
            data.append(-dj)
            rows.append(start)
            cols.append(i+1)
            data.append(-d_i)
            rows.append(start)
            cols.append(j+1)
            data.append(-d_j)
            rows.append(start)
            cols.append(n*2+i)
            data.append(-1)
            rows.append(start)
            cols.append(n*2+j)
            data.append(-1)
            start += 1

    jac_wall = sp.coo_matrix((data, (rows, cols)), shape=(n_wall, n_vars))
    jac_overlap = sp.coo_matrix((data, (rows, cols)), shape=(n_overlap, n_vars))

    bounds = [(0,1) for _ in range(2*n)] + [(0, 0.5) for _ in range(n)]

    x0 = np.concatenate([centers.flatten(), radii])
    try:
        res = minimize(obj, x0, method="trust-constr", bounds=bounds,
                       constraints=[NonlinearConstraint(wall_ineq, -np.inf, 0, jac=jac_wall),
                                    NonlinearConstraint(overlap_ineq, -np.inf, 0, jac=jac_overlap)],
                       options={"maxiter": 200})
        if res.success:
            new_x = res.x
            centers = new_x[:2*n].reshape((n,2))
            radii = new_x[2*n:]
            return centers, radii
    except:
        pass
    return centers, radii

def run_packing():
    centers, radii = init_hex_grid()
    radii = max_radii(centers, radii)
    centers, radii = hill_climb(centers, radii)
    centers, radii = refine_with_scipy(centers, radii)
    sum_r = float(radii.sum())
    return centers, radii, sum_r
