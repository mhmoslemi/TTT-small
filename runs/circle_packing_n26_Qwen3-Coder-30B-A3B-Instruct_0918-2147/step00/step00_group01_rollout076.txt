def generate_lattice(s):
    """
    Generate a hexagonal lattice of candidate centre points inside unit square.
    Only points with distance >= s/2 from all walls are included.
    """
    sqrt3 = sqrt(3)
    rows = int(1.0 / (s * sqrt3 / 2)) + 2
    cols = int(1.0 / s) + 2
    points = []
    for r in range(rows):
        y = r * s * sqrt3 / 2
        if y < s / 2 or y > 1 - s / 2:
            continue
        offset = 0.5 if r % 2 == 1 else 0.0
        for c in range(cols):
            x = (c + offset) * s
            if x < s / 2 or x > 1 - s / 2:
                continue
            points.append([x, y])
    return np.array(points)

def build_milp(P):
    """
    Build a cvxpy MILP model for the circle packing problem.
    Returns a function that solves the LP relaxation for a given binary selection.
    """
    K = P.shape[0]
    # Wall distances
    wL = P[:, 0]
    wR = 1 - P[:, 0]
    wB = P[:, 1]
    wT = 1 - P[:, 1]
    w = np.minimum(np.minimum(wL, wR), np.minimum(wB, wT))

    # Pairwise distances
    D = np.zeros((K, K))
    for i in range(K):
        for j in range(i + 1, K):
            d = np.linalg.norm(P[i] - P[j])
            D[i, j] = D[j, i] = d

    # Variables
    z = cp.Variable(K, boolean=True)
    r = cp.Variable(K, nonneg=True)

    # Constraints
    constraints = [cp.sum(z) == 26]
    for i in range(K):
        constraints.append(r[i] <= w[i] * z[i])
    
    # Non-overlap constraints
    M = 1.0
    for i in range(K):
        for j in range(i + 1, K):
            # d_ij >= r_i + r_j - M*(2 - z_i - z_j)
            constraints.append(D[i, j] >= r[i] + r[j] - M * (2 - z[i] - z[j]))

    # Objective
    objective = cp.Maximize(cp.sum(r))

    def solve_milp_fixed_z(fixed_z):
        # Fix some z variables and solve
        prob = cp.Problem(objective, constraints)
        # To create a copy, use values
        for i in range(K):
            z[i].value = fixed_z[i]
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                return True, [z[i].value for i in range(K)], [r[i].value for i in range(K)]
            else:
                return False, [z[i].value for i in range(K)], [r[i].value for i in range(K)]
        except:
            return False, [z[i].value for i in range(K)], [r[i].value for i in range(K)]

    return solve_milp_fixed_z
