import numpy as np
from scipy.optimize import minimize
import scipy.sparse as sp

def init_hex_grid(n=26, r_init=0.04):
    # Generate hexagonal grid points
    h = np.sqrt(3) * r_init
    rows = []
    y = r_init
    row_idx = 0
    while len(rows) < n:
        if row_idx % 2 == 0:
            x_start = r_init
        else:
            x_start = 2 * r_init
        x = x_start
        while x <= 1 - r_init and len(rows) < n:
            rows.append([x, y])
            x += 2 * r_init
        y += h
        row_idx += 1
    centers = np.array(rows[:n])
    radii = np.full(n, r_init)
    return centers, radii

def max_radii(centers, radii, max_iter=100, tol=1e-8):
    # Compute maximal radii given current centers and existing radii
    n = centers.shape[0]
    # Wall constraints
    wall = np.minimum(centers, 1 - centers).min(axis=1)
    # Compute distance matrix
    D = np.sqrt(((centers[:, None, :] - centers[None, :, :]) ** 2).sum(-1))
    # Set diagonal to large value
    np.fill_diagonal(D, 1e10)
    radii_old = radii.copy()
    for _ in range(max_iter):
        # Compute neighbor limits
        neighbor = D - radii[None, :]
        neighbor = np.maximum(neighbor, 0)  # ensure no negative distances
        neighbor_min = np.min(neighbor, axis=1)
        radii_new = np.minimum(wall, neighbor_min)
        radii_new = np.maximum(radii_new, 0)
        if np.max(np.abs(radii_new - radii_old)) < tol:
            break
        radii_old = radii_new
    return radii_new

def hill_climb(centers, radii, iterations=5000, step=0.01):
    # Stochastic hill-climbing to improve sum of radii
    n = centers.shape[0]
    radii_best = radii.copy()
    centers_best = centers.copy()
    sum_r_best = np.sum(radii_best)
    for _ in range(iterations):
        # Select a random circle
        i = np.random.randint(n)
        # Try small random displacement
        delta = np.random.uniform(-step, step, 2)
        c_new = centers[i] + delta
        # Ensure it stays within bounds
        r = radii[i]
        if (c_new[0] - r < 0 or c_new[0] + r > 1 or
                c_new[1] - r < 0 or c_new[1] + r > 1):
            continue
        centers_new = centers.copy()
        centers_new[i] = c_new
        # Compute new radii
        radii_new = max_radii(centers_new, radii_best)
        sum_r_new = np.sum(radii_new)
        if sum_r_new > sum_r_best:
            centers_best = centers_new
            radii_best = radii_new
            sum_r_best = sum_r_new
    return centers_best, radii_best

    n = centers.shape[0]
    # Setup constraints for scipy
    def wall_cons(centers_radii):

        x = centers_radii[:n]
        y = centers_radii[n:2*n]
        result = np.zeros(4*n)

        result[:n] = r - x
        # r + x - 1 <= 0
        result[n:2*n] = r + x - 1
        # r - y <= 0

        # r + y - 1 <= 0
        result[3*n:] = r + y - 1
        return result

        y = centers_radii[n:2*n]
        r = centers_radii[2*n:]
