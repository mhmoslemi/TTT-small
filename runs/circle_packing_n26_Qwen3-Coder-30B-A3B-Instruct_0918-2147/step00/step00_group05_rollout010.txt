import numpy as np
from scipy.optimize import minimize

def hex_lattice_points(n, d):
    """Generate at least n points in a hexagonal lattice with spacing d."""
    rows = []
    y = 0.0
    while len(rows) < n:
        row = []
        x = 0.0
        while x <= 1.0 + d:
            row.append((x, y))
            x += d
        rows.append(row)
        y += d * np.sqrt(3) / 2
        if len(rows) % 2 == 1:
            x = d / 2
        else:
            x = 0.0
        while x <= 1.0 + d:
            row.append((x, y))
            x += d
        rows.append(row)
        y += d * np.sqrt(3) / 2
        if len(rows) >= n:
            break
    points = []
    for row in rows:
        for x, y in row:
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append((x, y))
            if len(points) >= n:
                break
        if len(points) >= n:
            break
    return np.array(points[:n])

    """Minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def wall_constraints(vars):
    """Return list of inequalities for wall constraints."""
    result = []
    for i in range(0, len(vars), 3):
        x = vars[i]
        y = vars[i+1]
        r = vars[i+2]
        # x - r >= 0   =>   r - x <= 0
        result.append(r - x)
        # x + r <= 1   =>   x + r - 1 <= 0
        result.append(x + r - 1)
        # y - r >= 0   =>   r - y <= 0
        result.append(r - y)
        # y + r <= 1   =>   y + r - 1 <= 0
        result.append(y + r - 1)
    return result

def nonoverlap_constraints(vars):
    """Return list of inequalities for non-overlap constraints."""
    result = []
    n = len(vars) // 3
    for i in range(n):
        for j in range(i + 1, n):
            xi = vars[3*i]
            yi = vars[3*i+1]
            ri = vars[3*i+2]
            xj = vars[3*j]
            yj = vars[3*j+1]

            dist_sq = (xi - xj)**2 + (yi - yj)**2
            radii_sum_sq = (ri + rj)**2
            result.append(radii_sum_sq - dist_sq)
