import numpy as np
from scipy.optimize import minimize

def initial_lattice(n=26):
    # Generate hexagonal lattice points
    spacing = 0.2
    sqrt3 = np.sqrt(3)
    rows = int(np.ceil(np.sqrt(n * 2 / sqrt3)))
    cols = int(np.ceil(n / rows))
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = j * spacing + (i % 2) * spacing / 2
            y = i * spacing * sqrt3 / 2
            if x <= 1 and y <= 1:
                centers.append([x, y])
            if len(centers) >= n:
                break
        if len(centers) >= n:
            break
    centers = np.array(centers[:n])
    
    # Compute max radii for each center
    radii = []
    for i in range(n):
        x, y = centers[i]
        min_dist = min(x, 1 - x, y, 1 - y)
        for j in range(n):
            if i != j:
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.hypot(dx, dy)
                if dist > 0:
                    min_dist = min(min_dist, dist / 2)
        radii.append(min(min_dist, 0.5))
    
    return centers, np.array(radii)

def pack(centers, radii):
    z = np.empty(3 * len(centers))
    z[0::3] = centers[:, 0]
    z[1::3] = centers[:, 1]
    z[2::3] = radii
    return z

def unpack(z):
    n = len(z) // 3
    centers = np.empty((n, 2))
    centers[:, 0] = z[0::3]
    centers[:, 1] = z[1::3]
    radii = z[2::3]
    return centers, radii

def objective(z):
    _, radii = unpack(z)
    return -np.sum(radii)

def constraint_boundary(z):
    centers, radii = unpack(z)
    n = len(centers)
    c = []
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        c.append(x - r) # left
        c.append(1 - x - r) # right
        c.append(y - r) # bottom
        c.append(1 - y - r) # top
    return np.array(c)

def constraint_nonoverlap(z):
    centers, radii = unpack(z)
    n = len(centers)
    c = []
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.hypot(dx, dy)
            r_sum = radii[i] + radii[j]
            # Non-overlap: dist >= r_i + r_j
            # So we add constraint: (r_i + r_j) - dist <= 0
            c.append(r_sum - dist)
    return np.array(c)

def repair(centers, radii):
    n = len(centers)
    # For boundary repair
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        min_dist = min(x, 1 - x, y, 1 - y)
        if r > min_dist:
            radii[i] = min_dist
    
    # For overlap repair
    max_iter = 10
    for _ in range(max_iter):
        any_overlap = False
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.hypot(dx, dy)
                r_sum = radii[i] + radii[j]
                if dist < r_sum:
                    overlap = r_sum - dist
                    # Shrink both radii
                    shrink = overlap / 2
                    radii[i] = max(0, radii[i] - shrink)
                    radii[j] = max(0, radii[j] - shrink)
                    any_overlap = True
        if not any_overlap:
            break
    return centers, radii

def run_packing():
    # Initial feasible layout
    centers, radii = initial_lattice(26)
    z0 = pack(centers, radii)
    
    # Define constraints
    constraints = []
    constraints.append({'type': 'ineq', 'fun': lambda z: constraint_boundary(z)})
    constraints.append({'type': 'ineq', 'fun': lambda z: constraint_nonoverlap(z)})
    
    # Bounding box for variables
    bounds = [(0, 1) for _ in range(52)] + [(0, 0.5) for _ in range(26)]
    
    # Optimize
    res = minimize(objective, z0, method='SLSQP', bounds=bounds, constraints=constraints,
                   options={'ftol': 1e-10, 'maxiter': 1000, 'disp': False})
    
    # Unpack result
    centers, radii = unpack(res.x)
    
    # Repair for boundary violations and overlaps
    centers, radii = repair(centers, radii)
    
    # Ensure non-negativity: a small epsilon for numerical tricks
    radii = np.maximum(radii, 1e-10)
    
    # Validation
    is_valid, msg = validate_packing(centers, radii)
    if not is_valid:
        raise RuntimeError(f"Final packing invalid: {msg}")
    
    sum_radii = np.sum(radii)
    if sum_radii < 2.635:
        raise RuntimeError("Sum of radii below target 2.636")
    
    return centers, radii, sum_radii
