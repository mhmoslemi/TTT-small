import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
from itertools import combinations

def generate_lattice(s):
    """Generate a hexagonal lattice inside the unit square."""
    sqrt3 = np.sqrt(3)
    rows = int(np.ceil(1.0 / (s * sqrt3 / 2))) + 1
    cols = int(np.ceil(1.0 / s)) + 1
    points = []
    for r in range(rows):
        y = r * s * sqrt3 / 2
        if r % 2 == 0:
            start_col = 0
        else:
            start_col = 0.5
        for c in range(cols):
            x = (start_col + c) * s
            if s/2 <= x <= 1 - s/2 and s/2 <= y <= 1 - s/2:
                points.append([x, y])
    return np.array(points)

def build_milp(P):
    """Builds a cvxpy problem that returns the LP relaxation for a given binary selection."""
    K = len(P)
    z = cp.Variable(K, boolean=True)
    r = cp.Variable(K, nonneg=True)
    
    # Compute wall distances for each point
    w = np.zeros(K)
    for k in range(K):
        x, y = P[k]
        w[k] = min(x, 1 - x, y, 1 - y)
    
    # Compute pairwise distances
    d = np.zeros((K, K))
    for k in range(K):
        for l in range(K):
            d[k, l] = np.linalg.norm(P[k] - P[l])
    
    # Objective
    obj = cp.Maximize(cp.sum(r))
    
    # Constraints
    constrs = [cp.sum(z) == 26]
    for k in range(K):
        constrs.append(r[k] <= w[k] * z[k])
        constrs.append(r[k] >= 0)
    
    # Add overlap constraints using big-M
    M = 1.0
    for k, l in combinations(range(K), 2):
        constrs.append(d[k, l] >= r[k] + r[l] - M * (2 - z[k] - z[l]))
    
    # Build problem
    prob = cp.Problem(obj, constrs)
    
    def solve_lp_fixed_z(fixed_z):
        for i in range(K):
            z[i].value = fixed_z[i]
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except:
            return np.full(K, 0.0), -float('inf')
        return r.value.copy(), prob.value
    
    return solve_lp_fixed_z

def branch_and_bound(milp_solver, node_limit=20000):
    """Performs a branch-and-bound search on the MILP model."""
    best_solution = None
    best_value = -float('inf')
    node_queue = [(np.full(120, 0.5), 0)]  # (z_values, depth)
    nodes_processed = 0

    while node_queue and nodes_processed < node_limit:
        nodes_processed += 1
        z_vals, depth = node_queue.pop(0)
        # Decide how to branch
        branch_var = -1
        max_frac = 0.0
        for i in range(len(z_vals)):
            frac = abs(z_vals[i] - 0.5)
            if frac > max_frac and z_vals[i] != 1.0 and z_vals[i] != 0.0:
                max_frac = frac
                branch_var = i
        if branch_var < 0:
            # This is an integer node
            z_int = (z_vals > 0.5).astype(int)
            r_vals, obj_value = milp_solver(z_int)
            if obj_value > best_value:
                best_value = obj_value
                best_solution = (z_int.copy(), r_vals.copy())
            continue
        
        # Branch
        # Value is higher than 0.5, so try fixing to 1 first
        z_high = z_vals.copy()
        z_high[branch_var] = 1.0
        # The 'relaxation' is valid
        r_vals, obj_bound = milp_solver(z_high)
        if obj_bound > best_value:
            node_queue.append((z_high, depth + 1))
        # Now try fixing to 0
        z_low = z_vals.copy()
        z_low[branch_var] = 0.0
        r_vals2, obj_bound2 = milp_solver(z_low)
        if obj_bound2 > best_value:
            node_queue.append((z_low, depth + 1))
        node_queue = sorted(node_queue, key=lambda x: milp_solver(x[0])[1], reverse=True)
        if len(node_queue) > 1000:
            node_queue = node_queue[:1000]
    if best_solution is None:
        return None, None
    return best_solution

def refine_solution(centers, radii):
    """Refine centers and radii using local optimization."""
    n = len(centers)
    # Define the objective function
    def objective(x):
        total = 0
        for i in range(n):
            r = x[i + n]
            total += r
        return -total
    
    # Define the constraint
    def constraint(x):
        c = []
        for i in range(n):
            r = x[i + n]
            c.append(r)
            c.append(1 - r - x[i])   # x + r <= 1
            c.append(1 - r - x[i + n]) # y + r <= 1
            c.append(x[i] - r)      # x - r >= 0
            c.append(x[i + n] - r)  # y - r >= 0
        for i, j in combinations(range(n), 2):
            ri = x[i + n]
            rj = x[j + n]
            dx = x[i] - x[j]
            dy = x[i + n] - x[j + n]
            c.append(-(dx**2 + dy**2) + (ri + rj)**2)
        return np.array(c)
    
    # Derive bounds
    bounds = []
    for i in range(n):
        bounds.append((0, 1))   # x
        bounds.append((0, 1))   # y
    for i in range(n):
        bounds.append((0, 1))   # r

    # Initial guess
    x0 = []
    for i in range(n):
        x0.append(centers[i][0])
        x0.append(centers[i][1])
        x0.append(max(0.01, radii[i]))
    
    # Run the optimization
    result = minimize(objective, x0, bounds=bounds, constraints={'type': 'ineq', 'fun': constraint},
                      method='SLSQP', options={'ftol': 1e-9, 'maxiter': 2000})

    refined_centers = []
    refined_radii = []
    for i in range(n):
        refined_centers.append((result.x[i], result.x[i + n]))
        refined_radii.append(result.x[i + 2 * n])

    refined_centers = np.array(refined_centers)
    refined_radii = np.array(refined_radii)
    sum_radii = np.sum(refined_radii)

    if not valid:
        # Try a slight repulsion to fix the issue
        epsilon = 1e-7
        # Uniformly shrink radii until valid
        while not valid and sum_radii > 1e-12:
            factor = 1 - epsilon
            refined_radii *= factor
            sum_radii = np.sum(refined_radii)
            valid, msg = validate_packing(refined_centers, refined_radii)
    return refined_centers, refined_radii, sum_radii

def run_packing():
    # Generate a good number of candidate points for placing circles
    P = generate_lattice(0.13)
    if len(P) < 26:
        P = generate_lattice(0.12)
    if len(P) < 26:
        P = generate_lattice(0.11)

    # Build the MILP solver
    milp_solver = build_milp(P)

    # Run the branch-and-bound on it
    z_final, r_final = branch_and_bound(milp_solver, node_limit=10000)
    if z_final is None:
        raise RuntimeError("B&B failed to find any integer solution")

    # Get the selected centers and their radii
    selected_indices = np.where(z_final == 1)[0]
    centers_lattice = P[selected_indices]
    radii_lattice = r_final[selected_indices]

    # Refine
    centers_refined, radii_refined, sum_radii = refine_solution(centers_lattice, radii_lattice)
    return centers_refined, radii_refined, sum_radii
