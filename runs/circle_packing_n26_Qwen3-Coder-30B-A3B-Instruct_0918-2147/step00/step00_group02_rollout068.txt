import numpy as np
from scipy.stats import qmc

def initialize_seeds(n, margin=0.02):
    """Generate low-discrepancy Sobol points in [margin, 1-margin]^2."""
    sampler = qmc.Sobol(d=2, scramble=False)
    points = sampler.random(n)
    # Scale to [margin, 1 - margin]^2
    points = margin + (1 - 2 * margin) * points
    return points

def compute_feasible_radii(centers):
    """Compute the maximal feasible radius for each center."""
    n = centers.shape[0]
    radii = np.zeros(n)
    # Compute pairwise distances
    diff = centers[:, None, :] - centers[None, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    # For each center, find the minimum distance to any other center (excluding itself)
    nearest = np.min(dist + np.eye(n) * 1e10, axis=1)
    # Compute boundary distances (min distance to each edge)
    bx = np.minimum(centers[:, 0], 1 - centers[:, 0])
    by = np.minimum(centers[:, 1], 1 - centers[:, 1])
    boundary = np.minimum(bx, by)
    # Feasible radius is minimum of boundary and 0.5 * nearest neighbor distance
    radii = np.minimum(boundary, 0.5 * nearest)
    return radii

def redistribute_radii(radii, slack, alpha=0.4):
    """Redistribute radius from wall-limited to neighbor-limited circles."""
    total_slack = np.sum(slack)
    if total_slack <= 1e-12:
        return radii.copy()
    # Distribute alpha * total_slack among wall-limited circles (slack > 0)
    new_radii = radii.copy()
    wall_limited = slack > 1e-12
    if not np.any(wall_limited):
        return new_radii
    # Give each wall-limited circle a proportional share of the total budget
    weights = slack[wall_limited]
    weights = weights / np.sum(weights)
    delta = alpha * total_slack
    increments = delta * weights
    new_radii[wall_limited] += increments
    # Now reduce from neighbor-limited (slack <= 0)
    neighbor_limited = ~wall_limited
    if np.any(neighbor_limited):
        # Reduce proportionally to current radii
        weights = radii[neighbor_limited]
        weights = weights / np.sum(weights)
        reductions = delta * weights
        new_radii[neighbor_limited] -= reductions
        # Ensure no negative radii
        new_radii = np.maximum(0.0, new_radii)
    return new_radii

def push_away(centers, radii, beta=0.3, eps=1e-8):
    """Move centers toward centroids of their Voronoi cells via neighbor repulsion."""
    n = centers.shape[0]
    new_centers = centers.copy()
    for i in range(n):
        ci = centers[i]
        ri = radii[i]
        for j in range(n):
            if i == j:
                continue
            cj = centers[j]
            rj = radii[j]
            diff = ci - cj
            dist = np.sqrt(np.sum(diff**2))
            if dist < ri + rj + eps:
                # Overlap, move ci away from cj
                delta = (ri + rj + eps - dist)
                push = beta * delta * (diff / dist)
                new_centers[i] += push
    # Clamp to [eps, 1 - eps]
    new_centers = np.clip(new_centers, eps, 1 - eps)
    return new_centers

def iterate(centers, radii, alpha=0.4, beta=0.3, eps=1e-8, max_iter=200):
    """Run the main algorithmic loop."""
    for _ in range(max_iter):
        old_sum = np.sum(radii)
        # Compute slack
        diff = centers[:, None, :] - centers[None, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2))
        nearest = np.min(dist + np.eye(centers.shape[0]) * 1e10, axis=1)
        boundary = np.minimum(centers[:, 0], 1 - centers[:, 0])
        boundary = np.minimum(boundary, centers[:, 1])
        boundary = np.minimum(boundary, 1 - centers[:, 1])
        feasible_radii = np.minimum(boundary, 0.5 * nearest)
        slack = 0.5 * nearest - radii
        # Redistribute
        new_radii = redistribute_radii(radii, slack, alpha)
        # Push away
        new_centers = push_away(centers, new_radii, beta, eps)
        # Re-evaluate radii
        radii = compute_feasible_radii(new_centers)
        new_sum = np.sum(radii)
        # Check convergence
        if abs(new_sum - old_sum) < 1e-5:
            break
        centers = new_centers
    return centers, radii

def repair(centers, radii, eps_final=5e-10):
    """Apply final corrections to satisfy validation constraints strictly."""
    # Shrink radii slightly
    radii = np.maximum(0.0, radii - eps_final)
    # Clamp centers
    centers = np.clip(centers, eps_final, 1 - eps_final)
    # Final overlap check & fix
    diff = centers[:, None, :] - centers[None, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    # List pairs that violate the rule and fix
    for i in range(centers.shape[0]):
        for j in range(i + 1, centers.shape[0]):
            if dist[i, j] < radii[i] + radii[j] - 1e-12:
                overlap = radii[i] + radii[j] - dist[i, j]
                reduction = overlap / 2 + eps_final
                if radii[i] > radii[j]:
                    radii[i] -= reduction
                else:
                    radii[j] -= reduction
                radii = np.maximum(0.0, radii)
    return centers, radii
