import numpy as np
from scipy.optimize import linprog, minimize
from math import sqrt, ceil
import math

def build_hex_lattice(d):
    """Builds a hexagonal lattice with at least d points."""
    # Hexagonal lattice with spacing sqrt(3)/2 ≈ 0.866
    # This gives an efficient coverage of the square
    a = sqrt(3) / 2
    rows = ceil(1.0 / a)
    cols = ceil(1.0 / (a * sqrt(3) / 2)) + 1
    centers = []

    for i in range(rows):
        for j in range(cols):
            x = j * a + (i % 2) * a / 2
            y = i * a * sqrt(3) / 2
            if x <= 1.0 and y <= 1.0:
                centers.append([x, y])
        if len(centers) >= d:
            break

    return np.array(centers[:d])

def lp_max_radii(centers):
    """Solve LP to find maximal radii for fixed centers."""
    n = centers.shape[0]
    if n == 0:
        return np.array([]), 0

    # Inequalities: A_ub * radii <= b_ub
    # Wall constraints
    A_ub = []
    b_ub = []

    # Each r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
    for i in range(n):
        x, y = centers[i]
        r_max = min(x, 1 - x, y, 1 - y)
        A_ub.append([-1.0])  # -r_i <= -r_max
        b_ub.append(-r_max)

    # Pairwise non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = sqrt(dx * dx + dy * dy)
            A_ub.append([1.0, 1.0])
            b_ub.append(dist)

    # Coefficient matrix A_ub has shape (constraints, n)
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)

    # Objective: maximize sum of radii (minimize negative sum)
    c = np.ones(n) * -1

    # Bound on radii (keep >= 0 due to wall constraints)
    bounds = [(0, None) for _ in range(n)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', tol=1e-9)
        if res.status == 0:
            radii = res.x
            return radii, -res.fun
        else:
            # Failure: fallback to small radius
            return np.zeros(n), 0.0
    except Exception:
        return np.zeros(n), 0.0

def perturb_centers(centers, sigma):
    """Perturb each center with Gaussian noise and clamp to unit square."""
    new_centers = centers + np.random.normal(0, sigma, centers.shape)
    new_centers[:, 0] = np.clip(new_centers[:, 0], 0, 1)
    new_centers[:, 1] = np.clip(new_centers[:, 1], 0, 1)
    return new_centers

def anneal_centers(initial_centers, max_iter=10000):
    """Use simulated annealing to find good center positions."""
    # Annealing schedule
    steps = [
        (0.02, 0.05, 2000),
        (0.01, 0.03, 4000),
        (0.005, 0.015, 3000),
        (0.002, 0.008, 2000)
    ]

    current_centers = initial_centers.copy()
    current_radii, current_sum = lp_max_radii(current_centers)
    best_centers = current_centers.copy()
    best_radii = current_radii.copy()
    best_sum = current_sum
    last_improvement = 0

    np.random.seed(42)  # for reproducibility

    iteration = 0
    for temp, sigma, num_iter in steps:
        for _ in range(num_iter):
            iteration += 1
            new_centers = perturb_centers(current_centers, sigma)
            new_radii, new_sum = lp_max_radii(new_centers)
            if new_sum > current_sum or np.random.rand() < np.exp((new_sum - current_sum) / temp):
                current_centers = new_centers
                current_radii = new_radii
                current_sum = new_sum
                if current_sum > best_sum:
                    best_centers = current_centers.copy()
                    best_radii = current_radii.copy()
                    best_sum = current_sum
                    last_improvement = iteration
            # Early stopping
            if iteration - last_improvement > 1000:
                break
        if iteration - last_improvement > 1000:
            break

    return best_centers, best_radii, best_sum

def local_refine(centers, radii):
    """Refine using local non-linear optimization (trust-constr)."""
    n = centers.shape[0]

    def objective(radii_and_centers):
        return -np.sum(radii_and_centers[n * 2:])

    def constraint_func(radii_and_centers):
        start_r = n * 2
        rad = radii_and_centers[start_r:]
        centers_arr = radii_and_centers[:start_r].reshape((n, 2))
        constraints = []
        # Correctness - radii must be at least current values
        for i in range(n):
            constraints.append(rad[i] - radii[i])
        # Wall bounds
        for i in range(n):
            x, y = centers_arr[i]
            constraints.append(x - rad[i])
            constraints.append(1 - x - rad[i])
            constraints.append(y - rad[i])
            constraints.append(1 - y - rad[i])
        # Non-overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_arr[i, 0] - centers_arr[j, 0]
                dy = centers_arr[i, 1] - centers_arr[j, 1]
                dist2 = dx * dx + dy * dy
                rad_sum = rad[i] + rad[j]
                # Constraint: dist^2 >= (r_i + r_j)^2
                # i.e. (r_i + r_j)^2 - dist^2 <= 0
                constraints.append(rad_sum * rad_sum - dist2)
        return np.array(constraints)

    # Initial guess
    initial_guess = np.concatenate([centers.flatten(), radii])

    # Box bounds:
    # First 2n values: center coordinates (x1,y1,...,xn,yn)
    # Next n values: radii (r1,...,rn)
    bounds = []
    for _ in range(n):
        bounds.append((0, 1))  # x coordinates
        bounds.append((0, 1))  # y coordinates
    for r in radii:
        bounds.append((r, None))  # radii

    try:
        cons = {'type': 'ineq', 'fun': constraint_func}
        res = minimize(objective, initial_guess, method='trust-constr', constraints=cons, bounds=bounds, options={'maxiter': 500})
        if res.success:
            refined_centers = res.x[:2 * n].reshape(n, 2)
            refined_radii = res.x[2 * n:]
            refined_sum = -res.fun
            return refined_centers, refined_radii, refined_sum
    except Exception:
        pass

    # If optimization fails, return original
    return centers, radii, np.sum(radii)
