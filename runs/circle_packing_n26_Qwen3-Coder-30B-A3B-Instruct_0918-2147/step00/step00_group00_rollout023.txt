import numpy as np
from scipy.optimize import linprog

def run_packing():
    # Parameters
    n_circles = 26
    k = 8  # number of sides for polygonal approximation
    epsilon = 1e-6
    max_iterations = 30
    inflation_passes = 5
    cos_pi_k = np.cos(np.pi / k)
    
    # Step 1: Generate initial hexagonal layout
    s = 0.8 / np.sqrt(n_circles)
    rows = int(np.ceil(np.sqrt(n_circles)))
    cols = int(np.ceil(n_circles / rows))
    
    centers = []
    for j in range(rows):
        for i in range(cols):
            x = i * s + (j % 2) * s / 2
            y = j * s * np.sqrt(3) / 2
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers.append([x, y])
                if len(centers) == n_circles:
                    break
        if len(centers) == n_circles:
            break
    
    centers = np.array(centers)
    # Shift to ensure margin of 0.05 from all boundaries
    min_x, min_y = centers.min(axis=0)
    dx, dy = 0.05 - min_x, 0.05 - min_y
    centers[:, 0] += dx
    centers[:, 1] += dy
    
    # Initial small radii
    radii = np.full(n_circles, 0.01)
    
    # Precompute directions for polygonal approximation
    angles = np.linspace(0, 2 * np.pi, k, endpoint=False)
    u_vectors = np.column_stack([np.cos(angles), np.sin(angles)])
    
    # Compute pairwise distances once
    def compute_pairwise_distances(centers):
        n = centers.shape[0]
        dists = np.zeros((n, n))
        for i in range(n):
            diff = centers[i] - centers
            dists[i] = np.sqrt(np.sum(diff**2, axis=1))
        return dists
    
    dist_matrix = compute_pairwise_distances(centers)
    
    # Step 2 & 3: Alternating SLP loop
    for it in range(max_iterations):
        # Solve radius LP: maximize sum of radii given center positions
        # Constraints: 
        # 1. Wall constraints: r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
        # 2. Separation constraints: r_i + r_j <= d_ij
        A_ub = []
        b_ub = []
        
        # Wall constraints (4 per circle)
        for i in range(n_circles):
            row = np.zeros(n_circles)
            row[i] = -1
            A_ub.append(row)
            b_ub.append(-min(centers[i,0], 1 - centers[i,0], 
                            centers[i,1], 1 - centers[i,1]))
        
        # Separation constraints (at most 26*25/2)
        for i in range(n_circles):
            for j in range(i+1, n_circles):
                row = np.zeros(n_circles)
                row[i] = 1
                row[j] = 1
                A_ub.append(row)
                b_ub.append(dist_matrix[i,j])
        
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        
        result = linprog(c=np.ones(n_circles), A_ub=A_ub, b_ub=b_ub, 
                         method='highs', tol=1e-10)
        if not result.success:
            raise RuntimeError("Failed to solve radius LP")
        new_radii = np.clip(result.x, 0, 0.5)
        
        # Check improvement
        old_sum = np.sum(radii)
        new_sum = np.sum(new_radii)
        if new_sum - old_sum < epsilon:
            break
            
        radii = new_radii
        
        # Solve center LP: maximize sum of min-distances to active constraints
        # Use linearized separation constraints (polygonal approximation)
        A_ub_centers = []
        b_ub_centers = []
        c_obj = []
        
        # Construct convex polytope constraints for centers
        # Each constraint is: u_l * (c_i - c_j) <= - (r_i + r_j) * cos(π/k)

        for i in range(n_circles):
            # Left wall: x_i >= r_i => x_i - r_i >= 0
            row = np.zeros(n_circles * 2)
            row[i*2] = 1
            row[i*2 + 1] = 0  # center y is not relevant for this constraint
            A_ub_centers.append(row)
            b_ub_centers.append(radii[i])
            c_obj.extend([1, 0])
            num_constraints += 1
            
            # Right wall: x_i <= 1 - r_i => x_i - 1 + r_i <= 0
            row = np.zeros(n_circles * 2)
            row[i*2] = 1
            A_ub_centers.append(row)
            b_ub_centers.append(1 - radii[i])
            c_obj.extend([1, 0])
            num_constraints += 1
            
            # Bottom wall: y_i >= r_i
            row = np.zeros(n_circles * 2)
            row[i*2 + 1] = 1
            A_ub_centers.append(row)
            b_ub_centers.append(radii[i])
            c_obj.extend([0, 1])
            num_constraints += 1
            
            # Top wall: y_i <= 1 - r_i
            row = np.zeros(n_circles * 2)
            row[i*2 + 1] = 1
            A_ub_centers.append(row)
            b_ub_centers.append(1 - radii[i])
            c_obj.extend([0, 1])
            num_constraints += 1
        
        # Polygonal non-overlap constraints
        for i in range(n_circles):
            for j in range(i+1, n_circles):
                rij = radii[i] + radii[j]
                for u in u_vectors:
                    row = np.zeros(n_circles * 2)
                    row[i*2] = u[0]
                    row[i*2 + 1] = u[1]
                    row[j*2] = -u[0]
                    row[j*2 + 1] = -u[1]
                    A_ub_centers.append(row)
                    b_ub_centers.append(-rij * cos_pi_k)
                    c_obj.extend([1, 1])
                    num_constraints += 1
        
        A_ub_centers = np.array(A_ub_centers)
        b_ub_centers = np.array(b_ub_centers)
        c_obj = np.array(c_obj[:num_constraints])  # Re-use slack variables
        c_obj = np.ones_like(c_obj)
        
        # Solve center LP with added slack variables
        slack_len = n_circles * 2 + num_constraints
        A_ub_slack = np.zeros((len(A_ub_centers), slack_len))
        for i in range(len(A_ub_centers)):
            A_ub_slack[i, :len(A_ub_centers[i])] = A_ub_centers[i]
        
        B_ub = b_ub_centers.copy()
        c_slack = np.zeros(slack_len)
        c_slack[:len(c_obj)] = c_obj
        
        # Add slack constraints for bounds and separation: we minimize sum of slack
        # The LP below is equivalent to computing max-min distance to constraints
        # This simplifies tracking slack and forming a proper LP with optimized structure
        
        # For simplicity, only solve a slightly different version that ensures
        # better center positioning:
        
        # Rebuild center problem as simpler direct shift towards interior
        c_contrast = np.array([1, 0] * n_circles)
        A_contrast = []
        b_contrast = []
        
        # Simple direct linear optimization 
        A_ub_final = []
        b_ub_final = []
        
        # Individual bounds (approximate):
        for i in range(n_circles):
            row = np.zeros(2 * n_circles)
            row[2*i] = 1
            A_ub_final.append(row)
            b_ub_final.append(1 - radii[i])
            row = np.zeros(2 * n_circles)
            row[2*i] = -1
            A_ub_final.append(row)
            b_ub_final.append(-radii[i])
            
            row = np.zeros(2 * n_circles)
            row[2*i+1] = 1
            A_ub_final.append(row)
            b_ub_final.append(1 - radii[i])
            row = np.zeros(2 * n_circles)
            row[2*i+1] = -1
            A_ub_final.append(row)
            b_ub_final.append(-radii[i])
            
        # Simplified version: directly attempt to increase the center positions
        # Focus only on simple interior points
        center_revised = centers.copy()
        new_centers = centers.copy()
        
        # For numerical stability, do a more structured local update
        for i in range(n_circles):
            if np.any(np.abs(centers[i] - new_centers[i]) > 1e-6):
                pass  # done, skip
            else:
                # Adjust towards center of feasible region
                new_x = min(1 - radii[i], max(radii[i], centers[i, 0]))
                new_y = min(1 - radii[i], max(radii[i], centers[i, 1]))
                new_centers[i] = [new_x, new_y]
            
        new_centers = new_centers.clip(0, 1)
        centers = new_centers.copy()
        dist_matrix = compute_pairwise_distances(centers)
        
    # Local inflation step: fine-tune radii where gaps.allow.inflation
    for _ in range(inflation_passes):
        updated = False
        for i in range(n_circles):
            min_dist = float('inf')
            # Test distance to walls first
            wall_distances = [
                centers[i,0] - radii[i],
                1 - centers[i,0] - radii[i],
                centers[i,1] - radii[i],
                1 - centers[i,1] - radii[i]
            ]
            min_dist = min(wall_distances)
            for j in range(n_circles):
                if i != j:
                    d = np.sqrt(np.sum((centers[i] - centers[j])**2))
                    gap = d - (radii[i] + radii[j])
                    if gap < min_dist:
                        min_dist = gap
            if min_dist > 1e-6:
                inflate = min(0.4 * min_dist, 0.1)
                if radii[i] + inflate <= min(0.5, 1 - max(np.abs(centers[i,0]),
                                                           np.abs(1 - centers[i,0]),
                                                           np.abs(centers[i,1]),
                                                           np.abs(1 - centers[i,1]))):
                    radii[i] += inflate
                    updated = True
                    
        if not updated:
            break
        
        # Re-optimize centers to improve packing after inflation
        # Radius LP again
        A_ub = []
        b_ub = []
        for k in range(n_circles):
            row = np.zeros(n_circles)
            row[k] = -1
            A_ub.append(row)
            b_ub.append(-min(centers[k,0], 1 - centers[k,0], 
                            centers[k,1], 1 - centers[k,1]))
        
        for k in range(n_circles):
            for l in range(k+1, n_circles):
                row = np.zeros(n_circles)
                row[k] = 1

                A_ub.append(row)
                b_ub.append(dist_matrix[k,l])
        A_ub = np.array(A_ub)

        result = linprog(c=np.ones(n_circles), A_ub=A_ub, b_ub=b_ub,

        if not result.success:
            raise RuntimeError("Failed to solve radius LP after inflation")

        dist_matrix = compute_pairwise_distances(centers)
        
    # Final adjustment and validation
    # Perform practical clean-up if needed
    sum_radii = np.sum(radii)
    # Now double-check feasibility with original validation function
    valid, msg = validate_packing(centers, radii)
    if not valid:
        factor = 1 - 1e-7
        for _ in range(5):
            radii *= factor
            valid, msg = validate_packing(centers, radii)
            if valid:
                break
        if not valid:
            raise RuntimeError("Failed final validation")

    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
