import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
from itertools import combinations

def init_candidate_grid(R, margin=1e-6):
    candidates = []
    for r in R:
        # Grid spacing ensures no overlap for equal radius circles
        spacing = r * np.sqrt(2) 
        # Generate grid within (r, 1-r) to keep margin from walls
        x_coords = np.arange(r + margin, 1 - r, spacing)
        y_coords = np.arange(r + margin, 1 - r, spacing)
        for x in x_coords:
            for y in y_coords:
                candidates.append((x, y, r))
    return candidates

def build_overlap_matrix(candidates):
    n = len(candidates)
    overlaps = np.zeros((n, n), dtype=bool)
    coords = np.array([(x, y) for x, y, r in candidates])
    radii = np.array([r for x, y, r in candidates])
    # Precompute distances
    distances = cdist(coords, coords)
    # Detect overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = distances[i, j]
            if dist < radii[i] + radii[j] - 1e-12:
                overlaps[i, j] = True
                overlaps[j, i] = True
    return overlaps

def solve_discrete_bip(candidates, overlap_matrix, target_sum):
    # Greedy initial packing closest to target sum
    def greedy_pack():
        used = np.zeros(len(candidates), dtype=bool)
        selected = []
        total_sum = 0.0
        radii = np.array([r for x, y, r in candidates])
        coords = np.array([(x, y) for x, y, r in candidates])
        sorted_indices = np.argsort(-radii)  # descending
    
        for i in sorted_indices:
            if used[i]:
                continue
            if total_sum + radii[i] > target_sum:
                continue
                
            # check if circle is not overlapping
            valid = True
            for j in range(len(selected)):
                dist = np.sqrt(np.sum((coords[i] - coords[selected[j]])**2))
                if dist < radii[i] + radii[selected[j]] - 1e-12:
                    valid = False
                    break
                    
            if valid:
                total_sum += radii[i]
                selected.append(i)
                used[i] = True
                
        # Fill up to 26
        for i in sorted_indices:
            if len(selected) == 26:
                break
            if used[i]:
                continue
            selected.append(i)
            used[i] = True
            total_sum += radii[i]
        return selected, total_sum

    selected_indices, total_sum = greedy_pack()
    
    # Use exact BIP solver alternatively if needed
    if total_sum >= target_sum:
        return [candidates[i] for i in selected_indices]
    
    # Implement a simpler version for the final answer which just attempts greedy
    # If we aren't allowed to go better, we must retry or diverge, but for now we return a suboptimal but valid pick
    # Reiterate over greedy logic for exactly 26 circles
    used = np.zeros(len(candidates), dtype=bool)
    selected = []
    total_sum = 0.0
    radii = np.array([r for x, y, r in candidates])
    coords = np.array([(x, y) for x, y, r in candidates])
    
    # First pick all large ones that are staying within constraints
    sorted_indices = np.argsort(-radii)  # Descending order
    for i in sorted_indices:
        if len(selected) == 26:
            break
        if used[i]:
            continue

        valid = True
        for j in selected:
            dist = np.sqrt(np.sum((coords[i] - coords[j])**2))
            if dist < radii[i] + radii[j] - 1e-12:
                valid = False
                break

        if valid:
            selected.append(i)
            used[i] = True
            total_sum += radii[i]
            
    # If not filled yet, add safe ones
    unused = [i for i in range(len(candidates)) if not used[i]]
    extra = 26 - len(selected)
    for i in unused[:extra]:
        selected.append(i)
        total_sum += radii[i]
        
    return [candidates[i] for i in selected]

def max_radii_fixed_centres(centers, radii_initial):
    radii_new = radii_initial.copy()
    coords = np.array(centers)
    n = len(coords)
    radii = np.array(radii_initial)
    
    # For each circle, at most one of r_i can be increased,
    # defined by
    # - wall constraints: r_i ≤ x_i, r_i ≤ (1 - x_i), r_i ≤ y_i, r_i ≤ (1 - y_i)
    # - overlap constraints: for all j ≠ i, sqrt((x_i - x_j)^2 + (y_i - y_j)^2) >= r_i + r_j
    
    for i in range(n):
        # Get minimal distance to walls
        min_wall = min(coords[i][0], 1 - coords[i][0], coords[i][1], 1 - coords[i][1])
        
        # Get minimal gap to other circles
        min_gap = float('inf')
        for j in range(n):
            if i == j:
                continue
            dist = np.sqrt(np.sum((coords[i] - coords[j])**2))
            min_gap = min(min_gap, dist - radii[j])
        
        # candidate max radius is limited by the smaller of wall constraints and gap constraints
        radii_new[i] = min(min_wall, min_gap)
        
        # Ensure it's not negative
        radii_new[i] = max(0, radii_new[i])
    
    return radii_new

def nlp_refine(centers_init, radii_init):
    n = len(centers_init)
    x0 = np.concatenate((np.array(centers_init).flatten(), np.array(radii_init)))

    def obj(v):
        r = v[-n:]
        return -np.sum(r)

    def jac(v):
        r = v[-n:]
        J = np.zeros_like(v)
        J[-n:] = -1.0
        return J

    def cons_f(v):
        x = v[:2*n].reshape((n, 2))
        r = v[2*n:]
        result = []
        
        for i in range(n):
            # Wall constraints
            result.append(r[i] - x[i][0])              # r_i - x_i <= 0
            result.append(r[i] + x[i][0] - 1)         # r_i + x_i - 1 <= 0
            result.append(r[i] - x[i][1])              # r_i - y_i <= 0
            result.append(r[i] + x[i][1] - 1)         # r_i + y_i - 1 <= 0

        for i in range(n):
            for j in range(i+1, n):
                dist = np.sqrt(np.sum((x[i] - x[j])**2))
                result.append(r[i] + r[j] - dist)
        return np.array(result)

    def cons_jac(v):
        x = v[:2*n].reshape((n, 2))
        r = v[2*n:]
        J = np.zeros((4*n + n*(n-1)//2, 2*n + n))

        # Wall Jacobians
        start_idx = 0
        for i in range(n):
            J[start_idx, 2*i + 0] = -1.0
            J[start_idx, 2*n + i] = 1.0
            start_idx += 1
            
            J[start_idx, 2*i + 0] = 1.0
            J[start_idx, 2*n + i] = 1.0
            start_idx += 1
            
            J[start_idx, 2*i + 1] = -1.0
            J[start_idx, 2*n + i] = 1.0
            start_idx += 1
            
            J[start_idx, 2*i + 1] = 1.0
            J[start_idx, 2*n + i] = 1.0
            start_idx += 1

        # Non-overlap Jacobians
        for i in range(n):
            for j in range(i+1, n):
                dx = x[i][0] - x[j][0]
                dy = x[i][1] - x[j][1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist > 0:
                    ddrx = dx / dist
                    ddry = dy / dist
                    ddrjx = -dx / dist
                    ddrjy = -dy / dist
                    J[start_idx, 2*i + 0] = ddrx
                    J[start_idx, 2*i + 1] = ddry
                    J[start_idx, 2*j + 0] = ddrjx
                    J[start_idx, 2*j + 1] = ddrjy
                    J[start_idx, 2*n + i] = 1.0
                    J[start_idx, 2*n + j] = 1.0
                else:
                    # Fallback for numerical stability; shouldn't be happening
                    J[start_idx, 2*n + i] = 1.0
                    J[start_idx, 2*n + j] = 1.0
                start_idx += 1
        return J

    bounds = []
    for i in range(2*n):
        bounds.append((0, 1))
    for i in range(n):
        bounds.append((0, 1))

    # Solve the optimization problem
    res = minimize(
        obj,
        x0,
        method='trust-constr',
        jac=jac,
        hess=None,
        bounds=bounds,
        constraints={'type': 'ineq', 'fun': cons_f, 'jac': cons_jac},
        options={
            'maxiter': 300,
            'xtol': 1e-9,
            'gtol': 1e-9,
        }
    )

    if not res.success:
        raise RuntimeError("Minimization failed")

    x = res.x[:2*n].reshape((n, 2))
    r = res.x[2*n:]
    return x, r

def run_packing():
    # Define admissible discrete radii set
    R = np.arange(0.025, 0.12, 0.01)  # R = {0.025, ..., 0.115}

    np.random.seed(42)  # Reproducibility

    candidates = init_candidate_grid(R)

    # Build overlap matrix for candidates
    overlap_matrix = build_overlap_matrix(candidates)
    
    # Get a valid initial candidate selection approximately maximizing sum
    selected_candidates = solve_discrete_bip(candidates, overlap_matrix, 2.6)

    # Extract centers and radii from the selected candidates
    centers = np.array([(x, y) for x, y, r in selected_candidates])
    radii = np.array([r for x, y, r in selected_candidates])
    
    # First tight projection of radii assuming fixed centers
    radii = max_radii_fixed_centres(centers, radii)
    
    # NLP refinement with trust-constr to maximize radius sum
    centers_refined, radii_refined = nlp_refine(centers, radii)

    # Again compute exact max radii under fixed centers (to lock in to maximal radii)
    radii_refined = max_radii_fixed_centres(centers_refined, radii_refined)

    # Validate
    valid, msg = validate_packing(centers_refined, radii_refined)
    if not valid:
        raise ValueError(f"Validation failed: {msg}")
    
    sum_radii = np.sum(radii_refined)
    return centers_refined, radii_refined, sum_radii
