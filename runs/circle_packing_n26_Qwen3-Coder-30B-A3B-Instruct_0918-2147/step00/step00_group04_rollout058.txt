import numpy as np
import cvxpy as cp
from scipy.spatial.distance import cdist

def build_leaf_problem(cell, k):
    """Build a SOCP for k circles in a cell."""
    x_min, y_min, x_max, y_max = cell
    x = cp.Variable((k, 2))
    r = cp.Variable(k)
    
    constraints = []
    # Boundary constraints
    for i in range(k):
        constraints.append(x[i, 0] - r[i] >= x_min)
        constraints.append(x[i, 0] + r[i] <= x_max)
        constraints.append(x[i, 1] - r[i] >= y_min)
        constraints.append(x[i, 1] + r[i] <= y_max)
    
    # Non-overlap constraints
    for i in range(k):
        for j in range(i+1, k):
            dist = cp.norm(x[i] - x[j])
            constraints.append(dist >= r[i] + r[j])
    
    objective = cp.Maximize(cp.sum(r))
    prob = cp.Problem(objective, constraints)
    return prob, x, r

def solve_leaf(cell, k):
    """Solve the leaf SOCP and return (centers, radii)."""
    prob, x, r = build_leaf_problem(cell, k)
    prob.solve(solver=cp.ECOS, verbose=False)
    return x.value, r.value

def merge_clusters(cluster_a, cluster_b, shared_edge):
    """Merge two clusters with shared interface."""
    centers_a, radii_a = cluster_a
    centers_b, radii_b = cluster_b
    all_centers = np.vstack([centers_a, centers_b])
    all_radii = np.hstack([radii_a, radii_b])
    k_a, k_b = len(radii_a), len(radii_b)
    
    x = cp.Variable((k_a + k_b, 2))
    r = cp.Variable(k_a + k_b)
    
    constraints = []
    # Boundary constraints for all
    for i in range(k_a + k_b):
        x_i = x[i]
        r_i = r[i]
        # Approximate boundaries using cell info, if available;
        # or let the whole problem solve around a general container
        
        # For extreme approximations, assume unit square
        constraints.append(x_i[0] - r_i >= 0)
        constraints.append(x_i[0] + r_i <= 1)
        constraints.append(x_i[1] - r_i >= 0)
        constraints.append(x_i[1] + r_i <= 1)
    
    # Non-overlap constraints
    for i in range(k_a + k_b):
        for j in range(i+1, k_a + k_b):
            dist = cp.norm(x[i] - x[j])
            constraints.append(dist >= r[i] + r[j])
    
    # Objective
    objective = cp.Maximize(cp.sum(r))
    prob = cp.Problem(objective, constraints)
    
    # Solve component-wise
    prob.solve(solver=cp.ECOS, verbose=False)
    
    return x.value, r.value

def hierarchical_merge(clusters):
    """Merges clusters using binary tree merge."""
    # Simple binary tree: pair up, merge, repeat
    while len(clusters) > 1:
        new_clusters = []
        for i in range(0, len(clusters), 2):
            if i + 1 < len(clusters):
                merged_centers, merged_radii = merge_clusters(
                    clusters[i], clusters[i+1], None)
                new_clusters.append((merged_centers, merged_radii))
            else:
                new_clusters.append(clusters[i])
        clusters = new_clusters
    
    return clusters[0]

def global_polish(centers, radii, max_iter=3):
    """Apply gradient-based polishing using log-barrier."""
    # Helper for safe distance calculation
    def safe_dist(a, b):
        return np.sqrt(np.sum((a - b) ** 2) + 1e-15)
        
    # Prepare fixed parameters
    n = len(radii)
    c = centers.copy()
    r = radii.copy()
    
    # Dynamics parameters
    mu = 1e-4
    step_size = 1e-3
    
    for _ in range(max_iter):
        grads = np.zeros_like(c)
        obj = 0.0
        
        # Gradient of sum of radii
        obj -= np.sum(r)
        grads += 1.0  # no clear derivative here: so use control to keep at boundary
        
        # Gradients from constraints
        for i in range(n):
            for j in range(i+1, n):
                dist = safe_dist(c[i], c[j])
                if dist - r[i] - r[j] < 1e-12:
                    direc = (c[i] - c[j]) / (dist + 1e-12)
                    grad_term = (1.0 / (mu * (dist - r[i] - r[j] + 1e-12))) * direc
                    grads[i] += grad_term
                    grads[j] -= grad_term
                    obj -= np.log(dist - r[i] - r[j] + 1e-12) / mu
            
            # Wall constraints
            for i in range(n):
                # From walls
                dx_right = 1 - (c[i, 0] + r[i])
                dx_left = c[i, 0] - r[i]
                dy_top = 1 - (c[i, 1] + r[i])
                dy_bot = c[i, 1] - r[i]
                
                if dx_right < 1e-12:
                    obj -= np.log(dx_right) / mu
                    grads[i, 0] += 1.0 / (mu * dx_right)
                if dx_left < 1e-12:
                    obj -= np.log(dx_left) / mu
                    grads[i, 0] -= 1.0 / (mu * dx_left)
                if dy_top < 1e-12:
                    obj -= np.log(dy_top) / mu
                    grads[i, 1] += 1.0 / (mu * dy_top)
                if dy_bot < 1e-12:
                    obj -= np.log(dy_bot) / mu
                    grads[i, 1] -= 1.0 / (mu * dy_bot)
        
        # Update step
        c += step_size * grads
        r = np.maximum(0.0, r + step_size)
        
        # Project centers back to feasible domain
        c = np.clip(c, 1e-12, 1-1e-12)
        # Reproject radii to ensure they don't exceed allowed limits
        for i in range(n):
            r[i] = min(r[i], c[i, 0] - 1e-12, 1 - c[i, 0] - 1e-12,
                       c[i, 1] - 1e-12, 1 - c[i, 1] - 1e-12)
    
    return c, r

def final_refine(centers, radii):
    """High-precision final refinement to fix any constraint violations."""
    centers = centers.copy()
    radii = radii.copy()
    
    # Use exact SOCP on the full set to recover any slacks more precisely
    prob, point_vars, r_vars = build_full_packing(centers, radii)
    prob.solve(solver=cp.ECOS, verbose=False)
    
    new_centers = point_vars.value
    new_radii = r_vars.value
    
    sum_radii = np.sum(new_radii)
    return new_centers, new_radii, sum_radii

def build_full_packing(initial_centers, initial_radii):
    n = len(initial_radii)
    x = cp.Variable((n, 2))
    r = cp.Variable(n)
    
    constraints = []
    for i in range(n):
        constraints.append(x[i, 0] - r[i] >= 0)
        constraints.append(x[i, 0] + r[i] <= 1)
        constraints.append(x[i, 1] - r[i] >= 0)
        constraints.append(x[i, 1] + r[i] <= 1)

    for i in range(n):
        for j in range(i+1, n):
            dist = cp.norm(x[i] - x[j])
            constraints.append(dist >= r[i] + r[j])

    objective = cp.Maximize(cp.sum(r))
    prob = cp.Problem(objective, constraints)
    return prob, x, r

def run_packing():
    """Main function to run circle packing."""
    # Decompose square into 9 rectangles: 4 corners, 4 edges, 1 center
    # Half edge length (to define cell sizes)
    a = 0.25
    
    # Cells with their [x_min, y_min, x_max, y_max]
    cells = [
        [0, 0, a, a],          # bottom-left
        [1-a, 0, 1, a],        # bottom-right
        [0, 1-a, a, 1],        # top-left
        [1-a, 1-a, 1, 1],      # top-right
        [a, 0, 1-a, a],        # bottom edge
        [a, 1-a, 1-a, 1],      # top edge
        [0, a, a, 1-a],        # left edge
        [1-a, a, 1, 1-a],      # right edge
        [a, a, 1-a, 1-a]       # center square
    ]
    
    # Initial allocation of circles per cell
    alloc = [5, 5, 5, 5, 1, 1, 1, 1, 2]  # Total: 26
    
    # Solve each cell separately with small numbers of circles
    clusters = []
    for i, (cell, num_circles) in enumerate(zip(cells, alloc)):
        if num_circles > 0:
            c, r = solve_leaf(cell, num_circles)
            clusters.append((c, r))

    # Merge all clusters hierarchically
    final_centers, final_radii = hierarchical_merge(clusters)

    # Global polishing
    final_centers, final_radii = global_polish(final_centers, final_radii)
    
    # Final refinement using SOCP
    final_centers, final_radii, sum_radii = final_refine(final_centers, final_radii)
    
    return final_centers, final_radii, sum_radii
