import numpy as np
from scipy.optimize import minimize

def hex_lattice_points(n, d):
    """Generate at least n points in a hexagonal lattice with spacing d."""
    sqrt3 = np.sqrt(3)
    points = []
    row = 0
    while len(points) < n:
        y = row * sqrt3 * d / 2
        if y > 1:
            break
        offset = 0 if row % 2 == 0 else d / 2
        x = offset
        while x <= 1 and len(points) < n:
            points.append([x, y])
            x += d
        row += 1
    return np.array(points[:n])
