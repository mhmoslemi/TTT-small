import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(0)
    
    # 1. Initial layout: perturbed hexagonal grid
    def init_grid(n_circles=26, spacing=0.07):
        rows = int(np.ceil(np.sqrt(n_circles) * 2))
        cols = int(np.ceil(n_circles / rows))
        points = []
        for k in range(rows):
            y = spacing / 2 + k * spacing * np.sqrt(3) / 2
            if k % 2 == 0:
                x_offset = 0
            else:
                x_offset = spacing / 2
            for m in range(cols):
                x = spacing / 2 + x_offset + m * spacing
                if 0 <= x <= 1 and 0 <= y <= 1:
                    points.append([x, y])
                if len(points) >= n_circles:
                    break
            if len(points) >= n_circles:
                break
        points = points[:n_circles]
        # Add small random jitter
        points = np.array(points) + np.random.uniform(-0.001, 0.001, (n_circles, 2))
        # Clip to [0,1]
        points[:, 0] = np.clip(points[:, 0], 0, 1)
        points[:, 1] = np.clip(points[:, 1], 0, 1)
        return points
    
    centers = init_grid()
    radii = np.full(26, 0.001)
    
    # 2. Physics simulation loop
    def compute_forces(centers, radii):
        F = np.zeros_like(centers)
        k_w = 10.0  # wall stiffness
        k_p = 5.0   # pairwise stiffness
        for i in range(len(centers)):
            x, y = centers[i]
            r = radii[i]
            # Wall forces
            dL = x - r
            dR = (1 - x) - r
            dB = y - r
            dT = (1 - y) - r
            if dL < 0:
                F[i, 0] += k_w * (-dL)
            if dR < 0:
                F[i, 0] -= k_w * (-dR)
            if dB < 0:
                F[i, 1] += k_w * (-dB)
            if dT < 0:
                F[i, 1] -= k_w * (-dT)
            # Pairwise forces
            for j in range(i):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                gap = dist - (radii[i] + radii[j])
                if gap < 0:
                    scale = k_p * (-gap) / (dist + 1e-12)
                    F[i, 0] += scale * dx
                    F[i, 1] += scale * dy
                    F[j, 0] -= scale * dx
                    F[j, 1] -= scale * dy
        return F
    
    dt = 0.02
    gamma = 1.0
    v = np.zeros_like(centers)
    
    # Simulate
    for step in range(5000):
        F = compute_forces(centers, radii)
        v = v + dt * (F - gamma * v)
        centers = centers + dt * v
        # Clip centers
        centers[:, 0] = np.clip(centers[:, 0], 0, 1)
        centers[:, 1] = np.clip(centers[:, 1], 0, 1)
        # Compute growth rates
        dot_r = np.ones_like(radii) * 0.001
        # Check if overlaps or wall constraints
        for i in range(len(centers)):
            x, y = centers[i]
            r = radii[i]
            dL = x - r
            dR = (1 - x) - r
            dB = y - r
            dT = (1 - y) - r
            if dL < 0 or dR < 0 or dB < 0 or dT < 0:
                dot_r[i] = 0
                continue
            # Check overlaps
            for j in range(i):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                gap = dist - (radii[i] + radii[j])
                if gap < 0:
                    dot_r[i] = 0
                    break
        # Update radii
        radii = radii + dt * dot_r
        # Stop if nothing is growing
        if not np.any(dot_r > 1e-10):
            break

    # 3. Fixed-centre maximal radii calculation
    def max_radii_fixed_centers(centers, radii):
        new_radii = radii.copy()
        for _ in range(30):
            radii_old = new_radii.copy()
            for i in range(len(centers)):
                x, y = centers[i]
                r_max = min(x, 1 - x, y, 1 - y)
                for j in range(len(centers)):
                    if i == j:
                        continue
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    r_max = min(r_max, dist - radii[j])
                new_radii[i] = max(0, r_max)
            if np.allclose(new_radii, radii_old, rtol=1e-9):
                break
        return new_radii
    
    radii = max_radii_fixed_centers(centers, radii)
    
    # 4. Global L-BFGS-B refinement
    def objective_sum_radii(centers_flat):
        centers = centers_flat.reshape(-1, 2)
        radii_new = max_radii_fixed_centers(centers, np.zeros_like(centers[:, 0]))
        return -np.sum(radii_new)
    
    def constraint_box(x):
        # Soft box constraint
        c = np.zeros(52)
        for i in range(26):
            val = x[2*i]
            c[2*i] = max(0, -val)**2 + max(0, val-1)**2
            val = x[2*i+1]
            c[2*i+1] = max(0, -val)**2 + max(0, val-1)**2
        return np.sum(c)
    
    bounds = [(0, 1) for _ in range(52)]
    x0_flat = centers.flatten()
    
    res = minimize(
        objective_sum_radii,
        x0_flat,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 2000, 'gtol': 1e-9}
    )
    centers = res.x.reshape(-1, 2)
    # Final radii update
    radii = max_radii_fixed_centers(centers, np.zeros_like(centers[:, 0]))
    
    # Final validation
    _, msg = validate_packing(centers, radii)
    if not msg == "ok":
        raise ValueError(f"Packing invalid: {msg}")

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
