import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def initial_configuration(n=26):
    # Generate Sobol sequence using Halton for better low-discrepancy sampling
    # Using a simple pseudo-random Halton-like sequence for 2D
    sobol_points = []
    for i in range(n):
        x = (i * 0.618033988749895) % 1.0  # Golden ratio
        y = (i * 0.414213562373095) % 1.0  # sqrt(2) - 1
        sobol_points.append([x, y])
    centers = np.array(sobol_points)

    # Compute initial radii:
    # Each radius is min(dist_to_edge, half_distance_to_nearest_neighbor)
    radii = []
    for i in range(n):
        x, y = centers[i]
        r_wall = min(x, 1 - x, y, 1 - y)
        distances = np.sqrt(np.sum((centers - centers[i])**2, axis=1))
        distances[i] = np.inf  # exclude self
        r_nearest = 0.5 * np.min(distances)
        r = min(r_wall, r_nearest)
        r = max(0.0, r)
        radii.append(r)

    radii = np.array(radii)
    return centers, radii

def pack(centers, radii):  # Flatten to 1D array
    return np.hstack([centers.flat, radii])

def unpack(z):  # Unflatten from 1D to centers (n,2) and radii (n,)
    n = len(z) // 3
    centers = z[:2*n].reshape((n, 2))
    radii = z[2*n:]
    return centers, radii

def max_violation(centers, radii):
    n = len(radii)
    # Check wall violations
    wall_violation = 0.0
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        wall_violation = max(wall_violation,
                             max(0.0, r - x),
                             max(0.0, r - (1 - x)),
                             max(0.0, r - y),
                             max(0.0, r - (1 - y)))
    
    # Check overlap violations
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                wall_violation = max(wall_violation, overlap)

    return wall_violation

def energy_and_grad(z):
    # Unpack
    centers, radii = unpack(z)
    n = len(radii)

    # Compute all pairwise distances
    D = cdist(centers, centers)
    np.fill_diagonal(D, np.inf)  # avoid self-distances

    # Compute energies and gradients for walls
    E_wall = 0.0
    grad_wall = np.zeros_like(z)

    for i in range(n):
        x, y = centers[i]
        r = radii[i]

        # Wall penalties
        dx = max(0.0, r - x)
        dy = max(0.0, r - (1 - x))
        dz = max(0.0, r - y)
        dw = max(0.0, r - (1 - y))

        E_wall += dx*dx + dy*dy + dz*dz + dw*dw

        # Derivatives w.r.t. center
        grad_wall[2*i] += -2.0 * dx + 2.0 * dy  # dE/dx
        grad_wall[2*i + 1] += -2.0 * dz + 2.0 * dw  # dE/dy

        # Derivatives w.r.t. radius
        grad_wall[2*n + i] += 2.0 * dx - 2.0 * dy + 2.0 * dz - 2.0 * dw

    # Compute energy and gradients for overlaps
    E_pair = 0.0
    grad_pair = np.zeros_like(z)

    lambda_pair = 100.0  # weight for pair energy
    for i in range(n):
        for j in range(i + 1, n):
            d = D[i, j]
            r_i = radii[i]
            r_j = radii[j]
            overlap = max(0.0, r_i + r_j - d)

            E_pair += overlap * overlap

            if overlap > 0:
                # Compute normalized vector from i to j
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                scale = 2.0 * overlap / (d + 1e-12)  # add small number to prevent div by zero

                # Gradients w.r.t. centers
                grad_pair[2*i] += -scale * dx
                grad_pair[2*i + 1] += -scale * dy
                grad_pair[2*j] += scale * dx
                grad_pair[2*j + 1] += scale * dy

                # Gradients w.r.t. radii
                grad_pair[2*n + i] += 2.0 * overlap
                grad_pair[2*n + j] += 2.0 * overlap

    # Budget penalty
    B = 3.0  # Initial overestimate
    lambda_budget = 1000.0
    sum_r = np.sum(radii)
    E_budget = lambda_budget * (B - sum_r) ** 2
    grad_budget = np.zeros_like(z)
    grad_budget[2*n:] = -2.0 * lambda_budget * (B - sum_r)  # only gradient w.r.t. radii

    # Total energy and gradient
    E_total = E_wall + E_pair + E_budget
    grad_total = grad_wall + grad_pair + grad_budget

    return E_total, grad_total

def inner_optimize(z0, B, lambda_budget=1000.0):
    def fun(z):
        return energy_and_grad(z)[0]

    def grad(z):
        return energy_and_grad(z)[1]

    # Bounds
    n = len(z0) // 3
    bounds = [(0.0, 1.0)] * 2 * n + [(0.0, 0.5)] * n  # center bounds [0,1], radius bounds [0,0.5]

    # Optimize
    res = minimize(fun, z0, method='L-BFGS-B', bounds=bounds, jac=grad)

    return res.x

def max_feasible_budget(z_init):
    # Binary search approach to find max feasible budget
    B = 3.0
    delta = 5e-4  # Step size
    max_iter = 40

    prev_z = z_init.copy()

    for _ in range(max_iter):
        if delta < 1e-6:
            break

        # Try increasing budget
        z_test = inner_optimize(z_init, B)
        viol = max_violation(*unpack(z_test))
        if viol <= 1e-13:
            # Feasible
            prev_z = z_test.copy()
            B += delta
            delta *= 1.2  # adaptively increase step
        else:
            # Infeasible
            B -= delta
            if B < 2.0:
                B = 2.0
            delta *= 0.8  # reduce step
            z_init = prev_z  # revert to last feasible

    return prev_z

def postprocess(z):
    centers, radii = unpack(z)

    eps = 5e-10
    # Clamp centers to stay away from boundaries
    centers = np.clip(centers, eps, 1.0 - eps)
    # Shrink radii to be safe
    radii = np.maximum(0.0, radii - eps)

    # Repair overlaps if any remain (shouldn't happen)
    for i in range(len(radii)):
        for j in range(i + 1, len(radii)):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                half_overlap = overlap / 2.0
                radii[i] -= half_overlap + eps
                radii[j] -= half_overlap + eps
                radii[i] = max(0.0, radii[i])
                radii[j] = max(0.0, radii[j])

    # Recompute sum
    sum_r = np.sum(radii)
    return centers, radii, sum_r

def run_packing():
    # Generate initial configuration
    centers, radii = initial_configuration(26)
    z = pack(centers, radii)
    
    # Perform max feasible budget search
    z_opt = max_feasible_budget(z)
    
    # Final reconstruction and postprocessing
    centers, radii, sum_r = postprocess(z_opt)
    
    return centers, radii, sum_r
