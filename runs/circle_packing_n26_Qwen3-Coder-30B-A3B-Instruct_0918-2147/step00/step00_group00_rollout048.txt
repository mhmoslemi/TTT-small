import numpy as np
from scipy.optimize import linprog
from scipy.spatial.distance import cdist

def generate_hex_lattice(h):
    """Generate a hexagonal lattice of points within [0,1]x[0,1]."""
    sqrt3 = np.sqrt(3)
    points = []
    j_max = int(np.ceil(1 / (h * sqrt3 / 2))) + 1
    i_max = int(np.ceil(1 / h)) + 1
    for j in range(j_max):
        y = j * h * sqrt3 / 2
        if y > 1:
            break
        for i in range(i_max):
            x = i * h + (j % 2) * (h / 2)
            if x > 1:
                break
            points.append([x, y])
    return np.array(points)

def build_neighbour_list(centers, max_dist=0.4):
    """Build list of neighbouring point pairs."""
    dists = cdist(centers, centers)
    pairs = []
    n = centers.shape[0]
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < max_dist:
                pairs.append((i, j, dists[i, j]))
    return pairs

def solve_radius_lp(selected_indices, R_wall, neighbour_list):
    """Solve LP for max sum of radii given selected indices."""
    n_selected = len(selected_indices)
    if n_selected == 0:
        return np.array([]), True
    # Build constraint matrix: A_ub @ r <= b_ub
    # Wall constraints: r_i <= R_wall[i]
    # Separation constraints: r_i + r_j <= d_ij
    A_ub = []
    b_ub = []
    # Wall constraints
    for i, idx in enumerate(selected_indices):
        row = np.zeros(n_selected)
        row[i] = 1
        A_ub.append(row)
        b_ub.append(R_wall[idx])
    # Separation constraints
    selected_set = set(selected_indices)
    for i, j, d_ij in neighbour_list:
        if i in selected_set and j in selected_set:
            idx_i = selected_indices.index(i)
            idx_j = selected_indices.index(j)
            row = np.zeros(n_selected)
            row[idx_i] = 1
            row[idx_j] = 1
            A_ub.append(row)
            b_ub.append(d_ij)
    # Objective: maximize sum of radii
    c = -np.ones(n_selected)
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
        if result.success:
            return result.x, True
        else:
            return np.array([]), False
    except Exception:
        return np.array([]), False

def upper_bound(partial_set, remaining_indices, R_wall):
    """Quick upper bound on sum of radii for current partial set."""
    ub = sum(R_wall[i] for i in partial_set)
    # Add largest potential wall radii for remaining spots
    if len(partial_set) < 26:
        remaining_r = [R_wall[i] for i in remaining_indices]
        remaining_r.sort(reverse=True)
        for i in range(26 - len(partial_set)):
            ub += remaining_r[i]
    return ub

def branch_and_bound(centers, R_wall, neighbour_list):
    """Perform branch-and-bound search for best 26-circle selection."""
    best_sum = -np.inf
    best_selected = None
    best_radii = None
    # Initial search: Start with empty selection
    # Node: (S_partial, U, lb, ub)
    stack = [(set(), set(range(len(centers))), -np.inf, np.inf)]
    while stack:
        S_partial, U, lb, ub = stack[-1]
        if ub <= best_sum:
            stack.pop()
            continue
        if len(S_partial) == 26:
            # Check feasibility and solve LP
            selected_indices = list(S_partial)
            radii, feasible = solve_radius_lp(selected_indices, R_wall, neighbour_list)
            if feasible:
                total = np.sum(radii)
                if total > best_sum:
                    best_sum = total
                    best_selected = selected_indices
                    best_radii = radii
            stack.pop()
        else:
            if not U:
                stack.pop()
                continue
            # Pick next candidate with largest wall radius
            max_r = -1
            next_cand = -1
            for i in U:
                if R_wall[i] > max_r:
                    max_r = R_wall[i]
                    next_cand = i
            if next_cand == -1:
                stack.pop()
                continue
            # Split into two branches
            new_S = S_partial.copy()
            new_S.add(next_cand)
            new_U = U - {next_cand}
            # Compute relaxed upper bound
            ub_upper = upper_bound(list(new_S), list(new_U), R_wall)
            if ub_upper > best_sum:
                new_node = (new_S, new_U, lb, ub_upper)
                stack.append(new_node)
            # Exclude next_cand
            new_node2 = (S_partial, new_U, lb, ub)
            stack.append(new_node2)
    return best_selected, best_radii

def local_continuous_refinement(centers, radii):
    """Refine the discrete solution with local continuous optimization."""
    n = len(radii)
    from scipy.optimize import minimize
    def objective(x):
        return -np.sum(x[2*n:])
    def constraint_fn(x):
        # x = [centers_x, centers_y, radii]
        cxs = x[:n]
        cys = x[n:2*n]
        rs = x[2*n:]
        # Boundary constraints
        # Need to avoid installing negative radii
        # and keep circles inside
        for i in range(n):
            if rs[i] < 0:
                return -rs[i]
            if cxs[i] - rs[i] < 0:
                return 0.5 * (cxs[i] - rs[i]) ** 2  # penalty
            if cxs[i] + rs[i] > 1:
                return 0.5 * (cxs[i] + rs[i] - 1) ** 2  # penalty
            if cys[i] - rs[i] < 0:
                return 0.5 * (cys[i] - rs[i]) ** 2  # penalty
            if cys[i] + rs[i] > 1:
                return 0.5 * (cys[i] + rs[i] - 1) ** 2  # penalty
        # Non-overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                dx = cxs[i] - cxs[j]
                dy = cys[i] - cys[j]
                dist_sq = dx*dx + dy*dy
                r_sum = rs[i] + rs[j]
                if dist_sq < r_sum * r_sum - 1e-10:
                    return 0.5 * (np.sqrt(dist_sq) - r_sum) ** 2
        return 0.0

    x0 = np.concatenate([centers[:, 0], centers[:, 1], radii])
    # SLSQP is good for constrained optimization
    result = minimize(objective, x0, method='SLSQP', 
                      constraints={'type': 'ineq', 'fun': constraint_fn},
                      options={'maxiter': 200, 'ftol': 1e-6})
    if result.success:
        refined_centers = np.column_stack([result.x[:n], result.x[n:2*n]])
        refined_radii = result.x[2*n:]
        # Final validation
        valid, msg = validate_packing(refined_centers, refined_radii)
        if valid:
            return refined_centers, refined_radii
    # If refinement fails or on validation failure, return untouched
    return centers, radii

def run_packing():
    h = 0.07
    centers = generate_hex_lattice(h)
    R_wall = np.min(np.column_stack([centers[:, 0], 1 - centers[:, 0], centers[:, 1], 1 - centers[:, 1]]), axis=1)
    neighbour_list = build_neighbour_list(centers, max_dist=0.4)
    # Run branch and bound
    selected_indices, radii = branch_and_bound(centers, R_wall, neighbour_list)
    # No valid solution found with branch and bound alone
    if selected_indices is None:
        # Fallback to hard-coded solution or detect failure
        raise RuntimeError("Branch and bound failed to find a solution")
    # Reconstruct full centers array
    centers_full = centers[selected_indices]
    if radii is None:
        raise RuntimeError("LP solve failed for the selected solution")
    # Local refinement
    centers_refined, radii_refined = local_continuous_refinement(centers_full, radii)
    sum_radii = float(np.sum(radii_refined))
    return centers_refined, radii_refined, sum_radii
