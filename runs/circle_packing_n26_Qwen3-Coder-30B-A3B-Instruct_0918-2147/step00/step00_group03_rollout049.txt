import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # 0. deterministic RNG seed
    np.random.seed(0)
    
    # 1. generate initial centres (perturbed hex grid) and tiny radii
    centers, radii = init_grid(26, 0.07, 0.001)
    
    # 2. physics simulation loop (forces, growth, Euler integration)
    centers, radii = simulate_physics(centers, radii, 5000, 0.02, 1e-6)
    
    # 3. fixed-centre maximal radius iteration
    radii = max_radii_fixed_centers(centers, radii)
    
    # 4. L-BFGS-B refinement on centres only, using objective that recomputes radii internally
    centers = refine_centers(centers, radii)
    
    # 5. final maximal radius pass
    radii = max_radii_fixed_centers(centers, radii)
    
    # 6. compute sum_r = radii.sum()
    sum_r = radii.sum()
    
    # 7. validate with supplied validator
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise RuntimeError(f"Validation failed: {msg}")
    
    # 8. return centers, radii, sum_r
    return centers, radii, sum_r


def init_grid(n_circles, spacing, initial_radius):
    sqrt3 = np.sqrt(3)
    rows = int(np.ceil(np.sqrt(n_circles * 2 / sqrt3)))
    cols = int(np.ceil(n_circles / rows))
    
    centers = []
    radii = []
    
    for k in range(rows):
        y = spacing / 2 + k * spacing * sqrt3 / 2
        if k % 2 == 0:
            x_start = spacing / 2
            offset = 0
        else:
            x_start = spacing / 2 + spacing / 2
            offset = spacing / 2
            
        for m in range(cols):
            x = x_start + m * spacing
            if 0 <= x <= 1 and 0 <= y <= 1:
                if min(x, 1 - x, y, 1 - y) > 0.02:
                    centers.append([x, y])
                    radii.append(initial_radius)
            if len(centers) >= n_circles:
                break
        if len(centers) >= n_circles:
            break
    
    # add small jitter
    centers = np.array(centers[:n_circles]) + np.random.uniform(-0.001, 0.001, (n_circles, 2))
    radii = np.array(radii[:n_circles])
    
    return centers, radii


def compute_forces(centers, radii, wall_stiff=10.0, pair_stiff=5.0):
    n = centers.shape[0]
    forces = np.zeros_like(centers)
    wall_forces = np.zeros_like(centers)
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # wall repulsion
        dl = x - r
        dr = 1 - x - r
        db = y - r
        dt = 1 - y - r
        if dl < 0:
            wall_forces[i, 0] += wall_stiff * (-dl)
        if dr < 0:
            wall_forces[i, 0] -= wall_stiff * (-dr)
        if db < 0:
            wall_forces[i, 1] += wall_stiff * (-db)
        if dt < 0:
            wall_forces[i, 1] -= wall_stiff * (-dt)

    # pairwise repulsion
    dists = cdist(centers, centers)
    for i in range(n):
        for j in range(i+1, n):
            if dists[i, j] < radii[i] + radii[j]:
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist_ij = np.sqrt(dx**2 + dy**2)
                if dist_ij > 1e-12:
                    force_magnitude = pair_stiff * (radii[i] + radii[j] - dist_ij)
                    force_vec = force_magnitude * np.array([dx, dy]) / dist_ij
                    forces[i] += force_vec
                    forces[j] -= force_vec

    forces += wall_forces
    return forces


def grow_radii(centers, radii, forces, growth_rate=1e-3):
    n = centers.shape[0]
    dot_radii = np.zeros(n)
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # check if touching any walls or circles (in). If so, not grow.
        door = (x - r > 1e-12) and (1 - x - r > 1e-12) and (y - r > 1e-12) and (1 - y - r > 1e-12)
        pair_press = False
        for j in range(n):
            if i != j:
                dist_ij = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist_ij < radii[i] + radii[j] - 1e-12:
                    pair_press = True
                    break
        delta = 1 - (int(not door) + int(pair_press))
        if delta == 0:
            dot_radii[i] = growth_rate
        else:
            dot_radii[i] = 0.0
    return dot_radii


def simulate_physics(centers, radii, max_steps, delta_t, min_force_norm):
    damping = 1.0
    vel = np.zeros_like(centers)
    
    for step in range(max_steps):
        forces = compute_forces(centers, radii)
        norm_max_force = np.linalg.norm(forces, axis=1).max()
        if norm_max_force < min_force_norm:
            break
            
        # update velocity
        vel += delta_t * (forces - damping * vel)
        
        # update positions
        centers += delta_t * vel
        
        # clamp to box
        centers = np.clip(centers, 0, 1)
        
        # grow radii
        dot_r = grow_radii(centers, radii, forces)
        radii += delta_t * dot_r
        
        # clamp radii
        radii = np.clip(radii, 0, None)
        
    return centers, radii


def max_radii_fixed_centers(centers, radii, tol=1e-9, max_iter=30):
    n = centers.shape[0]
    old_radii = radii.copy()
    for _ in range(max_iter):
        new_radii = np.zeros(n)
        for i in range(n):
            x, y = centers[i]
            r_i = radii[i]
            min_dist = np.min(np.sqrt(np.sum((centers[i] - centers[np.arange(n) != i]) ** 2, axis=1))) - radii[np.arange(n) != i]
            wall-dist = min(x, 1 - x, y, 1 - y)
            new_radii[i] = min(wall-dist, min_dist)
        radii = new_radii
        if np.allclose(old_radii, radii, atol=tol):
            break
        old_radii = radii
    return radii


def objective_sum_radii(centers_flat, centers, radii):
    new_centers = centers_flat.reshape(-1, 2)
    new_radii = max_radii_fixed_centers(new_centers, radii.copy())
    return -new_radii.sum()


def refine_centers(centers, radii, maxiter=2000, eps=1e-9, barrier_weight=1e-4):
    # cast to flat vector
    centers_flat = centers.flatten()
    
    # define barrier function
    def barrier(x):
        x_reshaped = x.reshape(-1, 2)
        b = 0.0
        for i in range(x_reshaped.shape[0]):
            b += np.sum(np.maximum(0, -x_reshaped[i])**2)
            b += np.sum(np.maximum(0, x_reshaped[i] - 1)**2)
        return b
    
    # modified objective
    def obj(x):
        total_obj = -objective_sum_radii(x, centers, radii)
        total_obj += barrier_weight * barrier(x)
        return total_obj
    
    # perform minimization
    res = minimize(obj, centers_flat, method='L-BFGS-B', options={'maxiter': maxiter, 'gtol': eps})
    
    return res.x.reshape(-1, 2)


def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
