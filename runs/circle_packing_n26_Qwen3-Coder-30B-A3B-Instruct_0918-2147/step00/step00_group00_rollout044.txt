import numpy as np
from scipy.optimize import minimize

def make_hex_lattice(n, margin=0.05):
    """Generate a hexagonal lattice of n points within a square with margin."""
    sqrt3 = np.sqrt(3)
    dx = 1.0 / (np.ceil(np.sqrt(n)) + 0.5)
    dy = dx * sqrt3 / 2

    points = []
    y = margin
    j = 0
    while len(points) < n:
        x = margin + (j % 2) * dx / 2
        while len(points) < n and x <= 1 - margin:
            points.append([x, y])
            x += dx
        y += dy
        j += 1

    return np.array(points[:n])

def objective(z):
    """Objective function to minimize negative sum of radii."""
    n = len(z) // 3
    radii = z[2::3]
    return -np.sum(radii)

def boundary_constraints(z):
    """Return array of boundary constraint violations."""
    n = len(z) // 3
    result = []
    for i in range(n):
        x, y, r = z[3*i], z[3*i+1], z[3*i+2]
        result.append(x - r)  # left
        result.append(1 - x - r)  # right
        result.append(y - r)  # bottom
        result.append(1 - y - r)  # top
    return np.array(result)

def pairwise_constraints(z):
    """Return array of pairwise separation constraint violations."""
    n = len(z) // 3
    result = []
    for i in range(n):
        for j in range(i + 1, n):
            x1, y1, r1 = z[3*i], z[3*i+1], z[3*i+2]
            x2, y2, r2 = z[3*j], z[3*j+1], z[3*j+2]
            dist_sq = (x1 - x2)**2 + (y1 - y2)**2
            sum_r_sq = (r1 + r2)**2
            result.append(dist_sq - sum_r_sq)
    return np.array(result)

def solve_nlp(initial_z):
    """Solve the NLP using scipy.optimize.minimize with SLSQP."""
    n = len(initial_z) // 3

    # Define bounds for all variables
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0))  # x coordinate
        bounds.append((0.0, 1.0))  # y coordinate
        bounds.append((0.0, 0.5))  # radius

    # Constraints
    bnds = {'type': 'ineq', 'fun': boundary_constraints}
    pairs = {'type': 'ineq', 'fun': pairwise_constraints}

    # Use SLSQP solver with tight tolerances
    res = minimize(
        objective,
        initial_z,
        method='SLSQP',
        bounds=bounds,
        constraints=[bnds, pairs],
        options={'ftol': 1e-9, 'maxiter': 1000}
    )

    if not res.success:
        raise RuntimeError(f"Optimization failed: {res.message}")

    return res.x

def nudge_centers(centers, radii):
    """Slightly adjust centers of circles that are almost touching a wall."""
    # Find centers that are likely near walls
    distances_to_walls = np.array([
        [centers[i, 0] - radii[i], 1 - centers[i, 0] - radii[i],
         centers[i, 1] - radii[i], 1 - centers[i, 1] - radii[i]]
        for i in range(len(centers))
    ])
    # For circles within 1e-6 of a wall, nudge them away slightly
    tolerance = 1e-6
    for i in range(len(centers)):
        min_dist = np.min(distances_to_walls[i])
        if min_dist < tolerance:
            idx = np.argmin(distances_to_walls[i])
            dx, dy = 0.0, 0.0
            if idx == 0: dx = 1e-3  # left wall
            elif idx == 1: dx = -1e-3  # right wall
            elif idx == 2: dy = 1e-3  # bottom wall
            elif idx == 3: dy = -1e-3  # top wall
            centers[i, 0] += dx
            centers[i, 1] += dy

def run_packing():
    """Main function to run the packing algorithm."""
    n = 26
    margin = 0.05   # margin from edges to allow for radii
    base_centers = make_hex_lattice(n, margin)

    # Initial guess
    initial_radii = np.full(n, 0.02)
    initial_z = np.zeros(3 * n)
    initial_z[::3] = base_centers[:, 0]
    initial_z[1::3] = base_centers[:, 1]
    initial_z[2::3] = initial_radii

    # Solve the NLP
    final_z = solve_nlp(initial_z)
    centers = final_z[::3].reshape(-1, 1)
    centers = np.hstack((centers, final_z[1::3].reshape(-1, 1)))
    radii = final_z[2::3]

    # Iterative refinement loop
    for _ in range(5):
        prev_sum = np.sum(radii)
        final_z = solve_nlp(final_z)
        centers = final_z[::3].reshape(-1, 1)
        centers = np.hstack((centers, final_z[1::3].reshape(-1, 1)))
        radii = final_z[2::3]
        if np.abs(np.sum(radii) - prev_sum) < 1e-5:
            break
        # Optional: nudge centersightly
        nudge_centers(centers, radii)

    # Validate final packing
    valid, msg = validate_packing(centers, radii)
    if not valid:
        # If validation fails, try to adjust small radii
        for _ in range(10):
            radii -= 1e-6
            valid, msg = validate_packing(centers, radii)
            if valid:
                break
        if not valid:
            raise RuntimeError(f"Validation failed after adjustments: {msg}")

    sum_radii = float(np.sum(radii))
    return centers, radii, sum_radii
