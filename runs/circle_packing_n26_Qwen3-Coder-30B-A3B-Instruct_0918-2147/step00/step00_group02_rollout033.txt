import numpy as np
from scipy.optimize import minimize

def initial_lattice(n=26):
    # Generate a hexagonal lattice
    # For simplicity, we start with a square lattice and convert to hex
    # Using the hexagonal packing pattern in a square domain
    sqrt3 = np.sqrt(3)
    spacing = 1.0
    
    # Try to create enough points for 26 circles
    rows = int(np.ceil(np.sqrt(n) * 2 / sqrt3))
    cols = int(np.ceil(n / rows))
    
    centers = []
    for i in range(rows):
        y = i * spacing * sqrt3 / 2
        if y > 1:
            break
        for j in range(cols):
            if i % 2 == 0:
                x = j * spacing
            else:
                x = j * spacing + spacing / 2
            if x > 1:
                continue
            centers.append([x, y])
            if len(centers) >= n:
                break
        if len(centers) >= n:
            break
    
    # Take first n centers
    centers = np.array(centers[:n])
    
    # Compute radii based on proximity to boundaries and neighbors
    radii = []
    for i in range(n):
        x, y = centers[i]
        r = min(x, 1 - x, y, 1 - y)  # distance to boundaries
        for j in range(n):
            if i == j:
                continue
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist = np.hypot(dx, dy)
            # If the centers are close enough that a circle of radius r would overlap
            # with the other circle, limit the radius to avoid overlaps
            if dist < 2 * r:
                r = min(r, dist / 2 - 1e-9)  # leave some margin to avoid overlap
        radii.append(r)
    
    return centers, np.array(radii)

def pack(centers, radii):
    # Flatten centers and radii
    z = np.empty(2 * centers.shape[0] + radii.shape[0])
    z[0::3] = centers[:, 0]
    z[1::3] = centers[:, 1]
    z[2::3] = radii
    return z

def unpack(z):
    n = z.shape[0] // 3
    centers = z[0::3].reshape(n, 1), z[1::3].reshape(n, 1)
    centers = np.hstack(centers)
    radii = z[2::3]
    return centers, radii

def constraint_boundary(z):
    centers, radii = unpack(z)
    constraints = []
    for i in range(centers.shape[0]):
        x, y = centers[i]
        r = radii[i]
        # All four boundary constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
        c1 = x - r  # left constraint
        c2 = 1 - x - r  # right constraint
        c3 = y - r  # bottom constraint
        c4 = 1 - y - r  # top constraint
        constraints.append(c1)
        constraints.append(c2)
        constraints.append(c3)
        constraints.append(c4)
    return np.array(constraints)

def constraint_nonoverlap(z):
    centers, radii = unpack(z)
    constraints = []
    n = centers.shape[0]
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist = np.hypot(dx, dy)
            r_sum = radii[i] + radii[j]
            constraints.append(dist - r_sum)  # non-overlap as a >= 0 constraint
    return np.array(constraints)

def objective(z):
    centers, radii = unpack(z)
    return -np.sum(radii)

def repair(centers, radii):
    n = centers.shape[0]
    # Keep a copy for comparison
    new_centers = centers.copy()
    new_radii = radii.copy()

    # Ensure no overlaps or boundary violations
    changed = True
    iterations = 0
    while changed and iterations < 10:
        changed = False
        for i in range(n):
            # Check boundary constraints
            r = new_radii[i]
            x, y = new_centers[i]
            r_safe = min(x, 1 - x, y, 1 - y)
            if r > r_safe:
                new_radii[i] = r_safe
                changed = True
            
            # Check overlapping constraints
            for j in range(i + 1, n):
                dx = new_centers[i][0] - new_centers[j][0]
                dy = new_centers[i][1] - new_centers[j][1]
                dist = np.hypot(dx, dy)
                r_sum = new_radii[i] + new_radii[j]
                if dist < r_sum:
                    overlap = r_sum - dist
                    # Reduce the radii of both circles
                    reduction = overlap / 2
                    if new_radii[i] > reduction and new_radii[j] > reduction:
                        new_radii[i] -= reduction
                        new_radii[j] -= reduction
                        changed = True
                    else:
                        # Shrink the bigger one if we can't equally reduce
                        if new_radii[i] > new_radii[j]:
                            new_radii[i] = max(1e-9, new_radii[i] - overlap)
                        else:
                            new_radii[j] = max(1e-9, new_radii[j] - overlap)
                        changed = True
                        
        iterations += 1

    return new_centers, new_radii

def run_packing():
    # Initial feasible layout from hexagonal lattice
    centers, radii = initial_lattice(n=26)
    
    # Pack into a flat vector
    z = pack(centers, radii)
    
    # Set up constraints
    cons = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_nonoverlap}
    ]
    
    # Bounds: [x_i, y_i, r_i] for i in [0, 25]
    bounds = []
    for i in range(26):
        bounds.append((0, 1))  # x_i
        bounds.append((0, 1))  # y_i
        bounds.append((0, 0.5))  # r_i (max 0.5 to fit worst case)
    
    # Minimize with trust-constr method
    res = minimize(
        objective,
        z,
        method='trust-constr',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9},
        tol=1e-9
    )
    
    # Unpack
    centers, radii = unpack(res.x)
    
    # Repair if necessary
    centers, radii = repair(centers, radii)
    
    # Compute final sum of radii
    sum_radii = np.sum(radii)
    
    # Validate
    valid, message = validate_packing(centers, radii)
    assert valid, f"Packing invalid: {message}"
    
    return centers, radii, sum_radii
