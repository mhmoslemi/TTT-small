import numpy as np
from scipy.optimize import linprog

def run_packing():
    # Parameters
    n_circles = 26
    k_gon = 8  # for polygonal approximation of forbidden regions
    cos_pi_k = np.cos(np.pi / k_gon)
    max_iter = 30
    tol = 1e-6
    inflation_passes = 5
    shrink_factor = 1e-7

    # Step 1: Generate initial hexagonal-like layout
    # Scale the triangular lattice to fill ~0.8 of unit square
    spacing = 0.8 / np.sqrt(n_circles)
    lattice_points = []
    y = 0
    while len(lattice_points) < n_circles:
        x = 0
        row = []
        while len(row) < n_circles:
            if x > 1 or y > 1:
                break
            row.append([x, y])
            x += spacing
        if len(row) > 0:
            lattice_points.extend(row)
        y += spacing * np.sqrt(3) / 2

    centers = np.array(lattice_points[:n_circles])
    # Apply margin to ensure feasibility
    margin = 0.05
    centers[:, 0] = np.clip(centers[:, 0], margin, 1 - margin)
    centers[:, 1] = np.clip(centers[:, 1], margin, 1 - margin)
    radii = np.full(n_circles, 0.01)

    # Precompute direction vectors for k-gon
    angles = np.linspace(0, 2 * np.pi, k_gon, endpoint=False)
    directions = np.column_stack([np.cos(angles), np.sin(angles)])

    # Compute pairwise distances
    def compute_distances(centers):
        diff = centers[:, None, :] - centers[None, :, :]
        return np.sqrt(np.sum(diff**2, axis=2))

    def solve_radius_lp(centers, radii):
        n = centers.shape[0]
        distances = compute_distances(centers)
        A_ub = []
        b_ub = []
        # Wall constraints
        for i in range(n):
            A_ub.append([1, 0, 0, 0])  # x
            b_ub.append(centers[i, 0])
            A_ub.append([-1, 0, 0, 0])  # -x
            b_ub.append(-centers[i, 0] + 1)
            A_ub.append([0, 1, 0, 0])  # y
            b_ub.append(centers[i, 1])
            A_ub.append([0, -1, 0, 0])  # -y
            b_ub.append(-centers[i, 1] + 1)
        # Separation constraints
        for i in range(n):
            for j in range(i + 1, n):
                dt = distances[i, j]
                A_ub.append([1, 0, 1, 0])  # dr_i + dr_j <= dist[i,j]
                b_ub.append(dt)
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        # We solve min -sum(r) = max sum(r)
        c = np.ones(n)
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', tol=1e-10)
        if result.success:
            new_radii = np.maximum(0, result.x)
            return new_radii
        else:
            # inplace update to existing radii if optimization fails
            return radii

    def solve_center_lp(centers, radii):
        n = centers.shape[0]
        A_ub = []
        b_ub = []
        C = []
        # Each circle must stay inside the unit square
        for i in range(n):
            # Constraints for x
            A_ub.append([1, 0, 0, 0])  # (center.x); x >= r
            b_ub.append(radii[i])
            A_ub.append([-1, 0, 0, 0])  # -x <= -r
            b_ub.append(-radii[i] + 1)
            # Constraints for y
            A_ub.append([0, 1, 0, 0])  # y >= r
            b_ub.append(radii[i])
            A_ub.append([0, -1, 0, 0])  # -y <= -r
            b_ub.append(-radii[i] + 1)
        # Forbidden directions between centers (polygonal approximation)
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                for d in directions:
                    l = np.dot(d, diff) + radii[i] + radii[j]
                    A_ub.append([d[0], d[1], -d[0], -d[1]])  # direction constraint
                    b_ub.append(-l * cos_pi_k)
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        c = np.ones(2 * n)  # We're going to solve for the constraints and will use slack variable after inverting 2*dimensioning
        # We restructure the problem
        # If we have slack for every variable, we solve for maximizing sum of tx (to be for x,y of centers)
        # We define slack = (center - r) or (r - center). Actually, we want maximimize the slack s_i
        # We introduce new slack variables over four sides of a square. 
        # s_x_left - c_x + r <= 0 => s_x_left + c_x >= r  
        # s_x_right - (1 - c_x) + r <= 0 => s_x_right + c_x >= 1 - r 
        # And similar for s_y_left, s_y_right constraints. Then the objective is to maximize sum of all slack.
        # No efficient way to translate the general k-gon constraint in that way. So we revert back to traditional formulation with slack
        # Adding new slack variables for each center
        # Re-work of model:
        # maximize sum s
        # s_i <= c_i.x - r_i
        # s_i <= 1 - c_i.x - r_i
        # s_i <= c_i.y - r_i
        # s_i <= 1 - c_i.y - r_i
        # s_i <= dist(c_i, c_j) - r_i - r_j for all non-overlapping pairs

        result = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', tol=1e-10)
        if result.success:
            updated = centers.copy()
            next_centers = result.x.reshape((n, 2))
            updated[:, 0] = np.clip(next_centers[:, 0], radii, 1 - radii)
            updated[:, 1] = np.clip(next_centers[:, 1], radii, 1 - radii)
            return updated

        else:
            return centers

    # Alternate solving loops
    old_sum = 0
    for iteration in range(max_iter):
        new_radii = solve_radius_lp(centers, radii)
        old_sum = np.sum(radii)
        radii = new_radii
        centers = solve_center_lp(centers, radii)
        new_sum = np.sum(radii)
        if new_sum - old_sum < tol:
            break

    # Step 5: Local inflation
    # The purpose is to increase radii where exact geometry allows without overlaps.
    distances = compute_distances(centers)
    for _ in range(inflation_passes):
        any_changed = False
        for i in range(n_circles):
            # Check constraint with each neighbor
            max_allowable = float('inf')
            for j in range(n_circles):
                if i != j:
                    gap = distances[i, j] - radii[i] - radii[j]
                    if gap > 0:
                        max_allowable = min(max_allowable, gap)
            if max_allowable < float('inf'):
                added_radius = min(0.4 * max_allowable, 0.1)
                if added_radius > 1e-5:
                    new_rad = min(0.5, radii[i] + added_radius)
                    old_r = radii[i]
                    radii[i] = new_rad
                    any_changed = True
        if not any_changed:
            break

    # Handled possible boundary violations by shrinking radii
    # Validate before returning
    valid, msg = validate_packing(centers, radii)
    if not valid:
        for _ in range(5):
            radii *= (1 - shrink_factor)
            valid, msg = validate_packing(centers, radii)
            if valid:
                break

    # Final check to ensure valid packing
    valid, msg = validate_packing(centers, radii)
    assert valid, f"Validation failed: {msg}"

    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
