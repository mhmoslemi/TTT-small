import numpy as np
from scipy.optimize import differential_evolution
from scipy.spatial.distance import cdist

def compute_radii(centers):
    """Compute maximal radii for given centers, respecting boundaries and non-overlap."""
    n = centers.shape[0]
    radii = np.zeros(n)
    
    # Wall distances
    left = centers[:, 0]
    right = 1.0 - centers[:, 0]
    bottom = centers[:, 1]
    top = 1.0 - centers[:, 1]
    wall_distances = np.minimum.reduce([left, right, bottom, top])
    
    # Pairwise distances
    distances = cdist(centers, centers)
    np.fill_diagonal(distances, np.inf)
    
    # Neighbour-based distances (half of minimum distance to any other circle)
    neighbour_distances = 0.5 * np.min(distances, axis=1)
    
    # Radii are the minimum of wall and neighbour distances
    radii = np.minimum(wall_distances, neighbour_distances)
    
    return radii

def neg_sum_radii(z):
    """Objective function for differential evolution: negative sum of radii."""
    centers = z.reshape(-1, 2)
    radii = compute_radii(centers)
    return -np.sum(radii)
