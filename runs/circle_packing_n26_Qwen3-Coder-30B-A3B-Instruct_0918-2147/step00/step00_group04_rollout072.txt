import numpy as np
from scipy.optimize import minimize

NUM_CIRCLES = 26
NUM_RESTARTS = 5

def hex_lattice_points(n_rows, spacing):
    points = []
    for i in range(n_rows):
        y = i * spacing * np.sqrt(3) / 2
        if y > 1:
            break
        offset = (i % 2) * spacing / 2
        n_cols = int((1 + spacing / 2) / spacing)
        for j in range(n_cols):
            x = offset + j * spacing
            if x > 1:
                break
            points.append((x, y))
    return np.array(points)

def initial_guess(seed):
    np.random.seed(seed)
    # Create a hexagonal lattice that's large enough to fit 26 circles
    spacing = 0.2  # Start with spacing that should yield ~30 points
    initial_points = hex_lattice_points(10, spacing)
    if len(initial_points) < NUM_CIRCLES:
        spacing = 0.15
        initial_points = hex_lattice_points(10, spacing)
    
    # Trim to NUM_CIRCLES or pad with jitter if needed
    if len(initial_points) >= NUM_CIRCLES:
        points = initial_points[:NUM_CIRCLES]
    else:
        # Pad with jittered copies if needed
        points = np.vstack([initial_points,
                            np.random.rand(NUM_CIRCLES - len(initial_points), 2) * 0.8 + 0.1])
    
    # Add small jitter
    points += np.random.rand(NUM_CIRCLES, 2) * 0.02 - 0.01
    
    # Limit to [0,1]^2
    points = np.clip(points, 0, 1)
    
    # Set small initial radii
    radii = np.full(NUM_CIRCLES, 0.02)
    
    # Flatten to a single array: [x1, y1, r1, x2, y2, r2, ...]
    vars0 = np.zeros(NUM_CIRCLES * 3)
    vars0[0::3] = points[:, 0]
    vars0[1::3] = points[:, 1]
    vars0[2::3] = radii
    
    return vars0

def objective(vars):
    # minimize negative sum of radii
    return -np.sum(vars[2::3])

def constraint_boundary(vars):
    # each circle must stay within the square: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    margins = np.array([
        x - r,
        1 - x - r,
        y - r,
        1 - y - r
    ])
    return margins.T.flatten()

def constraint_nonoverlap(vars):
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    n = len(x)
    
    # For all pairs of circles: distance - (r_i + r_j) >= 0
    all_margins = []
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
            margin = dist - (r[i] + r[j])
            all_margins.append(margin)
    return np.array(all_margins)

def postprocess(vars):
    # Fix violations of the boundary constraints due to numerical errors
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # For boundary issues: clip centers to fit inside square and maintain radius
    x = np.clip(x, r, 1 - r)
    y = np.clip(y, r, 1 - r)
    
    # Reconstruct
    new_vars = np.zeros_like(vars)
    new_vars[0::3] = x
    new_vars[1::3] = y
    new_vars[2::3] = r
    
    return new_vars

def run_packing():
    best_sum = -np.inf
    best_vars = None

    for seed in range(NUM_RESTARTS):
        vars0 = initial_guess(seed)
        # Set bounds for x, y, r
        bounds = [(0,1), (0,1), (0,0.5)] * NUM_CIRCLES
        
        constraints = [
            {'type': 'ineq', 'fun': constraint_boundary},
            {'type': 'ineq', 'fun': constraint_nonoverlap}
        ]
        
        res = minimize(objective, vars0,
                       method='SLSQP',
                       bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': 3000, 'ftol': 1e-7})
        
        if not res.success:
            continue
            
        vars_corr = postprocess(res.x)
        s = np.sum(vars_corr[2::3])
        if s > best_sum:
            best_sum = s
            best_vars = vars_corr

    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii = best_vars[2::3]
    
    # Final validation
    is_valid, msg = validate_packing(centers, radii)
    if not is_valid:
        raise ValueError(f"Final packing invalid: {msg}")
    
    return centers, radii, float(best_sum)
