import numpy as np
from scipy.optimize import minimize

def initial_lattice(n=26):
    # Generate hexagonal lattice points
    # We'll use a grid and pick first n points that fit within a square
    # with radius s/2 around each point
    
    # For a hexagonal lattice, spacing along x-axis is s, 
    # and y-axis spacing is s * sqrt(3)/2
    # We'll use a trial and error method to find a suitable spacing
    spacing = 0.2
    while True:
        points = []
        i = 0
        while len(points) < 50:  # Generate plenty of points to sample
            j = 0
            while len(points) < 50 and j < 10:
                if i % 2 == 0:
                    x = i * spacing
                    y = j * spacing * np.sqrt(3) / 2
                else:
                    x = i * spacing + spacing / 2
                    y = j * spacing * np.sqrt(3) / 2
                    
                if 0 <= x <= 1 and 0 <= y <= 1:
                    # Check if a circle of radius spacing/2 would fit
                    if (x - spacing/2 >= 0 and x + spacing/2 <= 1 and 
                        y - spacing/2 >= 0 and y + spacing/2 <= 1):
                        points.append((x, y))
                    
                j += 1
            i += 1
        if len(points) >= n:
            break
        spacing *= 0.95
    
    # Now limit to first n points
    centers = np.array(points[:n])
    
    # Compute admissible radii
    radii = np.full(n, spacing / 2.0)
    
    # For each circle, compute actual admissible radius
    for i in range(n):
        x, y = centers[i]
        r = spacing / 2.0
        
        # Compute minimum distance to boundaries
        min_dist_to_bound = min(x, 1-x, y, 1-y)
        r = min(r, min_dist_to_bound)
        
        # Compute minimum of distance to other centers divided by 2
        for j in range(n):
            if i != j:
                x2, y2 = centers[j]
                dist = np.sqrt((x - x2)**2 + (y - y2)**2)
                if dist < 2 * r:
                    r_new = dist / 2
                    r = min(r, r_new)
        
        # Still need to ensure r is non-negative
        radii[i] = max(0, r)
    
    # Make sure we still have at least n points (should be true if spacing is reasonable)
    return centers, radii

def pack(centers, radii):
    z = np.zeros(3 * len(centers))
    for i in range(len(centers)):
        z[3*i] = centers[i, 0]
        z[3*i+1] = centers[i, 1]
        z[3*i+2] = radii[i]
    return z

def unpack(z):
    n = len(z) // 3
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = z[3*i]
        centers[i, 1] = z[3*i+1]
        radii[i] = z[3*i+2]
    return centers, radii

def objective(z):
    _, radii = unpack(z)
    # Return negative because we want to maximize
    return -np.sum(radii)

def constraint_boundary(z):
    centers, radii = unpack(z)
    n = len(centers)
    constraints = []
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Left constraint: r - x <= 0 
        constraints.append(r - x)
        # Right constraint: r + x - 1 <= 0
        constraints.append(r + x - 1)
        # Bottom constraint: r - y <= 0
        constraints.append(r - y)
        # Top constraint: r + y - 1 <= 0
        constraints.append(r + y - 1)
    return np.array(constraints)

def constraint_nonoverlap(z):
    centers, radii = unpack(z)
    n = len(centers)
    constraints = []
    
    for i in range(n):
        for j in range(i + 1, n):
            x1, y1 = centers[i]
            x2, y2 = centers[j]
            r1 = radii[i]
            r2 = radii[j]
            
            # Distance between centers
            dx = x1 - x2
            dy = y1 - y2
            
            # Using hypot to avoid numerical issues
            dist = np.hypot(dx, dy)
            
            # Non-overlap constraint
            constraints.append(dist - (r1 + r2))
    
    return np.array(constraints)

def repair(centers, radii):
    # Simple repair procedure
    n = len(centers)
    
    # Apply boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Clip r to stay within bounds
        r = min(r, x, 1 - x, y, 1 - y)
        radii[i] = max(0, r)
    
    # Resolve overlaps
    max_iter = 10
    for iteration in range(max_iter):
        changed = False
        for i in range(n):
            for j in range(i + 1, n):
                x1, y1 = centers[i]
                x2, y2 = centers[j]
                r1 = radii[i]
                r2 = radii[j]
                
                dist = np.hypot(x1 - x2, y1 - y2)
                overlap = r1 + r2 - dist
                
                if overlap > 0:
                    # Reduce both radii
                    reduction = overlap / 2
                    r1_new = max(0, r1 - reduction)
                    r2_new = max(0, r2 - reduction)
                    
                    # If any reduce to zero, we make the other one smaller accordingly
                    if r1_new == 0 or r2_new == 0:
                        # Try a different strategy to preserve one circle
                        max_reduction = min(r1, r2)
                        rad_ratio = r1 / (r1 + r2 + 1e-12)
                        reduction_i = max_reduction * rad_ratio
                        reduction_j = max_reduction * (1 - rad_ratio)
                        r1_new = max(0, r1 - reduction_i)
                        r2_new = max(0, r2 - reduction_j)
                    
                    # Apply changes
                    radii[i] = r1_new
                    radii[j] = r2_new
                    changed = True
    
        if not changed:
            break
            
    return centers, radii

def run_packing():
    centers, radii = initial_lattice(26)
    
    # Convert to flat vector
    z0 = pack(centers, radii)
    
    # Bounds for variables
    # x, y in [0, 1], r in [0, 0.5] (max possible for unit square)
    bounds = [(0, 1), (0, 1), (0, 0.5)] * 26
    
    # Build constraints
    constraints = []
    
    # Boundary constraints
    constraints.append({
        'type': 'ineq',
        'fun': constraint_boundary
    })
    
    # Non-overlap constraints
    constraints.append({
        'type': 'ineq',
        'fun': constraint_nonoverlap
    })
    
    # Optimize
    result = minimize(
        objective, 
        z0, 
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 500, 'ftol': 1e-9, 'disp': False}
    )
    
    if not result.success:
        raise RuntimeError("Optimization failed")
    
    centers_opt, radii_opt = unpack(result.x)
