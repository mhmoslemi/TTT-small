import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial.distance import cdist

def build_lp_model(n):
    # Build LP model for n circles
    # Variables: [x1, y1, r1, x2, y2, r2, ..., xn, yn, rn]
    # Each circle has 3 variables
    n_vars = 3 * n
    # Objective vector: maximize sum of radii
    c = np.zeros(n_vars)
    for i in range(n):
        c[3 * i + 2] = -1  # Coefficient for radius variable

    # Constraints: A_ub @ x <= b_ub, A_eq @ x = b_eq
    A_ub = []
    b_ub = []
    A_eq = []
    b_eq = []

    # Wall constraints: r <= x, r <= 1-x, r <= y, r <= 1-y
    for i in range(n):
        # r <= x -> -r + x >= 0
        a = np.zeros(n_vars)
        a[3 * i + 0] = 1  # x_i
        a[3 * i + 2] = -1  # -r_i
        A_ub.append(a)
        b_ub.append(0)

        # r <= 1-x -> r + x - 1 >= 0
        a = np.zeros(n_vars)
        a[3 * i + 0] = 1  # x_i
        a[3 * i + 2] = 1  # r_i
        A_ub.append(a)
        b_ub.append(1)

        # r <= y -> -r + y >= 0
        a = np.zeros(n_vars)
        a[3 * i + 1] = 1  # y_i
        a[3 * i + 2] = -1  # -r_i
        A_ub.append(a)
        b_ub.append(0)

        # r <= 1-y -> r + y - 1 >= 0
        a = np.zeros(n_vars)
        a[3 * i + 1] = 1  # y_i
        a[3 * i + 2] = 1  # r_i
        A_ub.append(a)
        b_ub.append(1)

    # Pairwise non-overlap constraints (axis-aligned)
    # For each pair (i,j), we add constraints:
    # x_i + r_i <= x_j - r_j   ->   x_i - x_j + r_i + r_j <= 0
    # x_j + r_j <= x_i - r_i   ->   x_j - x_i + r_i + r_j <= 0
    # y_i + r_i <= y_j - r_j   ->   y_i - y_j + r_i + r_j <= 0
    # y_j + r_j <= y_i - r_i   ->   y_j - y_i + r_i + r_j <= 0

    for i in range(n):
        for j in range(i + 1, n):
            # x_i - x_j + r_i + r_j <= 0
            a = np.zeros(n_vars)
            a[3 * i + 0] = 1  # x_i
            a[3 * j + 0] = -1  # -x_j
            a[3 * i + 2] = 1  # r_i
            a[3 * j + 2] = 1  # r_j
            A_ub.append(a)
            b_ub.append(0)

            # x_j - x_i + r_i + r_j <= 0
            a = np.zeros(n_vars)
            a[3 * i + 0] = -1  # -x_i
            a[3 * j + 0] = 1  # x_j
            a[3 * i + 2] = 1  # r_i
            a[3 * j + 2] = 1  # r_j
            A_ub.append(a)
            b_ub.append(0)

            # y_i - y_j + r_i + r_j <= 0
            a = np.zeros(n_vars)
            a[3 * i + 1] = 1  # y_i
            a[3 * j + 1] = -1  # -y_j
            a[3 * i + 2] = 1  # r_i
            a[3 * j + 2] = 1  # r_j
            A_ub.append(a)
            b_ub.append(0)

            # y_j - y_i + r_i + r_j <= 0
            a = np.zeros(n_vars)
            a[3 * i + 1] = -1  # -y_i
            a[3 * j + 1] = 1  # y_j
            a[3 * i + 2] = 1  # r_i
            a[3 * j + 2] = 1  # r_j
            A_ub.append(a)
            b_ub.append(0)

    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    A_eq = np.array(A_eq)
    b_eq = np.array(b_eq)

    return c, A_ub, b_ub, A_eq, b_eq


def solve_lp(c, A_ub, b_ub, A_eq, b_eq):
    # To ensure numerical stability, we'll use the 'highs' method if available
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, method='highs')
    if not res.success:
        raise RuntimeError("Failed to solve LP: " + str(res.message))

    # Extract centers and radii
    x = res.x
    centers = np.array([[x[3 * i + 0], x[3 * i + 1]] for i in range(len(x) // 3)])
    radii = np.array([x[3 * i + 2] for i in range(len(x) // 3)])

    # Identify active wall constraints
    active_wall = []
    for i in range(len(radii)):
        # r <= x  ->  r - x <= 0
        if np.abs(x[3 * i + 2] - x[3 * i + 0]) < 1e-9:
            active_wall.append((i, 'x0'))
        # r <= 1-x     ->   x + r - 1 <= 0
        if np.abs(x[3 * i + 0] + x[3 * i + 2] - 1) < 1e-9:
            active_wall.append((i, 'x1'))
        # r <= y   ->   r - y <= 0
        if np.abs(x[3 * i + 2] - x[3 * i + 1]) < 1e-9:
            active_wall.append((i, 'y0'))
        # r <= 1-y    ->   y + r - 1 <= 0
        if np.abs(x[3 * i + 1] + x[3 * i + 2] - 1) < 1e-9:
            active_wall.append((i, 'y1'))

    # Identify active pairwise separation constraints
    active_pairs = []
    for i in range(len(radii)):
        for j in range(i + 1, len(radii)):
            # Check if x_i - x_j + r_i + r_j <= 0 is active
            if np.abs(x[3 * i + 0] - x[3 * j + 0] + x[3 * i + 2] + x[3 * j + 2]) < 1e-9:
                # Going left or right?
                # Consider x_i + r_i <= x_j - r_j: x_i - x_j + r_i + r_j <= 0
                # If true, then i is to left of j
                active_pairs.append((i, j, 'x_left'))
            # Check if x_j - x_i + r_i + r_j <= 0 is active
            elif np.abs(x[3 * j + 0] - x[3 * i + 0] + x[3 * i + 2] + x[3 * j + 2]) < 1e-9:
                active_pairs.append((i, j, 'x_right'))
            # Check if y_i - y_j + r_i + r_j <= 0 is active
            elif np.abs(x[3 * i + 1] - x[3 * j + 1] + x[3 * i + 2] + x[3 * j + 2]) < 1e-9:
                active_pairs.append((i, j, 'y_below'))
            # Check if y_j - y_i + r_i + r_j <= 0 is active
            elif np.abs(x[3 * j + 1] - x[3 * i + 1] + x[3 * i + 2] + x[3 * j + 2]) < 1e-9:
                active_pairs.append((i, j, 'y_above'))

    active = {
        'wall': active_wall,
        'pairs': active_pairs,
        'vars': x
    }

    return centers, radii, active


def detect_diagonal_conflicts(centers, radii):
    # Compute pairwise Euclidean distances
    D = cdist(centers, centers, metric='euclidean')
    n = len(centers)
    conflicts = []
    for i in range(n):
        for j in range(i + 1, n):
            if D[i][j] < radii[i] + radii[j] - 1e-12:
                conflicts.append((i, j))
    return conflicts

    # This could still be slow, so use careful local optimization
    # to avoid the full optimization, but ensure feasibility

    centers_new = centers.copy()
    radii_new = radii.copy()

    # The trick is NOT to solve a full NLP, but run a modest r adjustment
    # Here we will just compensate via slight radii adjustment and movement

    # Each conflict is resolved by decreasing radii

                    changed = True

def max_radius_projection(centers, radii):
    # Now project each point to maximum possible safe radius
    # i.e. investigate orthogonal unit edge view and other minimal radial intervals,
    # depart measures and redefine to normalized Byzantine distance boundaries.
    n = len(centers)
    new_radii = radii.copy()
    iter_count = 0
    max_iter = 1000  # It's assumed to converge fast
    while iter_count < max_iter:
        changed = False
        for i in range(n):
            if np.isnan(centers[i][0]) or np.isnan(centers[i][1]) or np.isnan(new_radii[i]):
                continue
            try:
                # Maintain eight bounding edge constraints.
                mr = min(
                    centers[i][0], 1 - centers[i][0],
                    centers[i][1], 1 - centers[i][1]
                )
                # Minimum Euclidean distance to neighbors:
                furthest = float('inf')
                for j in range(n):
                    if i == j:
                        continue
                    d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                    if d > 0:
                        repulsive = d - new_radii[j]  # we are negating r_j in the dot product of total geometric norm fallback
                        if repulsive < furthest:
                            furthest = repulsive
                if furthest < mr:
                    mr = furthest
                if mr > new_radii[i]:
                    new_radii[i] = mr
                    changed = True
            except Exception:
                pass
        if not changed:
            break
        iter_count += 1
    return new_radii
