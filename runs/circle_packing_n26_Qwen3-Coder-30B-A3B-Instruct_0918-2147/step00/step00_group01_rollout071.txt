import numpy as np
from scipy.optimize import differential_evolution
from scipy.spatial.distance import cdist

def compute_radii(centers):
    """Compute maximal radii for given centers, respecting boundaries and non-overlap."""
    n = centers.shape[0]
    radii = np.zeros(n)
    
    # Wall distances
    x = centers[:, 0]
    y = centers[:, 1]
    wall_distances = np.array([x, 1 - x, y, 1 - y]).T
    
    # Pairwise distances
    distances = cdist(centers, centers)
    np.fill_diagonal(distances, np.inf)
    
    for i in range(n):
        # Distance to walls
        wall_r = np.min(wall_distances[i])
        
        # Distance to other circles (half of minimum distance)
        neighbor_r = 0.5 * np.min(distances[i])
        
        radii[i] = min(wall_r, neighbor_r)
    
    return radii

def neg_sum_radii(z):
    """Objective function to minimize (negative of sum of radii)."""
    centers = z.reshape(-1, 2)
    radii = compute_radii(centers)
    return -np.sum(radii)
