import numpy as np
import cvxpy as cp
from scipy.spatial.distance import cdist

def build_leaf_problem(cell, k, circles_in_cell):
    # cell: (x_min, y_min, x_max, y_max)
    x_min, y_min, x_max, y_max = cell
    n = len(circles_in_cell)
    
    # Variables for centers and radii
    centers = cp.Variable((n, 2))
    radii = cp.Variable(n)
    
    # Constraints for boundary
    constraints = [
        centers[:, 0] >= x_min + radii,
        centers[:, 0] <= x_max - radii,
        centers[:, 1] >= y_min + radii,
        centers[:, 1] <= y_max - radii
    ]
    
    # Non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist_sq = (centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2
            constraints.append(dist_sq >= (radii[i] + radii[j])**2)
    
    # Objective: maximize sum of radii
    objective = cp.Maximize(cp.sum(radii))
    
    # Create problem instance
    prob = cp.Problem(objective, constraints)
    return prob, centers, radii

def solve_leaf(cell, k):
    # Define leaf cell constraints and variables
    prob, centers, radii = build_leaf_problem(cell, k, list(range(k)))
    try:
        prob.solve(solver=cp.ECOS, abstol=1e-10, reltol=1e-10)
        if prob.status != cp.OPTIMAL:
            raise ValueError(f"Problem status: {prob.status}")
    except Exception as e:
        raise RuntimeError("Could not solve leaf problem") from e
    return centers.value, radii.value

def build_cluster_problem(clusters):
    # Combine clusters to form a merged cluster
    all_centers = []
    all_radii = []
    all_constraints = []
    all_variables = []
    violations = []
    
    # Collect all centers and radii
    total_circle_count = 0
    for cluster in clusters:
        centers, radii = cluster["centers"], cluster["radii"]
        all_centers.append(centers)
        all_radii.append(radii)
        total_circle_count += len(radii)
    
    # Flatten the directions and variables
    all_centers = np.vstack(all_centers)
    all_radii = np.hstack(all_radii) 
    
    # Build new variables for the combined system
    combined_centers = cp.Variable((total_circle_count, 2))
    combined_radii = cp.Variable(total_circle_count)
    
    # Penalty constraints
    max_edge = 1.0
    entry_constraints = []
    
    # Boundary constraints for inside unit square
    for i in range(total_circle_count):
        entry_constraints.append(combined_centers[i, 0] >= combined_radii[i])
        entry_constraints.append(combined_centers[i, 0] <= max_edge - combined_radii[i])
        entry_constraints.append(combined_centers[i, 1] >= combined_radii[i])
        entry_constraints.append(combined_centers[i, 1] <= max_edge - combined_radii[i])
    
    # Non-overlap constraints
    for i in range(total_circle_count):
        for j in range(i+1, total_circle_count):
            d_ij = cp.sqrt((combined_centers[i, 0] - combined_centers[j, 0])**2 + 
                        (combined_centers[i, 1] - combined_centers[j, 1])**2)
            entry_constraints.append(d_ij >= combined_radii[i] + combined_radii[j])
    
    # Minimize total negative sum of radii to maximize sum
    objective = cp.Minimize(-cp.sum(combined_radii))
    prob = cp.Problem(objective, entry_constraints)
    
    try:
        prob.solve(solver=cp.ECOS, abstol=1e-10, reltol=1e-10)
        if prob.status != cp.OPTIMAL:
            raise ValueError(f"Merged cluster problem status: {prob.status}")
    except Exception as e:
        raise RuntimeError("Could not solve cluster merge problem") from e
        
    return combined_centers.value, combined_radii.value

def hierarchical_merge(clusters, adjacency):
    # Simple binary tree merge approach
    # Use list of clusters as stack, merge pairs by pairs
    while len(clusters) > 1:
        # Merge the first two
        a, b = clusters[0], clusters[1]
        merged_centers, merged_radii = build_cluster_problem([a, b])
        merged_cluster = {"centers": merged_centers, "radii": merged_radii}
        # Remove first two and add result
        clusters = [merged_cluster] + clusters[2:]
    return clusters[0]

def global_polish(centers, radii):
    # Apply barrier-gradient refinement
    n = len(centers)
    x = centers[:, 0]
    y = centers[:, 1]
    r = radii
    
    # Objective with barrier terms
    def get_obj_grad():
        # Sum of radii
        obj = np.sum(r)
        # Barrier terms: log(x-r), log(1-x-r), etc., and log(dist-r1-r2) for overlaps
        # Soft constraint for box
        barrier = 0
        d = np.zeros((n, n))
        grad_x = np.zeros(n)
        grad_y = np.zeros(n)
        grad_r = np.zeros(n)
        
        # Boundary term
        for i in range(n):
            if r[i] >= x[i]:
                barrier -= np.log(x[i] - r[i])  # Negative for min/max convention
                grad_x[i] -= 1.0 / (x[i] - r[i])
            if r[i] >= 1 - x[i]:
                barrier -= np.log(1 - x[i] - r[i])
                grad_x[i] += 1.0 / (1 - x[i] - r[i])
            if r[i] >= y[i]:
                barrier -= np.log(y[i] - r[i])
                grad_y[i] -= 1.0 / (y[i] - r[i])
            if r[i] >= 1 - y[i]:
                barrier -= np.log(1 - y[i] - r[i])
                grad_y[i] += 1.0 / (1 - y[i] - r[i])
        
        # Distance constraint terms
        dists = cdist(np.column_stack([x, y]), np.column_stack([x, y]))
        for i in range(n):
            for j in range(i+1, n):
                d_ij = dists[i, j]
                r_sum = r[i] + r[j]
                if d_ij < r_sum:
                    dist_barrier = np.log(r_sum - d_ij)
                    barrier -= 0.0001 * dist_barrier  # small mu
                    g = (d_ij - r_sum) / (r_sum - d_ij) / (d_ij - r_sum)**2
                    dx_ij = (x[i] - x[j])/(d_ij + 1e-12)
                    dy_ij = (y[i] - y[j])/(d_ij + 1e-12)
                    grad_x[i] += 0.0001 * g * dx_ij
                    grad_x[j] += 0.0001 * g * dx_ij
                    grad_y[i] += 0.0001 * g * dy_ij
                    grad_y[j] += 0.0001 * g * dy_ij
                    grad_r[i] += 0.0001 * g
                    grad_r[j] += 0.0001 * g
        
        return -obj, -grad_x, -grad_y, -grad_r  # Minimizing negative of sum
    
    # Gradient ascent step
    step = 0.01
    for _ in range(3):
        obj, gx, gy, gr = get_obj_grad()
        # Update
        x = x + step * gx
        y = y + step * gy
        r = r + step * gr
        
        # Keep radii positive and inside bounds
        x = np.clip(x, r, 1-r)
        y = np.clip(y, r, 1-r)
        r = np.maximum(r, 0.0)
        
    return np.column_stack([x, y]), r

def run_packing():

    ]

    clusters = []

    merged_cluster = hierarchical_merge(clusters, None)
    
    # Final polishing using gradient ascent and barriers

    # Final optimization to clean up rounding errors

    # Force the new values in the constraints
    try:

        pass  # Keep the approximate version if not deterministic
    
    # Correction
