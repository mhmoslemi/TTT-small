import numpy as np
import random

def make_hex_seed(n, margin):
    """Generate initial hexagonal seed configuration."""
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / float(rows)))
    spacing = 1.0 / max(rows, cols)
    x0 = margin + spacing / 2.0
    y0 = margin + spacing / 2.0
    dx = spacing * 0.8660254037844386  # sqrt(3)/2
    dy = spacing * 0.5
    centers = []
    radii = []
    count = 0
    for i in range(rows):
        for j in range(cols):
            if count >= n:
                break
            x = x0 + j * spacing
            if i % 2 == 1:
                x += dx
            y = y0 + i * spacing * dy
            if x <= 1 - margin and y <= 1 - margin:
                centers.append([x, y])
                # Compute distance to nearest wall
                d = min(x, 1 - x, y, 1 - y)
                radii.append(min(0.1, d * 0.4))
                count += 1
        if count >= n:
            break
    # Maybe fill remaining slots with small random circles near center
    while len(centers) < n:
        x = random.uniform(0.1, 0.9)
        y = random.uniform(0.1, 0.9)
        d = min(x, 1 - x, y, 1 - y)
        r = min(0.02, d * 0.3)
        centers.append([x, y])
        radii.append(r)
    return np.array(centers), np.array(radii)

def max_feasible_radius(i, centers, radii):
    """Compute the maximum feasible radius for circle i given the current setup."""
    x, y = centers[i]
    r = radii[i]
    # Wall distances
    d_L = x
    d_R = 1 - x
    d_B = y
    d_T = 1 - y
    min_dist = min(d_L, d_R, d_B, d_T)
    # Neighbour distances
    for j in range(len(centers)):
        if i != j:
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            min_dist = min(min_dist, dist - radii[j])
    # Max feasible radius is the minimum of all constraints
    return min_dist

def inflate_radii(centers, radii, alpha):
    """Uniform global inflation."""
    mean_r = np.mean(radii)
    delta = alpha * mean_r
    new_radii = radii + delta
    updated_radii = []
    total_delta = 0.0
    for i in range(len(centers)):
        max_feas = max_feasible_radius(i, centers, new_radii)
        clamped = min(max_feas, new_radii[i])
        delta_i = clamped - radii[i]
        total_delta += delta_i
        updated_radii.append(clamped)
    return np.array(updated_radii), total_delta

def annl_neighbourhood(x, y, r, centers, radii):
    """Get circles that may be close to the circle at (x, y) with radius r."""
    neighbours = []
    for i in range(len(centers)):
        if i == len(centers):
            continue
        dx = centers[i][0] - x
        dy = centers[i][1] - y
        dist_sq = dx * dx + dy * dy
        if dist_sq < (r + radii[i]) ** 2:
            neighbours.append(i)
    return neighbours

def anneal_centers(centers, radii, temp0, gamma, sigma0, steps):
    """Stochastic Annealing to locally improve circle centers."""
    # Copy of centers to modify
    c_copy = centers.copy()
    r_copy = radii.copy()
    current_total = np.sum(r_copy)
    best_total = current_total
    best_centers = c_copy.copy()
    best_radii = r_copy.copy()
    temp = temp0
    sigma = sigma0
    for step in range(steps):
        # Occasionally shift the entire configuration slightly to break symmetry
        if step % 50 == 0 and step > 0:
            shift_x = random.uniform(-0.002, 0.002)
            shift_y = random.uniform(-0.002, 0.002)
            c_copy[:, 0] += shift_x
            c_copy[:, 1] += shift_y
            # Adjust all radii according to new positions
            for i in range(len(c_copy)):
                r_copy[i] = max_feasible_radius(i, c_copy, r_copy)
            current_total = np.sum(r_copy)
            if current_total > best_total:
                best_total = current_total
                best_centers[:] = c_copy[:]
                best_radii[:] = r_copy[:]
        # Select a random circle to perturb
        idx = random.randrange(len(c_copy))
        # Displacement
        dx = random.gauss(0, sigma)
        dy = random.gauss(0, sigma)
        old_x, old_y = c_copy[idx, 0], c_copy[idx, 1]
        new_x = old_x + dx
        new_y = old_y + dy
        c_copy[idx, 0] = new_x
        c_copy[idx, 1] = new_y
        # Update the feasible radius of this circle only
        old_r = r_copy[idx]
        new_r = max_feasible_radius(idx, c_copy, r_copy)
        # Accept or reject the move
        if new_r > old_r:
            # Accept
            r_copy[idx] = new_r
            current_total += (new_r - old_r)
            if current_total > best_total:
                best_total = current_total
                best_centers[:] = c_copy[:]
                best_radii[:] = r_copy[:]
        else:
            # Accept with probability
            delta = new_r - old_r
            if random.random() < np.exp(delta / temp):
                r_copy[idx] = new_r
                current_total += delta
                if current_total > best_total:
                    best_total = current_total
                    best_centers[:] = c_copy[:]
                    best_radii[:] = r_copy[:]
        # Decrement temperature
        temp *= gamma
        # Update sigma
        if temp > 0:
            sigma = sigma0 * temp / temp0
    return best_centers, best_radii, best_total

def polished_radius_inflation(centers, radii, beta):
    """Increasingly inflate radii to fill exact gaps."""
    updated_radii = radii.copy()
    for _ in range(20):  # Should terminate once no more inflation possible
        old_sum = np.sum(updated_radii)
        for i in range(len(centers)):
            x, y = centers[i]
            r = updated_radii[i]
            # Wall gap
            g_wall = min(x, 1 - x, y, 1 - y)
            # Neighbour gaps
            g_nei = float('inf')
            for j in range(len(centers)):
                if i == j:
                    continue
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                gap = dist - (r + radii[j])
                if gap < g_nei:
                    g_nei = gap
            g = min(g_wall, g_nei)
            if g > 0.0:
                new_r = r + beta * g
                # Also constrained by what would fit with local geometry
                max_feas = max_feasible_radius(i, centers, updated_radii)
                new_r = min(new_r, max_feas)
                updated_radii[i] = new_r
        new_sum = np.sum(updated_radii)
        if abs(new_sum - old_sum) < 1e-10:
            break
    return updated_radii

def polish_radii(centers, radii, beta=0.6, passes=3):
    """Final polishing pass to properly inflate all local gaps."""
    returned_radii = radii.copy()
    for _ in range(passes):
        returned_radii = polished_radius_inflation(centers, returned_radii, beta)
        # Post-process for overlaps
        changed = True
        while changed:
            changed = False
            overlaps = []
            for i in range(len(centers)):
                for j in range(i + 1, len(centers)):
                    dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                    overlap = dist - (returned_radii[i] + returned_radii[j])
                    if overlap < 0:
                        overlaps.append((i, j, overlap))
            if not overlaps:
                break
            for i, j, overlap in overlaps:
                if overlap < 0:
                    change = -overlap / 2.0
                    returned_radii[i] -= change
                    returned_radii[j] -= change
                    changed = True
    return returned_radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Set deterministic seed
    np.random.seed(42)
    random.seed(42)
    # Parameters
    n = 26
    margin = 0.03
    alpha_start = 0.04
    temp0 = 0.02
    gamma = 0.97
    sigma0 = 0.03
    steps_per_block = 200
    max_inflation_stagnation = 8
    max_cycles = 5
    # 1. Initial seed
    centers, radii = make_hex_seed(n, margin)
    # Initial sum
    current_sum = np.sum(radii)
    print(f"Initial sum: {current_sum:.6f}")
    # 2. Inflation
    alpha = alpha_start
    stagnation_count = 0
    # Main mixed loop
    cycle = 0
    while cycle < max_cycles:
        last_sum = current_sum
        inflation_round = 0
        # Inflation loop
        while inflation_round < 100:  # Avoid infinite loops
            new_radii, delta = inflate_radii(centers, radii, alpha)
            current_sum += delta
            radii = new_radii
            if abs(delta) < 1e-6:
                stagnation_count += 1
            else:
                stagnation_count = 0
            inflation_round += 1
            if stagnation_count >= max_inflation_stagnation:
                break
            alpha *= 0.95  # Decay alpha
        # If stagnated or hit cycle limit, break
        if stagnation_count >= max_inflation_stagnation:
            print(f"Entering SA phase after {cycle} full cycles, sum: {current_sum:.6f}")
            # Run one block of SA on centers
            centers, radii, new_sum = anneal_centers(centers, radii, temp0, gamma, sigma0, steps_per_block)
            current_sum = new_sum
            cycle += 1
            if cycle >= max_cycles:
                break
            # Re-inflate after SA
            alpha = alpha_start  # Reset inflation parameter
            stagnation_count = 0
        # Else, we are still in inflation phase, continue
        if stagnation_count < max_inflation_stagnation:
            if current_sum < last_sum:
                break
            continue
    # Final polishing
    final_radii = polish_radii(centers, radii)
    final_sum = np.sum(final_radii)
    # Validate result
    valid, msg = validate_packing(centers, final_radii)
    if not valid:
        raise RuntimeError("Validation failed after polishing: " + msg)
    return centers, final_radii, final_sum
