import numpy as np
from scipy.optimize import minimize

def hex_lattice_points(n, d):
    """Generate at least n points in a hexagonal lattice."""
    # Hexagonal lattice with spacing d
    sqrt3 = np.sqrt(3)
    rows = []
    y = 0.0
    row = 0
    while len(rows) < n:
        if row % 2 == 0:
            x_start = 0.0
        else:
            x_start = d / 2.0
        x = x_start
        while x <= 1.0 + d:
            rows.append((x, y))
            x += d
        y += sqrt3 * d / 2.0
        row += 1
    return np.array(rows[:n])

        radii.append(r)
    
    # Use the smallest wall distance as the uniform radius
    r0 = min(radii)

    vars = []

    return np.array(vars)

def objective(vars):
    """Minimize negative sum of radii."""
    n = len(vars) // 3
    r = vars[2::3]
    return -np.sum(r)

def wall_constraints(vars):
    """Return a list of constraint values for wall bounds."""
    n = len(vars) // 3
    constraints = []
    for i in range(n):
        x = vars[3*i]
        y = vars[3*i+1]
        r = vars[3*i+2]
        # x - r >= 0
        constraints.append(r - x)
        # x + r <= 1
        constraints.append(x + r - 1)
        # y - r >= 0
        constraints.append(r - y)
        # y + r <= 1
        constraints.append(y + r - 1)
    return constraints

def nonoverlap_constraints(vars):
    """Return a list of constraint values for non-overlapping."""
    n = len(vars) // 3
    constraints = []
    for i in range(n):
        for j in range(i+1, n):
            x_i = vars[3*i]
            y_i = vars[3*i+1]
            r_i = vars[3*i+2]
            x_j = vars[3*j]
            y_j = vars[3*j+1]
            r_j = vars[3*j+2]
            
            dx = x_i - x_j
            dy = y_i - y_j
            dist_sq = dx*dx + dy*dy
            radii_sum = r_i + r_j

            # We return -(dist^2 - (r_i + r_j)^2)
            # So it's <= 0 when it becomes active
            # We also impose a small tolerance
            constraints.append(-(dist_sq - radii_sum * radii_sum))
    return constraints
