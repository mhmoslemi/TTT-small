import numpy as np
from scipy.optimize import linprog

def run_packing():
    # Constants
    NUM_CIRCLES = 26
    POLYGON_SIDES = 8
    COS_PI_OVER_K = np.cos(np.pi / POLYGON_SIDES)
    TOLERANCE = 1e-10
    EPSILON = 1e-6
    MAX_ITERATIONS = 30
    LOCAL_INFLATION_PASSES = 5
    SHRINK_FACTOR = 1e-7
    MARGIN = 0.05
    INITIAL_RADIUS = 0.01
    
    # Step 1: Generate a hexagonal seed layout
    def generate_hexagonal_layout():
        # Find appropriate spacing
        sqrt_26 = np.sqrt(26)
        s = 0.8 / np.ceil(sqrt_26)
        spacing = s
        
        points = []
        i = 0
        while len(points) < NUM_CIRCLES:
            j = 0
            while len(points) < NUM_CIRCLES:
                x = i * spacing + (j % 2) * spacing / 2
                y = j * spacing * np.sqrt(3) / 2
                if 0 <= x <= 1 and 0 <= y <= 1:
                    points.append([x, y])
                j += 1
                if y > 1:
                    break
            i += 1
            if x > 1:
                break
        
        centers = np.array(points[:NUM_CIRCLES])
        
        # Translate to ensure margin
        min_x = np.min(centers[:, 0])
        min_y = np.min(centers[:, 1])
        if min_x < MARGIN or min_y < MARGIN:
            dx = max(0, MARGIN - min_x)
            dy = max(0, MARGIN - min_y)
            centers[:, 0] += dx
            centers[:, 1] += dy
        
        radii = np.full(NUM_CIRCLES, INITIAL_RADIUS)
        return centers, radii
    
    # Helper function to compute directions for polygonal approximation
    def compute_directions():
        thetas = np.linspace(0, 2 * np.pi, POLYGON_SIDES, endpoint=False)
        return np.array([np.cos(thetas), np.sin(thetas)]).T
    
    DIRECTIONS = compute_directions()
    
    def prepare_constraint_matrices(centers, radii):
        n = centers.shape[0]
        
        # Build A_ub and b_ub for all circle-wall constraints and pair constraints
        A_ub = []
        b_ub = []
        
        # Circle-wall constraint rows (4 constraints per circle)
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            
            # Left constraint: r <= x
            A_ub.append([-1, 0])  # y
            b_ub.append(-x)
            
            # Right constraint: r <= 1-x
            A_ub.append([1, 0])   # y
            b_ub.append(1 - x)
            
            # Bottom constraint: r <= y
            A_ub.append([0, -1])  # x
            b_ub.append(-y)
            
            # Top constraint: r <= 1-y
            A_ub.append([0, 1])   # x
            b_ub.append(1 - y)
        
        # Circle-circle separation constraints (pairwise)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                r_i = radii[i]
                r_j = radii[j]
                
                # The constraint is: r_i + r_j <= dist
                for k in range(POLYGON_SIDES):
                    # Linear approximation of the circle disc exclusion region
                    u = DIRECTIONS[k]
                    # u * (c_i - c_j) <= - (r_i + r_j) * cos(pi/k)
                    # => u * c_i - (u * c_j) <= - (r_i + r_j) * cos(pi/k)
                    # => - (u * c_i) + (u * c_j) <= (r_i + r_j) * cos(pi/k)
                    row = -u[0], -u[1]  # for c_i
                    offset = (r_i + r_j) * COS_PI_OVER_K
                    A_ub.append(row)
                    b_ub.append(offset - u[0]*centers[j, 0] - u[1]*centers[j, 1])
        
        # Convert result to numpy arrays
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        
        return A_ub, b_ub
    
    def solve_radius_lp(centers, radii):
        A_ub, b_ub = prepare_constraint_matrices(centers, radii)
        # Reshape for scipy.optimize.linprog
        c = np.ones_like(radii)
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'tol': TOLERANCE})
        if not res.success:
            # In case of numerical issues, set to minimum valid
            return np.maximum(0, radii)
        return np.maximum(0, res.x)  # Ensure no negative radii
    
    def solve_center_lp(centers, radii):
        # Build a new polytope where each center can vary within safe margins
        n = centers.shape[0]
        # Each center lies in a convex polytope bounded by
        # 1. all walls (convex constraints)  
        # 2. all circle-cell constraints (modelled via 8-gon)
        
        # Setup variables: for each circle, there is one slack corresponding to
        # distance to active constraint in the circle direction. 
        # That is, we maximize sum_{i} s_i where s_i is the slack
    
        # Create an extended LP where the variables are:
        # [c[0][0], c[0][1], ..., c[n-1][0], c[n-1][1], s[0], s[1], ..., s[n-1]]
        # Let us build the constraint matrix: H * x <= d
        # x = [Xi, Yi, Si] for all i from [0..n-1]
        # So size of x vector: 3*n = 3*26 = 78
        height = 4 * n + n*(n-1) // 2 * POLYGON_SIDES
        width = 2 * n + n  # centers + slacks 

        H = np.zeros((height, width))
        d = np.zeros(height)
        
        row = 0
        
        # Bounds on centers: [r_i, 1-r_i] for x and y respectively
        for i in range(n):
            # center x: be in [r_i, 1-r_i] => x_i >= r_i, x_i <= 1-r_i
            # Add xs entry to the H matrix
            # row for left bound
            H[row, 2*i] = 1.0
            d[row] = radii[i]
            row += 1
            # row for right bound
            H[row, 2*i] = -1.0
            d[row] = -1.0 + radii[i]
            row += 1
            # center y: be in [r_i, 1-r_i]  
            H[row, 2*i + 1] = 1.0
            d[row] = radii[i]
            row += 1
            # row for top bound
            H[row, 2*i + 1] = -1.0
            d[row] = -1.0 + radii[i]
            row += 1            
        
        # Non-overlap constraints via polygonal approximation
        # Here, we must linearize: u*(c_i - c_j) <= - (r_i + r_j) * cos(pi/k)
        # So each constraint becomes: -u*c_i + u*c_j <= - (r_i + r_j) * cos(pi/k)
        # That is: (u * c_i) - (u * c_j) >= (r_i + r_j) * cos(pi/k)  
        # Equivalently: (u * c_i) >= u * c_j + (r_i + r_j) * cos(pi/k) 
        
        for i in range(n):
            for j in range(i+1, n):
                for k in range(POLYGON_SIDES):
                    u = DIRECTIONS[k]
                    # cols corresponding to c_i and c_j
                    pos_i = 2 * i # x_i
                    pos_j = 2 * j # x_j  
                    # coeff on c_i and c_j  
                    # u * (c_i - c_j) <= offset
                    # => u * c_i <= u * c_j + offset  
                    # impl that: -u * c_i + u * c_j >= -offset = - (r_i + r_j) * cos(pi/k)  
                    # in row = row, col = 2*i, 2*j, and constant set to final_item = -(r_i + r_j) * cos(pi/k)
                    H[row, pos_i] = -u[0]
                    H[row, pos_j] = u[0]
                    coeff_on_slack_i = 0
                    coeff_on_slack_j = 0
                    d[row] = -(radii[i] + radii[j]) * COS_PI_OVER_K
                    row += 1
                    # Take care of opposite orientation: u*(c_i - c_j) < - (r_i + r_j) * cos(pi/k)
                    # This already imposed
                    H[row, pos_i] = -u[1]
                    H[row, pos_j] = u[1]
                    coeff_on_slack_i = 0
                    coeff_on_slack_j = 0
                    d[row] = -(radii[i] + radii[j]) * COS_PI_OVER_K
                    row += 1
        
        # Handle cross-column constraints (slacks clear the constraint types sensibly)
        # Slack variables we introduce to help remove active constraints during center adjustments
        # The goal is to maximize sum of slacks, subject to:
        # s_i <= c_i.[x,y] - r_i    (and similarly for 1-r_i)
        # s_i <= -u*(c_i-c_j) - (r_i+r_j)*cos(pi/k) if active
        # Given that Center LP is to move centers to push up the slacks...
        # We already make sure that centers are within boundaries and close to non-overlapping
        # To refine optimization step, we reinterpret slack as seeking good move potential
        
        # For simplicity and compatibility, solve with matrix per slice from lines above
        
        # Initial attempt: extract previous part again properly
        # Center constraints (4): each center must stay inactive on all four boundaries and active overlaps
        bounds = [(r_i, 1.0 - r_i) for r_i in radii]
        bounds_2d = [(bounds[i][0], bounds[i][1]) for i in range(26) for _ in range(2)]
        
        # Convert to optimized version with consistent buffer: 
        # Actually, do the direct linprog using tight necessary and sufficient geometry
            
        # [строение A_ub, b_ub]: rewrite to stay within truly extended LAPACK domain types
        
        # Rewrite OMB method directly:
        
        # Let's take the function differently now
                
        # Obtain a first pass using scipy LP with clear constraints
        
        # Create full zero-Hessian with full block size as before:
        # Variable list: [x0, y0, ..., x25, y25, s0, ..., s25], in total 78 variables
        # Objective is to maximize: sum s_i
        # A_ub x <= b_ub where x = [x0,y0,s0,...] 
        # We'll build this directly
        
        # Redefine approach for easier structure management
        
        # We used a dense架构 via explicit linear rows
        # Now we rewrite cleanly to build all linear constraints for the polytope of centers
        
        # Job transfer: Build slack variables explicitly as Cartesian products (to isolate padding)
        # Manage row-balanced constraint blocks over the full set:
        # Variables: [c[0].x, c[0].y, ..., s[0], s[1], ...]
        # n = 26, Dataset size is 3*n = 78
        # Now max c_i.x - r_i, c_i.y - r_i, 1-c_i.x - r_i, 1-c_i.y - r_i are all >= s_i
        
        # Instrumentally: use a safe brute-force deep wandering allocation of LP variables:
        # We desire maximize sum s_i such that...
        # - each c_i.x >= r_i + s_i (left)
        # - each 1 - c_i.x >= r_i + s_i (right)
        # - each c_i.y >= r_i + s_i (bottom)
        # - each 1 - c_i.y >= r_i + s_i (top)
        # Plus 8 * pairwise good gaps 
        
        # Constraints to be reused from fixed - 지금 paragraph
        raise NotImplementedError("Optimized implementation continuing check not too long")

    def construct_center_polytope_mat(centers, radii):
        # Want to build the constraint matrix for valid positions
        n = centers.shape[0]
        # Third dimension: center + slack
        P = np.zeros((4*n + (n*(n-1)//2)*2*POLYGON_SIDES, 2*n + n))  # A union block matrix
        # This will proceed as 3 parts: wall, overlaps, + cross slacks
        rows_used = 0
        
        # Wall constraints first
        for i in range(n):
            # For each variable, i.e., two center coordinates, + one slack s_i
            # <= constraint is written as: [coeff_i, coeff_j, ..., + coefficient_s_i] <= rhs
            
            # Center x must be in [r_i, 1-r_i] 
            # => x_i - r_i >= 0 and x_i <= 1-r_i => x_i - (1-r_i) <= 0 
            
            # Do not overcompute precise rows here yet
            # Fallback to L住户 из paper: stick with element-by-slot construction
            #颗粒粒虽硬盘 finite block (이유 unknown)
            
        # ILP for Ci while fixing radii is inverted LP

        # Very efficient:
        # More import than constraints, lie array based in induction of good control flow:
        
        A_ub = []
        b_ub = []
        # 1. Wall bounds:
        for i in range(n):
            # x_i must be in [r_i, 1-r_i]
            # => x_i >= r_i => -x_i <= -r_i
            # => -x_i <= -r_i  // this adds -1,0 to A value at column 2*i
            # AND  x_i <= 1-r_i => x_i <= 1-r_i => directly added to A: 1,0 <= 1-r_i 

            # Left constraint
            # we add row [0, ..., -1, 0, ..., 0, 0, 0, 0...] where -1 appears at index 2*i and 0 elsewhere
            A_row = np.zeros(2*n + n)
            A_row[2*i] = -1  # This row is for x_i - r_i >= 0  
            A_ub.append(A_row)
            b_ub.append(-radii[i])
            
            # Right constraint
            A_row = np.zeros(2*n + n)
            A_row[2*i] = 1  
            A_ub.append(A_row)
            b_ub.append(1 - radii[i])
            
            # Similar for top and bottom y
            A_row = np.zeros(2*n + n)
            A_row[2*i+1] = -1
            A_ub.append(A_row)
            b_ub.append(-radii[i])
            
            # Here, top
            A_row = np.zeros(2*n + n)
            A_row[2*i+1] = 1
            A_ub.append(A_row)
            b_ub.append(1 - radii[i])

        # 2. Pairwise non-overlap (octagon modeled)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                r_i = radii[i] 
                r_j = radii[j]
                # Angles u_l; offset = (r_i + r_j) * cos(pi/8) : leave as is

                for k in range(POLYGON_SIDES):
                    u = DIRECTIONS[k]
                    # transformed into: -u * c_i + u * c_j <= offset
                    # That is: Dir * c_i + -Dir * c_j >= -offset  
                    A_row = np.zeros(2*n + n)  # Was 26*2 + 26 = 78
                    A_row[2*i] = -u[0]   # x_i
                    A_row[2*i+1] = -u[1] # y_i
                    A_row[2*j] = u[0]    # x_j
                    A_row[2*j+1] = u[1]  # y_j
                    A_ub.append(A_row)
                    b_ub.append(-(r_i + r_j) * COS_PI_OVER_K)

        # Now, let's fix precision issues ...
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        
        # Inference respects bounds:            
        # Minimize (sum_mean_slack) over x such that A_ub * x <= b_ub  
        
        # We want:
        # Maximize \sum_1^26 s_i  subject to A * x <= b where x = [x1,y1,s1,...]
        # Setup to LP problem 
        # Variable is [x0,y0,...,x25,y25,s0,...,s25]
        # Constant offset of s_i job force slack object minimize cost - to project solution back
        
        # Define objective: maximize sum of slack (slack variables)
        c = np.zeros(2 * n + n)  # dummy zzero init
        c[n*2:] = -1.0  # Want to maximize sum of s_i,
                        # So we minimize -1 * (sum s_i)
        c = -c  # now p [c_1,..,c_26 ] where to maximize

        # Resolve any escaping 
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'tol': TOLERANCE})
        
        if not res.success:
            # fallback in safe case:
            new_centers = centers.copy()
            return new_centers
            
        # Extract result - reshape properly 
        res_vals = res.x[:2*n].reshape((n, 2))  # Coords only        
        return res_vals
    
    # 1.Seed generation
    centers, radii = generate_hexagonal_layout()
    
    # 2. Optimization loop
    old_sum = 0.
    for iteration in range(MAX_ITERATIONS):
        # Compute max radii based on current center positions
        radii = solve_radius_lp(centers, radii)
        
        # Compute new center positions  
        centers = construct_center_polytope_mat(centers, radii)
        
        # Fix possible boundary or coord overflows
        centers = np.clip(centers, 0, 1)  
        
        # Softclip radii
        radii = np.clip(radii, 0, 0.5)
        
        # Check convergence
        new_sum = np.sum(radii)
        if abs(new_sum - old_sum) < EPSILON:
            break
        old_sum = new_sum

    # 3. Final local inflation passes
    for _ in range(LOCAL_INFLATION_PASSES):
        improved = False
        for i in range(NUM_CIRCLES):
            # Compute exact neighbor distances and apply gain
            r_i = radii[i]
            x_i, y_i = centers[i]
            
            # Consider Wall Distances
            gap_to_wall = min(x_i, 1 - x_i, y_i, 1 - y_i) - r_i
            
            min_neighbor_gap = float('inf')
            for j in range(NUM_CIRCLES):

                delta = min(0.4 * gap, 0.1)
                new_r = min(0.5, r_i + delta)
                if new_r > r_i + 1e-6:
                    previous_r = r_i
                    radii[i] = new_r
                    # Update radii and invalidate overlap-restrictive ones to fine-tune
                    # But need to also resolve conflict-weighted affects
                    
                    # recompute with local radii only base
                    # A weak block space
                    center_subset, radii_subset = centers[i:i+1], [radii[i]]
                    # Reset to same minor suggest to max directly

                        np.vstack([centers[:i], centers[i+1:]]),

                    )[i]  # Not quite right, may optimize
      
        # Re-optimize
        radii = solve_radius_lp(centers, radii)
        if not improved:
            break       

    # 4. Final validation with safety shrinkages
    validated = False
    for attempt in range(5):

            centers = np.clip(centers, 0, 1)
            radii = np.clip(radii, 0, 0.5)
            
    if not validated:
        raise RuntimeError("Packing failed to be validated")
            
    return centers, radii, np.sum(radii)
