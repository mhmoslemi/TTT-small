import numpy as np
import cvxpy as cp
from scipy.spatial.distance import cdist

def build_leaf_problem(cell, k, max_radius=0.5):
    """For a given rectangle cell and number of circles k, build an SOCP."""
    x_min, y_min, x_max, y_max = cell
    width = x_max - x_min
    height = y_max - y_min
    
    # Define variables for centers and radii of k circles
    centers = cp.Variable((k, 2))
    radii = cp.Variable(k)
    
    # Constraints: Circle must be inside the cell
    constraints = []
    constraints.append(centers[:, 0] >= x_min + radii)
    constraints.append(centers[:, 0] <= x_max - radii)
    constraints.append(centers[:, 1] >= y_min + radii)
    constraints.append(centers[:, 1] <= y_max - radii)
    
    # Constraints: Non-overlap between circles in this cell
    for i in range(k):
        for j in range(i+1, k):
            dist = cp.norm(centers[i] - centers[j])
            constraints.append(dist >= radii[i] + radii[j])
    
    # Objective: maximize sum of radii
    objective = cp.Maximize(cp.sum(radii))
    
    # Solve the SOCP
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, verbose=False, abstol=1e-12, reltol=1e-12)
    
    if prob.status != cp.OPTIMAL:
        return None, None, None
        
    c = centers.value
    r = radii.value
    
    return c, r, constraints

def get_cell_indices(n_circles_by_cell, num_cells=9):
    """Generate cell assignment index lists."""
    indices = []
    start = 0
    for n in n_circles_by_cell:
        indices.append(list(range(start, start + n)))
        start += n
    return indices

def perturb_for_spacing(centers, radii, target_sum):
    """
    Gradient ascent with barrier penalties.
    Start from initial positions, refine using barrier gradient ascent.
    """
    num_circles = len(radii)
    x = centers[:, 0].copy()
    y = centers[:, 1].copy()
    r = radii.copy()

    mu = 1e-4
    max_iter = 50
    step_size = 0.01
    
    for _ in range(max_iter):
        # Compute gradients based on barrier function
        # ∇ variable: ~∂/∂x_i, ∂/∂y_i, ∂/∂r_i of total objective
        grad_x = np.zeros_like(x)
        grad_y = np.zeros_like(y)
        grad_r = np.ones_like(r) * -1.0

        # Form distances
        dist_matrix = cdist(np.column_stack([x, y]), np.column_stack([x, y]))

        # Add penalty terms to gradients (barrier approach)
        for i in range(num_circles):
            # Wall penalties:
            grad_x[i] -= mu * (1/(x[i] - r[i]) - 1/(1 - x[i] - r[i]))
            grad_y[i] -= mu * (1/(y[i] - r[i]) - 1/(1 - y[i] - r[i]))
            
            # Pairwise overlaps contribution:
            for j in range(num_circles):
                if i == j:
                    continue
                dx = x[i] - x[j]
                dy = y[i] - y[j]
                dist = np.sqrt(dx*dx + dy*dy)

                if dist < r[i] + r[j]:
                    # Avoid division by zero on near-touching
                    safe_dist = max(dist, 1e-10)
                    dist_grad = 1.0 / safe_dist
                    # Force direction toward preventing overlap
                    grad_x[i] += -mu * (dx / safe_dist) / (dist - r[i] - r[j] + 1e-10)
                    grad_y[i] += -mu * (dy / safe_dist) / (dist - r[i] - r[j] + 1e-10)
                    grad_r[i] += mu / (r[i] + r[j] - dist + 1e-10)
                    
                    # Also clip to theoretical minimum for consistency
                    if r[i] > dist - r[j] - 1e-10:
                        r[i] = dist - r[j] - 1e-10
                        r[i] = max(r[i], 1e-8)

        # Step update using gradient ascent
        x += step_size * grad_x
        y += step_size * grad_y
        r += step_size * grad_r

        # Enforce boundary conditions
        x = np.clip(x, r, 1 - r)
        y = np.clip(y, r, 1 - r)
        r = np.maximum(r, 1e-12)

    # Clamp final radii so sum doesn't overshoot target after refinement
    current_sum = np.sum(r)
    if current_sum > target_sum:
        # Linear adjustment to get sum to target
        scaling = target_sum / current_sum
        r *= scaling

    # Return updated centers and radii
    updated_centers = np.column_stack([x, y])
    return updated_centers, r
