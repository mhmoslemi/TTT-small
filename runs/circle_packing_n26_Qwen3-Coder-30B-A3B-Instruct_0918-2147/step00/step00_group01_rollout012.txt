import numpy as np
from scipy.optimize import differential_evolution
from scipy.spatial.distance import cdist

def compute_radii(centers):
    # Compute wall distances
    x, y = centers.T
    wall_distances = np.array([
        x, 1 - x, y, 1 - y
    ]).min(axis=0)
    
    # Compute pairwise distances
    distances = cdist(centers, centers)
    np.fill_diagonal(distances, np.inf)
    
    # Compute neighbour-based distances (half the minimum distance to other centers)
    neighbour_distances = 0.5 * distances.min(axis=1)
    
    # Compute radii as minimum of wall and neighbour distances
    radii = np.minimum(wall_distances, neighbour_distances)
    
    return radii

def neg_sum_radii(z):
    centers = z.reshape(-1, 2)
    radii = compute_radii(centers)
    return -np.sum(radii)

def run_packing():
    np.random.seed(42)
    
    # Define bounds for centers
    bounds = [(0, 1)] * 52
    
    # Differential Evolution to find good center configuration
    result = differential_evolution(
        neg_sum_radii,
        bounds,
        maxiter=2000,
        popsize=10,
        tol=1e-7,
        polish=False,
        strategy='best1bin',
        seed=42
    )
    
    # Extract centers from DE result
    best_centers = result.x.reshape(-1, 2)
    
    # Local refinement with Gaussian mutation
    current_centers = best_centers.copy()
    current_radii = compute_radii(current_centers)
    current_sum = np.sum(current_radii)
    
    sigma = 0.02
    min_sigma = 1e-5
    max_sigma = 0.1
    k_max = 3000
    
    for k in range(k_max):
        # Generate mutation
        mutation = np.random.normal(0, sigma, current_centers.shape)
        new_centers = current_centers + mutation
        new_centers = np.clip(new_centers, 0, 1)
        
        # Evaluate new solution
        new_radii = compute_radii(new_centers)
        new_sum = np.sum(new_radii)
        
        if new_sum > current_sum:
            current_centers = new_centers
            current_radii = new_radii
            current_sum = new_sum
            sigma = min(max_sigma, sigma * 1.1)
        else:
            sigma = max(min_sigma, sigma * 0.98)
    
    # Final computation of radii
    final_radii = compute_radii(current_centers)
    
    # Validation step
    valid, msg = validate_packing(current_centers, final_radii)
    if not valid:
        # If there's a validation issue, slightly shrink all radii
        shrink_factor = 1.0 - 1e-7
        final_radii = final_radii * shrink_factor
        valid, msg = validate_packing(current_centers, final_radii)
        if not valid:
            raise RuntimeError(f"Validation failed after shrinkage: {msg}")
    
    sum_radii = np.sum(final_radii)
    
    return current_centers, final_radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
