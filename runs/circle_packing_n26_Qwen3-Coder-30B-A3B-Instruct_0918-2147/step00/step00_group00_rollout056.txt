import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Set random seed for reproducibility
    np.random.seed(42)

    # Constants
    N_CIRCLES = 26
    K_GON = 8  # number of sides for polygonal approximation of disc
    POLYGONAL_SCALE = np.cos(np.pi / K_GON)  # factor to scale disc into polygon
    MAX_ITER = 30
    TOLERANCE = 1e-6
    LOCAL_INFLATION_PASSES = 5
    LOCAL_INFLATION_FACTOR = 0.4
    NUMERICAL_TOL = 1e-10

    # Step 1: Geometry-first layout (deterministic seed)
    square_side = 0.8
    spacing = square_side / np.ceil(np.sqrt(N_CIRCLES))
    sqrt3 = np.sqrt(3)

    # Generate triangular lattice points
    points = []
    i = 0
    while len(points) < N_CIRCLES:
        for j in range(int(np.ceil(N_CIRCLES / 2))):
            x = i * spacing + (j % 2) * spacing / 2
            y = j * spacing * sqrt3 / 2
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append((x, y))
            if len(points) >= N_CIRCLES:
                break
        i += 1

    # Collect first 26 points
    centers = np.array(points[:N_CIRCLES])
    
    # Shift by margin to avoid tight boundary conditions
    margin = 0.05
    centers[:, 0] += margin
    centers[:, 1] += margin

    # Initial radii
    radii = np.full(N_CIRCLES, 0.01, dtype=float)
    
    # Direction vectors for polygonal approximation
    thetas = np.linspace(0, 2*np.pi, K_GON, endpoint=False)
    unit_vectors = np.column_stack((np.cos(thetas), np.sin(thetas)))

    def compute_distance_matrix(centers):
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        return np.sqrt(np.sum(diff**2, axis=2))

    def solve_radius_lp(centers, radii):
        # Build LP constraint matrix A_ub * radii <= b_ub
        n = centers.shape[0]
        A_ub = []
        b_ub = []
        
        # Wall constraints: r_i <= min(c_i.x, 1-c_i.x, c_i.y, 1-c_i.y)
        for i in range(n):
            x, y = centers[i]
            r_max = min(x, 1 - x, y, 1 - y)
            A_ub.append([-1.0])  # -r_i <= -r_max (but we do r_i <= r_max, so we reverse sign)
            b_ub.append(r_max)
        
        dist_matrix = compute_distance_matrix(centers)
        
        # Separation constraints: r_i + r_j <= distance_ij
        for i in range(n):
            for j in range(i+1, n):
                A_row = np.zeros(n)
                A_row[i] = 1.0
                A_row[j] = 1.0
                A_ub.append(A_row)
                b_ub.append(dist_matrix[i, j])
                
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        
        c = np.ones(n)
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', 
                        options={'tol': NUMERICAL_TOL})
        
        if result.success:
            new_radii = np.maximum(0, result.x)
        else:
            new_radii = np.array(radii)
            
        return new_radii
    
    def solve_center_lp(centers, radii):
        # Define centers as variables
        n = centers.shape[0]
        
        # Collect all constraints and variables for LP
        # For each circle i, we define slack variable s_i
        # New variables: [c1[0], c1[1], ..., cn[0], cn[1], s1, ..., sn]
        # Variables = [x1, y1, ..., xn, yn, s1, ..., sn]
        
        # Constraint matrices for the linear program
        # For each circle, we have at most 2*n + 4 constraints:
        constr_count = 0  # actual number of constraints
        A_ub_rows = []
        b_ub_vals = []
        
        # Slacks for safety margin
        slack_indices = list(range(2*n, 2*n + n))
        
        # Wall constraints:
        for i in range(n):
            row = np.zeros(2*n + n)  # now [x1,y1,...,xn,yn,s1,...,sn]
            row[2*i] = 1  # x_i
            row[2*n + i] = -1  # -s_i
            A_ub_rows.append(row)
            b_ub_vals.append(radii[i])
            
            row = np.zeros(2*n + n)
            row[2*i+1] = 1  # y_i
            row[2*n + i] = -1  # -s_i
            A_ub_rows.append(row)
            b_ub_vals.append(radii[i])
            
            row = np.zeros(2*n + n)
            row[2*i] = -1  # -x_i
            row[2*n + i] = -1  # -s_i
            A_ub_rows.append(row)
            b_ub_vals.append(1 - radii[i] - 1)
            
            row = np.zeros(2*n + n)
            row[2*i+1] = -1  # -y_i
            row[2*n + i] = -1  # -s_i
            A_ub_rows.append(row)
            b_ub_vals.append(1 - radii[i] - 1)

        # Neighbour separation with polygonal approximation
                
        for i in range(n):
            for j in range(i + 1, n):
                # For each direction vector u_l, add constraint
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist > 0:
                    safety_dist = dist - (radii[i] + radii[j])
                    if safety_dist > 0.0001:  # Avoid degenerate cases or if already close
                    
                        # Perturb slightly to always be conservative
                        projected = (centers[i] - centers[j]) / dist
                        
                        for k in range(K_GON):
                            u = unit_vectors[k]
                            # constraint u^T * (c_i - c_j) <= -r_i - r_j * cos(π/k)
                            row = np.zeros(2*n + n)
                            row[2*i:2*i+2] = u
                            row[2*j:2*j+2] = -u
                            A_ub_rows.append(row)
                            b_ub_vals.append(- (radii[i] + radii[j]) * POLYGONAL_SCALE)
                            
        # Combination: A_ub*x <= b_ub
        A_ub = np.array(A_ub_rows)
        b_ub = np.array(b_ub_vals)
        
        # Objective: maximize sum of slacks
        c_obj = np.zeros(2*n + n)
        c_obj[2*n:] = 1  # Coefficients for slack variables
        
        result = linprog(-c_obj, A_ub=A_ub, b_ub=b_ub, method='highs',
                        options={'tol': NUMERICAL_TOL})

        if result.success:
            # Extract updated centers and slacks
            sol = result.x
            new_centers = sol[:2*n].reshape(n, 2)
            # Clip centers to [0,1] box
            new_centers = np.clip(new_centers, 0., 1.)
        else:
            # If solving fails, do not move centers
            new_centers = np.array(centers)
            
        return new_centers

    # Main alternating loop
    old_sum = 0.
    for _ in range(MAX_ITER):
        new_radii = solve_radius_lp(centers, radii)
        
        # Clip radii to [0, 0.5] to avoid oversized circles
        new_radii = np.clip(new_radii, 0.0, 0.5)

        # Compute the sum of radii
        rad_sum = np.sum(new_radii)
        if rad_sum - old_sum < TOLERANCE:
            break
        old_sum = rad_sum

        centers = solve_center_lp(centers, new_radii)
        
        radii = new_radii

    # Local inflation passes
    for _ in range(LOCAL_INFLATION_PASSES):
        # Exact distance matrix
        dist_matrix = compute_distance_matrix(centers)
        # For each circle, check slack in all directions
        new_radii = np.copy(radii)
        any_change = False

        for i in range(N_CIRCLES):
            current_r = new_radii[i]
            # Wall slacks
            left = centers[i][0] - current_r
            right = 1 - (centers[i][0] + current_r)
            bottom = centers[i][1] - current_r
            top = 1 - (centers[i][1] + current_r)
            wall_slack = min(left, right, bottom, top)
            
            # nearest neighbors
            min_dist = float('inf')
            for j in range(N_CIRCLES):
                if i != j:
                    d = dist_matrix[i, j]
                    if d < min_dist:
                        min_dist = d
                        
            new_r = current_r
            if min_dist > 0:
                gap = min_dist - current_r
                if gap > 1e-6:
                    add_r = LOCAL_INFLATION_FACTOR * gap
                    if add_r < 0.4 * wall_slack:
                        new_r += add_r
                        
            if abs(new_r - current_r) > 1e-5 and new_r <= 0.5:
                new_radii[i] = new_r
                any_change = True
                
        if not any_change:
            break
            
        # Re-solve the radius LP for only changed radii if needed
        radii = solve_radius_lp(centers, new_radii)  # This may be too aggressive
        
        # Force resolvability - we take a simple update since updating all radii might not be happy
        # Instead, use improved method:
        radii = new_radii

    # Clamp shapes after adjustment
    radii = np.clip(radii, 0.0, 0.5)
    centers = np.clip(centers, 0.0, 1.0)
    
    # Final validation with provided validation function
    valid, msg = validate_packing(centers, radii)
    if not valid:
        # Try shrinking all radii a bit if validation fails
        factor = 1.0 - 1e-7
        for _ in range(5):
            adjusted_radii = factor * radii
            validnext, msg = validate_packing(centers, adjusted_radii)
            if validnext:
                radii = adjusted_radii
                break
            factor -= 1e-8

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
