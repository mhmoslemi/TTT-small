import numpy as np
from scipy.optimize import linprog
from math import sqrt, floor, ceil

def initial_hex_centers(k, spacing):
    sqrt3 = sqrt(3)
    centers = []
    i = 0
    j = 0
    while len(centers) < k:
        x = i * spacing + (j % 2) * spacing / 2
        y = j * spacing * sqrt3 / 2
        if 0 <= x <= 1 and 0 <= y <= 1:
            centers.append([x, y])
        if x > 1:
            i = 0
            j += 1
        else:
            i += 1
    return np.array(centers[:k])

def initial_radii(k, r0):
    return np.full(k, r0)

def build_lp_constants(centers, radii):
    n = centers.shape[0]
    d_left = centers[:, 0]
    d_right = 1 - centers[:, 0]
    d_bottom = centers[:, 1]
    d_top = 1 - centers[:, 1]
    d_walls = np.stack([d_left, d_right, d_bottom, d_top], axis=1)
    d_walls_min = np.min(d_walls, axis=1)
    
    # Compute pairwise centre distances
    D = np.zeros((n, n))
    for i in range(n):
        for j in range(i+1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            D[i, j] = D[j, i] = dist
    
    return d_walls_min, D

def solve_radius_lp(consts, radii, graph, alpha=0.5):
    d_walls_min, D = consts
    n = len(radii)
    c = np.ones(n)
    
    # Construct A_ub and b_ub
    A_ub = []
    b_ub = []
    
    # Wall capacity constraints: Δr_i <= min(d_i_left, ...)
    for i in range(n):
        max_delta = d_walls_min[i] - radii[i]
        if max_delta < 0:
            max_delta = 0
        A_ub.append([-1.0 if j == i else 0.0 for j in range(n)])
        b_ub.append(-max_delta)
    
    # Pairwise capacity constraints: Δr_i + Δr_j <= D_ij - (r_i + r_j)
    for i in range(n):
        for j in range(i+1, n):
            val = D[i, j] - (radii[i] + radii[j])
            if val > 0:
                A_row = [0.0] * n
                A_row[i] = 1.0
                A_row[j] = 1.0
                A_ub.append(A_row)
                b_ub.append(val)
    
    # Graph-based transfer limits
    for i in range(n):
        neighbors = graph.get(i, [])
        if neighbors:
            tot_adjacent = sum(D[i, j] - (radii[i] + radii[j]) for j in neighbors)
            if tot_adjacent > 0:
                A_row = [0.0] * n
                A_row[i] = -1.0
                for j in neighbors:
                    A_row[j] = alpha
                A_ub.append(A_row)
                b_ub.append(0.0)
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'presolve': True})
    return res.x if res.success else np.zeros(n)

def apply_increment(radii, delta_radii):
    return radii + delta_radii

def push_feasibility(centers, radii):
    n = centers.shape[0]
    
    # Wall push
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            radii[i] = x
        if x + r > 1:
            radii[i] = 1 - x
        if y - r < 0:
            radii[i] = y
        if y + r > 1:
            radii[i] = 1 - y
    
    # Pairwise push
    new_edges = []
    for i in range(n):
        for j in range(i+1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            r_sum = radii[i] + radii[j]
            if dist < r_sum:
                overlap = r_sum - dist
                unit_vec = (centers[i] - centers[j]) / dist
                move_amount = overlap / 2.0
                centers[i] += -move_amount * unit_vec
                centers[j] += move_amount * unit_vec
                
                # Check wall boundaries
                for k, c in enumerate([centers[i], centers[j]]):
                    if c[0] - radii[i+k] < 0 or c[0] + radii[i+k] > 1:
                        # Clamp to wall
                        radii[i+k] = min(c[0], 1 - c[0])
                    if c[1] - radii[i+k] < 0 or c[1] + radii[i+k] > 1:
                        radii[i+k] = min(c[1], 1 - c[1])
                new_edges.append((i, j))
    
    return centers, radii, new_edges

def update_graph(graph, new_edges, wall_contacts):
    for i, j in new_edges:
        graph.setdefault(i, set()).add(j)
        graph.setdefault(j, set()).add(i)
    return graph

def run_iterations(centers, radii, max_iter=200, epsilon=1e-5, alpha=0.5):
    graph = {}
    
    for _ in range(max_iter):
        # Consts for LP
        consts = build_lp_constants(centers, radii)
        
        # Compute radius change
        delta_r = solve_radius_lp(consts, radii, graph, alpha)
        
        # Exit if no change
        total_delta = np.sum(delta_r)
        if total_delta < epsilon:
            break
        
        # Update radii
        radii = apply_increment(radii, delta_r)
        
        # Push feasibility
        centers, radii, new_edges = push_feasibility(centers, radii)
        
        # Update graph
        graph = update_graph(graph, new_edges, {})
    
    return centers, radii

def final_refinement(centers, radii):
    consts = build_lp_constants(centers, radii)
    
    # High precision LP solve
    c = np.ones(len(radii))
    A_ub = []
    b_ub = []
    
    d_walls_min, D = consts
    for i in range(len(radii)):
        max_delta = d_walls_min[i] - radii[i]
        if max_delta < 0:
            max_delta = 0
        A_ub.append([-1.0 if j == i else 0.0 for j in range(len(radii))])
        b_ub.append(-max_delta)
    
    for i in range(len(radii)):
        for j in range(i+1, len(radii)):
            val = D[i, j] - (radii[i] + radii[j])
            if val > 0:
                A_row = [0.0] * len(radii)
                A_row[i] = 1.0
                A_row[j] = 1.0
                A_ub.append(A_row)
                b_ub.append(val)
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'presolve': True, 'primal_feasible': True, 'dual_feasible': True, 'time_limit': 30})
        if res.success:
            radii = apply_increment(radii, res.x)
    except:
        pass
    
    # Final clipping
    for i in range(len(radii)):
        r = radii[i]
        x, y = centers[i]
        r_clamped = min(r, x, 1-x, y, 1-y)
        if r_clamped < r:
            radii[i] = r_clamped
    
    # Final pairwise correction
    for i in range(len(radii)):
        for j in range(i+1, len(radii)):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            r_sum = radii[i] + radii[j]
            if dist < r_sum:
                overlap = r_sum - dist
                radii[i] -= overlap / 2 + 1e-13
                radii[j] -= overlap / 2 + 1e-13
    
    return centers, radii

def run_packing():
    # 1. Seed with hexagonal lattice
    center_positions = initial_hex_centers(26, spacing=0.12)
    initial_radius = 0.005
    radii = initial_radii(26, initial_radius)
    
    # 2. Run iterations
    final_centers, final_radii = run_iterations(center_positions, radii, max_iter=100, epsilon=1e-5, alpha=0.5)
    
    # 3. Final refinement
    final_centers, final_radii = final_refinement(final_centers, final_radii)
    
    # Compute sum
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
