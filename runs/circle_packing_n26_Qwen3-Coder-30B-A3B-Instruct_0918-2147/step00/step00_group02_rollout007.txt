import numpy as np
from scipy.stats.qmc import Sobol

def initialize_seeds(n):
    sobol = Sobol(d=2, seed=42)
    points = sobol.random(n)
    delta = 0.02
    return delta + (1 - 2 * delta) * points

def compute_feasible_radii(centers):
    n = centers.shape[0]
    # Compute pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_matrix = np.sqrt(np.sum(diff**2, axis=2))
    
    # Nearest neighbor distances (excluding self)
    nn_distances = np.partition(dist_matrix, 1, axis=1)[:, 1]
    
    # Wall distances
    wall_dist = np.min([centers[:, 0], 1 - centers[:, 0], centers[:, 1], 1 - centers[:, 1]], axis=0)
    
    # Feasible radii
    radii = np.minimum(wall_dist, 0.5 * nn_distances)
    return radii

def redistribute_radii(radii, slack, alpha):
    total_slack = np.sum(slack)
    if total_slack <= 1e-10:
        return radii.copy()
    
    new_radii = radii.copy()
    # Add to wall-limited circles
    wall_limited = slack > 1e-10
    wall_indices = np.where(wall_limited)[0]
    
    if len(wall_indices) == 0:
        return radii.copy()
    
    # Compute redistribution amount
    to_add = alpha * slack[wall_limited]
    total_add = np.sum(to_add)
    
    # Reduce from neighbor-limited circles
    neighbor_limited = ~wall_limited
    neighbor_indices = np.where(neighbor_limited)[0]
    
    if len(neighbor_indices) == 0:
        new_radii[wall_indices] += to_add
        return new_radii
    
    # Distribute reduction proportionally to current radii
    current_radii = radii[neighbor_indices]
    reduced_radii = np.maximum(0, current_radii - total_add * current_radii / np.sum(current_radii))
    reduction = current_radii - reduced_radii
    new_radii[neighbor_indices] = reduced_radii
    new_radii[wall_indices] += to_add
    
    return new_radii

def push_away(centers, radii, beta, eps):
    n = centers.shape[0]
    new_centers = centers.copy()
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_matrix = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(n):
            if i != j:
                distance = dist_matrix[i, j]
                overlap = radii[i] + radii[j] + eps - distance
                if overlap > 0:
                    v_ij = centers[i] - centers[j]
                    dir_vec = v_ij / (np.linalg.norm(v_ij) + eps)
                    delta = beta * overlap
                    new_centers[i] += delta * dir_vec
    
    # Clip to interior
    clip_eps = 1e-10
    new_centers = np.clip(new_centers, clip_eps, 1 - clip_eps)
    return new_centers

def iterate(centers, radii, alpha, beta, max_iter=200):
    for _ in range(max_iter):
        # Compute slack
        nn_distances = np.partition(np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :])**2, axis=2)), 1, axis=1)[:, 1]
        wall_dist = np.min([centers[:, 0], 1 - centers[:, 0], centers[:, 1], 1 - centers[:, 1]], axis=0)
        feasible_radii = np.minimum(wall_dist, 0.5 * nn_distances)
        slack = 0.5 * nn_distances - radii
        
        # Redistribute
        new_radii = redistribute_radii(radii, slack, alpha)
        
        # Push away
        new_centers = push_away(centers, new_radii, beta, 1e-8)
        
        # Recompute radii
        radii = compute_feasible_radii(new_centers)
        sum_radii = np.sum(radii)
        old_sum = np.sum(radii)
        
        if np.abs(sum_radii - old_sum) < 1e-5:
            break
        centers = new_centers
    
    return centers, radii

def repair(centers, radii, eps):
    # Shrink radii slightly
    radii = np.maximum(0, radii - eps)
    # Clamp centers
    centers = np.clip(centers, eps, 1 - eps)
    # Ensure no overlaps
    n = centers.shape[0]
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_matrix = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i + 1, n):
            if dist_matrix[i, j] < radii[i] + radii[j]:
                overlap = radii[i] + radii[j] - dist_matrix[i, j]
                # Reduce the larger one
                if radii[i] > radii[j]:
                    radii[i] -= overlap / 2 + eps
                else:
                    radii[j] -= overlap / 2 + eps
    return centers, radii

def run_packing():
    n = 26
    centers = initialize_seeds(n)
    radii = compute_feasible_radii(centers)
    alpha = 0.4
    beta = 0.3
    max_iter = 200
    centers, radii = iterate(centers, radii, alpha, beta, max_iter)
    eps_final = 5e-10
    centers, radii = repair(centers, radii, eps_final)
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
