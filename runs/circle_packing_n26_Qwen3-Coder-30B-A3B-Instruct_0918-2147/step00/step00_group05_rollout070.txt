import numpy as np
from scipy.optimize import minimize

def hex_lattice_points(n, d):
    """Generate at least n points in a hexagonal lattice."""
    rows = []
    y = 0.0
    while len(rows) < n:
        row = []
        x = 0.0
        while x <= 1.0:
            row.append((x, y))
            x += d
        rows.append(row)
        y += d * np.sqrt(3) / 2
        if len(rows) % 2 == 1:
            x = d / 2
        else:
            x = 0.0
    points = []
    for i, row in enumerate(rows):
        for x, y in row:
            if len(points) < n:
                points.append((x, y))
            else:
                break
        if len(points) >= n:
            break
    return np.array(points)

def initial_guess():
    """Generate initial configuration for 26 circles."""
    d = 0.2  # lattice spacing
    points = hex_lattice_points(26, d)
    # center them
    xs = points[:, 0]
    ys = points[:, 1]
    x_center = np.mean(xs)
    y_center = np.mean(ys)
    shift = np.array([0.5 - x_center, 0.5 - y_center])
    points += shift
    # compute max radius that fits nicely
    radii = []
    for i, (x, y) in enumerate(points):
        r_wall = min(x, 1-x, y, 1-y)
        r = min(r_wall, d/2)
        radii.append(r)
    # ensure all fit inside square (considering all points)
    max_r = min(np.min(radii), 0.5)
    radii = [max_r] * 26
    # pack into variables in order [x0, y0, r0, x1, y1, r1, ...]
    vars = []
    for i in range(26):
        vars.extend([points[i, 0], points[i, 1], radii[i]])
    return np.array(vars)

def objective(vars):
    # maximize sum of radii -> minimize negative sum
    return -np.sum(vars[2::3])

def wall_constraints(vars):
    # Wall constraints: 4 per circle (x ± r >= 0, x ± r <= 1)
    cons = []
    for i in range(26):
        x = vars[3*i]
        y = vars[3*i+1]
        r = vars[3*i+2]
        # x: x - r >= 0   and   x + r <= 1
        cons.append(r - x)      # x - r <= 0
        cons.append(x + r - 1)  # x + r - 1 <= 0
        # y: y - r >= 0   and   y + r <= 1
        cons.append(r - y)      # y - r <= 0
        cons.append(y + r - 1)  # y + r - 1 <= 0
    return np.array(cons)

def nonoverlap_constraints(vars):
    # Non-overlap constraints: (x_i-x_j)^2 + (y_i-y_j)^2 >= (r_i+r_j)^2
    # We'll return -((x_i-x_j)^2 + (y_i-y_j)^2 - (r_i+r_j)^2)
    cons = []
    for i in range(26):
        for j in range(i+1, 26):
            x1 = vars[3*i]
            y1 = vars[3*i+1]
            r1 = vars[3*i+2]
            x2 = vars[3*j]
            y2 = vars[3*j+1]
            r2 = vars[3*j+2]
            dx = x1 - x2
            dy = y1 - y2
            dist_sq = dx*dx + dy*dy
            radii_sum = r1 + r2
            radii_sum_sq = radii_sum * radii_sum
            # Return negative of constraint value: - (dist_sq - radii_sum_sq)
            cons.append(-(dist_sq - radii_sum_sq))
    return np.array(cons)

def run_packing():
    # Initial guess
    init_vars = initial_guess()
    # Create bounds
    bounds = []
    for i in range(26):
        bounds.append((0.0, 1.0))     # x_i
        bounds.append((0.0, 1.0))     # y_i
        bounds.append((0.0, 0.5))     # r_i

    # Create constraints list
    constraints = [
        {'type': 'ineq', 'fun': wall_constraints},
        {'type': 'ineq', 'fun': nonoverlap_constraints}
    ]

    # Run optimization
    result = minimize(
        objective,
        init_vars,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 2000, 'ftol': 1e-10}
    )

    if not result.success:
        # return fallback
        init_vars = initial_guess()
        vars = init_vars

    else:
        vars = result.x

    # Build centers and radii arrays from optimized variable vector
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    for i in range(26):
        centers[i] = (vars[3*i], vars[3*i+1])
        radii[i] = vars[3*i+2]

    # Clip any tiny negative radii
    radii = np.clip(radii, 0.0, None)

    # Compute sum of radii
    sum_radii = np.sum(radii)

    # Verify feasibility
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise AssertionError(f"Generated packing invalid: {msg}")

    return centers, radii, sum_radii
