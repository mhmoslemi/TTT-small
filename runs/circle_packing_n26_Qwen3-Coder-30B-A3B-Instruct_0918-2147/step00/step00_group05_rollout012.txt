import numpy as np

def hex_lattice(d):
    """Generate hexagonal lattice points in [0,1] x [0,1]"""
    sqrt3 = np.sqrt(3)
    dx = d
    dy = sqrt3 * d / 2
    rows = []
    y = 0
    row_idx = 0
    while y <= 1:
        x = 0
        if row_idx % 2 == 1:
            x = dx / 2
        row = []
        while x <= 1:
            row.append([x, y])
            x += dx
        rows.extend(row)
        y += dy
        row_idx += 1
    return np.array(rows)

def add_border_circles(points, d):
    """Append 6 border circles (4 corners, 2 edge mid)"""
    r = d / 4
    border = [
        [r, r],                # bottom-left
        [1 - r, r],            # bottom-right
        [r, 1 - r],            # top-left
        [1 - r, 1 - r],        # top-right
        [0.5, r],              # middle-bottom
        [0.5, 1 - r]           # middle-top
    ]
    return np.vstack([points, border])

def initial_radii(points):
    """Compute uniform safe radius for all points"""
    n = points.shape[0]
    # Wall distances
    wall_distances = np.array([
        points[:, 0],           # x
        1 - points[:, 0],       # 1-x
        points[:, 1],           # y
        1 - points[:, 1]        # 1-y
    ]).min(axis=0)

    # Pairwise distances
    dist_mat = np.sqrt(
        np.sum((points[:, np.newaxis] - points[np.newaxis, :]) ** 2, axis=2)
    )
    np.fill_diagonal(dist_mat, np.inf)
    pairwise_distances = dist_mat.min(axis=1)
    # Safe initial radius is minimum of wall and pairwise distances / 2
    safe_radii = np.minimum(wall_distances, pairwise_distances / 2)
    return safe_radii.min()

def inflate_radii(centers, radii):
    """Inflate all radii to maximum feasible values."""
    n = centers.shape[0]
    # Compute distance matrix once
    dist_mat = np.sqrt(
        np.sum((centers[:, np.newaxis] - centers[np.newaxis, :]) ** 2, axis=2)
    )

    new_radii = np.copy(radii)
    # For each circle, compute max radius based on constraints
    for i in range(n):
        x, y = centers[i]
        # Wall limits
        wall_limits = [
            x,              # left wall
            1 - x,          # right wall
            y,              # bottom wall
            1 - y           # top wall
        ]
        # Pairwise distance limits
        pairwise_limits = []
        for j in range(n):
            if i != j:
                dist = dist_mat[i, j]
                max_radii = dist - radii[j]
                pairwise_limits.append(max_radii)
        
        # Compute the valid radius for this circle
        new_radius = min(wall_limits + pairwise_limits)
        
        if new_radius > radii[i] + 1e-6:
            new_radii[i] = new_radius
    return new_radii

def relax_centers(centers, radii, steps=10):
    """Apply repulsive forces to centers to improve packing."""
    n = centers.shape[0]
    # Force parameters
    kappa = 0.02
    lambda_wall = 0.01
    gamma = 0.02
    beta = 0.02

    new_centers = np.copy(centers)
    for step in range(steps):
        forces = np.zeros_like(centers)
        
        # Compute pairwise forces
        for i in range(n):
            for j in range(n):
                if i != j:
                    diff = new_centers[i] - new_centers[j]
                    dist = np.linalg.norm(diff)
                    if dist > 0:
                        f = kappa * max(0, beta - dist) * diff / dist
                        forces[i] += f
          
        # Wall forces
        for i in range(n):
            x, y = new_centers[i]
            r = radii[i]
            
            # x-direction forces
            fx = -lambda_wall * max(0, gamma - (x - r))  # push right
            fx += lambda_wall * max(0, gamma - (1 - x - r))  # push left
            
            # y-direction forces
            fy = -lambda_wall * max(0, gamma - (y - r))  # push up
            fy += lambda_wall * max(0, gamma - (1 - y - r))  # push down
            
            forces[i] += [fx, fy]

        # Update center positions
        alpha = 0.5  # damping
        new_centers += alpha * forces

        # Clamp centers to admissible intervals
        new_centers[:, 0] = np.clip(new_centers[:, 0], radii, 1 - radii)
        new_centers[:, 1] = np.clip(new_centers[:, 1], radii, 1 - radii)

    return new_centers

        # Only relax centers if at least one ratio increased
