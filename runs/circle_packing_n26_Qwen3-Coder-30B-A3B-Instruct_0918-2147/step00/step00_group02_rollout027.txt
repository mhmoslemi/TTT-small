import numpy as np
from scipy.stats.qmc import Sobol
from scipy.spatial.distance import cdist

def initialize_seeds(n):
    sobol = Sobol(d=2, seed=42)
    points = sobol.random(n)
    # Clip to interior with margin
    delta = 0.02
    points = delta + (1 - 2 * delta) * points
    return points

def compute_feasible_radii(centers):
    n = centers.shape[0]
    # Distance matrix
    D = cdist(centers, centers)
    # Nearest neighbor distances
    nn_dists = np.partition(D, kth=1, axis=1)[:, 1]
    # Wall distances
    wall_dists = np.minimum(centers[:, 0], 1 - centers[:, 0])
    wall_dists = np.minimum(wall_dists, centers[:, 1])
    wall_dists = np.minimum(wall_dists, 1 - centers[:, 1])
    # Feasible radius is min of wall dist and 0.5 * nn dist
    radii = np.minimum(wall_dists, 0.5 * nn_dists)
    return radii

def redistribute_radii(radii, slack, alpha):
    # Total slack
    B = np.sum(slack)
    if B <= 1e-15:
        return radii.copy()
    
    new_radii = radii.copy()
    # Find wall-limited circles (slack > 0)
    wall_limited = slack > 1e-15
    # Amount to increase in wall-limited
    increase = alpha * slack[wall_limited]
    # Total increase
    total_increase = np.sum(increase)
    # Reduce from neighbour-limited, proportionally
    neighbor_limited = ~wall_limited
    if np.any(neighbor_limited):
        current_radii = radii[neighbor_limited]
        total_current = np.sum(current_radii)
        if total_current > 1e-15:
            reduction = (current_radii / total_current) * total_increase
            new_radii[neighbor_limited] -= reduction
            # Clamp to avoid negatives
            new_radii[neighbor_limited] = np.maximum(0, new_radii[neighbor_limited])
    
    new_radii[wall_limited] += increase
    return new_radii

def push_away(centers, radii, beta, eps):
    n = centers.shape[0]
    new_centers = centers.copy()
    # For each circle, compute net displacement
    displacements = np.zeros_like(centers)
    D = cdist(centers, centers)
    for i in range(n):
        for j in range(n):
            if i == j:
                continue
            dist = D[i, j]
            if dist < radii[i] + radii[j] + eps:
                # Overlap - push away
                overlap = radii[i] + radii[j] + eps - dist
                direction = centers[i] - centers[j]
                norm_dir = np.linalg.norm(direction)
                if norm_dir > eps:
                    disp = beta * overlap * direction / norm_dir
                    displacements[i] += disp
    # Apply displacement
    new_centers += displacements
    # Clamp to [eps, 1-eps]
    new_centers = np.clip(new_centers, eps, 1 - eps)
    return new_centers

def iterate(centers, radii, alpha, beta, max_iter):
    for _ in range(max_iter):
        # Compute slack
        D = cdist(centers, centers)
        nn_dists = np.partition(D, kth=1, axis=1)[:, 1]
        wall_dists = np.minimum(centers[:, 0], 1 - centers[:, 0])
        wall_dists = np.minimum(wall_dists, centers[:, 1])
        wall_dists = np.minimum(wall_dists, 1 - centers[:, 1])
        feasible_radii = np.minimum(wall_dists, 0.5 * nn_dists)
        slack = 0.5 * nn_dists - radii
        slack = np.maximum(0, slack)

        # Redistribute
        new_radii = redistribute_radii(radii, slack, alpha)

        # Push away
        new_centers = push_away(centers, new_radii, beta, 1e-8)

        # Re-evaluate radii
        radii = compute_feasible_radii(new_centers)
        centers = new_centers

        # Check termination
        old_sum = np.sum(feasible_radii)
        new_sum = np.sum(radii)
        if abs(new_sum - old_sum) < 1e-5:
            break

    return centers, radii

def repair(centers, radii, eps_final):
    # Final shrink
    new_radii = np.maximum(0, radii - eps_final)
    # Clamp centers
    new_centers = np.clip(centers, eps_final, 1 - eps_final)
    # Final overlap check and repair
    D = cdist(new_centers, new_centers)
    eps_check = 1e-12
    for i in range(new_centers.shape[0]):
        for j in range(i+1, new_centers.shape[0]):
            if D[i, j] < new_radii[i] + new_radii[j] - eps_check:
                diff = (new_radii[i] + new_radii[j] - D[i, j]) / 2.0
                new_radii[i] -= diff
                new_radii[j] -= diff
                new_radii[i] = np.maximum(0, new_radii[i])
                new_radii[j] = np.maximum(0, new_radii[j])

    return new_centers, new_radii
