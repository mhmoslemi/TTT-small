import numpy as np
from scipy.spatial.distance import cdist

def penalty(z, alpha=10.0, beta=10.0):
    """Compute penalty function for a full configuration vector."""
    n = 26
    centers = z[:2*n].reshape(n, 2)
    radii = z[2*n:]
    
    # Wall penalties
    wall_pen = 0.0
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        wall_dist = min(x, 1-x, y, 1-y)
        wall_pen += alpha * max(0, r - wall_dist)
    
    # Overlap penalties
    overlap_pen = 0.0
    for i in range(n):
        for j in range(i+1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            r_i, r_j = radii[i], radii[j]
            overlap = max(0, r_i + r_j - dist)
            overlap_pen += beta * overlap
    
    return wall_pen + overlap_pen

def energy(z, lam, alpha=10.0, beta=10.0):
    """Compute merit energy with penalty weight lambda."""
    total_radius = np.sum(z[2*26:])
    return -total_radius + lam * penalty(z, alpha, beta)

def propose_move(z, sigma_c, sigma_r, t):
    """Propose a new configuration with Gaussian noise."""
    z_new = z.copy()
    n = 26
    choice = np.random.randint(0, 3)
    
    if choice == 0:  # Center jitter
        i = np.random.randint(0, n)
        noise = np.random.normal(0, sigma_c * np.sqrt(t))
        z_new[2*i] += noise[0]
        z_new[2*i+1] += noise[1]
        z_new[2*i] = np.clip(z_new[2*i], 0, 1)
        z_new[2*i+1] = np.clip(z_new[2*i+1], 0, 1)
    elif choice == 1:  # Radius perturb
        i = np.random.randint(0, n)
        noise = np.random.normal(0, sigma_r * np.sqrt(t))
        z_new[2*n+i] += noise
        z_new[2*n+i] = max(0, z_new[2*n+i])
    else:  # Swap pair
        i, j = np.random.choice(n, 2, replace=False)
        z_new[2*n+i], z_new[2*n+j] = z_new[2*n+j], z_new[2*n+i]
    
    return z_new

def anneal():
    """Run simulated annealing with penalty-weighted energy."""
    n = 26
    # Initialize on a coarse hex grid
    spacing = 0.18
    xs = np.arange(spacing, 1.0, spacing)
    ys = np.arange(spacing, 1.0, spacing)
    grid = [(x, y) for x in xs for y in ys if x <= 1 and y <= 1]
    if len(grid) < n:
        grid = [(0.5, 0.5)] * n
    else:
        grid = grid[:n]
    
    centers = np.array(grid)
    radii = np.min(np.stack([centers.T, 1 - centers.T]), axis=0).min(axis=0)
    z = np.hstack([centers.flatten(), radii])
    
    # Annealing configuration
    T0 = 1.0
    c_T = 0.995
    T = T0
    lam = 0.1
    c_lam = 1.03
    sigma_c = 0.03
    sigma_r = 0.02
    max_iter = 10000
    
    best_energy = energy(z, lam)
    best_z = z.copy()
    best_feasible = None
    best_feasible_energy = np.inf
    
    # Keep track of number of valid moves that increase the sum of radii
    accepted_sums = 0
    num_steps = 0
    
    for iteration in range(max_iter):
        num_steps += 1
        z_proposed = propose_move(z, sigma_c, sigma_r, T)
        e_proposed = energy(z_proposed, lam)
        
        delta_e = e_proposed - energy(z, lam)
        if delta_e <= 0 or np.random.rand() < np.exp(-delta_e / T):
            z = z_proposed
            if e_proposed < best_energy:
                best_energy = e_proposed
                best_z = z.copy()
        
        # Check feasibility and update best feasible
        penalty_z = penalty(z, 10.0, 10.0)
        if penalty_z < 1e-8 and e_proposed < best_feasible_energy:
            best_feasible_energy = e_proposed
            best_feasible = z.copy()
        
        # Update temperature and penalty weight
        T *= c_T
        if T < 1e-5:
            break
        lam *= c_lam
    
    # Return best feasible or fallback
    final_z = best_feasible if best_feasible is not None else best_z
    return final_z

def project_and_ascent(centers, radii):
    """Apply projected gradient ascent to improve configuration."""
    n = centers.shape[0]
    z = np.hstack([centers.flatten(), radii])
    
    # Gradient ascent with projection
    eta = 5e-3
    max_iter = 500
    tol = 1e-7
    prev_sum = np.sum(radii)
    
    for _ in range(max_iter):
        # Update gradient based on zero gradient for x,y, unit gradient for r
        grad = np.zeros_like(z)
        grad[2*n:] = 1.0  # Only radii contribute to gradient of sum
    
        z += eta * grad
        
        # Project radii to valid limits
        for i in range(n):
            r = z[2*n+i]
            x, y = z[2*i], z[2*i+1]
            wall_limit = min(x, 1-x, y, 1-y)
            z[2*n+i] = min(r, wall_limit)
        
        # Resolve Overlaps
        centers = z[:2*n].reshape(n, 2)
        radii = z[2*n:]
        
        overlap_resolved = True
        while overlap_resolved:
            overlap_resolved = False
            for i in range(n):
                for j in range(i+1, n):
                    dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                    r_i, r_j = radii[i], radii[j]
                    overlap = r_i + r_j - dist
                    if overlap > 1e-12:
                        overlap_resolved = True
                        # Shrink both radii by same amount
                        shrink = overlap / 2
                        radii[i] = max(0, r_i - shrink)
                        radii[j] = max(0, r_j - shrink)
                        # Re-project radii
                        x_i, y_i = centers[i]
                        x_j, y_j = centers[j]
                        wall_i = min(x_i, 1-x_i, y_i, 1-y_i)
                        wall_j = min(x_j, 1-x_j, y_j, 1-y_j)
                        radii[i] = min(radii[i], wall_i)
                        radii[j] = min(radii[j], wall_j)
        
        # Reform z
        z = np.hstack([centers.flatten(), radii])
        
        # Update centers and radii
        centers = z[:2*n].reshape(n, 2)
        radii = z[2*n:]
        
        # Check stopping criteria
        current_sum = np.sum(radii)
        if abs(current_sum - prev_sum) < tol:
            break
        prev_sum = current_sum
        eta *= 0.95
    
    return centers, radii
