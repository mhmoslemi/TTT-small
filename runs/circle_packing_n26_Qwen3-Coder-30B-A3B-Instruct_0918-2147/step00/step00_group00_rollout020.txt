import numpy as np
import random

def make_hex_seed(n, margin):
    rows = int(np.sqrt(n * 2 / 3)) + 1
    cols = int(n / rows) + 1
    spacing_x = (1 - 2 * margin) / (cols - 1) if cols > 1 else 1
    spacing_y = (1 - 2 * margin) / (rows - 1) if rows > 1 else 1
    centers = []
    radii = []
    for i in range(rows):
        for j in range(cols):
            if len(centers) >= n:
                break
            x = margin + j * spacing_x
            y = margin + i * spacing_y
            centers.append([x, y])
            # Compute wall distances: min of distances to all four walls
            d_left = x
            d_right = 1 - x
            d_bottom = y
            d_top = 1 - y
            min_wall_dist = min(d_left, d_right, d_bottom, d_top)
            # Using a principle: larger radius near center, smaller near corners
            center_x, center_y = 0.5, 0.5
            dx = abs(x - center_x)
            dy = abs(y - center_y)
            dist_to_center = np.sqrt(dx ** 2 + dy ** 2)
            # Map to a usable radius
            # Center is at 0, edges at 0.5; 0.5 means get small radius.
            # Use a factor that gives larger radius when near center.
            ratio = 1 - dist_to_center * 2  # maps center to 1, edges to 0
            ratio = max(0, ratio)
            base_radius = 0.015 + 0.03 * ratio # nominal range: 0.015 to 0.045
            calculated_radius = min(base_radius, min_wall_dist)
            if i == 0 and j == 0:
                # ensure minimum is 0.015 given some corners could be very tight
                calculated_radius = max(0.015, calculated_radius)
            radii.append(calculated_radius)
        if len(centers) >= n:
            break
    # trim to exactly n items
    centers = np.array(centers[:n])
    radii = np.array(radii[:n])
    return centers, radii

def max_feasible_radius(i, centers, radii):
    x_i, y_i = centers[i]
    r_i = radii[i]
    # Wall distances
    d_left = x_i
    d_right = 1 - x_i
    d_bottom = y_i
    d_top = 1 - y_i
    min_wall_dist = min(d_left, d_right, d_bottom, d_top)
    
    # Neighbour distances
    min_neighbour_dist = float('inf')
    for j in range(len(centers)):
        if i == j:
            continue
        x_j, y_j = centers[j]
        r_j = radii[j]
        dx = x_i - x_j
        dy = y_i - y_j
        dist = np.sqrt(dx * dx + dy * dy)
        # Check if this circle j is too close
        gap = dist - (r_i + r_j)
        if gap < min_neighbour_dist:
            min_neighbour_dist = gap
    
    # Feasible radius is the minimum of wall and neighbour constraints
    if min_neighbour_dist == float('inf'):
        # No neighbours, just far from walls
        r_max = min_wall_dist
    else:
        r_max = min_wall_dist
        if min_neighbour_dist < 0:  # A conflict exists, adjust accordingly
            r_max = min(r_max, min_neighbour_dist + r_i)  # But stay above 0
        else:
            r_max = min(r_max, min_neighbour_dist)
    if r_max < 0:
        r_max = 0.0
    return r_max

def inflate_radii(centers, radii, alpha):
    sum_r_before = np.sum(radii)
    new_radii = radii.copy()
    mean_r = np.mean(radii)
    delta = alpha * mean_r
    new_radii += delta
    new_sum = 0.0
    for i in range(len(radii)):
        r_max = max_feasible_radius(i, centers, new_radii)
        new_radii[i] = min(new_radii[i], r_max)
        new_sum += new_radii[i]
    return new_radii, new_sum - sum_r_before

def annihilation(centers, radii):
    # If necessary, resolve overlaps by shrinking
    changed = True
    iteration = 0
    max_iter = 5
    while changed and iteration < max_iter:
        changed = False
        for i in range(len(centers)):
            x_i, y_i = centers[i]
            r_i = radii[i]
            # Check for overlap with others
            for j in range(len(centers)):
                if i == j:
                    continue
                x_j, y_j = centers[j]
                r_j = radii[j]
                dx = x_i - x_j
                dy = y_i - y_j
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < r_i + r_j:
                    # Overlap exists, decrease both
                    overlap = (r_i + r_j) - dist
                    # Reduce each by overlap / 2
                    r_i_new = max(0.0, r_i - overlap / 2.0)
                    r_j_new = max(0.0, r_j - overlap / 2.0)
                    # Don't allow negative radii to get us stuck
                    # If we already had large r_i and r_j, make sure we can lower them
                    if r_i_new < r_i and r_j_new < r_j:
                        old_r_i = radii[i]
                        old_r_j = radii[j]
                        radii[i] = r_i_new
                        radii[j] = r_j_new
                        changed = True
                    elif r_i_new < r_i:
                        radii[i] = r_i_new
                        changed = True
                    elif r_j_new < r_j:
                        radii[j] = r_j_new
                        changed = True
        iteration += 1

def anneal_centers(centers, radii, temp0, gamma, sigma0, steps):
    # Make a deep copy of the input to avoid side effects
    current_centers = centers.copy()
    current_radii = radii.copy()
    best_sum = np.sum(current_radii)
    current_sum = best_sum
    temp = temp0
    sigma = sigma0
    # Be cautious to not take very large steps
    iter_step = 0

    for step in range(steps):
        # Occasionally do a global shift to break symmetry
        if step % 50 == 0 and step > 0:
            shift = np.random.normal(0, 0.002, size=2)
            current_centers += shift

        # Random circle to move
        i = random.randint(0, len(current_centers) - 1)
        old_x, old_y = current_centers[i][0], current_centers[i][1]
        # Proposal
        dx, dy = np.random.normal(0, sigma, size=2)
        new_x = old_x + dx
        new_y = old_y + dy

        # Ensure it's within bounds
        if new_x < 0.0 or new_x > 1.0 or new_y < 0.0 or new_y > 1.0:
            continue

        # Temporarily move
        current_centers[i][0] = new_x
        current_centers[i][1] = new_y
        
        # Compute max feasible radius for i
        r_max = max_feasible_radius(i, current_centers, current_radii)
        old_r = current_radii[i]
        if r_max > old_r:
            current_radii[i] = r_max
            current_sum += (r_max - old_r)
            if current_sum > best_sum:
                best_sum = current_sum

        # Apply Metropolis criterion
        elif temp > 0:
            delta = r_max - old_r
            if delta > 0:
                prob = np.exp(delta / temp)
                if random.random() < prob:
                    current_radii[i] = r_max
                    current_sum += delta
                    if current_sum > best_sum:
                        best_sum = current_sum
            else:
                # We have a decrease in radius
                # Accept with probability only if beneficial
                if delta > -1e-10:
                    if random.random() < np.exp(delta / temp):
                        current_radii[i] = r_max
                        current_sum += delta
                        if current_sum > best_sum:
                            best_sum = current_sum
        iter_step += 1
        
        # Update temperature
        if temp >= 1e-10:
            temp *= gamma
        # Adjust sigma based on temp if needed
        # For simplicity, we decouple it slightly
        if iter_step % 100 == 0 and sigma > 1e-5:
            sigma *= 0.95  # gives gradual damping
    return current_centers, current_radii, best_sum

def polish_radii(centers, radii, beta, passes):
    # Exact local optimization on all circles
    changed = True
    for _ in range(passes):
        if not changed:
            break
        changed = False
        for i in range(len(centers)):
            x_i, y_i = centers[i]
            old_r = radii[i]

            # Compute exact feasibility gap
            # Wall gaps
            dx = min(x_i, 1 - x_i)
            dy = min(y_i, 1 - y_i)
            g_wall = min(dx, dy)
            # Neighbour gaps
            g_nei = float('inf')
            for j in range(len(centers)):
                if i == j:
                    continue
                x_j, y_j = centers[j]
                r_j = radii[j]
                dx = x_i - x_j
                dy = y_i - y_j
                dist = np.sqrt(dx * dx + dy * dy)
                gap = dist - (r_i + r_j)
                if gap < g_nei:
                    g_nei = gap
            if g_nei == float('inf'):
                g_nei = 0
            g = min(g_wall, g_nei)
            if g > 0:
                delta_r = beta * g
                new_r = min(old_r + delta_r, max_feasible_radius(i, centers, radii))
                if new_r > old_r and new_r > 0:
                    radii[i] = new_r
                    changed = True
        # Do a final overlap resolution sweep
        annihilation(centers, radii)
    return centers, radii

def run_packing():
    np.random.seed(42)  # For reproducibility
    random.seed(42)

    # Phase A – Structured initialisation
    margin = 0.04  # Pre-added margin
    centers, radii = make_hex_seed(26, margin)
    
    # Make sure that minimum radius is acceptable (safety margin)
    min_radius = np.min(radii)
    if min_radius < 0.015:
        factor = 0.015 / min_radius
        radii *= factor
    alpha = 0.04  # Inflation factor
    epsilon = 1e-6
    K = 8  # Stops if no improvement in K consecutive iterations
    M = 200  # Number of SA steps per block
    P = 5  # Stop if no improvement in P full mixed cycles
    T0 = 0.02
    gamma = 0.97
    sigma0 = 0.03
    beta = 0.6
    passes = 3

    # Phase B – Adaptive Inflation Loop
    improved_count = 0
    stagnation_iter = 0
    last_sum = np.sum(radii)
    for _ in range(500):  # Prevent infinite loop
        new_radii, delta_sum = inflate_radii(centers, radii, alpha)
        if abs(delta_sum) < epsilon:
            stagnation_iter += 1
            if stagnation_iter >= K:
                break
        else:
            stagnation_iter = 0
        radii = new_radii
        alpha *= 0.95
        if abs(delta_sum) > epsilon:
            improved_count = 0
        else:
            improved_count += 1
        if improved_count > 100:  # Eventually slow down inflation dynamics
            alpha = max(alpha, 0.001)

    # Phase C – Stochastic Centre Reoptimisation
    tmp_centers, tmp_radii, sa_sum = anneal_centers(centers, radii, T0, gamma, sigma0, M)

    # Phase D – Mixed Inflation-Reposition Cycle
    last_total = sa_sum
    total_cycles = 0
    while True:
        # Run inflation step
        new_radii, delta_sum = inflate_radii(tmp_centers, tmp_radii, alpha)
        tmp_radii = new_radii
        if abs(delta_sum) < epsilon:
            stagnation_iter += 1
        else:
            stagnation_iter = 0
        if stagnation_iter >= K:
            # Proceed to SA block
            tmp_centers, tmp_radii, sa_sum = anneal_centers(tmp_centers, tmp_radii, T0, gamma, sigma0, M)
            # Check for convergence on total sum
            total_sum = np.sum(tmp_radii)
            if abs(total_sum - last_total) < epsilon:
                total_cycles += 1
            else:
                total_cycles = 0  # Reset if there was progress
            last_total = total_sum
            if total_cycles >= P:
                break
        # Regardless, continue

    # Phase E – Final Polishing (Local Exact Inflation)
    final_centers, final_radii = polish_radii(tmp_centers, tmp_radii, beta, passes)
    
    # Final validation
    valid, msg = validate_packing(final_centers, final_radii)
    if not valid:
        raise RuntimeError(f"Packing invalid: {msg}")

    sum_radii = np.sum(final_radii)
    return final_centers, final_radii, sum_radii
