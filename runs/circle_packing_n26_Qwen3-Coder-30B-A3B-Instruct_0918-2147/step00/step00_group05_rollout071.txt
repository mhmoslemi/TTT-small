import numpy as np

def hex_lattice(d):
    """Generate hexagonal lattice points in [0,1] x [0,1]"""
    sqrt3 = np.sqrt(3)
    dx = d
    dy = sqrt3 / 2 * d
    rows = []
    y = 0
    row_idx = 0
    while y <= 1 + 1e-12:
        x = 0
        if row_idx % 2 == 1:
            x = dx / 2
        row = []
        while x <= 1 + 1e-12:
            row.append([x, y])
            x += dx
        rows.extend(row)
        y += dy
        row_idx += 1
    points = np.array(rows)
    return points[:20]  # Take first 20 points


def add_border_circles(points, d):
    """Add 6 border circles to the hexagonal core."""
    r = d / 4
    border_points = [
        [r, r],                     # bottom-left
        [1 - r, r],                 # bottom-right
        [r, 1 - r],                 # top-left
        [1 - r, 1 - r],             # top-right
        [0.5, r],                   # bottom mid
        [0.5, 1 - r]                # top mid
    ]
    return np.vstack([points, border_points])


def initial_radii(points):
    """Compute initial uniform safe radius for all points."""
    n = points.shape[0]
    # Compute wall distances for each point
    wall_distances = np.array([
        points[:, 0],         # x
        1 - points[:, 0],     # 1-x
        points[:, 1],         # y
        1 - points[:, 1]      # 1-y
    ]).min(axis=0)
    # Compute minimum inter-center distances
    dists = np.linalg.norm(points[:, None, :] - points[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    min_distances = dists.min(axis=1)
    # Initial safe radius
    r0 = np.minimum(wall_distances, min_distances / 2).min()
    return np.full(n, r0)


def inflate_radii(centers, radii):
    """Inflate all radii to maximum allowable values."""
    n = centers.shape[0]
    new_radii = radii.copy()
    # Precompute distance matrix
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    # For each point, compute new radius
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Wall constraints
        max_r = min(x, 1 - x, y, 1 - y)
        # Pairwise constraints
        dist_to_others = dists[i, :]
        for j in range(n):
            if i != j:
                max_r = min(max_r, dist_to_others[j] - radii[j])
        new_radius = max_r
        if new_radius > r + 1e-6:
            new_radii[i] = new_radius
    return new_radii


def relax_centers(centers, radii, steps=10):
    """Relax center positions with repulsion and wall forces."""
    n = centers.shape[0]
    # Parameters
    kappa = 0.02
    lambda_wall = 0.01
    alpha = 0.5
    beta_threshold = 0.02
    gamma = 0.02
    # Clamp centers to safe region wrt current radii
    centers = centers.copy()
    centers[:, 0] = np.clip(centers[:, 0], radii, 1 - radii)
    centers[:, 1] = np.clip(centers[:, 1], radii, 1 - radii)
    for step in range(steps):
        forces = np.zeros_like(centers)
        # Compute repulsion forces
        dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
        for i in range(n):
            for j in range(n):
                if i == j:
                    continue
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                d = np.sqrt(dx*dx + dy*dy)
                if d > 1e-12:
                    overlap = (radii[i] + radii[j]) - d
                    if overlap > 0:
                        # Circle overlap is tiny due to inflation, just prevent divergence
                        overlap = 0
                    max_force_dist = radii[i] + radii[j] + beta_threshold
                    if d < max_force_dist:
                        force_magnitude = kappa * max(0, max_force_dist - d) / d
                        forces[i, 0] -= force_magnitude * dx
                        forces[i, 1] -= force_magnitude * dy
        # Compute wall forces
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            # x wall forces
            dx_right = x - r  # dist from left wall
            dx_left = (1 - x) - r  # dist from right wall
            forces[i, 0] += lambda_wall * max(0, gamma - dx_right)  # push right
            forces[i, 0] -= lambda_wall * max(0, gamma - dx_left)   # push left
            # y wall forces
            dy_bottom = y - r  # dist from bottom wall
            dy_top = (1 - y) - r  # dist from top wall
            forces[i, 1] += lambda_wall * max(0, gamma - dy_bottom)  # push up
            forces[i, 1] -= lambda_wall * max(0, gamma - dy_top)     # push down
        # Apply forces
        centers += alpha * forces
        # Clamp to [r, 1-r]
        centers[:, 0] = np.clip(centers[:, 0], radii, 1 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1 - radii)
    return centers


def run_packing():
    """Main packing function."""
    # Generate core double-hexagonal pattern
    d = 0.18  # spacing for core hexagon
    points = hex_lattice(d)
    points = add_border_circles(points, d)
    radii = initial_radii(points)
    centers = points

    # Run inflation and relaxation loop
    MAX_OUTER = 30
    for _ in range(MAX_OUTER):
        new_radii = inflate_radii(centers, radii)
        centers = relax_centers(centers, new_radii)
        if np.allclose(radii, new_radii, rtol=1e-6):
            break
        radii = new_radii

    # Final validation (should not fail)
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise AssertionError(f"Validation failed: {msg}")

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
