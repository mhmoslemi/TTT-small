import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
from math import sqrt

def generate_lattice(s):
    """Generate hexagonal lattice points inside unit square."""
    rows = int(1.0 / (s * np.sqrt(3) / 2)) + 2
    cols = int(1.0 / s) + 2
    points = []
    for r in range(rows):
        offset = (r % 2) * 0.5
        for c in range(cols):
            x = c * s + offset * s
            y = r * s * np.sqrt(3) / 2
            if s/2 <= x <= 1 - s/2 and s/2 <= y <= 1 - s/2:
                points.append([x, y])
    return np.array(points)

def build_milp(P):
    """Build CVXPY optimization model for given lattice points."""
    K = len(P)
    # Variables
    z = cp.Variable(K, boolean=True)
    r = cp.Variable(K, nonneg=True)
    
    # Wall constraints: each radius <= distance to nearest wall
    wL = P[:, 0]
    wR = 1 - P[:, 0]
    wB = P[:, 1]
    wT = 1 - P[:, 1]
    w = np.minimum(np.minimum(wL, wR), np.minimum(wB, wT))
    constraints = [r <= w * z]
    
    # Distance matrix between points (precompute maximum Euclidean dist)
    d = np.zeros((K, K))
    for i in range(K):
        for j in range(i+1, K):
            dist = np.linalg.norm(P[i] - P[j])
            d[i, j] = d[j, i] = dist
    
    # Non-overlap constraints using big-M method with M = 1
    M = 1.0
    for i in range(K):
        for j in range(i+1, K):
            # d[i,j] >= r[i] + r[j] - M * (2 - z[i] - z[j])
            constraints.append(d[i, j] >= r[i] + r[j] - M * (2 - z[i] - z[j]))
    
    # Card constraint
    constraints.append(cp.sum(z) == 26)
    
    # Objective
    objective = cp.Maximize(cp.sum(r))
    
    prob = cp.Problem(objective, constraints)
    
    def solve_subproblem(z_fixed):
        # Fix z variables for the subproblem
        z.value = z_fixed
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except:
            return None, None
        if prob.status == cp.OPTIMAL:
            return r.value, prob.value
        else:
            return None, None
            
    return solve_subproblem

def branch_and_bound(milp_solver, node_limit=20000):

    # Start with a heuristic approach to find a good setup
    # Generate large enough lattice

    K = len(P)
    
    # Build constraints for better estimation
    # For this problem, a greedy heuristic is acceptable
    
    # Compute pairwise distances
    d = np.zeros((K, K))
    for i in range(K):
        for j in range(i+1, K):
            dist = np.linalg.norm(P[i] - P[j])
            d[i, j] = d[j, i] = dist

    # Heuristic: start with farthest points and greedily add more
    # Choose 26 points starting with max separation
    selected = set()
    if K == 0:
        return [], np.zeros(0)
        
    # Start with a random point
    selected.add(0)
    
    # Greedy selection: select the point that maximizes distance to all selected points
    while len(selected) < min(26, K):
        best_dist = -1
        best_idx = -1
        for i in range(K):
            if i in selected:
                continue
            min_dist = float('inf')
            for j in selected:
                min_dist = min(min_dist, d[i, j])
            if min_dist > best_dist:
                best_dist = min_dist
                best_idx = i
        if best_idx != -1:
            selected.add(best_idx)
        else:
            # If no new point found try another approach
            break
            
    # If still not enough, fill with closest to existing
    while len(selected) < 26 and K > 0:
        # Prefer points in majority region
        curr_selected = list(selected)
        best_i = -1
        max_sum_dist = -1
        for i in range(K):
            if i in selected:
                continue
            sum_dist = sum(d[i, j] for j in curr_selected)
            if sum_dist > max_sum_dist:
                max_sum_dist = sum_dist
                best_i = i
        if best_i != -1:
            selected.add(best_i)
        else:
            # Ensure a fill
            for i in range(K):
                if i not in selected:
                    selected.add(i)

    if len(selected) != 26:
        # Fallback to a set
        final_selected = list(selected)[:26]

            for i in range(26 - len(final_selected)):

    selected_indices = sorted(selected)

            r_values[selected_indices] = 0.05  # simple fallback

def refine_solution(centers, radii):

    n = len(centers)
    # Slice down to 26
    if n < 26:
        # Add dummy points, not used
        return centers, radii, np.sum(radii)
    centers = centers[:26]
    radii = radii[:26]
    
    # Local optimization: use scipy minimize
    def obj(x):
        # x is [c1x, c1y, r1, c2x, c2y, r2, ..., c26x, c26y, r26]
        total = 0
        for i in range(26):
            total += x[3*i+2]
        return -total
    
    def grad(x):
        # Gradient of objective
        g = np.zeros_like(x)
        for i in range(26):

        return g
    
    def constraint1(x, i, j):
        # Distance constraint: (cx_i - cx_j)^2 + (cy_i - cy_j)^2 >= (r_i + r_j)^2
        ci = (x[3*i], x[3*i+1])
        cj = (x[3*j], x[3*j+1])
        ri = x[3*i+2]
        rj = x[3*j+2]
        diff = np.array(ci) - np.array(cj)
        dist_sq = np.sum(diff ** 2)
        const_val = (ri + rj)**2
        return dist_sq - const_val
    
    def constraint2(x, i):
        # Wall constraints
        cx = x[3*i]
        cy = x[3*i+1]
        r = x[3*i+2]
        c = np.array([cx, cy, r])
        contr = [
            c[2] - c[0],   # r <= cx
            c[2] - (1 - c[0]),   # r <= 1 - cx
            c[2] - c[1],   # r <= cy
            c[2] - (1 - c[1])   # r <= 1 - cy
        ]
        return np.array(contr)
    
    # Build bounds and constraints for scipy.minimize
    bounds = []
    for i in range(26):
        bounds.append((1e-6, 1 - 1e-6))  # cx
        bounds.append((1e-6, 1 - 1e-6))  # cy
        bounds.append((1e-6, 0.5))       # r, cap at 0.5 for safety
    
    cons = []
    for i in range(26):
        for j in range(i+1, 26):
            cons.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: constraint1(x, i, j)})
            
    # Add constraint per circle
    for i in range(26):
        con_wall = {'type': 'ineq', 'fun': lambda x, i=i: constraint2(x, i)}
        cons.append(con_wall)
    
    # Initial values
    x0 = []
    for i in range(26):
        x0.append(centers[i][0])
        x0.append(centers[i][1])
        x0.append(radii[i])
    x0 = np.array(x0)
    
    # Optimization
    result = minimize(obj, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                      jac=grad, options={'ftol': 1e-9, 'maxiter': 2000})
    
    # Extract new values
    new_centers = np.zeros((26, 2))
    new_radii = np.zeros(26)
    for i in range(26):
        new_centers[i] = [result.x[3*i], result.x[3*i+1]]
        new_radii[i] = result.x[3*i+2]

        if not valid2:
            # Crop to ensure feasibility
            shrink = 0.99
            new_radii = new_radii * shrink
            new_centers = new_centers.astype(float)
            new_radii = new_radii.astype(float)
    
    return new_centers, new_radii, np.sum(new_radii)
