import numpy as np
from scipy.optimize import linprog, minimize

def build_lp_model(n_circles):
    # Prepare LP model: maximize sum of radii with linear constraints
    # Variables: [x1, y1, r1, x2, y2, r2, ..., x_n, y_n, r_n]
    n_vars = 3 * n_circles
    c = np.zeros(n_vars)
    c[2::3] = -1.0  # maximize sum of radii (negative for minimization)

    # Build constraint matrix A_ub and vector b_ub
    # Wall inequality constraints: r_i <= x_i, 1-x_i, y_i, 1-y_i
    A_ub = []
    b_ub = []
    for i in range(n_circles):
        var_idx = i * 3
        # r_i <= x_i => -r_i + x_i >= 0
        row = np.zeros(n_vars)
        row[var_idx + 2] = -1
        row[var_idx] = 1
        A_ub.append(row)
        b_ub.append(0)
        # r_i <= 1 - x_i => -r_i - x_i >= -1
        row = np.zeros(n_vars)
        row[var_idx + 2] = -1
        row[var_idx] = -1
        A_ub.append(row)
        b_ub.append(-1)
        # r_i <= y_i => -r_i + y_i >= 0
        row = np.zeros(n_vars)
        row[var_idx + 2] = -1
        row[var_idx + 1] = 1
        A_ub.append(row)
        b_ub.append(0)
        # r_i <= 1 - y_i => -r_i - y_i >= -1
        row = np.zeros(n_vars)
        row[var_idx + 2] = -1
        row[var_idx + 1] = -1
        A_ub.append(row)
        b_ub.append(-1)

    # Pairwise axis-separated constraints (at most 4 per pair)
    # For each pair (i, j), add four constraints:
    # x_i + r_i <= x_j - r_j  => -x_i + x_j - r_i - r_j >= 0
    # x_j + r_j <= x_i - r_i  =>  x_i - x_j - r_i - r_j >= 0
    # y_i + r_i <= y_j - r_j  => -y_i + y_j - r_i - r_j >= 0
    # y_j + r_j <= y_i - r_i  =>  y_i - y_j - r_i - r_j >= 0

    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            var_i = i * 3
            var_j = j * 3

            # x_i + r_i <= x_j - r_j
            row = np.zeros(n_vars)
            row[var_i] = -1
            row[var_j] = 1
            row[var_i + 2] = -1
            row[var_j + 2] = -1
            A_ub.append(row)
            b_ub.append(0)

            # x_j + r_j <= x_i - r_i
            row = np.zeros(n_vars)
            row[var_i] = 1
            row[var_j] = -1
            row[var_i + 2] = -1
            row[var_j + 2] = -1
            A_ub.append(row)
            b_ub.append(0)

            # y_i + r_i <= y_j - r_j
            row = np.zeros(n_vars)
            row[var_i + 1] = -1
            row[var_j + 1] = 1
            row[var_i + 2] = -1
            row[var_j + 2] = -1
            A_ub.append(row)
            b_ub.append(0)

            # y_j + r_j <= y_i - r_i
            row = np.zeros(n_vars)
            row[var_i + 1] = 1
            row[var_j + 1] = -1
            row[var_i + 2] = -1
            row[var_j + 2] = -1
            A_ub.append(row)
            b_ub.append(0)

    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Optional: tighten constraints by scaling or adding diagonal cuts
    # Not doing in this implementation for brevity.
    
    return A_ub, b_ub, c

def solve_lp(A_ub, b_ub, c):
    # Solve the LP (using method='highs')
    res = linprog(c=c, A_ub=A_ub, b_ub=b_ub, method="highs", options={"presolve": False})
    if not res.success:
        raise RuntimeError(f"LP solver failed: {res.message}")

    # Extract variables
    n_vars = len(res.x)
    n_circles = n_vars // 3

    centers = np.zeros((n_circles, 2))
    radii = np.zeros(n_circles)
    
    for i in range(n_circles):
        centers[i] = [res.x[i * 3], res.x[i * 3 + 1]]
        radii[i] = res.x[i * 3 + 2]

    return centers, radii

def detect_diagonal_conflicts(centers, radii):
    n = len(centers)
    conflicts = []

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                conflicts.append((i, j))
                
    return conflicts

    for i in range(n):
