import numpy as np
from scipy.optimize import minimize

def initial_lattice(n=26):
    # Generate hexagonal lattice points
    # Use spacing s = 0.25 to ensure we get enough points inside the square
    s = 0.25
    sqrt3 = np.sqrt(3)
    y_spacing = s * sqrt3 / 2
    points = []
    
    # Generate points in a hexagonal pattern, clipped to ensure they fit
    y = 0
    row_offset = 0
    while len(points) < n:
        x = row_offset * s / 2
        while len(points) < n and x <= 1:
            points.append((x, y))
            x += s
        y += y_spacing
        row_offset = 1 - row_offset
    
    centers = np.array(points[:n])
    
    # Compute safe radii for each point
    radii = []
    for i, (x, y) in enumerate(centers):
        # Distance to boundaries
        d_left = x
        d_right = 1 - x
        d_bottom = y
        d_top = 1 - y
        
        min_dist_to_boundary = min(d_left, d_right, d_bottom, d_top)
        
        # Distance to nearest neighbors
        min_dist_to_neighbor = float('inf')
        for j in range(i):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            if dist < min_dist_to_neighbor:
                min_dist_to_neighbor = dist
                
        if min_dist_to_neighbor != float('inf'):
            min_dist_to_neighbor /= 2
            
        safe_radius = min(min_dist_to_boundary, min_dist_to_neighbor)
        radii.append(max(0, safe_radius))
        
    return centers, np.array(radii)

def pack(centers, radii):
    # Flatten centers and radii into z vector
    z = np.empty(3 * centers.shape[0])
    z[0::3] = centers[:, 0]  # x coordinates
    z[1::3] = centers[:, 1]  # y coordinates
    z[2::3] = radii          # radii
    return z

def unpack(z):
    n = z.shape[0] // 3
    centers = np.empty((n, 2))
    centers[:, 0] = z[0::3]
    centers[:, 1] = z[1::3]
    radii = z[2::3]
    return centers, radii

def objective(z):
    # Return -sum(radii) because scipy minimize minimizes
    return -np.sum(z[2::3])

def constraint_boundary(z):
    centers, radii = unpack(z)
    constraints = []
    for i in range(centers.shape[0]):
        x, y = centers[i]
        r = radii[i]
        # x - r >= 0
        constraints.append(x - r) 
        # r + x - 1 <= 0
        constraints.append(r + x - 1)
        # y - r >= 0
        constraints.append(y - r)
        # r + y - 1 <= 0
        constraints.append(r + y - 1)
    return np.array(constraints)

def constraint_nonoverlap(z):
    centers, radii = unpack(z)
    constraints = []
    for i in range(centers.shape[0]):
        for j in range(i + 1, centers.shape[0]):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.hypot(dx, dy)  # use hypot to avoid numerical issues
            r_sum = radii[i] + radii[j]
            # (r_i + r_j) - dist <= 0
            constraints.append(r_sum - dist)
    return np.array(constraints)

def repair(centers, radii):
    # Make sure all circles are within the square and non-overlapping
    n = centers.shape[0]
    max_iterations = 10
    
    # First pass: enforce boundary constraints
    for i in range(n):
        r = radii[i]
        x, y = centers[i]
        # Adjust radius to fit within square
        r = min(r, x, 1 - x, y, 1 - y)
        radii[i] = r
    
    # If any overlaps remain, shrink circles to fix them
    for _ in range(max_iterations):
        has_overlap = False
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.hypot(dx, dy)
                r_sum = radii[i] + radii[j]
                
                if dist < r_sum:
                    overlap = r_sum - dist
                    # Shrink both radii by half the overlap amount
                    shrink = overlap / 2
                    radii[i] = max(0, radii[i] - shrink)
                    radii[j] = max(0, radii[j] - shrink)
                    has_overlap = True
                
        if not has_overlap:
            break
            
    # Ensures no circles have negative radius
    radii = np.maximum(radii, 0)
    return centers, radii

    # Get initial feasible solution
    centers, radii = initial_lattice(n)
    
    # Flatten for optimization
    z = pack(centers, radii)
    
    # Optimization bounds
    bounds = []
    for i in range(n):
        bounds.append((0, 1))  # x
        bounds.append((0, 1))  # y
        bounds.append((0, 0.5))  # r (max radius is 0.5)
    
    # Optimization constraints
    constraints = []
    
    # Boundary constraints

        return constraint_boundary(z)
    constraints.append({'type': 'ineq', 'fun': boundary_func})

        return constraint_nonoverlap(z)
