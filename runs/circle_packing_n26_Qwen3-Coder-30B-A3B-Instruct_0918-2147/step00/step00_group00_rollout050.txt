import numpy as np
import random

def make_hex_seed(n, margin):
    """Generate a hexagonal seed packing with different initial radii."""
    sqrt3 = np.sqrt(3)
    spacing = 2 * margin
    rows = int(np.ceil(np.sqrt(n) * 2 / sqrt3))
    cols = int(np.ceil(n / rows))
    
    centers = []
    radii = []
    
    for i in range(rows):
        row_offset = (i % 2) * spacing / 2
        for j in range(cols):
            x = margin + row_offset + j * spacing
            y = margin + i * spacing * sqrt3 / 2
            if len(centers) < n:
                centers.append([x, y])
                # Initial radius is based on distance to nearest wall
                dist_to_walls = [x, 1-x, y, 1-y]
                min_wall_dist = min(dist_to_walls)
                r = min_wall_dist * 0.6
                radii.append(max(0.015, r))
            else:
                break
        if len(centers) >= n:
            break
    
    centers = np.array(centers[:n])
    radii = np.array(radii[:n])
    
    # Rescale to ensure minimum radius 0.015
    min_r = np.min(radii)
    if min_r < 0.015:
        scale = 0.015 / min_r
        centers *= scale
        radii *= scale
    
    return centers, radii

def max_feasible_radius(i, centers, radii):
    """Compute the maximum feasible radius for circle i."""
    x_i, y_i = centers[i]
    r_i = radii[i]
    
    d_L = x_i
    d_R = 1 - x_i
    d_B = y_i
    d_T = 1 - y_i
    
    min_dist = min(d_L, d_R, d_B, d_T)
    
    for j in range(centers.shape[0]):
        if i != j:
            dist = np.sqrt((centers[i][0] - centers[j][0])**2 + (centers[i][1] - centers[j][1])**2)
            min_dist = min(min_dist, dist - radii[j])
    
    return min(min_dist, r_i)

def inflate_radii(centers, radii, alpha):
    """Inflate all radii uniformly and clamp each one locally."""
    mean_r = np.mean(radii)
    delta = alpha * mean_r
    
    old_sum = np.sum(radii)
    new_radii = radii + delta
    
    for i in range(len(radii)):
        max_r = max_feasible_radius(i, centers, new_radii)
        new_radii[i] = min(new_radii[i], max_r)
    
    return new_radii, np.sum(new_radii) - old_sum

def anneal_centers(centers, radii, temp0, gamma, sigma0, steps):
    """Perform simulated annealing on circle centers."""
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = np.sum(radii)
    
    temp = temp0
    
    for step in range(steps):
        circle_idx = random.randint(0, len(centers) - 1)
        old_x, old_y = centers[circle_idx]
        delta_x = np.random.normal(0, sigma0 * temp / temp0)
        delta_y = np.random.normal(0, sigma0 * temp / temp0)
        new_x = old_x + delta_x
        new_y = old_y + delta_y
        
        # Test new center position with current radii
        test_centers = centers.copy()
        test_centers[circle_idx] = [new_x, new_y]
        
        new_r = max_feasible_radius(circle_idx, test_centers, radii)
        
        old_r = radii[circle_idx]
        
        if new_r > old_r:
            centers[circle_idx] = [new_x, new_y]
            radii[circle_idx] = new_r
            current_sum = np.sum(radii)
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = centers.copy()
                best_radii = radii.copy()
        else:
            # Accept with probability based on temperature
            prob = np.exp((new_r - old_r) / temp)
            if random.random() < prob:
                centers[circle_idx] = [new_x, new_y]
                radii[circle_idx] = new_r
                current_sum = np.sum(radii)
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = centers.copy()
                    best_radii = radii.copy()
        
        # Occasionally global shift
        if step % 50 == 0:
            shift = np.random.uniform(-0.002, 0.002, size=2)
            centers += shift
        
        # Update temperature
        temp *= gamma
    
    return best_centers, best_radii, best_sum

def mixed_cycle(centers, radii, params):
    """Conduct alternating inflation and SA cycles."""
    alpha = 0.04
    epsilon = 1e-6
    k_stagnation = 8
    m_sa_steps = 200
    p_cycles = 5
    k = 0
    
    current_sum = np.sum(radii)
    prev_sum = 0.0
    cycle_count = 0
    
    while cycle_count < p_cycles:
        # Inflation loop
        while k < k_stagnation:
            prev_sum = current_sum
            new_radii, delta = inflate_radii(centers, radii, alpha)
            radii = new_radii
            current_sum = np.sum(radii)
            if abs(delta) < epsilon:
                k += 1
            else:
                k = 0
            
            alpha *= 0.95  # decay
            if alpha < 0.002:
                alpha = 0.002

        # SA phase
        centers, radii, sa_sum = anneal_centers(centers, radii, params['temp0'],
                                               params['gamma'], params['sigma0'],
                                               m_sa_steps)
        current_sum = sa_sum
        cycle_count += 1
        k = 0
        
    return centers, radii

def polish_radii(centers, radii, beta, passes):
    """Apply final polishing to extract slack."""
    for _ in range(passes):
        updated = True
        iteration = 0
        while updated and iteration < 5:
            updated = False
            for i in range(len(centers)):
                # Compute gaps
                x, y = centers[i]
                r = radii[i]
                
                g_wall = min(x - r, 1 - x - r, y - r, 1 - y - r)
                
                g_nei = float('inf')
                for j in range(len(centers)):
                    if i != j:
                        dist = np.sqrt((centers[i][0] - centers[j][0])**2 + (centers[i][1] - centers[j][1])**2)
                        g = dist - (r + radii[j])
                        g_nei = min(g_nei, g)
                
                g = min(g_wall, g_nei)
                if g > 0:
                    new_r = r + beta * g
                    max_feasible = max_feasible_radius(i, centers, radii)
                    new_r = min(new_r, max_feasible)
                    if new_r > r:
                        radii[i] = new_r
                        updated = True
                        
            iteration += 1
        
        # Resolve overlaps after gap adjustments
        for i in range(len(centers)):
            change = True
            while change:
                change = False
                for j in range(i + 1, len(centers)):
                    dist = np.sqrt((centers[i][0] - centers[j][0])**2 + (centers[i][1] - centers[j][1])**2)
                    overlap = dist - (radii[i] + radii[j])
                    if overlap < -1e-12:
                        delta = -overlap / 2
                        radii[i] -= delta
                        radii[j] -= delta
                        change = True
        
        # Final recheck after overlap resolutions to enforce feasibility
        for i in range(len(centers)):
            max_feasible = max_feasible_radius(i, centers, radii)
            if radii[i] > max_feasible:
                radii[i] = max_feasible
        
    return centers, radii

    np.random.seed(42)  # Fix for reproducibility
