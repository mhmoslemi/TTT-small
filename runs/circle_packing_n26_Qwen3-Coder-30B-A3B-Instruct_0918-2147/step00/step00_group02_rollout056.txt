import numpy as np
from scipy.optimize import minimize
from ortools.linear_solver import pywraplp

def generate_candidates(g, radius_levels):
    candidates = []
    grid_points = np.arange(0, 1 + g/2, g)
    for x in grid_points:
        for y in grid_points:
            for r in radius_levels:
                if (x - r >= 0 and x + r <= 1 and y - r >= 0 and y + r <= 1):
                    candidates.append((x, y, r))
    return candidates

def build_milp(candidates):
    solver = pywraplp.Solver.CreateSolver('SCIP')
    if not solver:
        raise RuntimeError("SCIP solver not available")

    num_candidates = len(candidates)
    z = [solver.BoolVar(f'z_{i}') for i in range(num_candidates)]

    # Constraint: At most one circle per grid point
    grid_map = {}
    for i, (x, y, r) in enumerate(candidates):
        grid_key = (int(x/g), int(y/g))
        if grid_key not in grid_map:
            grid_map[grid_key] = []
        grid_map[grid_key].append(i)

    for grid_key, indices in grid_map.items():
        solver.Add(solver.Sum([z[i] for i in indices]) <= 1)

    # Constraint: Exactly 26 circles
    solver.Add(solver.Sum(z) == 26)

    # Constraint: Non-overlap using L-infinity approximations
    # For each pair of circles, we add constraints
    # We store actual distances to not double-count pairs
    for i in range(num_candidates):
        for j in range(i+1, num_candidates):
            x1, y1, r1 = candidates[i]
            x2, y2, r2 = candidates[j]
            dx = solver.NumVar(0, 1, f'dx_{i}_{j}')
            dy = solver.NumVar(0, 1, f'dy_{i}_{j}')
            
            solver.Add(dx >= x1 - x2)
            solver.Add(dx >= x2 - x1)
            solver.Add(dy >= y1 - y2)
            solver.Add(dy >= y2 - y1)
            
            # L-infinity non-overlap
            solver.Add(dx + dy >= r1 + r2)
            solver.Add(dx - dy >= r1 + r2 - 2)

            # Link variables with binary selection
            solver.Add(dx <= 1.0 * (z[i] + z[j]))
            solver.Add(dy <= 1.0 * (z[i] + z[j]))

    # Objective
    solver.Maximize(solver.Sum([r * z[i] for i, (_, _, r) in enumerate(candidates)]))

    return solver, z

def solve_milp(solver, z):
    status = solver.Solve()
    if status != pywraplp.Solver.OPTIMAL:
        raise RuntimeError(f"MILP solver did not converge: status {status}")

    selected = [i for i in range(len(z)) if z[i].solution_value() > 0.5]
    return selected

def extract_initial_solution(selected, candidates):
    centers = np.array([[candidates[i][0], candidates[i][1]] for i in selected])
    radii = np.array([candidates[i][2] for i in selected])
    return centers, radii

def shrink_initial(radii, delta):
    return np.maximum(radii - delta, 0)

def pack_variables(centers0, radii0):
    flat = []
    for i in range(len(centers0)):
        flat.extend([centers0[i][0], centers0[i][1], radii0[i]])
    return np.array(flat)

def unpack_solution(z):
    n = len(z) // 3
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i][0] = z[3*i]
        centers[i][1] = z[3*i+1]
        radii[i] = z[3*i+2]
    return centers, radii

def define_objective(z):
    return -np.sum(z[2::3])

def define_jacobian(z):
    n = len(z) // 3
    grad = np.zeros_like(z)
    grad[2::3] = -1.0
    return grad

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
