import numpy as np
import cvxpy as cp
from scipy.spatial.distance import cdist

def build_leaf_problem(cell, k):
    """Build a Second-Order Cone Program for k circles in a rectangular cell."""
    x_min, y_min, x_max, y_max = cell
    width = x_max - x_min
    height = y_max - y_min

    # Variables for centers and radii
    centers = cp.Variable((k, 2))
    radii = cp.Variable(k)

    # Constraints for each circle
    constraints = []

    # Boundary constraints
    for i in range(k):
        x, y = centers[i]
        r = radii[i]
        constraints.append(x - r >= x_min)
        constraints.append(x + r <= x_max)
        constraints.append(y - r >= y_min)
        constraints.append(y + r <= y_max)

    # Non-overlap constraints
    for i in range(k):
        for j in range(i+1, k):
            dist = cp.norm(centers[i] - centers[j])
            constraints.append(dist >= radii[i] + radii[j])

    # Objective: maximize sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Create the problem
    prob = cp.Problem(objective, constraints)
    return prob, centers, radii

def solve_leaf(cell, k):
    prob, centers, radii = build_leaf_problem(cell, k)
    try:
        prob.solve(solver=cp.ECOS, verbose=False, abstol=1e-12, reltol=1e-12)
    except Exception:
        prob.solve(solver=cp.SCS, verbose=False, abstol=1e-12, reltol=1e-12)
    return centers.value, radii.value

    # Let the relative positions be fixed where there is contact,

            constraints.append(x + r <= 1)
            constraints.append(y - r >= 0)

        for i in range(n_total):
            for j in range(i+1, n_total):

        # Since we are merging, and have some hypothesis of which circles touch
        # using soft constraint approach; for large clusters, more involved contact graph needed
        # For simplicity, solve as a block problem
