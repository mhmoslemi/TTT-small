import numpy as np
from scipy.optimize import linprog
from math import sqrt

def initial_hex_centers(k, spacing=0.12):
    """Generate k non-overlapping centers on a hexagonal lattice."""
    sqrt3 = sqrt(3)
    s = spacing
    centers = []
    i = 0
    j = 0
    while len(centers) < k:
        x = i * s + (j % 2) * s / 2
        y = j * s * sqrt3 / 2
        if 0 <= x <= 1 and 0 <= y <= 1:
            centers.append([x, y])
        if x > 1:
            j += 1
            i = 0
        else:
            i += 1
    return np.array(centers[:k])

def initial_radii(k, r0=0.005):
    """Return k identical small radii."""
    return np.full(k, r0)

def build_lp_constants(centers, radii):
    """Compute all constants needed for LP formulation."""
    n = centers.shape[0]
    d_left = centers[:, 0]
    d_right = 1 - centers[:, 0]
    d_bottom = centers[:, 1]
    d_top = 1 - centers[:, 1]
    wall_distances = np.stack([d_left, d_right, d_bottom, d_top], axis=1)

    # Pairwise distances
    D = np.zeros((n, n))
    for i in range(n):
        for j in range(i+1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            D[i, j] = D[j, i] = dist

    return wall_distances, D

def solve_radius_lp(consts, radii, graph, α=0.5):
    """Solve the radius LP using scipy.optimize.linprog."""
    wall_distances, D = consts
    n = len(radii)
    
    # Construct inequality matrix A_ub and vector b_ub.
    # Each row of A_ub corresponds to one constraint (linear inequality).
    # We have constraints from:
    # 1. Wall capacity: Δr_i ≤ min(d_i.left, d_i.right, d_i.bottom, d.i.top) - r_i
    # 2. Pairwise capacity: Δr_i + Δr_j ≤ D_ij - (r_i + r_j)
    # 3. Optional graph-based transfer limits (sparse)

    # Wall constraints
    init_capacities = np.min(wall_distances, axis=1) - radii
    
    # Compute pairwise constraints (excluding duplicates)
    constraints_ineq = []
    constraints_eq = []
    
    # All wall capacity constraints (n total)
    A_wall = np.zeros((n, n))
    for i in range(n):
        A_wall[i, i] = 1
    b_wall = init_capacities

    # Pairwise constraints
    row_idx = n
    A_pair = []
    b_pair = []
    for i in range(n):
        for j in range(i+1, n):
            if D[i,j] > radii[i] + radii[j]:
                A_row = np.zeros(n)
                A_row[i] = 1
                A_row[j] = 1
                A_pair.append(A_row)
                b_pair.append(D[i,j] - (radii[i] + radii[j]))

    A_ub = np.vstack([A_wall] + A_pair)
    b_ub = np.hstack([b_wall] + b_pair)
    c = np.ones(n)

    # Bound on Δr_i >= 0
    bounds = [(0, None) for _ in range(n)]

    # Solve using scipy
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
        # fallback if infeasible
        return np.zeros(n)
    except Exception:
        return np.zeros(n)

def apply_increment(radii, Δr):
    """Update radii by applying the increments."""
    return radii + Δr

def push_feasibility(centers, radii):
    """Perform geometric push to fix overlaps and wall violations."""
    n = len(radii)
    # Make copies to avoid in-place modification in case of error
    new_centers = centers.copy()
    new_radii = radii.copy()
    
    # Wall constraints
    for i in range(n):
        dx = new_radii[i] - new_centers[i, 0]
        if dx > 0:
            new_radii[i] -= dx
            new_centers[i, 0] = new_radii[i]
        dx = new_centers[i, 0] + new_radii[i] - 1
        if dx > 0:
            new_radii[i] -= dx
            new_centers[i, 0] = 1 - new_radii[i]
        dy = new_radii[i] - new_centers[i, 1]
        if dy > 0:
            new_radii[i] -= dy
            new_centers[i, 1] = new_radii[i]
        dy = new_centers[i, 1] + new_radii[i] - 1
        if dy > 0:
            new_radii[i] -= dy
            new_centers[i, 1] = 1 - new_radii[i]

    # Pairwise overlaps

            if dist < r_sum:
                delta = r_sum - dist
                # Move centers apart along line connecting them

                    unit = (new_centers[i] - new_centers[j]) / dist
                    new_centers[i] += (delta / 2) * unit
                    new_centers[j] -= (delta / 2) * unit
                else:
                    # When centers coincide, use arbitrary direction
                    unit = np.array([1.0, 0.0])
                # Recheck wall constraints after push
                if new_centers[i, 0] - new_radii[i] < 0:
                    new_radii[i] = new_centers[i, 0]
                if new_centers[i, 0] + new_radii[i] > 1:
                    new_radii[i] = 1 - new_centers[i, 0]
                if new_centers[i, 1] - new_radii[i] < 0:
                    new_radii[i] = new_centers[i, 1]
                if new_centers[i, 1] + new_radii[i] > 1:
                    new_radii[i] = 1 - new_centers[i, 1]
                if new_centers[j, 0] - new_radii[j] < 0:
                    new_radii[j] = new_centers[j, 0]

                if new_centers[j, 1] + new_radii[j] > 1:

def run_iterations(centers, radii, max_iter=200, ε=1e-5, α=0.5):
    """Run the main iterative optimization loop."""

    # Initialize contact graph
    n = len(radii)
    graph = np.zeros((n, n), dtype=int)

        consts = build_lp_constants(centers, radii)
        
        # Solve LP
        Δr = solve_radius_lp(consts, radii, graph, α)
        
        # Update radii
        new_radii = apply_increment(radii, Δr)
        
        # Check if we can stop
        Δ_sum = np.sum(Δr)
        if Δ_sum < ε:
            break

        # Push feasibility
        centers, new_radii = push_feasibility(centers, new_radii)
        
        # Update global radii
        radii = new_radii

    return centers, radii
