import numpy as np
from scipy.optimize import minimize
from ortools.linear_solver import pywraplp

def generate_candidates(g, radius_levels):
    candidates = []
    grid_x = np.arange(0, 1 + g/2, g)
    grid_y = np.arange(0, 1 + g/2, g)
    for x in grid_x:
        for y in grid_y:
            for r in radius_levels:
                if (x - r >= -1e-12 and x + r <= 1 + 1e-12 and
                    y - r >= -1e-12 and y + r <= 1 + 1e-12):
                    candidates.append((x, y, r))
    return candidates

def build_milp(candidates):
    solver = pywraplp.Solver.CreateSolver('SCIP')
    if not solver:
        raise RuntimeError("Failed to create SCIP solver")

    n = len(candidates)
    z = [solver.IntVar(0, 1, f'z_{i}') for i in range(n)]

    # Grid point constraint
    grid_points = {}
    for i, (x, y, r) in enumerate(candidates):
        key = (round(x/g), round(y/g))
        if key not in grid_points:
            grid_points[key] = []
        grid_points[key].append(i)

    for points in grid_points.values():
        solver.Add(sum(z[i] for i in points) <= 1)

    # Exactly 26 circles
    solver.Add(sum(z[i] for i in range(n)) == 26)

    # Compute the maximum range of x and y
    # Use 0.5 as the max distance in any direction to guarantee safe bounds
    max_dist = 0.5

    # Non-overlap constraints using L-infinity
    for i in range(n):
        for j in range(i+1, n):
            x1, y1, r1 = candidates[i]
            x2, y2, r2 = candidates[j]

            dx = solver.NumVar(0, max_dist, f'dx_{i}_{j}')
            dy = solver.NumVar(0, max_dist, f'dy_{i}_{j}')

            solver.Add(dx >= x1 - x2)
            solver.Add(dx >= x2 - x1)
            solver.Add(dy >= y1 - y2)
            solver.Add(dy >= y2 - y1)

            # Quadrant constraints
            solver.Add(dx + dy >= r1 + r2)
            solver.Add(dx - dy >= r1 + r2)
            solver.Add(-dx + dy >= r1 + r2)
            solver.Add(-dx - dy >= r1 + r2)

            # Link auxiliary variables only when both are selected
            solver.Add(dx <= max_dist * (z[i] + z[j]))
            solver.Add(dy <= max_dist * (z[i] + z[j]))

    # Objective
    solver.Maximize(sum(candidates[i][2] * z[i] for i in range(n)))

    return solver, z

def solve_milp(solver, z):
    status = solver.Solve()
    if status != pywraplp.Solver.OPTIMAL:
        raise RuntimeError("MILP solver did not find optimal solution")
    selected = [i for i in range(len(z)) if z[i].solution_value() > 0.5]
    return selected

def extract_initial_solution(selected_indices, candidates):
    centers = np.array([[candidates[i][0], candidates[i][1]] for i in selected_indices])
    radii = np.array([candidates[i][2] for i in selected_indices])
    return centers, radii

def shrink_initial(radii, delta=1e-5):
    return radii - delta

def pack_variables(centers, radii):
    return np.hstack([centers.flatten(), radii])

def unpack_solution(z):
    n_circles = len(z) // 3
    centers = z[:2*n_circles].reshape(-1, 2)
    radii = z[2*n_circles:]
    return centers, radii

def define_objective(z):
    _, radii = unpack_solution(z)
    return -np.sum(radii)

def define_jacobian(z):
    _, radii = unpack_solution(z)
    grad = np.zeros_like(z)
    n_circles = len(radii)
    grad[2*n_circles:] = -1.0
    return grad

def define_constraints(z):
    centers, radii = unpack_solution(z)
    n = len(centers)
    cons = []

    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # r <= x
        cons.append({
            'fun': lambda z, i=i: r - x,
            'jac': lambda z, i=i: np.concatenate([np.zeros(2*(i+1)), -1.0])
        })
        # r <= 1 - x
        cons.append({
            'fun': lambda z, i=i: r - (1 - x),
            'jac': lambda z, i=i: np.concatenate([np.zeros(2*(i+1)), -1.0])
        })
        # r <= y
        cons.append({
            'fun': lambda z, i=i: r - y,
            'jac': lambda z, i=i: np.concatenate([np.zeros(2*(i+1)), -1.0])
        })
        # r <= 1 - y
        cons.append({
            'fun': lambda z, i=i: r - (1 - y),
            'jac': lambda z, i=i: np.concatenate([np.zeros(2*(i+1)), -1.0])
        })

    # Non-overlap constraints
    for i in range(n):
        for j in range(i+1, n):
            def pairwise_constraint(z, i=i, j=j):
                x1, y1 = centers[i]
                x2, y2 = centers[j]
                r1 = radii[i]
                r2 = radii[j]
                dx = x1 - x2
                dy = y1 - y2
                dist = np.sqrt(dx*dx + dy*dy)
                return (r1 + r2) - dist

            def jacobian_pairwise(z, i=i, j=j):
                x1, y1 = centers[i]
                x2, y2 = centers[j]
                r1 = radii[i]
                r2 = radii[j]
                dx = x1 - x2
                dy = y1 - y2
                dist_sq = dx*dx + dy*dy
                dist = np.sqrt(dist_sq)
                if dist < 1e-12:
                    dist = 1e-12
                ddx = dx / dist
                ddy = dy / dist
                jac = np.zeros_like(z)
                jac[2*i] = -(ddx)
                jac[2*i + 1] = -(ddy)
                jac[2*j] = (ddx)
                jac[2*j + 1] = (ddy)
                jac[2*n + i] = -1.0
                jac[2*n + j] = -1.0
                return jac

            cons.append({
                'fun': pairwise_constraint,
                'jac': jacobian_pairwise
            })
    return cons

def repair(centers, radii):
    for i in range(len(radii)):
        r = radii[i]
        x, y = centers[i]
        radii[i] = min(r, x, 1-x, y, 1-y)
        # Re-initialize to all zeros in case of conflict
        # But not required since the solver handles this already.

    # Optional post processing for overlaps:
    # For very tight convergence only
    for _ in range(5):
        any_shrink = False
        for i in range(len(radii)):
            for j in range(i+1, len(radii)):
                x1, y1 = centers[i]
                x2, y2 = centers[j]
                r1 = radii[i]
                r2 = radii[j]
                dx = x1 - x2
                dy = y1 - y2
                dist = np.sqrt(dx*dx + dy*dy)
                require = r1 + r2
                if dist < require:
                    diff = require - dist
                    delta = diff / 2.0
                    if r1 > delta and r2 > delta:
                        radii[i] -= delta
                        radii[j] -= delta
                        any_shrink = True
        if not any_shrink:
            break
    return centers, radii

def run_packing():
    g = 0.04
    radius_levels = np.arange(0.02, 0.26, 0.02)
    candidates = generate_candidates(g, radius_levels)

    solver, z = build_milp(candidates)
    selected = solve_milp(solver, z)
    centers0, radii0 = extract_initial_solution(selected, candidates)
    radii0 = shrink_initial(radii0)
    z0 = pack_variables(centers0, radii0)

    def obj(z):
        return define_objective(z)

    def jac(z):
        return define_jacobian(z)

    constraints = define_constraints(z0)

    bounds = [(0, 1) for _ in range(len(z0))]

    # Restore initial centers and radii to ensure correct unpacking order
    def resample():
        return z0.copy()

    resampled = resample()
    result = minimize(
        obj,
        resampled,
        method='trust-constr',
        jac=jac,
        bounds=bounds,
        constraints=constraints,
        options={'disp': False}
    )
    if not result.success:
        raise RuntimeError("Continuous optimization failed")

    centers, radii = unpack_solution(result.x)
    centers, radii = repair(centers, radii)
    sum_radii = np.sum(radii)

    # Run validation check to ensure correctness
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise RuntimeError("Post-optimization validation failed: " + msg)

    return centers, radii, sum_radii
