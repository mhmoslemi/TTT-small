import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial.distance import cdist

def build_lp_model(n_circles):
    # Build LP model for packing circles in unit square
    # Variables: x_i, y_i, r_i for i=0..n_circles-1
    # Total variables: 2*n_circles + n_circles = 3*n_circles

    # Create coefficient matrix A_ub and vector b_ub for constraints
    # Will collect all inequality constraints Ax <= b
    A_ub_list = []
    b_ub_list = []

    # Wall constraints: r_i <= x_i, 1-x_i, y_i, 1-y_i
    # That's 4 constraints per circle
    for i in range(n_circles):
        # r_i <= x_i  -->  -r_i + x_i >= 0
        row = [0]*n_circles + [0]*n_circles + [0]*n_circles
        row[i] = -1
        row[n_circles + i] = 1
        A_ub_list.append(row)
        b_ub_list.append(0)  # 0 <= x_i - r_i

        # r_i <= 1 - x_i  -->  r_i + x_i <= 1
        row = [0]*n_circles + [0]*n_circles + [0]*n_circles
        row[i] = 1
        row[n_circles + i] = 1
        A_ub_list.append(row)
        b_ub_list.append(1)  # x_i + r_i <= 1

        # r_i <= y_i  -->  -r_i + y_i >= 0
        row = [0]*n_circles + [0]*n_circles + [0]*n_circles
        row[i] = -1
        row[n_circles*2 + i] = 1
        A_ub_list.append(row)
        b_ub_list.append(0)  # y_i - r_i >= 0

        # r_i <= 1 - y_i  -->  r_i + y_i <= 1
        row = [0]*n_circles + [0]*n_circles + [0]*n_circles
        row[i] = 1
        row[n_circles*2 + i] = 1
        A_ub_list.append(row)
        b_ub_list.append(1)  # y_i + r_i <= 1

    # Pairwise separation constraints from axis alignment
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            # x_i + r_i <= x_j - r_j
            row = [0]*n_circles + [0]*n_circles + [0]*n_circles
            row[i] = 1
            row[j] = -1
            row[n_circles + i] = 1
            row[n_circles + j] = -1
            A_ub_list.append(row)
            b_ub_list.append(0)  # x_i + r_i - x_j + r_j <= 0

            # x_j + r_j <= x_i - r_i
            row = [0]*n_circles + [0]*n_circles + [0]*n_circles
            row[i] = -1
            row[j] = 1
            row[n_circles + i] = -1
            row[n_circles + j] = 1
            A_ub_list.append(row)
            b_ub_list.append(0)  # x_j + r_j - x_i + r_i <= 0

            # y_i + r_i <= y_j - r_j
            row = [0]*n_circles + [0]*n_circles + [0]*n_circles
            row[i] = 1
            row[j] = -1
            row[n_circles*2 + i] = 1
            row[n_circles*2 + j] = -1
            A_ub_list.append(row)
            b_ub_list.append(0)  # y_i + r_i - y_j + r_j <= 0

            # y_j + r_j <= y_i - r_i
            row = [0]*n_circles + [0]*n_circles + [0]*n_circles
            row[i] = -1
            row[j] = 1
            row[n_circles*2 + i] = -1
            row[n_circles*2 + j] = 1
            A_ub_list.append(row)
            b_ub_list.append(0)  # y_j + r_j - y_i + r_i <= 0

    # Objective function: minimize -sum(r_i)
    c = [0]*n_circles + [0]*n_circles + [-1]*n_circles

    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)

    # No equality constraints
    A_eq = np.array([])
    b_eq = np.array([])

    return c, A_ub, b_ub, A_eq, b_eq

def solve_lp(c, A_ub, b_ub, A_eq, b_eq):
    # Solve the linear program
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, method='highs', options={'presolve': True})
    if not res.success:
        raise ValueError("LP solver failed")
    
    # Extract solution
    sol = res.x
    n_circles = len(sol) // 3
    centers = np.array([[sol[i], sol[n_circles + i]] for i in range(n_circles)])
    radii = sol[2*n_circles:]

    # Determine which constraints are active (within tolerance)
    active_set = []

    # Check wall constraints
    for i in range(n_circles):
        # r_i == min(x_i, 1 - x_i, y_i, 1 - y_i)
        x_i, y_i = centers[i]
        r_i = radii[i]
        if abs(r_i - x_i) < 1e-9:
            active_set.append((i, 'wall', 'x_pos'))
        if abs(r_i - (1 - x_i)) < 1e-9:
            active_set.append((i, 'wall', 'x_neg'))
        if abs(r_i - y_i) < 1e-9:
            active_set.append((i, 'wall', 'y_pos'))
        if abs(r_i - (1 - y_i)) < 1e-9:
            active_set.append((i, 'wall', 'y_neg'))

    # Check pairwise constraints
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            # Check if any of the separation constraints are active
            d_x = abs(centers[i][0] - centers[j][0])
            d_y = abs(centers[i][1] - centers[j][1])
            r_i = radii[i]
            r_j = radii[j]

            # Check constraints from axially aligned separation
            # Check x separation
            if abs(d_x - (r_i + r_j)) < 1e-9:
                # Determine which one is left
                if centers[i][0] < centers[j][0]:
                    active_set.append((i, j, 'x_left'))
                else:
                    active_set.append((j, i, 'x_left'))

            # Check y separation
            if abs(d_y - (r_i + r_j)) < 1e-9:
                # Determine which one is lower
                if centers[i][1] < centers[j][1]:
                    active_set.append((i, j, 'y_bottom'))
                else:
                    active_set.append((j, i, 'y_bottom'))

    return centers, radii, active_set

def detect_diagonal_conflicts(centers, radii):
    n = len(centers)
    conflicts = []
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            r_sum = radii[i] + radii[j]
            if dist < r_sum:
                conflicts.append((i, j))

    return conflicts

def max_radius_projection(centers, radii):
    # For each circle, compute maximum radius that doesn't overlap other circles
    # and is constrained by square edges
    n = len(centers)
    new_radii = radii.copy()
    updated = True
    while updated:
        updated = False
        for i in range(n):
            x, y = centers[i]
            r_i = new_radii[i]
            min_dist = min(x, 1 - x, y, 1 - y)
            for j in range(n):
                if i != j:
                    d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                    min_dist = min(min_dist, d - new_radii[j])
            if min_dist > r_i:
                new_radii[i] = min_dist
                updated = True
    return new_radii
