import numpy as np
from scipy.optimize import minimize
from math import sqrt, ceil
import itertools

def build_hex_lattice(spacing):
    """Generate a hexagonal lattice of points inside unit square"""
    # Hexagonal close packing, rows offset by sqrt(3)/2 spacing
    # We generate enough to have 24 points inside [0,1]^2
    rows = ceil(1.0 / (spacing * 0.5 * sqrt(3))) + 1
    cols = ceil(1.0 / spacing) + 1
    points = []
    for i in range(rows):
        y = i * spacing * 0.5 * sqrt(3)
        if y > 1:
            break
        for j in range(cols):
            x = j * spacing + (i % 2) * spacing * 0.5
            if x > 1:
                break
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append([x, y])
    # Return first 24 points
    return np.array(points[:24])

def augment_with_extra_points(lattice):
    """Add two extra points near center and midpoint of long edge"""
    # Center
    center = np.array([0.5, 0.5])
    # Midpoint of edge - this will be near a corner but not exact, could be 0.5,0 or 0.5,1
    top_mid = np.array([0.5, 0.0])
    bottom_mid = np.array([0.5, 1.0])
    # Push them inward to avoid over-constraint
    top_mid[1] = 0.1
    bottom_mid[1] = 0.9
    # Use the first two points on the x-axis as similar but simpler approach
    augmented = np.vstack([lattice, center])
    # Add another point for shape balance
    augmented = np.vstack([augmented, bottom_mid])
    return augmented

def generate_contact_graphs(centers):
    """
    Generate plausible contact graphs with:
    - Each circle touching 0 to 4 walls
    - Some being tangent to their neighbours via hexagon logic
    """
    graphs = []
    n = len(centers)
    # Enumerate plausible wall contact masks (0 to 4 walls)
    # Filter out impossible cases where both left/right or both top/bottom are touched
    wall_masks = []
    for i in range(16):
        mask = [bool(i & (1 << j)) for j in range(4)]
        if mask[0] and mask[1]: continue  # both left/right

        wall_masks.append(mask)

    # Create 26 node graph, but only some pairs based on distance
    edges = []

        for j in range(i+1, n):
            # Only connect if close
            dist_sq = np.sum((centers[i] - centers[j]) ** 2)
            if dist_sq < 0.15**2:  # adjustable threshold
                edges.append((i, j))

    # Choose up to 20 edges
    # This is a small sample to demonstrate
    edge_subset = edges[:15]

    # Now enumerate plausible wall contact combinations
    # Pick 4 mask for first few circles
    wall_indices = list(range(min(12, n)))
    
    # Sample some combinations; swap this with full enumeration if needed
    for i in range(20):
        temp = []
        # Assume 22 circles touch at most 1 wall each with some subset touching wall
        for j in range(n):
            mask = [False, False, False, False]  # left right bottom top
            if j < 4:
                if j == 0:  # left wall
                    mask[0] = True
                elif j == 1:  # right wall
                    mask[1] = True

            temp.append(mask)
        # Add placeholder neighbors too

    return graphs

def solve_contact_system(centers, wall_mask, edges):
    """Solve the contact equations analytically given wall/tangency constraints.
    Returns radii or None if infeasible.
    """
    n = len(centers)
    # Initialize radii, all zero in beginning
    radii = np.zeros(n)
    # We simulate a trial assignment of 'fixed' values for wall-driven constraints
    # and try to assign for tangent constraint circle solving
    
    # Create a mapping of (i,j) pairs representing neighbour contacts
    tangencies = {}
    for (i, j) in edges:
        tangencies[(i, j)] = True
        tangencies[(j, i)] = True
    
    # Set wall-bound radii
    for i in range(n):
        if wall_mask[i][0]:  # left
            radii[i] = centers[i][0]
        elif wall_mask[i][1]:  # right
            radii[i] = 1 - centers[i][0]
        elif wall_mask[i][2]:  # bottom
            radii[i] = centers[i][1]
        elif wall_mask[i][3]:  # top
            radii[i] = 1 - centers[i][1]

    # Try solving tangencies using partial substitution
    # Simulate linear situation with relaxed tangent computation
    
    # Simple iterative assignment using bulk updating
    max_iter = 100
    for _ in range(max_iter):
        updated = False
        for i in range(n):
            # skip if already fixed by wall
            if wall_mask[i][0] or wall_mask[i][1] or wall_mask[i][2] or wall_mask[i][3]:
                continue
            total = 0.0
            count = 0
            for j in range(n):
                if (i, j) in tangencies or (j, i) in tangencies:
                    # for now we add a fixed guess based on simple constraint
                    total += radii[j]
                    count += 1
            # Simple heuristic, adjust based on distance to neighbours if any
            if count > 0:
                # use 0.5 * mean of touches as approximation
                radii[i] = 0.5 * max(0.001, total / count)
                updated = True
            else:
                # assign small positive value to interior ones
                radii[i] = 0.01
        if not updated:
            break
    
    # Now enforce basic constraints
    all_good = True
    for i in range(n):
        if radii[i] < 0:
            return None
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return None
    if not all_good:
        return None
    return radii

def setup_constraint_bounds(centers, radii):
    """Set up bounds for constraints"""
    # Create bounds for each circle: (min_x, max_x, min_y, max_y, min_r, max_r)
    bounds = []
    for i in range(len(centers)):
        x, y = centers[i]
        r = radii[i]
        bounds.append((max(0, x - r), min(1, x + r),
                      max(0, y - r), min(1, y + r),
                      max(0, r), 1.0))
    return bounds

def norm2(x):
    return np.sqrt(np.sum(x**2))

def compute_gradients(centers, radii):
    """Compute gradients of constraints for optimizer"""
    n = len(centers)
    # Wall gradients per circle
    wall_grads = []
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        g = np.zeros((6, len(centers) * 3 + 1))  # 6 constraints per circle
        # left, right, bottom, top, non-overlap to others
        g[0, 0 + i*3 + 0] = -1  # d(x)/d(x) = -1 (x >= r constraint)
        g[1, 0 + i*3 + 0] = 1   # d(x)/d(x) = 1 (x <= 1 - r constraint)
        g[2, 0 + i*3 + 1] = -1
        g[3, 0 + i*3 + 1] = 1
        wall_grads.append(g)

    # For non-overlap constraint, assume we generate all pairs
    non_overlap_grads = []
    for i in range(n):
        for j in range(i+1, n):
            xi, yi = centers[i]
            xj, yj = centers[j]
            ri, rj = radii[i], radii[j]
            dist = norm2(np.array([xi - xj, yi - yj]))
            diff = dist - (ri + rj)
            # force larger before convergence
            if diff < 0:
                # compute Jac of constraint: (dist - ri - rj)
                grad = np.zeros(len(centers) * 3 + 1)
                if dist > 1e-8:
                    dx = (xi - xj) / dist
                    dy = (yi - yj) / dist
                    grad[0 + i*3 + 0] = dx
                    grad[0 + i*3 + 1] = dy
                    grad[0 + i*3 + 2] = -1
                    grad[0 + j*3 + 0] = -dx
                    grad[0 + j*3 + 1] = -dy
                    grad[0 + j*3 + 2] = -1
                else:
                    # already overlapped, gradient is zero
                    pass
                # use [-c, 0, 0]
                non_overlap_grads.append(grad)

    # Each variable is [x, y, r] = (3n+1,) vector
    return wall_grads, non_overlap_grads

def local_optimize(centers, radii):
    """Use scipy local optimizer to improve packing with new radii/radius assignment"""
    n = len(centers)
    x0 = np.reshape(centers, -1)
    r0 = np.array(radii)
    th = np.hstack([x0, r0])
    
    def obj_func(x):
        # x = [x0, y0, ..., xn, yn, r0, ..., rn]
        xvec = x[:-n]  # first 2n elements
        rvec = x[-n:]
        return -np.sum(rvec)  # maximize sum
    
    # Box bounds for variables
    bounds = []
    # for each (x,y,r)
    for i in range(n):
        bounds.append((0.0, 1.0))   # x
        bounds.append((0.0, 1.0))   # y
        bounds.append((0.0, 0.5))   # r (arbitrary ceiling)
    bounds = bounds[:2*n] + [(0., 1.0)] * n   # make proper length

    def constraint_func(x):
        # x = [x0, y0, ..., xn, yn, r0, ..., rn]
        xvec = x[0:n*2:2]
        yvec = x[1:n*2:2]
        rvec = x[n*2:]
        # compute all non-overlap constraints (pairwise inequalities)
        g = []
        for i in range(n):
            for j in range(i+1, n):
                dist = sqrt((xvec[i] - xvec[j])**2 + (yvec[i] - yvec[j])**2)
                gap = dist - (rvec[i] + rvec[j])
                g.append(gap)
        # and check wall constraints
        for i in range(n):
            gp = [
                xvec[i] - rvec[i],             # left constraint
                1 - xvec[i] - rvec[i],         # right constraint
                yvec[i] - rvec[i],             # bottom constraint
                1 - yvec[i] - rvec[i]          # top constraint
            ]
            g.extend(gp)
        return np.array(g)

    constraints = {'type': 'ineq', 'fun': constraint_func,
                  'jac': lambda x: np.eye(len(constraint_func(x)))}
    try:
        result = minimize(obj_func, th,
                          bounds=bounds,
                          constraints=[constraints],
                          method='trust-constr',
                          tol=1e-12,
                          options={'maxiter': 500}
                          )
        if result.success:
            # extract refined centers and radii
            new_centers = result.x[:2*n].reshape(n, 2)
            new_radii = result.x[2*n:]
            return new_centers, new_radii
        else:
            return centers, radii  # not improved; just return original
    except Exception as e:
        return centers, radii
