import numpy as np
import cvxpy as cp
from scipy.spatial.distance import cdist


def build_leaf_problem(cell, k):
    """Builds an SOCP for a leaf cell."""
    x_min, y_min, x_max, y_max = cell
    # Variables for centers and radii
    centers = cp.Variable((k, 2))
    radii = cp.Variable(k, nonneg=True)

    # Constraints
    constraints = []
    # Cell boundary constraints
    for i in range(k):
        constraints += [
            centers[i, 0] - radii[i] >= x_min,
            centers[i, 0] + radii[i] <= x_max,
            centers[i, 1] - radii[i] >= y_min,
            centers[i, 1] + radii[i] <= y_max,
        ]

    # Non-overlap constraints
    for i in range(k):
        for j in range(i + 1, k):
            dist = cp.norm(centers[i] - centers[j])
            constraints += [dist >= radii[i] + radii[j]]

    # Objective
    objective = cp.Maximize(cp.sum(radii))

    prob = cp.Problem(objective, constraints)
    return prob, centers, radii


def solve_leaf(cell, k):
    """Solves an exact SOCP for a leaf cell."""
    prob, centers, radii = build_leaf_problem(cell, k)
    prob.solve(solver=cp.ECOS, verbose=False, abstol=1e-12, reltol=1e-12)
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        raise RuntimeError("Leaf SOCP infeasible")
    return centers.value, radii.value


def merge_clusters(cluster_a, cluster_b, shared_edge):
    """Merges two clusters based on shared edge."""
    # Combine all centers and radii
    all_centers = np.vstack([cluster_a[0], cluster_b[0]])
    all_radii = np.hstack([cluster_a[1], cluster_b[1]])
    k_total = len(all_radii)

    # Build merged problem variables
    merged_centers = cp.Variable((k_total, 2))
    merged_radii = cp.Variable(k_total, nonneg=True)

    # Constraints
    constraints = []

    # Boundary constraints
    for i in range(k_total):
        cx, cy = merged_centers[i]
        r = merged_radii[i]
        constraints += [
            cx - r >= 0,
            cx + r <= 1,
            cy - r >= 0,
            cy + r <= 1,
        ]

    # Non-overlap constraints
    for i in range(k_total):
        for j in range(i+1, k_total):
            dist = cp.norm(merged_centers[i] - merged_centers[j])
            constraints += [dist >= merged_radii[i] + merged_radii[j]]

    # Objectives
    objective = cp.Maximize(cp.sum(merged_radii))

    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, verbose=False, abstol=1e-12, reltol=1e-12)

    if prob.status not in ['optimal', 'optimal_inaccurate']:
        raise RuntimeError("Merge SOCP infeasible")

    return merged_centers.value, merged_radii.value


def migrate_circle(source_cluster, target_cluster):
    """Performs migration of one circle from source to target."""
    # Drop one from source cluster
    source_centers, source_radii = source_cluster
    # Remove smallest radius circle
    min_idx = np.argmin(source_radii)
    new_centers = np.delete(source_centers, min_idx, axis=0)
    new_radii = np.delete(source_radii, min_idx)
    # Add to target
    target_centers, target_radii = target_cluster
    new_target_centers = np.vstack([target_centers, source_centers[min_idx]])
    new_target_radii = np.hstack([target_radii, source_radii[min_idx]])
    return (new_centers, new_radii), (new_target_centers, new_target_radii)

def global_polish(centers, radii, max_iter=3, mu=1e-4):
    """Implements barrier-based gradient until convergence."""
    centers = np.copy(centers)
    radii = np.copy(radii)
    for _ in range(max_iter):
        # Compute all pairwise distances
        dist_matrix = cdist(centers, centers)
        n = len(centers)

        # Compute gradient components
        grad = np.zeros_like(centers)
        grad_r = np.zeros_like(radii)

        for i in range(n):
            # For radii gradients: watch all other circles
            for j in range(n):
                if i == j:
                    continue
                dij = np.linalg.norm(centers[i] - centers[j])
                diff = centers[i] - centers[j]
                # Barrier term
                if dij > radii[i] + radii[j]:
                    grad_r[i] += (1 / (dij - radii[i] - radii[j]))  # ring of constraint
                # Gradient of radii sum (attach parameter to penalize size)
                grad_r[i] -= 1  # Constant decrease per radius (maximize)

        # Below, we compute total energy corrected with barrier penalties
        # We simply apply large steps with truncation
        step_size = 0.01
        centers -= step_size * grad
        radii += step_size * grad_r
        radii[radii < 0.0] = 0.0
        centers[:, 0] = np.clip(centers[:, 0], radii, 1 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1 - radii)

    return centers, radii


def final_refine(centers, radii):
    """Final refinement using high-precision SOCP."""
    # Reduce problem size for final solve
    n = len(centers)
    prob, cps, cpr = build_leaf_problem((0, 0, 1, 1), n)
    # Some vars fixed for contacts (by calling solve)
    # Use current if good enough
    prob.solve(solver=cp.ECOS, verbose=False, abstol=1e-12, reltol=1e-12)
    if prob.status in ['optimal', 'optimal_inaccurate']:
        new_centers = cps.value
        new_radii = cpr.value
        return new_centers, new_radii

    # If above failed, fall back to repair
    new_centers = np.array(centers)
    new_radii = np.array(radii)
    # Safely clip small overlaps and repair
    dists = cdist(new_centers, new_centers)
    for i in range(n):
        for j in range(i+1, n):
            overlap = dists[i,j] - (new_radii[i]+new_radii[j])
            if overlap < -1e-12:
                # Adjust radii inward
                compensation = -overlap / 2.0
                new_radii[i] -= compensation
                new_radii[j] -= compensation
                if new_radii[i] < 0 or new_radii[j] < 0:
                    raise RuntimeError("Degenerate radius below zero")
    return new_centers, new_radii

def hierarchy(clusters):
    """Simplified hierarchical merge."""
    try:
        # Merge all pairs step-by-step for simplicity
        if len(clusters) <= 1:
            return clusters[0] if clusters else (np.empty((0,2)), np.array([]))
        
        merged = clusters[0]
        for i in range(1, len(clusters)):
            merged = merge_clusters(merged, clusters[i], None)
        return merged
    except Exception as e:
        raise RuntimeError("Hierarchical merge failure") from e
