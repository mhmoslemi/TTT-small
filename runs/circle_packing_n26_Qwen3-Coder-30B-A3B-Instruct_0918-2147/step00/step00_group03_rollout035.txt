def build_lp_model(n):

    # Variables: x1,y1,r1,x2,y2,r2,...,x26,y26,r26
    # Total: 26*2 + 26 = 78 variables

    x_indices = list(range(0, 2*n, 2))
    y_indices = list(range(1, 2*n, 2))
    r_indices = list(range(2*n, 3*n))

    c = np.zeros(3*n)

    # Constraints matrix A_ub and vector b_ub

            row[r_indices[j]] = 1
            A_ub.append(row)

            # y_j + r_j <= y_i - r_i => y_j - y_i + r_i + r_j <= 0
            row = [0]*3*n
            row[y_indices[i]] = -1

            row[r_indices[i]] = 1

            A_ub.append(row)
            b_ub.append(0)
        
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    return c, A_ub, b_ub, x_indices, y_indices, r_indices

def solve_lp(c, A_ub, b_ub, x_indices, y_indices, r_indices, max_iter=1000):
    # Solve the LP
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'maxiter': max_iter})
    
    if not result.success:
        raise RuntimeError("LP solver failed to find an optimal solution")

    centers = np.zeros((len(x_indices), 2))

        centers[i, 0] = result.x[x_indices[i]]
        centers[i, 1] = result.x[y_indices[i]]
        radii[i] = result.x[r_indices[i]]
        
    return centers, radii
