import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
from scipy.optimize import milp
from scipy.sparse import csr_matrix
import itertools

def init_candidate_grid(R, Δ):
    candidates = []
    for r in R:
        # Grid spacing such that adjacent points ensure non-overlap for same radius
        s = r * np.sqrt(2)
        # Only consider valid centers within margin r
        x_range = np.arange(r, 1 - r + Δ / 10, s)
        y_range = np.arange(r, 1 - r + Δ / 10, s)
        for x in x_range:
            for y in y_range:
                candidates.append((x, y, r))
    return candidates

def build_overlap_matrix(candidates):
    n = len(candidates)
    # Precompute pairwise distances for all possible pairs
    centers = np.array([[c[0], c[1]] for c in candidates])
    radii = np.array([c[2] for c in candidates])
    distances = cdist(centers, centers)
    overlap_matrix = np.zeros((n, n), dtype=bool)
    for i in range(n):
        for j in range(i+1, n):
            d = distances[i, j]
            r_sum = radii[i] + radii[j]
            overlap_matrix[i, j] = d < r_sum
            overlap_matrix[j, i] = d < r_sum
    return overlap_matrix

def solve_discrete_bip(candidates, overlap_matrix, target_sum):
    n = len(candidates)
    # Create binary variables z_k (k=0..n-1): 1 iff candidate k is selected
    # Binary variables vector z of size n
    # Constraint 1: exactly 26 selected
    # Constraint 2: no overlaps (z_i + z_j <= 1 for overlapping pairs)
    # Objective: maximize sum of radii, which is linear in z
    # Linear programming approach
    c = [-c[2] for c in candidates]
    # Constraint 1: sum(z) = 26
    A_eq = np.ones((1, n))
    b_eq = [26]
    # Constraint 2: z_i + z_j <= 1 for overlapping i, j
    rows = []
    cols = []
    data = []
    for i in range(n):
        for j in range(i+1, n):
            if overlap_matrix[i, j]:
                rows.append(len(data))
                cols.append(i)
                data.append(1)
                rows.append(len(data))
                cols.append(j)
                data.append(1)
    A_ub = csr_matrix((data, (rows, cols)), shape=(len(data) // 2, n))
    b_ub = np.ones(len(data) // 2)
    # Bounds: z_i binary
    bounds = [(0, 1)] * n
    # Solve
    try:
        result = milp(c=c, bounds=bounds, A_eq=A_eq, b_eq=b_eq, A_ub=A_ub, b_ub=b_ub, integrality=1)
        if result.success:
            selected_indices = np.where(np.round(result.x) > 0.5)[0]
            if len(selected_indices) == 26:
                selected_centers = np.array([[candidates[i][0], candidates[i][1]] for i in selected_indices])
                selected_radii = np.array([candidates[i][2] for i in selected_indices])
                return selected_centers, selected_radii
            else:
                raise RuntimeError("Failed to select exact 26 candidates")
        else:
            raise RuntimeError("Discrete optimization failed")
    except Exception as e:
        # Fallback to greedy assignment if MILP fails
        selected_indices = []
        remaining = list(range(n))
        radii = np.array([c[2] for c in candidates])
        while len(selected_indices) < 26 and remaining:
            # Greedily pick the largest potential candidate satisfying overlap constraints
            best_i = -1
            best_radius = -1
            for i in remaining:
                can_select = True
                for j in selected_indices:
                    dist = np.sqrt((candidates[i][0]-candidates[j][0])**2 + (candidates[i][1]-candidates[j][1])**2)
                    if dist < candidates[i][2] + candidates[j][2]:
                        can_select = False
                        break
                if can_select:
                    if candidates[i][2] > best_radius:
                        best_radius = candidates[i][2]
                        best_i = i
            if best_i >= 0:
                selected_indices.append(best_i)
                remaining.remove(best_i)
            else:
                break
        if len(selected_indices) < 26:
            raise RuntimeError("Could not reach 26 selections via greedy fallback")
        return np.array([[candidates[i][0], candidates[i][1]] for i in selected_indices]), \
               np.array([candidates[i][2] for i in selected_indices])

def max_radii_fixed_centres(centers, radii_initial):
    n = centers.shape[0]
    radii = radii_initial.copy()
    for i in range(n):
        # Compute feasible radius for circle i with all other circles fixed
        r_new = 1 - max(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
        for j in range(n):
            if i != j:
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                r_j = radii[j]
                r_new = min(r_new, dist - r_j)
        if r_new < 0:
            r_new = 0
        radii[i] = r_new
    return radii

def nlp_refine(centers, radii_initial):
    n = centers.shape[0]
    # Concatenate all variables in order: [x1, y1, x2, y2, ..., x26, y26, r1, r2, ..., r26]
    x0 = np.hstack([centers.ravel(), radii_initial])
    
    def f(x):
        # Minimize negative sum of radii
        return -np.sum(x[2 * n:])
    
    def df(x):
        # Derivative of f w.r.t. all variables
        grad = np.zeros_like(x)
        grad[2 * n:] = -1.0
        return grad
    
    def constraint_func(x):
        # Indexing: first 26 are first x coordinates, next 26 are first y coordinates, next 26 are radii
        X = x[:2 * n].reshape((n, 2))
        R = x[2 * n:]
        # Wall constraints
        wall_constraints = []
        for i in range(n):
            wall_constraints.append(R[i] - X[i, 0])
            wall_constraints.append(R[i] - (1 - X[i, 0]))
            wall_constraints.append(R[i] - X[i, 1])
            wall_constraints.append(R[i] - (1 - X[i, 1]))
        # Non-overlap constraints
        for i in range(n):
            for j in range(i+1, n):
                dx = X[i, 0] - X[j, 0]
                dy = X[i, 1] - X[j, 1]
                dist_square = dx*dx + dy*dy
                dist = np.sqrt(dist_square)
                overlap_residual = R[i] + R[j] - dist
                wall_constraints.append(overlap_residual)
        return np.array(wall_constraints)
    
    def dconstraint(x):
        # Analytic Jacobian of constraints
        # (includes wall and overlap)
        n = centers.shape[0] 
        J = []
        X = x[:2 * n].reshape((n, 2))
        R = x[2 * n:]
        num_constraints = 4*n + n*(n-1)//2
        
        for i in range(n): # wall boundary constraints
            row = np.zeros(2*n + n)
            row[2*i] = -1.0  # d/dx of (r - x)
            row[2*n + i] = 1.0  # d/dr of (r - x)
            J.append(row)
            
            row = np.zeros(2*n + n)
            row[2*i] = 1.0   # d/dx of (r + x - 1)
            row[2*n + i] = 1.0  # d/dr of (r + x - 1)
            J.append(row)
            
            row = np.zeros(2*n + n)
            row[2*i+1] = -1.0  # d/dy of (r - y)
            row[2*n + i] = 1.0  # d/dr of (r - y)
            J.append(row)
            
            row = np.zeros(2*n + n)
            row[2*i+1] = 1.0   # d/dy of (r + y - 1)
            row[2*n + i] = 1.0  # d/dr of (r + y - 1)
            J.append(row)
            
        for i in range(n):
            for j in range(i+1, n):
                # dist = sqrt((x_i-x_j)^2 + (y_i-y_j)^2)
                dx = X[i, 0] - X[j, 0]
                dy = X[i, 1] - X[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < 1e-10:
                    # Avoid division by zero
                    grad = np.zeros(2*n + n)
                    grad[2*i] = 0
                    grad[2*j] = 0
                    grad[2*i+1] = 0
                    grad[2*j+1] = 0
                    grad[2*n + i] = 1.0
                    grad[2*n + j] = 1.0
                    J.append(grad)
                else:
                    inv_dist = 1 / dist
                    grad = np.zeros(2*n + n)
                    # Gradient w.r.t. x_i
                    grad[2*i] = dx * -inv_dist
                    grad[2*n + i] = 1.0
                    # Gradient w.r.t. x_j
                    grad[2*j] = dx * inv_dist
                    grad[2*n + j] = 1.0
                    # Gradient w.r.t. y_i
                    grad[2*i+1] = dy * -inv_dist
                    grad[2*n + i] = 1.0
                    # Gradient w.r.t. y_j
                    grad[2*j+1] = dy * inv_dist
                    grad[2*n + j] = 1.0
                    J.append(grad)
        return np.array(J)
    
    # No bounds needed; constraints enforced within recurrence
    cons = {'type': 'ineq', 'fun': constraint_func, 'jac': dconstraint}
    
    # Optimizer parameters
    res = minimize(f, x0, method='trust-constr', jac=df, constraints=cons, options={'maxiter': 300, 'xtol': 1e-9, 'gtol': 1e-9})
    
    if res.success:
        x_final = res.x
        refined_centers = x_final[:2*n].reshape((n,2))
        refined_radii = x_final[2*n:]
        # Clamp extremely tiny radii to zero (numerics sometimes produce small negatives)
        np.clip(refined_radii, 0, None, out=refined_radii)
        return refined_centers, refined_radii
    else:
        # If optimization failed, return the initial points, albeit lossy
        return centers, radii_initial

    R = np.array([0.025, 0.035, 0.045, 0.055, 0.065, 0.075, 0.085, 0.095, 0.105, 0.115])
