import numpy as np
from scipy.optimize import minimize
from math import sqrt

def initial_lattice(n=26):
    """Generate an initial hexagonal lattice arrangement."""
    # We want a hexagonal packing with at least n circles
    # In a hexagonal lattice with spacing s, the number of circles in a square of side L is approximately (L/s)^2 * 2/sqrt(3)
    # We want ~26 circles, so try s = 0.25, which places circles in a roughly 5x5 grid with offset rows
    spacing = 0.25
    sqrt3 = sqrt(3)
    
    # Generate points in a hexagonal pattern
    points = []
    row = 0
    y = 0.0
    while len(points) < n:
        offset = (row % 2) * spacing / 2.0
        x = offset
        col = 0
        while x < 1.0 and len(points) < n:
            points.append((x, y))
            x += spacing
            col += 1
        y += spacing * sqrt3 / 2.0
        row += 1
        if y >= 1.0:
            break
    points = points[:n]
    
    centers = np.array(points)
    radii = np.zeros(n)
    
    for i in range(n):
        x, y = centers[i]
        # Compute admissible radius for this point within the unit square
        r_max = min(x, 1 - x, y, 1 - y)
        # Compute minimum distance to any other point (excluding itself)
        r_min = 1.0
        for j in range(n):
            if i != j:
                dist = np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
                if dist < r_min:
                    r_min = dist
        # The maximum radius is the smaller of (r_max) and (r_min / 2)
        if r_min < 1e-6:
            radius = r_max
        else:
            radius = min(r_max, r_min / 2.0)
        radii[i] = radius
    
    # Cap all radii at 0.5 to ensure they can fit in the unit square
    radii = np.minimum(radii, 0.5)
    return centers, radii

def unpack(z):
    """Reshape flat vector to centers and radii."""
    n = len(z) // 3
    centers = z[0:2*n].reshape((n, 2))
    radii = z[2*n:3*n]
    return centers, radii

def pack(centers, radii):
    """Flatten centers and radii to a vector."""
    n = centers.shape[0]
    z = np.zeros(3*n)
    z[0:2*n] = centers.flatten()
    z[2*n:3*n] = radii
    return z

def objective(z):
    """Minimize negative sum of radii."""
    _, radii = unpack(z)
    return -np.sum(radii)

def constraint_boundary(z):
    """Return list of boundary constraints (>=0)."""
    centers, radii = unpack(z)
    n = len(radii)
    constraints = []
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Left boundary
        constraints.append(x - r)
        # Right boundary
        constraints.append(1 - x - r)
        # Bottom boundary
        constraints.append(y - r)
        # Top boundary
        constraints.append(1 - y - r)
    return constraints

def constraint_nonoverlap(z):
    """Return list of non-overlap constraints (>=0)."""
    centers, radii = unpack(z)
    n = len(radii)
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.hypot(dx, dy)
            r_sum = radii[i] + radii[j]
            # Non-overlap means distance >= sum of radii
            constraints.append(dist - r_sum)
    return constraints

def repair(centers, radii):
    """Iteratively repair overlaps and boundary violations."""
    max_iter = 10
    for _ in range(max_iter):
        changed = False
        # Repair boundary violations
        for i in range(len(radii)):
            x, y = centers[i]
            r = radii[i]
            new_r = min(r, x, 1 - x, y, 1 - y)
            if new_r < r - 1e-9:
                radii[i] = new_r
                changed = True

        # Repair overlaps
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.hypot(dx, dy)
                r_sum = radii[i] + radii[j]
                if dist < r_sum - 1e-12:
                    overlap = r_sum - dist
                    change = overlap / 2.0
                    if radii[i] - change > 0 and radii[j] - change > 0:
                        radii[i] -= change
                        radii[j] -= change
                        changed = True
        if not changed:
            break
    return centers, radii

def run_packing():
    # Generate initial lattice
    centers, radii = initial_lattice(26)
    
    # Flatten to vector
    z = pack(centers, radii)
    
    # Build bounds
    bounds = [(0, 1) for _ in range(52)] + [(0, 0.5) for _ in range(26)]
    
    # Define constraints
    constraints = []
    constraints.extend([{'type': 'ineq', 'fun': lambda z: np.array(constraint_boundary(z))}])
    constraints.extend([{'type': 'ineq', 'fun': lambda z: np.array(constraint_nonoverlap(z))}])
    
    # Run optimization
    result = minimize(
        objective,
        z,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 500, 'ftol': 1e-9, 'disp': False},
    )
    if not result.success:
        raise RuntimeError("Optimization failed: " + result.message)
    
    # Unpack solution
    best_centers, best_radii = unpack(result.x)
    
    # Repair if needed
    best_centers, best_radii = repair(best_centers, best_radii)
    
    sum_radii = np.sum(best_radii)
    
    # Final validation
    valid, msg = validate_packing(best_centers, best_radii)
    if not valid:
        raise RuntimeError("Invalid packing after optimization: " + msg)
    
    # If sum is not satisfactory, perturb and retry
    if sum_radii < 2.636 - 1e-3:
        # Try again with slight random perturbation
        np.random.seed(42)
        perturbed_centers = best_centers + np.random.uniform(-0.02, 0.02, size=(26, 2))
        # Clip to [0,1]
        perturbed_centers[:, 0] = np.clip(perturbed_centers[:, 0], 0, 1)
        perturbed_centers[:, 1] = np.clip(perturbed_centers[:, 1], 0, 1)
        
        # Perform optimization again with this new starting point
        z_start = pack(perturbed_centers, best_radii)
        
        result = minimize(
            objective,
            z_start,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 500, 'ftol': 1e-9, 'disp': False},
        )
        if result.success:
            best_centers, best_radii = unpack(result.x)
            # Repair
            best_centers, best_radii = repair(best_centers, best_radii)
            sum_radii = np.sum(best_radii)
            # Final validation
            valid, msg = validate_packing(best_centers, best_radii)
            if not valid:
                raise RuntimeError("Invalid packing after second optimization: " + msg)
    
    return best_centers, best_radii, sum_radii
