import numpy as np
from scipy.optimize import minimize

def hex_lattice_points(n, d):
    """Generate at least n points on a hexagonal lattice with spacing d."""
    rows = int(np.ceil(np.sqrt(n * 2 / np.sqrt(3))))
    points = []
    for i in range(rows):
        y = i * d * np.sqrt(3) / 2
        if y > 1:
            break
        offset = (i % 2) * d / 2
        for j in range(int(np.ceil(1.0 / d)) + 1):
            x = j * d + offset
            if x > 1:
                break
            if len(points) >= n:
                break
            points.append([x, y])
        if len(points) >= n:
            break
    return np.array(points[:n])

    d = 0.2
    points = hex_lattice_points(26, d)
    # Center the lattice
    x_center = np.mean(points[:, 0])
    y_center = np.mean(points[:, 1])
    points[:, 0] -= x_center
    points[:, 1] -= y_center
    points[:, 0] += 0.5
    points[:, 1] += 0.5
    # Compute safe radius based on distance to boundaries
    min_radius = 1.0
    for x, y in points:
        r = min(x, 1 - x, y, 1 - y)
        if r < min_radius:
            min_radius = r
    # Set all radii to min_radius/1.5 to give some room for optimization
    radii = np.full(26, min_radius * 0.6)
    # Flatten centers and radii for optimization vector
    vars0 = []
    for i in range(26):
        vars0.append(points[i, 0])
        vars0.append(points[i, 1])
        vars0.append(radii[i])
    return np.array(vars0)

def objective(vars):
    """Minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def wall_constraints(vars):
    """Return list of wall inequality constraints (g(x) >= 0)."""
    result = []
    for i in range(26):
        x = vars[3*i]

        result.append(r - x)

        result.append(x + r - 1)
        # y - r >= 0
        result.append(r - y)
        # y + r <= 1
        result.append(y + r - 1)
    return result

def nonoverlap_constraints(vars):
    """Return list of non-overlap inequality constraints (g(x) >= 0)."""
    result = []
    for i in range(26):
        for j in range(i + 1, 26):

            yi = vars[3*i + 1]

            xj = vars[3*j]
            yj = vars[3*j + 1]
            rj = vars[3*j + 2]

            sum_radii_sq = (ri + rj)**2
            # Constraint: dist^2 >= (r_i + r_j)^2
            result.append(dist_sq - sum_radii_sq)
    return result
