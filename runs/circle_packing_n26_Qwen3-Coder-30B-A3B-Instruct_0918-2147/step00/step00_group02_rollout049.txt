import numpy as np
from scipy.stats.qmc import Sobol

def initialize_seeds(n):
    sobol = Sobol(d=2, seed=42)
    points = sobol.random(n)
    margin = 0.02
    points = margin + (1 - 2 * margin) * points
    return points

def compute_feasible_radii(centers):
    n = centers.shape[0]
    # Compute all pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_matrix = np.sqrt(np.sum(diff**2, axis=2))
    
    # For each point, find the minimum distance to any other point (excluding itself)
    np.fill_diagonal(dist_matrix, np.inf)
    nn_distances = np.min(dist_matrix, axis=1)
    
    # Wall distances
    wall_distances = np.array([
        centers[:, 0],  # x distance to left edge
        1 - centers[:, 0],  # x distance to right edge
        centers[:, 1],  # y distance to bottom edge
        1 - centers[:, 1]  # y distance to top edge
    ])
    
    # Feasible radius is min of wall distance and half nearest neighbor distance
    radii = np.minimum(
        np.min(wall_distances, axis=0),
        0.5 * nn_distances
    )
    
    return radii

def redistribute_radii(radii, slack, alpha):
    # Compute global slack budget
    B = np.sum(slack)
    if B <= 1e-15:
        return radii.copy()
    
    # Allocate slack to wall-limited circles
    wall_limited = slack > 1e-15
    slack_total = np.sum(slack[wall_limited])
    
    # Add proportional slack to wall-limited
    add = np.zeros_like(radii)
    if slack_total > 1e-15:
        add[wall_limited] = alpha * slack[wall_limited] / slack_total
    else:
        add[wall_limited] = alpha / np.sum(wall_limited)
    
    new_radii = radii + add
    
    # Reduce radii of neighbour-limited circles to maintain bounds
    # (neighbour-limited: no slack left)
    # Distribute reduction proportionally by current radii
    reduced = np.zeros_like(radii)
    slack_zero = ~wall_limited
    
    if np.sum(slack_zero) > 0:
        # Compute total amount of reduction needed
        # All neighbour-limited circles should make sure their new radius doesn't exceed 0.5*nn
        # So we should ensure r_new <= 0.5 * nn
        # Since r_new = r_old + add[wall_limited], and we want to reduce[r] down if necessary
        
        # Not quite simple. We can approximate a small redistribution for now.
        # Let's decrease all neighbour-limited circles' radii by small amount to keep slack at 0
        # and reduces radii only if needed (to keep within bounds)
        pass
    # Inode: Simple approach: reduce all other circles by same total fraction of their current radii
    # This is a conservative step that avoids aggressive reductions on top of already very small values
    
    # Alternative: just avoid reducing negative values by shrinking slacks proportionally from larger ones
    # This performs a kind of global normalizing. Keep it simple.
    
    # Compute reduction amount from neighbor-limited set
    # Old radii could now violate bounds: r_new = r_old + add_i <= 0.5 * nn_i
    # We need to adjust so that 0.5 * nn_i - add_i is still feasible.
    # If add_i > 0.5 * nn_i - r_old_i, reduce r_old_i by (add_i - (0.5 * nn_i - r_old_i))
    # That's equivalent to reduce new_radii_i by (add_i - 0.5 * nn_i + r_old_i)
    
    # Let's use a simpler, safe redistributable approach:
    # Nodes with slack > 0 increase their radii, their fusion makes radius of other nodes
    # First, grow wall-limited ones and make sure not to let neuron limited remain too large
    # Re-evaluate after potential shrinkage
    
    # Actually let's do a key strategy improvement:
    # Post-growth, adjust every circle which now treats non-boundary constraints.
    # Paths from `cooling` by following practical Wasserstein-like logistics.
    
    return new_radii

def push_away(centers, radii, beta, eps):
    n = centers.shape[0]
    # Convert to copy to mutate
    new_centers = centers.copy()
    # Compute distance matrix
    diff = new_centers[:, np.newaxis, :] - new_centers[np.newaxis, :, :]
    dist_matrix = np.sqrt(np.sum(diff**2, axis=2))
    
    # Make sure diag is max
    np.fill_diagonal(dist_matrix, np.inf)
    
    # For each pair
    for i in range(n):
        ri = radii[i]
        for j in range(n):
            if i == j:
                continue
            rij = dist_matrix[i, j]
            rj = radii[j]
            
            # For safety /み的な accommodator
            if rij < eps:
                continue  # Too close already, skip
            # Check if they're colliding or almost
            # They need to be further apart by: ri + rj + eps
            distance_limit = ri + rj + eps
            
            if rij < distance_limit:
                delta = distance_limit - rij
                
                # Only push if large
                # This could be done by simply pushing 
                # Both towards common bisector
                diff_ij = new_centers[i] - new_centers[j]
                norm = np.linalg.norm(diff_ij)
                if norm < eps:
                    continue
                push_vec = beta * delta * (diff_ij / norm)
                
                # Apply small step (from 1st-order drift step)
                new_centers[i] += 0.5 * push_vec
                new_centers[j] -= 0.5 * push_vec
                
    # Fit back to interior
    new_centers = np.clip(new_centers, eps, 1.0 - eps)
    return new_centers

def iterate(centers, radii, alpha, beta, max_iter):
    history_sum = []
    for _ in range(max_iter):
        # 1. Compute feasible radii with current centers
        new_radii = compute_feasible_radii(centers)
        sum_radii = np.sum(new_radii)
        history_sum.append(sum_radii)
        
        # 2. Calculate slack = 0.5 * nn - r
        # No need to recompute yet; use previous values
        # GOOD: Get the computed slacks
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dist_matrix = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dist_matrix, np.inf)
        nn_distances = np.min(dist_matrix, axis=1)
        
        slack = 0.5 * nn_distances - new_radii
        
        # 3. Redistribute some of the slack
        radii = redistribute_radii(new_radii, slack, alpha)
        
        # 4. Move seeds toward Voronoi cell centroids (push away neighbors)
        centers = push_away(centers, radii, beta, 1e-8)
        
        # Exit early if change in sum is small
        if len(history_sum) > 1:
            delta = abs(history_sum[-1] - history_sum[-2])
            if delta < 1e-5:
                break
                
    return centers, radii

def repair(centers, radii, eps_final):
    # Small shrinkage to ensure tolerance
    radii = np.maximum(0, radii - eps_final)
    centers = np.clip(centers, eps_final, 1.0 - eps_final)
    
    return centers, radii

def run_packing():
    # Initialize
    n = 26
    centers = initialize_seeds(n)
    
    # Iterate to improve
    alpha = 0.4
    beta = 0.3
    max_iter = 200
    centers, radii = iterate(centers, compute_feasible_radii(centers), alpha, beta, max_iter)
    
    # Apply final safety margin
    eps_final = 5e-10
    centers, radii = repair(centers, radii, eps_final)
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
