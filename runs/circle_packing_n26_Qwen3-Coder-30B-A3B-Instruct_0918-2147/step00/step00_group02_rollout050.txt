import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def initial_configuration(n=26):
    # Generate Sobol points in [0,1]^2
    # Using a simple quasi-Monte Carlo approach (low-discrepancy)
    # Generate points using a simple Halton or similar - here a simple pattern
    # For simplicity, this creates a structured but decent spread
    points = []
    for i in range(n):
        x = (i * 0.618033988749895) % 1.0
        y = (i * 0.381966011250105) % 1.0
        points.append([x, y])
    centers = np.array(points)
    
    # Compute initial radii
    radii = np.zeros(n)
    for i in range(n):
        x, y = centers[i]
        r = min(x, 1-x, y, 1-y)
        # Also consider proximity to neighbors
        distances = np.sqrt(np.sum((centers - centers[i])**2, axis=1))
        distances[i] = np.inf  # Ignore self-distance
        min_dist = np.min(distances)
        r = min(r, min_dist / 2.0)
        radii[i] = max(0, r)
    
    return centers, radii

def unpack(z):
    n = len(z) // 3
    centers = z[:2*n].reshape(n, 2)
    radii = z[2*n:]
    return centers, radii

def pack(centers, radii):
    return np.concatenate([centers.flatten(), radii])

def max_violation(centers, radii):
    n = len(radii)
    max_wall = 0
    max_overlap = 0
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        max_wall = max(max_wall, max(0, r - x), max(0, r - (1-x)), max(0, r - y), max(0, r - (1-y)))
    
    # Check overlaps
    D = cdist(centers, centers)
    for i in range(n):
        for j in range(i+1, n):
            d = D[i, j]
            r_sum = radii[i] + radii[j]
            if d < r_sum:
                overlap = r_sum - d
                max_overlap = max(max_overlap, overlap)
    return max(max_wall, max_overlap)

def energy_and_grad(z):
    centers, radii = unpack(z)
    n = len(radii)
    
    # Wall energy
    E_wall = 0.0
    grad_wall = np.zeros(z.shape)
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        dx = max(0, r - x)
        dy = max(0, r - (1 - x))
        dz = max(0, r - y)
        dw = max(0, r - (1 - y))
        E_wall += dx**2 + dy**2 + dz**2 + dw**2
        
        if dx > 0:
            grad_wall[2*i] -= 2 * dx
        if dy > 0:
            grad_wall[2*i + 1] -= 2 * dy
        if dz > 0:
            grad_wall[2*n + i] -= 2 * dz
        if dw > 0:
            grad_wall[2*n + i] -= 2 * dw
    
    # Pairwise overlap energy
    E_pair = 0.0
    grad_pair = np.zeros(z.shape)
    
    D = cdist(centers, centers)
    for i in range(n):
        for j in range(i+1, n):
            d = D[i, j]
            r_sum = radii[i] + radii[j]
            overlap = max(0, r_sum - d)
            E_pair += overlap**2
            
            if overlap > 0:
                # Gradient w.r.t. centers
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < 1e-12:  # Avoid division by zero
                    dist = 1e-12
                scale = 2 * overlap / dist
                
                grad_pair[2*i] += scale * dx
                grad_pair[2*i+1] += scale * dy
                grad_pair[2*j] -= scale * dx
                grad_pair[2*j+1] -= scale * dy
                
                # Gradient w.r.t. radii
                grad_pair[2*n+i] += 2 * overlap
                grad_pair[2*n+j] += 2 * overlap
    
    # Budget energy
    B = 3.0
    lambda_budget = 1000.0
    S = np.sum(radii)
    E_budget = lambda_budget * (B - S)**2
    grad_budget = np.zeros_like(z)
    grad_budget[2*n:] = -2 * lambda_budget * (B - S)
    
    E = E_wall + E_pair + E_budget
    grad = grad_wall + grad_pair + grad_budget
    return E, grad

def inner_optimize(z0, B, lambda_budget):
    result = minimize(
        lambda z: energy_and_grad(z)[0],
        z0,
        method='L-BFGS-B',
        jac=lambda z: energy_and_grad(z)[1],
        bounds=[(0,1)] * (2*len(z0)//3) + [(0, 0.5)] * (len(z0)//3),
        options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12}
    )
    return result.x

def max_feasible_budget(z_start):
    B = 3.0
    lambda_budget = 1000.0
    delta = 5e-4
    min_delta = 1e-6
    max_iter = 40
    iteration = 0
    best_feasible = z_start
    best_sum = 0.0
    
    while True:
        if delta < min_delta or iteration > max_iter:
            break
        z = inner_optimize(z_start, B, lambda_budget)
        centers, radii = unpack(z)
        v = max_violation(centers, radii)
        if v <= 1e-13:
            best_feasible = z
            best_sum = np.sum(radii)
            B += delta
        else:
            B -= delta
        iteration += 1
    return best_feasible

def postprocess(z):
    centers, radii = unpack(z)
    eps = 5e-10
    centers = np.clip(centers, eps, 1 - eps)
    radii = np.maximum(0, radii - eps)
    
    # Final correction loop
    done = False
    max_iter = 5
    for _ in range(max_iter):
        done = True
        D = cdist(centers, centers)
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                d = D[i, j]
                r_sum = radii[i] + radii[j]
                if d < r_sum:
                    overlap = r_sum - d
                    # Reduce both radii
                    reduction = overlap / 2 + eps
                    radii[i] = max(0, radii[i] - reduction)
                    radii[j] = max(0, radii[j] - reduction)
                    done = False
        if done:
            break
    return centers, radii
