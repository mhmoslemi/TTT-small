import numpy as np
from scipy.optimize import minimize

def hex_lattice_points(n, d):
    # Generate points in a hexagonal lattice
    # horizontal spacing is d, vertical spacing is d * sqrt(3)/2
    rows = int(np.ceil(np.sqrt(n * 2 / np.sqrt(3))))
    y_spacing = d * np.sqrt(3) / 2
    x_spacing = d
    points = []
    for i in range(rows):
        offset = (i % 2) * x_spacing / 2
        for j in range(int(np.ceil(n / rows)) + 1):
            x = offset + j * x_spacing
            y = i * y_spacing
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append([x, y])
                if len(points) >= n:
                    break
        if len(points) >= n:
            break
    return np.array(points[:n])

def initial_guess():
    # Start with a hexagonal lattice with a spacing of d
    n_circles = 26
    d = 0.25  # initial guess for spacing
    points = hex_lattice_points(n_circles, d)
    
    # Compute the radius that just touches neighbors
    # The minimum neighbor distance is d, so radius = d/2
    r_neighbor = d / 2
    
    # Compute the radius limited by walls
    r_wall = min(np.min(points[:, 0]), np.min(1 - points[:, 0]),
                 np.min(points[:, 1]), np.min(1 - points[:, 1]))
    # Safe to use the minimum of both
    r = min(r_neighbor, r_wall)
    
    # Ensure all circles fit within the square
    if r <= 0:
        r = 0.01  # fallback to small radius
    
    # Pack into 1D vector: x0, y0, r0, x1, y1, r1, ...
    vars = np.zeros(3 * n_circles)
    for i in range(n_circles):
        vars[3*i] = points[i, 0]
        vars[3*i+1] = points[i, 1]
        vars[3*i+2] = r
    
    return vars

def objective(vars):
    # Minimize the negative sum of radii
    return -np.sum(vars[2::3])

def wall_constraints(vars):
    # Return g(x) for wall constraints: g(x) >= 0
    # x - r >= 0  and  x + r <= 1
    # y - r >= 0  and  y + r <= 1
    result = []
    n_circles = len(vars) // 3
    for i in range(n_circles):
        x = vars[3*i]
        y = vars[3*i+1]
        r = vars[3*i+2]
        # x - r >= 0  ->  r - x <= 0
        result.append(r - x)
        # x + r <= 1  ->  x + r - 1 <= 0
        result.append(x + r - 1)
        # y - r >= 0  ->  r - y <= 0
        result.append(r - y)
        # y + r <= 1  ->  y + r - 1 <= 0
        result.append(y + r - 1)
    return result

def nonoverlap_constraints(vars):
    # Return g(x) for non-overlap constraints: g(x) >= 0
    # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    result = []
    n_circles = len(vars) // 3
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            x_i = vars[3*i]
            y_i = vars[3*i+1]
            r_i = vars[3*i+2]
            x_j = vars[3*j]
            y_j = vars[3*j+1]
            r_j = vars[3*j+2]
            dist_sq = (x_i - x_j)**2 + (y_i - y_j)**2
            sum_radii_sq = (r_i + r_j)**2
            # We want dist_sq - sum_radii_sq >= 0
            result.append(dist_sq - sum_radii_sq)
    return result

def run_packing():
    n_circles = 26
    # Initial guess
    x0 = initial_guess()
    
    # Bounds: x in [0,1], y in [0,1], r in [0,0.5]
    bounds = []
    for i in range(n_circles):
        bounds.append((0, 1))   # x_i
        bounds.append((0, 1))   # y_i
        bounds.append((0, 0.5)) # r_i
    bounds = tuple(bounds)
    
    # Constraints
    constraints = []
    # Wall constraints
    constraints.append({
        'type': 'ineq',
        'fun': wall_constraints
    })
    # Non-overlap constraints
    constraints.append({
        'type': 'ineq',
        'fun': nonoverlap_constraints
    })
    
    # Optimize
    res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                   constraints=constraints, options={'maxiter': 2000, 'ftol': 1e-10})
    
    if not res.success:

        centers = np.zeros((n_circles, 2))

        for i in range(n_circles):
            centers[i, 0] = x0[3*i]
            centers[i, 1] = x0[3*i+1]
            radii[i] = x0[3*i+2]
    else:

        radii = np.zeros(n_circles)
        for i in range(n_circles):
            centers[i, 0] = res.x[3*i]
