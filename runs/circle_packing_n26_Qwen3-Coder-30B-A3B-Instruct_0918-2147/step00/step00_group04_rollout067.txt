import numpy as np
import cvxpy as cp
from scipy.spatial.distance import pdist, squareform

def build_socp_problem(n):
    r = cp.Variable(n, nonneg=True)
    # We'll set up parameters for distances later
    dist_params = []
    constraints = []
    # Boundary constraints (these are linear)
    for i in range(n):
        constraints.append(r[i] <= 1)  # dummy for shape; actual will be set later
    # Non-overlap constraints will be added in solve_radii
    # We'll return a problem, and can update the distance parameters each time
    prob = cp.Problem(cp.Maximize(cp.sum(r)), constraints)
    return prob, r, dist_params

def solve_radii(centres, prob, r, dist_params):
    n = centres.shape[0]
    # Update distance parameters for non-overlap constraints
    # The distance matrix between centres
    distances = squareform(pdist(centres, 'euclidean'))
    
    # We configure each constraint
    old_constraints = prob.constraints
    new_constraints = []
    
    # Add back boundary constraints with actual values
    for i in range(n):
        x, y = centres[i]
        new_constraints.append(r[i] <= x)
        new_constraints.append(r[i] <= 1 - x)
        new_constraints.append(r[i] <= y)
        new_constraints.append(r[i] <= 1 - y)
    
    # Add non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            # Make this a constraint. We skip building parameters explicitly
            # because in this case, we need to re-set the constraints each time.
            # Using cvxpy's "Problem" with parameters is cleaner.
            dist_ij = distances[i, j]
            new_constraints.append(r[i] + r[j] <= dist_ij)

    # Set problem with new constraints
    prob.constraints = new_constraints
    
    # Solve
    prob.solve(solver=cp.ECOS, abstol=1e-12, reltol=1e-12)
    
    if prob.status != cp.OPTIMAL:
        raise RuntimeError(f"SOCP solver failed with status: {prob.status}")
    
    radii = r.value
    return radii

def energy(centres, prob, r, dist_params):
    radii = solve_radii(centres, prob, r, dist_params)
    return -np.sum(radii)

def random_move(state, rng, with_dual_bias=False):
    new_state = state.copy()
    n = len(new_state) // 2
    
    move_type = rng.choice(['single', 'cluster', 'edge', 'swap'])
    
    if move_type == 'single':
        idx = rng.integers(0, n)
        dx = rng.normal(0, 0.02)
        dy = rng.normal(0, 0.02)
        new_state[2*idx] = np.clip(new_state[2*idx] + dx, 0, 1)
        new_state[2*idx+1] = np.clip(new_state[2*idx+1] + dy, 0, 1)
    
    elif move_type == 'cluster':
        # pick 3-5 nearby centers
        num = rng.integers(3, 6)
        indices = rng.choice(n, size=num, replace=False)
        # Calculate mean and delta from that mean
        mean_x = np.mean([new_state[2*i] for i in indices])
        mean_y = np.mean([new_state[2*i+1] for i in indices])
        # small correlated move
        dx = rng.normal(0, 0.02)
        dy = rng.normal(0, 0.02)
        for i in indices:
            new_state[2*i] = np.clip(new_state[2*i] + dx, 0, 1)
            new_state[2*i+1] = np.clip(new_state[2*i+1] + dy, 0, 1)
    
    elif move_type == 'edge':
        if rng.random() < 0.1:
            idx = rng.integers(0, n)
            x, y = new_state[2*idx], new_state[2*idx+1]
            # push towards nearest wall
            if x < 0.5:
                new_state[2*idx] = np.clip(new_state[2*idx] - 0.02, 0, 1)
            else:
                new_state[2*idx] = np.clip(new_state[2*idx] + 0.02, 0, 1)
            if y < 0.5:
                new_state[2*idx+1] = np.clip(new_state[2*idx+1] - 0.02, 0, 1)
            else:
                new_state[2*idx+1] = np.clip(new_state[2*idx+1] + 0.02, 0, 1)
    
    else:  # swap (do not swap)
        pass
    
    return new_state

def anneal_chain(seed, rng):
    n = 26
    # Initial centers (hexagonal setup that will be refined)
    try:
        from scipy.spatial import distance_matrix
        init = _hex_lattice_init(n)
        if init.shape[0] < n:
            # extend with random
            extra = n - init.shape[0]
            rand = rng.random((extra, 2))
            init = np.vstack([init, rand])
    except Exception:
        init = rng.random((n, 2))
    
    state = init.flatten()
    
    T0 = 0.05
    alpha = 0.995
    T = T0
    best_state = state.copy()
    best_energy = energy(state.reshape(-1, 2), prob, r, dist_params)
    
    for step in range(100000):
        new_state = random_move(state, rng)
        Delta = energy(new_state.reshape(-1, 2), prob, r, dist_params) - best_energy
        if Delta < 0 or rng.random() < np.exp(-Delta / T):
            state = new_state
            if Delta < 0:
                best_energy = energy(state.reshape(-1, 2), prob, r, dist_params)
                best_state = state.copy()
        
        # cooling
        T *= alpha
        
        if T < 1e-6:
            break
    
    centres = best_state.reshape(-1, 2)
    return centres

# Helper: Initialize with a hexagonal grid of circles
def _hex_lattice_init(n):
    # Areasonable hexagonal setup within a square
    # Parameters: grid spacing
    spacing = 0.15
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))
    centres = []
    for i in range(rows):
        for j in range(cols):
            if len(centres) >= n:
                break
            x = (j + 0.5 * (i % 2)) * spacing + 0.05
            y = i * spacing * np.sqrt(3) / 2 + 0.05
            if 0 <= x <= 1 and 0 <= y <= 1:
                centres.append([x, y])
        if len(centres) >= n:
            break
    return np.array(centres[:n])

def local_refine(centres, prob, r, dist_params):
    old_sum = np.inf
    for _ in range(10):
        temps = []  # store temporaries
        try:
            radii = solve_radii(centres, prob, r, dist_params)
            sum_radii = np.sum(radii)
            if old_sum - sum_radii < 1e-5:
                break
            old_sum = sum_radii
        except Exception:
            break

    return centres, radii

def finalise(centres, prob, r, dist_params):
    # final strict check and refine
    radii = solve_radii(centres, prob, r, dist_params)
    # Clamp any violation
    for i in range(len(centres)):
        x, y = centres[i]
        r_i = radii[i]
        # ensure within bounds
        max_rad = min(x, 1-x, y, 1-y)
        if r_i > max_rad + 1e-12:
            radii[i] = max_rad
    # make sure no overlaps
    distances = squareform(pdist(centres, 'euclidean'))
    for i in range(len(centres)):
        for j in range(i + 1, len(centres)):
            dist_ij = distances[i, j]
            overlap = radii[i] + radii[j] - dist_ij
            if overlap > 1e-12:
                # adjust radii to stop overlap
                delta = overlap * 0.5
                radii[i] -= delta
                radii[j] -= delta
    return centres, radii

def run_packing():
    np.random.seed(42)
    # Build problem once (this is fixed)
    prob, r, dist_params = build_socp_problem(26)
    
    # Initialize 8 chains
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    for seed in range(8):
        rng = np.random.default_rng(seed)
        centres = anneal_chain(seed, rng)
        centres, radii = local_refine(centres, prob, r, dist_params)
        centres, radii = finalise(centres, prob, r, dist_params)
        sum_radii = np.sum(radii)
        if sum_radii > best_sum:
            best_sum = sum_radii
            best_centers = centres.copy()
            best_radii = radii.copy()

    return best_centers, best_radii, best_sum
