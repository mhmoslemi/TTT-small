import numpy as np
from scipy.optimize import linprog
from itertools import combinations

def generate_hex_lattice(h):
    sqrt3 = np.sqrt(3)
    points = []
    for j in range(int(1.0 / (h * sqrt3 / 2)) + 2):
        for i in range(int(1.0 / h) + 2):
            x = i * h + (j % 2) * (h / 2)
            y = j * h * sqrt3 / 2
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append([x, y])
    return np.array(points)

def build_neighbour_list(centers, max_dist):
    edges = []
    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist <= max_dist:
                edges.append((i, j, dist))
    return edges

def solve_radius_lp(selected_indices, R_wall, neighbour_list):
    n = len(selected_indices)
    if n == 0:
        return np.array([]), True

    # Build constraints
    # Wall constraints: 0 <= r_i <= R_wall[i]
    A_ub = []
    b_ub = []
    # Separation constraints: r_i + r_j <= d_ij
    for i, j, d_ij in neighbour_list:
        if i in selected_indices and j in selected_indices:
            idx_i = selected_indices.index(i)
            idx_j = selected_indices.index(j)
            a = np.zeros(n)
            a[idx_i] = 1
            a[idx_j] = 1
            A_ub.append(a)
            b_ub.append(d_ij)
    
    # Wall inequalities: r_i <= R_wall[i]
    for i, idx in enumerate(selected_indices):
        a = np.zeros(n)
        a[i] = -1
        A_ub.append(a)
        b_ub.append(-R_wall[idx])
    
    # Objective: maximize sum of radii
    c = -np.ones(n)

    # Solve
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    
    if not res.success:
        return np.array([]), False

    return res.x, True

def upper_bound(partial_set, remaining_indices, R_wall):
    ub = 0.0
    for i in partial_set:
        ub += R_wall[i]
    if len(remaining_indices) > 0:
        max_wall = max(R_wall[i] for i in remaining_indices)
        ub += max_wall * (26 - len(partial_set))
    return ub

def branch_and_bound(centers, R_wall, neighbour_list):
    best_sum = -np.inf
    best_selection = []
    best_radii = []

    def search(partial_set, remaining_indices):
        nonlocal best_sum, best_selection, best_radii
        if len(partial_set) == 26:
            radii, feasible = solve_radius_lp(partial_set, R_wall, neighbour_list)
            if feasible and len(radii) == 26:
                s = np.sum(radii)
                if s > best_sum:
                    best_sum = s
                    best_selection = partial_set.copy()
                    best_radii = radii.copy()
            return
        
        if len(partial_set) + len(remaining_indices) < 26:
            return

        ub = upper_bound(partial_set, remaining_indices, R_wall)
        if ub <= best_sum:
            return

        if len(remaining_indices) == 0:
            return

        # Select next candidate with largest wall radius
        max_idx = max(remaining_indices, key=lambda i: R_wall[i])
        
        # Branch: Include
        new_partial = partial_set + [max_idx]
        new_remaining = [i for i in remaining_indices if i != max_idx]
        search(new_partial, new_remaining)
        
        # Branch: Exclude
        search(partial_set, new_remaining)

    search([], list(range(len(centers))))
    return best_selection, best_radii
