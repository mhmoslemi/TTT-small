import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
from itertools import combinations
import math

def init_candidate_grid(R, Δ):
    candidates = []
    for r in R:
        s = r * np.sqrt(2)
        if s > 1:
            continue
        x_coords = np.arange(r, 1 - r + Δ / 10, s)
        y_coords = np.arange(r, 1 - r + Δ / 10, s)
        for x in x_coords:
            for y in y_coords:
                candidates.append((x, y, r))
    return candidates

def build_overlap_matrix(candidates):
    n = len(candidates)
    overlap = np.zeros((n, n), dtype=bool)
    for i in range(n):
        (x1, y1, r1) = candidates[i]
        for j in range(i+1, n):
            (x2, y2, r2) = candidates[j]
            dx = x1 - x2
            dy = y1 - y2
            dist = np.sqrt(dx*dx + dy*dy)
            if dist < r1 + r2:
                overlap[i, j] = True
                overlap[j, i] = True
    return overlap

def solve_discrete_bip(candidates, overlap_matrix, target_sum):
    # Greedy initialization
    selected = []
    selected_indices = set()
    radii = [c[2] for c in candidates]
    remaining = list(range(len(candidates)))
    total = 0
    while len(selected) < 26 and remaining:
        best = -1
        best_i = -1
        for i in remaining:
            if total + radii[i] < target_sum - 1e-6:
                if best == -1 or radii[i] > best:
                    best = radii[i]
                    best_i = i
        if best_i == -1:
            break
        selected.append(candidates[best_i])
        selected_indices.add(best_i)
        total += radii[best_i]
        new_remaining = []
        for i in remaining:
            if not overlap_matrix[best_i, i]:
                new_remaining.append(i)
        remaining = new_remaining
    while len(selected) < 26 and remaining:
        i = remaining[0]
        selected.append(candidates[i])
        selected_indices.add(i)
        total += radii[i]
        new_remaining = []
        for j in remaining:
            if not overlap_matrix[i, j]:
                new_remaining.append(j)
        remaining = new_remaining
    return np.array(selected)

def max_radii_fixed_centres(centers, radii_initial):
    n = len(centers)
    radii = radii_initial.copy()
    changed = True
    while changed:
        changed = False
        for i in range(n):
            min_radius = min(
                centers[i][0],
                1 - centers[i][0],
                centers[i][1],
                1 - centers[i][1]
            )
            min_radius = min(min_radius, min_radius)
            for j in range(n):
                if i == j:
                    continue
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx*dx + dy*dy)
                max_radius = dist - radii[j]
                if max_radius < min_radius:
                    min_radius = max_radius
            if min_radius > radii[i]:
                radii[i] = min_radius
                changed = True
    return radii

def nlp_refine(centers, radii_initial):
    n = len(centers)
    # Initial values
    x0 = np.concatenate([centers.flatten(), radii_initial])
    bounds = []
    # Position bounds [x1, y1, ..., x26, y26]
    for i in range(n):
        bounds.append((1e-12, 1 - 1e-12))  # x bounds
        bounds.append((1e-12, 1 - 1e-12))  # y bounds
    # Radius bounds [r1, ..., r26]
    for i in range(n):
        bounds.append((1e-12, 1 - 1e-12))

    # Objective
    def obj(x):
        return -np.sum(x[2*n:])

    # Constraints
    def wall_bounds(x):
        # for circle i:
        # constraint enforcement: x_i >= r_i, 1-x_i >= r_i, y_i >= r_i, 1-y_i >= r_i
        result = []
        for i in range(n):
            r = x[2*n + i]
            result.append(r - x[i])  # r - x_i >= 0
            result.append(r + x[i] - 1)  # r + x_i - 1 >= 0
            result.append(r - x[n + i])  # r - y_i >= 0
            result.append(r + x[n + i] - 1)  # r + y_i - 1 >= 0
        return np.array(result)

    def overlaps(x):
        result = []
        for i, j in combinations(range(n), 2):
            xi = x[i]
            yi = x[n + i]
            xj = x[j]
            yj = x[n + j]
            ri = x[2*n + i]
            rj = x[2*n + j]
            d = np.sqrt((xi - xj)**2 + (yi - yj)**2)
            result.append(d - ri - rj)  # d - ri - rj <= 0
        return np.array(result)

    cons = [
        {'type': 'ineq', 'fun': wall_bounds},
        {'type': 'ineq', 'fun': overlaps}
    ]

    res = minimize(obj, x0, method='trust-constr', bounds=bounds, constraints=cons,
                   options={'maxiter': 300, 'xtol': 1e-9, 'gtol': 1e-9})
    
    if res.success:
        x_sol = res.x
        new_centers = x_sol[:2*n].reshape((n, 2))
        new_radii = np.maximum(0, x_sol[2*n:])
        return new_centers, new_radii
    else:
        # If optimization fails, return initial guess
        return centers.copy(), radii_initial.copy()

def run_packing():
    radii_list = [
        0.025, 0.035, 0.045, 0.055, 0.065, 0.075, 0.085, 0.095, 0.105, 0.115
    ]
    Δ = 0.01
    candidates = init_candidate_grid(radii_list, Δ)
    overlap_matrix = build_overlap_matrix(candidates)
    selected = solve_discrete_bip(candidates, overlap_matrix, 2.636)
    centers = np.array([c[:2] for c in selected])
    radii = np.array([c[2] for c in selected])
    radii = max_radii_fixed_centres(centers, radii)
    centers_refined, radii_refined = nlp_refine(centers, radii)
    radii_refined = max_radii_fixed_centres(centers_refined, radii_refined)
    sum_radii = np.sum(radii_refined)
    assert sum_radii >= 2.636, f"Sum of radii {sum_radii} is less than 2.636"
    return centers_refined, radii_refined, sum_radii
