import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import math

def generate_lattice(s):
    """Generate a hexagonal lattice of candidate center points."""
    sqrt3 = math.sqrt(3)
    rows = int(1.0 / (s * sqrt3 / 2)) + 2
    cols = int(1.0 / s) + 2
    points = []
    for r in range(rows):
        y = r * s * sqrt3 / 2
        if y > 1 - s / 2:
            break
        for c in range(cols):
            if r % 2 == 0:
                x = c * s
            else:
                x = (c + 0.5) * s
            if x > 1 - s / 2:
                break
            if x >= s / 2 and y >= s / 2 and x <= 1 - s / 2 and y <= 1 - s / 2:
                points.append([x, y])
    return np.array(points)

def build_milp(P):
    """Construct the MILP model with binary selection and radius variables."""
    K = P.shape[0]
    z = cp.Variable(K, boolean=True)
    r = cp.Variable(K, nonneg=True)
    w = np.minimum(np.minimum(P[:, 0], 1 - P[:, 0]), np.minimum(P[:, 1], 1 - P[:, 1]))
    d = np.zeros((K, K))
    for i in range(K):
        for j in range(i + 1, K):
            d[i, j] = d[j, i] = np.linalg.norm(P[i] - P[j])
    
    constraints = [cp.sum(z) == 26]
    for k in range(K):
        constraints.append(r[k] <= w[k] * z[k])
    for i in range(K):
        for j in range(i + 1, K):
            M = 1.0
            constraints.append(d[i, j] >= r[i] + r[j] - M * (2 - z[i] - z[j]))
    objective = cp.Maximize(cp.sum(r))
    prob = cp.Problem(objective, constraints)
    
    def solve_lp(fixed_z=None):
        if fixed_z is not None:
            for k in range(K):
                if fixed_z[k] is not None:
                    z[k].value = fixed_z[k]
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            return None, None
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return r.value, prob.value
        else:
            return None, None
    
    return solve_lp

def branch_and_bound(milp_solver, node_limit=20000):
    """Custom branch and bound solver for the MILP."""
    best_solution = None
    best_value = -1e300
    bound_queue = []
    def add_node(z_assigned, node_bound):
        if node_bound > best_value:
            bound_queue.append((node_bound.copy(), z_assigned.copy()))
            bound_queue.sort(key=lambda x: -x[0])
    
    root_bound, _ = milp_solver()
    if root_bound is not None:
        add_node([None] * 26, root_bound)
    
    nodes_processed = 0
    while bound_queue and nodes_processed < node_limit:
        nodes_processed += 1
        current_bound, current_z = bound_queue.pop(0)
        if np.all([x is not None for x in current_z]):
            if current_bound > best_value:
                best_value = current_bound
                best_solution = current_z
            continue
        unfixed_indices = [i for i in range(len(current_z)) if current_z[i] is None]
        j = unfixed_indices[np.argmin([abs(0.5 - current_bound[i]) for i in unfixed_indices])]
        z_copy = current_z.copy()
        z_copy[j] = 0
        bound, _ = milp_solver(fixed_z=z_copy)
        if bound is not None:
            add_node(z_copy, bound)
        z_copy = current_z.copy()
        z_copy[j] = 1
        bound, _ = milp_solver(fixed_z=z_copy)
        if bound is not None:
            add_node(z_copy, bound)
        if nodes_processed % 100 == 0:
            pass
    
    if best_solution is None:
        raise RuntimeError("No feasible integer solution found")
    
    selected_indices = [i for i in range(len(best_solution)) if best_solution[i]]
    _, total_value = milp_solver(fixed_z=best_solution)
    return selected_indices, total_value

def refine_solution(centers_initial, radii_initial):
    """Refine by allowing free movement of centers and radii."""
    n = centers_initial.shape[0]
    def objective(x):
        radii = x[n*2:n*2+n]
        return -np.sum(radii)
    
    def constraint(x):
        centers = x[:n*2].reshape((n,2))
        radii = x[n*2:n*2+n]
        cons = []
        for i in range(n):
            cons.append(centers[i,0] - radii[i])  # left
            cons.append(1 - centers[i,0] - radii[i])  # right
            cons.append(centers[i,1] - radii[i])  # bottom
            cons.append(1 - centers[i,1] - radii[i])  # top
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[i,0] - centers[j,0]
                dy = centers[i,1] - centers[j,1]
                dist_sq = dx*dx + dy*dy
                sum_radii_sq = (radii[i] + radii[j])*(radii[i] + radii[j])
                cons.append(dist_sq - sum_radii_sq)
        return np.array(cons)
    
    x0 = np.concatenate([centers_initial.flatten(), radii_initial])
    bounds = [(0,1) for _ in range(n*2)] + [(0,None) for _ in range(n)]
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': constraint}, 
                      options={'ftol': 1e-12, 'maxiter': 1000})
    
    final_centers = result.x[:n*2].reshape((n,2))
    final_radii = np.maximum(result.x[n*2:n*2+n], 0)
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii

def run_packing():
    """Main function that returns centers, radii, sum of radii."""
    P = generate_lattice(s=0.13)
    milp_solver = build_milp(P)
    selected_indices, _ = branch_and_bound(milp_solver, node_limit=20000)
    centers_lattice = P[selected_indices]
    radii_lattice = np.minimum(np.minimum(centers_lattice[:, 0], 1 - centers_lattice[:, 0]),
                                np.minimum(centers_lattice[:, 1], 1 - centers_lattice[:, 1]))
    centers_refined, radii_refined, sum_radii = refine_solution(centers_lattice, radii_lattice)
    return centers_refined, radii_refined, sum_radii
