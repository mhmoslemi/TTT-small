import numpy as np
from scipy.stats.qmc import Sobol

def initialize_seeds(n):
    sobol = Sobol(d=2, seed=42)
    points = sobol.random(n)
    margin = 0.02
    points = margin + (1 - 2 * margin) * points
    return points

def compute_feasible_radii(centers):
    n = centers.shape[0]
    # Compute pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Set diagonal to large value to ignore self-distance
    np.fill_diagonal(dists, np.inf)
    
    # Get nearest neighbor distances
    nn_dists = np.min(dists, axis=1)
    
    # Compute wall distances
    wall_dists = np.min([centers[:, 0], 1 - centers[:, 0], centers[:, 1], 1 - centers[:, 1]], axis=0)
    
    # Compute feasible radii
    radii = np.minimum(0.5 * nn_dists, wall_dists)
    
    return radii

def redistribute_radii(radii, slack, alpha):
    # Total slack
    total_slack = np.sum(slack)
    
    # Increase radii of wall-limited circles
    delta = alpha * slack
    
    # Distribute the decrease among neighbor-limited circles (proportional to current radii)
    # Only reduce circles that have a slack of 0 (i.e., are neighbor-limited)
    neighbor_limited = (slack <= 1e-12)
    if np.sum(neighbor_limited) > 0:
        total_radius = np.sum(radii[neighbor_limited])
        if total_radius > 1e-15:
            # Reduce neighbor-limited circles proportionally to their current radii
            reduction = delta[neighbor_limited] * (radii[neighbor_limited] / total_radius)
            radii[neighbor_limited] -= reduction
    
    # Increase wall-limited circles
    wall_limited = (slack > 1e-12)
    radii[wall_limited] += delta[wall_limited]
    
    return radii

def push_away(centers, radii, beta, eps):
    n = centers.shape[0]
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Update centers based on neighbor exclusion
    new_centers = centers.copy()
    for i in range(n):
        for j in range(n):
            if i == j:
                continue
            d = dists[i, j]
            if d < radii[i] + radii[j] + eps:
                # Push i away from j
                vec = centers[i] - centers[j]
                vec_len = np.sqrt(np.sum(vec**2))
                if vec_len > eps:
                    delta = (radii[i] + radii[j] + eps - d)
                    new_centers[i] += beta * delta * (vec / vec_len)
    
    # Clip to boundary again
    new_centers = np.clip(new_centers, eps, 1 - eps)
    
    return new_centers

def iterate(centers, radii, alpha, beta, max_iter):
    for _ in range(max_iter):
        old_sum = np.sum(radii)
        
        # Compute slack
        nn_dists = np.min(
            np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :])**2, axis=2)),
            axis=1
        )
        nn_dists = np.where(nn_dists == np.inf, 0, nn_dists)
        wall_dists = np.min([centers[:, 0], 1 - centers[:, 0], centers[:, 1], 1 - centers[:, 1]], axis=0)
        max_radii = np.minimum(0.5 * nn_dists, wall_dists)
        slack = 0.5 * nn_dists - radii
        
        # Redistribute radii
        radii = redistribute_radii(radii, slack, alpha)
        
        # Push away
        centers = push_away(centers, radii, beta, 1e-8)
        
        # Recompute radii
        radii = compute_feasible_radii(centers)
        
        # Check for convergence
        new_sum = np.sum(radii)
        if np.abs(new_sum - old_sum) < 1e-5:
            break
    
    return centers, radii

def repair(centers, radii, eps_final):
    # Shrink radii
    radii = np.maximum(0, radii - eps_final)
    # Clip centers
    centers = np.clip(centers, eps_final, 1 - eps_final)
    
    # Ensure no overlaps
    n = centers.shape[0]
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i+1, n):
            overlap = dists[i, j] - radii[i] - radii[j]
            if overlap < -1e-12:
                reduction = (radii[i] + radii[j] - dists[i, j]) / 2 + eps_final
                radii[i] = max(0, radii[i] - reduction)
                radii[j] = max(0, radii[j] - reduction)
    
    return centers, radii
