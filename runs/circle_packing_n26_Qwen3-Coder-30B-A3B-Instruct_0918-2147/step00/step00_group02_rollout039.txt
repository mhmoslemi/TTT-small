import numpy as np
from scipy.stats.qmc import Sobol
from scipy.spatial.distance import cdist

def initialize_seeds(n):
    sobol = Sobol(d=2, seed=42)
    samples = sobol.random(n)
    # Clip to interior to ensure all seeds have positive initial radius
    margin = 0.02
    samples = margin + (1 - 2*margin) * samples
    return samples

def compute_feasible_radii(centers):
    n = centers.shape[0]
    # Euclidean distance matrix from scipy
    D = cdist(centers, centers)
    np.fill_diagonal(D, np.inf)
    nn = np.min(D, axis=1)
    # Wall distances: distance to each side of square
    wall_distances = np.array([
        centers[:, 0],  # distance to left edge
        1 - centers[:, 0],  # distance to right edge
        centers[:, 1],  # distance to bottom edge
        1 - centers[:, 1]  # distance to top edge
    ]).min(axis=0)
    # Feasible radius is min of wall distance and half of nearest neighbor distance
    radii = np.minimum(wall_distances, 0.5 * nn)
    return radii

def redistribute_radii(radii, slack, alpha):
    # total slack to redistribute
    B = np.sum(slack)
    if B <= 1e-12:
        # No redistribution possible
        return radii.copy()
    
    # Compute amount to add to each circle
    # Add alpha * slack[i] to circle i if it's wall-limited (slack[i] > 0)
    delta = np.zeros_like(radii)
    wall_limited = slack > 1e-12
    if np.sum(wall_limited) == 0:
        return radii.copy()
    
    delta[wall_limited] = alpha * slack[wall_limited]
    # Subtract this total from the neighbour-limited circles proportionally by radius
    total_add = np.sum(delta[wall_limited])
    if total_add > 1e-12:
        # For neighbour-limited circles, subtract proportional to current radius
        neighbour_limited = ~wall_limited
        radii_nl = radii[neighbour_limited]
        total_radii_nl = np.sum(radii_nl)
        if total_radii_nl > 1e-12:
            frac = total_add / total_radii_nl
            delta[neighbour_limited] -= frac * radii_nl
            # Ensure no negative radii
            delta[neighbour_limited] = np.maximum(delta[neighbour_limited], -radii_nl)
            
    new_radii = radii + delta
    new_radii = np.maximum(0.0, new_radii)
    return new_radii

def push_away(centers, radii, beta, eps):
    n = centers.shape[0]
    new_centers = centers.copy()
    for i in range(n):
        c_i = new_centers[i]
        r_i = radii[i]
        for j in range(n):
            if i == j:
                continue
            c_j = centers[j]
            r_j = radii[j]
            v_ij = c_i - c_j
            dist_ij = np.sqrt(np.sum(v_ij ** 2))
            if dist_ij < r_i + r_j + eps:
                # compute overlap amount
                overlap = (r_i + r_j + eps) - dist_ij
                if dist_ij > eps:
                    move = beta * overlap * (v_ij / dist_ij)
                else:
                    # near coincident: move in random direction
                    move = np.array([beta * overlap, 0.0])
                new_centers[i] += move
        # project back into unit square interior with a margin
        new_centers[i, 0] = np.clip(new_centers[i, 0], eps, 1 - eps)
        new_centers[i, 1] = np.clip(new_centers[i, 1], eps, 1 - eps)
    return new_centers

def iterate(centers, radii, alpha=0.4, beta=0.3, max_iter=200):
    old_sum = np.sum(radii)
    for _ in range(max_iter):
        # Compute slack for each circle: slack = 0.5*nn[i] - r[i]
        D = cdist(centers, centers)
        np.fill_diagonal(D, np.inf)
        nn = np.min(D, axis=1)
        wall_distances = np.array([
            centers[:, 0],
            1 - centers[:, 0],
            centers[:, 1],
            1 - centers[:, 1]
        ]).min(axis=0)
        # feasible radii
        r = np.minimum(wall_distances, 0.5 * nn)
        # Slack
        slack = 0.5 * nn - r
        # redistribute
        new_radii = redistribute_radii(radii, slack, alpha)
        # push away
        new_centers = push_away(centers, new_radii, beta, 1e-8)
        # compute new radii
        r_new = compute_feasible_radii(new_centers)
        new_sum = np.sum(r_new)
        if abs(new_sum - old_sum) < 1e-5:
            break
        centers, radii = new_centers, r_new
        old_sum = new_sum
    return centers, radii

def run_packing():
    n_circles = 26
    # 1. Initialize seeds
    centers = initialize_seeds(n_circles)
    # 2. Compute feasible radii
    radii = compute_feasible_radii(centers)
    # 3. Iterate to increase sum of radii
    centers, radii = iterate(centers, radii, alpha=0.4, beta=0.3, max_iter=100)
    # 4. Final repair step for validation
    eps_final = 5e-10
    radii = np.maximum(0.0, radii - eps_final)
    centers = np.clip(centers, eps_final, 1 - eps_final)
    return centers, radii, np.sum(radii)
