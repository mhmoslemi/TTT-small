import numpy as np
from scipy.optimize import linprog, minimize
from math import sqrt, ceil

def build_hex_lattice(d):
    # Generate a hexagonal lattice inside unit square
    # d is the desired distance between centers
    # generate at least 26 points

    # Hexagonal grid spacing
    spacing = d
    radius = spacing / (2 * sqrt(3))  # radius of inscribed circle in hexagon

    # compute the required number of rows and cols
    rows = ceil(1.0 / spacing)
    cols = ceil(1.0 / spacing)

    centers = []
    for i in range(rows):
        for j in range(cols):
            x = (j + 0.5 * (i % 2)) * spacing
            y = i * spacing * sqrt(3) / 2
            if x <= 1 and y <= 1 and len(centers) < 26:
                centers.append([x, y])

    # ensure we have at least 26 points
    while len(centers) < 26:
        # add points along the boundary
        x = 0.01 + len(centers) * 0.03
        y = 0.01
        if x <= 1:
            centers.append([x, y])

    # clip to 26 exactly
    return np.array(centers[:26], dtype=np.float64)

def lp_max_radii(centers):
    # For given centers return the optimal radii that maximize sum
    # This is a linear programming problem
    n = centers.shape[0]
    
    # Each circle has 4 boundary constraints (x-r >= 0, etc)
    # Plus n*(n-1)/2 overlap constraints
    # Total: 4n + n*(n-1)/2 constraints
    
    A_ub = []
    b_ub = []
    
    # Wall constraints for all circles
    for i in range(n):
        x, y = centers[i]
        # r_i <= x
        A_ub.append([0]*i + [-1] + [0]*(n-i-1) + [0]*4 + [0]*((n*(n-1)//2) - 4*i - 4))
        b_ub.append(-x)
        # r_i <= 1-x
        A_ub.append([0]*i + [-1] + [0]*(n-i-1) + [1]*4 + [0]*((n*(n-1)//2) - 4*i - 4))
        b_ub.append(1-x)
        # r_i <= y
        A_ub.append([0]*i + [-1] + [0]*(n-i-1) + [2]*4 + [0]*((n*(n-1)//2) - 4*i - 4))
        b_ub.append(-y)
        # r_i <= 1-y
        A_ub.append([0]*i + [-1] + [0]*(n-i-1) + [3]*4 + [0]*((n*(n-1)//2) - 4*i - 4))
        b_ub.append(1-y)

    # Non-overlap constraints
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            # r_i + r_j <= d_ij 
            d = np.linalg.norm(centers[i] - centers[j])
            row = [0]*n + [0]*4 + [0]*((n*(n-1)//2))
            row[i] = -1  # coefficient of r_i
            row[j] = -1  # coefficient of r_j
            row[4 + idx] = 1  # row number for constraint (d_ij)
            A_ub.append(row)
            b_ub.append(-d)
            idx += 1

    # Set up objective coefficients (negative because linprog minimizes)
    c = [-1.0] * n + [0]*4 + [0]*((n*(n-1)//2))

    # Convert to arrays
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    c = np.array(c)

    # bounds: r_i >= 0
    bounds = [(0.0, None)] * n + [(None, None)] * (4 + (n*(n-1)//2))

    # Solve LP
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', tol=1e-9)
    except:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')

    if not res.success:
        # fallback: set all radii to zero.
        # This should never happen with a valid initial configuration
        return np.zeros(n), 0.0

    # Extract radii from solution
    radii = res.x[:n]
    
    # Check for numerical errors that make radii negative
    radii = np.maximum(radii, 0.0)

    sum_radii = np.sum(radii)
    return radii, sum_radii

def perturb_centers(centers, sigma):
    # Add Gaussian noise, then clamp to [0,1]
    perturbed = centers + np.random.normal(0.0, sigma, centers.shape)
    perturbed[:, 0] = np.clip(perturbed[:, 0], 0.0, 1.0)
    perturbed[:, 1] = np.clip(perturbed[:, 1], 0.0, 1.0)
    return perturbed

def anneal_centers(initial_centers):
    np.random.seed(42)
    best_centers = initial_centers.copy()
    best_radii, best_sum = lp_max_radii(initial_centers)
    
    # Annealing schedule
    temp_schedule = [
        (0.02, 0.05, 2000),
        (0.01, 0.03, 4000),
        (0.005, 0.015, 3000),
        (0.002, 0.008, 2000)
    ]
    
    for temp, sigma, num_iter in temp_schedule:
        for _ in range(num_iter):
            new_centers = perturb_centers(best_centers, sigma)
            new_radii, new_sum = lp_max_radii(new_centers)
            
            # Accept with Metropolis criterion
            if new_sum >= best_sum or np.random.rand() < np.exp((new_sum - best_sum) / temp):
                best_centers = new_centers.copy()
                best_radii = new_radii.copy()
                best_sum = new_sum

    # Ensure we save the latest best if it's better
    return best_centers, best_radii, best_sum

def local_refine(centers, radii):
    # Refine with non-linear constraints using trust-constr
    n = centers.shape[0]
    
    # Objective: maximize sum of radii
    # We'll minimize negative sum to be consistent with scipy.optimize minimize
    def obj(x):
        # x contains [center_x1, center_y1, ..., center_xn, center_yn, radii1, ..., radii_n]
        center_vec = x[:2*n]
        radius_vec = x[2*n:]
        return - np.sum(radius_vec)

    # Constraints for trust-constr
    # 1. Circle boundary constraints: x_i - r_i >= 0, x_i + r_i <= 1, y_i - r_i >= 0, y_i + r_i <= 1
    def bound_cons(x):
        center_vec = x[:2*n]
        radius_vec = x[2*n:]
        # x_i - r_i >= 0
        c1 = center_vec[0::2] - radius_vec
        # x_i + r_i <= 1
        c2 = 1 - (center_vec[0::2] + radius_vec)
        # y_i - r_i >= 0
        c3 = center_vec[1::2] - radius_vec
        # y_i + r_i <= 1
        c4 = 1 - (center_vec[1::2] + radius_vec)
        return np.concatenate([c1, c2, c3, c4])

    # 2. Circle non-overlap constraints: ||c_i - c_j||^2 >= (r_i + r_j)^2
    def overlap_cons(x):
        center_vec = x[:2*n]
        radius_vec = x[2*n:]
        result = []
        for i in range(n):
            for j in range(i+1, n):
                dx = center_vec[2*i] - center_vec[2*j]
                dy = center_vec[2*i+1] - center_vec[2*j+1]
                dist_sq = dx*dx + dy*dy
                r_sum = radius_vec[i] + radius_vec[j]
                result.append(dist_sq - r_sum * r_sum)
        return np.array(result)

    # Bounds: center coordinates between [0,1], radii between [0, 1]
    bounds = []
    
    # Centers: [0,1] x 2n
    for _ in range(2*n):
        bounds.append((0.0, 1.0))
    
    # Radii: [0, 1] * n (the maximum possible is 1, but realistic may be 0.5)
    for _ in range(n):
        bounds.append((0.0, 1.0))
    
    # Actual x vector:
    x0 = np.concatenate([centers.flatten(), radii])

    # Constraint objects
    bound_constraint = {'type': 'ineq', 'fun': bound_cons}
    overlap_constraint = {'type': 'ineq', 'fun': overlap_cons}

    # Optimize
    try:
        res = minimize(obj, x0, method='trust-constr', 
                       constraints=[bound_constraint, overlap_constraint],
                       bounds=bounds, options={'maxiter': 500, 'disp': False})
    except:
        res = minimize(obj, x0, method='trust-constr', 
                       constraints=[bound_constraint, overlap_constraint],
                       bounds=bounds, options={'maxiter': 100, 'disp': False})

    # Extract final result
    if res.success:
        x_final = res.x
        refined_centers = x_final[:2*n].reshape(n, 2)
        refined_radii = x_final[2*n:]
    else:
        # On failure return original 
        refined_centers = centers
        refined_radii = radii
    
    # Reapply LP to get optimal alignment with constraints
    _, sum_radii = lp_max_radii(refined_centers)
    
    return refined_centers, refined_radii, sum_radii
