import numpy as np

def hex_lattice(d):
    """Generate hexagonal lattice points within [0,1]^2."""
    sqrt3 = np.sqrt(3)
    dy = sqrt3 * d / 2
    points = []
    y = 0
    row = 0
    while y <= 1 + 1e-12:
        x = 0
        offset = (row % 2) * d / 2
        while x <= 1 + 1e-12:
            points.append((x, y))
            x += d
        y += dy
        row += 1
    return np.array(points)

def add_border_circles(hex_points, d):
    """Add 6 border-enhanced circles to the hexagonal core."""
    r = d / 4
    border_points = [
        (r, r), (1 - r, r), (r, 1 - r), (1 - r, 1 - r),
        (0.5, r), (0.5, 1 - r)
    ]
    return np.vstack([hex_points[:20], border_points])

def initial_radii(points):
    """Compute a uniform initial safe radius for all points."""
    # Wall distances for each point
    wall_distances = np.array([
        points[:, 0], 1 - points[:, 0],
        points[:, 1], 1 - points[:, 1]
    ]).min(axis=0)
    # Pairwise distances
    distances = np.linalg.norm(
        points[:, None, :] - points[None, :, :], axis=2
    )
    # Minimum distance to any other point (excluding self)
    min_distances = np.min(distances + np.eye(distances.shape[0]) * 1e10, axis=1)
    # Take minimum of wall and half pairwise
    radii = np.minimum(wall_distances, min_distances / 2)
    return np.min(radii)

def inflate_radii(centers, radii):
    """Inflate radii to their maximum feasible values."""
    new_radii = radii.copy()
    distances = np.linalg.norm(
        centers[:, None, :] - centers[None, :, :], axis=2
    )
    for i in range(centers.shape[0]):
        # Wall constraints
        wall_caps = [
            centers[i, 0],
            1 - centers[i, 0],
            centers[i, 1],
            1 - centers[i, 1]
        ]
        # Pairwise constraints
        pairwise_caps = []
        for j in range(centers.shape[0]):
            if i != j:
                pairwise_caps.append(distances[i, j] - radii[j])
        caps = wall_caps + pairwise_caps
        new_radius = np.min(caps)
        if new_radius > radii[i] + 1e-6:
            new_radii[i] = new_radius
    return new_radii

def relax_centers(centers, radii, steps=10):
    """Perform force-based relaxation of centers."""
    κ = 0.02
    λ = 0.01
    α = 0.5
    β = 0.02
    γ = 0.02
    c = centers.copy()
    for _ in range(steps):
        f_total = np.zeros_like(c)
        # Repulsive forces
        distances = np.linalg.norm(
            c[:, None, :] - c[None, :, :], axis=2
        )
        for i in range(c.shape[0]):
            for j in range(c.shape[0]):
                if i == j:
                    continue
                dist = distances[i, j]
                delta = radii[i] + radii[j] - dist
                if dist > 1e-10:
                    f_ij = κ * max(0, β - dist) * (c[i] - c[j]) / dist
                    f_total[i] += f_ij
        # Wall forces
        for i in range(c.shape[0]):
            f_wall_x = 0
            f_wall_y = 0
            if c[i, 0] - radii[i] < γ:
                f_wall_x += -λ * max(0, γ - (c[i, 0] - radii[i]))
            if 1 - c[i, 0] - radii[i] < γ:
                f_wall_x += λ * max(0, γ - (1 - c[i, 0] - radii[i]))
            if c[i, 1] - radii[i] < γ:
                f_wall_y += -λ * max(0, γ - (c[i, 1] - radii[i]))
            if 1 - c[i, 1] - radii[i] < γ:
                f_wall_y += λ * max(0, γ - (1 - c[i, 1] - radii[i]))
            f_total[i] += [f_wall_x, f_wall_y]
        # Update positions
        c += α * f_total
        # Clamp to [r, 1-r]
        c[:, 0] = np.clip(c[:, 0], radii, 1 - radii)
        c[:, 1] = np.clip(c[:, 1], radii, 1 - radii)
    return c

def run_packing():

    d = 0.18  # spacing for ~20 core circles
    hex_points = hex_lattice(d)
    centers = add_border_circles(hex_points, d)

    r0 = initial_radii(centers)
    radii = np.full(centers.shape[0], r0)
    # Step 3: Iterative optimization
    MAX_OUTER = 30
    previous_sum = np.sum(radii)

        centers = relax_centers(centers, radii)
        current_sum = np.sum(radii)
