def solve_leaf(cell_coords, k):
    # cell_coords: (x_min, y_min), (x_max, y_max)
    x_min, y_min = cell_coords[0]
    x_max, y_max = cell_coords[1]
    # Create variables for k circles in cell
    centers = []
    radii = []
    for i in range(k):
        x = cp.Variable()
        y = cp.Variable()
        r = cp.Variable()
        centers.append((x, y))
        radii.append(r)

    # Constraints for bounding box
    constraints = []
    for i in range(k):
        x, y = centers[i]
        r = radii[i]
        constraints.append(x - r >= x_min)
        constraints.append(x + r <= x_max)
        constraints.append(y - r >= y_min)
        constraints.append(y + r <= y_max)

    # Non-overlap constraints
    for i in range(k):
        for j in range(i):
            x_i, y_i = centers[i]
            x_j, y_j = centers[j]
            r_i = radii[i]
            r_j = radii[j]
            # Second-order cone constraint: ||(x_i - x_j, y_i - y_j)|| >= r_i + r_j
            dist = cp.norm(cp.vstack([x_i - x_j, y_i - y_j]), 2)
            constraints.append(dist >= r_i + r_j)

    # Objective
    objective = cp.Maximize(cp.sum(radii))
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.ECOS, abstol=1e-12, reltol=1e-12)

    # Extract solution
    x_vals = [c[0].value for c in centers]
    y_vals = [c[1].value for c in centers]
    r_vals = [r.value for r in radii]
    return np.array(list(zip(x_vals, y_vals))), np.array(r_vals)

def global_polish(centers, radii, max_iter=5):
    # Use simple projected gradient ascent with dual barrier penalties
    n = len(centers)
    # Variables: x_i, y_i, r_i
    x = centers[:, 0].copy()
    y = centers[:, 1].copy()
    r = radii.copy()
    mu = 1e-5  # barrier weight

    # Define penalty function
    def penalty(x, y, r):
        p = -np.sum(r)
        # Apply log-barrier on boundary constraints
        p -= mu * np.sum(np.log(np.maximum(x - r, 1e-12)) +
                        np.log(np.maximum(1 - x - r, 1e-12)) +
                        np.log(np.maximum(y - r, 1e-12)) +
                        np.log(np.maximum(1 - y - r, 1e-12)))
        # Apply log-barrier on overlap constraints
        for i in range(n):
            for j in range(i):
                d = np.sqrt((x[i]-x[j])**2 + (y[i]-y[j])**2)
                p -= mu * np.log(np.maximum(d - r[i] - r[j], 1e-12))
        return p + 1e-10  # Prevent zero

    for _ in range(max_iter):
        # Gradient of penalty
        # This is approximate
        dp_dr = np.ones(n) - mu * (1.0 / (x - r + 1e-12) +
