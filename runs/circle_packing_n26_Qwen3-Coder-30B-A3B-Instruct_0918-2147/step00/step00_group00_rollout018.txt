import numpy as np
import random

def make_hex_seed(n, margin):
    # Generate hexagonal seed pattern
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))
    
    centers = []
    for i in range(rows):
        for j in range(cols):
            if len(centers) >= n:
                break
            x = margin + (j + 0.5 * (i % 2)) * (1 - 2 * margin) / cols
            y = margin + i * (1 - 2 * margin) / rows
            centers.append([x, y])
        if len(centers) >= n:
            break
    
    centers = np.array(centers[:n])
    
    # Assign radii based on distance to walls
    radii = []
    for center in centers:
        x, y = center
        min_dist = min(x, 1 - x, y, 1 - y)
        # Prefer larger radii near center
        r = min_dist * 0.6 + 0.01  # Ensure some minimum
        radii.append(r)
    
    radii = np.array(radii)
    
    # Scale so that min radius is at least 0.015
    min_radius = np.min(radii)
    if min_radius < 0.015:
        scale = 0.015 / min_radius
        radii *= scale
        centers[:, 0] *= scale
        centers[:, 1] *= scale
    
    return centers, radii

def max_feasible_radius(i, centers, radii):
    # Compute the maximum feasible radius for circle i
    x, y = centers[i]
    r = radii[i]
    
    # Wall constraints
    d_L = x
    d_R = 1 - x
    d_B = y
    d_T = 1 - y
    
    min_dist = min(d_L, d_R, d_B, d_T)
    
    # Neighbour constraints
    for j in range(len(centers)):
        if i != j:
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            d_ij = dist - radii[j]
            if d_ij < min_dist:
                min_dist = d_ij
    
    # Return safe value
    return min_dist

def inflate_radii(centers, radii, alpha):
    old_sum = np.sum(radii)
    mean_r = np.mean(radii)
    delta = alpha * mean_r
    
    # Apply uniform inflation
    new_radii = radii + delta
    
    # Clamp each radius to its feasible maximum
    for i in range(len(centers)):
        max_r = max_feasible_radius(i, centers, new_radii)
        new_radii[i] = min(new_radii[i], max_r)
    
    new_sum = np.sum(new_radii)
    return new_radii, new_sum - old_sum

def anneal_centers(centers, radii, temp0, gamma, sigma0, steps):
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = np.sum(radii)
    
    temp = temp0
    sigma = sigma0
    
    for step in range(steps):
        i = random.randint(0, len(centers) - 1)
        
        # Save old state
        old_center = centers[i].copy()
        old_radius = radii[i]
        
        # Propose new center
        displacement = np.random.normal(0, sigma, 2)
        centers[i] += displacement
        
        # Make sure it stays in bounds
        centers[i][0] = np.clip(centers[i][0], 0, 1)
        centers[i][1] = np.clip(centers[i][1], 0, 1)
        
        # Compute new feasible radius for this circle
        new_radius = max_feasible_radius(i, centers, radii)
        
        # Accept or reject
        delta_radius = new_radius - old_radius
        if delta_radius > 0 or random.random() < np.exp(delta_radius / temp):
            radii[i] = new_radius
        else:
            centers[i] = old_center
        
        # Occasionally do global shift
        if step % 50 == 0:
            shift = np.random.normal(0, sigma/10, 2)
            for c in centers:
                c += shift
                c[0] = np.clip(c[0], 0, 1)
                c[1] = np.clip(c[1], 0, 1)
        
        # Update best
        current_sum = np.sum(radii)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
        
        # Decay temperature and sigma
        temp *= gamma
        sigma *= gamma
    
    return best_centers, best_radii, best_sum

def mixed_cycle(centers, radii, params):
    inflation_tolerance = params['inflation_tolerance']
    stagnation_threshold = params['stagnation_threshold']
    sa_steps_per_cycle = params['sa_steps_per_cycle']
    cycles_before_stop = params['cycles_before_stop']
    alpha_start = params['alpha_start']
    alpha_decay = params['alpha_decay']
    
    alpha = alpha_start
    stagnation_count = 0
    cycles_count = 0
    prev_sum = np.sum(radii)
    
    while cycles_count < cycles_before_stop:
        # Inflation phase
        for round_ in range(100):
            new_radii, delta_sum = inflate_radii(centers, radii, alpha)
            radii = new_radii
            
            if delta_sum < inflation_tolerance:
                stagnation_count += 1
                if stagnation_count > stagnation_threshold:
                    break
            else:
                stagnation_count = 0
            
            prev_sum = np.sum(radii)
        else:
            break  # Did not stagnate, keep going...
        
        # Now SA repositioning
        temp0 = 0.02
        gamma = 0.97
        sigma0 = 0.03
        best_centers, best_radii, best_sum = anneal_centers(centers, radii, temp0, gamma, sigma0, sa_steps_per_cycle)
        centers = best_centers
        radii = best_radii
        
        # Check if module converged
        current_sum = np.sum(radii)
        if abs(current_sum - prev_sum) / prev_sum < 1e-6:
            cycles_count += 1
        else:
            cycles_count = 0
        
        alpha *= alpha_decay
    
    return centers, radii

def polish_radii(centers, radii, beta, passes):
    # Exact polishing: increase radii based on gap analysis
    for _ in range(passes):
        stable = False
        while not stable:
            stable = True
            for i in range(len(centers)):
                x, y = centers[i]
                r = radii[i]
                
                # Wall gaps
                g_wall = min(x, 1 - x, y, 1 - y)
                
                # Neighbour gaps
                g_nei = float('inf')
                for j in range(len(centers)):
                    if i != j:
                        dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                        gap = dist - (r + radii[j])
                        if gap < g_nei:
                            g_nei = gap
                
                g = min(g_wall, g_nei)
                if g > 1e-8:
                    # Try increasing slightly
                    r_new = r + beta * g
                    r_max = max_feasible_radius(i, centers, radii)
                    r_new = min(r_new, r_max)
                    if r_new > r:
                        radii[i] = r_new
                        stable = False
            
            # Handle overlaps using pairwise shrink
            changed = True
            for _ in range(3):  # Allow a few rounds of overlap resolution
                changed = False
                for i in range(len(centers)):
                    for j in range(i+1, len(centers)):
                        dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                        overlap = (radii[i] + radii[j]) - dist
                        if overlap > 0:
                            delta = overlap / 2
                            radii[i] = max(1e-8, radii[i] - delta)
                            radii[j] = max(1e-8, radii[j] - delta)
                            changed = True
                if not changed:
                    break
    
    return radii

    # Parameters
    params = {
