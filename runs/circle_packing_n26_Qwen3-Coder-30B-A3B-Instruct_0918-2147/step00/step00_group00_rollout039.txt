import numpy as np
from scipy.optimize import linprog
from scipy.spatial.distance import cdist

def generate_hex_lattice(h):
    sqrt3 = np.sqrt(3)
    points = []
    i_max = int(np.ceil(1 / h)) + 1
    j_max = int(np.ceil(1 / (h * sqrt3 / 2))) + 1

    for j in range(j_max):
        for i in range(i_max):
            x = i * h + (j % 2) * (h / 2)
            y = j * h * sqrt3 / 2
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append([x, y])

    centers = np.array(points)
    wall_radii = np.minimum(centers[:, 0], 1 - centers[:, 0])
    wall_radii = np.minimum(wall_radii, centers[:, 1])
    wall_radii = np.minimum(wall_radii, 1 - centers[:, 1])

    return centers, wall_radii

def build_neighbour_list(centers, max_dist=0.4):
    D = cdist(centers, centers)
    E = []
    n = centers.shape[0]
    for i in range(n):
        for j in range(i + 1, n):
            if D[i, j] < max_dist:
                E.append((i, j, D[i, j]))
    return E

def solve_radius_lp(selected_indices, R_wall, neighbour_list):
    n_selected = len(selected_indices)
    if n_selected == 0:
        return np.zeros(0), True

    # Build LP
    # Objective: maximize sum of r_i
    c = -np.ones(n_selected)

    # Variables: r_i for i in selected_indices
    # Bounds: 0 <= r_i <= R_wall[i]
    bounds = [(0, R_wall[i]) for i in selected_indices]

    # Inequalities: r_i + r_j <= d_ij for (i,j,d_ij) in neighbour_list
    A_ub = []
    b_ub = []

    idx_map = {selected_indices[i]: i for i in range(n_selected)}

    for i, j, d_ij in neighbour_list:
        if i in idx_map and j in idx_map:
            row = np.zeros(n_selected)
            row[idx_map[i]] = 1
            row[idx_map[j]] = 1
            A_ub.append(row)
            b_ub.append(d_ij)

    # Solve
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

    if result.success:
        return -result.fun, result.x
    else:
        return -np.inf, None

def upper_bound(partial_set, remaining_indices, R_wall):
    ub_est = sum(R_wall[i] for i in partial_set)
    remaining_count = 26 - len(partial_set)
    if remaining_count > 0:
        # Find max R_wall among remaining
        max_r = max(R_wall[i] for i in remaining_indices)
        ub_est += remaining_count * max_r
    return ub_est

def branch_and_bound(centers, R_wall, neighbour_list):
    global_best_sum = -np.inf
    global_best_centers = None
    global_best_radii = None

    def search(partial_set, remaining_indices, lb, ub):
        nonlocal global_best_sum, global_best_centers, global_best_radii

        # Pruning
        if ub <= global_best_sum:
            return

        if len(partial_set) == 26:
            # Solve LP for this selection
            sum_rad, radii = solve_radius_lp(partial_set, R_wall, neighbour_list)
            if sum_rad > global_best_sum:
                global_best_sum = sum_rad
                global_best_centers = centers[partial_set].copy()
                global_best_radii = radii.copy()
            return

        # Select next best candidate
        if not remaining_indices:
            return

        # Choose point with max R_wall
        best_idx = max(remaining_indices, key=lambda i: R_wall[i])
        remaining_indices.remove(best_idx)

        # Branch: include
        new_partial = partial_set + [best_idx]
        new_remaining = remaining_indices.copy()
        # Compute new ub based on current state
        ub_new = upper_bound(new_partial, new_remaining, R_wall)
        search(new_partial, new_remaining, lb, ub_new)

        # Branch: exclude
        search(partial_set, remaining_indices, lb, ub)

    initial_remaining = set(range(len(centers)))
    search([], initial_remaining, -np.inf, np.inf)
    return global_best_centers, global_best_radii, global_best_sum
