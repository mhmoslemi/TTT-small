import numpy as np

def hex_lattice(d):
    """Generate hexagonal lattice points in unit square."""
    rows = []
    y = 0.0
    dy = d * np.sqrt(3) / 2
    while y <= 1.0:
        row = []
        x = 0.0
        if len(rows) % 2 == 1:
            x = d / 2
        while x <= 1.0:
            row.append([x, y])
            x += d
        rows.append(row)
        y += dy
    # Flatten and filter
    points = []
    for i, row in enumerate(rows):
        for point in row:
            if 0 <= point[0] <= 1 and 0 <= point[1] <= 1:
                points.append(point)
    return np.array(points)

def add_border_circles(points, d):
    """Add 6 border circles."""
    r = d / 4
    new_points = points.tolist()
    # Corner circles
    new_points.append([r, r])
    new_points.append([1 - r, r])
    new_points.append([r, 1 - r])
    new_points.append([1 - r, 1 - r])
    # Edge circles
    new_points.append([0.5, r])
    new_points.append([0.5, 1 - r])
    return np.array(new_points)

def initial_radii(points):
    """Compute uniform safe initial radius."""
    n = points.shape[0]
    # Compute wall distances (min distance to each of 4 walls)
    wall_distances = np.minimum(points[:, 0], 1 - points[:, 0])
    wall_distances = np.minimum(wall_distances, points[:, 1])
    wall_distances = np.minimum(wall_distances, 1 - points[:, 1])
    
    # Compute minimum distances to other points
    min_neighbor_distances = np.full(n, np.inf)
    for i in range(n):
        distances = np.linalg.norm(points[i] - points, axis=1)
        distances[i] = np.inf  # Ignore self-distance
        min_neighbor_distances[i] = np.min(distances)

    # Safe initial radius is min of wall and neighbor distances (halved for overlap consideration)
    safe_radii = np.minimum(wall_distances, min_neighbor_distances / 2)
    return np.min(safe_radii)

def inflate_radii(centers, radii):
    """Inflate radii to maximum while maintaining non-overlap."""
    n = centers.shape[0]
    new_radii = radii.copy()
    # Compute distance matrix
    dist_matrix = np.sqrt(np.sum((centers[:, None] - centers[None, :]) ** 2, axis=2))
    
    epsilon = 1e-7
    improved = True
    iteration = 0
    max_iter = 200

    while improved and iteration < max_iter:
        improved = False
        for i in range(n):
            # Wall constraints
            max_radius = min(
                centers[i, 0],
                1 - centers[i, 0],
                centers[i, 1],
                1 - centers[i, 1]
            )
            # Neighbour constraints
            for j in range(n):
                if i != j:
                    max_radius = min(max_radius, dist_matrix[i, j] - radii[j])
            if max_radius > radii[i] + epsilon:
                new_radii[i] = max_radius
                improved = True
        if improved:
            radii = new_radii.copy()
        iteration += 1
    return radii

def relax_centers(centers, radii, steps=10):
    """Apply force-based relaxation."""
    n = centers.shape[0]

    # Parameters
    kappa = 0.02
    lambda_force = 0.01
    alpha = 0.5
    beta = 0.02
    gamma = 0.02

    for _ in range(steps):
        # Store old centers for possible revert
        old_centers = centers.copy()
        forces = np.zeros_like(centers)

        # Compute pairwise forces
        for i in range(n):
            for j in range(n):
                if i != j:
                    diff = centers[i] - centers[j]
                    dist = np.linalg.norm(diff)
                    if dist > 1e-12:
                        force_magnitude = kappa * max(0, beta - dist)
                        force_vector = force_magnitude * diff / dist
                        forces[i] += force_vector

        # Add wall forces
        for i in range(n):
            # Left wall
            if centers[i, 0] - radii[i] < gamma:
                forces[i, 0] += lambda_force * max(0, gamma - (centers[i, 0] - radii[i]))
            # Right wall
            if 1 - centers[i, 0] - radii[i] < gamma:
                forces[i, 0] -= lambda_force * max(0, gamma - (1 - centers[i, 0] - radii[i]))
            # Bottom wall
            if centers[i, 1] - radii[i] < gamma:
                forces[i, 1] += lambda_force * max(0, gamma - (centers[i, 1] - radii[i]))
            # Top wall
            if 1 - centers[i, 1] - radii[i] < gamma:
                forces[i, 1] -= lambda_force * max(0, gamma - (1 - centers[i, 1] - radii[i]))

        # Update positions
        new_centers = centers + alpha * forces

        # Clamp to allowable region
        for i in range(n):
            new_centers[i, 0] = np.clip(new_centers[i, 0], radii[i], 1 - radii[i])
            new_centers[i, 1] = np.clip(new_centers[i, 1], radii[i], 1 - radii[i])

        # Validate after update
        check_centers = new_centers.copy()
        check_radii = radii.copy()
        all_valid = True
        # Do a quick overlap check
        dist_matrix = np.sqrt(np.sum((check_centers[:, None] - check_centers[None, :]) ** 2, axis=2))
        for i in range(n):
            for j in range(i + 1, n):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    all_valid = False
                    break
            if not all_valid:
                break

        if all_valid:
            centers = new_centers.copy()
        else:
            centers = old_centers.copy()

    return centers

def run_packing():
    # Generate initial pattern
    d = 0.18  # Spacing
    core_points = hex_lattice(d)
    all_points = add_border_circles(core_points[:20], d)
    
    # Initial uniform radius
    r0 = initial_radii(all_points)
    initial_radii_array = np.full(all_points.shape[0], r0)
    
    # Start with a somewhat larger uniform radius to get a better base
    r0 *= 1.2  # Adjusted for better convergence
    radii = np.full(all_points.shape[0], r0)
    
    centers = all_points.copy()
    
    # Outer optimization loop
    previous_sum = np.sum(radii)
    max_outer = 30
    for _ in range(max_outer):
        # Inflate radii
        radii = inflate_radii(centers, radii)
        # Relax centers
        centers = relax_centers(centers, radii)
        # Evaluate sum
        current_sum = np.sum(radii)
        if current_sum - previous_sum < 1e-6:
            break
        previous_sum = current_sum

    # Final validation
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise AssertionError(f"Final packing invalid: {msg}")

    # Ensure non-negative radii
    radii = np.maximum(radii, 0)
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
