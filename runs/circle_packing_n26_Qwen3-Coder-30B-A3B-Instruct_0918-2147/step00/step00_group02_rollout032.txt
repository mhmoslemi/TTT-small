import numpy as np
from scipy.optimize import minimize

def initial_lattice(n=26):
    # Generate a hexagonal lattice
    s = 0.3  # Initial spacing guess, may need adjustment
    while True:
        rows = int(np.ceil(1.0 / (s * np.sqrt(3)/2)))
        cols = int(np.ceil(1.0 / s))
        total = rows * cols
        if total >= n:
            break
        s -= 0.001
        if s < 0.01:
            s = 0.01
    
    centers = []
    for i in range(rows):
        y = i * s * np.sqrt(3)/2
        if y + s > 1:
            continue
        for j in range(cols):
            x = j * s + (i % 2) * s/2
            if x + s > 1:
                continue
            centers.append([x, y])
            if len(centers) == n:
                break
        if len(centers) == n:
            break
    
    # Ensure we have exactly n centers
    if len(centers) < n:
        # If we didn't get enough, adjust and expand
        s = 0.3
        centers = []
        for i in range(100):  # Large enough range
            for j in range(100):
                x = j * s + (i % 2) * s/2
                y = i * s * np.sqrt(3)/2
                if x + s > 1 or y + s > 1:
                    continue
                centers.append([x, y])
                if len(centers) == n:
                    break
            if len(centers) == n:
                break

    centers = np.array(centers[:n])
    radii = np.full(n, 0.0)

    # Compute admissible radii for each circle
    for i in range(n):
        x, y = centers[i]
        r = min(x, 1 - x, y, 1 - y)
        for j in range(n):
            if i == j:
                continue
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            r = min(r, dist / 2)
        radii[i] = max(0, r)

    # Bound each radius to 0.5 or less
    radii = np.minimum(radii, 0.5)
    return centers, radii

def pack(centers, radii):
    z = np.empty(3 * len(centers))
    z[0::3] = centers[:, 0]
    z[1::3] = centers[:, 1]
    z[2::3] = radii
    return z

def unpack(z):
    n = len(z) // 3
    centers = np.empty((n, 2))
    radii = np.empty(n)
    centers[:, 0] = z[0::3]
    centers[:, 1] = z[1::3]
    radii = z[2::3]
    return centers, radii

def objective(z):
    _, radii = unpack(z)
    return -np.sum(radii)

def constraint_boundary(z):
    centers, radii = unpack(z)
    c = []
    for i in range(len(centers)):
        x, y = centers[i]
        r = radii[i]
        # Left boundary
        c.append(x - r)
        # Right boundary
        c.append(1 - x - r)
        # Bottom boundary
        c.append(y - r)
        # Top boundary
        c.append(1 - y - r)
    return np.array(c)

def constraint_nonoverlap(z):
    centers, radii = unpack(z)
    c = []
    n = len(centers)
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            r_sum = radii[i] + radii[j]
            c.append(dist - r_sum)
    return np.array(c)

def repair(centers, radii):
    # Bound each radius to what it can be
    for i in range(len(centers)):
        x, y = centers[i]
        r = radii[i]
        new_r = min(x, 1 - x, y, 1 - y)
        if new_r < r:
            radii[i] = new_r
    
    # Iteratively fix overlaps
    for _ in range(10):
        any_overlap = False
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                r_sum = radii[i] + radii[j]
                if dist < r_sum:
                    any_overlap = True
                    overlap = r_sum - dist
                    radii[i] -= overlap / 2
                    radii[j] -= overlap / 2
        if not any_overlap:
            break
    
    # Clamp radii to be non-negative
    radii = np.maximum(radii, 0)
    
    # Re-bound radii to avoid boundary violations after repair
    for i in range(len(centers)):
        x, y = centers[i]
        r = radii[i]
        new_r = min(x, 1 - x, y, 1 - y)
        radii[i] = min(new_r, r)
    
    return centers, radii
