import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def init_grid():
    s = 0.07
    points = []
    for k in range(10):
        y = s/2 + k * s * np.sqrt(3)/2
        if y > 1 - 0.02:
            break
        for m in range(10):
            if k % 2 == 0:
                x = s/2 + m * s
            else:
                x = s/2 + m * s + s/2
            if x > 1 - 0.02:
                break
            if 0.02 <= x <= 1 - 0.02 and 0.02 <= y <= 1 - 0.02:
                points.append([x, y])
            if len(points) >= 26:
                break
        if len(points) >= 26:
            break
    points = np.array(points[:26])
    noise = np.random.normal(0, 0.001, (26, 2))
    points += noise
    return points

def compute_forces(centers, radii, k_w=10.0, k_p=5.0):
    n = centers.shape[0]
    forces = np.zeros_like(centers)
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Wall forces
        dL = x - r
        dR = 1 - x - r
        dB = y - r
        dT = 1 - y - r
        if dL < 0:
            forces[i, 0] += k_w * (-dL)
        if dR < 0:
            forces[i, 0] -= k_w * (-dR)
        if dB < 0:
            forces[i, 1] += k_w * (-dB)
        if dT < 0:
            forces[i, 1] -= k_w * (-dT)
        # Pairwise repulsion
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            if dist > 0:
                gap = dist - (radii[i] + radii[j])
                if gap < 0:
                    f = k_p * (-gap) / dist
                    forces[i, 0] += f * dx
                    forces[i, 1] += f * dy
                    forces[j, 0] -= f * dx
                    forces[j, 1] -= f * dy
    return forces

def grow_radii(centers, radii, forces, alpha=1e-3):
    n = centers.shape[0]
    dot_r = np.ones(n) * alpha
    for i in range(n):
        # If any wall or overlap exists, no growth
        x, y = centers[i]
        r = radii[i]
        dL = x - r
        dR = 1 - x - r
        dB = y - r
        dT = 1 - y - r
        if dL < 0 or dR < 0 or dB < 0 or dT < 0:
            dot_r[i] = 0
        # Check overlaps
        for j in range(n):
            if i != j:
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                gap = dist - (radii[i] + radii[j])
                if gap < 0:
                    dot_r[i] = 0
                    break
    return dot_r

def max_radii_fixed_centers(centers, radii, max_iter=30, tol=1e-9):
    n = centers.shape[0]
    r_new = radii.copy()
    for _ in range(max_iter):
        r_old = r_new.copy()
        for i in range(n):
            x, y = centers[i]
            r = r_new[i]
            r_max = min(x, 1 - x, y, 1 - y)
            for j in range(n):
                if i != j:
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.sqrt(dx*dx + dy*dy)
                    r_max = min(r_max, dist - r_new[j])
            r_new[i] = max(0, r_max)
        if np.allclose(r_new, r_old, atol=tol):
            break
    return r_new

def objective_sum_radii(centers_flat, centers, radii):
    n = len(centers)
    centers = centers_flat.reshape((n, 2))
    # Helper function to compute radii internally
    r_temp = max_radii_fixed_centers(centers, radii)
    return -r_temp.sum()

def run_packing():
    np.random.seed(0)
    # Initial grid placement
    centers = init_grid()
    radii = np.ones(26) * 0.001
    # Physics simulation
    dt = 0.02
    gamma = 1.0
    v = np.zeros_like(centers)
    i = 0
    while True:
        forces = compute_forces(centers, radii)
        dot_r = grow_radii(centers, radii, forces)
        # Update
        v += dt * (forces - gamma * v)
        centers += dt * v
        radii += dt * dot_r
        # Clip centers
        centers = np.clip(centers, 0, 1)
        # Check termination
        if np.allclose(dot_r, 0, atol=1e-6) and np.linalg.norm(forces) < 1e-6:
            break
        i += 1
        if i > 5000:
            break
    # Compute exact max radii for fixed centers
    radii = max_radii_fixed_centers(centers, radii)
    # Refinement via L-BFGS-B
    centers_flat = centers.flatten()
    x0 = centers_flat.copy()
    bounds = [(0, 1) for _ in range(52)]
    # Objective function

        centers_attempt = x.reshape((26, 2))
        r_attempt = max_radii_fixed_centers(centers_attempt, radii)
        return -r_attempt.sum()
    res = minimize(obj, x0, method='L-BFGS-B', bounds=bounds, options={'maxiter': 2000, 'ftol': 1e-9, 'gtol': 1e-9})
