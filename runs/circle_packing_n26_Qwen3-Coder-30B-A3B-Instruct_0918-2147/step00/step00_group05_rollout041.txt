import numpy as np

def hex_lattice(d):
    """Generate hexagonal lattice points within [0,1]x[0,1]."""
    sqrt3 = np.sqrt(3)
    dy = sqrt3 * d / 2
    rows = []
    y = 0
    row_offset = 0
    while y <= 1:
        row = []
        x = row_offset
        while x <= 1:
            row.append([x, y])
            x += d
        rows.append(row)
        y += dy
        row_offset = (row_offset + d/2) % d
    return np.vstack(rows)

def add_border_circles(points, d):
    """Add six border-enhanced circles."""
    r = d / 4
    border_points = [
        [r, r], [1-r, r], [r, 1-r], [1-r, 1-r],
        [0.5, r], [0.5, 1-r]
    ]
    return np.vstack([points, border_points])

def initial_radii(points):
    """Compute initial uniform feasible radius."""
    # Compute minimum distance to any wall
    wall_distances = np.minimum(points[:, 0], 1 - points[:, 0])
    wall_distances = np.minimum(wall_distances, points[:, 1])
    wall_distances = np.minimum(wall_distances, 1 - points[:, 1])
    # Compute minimum distances to other points
    n = points.shape[0]
    min_distances = np.full(n, np.inf)
    for i in range(n):
        dists = np.linalg.norm(points[i] - points, axis=1)
        dists[i] = np.inf
        min_distances[i] = np.min(dists)
    # Initial radius is min over all of these
    r0 = np.min(np.minimum(wall_distances, min_distances / 2))
    return np.full(n, r0)

def inflate_radii(centers, radii):
    """Increase radii as much as possible while keeping non-overlap."""
    n = centers.shape[0]
    new_radii = radii.copy()
    # Precompute distance matrix
    D = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    epsilon = 1e-7
    for i in range(n):
        # Wall limits
        x, y = centers[i]
        wall_caps = [x, 1-x, y, 1-y]
        # Pairwise constraints
        pairwise_caps = []
        for j in range(n):
            if i != j:
                pairwise_caps.append(D[i, j] - radii[j])
        # Global cap
        caps = wall_caps + pairwise_caps
        max_r = min(caps)
        if max_r > new_radii[i] + epsilon:
            new_radii[i] = max_r
    return new_radii

def relax_centers(centers, radii, steps=10):
    """Apply force-based relaxation."""
    n = centers.shape[0]
    # Parameters
    kappa = 0.02
    lambda_wall = 0.01
    alpha = 0.5
    beta = 0.02
    gamma = 0.02
    
    for _ in range(steps):
        # Save old centers to detect overlaps
        old_centers = centers.copy()
        forces = np.zeros_like(centers)
        
        # Compute pairwise forces
        D = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
        for i in range(n):
            for j in range(n):
                if i == j:
                    continue
                dist_ij = D[i, j]
                r_i, r_j = radii[i], radii[j]
                delta = (r_i + r_j) - dist_ij
                if delta <= 0:
                    # Clearance; apply repulsion
                    if dist_ij > 1e-12:
                        force_magnitude = kappa * max(0, beta - dist_ij)
                        direction = (centers[i] - centers[j]) / dist_ij
                        forces[i] += force_magnitude * direction
        
        # Apply wall forces
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            f_x = 0
            f_y = 0
            f_x += -lambda_wall * max(0, gamma - (x - r))  # push right
            f_x += lambda_wall * max(0, gamma - (1 - x - r))  # push left
            f_y += -lambda_wall * max(0, gamma - (y - r))  # push up
            f_y += lambda_wall * max(0, gamma - (1 - y - r))  # push down
            forces[i] += np.array([f_x, f_y])
        
        # Update positions
        centers += alpha * forces
        # Clamp centers
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            centers[i, 0] = np.clip(x, r, 1 - r)
            centers[i, 1] = np.clip(y, r, 1 - r)
        
        # Check for overlaps
        D_new = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
        overlap_exists = False
        for i in range(n):
            for j in range(i+1, n):
                if D_new[i, j] < radii[i] + radii[j] - 1e-8:
                    overlap_exists = True
                    break
            if overlap_exists:
                break
        if overlap_exists:
            centers = old_centers.copy()

    return centers

def generate_initial_pattern():
    """Generate initial pattern of 26 circles."""
    # Hexagonal core spacing
    d = 0.18
    points = hex_lattice(d)
    # Keep first 20 points
    points = points[:20, :]
    # Add border circles
    points = add_border_circles(points, d)
    # Initial uniform radius
    radii = initial_radii(points)
    return points, radii

def run_packing():
    """Main function to pack 26 circles in unit square to maximize sum of radii."""
    # Generate initial pattern
    centers, radii = generate_initial_pattern()
    previous_sum = np.sum(radii)
    MAX_OUTER = 30
    epsilon_sum = 1e-6
    for _ in range(MAX_OUTER):
        # Inflate radii
        radii = inflate_radii(centers, radii)
        # Relax centers
        centers = relax_centers(centers, radii)
        # Compute new sum
        current_sum = np.sum(radii)
