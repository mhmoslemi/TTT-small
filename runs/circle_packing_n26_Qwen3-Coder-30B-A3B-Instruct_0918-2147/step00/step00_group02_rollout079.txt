import numpy as np
from scipy.optimize import minimize
from scipy.stats import qmc

def run_packing():
    # Generate 26 points using Sobol sequence
    sampler = qmc.Sobol(d=2, scramble=True)
    points = sampler.random(n=26)
    
    # Initial centers
    centers = points.copy()
    
    # Compute initial radii: min wall distance and half nearest neighbor distance
    radii = np.zeros(26)
    for i in range(26):
        x, y = centers[i]
        # Wall distances
        wall_dists = [x, 1-x, y, 1-y]
        # Nearest neighbor distances
        min_nn_dist = float('inf')
        for j in range(26):
            if i != j:
                dist = np.sqrt((centers[i][0] - centers[j][0])**2 + (centers[i][1] - centers[j][1])**2)
                min_nn_dist = min(min_nn_dist, dist)
        # Set radius to min of wall distances and half of nearest neighbor
        radii[i] = min(wall_dists)  # Start with wall-only constraint
        if min_nn_dist > 0:
            radii[i] = min(radii[i], min_nn_dist/2)
    
    # Adjust so that all radii are positive and set feasible starting configuration
    radii = np.maximum(radii, 1e-6)
    
    # Pack initial configuration into flat vector
    z_start = pack(centers, radii)
    
    # Run outer optimization to find maximal feasible sum of radii
    final_z = max_feasible_budget(z_start, B_initial=3.0, stepsize=5e-4)
    
    # Post-process with repair and shrinkage
    centers, radii, sum_r = postprocess(final_z)
    
    return centers, radii, sum_r

def initial_configuration(n=26):
    # Generate points using Sobol sequence
    sampler = qmc.Sobol(d=2, scramble=True)
    points = sampler.random(n=n)
    
    # Initial centers
    centers = points.copy()
    
    # Compute initial radii: min wall distance and half nearest neighbor distance
    radii = np.zeros(n)
    for i in range(n):
        x, y = centers[i]
        # Wall distances
        wall_dists = [x, 1-x, y, 1-y]
        # Nearest neighbor distances
        min_nn_dist = float('inf')
        for j in range(n):
            if i != j:
                dist = np.sqrt((centers[i][0] - centers[j][0])**2 + (centers[i][1] - centers[j][1])**2)
                min_nn_dist = min(min_nn_dist, dist)
        # Set radius to min of wall distances and half of nearest neighbor
        radii[i] = min(wall_dists)  # Start with wall-only constraint
        if min_nn_dist > 0:
            radii[i] = min(radii[i], min_nn_dist/2)
    
    # Adjust so that all radii are positive and set feasible starting configuration
    radii = np.maximum(radii, 1e-6)
    
    return centers, radii

def pack(centers, radii):
    assert centers.shape == (26, 2)
    assert radii.shape == (26,)
    z = np.empty(78)
    z[:52] = centers.flatten()
    z[52:] = radii
    return z

def unpack(z):
    assert z.shape == (78,)
    centers = z[:52].reshape((26,2))
    radii = z[52:]
    return centers, radii

def energy_and_grad(z):
    centers, radii = unpack(z)
    
    # Compute pairwise distances
    # Vectorized computation of all distances
    # centers.shape = (26, 2)
    # radii.shape = (26,)
    # diff = centers[i] - centers[j]
    # dists = sum(diff**2, axis=1) ** 0.5
    
    # Compute square distances as matrix
    diff = centers[:, None, :] - centers[None, :, :]
    dist_matrix = np.sqrt(np.sum(diff**2, axis=2))
    
    # Pairwise overlap penalties and forces
    sum_overlap = 0.0
    grad_c = np.zeros_like(centers)
    grad_r = np.zeros_like(radii)
    
    # Enumerate pairs (i,j) where i < j
    for i in range(26):
        for j in range(i+1, 26):
            d = dist_matrix[i,j]
            r_i = radii[i]
            r_j = radii[j]
            overlap = max(0, r_i + r_j - d)
            sum_overlap += overlap**2
            
            if overlap > 0:
                # Gradient w.r.t. centers
                # dp = centers[i] - centers[j]
                dp = diff[i,j]
                norm_dp = np.linalg.norm(dp)
                if norm_dp > 1e-12:
                    direction = dp / norm_dp
                    grad_c[i] -= 2 * overlap * direction
                    grad_c[j] += 2 * overlap * direction
                
                # Gradient w.r.t. radii
                grad_r[i] += 2 * overlap
                grad_r[j] += 2 * overlap
    
    # Wall repulsion energy and its gradient
    # For each circle, penalize radius exceeding distance to walls
    wall_energy = 0.0
    
    for i in range(26):
        x, y = centers[i]
        r = radii[i]
        wall_penalties = [
            max(0, r - x)**2,
            max(0, r - (1 - x))**2,
            max(0, r - y)**2,
            max(0, r - (1 - y))**2
        ]
        wall_energy += sum(wall_penalties)
        
        # Gradient w.r.t. r
        d_len = 0.0
        for term in wall_penalties:
            if term > 0:  # Because term = (r - ...) ** 2 if the difference is positive
                # derivative of (r - ...)^2 = 2 * (r - ...)
                # if r > ..., then the derivative contributes 2 * (r - ...)
                # so if the term is non-zero, r must be greater than the wall distance
                
                # If we analyze
                if r > x:
                    grad_r[i] += 2 * (r - x)
                if r > (1 - x):
                    grad_r[i] += 2 * (r - (1 - x))
                if r > y:
                    grad_r[i] += 2 * (r - y)
                if r > (1 - y):
                    grad_r[i] += 2 * (r - (1 - y))
    
    # Budget energy = lambda * (B - S)^2
    budget_energy = 0.0
    budget_grad = np.zeros_like(z)
    
    # Assume lambda = 1e6 for numerical stability (can be tuned)
    lambda_budget = 1e6
    # The value of B is passed in training phase
    # Here we use a placeholder that will be overwritten later
    
    # The actual gradient is zero for all but indices 52:78 corresponding to radii
    # grad_r is what we should compute
    return wall_energy + sum_overlap + budget_energy, np.concatenate([grad_c.flatten(), grad_r])

def inner_optimize(z0, B, lambda_budget):
    # z0 = initial flat vector of (centers, radii)
    # B = target sum of radii
    # lambda_budget = penalty for deviating from B (coefficient)
    
    def obj_and_grad(z):
        # Add budget term to penalty
        _, grad = energy_and_grad(z)
        # Add budget penalty term to gradient
        centers, radii = unpack(z)
        sum_radii = np.sum(radii)
        # Gradient of (B - sum_radii)^2 w.r.t. radii
        grad_radii = -2 * lambda_budget * (B - sum_radii) * np.ones_like(radii)
        # Gradient of (B - sum_radii)^2 w.r.t. centers = 0
        grad_centers = np.zeros_like(centers)
        # Combine gradients
        grad_full = np.concatenate([grad_centers.flatten(), grad_radii])
        # Return total energy as function value
        wall_energy, overlap_energy = energy_and_grad(z)[0], 0
        return wall_energy + overlap_energy + lambda_budget * (B - sum_radii)**2, grad_full
    
    # Bounds
    bounds = []
    for i in range(26):  # x, y centers
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 1.0))
    for i in range(26):  # radii
        bounds.append((0.0, 0.5))  # no radii can exceed 0.5
    
    # Optimize
    try:
        res = minimize(obj_and_grad, z0,
                       method='L-BFGS-B',
                       bounds=bounds,
                       jac=True,
                       options={'maxiter': 500})
        if res.success:
            return res.x
    except:
        pass
    # Fallback: just return input
    return z0

def max_feasible_budget(z_start, B_initial=3.0, stepsize=5e-4):
    # B_start = 3.0 is intentionally high
    B = B_initial
    lambda_budget = 1e6
    z = z_start.copy()
    
    max_iterations = 40
    step = stepsize
    
    for iteration in range(max_iterations):
        # Inner optimization
        z = inner_optimize(z, B, lambda_budget)
        # Compute violation
        centers, radii = unpack(z)
        v_max = max_violation(centers, radii)
        
        # Check feasibility
        if v_max <= 1e-13:
            # Feasible, try increasing budget
            B = min(B + step, B_initial)
        else:
            # Infeasible, decrease budget
            B = max(B - step, 1e-6)
        
        # If step size is small enough, stop
        if step < 1e-6:
            break
        # Adjust step size
        if v_max <= 1e-13:
            step = min(step * 1.2, 5e-4)
        else:
            step = max(step * 0.9, 1e-7)
    
    return z

def max_violation(centers, radii):
    # Test hard constraints:
    # wall distances and pair overlaps
    v_max = 0.0
    
    # Wall distances
    for i in range(26):
        x, y = centers[i]
        r = radii[i]
        wall_violation = max(
            max(0, r - x),
            max(0, r - (1-x)),
            max(0, r - y),
            max(0, r - (1-y))
        )
        v_max = max(v_max, wall_violation)
    
    # Pairwise violations
    for i in range(26):
        for j in range(i+1, 26):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            pair_violation = max(0, radii[i] + radii[j] - dist)
            v_max = max(v_max, pair_violation)
    
    return v_max

def postprocess(z):
    """
    Apply repairs and formatting to final result.
    """
    centers, radii = unpack(z)
    eps = 5e-10
    
    # Shrink all radii slightly
    radii = np.maximum(0.0, radii - eps)
    
    # Clip centers to be away from walls
    centers[:, 0] = np.clip(centers[:, 0], eps, 1 - eps)
    centers[:, 1] = np.clip(centers[:, 1], eps, 1 - eps)
    
    # Optional repair pass — make sure there are no overlaps
    # Only needed for robustness, but this is unlikely to trigger.
    
    MAX_REPAIR = 10
    for _ in range(MAX_REPAIR):
        # Compute all pairwise distances
        diff = centers[:, None, :] - centers[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        
        fixed = False
        for i in range(26):
            for j in range(i+1, 26):
                d = dists[i,j]
                r_i = radii[i]
                r_j = radii[j]
                overlap = max(0, r_i + r_j - d)
                if overlap > 0:
                    # Reduce both radii by half overlap + eps
                    reduction = overlap/2 + eps
                    if r_i - reduction > 0 and r_j - reduction > 0:
                        radii[i] -= reduction
                        radii[j] -= reduction
                        fixed = True
                    else:
                        # Just make one smaller
                        radii[i] = max(0, radii[i] - reduction)
                        radii[j] = max(0, radii[j] - reduction)
                        fixed = True
        
        if not fixed:
            break
    
    # Final sum of radii
    sum_r = np.sum(radii)
    
    return centers, radii, sum_r
