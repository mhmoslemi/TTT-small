import numpy as np
from scipy.optimize import linprog

def run_packing():
    # Constants
    n_circles = 26
    margin = 0.05
    polygon_sides = 8
    cos_pi_over_k = np.cos(np.pi / polygon_sides)
    max_iter = 30
    tol = 1e-6
    
    # Generate initial hexagonal lattice
    scale = 0.8
    spacing = scale / np.ceil(np.sqrt(n_circles))
    sqrt3 = np.sqrt(3)
    
    centers = []
    j = 0
    while len(centers) < n_circles:
        i = 0
        while len(centers) < n_circles:
            x = i * spacing + (j % 2) * spacing / 2
            y = j * spacing * sqrt3 / 2
            if x <= 1 and y <= 1:
                centers.append([x, y])
            else:
                break
            i += 1
        j += 1

    # Select first 26 centers
    centers = np.array(centers[:n_circles])
    
    # Apply margin
    min_x = np.min(centers[:, 0])
    min_y = np.min(centers[:, 1])
    centers[:, 0] += margin - min_x
    centers[:, 1] += margin - min_y
    
    # Initial radii
    radii = np.full(n_circles, 0.01)
    
    # Precompute direction vectors for polygonal approximation
    angles = np.linspace(0, 2*np.pi, polygon_sides, endpoint=False)
    unit_vectors = np.array([np.cos(angles), np.sin(angles)]).T
    
    # Precompute center distances for neighbour constraints
    # We will derive these during the process
    
    def compute_radius_lp(centers_current, radii_current):
        """Compute maximum radii while keeping centers fixed"""
        n = len(centers_current)
        # Wall constraints for each circle
        # r_i <= min(c_i.x, 1-c_i.x, c_i.y, 1-c_i.y)
        # => r_i <= min(x, 1-x, y, 1-y)
        # This is equivalent to:
        # r_i <= x_lower, r_i <= 1 - x_upper, r_i <= y_lower, r_i <= 1 - y_upper
        # But we rewrite as matrix inequalities
        
        # Row 1: r_i <= c_i.x
        A_ub1 = np.zeros((n, n))
        for i in range(n):
            A_ub1[i, i] = -1  #   -r_i <= -c_i.x
        b_ub1 = -centers_current[:, 0]

        # Row 2: r_i <= 1 - c_i.x  
        A_ub2 = np.zeros((n, n))
        for i in range(n):
            A_ub2[i, i] = -1
        b_ub2 = - (1 - centers_current[:, 0])

        # Row 3: r_i <= c_i.y
        A_ub3 = np.zeros((n, n))
        for i in range(n):
            A_ub3[i, i] = -1
        b_ub3 = - centers_current[:, 1]

        # Row 4: r_i <= 1 - c_i.y
        A_ub4 = np.zeros((n, n))
        for i in range(n):
            A_ub4[i, i] = -1
        b_ub4 = - (1 - centers_current[:, 1])

        # Distances for separation constraints
        # We will compute pairs during iteration
        # In principle, to have a completely separated circle we compute pair distances
        # Each circle i and j must have precondition: dist_ij >= r_i + r_j - epsilon
        # i.e, normalized_sum_constraints = r_i + r_j <= dist_ij
        # But that gives row: 1*r_i + 1*r_j <= dist_ij
        A_ub_pairs = []
        b_ub_pairs = []
        for i in range(n):
            for j in range(i+1, n):
                dist_ij = np.linalg.norm(centers_current[i] - centers_current[j])
                if dist_ij > 0:  # Avoid division by zero
                    A_ub_pairs.append([0]*i + [1] + [0]*(j-i-1) + [1] + [0]*(n-j-1))
                    b_ub_pairs.append(dist_ij - 1e-12)  # Account for precision loss
        
        if A_ub_pairs:
            A_ub_pairs = np.array(A_ub_pairs)
            b_ub_pairs = np.array(b_ub_pairs)
            A_combined = np.vstack([A_ub1, A_ub2, A_ub3, A_ub4, A_ub_pairs])
            b_combined = np.hstack([b_ub1, b_ub2, b_ub3, b_ub4, b_ub_pairs])
        else:
            A_combined = np.vstack([A_ub1, A_ub2, A_ub3, A_ub4])
            b_combined = np.hstack([b_ub1, b_ub2, b_ub3, b_ub4])

        # LP bounds for radii
        bounds = [(0, None) for _ in range(n)]
        
        # Objective: maximize sum of radii, i.e. [1,1,...,1] @ r
        c = np.ones(n)

        # Solve LP
        result = linprog(c, A_ub=A_combined, b_ub=b_combined, bounds=bounds, method='highs', tol=1e-10)
        
        if result.success:
            return np.maximum(0, result.x)
        else:
            # Return current values if not successful
            return np.maximum(0, radii_current)
    
    def compute_center_lp(centers_current, radii_current):
        """Compute new center positions to move from conflict situation (within valid polytope)"""
        n = len(centers_current)
        slack_vars = n  # For slack variables s_i
        # Total number of variables: n (center x/y) + n (slack)
        total_vars = 2*n + n
        
        # LP objective: maximize sum of slack,
        # i.e. sum s_i = c @ [r1,r2,...,rn,s1,s2,...,sn]
        # But we are working in r,c workspace, so we need to add slack as additional 
        # independent variables but express constraints properly.
        # Here, we use standard practice:
        # Transform to original 2n variables of centers, but manage slack in matrix setup
        
        # In fact, let's redefine optimization just to find directions to push centers.
        # Simplification: maximize sum of s_i with geometric conditions written as linear inequalities
        
        # For each center:
        # Let c = [x, y] be center coordinate vector (two entries)
        # Let s = slack value for this center
        
        # In standard linear programming, treat s_i as scalar variables to maximize
        # which requires that:
        # 1. s_i <= c_i.x - r_i         =>  -c_i.x + s_i <= -r_i 
        # 2. s_i <= 1 - c_i.x - r_i    =>   c_i.x + s_i <= 1-r_i 
        # 3. s_i <= c_i.y - r_i        =>  -c_i.y + s_i <= -r_i
        # 4. s_i <= 1 - c_i.y - r_i    =>   c_i.y + s_i <= 1-r_i
        
        # Cases 1 and 2 become 2*n inequalities
        # Rest are n inequalities, so we need at least 2n + n = 3n variables total
        # But we optimize over x,y, s for each circle
        
        # Variable order: center_x1, center_y1, ..., center_xn, center_yn, slack1, ..., slackn
        
        # Build matrix for inequalities:
        # Row 1-n: [ -1, 0, 0, 0, ... , 1, 0, 0, ..., 0 ] @ [x, y, x, y, ..., s] <= -r_i
        # Row n+1-2n: [ 1, 0, 0, 0, ..., 1, 0, 0, ..., 0 ] @ [x, y, x, y, ..., s] <= 1-r_i
        # Row 2n+1-3n: [ 0, -1, 0, 0, ..., 1, 0, 0, ..., 0 ] @ [x, y, x, y, ..., s] <= -r_i
        # Row 3n+1-4n: [ 0, 1, 0, 0, ..., 1, 0, 0, ..., 0 ] @ [x, y, x, y, ..., s] <= 1-r_i
        # Then for each pair i,j constraints from polygonal edges
        
        # We will partition problem into "core" constraints around 4 walls per circle,
        # followed by "pairwise" constraints from non-overlap
        
        # Start by creating intermediate list of constraints
        
        # Generate constraint matrix and vector:
        # Each pair of circles generates up to N constraints where N = polygon sides
        
        # We use the projection of full matrix form:
        # A_ub*x <= b_ub
        
        # But for performance, we keep walk through
        # Our approach is slightly different:
        # We write full vector of all centers and slacks as one array
        # and enforce all relevant constraints to stay inside the polytope
        
        # So total variables = 2*n + n = 3*n
        # xvec = [c1x, c1y, ..., c_nx, c_ny, s1, ..., sn] 
        # Outer product between variables in inequalities not computationally clean for large n
        # So we build the row-wise constraint matrix
    
        # Step 1: Core wall constraints:
        # Remember that variable indices are [x0,y0,x1,y1,...,sn] 
        # We must work with design matrix accordingly
        # This is a sub-problem that represents constraint perimeter
        # We now build total constraint matrix
        
        # Maximize sum of slacks: c = [0]*2n + [1]*n 
        c_center = np.zeros(2*n + n)
        c_center[2*n:] = 1
        
        # Inequality system of evolved form:
        # First create core inequalities from boundaries: s_i <= c_i.x - r_i  
        # etc.
        A_ub = []
        b_ub = []
        
        for i in range(n):
            # Wall-related constraints for center i (indexes 2*i, 2*i+1)
            # s <= c_i.x - r;   need: v1 = [-1, 0, ... , 0, 1, 0, ... , 0] (1 at position 2*i)
            # But live in extended dimension [x0, y0, ...., x_n, y_n, s1, ..., s_n] 
            # So we mimic transpose structure
            
            row = np.zeros(2*n + n)
            row[2*i] = -1  # -c_i.x <= - c_i.x
            row[2*n + i] = 1  # s_i <= -r_i  
            A_ub.append(row)
            b_ub.append(-radii_current[i])

            row = np.zeros(2*n + n)
            row[2*i] = 1  # c_i.x <= 1 - r_i + c_i.x   -> c_i.x + s_i <= 1 - r_i 
            row[2*n + i] = 1  # + s_i <= 1 - r_i  
            A_ub.append(row)
            b_ub.append(1 - radii_current[i])

            row = np.zeros(2*n + n)
            row[2*i+1] = -1  # -c_i.y <= -r_i
            row[2*n + i] = 1  # s_i <= -r_i
            A_ub.append(row)
            b_ub.append(-radii_current[i])
                
            row = np.zeros(2*n + n)
            row[2*i+1] = 1  # c_i.y + s_i <= 1 - r_i
            row[2*n + i] = 1  # + s_i <= 1 - r_i
            A_ub.append(row)
            b_ub.append(1 - radii_current[i])
            
        # Pair-wise constraints:
        # For i and j we add linearization of encroached disc using polygonal disk approx
        # Create a parallel version of the large problem we outlined earlier
        for i in range(n):
            for j in range(i+1, n):
                if np.linalg.norm(centers_current[i] - centers_current[j]) >= radii_current[i] + radii_current[j]:
                    continue  # No barrier needed if already separated
                    
                # For each of the k polygonal edges, calculate support between circles
                # Dirac vectors as previously computed
                for u_dir in unit_vectors: 
                    # For supporting plane, we use u_dir*(c_i - c_j) <= (r_i + r_j) * cos_pi_over_k
                    # Or equivalently, u_dir*c_i - u_dir*c_j <= (r_i + r_j)*cos_pi_over_k
                    # Rewrite as:
                    #  -u_dir*c_i + u_dir*c_j + (r_i + r_j)*cos_pi_over_k <= 0
                    # u_dir*(-c_i) + (u_dir*c_j) <=  (r_i + r_j)*cos_pi_over_k
                    # We need to express this in standard matrix form.
                    # Need indices for i-th center (2*i, 2*i+1) and j-th center (2*j, 2*j+1)
                    row = np.zeros(2*n + n)
                    row[2*i] = -u_dir[0]  # -u_dir[0]*c_i.x
                    row[2*i+1] = -u_dir[1]  # -u_dir[1]*c_i.y
                    row[2*j] = u_dir[0]  # u_dir[0]*c_j.x
                    row[2*j+1] = u_dir[1]  # u_dir[1]*c_j.y
                    # Then slack terms are zero here, since we don't directly involve them.
                    A_ub.append(row)
                    b_ub.append((radii_current[i] + radii_current[j]) * cos_pi_over_k)
        
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        
        # Bounds for the first 2*n variables (center coordinates)
        bounds = [(None, None) for _ in range(2*n)] + [(0, None) for _ in range(n)]
        
        # Solve full LP to move centers toward optimal distribution
        result = linprog(c_center, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', tol=1e-10)
        
        if result.success and np.allclose(result.fun, 0) and len(result.x) == 3*n:
            # Reconstruct the new centers
            new_centers = result.x[:2*n].reshape(n, 2)
            # Return copy of centers ( clipped to [0,1] but this is implied by linprog bounds) 
            return new_centers
        else:
            # If programming fails or unstable, just return the fixed positions
            return centers_current.copy()

    def local_inflation(centers_current, radii_current):
        """Local inflation step to tight or conservative portion of conflict regions"""
        n = len(centers_current)
        new_radii = radii_current.copy()
        
        # Compute distance matrix times 2
        dmatrix = np.zeros((n, n))
        for i in range(n):
            for j in range(i+1, n):
                d = np.linalg.norm(centers_current[i] - centers_current[j])
                dmatrix[i, j] = d
                dmatrix[j, i] = d
                
        updated = False
        # For each circle, check how much the real gap over the constraint is
        for i in range(n):
            min_gap = float('inf')
            for j in range(n):
                if i != j:
                    gap = dmatrix[i, j] - (radii_current[i] + radii_current[j])
                    if gap < min_gap:
                        min_gap = gap
            
            # Allow to assimilate a fraction of extra space
            if min_gap > 1e-6:
                inc = 0.4 * min_gap
                new_radii[i] = np.min([0.5, new_radii[i] + inc])
                updated = True
                
        # If we changed any radii, recompute using a small LP restricted to those circles
        if updated:
            # Find which ones changed significantly
            changed = np.where(np.abs(radii_current - new_radii) > 1e-5)[0]
            
            # If too many are changed, use full problem:
            if len(changed) > 20:
                new_radii = compute_radius_lp(centers_current, new_radii)
            else:
                # Handle only the locally affected circles with simplified constraints
                pass  # In a full implementation, we'd solve in smaller subproblem using neighbors
            
        return new_radii

    # Main loop
    for iteration in range(max_iter):
        old_sum = np.sum(radii)
        radii = compute_radius_lp(centers, radii)
        centers = compute_center_lp(centers, radii)
        new_sum = np.sum(radii)
        if new_sum - old_sum < tol:
            break

    # Local inflation
    radii = local_inflation(centers, radii)
    radii = compute_radius_lp(centers, radii)
    
    # Clean up numerical noise –Clip coordinates to [0,1] range 
    centers = np.clip(centers, 0, 1)
    radii = np.clip(radii, 0, 0.5)  # ensure all are feasible without negative
    
    # Final validation step
    valid, msg = validate_packing(centers, radii)
    if not valid:
        print(msg)
        # Try shrinking slightly to satisfy validation
        shrink_factor = 1e-7
        r_shrink = radii * (1 - shrink_factor)
        if np.all(r_shrink > 0):
            # Ensure they dont become negative
            r_shrink = np.clip(r_shrink, 0, 0.5)
            # Validate once
            _, msg = validate_packing(centers, r_shrink)
            if msg == "ok":
                radii = r_shrink
            else:
                # If still invalid, should be last resort or bad condition
                pass  # Would normally raise error
    
    return centers, radii, np.sum(radii)
