import numpy as np

def hex_lattice(d):
    """Generate a hexagonal lattice within [0,1]x[0,1] with spacing d."""
    sqrt3 = np.sqrt(3)
    dy = sqrt3 * d / 2
    rows = []
    y = 0
    row_id = 0
    while y <= 1 + 1e-12:
        x_offset = (row_id % 2) * d / 2
        x = x_offset
        row = []
        while x <= 1 + 1e-12:
            row.append([x, y])
            x += d
        rows.extend(row)
        y += dy
        row_id += 1
    return np.array(rows)

def add_border_circles(core_points, d):
    """Add 6 border circles: 4 corners, 2 edges."""
    r = d / 4
    border = [
        [r, r], [1 - r, r], [r, 1 - r], [1 - r, 1 - r],
        [0.5, r], [0.5, 1 - r]
    ]
    return np.vstack([core_points, border])

def initial_radii(points):
    """Compute a uniform safe initial radius for all points."""
    n = points.shape[0]
    # Wall distances
    wall_distances = np.minimum(points[:, 0], 1 - points[:, 0])
    wall_distances = np.minimum(wall_distances, points[:, 1])
    wall_distances = np.minimum(wall_distances, 1 - points[:, 1])
    # Pairwise distances
    distances = np.linalg.norm(points[:, None, :] - points[None, :, :], axis=2)
    # Min distance to other centers (excluding self)
    min_other = np.min(distances + np.eye(n) * 1e10, axis=1)
    # Safe radius is minimum of wall distance and half min pairwise distance
    min_radius = np.minimum(wall_distances, min_other / 2)
    # Uniform initial radius
    return np.min(min_radius)

def inflate_radii(centers, radii):
    """Increase all radii to feasible maximum."""
    n = centers.shape[0]
    radii = radii.copy()
    # Precompute distances
    diff = centers[:, None, :] - centers[None, :, :]
    distances = np.linalg.norm(diff, axis=2)
    # Clip self-distances (to avoid very small positive values from numerical error)
    np.fill_diagonal(distances, 1e10)
    # Update each radius
    updated = False
    for i in range(n):
        # Wall limits
        x, y = centers[i]
        wall_limit = min(x, 1 - x, y, 1 - y)
        # Pairwise limits
        limits = distances[i, :] - radii  # r_j must be <= dist_ij - r_i
        limits[i] = 1e10  # ignore self
        pairwise_limit = np.min(limits)
        max_radius = min(wall_limit, pairwise_limit)
        if max_radius > radii[i] + 1e-7:
            radii[i] = max_radius
            updated = True
    return radii, updated

def relax_centers(centers, radii, steps=10):
    """Apply force-based relaxation to centers."""
    n = centers.shape[0]
    kappa = 0.02
    lam = 0.01
    alpha = 0.5
    beta = 0.02
    gamma = 0.02
    centers = centers.copy()
    for _ in range(steps):
        old_centers = centers.copy()
        forces = np.zeros_like(centers)
        # Compute pairwise forces
        diff = centers[:, None, :] - centers[None, :, :]
        distances = np.linalg.norm(diff, axis=2)
        np.fill_diagonal(distances, 1e10)
        # Never touch is faster, could replace with the correct indices later if needed
        mask = (distances < beta + radii[:, None] + radii[None, :]) & (distances > 1e-12)
        # Forces from neighbors
        for i in range(n):
            for j in range(n):
                if i == j or not mask[i, j]:
                    continue
                dir_vec = centers[i] - centers[j]
                dist_ij = distances[i, j]
                if dist_ij < 1e-12:
                    continue
                unit_vec = dir_vec / dist_ij
                overlap = (radii[i] + radii[j]) - dist_ij
                if overlap > 0:
                    raise AssertionError("Overlap in relaxation")
                force_magnitude = kappa * max(0, beta - dist_ij)
                forces[i] += force_magnitude * unit_vec
        # Wall forces
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            # Repulse from walls if near
            dx = 0
            dy = 0
            if x - r < gamma:
                dx += lam * max(0, gamma - (x - r))
            if 1 - x - r < gamma:
                dx -= lam * max(0, gamma - (1 - x - r))
            if y - r < gamma:
                dy += lam * max(0, gamma - (y - r))
            if 1 - y - r < gamma:
                dy -= lam * max(0, gamma - (1 - y - r))
            forces[i] += [dx, dy]
        # Update centers
        centers += alpha * forces
        # Clamp to valid region
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            centers[i, 0] = np.clip(x, r, 1 - r)
            centers[i, 1] = np.clip(y, r, 1 - r)
        # Recheck for overlap after update
        diff = centers[:, None, :] - centers[None, :, :]
        distances = np.linalg.norm(diff, axis=2)
        # Loop to prevent renormalizing some non-overlapping and avoid infinite loop
        conflict = False
        for i in range(n):
            for j in range(i + 1, n):
                if distances[i, j] < radii[i] + radii[j] - 1e-8:
                    conflict = True
                    break
            if conflict:
                break
        if conflict:
            centers[:] = old_centers[:]
            break
    return centers

def run_packing():
    # Parameters
    d = 0.18  # hex grid spacing
    MAX_OUTER = 30
    MIN_RADIUS_INCREASE = 1e-7

    # Generate initial pattern
    core = hex_lattice(d)
    # Keep only first 20 for core
    core = core[:20]
    # Add border circles
    centers = add_border_circles(core, d)
    # Compute uniform initial radius
    radii = np.full(26, initial_radii(centers))
    # Run the iterative improvement
    previous_sum = np.sum(radii)
    for iteration in range(MAX_OUTER):
        # Inflate radii
        radii, updated = inflate_radii(centers, radii)
        if not updated:
            break
        # Relax centers
        centers = relax_centers(centers, radii)
        current_sum = np.sum(radii)
        if current_sum - previous_sum < MIN_RADIUS_INCREASE:
            break
        previous_sum = current_sum

    # Final check

    if not is_valid:
        raise AssertionError(f"Packing validation failed: {msg}")

    sum_radii = np.sum(radii)
