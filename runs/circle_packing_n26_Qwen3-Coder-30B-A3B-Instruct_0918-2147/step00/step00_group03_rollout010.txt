import numpy as np
from scipy.optimize import minimize
import scipy.sparse as sp

def init_hex_grid(n=26, r_init=0.04):
    """Generate initial placement using hexagonal lattice."""
    # Hexagonal lattice spacing
    h = np.sqrt(3) * r_init
    rows = []
    y = r_init
    row_idx = 0
    
    while len(rows) < n:
        # Even rows are shifted by r_init
        x_offset = (row_idx % 2) * r_init
        x = x_offset + r_init
        row = []
        while x <= 1 - r_init:
            if len(rows) + len(row) >= n:
                break
            row.append([x, y])
            x += 2 * r_init
        if row:
            rows.extend(row)
        y += h
        row_idx += 1
        
        if y > 1 - r_init:
            break
    
    centers = np.array(rows[:n])
    radii = np.full(n, r_init)
    return centers, radii

def max_radii(centers, radii):
    """Update radii to maximum allowable values given current positions."""
    n = len(centers)
    max_iter = 100
    tol = 1e-8
    
    for _ in range(max_iter):
        # Wall limits
        wall = np.minimum(centers, 1 - centers).min(axis=1)
        
        # Pairwise distances
        diff = centers[:, None, :] - centers[None, :, :]
        D = np.sqrt(np.sum(diff**2, axis=2))
        
        # Set diagonal to large value to ignore self-distances
        np.fill_diagonal(D, 1e10)
        
        # Neighbour limits
        neighbor = D - radii[None, :]  # Each row is a circle's distance to all others minus radii
        neighbor = np.min(neighbor, axis=1)  # Min over all neighbours
        
        # New radii
        new_radii = np.minimum(wall, neighbor)
        new_radii = np.maximum(new_radii, 0)
        
        if np.max(np.abs(new_radii - radii)) < tol:
            break
        radii = new_radii
        
    return radii

def hill_climb(centers, radii, iterations=5000, step=0.01):
    """Stochastic hill-climbing with random moves."""
    n = len(centers)
    current_sum = np.sum(radii)
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = current_sum
    
    for _ in range(iterations):
        # Pick random circle
        i = np.random.randint(n)
        
        # Random displacement
        delta = (np.random.rand(2) - 0.5) * 2 * step
        
        # New center
        new_centers = centers.copy()
        new_centers[i] += delta
        
        # Check bounds
        if (new_centers[i, 0] - radii[i] < 0 or new_centers[i, 0] + radii[i] > 1 or
            new_centers[i, 1] - radii[i] < 0 or new_centers[i, 1] + radii[i] > 1):
            continue
            
        # Update radii based on new center
        old_radii = radii.copy()
        radii = max_radii(new_centers, radii)
        
        new_sum = np.sum(radii)
        
        if new_sum > current_sum:
            centers = new_centers
            current_sum = new_sum
            if new_sum > best_sum:
                best_centers = new_centers.copy()
                best_radii = radii.copy()
                best_sum = new_sum
        else:
            radii = old_radii
    return best_centers, best_radii

def refine_with_scipy(centers, radii):
    """Use scipy optimizer to further refine."""
    n = len(centers)
    # Flatten for optimizer
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Constraints
    def wall_constraints(x):
        c = []
        for i in range(n):
            ci = i * 2
            ri = n * 2 + i
            c.extend([
                x[ri] - x[ci],           # r - x
                x[ri] + x[ci] - 1,       # r + x - 1
                x[ri] - x[ci + 1],       # r - y
                x[ri] + x[ci + 1] - 1    # r + y - 1
            ])
        return np.array(c)
    
    def overlap_constraints(x):
        c = []
        for i in range(n):
            ci = i * 2
            ri = n * 2 + i
            for j in range(i + 1, n):
                cj = j * 2
                rj = n * 2 + j
                dx = x[ci] - x[cj]
                dy = x[ci + 1] - x[cj + 1]
                dist = np.sqrt(dx * dx + dy * dy)
                c.append(x[ri] + x[rj] - dist)
        return np.array(c)
    
    # JACOBIAN
    def wall_jacobian(x):
        rows = []
        cols = []
        data = []
        for i in range(n):
            ri = n * 2 + i
            ci = i * 2
            
            # r - x
            rows.extend([0 + 4 * i, 1 + 4 * i])
            cols.extend([ci, ri])
            data.extend([-1, 1])

            # r + x - 1
            rows.extend([2 + 4 * i, 3 + 4 * i])
            cols.extend([ci, ri])
            data.extend([1, 1])

            # r - y
            rows.extend([0 + 4 * i, 2 + 4 * i])
            cols.extend([ci + 1, ri])
            data.extend([-1, 1])

            # r + y - 1
            rows.extend([1 + 4 * i, 3 + 4 * i])
            cols.extend([ci + 1, ri])
            data.extend([1, 1])

        return sp.coo_matrix((data, (rows, cols)), shape=(4 * n, 2 * n + n))
    
    def overlap_jacobian(x):
        rows = []
        cols = []
        data = []
        idx = 0
        for i in range(n):
            ci = i * 2
            ri = n * 2 + i
            for j in range(i + 1, n):
                cj = j * 2
                rj = n * 2 + j
                dx = x[ci] - x[cj]
                dy = x[ci + 1] - x[cj + 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if abs(dist) > 1e-10:
                    dx /= dist
                    dy /= dist
                else:
                    dx, dy = 0, 0
                    
                # For the constraint: r_i + r_j - sqrt((x_i-x_j)^2 + (y_i-y_j)^2)
                rows.extend([idx, idx, idx])
                cols.extend([ci, ci+1, ri])
                data.extend([-dx, -dy, 1])
                
                rows.append(idx)
                cols.append(cj)
                data.append(dx)
                
                rows.append(idx)
                cols.append(cj+1)
                data.append(dy)
                
                rows.append(idx)
                cols.append(rj)
                data.append(1)
                idx += 1
        return sp.coo_matrix((data, (rows, cols)), shape=(len(overlap_constraints(x)), 2 * n + n))
    
    # Bounds
    bnds = []
    for i in range(n):
        bnds.append((0, 1))
        bnds.append((0, 1))
        bnds.append((1e-10, 0.5))
    bounds = bnds
    
    cons = (
        NonlinearConstraint(wall_constraints, -np.inf, 0, jac=wall_jacobian),
        NonlinearConstraint(overlap_constraints, -np.inf, 0, jac=overlap_jacobian)
    )
    
    # Optimize
    res = minimize(lambda v: -np.sum(v[2 * n:]), x0, method='trust-constr', 
                   bounds=bounds, constraints=cons, options={'maxiter': 200})
    
    if not res.success:
        return centers, radii
    
    x = res.x
    new_centers = x[:2 * n].reshape(-1, 2)
    new_radii = x[2 * n:]
    return new_centers, new_radii

def run_packing():
    centers, radii = init_hex_grid()
    # Ensure we get enough circles
    while len(centers) < 26:
        # Raise initial radius to build more circles
        r_new = 0.05  # Increase slightly
        centers, radii = init_hex_grid(26, r_new)
    centers = centers[:26]
    radii = radii[:26]
    
    radii = max_radii(centers, radii)
    centers, radii = hill_climb(centers, radii)
    centers, radii = refine_with_scipy(centers, radii)
    
    # Final validation
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise RuntimeError(f"Validation failed: {msg}")
    
    sum_r = float(radii.sum())
    return centers, radii, sum_r
