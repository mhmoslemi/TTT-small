import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def initial_configuration(n=26):
    # Generate Sobol points for initial seed centres
    # A simple approximation using a low-discrepancy sequence-like approach
    phi = (1 + np.sqrt(5)) / 2
    indices = np.arange(n)
    golden_ratio = phi ** np.arange(n)
    x = (golden_ratio * 0.5) % 1
    y = (golden_ratio * 0.7) % 1
    # Since proper Sobol is not available, use this pseudo-random low-discrepancy
    centers = np.column_stack([x[:n], y[:n]])

    # Compute min distance to walls
    dist_to_walls = np.minimum(centers, 1 - centers)
    min_wall_dist = np.min(dist_to_walls, axis=1)

    # Compute minimum distances to other points
    distances = cdist(centers, centers)
    np.fill_diagonal(distances, np.inf)
    min_neighbor_dist = np.min(distances, axis=1)

    # Initial radii are limited by wall distances and neighbor distances
    radii = 0.5 * np.minimum(min_wall_dist, min_neighbor_dist)

    return centers, radii

def unpack(z):
    n = len(z) // 3
    centers = z[:2*n].reshape((n, 2))
    radii = z[2*n:]
    return centers, radii

def pack(centers, radii):
    return np.concatenate([centers.flatten(), radii])

def energy_and_grad(z):
    centers, radii = unpack(z)
    n = len(radii)
    
    # Precompute distances
    dists = cdist(centers, centers)
    np.fill_diagonal(dists, np.inf)

    E = 0.0
    grad = np.zeros_like(z)
    
    # Wall energy terms
    wall_energy = 0.0
    wall_grads = np.zeros_like(centers)
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Penalties for wall constraints
        penalties = [
            max(0, r - x)**2,
            max(0, r - (1 - x))**2,
            max(0, r - y)**2,
            max(0, r - (1 - y))**2
        ]
        wall_energy += sum(penalties)
        
        # Gradients w.r.t. x, y
        grad_contrib_x = 0
        grad_contrib_y = 0
        if r > x:
            grad_contrib_x -= 2 * (r - x)
        if r > 1 - x:
            grad_contrib_x += 2 * (r - (1 - x))
        if r > y:
            grad_contrib_y -= 2 * (r - y)
        if r > 1 - y:
            grad_contrib_y += 2 * (r - (1 - y))
            
        wall_grads[i, 0] = grad_contrib_x
        wall_grads[i, 1] = grad_contrib_y
    
    # Pairwise overlap energy
    pair_energy = 0.0
    pair_grads = np.zeros_like(centers)
    radius_grads = np.zeros(n)
    
    for i in range(n):
        for j in range(i + 1, n):
            d = dists[i, j]
            r_ij = radii[i] + radii[j]
            overlap = max(0, r_ij - d)
            if overlap > 0:
                pair_energy += overlap**2
                
                # Gradients w.r.t. positions
                if d > 1e-15:
                    dir_vec = (centers[i] - centers[j]) / d
                    grad_delta = 2 * overlap * dir_vec
                    pair_grads[i] -= grad_delta
                    pair_grads[j] += grad_delta
                # Gradients w.r.t. radii
                radius_grads[i] += 2 * overlap
                radius_grads[j] += 2 * overlap

    # Budget penalty – designed to be 0.0 when sum(radii) = B
    B = 3.0  # Initial generous budget
    sum_radii = np.sum(radii)
    radius_delta = B - sum_radii
    budget_energy = 1e6 * radius_delta**2
    budget_grad = np.zeros(n)
    budget_grad[:] = -2e6 * radius_delta  # Equal for all circle radii

    E = wall_energy + pair_energy + budget_energy
    grad[:2*n] = pair_grads.flatten() + wall_grads.flatten()
    grad[2*n:] = radius_grads + budget_grad

    return E, grad

def inner_optimize(z0, B, lambda_budget=1e6):
    # Use L-BFGS-B optimiser from SciPy
    bounds = [(0, 1) for _ in range(26*2)] + [(0, 0.5) for _ in range(26)]
    res = minimize(
        fun=lambda z: energy_and_grad(z)[0],
        x0=z0,
        method='L-BFGS-B',
        jac=lambda z: energy_and_grad(z)[1],
        bounds=bounds,
        options={'maxiter': 500},
    )
    return res.x

def max_feasible_budget(z0):
    B = 3.0  # Initial high budget
    prev_B = B
    delta = 5e-4
    lambda_budget = 1e6
    best_z = z0
    best_B = 2.0
    num_iter = 0
    
    while delta > 1e-6 and num_iter < 100:
        num_iter += 1
        z_opt = inner_optimize(z0, B, lambda_budget)
        centers, radii = unpack(z_opt)
        sum_radii = np.sum(radii)
        # Check violation using the validate function logic roughly
        violations = []
        for i in range(len(radii)):
            x, y = centers[i]
            r = radii[i]
            constraints = [
                r - x,
                r - (1 - x),
                r - y,
                r - (1 - y)
            ]
            violations.extend(constraints)
            
            # Check overlap with others
            for j in range(i + 1, len(radii)):
                d = np.linalg.norm(centers[i] - centers[j])
                overlap = radii[i] + radii[j] - d
                violations.append(-overlap)
                
        max_violation = max(violations) if violations else -1e12
        is_feasible = max_violation < 1e-13

        if is_feasible:
            best_z = z_opt
            best_B = B
            B += delta
            prev_B = B
        else:
            B = prev_B - delta  # Backtrack
            if B < 2.0:
                delta /= 2
            if delta < 1e-15:
                break
    return best_z, best_B

def postprocess(z):
    centers, radii = unpack(z)
    eps = 5e-10
    radii = np.maximum(0, radii - eps)
    centers = np.clip(centers, eps, 1 - eps)
    
    # Final safety check & minor repair
    dists = cdist(centers, centers)
    np.fill_diagonal(dists, np.inf)
    for i in range(len(radii)):
        for j in range(i + 1, len(radii)):
            overlap = radii[i] + radii[j] - dists[i,j]
            if overlap > 0:
                # Reduce larger radius
                diff = overlap / 2 + eps
                if radii[i] > radii[j]:
                    radii[i] -= diff
                else:
                    radii[j] -= diff
    sum_r = np.sum(radii)
    return centers, radii, sum_r

def run_packing():
    centers, radii = initial_configuration(26)
    z0 = pack(centers, radii)
    z_opt, _ = max_feasible_budget(z0)
    centers, radii, sum_r = postprocess(z_opt)
    return centers, radii, sum_r
