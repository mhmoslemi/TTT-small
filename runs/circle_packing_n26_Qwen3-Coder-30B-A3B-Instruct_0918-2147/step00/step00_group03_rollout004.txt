import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
from itertools import combinations

def init_candidate_grid(R, Δ):
    candidates = []
    for r in R:
        # Create grid with spacing s = r * sqrt(2) to ensure no overlap for same radii
        s = r * np.sqrt(2)
        x_coords = np.arange(r, 1 - r + s / 2, s)
        y_coords = np.arange(r, 1 - r + s / 2, s)
        for x in x_coords:
            for y in y_coords:
                candidates.append((x, y, r))
    return candidates

def build_overlap_matrix(candidates):
    n = len(candidates)
    overlap = np.zeros((n, n), dtype=bool)
    for i in range(n):
        x1, y1, r1 = candidates[i]
        for j in range(i + 1, n):
            x2, y2, r2 = candidates[j]
            dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
            if dist < r1 + r2:
                overlap[i, j] = True
                overlap[j, i] = True
    return overlap

def solve_discrete_bip(candidates, overlap_matrix, target_sum, primal_bound):
    # Simple greedy fallback to ensure feasibility
    selected = []
    used_indices = set()
    available_indices = list(range(len(candidates)))
    total_sum = 0.0
    while len(selected) < 26 and available_indices:
        # Greedy selection based on radius
        best_i = -1
        best_r = -1
        for i in available_indices:
            x, y, r = candidates[i]
            # Check if this candidate overlaps with any selected
            valid = True
            for j in used_indices:
                if overlap_matrix[i, j]:
                    valid = False
                    break
            if valid and r > best_r:
                best_i = i
                best_r = r
        if best_i == -1:
            break
        x, y, r = candidates[best_i]
        selected.append((x, y, r))
        total_sum += r
        used_indices.add(best_i)
        available_indices.remove(best_i)
        if total_sum >= primal_bound:
            break
    # If we still don't have 26, truncate or pad as needed
    while len(selected) < 26:
        _, _, r = selected[-1]
        selected.append((0.5, 0.5, r))
    return np.array(selected)

def max_radii_fixed_centres(centers, radii_initial):
    n = centers.shape[0]
    radii = radii_initial.copy()
    # compute radii that maximize sum subject to constraints
    for i in range(n):
        # boundary constraints
        x, y = centers[i]
        r = radii[i]
        min_radius = min(x, y, 1 - x, 1 - y)
        if min_radius < r:
            radii[i] = min_radius

        # pairwise overlap constraints
        for j in range(i):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            if dist < radii[i] + radii[j]:
                max_radius_i = dist - radii[j]
                if max_radius_i < radii[i]:
                    radii[i] = max_radius_i
    return np.maximum(radii, 0)

def nlp_refine(centers, radii_initial):
    n = centers.shape[0]
    # Variables: [x1, y1, ..., x26, y26, r1, ..., r26]
    x0 = np.hstack([centers.flatten(), radii_initial])

    def obj_and_grad(x):
        # Extract radii
        r = x[2*n:]
        return -np.sum(r), -np.ones_like(r)

    def ineq_constr(x):
        # Wall constraints
        # x_*: [x1,...,x26, y1,...,y26, r1,...,r26]
        x_coords = x[:n]
        y_coords = x[n:2*n]
        r = x[2*n:]
        # wall constraints for each circle: r_i - x_i <= 0, etc
        # We will compute f_i for each constraint
        f = []
        for i in range(n):
            f.append(r[i] - x_coords[i])        # r - x <= 0
            f.append(r[i] + x_coords[i] - 1)    # r + x - 1 <= 0
            f.append(r[i] - y_coords[i])        # r - y <= 0
            f.append(r[i] + y_coords[i] - 1)    # r + y - 1 <= 0

        # Non-overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt((x_coords[i] - x_coords[j])**2 + (y_coords[i] - y_coords[j])**2)
                val = r[i] + r[j] - dist
                f.append(val)
        return np.array(f)

    def ineq_jac(x):
        # Compute analytical Jacobian for non-overlap constraints and boundary constraints
        # Extract all variable values
        x_coords = x[:n]
        y_coords = x[n:2*n]
        r = x[2*n:]
        jac = np.zeros((4*n + n*(n-1)//2, 2*n + n))

        # Boundary constraints
        for i in range(n):
            xi = x_coords[i]
            yi = y_coords[i]
            ri = r[i]
            # each constraint:
            # ri - xi <= 0
            jac[4*i, i] = -1.0
            jac[4*i, 2*n+i] = 1.0

            # ri + xi - 1 <= 0
            jac[4*i+1, i] = 1.0
            jac[4*i+1, 2*n+i] = 1.0

            # ri - yi <= 0
            jac[4*i+2, n+i] = -1.0
            jac[4*i+2, 2*n+i] = 1.0

            # ri + yi - 1 <= 0
            jac[4*i+3, n+i] = 1.0
            jac[4*i+3, 2*n+i] = 1.0

        # Non-overlap constraints
        idx = 4*n
        for i in range(n):
            for j in range(i + 1, n):
                dx = x_coords[i] - x_coords[j]
                dy = y_coords[i] - y_coords[j]
                dist = np.sqrt(dx**2 + dy**2)
                if dist > 1e-12:
                    drx = dx / dist
                    dry = dy / dist
                    # Constraint: ri + rj - dist <= 0
                    # Gradient w.r.t. xi
                    jac[idx, i] = -drx
                    # Gradient w.r.t. xj
                    jac[idx, j] = drx
                    # Gradient w.r.t. yi
                    jac[idx, n+i] = -dry
                    # Gradient w.r.t. yj
                    jac[idx, n+j] = dry
                    # Gradient w.r.t. ri
                    jac[idx, 2*n+i] = 1.0
                    # Gradient w.r.t. rj
                    jac[idx, 2*n+j] = 1.0
                else:
                    # degenerate case, skip
                    pass
                idx += 1
        return jac

    # Use trust-constr to solve
    bounds = [(-1e10, 1+1e-5)]*(2*n) + [(0, 1+1e-5)]*n
    res = minimize(obj_and_grad, x0, method='trust-constr', jac=True,
                   bounds=bounds,
                   constraints={'type': 'ineq', 'fun': ineq_constr, 'jac': ineq_jac},
                   options={'gtol': 1e-9, 'xtol': 1e-9, 'maxiter': 400})

    if not res.success:
        print("NLP solver failure:", res.message)
    refined_centers = res.x[:2*n].reshape((-1, 2))
    refined_radii = res.x[2*n:]
    return refined_centers, refined_radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    R = [0.025, 0.035, 0.045, 0.055, 0.065, 0.075, 0.085, 0.095, 0.105, 0.115]
