import numpy as np
from scipy.optimize import linprog

def run_packing():
    # Constants
    n_circles = 26
    polygon_sides = 8
    cos_pi_over_k = np.cos(np.pi / polygon_sides)
    epsilon = 1e-6
    max_iterations = 30
    inflation_factor = 0.4
    shrink_factor = 1 - 1e-7

    # Step 1: Geometry-first layout (deterministic seed)
    # Create triangular lattice
    sqrt_3 = np.sqrt(3)
    spacing = 0.8 / np.floor(np.sqrt(n_circles))
    cols = int(np.ceil(1.0 / spacing))
    rows = int(np.ceil(1.0 / (spacing * sqrt_3 / 2)))
    
    points = []
    for j in range(rows):
        y = j * spacing * sqrt_3 / 2
        if y > 1.0:
            break
        for i in range(cols):
            x = i * spacing + (j % 2) * spacing / 2
            if x > 1.0:
                break
            if x >= 0 and y >= 0 and x <= 1 and y <= 1:
                points.append([x, y])
    
    if len(points) < n_circles:
        # Extend pattern if needed
        while len(points) < n_circles:
            points.append([x + spacing, y])

    # Trim to exact number and apply margin
    points = points[:n_circles]
    margin = 0.05
    for p in points:
        p[0] = max(margin, min(1 - margin, p[0]))
        p[1] = max(margin, min(1 - margin, p[1]))

    centers = np.array(points)
    radii = np.full(n_circles, 0.01)

    # Step 2 & 3: Alternate LP solving
    for iteration in range(max_iterations):
        # Solve for optimal radii based on current centers
        A_ub = []
        b_ub = []
        # Wall constraints
        for i in range(n_circles):
            A_ub.append([-1, 0])  # x constraint
            b_ub.append(-centers[i][0])
            A_ub.append([1, 0])   # 1-x constraint
            b_ub.append(centers[i][0] - 1)
            A_ub.append([0, -1])  # y constraint
            b_ub.append(-centers[i][1])
            A_ub.append([0, 1])   # 1-y constraint
            b_ub.append(centers[i][1] - 1)
        
        # Overlap constraints
        for i in range(n_circles):
            for j in range(i+1, n_circles):
                diff = centers[i] - centers[j]
                dist = np.linalg.norm(diff)
                # Add 8 directional bounding planes
                for k in range(polygon_sides):
                    theta = k * 2 * np.pi / polygon_sides
                    u = np.array([np.cos(theta), np.sin(theta)])
                    left_side = u.dot(centers[i] - centers[j])
                    right_side = -(radii[i] + radii[j]) * cos_pi_over_k
                    A_ub.append([-u[0], -u[1]])  # negate because of constraints form A x <= b
                    b_ub.append(right_side - left_side)
        
        # Formulate and solve LP
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        res = linprog(
            c=np.ones(n_circles),
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=[(0, 0.5)] * n_circles,
            method='highs',
            tol=1e-10
        )
        if not res.success:
            break
        new_radii = np.clip(res.x, 0, 0.5)

        # Compute improvement
        diff = np.sum(new_radii) - np.sum(radii)
        if diff < epsilon:
            break

            A_ub.append([-1, 0, 1]) # x + r >= 1 - s
            b_ub.append(1 - centers[i][0] - radii[i])
            A_ub.append([0, 1, 1])  # y - r <= s
            b_ub.append(centers[i][1] - radii[i])
            A_ub.append([0, -1, 1]) # y + r >= 1 - s
            b_ub.append(1 - centers[i][1] - radii[i])

            # Non-overlap bounds by polygons
            for j in range(i+1, n_circles):
                diff = centers[i] - centers[j]
                dist = np.linalg.norm(diff)
                for k in range(polygon_sides):
                    theta = k * 2 * np.pi / polygon_sides
                    u = np.array([np.cos(theta), np.sin(theta)])
                    # Convex outer bound for overlap constraint
                    left_side = u.dot(centers[i] - centers[j])
                    right_side = -(radii[i] + radii[j]) * cos_pi_over_k

        # Step 4: Re-positioning with greedy heuristic (replace LP for speed and simplicity)
        # Compute current distances and move towards max slack
        old_centers = centers.copy()

        for i in range(n_circles):
            x, y = centers[i]
            r = radii[i]
            # Try to improve position a little by moving center within safe region and keeping it inside
            dx = dy = 0
            # Greedy movement to avoid walls
            if x - r < 0.01:
                dx += 0.001
            if 1 - x - r < 0.01:
                dx -= 0.001
            if y - r < 0.01:
                dy += 0.001
            if 1 - y - r < 0.01:
                dy -= 0.001

                        gap = real_dist - (radii[i] + radii[j])
                        if gap < min_dist_to_neighbor:
                            min_dist_to_neighbor = gap
                        
                # Compute potential extra gain
                if min_dist_to_neighbor > 0:
                    max_increase = 0.4 * min(
                        min_dist_to_wall,
                        min_dist_to_neighbor / 2.0
                    )
                    if max_increase > 1e-5:
