import numpy as np
from scipy.optimize import minimize
from ortools.linear_solver import pywraplp

def generate_candidates(g, radius_levels):
    """Generate candidate circles with discrete positions and radii."""
    candidates = []
    grid_x = np.arange(0, 1 + g/2, g)
    grid_y = np.arange(0, 1 + g/2, g)
    
    for x in grid_x:
        for y in grid_y:
            for r in radius_levels:
                if (x - r >= -1e-12 and x + r <= 1 + 1e-12 and
                    y - r >= -1e-12 and y + r <= 1 + 1e-12):
                    candidates.append((x, y, r))
    return candidates

def build_milp(candidates):
    """Build and solve MILP model."""
    solver = pywraplp.Solver.CreateSolver('SCIP')
    if not solver:
        raise RuntimeError("SCIP solver not available")
        
    n = len(candidates)
    z = [solver.BoolVar(f'z_{i}') for i in range(n)]
    
    # At most one circle per grid point
    grid_dict = {}
    for i, (x, y, r) in enumerate(candidates):
        key = (round(x/g, 5), round(y/g, 5))
        if key not in grid_dict:
            grid_dict[key] = []
        grid_dict[key].append(i)
    
    for indices in grid_dict.values():
        solver.Add(solver.Sum([z[i] for i in indices]) <= 1)
    
    # Exactly 26 circles
    solver.Add(solver.Sum(z) == 26)
    
    # Pairwise non-overlap using L-infinity
    M = 1.0
    dx_vars = {}
    dy_vars = {}
    
    for i in range(n):
        for j in range(i+1, n):
            xi, yi, ri = candidates[i]
            xj, yj, rj = candidates[j]
            
            dx = solver.NumVar(0, 1, f'dx_{i}_{j}')
            dy = solver.NumVar(0, 1, f'dy_{i}_{j}')
            
            dx_vars[(i,j)] = dx
            dy_vars[(i,j)] = dy
            
            solver.Add(dx >= xi - xj)
            solver.Add(dx >= xj - xi)
            solver.Add(dy >= yi - yj)
            solver.Add(dy >= yj - yi)
            solver.Add(dx + dy >= ri + rj)
            solver.Add(dx - dy >= ri + rj - 2*M*(1 - z[i] - z[j]))
    
    # Objective
    solver.Maximize(solver.Sum([z[i] * candidates[i][2] for i in range(n)]))
    
    return solver, z, dx_vars, dy_vars

def solve_milp(solver, z_vars):
    """Solve the MILP and return selected indices."""
    status = solver.Solve()
    if status != pywraplp.MPSolver.OPTIMAL:
        raise RuntimeError(f"MILP solver failed with status {status}")
        
    selected = [i for i, var in enumerate(z_vars) if var.solution_value() > 0.5]
    return selected

def extract_initial_solution(selected_indices, candidates):
    """Extract centers and radii from selected candidates."""
    centers = np.array([[candidates[i][0], candidates[i][1]] for i in selected_indices])
    radii = np.array([candidates[i][2] for i in selected_indices])
    return centers, radii

def shrink_initial(radii, delta=1e-5):
    """Shrink radii slightly for numerical stability"""
    return np.maximum(0, radii - delta)

def pack_variables(centers0, radii0):
    """Pack centers and radii into flattened vector"""
    return np.hstack([centers0.flatten(), radii0])

def unpack_solution(z_vector, n=26):
    """Unpack z_vector into centers and radii"""
    x = z_vector[0:2*n].reshape(n, 2)
    r = z_vector[2*n:]
    return x, r

def define_objective(z_vector, n=26):
    """Define the objective function (negative sum of radii)"""
    centers, radii = unpack_solution(z_vector, n)
    return -np.sum(radii)

def define_jacobian(z_vector, n=26):
    """Define the jacobian of the objective"""
    j = np.zeros_like(z_vector)
    j[2*n:] = -1.0
    return j

def define_constraints(z_vector, centers0, radii0, n=26):
    """Define all constraints for the nonlinear optimization"""
    centers, radii = unpack_solution(z_vector, n)
    
    constraints = []
    
    # Boundary constraints
    for i in range(n):
        # r <= x, 1-x, y, 1-y
        constraints.append({
            'fun': lambda z, i=i: -radii[i] + centers[i,0],
            'jac': lambda z, i=i: np.hstack([np.zeros(2), -1.0 if i < n else 0])
        })
        constraints.append({
            'fun': lambda z, i=i: -radii[i] + (1 - centers[i,0]),
            'jac': lambda z, i=i: np.hstack([np.zeros(2), -1.0 if i < n else 0])
        })
        constraints.append({
            'fun': lambda z, i=i: -radii[i] + centers[i,1],
            'jac': lambda z, i=i: np.hstack([np.zeros(2), -1.0 if i < n else 0])
        })
        constraints.append({
            'fun': lambda z, i=i: -radii[i] + (1 - centers[i,1]),
            'jac': lambda z, i=i: np.hstack([np.zeros(2), -1.0 if i < n else 0])
        })
    
    # Non-overlap constraints (Euclidean distance)
    for i in range(n):
        for j in range(i+1, n):
            def constraint_func(z, i=i, j=j):
                x_i, y_i = centers[i]
                x_j, y_j = centers[j]
                r_i, r_j = radii[i], radii[j]
                dist = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)
                return (r_i + r_j - dist)

            def constraint_jac(z, i=i, j=j):
                x_i, y_i = centers[i]
                x_j, y_j = centers[j]
                r_i, r_j = radii[i], radii[j]
                dx = x_i - x_j
                dy = y_i - y_j
                dist = np.sqrt(dx*dx + dy*dy) + 1e-15
                
                dxi = (dx / dist)
                dyi = (dy / dist)
                dxj = -(dx / dist)
                dyj = -(dy / dist)
                
                jac = np.zeros_like(z_vector)
                jac[2*i   ] = -dxi
                jac[2*i+1 ] = -dyi
                jac[2*j   ] = -dxj
                jac[2*j+1 ] = -dyj
                jac[2*n+i] = -1.0
                jac[2*n+j] = -1.0
                return jac
            
            constraints.append({
                'fun': constraint_func,
                'jac': constraint_jac
            })
    
    return constraints

def run_packing():
    # Stage 1: Discrete Global Search
    g = 0.04
    radius_levels = np.arange(0.02, 0.26, 0.02)
    candidates = generate_candidates(g, radius_levels)
    
    # Build and solve MILP
    solver, z_vars, dx_vars, dy_vars = build_milp(candidates)
    selected_indices = solve_milp(solver, z_vars)
    
    # Extract initial solution
    centers0, radii0 = extract_initial_solution(selected_indices, candidates)
    
    # Shrink radii
    radii0 = shrink_initial(radii0)
    
    # Stage 2: Continuous Local Optimization
    z0 = pack_variables(centers0, radii0)
    n = len(selected_indices)
    
    # Define constraints
    cons = define_constraints(z0, centers0, radii0, n)
    
    # Bounds
    bounds = []
    # x, y boundary
    bounds.extend([(0,1), (0,1)]*n)
    # r boundary
    bounds.extend([(0, 0.5)]*n)
    
    # Minimize objective
    res = minimize(
        fun=lambda z: define_objective(z, n),
        x0=z0,
        method='trust-constr',
        jac=lambda z: define_jacobian(z, n),
        constraints=cons,
        bounds=bounds,
        options={'gtol': 1e-9, 'xtol': 1e-9}
    )
    
    if not res.success:
        raise RuntimeError(f"Optimization failed: {res.message}")
    
    centers, radii = unpack_solution(res.x, n)
    
    # Repair loop
    for _ in range(5):
        changed = False
        for i in range(n):
            # Clip radii to boundary
            new_r = min(radii[i], centers[i,0], 1-centers[i,0], 
                        centers[i,1], 1-centers[i,1])
            if new_r < radii[i] - 1e-10:
                radii[i] = new_r
                changed = True
        
        # Check overlaps and adjust
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[i,0] - centers[j,0]
                dy = centers[i,1] - centers[j,1]
                dist = np.sqrt(dx*dx + dy*dy)
                r_sum = radii[i] + radii[j]
                if dist < r_sum:
                    overlap = r_sum - dist
                    radii[i] -= overlap / 2
                    radii[j] -= overlap / 2
                    changed = True
        if not changed:
            break
    
    # Final validation
    centers = np.array(centers)
    radii = np.array(radii)
    _, msg = validate_packing(centers, radii)
    if not msg == "ok":
        raise RuntimeError(f"Validation failed: {msg}")
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
