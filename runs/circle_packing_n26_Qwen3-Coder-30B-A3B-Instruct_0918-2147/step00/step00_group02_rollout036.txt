import numpy as np
from scipy.optimize import minimize
from ortools.linear_solver import pywraplp
import itertools

def generate_candidates(g, radius_levels):
    candidates = []
    grid_points = np.arange(0, 1 + g/2, g)
    for x in grid_points:
        for y in grid_points:
            for r in radius_levels:
                if (x >= r and x <= 1-r and y >= r and y <= 1-r):
                    candidates.append((x, y, r))
    return candidates

def build_milp(candidates):
    solver = pywraplp.Solver.CreateSolver('SCIP')
    if not solver:
        solver = pywraplp.Solver.CreateSolver('CBC')

    n = len(candidates)
    z = [solver.IntVar(0, 1, f'z_{i}') for i in range(n)]

    # Grid point constraints
    grid_dict = {}
    for i, (x, y, r) in enumerate(candidates):
        key = (round(x/g), round(y/g))
        if key not in grid_dict:
            grid_dict[key] = []
        grid_dict[key].append(i)

    for key, indices in grid_dict.items():
        solver.Add(solver.Sum([z[i] for i in indices]) <= 1)

    # Exactly 26 circles
    solver.Add(solver.Sum(z) == 26)

    # Pairwise non-overlap constraints (using L-infinity approximation)
    for i, j in itertools.combinations(range(n), 2):
        x1, y1, r1 = candidates[i]
        x2, y2, r2 = candidates[j]

        dx = solver.NumVar(0, 1, f'dx_{i}_{j}')
        dy = solver.NumVar(0, 1, f'dy_{i}_{j}')

        solver.Add(dx >= x1 - x2)
        solver.Add(dx >= x2 - x1)
        solver.Add(dy >= y1 - y2)
        solver.Add(dy >= y2 - y1)

        # L-infinity norm >= r1 + r2
        solver.Add(dx + dy >= r1 + r2)
        solver.Add(dx - dy >= r1 + r2)
        solver.Add(dy - dx >= r1 + r2)

        # Big-M constraints
        M = 1.0
        solver.Add(dx <= M * (z[i] + z[j]))
        solver.Add(dy <= M * (z[i] + z[j]))

    # Objective
    solver.Maximize(solver.Sum([r * z[i] for i, (x,y,r) in enumerate(candidates)]))
    return solver, z

def solve_milp(solver, z):
    status = solver.Solve()
    if status != pywraplp.MPSolver.OPTIMAL:
        raise RuntimeError("MILP solver did not find an optimal solution")
    selected = [i for i in range(len(z)) if z[i].solution_value() > 0.5]
    return selected

def extract_initial_solution(selected_indices, candidates):
    centers = np.array([[candidates[i][0], candidates[i][1]] for i in selected_indices])
    radii = np.array([candidates[i][2] for i in selected_indices])
    return centers, radii

def shrink_initial(radii, delta):
    return radii - delta

def pack_variables(centers, radii):
    z = np.zeros(3 * len(centers))
    for i in range(len(centers)):
        z[3*i] = centers[i, 0]
        z[3*i+1] = centers[i, 1]
        z[3*i+2] = radii[i]
    return z

def unpack_solution(z):
    n = len(z) // 3
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = z[3*i]
        centers[i, 1] = z[3*i+1]
        radii[i] = z[3*i+2]
    return centers, radii

def define_objective(z):
    _, radii = unpack_solution(z)
    return -np.sum(radii)

def compute_boundary_violations(centers, radii):
    """Detect boundary violations"""
    violations = []
    for i, (x, y) in enumerate(centers):
        r = radii[i]
        bounds = [r, 1 - r, r, 1 - r]
        violations.extend([b - r for b in bounds])
    return np.array(violations)

def compute_pairwise_violations(centers, radii):
    """Compute pairwise distance violations"""
    violations = []
    n = len(centers)
    for i in range(n):
        for j in range(i+1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            r_sum = radii[i] + radii[j]
            violations.append(r_sum - dist)
    return np.array(violations)

def define_jacobian(z):
    n = len(z) // 3
    grad = np.zeros_like(z)
    for i in range(n):
        grad[3*i+2] = -1.0  # gradient w.r.t. radius
    return grad

def define_constraints(z):
    "Define constraints for trust-constr"
    centers, radii = unpack_solution(z)
    n = len(centers)
    constraints = []

    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # 1. r <= x  -> -r + x >= 0
        def bound_constraint_1(z):
            x, y = centers[i]
            r = radii[i]
            return x - r
        def bound_jac_1(z):
            jac = np.zeros_like(z)
            jac[3*i] = 1.0
            jac[3*i+2] = -1.0
            return jac
        constraints.append({'fun': bound_constraint_1, 'jac': bound_jac_1})

        # 2. r <= 1 - x  -> -r + (1-x) >= 0
        def bound_constraint_2(z):
            x, y = centers[i]
            r = radii[i]
            return 1 - x - r
        def bound_jac_2(z):
            jac = np.zeros_like(z)
            jac[3*i] = -1.0
            jac[3*i+2] = -1.0
            return jac
        constraints.append({'fun': bound_constraint_2, 'jac': bound_jac_2})

        # 3. r <= y  -> -r + y >= 0
        def bound_constraint_3(z):
            x, y = centers[i]
            r = radii[i]
            return y - r
        def bound_jac_3(z):
            jac = np.zeros_like(z)
            jac[3*i+1] = 1.0
            jac[3*i+2] = -1.0
            return jac
        constraints.append({'fun': bound_constraint_3, 'jac': bound_jac_3})

        # 4. r <= 1 - y  -> -r + (1-y) >= 0
        def bound_constraint_4(z):
            x, y = centers[i]
            r = radii[i]
            return 1 - y - r
        def bound_jac_4(z):
            jac = np.zeros_like(z)
            jac[3*i+1] = -1.0
            jac[3*i+2] = -1.0
            return jac
        constraints.append({'fun': bound_constraint_4, 'jac': bound_jac_4})

    # Non-overlap
    for i in range(n):
        for j in range(i+1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            r_sum = radii[i] + radii[j]

            def overlap_constraint(z):
                centers_new, radii_new = unpack_solution(z)
                dx_new = centers_new[i, 0] - centers_new[j, 0]
                dy_new = centers_new[i, 1] - centers_new[j, 1]
                dist_new = np.sqrt(dx_new*dx_new + dy_new*dy_new)
                r_sum_new = radii_new[i] + radii_new[j]
                return r_sum_new - dist_new

            def overlap_jac(z):
                centers_new, radii_new = unpack_solution(z)
                dx_new = centers_new[i, 0] - centers_new[j, 0]
                dy_new = centers_new[i, 1] - centers_new[j, 1]
                dist_new = np.sqrt(dx_new*dx_new + dy_new*dy_new) + 1e-12  # Avoid division by zero
                if dist_new == 0: dist_new = 1e-12
                jac = np.zeros_like(z)
                dx = centers_new[i, 0] - centers_new[j, 0]
                dy = centers_new[i, 1] - centers_new[j, 1]
                jac[3*i] = (dx/dist_new)
                jac[3*i+1] = (dy/dist_new)
                jac[3*i+2] = 1.0
                jac[3*j] = -(dx/dist_new)
                jac[3*j+1] = -(dy/dist_new)
                jac[3*j+2] = 1.0
                return jac

            constraints.append({'fun': overlap_constraint, 'jac': overlap_jac})
    return constraints

def run_packing():
    g = 0.04
    radius_levels = [0.02 * i for i in range(1, 26)]

    candidates = generate_candidates(g, radius_levels)
    solver, z_vars = build_milp(candidates)
    selected = solve_milp(solver, z_vars)
    centers0, radii0 = extract_initial_solution(selected, candidates)

    radius_delta = 1e-5
    radii0_shrunk = shrink_initial(radii0, radius_delta)

    z0 = pack_variables(centers0, radii0_shrunk)

    def objective_func(z):
        return define_objective(z)

    def jac_func(z):
        return define_jacobian(z)

    constraints = define_constraints(z0)
    bounds = [(0, 1.0) for _ in range(len(z0))]  # x, y, r for each circle
    for i in range(0, len(bounds), 3):
        bounds[i] = (0, 1)
        bounds[i+1] = (0, 1)
        bounds[i+2] = (0, 0.5)  # arbitrarily bounded

    res = minimize(
        fun=objective_func,
        x0=z0,
        method='trust-constr',
        jac=jac_func,
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 1000, 'disp': False, 'gtol': 1e-9, 'xtol': 1e-9}
    )
    
    if not res.success:
        raise RuntimeError(f"Optimization failed: {res.message}")
    
    centers, radii = unpack_solution(res.x)
    
    # Post-processing to fix any violations
    n = len(centers)
    for i in range(n):
        r = radii[i]
        x, y = centers[i]
        new_r = min(r, x, 1-x, y, 1-y)
        if new_r < 0:
            new_r = 1e-12
        radii[i] = new_r

    # Final overlap check and adjustment
    for _ in range(5):
        changed = False
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                r_sum = radii[i] + radii[j]
                if dist < r_sum:
                    overlap = r_sum - dist
                    if radii[i] >= radii[j]:
                        radii[i] -= overlap/2
                        radii[j] -= overlap/2
                    else:
                        radii[j] -= overlap/2
                        radii[i] -= overlap/2
                    changed = True
        if not changed:
            break

    for i in range(n):
        r = radii[i]
        x, y = centers[i]
        radii[i] = min(r, x, 1-x, y, 1-y) - 1e-10

    # Validate
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise RuntimeError(f"Validation failed: {msg}")

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
