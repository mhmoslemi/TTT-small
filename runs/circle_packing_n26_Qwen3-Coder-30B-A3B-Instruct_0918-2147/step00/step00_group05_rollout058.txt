import numpy as np
from scipy.optimize import minimize

def hex_lattice_points(n, d):
    """Generate at least n points in a hexagonal lattice."""
    points = []
    sqrt3 = np.sqrt(3)
    y_step = sqrt3 * d / 2
    x_step = d
    y = 0
    row = 0
    while len(points) < n:
        y = row * y_step
        if y > 1:
            break
        x_offset = 0 if row % 2 == 0 else x_step / 2
        x = x_offset
        while x <= 1:
            if len(points) >= n:
                break
            points.append([x, y])
            x += x_step
        row += 1
    return np.array(points[:n])

def initial_guess():
    """Generate initial feasible configuration."""
    d = 0.2  # spacing
    points = hex_lattice_points(26, d)
    # Center lattice
    x_min, x_max = points[:, 0].min(), points[:, 0].max()
    y_min, y_max = points[:, 1].min(), points[:, 1].max()
    x_center = (x_min + x_max) / 2
    y_center = (y_min + y_max) / 2
    points[:, 0] -= x_center
    points[:, 0] += 0.5
    points[:, 1] -= y_center
    points[:, 1] += 0.5
    # Compute min distance to walls
    r_wall = np.inf
    for x, y in points:
        r_wall = min(r_wall, x, 1 - x, y, 1 - y)
    # Uniform radius based on spacing and wall distance
    r0 = min(d / 2, r_wall)
    # Initialize all radii equal
    radii = np.full(26, r0)
    # Pack into variables [x0, y0, r0, x1, y1, r1, ...]
    vars = np.empty(3 * 26)
    for i in range(26):
        vars[3 * i] = points[i, 0]
        vars[3 * i + 1] = points[i, 1]
        vars[3 * i + 2] = radii[i]
    return vars

def objective(vars):
    # Objective: minimize -sum of radii
    return -np.sum(vars[2::3])

def wall_constraints(vars):
    """Inequality constraints for walls: x - r >= 0, x + r <= 1, y - r >= 0, y + r <= 1."""
    constraints = []
    for i in range(26):
        x = vars[3 * i]
        y = vars[3 * i + 1]
        r = vars[3 * i + 2]
        # x - r >= 0 → r - x <= 0
        constraints.append(r - x)
        # x + r <= 1 → x + r - 1 <= 0
        constraints.append(x + r - 1)
        # y - r >= 0 → r - y <= 0
        constraints.append(r - y)
        # y + r <= 1 → y + r - 1 <= 0
        constraints.append(y + r - 1)
    return constraints

def nonoverlap_constraints(vars):
    """Inequality constraints for non-overlap: (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2"""
    constraints = []
    for i in range(26):
        x_i = vars[3 * i]
        y_i = vars[3 * i + 1]
        r_i = vars[3 * i + 2]
        for j in range(i + 1, 26):
            x_j = vars[3 * j]
            y_j = vars[3 * j + 1]
            r_j = vars[3 * j + 2]
            # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
            dist_sq = (x_i - x_j)**2 + (y_i - y_j)**2
            radii_sum_sq = (r_i + r_j)**2
            constraints.append(dist_sq - radii_sum_sq)
    return constraints

def run_packing():
    # Initial guess
    x0 = initial_guess()
    # Build bounds
    bounds = []
    for i in range(26):
        bounds.append((0, 1))   # x bounds
        bounds.append((0, 1))   # y bounds
        bounds.append((0, 0.5)) # r bounds
    # Build constraint list
    constraints = []
    # Wall constraints
    constraints.append({'type': 'ineq', 'fun': lambda x: wall_constraints(x)})
    # Nonoverlap constraints
    constraints.append({'type': 'ineq', 'fun': lambda x: nonoverlap_constraints(x)})
    # Solve
    result = minimize(objective, x0, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': 2000, 'ftol': 1e-10})
    # Extract final solution
    final_vars = result.x
    centers = np.empty((26, 2))
    for i in range(26):
        centers[i, 0] = final_vars[3 * i]
        centers[i, 1] = final_vars[3 * i + 1]
    radii = final_vars[2::3]
