import numpy as np
from scipy.optimize import minimize
from ortools.linear_solver import pywraplp

def generate_candidates(g, radius_levels):
    candidates = []
    grid_points = np.arange(0, 1 + g/2, g)
    for x in grid_points:
        for y in grid_points:
            for r in radius_levels:
                if (x >= r and x <= 1 - r and y >= r and y <= 1 - r):
                    candidates.append((x, y, r))
    return candidates

def build_milp(candidates):
    num_candidates = len(candidates)
    solver = pywraplp.Solver.CreateSolver('SCIP')
    if not solver:
        raise RuntimeError("Failed to create solver")

    z = [solver.IntVar(0, 1, f'z_{i}') for i in range(num_candidates)]

    # Each grid point allows at most one circle
    grid_map = {}
    for i, (x, y, r) in enumerate(candidates):
        pos = (round(x/g), round(y/g))
        if pos not in grid_map:
            grid_map[pos] = []
        grid_map[pos].append(i)

    for pos, indices in grid_map.items():
        solver.Add(solver.Sum([z[i] for i in indices]) <= 1)

    # Exactly 26 circles
    solver.Add(solver.Sum(z) == 26)

    # Non-overlap constraints
    for i in range(num_candidates):
        for j in range(i+1, num_candidates):
            x1, y1, r1 = candidates[i]
            x2, y2, r2 = candidates[j]
            
            dx = solver.NumVar(0, 1, f'dx_{i}_{j}')
            dy = solver.NumVar(0, 1, f'dy_{i}_{j}')
            
            solver.Add(dx >= x1 - x2)
            solver.Add(dx >= x2 - x1)
            solver.Add(dy >= y1 - y2)
            solver.Add(dy >= y2 - y1)
            
            # L-infinity norm of distance
            solver.Add(dx + dy >= r1 + r2)
            solver.Add(dx - dy >= r1 + r2 - 1)
            
            # Big-M constraints to only apply when both are selected
            solver.Add(dx <= 1 * (z[i] + z[j]))
            solver.Add(dy <= 1 * (z[i] + z[j]))

    # Objective
    solver.Maximize(solver.Sum([r * z[i] for i, (_, _, r) in enumerate(candidates)]))

    return solver, z

def solve_milp(solver, z):
    status = solver.Solve()
    if status != pywraplp.Solver.OPTIMAL:
        raise RuntimeError("MILP solver did not find optimal solution")

    selected_indices = [i for i in range(len(z)) if z[i].solution_value() > 0.5]
    return selected_indices

def extract_initial_solution(selected_indices, candidates):
    centers = np.array([[candidates[i][0], candidates[i][1]] for i in selected_indices])
    radii = np.array([candidates[i][2] for i in selected_indices])
    return centers, radii

def shrink_initial(radii, delta=1e-5):
    return np.maximum(radii - delta, 0)

def pack_variables(centers, radii):
    return np.concatenate([centers.flatten(), radii])

def unpack_solution(z_vec, n=26):
    centers = z_vec[:2*n].reshape(n, 2)
    radii = z_vec[2*n:]
    return centers, radii

def define_objective(z_vec):
    _, radii = unpack_solution(z_vec)
    return -np.sum(radii)

def define_jacobian(z_vec):
    _, radii = unpack_solution(z_vec)
    n = len(radii)
    grad = np.zeros_like(z_vec)
    grad[2*n:] = -1.0
    return grad

def define_constraints(z_vec):
    centers, radii = unpack_solution(z_vec)
    n = len(centers)
    constraints = []

    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # r ≤ x
        def c1(z):
            _, radii = unpack_solution(z)
            return radii[i] - x
        
        # r ≤ 1 - x
        def c2(z):
            _, radii = unpack_solution(z)
            return radii[i] - (1 - x)
        
        # r ≤ y
        def c3(z):
            _, radii = unpack_solution(z)
            return radii[i] - y
        
        # r ≤ 1 - y
        def c4(z):
            _, radii = unpack_solution(z)
            return radii[i] - (1 - y)

        constraints.append({
            'fun': c1,
            'jac': lambda z: np.zeros_like(z_vec)
        })
        constraints.append({
            'fun': c2,
            'jac': lambda z: np.zeros_like(z_vec)
        })
        constraints.append({
            'fun': c3,
            'jac': lambda z: np.zeros_like(z_vec)
        })
        constraints.append({
            'fun': c4,
            'jac': lambda z: np.zeros_like(z_vec)
        })

    # Non-overlap constraints
    for i in range(n):
        for j in range(i+1, n):
            def c_ij(z):
                centers, radii = unpack_solution(z)
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                distance = np.sqrt(dx*dx + dy*dy + 1e-12)
                overlap = (radii[i] + radii[j]) - distance
                return overlap
            
            def j_ij(z):
                centers, radii = unpack_solution(z)
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                distance = np.sqrt(dx*dx + dy*dy + 1e-12)
                ddx = dx / distance
                ddy = dy / distance
                jac = np.zeros_like(z_vec)
                
                # Gradient w.r.t. centers and radii
                jac[2*i] -= ddx
                jac[2*i+1] -= ddy
                jac[2*j] += ddx
                jac[2*j+1] += ddy
                jac[2*n+i] -= 1.0
                jac[2*n+j] -= 1.0
                return jac
            
            constraints.append({
                'fun': c_ij,
                'jac': j_ij
            })

    return constraints

def run_packing():
    g = 0.035
    radius_levels = np.arange(0.02, 0.26, 0.02)
    candidates = generate_candidates(g, radius_levels)
    solver, z = build_milp(candidates)
    selected_indices = solve_milp(solver, z)
    centers0, radii0 = extract_initial_solution(selected_indices, candidates)
    radii0 = shrink_initial(radii0, 1e-5)
    z0 = pack_variables(centers0, radii0)
    bounds = [(0, 1)] * len(z0)
    
    constraints = define_constraints(z0)
    
    result = minimize(
        fun=define_objective,
        x0=z0,
        method='trust-constr',
        jac=define_jacobian,
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 1000}
    )
    
    centers, radii = unpack_solution(result.x)
    sum_radii = np.sum(radii)
    
    # Apply repair
    for i in range(len(centers)):
        r = radii[i]
        x, y = centers[i]
        new_r = min(r, x, 1-x, y, 1-y)
        radii[i] = new_r

    for _ in range(5):
        changed = False
        for i in range(len(centers)):
            for j in range(i+1, len(centers)):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx*dx + dy*dy + 1e-12)
                overlap = (radii[i] + radii[j]) - dist
                if overlap > 0:
                    reduction = overlap / 2
                    if radii[i] >= radii[j]:
                        radii[i] -= reduction
                    else:
                        radii[j] -= reduction
                    changed = True
        if not changed:
            break
