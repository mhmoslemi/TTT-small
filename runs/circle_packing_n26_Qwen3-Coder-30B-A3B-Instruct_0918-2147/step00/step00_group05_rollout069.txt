import numpy as np
from scipy.optimize import minimize

def hex_lattice_points(n, d):
    """Generate at least n points in a hexagonal lattice with spacing d."""
    rows = int(np.ceil(np.sqrt(n / (np.sqrt(3)/2))))
    points = []
    for i in range(rows):
        y = i * np.sqrt(3)/2 * d
        if y > 1:
            break
        offset = (i % 2) * d / 2
        for j in range(int(np.ceil(1.0 / d)) + 1):
            x = j * d + offset
            if x > 1:
                break
            points.append([x, y])
            if len(points) >= n:
                break
        if len(points) >= n:
            break
    return np.array(points[:n])

def initial_guess():
    """Generate initial layout: hexagonal lattice with equal radii."""
    # Start with spacing d = 0.2, which will generate enough points
    d = 0.2
    points = hex_lattice_points(26, d)
    x = points[:, 0]
    y = points[:, 1]
    # Compute minimal distance to wall for each point
    r_wall = np.minimum(np.minimum(x, 1-x), np.minimum(y, 1-y))
    r_min = np.min(r_wall)
    # Use half the lattice spacing for initial radius to make sure circles do not overlap
    initial_r = min(r_min, d / 2)
    # Pack variables: [x0, y0, r0, x1, y1, r1, ...]
    vars = np.zeros(3 * 26)
    for i in range(26):
        vars[3*i] = x[i]
        vars[3*i+1] = y[i]
        vars[3*i+2] = initial_r
    return vars

def objective(vars):
    # Sum of radii (negative because we minimize)
    return -np.sum(vars[2::3])

def wall_constraints(vars):
    """Return list of wall constraint values (should be >= 0)"""
    result = []
    for i in range(26):
        x = vars[3*i]
        y = vars[3*i+1]
        r = vars[3*i+2]
        # x - r >= 0
        result.append(r - x)
        # x + r <= 1
        result.append(x + r - 1)
        # y - r >= 0
        result.append(r - y)
        # y + r <= 1
        result.append(y + r - 1)
    return result

def nonoverlap_constraints(vars):
    """Return list of non-overlap constraint values (should be >= 0)"""
    result = []
    for i in range(26):
        for j in range(i + 1, 26):
            x1 = vars[3*i]
            y1 = vars[3*i+1]
            r1 = vars[3*i+2]
            x2 = vars[3*j]
            y2 = vars[3*j+1]
            r2 = vars[3*j+2]
            dx = x1 - x2
            dy = y1 - y2
            dist_sq = dx*dx + dy*dy
            sum_radii = r1 + r2
            # We want dist^2 >= (r1 + r2)^2
            # So we want (r1 + r2)^2 - dist^2 <= 0 (we need it negative for minimize)
            result.append(sum_radii * sum_radii - dist_sq)
    return result

def run_packing():
    # Initial guess
    vars_initial = initial_guess()
    
    # Build constraints
    constraints = []

    # Wall constraints
    constraints.append({
        'type': 'ineq',
        'fun': lambda v: wall_constraints(v)
    })

    # Non-overlap constraints
    constraints.append({
        'type': 'ineq',
        'fun': lambda v: nonoverlap_constraints(v)
    })

    # Bounds
    bounds = []
    for i in range(26):
        bounds.append((0, 1))   # x_i
        bounds.append((0, 1))   # y_i
        bounds.append((0, 0.5)) # r_i

    # Optimize
    res = minimize(
        objective,
        vars_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 2000, 'ftol': 1e-10}
    )

    if not res.success:
        raise RuntimeError(f"Optimization failed: {res.message}")

    # Extract final values
    x_final = res.x[0::3]
    y_final = res.x[1::3]
    r_final = res.x[2::3]
    
    # Clip tiny negative values to zero
    r_final = np.maximum(r_final, 0.0)
