import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(0)
    
    def init_grid():
        s = 0.07
        centers = []
        for k in range(6):
            y = s / 2 + k * s * np.sqrt(3) / 2
            if y > 1 - s:
                break
            for m in range(10):
                if k % 2 == 0:
                    x = s / 2 + m * s
                else:
                    x = s / 2 + m * s + s / 2
                if x > 1 - s:
                    break
                if 0 < x < 1 and 0 < y < 1:
                    centers.append([x, y])
        centers = np.array(centers[:26])
        # Add small noise
        centers += np.random.uniform(-0.001, 0.001, centers.shape)
        centers = np.clip(centers, 0, 1)
        radii = np.full(26, 0.001)
        return centers, radii

    def compute_forces(centers, radii, k_w=10.0, k_p=5.0):
        n = centers.shape[0]
        forces = np.zeros_like(centers)
        for i in range(n):
            # Wall forces
            x, y = centers[i]
            r = radii[i]
            dL = x - r
            dR = (1 - x) - r
            dB = y - r
            dT = (1 - y) - r

            if dL < 0:
                forces[i, 0] += k_w * (-dL)
            if dR < 0:
                forces[i, 0] -= k_w * (-dR)
            if dB < 0:
                forces[i, 1] += k_w * (-dB)
            if dT < 0:
                forces[i, 1] -= k_w * (-dT)

            # Pairwise forces
            for j in range(n):
                if i == j:
                    continue
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                gap = dist - (radii[i] + radii[j])
                if gap < 0:
                    f = k_p * (-gap) / (dist + 1e-12)
                    forces[i, 0] += f * dx
                    forces[i, 1] += f * dy
        return forces

    def grow_radii(centers, radii, forces, alpha=1e-3):
        n = centers.shape[0]
        dr = np.full(n, alpha)
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            dL = x - r
            dR = (1 - x) - r
            dB = y - r
            dT = (1 - y) - r
            # Check if touching any constraint
            touching_constraint = (dL < 0 or dR < 0 or dB < 0 or dT < 0)
            for j in range(n):
                if i == j:
                    continue
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                gap = dist - (radii[i] + radii[j])
                if gap < 0:
                    touching_constraint = True
                    break
            if touching_constraint:
                dr[i] = 0.0
        return dr

    def simulate_physics():
        centers, radii = init_grid()
        vel = np.zeros_like(centers)
        dt = 0.02
        gamma = 1.0
        max_iter = 5000
        for _ in range(max_iter):
            forces = compute_forces(centers, radii)
            dr = grow_radii(centers, radii, forces)
            vel = vel + dt * (forces - gamma * vel)
            centers = centers + dt * vel
            radii = radii + dt * dr
            # Clip centers
            centers = np.clip(centers, 0, 1)
            # Clamp radii to non-negative values
            radii = np.maximum(radii, 0)
            # Early stop criterion
            if np.all(dr < 1e-8) and np.linalg.norm(forces) < 1e-6:
                break
        return centers, radii

    def max_radii_fixed_centers(centers, radii):
        old_radii = radii.copy()
        for _ in range(50):
            new_radii = np.zeros_like(radii)
            for i in range(26):
                x, y = centers[i]
                # Wall distances
                r_min = min(x, 1 - x, y, 1 - y)
                # Pairwise distances
                for j in range(26):
                    if i != j:
                        dx = centers[i, 0] - centers[j, 0]
                        dy = centers[i, 1] - centers[j, 1]
                        dist = np.sqrt(dx*dx + dy*dy)
                        if dist > 0:
                            gap = dist - radii[j]
                            if gap < r_min:
                                r_min = gap
                new_radii[i] = max(0, r_min)
            radii = new_radii
            if np.allclose(radii, old_radii, atol=1e-9):
                break
            old_radii = radii.copy()
        return radii

    def objective_sum_radii(centers_flat):
        centers = centers_flat.reshape(-1, 2)
        # We need to compute max radii for this specific set of centers
        radii = np.zeros(26)
        for i in range(26):
            x, y = centers[i]
            r_min = min(x, 1 - x, y, 1 - y)
            for j in range(26):
                if i == j:
                    continue
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                gap = dist - radii[j]
                if gap < r_min:
                    r_min = gap
            radii[i] = max(0, r_min)
        return -radii.sum()

    # Step 1: Physics simulation to get jammed configuration
    centers, radii = simulate_physics()
    # Step 2: Refine radii
    radii = max_radii_fixed_centers(centers, radii)
    # Step 3: Global optimization
    centers_flat = centers.reshape(-1)
    bounds = []
    for i in range(26):
        bounds.append((0.0, 1.0))  # x
        bounds.append((0.0, 1.0))  # y

    # Add a small penalty to avoid edges
    def barrier(centers_flat):
        centers = centers_flat.reshape(-1, 2)
        penalty = 0.0
        eps = 1e-4
        for i in range(26):
            x, y = centers[i]
            penalty += max(0, -x)**2 + max(0, x - 1)**2 + max(0, -y)**2 + max(0, y - 1)**2
        return penalty * 1e-4

    def combined_objective(centers_flat):
        return objective_sum_radii(centers_flat) + barrier(centers_flat)

    result = minimize(combined_objective, centers_flat, bounds=bounds, method='L-BFGS-B', options={'maxiter': 2000, 'gtol': 1e-9})
    centers_flat = result.x
    centers = centers_flat.reshape(-1, 2)
    # Final refinement of radii
    radii = max_radii_fixed_centers(centers, np.zeros(26))
    sum_radii = radii.sum()
    
    # Validate final packing
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise RuntimeError(f"Invalid packing: {msg}")
    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
