import numpy as np
import random

def make_hex_seed(n, margin):
    # Generate a hexagonal grid pattern with some margin from boundaries
    sqrt3 = np.sqrt(3)
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))
    
    # Create grid points with spacing that allows for margin
    spacing_x = (1 - 2*margin) / (cols - 1) if cols > 1 else 0.5
    spacing_y = (1 - 2*margin) / (rows - 1) if rows > 1 else 0.5
    
    centers = []
    radii = []
    
    for i in range(rows):
        for j in range(cols):
            if len(centers) >= n:
                break
            x = margin + j * spacing_x
            y = margin + i * spacing_y
            
            # Adjust odd rows to create hexagonal structure
            if i % 2 == 1:
                x += spacing_x / 2
            
            # Distance to nearest wall
            d_min = min(x, 1-x, y, 1-y)
            
            # Use a function of wall distances to assign initial radius
            rad = 0.015 + 0.01 * d_min
            
            centers.append([x, y])
            radii.append(rad)
    
    # Trim to exactly n
    centers = np.array(centers[:n])
    radii = np.array(radii[:n])
    
    return centers, radii

def max_feasible_radius(i, centers, radii):
    # Determine maximum allowable radius for circle i
    x, y = centers[i]
    r = radii[i]
    
    # Walls
    d_left = x
    d_right = 1 - x
    d_bottom = y
    d_top = 1 - y
    
    # Neighbors
    min_dist = float('inf')
    for j in range(len(centers)):
        if i == j:
            continue
        dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
        min_dist = min(min_dist, dist - radii[j])
    
    # Return minimum of all constraints
    return min(d_left, d_right, d_bottom, d_top, min_dist)

def inflate_radii(centers, radii, alpha):
    old_sum = np.sum(radii)
    mean_r = np.mean(radii)
    delta = alpha * mean_r
    
    new_radii = radii + delta
    for i in range(len(centers)):
        max_r = max_feasible_radius(i, centers, radii)
        new_radii[i] = min(new_radii[i], max_r)
    
    new_sum = np.sum(new_radii)
    return new_radii, new_sum - old_sum

def anneal_centers(centers, radii, temp0, gamma, sigma0, steps):
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = np.sum(radii)
    
    temp = temp0
    sigma = sigma0
    
    for _ in range(steps):
        # Global translation occasionally to break symmetry
        if _ % 50 == 0:
            delta = np.random.normal(0, 0.002, 2)
            centers += delta
        
        # Random circle selection
        i = random.randint(0, len(centers) - 1)
        
        # Proposal
        displacement = np.random.normal(0, sigma, 2)
        old_pos = centers[i].copy()
        old_radius = radii[i]
        
        centers[i] += displacement
        
        # Ensure the new center stays within the square
        centers[i] = np.clip(centers[i], 0, 1)
        
        # Recompute max allowable radius for moved circle
        new_radius = max_feasible_radius(i, centers, radii)
        
        # Accept with acceptance probability
        delta_radius = new_radius - old_radius
        if delta_radius > 0 or np.random.rand() < np.exp(delta_radius / temp):
            radii[i] = new_radius
        else:
            # Restore previous state
            centers[i] = old_pos
            radii[i] = old_radius
        
        # Update best
        curr_sum = np.sum(radii)
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers[:] = centers[:]
            best_radii[:] = radii[:]
        
        # Anneal temperature and sigma
        temp *= gamma
        sigma = sigma0 * temp / temp0
        
        if temp < 1e-10:
            break
    
    return best_centers, best_radii, best_sum

def polish_radii(centers, radii, beta, passes):
    old_sum = np.sum(radii)
    for _ in range(passes):
        changed = False
        for i in range(len(centers)):
            x, y = centers[i]
            r = radii[i]
            
            # Calculate gaps
            d_left = x
            d_right = 1 - x
            d_bottom = y
            d_top = 1 - y
            
            g_wall = min(d_left, d_right, d_bottom, d_top)
            
            g_nei = float('inf')
            for j in range(len(centers)):
                if i == j:
                    continue
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                gap = dist - (radii[i] + radii[j])
                g_nei = min(g_nei, gap)
            
            g = min(g_wall, g_nei)
            if g > 0:
                new_r = min(r + beta * g, max_feasible_radius(i, centers, radii))
                if new_r > r:
                    radii[i] = new_r
                    changed = True
        
        # Resolve overlaps if any
        overlap_resolved = True
        while overlap_resolved:
            overlap_resolved = False
            for i in range(len(centers)):
                for j in range(i + 1, len(centers)):
                    dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                    overlap = (radii[i] + radii[j]) - dist
                    if overlap > 0:
                        # Shrink both circles
                        delta = overlap / 2
                        radii[i] = max(0, radii[i] - delta)
                        radii[j] = max(0, radii[j] - delta)
                        overlap_resolved = True
        
        if not changed:
            break
    
    return radii
