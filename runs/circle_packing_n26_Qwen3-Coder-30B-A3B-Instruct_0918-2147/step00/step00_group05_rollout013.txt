import numpy as np

def hex_lattice(d):
    """Generate a hexagonal lattice within [0,1]x[0,1]"""
    sqrt3 = np.sqrt(3)
    dy = sqrt3 * d / 2
    rows = []
    y = 0
    row_idx = 0
    while y <= 1:
        row = []
        x = 0
        if row_idx % 2 == 1:
            x = d / 2
        while x <= 1:
            row.append([x, y])
            x += d
        rows.extend(row)
        y += dy
        row_idx += 1
    return np.array(rows)

def add_border_circles(hex_points, d):
    """Append 6 border-enhanced circles"""
    r = d / 4
    border_points = [
        [r, r], [1 - r, r], [r, 1 - r], [1 - r, 1 - r],
        [0.5, r], [0.5, 1 - r]
    ]
    return np.vstack([hex_points, border_points])

def initial_radii(points):
    """Compute uniform safe radius for all points"""
    n = points.shape[0]
    # Compute min wall distance for each point
    wall_distances = np.minimum(
        points[:, 0], 1 - points[:, 0],
        points[:, 1], 1 - points[:, 1]
    )
    # Compute minimum distance to other points
    min_distances = np.full(n, np.inf)
    for i in range(n):
        dists = np.linalg.norm(points[i] - points, axis=1)
        dists[i] = np.inf
        min_distances[i] = np.min(dists)
    # Safe radius is min of wall and half min distance to others
    safe_radii = np.minimum(wall_distances, min_distances / 2)
    return np.min(safe_radii)

def inflate_radii(centers, radii):
    """Increase radii as much as possible while maintaining constraints"""
    n = centers.shape[0]
    new_radii = radii.copy()
    # Precompute distance matrix
    D = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    eps = 1e-6
    
    for i in range(n):
        # Wall limits
        x, y = centers[i]
        wall_limits = [x, 1 - x, y, 1 - y]
        # Neighbour limits
        neighbour_limits = []
        for j in range(n):
            if i != j:
                neighbour_limits.append(D[i, j] - radii[j])
        if neighbour_limits:
            all_limits = wall_limits + neighbour_limits
            new_radius = min(all_limits)
            if new_radius - radii[i] > eps:
                new_radii[i] = new_radius
    return new_radii

def relax_centers(centers, radii, steps=10):
    """Apply force-based relaxation to centers"""
    n = centers.shape[0]
    # Parameters
    kappa = 0.02
    gamma = 0.02
    lambda_wall = 0.01
    alpha = 0.5
    beta = 0.02
    
    current_centers = centers.copy()
    for _ in range(steps):
        force = np.zeros_like(current_centers)
        # Compute distance matrix
        D = np.linalg.norm(current_centers[:, None, :] - current_centers[None, :, :], axis=2)
        
        # Compute repulsive forces
        for i in range(n):
            for j in range(n):
                if i == j:
                    continue
                dist = D[i, j]
                r_i = radii[i]
                r_j = radii[j]
                delta = r_i + r_j - dist
                if dist < beta:
                    # Repulsive force
                    force_ij = kappa * max(0, beta - dist) * (current_centers[i] - current_centers[j]) / max(dist, 1e-10)
                    force[i] += force_ij
        
        # Compute wall forces
        for i in range(n):
            x, y = current_centers[i]
            r = radii[i]
            # X forces
            f_x = 0
            f_x -= lambda_wall * max(0, gamma - (x - r)) # push right
            f_x += lambda_wall * max(0, gamma - (1 - x - r)) # push left
            force[i, 0] += f_x
            
            # Y forces
            f_y = 0
            f_y -= lambda_wall * max(0, gamma - (y - r)) # push up
            f_y += lambda_wall * max(0, gamma - (1 - y - r)) # push down
            force[i, 1] += f_y
        
        # Update positions
        new_centers = current_centers + alpha * force
        # Clamp to safe positions
        new_centers[:, 0] = np.clip(new_centers[:, 0], radii, 1 - radii)
        new_centers[:, 1] = np.clip(new_centers[:, 1], radii, 1 - radii)
        
        # Check for overlap
        new_D = np.linalg.norm(new_centers[:, None, :] - new_centers[None, :, :], axis=2)
        if np.any(new_D < radii[:, None] + radii[None, :] - 1e-8):
            # Revert
            break
        current_centers = new_centers
        
    return current_centers

def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"
    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"
    return True, "ok"

def run_packing():
    # Parameters
    d = 0.18
    MAX_OUTER = 30
    eps_sum = 1e-6
    # Generate hexagonal lattice
    hex_points = hex_lattice(d)
    # Keep only first 20
    hex_points = hex_points[:20]
    # Add border circles
    centers = add_border_circles(hex_points, d)
    # Compute initial radius
    r0 = initial_radii(centers)
    radii = np.full(26, r0)
    # Initial sum
    prev_sum = np.sum(radii)
    
    for _ in range(MAX_OUTER):
        radii = inflate_radii(centers, radii)
        centers = relax_centers(centers, radii)
        current_sum = np.sum(radii)
        if current_sum - prev_sum < eps_sum:
            break
        prev_sum = current_sum
    
    # Final validation
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise AssertionError(msg)
    
    return centers, radii, np.sum(radii)
