```python
import numpy as np
from scipy.optimize import minimize

NUM_CIRCLES = 26
NUM_RESTARTS = 5
HEX_ROWS = 5
HEX_COLS = 6
HEX_SPACING = 0.2
INITIAL_RADIUS = 0.02


def hex_lattice_points(n_rows, spacing):
    points = []
    for i in range(n_rows):
        y = i * spacing * np.sqrt(3) / 2
        for j in range(HEX_COLS):
            x = j * spacing + (i % 2) * spacing / 2
            if x <= 1 and y <= 1:
                points.append([x, y])
    return np.array(points)[:NUM_CIRCLES]


def initial_guess(seed):
    np.random.seed(seed)
    centers = hex_lattice_points(HEX_ROWS, HEX_SPACING)
    # jitter
    centers += np.random.uniform(-0.02, 0.02, centers.shape)
    centers = np.clip(centers, 0, 1)
    radii = np.full(NUM_CIRCLES, INITIAL_RADIUS)
    vars0 = np.empty(3 * NUM_CIRCLES)
    vars0[0::3] = centers[:, 0]
    vars0[1::3] = centers[:, 1]
    vars0[2::3] = radii
    return vars0


def objective(vars):
    return -np.sum(vars[2::3])


def constraint_boundary(vars):
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    # x - r, 1 - x - r, y - r, 1 - y - r
    return np.concatenate([
        x - r,
        1 - x - r,
        y - r,
        1 - y - r
    ])


def constraint_nonoverlap(vars):
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    # All pairs (i,j) with i < j
    margins = []
    for i in range(NUM_CIRCLES):
        for j in range(i + 1, NUM_CIRCLES):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dist_sq = dx * dx + dy * dy
            dist = np.sqrt(dist_sq)
            sum_radii = r[i] + r[j]
            margin = dist - sum_radii
            margins.append(margin)
    return np.array(margins)


def postprocess(vars):
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    # Temporary arrays
    x_new = x.copy()
    y_new = y.copy()
    r_new = r.copy()

    # Ensure not outside
    for i in range(NUM_CIRCLES):
        if x_new[i] < r_new[i]:
            x_new[i] = r_new[i]
        if x_new[i] > 1 - r_new[i]:
            x_new[i] = 1 - r_new[i]
        if y_new[i] < r_new[i]:
            y_new[i] = r_new[i]
        if y_new[i] > 1 - r_new[i]:
            y_new[i] = 1 - r_new[i]

    # Fix overlaps if any
    for i in range(NUM_CIRCLES):
        for j in range(i + 1, NUM_CIRCLES):
            dx = x_new[i] - x_new[j]
            dy = y_new[i] - y_new[j]
            dist_sq = dx * dx + dy * dy
            dist = np.sqrt(dist_sq)
            sum_radii = r_new[i] + r_new[j]
            if dist < sum_radii:
                overlap = sum_radii - dist
                # Reduce the larger radius first
                if r_new[i] >= r_new[j]:
                    r_new[i] -= overlap / 2.0
                    r_new[j] += overlap / 2.0
                else:
                    r_new[i] += overlap / 2.0
                    r_new[j] -= overlap / 2.0

    # Rebuild output vector
    vars_corr = np.empty(3 * NUM_CIRCLES)
    vars_corr[0::3] = x_new
    vars_corr[1::3] = y_new
    vars_corr[2::3] = r_new
    return vars_corr


def run_packing():
    best_sum = -np.inf
    best_vars = None

    # Build global constraints and bounds
    cons = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_nonoverlap}
    ]
    bnds = [(0, 1), (0, 1), (0, 0.5)] * NUM_CIRCLES

    for seed in range(NUM_RESTARTS):
        vars0 = initial_guess(seed)
        res = minimize(
            objective,
            vars0,
            method='SLSQP',
            bounds=bnds,
            constraints=cons,
            options={'maxiter': 3000, 'ftol': 1e-7}
        )
        if not res.success:
            continue
        vars_corr = postprocess(res.x)
        s = np.sum(vars_corr[2::3])
        if s > best_sum:
            best_sum = s
            best_vars = vars_corr

    # reshape
    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii = best_vars[2::3]
    return centers, radii, float(best_sum)
```
