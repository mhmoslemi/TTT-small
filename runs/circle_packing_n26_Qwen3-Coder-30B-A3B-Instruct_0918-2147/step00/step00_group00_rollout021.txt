import numpy as np
from scipy.optimize import minimize

def make_hex_lattice(n, margin=0.05):
    # Generate a hexagonal lattice that fits in [margin, 1-margin] x [margin, 1-margin]
    side = 1 - 2 * margin
    sqrt3 = np.sqrt(3)
    dx = side / np.ceil(np.sqrt(n))
    dy = dx * sqrt3 / 2
    
    points = []
    i = 0
    y = margin
    while i < n:
        offset = (y % (dy * 2)) / dy
        x = margin + offset * dx / 2
        while x < 1 - margin and i < n:
            points.append([x, y])
            x += dx
            i += 1
        y += dy
    return np.array(points[:n])

def objective(z):
    # z is arranged as [x0, y0, r0, x1, y1, r1, ...] 
    radii = z[2::3]
    return -np.sum(radii)

def boundary_constraints(z):
    # For each circle i: x_i - r_i >= 0, x_i + r_i <= 1, y_i - r_i >= 0, y_i + r_i <= 1
    # Return array of size 4*n
    n = len(z) // 3
    result = np.zeros(4 * n)
    for i in range(n):
        x, y, r = z[3*i], z[3*i+1], z[3*i+2]
        result[4*i] = x - r  # x - r >= 0
        result[4*i+1] = 1 - (x + r)  # x + r <= 1
        result[4*i+2] = y - r  # y - r >= 0
        result[4*i+3] = 1 - (y + r)  # y + r <= 1
    return result

def pairwise_constraints(z):
    # For all pairs of circles, (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    # Return array of size n*(n-1)/2
    n = len(z) // 3
    num_pairs = n * (n - 1) // 2
    result = np.zeros(num_pairs)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            xi, yi, ri = z[3*i], z[3*i+1], z[3*i+2]
            xj, yj, rj = z[3*j], z[3*j+1], z[3*j+2]
            dist_sq = (xi - xj)**2 + (yi - yj)**2
            sum_radii_sq = (ri + rj)**2
            result[idx] = dist_sq - sum_radii_sq
            idx += 1
    return result

def jacobian_boundary(z):
    # Analytic Jacobian for boundary constraints
    n = len(z) // 3
    J = np.zeros((4*n, 3*n))
    for i in range(n):
        row = 4*i
        col = 3*i
        J[row, col:col+3] = [1, 0, -1]      # x - r >= 0
        J[row+1, col:col+3] = [-1, 0, -1]   # x + r <= 1
        J[row+2, col:col+3] = [0, 1, -1]    # y - r >= 0
        J[row+3, col:col+3] = [0, -1, -1]   # y + r <= 1
    return J

def jacobian_pairwise(z):
    # Analytic Jacobian for pairwise constraints 
    n = len(z) // 3
    num_pairs = n * (n - 1) // 2
    J = np.zeros((num_pairs, 3*n))
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            xi, yi, ri = z[3*i], z[3*i+1], z[3*i+2]
            xj, yj, rj = z[3*j], z[3*j+1], z[3*j+2]
            # Constraint (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2
            # d/dx_i = 2(x_i - x_j)
            # d/dy_i = 2(y_i - y_j) 
            # d/dr_i = -2(r_i + r_j)
            # d/dx_j = -2(x_i - x_j)
            # d/dy_j = -2(y_i - y_j)
            # d/dr_j = -2(r_i + r_j)
            J[idx, 3*i] = 2*(xi - xj)
            J[idx, 3*i+1] = 2*(yi - yj)
            J[idx, 3*i+2] = -2*(ri + rj)
            J[idx, 3*j] = -2*(xi - xj)
            J[idx, 3*j+1] = -2*(yi - yj)
            J[idx, 3*j+2] = -2*(ri + rj)
            idx += 1
    return J

def solve_nlp(initial_z):
    n = len(initial_z) // 3
    def obj(z):
        return objective(z)
    def bnd(z):
        return boundary_constraints(z)
    def pair(z):
        return pairwise_constraints(z)

    bnds = [(0, 1), (0, 1), (0, 0.5)] * n
    constraints = [
        {'type': 'ineq', 'fun': bnd},
        {'type': 'ineq', 'fun': pair}
    ]
    jac_opt = lambda z: -np.ones_like(z)  # We'll add derivative of obj wrt radii
    jac_bnd = jacobian_boundary
    jac_pair = jacobian_pairwise

    sol = minimize(
        obj,
        initial_z,
        method='SLSQP',
        bounds=bnds,
        constraints=constraints,
        jac=jac_opt,
        options={'ftol': 1e-10, 'maxiter': 1000}
    )
    return sol.x

def nudge_centers(centers, radii):
    # For each circle near a boundary, nudge it slightly away
    margin = 1e-4
    for i in range(len(centers)):
        x, y = centers[i]
        r = radii[i]
        # Check which wall it's near
        near_wall = (x < margin + r) or (x > 1 - margin - r) or (y < margin + r) or (y > 1 - margin - r)
        if near_wall:
            # Nudge towards the interior
            if x < margin + r:
                centers[i][0] += 0.001
            elif x > 1 - margin - r:
                centers[i][0] -= 0.001
            if y < margin + r:
                centers[i][1] += 0.001
            elif y > 1 - margin - r:
                centers[i][1] -= 0.001
    return centers
