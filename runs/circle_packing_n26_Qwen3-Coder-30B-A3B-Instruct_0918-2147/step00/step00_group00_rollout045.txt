import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Parameters
    num_circles = 26
    max_iterations = 30
    tolerance = 1e-6
    polygon_sides = 8
    inflation_factor = 0.4
    shrink_factor = 1e-7
    margin = 0.05
    initial_radius = 0.01

    # Generate hexagonal lattice layout
    def generate_hexagonal_layout(n):
        sqrt3 = np.sqrt(3)
        spacing = 0.8 / np.ceil(np.sqrt(n))
        centers = []
        i = 0
        j = 0
        while len(centers) < n:
            x = i * spacing + (j % 2) * spacing / 2
            y = j * spacing * sqrt3 / 2
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers.append([x, y])
            i += 1
            if i * spacing > 1:
                i = 0
                j += 1
        centers = np.array(centers[:n], dtype=np.float64)
        # Add margin
        x_min = np.min(centers[:, 0])
        y_min = np.min(centers[:, 1])
        centers[:, 0] -= x_min - margin
        centers[:, 1] -= y_min - margin
        return centers

    # Compute direction vectors for polygonal approximation
    def compute_directions(k):
        angles = np.linspace(0, 2*np.pi, k, endpoint=False)
        return np.column_stack([np.cos(angles), np.sin(angles)])

    # Initial centers and radii
    centers = generate_hexagonal_layout(num_circles)
    radii = np.full(num_circles, initial_radius, dtype=np.float64)

    directions = compute_directions(polygon_sides)
    cos_pi_over_k = np.cos(np.pi / polygon_sides)

    # For polyhedral approx. of non-overlap constraints
    def compute_polyhedral_bounds(centers, radii):
        """
        Compute the matrix A_ub and vector b_ub representing the constraints:
        A_ub @ radii <= b_ub
        """
        n = centers.shape[0]
        # Wall constraints
        # 4 constraints per circle
        num_wall_constraints = 4 * n
        # Neighbour constraints
        # Pair-wise constraints for each pair
        num_pair_constraints = 0
        for i in range(n):
            for j in range(i + 1, n):
                d = np.linalg.norm(centers[i] - centers[j])
                if d < radii[i] + radii[j] + 1e-6:
                    num_pair_constraints += 1

        total_constraints = num_wall_constraints + num_pair_constraints

        # Allocate matrices
        A_ub = np.zeros((total_constraints, n))
        b_ub = np.zeros(total_constraints)
        constraint_count = 0

        # Wall constraints
        for i in range(n):
            # Left walls
            A_ub[constraint_count, i] = -1
            b_ub[constraint_count] = -centers[i, 0]
            constraint_count += 1
            # Right walls
            A_ub[constraint_count, i] = -1
            b_ub[constraint_count] = centers[i, 0] - 1
            constraint_count += 1
            # Bottom walls
            A_ub[constraint_count, i] = -1
            b_ub[constraint_count] = -centers[i, 1]
            constraint_count += 1
            # Top walls
            A_ub[constraint_count, i] = -1
            b_ub[constraint_count] = centers[i, 1] - 1
            constraint_count += 1

        # Pairwise constraints
        for i in range(n):
            for j in range(i + 1, n):
                d = np.linalg.norm(centers[i] - centers[j])
                if d < radii[i] + radii[j] + 1e-6:
                    # for each direction vector
                    for k in range(directions.shape[0]):
                        u = directions[k]
                        # Constraint: ucdot(ci - cj) <= -(ri + rj) * cos(π/k)
                        A_ub[constraint_count, i] = u[0]
                        A_ub[constraint_count, j] = -u[0]
                        A_ub[constraint_count + 1, i] = u[1]
                        A_ub[constraint_count + 1, j] = -u[1]
                        val = -(radii[i] + radii[j]) * cos_pi_over_k
                        b_ub[constraint_count] = val
                        b_ub[constraint_count + 1] = val
                        constraint_count += 2

        return A_ub[:constraint_count], b_ub[:constraint_count]

    # Solve for max radii given fixed centers
    def solve_max_radii(centers, radii):
        n = centers.shape[0]
        A_ub, b_ub = compute_polyhedral_bounds(centers, radii)
        # Objective is to maximize sum of radii
        c = np.ones(n)
        out = linprog(
            -c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, 0.5)]*n,
            method='highs', options={'tol': 1e-10}
        )
        if not out.success:
            raise RuntimeError("Linear programming for radii failed")
        return np.clip(out.x, 0, 0.5)

        # We will generate the constraint matrix for the optimization:
        # maximize sum of slack variables s_i
        # subject to:
        #  s_i <= c_i.x - r_i
        #  s_i <= 1 - c_i.x - r_i
        #  s_i <= c_i.y - r_i
        #  s_i <= 1 - c_i.y - r_i
        #  s_i + s_j <= u_lcdot(c_i - c_j) + (r_i + r_j) * cos(π/k) for neighbors
        # Slacks added to using extended variable: s_i + 0.5 * (x_i + y_i) -> maximize s_i
        slack_vars = 1  # We're optimizing sum of s_i
        # We're going to build dual LP formulation for small number of constraints.
        # Num variables: n (centers) + n (slacks) + n*2 (also add 2 more with directionals)
        # Better to solve manually for coefficient matrix.
        # Better approach: use inverse formulation.
        # we compute g_i = min( dist(i,j) - (r_i + r_j) ) for all neighbors
        # and if g_i > 0, allow i to expand by fraction of that in final inflation.
        # The real optimization is to avoid degenerate solutions.
        # Instead of solving ideal jacobian or QP, let's formulate a feasible LP for center moves.

    def solve_center_slack_optimization(centers, radii):
        n = centers.shape[0]
        # slate variables s_i >= 0 for slack at each circle center
        # constraints for s_i from wall separation: 
        #  s_i <= c_i.x - r_i     =>  -c_i.x + s_i <= -r_i  
        #  s_i <= 1 - c_i.x - r_i =>   c_i.x + s_i <= 1 - r_i
        #  s_i <= c_i.y - r_i     =>  -c_i.y + s_i <= -r_i
        #  s_i <= 1 - c_i.y - r_i =>   c_i.y + s_i <= 1 - r_i

        # From neighbor pair distance constraint (polygonal approx): 
        # u_lcdot(c_i - c_j) <= -(r_i + r_j) * cos(π/k)
        # we use u_lcdot(c_i - c_j) + (r_i + r_j) * cos(π/k) <= 0
        # which becomes s_i + s_j <= u_lcdot(c_i - c_j) + (r_i + r_j)*cos(π/k) as a surrogate
        # by adding large constant we maximize just s_i (varilde u)
        
        total_constraints = 0
        # Initial wall constraints for every circle
        wall_constraints = 4*n
        # Pairwise overlap constraints for neighbors
        neighbour_pairs = 0
        for i in range(n):
            for j in range(i + 1, n):
                d = np.linalg.norm(centers[i] - centers[j])
                if d < radii[i] + radii[j] + 1e-5:
                    neighbour_pairs += 1
        pair_constraints = neighbour_pairs * 2 * polygon_sides
        total_constraints = wall_constraints + pair_constraints # Approx.
        
        # Cheap LP - work in two parts: box+projection  
        # We use `linprog` with equality bounds 
        # Sample current behavior to estimate
        # Ultimately the dual WP is to avoid obstacles due to overlaps.
        # Once aligned, let's possibly try to refine.

        # Better design: manually frame as an optimizer that increments minimal clearances.
        # Since each continuous variable can be moved to maximize slack marginal 
        # as long as constraints hold. Surprisingly: solution is to scale the problem with custom basis.

        # We define a simple heuristic that moves centers towards safer zones
        # Skip full 3D iteration; let it be simple weighted shuffle + relaxation

        new_centers = centers.copy()
        for i in range(n):

            r = radii[i] 
            safe_x_min = r + 1e-6
            safe_x_max = 1 - r - 1e-6
            safe_y_min = r + 1e-6
            safe_y_max = 1 - r - 1e-6
            
            new_x = np.clip(x, safe_x_min, safe_x_max)
            new_y = np.clip(y, safe_y_min, safe_y_max)
            new_centers[i] = [new_x, new_y]            
        # Repel neighbors slightly
        # For full correctness, rebuild LP or proceed a maximum of few steps on both directions
        # but for practical design and clean code:
        # These grpahic algorithms should in some LPSM/flatten MAP need re-aligning in cycle
        return new_centers

    # Initial solution
    total_sum_old = 0.0
    for i in range(max_iterations):
        # Solve Radius LP
        radii_new = solve_max_radii(centers, radii)
        # Solve Center LP
        centers_new = solve_center_slack_optimization(centers, radii_new)
        # Compute new sum
        sum_new = np.sum(radii_new)
        # Break if no change
        if abs(sum_new - total_sum_old) < tolerance:
            break
        total_sum_old = sum_new
        centers = centers_new
        radii = radii_new

    # Perform Final inflation passes 
    # Compute effective gaps
    for infl_iter in range(5):
        changed = 0
        for i in range(num_circles):
            # Compute min distance to all other circles
            min_gap = 1e9
            # Wall constraints
            x, y = centers[i]
            r = radii[i]
            gap_to_wall = [
                x - r, 1 - x - r,  # left, right walls
                y - r, 1 - y - r   # bottom, top walls
            ]
            min_gap_wall = min(gap_to_wall)
            if min_gap_wall < min_gap:
                min_gap = min_gap_wall
            
            # Overlap constraints
            for j in range(num_circles):
                if i != j:
                    d = np.linalg.norm(centers[i] - centers[j])
                    gap_to_neighbor = d - (radii[i] + radii[j])
                    if gap_to_neighbor < min_gap:
                        min_gap = gap_to_neighbor

            # Allow small inflation if still positive
            if min_gap > 1e-5:
                # Inflation + allowance preserves more
                inflate_by = inflation_factor * min_gap
                # Don't push > 0.5 radius; cap radii
                if radii[i] + inflate_by <= 0.5:
                    radii[i] += inflate_by
                    changed += 1
        if changed == 0:

    attempts = 0
    while not valid and attempts < 5:
        radii *= (1 - shrink_factor)
        valid, msg = validate_packing(centers, radii)
        attempts += 1

    if not valid:
        raise RuntimeError(f"Final packing validation failed: {msg}")
    
    return centers, radii, np.sum(radii)
