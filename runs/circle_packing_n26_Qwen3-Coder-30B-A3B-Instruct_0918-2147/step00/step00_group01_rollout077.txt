import numpy as np
from scipy.spatial.distance import cdist

def penalty(z, alpha=10.0, beta=10.0):
    n = 26
    centers = z[:2*n].reshape(n, 2)
    radii = z[2*n:]
    
    # Wall penalties
    walls = np.array([
        centers[:, 0],           # x
        1 - centers[:, 0],       # 1 - x
        centers[:, 1],           # y
        1 - centers[:, 1]        # 1 - y
    ])
    wall_viol = np.maximum(0, radii - np.min(walls, axis=0))
    wall_penalty = alpha * np.sum(wall_viol)

    # Overlap penalties
    dists = cdist(centers, centers)
    np.fill_diagonal(dists, np.inf)
    overlap_viol = np.maximum(0, radii[:, None] + radii[None, :] - dists)
    overlap_penalty = beta * np.sum(overlap_viol) / 2  # divide by 2 to avoid double counting

    return wall_penalty + overlap_penalty

def energy(z, lam, alpha=10.0, beta=10.0):
    sum_radii = np.sum(z[2*26:])
    return -sum_radii + lam * penalty(z, alpha, beta)

def propose_move(z, sigma_c, sigma_r):
    z_new = z.copy()
    n = 26
    move_type = np.random.choice(3)
    if move_type == 0:  # Center jitter
        i = np.random.randint(n)
        z_new[2*i] += np.random.normal(0, sigma_c)
        z_new[2*i+1] += np.random.normal(0, sigma_c)
        z_new[2*i] = np.clip(z_new[2*i], 0, 1)
        z_new[2*i+1] = np.clip(z_new[2*i+1], 0, 1)
    elif move_type == 1:  # Radius perturb
        i = np.random.randint(n)
        z_new[2*n + i] += np.random.normal(0, sigma_r)
        z_new[2*n + i] = np.maximum(0, z_new[2*n + i])
    else:  # Swap pair
        i, j = np.random.choice(n, size=2, replace=False)
        z_new[2*n + i], z_new[2*n + j] = z_new[2*n + j], z_new[2*n + i]
    return z_new

def anneal():
    n = 26
    T0 = 1.0
    c_T = 0.995
    T_min = 1e-5
    lam0 = 0.1
    c_lam = 1.03
    max_iter = 10000
    sigma_c = 0.03
    sigma_r = 0.02

    # Initial hexagonal grid placement
    spacing = 0.18
    grid_x = np.arange(0.05, 1.0, spacing)
    grid_y = np.arange(0.05, 1.0, spacing)
    centers = np.array([[x, y] for y in grid_y for x in grid_x if (x - 0.05)**2 + (y - 0.05)**2 < 0.9**2][:n])
    if len(centers) < n:
        # Try again with smaller spacing
        spacing = 0.16
        grid_x = np.arange(0.05, 1.0, spacing)
        grid_y = np.arange(0.05, 1.0, spacing)
        centers = np.array([[x, y] for y in grid_y for x in grid_x if (x - 0.05)**2 + (y - 0.05)**2 < 0.9**2][:n])
    
    # Pad or truncate to exactly n centers
    if len(centers) < n:
        extra = n - len(centers)
        for i in range(extra):
            centers = np.vstack([centers, np.random.rand(2)])
    else:
        centers = centers[:n]
    
    # Assign radii
    radii = np.array([min(c[0], 1 - c[0], c[1], 1 - c[1]) for c in centers])
    
    z = np.hstack([centers.flatten(), radii])
    
    best_feasible = z.copy()
    best_feasible_sum = np.sum(radii)
    current_z = z.copy()
    current_energy = energy(z, lam0)
    
    T = T0
    lam = lam0
    
    iter = 0
    while T > T_min and iter < max_iter:
        z_proposed = propose_move(current_z, sigma_c * np.sqrt(T / T0), sigma_r * np.sqrt(T / T0))
        # Clip to bounds for proposed
        z_proposed[:2*n] = np.clip(z_proposed[:2*n], 0, 1)
        z_proposed[2*n:] = np.maximum(0, z_proposed[2*n:])
        
        E_proposed = energy(z_proposed, lam)
        dE = E_proposed - current_energy
        
        accept = (dE <= 0) or (np.random.rand() < np.exp(-dE / T))
        
        if accept:
            current_z = z_proposed
            current_energy = E_proposed
        
        # Check feasibility of current best
        p = penalty(current_z)
        if p < 1e-10 and np.sum(current_z[2*n:]) > best_feasible_sum:
            best_feasible = current_z.copy()
            best_feasible_sum = np.sum(current_z[2*n:])
        
        T *= c_T
        lam *= c_lam
        iter += 1
    
    return best_feasible[:2*n].reshape(n, 2), best_feasible[2*n:]

def project_and_ascent(centers, radii):
    n = centers.shape[0]
    z = np.hstack([centers.flatten(), radii])
    
    eta = 5e-3
    max_iter = 500
    tol = 1e-7
    
    for _ in range(max_iter):
        # Gradient step: increase radii, keep centers same
        z_new = z.copy()
        z_new[2*n:] += eta
        
        # Apply projections to new values
        centers_new = z_new[:2*n].reshape(n, 2)
        radii_new = z_new[2*n:]
        
        # Clamp radii to wall bounds
        wall_bounds = np.array([
            centers_new[:, 0],           # x
            1 - centers_new[:, 0],       # 1 - x
            centers_new[:, 1],           # y
            1 - centers_new[:, 1]        # 1 - y
        ])
        max_radii = np.min(wall_bounds, axis=0)
        radii_new = np.minimum(radii_new, max_radii)
        radii_new = np.maximum(radii_new, 0)
        
        # Handle overlaps
        centers_new = centers_new.copy()
        radii_new = radii_new.copy()
        
        # Repeatedly resolve overlaps
        max_overlap_iter = 10
        for _ in range(max_overlap_iter):
            dists = cdist(centers_new, centers_new)
            np.fill_diagonal(dists, np.inf)
            
            overlap_amounts = radii_new[:, None] + radii_new[None, :] - dists
            overlap_indices = np.triu(np.where(overlap_amounts > 0), k=1)
            if not overlap_indices[0].size:
                break
            
            # Reduce radii proportionally
            for i, j in zip(overlap_indices[0], overlap_indices[1]):
                o = overlap_amounts[i, j]
                if o > 0:
                    radii_new[i] -= o / 2
                    radii_new[j] -= o / 2
            
            # Reclamp
            wall_bounds = np.array([
                centers_new[:, 0],
                1 - centers_new[:, 0],
                centers_new[:, 1],
                1 - centers_new[:, 1]
            ])
            max_radii = np.min(wall_bounds, axis=0)
            radii_new = np.minimum(radii_new, max_radii)
            radii_new = np.maximum(radii_new, 0)
        
        z_new = np.hstack([centers_new.flatten(), radii_new])
        
        # Apply constraint to converge to feasible region if there are any infeasibilities
        new_sum = np.sum(radii_new)
        delta = new_sum - np.sum(z[2*n:])
        
        if delta < tol:
            break
            
        z = z_new
        eta *= 0.95
    
    return z[:2*n].reshape(n, 2), z[2*n:]

def run_packing():
    # Annealing phase
    centers, radii = anneal()
    # Local refinement
    centers, radii = project_and_ascent(centers, radii)
    
    # Validate
    success, msg = validate_packing(centers, radii)
    if not success:
        # Make a minimal adjustment
        radii = radii * (1 - 1e-7)
        success, msg = validate_packing(centers, radii)
        if not success:
            raise AssertionError("Validation failed after adjustment: " + msg)
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
