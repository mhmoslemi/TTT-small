import numpy as np
from scipy.optimize import linprog
import math

def initial_hex_centers(k, spacing):
    """Generate k non-overlapping centers in a hexagonal lattice."""
    sqrt3 = math.sqrt(3)
    centers = []
    i = 0
    j = 0
    while len(centers) < k:
        x = i * spacing + (j % 2) * spacing / 2
        y = j * spacing * sqrt3 / 2
        if 0 <= x <= 1 and 0 <= y <= 1:
            centers.append([x, y])
        if x > 1:
            j += 1
            i = -j // 2
        else:
            i += 1
    return np.array(centers[:k])

def initial_radii(k, r0):
    """Return array of initial radii."""
    return np.full(k, r0)

def build_lp_constants(centers, radii):
    """Compute all constants needed for the LP."""
    n = centers.shape[0]
    d_left = centers[:, 0]
    d_right = 1 - centers[:, 0]
    d_bottom = centers[:, 1]
    d_top = 1 - centers[:, 1]
    
    # Compute pairwise distances
    D_ij = np.zeros((n, n))
    for i in range(n):
        for j in range(i+1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            D_ij[i, j] = dist
            D_ij[j, i] = dist
            
    return {
        'd_left': d_left,
        'd_right': d_right,
        'd_bottom': d_bottom,
        'd_top': d_top,
        'D_ij': D_ij
    }

def solve_radius_lp(consts, radii, graph, alpha=0.5):
    """Solve linear program to find optimal radius increments."""
    n = len(radii)
    # Build constraints matrix and bounds
    A_ub = []
    b_ub = []
    
    # Wall constraints
    for i in range(n):
        rhs = min(consts['d_left'][i], consts['d_right'][i],
                  consts['d_bottom'][i], consts['d_top'][i]) - radii[i]
        if rhs < 0:
            rhs = 0
        A_ub.append([-1.0])  # -Δr_i <= -rhs
        b_ub.append(-rhs)
        
    # Pairwise constraints
    for i in range(n):
        for j in range(i+1, n):
            if consts['D_ij'][i, j] > 0:
                rhs = consts['D_ij'][i, j] - (radii[i] + radii[j])
                if rhs >= 0:
                    coef = np.zeros(n)
                    coef[i] = 1.0
                    coef[j] = 1.0
                    A_ub.append(-coef)
                    b_ub.append(-rhs)
    
    # Optional graph-based constraints (transfer limits)
    if graph is not None and len(graph) > 0:
        for i in range(n):
            neighbor_sum = 0.0
            for (u, v) in graph:
                if u == i:
                    neighbor_sum += consts['D_ij'][u, v] - (radii[u] + radii[v])
                elif v == i:
                    neighbor_sum += consts['D_ij'][u, v] - (radii[u] + radii[v])
            if neighbor_sum > 0:
                A_ub.append([0.0] * i + [-1.0] + [0.0] * (n - i - 1))  # -Δr_i <= -alpha * neighbor_sum
                b_ub.append(-alpha * neighbor_sum)
                
    # Objective: maximize sum of Δr_i, i.e., minimize -sum(Δr_i)
    c = np.full(n, -1.0)
    
    # Solve LP
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
        
    return np.zeros(n)

def apply_increment(radii, Δr):
    """Apply Δr increments."""
    return radii + Δr

def push_feasibility(centers, radii):
    """Apply deterministic push to satisfy constraints."""
    n = len(radii)
    # First ensure circles are within bounds
    r_new = np.copy(radii)
    c_new = np.copy(centers)
    
    updated = True
    while updated:
        updated = False
        for i in range(n):
            # Enforce bounds
            if c_new[i, 0] - r_new[i] < 0:
                r_new[i] = c_new[i, 0]
                updated = True
            if c_new[i, 0] + r_new[i] > 1:
                r_new[i] = 1 - c_new[i, 0]
                updated = True
            if c_new[i, 1] - r_new[i] < 0:
                r_new[i] = c_new[i, 1]
                updated = True
            if c_new[i, 1] + r_new[i] > 1:
                r_new[i] = 1 - c_new[i, 1]
                updated = True
                
        # Resolve any overlaps by moving circles
        for i in range(n):
            for j in range(i+1, n):
                dist = np.sqrt(np.sum((c_new[i] - c_new[j])**2))
                if dist < r_new[i] + r_new[j]:
                    overlap = r_new[i] + r_new[j] - dist
                    if dist > 1e-15:
                        dx = (c_new[i, 0] - c_new[j, 0]) / dist
                        dy = (c_new[i, 1] - c_new[j, 1]) / dist
                        drx = (overlap / 2) * dx
                        dry = (overlap / 2) * dy
                        c_new[i, 0] -= drx
                        c_new[i, 1] -= dry
                        c_new[j, 0] += drx
                        c_new[j, 1] += dry
                        updated = True
                        
    return c_new, r_new

def run_iterations(centers, radii, max_iter, ε, α):
    """Main iterative algorithm."""
    graph = set()
    
    for iteration in range(max_iter):
        # Build LP constants
        consts = build_lp_constants(centers, radii)
        
        # Solve LP for Δr
        Δr = solve_radius_lp(consts, radii, list(graph), α)
        
        # Check convergence
        total_delta = np.sum(Δr)
        if total_delta < ε:
            break
            
        # Apply increments
        new_radii = apply_increment(radii, Δr)
        
        # Micro-push: apply feasibility restoration
        centers, new_radii = push_feasibility(centers, new_radii)
        
        # Update radii
        radii = new_radii
            
    return centers, radii

def final_refinement(centers, radii):
    """Apply final high-precision LP and clipping."""
    n = len(radii)
    consts = build_lp_constants(centers, radii)
    Δr = solve_radius_lp(consts, radii, [], α=0.0)
    final_radii = apply_increment(radii, Δr)
    
    # Final clipping for strict validation
    for i in range(n):
        final_radii[i] = min(final_radii[i], 
                            centers[i, 0], 
                            1 - centers[i, 0], 
                            centers[i, 1], 
                            1 - centers[i, 1])
    
    # Ensure no overlaps
    for i in range(n):
        for j in range(i+1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            if dist < final_radii[i] + final_radii[j]:
                overlap = final_radii[i] + final_radii[j] - dist
                final_radii[i] -= overlap * 0.5
                final_radii[j] -= overlap * 0.5
    
    return centers, final_radii

def run_packing():
    """Run the full packing algorithm."""
    # Initial hexagonal configuration
    initial_spacing = 0.13
    centers = initial_hex_centers(26, initial_spacing)
    radii = initial_radii(26, 0.005)
    
    # Run iterations
    centers, radii = run_iterations(centers, radii, 100, 1e-5, 0.5)
    
    # Final refinement
    centers, radii = final_refinement(centers, radii)
    
    # Compute sum of radii
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
