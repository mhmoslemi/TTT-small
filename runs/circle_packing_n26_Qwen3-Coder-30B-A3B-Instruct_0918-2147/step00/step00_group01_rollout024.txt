import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
from itertools import combinations

def generate_lattice(s):
    """Generate a hexagonal lattice with spacing s inside unit square."""
    sqrt3 = np.sqrt(3)
    rows = int(np.ceil(1.0 / (s * sqrt3 / 2))) + 2
    cols = int(np.ceil(1.0 / s)) + 2
    points = []
    for r in range(rows):
        y = r * s * sqrt3 / 2
        if y > 1 - s/2:
            break
        offset = 0.5 if r % 2 == 1 else 0.0
        for c in range(cols):
            x = (c + offset) * s
            if x > 1 - s/2:
                break
            if s/2 <= x <= 1 - s/2 and s/2 <= y <= 1 - s/2:
                points.append([x, y])
    return np.array(points)

def build_milp(P):
    """Build a MILP model given a list of candidate points P."""
    K = len(P)
    z = cp.Variable(K, boolean=True)
    r = cp.Variable(K, nonneg=True)
    
    def compute_wall_distances(p):
        return np.array([p[0], 1 - p[0], p[1], 1 - p[1]])
    
    w = np.array([np.min(compute_wall_distances(p)) for p in P])
    
    # Compute all pairwise distances
    d = np.zeros((K, K))
    for i in range(K):
        for j in range(i+1, K):
            dist = np.linalg.norm(P[i] - P[j])
            d[i, j] = d[j, i] = dist
    
    # Constraints: exactly 26 circles
    constraints = [cp.sum(z) == 26]
    
    # Link radius to selection
    for i in range(K):
        constraints.append(r[i] <= w[i] * z[i])
    
    # Non-overlap constraints
    M = 1.0
    for i, j in combinations(range(K), 2):
        if d[i, j] > 0:
            constraints.append(d[i, j] >= r[i] + r[j] - M * (2 - z[i] - z[j]))
    
    # Objective
    objective = cp.Maximize(cp.sum(r))
    
    # Return solver function
    def solve_model(fixed_z=None):
        prob = cp.Problem(objective, constraints)
        if fixed_z is not None:
            for i in range(K):
                z[i].value = fixed_z[i]
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status != cp.OPTIMAL:
            return None, None
        return [r_val.value for r_val in r], prob.value
    
    return solve_model

def branch_and_bound(milp_solver, node_limit=20000):
    """Custom branch-and-bound solver."""
    best_obj = -np.inf
    best_z = None
    best_r = None
    
    # Priority queue: list of (bound, z_fixed)
    pq = []
    
    # Initialize with all z = 0
    initial_z = np.zeros(len(milp_solver.__code__.co_consts), dtype=int)
    root_bound, _ = milp_solver(initial_z)
    if root_bound is not None:
        pq.append((-root_bound, initial_z.copy()))
    
    node_count = 0
    while pq and node_count < node_limit:
        node_count += 1
        _, z = pq.pop(0)
        
        # Check if all binary variables are fixed
        if z.sum() == 26:
            # Solve with current z
            r_vals, obj_val = milp_solver(z)
            if obj_val > best_obj:
                best_obj = obj_val
                best_r = r_vals.copy()
                best_z = z.copy()
            continue
        
        # Choose branching variable (closest to 0.5)
        free_vars = np.where(z == -1)[0] if any(z == -1) else np.where(np.logical_and(z == 0, np.arange(len(z)) < 26))[0]
        if not free_vars.size:
            # Fix unassigned variables to 1
            z_copy = z.copy()
            for i in range(len(z_copy)):
                if z_copy[i] == -1 and z_copy.sum() < 26:
                    z_copy[i] = 1
                    if z_copy.sum() == 26:
                        break
            r_vals, obj_val = milp_solver(z_copy)
            if obj_val > best_obj:
                best_obj = obj_val
                best_r = r_vals.copy()
                best_z = z_copy.copy()
            continue
            
        branch_var = free_vars[int(len(free_vars)/2)]  # simple heuristic
        
        z0 = z.copy()
        z1 = z.copy()
        z0[branch_var] = 0
        z1[branch_var] = 1
        
        r0, obj0 = milp_solver(z0)
        if obj0 is not None: # This is likely quadratic, but for now this line is necessary
            if obj0 > best_obj:
                # Too costly, but we're OK with this since branching isn't perfect.
                pass
            # Consider if we want to maintain this info later or not.
            bound0 = -obj0
            pq.append((bound0, z0.copy()))
        
        r1, obj1 = milp_solver(z1)
        if obj1 is not None:
            # Update if better.
            if obj1 > best_obj:
                best_obj = obj1
                best_r = r1.copy()
                best_z = z1.copy()
            bound1 = -obj1
            pq.append((bound1, z1.copy()))
        
        # Sort queue by bound
        pq.sort(key=lambda x: x[0])
    
    # If no valid solution found, return the best one
    return best_z, best_r

def refine_solution(centers, radii):
    """Refine centers and radii using continuous optimization."""
    def obj_func(x):
        # Input x: [c1_x, c1_y, r1, c2_x, c2_y, r2, ...]
        total = 0
        n = len(centers)
        for i in range(n):
            total += x[2*i + 2]
        return -total  # Maximize sum of radii (minimize negative)
    
    def constraint_func(x):
        # Non-overlap constraints: (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
        n = len(centers)
        ineqs = []
        for i in range(n):
            for j in range(i+1, n):
                xi = x[2*i]
                yi = x[2*i + 1]
                ri = x[2*i + 2]
                xj = x[2*j]
                yj = x[2*j + 1]
                rj = x[2*j + 2]
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                penalty = dist_sq - (ri + rj)**2
                ineqs.append(penalty)
                
        # Wall constraints
        for i in range(n):
            xi = x[2*i]
            yi = x[2*i + 1]
            ri = x[2*i + 2]
            ineqs.append(xi - ri)
            ineqs.append(1 - xi - ri)
            ineqs.append(yi - ri)
            ineqs.append(1 - yi - ri)
            
        return np.array(ineqs)
    
    # Prepare variables
    n = len(centers)
    bounds = []
    x0 = []
    for i in range(n):
        # Add center (x_i, y_i)
        bounds.extend([(0, 1), (0, 1)])
        x0.extend([centers[i][0], centers[i][1]])
        # Add radius r_i
        bounds.append((0, 1))
        x0.append(float(radii[i]))
    
    # Set up optimization
    constraints = {"type": "ineq", "fun": constraint_func}
    res = minimize(
        obj_func,
        x0,
        bounds=bounds,
        constraints=constraints,
        method="SLSQP",
        options={"ftol": 1e-9, "maxiter": 2000}
    )
    
    if not res.success:
        return centers, radii, np.sum(radii)
        
    # Parse result
    refined_centers = []
    refined_radii = []
    for i in range(n):
        refined_centers.append([res.x[2*i], res.x[2*i + 1]])
        refined_radii.append(res.x[2*i + 2])
    
    refined_centers = np.array(refined_centers)
    refined_radii = np.array(refined_radii)
    
    # Validate against small numerical accidents
    sum_radii = np.sum(refined_radii)
    return refined_centers, refined_radii, sum_radii
