def build_hex_lattice(spacing):
    """Create a hexagonal lattice of points inside the unit square."""
    points = []
    y = spacing
    while y < 1 - spacing:
        x = spacing
        while x < 1 - spacing:
            points.append([x, y])
            x += spacing * 1.5
        y += spacing * sqrt(3) / 2
    return np.array(points)

def generate_combinations(items, k):
    """Generate combinations of size k from items."""
    if k == 0:
        yield []
    elif len(items) == k:
        yield items
    elif k == 1:
        for i in items:
            yield [i]
    else:
        for i in range(len(items)):
            for subcombo in generate_combinations(items[i+1:], k-1):
                yield [items[i]] + subcombo

def local_optimize(C, R0):
    """Refine with scipy."""
    n = C.shape[0]
    x0 = np.hstack([C.flatten(), R0])
    bounds = []
    for i in range(n):
        bounds.extend([(0, 1), (0, 1), (0, 0.5)])  # x, y, r
    bounds = bounds[:3*n]
    
    def obj(x):
        return -sum(x[2::3])  # maximize negated sum
        
    def constraint_func(x):
        c = C.copy()
        r = x[2::3]
        c[:,0] = x[0::3]
        c[:,1] = x[1::3]
        result = []
        for i in range(n):
            # Wall constraints
            if c[i,0] < r[i]:
                result.append(r[i] - c[i,0])  # r >= x (left)
            if c[i,0] > 1 - r[i]:
                result.append(1 - r[i] - c[i,0])  # 1-r >= x (right)
            if c[i,1] < r[i]:
                result.append(r[i] - c[i,1])  # r >= y (bottom)
            if c[i,1] > 1 - r[i]:
                result.append(1 - r[i] - c[i,1])  # 1-r >= y (top)
        return np.array(result)
    
    def constraint_func2(x):
        # Non-overlap constraints
        c = C.copy()
        r = x[2::3]
        c[:,0] = x[0::3]
        c[:,1] = x[1::3]
        result = []
        for i in range(n):
            for j in range(i+1, n):
                dx = c[i,0] - c[j,0]
                dy = c[i,1] - c[j,1]
                dist_sq = dx*dx + dy*dy
                sum_radii = r[i] + r[j]
                result.append(sum_radii - np.sqrt(dist_sq))
        return np.array(result)
    
    constraints = [
        {'type': 'ineq', 'fun': constraint_func},
        {'type': 'ineq', 'fun': constraint_func2}
    ]
    
    res = minimize(obj, x0, bounds=bounds, constraints=constraints,
                   method='trust-constr', options={'maxiter': 500})
    
    r_final = res.x[2::3]
    c_final = res.x[0::3].reshape(n, 2)
    return c_final, r_final
