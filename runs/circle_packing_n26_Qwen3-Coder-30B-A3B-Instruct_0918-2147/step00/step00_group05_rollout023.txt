import numpy as np

def hex_lattice(d):
    """Generate a hexagonal lattice within [0,1]x[0,1] with spacing d."""
    sqrt3_2 = np.sqrt(3) / 2
    rows = []
    y = 0
    row = 0
    while y <= 1:
        x = 0
        offset = (row % 2) * d / 2
        while x <= 1:
            rows.append([x + offset, y])
            x += d
        y += sqrt3_2 * d
        row += 1
    return np.array(rows)

def add_border_circles(points, d):
    """Add 6 border-enhanced circles to the core pattern."""
    r = d / 4
    border_points = [
        [r, r],                        # bottom-left
        [1 - r, r],                    # bottom-right
        [r, 1 - r],                    # top-left
        [1 - r, 1 - r],                # top-right
        [0.5, r],                      # bottom mid
        [0.5, 1 - r]                   # top mid
    ]
    return np.vstack([points, border_points])

def initial_radii(points):
    """Assign a safe initial radius to each point."""
    wall_distances = np.minimum(points[:, 0], 1 - points[:, 0])
    wall_distances = np.minimum(wall_distances, points[:, 1])
    wall_distances = np.minimum(wall_distances, 1 - points[:, 1])
    n = points.shape[0]
    min_distances = np.full(n, np.inf)
    for i in range(n):
        diffs = points[i] - points
        distances = np.sqrt(np.sum(diffs**2, axis=1))
        distances[i] = np.inf  # Ignore self
        min_distances[i] = np.min(distances)
    safe_radii = np.minimum(wall_distances, min_distances / 2)
    return np.min(safe_radii) * np.ones(n)

def inflate_radii(centers, radii):
    """Increase all radii to maximum feasible values."""
    n = centers.shape[0]
    new_radii = radii.copy()
    # Precompute all distances
    D = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    epsilon = 1e-12
    for i in range(n):
        # Wall constraints
        wall_constraints = [
            centers[i, 0] - radii[i],
            1 - centers[i, 0] - radii[i],
            centers[i, 1] - radii[i],
            1 - centers[i, 1] - radii[i]
        ]
        # Pairwise constraints
        pairwise_constraints = []
        for j in range(n):
            if i != j:
                pairwise_constraints.append(D[i, j] - radii[j])
        # The new radius can be no larger than the minimum of all constraints
        max_radius = min(wall_constraints + pairwise_constraints)
        if max_radius > new_radii[i] + epsilon:
            new_radii[i] = max_radius
    return new_radii

def relax_centers(centers, radii, steps=10):
    """Apply a force-based relaxation to improve spacing."""
    n = centers.shape[0]
    kappa = 0.02
    lambda_wall = 0.01
    alpha = 0.5
    gamma = 0.02
    beta = 0.02
    moved_centers = centers.copy()
    clamped_centers = centers.copy()
    for step in range(steps):
        forces = np.zeros((n, 2))
        # Wall forces
        for i in range(n):
            x, y = moved_centers[i]
            r = radii[i]
            fx = 0.0
            fy = 0.0
            if x - r < gamma:
                fx += kappa * (gamma - (x - r))
            if 1 - x - r < gamma:
                fx -= kappa * (gamma - (1 - x - r))
            if y - r < gamma:
                fy += kappa * (gamma - (y - r))
            if 1 - y - r < gamma:
                fy -= kappa * (gamma - (1 - y - r))
            forces[i, 0] += fx
            forces[i, 1] += fy
        # Pairwise repulsion
        distances = np.linalg.norm(
            moved_centers[:, None, :] - moved_centers[None, :, :], axis=2)
        for i in range(n):
            for j in range(n):
                if i == j:
                    continue
                dist = distances[i, j]
                r_i, r_j = radii[i], radii[j]
                if dist < r_i + r_j + beta:
                    delta = (r_i + r_j) - dist
                    if delta > 0:
                        f = kappa * delta * (moved_centers[i] - moved_centers[j]) / (dist + 1e-12)
                        forces[i, :] += f
        # Update
        moved_centers += alpha * forces
        # Clamp to space
        for i in range(n):
            x, y = moved_centers[i]
            r = radii[i]
            moved_centers[i, 0] = np.clip(x, r, 1 - r)
            moved_centers[i, 1] = np.clip(y, r, 1 - r)
    return moved_centers

    d = 0.18

    # 1. Generate initial pattern
    core_points = hex_lattice(d)

    previous_sum = np.sum(radii)
    for _ in range(MAX_OUTER):
        # Inflate radii
        radii = inflate_radii(centers, radii)
        # Relax centers
        centers = relax_centers(centers, radii)
        # Check stopping condition
