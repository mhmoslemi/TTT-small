import numpy as np
from scipy.spatial.distance import cdist

def penalty(z, alpha, beta):
    n = 26
    centers = z[:2*n].reshape(n, 2)
    radii = z[2*n:]
    
    # Wall penalty
    wall_pen = 0.0
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        dists = [x, 1-x, y, 1-y]
        for d in dists:
            if r > d:
                wall_pen += alpha * (r - d)
    
    # Overlap penalty
    overlap_pen = 0.0
    dist_matrix = cdist(centers, centers)
    for i in range(n):
        for j in range(i+1, n):
            dist = dist_matrix[i, j]
            r_i = radii[i]
            r_j = radii[j]
            if dist < r_i + r_j:
                overlap_pen += beta * (r_i + r_j - dist)
    
    return wall_pen + overlap_pen

def energy(z, lam, alpha, beta):
    sum_radii = np.sum(z[2*26:])
    return -sum_radii + lam * penalty(z, alpha, beta)

def propose_move(z, sigma_c, sigma_r):
    z_new = z.copy()
    n = 26
    choice = np.random.choice(3)
    
    if choice == 0:  # Center jitter
        i = np.random.randint(n)
        z_new[2*i] += np.random.normal(0, sigma_c)
        z_new[2*i+1] += np.random.normal(0, sigma_c)
        z_new[2*i] = np.clip(z_new[2*i], 0, 1)
        z_new[2*i+1] = np.clip(z_new[2*i+1], 0, 1)
    elif choice == 1:  # Radius perturb
        i = np.random.randint(n)
        z_new[2*n + i] += np.random.normal(0, sigma_r)
        z_new[2*n + i] = max(0, z_new[2*n + i])
    else:  # Swap pair
        i, j = np.random.choice(n, 2, replace=False)
        r_i = z_new[2*n + i]
        r_j = z_new[2*n + j]
        z_new[2*n + i] = r_j
        z_new[2*n + j] = r_i
    
    return z_new

def anneal():
    n = 26
    # Initial hex grid configuration
    spacing = 0.18
    x_grid = np.arange(spacing, 1 - spacing, spacing)
    y_grid = np.arange(spacing, 1 - spacing, spacing)
    centers = []
    for x in x_grid:
        for y in y_grid:
            if len(centers) < n:
                centers.append([x, y])
    if len(centers) < n:
        # Fill remaining positions
        # Place in a pattern to get 26 circles approximately
        centers.extend([[0.5, 0.5]] * (n - len(centers)))
    
    centers = np.array(centers)
    radii = np.minimum(centers[:, 0], 1 - centers[:, 0])
    radii = np.minimum(radii, centers[:, 1])
    radii = np.minimum(radii, 1 - centers[:, 1])
    
    # Initialize z vector
    z = np.concatenate([centers.flatten(), radii])
    
    # Parameters
    alpha = 10.0
    beta = 10.0
    T0 = 1.0
    c_T = 0.995
    c_lambda = 1.03
    T = T0
    lam = 0.1
    sigma_c = 0.03
    sigma_r = 0.02
    
    # Annealing loop
    best_feasible_z = None
    best_feasible_sum = -np.inf
    best_z = z.copy()
    best_energy = energy(z, lam, alpha, beta)
    
    # Run the loop for a fixed number of outer iterations
    for i in range(10000):
        # Scale sigma by temperature
        scaled_sigma_c = sigma_c * np.sqrt(T/T0)
        scaled_sigma_r = sigma_r * np.sqrt(T/T0)
        
        # Propose a move
        z_proposed = propose_move(z, scaled_sigma_c, scaled_sigma_r)
        
        # Calculate energy change
        e_proposed = energy(z_proposed, lam, alpha, beta)
        delta_e = e_proposed - energy(z, lam, alpha, beta)
        
        # Accept or reject
        if delta_e <= 0 or np.random.rand() < np.exp(-delta_e / T):
            z = z_proposed
        
        # Update best if feasible
        p = penalty(z, alpha, beta)
        if p < 1e-10:  # Feasible (no penalty)
            sum_radii = np.sum(z[2*n:])
            if sum_radii > best_feasible_sum:
                best_feasible_sum = sum_radii
                best_feasible_z = z.copy()
        
        # Update best overall
        if e_proposed < best_energy:
            best_energy = e_proposed
            best_z = z.copy()
        
        # Update temperature and penalty weight
        T *= c_T
        lam *= c_lambda
        
        if T < 1e-5:
            break
    
    # Use the best feasible configuration if available
    if best_feasible_z is not None:
        z = best_feasible_z
    
    return z

def project_and_ascent(centers, radii, max_iters=500):
    n = len(centers)
    # Start with a copy to avoid modifying the arrays in place
    centers = centers.copy()
    radii = radii.copy()
    
    # Gradient ascent with projection
    eta = 5e-3
    for it in range(max_iters):
        # Compute overlap pairs, note that we consider all pairs
        # even though we don't modify the gradient
        changed = False
        # Project to boundaries
        for i in range(n):
            r = radii[i]
            x, y = centers[i]
            r = min(r, x, 1-x, y, 1-y)
            if r != radii[i]:
                radii[i] = r
                changed = True
        
        # Make sure no overlap exists
        max_overlaps = 10  # arbitrary max passes to prevent infinite loop
        pass_count = 0
        while pass_count < max_overlaps:
            overlap_found = False
            # Detect all pairs with overlaps
            dist_matrix = cdist(centers, centers)
            for i in range(n):
                for j in range(i+1, n):
                    dist = dist_matrix[i, j]
                    # Use original radii for safe computation
                    r_i = radii[i]
                    r_j = radii[j]
                    if dist < r_i + r_j - 1e-10:
                        overlap = (r_i + r_j - dist)
                        # Reduce the two radii proportionally to remove the overlap
                        # For computational simplicity, simply reduce both by overlap/2
                        r_i_new = max(0, r_i - overlap/2)
                        r_j_new = max(0, r_j - overlap/2)
                        overlap_found = True
                        changed = True
                        # Update the radii
                        radii[i] = r_i_new
                        radii[j] = r_j_new
            if not overlap_found:
                break
            pass_count += 1
        
        # Apply boundary clamping after overlap resolution
        for i in range(n):
            r = radii[i]
            x, y = centers[i]
            r = min(r, x, 1-x, y, 1-y)
            radii[i] = r
        
        # If no actual change occurred, stop
        if not changed:
            break
        
        # Grad step: maximize sum of radii
        eta *= 0.95
        if eta < 1e-6:
            break
    
    return centers, radii
