import numpy as np
from scipy.optimize import minimize
import scipy.sparse as sp

def init_hex_grid(n=26, r_init=0.04):
    # Generate hexagonal grid points
    h = np.sqrt(3) * r_init
    rows = []
    y = r_init
    row_idx = 0
    while len(rows) < n:
        row = []
        x = r_init
        if row_idx % 2 == 1:
            x += r_init / 2
        while len(row) < 5 and len(rows) + len(row) < n:
            row.append((x, y))
            x += 2 * r_init
        if row:
            rows.append(row)
        y += h
        row_idx += 1
        if y + r_init > 1:
            break

    # Flatten and select only first n points
    points = []
    for row in rows:
        points.extend(row)
    points = points[:n]
    
    centers = np.array(points)
    radii = np.full(n, r_init)
    return centers, radii

def max_radii(centers, radii):
    # Compute wall limits
    wall = np.minimum(centers, 1 - centers).min(axis=1)
    
    # Compute inter-circle distances
    D = np.sqrt(((centers[:,None,:] - centers[None,:,:])**2).sum(-1))
    
    # Compute neighbour limits: for each circle, what's the max radius it can have
    # such that it doesn't overlap with any other circle
    neighbor = D - radii[:,None]
    np.fill_diagonal(neighbor, np.inf)
    neighbor = np.maximum(neighbor, 0).min(axis=1)
    
    new_radii = np.minimum(wall, neighbor)
    new_radii = np.maximum(new_radii, 0)
    
    return new_radii

def hill_climb(centers, radii, iterations=5000, step=0.01):
    # Create a copy to work with
    centers = centers.copy()
    radii = radii.copy()
    
    # Precompute distance matrix for efficiency
    D = np.sqrt(((centers[:,None,:] - centers[None,:,:])**2).sum(-1))
    
    for _ in range(iterations):
        # Shuffle indices
        indices = np.random.permutation(len(centers))
        for i in indices:
            old_sum = np.sum(radii)
            
            # Compute max radius with current configuration
            max_rad = np.minimum(
                np.minimum(centers[i], 1 - centers[i]).min(),
                np.maximum(D[i] - radii, 0).min()
            )
            
            old_rad = radii[i]
            
            # Try small moves in random directions
            delta_x = np.random.uniform(-step, step)
            delta_y = np.random.uniform(-step, step)
            
            new_x = centers[i, 0] + delta_x
            new_y = centers[i, 1] + delta_y
            
            # Check if move is within bounds
            if new_x < radii[i] or new_x > 1 - radii[i] or \
               new_y < radii[i] or new_y > 1 - radii[i]:
                continue  # Reject move that violates wall constraint
            
            # Temporarily adjust center
            old_center = centers[i].copy()
            centers[i] = [new_x, new_y]
            
            # Recompute max radius for circle i given the new center
            # and all other updated radii
            new_radii = max_radii(centers, radii)
            
            new_sum = np.sum(new_radii)
            
            # Accept if improvement
            if new_sum > old_sum:
                radii = new_radii
                continue
            else:
                # Revert change
                centers[i] = old_center
                
    return centers, radii

    x0 = np.concatenate([centers.flatten(), radii])

    def wall_constraints(v):
        centers_v = v[:2*n].reshape((n,2))
        radii_v = v[2*n:]
        c1 = radii_v - centers_v[:,0]  # r - x
        c2 = radii_v + centers_v[:,0] - 1  # r + x - 1
        c3 = radii_v - centers_v[:,1]  # r - y
        c4 = radii_v + centers_v[:,1] - 1  # r + y - 1
        return np.hstack([c1,c2,c3,c4])

    def layer_constraints(v):
        centers_v = v[:2*n].reshape((n,2))
        radii_v = v[2*n:]
        # For each pair (i,j) if i != j compute constraint r_i + r_j - d_{ij}
        constraint_vals = []
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers_v[i] - centers_v[j]) ** 2))
                val = radii_v[i] + radii_v[j] - dist
                constraint_vals.append(val)
        return np.array(constraint_vals)

        cons.append(NonlinearConstraint(lambda v: wall_constraints(v)[4*i + 1], -np.inf, 0))  # r + x - 1 <= 0
        cons.append(NonlinearConstraint(lambda v: wall_constraints(v)[4*i + 2], -np.inf, 0))  # r - y <= 0
        cons.append(NonlinearConstraint(lambda v: wall_constraints(v)[4*i + 3], -np.inf, 0))  # r + y - 1 <= 0

    # Add layer constraints
    for i in range(n):
        for j in range(i + 1, n):
            def constraint_func(v): return layer_constraints(v)[i*(n-1) + j - i - 1]
            cons.append(NonlinearConstraint(constraint_func, -np.inf, 0))

    bounds = [(0, 1) for _ in range(2*n)] + [(0, 0.5) for _ in range(n)]

    # Objective to minimize -sum(radii)
    def obj_func(v):
        return -np.sum(v[2*n:])
    
    res = minimize(obj_func, x0, method='trust-constr', constraints=cons, bounds=bounds, options={'maxiter': 200})

    radii_opt = res.x[2*n:]
    
    # Specifically adjust radii to be non-negative and valid
    radii_opt = np.maximum(radii_opt, 0)
    
    return centers_opt, radii_opt
