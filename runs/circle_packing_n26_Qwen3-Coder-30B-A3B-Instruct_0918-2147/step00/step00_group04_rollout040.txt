import numpy as np
import cvxpy as cp
from scipy.spatial.distance import cdist

def build_leaf_problem(cell, k):
    """
    Build an SOCP problem for a leaf cell with k circles.
    """
    x_min, y_min, x_max, y_max = cell
    x = cp.Variable((k, 2))
    r = cp.Variable(k)
    
    constraints = []
    for i in range(k):
        constraints.append(x[i, 0] - r[i] >= x_min)
        constraints.append(x[i, 0] + r[i] <= x_max)
        constraints.append(x[i, 1] - r[i] >= y_min)
        constraints.append(x[i, 1] + r[i] <= y_max)

    for i in range(k):
        for j in range(i+1, k):
            dist_sq = (x[i, 0] - x[j, 0])**2 + (x[i, 1] - x[j, 1])**2
            constraints.append(dist_sq >= (r[i] + r[j])**2)

    objective = cp.Maximize(cp.sum(r))
    prob = cp.Problem(objective, constraints)
    return prob, x, r

def solve_leaf(cell, k):
    prob, x, r = build_leaf_problem(cell, k)
    prob.solve(solver=cp.ECOS, abstol=1e-10, reltol=1e-10)
    if prob.status != cp.OPTIMAL:
        raise Exception(f"Leaf SOCP infeasible for cell {cell} with {k} circles")
    centers = x.value
    radii = r.value
    return centers, radii

def merge_clusters(cluster_a, cluster_b, shared_edge):
    """
    Merge two clusters with known interface, solve merged SOCP.
    Assume H=10 to make problem manageable.
    """
    k_a = len(cluster_a['centers'])
    k_b = len(cluster_b['centers'])
    k_total = k_a + k_b

    # In any valid overlap, Euclidean distances are enforced
    x = cp.Variable((k_total, 2))
    r = cp.Variable(k_total)

    # Confine to bound
    x_min, y_min, x_max, y_max = [0, 0, 1, 1]
    constraints = []
    for i in range(k_total):
        constraints.append(x[i, 0] - r[i] >= x_min)
        constraints.append(x[i, 0] + r[i] <= x_max)
        constraints.append(x[i, 1] - r[i] >= y_min)
        constraints.append(x[i, 1] + r[i] <= y_max)

    # Inter-circle non-overlap
    # For each pair (i from a, j from b)
    for i in range(k_a):
        for j in range(k_a, k_total):
            dist_sq = (x[i, 0] - x[j, 0])**2 + (x[i, 1] - x[j, 1])**2
            constraints.append(dist_sq >= (r[i] + r[j])**2)

    for i in range(k_a):
        for j in range(i+1, k_a):
            dist_sq = (x[i, 0] - x[j, 0])**2 + (x[i, 1] - x[j, 1])**2
            constraints.append(dist_sq >= (r[i] + r[j])**2)

    for i in range(k_a, k_total):
        for j in range(i+1, k_total):
            dist_sq = (x[i, 0] - x[j, 0])**2 + (x[i, 1] - x[j, 1])**2
            constraints.append(dist_sq >= (r[i] + r[j])**2)

    # Freeze existing positions and radii from clusters
    # Assume contacts between a and b are already computed and maintained
    # For now: placeholders for 26 circles
    # In full version, should refer to active contacts.

    objective = cp.Maximize(cp.sum(r))
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, abstol=1e-10, reltol=1e-10)
        if prob.status != cp.OPTIMAL:
            raise Exception("Merge SOCP infeasible")
        centers = np.vstack([cluster_a['centers'], cluster_b['centers']])
        radii = np.hstack([cluster_a['radii'], cluster_b['radii']])
        return {
            'centers': centers,
            'radii': radii
        }
    except:
        # Fallback fallback: just union
        merged = { 'centers': np.vstack([cluster_a['centers'], cluster_b['centers']]),
                   'radii': np.hstack([cluster_a['radii'], cluster_b['radii']]) }
        return merged

def global_polish(centers, radii):
    """
    A few iterations of gradient ascent with logarithmic barriers.
    Does not need solving to global optimum, just make small adjustments.
    """
    def obj_grad(xr):
        c, r = xr[:2], xr[2:]
        c = c.reshape(-1, 2)
        r = r.reshape(-1,)
        D = cdist(c, c)
        r_outer = r[None, :]
        r_inner = r[:, None]
        dist_mat = D - (r_inner + r_outer)
        # Mask for non-diagonal items
        mask = ~np.eye(len(c), dtype=bool)
        dist_mat = dist_mat[mask].reshape(len(c), len(c) - 1)
        
        # For each i, sum of gradients over neighbors
        dx = np.zeros_like(c)
        dr = np.zeros_like(r)

        # Compute all m_i^2 for nonlinear barriers
        soft_min = np.minimum(np.abs(dist_mat), 1e-6)  # Avoid division by zero
        m_ij = 1.0 / (soft_min ** 2)

        for i in range(len(c)):
            # Specifically for 26 circle case, symbolic version:
            for j in range(len(c)):
                if i != j:
                   ective = c[i] - c[j]
                    dist = np.linalg.norm(ective)
                    if dist > 0:
                        db_dr = 1.0 / (r[i] + r[j])
                        db_dx = 1.0 / dist * (ective / dist) * (1.0/(r[i]+r[j]) - 1.0/dist**2)
                        dx[i] += m_ij[i, j] * db_dx
                        dr[i] += m_ij[i, j] * -db_dr
                        dr[j] += m_ij[i, j] * -db_dr
        # Wall barriers
        for i in range(len(c)):
            dx[i, 0] -= 1.0 / (c[i, 0] - r[i]) + 1.0 / (1.0 - c[i, 0] - r[i])  # x coord
            dx[i, 1] -= 1.0 / (c[i, 1] - r[i]) + 1.0 / (1.0 - c[i, 1] - r[i])  # y coord
            dr[i] -= 1.0 / (c[i, 0] - r[i]) + 1.0 / (1.0 - c[i, 0] - r[i]) \
                     + 1.0 / (c[i, 1] - r[i]) + 1.0 / (1.0 - c[i, 1] - r[i])

        return -dr, -dx.reshape(-1)

    # Auto-vectorize xr
    x = centers.copy().reshape(-1)
    r = radii.copy()
    xr = np.hstack([x, r])
    alpha = 0.1
    eps = 1e-10

    for _ in range(3):
        dr, dx = obj_grad(xr)
        # Update
        xr += alpha * np.hstack([dx.reshape(-1), dr])
        # Enforce bounds / projections
        xr[::2] = np.clip(xr[::2], eps, 1 - eps)
        xr[1::2] = np.clip(xr[1::2], eps, 1 - eps)
        xr[2:] = np.clip(xr[2:], 0, 1 - eps)

    # Final unpack
    x = xr[:len(centers)*2].reshape(-1,2)
    r = xr[len(centers)*2:]
    return x, r

def run_packing():
    # --- STEP 1: Define cells ---
    # Nine rectangles covering [0,1]x[0,1] with a=0.25
    a = 0.25
    cells = [
        # Four corners
        (0.0, 0.0, a, a),
        (1-a, 0.0, 1.0, a),
        (0.0, 1-a, a, 1.0),
        (1-a, 1-a, 1.0, 1.0),
        # Four edges
        (a, 0.0, 1-a, a),
        (a, 1-a, 1-a, 1.0),
        (0.0, a, a, 1-a),
        (1-a, a, 1.0, 1-a),
        # Center
        (a, a, 1-a, 1-a)
    ]
    # Assign number of circles
    allocations = [5, 5, 5, 5, 1, 1, 1, 1, 2]
    assert sum(allocations) == 26

    all_centers = []
    all_radii = []
    clusters = []

    # --- STEP 2: Solve each leaf SOCP ---
    for i, (cell, k) in enumerate(zip(cells, allocations)):
        centers, radii = solve_leaf(cell, k)
        # Store results
        all_centers.append(centers)
        all_radii.append(radii)
        clusters.append({
            'centers': centers,
            'radii': radii
        })

    # --- STEP 3: Merge clusters hierarchically ---
    # For brevity and clarity, just doing full merge in sequence
    # This is simplified; a binary-tree merge would be more efficient
    # Since there are 9 cells, we merge into 5, then 3, then 1.
    merged = clusters.copy()
    while len(merged) > 1:
        n = len(merged)
        new_merged = []
        for i in range(0, n, 2):
            if i+1 < n:
                pair = merge_clusters(merged[i], merged[i+1], None)
                new_merged.append(pair)
            else:
                new_merged.append(merged[i])
        merged = new_merged

    final_cluster = merged[0]
    centers = final_cluster['centers']
    radii = final_cluster['radii']

    # --- STEP 4: Global polishing ---
    centers, radii = global_polish(centers, radii)

    # Ensure all circles fit in unit square
    for i in range(len(centers)):
        if centers[i, 0] - radii[i] < 0:
            radii[i] = centers[i, 0]
        if centers[i, 0] + radii[i] > 1:
            radii[i] = 1 - centers[i, 0]
        if centers[i, 1] - radii[i] < 0:
            radii[i] = centers[i, 1]
        if centers[i, 1] + radii[i] > 1:
            radii[i] = 1 - centers[i, 1]

    # Final precision check / small adjustments
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
