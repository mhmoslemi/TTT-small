import numpy as np
from scipy.optimize import linprog
import math

def initial_hex_centers(k, spacing):
    """Generate k centers in a hexagonal pattern."""
    sqrt3 = math.sqrt(3)
    centers = []
    i = 0
    j = 0
    while len(centers) < k:
        x = i * spacing + (j % 2) * spacing / 2
        y = j * spacing * sqrt3 / 2
        if 0 <= x <= 1 and 0 <= y <= 1:
            centers.append([x, y])
        if x > 1:
            j += 1
            i = 0
        else:
            i += 1
    return np.array(centers[:k])

def initial_radii(k, r0):
    """Return an array of k initial radii."""
    return np.full(k, r0)

def build_lp_constants(centers, radii):
    """Return wall distances and pairwise distances."""
    n = centers.shape[0]
    d_left = centers[:, 0]
    d_right = 1 - centers[:, 0]
    d_bottom = centers[:, 1]
    d_top = 1 - centers[:, 1]
    d_wall = np.minimum.reduce([d_left, d_right, d_bottom, d_top])
    # Pairwise distances
    D_ij = np.zeros((n, n))
    for i in range(n):
        for j in range(i+1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            D_ij[i, j] = dist
            D_ij[j, i] = dist
    return d_wall, D_ij

def solve_radius_lp(consts, radii, graph, alpha):
    """Solve linear program for radius increments."""
    d_wall, D_ij = consts
    n = len(radii)
    # Build constraints
    A_ub = []
    b_ub = []
    # Wall constraints
    for i in range(n):
        slack = d_wall[i] - radii[i]
        if slack >= 0:
            A_ub.append([-1.0])  # Δr_i <= slack
            b_ub.append(-slack)
        else:
            A_ub.append([-1.0])
            b_ub.append(-0.0)
    
    # Pairwise constraints
    for i in range(n):
        for j in range(i+1, n):
            if D_ij[i, j] > 0:
                dist_constraint = 1.0 + 1.0  # Δr_i + Δr_j <= (D_ij - (r_i + r_j))
                A_ub.append([1.0, 1.0])
                b_ub.append(D_ij[i, j] - (radii[i] + radii[j]))
    
    # Variables to maximize sum of Δr_i
    c = np.ones(n)
    result = linprog(-c, A_ub=A_ub, b_ub=b_ub, method='highs')
    if not result.success:
        # Fallback to no growth
        return np.zeros(n)
    return result.x

def apply_increment(radii, Δr):
    """Update radii with increments."""
    return radii + Δr

def push_feasibility(centers, radii):
    """Adjust centers and radii to satisfy constraints."""
    n = len(radii)
    # Wall constraints
    for i in range(n):
        r = radii[i]
        x, y = centers[i]
        if x - r < 0:
            radii[i] = x
        if x + r > 1:
            radii[i] = 1 - x
        if y - r < 0:
            radii[i] = y
        if y + r > 1:
            radii[i] = 1 - y
        # Update center to be consistent with new radius
        centers[i] = [x, y]

    # Pairwise overlaps
    updated = True
    while updated:
        updated = False
        for i in range(n):
            for j in range(i+1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                r_sum = radii[i] + radii[j]
                if dist < r_sum:
                    overlap = r_sum - dist
                    if dist > 1e-15:
                        delta = overlap / 2
                        unit_vec = (centers[i] - centers[j]) / dist
                        centers[i] += delta * unit_vec
                        centers[j] -= delta * unit_vec
                        updated = True
    return centers, radii

def run_iterations(centers, radii, max_iter=100, ε=1e-5, α=0.5):
    """Main iterative loop."""
    for iteration in range(max_iter):
        # Build LP constants
        consts = build_lp_constants(centers, radii)
        # Solve LP
        Δr = solve_radius_lp(consts, radii, {}, α)
        # Update radii
        new_radii = apply_increment(radii, Δr)
        # Push feasibility
        centers, new_radii = push_feasibility(centers, new_radii)
        # Check for stopping criterion
        Δ_sum = np.sum(Δr)
        if Δ_sum < ε:
            break
        radii = new_radii
    return centers, radii

def final_refinement(centers, radii):
    """Final high-precision refinement."""
    # Rebuild LP with tighter tolerance
    consts = build_lp_constants(centers, radii)
    Δr = solve_radius_lp(consts, radii, {}, 0.5)
    radii = apply_increment(radii, Δr)
    # Final clipping
    for i in range(len(radii)):
        r = radii[i]
        x, y = centers[i]
        r = min(r, x, 1 - x, y, 1 - y)
        radii[i] = r
    return centers, radii

def run_packing():
    """Run the full packing process."""
    # Initial setup
    k = 26
    spacing = 0.12
    r0 = 0.005

    # Step 1: Initial centers
    centers = initial_hex_centers(k, spacing)

    # Step 2: Initial radii
    radii = initial_radii(k, r0)

    # Step 3: Main iterations
    centers, radii = run_iterations(centers, radii, max_iter=50, ε=1e-5, α=0.5)

    # Step 4: Final refinement
    centers, radii = final_refinement(centers, radii)

    # Step 5: Sum of radii
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
