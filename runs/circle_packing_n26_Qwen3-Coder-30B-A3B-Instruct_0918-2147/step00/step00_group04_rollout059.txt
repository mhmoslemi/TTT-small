import numpy as np
import cvxpy as cp

def build_leaf_problem(cell, k):
    # Variables for k circles
    centers = cp.Variable((k, 2))
    radii = cp.Variable(k)
    
    # Constraints from cell walls
    x_min, x_max, y_min, y_max = cell
    constraints = [
        centers[:, 0] >= x_min + radii,
        centers[:, 0] <= x_max - radii,
        centers[:, 1] >= y_min + radii,
        centers[:, 1] <= y_max - radii,
    ]
    
    # Non-overlap constraints (SOC)
    for i in range(k):
        for j in range(i + 1, k):
            diff = centers[i] - centers[j]
            dist = cp.norm(diff)
            constraints.append(dist >= radii[i] + radii[j])
    
    # Maximize sum of radii
    objective = cp.Maximize(cp.sum(radii))
    
    return cp.Problem(objective, constraints), centers, radii

def solve_leaf(cell, k):
    problem, centers, radii = build_leaf_problem(cell, k)
    problem.solve(solver=cp.ECOS, abstol=1e-12, reltol=1e-12)
    
    if problem.status != cp.OPTIMAL:
        raise RuntimeError(f"Leaf SOCP problem failed to solve: {problem.status}")
        
    center_vals = centers.value
    radius_vals = radii.value
    
    # Trim result to remove degenerate circles
    good_indices = radius_vals >= 1e-12
    if not np.all(good_indices):
        center_vals = center_vals[good_indices]
        radius_vals = radius_vals[good_indices]
        
    return center_vals, radius_vals

def get_cell_bounds(cell_idx):
    if cell_idx == 0:
        return (0, 0.25, 0, 0.25)
    elif cell_idx == 1:
        return (0.75, 1.0, 0, 0.25)
    elif cell_idx == 2:
        return (0, 0.25, 0.75, 1.0)
    elif cell_idx == 3:
        return (0.75, 1.0, 0.75, 1.0)
    elif cell_idx == 4:
        return (0.25, 0.75, 0, 0.25)
    elif cell_idx == 5:
        return (0.25, 0.75, 0.75, 1.0)
    elif cell_idx == 6:
        return (0.0, 0.25, 0.25, 0.75)
    elif cell_idx == 7:
        return (0.75, 1.0, 0.25, 0.75)
    elif cell_idx == 8:
        return (0.25, 0.75, 0.25, 0.75)
    else:
        raise ValueError("Invalid cell index")

def run_packing():
    # Define cells
    cells = [get_cell_bounds(i) for i in range(9)]
    
    # Circle allocation by cell
    allocations = [
        5,  # corner (0,0)
        5,  # corner (1,0)
        5,  # corner (0,1)
        5,  # corner (1,1)
        1,  # edge top
        1,  # edge bottom
        1,  # edge left
        1,  # edge right
        2   # center
    ]
    
    # Solve each leaf cell optimally
    clusters = []
    for i, (cell, count) in enumerate(zip(cells, allocations)):
        center_vals, radius_vals = solve_leaf(cell, count)
        clusters.append((center_vals, radius_vals, cell))
        
    def merge_clusters(clusters_left, clusters_right):
        # Create the merged cluster
        all_centers = np.vstack([c[0] for c in [clusters_left, clusters_right]])
        all_radii = np.hstack([c[1] for c in [clusters_left, clusters_right]])
        
        # Form new cell bounds from bounds of both
        lbs = [c[2] for c in [clusters_left, clusters_right]]
        min_x = min([b[0] for b in lbs])
        max_x = max([b[1] for b in lbs])
        min_y = min([b[2] for b in lbs])
        max_y = max([b[3] for b in lbs])
        
        merged_cell = (min_x, max_x, min_y, max_y)
        
        # Define the problem with all centers and radii
        k_total = all_centers.shape[0]
        new_centers = cp.Variable((k_total, 2))
        new_radii = cp.Variable(k_total)
        
        # Boundary constraints
        constraints = [
            new_centers[:, 0] >= min_x + new_radii,
            new_centers[:, 0] <= max_x - new_radii,
            new_centers[:, 1] >= min_y + new_radii,
            new_centers[:, 1] <= max_y - new_radii,
        ]
        
        # Non-overlap constraints
        for i in range(k_total):
            for j in range(i + 1, k_total):
                diff = new_centers[i] - new_centers[j]
                dist = cp.norm(diff)
                constraints.append(dist >= new_radii[i] + new_radii[j])
        
        # Optimization
        objective = cp.Maximize(cp.sum(new_radii))
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, abstol=1e-12, reltol=1e-12)
        except:
            raise RuntimeError("Merging failed due to infeasible bounds.")
            
        if prob.status != cp.OPTIMAL:
            raise RuntimeError(f"Merging failed with status: {prob.status}")
            
        final_centers = new_centers.value
        final_radii = new_radii.value
        
        # Housekeeping
        good_indices = final_radii > 1e-12
        final_centers = final_centers[good_indices]
        final_radii = final_radii[good_indices]
        
        return (final_centers, final_radii, merged_cell)
        
    while len(clusters) > 1:
        new_clusters = []
        for i in range(0, len(clusters), 2):
            if i + 1 < len(clusters):
                merged = merge_clusters(clusters[i], clusters[i+1])
                new_clusters.append(merged)
            else:
                new_clusters.append(clusters[i])
        clusters = new_clusters

    # Final cluster is now [centers, radii, cell]
    all_centers = clusters[0][0]
    all_radii = clusters[0][1]
    
    if all_centers.shape[0] != 26:
        raise RuntimeError(f"Expected 26 circles, got {all_centers.shape[0]}")
    
    # Final refinement with barrier method to get final value
    # a simple gradient solver on a tight log-barrier approximates the solution well
    def apply_barrier_step(centers, radii, step_size=0.01, mu=1e-4):
        """Apply a single optimization step using log barrier"""
        n = len(radii)
        grad = np.zeros_like(radii)
        grad += 1.0  # gradient for radius (weight 1 for sum of radii)

        for i in range(n):
            # Compute barrier contributions
            x, y = centers[i]
            r = radii[i]
            # Weights for barrier
            penalty_weight = mu * (
                1.0 / (x - r + 1e-15) +
                1.0 / (1 - x - r + 1e-15) +
                1.0 / (y - r + 1e-15) +
                1.0 / (1 - y - r + 1e-15)
            )
            grad[i] += -penalty_weight

            # Backward compatibility check: not necessary, but added for completeness.
            for j in range(i+1, n):
                diff = centers[i] - centers[j]
                d = np.linalg.norm(diff)
                r_sum = radii[i] + radii[j]
                if d > r_sum + 1e-10:
                    barrier_deriv = -mu / (d - r_sum + 1e-15)
                    grad[i] += barrier_deriv
                    grad[j] += barrier_deriv
                    
        new_radii = radii + step_size * grad
        idx = new_radii > 1e-15
        new_radii = np.where(idx, new_radii, 1e-15)

        # update centers and radii to valid domain
        for i in range(n):
            centers[i] = np.clip(centers[i], [radii[i], radii[i]], [1-radii[i], 1-radii[i]])

        # Project back to [0,1] x [0,1]
        for i in range(n):
            centers[i][0] = np.clip(centers[i][0], radii[i], 1 - radii[i])
            centers[i][1] = np.clip(centers[i][1], radii[i], 1 - radii[i])

        return centers, new_radii
    
    # store original centers and radii
    final_centers = all_centers.copy()
    final_radii = all_radii.copy()
    
    # Let's take 3 iterative steps of barrier gradient
    for _ in range(3):
        final_centers, final_radii = apply_barrier_step(final_centers, final_radii)
        
    # final correction: make sure areas are consistent
    sum_radii = np.sum(final_radii)
    
    # final validation
    is_valid, msg = validate_packing(final_centers, final_radii)
    if not is_valid:
        raise RuntimeError(f"Generated solution failed validation: {msg}")

    return (final_centers, final_radii, sum_radii)
