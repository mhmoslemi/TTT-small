import numpy as np
from scipy.spatial.distance import cdist

def penalty(z, alpha=10.0, beta=10.0):
    # z is (78,)
    centers = z.reshape(-1, 3)[:, :2]
    radii = z.reshape(-1, 3)[:, 2]
    
    # Wall penalties
    wall_distances = np.array([
        centers[:, 0],
        1 - centers[:, 0],
        centers[:, 1],
        1 - centers[:, 1]
    ])
    wall_violations = np.maximum(0, radii[:, np.newaxis] - wall_distances.T).sum(0)
    
    # Overlap penalties
    dist_matrix = cdist(centers, centers)
    radii_matrix = radii[:, np.newaxis] + radii[np.newaxis, :]
    overlap_violations = np.maximum(0, radii_matrix - dist_matrix)
    overlap_violations = overlap_violations[np.triu(np.ones_like(overlap_violations), k=1)].sum()
    
    return alpha * wall_violations.sum() + beta * overlap_violations

def energy(z, lam, alpha=10.0, beta=10.0):
    total_radius = z.reshape(-1, 3)[:, 2].sum()
    return -total_radius + lam * penalty(z, alpha, beta)

def propose_move(z, sigma_c, sigma_r):
    z_new = z.copy()
    move_type = np.random.choice(3)
    n_circles = z.shape[0] // 3
    
    if move_type == 0:  # Center jitter
        i = np.random.randint(n_circles)
        z_new[i*3] += np.random.normal(0, sigma_c)
        z_new[i*3+1] += np.random.normal(0, sigma_c)
        z_new[i*3] = np.clip(z_new[i*3], 0, 1)
        z_new[i*3+1] = np.clip(z_new[i*3+1], 0, 1)
    elif move_type == 1:  # Radius perturb
        i = np.random.randint(n_circles)
        z_new[i*3+2] += np.random.normal(0, sigma_r)
        z_new[i*3+2] = np.maximum(0, z_new[i*3+2])
    else:  # Swap pair
        i = np.random.randint(n_circles)
        j = np.random.randint(n_circles)
        if i != j:
            z_new[i*3+2], z_new[j*3+2] = z_new[j*3+2], z_new[i*3+2]
    
    return z_new

def anneal():
    n_circles = 26
    np.random.seed(42)  # For reproducibility
    
    # Create initial hexagonal grid
    spacing = 0.18
    rows = int(1.0 / spacing) + 1
    cols = int(1.0 / spacing) + 1
    centers = np.array([
        (i * spacing, j * spacing) 
        for i in range(cols) 
        for j in range(rows) 
        if i * spacing < 1.0 and j * spacing < 1.0
    ][:n_circles])
    
    # Ensure we have exactly n_circles centers
    if len(centers) < n_circles:
        # Add more random points
        while len(centers) < n_circles:
            centers = np.vstack([centers, np.random.rand(1, 2)])
    centers = centers[:n_circles]
    
    # Initial radii based on wall distances
    radii = np.minimum(
        centers[:, 0],
        1 - centers[:, 0],
        centers[:, 1],
        1 - centers[:, 1]
    )
    z = np.column_stack([centers, radii]).flatten()
    
    # Annealing parameters
    T0 = 1.0
    c_T = 0.995
    T = T0
    lam = 0.1
    c_lam = 1.03
    max_iter = 10000
    sigma_c = 0.03
    sigma_r = 0.02
    
    # Track best feasible solution
    best_feasible_z = z.copy()
    best_feasible_sum = z.reshape(-1, 3)[:, 2].sum()
    
    # Annealing loop
    for _ in range(max_iter):
        if T < 1e-5:
            break
        z_new = propose_move(z, sigma_c * np.sqrt(T/T0), sigma_r * np.sqrt(T/T0))
        delta_E = energy(z_new, lam) - energy(z, lam)
        if delta_E <= 0 or np.random.rand() < np.exp(-delta_E / T):
            z = z_new
        T *= c_T
        lam *= c_lam
        
        # Check if the current solution is feasible (penalty = 0)
        if penalty(z) < 1e-8:
            current_sum = z.reshape(-1, 3)[:, 2].sum()
            if current_sum > best_feasible_sum:
                best_feasible_sum = current_sum
                best_feasible_z = z.copy()
    
    return best_feasible_z.reshape(-1, 3)

def project_and_ascent(centers, radii):
    # Set up optimization parameters
    eta = 5e-3
    n_iter = 500
    max_project_passes = 10
    
    # Concatenate all for easy manipulation
    z = np.column_stack([centers, radii]).flatten()
    step_size = eta
    lambda_ = 1.0
    
    for _ in range(n_iter):
        # Gradient ascent: increase all radii (one step)
        z.reshape(-1, 3)[:, 2] += step_size  # Increase all radii
        
        # Project (simple wall constraint)
        centers = z.reshape(-1, 3)[:, :2]
        radii = z.reshape(-1, 3)[:, 2]
        max_radius = np.minimum(centers[:, 0], 1 - centers[:, 0], centers[:, 1], 1 - centers[:, 1])
        radii = np.minimum(radii, max_radius)
        z.reshape(-1, 3)[:, 2] = radii
        
        # Handle overlaps with repeated projection
        for _ in range(max_project_passes):
            dist_matrix = cdist(centers, centers)
            radii_matrix = radii[:, np.newaxis] + radii[np.newaxis, :]
            overlaps = np.maximum(0, radii_matrix - dist_matrix)
            imbalance = np.maximum(0, overlaps[np.triu(np.ones_like(overlaps), k=1)])
            if imbalance.sum() < 1e-10:
                break
            
            # Reduce radii proportionally to overlap
            overlap_sum = overlaps.sum(0)
            overlap_reduction = (overlap_sum / 2) * 0.8  # A bit of damping
            radii = np.maximum(0, radii - overlap_reduction)
            z.reshape(-1, 3)[:, 2] = radii
            
            # Clip radii again
            max_radius = np.minimum(centers[:, 0], 1 - centers[:, 0], centers[:, 1], 1 - centers[:, 1])
            radii = np.minimum(radii, max_radius)
            z.reshape(-1, 3)[:, 2] = radii
            
        step_size *= 0.95
        
    # Final calculation
    centers = z.reshape(-1, 3)[:, :2]
    radii = z.reshape(-1, 3)[:, 2]
    sum_radii = radii.sum()
    return centers, radii, sum_radii
