import numpy as np
from scipy.optimize import minimize

def make_hex_lattice(n, margin=0.05):
    # Create a hexagonal lattice that fits in [margin, 1-margin]^2
    # Scale to fit roughly n circles
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))
    
    dx = (1 - 2 * margin) / cols
    dy = dx * np.sqrt(3) / 2
    
    centers = []
    for i in range(cols):
        for j in range(rows):
            x = margin + i * dx + (j % 2) * dx / 2
            y = margin + j * dy
            if x <= 1 - margin and y <= 1 - margin:
                centers.append([x, y])
                if len(centers) == n:
                    break
        if len(centers) == n:
            break

    centers = np.array(centers)
    return centers

def objective(z):
    # z = [x0, y0, r0, x1, y1, r1, ...]
    radii = z[2::3]
    return -np.sum(radii)

def boundary_constraints(z):
    # Each circle must stay inside [0,1]^2
    centers = z[0::3].reshape(-1, 1), z[1::3].reshape(-1, 1)
    radii = z[2::3].reshape(-1, 1)
    # Corner constraints:
    # x - r >= 0
    # x + r <= 1
    # y - r >= 0
    # y + r <= 1
    x, y = centers
    c1 = x - radii    # x - r >= 0
    c2 = 1 - (x + radii)  # x + r <= 1
    c3 = y - radii    # y - r >= 0
    c4 = 1 - (y + radii)  # y + r <= 1
    return np.vstack((c1, c2, c3, c4)).flatten()

def pairwise_constraints(z):
    # List of (i,j) pairs
    n = len(z) // 3
    result = []
    for i in range(n):
        for j in range(i+1, n):
            x1, y1 = z[3*i], z[3*i+1]
            x2, y2 = z[3*j], z[3*j+1]
            r1, r2 = z[3*i+2], z[3*j+2]
            dist_sq = (x1 - x2)**2 + (y1 - y2)**2
            sep_sq = (r1 + r2)**2
            result.append(dist_sq - sep_sq)
    return np.array(result)

def jacobian_boundary(z):
    n = len(z) // 3
    J = np.zeros((4*n, 3*n))
    for i in range(n):
        # All rows for circle i
        row = 4*i
        J[row:row+4, 3*i:3*i+3] = np.array([
            [1, 0, -1],   # x - r >= 0
            [-1, 0, -1],  # x + r <= 1
            [0, 1, -1],   # y - r >= 0
            [0, -1, -1]   # y + r <= 1
        ])
    return J

def jacobian_pairwise(z):
    n = len(z) // 3
    J = np.zeros((n*(n-1)//2, 3*n))
    k = 0
    for i in range(n):
        for j in range(i+1, n):
            x1, y1, r1 = z[3*i], z[3*i+1], z[3*i+2]
            x2, y2, r2 = z[3*j], z[3*j+1], z[3*j+2]
            dfdx1 = 2*(x1 - x2)
            dfdy1 = 2*(y1 - y2)
            dfdr1 = 2*(r1 + r2)
            dfdx2 = -2*(x1 - x2)
            dfdy2 = -2*(y1 - y2)
            dfdr2 = 2*(r1 + r2)
            J[k, 3*i]   = dfdx1
            J[k, 3*i+1] = dfdy1
            J[k, 3*i+2] = dfdr1
            J[k, 3*j]   = dfdx2
            J[k, 3*j+1] = dfdy2
            J[k, 3*j+2] = dfdr2
            k += 1
    return J

def solve_nlp(initial_z):
    constraints = [
        {'type': 'ineq', 'fun': boundary_constraints, 'jac': jacobian_boundary},
        {'type': 'ineq', 'fun': pairwise_constraints, 'jac': jacobian_pairwise}
    ]
    bounds = [(0, 1)] * (2 * 26) + [(0, 0.5)] * 26
    res = minimize(
        objective,
        initial_z,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 1000, 'ftol': 1e-9}
    )
    return res.x

def nudge_centers(centers, radii):
    centers = centers.copy()
    nudged = False
    for i in range(26):
        x, y = centers[i]
        r = radii[i]
        # Check if circle is tight against any wall
        # If so, nudge slightly away from that wall
        # This also prevents getting stuck in degenerate solutions
        if abs(x - r) < 1e-5:
            centers[i, 0] += 0.001
            nudged = True
        elif abs(x + r - 1) < 1e-5:
            centers[i, 0] -= 0.001
            nudged = True
        if abs(y - r) < 1e-5:
            centers[i, 1] += 0.001
            nudged = True
        elif abs(y + r - 1) < 1e-5:
            centers[i, 1] -= 0.001
            nudged = True
    if nudged:
        centers = np.clip(centers, 0, 1)
    return centers

def run_packing():
    # Step 1: Generate initial hex lattice with margin
    centers = make_hex_lattice(26, margin=0.05)
    radii = np.full(26, 0.02)
    
    # Step 2: Create initial z vector
    initial_z = np.zeros(3 * 26)
    initial_z[0::3] = centers[:, 0]
    initial_z[1::3] = centers[:, 1]
    initial_z[2::3] = radii
    
    # Step 3: Optimize and iterate
    for _ in range(5):
        new_z = solve_nlp(initial_z)
        new_centers = new_z[0::3].reshape(26, 1), new_z[1::3].reshape(26, 1)
        new_centers = np.column_stack(new_centers)
        new_radii = new_z[2::3]
        if abs(np.sum(new_radii) - np.sum(radii)) < 1e-5:
            break
        # Nudge circles that are pinned against a wall
        new_centers = nudge_centers(new_centers, new_radii)
        # Reconstruct z for next iteration
        initial_z[0::3] = new_centers[:, 0]
        initial_z[1::3] = new_centers[:, 1]
        initial_z[2::3] = new_radii
        radii = new_radii

    # Final validation and cleanup
    assert validate_packing(new_centers, new_radii)[0], "Final packing invalid"
    
    sum_radii = float(np.sum(new_radii))
    return new_centers, new_radii, sum_radii
