def build_hex_lattice(spacing):
    # Build a hexagonal lattice of 24 points in [0.05, 0.95]^2
    rows = 4
    cols = 6
    lattice = []
    for i in range(rows):
        for j in range(cols):
            x = 0.05 + j * spacing
            y = 0.05 + i * spacing * sqrt(3) / 2
            if i % 2 == 1:
                x += spacing / 2
            if 0.05 <= x <= 0.95 and 0.05 <= y <= 0.95:
                lattice.append([x, y])
    return np.array(lattice)
