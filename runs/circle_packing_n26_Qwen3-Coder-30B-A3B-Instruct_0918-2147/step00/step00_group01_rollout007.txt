import numpy as np
from scipy.spatial.distance import pdist, squareform

def penalty(z, alpha=10.0, beta=10.0):
    """Compute the penalty function P(z) for the packing."""
    n = 26
    centers = z[:2*n].reshape((n, 2))
    radii = z[2*n:]

    # Wall penalties
    wall_pen = 0.0
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        wall_dist = min(x, 1 - x, y, 1 - y)
        wall_pen += alpha * max(0.0, r - wall_dist)

    # Overlap penalties
    overlap_pen = 0.0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            overlap = radii[i] + radii[j] - dist
            overlap_pen += beta * max(0.0, -overlap)

    return wall_pen + overlap_pen

def energy(z, lam, alpha=10.0, beta=10.0):
    """Compute the energy function E(z) = -sum(radii) + lam * penalty."""
    n = 26
    radii = z[2*n:]
    obj = np.sum(radii)
    pen = penalty(z, alpha, beta)
    return -obj + lam * pen

def propose_move(z, sigma_c, sigma_r, T=1.0):
    """Propose a new configuration by applying one of three move types."""
    n = 26
    centers = z[:2*n].reshape((n, 2))
    radii = z[2*n:]
    
    # Adjust noise scales with current temperature
    noise_c = sigma_c * np.sqrt(T)
    noise_r = sigma_r * np.sqrt(T)
    
    move_type = np.random.choice(3)
    z_new = z.copy()
    centers_new = z_new[:2*n].reshape((n, 2))
    radii_new = z_new[2*n:]

    if move_type == 0:  # Center jitter
        idx = np.random.randint(n)
        centers_new[idx] += np.random.normal(0.0, noise_c, 2)
        centers_new[idx, 0] = np.clip(centers_new[idx, 0], 0.0, 1.0)
        centers_new[idx, 1] = np.clip(centers_new[idx, 1], 0.0, 1.0)
    elif move_type == 1:  # Radius perturb
        idx = np.random.randint(n)
        radii_new[idx] += np.random.normal(0.0, noise_r, 1)[0]
        radii_new[idx] = np.clip(radii_new[idx], 0.0, None)
    else:  # Swap pair
        i, j = np.random.choice(n, 2, replace=False)
        radii_new[i], radii_new[j] = radii_new[j], radii_new[i]

    return z_new

def anneal():
    """Run the simulated annealing algorithm."""
    np.random.seed(42)
    
    n = 26
    sigma_c = 0.03
    sigma_r = 0.02
    T0 = 1.0
    c_T = 0.995
    T_min = 1e-5
    lam0 = 0.1
    c_lam = 1.03
    max_iter = 10000
    T = T0
    lam = lam0

    # Initial hexagonal layout
    spacing = 0.18
    rows = int(np.ceil(1.0 / spacing)) + 1
    cols = int(np.ceil(1.0 / spacing)) + 1
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = (i + 0.5) * spacing
            y = (j + 0.5) * spacing
            if x <= 1.0 and y <= 1.0:
                centers.append([x, y])
    centers = np.array(centers[:n])

    # Assign initial radii
    radii = np.zeros(n)
    for i in range(n):
        x, y = centers[i]
        radii[i] = min(x, 1 - x, y, 1 - y)

    z_current = np.concatenate([centers.flatten(), radii])
    z_best_feasible = z_current.copy()
    best_sum = np.sum(radii)
    best_feasible_sum = best_sum
    z_lowest_energy = z_current.copy()
    lowest_energy = energy(z_current, lam)

    iter_count = 0
    while T >= T_min and iter_count < max_iter:
        # Propose and evaluate move
        z_proposed = propose_move(z_current, sigma_c, sigma_r, T)
        e_current = energy(z_current, lam)
        e_proposed = energy(z_proposed, lam)
        delta_e = e_proposed - e_current

        # Accept or reject
        if delta_e <= 0 or np.random.random() < np.exp(-delta_e / T):
            z_current = z_proposed

        # Keep track of best feasible
        p = penalty(z_current)
        if p <= 1e-8:
            sum_radii = np.sum(z_current[2*n:])
            if sum_radii > best_feasible_sum:
                best_feasible_sum = sum_radii
                z_best_feasible = z_current.copy()

        # Always keep the lowest energy
        if e_proposed < lowest_energy:
            lowest_energy = e_proposed
            z_lowest_energy = z_proposed.copy()

        # Update iteration and temperature
        iter_count += 1
        T *= c_T
        if iter_count % 100 == 0:
            lam *= c_lam

    # Set final based on best feasible
    z_final = z_best_feasible if best_feasible_sum > 0 else z_lowest_energy
    centers_final = z_final[:2*n].reshape((n,2))
    radii_final = z_final[2*n:]
    sum_radii = np.sum(radii_final)

    return centers_final, radii_final, sum_radii

def project_and_ascent(centers, radii, max_iter=500, eta=0.005, decay=0.95, tol=1e-7):
    """Perform projected gradient ascent to improve feasible packing."""
    n = 26
    z = np.concatenate([centers.flatten(), radii])
    eta_current = eta

    for iteration in range(max_iter):
        # Gradient is just [0,...,1,...] for radii
        grad = np.zeros_like(z)
        grad[2*n:] = 1.0  # Gradients w.r.t. radii

        # Apply step
        z_new = z + eta_current * grad

        # Clamp centers to the square
        centers_new = z_new[:2*n].reshape((n,2))
        centers_new[:, 0] = np.clip(centers_new[:, 0], 0.0, 1.0)
        centers_new[:, 1] = np.clip(centers_new[:, 1], 0.0, 1.0)
        z_new[:2*n] = centers_new.flatten()

        # Clamp radii to valid wall-based range
        radii_new = z_new[2*n:]
        for i in range(n):
            x, y = centers_new[i]
            wall_dist = min(x, 1 - x, y, 1 - y)
            radii_new[i] = max(0.0, min(radii_new[i], wall_dist))
        
        # Resolve overlaps
        centers_new = z_new[:2*n].reshape((n,2))
        radii_new = z_new[2*n:]
        dist_matrix = squareform(pdist(centers_new))
        for i in range(n):
            for j in range(i + 1, n):
                if dist_matrix[i,j] < radii_new[i] + radii_new[j]:
                    overlap = radii_new[i] + radii_new[j] - dist_matrix[i,j]
                    # Reduce both radii equally
                    radii_new[i] -= overlap / 2.0
                    radii_new[j] -= overlap / 2.0
                    # Stay within wall limits
                    x, y = centers_new[i]
                    wall_dist = min(x, 1 - x, y, 1 - y)
                    radii_new[i] = max(0.0, min(radii_new[i], wall_dist))
                    x, y = centers_new[j]
                    wall_dist = min(x, 1 - x, y, 1 - y)
                    radii_new[j] = max(0.0, min(radii_new[j], wall_dist))
        
        # Reassemble Z
        z_new = np.concatenate([centers_new.flatten(), radii_new])

        # Update z
        z = z_new
        eta_current *= decay

        # Stop if no meaningful increase
        new_sum = np.sum(radii_new)
        if iteration > 10 and np.abs(new_sum - np.sum(radii)) < tol:
            break

    centers_final = z[:2*n].reshape((n,2))
    radii_final = z[2*n:]
    return centers_final, radii_final

def run_packing():
    """Run the full packing algorithm."""
