import numpy as np
import cvxpy as cp
from scipy.spatial.distance import pdist, squareform

def build_socp_problem(n):
    r = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(r)), [])
    dist_params = []
    return prob, r, dist_params

def solve_radii(centres, prob, r, dist_params):
    n = centres.shape[0]
    constraints = []
    
    # Add boundary constraints
    x = centres[:, 0]
    y = centres[:, 1]
    constraints += [r <= x]
    constraints += [r <= 1 - x]
    constraints += [r <= y]
    constraints += [r <= 1 - y]
    
    # Add pairwise non-overlap constraints 
    distances = pdist(centres)
    distance_matrix = squareform(distances)
    for i in range(n):
        for j in range(i + 1, n):
            dist_ij = distance_matrix[i, j]
            constraints += [r[i] + r[j] <= dist_ij]
    
    # Update problem constraints
    prob.constraints = constraints
    
    # Solve
    prob.solve(solver=cp.ECOS, abstol=1e-12, reltol=1e-12)
    
    if prob.status != cp.OPTIMAL:
        # Likely infeasible due to small numerical errors;
        # Resolving with slight relaxation may help
        # But for now, just fail
        raise RuntimeError("SOCP solver did not converge to optimal")

    radii = r.value
    return radii

def energy(centres, prob, r, dist_params):
    radii = solve_radii(centres, prob, r, dist_params)
    return -np.sum(radii)

def random_move(state, rng):
    # state is flattened [x0,y0,x1,y1,...,x25,y25]
    new_state = state.copy()
    move_type = rng.choice(4)
    
    if move_type == 0:  # single centre jitter
        idx = rng.integers(0, 26)
        new_state[2*idx:2*idx+2] += rng.normal(0, 0.02, 2)
        new_state[2*idx:2*idx+2] = np.clip(new_state[2*idx:2*idx+2], 0, 1)
        
    elif move_type == 1:  # cluster jitter
        indices = rng.choice(26, rng.integers(3, 6), replace=False)
        offset = rng.normal(0, 0.02, 2)
        for idx in indices:
            new_state[2*idx:2*idx+2] += offset
            new_state[2*idx:2*idx+2] = np.clip(new_state[2*idx:2*idx+2], 0, 1)
            
    elif move_type == 2:  # edge push
        if rng.random() < 0.1:
            idx = rng.integers(0, 26)
            x, y = new_state[2*idx], new_state[2*idx+1]
            # Move towards closest wall
            dx = -np.sign(x - 0.5) * 0.01
            dy = -np.sign(y - 0.5) * 0.01
            new_state[2*idx] += dx
            new_state[2*idx+1] += dy
            new_state[2*idx] = np.clip(new_state[2*idx], 0, 1)
            new_state[2*idx+1] = np.clip(new_state[2*idx+1], 0, 1)
            
    else:  # swap order (just shuffle)
        temp = new_state.reshape(-1, 2)
        rng.shuffle(temp)
        new_state = temp.flatten()
        
    return new_state

def anneal_chain(seed, prob, r, dist_params):
    rng = np.random.default_rng(seed)
    state = rng.uniform(0, 1, 52)
    best_state = state.copy()
    best_energy = energy(state.reshape(-1, 2), prob, r, dist_params)
    
    T = 0.05
    alpha = 0.995
    N_iter = 150000
    accepted_count = 0
    
    current_energy = best_energy
    current_state = state.copy()

    while accepted_count < N_iter and T > 1e-6:
        new_state = random_move(current_state, rng)
        new_energy = energy(new_state.reshape(-1, 2), prob, r, dist_params)
        delta = new_energy - current_energy
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            current_energy = new_energy
            current_state = new_state.copy()
            
            if current_energy < best_energy:
                best_energy = current_energy
                best_state = current_state.copy()
                
            accepted_count += 1
            
        T *= alpha
        
    centers = best_state.reshape(-1, 2)
    radii = solve_radii(centers, prob, r, dist_params)
    return centers, radii, -best_energy

def local_refine(centres, radii, prob, r, dist_params):
    # Identify binding constraints from duals
    # Duals can be retrieved from constraints
    # But cvxpy doesn't directly expose duality; we'll re-solve to get constraint info

    n = len(radii)
    
    # Re-solve the SOCP to get dual values
    constraints = []
    
    # Boundary constraints
    x = centres[:, 0]
    y = centres[:, 1]
    constraints += [r <= x]
    constraints += [r <= 1 - x]
    constraints += [r <= y]
    constraints += [r <= 1 - y]
    
    distances = pdist(centres)
    distance_matrix = squareform(distances)
    for i in range(n):
        for j in range(i + 1, n):
            dist_ij = distance_matrix[i, j]
            constraints += [r[i] + r[j] <= dist_ij]
    
    # Update problem constraints
    prob.constraints = constraints
    
    # Solve again
    prob.solve(solver=cp.ECOS, abstol=1e-12, reltol=1e-12)
    if prob.status != cp.OPTIMAL:
        raise RuntimeError("SOCP solver failed in refinement")
    
    # For simplicity, we treat it as a heuristic: slightly refine positions 
    # to enforce active constraints more strictly
    
    # Simpler approach: just return original with good radii  
    return centres, radii

def finalise(centres, radii):
    # Return again with high precision for the enclosing check
    n = centres.shape[0]
    r = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(r)), [])
    dist_params = []
    
    # Add boundary constraints
    x = centres[:, 0]
    y = centres[:, 1]
    constraints = [r <= x, r <= 1 - x, r <= y, r <= 1 - y]
    
    # Add pairwise non-overlap constraints 
    distances = pdist(centres)
    distance_matrix = squareform(distances)
    for i in range(n):
        for j in range(i + 1, n):
            dist_ij = distance_matrix[i, j]
            constraints += [r[i] + r[j] <= dist_ij]
    
    # Update problem constraints
    prob.constraints = constraints
    
    # Solve  
    prob.solve(solver=cp.ECOS, abstol=1e-12, reltol=1e-12)

    if prob.status != cp.OPTIMAL:
        raise RuntimeError("Final SOCP solve failed")

    final_radii = r.value
    return centres, final_radii

def run_packing():
    n = 26
    prob, r, dist_params = build_socp_problem(n)
    
    # Format seeds for different strategies
    seeds = []
    rng = np.random.default_rng(42)
    # 1. Random uniform
    seeds.append(rng.uniform(0, 1, 52))
    # 2. Corner heuristic: place 4 circles in corners
    corner_centers = np.array([
        [0.05, 0.05],
        [0.95, 0.05],
        [0.05, 0.95],
        [0.95, 0.95]
    ])
    # Add 22 randomly arranged ones for better coverage of parameter space
    random_part = rng.uniform(0.1, 0.9, (22, 2))
    full_seeds = np.vstack([corner_centers, random_part]).flatten()
    seeds.append(full_seeds)
    
    # 3. Regular grid (like hex but latticed)
    grid_points = []
    steps = 5
    for i in range(steps):
        for j in range(steps):
            x = (i + 0.5) / steps
            y = (j + 0.5) / steps
            grid_points.append([x, y])
    grid_centers = np.array(grid_points[:26])
    seeds.append(grid_centers.flatten())
    
    # 4. Hexagonal-like arrangement (simplified)
    hex_points = []
    rows = 5
    offset = 0
    for i in range(rows):
        if i % 2 == 0:
            offset = 0
        else:
            offset = 1 / (5 * 2)
        for j in range(5):
            x = offset + (j / 5)
            y = (i / (rows - 1)) if rows > 1 else 0.5
            hex_points.append([x, y])
    hex_centers = np.array(hex_points[:26])
    seeds.append(hex_centers.flatten())
    
    best_result = None
    best_sum = -np.inf
    
    for i, seed in enumerate(seeds):
        try:
            centers, radii, sum_radii = anneal_chain(i * 1000, prob, r, dist_params)
            refined_centers, refined_radii = local_refine(centers, radii, prob, r, dist_params)
            final_centers, final_radii = finalise(refined_centers, refined_radii)
            final_sum = np.sum(final_radii)
            
            if final_sum > best_sum:
                best_sum = final_sum
                best_result = (final_centers, final_radii, final_sum)
        except Exception as e:
            continue
    
    if best_result is None:
        raise RuntimeError("All annealing chains failed")
        
    return best_result
