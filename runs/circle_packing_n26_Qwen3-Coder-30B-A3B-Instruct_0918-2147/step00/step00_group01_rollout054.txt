import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
from math import sqrt

def generate_lattice(s):
    """Generate a hexagonal lattice of candidate center points."""
    sqrt3 = sqrt(3)
    spacing = s
    y_spacing = spacing * sqrt3 / 2
    x_spacing = spacing
    candidates = []
    y = spacing / 2
    row = 0
    while y <= 1 - spacing / 2:
        x_start = spacing / 2 if row % 2 == 0 else spacing
        x = x_start
        while x <= 1 - spacing / 2:
            candidates.append([x, y])
            x += x_spacing
        y += y_spacing
        row += 1
    return np.array(candidates)

def build_milp(P):
    """Return a function that solves the LP relaxation for given z."""
    K = P.shape[0]
    z = cp.Variable(K, boolean=True)
    r = cp.Variable(K, nonneg=True)
    # Compute wall distances
    wL = P[:, 0]
    wR = 1 - P[:, 0]
    wB = P[:, 1]
    wT = 1 - P[:, 1]
    w = np.minimum(np.minimum(wL, wR), np.minimum(wB, wT))
    # Precompute distances
    D = np.zeros((K, K))
    for i in range(K):
        for j in range(i + 1, K):
            d = np.linalg.norm(P[i] - P[j])
            D[i, j] = D[j, i] = d
    # Build constraints
    constraints = [cp.sum(z) == 26]
    for k in range(K):
        constraints.append(r[k] <= w[k] * z[k])
    # Big-M constraints for overlap
    M = 1.0
    for i in range(K):
        for j in range(i + 1, K):
            constraints.append(D[i, j] >= r[i] + r[j] - M * (2 - z[i] - z[j]))
    objective = cp.Maximize(cp.sum(r))
    prob = cp.Problem(objective, constraints)
    
    def solve_lp(z_fixed=None):
        if z_fixed is not None:
            for k in range(K):
                z[k].fix(z_fixed[k])
        try:
            prob.solve(solver=cp.OSQP, verbose=False, eps_abs=1e-9, eps_rel=1e-9)
        except:
            return None, np.full(K, 0.0)
        if prob.status != cp.OPTIMAL:
            return None, np.full(K, 0.0)
        return prob.value, r.value
    
    return solve_lp

def branch_and_bound(milp_solver, node_limit=20000):
    import heapq
    best_obj = -np.inf
    best_z = None
    best_r = None
    
    # Priority queue: (-bound, node_id, z_fixed)
    queue = []
    node_id = 0
    heapq.heappush(queue, (0.0, node_id, np.full(120, -1, dtype=int)))  # all free
    node_id += 1
    visited = set()
    
    while queue and len(visited) < node_limit:
        neg_bound, _, z_fixed = heapq.heappop(queue)
        bound = -neg_bound
        for i in range(len(z_fixed)):
            if z_fixed[i] == -1:
                break
        else:
            # All binary variables are fixed. Check if integer.
            if np.all(z_fixed == 0):
                continue
            # Solve LP with fixed z
            obj, r_sol = milp_solver(z_fixed)
            if obj is None:
                continue
            if obj > best_obj:
                best_obj = obj
                best_z = z_fixed.copy()
                best_r = r_sol.copy()
            continue
            
        # Branch on variable i
        z_new = z_fixed.copy()
        # Try z[i] = 1
        z_new[i] = 1
        obj1, _ = milp_solver(z_new)
        if obj1 is not None and obj1 > best_obj - 1e-10:
            heapq.heappush(queue, (-obj1, node_id, z_new))
            node_id += 1
        # Try z[i] = 0
        z_new[i] = 0
        obj0, _ = milp_solver(z_new)
        if obj0 is not None and obj0 > best_obj - 1e-10:
            heapq.heappush(queue, (-obj0, node_id, z_new))
            node_id += 1
    
    if best_z is None:
        # Fall back to just pick 26 smallest indices
        best_z = np.zeros(120, dtype=int)
        best_z[:26] = 1
        obj, best_r = milp_solver(best_z)
        if obj is None:
            raise RuntimeError("No solution found")
    return best_z, best_r

def refine_solution(centers, radii):
    n = len(centers)
    # Prepare variable bounds and constraints
    bounds = []
    for i in range(n):
        r = radii[i]
        x, y = centers[i]
        bounds.append((0, x))  # r <= x
        bounds.append((0, 1 - x))  # r <= 1 - x
        bounds.append((0, y))  # r <= y
        bounds.append((0, 1 - y))  # r <= 1 - y

    def objective(r_vec):
        return -np.sum(r_vec)  # maximize sum of radii

    def gradient(r_vec):
        return -np.ones_like(r_vec)

    constraints = []

    def barrier_term(r_vec):
        eps = 1e-4
        return -eps * np.sum(np.log(r_vec + 1e-12))

    def barrier_grad(r_vec):
        eps = 1e-4
        return -eps * (1.0 / (r_vec + 1e-12))

    def constraint_func(x_vec):
        # x_vec = [x1, y1, r1, x2, y2, r2, ...]
        res = []
        for i in range(len(centers)):
            x, y = x_vec[2*i], x_vec[2*i+1]
            r = x_vec[2*n + i]  # r_i
            # Wall constraints
            res.append(1.0 - x - r)  # r <= 1 - x
            res.append(1.0 - y - r)  # r <= 1 - y
            res.append(x - r)  # r <= x
            res.append(y - r)  # r <= y
            # Inter circle constraints
            for j in range(i+1, len(centers)):
                xj, yj = x_vec[2*j], x_vec[2*j+1]
                rj = x_vec[2*n + j]
                dist_sq = (x - xj)**2 + (y - yj)**2
                res.append(dist_sq - (r + rj)**2)
        return np.array(res)

    x0 = []
    for i in range(n):
        x0.append(centers[i][0])
        x0.append(centers[i][1])
        x0.append(radii[i])
    x0 = np.array(x0)
    
    # Constraint bounds
    ub = [0.0] * (4*n + n*(n-1)//2)  # inequality <= 0
    cons = {'type': 'ineq', 'fun': constraint_func, 'jac': lambda x: np.zeros(len(ub))}

    result = minimize(
        objective,
        x0,
        method='SLSQP',
        jac=gradient,
        bounds=[(1e-9, 0.5)] * len(x0),
        constraints=cons,
        options={'ftol': 1e-9, 'maxiter': 2000, 'disp': False}
    )
    
    # Extract result
    if not result.success:
        # If optimization failed, return original
        return centers.copy(), radii.copy(), np.sum(radii)
    
    x_result = result.x
    refined_centers = np.zeros((n, 2))
    refined_radii = np.zeros(n)
    for i in range(n):
        refined_centers[i][0] = x_result[2*i]
        refined_centers[i][1] = x_result[2*i+1]
        refined_radii[i] = x_result[2*n + i]
    
    excessive = 1e-12
    mask = refined_radii < excessive
    refined_radii[mask] = 0.0
    sum_radii = np.sum(refined_radii)
    
    # Post-process to check feasibility
    valid, msg = validate_packing(refined_centers, refined_radii)
    if not valid:
        # Try uniform shrink
        factor = 1.0 - 1e-7
        refined_radii *= factor
        valid, msg = validate_packing(refined_centers, refined_radii)
        if not valid:
            # Still invalid; shrink more and try default
            refined_radii *= 0.999
            valid, msg = validate_packing(refined_centers, refined_radii)
            if not valid:
                # Revert to original or raise?
                return centers, radii, np.sum(radii)
    
    return refined_centers, refined_radii, sum_radii

def run_packing():
    P = generate_lattice(0.13)
    milp = build_milp(P)
    best_z, best_r = branch_and_bound(milp, node_limit=20000)
    best_centers = P[best_z == 1]
    best_radii = best_r[best_z == 1]
    # Refine
    final_centers, final_radii, sum_radii = refine_solution(best_centers, best_radii)
    return final_centers, final_radii, sum_radii
