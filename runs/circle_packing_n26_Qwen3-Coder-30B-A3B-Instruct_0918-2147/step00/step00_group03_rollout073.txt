import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(0)  # For reproducibility

    # 1. Initial placement: perturbed hexagonal grid
    def init_grid(n_circles=26, spacing=0.07):
        rows = int(np.ceil(np.sqrt(n_circles) * 2))
        cols = int(np.ceil(n_circles / rows))
        y_coords = np.array([spacing/2 + k * spacing * np.sqrt(3)/2 for k in range(rows)])
        x_coords = []
        y_coords_filtered = []
        for k, y in enumerate(y_coords):
            if k % 2 == 0:
                xs = np.arange(spacing/2, 1, spacing)
            else:
                xs = np.arange(spacing/2 + spacing/2, 1, spacing)
            for x in xs:
                if 0 <= x <= 1 and 0 <= y <= 1:
                    x_coords.append(x)
                    y_coords_filtered.append(y)
                if len(x_coords) >= n_circles:
                    break
            if len(x_coords) >= n_circles:
                break
        centers = np.array(list(zip(x_coords[:n_circles], y_coords_filtered[:n_circles]))).astype(np.float64)
        # Add random jitter to break symmetry
        centers += np.random.uniform(-0.001, 0.001, centers.shape)
        return centers

    centers = init_grid(26)
    radii = np.full(26, 0.001, dtype=np.float64)  # Initially tiny radii

    # 2. Physics-based simulation with forces and radius growth
    def compute_forces(centers, radii):
        N = len(centers)
        forces = np.zeros_like(centers)
        k_w = 10.0
        k_p = 5.0
        for i in range(N):
            x_i, y_i = centers[i]
            r_i = radii[i]
            # Wall repulsion (soft, unless circle touches)
            fx, fy = 0.0, 0.0
            d_L = x_i - r_i
            d_R = (1.0 - x_i) - r_i
            d_B = y_i - r_i
            d_T = (1.0 - y_i) - r_i
            if d_L < 0:
                fx -= k_w * (-d_L)
            if d_R < 0:
                fx += k_w * (-d_R)
            if d_B < 0:
                fy -= k_w * (-d_B)
            if d_T < 0:
                fy += k_w * (-d_T)
            # Pairwise repulsion
            for j in range(N):
                if i == j:
                    continue
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                gap = np.sqrt(dx**2 + dy**2) - (radii[i] + radii[j])
                if gap < 0:
                    f = k_p * (-gap)
                    fx += f * dx / (np.sqrt(dx**2 + dy**2) + 1e-12)
                    fy += f * dy / (np.sqrt(dx**2 + dy**2) + 1e-12)
            forces[i] = np.array([fx, fy])
        return forces

    def grow_radii(centers, radii, forces, dt=0.02, alpha=1e-3):
        dot_r = np.ones_like(radii) * alpha  # Base growth rate
        for i in range(len(centers)):
            if forces[i].any() or units := (
                centers[i, 0] - radii[i] < 1e-12 or
                1.0 - centers[i, 0] - radii[i] < 1e-12 or
                centers[i, 1] - radii[i] < 1e-12 or
                1.0 - centers[i, 1] - radii[i] < 1e-12):
                continue
            for j in range(len(centers)):
                if i == j:
                    continue
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    dot_r[i] = 0
                    break
        new_radii = radii + dt * dot_r
        new_radii = np.maximum(new_radii, 0)
        return new_radii

    # Simulation loop
    dt = 0.02
    gamma = 1.0
    v = np.zeros_like(centers)  # velocities
    stagnant_count = 0

    for step in range(5000):
        forces = compute_forces(centers, radii)
        # Velocity update
        v += dt * (forces - gamma * v)
        # Position update
        centers += dt * v
        # Clip positions to unit square
        centers = np.clip(centers, 0, 1)
        # Grow radii
        new_radii = grow_radii(centers, radii, forces, dt, alpha=1e-3)
        # Check if converged
        if np.allclose(radii, new_radii, rtol=1e-12):
            stagnant_count += 1
        else:
            stagnant_count = 0
        radii = new_radii
        # Stop if stable
        if stagnant_count > 100 and np.max(np.linalg.norm(forces, axis=1)) < 1e-6:
            break

    # 3. Compute maximal radii for fixed centers (in exact manner)
    def max_radii_fixed_centers(centers, radii):
        max_iter = 30
        for _ in range(max_iter):
            new_radii = np.zeros_like(radii)
            for i in range(len(centers)):
                c_i = centers[i]
                # Wall constraints
                min_dist = min(c_i[0], 1-c_i[0], c_i[1], 1-c_i[1])
                # Circle constraints
                for j in range(len(centers)):
                    if i == j:
                        continue
                    c_j = centers[j]
                    dist = np.sqrt(np.sum((c_i - c_j)**2))
                    min_dist = min(min_dist, dist - radii[j])
                new_radii[i] = min_dist
            # Update
            if np.allclose(radii, new_radii, rtol=1e-9):
                break
            radii = new_radii
        return radii

    radii = max_radii_fixed_centers(centers, radii)

    # 4. L-BFGS-B refinement of centers to maximize sum of radii
    def objective_sum_radii(x_flat):
        # Reconstruct centers from flat vector
        centers = x_flat.reshape(-1, 2)
        # Fix radii as if they were maximal given these centers
        radii_new = max_radii_fixed_centers(centers, np.zeros(len(centers)))
        return -np.sum(radii_new)

    def callback(xk):
        pass

    # Minimize -sum(radii) (i.e. maximize sum(radii))
    x0 = centers.flatten()
    bounds = [(0, 1) for _ in range(52)]
    res = minimize(objective_sum_radii, x0, method='L-BFGS-B', bounds=bounds, tol=1e-9, options={'maxiter': 2000}, callback=callback)
    optimized_centers = res.x.reshape(-1, 2).astype(np.float64)
    optimized_radii = max_radii_fixed_centers(optimized_centers, np.zeros(26))

    # 5. Final verification
    final_sum = np.sum(optimized_radii)
    validate_packing(optimized_centers, optimized_radii)

    return optimized_centers, optimized_radii, final_sum
