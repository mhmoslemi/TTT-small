import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial.distance import cdist

def build_lp_model(n_circles):
    # Build the LP model for the circle packing problem
    # Variables: [x1, y1, r1, x2, y2, r2, ..., x_n, y_n, r_n]
    n_vars = 3 * n_circles
    A_ub = []
    b_ub = []
    A_eq = []
    b_eq = []
    
    # Pairwise separation constraints (axis-aligned)
    for i in range(n_circles):
        for j in range(i+1, n_circles):
            # x-axis separation - two inequalities for each pair
            row1 = np.zeros(n_vars)
            row1[3*i] = 1.0  # x_i
            row1[3*j] = -1.0 # -x_j
            row1[3*i+2] = 1.0 # r_i
            row1[3*j+2] = 1.0 # r_j
            A_ub.append(row1)
            b_ub.append(0.0)
            
            row2 = np.zeros(n_vars)
            row2[3*j] = 1.0  # x_j
            row2[3*i] = -1.0 # -x_i
            row2[3*j+2] = 1.0 # r_j
            row2[3*i+2] = 1.0 # r_i
            A_ub.append(row2)
            b_ub.append(0.0)
            
            # y-axis separation - two inequalities for each pair
            row3 = np.zeros(n_vars)
            row3[3*i+1] = 1.0  # y_i
            row3[3*j+1] = -1.0 # -y_j
            row3[3*i+2] = 1.0 # r_i
            row3[3*j+2] = 1.0 # r_j
            A_ub.append(row3)
            b_ub.append(0.0)
            
            row4 = np.zeros(n_vars)
            row4[3*j+1] = 1.0  # y_j
            row4[3*i+1] = -1.0 # -y_i
            row4[3*j+2] = 1.0 # r_j
            row4[3*i+2] = 1.0 # r_i
            A_ub.append(row4)
            b_ub.append(0.0)
    
    # Wall constraints
    for i in range(n_circles):
        # r_i <= x_i
        row1 = np.zeros(n_vars)
        row1[3*i] = -1.0
        row1[3*i+2] = 1.0
        A_ub.append(row1)
        b_ub.append(0.0)
        
        # r_i <= 1 - x_i
        row2 = np.zeros(n_vars)
        row2[3*i] = 1.0
        row2[3*i+2] = 1.0
        A_ub.append(row2)
        b_ub.append(1.0)
        
        # r_i <= y_i
        row3 = np.zeros(n_vars)
        row3[3*i+1] = -1.0
        row3[3*i+2] = 1.0
        A_ub.append(row3)
        b_ub.append(0.0)
        
        # r_i <= 1 - y_i
        row4 = np.zeros(n_vars)
        row4[3*i+1] = 1.0
        row4[3*i+2] = 1.0
        A_ub.append(row4)
        b_ub.append(1.0)
    
    # Objective: maximize sum of radii
    c = np.zeros(n_vars)
    c[2::3] = -1.0  # coefficients for r_i in objective
    
    return np.array(A_ub), np.array(b_ub), c

def solve_lp(A_ub, b_ub, c):
    # Solve the LP problem
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    if not result.success:
        raise RuntimeError("Linear programming failed to solve")
    
    # Extract solution
    solution = result.x
    centers = solution[0::3][:26].reshape((26, 2))
    radii = solution[2::3][:26]
    
    return centers, radii

def find_active_constraints(centers, radii, A_ub, b_ub):
    # Build boolean mask of active constraints
    n_circles = centers.shape[0]
    n_constraints = A_ub.shape[0]
    active_indices = []
    
    # Calculate constraint violations
    violation = A_ub @ np.concatenate((centers.flatten(), radii))
    # Use that constraints are less than or equal
    # So if violation is close to right hand side, it's active
    for i in range(n_constraints):
        if abs(violation[i] - b_ub[i]) < 1e-9:
            active_indices.append(i)
    
    return active_indices

def detect_diagonal_conflicts(centers, radii):
    # Check for diagonal overlaps
    conflict_pairs = []
    pairs = []
    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):
            dist = np.linalg.norm(centers[i] - centers[j])
            if dist < radii[i] + radii[j] - 1e-12:
                conflict_pairs.append((i, j))
                
    return conflict_pairs

def resolve_conflicts(centers, radii, active_set, conflict_pairs):
    # Solves a small quadratic subproblem to resolve conflicts
    # Relax problem such that only active walls and axes are fixed
    def squared_distance(xi, yi, xj, yj):
        return (xi - xj)**2 + (yi - yj)**2
    
    def penalty_function(centers_seq, radii_seq):
        val = 0.0
        for i, j in conflict_pairs:
            dx = centers_seq[i][0] - centers_seq[j][0]
            dy = centers_seq[i][1] - centers_seq[j][1]
            d = np.sqrt(dx*dx + dy*dy)
            overlap = d - (radii_seq[i] + radii_seq[j])
            if overlap < 0:
                val += overlap * overlap
        return val
    
    # Try iterative solution
    for _ in range(100): # limit iterations
        # For small number of conflicts this approximation works
        # A bit ad-hoc but should work reasonably well
        if len(conflict_pairs) == 0:
            break
            
        # Simple constraint-satisfying move
        changed = False
        for a, b in conflict_pairs:
            dx = centers[a][0] - centers[b][0]
            dy = centers[a][1] - centers[b][1]
            d = np.sqrt(dx*dx + dy*dy)
            if d < radii[a] + radii[b]:
                correction = (radii[a] + radii[b] - d) / 2.0
                
                # Move away from each other
                move_factor = 1.0
                dx_normalize = dx / (d + 1e-12)
                dy_normalize = dy / (d + 1e-12)
                
                centers[a][0] += dx_normalize * move_factor * correction
                centers[a][1] += dy_normalize * move_factor * correction
                centers[b][0] -= dx_normalize * move_factor * correction
                centers[b][1] -= dy_normalize * move_factor * correction
                
                changed = True
        
        # Check conflicts again
        new_conflicts = detect_diagonal_conflicts(centers, radii)
        if len(new_conflicts) == len(conflict_pairs) or not changed:
            break
        conflict_pairs = new_conflicts
    
    # Clip to unit square with minimal radius adjustments
    n = len(centers)
    for i in range(n):
        # Adjust centers to stay in unit square
        for dim in range(2):
            if centers[i][dim] < radii[i]:
                centers[i][dim] = radii[i]
            elif centers[i][dim] > 1 - radii[i]:
                centers[i][dim] = 1 - radii[i]
                
    return centers, radii

def polish_global(centers, radii, active_set):
    # Global polishing keeping constraint structure fixed
    # We want to maximize sum of radii, but start from current configuration which respected
    # the intersection of active constraints.
    
    # Objective function
    def objective(radii):
        return -np.sum(radii)
    
    # Initialize combined constraints list
    n_circles = len(centers)
    
    # Convert constraints into bound/linear form for scipy.optimize
    # Define different sorting logic here by asserting the structure
    
    # Exactly because this SIMDified implementation is much faster,
    # we'll loosely aggregate non-overlapping constraints with simple subproblem solving:
    
    # Use scipy minimizer, but many bounds are just linker-enabled constants.
    
    # First make sure we satisfy non-negative radius
    bounds = [(0, 1) for _ in range(n_circles)]
    
    # No explicit linear or nonlinear inequality solutions for core implementation,
    # defer to non-constrained severe cleaning:
    def noise_naive(x):
        return -np.sum(x)
    
    res = minimize(noise_naive, radii, method='Nelder-Mead', bounds=bounds,
                   options={'ftol': 1e-10, 'xtol': 1e-10, 'maxiter': 1000})
    
    return res.x

def max_radius_projection(centers, radii):
    # Project each circle to its maximum admissible radius given current placements
    max_radii = radii.copy()
    converged = False
    max_iter = 100
    
    for _ in range(max_iter):
        old_radii = max_radii.copy()
        
        # Compute minimal distances to all other circles and to walls
        for i in range(len(centers)):
            # Distance to walls
            near_wall = [
                centers[i][0],         # left
                1 - centers[i][0],     # right
                centers[i][1],         # bottom
                1 - centers[i][1]      # top
            ]
            
            # Distance to other circles
            distances = []
            for j in range(len(centers)):
                if i != j:
                    dist = np.linalg.norm(centers[i] - centers[j])
                    distances.append(dist)
                    
            min_dist_to_others = min(distances) if distances else 1e6
            min_to_wall = min(near_wall)
            max_radii[i] = min(min_to_wall, min_dist_to_others)
            
        if np.allclose(max_radii, old_radii, atol=1e-10):
            break
            
    return max_radii
