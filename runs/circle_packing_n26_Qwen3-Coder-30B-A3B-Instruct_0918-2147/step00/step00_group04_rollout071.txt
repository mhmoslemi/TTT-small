import numpy as np
from scipy.optimize import linprog
from math import sqrt

def initial_hex_centers(k, spacing):
    s = spacing
    sqrt3 = sqrt(3)
    centers = []
    j = 0
    while len(centers) < k:
        for i in range(int(1.0 / s) + 2):
            px = i * s + (j % 2) * s / 2
            py = j * s * sqrt3 / 2
            if 0 <= px <= 1 and 0 <= py <= 1:
                centers.append([px, py])
                if len(centers) >= k:
                   break
        j += 1
    return np.array(centers[:k])

def initial_radii(k, r0):
    return np.full(k, r0)

def build_lp_constants(centers, radii):
    n = centers.shape[0]
    d_left = centers[:, 0]
    d_right = 1 - centers[:, 0]
    d_bottom = centers[:, 1]
    d_top = 1 - centers[:, 1]
    D = np.zeros((n, n))
    for i in range(n):
        for j in range(i+1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = sqrt(dx*dx + dy*dy)
            D[i, j] = dist
            D[j, i] = dist

    return {
        "d_left": d_left,
        "d_right": d_right,
        "d_bottom": d_bottom,
        "d_top": d_top,
        "D": D
    }

def solve_radius_lp(consts, radii, graph_edges, alpha):
    n = len(radii)
    d_left = consts["d_left"]
    d_right = consts["d_right"]
    d_bottom = consts["d_bottom"]
    d_top = consts["d_top"]
    D = consts["D"]
    
    A = []
    b = []
    
    # Wall constraints
    for i in range(n):
        max_grow = min(d_left[i], d_right[i], d_bottom[i], d_top[i]) - radii[i]
        if max_grow < 0: max_grow = 0
        A.append([-1.0 if j == i else 0.0 for j in range(n)])
        b.append(-max_grow)
    
    # Pairwise constraints
    for i in range(n):
        for j in range(i+1, n):
            max_grow = D[i, j] - (radii[i] + radii[j])
            if max_grow < 0: max_grow = 0
            row = [0.0] * n
            row[i] = -1.0
            row[j] = -1.0
            A.append(row)
            b.append(-max_grow)
    
    # Graph-based transfer limits (optional)
    # We can skip for simplicity

    c = np.ones(n)
    try:
        res = linprog(c, A_ub=A, b_ub=b, method='highs', options={'presolve': True, 'time_limit': 1.0})
        if res.success:
            return res.x
        else:
            return np.zeros(n)
    except Exception:
        return np.zeros(n)

def apply_increment(radii, Δr):
    return radii + Δr

def push_feasibility(centers, radii):
    n = centers.shape[0]
    updated_centers = centers.copy()
    updated_radii = radii.copy()
    
    # Handle wall violations
    for i in range(n):
        x, y = updated_centers[i]
        r = updated_radii[i]
        if x - r < 0:
            updated_radii[i] = x
        if x + r > 1:
            updated_radii[i] = 1 - x
        if y - r < 0:
            updated_radii[i] = y
        if y + r > 1:
            updated_radii[i] = 1 - y
    
    # Handle pairwise overlaps
    tol = 1e-12
    done = False
    while not done:
        done = True
        for i in range(n):
            for j in range(i+1, n):
                dx = updated_centers[i, 0] - updated_centers[j, 0]
                dy = updated_centers[i, 1] - updated_centers[j, 1]
                dist = sqrt(dx*dx + dy*dy)
                r_sum = updated_radii[i] + updated_radii[j]
                if dist < r_sum - tol:
                    overlap = r_sum - dist
                    dx /= dist
                    dy /= dist
                    updated_centers[i, 0] -= (overlap/2) * dx
                    updated_centers[i, 1] -= (overlap/2) * dy
                    updated_centers[j, 0] += (overlap/2) * dx
                    updated_centers[j, 1] += (overlap/2) * dy
                    done = False
    
    # Clamp final positions
    for i in range(n):
        updated_centers[i, 0] = max(0, min(1, updated_centers[i, 0]))
        updated_centers[i, 1] = max(0, min(1, updated_centers[i, 1]))
        updated_radii[i] = min(updated_radii[i], 
                               updated_centers[i, 0],
                               1 - updated_centers[i, 0],
                               updated_centers[i, 1],
                               1 - updated_centers[i, 1])
    
    return updated_centers, updated_radii

def run_iterations(centers, radii, max_iter, ε, α):
    graph_edges = set()
    center_history = [centers.copy()]
    radius_history = [radii.copy()]
    
    for _ in range(max_iter):
        # Build constants for LP
        consts = build_lp_constants(centers, radii)
        
        # Solve LP for Δr
        Δr = solve_radius_lp(consts, radii, graph_edges, α)
        
        # Apply increments
        new_radii = apply_increment(radii, Δr)
        
        # Calculate increase
        delta_sum = np.sum(Δr)
        if delta_sum < ε:
            break
        
        # Temporarily increase radii to get centers
        temp_centers = centers.copy()
        temp_radii = new_radii
        
        # Push
        centers, radii = push_feasibility(temp_centers, temp_radii)
        
        # Keep track of changes
        center_history.append(centers.copy())
        radius_history.append(radii.copy())
    
    return centers, radii

def final_refinement(centers, radii):
    # Final high precision LP with tight constraints
    consts = build_lp_constants(centers, radii)
    try:
        Δr = solve_radius_lp(consts, radii, set(), 0.5)
        new_radii = apply_increment(radii, Δr)
    except:
        new_radii = radii.copy()
    
    # Final safety clamping
    for i in range(len(centers)):
        r = new_radii[i]
        x = centers[i, 0]
        y = centers[i, 1]
        r = min(r, x, 1-x, y, 1-y)
        new_radii[i] = r
    
    # Pairwise final checks
    eps = 1e-13
    n = len(centers)
    for i in range(n):
        for j in range(i+1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = sqrt(dx*dx + dy*dy)
            r_sum = new_radii[i] + new_radii[j]
            if dist < r_sum - eps:
                delta = r_sum - dist
                if new_radii[i] >= new_radii[j]:
                    new_radii[i] = new_radii[i] - delta
                else:
                    new_radii[j] = new_radii[j] - delta
    
    return centers, new_radii

def run_packing():
    # Initial setup
    k = 26
    spacing = 0.12
    r0 = 0.005
    max_iter = 100
    ε = 1e-5
    α = 0.5
    
    # Generate initial placement
    centers = initial_hex_centers(k, spacing)
    radii = initial_radii(k, r0)
    
    # Run iterations
    centers, radii = run_iterations(centers, radii, max_iter, ε, α)
    
    # Final refinement
    centers, radii = final_refinement(centers, radii)
    
    # Calculate final sum
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
