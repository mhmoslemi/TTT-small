import numpy as np
from scipy.optimize import minimize

def hex_lattice_points(n_rows, spacing):
    """Generate points in a hexagonal lattice inside unit square"""
    points = []
    for i in range(n_rows):
        y = i * spacing * np.sqrt(3) / 2
        if y > 1:
            break
        offset = (i % 2) * spacing / 2
        n_cols = int((1 - offset) / spacing) + 1
        for j in range(n_cols):
            x = offset + j * spacing
            if x > 1:
                break
            points.append([x, y])
    return np.array(points)

def initial_guess(seed):
    """Generate initial guess with jittered hexagonal lattice"""
    np.random.seed(seed)
    
    # Use a hexagonal lattice with 26 points
    spacing = 0.2
    lattice_points = hex_lattice_points(10, spacing)
    
    # If we don't have enough, pad with jittered points
    if len(lattice_points) < 26:
        # Generate more
        lattice_points = hex_lattice_points(12, spacing)
        
    # Take first 26 points if necessary
    if len(lattice_points) > 26:
        lattice_points = lattice_points[:26]
        
    # Pad if necessary by generating random points
    if len(lattice_points) < 26:
        while len(lattice_points) < 26:
            x = np.random.uniform(0.01, 0.99)
            y = np.random.uniform(0.01, 0.99)
            lattice_points = np.vstack((lattice_points, [x, y]))
            
    # Start with small uniform radius
    initial_radii = np.full(26, 0.02)
    
    # Add jitter to positions
    jitter = np.random.uniform(-0.02, 0.02, (26, 2))
    positions = lattice_points + jitter
    positions = np.clip(positions, 0.01, 0.99)
    
    # Concatenate to get variables (x, y, r) for each circle
    vars0 = np.empty(78)
    vars0[0::3] = positions[:, 0]
    vars0[1::3] = positions[:, 1]
    vars0[2::3] = initial_radii
    
    return vars0

def objective(vars):
    """Minimize negative sum of radii"""
    radii = vars[2::3]
    return -np.sum(radii)

def constraint_boundary(vars):
    """Ensure each circle is inside the unit square"""
    xs = vars[0::3]
    ys = vars[1::3]
    rs = vars[2::3]
    # x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    boundaries = np.hstack([
        xs - rs, 
        1 - xs - rs,
        ys - rs,
        1 - ys - rs
    ])
    return boundaries

def constraint_nonoverlap(vars):
    """Ensure no overlap between circles"""
    n = len(vars) // 3
    xs = vars[0::3]
    ys = vars[1::3]
    rs = vars[2::3]
    
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            dx = xs[i] - xs[j]
            dy = ys[i] - ys[j]
            dist_sq = dx*dx + dy*dy
            sum_radii = rs[i] + rs[j]
            # We want dist >= sum_radii -> dist - sum_radii >= 0
            constraints.append(np.sqrt(dist_sq) - sum_radii)
    
    return np.array(constraints)

def postprocess(vars):
    """Fix small constraint violations"""
    new_vars = vars.copy()
    xs = new_vars[0::3]
    ys = new_vars[1::3]
    rs = new_vars[2::3]
    
    # Fix boundary violations
    for i in range(len(xs)):
        if xs[i] - rs[i] < 0:
            xs[i] = rs[i]
        elif xs[i] + rs[i] > 1:
            xs[i] = 1 - rs[i]
        if ys[i] - rs[i] < 0:
            ys[i] = rs[i]
        elif ys[i] + rs[i] > 1:
            ys[i] = 1 - rs[i]
    
    # Repair overlaps: reduce radii of larger circles
    for i in range(len(xs)):
        for j in range(i + 1, len(xs)):
            dx = xs[i] - xs[j]
            dy = ys[i] - ys[j]
            dist = np.sqrt(dx*dx + dy*dy)
            sum_radii = rs[i] + rs[j]
            if dist < sum_radii:
                overlap = sum_radii - dist
                # Reduce the larger one
                if rs[i] > rs[j]:
                    rs[i] -= overlap * 0.5
                else:
                    rs[j] -= overlap * 0.5
                # Validate radii positive
                rs[i] = max(1e-9, rs[i])
                rs[j] = max(1e-9, rs[j])
    
    return new_vars

def run_packing():
    NUM_RESTARTS = 5
    best_sum = -np.inf
    best_vars = None
    
    # Create bounds and constraints
    bounds = []
    for i in range(26):
        bounds.append((0, 1))  # x bounds
        bounds.append((0, 1))  # y bounds
        bounds.append((0, 0.5))  # r bounds
    
    # Global constraints
    global_constraints = []
    global_constraints.append({
        'type': 'ineq',
        'fun': constraint_boundary
    })
    
    # Time limit: 1000 iterations per restart is reasonable
    for seed in range(NUM_RESTARTS):
        vars0 = initial_guess(seed)
        res = minimize(objective, vars0,
                       method='SLSQP',
                       bounds=bounds,
                       constraints=global_constraints,
                       options={'maxiter': 2000, 'ftol': 1e-7})
        
        if not res.success:
            continue
        
        vars_corr = postprocess(res.x)
        s = np.sum(vars_corr[2::3])
        if s > best_sum:
            best_sum = s
            best_vars = vars_corr
    
    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii = best_vars[2::3]
    
    # Final validation
    valid, msg = validate_packing(centers, radii)
    assert valid, msg
    
    return centers, radii, float(best_sum)
