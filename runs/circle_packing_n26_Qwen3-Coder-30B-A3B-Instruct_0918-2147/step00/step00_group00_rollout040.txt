import numpy as np
from scipy.optimize import linprog
from scipy.spatial.distance import cdist

def generate_hex_lattice(h):
    """Generate hexagonal lattice points in [0,1]x[0,1] with spacing h"""
    sqrt3 = np.sqrt(3)
    points = []
    j_max = int(np.ceil(1.0 / (h * sqrt3 / 2))) + 1
    for j in range(j_max):
        y = j * h * sqrt3 / 2
        if y > 1:
            break
        i_max = int(np.ceil(1.0 / h)) + 1
        for i in range(i_max):
            x = i * h + (j % 2) * (h / 2)
            if x > 1:
                break
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append([x, y])
    return np.array(points)

def build_neighbour_list(centers, max_dist=0.4):
    """Build list of close neighbours with distances"""
    distances = cdist(centers, centers)
    edges = []
    n = len(centers)
    for i in range(n):
        for j in range(i + 1, n):
            d = distances[i, j]
            if d < max_dist:
                edges.append((i, j, d))
    return edges

def solve_radius_lp(selected_indices, R_wall, neighbour_list):
    """Solve LP for selected circle radii, return optimal radii and feasibility"""
    # LP setup: maximize sum of radii, with constraints:
    #   - r_i <= R_wall[i]
    #   - r_i + r_j <= d_ij for all neighbours (i,j) in selected indices
    n_selected = len(selected_indices)
    if n_selected == 0:
        return np.array([]), True

    # Build constraint matrix A_ub x <= b_ub
    # Each variable is indexed from 0 to n_selected-1
    A_ub = []
    b_ub = []
    # Wall constraints: r_i <= R_wall[i] --> -r_i >= -R_wall[i]
    for i in range(n_selected):
        row = [0] * n_selected
        row[i] = -1
        A_ub.append(row)
        b_ub.append(-R_wall[selected_indices[i]])

    # Neighbour constraints: r_i + r_j <= d_ij
    selected_set = set(selected_indices)
    for i, j, d_ij in neighbour_list:
        if i in selected_set and j in selected_set:
            idx_i = selected_indices.index(i)
            idx_j = selected_indices.index(j)
            row = [0] * n_selected
            row[idx_i] = 1
            row[idx_j] = 1
            A_ub.append(row)
            b_ub.append(d_ij)

    # Objective: maximize sum of radii --> minimize -sum of radii
    c = np.array([-1.0] * n_selected)

    # Variable bounds [0, R_wall[i]]
    bounds = [(0, R_wall[i]) for i in selected_indices]

    # Solve using scipy.optimize.linprog
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, True
    else:
        return np.array([]), False

def upper_bound(partial_set, remaining_indices, R_wall):
    """Compute optimistic upper bound for total sum of radii from this partial set"""
    total = sum(R_wall[i] for i in partial_set)
    # Get the largest R_wall for remaining indices
    remaining_r_walls = [R_wall[i] for i in remaining_indices]
    if len(remaining_r_walls) > 0:
        total += sum(remaining_r_walls)  # assume we can assign best remaining
    return total

def branch_and_bound(centers, R_wall, neighbour_list):
    """Branch and bound search for best set of 26 circles"""
    n_points = len(centers)
    best_solution = None
    best_sum = -np.inf

    # Heuristic ordering: start with points having highest wall radii
    sorted_indices = np.argsort([-R_wall[i] for i in range(n_points)])[::-1]

    # Stack: State is (partial_indices, unassigned_indices, lb, ub)
    stack = [([], sorted_indices, -np.inf, np.inf)]

    while stack:
        S_partial, U, lb, ub = stack.pop()
        remaining_count = 26 - len(S_partial)

        # If we have too few remaining points or too optimistic, skip
        if remaining_count == 0:
            # Leaf node: solve exact LP
            radii, feasible = solve_radius_lp(S_partial, R_wall, neighbour_list)
            if feasible and len(radii) == 26:
                total_sum = np.sum(radii)
                if total_sum > best_sum:
                    best_sum = total_sum
                    best_solution = (np.array(S_partial), radii.copy())
        elif remaining_count > 0 and ub > best_sum:
            # Pruning based on remaining
            if len(U) < remaining_count:
                # Not enough candidates
                continue

            # Branch on first element of U
            if len(U) == 0:
                continue

            c = U[0]
            new_S_partial = S_partial + [c]
            new_U = U[1:]
            new_lb = lb
            new_ub = upper_bound(new_S_partial, new_U, R_wall)

            # Add the subtree nodes
            # Include c
            if new_ub > lb:
                stack.append((new_S_partial, new_U, new_lb, new_ub))

            # Exclude c => just pop c from U and explore further
            # Don't include new_S_partial + [c] but instead just explore with U[1:]
            if new_ub > best_sum:
                stack.append((S_partial, new_U, lb, new_ub))
        else:
            # Reconstruct
            continue

    if best_solution is None:
        raise RuntimeError("No valid solution found in branch-and-bound")

    selected_indices, radii = best_solution
    centers_selected = centers[selected_indices]
    return centers_selected, radii

def local_continuous_refinement(start_centers, start_radii, centers, R_wall, neighbour_list):
    """Run local continuous optimization to refine the discrete solution"""
    # Use SLSQP as more suitable for constraints
    from scipy.optimize import minimize

    initial_x = np.concatenate([start_centers.flatten(), start_radii])
    n = len(start_centers)
    bounds = []
    for i in range(n):
        bounds.append((0, 1))  # x coordinates
        bounds.append((0, 1))  # y coordinates
    for i in range(n):
        bounds.append((0, R_wall[i]))  # r coordinates

    def objective(x):
        radii = x[2*n:]
        return -np.sum(radii)  # Negative because we maximize sum of radii

    def constraint_func(x):
        centers_combined = x[:2*n].reshape(-1, 2)
        radii = x[2*n:]
        results = []

        # Wall constraints
        for i in range(n):
            x_c, y_c = centers_combined[i]
            r = radii[i]
            g1 = x_c - r  # x - r >= 0
            g2 = 1 - x_c - r  # 1 - x - r >= 0
            g3 = y_c - r  # y - r >= 0
            g4 = 1 - y_c - r  # 1 - y - r >= 0
            results.append(g1)
            results.append(g2)
            results.append(g3)
            results.append(g4)

        # Neighbour constraints
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_combined[i, 0] - centers_combined[j, 0]
                dy = centers_combined[i, 1] - centers_combined[j, 1]
                dist_ij = np.sqrt(dx**2 + dy**2)
                r_i = radii[i]
                r_j = radii[j]
                # r_i + r_j <= dist_ij => r_i + r_j - dist_ij <= 0
                results.append(r_i + r_j - dist_ij)

        return np.array(results)

    # The number of constraints: 4*n wall + (n*n-n)/2 neighbour
    num_constraints = 4 * n + (n * (n - 1)) // 2

    # Create a simple lottery initial point for robust start
    def generate_initial_guess():
        # Perturb from original
        noise = np.random.uniform(-0.002, 0.002, size=(n, 2))
        center_perturbed = np.clip(start_centers + noise, 0, 1)
        radii_perturbed = np.clip(start_radii + np.random.uniform(-0.001, 0.001, size=n), 0, 1)
        return np.concatenate([center_perturbed.flatten(), radii_perturbed])

    x0 = generate_initial_guess()
    for retry in range(3):
        try:
            res = minimize(
                objective,
                x0,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'ineq', 'fun': constraint_func},
                options={'maxiter': 200, 'ftol': 1e-6}
            )
            if res.success and np.isfinite(res.fun):
                new_centers = res.x[:2*n].reshape(n, 2)
                new_radii = res.x[2*n:]
                break
            else:
                x0 = generate_initial_guess()
        except Exception:
            x0 = generate_initial_guess()

    return new_centers, new_radii

def run_packing():
    # Generate hexagonal lattice
    h = 0.07
    centers = generate_hex_lattice(h)
    n_points = len(centers)
    R_wall = np.array([min(x, 1 - x, y, 1 - y) for x, y in centers], dtype=float)

    # Build neighbour list
    neighbour_list = build_neighbour_list(centers, max_dist=0.4)

    # Branch and bound
    chosen_centers, radii = branch_and_bound(centers, R_wall, neighbour_list)

    # Continuous refinement for better total sum
    refined_centers, refined_radii = local_continuous_refinement(chosen_centers, radii, centers, R_wall, neighbour_list)

    # Validate
    valid, msg = validate_packing(refined_centers, refined_radii)

            scale_factor = 1.0 - 1e-7
