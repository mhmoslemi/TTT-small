import numpy as np
from scipy.optimize import linprog

def initial_hex_centers(k, spacing):
    sqrt3 = np.sqrt(3)
    centers = []
    i = 0
    j = 0
    while len(centers) < k:
        x = i * spacing + (j % 2) * spacing / 2
        y = j * spacing * sqrt3 / 2
        if 0 <= x <= 1 and 0 <= y <= 1:
            centers.append([x, y])
        if x > 1:
            j += 1
            i = 0
        else:
            i += 1
    return np.array(centers[:k])

def initial_radii(k, r0):
    return np.full(k, r0)

def build_lp_constants(centers, radii):
    n = centers.shape[0]
    wall_dists = np.array([
        [centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1]]
        for i in range(n)
    ])
    d_ij = np.array([
        [np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
         for j in range(i + 1, n)]
        for i in range(n)
    ])
    return wall_dists, d_ij

def solve_radius_lp(consts, radii, graph, alpha=0.5):
    wall_dists, d_ij = consts
    n = len(radii)
    # Build constraint matrix and bounds
    A_eq = []
    b_eq = []
    A_ub = []
    b_ub = []
    c = np.ones(n)
    # Wall capacity constraints
    for i in range(n):
        wall_cap = np.min(wall_dists[i]) - radii[i]
        if wall_cap < 0:
            wall_cap = 0
        A_ub.append(-np.eye(1, n, i, dtype=float).flatten())
        b_ub.append(-wall_cap)
    # Pairwise capacity constraints
    for i in range(n):
        for j in range(i + 1, n):
            pair_cap = d_ij[i, j - i - 1] - (radii[i] + radii[j])
            A_ub.append(np.zeros(n))
            A_ub[-1][i] = 1
            A_ub[-1][j] = 1
            b_ub.append(pair_cap)
    # Solve LP
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'presolve': True, 'time_limit': 5})
    except:
        # fallback if highs fails
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs-ds', options={'presolve': True, 'time_limit': 5})
    if not res.success:
        return np.zeros(n)
    return res.x

def apply_increment(radii, Δr):
    return radii + Δr

def push_feasibility(centers, radii):
    n = centers.shape[0]
    # Init new centers and radii
    new_centers = centers.copy()
    new_radii = radii.copy()
    # Store which circles touch each wall
    wall_contacts = {}
    # Process wall constraints first
    for i in range(n):
        x, y = new_centers[i]
        r = new_radii[i]
        # Left wall
        if x - r < 0:
            new_radii[i] = x
        # Right wall
        if x + r > 1:
            new_radii[i] = 1 - x
        # Bottom wall
        if y - r < 0:
            new_radii[i] = y
        # Top wall
        if y + r > 1:
            new_radii[i] = 1 - y
        # Record which walls contacts
        if new_radii[i] == x:
            wall_contacts[i] = 'left'
        elif new_radii[i] == 1 - x:
            wall_contacts[i] = 'right'
        elif new_radii[i] == y:
            wall_contacts[i] = 'bottom'
        elif new_radii[i] == 1 - y:
            wall_contacts[i] = 'top'
    # Then return after clamping
    # Reclamp the centers to ensure they are within bounds
    new_centers = np.clip(new_centers, 0, 1)
    # Now do pairwise push for overlaps
    # For each pair
    changed = True
    while changed:
        changed = False
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((new_centers[i] - new_centers[j]) ** 2))
                sum_r = new_radii[i] + new_radii[j]
                if dist < sum_r - 1e-12:
                    changed = True
                    overlap = sum_r - dist
                    dx = new_centers[i] - new_centers[j]
                    dx_norm = np.sqrt(np.sum(dx ** 2))
                    if dx_norm > 1e-15:
                        dx = dx / dx_norm
                        # Push both in opposite directions
                        push_amount = overlap / 2
                        new_centers[i] -= push_amount * dx
                        new_centers[j] += push_amount * dx
                        # Clamp to bounds if needed
                        new_centers[i] = np.clip(new_centers[i], 0, 1)
                        new_centers[j] = np.clip(new_centers[j], 0, 1)
    # Renormalize radii to be within bounds
    for i in range(n):
        x, y = new_centers[i]
        if i in wall_contacts:
            # Already clamped properly
            pass
        else:
            # Balls should have their radii inwardmax capped
            new_radii[i] = min(new_radii[i], x, 1 - x, y, 1 - y)
    # Reclamp any too-large radii (necessary after geometric adjustments)
    for i in range(n):
        x, y = new_centers[i]
        max_rad = min(x, 1 - x, y, 1 - y)
        new_radii[i] = min(new_radii[i], max_rad)
    return new_centers, new_radii

def update_graph(graph, new_edges, wall_contacts):
    for e in new_edges:
        graph[tuple(e)] = True
    return graph

def run_iterations(centers, radii, max_iter=200, ε=1e-5, α=0.5):
    graph = {}
    for i in range(len(centers)):
        wall_contacts = {}
    for _ in range(max_iter):
        # Build constraints
        consts = build_lp_constants(centers, radii)
        Δr = solve_radius_lp(consts, radii, graph, α)
        # Apply increment
        new_radii = apply_increment(radii, Δr)
        # Determine total increase
        delta_sum = np.sum(Δr)
        if delta_sum < ε:
            break
        # Temporarily update with inc radii for push
        temp_radii = new_radii
        centers, radii = push_feasibility(centers, temp_radii)
        # Update the graph, but skip for now as we simplify
    return centers, radii

def final_refinement(centers, radii):
    # Final adjustment
    consts = build_lp_constants(centers, radii)
    Δr = solve_radius_lp(consts, radii, {}, 0.5)
    final_radii = apply_increment(radii, Δr)
    # One more push;
    centers, final_radii = push_feasibility(centers, final_radii)
    # Optional final substantial clamping
    n = len(final_radii)
    for i in range(n):
        x, y = centers[i]
        final_radii[i] = min(final_radii[i], x, 1 - x, y, 1 - y)
    # Final pairwise checks to correct very slight overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            sum_r = final_radii[i] + final_radii[j]
            if dist < sum_r - 1e-13:
                overlap = sum_r - dist
                final_radii[i] -= overlap / 2
                final_radii[j] -= overlap / 2
    return centers, final_radii

def run_packing():
    centers = initial_hex_centers(26, 0.12)
    radii = initial_radii(26, 0.005)
    # Run iterations
    centers, radii = run_iterations(centers, radii, max_iter=150, ε=1e-5, α=0.5)
    # Final refinement
    centers, radii = final_refinement(centers, radii)
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
