import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
from itertools import combinations

def init_candidate_grid(R, Δ):
    """Initialize grid of candidate (center, radius) tuples."""
    candidates = []
    for r in R:
        # Grid spacing to ensure non-overlap for same radius
        s = r * np.sqrt(2)
        if s > 1:
            continue  # Impossible to fit even one circle of this size
        # Generate candidates inside feasible region [r, 1-r]^2
        x_coords = np.arange(r, 1 - r + 1e-10, s)
        y_coords = np.arange(r, 1 - r + 1e-10, s)
        for x in x_coords:
            for y in y_coords:
                candidates.append((x, y, r))
    return candidates

def build_overlap_matrix(candidates):
    """
    Return a boolean matrix indicating which pairs of candidates overlap.
    """
    n = len(candidates)
    overlap = np.zeros((n, n), dtype=bool)
    # Extract centers and radii
    centers = np.array([[c[0], c[1]] for c in candidates])
    radii = np.array([c[2] for c in candidates])
    # Compute pairwise distances
    dists = cdist(centers, centers)
    # Overlap when dist <= r_i + r_j
    overlap_threshold = radii[:, None] + radii[None, :]
    overlap = dists < overlap_threshold - 1e-12
    np.fill_diagonal(overlap, False)
    return overlap

def solve_discrete_bip(candidates, overlap_matrix, target_sum):
    """
    Solve a (simplified) BIP to select 26 non-overlapping candidates.
    Approximates the solution as a greedy heuristic with fallback logic.
    """
    n = len(candidates)
    # Hacky direct approach due to constraints.
    # Start with largest radii and then greedily assign non-overlapping ones.
    radii = np.array([c[2] for c in candidates])
    idxs = np.argsort(radii)[::-1]
    selected_mask = np.zeros(n, dtype=bool)
    selected_indices = []
    total_sum = 0.0

    for i in idxs:
        if len(selected_indices) >= 26:
            break
        if selected_mask[i]:
            continue
        # Check if it overlaps with already selected
        overlaps = overlap_matrix[i, selected_mask]
        if overlaps.any():
            continue
        selected_mask[i] = True
        selected_indices.append(i)
        total_sum += radii[i]

    # If we couldn't find 26 good candidates, we're losing.
    # Retain the top 26 with non-overlap, or just cut to 26.
    if len(selected_indices) < 26:
        # Greedy but better: add more to get 26 if we can without errors.
        candidates_left = []
        for i in range(n):
            if not selected_mask[i]:
                candidates_left.append(i)
        selected_indices.extend(candidates_left[:26 - len(selected_indices)])

    chosen_candidates = [candidates[i] for i in selected_indices]
    centers = np.array([[c[0], c[1]] for c in chosen_candidates])
    radii = np.array([c[2] for c in chosen_candidates])
    return centers, radii

def max_radii_fixed_centres(centers, radii_initial):
    """For fixed centers, compute max radius for each circle."""
    n = centers.shape[0]
    radii = radii_initial.copy()
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Maximal radius at this position
        r_max = min(x, 1 - x, y, 1 - y)
        # Check overlap constraints with all others
        for j in range(n):
            if i == j:
                continue
            xj, yj = centers[j]
            rj = radii[j]
            dist = np.sqrt((x - xj)**2 + (y - yj)**2)
            rmax_allowed = dist - rj
            r_max = min(r_max, rmax_allowed)
        radii[i] = max(0.0, r_max)
    return radii

def nlp_refine(centers_initial, radii_initial):
    """Run continuous refinement using trust-constr."""
    n = centers_initial.shape[0]
    x0 = np.concatenate([centers_initial.flat, radii_initial])
    bounds = []
    # Bounds for x_i, y_i: [r_i, 1 - r_i] for each
    # So we allow center to move up to the wall.
    for i in range(n):
        bounds.append((radii_initial[i], 1 - radii_initial[i]))  # x_i
        bounds.append((radii_initial[i], 1 - radii_initial[i]))  # y_i
    for i in range(n):
        bounds.append((0.0, None))  # r_i >= 0

    def obj_func(v):
        # v is [x1, y1, ..., x26, y26, r1, ..., r26]
        # Minimize -sum(r_i)
        return -sum(v[-n:])

    def jac_obj(v):
        # Jacobian of obj_func
        j = np.zeros_like(v)
        j[-n:] = -1.0
        return j

    def constraints_func(v):
        # Return array of constraint values <= 0.
        # Precompute centers and radii
        centers = v[:2*n].reshape(-1, 2)
        radii = v[2*n:]
        out = []
        # Wall constraints for each circle (i)
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            # r - x <= 0
            out.append(r - x)
            # r + x - 1 <= 0
            out.append(r + x - 1)
            # r - y <= 0
            out.append(r - y)
            # r + y - 1 <= 0
            out.append(r + y - 1)

        # Non-overlap constraints for all pairs (i < j)
        for i, j in combinations(range(n), 2):
            x1, y1 = centers[i]
            x2, y2 = centers[j]
            r1 = radii[i]
            r2 = radii[j]
            dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
            out.append(r1 + r2 - dist)

        return np.array(out)

    def jac_constraints(v):
        # Jacobian of all constraints
        n = centers_initial.shape[0]
        centers = v[:2*n].reshape(-1, 2)
        radii = v[2*n:]
        # Number of constraints:
        # 4*n (walls)
        # C(26,2) = 325 (non-overlaps)
        n_cons = 4 * n + n * (n - 1) // 2

        J = np.zeros((n_cons, 3 * n))
        offset = 0

        # Wall constraints: 4 per circle
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            # (r - x) <= 0
            J[offset][2*i] = 1.0
            J[offset][2*n + i] = -1.0
            offset += 1
            # (r + x - 1) <= 0
            J[offset][2*i] = 1.0
            J[offset][2*n + i] = 1.0
            offset += 1
            # (r - y) <= 0
            J[offset][2*i + 1] = 1.0
            J[offset][2*n + i] = -1.0
            offset += 1
            # (r + y - 1) <= 0
            J[offset][2*i + 1] = 1.0
            J[offset][2*n + i] = 1.0
            offset += 1

        # Non-overlap constraints
        for i, j in combinations(range(n), 2):
            x1, y1 = centers[i]
            x2, y2 = centers[j]
            r1 = radii[i]
            r2 = radii[j]
            dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
            if dist < 1e-12:
                continue  # At least avoid divide-by-zero paths with very close centers

            # Gradient w.r.t. centers
            dx = (x1 - x2) / dist
            dy = (y1 - y2) / dist
            J[offset][2*i] = -1.0 * dx
            J[offset][2*i + 1] = -1.0 * dy
            J[offset][2*j] = 1.0 * dx
            J[offset][2*j + 1] = 1.0 * dy

            # Gradient w.r.t. radii
            J[offset][2*n + i] = -1.0
            J[offset][2*n + j] = -1.0
            offset += 1

        return J

    # Optimization
    try:
        # Solve the NLP
        res = minimize(
            fun=obj_func,
            x0=x0,
            method='trust-constr',
            jac=jac_obj,
            constraints={'type': 'ineq', 'fun': constraints_func, 'jac': jac_constraints},
            bounds=bounds,
            options={'maxiter': 300, 'gtol': 1e-9, 'xtol': 1e-9}
        )

        # Return updated values
        values = res.x
        centers = values[:2*n].reshape(-1, 2)
        radii = values[2*n:]
        return centers, radii

    except Exception:
        return centers_initial, radii_initial

def run_packing():
    np.random.seed(42)

    # Discrete phase 1: Generate candidate grid
    R = [0.025, 0.035, 0.045, 0.055, 0.065, 0.075, 0.085, 0.095, 0.105, 0.115]
    candidates = init_candidate_grid(R, 0.01)
    n_candidates = len(candidates)

    # Compute overlap matrix in discrete space
    overlap_matrix = build_overlap_matrix(candidates)

    # Exact discretized solution
    centers_discrete, radii_discrete = solve_discrete_bip(candidates, overlap_matrix, 2.636)

    # Continuous refinement using trust-constr
    centers_refined, radii_refined = nlp_refine(centers_discrete, radii_discrete)

    # Final projection to exact max radii on fixed centers
    radii_final = max_radii_fixed_centres(centers_refined, radii_refined)

    # Sum of radii
    sum_radii = np.sum(radii_final)

    return centers_refined, radii_final, sum_radii
