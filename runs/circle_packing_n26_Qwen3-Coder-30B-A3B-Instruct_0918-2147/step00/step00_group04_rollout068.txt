import numpy as np
import cvxpy as cp
from scipy.spatial.distance import cdist

def build_leaf_problem(cell, k):
    """Build an SOCP for a leaf cell with k circles."""
    x_min, y_min, x_max, y_max = cell
    width = x_max - x_min
    height = y_max - y_min

    # Variables for k circles
    centers = cp.Variable((k, 2))
    radii = cp.Variable(k)

    # Constraints for boundary
    constraints = []
    for i in range(k):
        constraints += [
            centers[i, 0] - radii[i] >= x_min,
            centers[i, 0] + radii[i] <= x_max,
            centers[i, 1] - radii[i] >= y_min,
            centers[i, 1] + radii[i] <= y_max
        ]

    # Non-overlap constraints as SOCP
    for i in range(k):
        for j in range(i + 1, k):
            d = cp.Variable()
            constraints += [
                d >= cp.norm(centers[i] - centers[j]),
                d >= radii[i] + radii[j]
            ]

    # Objective to maximize sum of radii
    objective = cp.Maximize(cp.sum(radii))

    return cp.Problem(objective, constraints), centers, radii

def solve_leaf(cell, k):
    """Solve the exact SOCP for the leaf cell with k circles."""
    problem, centers, radii = build_leaf_problem(cell, k)
    problem.solve(solver=cp.ECOS, abstol=1e-12, reltol=1e-12)
    return centers.value, radii.value

def merge_clusters(cluster_a, cluster_b, shared_interface):
    """Merge clusters with adjustment along shared boundary."""
    joints = shared_interface
    all_centers = np.vstack([cluster_a['centers'], cluster_b['centers']])
    all_radii = np.hstack([cluster_a['radii'], cluster_b['radii']])
    all_k = len(all_centers)

    # Create variables
    centers = cp.Variable((all_k, 2))
    radii = cp.Variable(all_k)

    # Constraints from boundaries
    constraints = []
    for i in range(all_k):
        x_min, y_min, x_max, y_max = (0, 0, 1, 1)
        constraints += [
            centers[i, 0] - radii[i] >= x_min,
            centers[i, 0] + radii[i] <= x_max,
            centers[i, 1] - radii[i] >= y_min,
            centers[i, 1] + radii[i] <= y_max
        ]

    # Contact constraints (fixed from original solution)
    for i in range(len(cluster_a['centers'])):
        for j in range(len(cluster_a['centers']), len(cluster_a['centers']) + len(cluster_b['centers'])):
            # Only enforce constraints between clusters
            d = cp.Variable()
            constraints += [
                d >= cp.norm(centers[i] - centers[j]),
                d >= radii[i] + radii[j]
            ]

    # Objective
    objective = cp.Maximize(cp.sum(radii))
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.ECOS, abstol=1e-12, reltol=1e-12)
    return centers.value, radii.value

def global_polish(centers, radii, steps=2):
    """Apply gradient ascent with log barrier."""
    def barrier_objective(c_arr, r_arr):
        obj = np.sum(r_arr)
        for i in range(len(c_arr)):
            for j in range(i + 1, len(c_arr)):
                dist = np.linalg.norm(c_arr[i] - c_arr[j])
                obj -= 1e-4 * np.log(max(dist - r_arr[i] - r_arr[j], 1e-15))
        for i in range(len(c_arr)):
            x, y = c_arr[i]
            r = r_arr[i]
            obj -= 1e-4 * (
                np.log(max(x - r, 1e-15)) +
                np.log(max(1 - x - r, 1e-15)) +
                np.log(max(y - r, 1e-15)) +
                np.log(max(1 - y - r, 1e-15))
            )
        return -obj

    # Vectors for gradients (simple estimator)
    def grad_centers_r(c_arr, r_arr):
        grad_c = np.zeros_like(c_arr)
        grad_r = np.zeros_like(r_arr)
        for i in range(len(c_arr)):
            # Gradient with respect to centers
            for j in range(len(c_arr)):
                if i != j:
                    d_vec = c_arr[i] - c_arr[j]
                    dist = np.linalg.norm(d_vec) + 1e-15
                    unit = d_vec / dist
                    if dist > r_arr[i] + r_arr[j]:
                        grad_c[i] += -1e-4 * unit * 1 / dist
                        grad_c[j] += 1e-4 * unit * 1 / dist
            # Gradient with respect to radii
            for j in range(len(c_arr)):
                if i != j:
                    dist = np.linalg.norm(c_arr[i] - c_arr[j])
                    if dist > r_arr[i] + r_arr[j]:
                        grad_r[i] += -1e-4 * 1 / (dist - r_arr[i] - r_arr[j])
                        grad_r[j] += -1e-4 * 1 / (dist - r_arr[i] - r_arr[j])
            # Boundary gradients
            x, y = c_arr[i]
            r = r_arr[i]
            grad_r[i] += 1e-4 * (
                1 / (x - r) +
                1 / (1 - x - r) +
                1 / (y - r) +
                1 / (1 - y - r)
            )
        return grad_c, grad_r

    for step in range(steps):
        grad_c, grad_r = grad_centers_r(centers, radii)
        eta = 0.02
        centers += eta * grad_c
        radii += eta * grad_r
        # Project back to feasibility
        for i in range(len(centers)):
            x, y = centers[i]
            r = radii[i]
            x = np.clip(x, r, 1 - r)
            y = np.clip(y, r, 1 - r)
            r = max(r, 1e-15)
            centers[i] = [x, y]
            radii[i] = r

    return centers, radii

def run_packing():

    # Merge clusters in binary tree fashion
    def recursive_merge(clusters):

                    'active': None

            else:
                merged_cluster.append(clusters[i])
        return recursive_merge(merged_cluster)

    final_cluster = recursive_merge(clusters)
    centers = final_cluster['centers']
