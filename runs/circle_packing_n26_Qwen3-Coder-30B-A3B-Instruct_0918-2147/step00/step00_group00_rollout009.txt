import numpy as np
import random

def make_hex_seed(n, margin):
    # Generate hexagonal pattern
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))
    
    centers = []
    radii = []
    
    for i in range(rows):
        for j in range(cols):
            if len(centers) >= n:
                break
            x = margin + (j + 0.5 * (i % 2)) * (1 - 2 * margin) / (cols - 0.5 * (cols > 1))
            y = margin + i * (1 - 2 * margin) / (rows - 1)
            centers.append([x, y])
            # Initial radius based on proximity to walls
            r = min(x, 1 - x, y, 1 - y) * 0.4
            radii.append(max(r, 0.015))  # Avoid zero/very small radii
        if len(centers) >= n:
            break
    
    centers = np.array(centers[:n])
    radii = np.array(radii[:n])
    return centers, radii

def max_feasible_radius(i, centers, radii):
    # Calculate max feasible radius for circle i
    x, y = centers[i]
    r_i = radii[i]
    
    # Wall distances
    d_left = x
    d_right = 1 - x
    d_bottom = y
    d_top = 1 - y
    
    # Neighbour distances
    d_neighbour = float('inf')
    for j in range(centers.shape[0]):
        if i != j:
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            d_neighbour = min(d_neighbour, dist - radii[j])
    
    # The limiting factor
    max_r = min(d_left, d_right, d_bottom, d_top, d_neighbour)
    return max_r

def inflate_radii(centers, radii, alpha):
    # Add global step to all radii and clamp with local feasibility
    old_sum = np.sum(radii)
    mean_r = np.mean(radii)
    delta = alpha * mean_r
    
    new_radii = radii + delta
    new_sum = old_sum + len(radii) * delta
    
    # Clamp each to local maximum
    for i in range(len(centers)):
        max_r = max_feasible_radius(i, centers, new_radii)
        new_radii[i] = min(new_radii[i], max_r)
    
    new_sum = np.sum(new_radii)
    return new_radii, new_sum - old_sum

def anneal_centers(centers, radii, temp0, gamma, sigma0, steps):
    # Simulated Annealing for center positions only, not changing radii
    
    # Keep a copy of current state
    current_centers = centers.copy()
    current_radii = radii.copy()
    best_centers = current_centers.copy()
    best_radii = current_radii.copy()
    best_sum = np.sum(current_radii)
    
    temp = temp0
    sigma = sigma0
    
    # Initial total sum
    last_sum = np.sum(current_radii)
    
    for step in range(steps):
        # Occasionally do a global shift to escape locals
        if step % 50 == 0:
            shift = np.random.normal(0, 0.005, 2)
            current_centers += shift
        
        # Select random circle
        i = random.randint(0, len(current_centers) - 1)
        
        # Sample displacement
        delta = np.random.normal(0, sigma, 2)
        new_center = current_centers[i] + delta
        
        # Reflect if out of bounds
        new_center[0] = np.clip(new_center[0], 0, 1)
        new_center[1] = np.clip(new_center[1], 0, 1)
        
        # Determine new legal radius for circle i
        old_r_i = current_radii[i]
        current_centers[i] = new_center  # temporarily set the new center
        new_r_i = max_feasible_radius(i, current_centers, current_radii)
        current_centers[i] = new_center  # put back
        diff = new_r_i - old_r_i
        
        # Accept move (according to Metropolis criterion)
        if diff > 0 or np.random.rand() < np.exp(diff / temp):
            current_radii[i] = new_r_i
            sum_r = np.sum(current_radii)
            if sum_r > best_sum:
                best_centers = current_centers.copy()
                best_radii = current_radii.copy()
                best_sum = sum_r
        else:
            # Leave unchanged
            pass
        
        # Update temperature and sigma
        temp *= gamma
        sigma = sigma0 * temp / temp0
        
        # Occasionally restart to avoid local minima trapping
        if step % 1000 == 0 and step > 0:
            temp = temp0
    
    return best_centers, best_radii, best_sum

def mixed_cycle(centers, radii, params):
    # Run mixture of inflation and SA
    alpha = params["alpha"]
    min_delta = params["min_delta"]
    convergence_steps = params["convergence_steps"]
    sa_steps = params["sa_steps"]
    sa_temp0 = params["sa_temp0"]
    sa_gamma = params["sa_gamma"]
    sa_sigma0 = params["sa_sigma0"]
    max_cycles = params["max_cycles"]
    
    current_centers = centers.copy()
    current_radii = radii.copy()
    current_sum = np.sum(current_radii)
    
    inactive_count = 0
    full_cycles = 0
    last_sum = float('-inf')
    
    while full_cycles < max_cycles:
        # Inflation
        while True:
            new_radii, delta = inflate_radii(current_centers, current_radii, alpha)
            current_radii = new_radii
            if delta < min_delta or np.abs(delta) < 1e-12:
                inactive_count += 1
            else:
                inactive_count = 0
            current_sum = np.sum(current_radii)
            if inactive_count >= convergence_steps:
                break
        # Increase alpha a bit for next inflation phase
        alpha *= 0.95  # decay factor
        
        # SA to optimize centers
        current_centers, current_radii, new_sum = anneal_centers(
            current_centers, current_radii,
            sa_temp0, sa_gamma, sa_sigma0, sa_steps)
        current_sum = new_sum
        
        # Check for cycle termination
        if np.abs(current_sum - last_sum) < 1e-6:
            full_cycles += 1
        else:
            full_cycles = 0
        last_sum = current_sum
        
    return current_centers, current_radii

def polish_radii(centers, radii, beta, passes):
    # Exactly inflate and resolve overlaps
    
    new_radii = radii.copy()
    for _ in range(passes):
        modified = True
        while modified:
            modified = False
            for i in range(len(centers)):
                # Compute gaps
                x, y = centers[i]
                r = new_radii[i]
                g_wall = min(x - r, 1 - x - r, y - r, 1 - y - r)
                
                g_nei = float('inf')
                for j in range(len(centers)):
                    if i != j:
                        dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                        g = dist - (r + new_radii[j])
                        g_nei = min(g_nei, g)
                
                g = min(g_wall, g_nei)
                if g > 0:
                    new_r = min(new_radii[i] + beta * g, max_feasible_radius(i, centers, new_radii))
                    if new_r > new_radii[i]:
                        new_radii[i] = new_r
                        modified = True
                        
            # After inflation, fix any overlaps
            for i in range(len(centers)):
                for j in range(i+1, len(centers)):
                    dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                    overlap = (new_radii[i] + new_radii[j]) - dist
                    if overlap > 0:
                        # Shrink both
                        shrink = overlap / 2
                        new_radii[i] = max(0, new_radii[i] - shrink)
                        new_radii[j] = max(0, new_radii[j] - shrink)
                        modified = True
    return new_radii

def run_packing():

    }
    
    # Run mixed cycles
    final_centers, final_radii = mixed_cycle(initial_centers, initial_radii, params)
    
    # Polish radii
    final_radii = polish_radii(final_centers, final_radii, 0.6, 3)
