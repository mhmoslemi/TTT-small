import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import math

def generate_lattice(s):
    """Generate hexagonal lattice of candidate center points."""
    sqrt3 = math.sqrt(3)
    rows = int(1.0 / (s * sqrt3 / 2)) + 2
    cols = int(1.0 / s) + 2
    
    points = []
    for r in range(rows):
        y = r * s * sqrt3 / 2
        if r % 2 == 0:
            start_col = 0
        else:
            start_col = 0.5
            
        for c in range(int(cols)):
            x = (start_col + c) * s
            if s/2 <= x <= 1 - s/2 and s/2 <= y <= 1 - s/2:
                points.append([x, y])
    
    return np.array(points)

def build_milp(P):
    """Build the MILP model for circle packing."""
    K = len(P)
    # Variables
    z = cp.Variable(K, boolean=True)
    r = cp.Variable(K, nonneg=True)
    
    # Distances between all pairs of points
    distances = np.zeros((K, K))
    for i in range(K):
        for j in range(i+1, K):
            dist = np.linalg.norm(P[i] - P[j])
            distances[i, j] = distances[j, i] = dist
    
    # Wall distances for each point
    wall_distances = np.zeros(K)
    for i in range(K):
        x, y = P[i]
        wall_distances[i] = min(x, 1-x, y, 1-y)
    
    # Constraints
    constraints = [
        cp.sum(z) == 26,
    ]
    
    # Link radii to selection
    for i in range(K):
        constraints.append(r[i] <= wall_distances[i] * z[i])
        constraints.append(r[i] >= 0)
        
    # Non-overlap constraints (big-M formulation)
    M = 1.0
    for i in range(K):
        for j in range(i+1, K):
            constraints.append(distances[i, j] >= r[i] + r[j] - M*(2 - z[i] - z[j]))
    
    # Objective
    objective = cp.Maximize(cp.sum(r))
    
    def solve_with_selection(fixed_z):
        # Update the binary variables to fixed values
        for i in range(K):
            if fixed_z[i] is not None:
                z[i].fix(fixed_z[i])
        
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        
        sol_r = r.value.copy() if r.value is not None else np.zeros(K)
        bound = prob.value if prob.value is not None else -np.inf
        
        # Reset z variables
        for i in range(K):
            z[i].unfix()
            if fixed_z[i] is not None:
                # Re-apply the fix
                z[i].fix(fixed_z[i])
        
        return bound, sol_r
    
    return solve_with_selection

def branch_and_bound(milp_solver, node_limit=20000):
    """Custom branch-and-bound solver."""
    best_objective = -np.inf
    best_z = None
    best_r = None
    
    # Priority queue: list of (lb, fixed, remaining_indices)
    queue = [(float('-inf'), [None] * 26, list(range(26)))]
    
    node_count = 0
    while queue and node_count < node_limit:
        queue.sort(key=lambda x: x[0], reverse=True)
        lb, fixed, free = queue.pop(0)
        
        if lb <= best_objective:
            continue
        
        # Solve LP relaxation
        if not free:
            # All binary variables determined
            bound, sol_r = milp_solver(fixed)
            if bound > best_objective:
                best_objective = bound
                best_z = fixed[:]
                best_r = sol_r[:]
            continue
        
        # Pick branching variable
        branching_idx = max(free, key=lambda i: abs(fixed[i] - 0.5) if fixed[i] is not None else 0.5)
        
        # Branch on 0
        fixed_0 = fixed[:]
        fixed_0[branching_idx] = 0
        bound_0, _ = milp_solver(fixed_0)
        if bound_0 > best_objective:
            queue.append((bound_0, fixed_0, [i for i in free if i != branching_idx]))
        
        # Branch on 1
        fixed_1 = fixed[:]
        fixed_1[branching_idx] = 1
        bound_1, _ = milp_solver(fixed_1)
        if bound_1 > best_objective:
            queue.append((bound_1, fixed_1, [i for i in free if i != branching_idx]))
            
        node_count += 1
        
    return best_z, best_r

def refine_solution(centers, radii):
    """Refine the solution with continuous optimization."""
    def obj_func(x):
        centers_r = x[:52].reshape((-1, 2))
        radii_r = x[52:]
        total = np.sum(radii_r)
        return -total  # to maximize

    def constraint_func(x):
        centers_r = x[:52].reshape((-1, 2))
        radii_r = x[52:]
        constraints = []
        for i in range(26):
            constraints.append(centers_r[i][0] - radii_r[i])  # x - r >= 0
            constraints.append(1 - centers_r[i][0] - radii_r[i])  # 1 - x - r >= 0
            constraints.append(centers_r[i][1] - radii_r[i])  # y - r >= 0
            constraints.append(1 - centers_r[i][1] - radii_r[i])  # 1 - y - r >= 0
        for i in range(26):
            for j in range(i+1, 26):
                dist_sq = (centers_r[i][0] - centers_r[j][0])**2 + (centers_r[i][1] - centers_r[j][1])**2
                constraints.append(-(dist_sq - (radii_r[i] + radii_r[j])**2))  # dist_sq >= (r_i + r_j)^2
        return np.array(constraints)

    x0 = np.concatenate([centers.flatten(), radii])
    
    # Add tiny penalties to avoid numerical issues
    def f_with_penalty(x):
        obj = obj_func(x)
        return obj - 1e-4 * np.sum(np.log(np.maximum(radii, 1e-12)))

    # Use SLSQP as it's suitable for nonlinear constraints
    options = {"maxiter": 2000, "ftol": 1e-9}
    res = minimize(f_with_penalty, x0, method='SLSQP', constraints=[{"type": "ineq", "fun": constraint_func}], options=options)

    refined_centers = res.x[:52].reshape((-1, 2))
    refined_radii = np.maximum(res.x[52:], 0)
    sum_radii = np.sum(refined_radii)
    
    return refined_centers, refined_radii, sum_radii

def run_packing():
    """Main function to run the packing algorithm."""
    # Stage 1: Generate lattice
    lattice = generate_lattice(0.13)
    K = len(lattice)
    
    # Stage 2: Build MILP model
    milp_func = build_milp(lattice)
    
    # Stage 3: Branch-and-bound to select best 26 centers
    best_z, best_r = branch_and_bound(milp_func)
    
    # Stage 4: Extract 26 centers
    selected_indices = [i for i in range(len(best_z)) if best_z[i] == 1]
    centers = lattice[selected_indices]
    radii = best_r[selected_indices]
    
    # Stage 5: Local continuous refinement
    refined_centers, refined_radii, sum_radii = refine_solution(centers, radii)
    
    return refined_centers, refined_radii, sum_radii
