import numpy as np

def hex_lattice(d):
    """Generate hexagonal lattice points within unit square."""
    rows = []
    y = 0
    dy = d * np.sqrt(3) / 2
    while y <= 1:
        row = []
        x = 0
        dx = d
        offset = (len(rows) % 2) * dx / 2
        while x <= 1:
            row.append([x + offset, y])
            x += dx
        rows.append(row)
        y += dy
    points = []
    for row in rows:
        points.extend(row)
    return np.array(points)

def add_border_circles(points, d):
    """Add 6 border-enhanced circles to the existing points."""
    r = d / 4
    border_points = [
        [r, r],                    # bottom-left
        [1 - r, r],                # bottom-right
        [r, 1 - r],                # top-left
        [1 - r, 1 - r],            # top-right
        [0.5, r],                  # mid-left
        [0.5, 1 - r]               # mid-right
    ]
    return np.vstack([points, border_points])

def initial_radii(points):
    """Compute the initial feasible uniform radius for all circles."""
    wall_distances = np.minimum(
        points[:, 0], 1 - points[:, 0],
        np.minimum(points[:, 1], 1 - points[:, 1])
    )
    n = points.shape[0]
    min_distances = np.full(n, np.inf)
    for i in range(n):
        dists = np.linalg.norm(points[i] - points, axis=1)
        dists[i] = np.inf
        min_distances[i] = np.min(dists)
    safe_radii = np.minimum(wall_distances, min_distances / 2)
    return np.min(safe_radii)

def inflate_radii(centers, radii):
    """Inflate all radii as much as possible while maintaining feasibility."""
    n = centers.shape[0]
    new_radii = radii.copy()
    # Compute all pairwise distances
    distances = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    for i in range(n):
        # Wall constraints
        wall_caps = [
            centers[i, 0], 1 - centers[i, 0],
            centers[i, 1], 1 - centers[i, 1]
        ]
        # Pairwise constraints
        pairwise_caps = []
        for j in range(n):
            if i != j:
                pairwise_caps.append(distances[i, j] - radii[j])
        caps = wall_caps + pairwise_caps
        new_radius = min(caps)
        if new_radius > radii[i] + 1e-6:
            new_radii[i] = new_radius
    return new_radii

def relax_centers(centers, radii, steps=10):
    """Apply force-based relaxation on circle centers."""
    n = centers.shape[0]
    # Force parameters
    kappa = 0.02
    lambda_wall = 0.01
    gamma = 0.02
    beta = 0.02
    alpha = 0.5

    for _ in range(steps):
        forces = np.zeros_like(centers)
        # Compute pairwise forces
        distances = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
        for i in range(n):
            for j in range(n):
                if i != j:
                    dist = distances[i, j]
                    if dist > 0:
                        delta = radii[i] + radii[j] - dist
                        if delta < 0:  # Overlap
                            # Skip actual negative forces for invalid ones
                            continue
                        # Repulsive force
                        if dist < beta:
                            force = kappa * max(0, beta - dist) * (centers[i] - centers[j]) / dist
                            forces[i] += force
        # Wall forces
        for i in range(n):
            fx = 0.0
            fy = 0.0
            # Left wall
            if centers[i, 0] - radii[i] < gamma:
                fx += lambda_wall * max(0, gamma - (centers[i, 0] - radii[i]))
            # Right wall
            if 1 - (centers[i, 0] + radii[i]) < gamma:
                fx -= lambda_wall * max(0, gamma - (1 - (centers[i, 0] + radii[i])))
            # Bottom wall
            if centers[i, 1] - radii[i] < gamma:
                fy += lambda_wall * max(0, gamma - (centers[i, 1] - radii[i]))
            # Top wall
            if 1 - (centers[i, 1] + radii[i]) < gamma:
                fy -= lambda_wall * max(0, gamma - (1 - (centers[i, 1] + radii[i])))
            forces[i] += np.array([fx, fy])
        # Apply forces
        centers += alpha * forces
        # Clamp to avoid violating constraints due to numerical inaccuracy
        centers = np.clip(centers, radii, 1 - radii)
    return centers
