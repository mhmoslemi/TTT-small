import numpy as np
from scipy.optimize import minimize

def hex_lattice_points(n, d):
    """Generate at least n points in a hexagonal lattice inside [0,1]x[0,1]."""
    sqrt3 = np.sqrt(3)
    y_spacing = sqrt3 * d / 2
    y = 0.0
    row = 0
    coords = []
    while len(coords) < n:
        x_offset = (row % 2) * d / 2
        x = x_offset
        while x <= 1.0:
            coords.append((x, y))
            x += d
        y += y_spacing
        row += 1
        if y > 1.0:
            break
    return np.array(coords[:n])

    x = np.clip(x, d/2, 1.0 - d/2)
    y = np.clip(y, d/2, 1.0 - d/2)
    # Compute feasible radius (limited by distance to closest boundary)
    r_wall = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    r = np.min(r_wall)
    # Ensure all radii are at least a small positive value
    r = max(r, 1e-6)

    vars = np.zeros(3 * 26)
    for i in range(26):
        vars[3*i] = x[i]
        vars[3*i+1] = y[i]
        vars[3*i+2] = r
    return vars

    """Minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def wall_constraints(vars):

def nonoverlap_constraints(vars):

            radius_sum = r_i + r_j

            cons.append(dist_sq - radius_sum * radius_sum)
    return cons
