import numpy as np
from scipy.optimize import linprog
from itertools import combinations

def generate_hex_lattice(h):
    sqrt3 = np.sqrt(3)
    cx = []
    cy = []
    rx = []
    j_start = 0
    if h * sqrt3 / 2 > 1:
        raise ValueError("Lattice spacing too large for unit square")
    j_max = int(np.ceil(1.0 / (h * sqrt3 / 2))) + 1
    for j in range(j_start, j_max):
        y = j * h * sqrt3 / 2
        if y > 1:
            break
        i_start = 0 if j % 2 == 0 else 1
        i_max = int(np.ceil(1.0 / h)) + 1
        for i in range(i_start, i_max):
            x = i * h + (j % 2) * (h / 2)
            if x > 1:
                break
            cx.append(x)
            cy.append(y)
    centers = np.array([cx, cy]).T
    R_wall = np.minimum(np.minimum(centers[:, 0], 1 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1 - centers[:, 1]))
    return centers, R_wall

def build_neighbour_list(centers, max_dist):
    n = centers.shape[0]
    edges = []
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < max_dist:
                edges.append((i, j, dist))
    return edges

def solve_radius_lp(selected_indices, R_wall, neighbour_list):
    n_sel = len(selected_indices)
    if n_sel == 0:
        return np.array([]), True
    # Build LP coefficients and constraints
    # Objective: maximize sum of radii
    c = -np.ones(n_sel)
    # Inequality matrix A_ub * x <= b_ub
    A_ub = []
    b_ub = []
    # Wall constraints: 0 <= r_i <= R_wall[i]
    for i in range(n_sel):
        # r[i] >= 0: -r[i] <= 0
        A_ub.append([0.] * n_sel)
        A_ub[-1][i] = -1
        b_ub.append(0.)
        # r[i] <= R_wall[selected_indices[i]]
        A_ub.append([0.] * n_sel)
        A_ub[-1][i] = 1
        b_ub.append(R_wall[selected_indices[i]])
    # Separation constraints: r_i + r_j <= d_ij
    for i, j, dist in neighbour_list:
        if i in selected_indices and j in selected_indices:
            idx_i = selected_indices.index(i)
            idx_j = selected_indices.index(j)
            A_ub.append([0.] * n_sel)
            A_ub[-1][idx_i] = 1
            A_ub[-1][idx_j] = 1
            b_ub.append(dist)
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    # Solve using linprog (dual simplex)
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    if result.success:
        return -result.fun, True, result.x
    else:
        return -1e-10, False, None

def upper_bound(partial_set, remaining_indices, R_wall, selected_indices):
    ub = 0.
    for i in partial_set:
        ub += R_wall[i]
    # For remaining, greedily pick the best available
    if len(remaining_indices) + len(partial_set) < 26:
        # This shouldn't happen if called correctly
        return ub
    remaining_wall = [R_wall[i] for i in remaining_indices]
    remaining_wall.sort(reverse=True)
    n_still_needed = 26 - len(partial_set)
    ub += sum(remaining_wall[:n_still_needed])
    return ub

def branch_and_bound(centers, R_wall, neighbour_list, n_circles=26):
    best_sum = -np.inf
    best_indices = []
    best_radii = []

    def backtrack(current_indices, remaining_indices, lb):
        if len(current_indices) == n_circles:
            # Final check: solve exact LP
            total_radius, feasible, radii = solve_radius_lp(current_indices, R_wall, neighbour_list)
            if feasible:
                if total_radius > lb:
                    return total_radius, current_indices, radii
            return lb, None, None
        elif len(current_indices) > n_circles:
            return lb, None, None

        # Upper bound estimate
        ub = upper_bound(current_indices, remaining_indices, R_wall, current_indices)
        if ub <= lb:
            return lb, None, None

        # Select next point with largest wall radius among unselected
        best_idx = -1
        best_wall = -1
        for i in remaining_indices:
            if R_wall[i] > best_wall:
                best_wall = R_wall[i]
                best_idx = i
        if best_idx < 0:
            return lb, None, None
        # Try adding it
        new_indices = current_indices + [best_idx]
        new_remaining = [i for i in remaining_indices if i != best_idx]
        new_lb, new_indices_sol, new_radii_sol = backtrack(new_indices, new_remaining, lb)
        if new_lb > lb:
            lb = new_lb
            if new_indices_sol is not None:
                best_indices = new_indices_sol
                best_radii = new_radii_sol
        # Try not adding it
        for idx in remaining_indices:
            if idx == best_idx:
                continue
            new_indices = current_indices
            new_remaining = [i for i in remaining_indices if i != idx]
            new_lb, new_indices_sol, new_radii_sol = backtrack(new_indices, new_remaining, lb)
            if new_lb > lb:
                lb = new_lb
                if new_indices_sol is not None:
                    best_indices = new_indices_sol
                    best_radii = new_radii_sol
        return lb, best_indices, best_radii

    # Start from full list
    all_indices = list(range(len(centers)))
    final_lb, final_indices, final_radii = backtrack([], all_indices, best_sum)
    return final_indices, final_radii

def local_continuous_refinement(centers, radii, R_wall):
    from scipy.optimize import minimize

    def objective(radii_and_centers):
        return -np.sum(radii_and_centers[26:])

    def constraint_eq(radii_and_centers):
        # Circle containment
        result = []
        n_circles = len(radii_and_centers) // 2
        for i in range(n_circles):
            x = radii_and_centers[i]
            y = radii_and_centers[i + n_circles]
            r = radii_and_centers[i + 2 * n_circles]
            result.append(x - r)  # x - r >= 0
            result.append(1 - x - r)  # x + r <= 1
            result.append(y - r)  # y - r >= 0
            result.append(1 - y - r)  # y + r <= 1
        return np.array(result)

    def constraint_ineq(radii_and_centers):
        result = []
        n_circles = len(radii_and_centers) // 2
        # Non-overlapping constraints
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                x_i = radii_and_centers[i]
                y_i = radii_and_centers[i + n_circles]
                r_i = radii_and_centers[i + 2 * n_circles]
                x_j = radii_and_centers[j]
                y_j = radii_and_centers[j + n_circles]
                r_j = radii_and_centers[j + 2 * n_circles]
                dx = x_i - x_j
                dy = y_i - y_j
                dist = np.sqrt(dx * dx + dy * dy)
                # dist >= r_i + r_j implies dist - r_i - r_j >= 0
                result.append(dist - r_i - r_j)
        return np.array(result)

    n_vars = 26 * 3  # x[i], y[i], r[i]
    bounds = []
    # x, y in [0,1], r in [0, R_wall[i]]
    for i in range(26):
        bounds.append((0, 1))  # x_i
        bounds.append((0, 1))  # y_i
        bounds.append((0, R_wall[i]))  # r_i

    # Initial guess: structure is [x0, y0, r0, x1, y1, r1, ...]
    init = np.zeros(n_vars)
    for i in range(26):
        init[i] = centers[i][0]
        init[i + 26] = centers[i][1]
        init[i + 52] = radii[i]

    # Options
    options = {'maxiter': 1000}

    result = minimize(objective, init, method='trust-constr', bounds=bounds,
                     constraints=[{'type': 'eq', 'fun': constraint_eq},
                                  {'type': 'ineq', 'fun': constraint_ineq}],
                     options=options)

    if not result.success:
        # Fall back to original
        print("Refinement did not converge, returning original")
        return centers, radii

    # Extract final values
    final_x = result.x[:26]
    final_y = result.x[26:52]
    final_r = result.x[52:]
    final_centers = np.stack([final_x, final_y], axis=1)
    return final_centers, final_r

def run_packing():
    # Generate candidate centers on hex lattice with h=0.07
    centers, R_wall = generate_hex_lattice(0.07)
    print(f"Generated {len(centers)} candidate centers")
    # Build neighbour list (with max distance 0.4)
    edges = build_neighbour_list(centers, 0.4)
    print(f"Built {len(edges)} neighbour constraints")
    # Solve with branch-and-bound
    final_indices, final_radii = branch_and_bound(centers, R_wall, edges)
    if final_indices is None or final_radii is None:
        raise RuntimeError("Branch and bound failed")
    # Assign centers and final radii
    selected_centers = centers[final_indices]
    radii = final_radii
    # Local refinement
    refined_centers, refined_radii = local_continuous_refinement(selected_centers, radii, R_wall[final_indices])
    # Final validation
    success, msg = validate_packing(refined_centers, refined_radii)
    if not success:
        # Try shrinking radii
        shrink_factor = 1e-7
        for _ in range(5):
            refined_radii = np.maximum(1e-10, refined_radii - shrink_factor)
            success, msg = validate_packing(refined_centers, refined_radii)
            if success:
                break
        if not success:
            raise RuntimeError(f"Final validation failed: {msg}")
    sum_radii = float(np.sum(refined_radii))
    return refined_centers, refined_radii, sum_radii
