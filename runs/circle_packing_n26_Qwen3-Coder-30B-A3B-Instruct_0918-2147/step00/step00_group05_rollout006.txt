import numpy as np
from scipy.optimize import linprog, minimize
from math import sqrt, ceil

def build_hex_lattice(d):
    """Generates a hexagonal lattice with at least d points in [0,1]^2."""
    # Approximate spacing needed for d points
    side = sqrt(1.0 / (d * sqrt(3) / 2.0))
    rows = ceil(1.0 / side)
    cols = ceil(1.0 / (side * sqrt(3) / 2.0))
    if rows * cols < d:
        # Increase density slightly
        side *= 0.9
        rows = ceil(1.0 / side)
        cols = ceil(1.0 / (side * sqrt(3) / 2.0))

    centers = []
    for i in range(rows):
        y = side * sqrt(3) / 2.0 * i
        if y > 1.0:
            break
        for j in range(cols):
            offset = 0.0 if (i % 2) == 0 else side * 0.5
            x = side * j + offset
            if x > 1.0:
                break
            centers.append([x, y])
            if len(centers) >= d:
                break
        if len(centers) >= d:
            break

    # Ensure at least d points.
    while len(centers) < d:
        centers.append([0.5, 0.5])

    return np.array(centers[:d])

def lp_max_radii(centers):
    """Solves the LP for optimal radii given fixed centers."""
    n = centers.shape[0]

    # Wall constraints for radii:
    # r_i <= min(x_i, 1 - x_i, y_i, 1 - y_i)
    A_ub = []
    b_ub = []

    for i in range(n):
        x, y = centers[i]
        r_max = min(x, 1 - x, y, 1 - y)
        # -r_i <= -r_max  =>  r_i >= r_max
        A_ub.append([-1.0])
        b_ub.append(-r_max)

    # Pairwise overlap constraints:
    # r_i + r_j <= d_ij
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d_ij = sqrt(dx * dx + dy * dy)
            A_ub.append([1.0 if k == i else 1.0 if k == j else 0.0 for k in range(n)])
            b_ub.append(d_ij)

    c = np.ones(n)  # maximize sum of radii

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', tol=1e-9)
    except:
        # Fallback to uniform radius
        r = np.array([0.0] * n)
        return r, 0.0

    if res.status != 0:
        # Infeasible or unbounded; return minimal radii
        return np.array([0.0] * n), 0.0

    return res.x, np.sum(res.x)

def perturb_centers(centers, sigma):
    """Add Gaussian noise to centers and clamp to [0,1]."""
    new_centers = centers + np.random.normal(0, sigma, centers.shape)
    new_centers[:, 0] = np.clip(new_centers[:, 0], 0, 1)
    new_centers[:, 1] = np.clip(new_centers[:, 1], 0, 1)
    return new_centers

def anneal_centers(initial_centers):
    """Run the annealing algorithm to improve center positions via simulated annealing."""
    np.random.seed(42)  # Reproducibility
    centers = initial_centers.copy()
    best_sum = -1.0
    best_centers = initial_centers.copy()
    best_radii = None

    # Annealing schedule
    schedule = [
        {"T": 0.02, "sigma": 0.05, "iters": 2000},
        {"T": 0.01, "sigma": 0.03, "iters": 4000},
        {"T": 0.005, "sigma": 0.015, "iters": 3000},
        {"T": 0.002, "sigma": 0.008, "iters": 2000}
    ]

    for stage in schedule:
        T = stage["T"]
        sigma = stage["sigma"]
        iters = stage["iters"]

        for _ in range(iters):
            new_centers = perturb_centers(centers, sigma)
            radii, sum_radii = lp_max_radii(new_centers)
            if sum_radii > best_sum:
                best_sum = sum_radii
                best_centers = new_centers.copy()
                best_radii = radii.copy()
            if sum_radii >= best_sum:
                centers = new_centers.copy()
            else:
                prob = np.exp((sum_radii - best_sum) / T)
                if np.random.rand() < prob:
                    centers = new_centers.copy()

    return best_centers, best_radii

def local_refine(centers, radii):
    """Use deterministic optimization to improve the final configuration."""
    n = centers.shape[0]
    w = np.zeros(n)
    for i in range(n):
        x, y = centers[i]
        w[i] = min(x, 1 - x, y, 1 - y)

    def fun(x):
        # x is [c1_x, c1_y, ..., c26_x, c26_y], so 2*n values
        # reshape to centers (n, 2)
        c = x.reshape(n, 2)
        r = np.array([0.0]*n)
        for i in range(n):
            r[i] = min(
                c[i,0], 1-c[i,0],  # wall constraints for x
                c[i,1], 1-c[i,1]   # wall constraints for y
            )
        # Compute overlap radius from LP again
        # This is a lower bound using current centers' radii
        for i in range(n):
            for j in range(i+1, n):
                dx = c[i,0] - c[j,0]
                dy = c[i,1] - c[j,1]
                d = sqrt(dx*dx + dy*dy)
                min_radius = min(r[i], r[j]) if d > 0 and d < (r[i] + r[j]) else -1
                # Don't hurt the constraint if it's always feasible
                if min_radius > 0:
                    r[i] = max(r[i], min_radius)
                    r[j] = max(r[j], min_radius)

        # Add small clip to check if constraints are straightforwardly satisfied
        for i in range(n):
            r[i] = min(r[i], w[i])

        return -np.sum(r)  # maximize sum of radii -> minimize negate

    def jac(x):
        # Use finite difference for derivatives
        eps = 1e-8
        grad = np.zeros_like(x)
        for i in range(x.shape[0]):
            x_plus = x.copy()
            x_plus[i] += eps
            x_minus = x.copy()
            x_minus[i] -= eps
            grad[i] = (fun(x_plus) - fun(x_minus)) / (2*eps)
        return grad

    def constraint_func(x):
        c = x.reshape(n, 2)
        violations = []

        for i in range(n):
            # Wall constraints: r_i <= min(x_i, 1 - x_i, y_i, 1 - y_i)
            r_i = min(c[i,0], 1 - c[i,0], c[i,1], 1 - c[i,1])
            # Currently not applying r_i >= LP r_i since it's already satisfied
            # We're just ensuring centers stay inside the square
            violations.append(c[i,0] - 0.0)    # x_i >= 0
            violations.append(1.0 - c[i,0])    # x_i <= 1
            violations.append(c[i,1] - 0.0)    # y_i >= 0
            violations.append(1.0 - c[i,1])    # y_i <= 1

        # Overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                dx = c[i,0] - c[j,0]
                dy = c[i,1] - c[j,1]
                d = sqrt(dx*dx + dy*dy)
                # (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
                # So we have (r_i + r_j)^2 - d^2 <= 0
                # But we want f(x) <= 0 for constraint
                violations.append(d * d - (r[i] + r[j])*(r[i] + r[j]))

        return np.array(violations)

    constraints = {'type': 'ineq', 'fun': constraint_func}
    # Initial guess for lagrangian multipliers (are not passed to minimize directly)
    x0 = centers.reshape(-1)

    # slightly perturb to ensure proper trust-region behavior
    # these optimistic initial radii are probably conservative
    try:
        res = minimize(fun, x0=x0, method='trust-constr', jac=jac, constraints=constraints, options={'maxiter': 100})
        if res.success:
            refined_centers = res.x.reshape(n, 2)
            refined_radii, refined_sum = lp_max_radii(refined_centers)
            return refined_centers, refined_radii, refined_sum
    except Exception:
        pass
    # fallback: just return the LP radii at refined centers
    refined_radii, refined_sum = lp_max_radii(centers)
    return centers, refined_radii, refined_sum
