import numpy as np
from scipy.optimize import linprog
from scipy.spatial.distance import cdist

def generate_hex_lattice(h):
    sqrt3 = np.sqrt(3)
    points = []
    max_x = 1.0
    max_y = 1.0
    
    j_start = 0
    j_end = int(np.ceil(max_y / (h * sqrt3 / 2))) + 1
    
    for j in range(j_start, j_end):
        y = j * h * sqrt3 / 2
        if y > max_y:
            break
            
        i_start = 0
        if j % 2 == 1:
            i_start = 1
            
        i_end = int(np.ceil(max_x / h)) + 1
        for i in range(i_start, i_end, 2):
            x = i * h + (j % 2) * h / 2
            if x > max_x:
                break
            points.append([x, y])
    
    centers = np.array(points)
    R_wall = np.minimum(centers[:, 0], 1 - centers[:, 0])
    R_wall = np.minimum(R_wall, centers[:, 1])
    R_wall = np.minimum(R_wall, 1 - centers[:, 1])
    
    return centers, R_wall

def build_neighbour_list(centers, max_dist=0.4):
    distances = cdist(centers, centers)
    pairs = []
    n = centers.shape[0]
    
    for i in range(n):
        for j in range(i+1, n):
            d = distances[i, j]
            if d < max_dist:
                pairs.append((i, j, d))
    
    return pairs

def solve_radius_lp(selected_indices, R_wall, neighbour_list):
    n_selected = len(selected_indices)
    if n_selected == 0:
        return np.array([]), True
    
    # Create the LP: maximize sum of radii
    c = -np.ones(n_selected)
    
    # Wall constraints: r_i <= R_wall[i]
    A_ub_wall = np.eye(n_selected)
    b_ub_wall = R_wall[selected_indices]
    
    # Separation constraints: r_i + r_j <= d_ij
    n_neighbours = len(neighbour_list)
    A_ub_sep = []
    b_ub_sep = []
    
    # Map indices to local indices
    index_map = {idx: i for i, idx in enumerate(selected_indices)}
    
    for i, j, dist in neighbour_list:
        if i in index_map and j in index_map:
            # Add constraint: r_i + r_j <= d_ij
            row = np.zeros(n_selected)
            row[index_map[i]] = 1.0
            row[index_map[j]] = 1.0
            A_ub_sep.append(row)
            b_ub_sep.append(dist)
    
    if len(A_ub_sep) > 0:
        A_ub = np.vstack([A_ub_wall] + A_ub_sep)
        b_ub = np.hstack([b_ub_wall] + b_ub_sep)
    else:
        A_ub = A_ub_wall
        b_ub = b_ub_wall
    
    # Solve LP
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    
    if res.success:
        return -res.fun, True
    else:
        return np.array([]), False

def upper_bound(partial_set, remaining_indices, R_wall):
    total = sum(R_wall[i] for i in partial_set)
    # For remaining slots, assign largest available wall radius
    unselected_radii = [R_wall[i] for i in remaining_indices]
    unselected_radii.sort(reverse=True)
    remaining_slots = 26 - len(partial_set)
    total += sum(unselected_radii[:remaining_slots])
    return total

def branch_and_bound(centers, R_wall, neighbour_list):
    best_solution = []
    best_sum = -np.inf
    stack = []
    # Initial node: empty partial set
    # Partial set, remaining indices, lower bound, upper bound
    stack.append((set(), list(range(len(centers))), -np.inf, np.inf))
    
    while stack:
        partial_set, remaining_indices, lb, ub = stack.pop()
        
        if ub <= best_sum:
            continue
            
        # If we have enough selected
        if len(partial_set) == 26:
            # Compute actual sum
            indices_list = list(partial_set)
            sum_radii, feasible = solve_radius_lp(indices_list, R_wall, neighbour_list)
            if feasible and sum_radii > best_sum:
                best_sum = sum_radii
                best_solution = indices_list[:]
            continue
            
        # Feasibility check
        if len(partial_set) > 0:
            # Compute optimistic bound and early prune
            optimistic_ub = upper_bound(partial_set, remaining_indices, R_wall)
            if optimistic_ub <= best_sum:
                continue
    
        if not remaining_indices:
            continue
            
        # Choose next item to add (largest wall radius first)
        # Sort remaining indices by R_wall in descending order
        remaining_sorted = sorted(remaining_indices, key=lambda i: R_wall[i], reverse=True)
        next_idx = remaining_sorted[0]
        new_remaining = [i for i in remaining_indices if i != next_idx]
        
        # Option 1: include this item
        new_partial = partial_set.copy()
        new_partial.add(next_idx)
        new_lb = lb if lb > -np.inf else upper_bound(new_partial, new_remaining, R_wall)
        new_ub = upper_bound(new_partial, new_remaining, R_wall)
        stack.append((new_partial, new_remaining, new_lb, new_ub))
        
        # Option 2: skip this item
        stack.append((partial_set, new_remaining, lb, ub))
    
    return best_solution, best_sum
