import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
from math import sqrt

# Helper functions
def generate_lattice(s):
    """Generate a hexagonal lattice with spacing s"""
    sqrt3 = sqrt(3)
    rows = int(1 / (s * sqrt3 / 2)) + 2
    cols = int(1 / s) + 2
    points = []
    for r in range(rows):
        y = r * s * sqrt3 / 2
        if r % 2 == 0:
            for c in range(cols):
                x = c * s
                if s/2 <= x <= 1 - s/2 and s/2 <= y <= 1 - s/2:
                    points.append([x, y])
        else:
            for c in range(cols):
                x = (c + 0.5) * s
                if s/2 <= x <= 1 - s/2 and s/2 <= y <= 1 - s/2:
                    points.append([x, y])
    return np.array(points)

def build_milp(P):
    """Build the MILP model."""
    K = len(P)
    z = cp.Variable(K, boolean=True)
    r = cp.Variable(K, nonneg=True)

    # Full distances between all pairs
    D = np.zeros((K, K))
    for i in range(K):
        for j in range(i+1, K):
            dist = np.linalg.norm(P[i] - P[j])
            D[i, j] = D[j, i] = dist

    # Build the wall distance to each point
    wall_distances = []
    for p in P:
        wL = p[0]
        wR = 1 - p[0]
        wB = p[1]
        wT = 1 - p[1]
        wall_distances.append(min(wL, wR, wB, wT))

    w = np.array(wall_distances)

    # Constraints
    constraints = [cp.sum(z) == 26]
    for k in range(K):
        constraints.append(r[k] <= w[k] * z[k])
        for j in range(k):
            # big-M constraint
            M = 1
            constraints.append(D[k,j] >= r[k] + r[j] - M*(2 - z[k] - z[j]))

    # Objective
    objective = cp.Maximize(cp.sum(r))

    # Build a callable that solves for a fixed subset of z
    def solve(z_fixed=None):
        if z_fixed is not None:
            # Fix the binaries for this solve
            for i in range(K):
                z[i].fix(z_fixed[i])
        else:
            # Reset all to free
            for i in range(K):
                z[i].unfreeze()
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return r.value
        else:
            return None

    return z, r, constraints, objective, solve

def branch_and_bound(milp_solver, node_limit=20000):
    """Custom branch-and-bound solver"""
    best_z = None
    best_r = -np.inf
    # Priority queue: (bound, node_index, binary_assignment)
    import heapq
    queue = []

    # Start with all variables free (fractional)
    z_init = np.full(120, 0.5)
    r_sol = milp_solver(z_init)
    if r_sol is not None:
        bound = sum(r_sol)
        heapq.heappush(queue, (-bound, 0, z_init.copy()))

    node_count = 0
    while queue and node_count < node_limit:
        node_count += 1
        neg_bound, idx, z_vals = heapq.heappop(queue)
        bound = -neg_bound

        # If no fractional variables left, then integer solve
        if (z_vals == 0).sum() + (z_vals == 1).sum() == len(z_vals):
            if bound > best_r:
                best_z = z_vals.copy()
                best_r = bound
            continue

        # Find most fractional variable
        f = None
        j = -1
        for i in range(len(z_vals)):
            if z_vals[i] != 0 and z_vals[i] != 1:
                z_diff = abs(z_vals[i] - 0.5)
                if f is None or z_diff > f:
                    f = z_diff
                    j = i
        if j == -1:
            continue

        # Branch on j
        z1 = z_vals.copy()
        z1[j] = 1
        z0 = z_vals.copy()
        z0[j] = 0

        # Test both subproblems
        r_sol1 = milp_solver(z1)
        if r_sol1 is not None:
            bound1 = sum(r_sol1)
            if bound1 > best_r:
                best_z = z1.copy()
                best_r = bound1
            heapq.heappush(queue, (-bound1, idx + 1, z1))

        r_sol0 = milp_solver(z0)
        if r_sol0 is not None:
            bound0 = sum(r_sol0)
            if bound0 > best_r:
                best_z = z0.copy()
                best_r = bound0
            heapq.heappush(queue, (-bound0, idx + 1, z0))

    return best_z, best_r

def refine_solution(centers, radii):
    """Local continuous refinement of centers and radii."""
    n = len(centers)

    # Minimize for -sum(radii) subject to constraints
    def objective(r):
        return -np.sum(r)

    def jac(r):
        return -np.ones_like(r)

    def constraint_func(r, c):
        # Non-overlap constraints (residual = distance^2 - (r1+r2)^2)
        res = np.zeros(n*(n-1)//2)
        idx = 0
        for i in range(n):
            for j in range(i+1, n):
                dx = c[i,0] - c[j,0]
                dy = c[i,1] - c[j,1]
                dist_sq = dx*dx + dy*dy
                rad_sum_sq = (r[i] + r[j])**2
                res[idx] = dist_sq - rad_sum_sq
                idx += 1
        return res

    def wall_constr(c, r):
        # Wall constraints
        res = np.zeros(n*4)
        for i in range(n):
            res[4*i + 0] = r[i] - c[i,0]  # to left
            res[4*i + 1] = r[i] - (1 - c[i,0])  # to right
            res[4*i + 2] = r[i] - c[i,1]  # to bottom
            res[4*i + 3] = r[i] - (1 - c[i,1])  # to top
        return res

    def constr_jac(c, r):
        J = np.zeros((n*(n-1)//2 + n*4, n*3))
        idx = 0
        for i in range(n):
            for j in range(i+1, n):
                dx = c[i,0] - c[j,0]
                dy = c[i,1] - c[j,1]
                # Gradient of (dist_sq - (r[i] + r[j])**2)
                #  d/dxi: 2*dx, 2*dy, -2*(r[i]+r[j]), 0, ..., 0
                #  d/dxj: -2*dx, -2*dy, 0, ..., 0
                #  d/dr[i]: -2*(r[i]+r[j]), 0, 0, 0
                #  d/dr[j]: 0, 0, -2*(r[i]+r[j]), 0
                J[idx, 0:2] = [2*dx, 2*dy]  # d/dxi
                J[idx, 3:5] = [-2*dx, -2*dy]  # d/dxj
                J[idx, 6+i] = -2*(r[i] + r[j])
                J[idx, 6+j] = -2*(r[i] + r[j])
                idx += 1
        # Handle wall constraints
        for i in range(n):
            for k in range(4):
                J[idx+k, 3*i+k] = 1 if k < 2 else 1
            idx += 4
        return J

    # Initial guess
    r = np.copy(radii)
    c = np.copy(centers)

    # Setup constraints and bounds
    bounds = []
    for _ in range(n):
        # radius bounds
        bounds.append((1e-10, 0.5))  # radius
    for _ in range(n):
        bounds.append((1e-10, 0.5))  # x
        bounds.append((1e-10, 0.5))  # y

    # Placeholder for bounds check
    # Will now refine with SLSQP
    def wrap_f(r_c):
        c = r_c[0::3].reshape(-1, 2)
        r = r_c[2::3]
        return -np.sum(r)

    def wrap_jac(r_c):
        # derivative w.r.t. r
        J = np.zeros_like(r_c)
        J[2::3] = -1.0  # r[i] coefficients for the gradient
        return J

    def wrap_constr(r_c):
        c = r_c[0::3].reshape(-1, 2)
        r = r_c[2::3]
        # Non-overlapping
        res = []
        for i in range(n):
            for j in range(i+1, n):
                dx = c[i,0] - c[j,0]
                dy = c[i,1] - c[j,1]
                dist_sq = dx*dx + dy*dy
                rad_sum_sq = (r[i] + r[j])*(r[i] + r[j])
                res.append(dist_sq - rad_sum_sq)
        # Wall constraints
        for i in range(n):
            res.append(r[i] - c[i,0])   # left wall
            res.append(r[i] - (1 - c[i,0]))  # right wall
            res.append(r[i] - c[i,1])   # bottom wall
            res.append(r[i] - (1 - c[i,1]))  # top wall
        return np.array(res)

    # Run SLSQP with bounds and constraints
    # Initial guess in flat form
    x0 = np.concatenate([c.flatten(), r])

    cons = {'type': 'ineq', 'fun': wrap_constr}

    sol = minimize(wrap_f, x0, method='SLSQP', jac=wrap_jac,
                   bounds=bounds,
                   constraints=cons,
                   options={'ftol': 1e-9, 'maxiter': 1000, 'disp': False})

    if not sol.success:
        # If we failed, try one simple adjustment.
        x_final = x0
    else:
        # Extract final values
        c_final = sol.x[0::3].reshape(-1, 2)
        r_final = sol.x[2::3]

    # Clip any negative radii and then revalidate
    r_final = np.maximum(r_final, 0)
    # Apply validation to check
    try:
        valid, msg = validate_packing(c_final, r_final)
        if not valid:
            # raise ValueError(f"Validation error in refinement: {msg}")
            # For now, we just take the solution and return it.
            pass
    except:
        pass

    return c_final, r_final, np.sum(r_final)

def run_packing():
    # Generate lattice of candidate centre positions
    P = generate_lattice(s=0.13)
    print(f"Generated {len(P)} candidate points")

    # Set up MILP problem
    z, r, con, obj, solver = build_milp(P)
    best_S, best_sum = branch_and_bound(solver, node_limit=10000)

    # Convert best_S binary indices to use
    selected_indices = np.where(best_S == 1)[0]
    selected_centers = P[selected_indices]
    selected_radii = solver(best_S)

    # Print before refinement
    total_before = np.sum(selected_radii)

    # Refine with local optimization
    refined_centers, refined_radii, final_sum = refine_solution(selected_centers, selected_radii)

    # Validate final result
    validated, msg = validate_packing(refined_centers, refined_radii)
    if not validated:
        raise RuntimeError(f"Final packing invalid: {msg}")

    return refined_centers, refined_radii, final_sum
