import numpy as np

def hex_lattice(d):
    """Generate a hexagonal lattice of points in [0,1]x[0,1]"""
    sqrt3 = np.sqrt(3)
    dx = d
    dy = sqrt3 / 2 * d
    rows = []
    y = 0
    row_idx = 0
    while y <= 1 + 1e-12:
        x = 0
        if row_idx % 2 == 1:
            x = dx / 2
        row = []
        while x <= 1 + 1e-12:
            row.append([x, y])
            x += dx
        rows.append(row)
        y += dy
        row_idx += 1
    points = []
    for row in rows:
        for p in row:
            if 0 <= p[0] <= 1 and 0 <= p[1] <= 1:
                points.append(p)
    return np.array(points)

def add_border_circles(points, d):
    """Add 6 border-enhanced circles"""
    r = d / 4
    border_points = []
    # Four corners
    border_points.append([r, r])
    border_points.append([1 - r, r])
    border_points.append([r, 1 - r])
    border_points.append([1 - r, 1 - r])
    # Two edges
    border_points.append([0.5, r])
    border_points.append([0.5, 1 - r])
    return np.vstack([points, border_points])

def initial_radii(points):
    """Compute safe initial uniform radius"""
    n = points.shape[0]
    wall_distances = np.minimum(points[:, 0], 1 - points[:, 0])
    wall_distances = np.minimum(wall_distances, points[:, 1])
    wall_distances = np.minimum(wall_distances, 1 - points[:, 1])
    
    # Compute min distances to other centers
    min_distances = np.full(n, np.inf)
    for i in range(n):
        dists = np.linalg.norm(points[i] - points, axis=1)
        dists[i] = np.inf
        min_distances[i] = np.min(dists)
        
    safe_radii = np.minimum(wall_distances, 0.5 * min_distances)
    return np.min(safe_radii)

def inflate_radii(centers, radii):
    """Increase radii as much as possible while maintaining feasibility"""
    n = centers.shape[0]
    new_radii = radii.copy()
    distances = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    
    epsilon = 1e-7
    for i in range(n):
        # Wall constraints
        wall_caps = [
            centers[i, 0], 
            1 - centers[i, 0], 
            centers[i, 1], 
            1 - centers[i, 1]
        ]
        
        # Pairwise constraints
        caps = []
        for j in range(n):
            if i != j:
                caps.append(distances[i, j] - radii[j])
        
        # Final cap
        max_radius = min(wall_caps + caps)
        if max_radius > radii[i] + epsilon:
            new_radii[i] = max_radius
            
    return new_radii

def relax_centers(centers, radii, steps=10):
    """Apply force-based relaxation"""
    n = centers.shape[0]
    alpha = 0.5
    kappa = 0.02
    lambda_wall = 0.01
    beta = 0.02
    gamma = 0.02
    new_centers = centers.copy()
    
    for _ in range(steps):
        forces = np.zeros_like(centers)
        
        # Compute pairwise forces
        for i in range(n):
            for j in range(n):
                if i == j: continue
                diff = new_centers[i] - new_centers[j]
                dist = np.linalg.norm(diff)
                if dist > 1e-12:
                    unit_diff = diff / dist
                    force_magnitude = kappa * max(0, beta - dist)
                    forces[i] += force_magnitude * unit_diff
        
        # Wall forces
        for i in range(n):
            # x-axis forces
            dx_left = new_centers[i, 0] - radii[i]
            dx_right = 1 - (new_centers[i, 0] + radii[i])
            forces[i, 0] -= lambda_wall * max(0, gamma - dx_left)
            forces[i, 0] += lambda_wall * max(0, gamma - dx_right)
            
            # y-axis forces
            dy_bottom = new_centers[i, 1] - radii[i]
            dy_top = 1 - (new_centers[i, 1] + radii[i])
            forces[i, 1] -= lambda_wall * max(0, gamma - dy_bottom)
            forces[i, 1] += lambda_wall * max(0, gamma - dy_top)
        
        # Apply forces
        new_centers += alpha * forces
        
        # Clamp to valid range
        new_centers[:, 0] = np.clip(new_centers[:, 0], radii, 1 - radii)
        new_centers[:, 1] = np.clip(new_centers[:, 1], radii, 1 - radii)
    
    return new_centers
