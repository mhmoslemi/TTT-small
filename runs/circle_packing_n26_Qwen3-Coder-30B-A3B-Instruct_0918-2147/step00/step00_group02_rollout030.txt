import numpy as np
from scipy.optimize import minimize
from numpy.polynomial.hermite_e import hermite_eval2d

def run_packing():
    # Parameters
    n_circles = 26
    initial_budget = 3.0
    budget_step = 5e-4
    min_budget_step = 1e-6
    max_outer_iter = 40
    lambda_budget = 1000.0
    eps = 5e-10
    wall_eps = 1e-12

    # Generate Sobol-like initial points using low-discrepancy sequence approach
    def sobol_like(n=26, seed=42):
        # Simple quasi-random sequence generator
        np.random.seed(seed)
        points = np.random.rand(n, 2)
        return points

    def initial_configuration(n=26):
        centers = sobol_like(n)
        radii = np.zeros(n)
        # For each point, compute minimum to walls and to nearest neighbor
        for i in range(n):
            x, y = centers[i]
            dist_to_walls = [x, 1-x, y, 1-y]
            radii[i] = min(dist_to_walls)
            min_dist = np.inf
            for j in range(n):
                if i != j:
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.sqrt(dx*dx + dy*dy)
                    if dist < min_dist:
                        min_dist = dist
            if min_dist < np.inf:
                # Use half of min distance (but check against wall constraint)
                radii[i] = min(radii[i], min_dist / 2.0)
        return centers, radii

    def unpack(z):
        # z = [x0,y0,r0,x1,y1,r1,...,x25,y25,r25]
        centers = z[0::3].reshape(-1, 2)
        radii = z[2::3]
        return centers, radii

    def pack(centers, radii):
        z = np.empty(3 * len(centers))
        z[0::3] = centers[:, 0]
        z[1::3] = centers[:, 1]
        z[2::3] = radii
        return z

    def max_violation(centers, radii):
        n = len(centers)
        v_max = 0.0

        # Wall violations
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            v_max = max(v_max, x - r, 1 - x - r, y - r, 1 - y - r)

        # Overlap violations
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                overlap = radii[i] + radii[j] - dist
                v_max = max(v_max, overlap)

        return v_max

    def energy_and_grad(z):
        # z is flat vector [x0,y0,r0,x1,y1,r1,...]
        centers, radii = unpack(z)
        n = len(centers)
        E = 0.0
        grad = np.zeros_like(z)

        # Wall energy terms
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            # Split into individual terms for gradients
            dx = max(0, r - x)
            dy = max(0, r - (1 - x))
            dz = max(0, r - y)
            dw = max(0, r - (1 - y))
            E += dx*dx + dy*dy + dz*dz + dw*dw
            # Gradient
            if dx > 0:
                grad[3*i] -= 2*dx
                grad[3*i+2] += 2*dx
            if dy > 0:
                grad[3*i+1] -= 2*dy  # <- See corrected vector indexing
                grad[3*i+2] += 2*dy
            if dz > 0:
                grad[3*i+1] -= 2*dz  # <- See corrected vector indexing
                grad[3*i+2] += 2*dz
            if dw > 0:
                grad[3*i+1] -= 2*dw  # <- See corrected vector indexing
                grad[3*i+2] += 2*dw

        # Pairwise overlap terms
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < 1e-12:
                    dist = 1e-12  # Avoid division by zero
                overlap = radii[i] + radii[j] - dist
                if overlap > 0:
                    E += overlap*overlap
                    # Gradient w.r.t. centers and radii
                    scale = 2.0 * overlap / dist
                    dx_grad = -scale * dx
                    dy_grad = -scale * dy
                    grad[3*i]   += dx_grad
                    grad[3*i+1] += dy_grad
                    grad[3*j]   -= dx_grad
                    grad[3*j+1] -= dy_grad
                    grad[3*i+2] += overlap
                    grad[3*j+2] += overlap

        # Budget penalty
        total_r = np.sum(radii)
        budget_violation = initial_budget - total_r
        E += lambda_budget * budget_violation * budget_violation
        # Gradient for budget
        grad[2::3] -= 2 * lambda_budget * budget_violation  # for all radii

        return E, grad

    def inner_optimize(z0, B, lambda_budget):
        # Constraints for optimization
        bounds = []
        n = len(z0) // 3
        for i in range(n):
            bounds.append((0, 1))  # x
            bounds.append((0, 1))  # y
            bounds.append((0, 0.5))  # r (at most 0.5)
        result = minimize(
            energy_and_grad,
            z0,
            method='L-BFGS-B',
            bounds=bounds,
            jac=True,
            options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        return result.x

    def max_feasible_budget(z_start):
        B = initial_budget
        last_feasible_z = None
        delta = budget_step

        for outer_iter in range(max_outer_iter):
            # Adjust budget based on feasibility of previous result
            z_opt = inner_optimize(z_start, B, lambda_budget)
            centers, radii = unpack(z_opt)
            violation = max_violation(centers, radii)
            if violation <= 1e-13:
                # Feasible
                last_feasible_z = z_opt
                B += delta
                # Check if we’re done
                if delta < min_budget_step:
                    break
            else:
                # Infeasible
                B -= delta
                if delta < min_budget_step:
                    break
            z_start = z_opt.copy()  # use last solution to start next

        if last_feasible_z is None:
            raise RuntimeError("No feasible configuration found")

        return last_feasible_z

    # Main procedure
    initial_centers, initial_radii = initial_configuration(n_circles)
    z0 = pack(initial_centers, initial_radii)
    z_final = max_feasible_budget(z0)
    final_centers, final_radii = unpack(z_final)

    # Post-process to ensure strict boundaries
    final_radii = np.maximum(0.0, final_radii - eps)
    final_centers = np.clip(final_centers, eps, 1 - eps)

    # Repair if necessary
    for _ in range(10):  # few passes
        need_repair = False
        for i in range(len(final_centers)):
            for j in range(i+1, len(final_centers)):
                dx = final_centers[i, 0] - final_centers[j, 0]
                dy = final_centers[i, 1] - final_centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                overlap = final_radii[i] + final_radii[j] - dist
                if overlap > 0:
                    reduction = 0.5 * overlap + eps
                    final_radii[i] = max(eps, final_radii[i] - reduction)
                    final_radii[j] = max(eps, final_radii[j] - reduction)
                    need_repair = True
        if not need_repair:
            break

    sum_radii = np.sum(final_radii)
    centers = final_centers
    radii = final_radii

    # Final validation
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise RuntimeError(f"Invalid packing: {msg}")

    return centers, radii, sum_radii
