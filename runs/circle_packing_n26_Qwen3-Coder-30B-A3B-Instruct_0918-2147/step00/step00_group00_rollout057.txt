import numpy as np
from scipy.optimize import linprog

def run_packing():
    # Constants
    num_circles = 26
    max_iterations = 30
    epsilon = 1e-6
    polygon_sides = 8
    inflation_factor = 0.4
    max_inflation_passes = 5
    
    # Step 1: Geometry-first layout (deterministic seed)
    
    # Create triangular lattice pattern (hexagonal packing)
    sqrt3 = np.sqrt(3)
    # Estimate spacing to fit ~26 circles in square of side 0.8
    spacing = 0.8 / np.sqrt(num_circles)
    rows = int(np.ceil(np.sqrt(num_circles)))
    cols = int(np.ceil(num_circles / rows))
    
    centers_template = []
    for i in range(rows):
        for j in range(cols):
            x = j * spacing + (i % 2) * spacing / 2
            y = i * spacing * sqrt3 / 2
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers_template.append([x, y])
                if len(centers_template) == num_circles:
                    break
        if len(centers_template) == num_circles:
            break
    
    centers = np.array(centers_template)
    
    # Ensure the pattern is within safe margin
    margin = 0.05
    centers[:, 0] = np.clip(centers[:, 0], margin, 1 - margin)
    centers[:, 1] = np.clip(centers[:, 1], margin, 1 - margin)
    
    # Initial small radii
    radii = np.full(num_circles, 0.01)
    
    # Precompute direction vectors for polygonal approximation
    theta_vec = np.linspace(0, 2*np.pi, polygon_sides, endpoint=False)
    direction_vectors = np.column_stack([np.cos(theta_vec), np.sin(theta_vec)])
    cos_pi_over_k = np.cos(np.pi / polygon_sides)
    
    # Compute all pairwise distances once for efficiency
    def compute_distance_matrix(c):
        diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
        distances = np.sqrt(np.sum(diff**2, axis=2))
        return distances
    
    # Step 2 and 3: Alternating LP loop
    prev_sum = 0
    
    for iteration in range(max_iterations):
        # Compute distances for constraints
        distance_matrix = compute_distance_matrix(centers)
        
        # Solve for optimal radii given current centers (radius LP)
        # Constraints are:
        # r_i <= c_i.x, 1 - c_i.x, c_i.y, 1 - c_i.y (4 wall constraints)  
        # r_i + r_j <= d_ij (separation constraints)
        
        A_ub = []
        b_ub = []
        
        # Wall constraints
        for i in range(num_circles):
            # r_i <= c_i.x, 1 - c_i.x, c_i.y, 1 - c_i.y
            A_ub.append([1.0 if j == i else 0.0 for j in range(num_circles)])
            b_ub.append(centers[i, 0])
            A_ub.append([1.0 if j == i else 0.0 for j in range(num_circles)])
            b_ub.append(1 - centers[i, 0])
            A_ub.append([1.0 if j == i else 0.0 for j in range(num_circles)])
            b_ub.append(centers[i, 1])
            A_ub.append([1.0 if j == i else 0.0 for j in range(num_circles)])
            b_ub.append(1 - centers[i, 1])
        
        # Separation constraints
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                # r_i + r_j <= d_ij
                a_row = [0.0] * num_circles
                a_row[i] = 1.0
                a_row[j] = 1.0
                A_ub.append(a_row)
                b_ub.append(distance_matrix[i, j])
        
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        
        # Solve LP: maximize sum of radii = 1^T * r
        # subject to A_ub * r <= b_ub, r >= 0 
        sol = linprog(c=-np.ones(num_circles), A_ub=A_ub, b_ub=b_ub, 
                      method='highs', options={'tol': 1e-10})
        
        if not sol.success:
            break
            
        radii = np.clip(sol.x, 0, 0.5)
        
        # Now solve for optimal centers given fixed radii (center LP)
        # The goal is to maximize the sum of slack distances to constraints
        
        # Build large constraint set
        num_constraints = 4 * num_circles + num_circles * (num_circles - 1) // 2 * polygon_sides
        
        A_ub_center = []
        b_ub_center = []
        A_eq_center = []
        b_eq_center = []
        
        # Start with a slack variable for each circle
        # We are doing: maximize sum of slack vars s_i
        # Slack x-direction: c_i.x - r_i >= s_i => -c_i.x + s_i <= -r_i
        # Slack y-direction: c_i.y - r_i >= s_i => -c_i.y + s_i <= -r_i
        # etc. for both walls and neighbors
        
        # Variables: 2 * num_circles (centers) + num_circles (slack)
        num_vars = 2 * num_circles + num_circles
        
        # Wall constraints on 2 * num_circles variables (for each center)
        for i in range(num_circles):
            # x variable
            a_row = [0.0] * num_vars
            a_row[2*i] = -1.0
            a_row[2*num_circles + i] = 1.0
            b_ub_center.append(-radii[i])
            
            # y variable
            a_row = [0.0] * num_vars
            a_row[2*i + 1] = -1.0
            a_row[2*num_circles + i] = 1.0
            b_ub_center.append(-radii[i])
            
            # For non-left/right/bottom/top (so avoid locking in)
            a_row = [0.0] * num_vars
            a_row[2*i] = 1.0
            a_row[2*num_circles + i] = -1.0
            b_ub_center.append(radii[i])
            
            a_row = [0.0] * num_vars
            a_row[2*i + 1] = 1.0
            a_row[2*num_circles + i] = -1.0
            b_ub_center.append(radii[i])
            
            a_row = [0.0] * num_vars
            a_row[2*i] = 1.0
            a_row[2*num_circles + i] = -1.0
            b_ub_center.append(1 - radii[i])
            
            a_row = [0.0] * num_vars
            a_row[2*i + 1] = 1.0
            a_row[2*num_circles + i] = -1.0
            b_ub_center.append(1 - radii[i])
            
        # Non-overlap constraints: for each neighbor, linearize polygonal inclusion
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                for direction in direction_vectors:
                    a_row = [0.0] * num_vars
                    a_row[2*i] = direction[0]
                    a_row[2*i + 1] = direction[1]
                    a_row[2*j] = -direction[0]
                    a_row[2*j + 1] = -direction[1]
                    a_row[2*num_circles + i] = cos_pi_over_k
                    a_row[2*num_circles + j] = cos_pi_over_k
                    b_ub_center.append(-(radii[i] + radii[j]) * cos_pi_over_k)
                    A_ub_center.append(a_row)
        
        A_ub_center = np.array(A_ub_center)
        b_ub_center = np.array(b_ub_center)
        
        # Objective: maximize sum of slacks
        c = np.zeros(num_vars)
        c[-num_circles:] = 1.0
        
        # solve with equality constraints to maintain bounds (can be skipped)
        # because we can trust bounds from clipping and triangular lattice
        
        sol = linprog(c=c, A_ub=A_ub_center, b_ub=b_ub_center,
                      method='highs', options={'tol': 1e-10})
        
        if not sol.success:
            break
        
        center_vars = sol.x[:-num_circles]
        centers = center_vars.reshape((num_circles, 2))
        centers = np.clip(centers, 0, 1)
        
        current_sum = np.sum(radii)
        if current_sum - prev_sum < epsilon:
            break
        prev_sum = current_sum
        
    # Step 5: Local inflation via exact Euclidean distances
    for _ in range(max_inflation_passes):
        prev_radii = radii.copy()
        for i in range(num_circles):
            # Compute exact distances to all other circles and walls
            dist_to_walls = np.array([
                centers[i, 0] - radii[i], 
                1 - centers[i, 0] - radii[i],
                centers[i, 1] - radii[i],
                1 - centers[i, 1] - radii[i]
            ])
            min_wall_gap = np.min(dist_to_walls[dist_to_walls > 0])
            
            min_neighbor_gap = float('inf')
            for j in range(num_circles):
                if i == j:
                    continue
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                gap = dist - (radii[i] + radii[j])
                if gap > 0 and gap < min_neighbor_gap:
                    min_neighbor_gap = gap
            
            # take a conservative overlap of min of wall and neighbor gaps  
            # (for pushing to keep spacing valid)
            exact_gap = min(min_wall_gap, min_neighbor_gap) if min_neighbor_gap < float('inf') else min_wall_gap
            
            if exact_gap > 0:
                # Try increasing the radius by a small fraction of the gap
                inc = inflation_factor * exact_gap
                if inc > 1e-5:
                    # Re-solve for radius LP considering change in i only
                    old_r = radii[i]
                    radii[i] = min(old_r + inc, 0.5)
                    
                # Check if there was improvement
                if radii[i] > old_r:
                    # Recompute appropriate LP fixes only radii affected by neighbor interactions
                    # We do this naively: recompute the radius LP with updated radius
                    new_distance_matrix = compute_distance_matrix(centers)
                    A_ub = []
                    b_ub = []
                    
                    # Wall constraints for all
                    for k in range(num_circles):
                        # r_i <= c_i.x, 1 - c_i.x, c_i.y, 1 - c_i.y
                        A_ub.append([1.0 if j == k else 0.0 for j in range(num_circles)])
                        b_ub.append(centers[k, 0])
                        A_ub.append([1.0 if j == k else 0.0 for j in range(num_circles)])
                        b_ub.append(1 - centers[k, 0])
                        A_ub.append([1.0 if j == k else 0.0 for j in range(num_circles)])
                        b_ub.append(centers[k, 1])
                        A_ub.append([1.0 if j == k else 0.0 for j in range(num_circles)])
                        b_ub.append(1 - centers[k, 1])
                    
                    # Separation constraints
                    for k in range(num_circles):
                        for l in range(k + 1, num_circles):
                            # r_k + r_l <= d_kl
                            a_row = [0.0] * num_circles
                            a_row[k] = 1.0
                            a_row[l] = 1.0
                            A_ub.append(a_row)
                            b_ub.append(new_distance_matrix[k, l])
                    
                    A_ub = np.array(A_ub)
                    b_ub = np.array(b_ub)
                    
                    try:
                        sol = linprog(c=-np.ones(num_circles), A_ub=A_ub, b_ub=b_ub,
                                      method='highs', options={'tol': 1e-10})
                        if sol.success:
                            # Re-clip final radii
                            radii = np.clip(sol.x, 0, 0.5)
                    except:
                        pass  # fallback
        
        if np.allclose(prev_radii, radii, atol=1e-5):
            break
    
    # Step 6: Final validation and cleanup
    valid, msg = validate_packing(centers, radii)
    if not valid:
        # Shrink radii slightly if needed
        for _ in range(5):
            radii = np.clip(radii * (1.-1e-7), 0, 0.5)
            valid, msg = validate_packing(centers, radii)
            if valid: break
    
    assert valid, msg
    
    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
