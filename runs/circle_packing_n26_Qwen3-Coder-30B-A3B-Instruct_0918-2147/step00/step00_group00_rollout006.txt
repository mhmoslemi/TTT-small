import numpy as np
from scipy.optimize import linprog
from scipy.spatial.distance import cdist

def generate_hex_lattice(h):
    sqrt3 = np.sqrt(3)
    centers = []
    y_coord = 0.0
    i = 0
    while y_coord <= 1.0 + h:
        x_offset = (i % 2) * h / 2.0
        x_coord = x_offset
        j = 0
        while x_coord <= 1.0 + h:
            if 0.0 <= x_coord <= 1.0 and 0.0 <= y_coord <= 1.0:
                centers.append([x_coord, y_coord])
            x_coord += h
            j += 1
        y_coord += h * sqrt3 / 2.0
        i += 1
    return np.array(centers)

def build_neighbour_list(centers, max_dist):
    dist_matrix = cdist(centers, centers)
    edges = []
    n = centers.shape[0]
    for i in range(n):
        for j in range(i+1, n):
            d = dist_matrix[i, j]
            if d < max_dist:
                edges.append((i, j, d))
    return edges

def solve_radius_lp(selected_indices, R_wall, neighbour_list):
    n_selected = len(selected_indices)
    if n_selected == 0:
        return np.array([]), False

    # Build LP problem
    # c = -1 * np.ones(n_selected)
    c = np.ones(n_selected)
    
    # Inequality constraints: A_ub @ x <= b_ub
    A_ub = []
    b_ub = []
    
    # Wall constraints: 0 <= r_i <= R_wall[i]
    for i in range(n_selected):
        row = np.zeros(n_selected)
        row[i] = -1.0  # r_i >= 0
        A_ub.append(row)
        b_ub.append(0.0)
        row = np.zeros(n_selected)
        row[i] = 1.0   # r_i <= R_wall[i]
        A_ub.append(row)
        b_ub.append(R_wall[selected_indices[i]])
    
    # Separation constraints: r_i + r_j <= d_ij
    for i, j, d in neighbour_list:
        if i in selected_indices and j in selected_indices:
            idx_i = selected_indices.index(i)
            idx_j = selected_indices.index(j)
            row = np.zeros(n_selected)
            row[idx_i] = 1.0
            row[idx_j] = 1.0
            A_ub.append(row)
            b_ub.append(d)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'presolve': True})
    if res.success:
        return -res.x, True
    else:
        return np.array([]), False

def upper_bound(partial_set, remaining_indices, R_wall):
    total = sum(R_wall[i] for i in partial_set)
    remaining_count = 26 - len(partial_set)
    if remaining_count <= 0:
        return total
    max_remaining = max(R_wall[i] for i in remaining_indices)
    return total + max_remaining * remaining_count

def branch_and_bound(centers, R_wall, neighbour_list):
    best_total = -np.inf
    best_indices = []
    best_radii = []

    # For efficiency, sort by wall radii descending to try large radii first
    wall_indices_sorted = [(R_wall[i], i) for i in range(len(R_wall))]
    wall_indices_sorted.sort(reverse=True)
    
    # Node: (selected_indices, unselected_indices, lb, ub)
    # Use a stack for iterative DFS (or a priority queue if using best-first)
    stack = [([], list(range(len(centers))), -np.inf, np.inf)]

    while stack:
        partial_indices, unselected, lb, ub = stack.pop()
        current_size = len(partial_indices)
        
        # Pruning
        if ub <= best_total:
            continue
        
        if current_size == 26:
            if len(partial_indices) != 26:
                continue
            radii, feasible = solve_radius_lp(partial_indices, R_wall, neighbour_list)
            if feasible:
                total = np.sum(radii)
                if total > best_total:
                    best_total = total
                    best_indices = partial_indices.copy()
                    best_radii = radii.copy()
        else:
            # Branch
            # In this simple version, we just take the best
            remaining = unselected[:]
            if len(remaining) == 0:
                continue
            
            # Use upper bound to guide which to try next
            # We'll do a greedy-ish search, preferring large wall radii
            remaining.sort(key=lambda i: R_wall[i], reverse=True)
            # Try the first few candidates to see if they cause any problems
            for idx in remaining[:min(10, len(remaining))]:
                new_partial = partial_indices + [idx]
                new_unselected = [i for i in unselected if i != idx]
                new_ub = upper_bound(new_partial, new_unselected, R_wall)
                if new_ub > best_total:
                    # Compute refined bound (partial LP with walls only)
                    # Just compute an optimistic guess: use wall values if manageable
                    # instead of solving sub-LP, skip small validation details.
                    stack.append((new_partial, new_unselected, lb, new_ub))
    
    if len(best_indices) == 26 and len(best_radii) > 0:
        return np.array(best_indices), np.array(best_radii)
    else:
        # fallback if no solution found, return empty
        return np.array([]), np.array([])
