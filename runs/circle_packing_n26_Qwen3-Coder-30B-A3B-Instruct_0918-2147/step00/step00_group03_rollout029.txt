import numpy as np
from scipy.optimize import minimize
import scipy.sparse as sp

def init_hex_grid(n=26, r_init=0.04):
    # Generate hexagonal grid points
    sqrt3 = np.sqrt(3)
    h = sqrt3 * r_init
    rows = []
    y = r_init
    row_num = 0
    while len(rows) < n:
        # Even rows: x = r_init + 2*i*r_init
        # Odd rows: x = 2*r_init + 2*i*r_init
        x_start = r_init if row_num % 2 == 0 else 2 * r_init
        x = x_start
        i = 0
        while True:
            x_pos = x + i * 2 * r_init
            if x_pos > 1 - r_init:
                break
            if len(rows) < n:
                rows.append((x_pos, y))
                i += 1
            else:
                break
        y += h
        if y > 1 - r_init:
            break
        row_num += 1
    centers = np.array(rows[:n])
    radii = np.full(n, r_init)
    return centers, radii

def max_radii(centers, radii, max_iter=100, tol=1e-8):
    # ceiling for preventing numerical issues
    max_radius = 0.5
    for _ in range(max_iter):
        # Compute wall limits
        wall_limits = np.minimum(centers, 1 - centers).min(axis=1)
        # Distance matrix
        D = np.sqrt(((centers[:, None, :] - centers[None, :, :]) ** 2).sum(axis=2))
        # Set diagonal to large value to ignore self-distance
        np.fill_diagonal(D, 1e10)
        # Neighbor limits (each circle's max radius)
        neighbor_limits = D - radii[:, None]
        neighbor_limits[np.arange(len(radii)), np.arange(len(radii))] = 1e10
        min_neighbor = np.min(neighbor_limits, axis=1)
        # Combined limits
        new_radii = np.minimum(wall_limits, min_neighbor)
        new_radii = np.clip(new_radii, 0, max_radius)
        if np.max(np.abs(new_radii - radii)) < tol:
            break
        radii = new_radii
    return radii

def hill_climb(centers, radii, iterations=5000, step=0.01, tolerance=1e-12):
    # Store initial state
    current_centers = centers.copy()
    current_radii = radii.copy()
    current_sum = np.sum(current_radii)
    for _ in range(iterations):
        # Shuffle circles for stochastic update
        indices = np.random.permutation(len(current_centers))
        for idx in indices:
            # Generate proposal
            delta = np.random.uniform(-step, step, 2)
            new_centers = current_centers.copy()
            new_centers[idx] += delta
            # Check new centers are valid
            if (new_centers[idx][0] - current_radii[idx] < -tol or 
                new_centers[idx][0] + current_radii[idx] > 1 + tol or
                new_centers[idx][1] - current_radii[idx] < -tol or 
                new_centers[idx][1] + current_radii[idx] > 1 + tol):
                continue
            new_radii = current_radii.copy()
            # Apply max radii to new centers
            new_radii = max_radii(new_centers, new_radii)
            new_sum = np.sum(new_radii)
            if new_sum > current_sum + tolerance:
                current_centers = new_centers
                current_radii = new_radii
                current_sum = new_sum
    return current_centers, current_radii

def refine_with_scipy(centers, radii):
    n = len(radii)
    # Flatten centers and radii into a single vector
    x0 = np.concatenate([centers.flatten(), radii])
    # Define constraints
    def wall_constraint(c):
        # c: [x1, y1, ..., x26, y26, r1, ..., r26]
        x = c[:n]
        y = c[n:2*n]
        r = c[2*n:]
        # For each circle i:
        # r_i - x_i <= 0
        # r_i + x_i - 1 <= 0
        # r_i - y_i <= 0
        # r_i + y_i - 1 <= 0
        return np.concatenate([
            r - x,           # r_i - x_i <= 0
            r + x - 1,       # r_i + x_i - 1 <= 0
            r - y,           # r_i - y_i <= 0
            r + y - 1        # r_i + y_i - 1 <= 0
        ])

    def overlap_constraint(c):
        # Return difference between sum of radii and norm of center differences
        x = c[:n]
        y = c[n:2*n]
        r = c[2*n:]
        centers = np.column_stack([x, y])
        diffs = centers[:, None, :] - centers[None, :, :]
        norms = np.sqrt(np.sum(diffs**2, axis=2))
        # For each pair i, j such that i < j,
        # we want r_i + r_j - norms[i,j] <= 0
        constraints = []
        for i in range(n):
            for j in range(i+1, n):
                constraints.append(r[i] + r[j] - norms[i,j])
        return np.array(constraints)

    # Jacobian for wall
    def wall_jacobian(c):
        x = c[:n]
        y = c[n:2*n]
        r = c[2*n:]
        # Each of the 4*n wall constraints a single variable
        Irow = []
        Icol = []
        data = []
        # 0 - n-1: r_i - x_i
        # n - 2*n-1: r_i + x_i - 1
        # 2*n - 3*n-1: r_i - y_i
        # 3*n - 4*n-1: r_i + y_i - 1
        for i in range(n):
            # r_i - x_i
            Irow.extend([i]*2)
            Icol.extend([2*n+i, i])
            data.extend([1, -1])
            # r_i + x_i - 1
            Irow.extend([n+i]*2)
            Icol.extend([2*n+i, i])
            data.extend([1, 1])
            # r_i - y_i
            Irow.extend([2*n+i]*2)
            Icol.extend([2*n+i, n+i])
            data.extend([1, -1])
            # r_i + y_i - 1
            Irow.extend([3*n+i]*2)
            Icol.extend([2*n+i, n+i])
            data.extend([1, 1])
        # Sparse matrix representation
        return sp.coo_matrix((data, (Irow, Icol)), shape=(4*n, 3*n))

    # Jacobian for overlap
    def overlap_jacobian(c):
        x = c[:n]
        y = c[n:2*n]
        r = c[2*n:]
        centers = np.column_stack([x, y])
        diffs = centers[:, None, :] - centers[None, :, :]
        norms = np.sqrt(np.sum(diffs**2, axis=2))
        # for all i<j pairs:
        #  (d/dx_i + d/dx_j) overlap
        #  (d/dy_i + d/dy_j) overlap
        #  (d/dr_i + d/dr_j) overlap
        # Row for (r_i + r_j - ||c_i - c_j||) <= 0
        Irow = []
        Icol = []
        data = []
        for i in range(n):
            for j in range(i+1, n):
                # Compute gradient:
                # dp = d/dr_i (r_i + r_j - norm)
                dp = 1.0
                Irow.append(len(Irow))
                Icol.append(2*n+i)
                data.append(dp)

                dp = 1.0
                Irow.append(len(Irow))
                Icol.append(2*n+j)
                data.append(dp)

                # Connection from c_i to c_j
                dp = -(centers[i,0] - centers[j,0]) / (norms[i,j] + 1e-10)
                Irow.append(len(Irow))
                Icol.append(i)
                data.append(dp)
                
                dp = - (centers[i,1] - centers[j,1]) / (norms[i,j] + 1e-10)
                Irow.append(len(Irow))
                Icol.append(n+i)
                data.append(dp)

                dp = (centers[j,0] - centers[i,0]) / (norms[i,j] + 1e-10)
                Irow.append(len(Irow))
                Icol.append(j)
                data.append(dp)

                dp = (centers[j,1] - centers[i,1]) / (norms[i,j] + 1e-10)
                Irow.append(len(Irow))
                Icol.append(n+j)
                data.append(dp)
        # Sparse matrix representation
        return sp.coo_matrix((data, (Irow, Icol)))

    # Define bounds
    bounds = []
    for i in range(2*n):
        bounds.append((0, 1))
    for i in range(n):
        bounds.append((0, 0.5))

    # Nonlinear constraints
    wall_con = NonlinearConstraint(wall_constraint, np.full(4*n, -np.inf), 0, jac=wall_jacobian)
    overlap_con = NonlinearConstraint(overlap_constraint, np.full(n*(n-1)//2, -np.inf), 0, jac=overlap_jacobian)

    res = minimize(lambda x: -np.sum(x[2*n:]), x0, method='trust-constr', bounds=bounds,
                   constraints=[wall_con, overlap_con], options={'maxiter': 200})

    # Extract final solution
    new_centers = res.x[:2*n].reshape((n, 2))
    new_radii = res.x[2*n:]
    return new_centers, new_radii

def run_packing():
    # Initial layout
    centers, radii = init_hex_grid()
    # Maximize radii (inner loop)
    radii = max_radii(centers, radii)
    # Stochastic hill climbing
    centers, radii = hill_climb(centers, radii)
    # Final refinement
    centers, radii = refine_with_scipy(centers, radii)
