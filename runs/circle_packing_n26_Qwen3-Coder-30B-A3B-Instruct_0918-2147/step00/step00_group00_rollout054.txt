import numpy as np
from scipy.optimize import minimize

def make_hex_lattice(n, margin=0.05):
    """Generate a hexagonal lattice of n points inside the unit square with margin."""
    sqrt3 = np.sqrt(3)
    dx = 1.0 / (int(np.ceil(np.sqrt(n))) + 0.5)
    dy = dx * sqrt3 / 2
    points = []
    i, j = 0, 0
    while len(points) < n:
        x = i * dx + (j % 2) * dx / 2
        y = j * dy
        if 0 <= x <= 1 and 0 <= y <= 1:
            points.append([x, y])
        if x + dx / 2 > 1:
            j += 1
            i = 0
        else:
            i += 1
    points = np.array(points[:n])
    # center the pattern
    x_min, x_max = points[:, 0].min(), points[:, 0].max()
    y_min, y_max = points[:, 1].min(), points[:, 1].max()
    dx_center = (0.5 * (x_min + x_max) - 0.5)
    dy_center = (0.5 * (y_min + y_max) - 0.5)
    points[:, 0] -= dx_center
    points[:, 1] -= dy_center
    # ensure margin
    points[:, 0] += margin
    points[:, 1] += margin
    return points

def objective(z):
    """Compute negative sum of radii."""
    return -np.sum(z[2::3])

def boundary_constraints(z):
    """Generate boundary constraint values. For each circle (x,y,r):
    x - r >= 0
    x + r <= 1
    y - r >= 0
    y + r <= 1
    """
    n = len(z) // 3
    cons = np.zeros(4 * n)
    for i in range(n):
        x, y, r = z[3*i], z[3*i + 1], z[3*i + 2]
        cons[4*i] = x - r  # x - r >= 0
        cons[4*i + 1] = 1 - x - r  # x + r <= 1
        cons[4*i + 2] = y - r  # y - r >= 0
        cons[4*i + 3] = 1 - y - r  # y + r <= 1
    return cons

def pairwise_constraints(z):
    """Generate pairwise distance constraint values. For each unordered pair (i,j):
    (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    """
    n = len(z) // 3
    cons = []
    for i in range(n):
        for j in range(i + 1, n):
            x1, y1, r1 = z[3*i], z[3*i + 1], z[3*i + 2]
            x2, y2, r2 = z[3*j], z[3*j + 1], z[3*j + 2]
            dist_sq = (x1 - x2)**2 + (y1 - y2)**2
            sum_radii_sq = (r1 + r2)**2
            cons.append(dist_sq - sum_radii_sq)
    return np.array(cons)

def jacobian_boundary(z):
    """Jacobian of boundary constraints."""
    n = len(z) // 3
    jac = np.zeros((4*n, 3*n))
    for i in range(n):
        jac[4*i, 3*i] = 1  # dx / dx_i
        jac[4*i+1, 3*i] = -1  # dx / dx_i
        jac[4*i+2, 3*i+1] = 1  # dy / dy_i
        jac[4*i+3, 3*i+1] = -1  # dy / dy_i
        jac[4*i, 3*i+2] = -1  # dr / dr_i
        jac[4*i+1, 3*i+2] = -1  # dr / dr_i
        jac[4*i+2, 3*i+2] = -1  # dr / dr_i
        jac[4*i+3, 3*i+2] = -1  # dr / dr_i
    return jac

def jacobian_pairwise(z):
    """Jacobian of pairwise constraints."""
    n = len(z) // 3
    jac = np.zeros((n*(n-1)//2, 3*n))
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            x1, y1, r1 = z[3*i], z[3*i+1], z[3*i+2]
            x2, y2, r2 = z[3*j], z[3*j+1], z[3*j+2]
            # derivative w.r.t. i
            jac[idx, 3*i] = 2 * (x1 - x2)
            jac[idx, 3*i+1] = 2 * (y1 - y2)
            jac[idx, 3*i+2] = 2 * (r1 + r2)
            # derivative w.r.t. j
            jac[idx, 3*j] = -2 * (x1 - x2)
            jac[idx, 3*j+1] = -2 * (y1 - y2)
            jac[idx, 3*j+2] = 2 * (r1 + r2)
            idx += 1
    return jac

def solve_nlp(initial_z):
    """Solve the NLP using SLSQP with constraints."""
    n = len(initial_z) // 3
    # Objective function
    def obj_func(z):
        return objective(z)
    # Boundary constraints
    def bnd_cons_func(z):
        return boundary_constraints(z)
    # Pairwise constraints
    def pw_cons_func(z):
        return pairwise_constraints(z)
    # Jacobians
    def jacob_bnd(z):
        return jacobian_boundary(z)
    def jacob_pw(z):
        return jacobian_pairwise(z)
        
    # Define constraints
    bnd_cons = {'type': 'ineq', 'fun': bnd_cons_func, 'jac': jacob_bnd}
    pw_cons = {'type': 'ineq', 'fun': pw_cons_func, 'jac': jacob_pw}
    constraints = [bnd_cons, pw_cons]
    
    # Variable bounds
    bounds = []
    for i in range(n):
        # x, y, r
        bounds.append((0, 1))
        bounds.append((0, 1))
        bounds.append((0, 0.5))  # radii maximum is 0.5
    res = minimize(obj_func, initial_z, method='SLSQP', bounds=bounds, 
                   constraints=constraints, options={'ftol': 1e-9, 'maxiter': 1000})
    return res.x

def nudge_centers(centers, radii):
    """Nudge any circle that's tight against a boundary."""
    eps = 0.001
    new_centers = centers.copy()
    for i in range(len(centers)):
        x, y = centers[i]
        r = radii[i]
        # Check for tight wall contacts
        if abs(x - r) < 1e-4:
            new_centers[i] = [x + eps, y]
        elif abs(1 - x - r) < 1e-4:
            new_centers[i] = [x - eps, y]
        elif abs(y - r) < 1e-4:
            new_centers[i] = [x, y + eps]
        elif abs(1 - y - r) < 1e-4:
            new_centers[i] = [x, y - eps]
    return new_centers

def run_packing():
    """Main function to generate and optimize circle packing."""
    # Initial hexagonal lattice
    centers = make_hex_lattice(26)
    # Initial radii
    radii = np.full(26, 0.02)
    # Initial guess
    initial_z = np.array([coord for center in centers for coord in center] + list(radii))
    
    # Solve a few times with nudging
    old_sum = -np.inf
    for _ in range(5):
        # Solve with current configuration
        result = solve_nlp(initial_z)
        # Extract centers and radii
        new_centers = result[::3].reshape(26, 2)
        new_radii = result[2::3]
        new_sum = np.sum(new_radii)
        if new_sum - old_sum < 1e-5:
            break
        old_sum = new_sum
        # Nudge centers that are close to a boundary
        new_centers = nudge_centers(new_centers, new_radii)
        # Rebuild initial_z
        initial_z = np.array([coord for center in new_centers for coord in center] + list(new_radii))

    # Final validation to ensure correct format
    assert validate_packing(new_centers, new_radii)[0], "Validation failed"
    
    # Final output
    sum_radii = float(np.sum(new_radii))
    return new_centers, new_radii, sum_radii

def validate_packing(centers, radii):
    """Run-only validation function."""
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
