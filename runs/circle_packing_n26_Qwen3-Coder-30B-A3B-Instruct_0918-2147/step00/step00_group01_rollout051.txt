import numpy as np
from scipy.optimize import differential_evolution
from scipy.spatial.distance import cdist

def compute_radii(centers):
    n = centers.shape[0]
    radii = np.zeros(n)
    
    # Wall distances
    x = centers[:, 0]
    y = centers[:, 1]
    wall_distances = npMinimum.reduce([x, 1 - x, y, 1 - y], axis=0)
    
    # Pairwise distances
    distances = cdist(centers, centers)
    # Set diagonal to large value to ignore self-distances
    np.fill_diagonal(distances, np.inf)
    
    # Half the minimum pairwise distance for each circle
    min_distances = 0.5 * np.min(distances, axis=1)
    
    # Combine constraints
    radii = np.minimum(wall_distances, min_distances)
    
    return radii

def neg_sum_radii(z):
    centers = z.reshape(-1, 2)
    radii = compute_radii(centers)
    return -np.sum(radii)

    # Differential Evolution

        neg_sum_radii,
        bounds,
        maxiter=2000,
        popsize=10,
        tol=1e-7,
        polish=False,
        strategy='best1bin',
        seed=42
    )

    # Extract best center positions from DE
    best_centers_flat = result.x
    best_centers = best_centers_flat.reshape(-1, 2)
    
    # Apply hill-climbing (CMA-ES style)

        # Evaluate

        new_sum = -neg_sum_radii(new_flat)
        
        if new_sum > best_sum:
            best_centers = new_centers
            best_sum = new_sum
            sigma = min(1.1 * sigma, 0.5)  # Increase step size

def npMinimum(a, b, axis=None):
    return np.minimum(a, b)
