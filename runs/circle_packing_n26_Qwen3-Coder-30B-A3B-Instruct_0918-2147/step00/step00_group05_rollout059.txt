import numpy as np
from scipy.optimize import minimize

def hex_lattice_points(n, d):
    """Generate hexagonal lattice points in unit square."""
    rows = []
    y = 0.0
    row = 0
    while len(rows) < n:
        x_start = (row % 2) * d / 2
        x = x_start
        while x <= 1.0:
            rows.append((x, y))
            x += d
        y += d * np.sqrt(3) / 2
        row += 1
        if y > 1.0:
            break
    return np.array(rows[:n])

    points += [0.5, 0.5]

    r_wall = 1.0
    for x, y in points:

    for x, y in points:
        vars.extend([x, y, r_uniform])

def objective(vars):
    """Minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def wall_constraints(vars):
    """Return wall constraints (all ≤ 0)."""
    n = len(vars) // 3
    constraints = []
    for i in range(n):
        x, y, r = vars[3*i], vars[3*i+1], vars[3*i+2]
        constraints.extend([
            r - x,           # x - r >= 0
            x + r - 1,       # x + r <= 1
            r - y,           # y - r >= 0
            y + r - 1        # y + r <= 1
        ])
    return np.array(constraints)

def nonoverlap_constraints(vars):
    """Return non-overlap constraints (all ≤ 0)."""
    n = len(vars) // 3
