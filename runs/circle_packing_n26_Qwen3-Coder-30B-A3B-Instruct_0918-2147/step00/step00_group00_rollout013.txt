import numpy as np
from scipy.optimize import minimize

def make_hex_lattice(n, margin=0.05):
    """Generate a hexagonal lattice of n points within the unit square."""
    # The hexagonal packing is most efficient for equal circles
    # The horizontal spacing is 1.0, vertical is sqrt(3)/2
    dx = 1.0 / (np.ceil(np.sqrt(n)) + 0.5)
    dy = dx * np.sqrt(3) / 2

    # Generate a sufficiently dense grid
    rows = np.ceil(1.0 / dy) + 2
    cols = np.ceil(1.0 / dx) + 2

    points = []
    for j in range(int(rows)):
        for i in range(int(cols)):
            x = i * dx + (j % 2) * dx / 2
            y = j * dy
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append([x, y])
            if len(points) >= n:
                break
        if len(points) >= n:
            break

    # Trim to exactly n points
    points = points[:n]
    points = np.array(points)
    
    # Center the pattern with margin
    x_min, x_max = np.min(points[:, 0]), np.max(points[:, 0])
    y_min, y_max = np.min(points[:, 1]), np.max(points[:, 1])
    
    cx = (x_min + x_max) / 2 
    cy = (y_min + y_max) / 2
    dx_shift = 0.5 - cx
    dy_shift = 0.5 - cy
    
    points[:, 0] += dx_shift
    points[:, 1] += dy_shift

    # Apply margin
    points[:, 0] = np.clip(points[:, 0], margin, 1 - margin)
    points[:, 1] = np.clip(points[:, 1], margin, 1 - margin)

    return points

def objective(z):
    """Objective function: -sum of radii."""
    n = len(z) // 3
    radii = z[2::3]
    return -np.sum(radii)

def boundary_constraints(z):
    """Returns inequality values of boundary constraints: positive means violation."""
    n = len(z) // 3
    out = []
    for i in range(n):
        x, y, r = z[3*i], z[3*i+1], z[3*i+2]
        # x - r >= 0
        out.append(x - r)
        # x + r <= 1  
        out.append(1 - x - r)
        # y - r >= 0
        out.append(y - r)
        # y + r <= 1
        out.append(1 - y - r)
    return np.array(out)

def pairwise_constraints(z):
    """Returns inequality values of pairwise separation constraints."""
    n = len(z) // 3
    out = []
    for i in range(n):
        for j in range(i + 1, n):
            x1, y1, r1 = z[3*i], z[3*i+1], z[3*i+2]
            x2, y2, r2 = z[3*j], z[3*j+1], z[3*j+2]
            dist_sq = (x1 - x2)**2 + (y1 - y2)**2
            r_sum = r1 + r2
            out.append(dist_sq - r_sum**2)
    return np.array(out)

def solve_nlp(initial_z):
    """Solve the NLP with inequality constraints."""
    # Combine ineq constraints
    cons = []
    cons.append({'type': 'ineq', 'fun': boundary_constraints})
    cons.append({'type': 'ineq', 'fun': pairwise_constraints})

    # Bounds for all variables: [x, y, r]
    bounds = []
    for i in range(len(initial_z) // 3):
        bounds.append((0.0, 1.0))  # x
        bounds.append((0.0, 1.0))  # y
        bounds.append((1e-8, 0.5)) # r

    # Solve using SLSQP
    res = minimize(objective, initial_z, method='SLSQP', constraints=cons,
                          bounds=bounds, options={'maxiter': 1000, 'ftol': 1e-9})
    if not res.success:
        raise RuntimeError("Optimization failed: " + res.message)
    return res.x

def nudge_centers(centers, radii):
    """Nudge centers of circles that are tight against boundaries."""
    n = centers.shape[0]
    # Create a copy to avoid editing in-place initially  
    new_centers = centers.copy()

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Check margin to boundaries
        dist_to_left = x - r
        dist_to_right = 1 - x - r
        dist_to_bottom = y - r  
        dist_to_top = 1 - y - r

        min_dist = min(dist_to_left, dist_to_right, dist_to_bottom, dist_to_top)
        if min_dist < 1e-5:  # Tight against boundary
            # Nudge away from the closest boundary
            if dist_to_left == min_dist:
                new_centers[i, 0] += 0.001
            elif dist_to_right == min_dist:
                new_centers[i, 0] -= 0.001 
            elif dist_to_bottom == min_dist:
                new_centers[i, 1] += 0.001
            else:  # top
                new_centers[i, 1] -= 0.001
    return new_centers

def run_packing():
    # 1. Initial lattice
    centers = make_hex_lattice(26, margin=0.05)
    radii = np.full(26, 0.02, dtype=float)
    
    # 2. Initial guess vector: [x0, y0, r0, x1, y1, r1, ...]
    initial_z = np.zeros(3 * 26)
    for i in range(26):
        initial_z[3*i] = centers[i, 0]
        initial_z[3*i+1] = centers[i, 1]
        initial_z[3*i+2] = radii[i]

    # 3. Solve with optimization
    final_z = solve_nlp(initial_z)
    
    # Extract results
    final_centers = np.zeros((26, 2))
    final_radii = np.zeros(26)
    for i in range(26):
        final_centers[i, 0] = final_z[3*i]
        final_centers[i, 1] = final_z[3*i+1]
        final_radii[i] = final_z[3*i+2]

    # Iterative refinement
    for _ in range(5):
        # Update the lattice points (via nudging)
        final_centers = nudge_centers(final_centers, final_radii)
        
        # Reassemble to z vector for NLP
        test_z = np.zeros(3 * 26)
        for i in range(26):
            test_z[3*i] = final_centers[i, 0]
            test_z[3*i+1] = final_centers[i, 1]
            test_z[3*i+2] = final_radii[i]
        
        # Final solve
        final_z = solve_nlp(test_z)
        
        # Extract new radii
        for i in range(26):
            final_radii[i] = final_z[3*i+2]
            
        # Termination check
        total = np.sum(final_radii)
        if total - np.sum(radii) < 1e-5:
            break
        radii = final_radii.copy()
