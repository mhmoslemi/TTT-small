import numpy as np
from scipy.optimize import linprog, minimize
from math import sqrt, ceil

def build_hex_lattice(d):
    points = []
    spacing = d / sqrt(3)
    rows = ceil(1.0 / spacing) + 1
    cols = ceil(1.0 / (spacing * sqrt(3))) + 1
    for i in range(rows):
        y = i * spacing
        if y > 1:
            break
        for j in range(cols):
            x = j * spacing * sqrt(3)
            if i % 2 == 1:
                x += spacing * sqrt(3) / 2
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append([x, y])
    return np.array(points[:26])

def lp_max_radii(centers):
    n = centers.shape[0]
    # Build LP: maximize sum of radii subject to:
    #   r_i <= x_i, 1-x_i, y_i, 1-y_i (wall constraints)
    #   r_i + r_j <= dist(c_i, c_j) (non-overlap constraints)
    A_ub = []
    b_ub = []

    # Wall constraints (4 per circle)
    for i in range(n):
        x, y = centers[i]
        A_ub.extend([
            [1, 0, 0, 0],   # r_i <= x_i
            [-1, 0, 0, 0],  # r_i <= 1-x_i
            [0, 1, 0, 0],   # r_i <= y_i
            [0, -1, 0, 0]   # r_i <= 1-y_i
        ])
        b_ub.extend([x, 1-x, y, 1-y])

    # Pairwise non-overlap constraints
    for i in range(n):
        for j in range(i+1, n):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist = sqrt(dx*dx + dy*dy)
            A_ub.extend([[-1, 0, -1, 0], [0, -1, 0, -1]])
            b_ub.extend([dist, dist])
            # for k in range(4*(i*n + j)):
            #   A_ub[k][i] = -1
            #   A_ub[k][j] = -1
            A_ub.append([1, 0, 1, 0])
            A_ub.append([0, 1, 0, 1])
            b_ub.append(dist)
            b_ub.append(dist)

    # Convert to np arrays with correct number of variables
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    c = np.ones(n)
    
    # Make sure the dimensions match
    num_vars = n
    padded_A_ub = np.zeros((A_ub.shape[0], num_vars))
    padded_A_ub[:, :A_ub.shape[1]] = A_ub
    A_ub = padded_A_ub
    
    # Solve the linear program
    result = linprog(-c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'tol': 1e-9})
    if result.success:
        return result.x, -result.fun
    else:
        # Return all zeros if not successful; should not happen in valid configurations
        return np.zeros(n), 0.0

def perturb_centers(centers, sigma):
    new_centers = centers + np.random.normal(0, sigma, centers.shape)
    new_centers = np.clip(new_centers, 0, 1)
    return new_centers

def anneal_centers(initial_centers):
    np.random.seed(42)
    current_centers = initial_centers.copy()
    best_centers = initial_centers.copy()
    best_sum = 0.0
    
    # Annealing schedule
    temperatures = [0.02, 0.01, 0.005, 0.002]
    sigmas = [0.05, 0.03, 0.015, 0.008]
    steps = [2000, 4000, 3000, 2000]

    for temp, sigma, step_count in zip(temperatures, sigmas, steps):
        for step in range(step_count):
            candidate_centers = perturb_centers(current_centers, sigma)
            radii, sum_radii = lp_max_radii(candidate_centers)
            if sum_radii > best_sum:
                best_sum = sum_radii
                best_centers = candidate_centers.copy()
            if sum_radii >= best_sum:
                current_centers = candidate_centers.copy()
            else:
                acceptance_prob = np.exp((sum_radii - best_sum) / temp)
                if np.random.rand() < acceptance_prob:
                    current_centers = candidate_centers.copy()
    return best_centers, best_sum
