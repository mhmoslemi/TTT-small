import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
from math import sqrt

def generate_lattice(s):
    """Generate a hexagonal lattice with spacing s in [0,1]x[0,1]."""
    sqrt3 = sqrt(3)
    rows = int(1.0 / (s * sqrt3 / 2)) + 1
    cols = int(1.0 / s) + 1
    points = []
    for r in range(rows):
        y = r * s * sqrt3 / 2
        if y + s/2 > 1:
            continue
        for c in range(cols):
            if r % 2 == 0:
                x = c * s
            else:
                x = (c + 0.5) * s
            if x + s/2 > 1:
                continue
            if x - s/2 >= 0 and y - s/2 >= 0:
                points.append([x, y])
    return np.array(points)

def build_milp(P):
    """Build a cvxpy optimization problem for the MILP."""
    K = len(P)
    z = cp.Variable(K, boolean=True)
    r = cp.Variable(K, nonneg=True)
    # wall distances
    wL = P[:, 0]
    wR = 1 - P[:, 0]
    wB = P[:, 1]
    wT = 1 - P[:, 1]
    w = np.minimum(np.minimum(wL, wR), np.minimum(wB, wT))
    # distance matrix
    D = np.zeros((K, K))
    for i in range(K):
        for j in range(i+1, K):
            d = np.linalg.norm(P[i] - P[j])
            D[i, j] = d
            D[j, i] = d
    # constraints
    constraints = [
        cp.sum(z) == 26,
        r <= w * z,
    ]
    # overlap constraints
    M = 1.0
    for i in range(K):
        for j in range(i+1, K):
            constraints.append(D[i, j] >= r[i] + r[j] - M * (2 - z[i] - z[j]))
    # objective
    objective = cp.Maximize(cp.sum(r))
    # solver settings
    prob = cp.Problem(objective, constraints)
    def solve_node(z_fixed):
        for k in range(K):
            if z_fixed[k] is not None:
                z[k].value = z_fixed[k]
            else:
                z[k].value = None
        try:
            prob.solve(solver=cp.ECOS, verbose=False, feastol=1e-9, reltol=1e-9)
        except:
            return None, None
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return r.value, prob.value
        return None, None
    return solve_node

def branch_and_bound(milp_solver, node_limit=20000):
    """Custom branch and bound solver."""
    from heapq import heappush, heappop
    from collections import deque
    best_obj = -np.inf
    best_z = None
    best_r = None
    pq = []
    root_z = [None]*len(milp_solver.__code__.co_consts)
    obj, ub = milp_solver(root_z)
    heappush(pq, (-ub, 0, root_z))
    node_count = 0
    while pq and node_count < node_limit:
        _, depth, z_curr = heappop(pq)
        node_count += 1
        # Find first unfixed variable
        var_idx = -1
        for i in range(len(z_curr)):
            if z_curr[i] is None:
                var_idx = i
                break
        if var_idx == -1:  # All fixed
            # Check feasibility
            r_vals, obj_val = milp_solver(z_curr)
            if obj_val is None:
                continue
            if obj_val > best_obj:
                best_obj = obj_val
                best_z = list(z_curr)
                best_r = r_vals
            continue
        # Try True (1)
        z_new1 = list(z_curr)
        z_new1[var_idx] = 1
        r1, ub1 = milp_solver(z_new1)
        if r1 is not None and -ub1 > best_obj:
            heappush(pq, (-ub1, depth+1, z_new1))
        # Try False (0)
        z_new0 = list(z_curr)
        z_new0[var_idx] = 0
        r0, ub0 = milp_solver(z_new0)
        if r0 is not None and -ub0 > best_obj:
            heappush(pq, (-ub0, depth+1, z_new0))
    return best_z, best_r

def refine_solution(centers, radii):
    """Use local optimization to improve centers and radii."""
    n = len(centers)
    # Helper function to compute distances
    def dist_sq(a, b):
        return np.sum((a - b)**2)
    # Objective function
    def objective(xr):
        c = xr[:2*n].reshape(-1, 2)
        r = xr[2*n:]
        obj = -np.sum(r)
        return obj
    # Constraints
    def con_wall(xr):
        C = []
        c = xr[:2*n].reshape(-1, 2)
        r = xr[2*n:]
        for i in range(n):
            C.append(c[i, 0] - r[i])  # left
            C.append(1 - c[i, 0] - r[i])  # right
            C.append(c[i, 1] - r[i])  # bottom
            C.append(1 - c[i, 1] - r[i])  # top
        return np.array(C)
    # Non-overlap
    def con_overlap(xr):
        C = []
        c = xr[:2*n].reshape(-1, 2)
        r = xr[2*n:]
        for i in range(n):
            for j in range(i+1, n):
                dist_sq_ij = dist_sq(c[i], c[j])
                sep = (r[i] + r[j])**2
                C.append(sep - dist_sq_ij)
        return np.array(C)
    # Initial guess
    x0 = np.concatenate([centers.flatten(), radii])
    cons = [
        {'type': 'ineq', 'fun': con_wall},
        {'type': 'ineq', 'fun': con_overlap},
    ]
    # Optimize using SLSQP
    res = minimize(objective, x0, method='SLSQP', constraints=cons,
                   options={'ftol': 1e-9, 'maxiter': 2000})
    if not res.success:
        res = minimize(objective, x0, method='SLSQP', constraints=cons,
                       options={'ftol': 1e-12, 'maxiter': 5000})
    if not res.success:
        return centers, radii, np.sum(radii)
    # Extract solution

def run_packing():

    P = generate_lattice(s=0.13)
    # Build solver
    milp_solver = build_milp(P)
    # Branch-and-bound to get integer solution

    selected_indices = [i for i in range(len(z_sol)) if z_sol[i] == 1]
