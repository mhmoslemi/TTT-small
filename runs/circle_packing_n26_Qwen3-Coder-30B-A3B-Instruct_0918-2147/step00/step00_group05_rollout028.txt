import numpy as np
from scipy.optimize import minimize

def hex_lattice_points(n, d):
    """Generate at least n points in a hexagonal lattice with spacing d."""
    rows = []
    y = 0.0
    row = 0
    while len(rows) < n:
        row += 1
        x_offset = (row % 2) * d / 2.0
        x = x_offset
        while x <= 1.0 + d:
            rows.append((x, y))
            x += d
        y += d * np.sqrt(3) / 2.0
    return np.array(rows[:n])

def initial_guess():
    """Generate initial layout using hexagonal lattice."""
    d = 0.2  # lattice spacing
    points = hex_lattice_points(26, d)
    centers = points[:26]

    margin = d / 2.0
    radii = np.full(26, margin)
    
    # Adjust to make sure no circle is outside the square
    for i in range(26):
        x, y = centers[i]
        r = radii[i]
        r_safe = min(x, 1 - x, y, 1 - y)
        radii[i] = min(r, r_safe)
    
    # Flatten centers and radii to 1D vector
    vars_flat = []
    for i in range(26):
        vars_flat.extend([centers[i, 0], centers[i, 1], radii[i]])
    return np.array(vars_flat)

def objective(vars):
    """Objective function: minimize negative of sum of radii."""
    return -np.sum(vars[2::3])

def wall_constraints(vars):
    """Return list of wall inequality constraints."""
    constraints = []
    for i in range(26):
        x = vars[3*i]
        y = vars[3*i+1]
        r = vars[3*i+2]
        # x - r >= 0
        constraints.append(r - x)
        # x + r <= 1
        constraints.append(x + r - 1)
        # y - r >= 0
        constraints.append(r - y)
        # y + r <= 1
        constraints.append(y + r - 1)
    return constraints

def nonoverlap_constraints(vars):
    """Return list of non-overlap inequality constraints."""

            constraints.append(dist_sq - sum_radii_sq)
