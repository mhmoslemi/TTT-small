import numpy as np
from scipy.optimize import minimize
from ortools.linear_solver import pywraplp
import itertools

def generate_candidates(g, radius_levels):
    candidates = []
    grid_points = np.arange(0, 1 + g/2, g)
    for x in grid_points:
        for y in grid_points:
            for r in radius_levels:
                if (x >= r and x <= 1 - r and y >= r and y <= 1 - r):
                    candidates.append((x, y, r))
    return candidates

def build_milp(candidates):
    solver = pywraplp.Solver.CreateSolver("SCIP")
    if not solver:
        raise RuntimeError("Failed to create solver")

    n = len(candidates)
    
    # Create binary variables
    z = [solver.BoolVar(f"z_{i}") for i in range(n)]

    # Each grid point at most one circle
    grid_dict = {}
    for i, (x, y, r) in enumerate(candidates):
        key = (round(x/g), round(y/g))
        if key not in grid_dict:
            grid_dict[key] = []
        grid_dict[key].append(i)

    for key, indices in grid_dict.items():
        solver.Add(solver.Sum([z[i] for i in indices]) <= 1)

    # Exactly 26 circles
    solver.Add(solver.Sum(z) == 26)

    # Non-overlap constraints (L-infinity linearization)
    big_M = 1.0
    dx_vars = {}
    dy_vars = {}

    for i, j in itertools.combinations(range(n), 2):
        xi, yi, ri = candidates[i]
        xj, yj, rj = candidates[j]
        
        dx = solver.NumVar(0, 1, f'dx_{i}_{j}')
        dy = solver.NumVar(0, 1, f'dy_{i}_{j}')
        dx_vars[(i,j)] = dx
        dy_vars[(i,j)] = dy
                
        solver.Add(dx >= xi - xj)
        solver.Add(dx >= xj - xi)
        solver.Add(dy >= yi - yj)
        solver.Add(dy >= yj - yi)
        
        solver.Add(dx + dy >= ri + rj - big_M * (1 - z[i] - z[j]))
        solver.Add(dx - dy >= ri + rj - big_M * (1 - z[i] - z[j]))

    # Objective
    solver.Maximize(solver.Sum([z[i] * candidates[i][2] for i in range(n)]))
    
    return solver, z

def solve_milp(solver, z):
    status = solver.Solve()
    if status != pywraplp.MPSolver.OPTIMAL:
        raise RuntimeError(f"MILP solver did not find optimal solution: {status}")
        
    selected_indices = [i for i in range(len(z)) if z[i].solution_value() > 0.5]
    
    return selected_indices

def extract_initial_solution(selected_indices, candidates):
    centers = []
    radii = []
    for i in selected_indices:
        x, y, r = candidates[i]
        centers.append([x, y])
        radii.append(r)
    return np.array(centers), np.array(radii)

def shrink_initial(radii, delta=1e-5):
    return np.maximum(radii - delta, 0.0)

def pack_variables(centers0, radii0):
    z0 = np.empty(3 * len(centers0))
    for i, (x, y) in enumerate(centers0):
        z0[3*i] = x
        z0[3*i+1] = y
        z0[3*i+2] = radii0[i]
    return z0

def define_objective(z):
    return -np.sum(z[2::3])

def define_jacobian(z):
    grad = np.zeros_like(z)
    grad[2::3] = -1.0
    return grad

def define_constraints(z, n_circles, centers0, radii0):
    # Initialize list of constraint dicts
    constrs = []
    
    # Ball constraints (boundary)
    for i in range(n_circles):
        x = z[3*i]
        y = z[3*i+1]
        r = z[3*i+2]
        def ball_constr():
            out = [
                x - r,           # x - r >= 0
                1 - x - r,       # (1 - x) - r >= 0
                y - r,           # y - r >= 0
                1 - y - r        # (1 - y) - r >= 0
            ]
            return out
        
        def ball_jac():
            J = np.zeros((4, 3*n_circles))
            J[0, 3*i] = 1.0
            J[0, 3*i+2] = -1.0
            J[1, 3*i] = -1.0
            J[1, 3*i+2] = -1.0
            J[2, 3*i+1] = 1.0
            J[2, 3*i+2] = -1.0
            J[3, 3*i+1] = -1.0
            J[3, 3*i+2] = -1.0
            return J

        constrs.append({
            'fun': ball_constr,
            'jac': ball_jac
        })
        
    # Pairwise non-overlap constraints
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            x1 = z[3*i]
            y1 = z[3*i+1]
            r1 = z[3*i+2]
            x2 = z[3*j]
            y2 = z[3*j+1]
            r2 = z[3*j+2]
            
            def pairwise_constr():
                dx = x1 - x2
                dy = y1 - y2
                dist = np.sqrt(dx*dx + dy*dy)
                f = (r1 + r2) - dist
                return f

            def pairwise_jac():
                dx = x1 - x2
                dy = y1 - y2
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < 1e-12:
                    dist = 1e-12
                df_dx = dx / dist
                df_dy = dy / dist
                
                J = np.zeros((1, 3*n_circles))
                J[0, 3*i] =  df_dx
                J[0, 3*i+1] =  df_dy
                J[0, 3*i+2] =  1.0
                J[0, 3*j] = -df_dx
                J[0, 3*j+1] = -df_dy
                J[0, 3*j+2] =  1.0
                return J
                
            constrs.append({
                'fun': pairwise_constr,
                'jac': pairwise_jac
            })

    return constrs

def run_packing():
    # Stage 1: discrete global search
    g = 0.04
    radius_levels = [0.02 * i for i in range(1, 13)]  # 0.02 to 0.24
    candidates = generate_candidates(g, radius_levels)
    
    solver, z = build_milp(candidates)
    selected_indices = solve_milp(solver, z)
    
    centers0, radii0 = extract_initial_solution(selected_indices, candidates)
    radii0 = shrink_initial(radii0)
    
    # Stage 2: continuous local refinement
    n_circles = len(centers0)
    z0 = pack_variables(centers0, radii0)
    
    constraints = define_constraints(z0, n_circles, centers0, radii0)
    bounds = [(0, 1) for _ in range(3*n_circles)]  # each of x,y,r has bounds [0,1]
    for i in range(n_circles):
        bounds[3*i+2] = (0, 0.5)  # r should be <= 0.5 because no circles can overlap
    
    def objective(z):
        return define_objective(z)
    
    def jacobian(z):
        return define_jacobian(z)

    # Optimize in parts - each constraint evaluates a scalar so we use a list of constraint dicts
    def fun_and_jac(z):
        f = objective(z)
        J = jacobian(z)
        return f, J
    
    # Run the optimization
    res = minimize(
        fun=objective,
        x0=z0,
        method='trust-constr',
        jac=jacobian,
        bounds=bounds,
        constraints=constraints,
        options={'gtol': 1e-9, 'xtol': 1e-9, 'maxiter': 1000}
    )
    
    if not res.success:
        raise RuntimeError("Optimization failed", res.message)
        
    # Unpack the solution
    centers = np.zeros((n_circles, 2))
    radii = np.zeros(n_circles)
    for i in range(n_circles):
        centers[i] = [res.x[3*i], res.x[3*i+1]]
        radii[i] = res.x[3*i+2]
 
    # Repair:
    # Boundary check
    max_r = np.minimum(centers[:,0], 1 - centers[:,0], centers[:,1], 1 - centers[:,1])
    radii = np.minimum(radii, max_r)
    
    # Overlap check
    for _ in range(10):
        any_overlap = False
        for i in range(n_circles):
            for j in range(i+1, n_circles):
                dx = centers[i,0] - centers[j,0]
                dy = centers[i,1] - centers[j,1]
                dist = np.sqrt(dx*dx + dy*dy)
                sum_r = radii[i] + radii[j]
                overlap = sum_r - dist
                
                if overlap > 1e-12:
                    any_overlap = True
                    # Reduce both radii
                    reduce = overlap / 2.0
                    if radii[i] > reduce and radii[j] > reduce:
                        radii[i] -= reduce
                        radii[j] -= reduce
                    else:
                        # If too small, reduce larger one more
                        if radii[i] >= radii[j]:
                            radii[i] -= overlap
                        else:
                            radii[j] -= overlap
                        
        if not any_overlap:
            break
    
    radii = np.maximum(radii, 0.0)
    
    sum_radii = np.sum(radii)

    # Validate
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise RuntimeError("Validation failed after optimization: ", msg)

    return (centers, radii, sum_radii)
