import numpy as np
from scipy.optimize import minimize
from ortools.linear_solver import pywraplp
import itertools

def generate_candidates(g, radius_levels):
    """Generate candidate circles with fixed centers on a grid."""
    candidates = []
    grid_points = np.arange(0, 1 + g/2, g)
    for x in grid_points:
        for y in grid_points:
            for r in radius_levels:
                if x - r >= 0 and x + r <= 1 and y - r >= 0 and y + r <= 1:
                    candidates.append((x, y, r))
    return candidates

def build_milp(candidates):
    """Build a MILP model for global layout selection."""
    solver = pywraplp.Solver.CreateSolver('CBC')
    if not solver:
        raise RuntimeError("CBC solver not available")

    n = len(candidates)
    z = [solver.BoolVar(f'z_{i}') for i in range(n)]

    # Each grid point has at most one circle
    grid_dict = {}
    for i, (x, y, r) in enumerate(candidates):
        key = (round(x/g), round(y/g))
        if key not in grid_dict:
            grid_dict[key] = []
        grid_dict[key].append(i)

    for indices in grid_dict.values():
        solver.Add(solver.Sum([z[i] for i in indices]) <= 1)

    # Exactly 26 circles selected
    solver.Add(solver.Sum(z) == 26)

    # Non-overlapping constraints using L-infinity approximation
    # We add constraints for all pairs
    for i, j in itertools.combinations(range(n), 2):
        x1, y1, r1 = candidates[i]
        x2, y2, r2 = candidates[j]

        dx = solver.NumVar(0, 1.0, f'dx_{i}_{j}')
        dy = solver.NumVar(0, 1.0, f'dy_{i}_{j}')

        solver.Add(dx >= x1 - x2)
        solver.Add(dx >= x2 - x1)
        solver.Add(dy >= y1 - y2)
        solver.Add(dy >= y2 - y1)

        # L-infinity distance >= r1 + r2
        solver.Add(dx + dy >= r1 + r2)
        solver.Add(dx - dy >= r1 + r2)  # Rotated
        solver.Add(-dx + dy >= r1 + r2)

        # Big-M constraints
        M = 1.0
        solver.Add(dx <= M * (z[i] + z[j]))
        solver.Add(dy <= M * (z[i] + z[j]))

    # Objective
    solver.Maximize(solver.Sum([r * z[i] for i, (_, _, r) in enumerate(candidates)]))

    return solver, z

def solve_milp(solver, z):
    """Solve MILP and return the selected indices."""
    status = solver.Solve()
    if status not in [pywraplp.Solver.OPTIMAL, pywraplp.Solver.FEASIBLE]:
        raise RuntimeError("MILP solver failed to find feasible solution")
    selected = [i for i in range(len(z)) if z[i].solution_value() > 0.5]
    return selected

def extract_initial_solution(selected_indices, candidates):
    """Extract initial configuration."""
    centers = np.array([[candidates[i][0], candidates[i][1]] for i in selected_indices])
    radii = np.array([candidates[i][2] for i in selected_indices])
    return centers, radii

def shrink_initial(radii0, delta=1e-5):
    """Prevent numerical issues by slightly shrinking radii."""
    return np.maximum(radii0 - delta, 1e-9)

def pack_variables(centers0, radii0):
    """Pack centers and radii into a vector."""
    n = centers0.shape[0]
    z0 = np.zeros(3 * n)
    z0[0::3] = centers0[:, 0]
    z0[1::3] = centers0[:, 1]
    z0[2::3] = radii0
    return z0

def unpack_solution(z):
    """Unpack a flattened solution into centers and radii."""
    n = len(z) // 3
    centers = np.reshape(z[0:3*n:3], (n, 2))
    radii = z[2::3]
    return centers, radii

def define_objective(z):
    """Compute the negative sum of radii."""
    return -np.sum(z[2::3])

def define_jacobian(z):
    """Compute the Jacobian (gradient) of objective."""
    grad = np.zeros_like(z)
    grad[2::3] = -1.0
    return grad

def define_constraints(z, centers0, radii0):
    """Define constraints for trust-constr."""
    n = len(z) // 3
    centers, radii = unpack_solution(z)
    constraints = []

    # Boundary constraints
    for i in range(n):
        # r <= x, 1-x, y, 1-y
        def bound_ineq(x, i):
            return (x[2 + 3*i] - x[0 + 3*i],  # r - x
                    x[2 + 3*i] - (1 - x[0 + 3*i]),  # r - (1-x)
                    x[2 + 3*i] - x[1 + 3*i],  # r - y
                    x[2 + 3*i] - (1 - x[1 + 3*i]))  # r - (1-y)

        def bound_jac(x, i):
            j = np.zeros(len(x))
            j[2 + 3*i] = 1.0
            j[0 + 3*i] = -1.0
            j[1 + 3*i] = -1.0
            return j

        # Each inequality is of the form x_i - x_j < 0
        # We compute r - x, r - (1-x), r - y, r - (1-y)
        # for simplicity just add 4 inequalities, ordered: (r-x), (r-(1-x)), (r-y), (r-(1-y))
        # But allow arbitrary signs to get sign comparison
        fvals = bound_ineq(z, i)
        jacs = bound_jac(z, i)
        # Each constraint is: r - x <= 0    =>  fval = r - x    =>   j = [0,0,-1,0,1,0]
        for k, (val, jac_row) in enumerate(zip(fvals, [jacs] * 4)):
            constraints.append({
                'fun': lambda x, k=k: val,
                'jac': lambda x, k=k: jac_row,
                'type': 'ineq',
                'ub': 0.0
            })

    # Overlap constraints (Euclidean)
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx ** 2 + dy ** 2)
            r_sum = radii[i] + radii[j]
            gap = r_sum - dist

            # Constraint is: (r_i + r_j)^2 - dist^2 >= 0
            # Or equivalently: freq <= 0 where freq = dist - r_sum <= 0
            # We compute d_dist / dz = [dx, dy, dx, dy] for gradient
            def overlap_f(x, i=i, j=j):
                c_i = x[0 + 3*i:2 + 3*i]
                c_j = x[0 + 3*j:2 + 3*j]
                r_i = x[2 + 3*i]
                r_j = x[2 + 3*j]
                dx = c_i[0] - c_j[0]
                dy = c_i[1] - c_j[1]
                dist = np.sqrt(dx*dx + dy*dy)
                return dist - (r_i + r_j)

            def overlap_jac(x, i=i, j=j):
                c_i = x[0 + 3*i:2 + 3*i]
                c_j = x[0 + 3*j:2 + 3*j]
                r_i = x[2 + 3*i]
                r_j = x[2 + 3*j]
                dx = c_i[0] - c_j[0]
                dy = c_i[1] - c_j[1]
                dist = np.sqrt(dx*dx + dy*dy)
                # Correct sign for consistency with implicit equality: dist - (r_i + r_j) <= 0
                d_dist = np.array([dx, dy, dx, dy]) / (dist + 1e-12)
                j = np.zeros_like(x)
                j[0 + 3*i] = d_dist[0]
                j[1 + 3*i] = d_dist[1]
                j[0 + 3*j] = -d_dist[2]
                j[1 + 3*j] = -d_dist[3]
                j[2 + 3*i] = -1.0
                j[2 + 3*j] = -1.0
                return -j  # Flip sign for inequality to be <= 0

            constraints.append({
                'fun': lambda x, i=i, j=j: overlap_f(x, i, j),
                'jac': lambda x, i=i, j=j: overlap_jac(x, i, j),
                'type': 'ineq',
                'ub': 0.0
            })

    return constraints

def run_packing():
    g = 0.03
    radius_levels = np.arange(0.02, 0.25, 0.02)
    candidates = generate_candidates(g, radius_levels)

    # Stage 1: Solve MILP
    solver, z_vars = build_milp(candidates)
    selected_indices = solve_milp(solver, z_vars)

    # Extract starting configuration
    centers0, radii0 = extract_initial_solution(selected_indices, candidates)
    radii0 = shrink_initial(radii0)

    # Stage 2: Local optimization using trust-constr
    z0 = pack_variables(centers0, radii0)

    def objective(z):
        return define_objective(z)

    def jac(z):
        return define_jacobian(z)

    bounds = [(0, 1), (0, 1), (0, 0.5)] * len(centers0)
    constraints = define_constraints(z0, centers0, radii0)

    result = minimize(objective, z0, method='trust-constr', jac=jac, bounds=bounds, constraints=constraints)

    if not result.success:
        raise RuntimeError("Optimization did not converge")

    centers, radii = unpack_solution(result.x)

    # Repair small numerical violations
    for i in range(len(centers)):
        radii[i] = min(radii[i], centers[i][0], 1 - centers[i][0], centers[i][1], 1 - centers[i][1])
        if radii[i] < 0:
            radii[i] = 1e-9

    # Re-ensure no overlaps
    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):
            dist = np.sqrt((centers[i][0] - centers[j][0])**2 + (centers[i][1] - centers[j][1])**2)
            r_sum = radii[i] + radii[j]
            if dist < r_sum - 1e-9:
                gap = r_sum - dist
                # Reduce both radii
                radii[i] = radii[i] - 0.5 * gap
                radii[j] = radii[j] - 0.5 * gap

    sum_radii = np.sum(radii)

    # Final validate
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise RuntimeError(f"Validation failed: {msg}")

    return centers, radii, sum_radii
