import numpy as np
from scipy.optimize import minimize
import scipy.sparse as sp

def init_hex_grid(n=26, r_init=0.04):
    # Generate hexagonal grid points
    h = np.sqrt(3) * r_init
    rows = []
    y = r_init
    row_num = 0
    while len(rows) < n:
        if row_num % 2 == 0:
            x_start = r_init
            step = 2 * r_init
        else:
            x_start = 2 * r_init
            step = 2 * r_init
        row = []
        x = x_start
        while x <= 1 - r_init and len(row) < n - len(rows):
            row.append([x, y])
            x += step
        if row:
            rows.extend(row)
        y += h
        row_num += 1
    # Trim to exactly n points and create arrays
    centers = np.array(rows[:n])
    radii = np.full(n, r_init)
    return centers, radii

def max_radii(centers, radii):
    n = centers.shape[0]
    max_iter = 100
    eps = 1e-12
    for _ in range(max_iter):
        # Compute wall limits
        wall_limits = np.minimum(centers, 1 - centers).min(axis=1)

        # Compute distance matrix
        diff = centers[:, None, :] - centers[None, :, :]
        D = np.sqrt(np.sum(diff ** 2, axis=2))

        # Compute neighbor limits
        # Set diagonal to large value to ignore self-distance
        D[np.arange(n), np.arange(n)] = 1e10
        neighbor_limits = np.min(D - radii[None, :], axis=1)

        # Compute new radii
        new_radii = np.minimum(wall_limits, neighbor_limits)
        new_radii = np.maximum(new_radii, 0)

        # Check for convergence
        if np.max(np.abs(new_radii - radii)) < 1e-8:
            break

        radii = new_radii

    return radii

def hill_climb(centers, radii, iterations=5000, step=0.01):
    n = centers.shape[0]
    current_sum = np.sum(radii)
    for _ in range(iterations):
        # Random order
        idxs = np.random.permutation(n)
        # Try each circle
        for i in idxs:
            # Try small random displacement
            delta = np.random.uniform(-step, step, 2)
            new_centers = centers.copy()
            new_centers[i] += delta
            # Check bounds
            if (new_centers[i, 0] - radii[i] < -1e-12 or 
                new_centers[i, 0] + radii[i] > 1 + 1e-12 or 
                new_centers[i, 1] - radii[i] < -1e-12 or 
                new_centers[i, 1] + radii[i] > 1 + 1e-12):
                continue
            # Perform inner loop on this modified layout
            old_radii = radii.copy()
            try:
                radii = max_radii(new_centers, radii)
                new_sum = np.sum(radii)
                if new_sum > current_sum:
                    centers = new_centers
                    current_sum = new_sum
                else:
                    radii = old_radii
            except Exception:
                radii = old_radii
    return centers, radii

def refine_with_scipy(centers, radii):
    n = centers.shape[0]
    # Build variables vector: [x1,y1,...,xn,yn,r1,...,rn]
    x0 = np.concatenate([centers.flatten(), radii])

    # Define constraints
    def wall_constraints(v):
        # Extract all variables
        c = v[:2*n].reshape((n,2))
        r = v[2*n:]
        # Wall constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
        # i.e. max(0, r_i - x_i) <= 0, max(0, r_i - (1-x_i)) <= 0, etc.
        # But in form of inequality ( <= 0 ), we return these:
        # r - x, r - (1-x), r - y, r - (1-y)
        constraints = np.empty(4*n)
        constraints[0::4] = r - c[:, 0]  # r - x
        constraints[1::4] = r - (1 - c[:, 0])  # r -(1-x)
        constraints[2::4] = r - c[:, 1]  # r - y
        constraints[3::4] = r - (1 - c[:, 1])  # r -(1-y)
        return constraints

    def overlap_constraints(v):
        # Extract all variables
        c = v[:2*n].reshape((n,2))
        r = v[2*n:]
        # Distance matrix
        diff = c[:, None, :] - c[None, :, :]
        D = np.sqrt(np.sum(diff ** 2, axis=2))
        # Non-overlap constraints: r_i + r_j - ||c_i - c_j|| <= 0
        # We need to return a vector of all such constraints
        constraints = []
        for i in range(n):
            for j in range(i+1, n):
                constraints.append(r[i] + r[j] - D[i, j])
        return np.array(constraints)

    # Objective: -sum(r)
    def objective(v):
        return -np.sum(v[2*n:])

    # Create constraints as scipy constraints
    n_wall = 4*n
    n_overlap = n*(n-1)//2
    # For wall constraints
    def wall_jacobian(v):
        c = v[:2*n].reshape((n,2))
        r = v[2*n:]
        # Jacobian is (n_wall, 2*n + n) = (4*n, 3*n)
        J = sp.coo_matrix((4*n, 3*n))
        # Fill in
        rows = []
        cols = []
        data = []
        idx = 0
        # r - x
        for i in range(n):
            rows.extend([idx, idx])
            cols.extend([2*i + 2*n, 2*i])
            data.extend([1.0, -1.0])
            idx += 1
        # r - (1-x)
        for i in range(n):
            rows.extend([idx, idx])
            cols.extend([2*i + 2*n, 2*i])
            data.extend([1.0, 1.0])
            idx += 1
        # r - y
        for i in range(n):
            rows.extend([idx, idx])
            cols.extend([2*i + 2*n, 2*i + 1])
            data.extend([1.0, -1.0])
            idx += 1
        # r - (1-y)
        for i in range(n):
            rows.extend([idx, idx])
            cols.extend([2*i + 2*n, 2*i + 1])
            data.extend([1.0, 1.0])
            idx += 1
        J = J.tocsr()
        J.data[:len(data)] = data
        J.indices[:len(cols)] = cols
        J.indptr[:len(rows) + 1] = np.cumsum([0] + [1]*len(rows))
        return J

    def overlap_jacobian(v):
        # Positions and radii
        c = v[:2*n].reshape((n,2))
        r = v[2*n:]
        # Compute distances
        diff = c[:, None, :] - c[None, :, :]
        D = np.sqrt(np.sum(diff ** 2, axis=2))
        # Jacobian size: n_overlap x (2*n + n)
        n_col = 2*n + n
        J = sp.coo_matrix((n_overlap, n_col))
        rows = []
        cols = []
        data = []
        idx = 0
        for i in range(n):
            for j in range(i+1, n):
                d = D[i, j]
                if d > 1e-12:
                    # For x_i, y_i
                    dx = diff[i, j, 0] / d
                    dy = diff[i, j, 1] / d
                    # For x_j, y_j
                    dxj = -dx
                    dyj = -dy
                    # Row idx in J
                    row = idx
                    # For (r_i + r_j - D)
                    # Gradients w.r.t. x_i, y_i
                    rows.extend([row, row])
                    cols.extend([2*i, 2*i + 1])
                    data.extend([dx, dy])
                    # Gradients w.r.t. x_j, y_j
                    rows.extend([row, row])
                    cols.extend([2*j, 2*j + 1])
                    data.extend([dxj, dyj])
                    # Gradients w.r.t r_i, r_j
                    rows.extend([row, row])
                    cols.extend([2*n + i, 2*n + j])
                    data.extend([1.0, 1.0])
                    idx += 1
        # Finalize sparse matrix
        J = J.tocsr()
        J.data[:len(data)] = data
        J.indices[:len(cols)] = cols
        J.indptr[:len(rows) + 1] = np.cumsum([0] + [1]*len(rows))
        return J

    wall_con = NonlinearConstraint(wall_constraints, -np.inf, 0, jac=wall_jacobian)
    overlap_con = NonlinearConstraint(overlap_constraints, -np.inf, 0, jac=overlap_jacobian)

    # Bounds: 0 < x_i < 1, 0 < y_i < 1, 0 < r_i < 0.5
    bounds = []
    for i in range(2*n):
        bounds.append((0.0, 1.0))
    for i in range(n):
        bounds.append((0.0, 0.5))

    # Call minimize
    res = minimize(objective, x0, method='trust-constr', jac=True,
                   constraints=[wall_con, overlap_con], bounds=bounds,
                   options={'maxiter': 200, 'verbose': 0})

    if res.success:
        final_centers = res.x[:2*n].reshape((n,2))
        final_radii = res.x[2*n:]
        return final_centers, final_radii
    else:
        # If optimization fails, return original result
        return centers, radii

def run_packing():
    centers, radii = init_hex_grid()
    radii = max_radii(centers, radii)
    centers, radii = hill_climb(centers, radii)
    centers, radii = refine_with_scipy(centers, radii)
    sum_r = float(radii.sum())
    return centers, radii, sum_r
