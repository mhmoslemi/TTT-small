import numpy as np
from scipy.optimize import linprog
from scipy.spatial.distance import cdist

def generate_hex_lattice(h):
    sqrt3 = np.sqrt(3)
    points = []
    for j in range(int(1.0 / (h * sqrt3 / 2)) + 1):
        y = j * h * sqrt3 / 2
        if y > 1.0:
            break
        for i in range(int(1.0 / h) + 1):
            x = i * h + (j % 2) * (h / 2)
            if x > 1.0:
                break
            points.append([x, y])
    return np.array(points)

def build_neighbour_list(centers, max_dist):
    dist_matrix = cdist(centers, centers)
    edges = []
    n = centers.shape[0]
    for i in range(n):
        for j in range(i + 1, n):
            d = dist_matrix[i, j]
            if d < max_dist:
                edges.append((i, j, d))
    return edges

def solve_radius_lp(selected_indices, R_wall, neighbour_list):
    n_selected = len(selected_indices)
    if n_selected == 0:
        return np.array([]), True
    # To avoid index issues, we map selected indices to [0, n_selected-1]
    idx_map = {orig: i for i, orig in enumerate(selected_indices)}
    # Variables
    c = np.ones(n_selected)  # maximize sum of radii
    # Bounds
    bounds = [(0, R_wall[i]) for i in selected_indices]
    # Inequality constraints A_ub * x <= b_ub
    A_ub = []
    b_ub = []
    for i, j, d_ij in neighbour_list:
        if i in idx_map and j in idx_map:
            row = [0] * n_selected
            row[idx_map[i]] = 1
            row[idx_map[j]] = 1
            A_ub.append(row)
            b_ub.append(d_ij)
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    # Solve
    res = linprog(-c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, True
    else:
        return np.array([]), False

def upper_bound(partial_set, remaining_indices, R_wall):
    ub_est = sum(R_wall[i] for i in partial_set)
    needed = 26 - len(partial_set)
    if needed <= 0:
        return ub_est
    remaining_radii = [R_wall[i] for i in remaining_indices]
    remaining_radii.sort(reverse=True)
    ub_est += sum(remaining_radii[:needed])
    return ub_est

def branch_and_bound(centers, R_wall, neighbour_list):
    best_solution = None
    best_sum = -np.inf
    # Use a stack to simulate recursion
    # Each element is (S_partial, U, lb, ub)
    # S_partial: list of selected indices
    # U: list of unselected candidates
    # lb: best sum found in subtree
    # ub: current optimistic upper bound
    stack = [([], list(range(centers.shape[0])), -np.inf, np.inf)]
    while stack:
        S_partial, U, lb, ub = stack.pop()
        if ub <= best_sum:
            continue
        if len(S_partial) == 26:
            radii, feasible = solve_radius_lp(S_partial, R_wall, neighbour_list)
            if feasible:
                total = np.sum(radii)
                if total > best_sum:
                    best_sum = total
                    best_solution = (S_partial[:], radii.copy())
            continue
        # heuristics for branching: pick the one with highest wall radius
        if not U:
            continue
        candidate_indices = sorted(U, key=lambda idx: R_wall[idx], reverse=True)
        c = candidate_indices[0]
        U_new = [idx for idx in U if idx != c]
        # Exclude c
        S_partial_new = S_partial[:]
        if S_partial_new:
            # Compute a quick feasibility check for current partial set + c
            new_set_with_c = S_partial_new + [c]
            # Try a small LP just with wall constraints
            _, is_feasible = solve_radius_lp(new_set_with_c, R_wall, neighbour_list)
            if not is_feasible:
                pass  # Don't add c
            else:
                ub_c = upper_bound(S_partial_new, U_new, R_wall)
                if ub_c > best_sum:
                    stack.append((S_partial_new, U_new, lb, ub_c))
        else:
            U_new_no_c = U_new[:]
            ub_no_c = upper_bound(S_partial, U_new_no_c, R_wall)
            if ub_no_c > best_sum:
                stack.append((S_partial, U_new_no_c, lb, ub_no_c))
        # Include c
        S_partial_incl = S_partial + [c]
        # Add a quick check of infeasibility for current partial + c
        _, is_feasible = solve_radius_lp(S_partial_incl, R_wall, neighbour_list)
        if not is_feasible:
            continue
        ub_incl = upper_bound(S_partial_incl, U_new, R_wall)
        if ub_incl > best_sum:
            stack.append((S_partial_incl, U_new, lb, ub_incl))
    return best_solution

def local_continuous_refinement(centers, radii):
    # start from current discrete solution. We optimize over centers + radii
    from scipy.optimize import minimize
    def objective(radii_centers):
        num_centers = centers.shape[0]
        rad = radii_centers[num_centers:]
        return -np.sum(rad)

    def constraint_fun(radii_centers):
        num_centers = centers.shape[0]
        locs = radii_centers[:num_centers * 2].reshape((num_centers, 2))
        rad = radii_centers[num_centers * 2:]
        # Containment
        xi = locs[:, 0]
        yi = locs[:, 1]
        r = rad

        contain_bounds = np.concatenate([
            xi - r, 1 - xi - r,
            yi - r, 1 - yi - r])
        # Non-overlap
        dists = cdist(locs, locs)
        nonovlp = []
        for i in range(num_centers):
            for j in range(i + 1, num_centers):
                d = dists[i, j]
                nonovlp.append(d - r[i] - r[j])
        return np.concatenate([contain_bounds, nonovlp])

    def constraint_fun_jac(radii_centers):
        # return gradient of constraint function:
        from scipy.sparse import coo_array
        # not needed yet
        return np.zeros((0, len(radii_centers)))

    num_centers = centers.shape[0]
    # initial guess
    initial_guess = np.concatenate([centers.flatten(), radii])
    # Add small random perturbation to centers
    initial_guess[:num_centers * 2] += np.random.uniform(-0.002, 0.002, size=num_centers * 2)
    # bounds: 0 <= x,y <= 1, 0 <= r
    bounds = [(0, 1) for _ in range(2 * num_centers)] + [(0, None) for _ in range(num_centers)]
    # constraints
    con = {'type': 'ineq', 'fun': constraint_fun}
    # minimize
    result = minimize(objective, initial_guess, method='trust-constr', bounds=bounds, constraints=con, options={'maxiter': 200})
    if not result.success:
        # fallback to original
        return centers, radii
    new_centers = result.x[:num_centers * 2].reshape((num_centers, 2))
    new_radii = result.x[num_centers * 2:]
    return new_centers, new_radii

    R_wall = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),

        raise RuntimeError("Branch and bound failed to find a solution")
    selected_indices, radii = best_solution
    selected_centers = centers[selected_indices]
