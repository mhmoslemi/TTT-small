import numpy as np
from scipy.optimize import minimize
from ortools.linear_solver import pywraplp
import itertools

def generate_candidates(g, radius_levels):
    candidates = []
    grid_points = []
    for x in np.arange(0, 1 + g/2, g):
        for y in np.arange(0, 1 + g/2, g):
            grid_points.append((x, y))

    for x, y in grid_points:
        for r in radius_levels:
            if (x - r >= 0 and x + r <= 1 and y - r >= 0 and y + r <= 1):
                candidates.append((x, y, r))
    
    return candidates

def build_milp(candidates):
    num_candidates = len(candidates)
    solver = pywraplp.Solver.CreateSolver('SCIP')
    if not solver:
        solver = pywraplp.Solver.CreateSolver('CBC')
    
    # Binary variables for selecting candidates
    z = [solver.IntVar(0, 1, f'z_{i}') for i in range(num_candidates)]
    
    # Constraint: Each grid point has at most one circle
    grid_dict = {}
    for i, (x, y, r) in enumerate(candidates):
        grid_key = (round(x/g), round(y/g))
        if grid_key not in grid_dict:
            grid_dict[grid_key] = []
        grid_dict[grid_key].append(i)
    
    for grid_key, indices in grid_dict.items():
        solver.Add(sum(z[i] for i in indices) <= 1)
    
    # Constraint: Exactly 26 circles
    solver.Add(sum(z) == 26)
    
    # Constraint: Non-overlapping
    for i, j in itertools.combinations(range(num_candidates), 2):
        xi, yi, ri = candidates[i]
        xj, yj, rj = candidates[j]
        
        # Use L-infinity distance approximation
        dx = solver.NumVar(0, 1, f'dx_{i}_{j}')
        dy = solver.NumVar(0, 1, f'dy_{i}_{j}')
        
        solver.Add(dx >= xi - xj)
        solver.Add(dx >= xj - xi)
        solver.Add(dy >= yi - yj)
        solver.Add(dy >= yj - yi)
        
        solver.Add(dx + dy >= ri + rj - 1e-6)
        solver.Add(dx - dy >= ri + rj - 1e-6)
        solver.Add(dx - dy <= -(ri + rj - 1e-6))
        # Linking constraint to only enforce when both selected
        M = 1.0
        solver.Add(dx <= M * (z[i] + z[j]))
        solver.Add(dy <= M * (z[i] + z[j]))
    
    # Objective: maximize sum of radii
    solver.Maximize(sum(candidates[i][2] * z[i] for i in range(num_candidates)))
    
    return solver, z

def solve_milp(solver, z):
    status = solver.Solve()
    selected_indices = [i for i in range(len(z)) if z[i].solution_value() > 0.5]
    return selected_indices

def extract_initial_solution(selected_indices, candidates):
    centers = np.array([[candidates[i][0], candidates[i][1]] for i in selected_indices])
    radii = np.array([candidates[i][2] for i in selected_indices])
    return centers, radii

def shrink_initial(radii, delta):
    return radii - delta

def pack_variables(centers, radii):
    return np.hstack([centers.flatten(), radii])

def unpack_solution(z_opt):
    n = len(z_opt) // 3
    centers = z_opt[:2*n].reshape(-1, 2)
    radii = z_opt[2*n:]
    return centers, radii

def define_objective(z):
    return -np.sum(z[2*n:])

def define_jacobian(z):
    n = len(z) // 3
    jac = np.full(len(z), -1.0)  # gradient of sum of radii - as we minimize -sum
    return jac

def define_constraints(z):
    n = len(z) // 3
    centers = z[:2*n].reshape(-1, 2)
    radii = z[2*n:]
    
    constraints = []
    # Boundary constraints
    for i in range(n):
        constraints.append({
            'fun': lambda z, i=i: -radii[i] + centers[i, 0],
            'jac': lambda z, i=i: np.concatenate([np.zeros(2*n), -np.eye(n)[i]])
        })
        constraints.append({
            'fun': lambda z, i=i: -radii[i] + (1 - centers[i, 0]),
            'jac': lambda z, i=i: np.concatenate([np.zeros(2*n), -np.eye(n)[i]])
        })
        constraints.append({
            'fun': lambda z, i=i: -radii[i] + centers[i, 1],
            'jac': lambda z, i=i: np.concatenate([np.zeros(2*n), -np.eye(n)[i]])
        })
        constraints.append({
            'fun': lambda z, i=i: -radii[i] + (1 - centers[i, 1]),
            'jac': lambda z, i=i: np.concatenate([np.zeros(2*n), -np.eye(n)[i]])
        })
    
    # Pairwise non-overlap constraints
    for i in range(n):
        for j in range(i+1, n):
            def pairwise_constraint(z, i=i, j=j):
                x1, y1 = centers[i, 0], centers[i, 1]
                x2, y2 = centers[j, 0], centers[j, 1]
                r1 = radii[i]
                r2 = radii[j]
                dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
                return (r1 + r2) - dist
                
            def pairwise_jac(z, i=i, j=j):
                x1, y1 = centers[i, 0], centers[i, 1]
                x2, y2 = centers[j, 0], centers[j, 1]
                r1 = radii[i]
                r2 = radii[j]
                dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2) + 1e-12
                dx = (x1 - x2) / dist
                dy = (y1 - y2) / dist
                jac = np.zeros(len(z))
                jac[2*i] = -dx
                jac[2*i + 1] = -dy
                jac[2*j] = dx
                jac[2*j + 1] = dy
                jac[2*n + i] = -1
                jac[2*n + j] = -1
                return jac
                
            constraints.append({
                'fun': pairwise_constraint,
                'jac': pairwise_jac
            })
    
    return constraints

def run_packing():
    g = 0.04
    radius_levels = np.arange(0.02, 0.25, 0.02)
    
    candidates = generate_candidates(g, radius_levels)
    
    solver, z = build_milp(candidates)
    selected_indices = solve_milp(solver, z)
    centers0, radii0 = extract_initial_solution(selected_indices, candidates)
    
    # Ensure compactness by shrinking radii slightly
    radii0 = shrink_initial(radii0, 1e-5)
    
    # Pack into vector for optimization
    z0 = pack_variables(centers0, radii0)
    
    # Use trust-constr to refine
    constraints = define_constraints(z0)
    bounds = [(0, 1)] * (2 * len(centers0)) + [(0, 0.5)] * len(radii0)
    
    result = minimize(
        fun=define_objective,
        x0=z0,
        method='trust-constr',
        jac=define_jacobian,
        bounds=bounds,
        constraints=constraints,
        options={'gtol': 1e-9, 'xtol': 1e-9}
    )
    
    centers, radii = unpack_solution(result.x)
    
    # Repair violations
    for i in range(len(centers)):
        radii[i] = min(radii[i], centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
        
    # Repair overlapping
    for i in range(len(centers)):
        for j in range(i+1, len(centers)):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            total_r = radii[i] + radii[j]
            if dist < total_r + 1e-9:
                overlap = total_r - dist
                radii[i] -= overlap / 2
                radii[j] -= overlap / 2
                radii[i] = max(0, radii[i])
                radii[j] = max(0, radii[j])
    
    # Final validation
    assert validate_packing(centers, radii)[0], "Final packing is not valid"
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
