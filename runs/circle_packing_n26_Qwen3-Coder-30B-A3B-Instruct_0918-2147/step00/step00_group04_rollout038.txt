import numpy as np

def build_leaf_problem(cell, k):
    """Build SOCP for a leaf cell with k circles"""
    x_min, y_min, x_max, y_max = cell
    x = cp.Variable((k, 2))
    r = cp.Variable(k)
    
    # Initialize constraints
    constraints = []
    
    # Boundary constraints
    for i in range(k):
        constraints.append(x[i, 0] - r[i] >= x_min)
        constraints.append(x[i, 0] + r[i] <= x_max)
        constraints.append(x[i, 1] - r[i] >= y_min)
        constraints.append(x[i, 1] + r[i] <= y_max)
        
    # Non-overlap constraints
    for i in range(k):
        for j in range(i+1, k):
            dist = cp.norm(x[i] - x[j])
            constraints.append(dist >= r[i] + r[j])
    
    # Objective
    objective = cp.Maximize(cp.sum(r))
    
    # Create problem
    prob = cp.Problem(objective, constraints)
    return prob, x, r

def solve_leaf(cell, k):
    """Solve leaf cell subproblem with k circles"""
    if k == 0:
        return np.array([]), np.array([]), []
    prob, x, r = build_leaf_problem(cell, k)
    prob.solve(solver=cp.ECOS, verbose=False)
    return x.value, r.value, prob
