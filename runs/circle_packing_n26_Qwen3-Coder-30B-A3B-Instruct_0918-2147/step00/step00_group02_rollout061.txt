import numpy as np
from scipy.optimize import minimize
from scipy.stats import qmc

def initial_configuration(n=26):
    # Generate Sobol sequence for initial seed centres
    sampler = qmc.Sobol(d=2, seed=42)
    points = sampler.random(n)
    
    centers = points.copy()
    radii = np.zeros(n)
    
    # Compute initial radii
    for i in range(n):
        x, y = centers[i]
        r_wall = min(x, 1-x, y, 1-y)
        
        # Find distance to nearest neighbour
        min_dist = np.inf
        for j in range(n):
            if i != j:
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                min_dist = min(min_dist, dist)
                
        r_neighbour = min_dist / 2.0
        radii[i] = min(r_wall, r_neighbour)
    
    # Ensure radii are non-negative
    radii = np.maximum(radii, 0.0)
    
    return centers, radii

def pack(centers, radii):
    # Flatten into a single array: [x0, y0, r0, x1, y1, r1, ...]
    z = np.empty(3 * len(centers))
    for i in range(len(centers)):
        z[3*i] = centers[i, 0]
        z[3*i+1] = centers[i, 1]
        z[3*i+2] = radii[i]
    return z

def unpack(z):
    n = len(z) // 3
    centers = np.empty((n, 2))
    radii = np.empty(n)
    for i in range(n):
        centers[i, 0] = z[3*i]
        centers[i, 1] = z[3*i+1]
        radii[i] = z[3*i+2]
    return centers, radii

def max_violation(centers, radii):
    n = len(centers)
    
    # Check wall constraints
    max_wall_violation = 0.0
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        wall_violation = max(0.0, r - x, r - (1 - x), r - y, r - (1 - y))
        max_wall_violation = max(max_wall_violation, wall_violation)
    
    # Check overlap constraints  
    max_overlap_violation = 0.0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            overlap = max(0.0, radii[i] + radii[j] - dist)
            max_overlap_violation = max(max_overlap_violation, overlap)
    
    return max(max_wall_violation, max_overlap_violation)

def energy_and_grad(z):
    np.seterr(all='ignore')
    centers, radii = unpack(z)
    n = len(centers)
    
    # Setup
    E = 0.0
    dE = np.zeros_like(z)
    eps = 1e-12
    
    # Wall repulsion energy and gradient
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Wall distances (min of distance to each wall)
        dx_left = x
        dx_right = 1 - x
        dy_down = y
        dy_up = 1 - y
        
        # Compute energy and gradient contribution per wall
        # Left wall
        if r > dx_left:
            E += (r - dx_left) ** 2
            dE[3*i+2] += 2 * (r - dx_left)
            dE[3*i] -= 2 * (r - dx_left)
            
        # Right wall
        if r > dx_right:
            E += (r - dx_right) ** 2
            dE[3*i+2] += 2 * (r - dx_right)
            dE[3*i] += 2 * (r - dx_right)
            
        # Down wall
        if r > dy_down:
            E += (r - dy_down) ** 2
            dE[3*i+2] += 2 * (r - dy_down)
            dE[3*i+1] -= 2 * (r - dy_down)
            
        # Up wall
        if r > dy_up:
            E += (r - dy_up) ** 2
            dE[3*i+2] += 2 * (r - dy_up)
            dE[3*i+1] += 2 * (r - dy_up)
            
    # Pairwise overlap energy and gradient
    lambda_pair = 1e5  # Penalty weight
    for i in range(n):
        for j in range(i + 1, n):
            dij = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            r_sum = radii[i] + radii[j]
            overlap = max(0.0, r_sum - dij)
            
            if overlap > 0.0:
                E += overlap ** 2
                
                # This contributes to gradients for both i and j
                dE[3*i+2] += 2 * lambda_pair * overlap
                dE[3*j+2] += 2 * lambda_pair * overlap
                
                # Gradient w.r.t. centres
                if dij > eps:
                    factor = 2 * lambda_pair * overlap / dij
                    dx = (centers[i] - centers[j]) * factor
                    dE[3*i] -= dx[0]
                    dE[3*i+1] -= dx[1]
                    dE[3*j] += dx[0]
                    dE[3*j+1] += dx[1]
    
    # Budget term (enforce that we use all radius budget)
    B = np.sum(radii)  # Current sum
    dE[3*i+2] -= 2 * lambda_pair * (B - 3.0)  # Just to make sure equilibrium point is at B=3 for trial setup
    
    return E, dE

def inner_optimize(z0, B, lambda_budget=1e5, maxiter=1000):
    def obj_and_grad(z):
        E, dE = energy_and_grad(z)
        return E, dE
    
    bounds = [(0, 1) for _ in range(len(z0))]
    for i in range(0, len(z0), 3):
        bounds[i] = (0, 1)       # x
        bounds[i+1] = (0, 1)     # y
        bounds[i+2] = (0, 0.5)   # r
    
    res = minimize(obj_and_grad, z0, method='L-BFGS-B', bounds=bounds,
                   jac=True, options={'maxiter': maxiter, 'ftol': 1e-10, 'gtol': 1e-10})
    
    return res.x

def max_feasible_budget(z_start, B0=3.0):
    B = B0
    step = 5e-4
    lambda_budget = 1e5
    maxiter = 1000
    
    z_current = z_start
    
    # Simple binary search approach to find best B
    for _ in range(100):
        z_new = inner_optimize(z_current, B, lambda_budget, maxiter)
        centers_new, radii_new = unpack(z_new)
        
        # Check feasibility with tolerance slightly tighter than validator
        violation = max_violation(centers_new, radii_new)
        
        if violation <= 1e-13:
            B += step
        else:
            B -= step
            
        z_current = z_new
        if np.abs(step) < 1e-6:
            break
    
    return z_current, B

def postprocess(z):
    centers, radii = unpack(z)
    eps = 5e-10
    
    # Shrink radii slightly
    radii = np.maximum(0.0, radii - eps)
    
    # Clamp centres
    centers[:, 0] = np.clip(centers[:, 0], eps, 1 - eps)
    centers[:, 1] = np.clip(centers[:, 1], eps, 1 - eps)
    
    # Apply final repair pass
    for _ in range(5):
        changed = False
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                required_sum = radii[i] + radii[j]
                if dist < required_sum:
                    overlap = required_sum - dist
                    shrink = overlap / 2.0 + eps
                    if radii[i] > shrink and radii[j] > shrink:
                        radii[i] -= shrink
                        radii[j] -= shrink
                        changed = True
                    elif radii[i] > shrink:
                        radii[i] -= shrink
                        changed = True
                    elif radii[j] > shrink:
                        radii[j] -= shrink
                        changed = True
        if not changed:
            break
    
    # Final validity check
    violation = max_violation(centers, radii)
    if violation > 1e-12:
        raise RuntimeError(f"Postprocessing failed, violation = {violation}")
        
    return centers, radii

def run_packing():
    centers, radii = initial_configuration(26)
    
    z_init = pack(centers, radii)
    z_final, _ = max_feasible_budget(z_init)
    centers_final, radii_final = postprocess(z_final)
    
    sum_radii = np.sum(radii_final)
    
    return centers_final, radii_final, sum_radii
