import numpy as np
from scipy.spatial.distance import cdist

def penalty(z, alpha=10.0, beta=10.0):
    n = 26
    centers = z[:2*n].reshape(n, 2)
    radii = z[2*n:]
    
    # Wall penalties
    wall_pen = 0.0
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        wall_pen += alpha * max(0, r - x)
        wall_pen += alpha * max(0, r - (1 - x))
        wall_pen += alpha * max(0, r - y)
        wall_pen += alpha * max(0, r - (1 - y))
    
    # Overlap penalties
    dists = cdist(centers, centers)
    for i in range(n):
        for j in range(i + 1, n):
            dist = dists[i, j]
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                wall_pen += beta * overlap
    
    return wall_pen


def energy(z, lam, alpha=10.0, beta=10.0):
    sum_radii = np.sum(z[2*26:])
    pen = penalty(z, alpha, beta)
    return -sum_radii + lam * pen


def propose_move(z, sigma_c, sigma_r):
    n = 26
    z_new = z.copy()
    move_type = np.random.randint(3)
    
    if move_type == 0:  # Center jitter
        idx = np.random.randint(n)
        z_new[2*idx] += np.random.normal(0, sigma_c)
        z_new[2*idx + 1] += np.random.normal(0, sigma_c)
        z_new[2*idx] = np.clip(z_new[2*idx], 0, 1)
        z_new[2*idx + 1] = np.clip(z_new[2*idx + 1], 0, 1)
    elif move_type == 1:  # Radius perturb
        idx = np.random.randint(n)
        z_new[2*n + idx] += np.random.normal(0, sigma_r)
        z_new[2*n + idx] = max(0, z_new[2*n + idx])
    elif move_type == 2:  # Swap radii
        idx1, idx2 = np.random.choice(n, 2, replace=False)
        r1 = z_new[2*n + idx1]
        r2 = z_new[2*n + idx2]
        z_new[2*n + idx1] = r2
        z_new[2*n + idx2] = r1
    
    return z_new


def anneal():
    n = 26
    # Generate initial configuration on a coarse hexagonal grid
    spacing = 0.18
    rows = int(1.0 / spacing) + 1
    cols = int(1.0 / spacing) + 1
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = (i + 0.5) * spacing
            y = (j + 0.5) * spacing
            if x <= 1 and y <= 1:
                centers.append([x, y])
    
    # Ensure we have exactly 26 centers, just in case
    while len(centers) < 26:
        centers.append([np.random.uniform(0., 1.), np.random.uniform(0., 1.)])
    centers = np.array(centers[:26])
    
    # Assign initial radii based on wall distances
    radii = []
    for i in range(26):
        x, y = centers[i]
        r = min(x, 1 - x, y, 1 - y)
        radii.append(r)
    
    radii = np.array(radii)
    
    # Set up the initial state vector: [x0, y0, ..., x25, y25, r0, ..., r25]
    z = np.concatenate([centers.flatten(), radii])
    
    # Simulated Annealing parameters
    T0 = 1.0
    c_T = 0.995
    T = T0
    lam = 0.1
    c_lam = 1.03
    sigma_c = 0.03
    sigma_r = 0.02
    max_iter = 10000
    k = 0
    best_feasible_z = None
    best_feasible_sum = -np.inf
    best_sum = -np.inf
    best_z = z.copy()
    
    # Annealing loop
    while T > 1e-5 and k < max_iter:
        T *= c_T
        lam *= c_lam
        sigma_c_temp = sigma_c * np.sqrt(T / T0)
        sigma_r_temp = sigma_r * np.sqrt(T / T0)
        
        z_prop = propose_move(z, sigma_c_temp, sigma_r_temp)
        delta_E = energy(z_prop, lam) - energy(z, lam)
        if delta_E <= 0 or np.random.rand() < np.exp(-delta_E / T):
            z = z_prop
        
        # Check if z is feasible
        if penalty(z) < 1e-8:
            sum_radii = np.sum(z[2*26:])
            if sum_radii > best_feasible_sum:
                best_feasible_sum = sum_radii
                best_feasible_z = z.copy()
        
        sum_radii = np.sum(z[2*26:])
        if sum_radii > best_sum:
            best_sum = sum_radii
            best_z = z.copy()
        
        k += 1
    
    if best_feasible_z is not None:
        return best_feasible_z[:2*26].reshape(26, 2), best_feasible_z[2*26:]
    else:
        # Fallback to best solution found
        return best_z[:2*26].reshape(26, 2), best_z[2*26:]


def project_and_ascent(centers, radii):
    n = 26
    eta = 5e-3
    max_iter = 500
    decay = 0.95
    min_delta = 1e-7
    
    # Continue until divergence or maximum iterations
    for _ in range(max_iter):
        old_sum = np.sum(radii)
        
        # Gradient-ascent step: increase all radii by rotation of 1
        radii = radii + eta * 1.0
        
        # Apply projection constraints
        # 1. Wall constraints
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            r = min(r, x, 1 - x, y, 1 - y)
            radii[i] = max(0, r)
        
        # 2. Overlap constraints
        dists = cdist(centers, centers)
        for i in range(n):
            for j in range(i + 1, n):
                dist = dists[i, j]
                overlap = radii[i] + radii[j] - dist
                if overlap > 0:
                    # Reduce both radii
                    reduction = overlap / 2
                    if radii[i] >= reduction:
                        radii[i] -= reduction
                    else:
                        # If one of the radii goes negative we just reduce the other
                        reduction = radii[i]
                        radii[i] = 0
                        radii[j] -= reduction
                    if radii[j] >= reduction:
                        radii[j] -= reduction
                    else:
                        # Ensure we don't go negative
                        radii[j] = 0
                    
                    # Re-clamp to wall limits  
                    x, y = centers[i]
                    r = radii[i]
                    radii[i] = min(r, x, 1 - x, y, 1 - y)
                    radii[i] = max(0, radii[i])
                    x, y = centers[j]
                    r = radii[j]
                    radii[j] = min(r, x, 1 - x, y, 1 - y)
                    radii[j] = max(0, radii[j])
        
        new_sum = np.sum(radii)
        if new_sum - old_sum < min_delta:
            break
        
        eta *= decay
    
    return centers, radii
