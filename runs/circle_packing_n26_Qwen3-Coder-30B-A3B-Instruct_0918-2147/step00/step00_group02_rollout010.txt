import numpy as np
from scipy.optimize import minimize
from scipy.stats import qmc

def initial_configuration(n=26):
    # Generate Sobol sequence for initial centres
    sampler = qmc.Sobol(d=2, scramble=True)
    points = sampler.random(n)
    centers = points.copy()
    
    # Compute initial radii based on distances to walls and nearest neighbour
    radii = np.zeros(n)
    for i in range(n):
        x, y = centers[i]
        min_dist_to_wall = min(x, 1 - x, y, 1 - y)
        min_dist_to_neighbour = float('inf')
        for j in range(n):
            if i != j:
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < min_dist_to_neighbour:
                    min_dist_to_neighbour = dist
        radii[i] = min(min_dist_to_wall, min_dist_to_neighbour * 0.5)
    
    # Ensure all radii are valid
    radii = np.maximum(radii, 1e-10)
    
    return centers, radii

def pack(centers, radii):
    # Flatten to a 1-D vector
    return np.hstack([centers.flatten(), radii])

def unpack(z):
    # Unpack from flattened vector
    n = len(z) // 3  # 26 centres + 26 radii
    centers = z[:2*n].reshape((n, 2))
    radii = z[2*n:]
    return centers, radii

def energy_and_grad(z):
    centers, radii = unpack(z)
    n = centers.shape[0]
    
    # Compute distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    distances = np.sqrt(np.sum(diff**2, axis=2))
    
    # Wall energy and gradient
    wall_energy = 0.0
    wall_grad = np.zeros_like(z)
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Wall forces
        dx = max(0, r - x)
        dy = max(0, r - (1 - x))
        dz = max(0, r - y)
        dw = max(0, r - (1 - y))
        
        wall_energy += dx**2 + dy**2 + dz**2 + dw**2
        
        # Gradient w.r.t. (x_i, y_i, r_i)
        if dx > 0:
            wall_grad[2*i] -= 2 * dx  # dE/dx
            wall_grad[2*i + 1] -= 2 * dz  # dE/dy 
        if dy > 0:
            wall_grad[2*i] -= 2 * dy  # dE/dx
            wall_grad[2*i + 1] -= 2 * dw  # dE/dy
        if dz > 0:
            wall_grad[2*i + 1] -= 2 * dz  # dE/dy
        if dw > 0:
            wall_grad[2*i + 1] -= 2 * dw  # dE/dy
        
        # dE/dr
        if dx > 0:
            wall_grad[2*n + i] += 2 * dx
        if dy > 0:
            wall_grad[2*n + i] += 2 * dy
        if dz > 0:
            wall_grad[2*n + i] += 2 * dz
        if dw > 0:
            wall_grad[2*n + i] += 2 * dw

    # Pairwise overlap and gradient
    pair_energy = 0.0
    pair_grad = np.zeros_like(z)
    
    lambda_budget = 1e3  # penalty for the radius budget
    total_radius = np.sum(radii)
    budget_energy = lambda_budget * (1.0 - total_radius)**2
    budget_grad = -2 * lambda_budget * (1.0 - total_radius) * np.ones(n)
    
    for i in range(n):
        for j in range(i + 1, n):
            d = distances[i, j]
            r_sum = radii[i] + radii[j]
            
            overlap = max(0, r_sum - d)
            pair_energy += overlap**2

            # Gradient w.r.t. neighbors
            if overlap > 0:
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                if d > 1e-12:
                    scale = 2*overlap / d
                    pair_grad[2*i] -= scale * dx
                    pair_grad[2*j] += scale * dx
                    pair_grad[2*i + 1] -= scale * dy
                    pair_grad[2*j + 1] += scale * dy
                else:
                    # Avoid division by zero
                    pair_grad[2*i] -= 2*overlap * 1e-12
                    pair_grad[2*j] += 2*overlap * 1e-12
                    pair_grad[2*i + 1] -= 2*overlap * 1e-12
                    pair_grad[2*j + 1] += 2*overlap * 1e-12
            
            # Gradient w.r.t. radii
            pair_grad[2*n + i] += 2 * overlap
            pair_grad[2*n + j] += 2 * overlap

    # Combine all components
    total_energy = wall_energy + pair_energy + budget_energy
    total_grad = wall_grad + pair_grad

    # Include the budget gradient as dE/dr
    total_grad[2*n:] += budget_grad

    return total_energy, total_grad

def inner_optimize(z0, B, lambda_budget=1e3):
    def obj_and_grad(z):
        return energy_and_grad(z)

    bounds = []
    n = len(z0) // 3
    # Bounds for x and y
    for i in range(n):
        bounds.append((0, 1))  # x_i
        bounds.append((0, 1))  # y_i
    # Bounds for radii
    for i in range(n):
        bounds.append((1e-10, 0.5))  # r_i

    res = minimize(
        obj_and_grad,
        z0,
        method='L-BFGS-B',
        bounds=bounds,
        jac=True,
        options={'maxiter': 200, 'gtol': 1e-10, 'ftol': 1e-12}
    )
    return res.x

def max_feasible_budget(z_start, B=3.0, step=5e-4, max_iter=50):
    z = z_start.copy()
    B_current = B

    for _ in range(max_iter):
        z_optimized = inner_optimize(z, B_current)
        centers, radii = unpack(z_optimized)
        violation = np.max([
            np.max(np.maximum(0, radii - centers[:, 0])),
            np.max(np.maximum(0, radii - (1 - centers[:, 0]))),
            np.max(np.maximum(0, radii - centers[:, 1])),
            np.max(np.maximum(0, radii - (1 - centers[:, 1]))),
            np.max([radii[i] + radii[j] - np.linalg.norm(centers[i] - centers[j]) 
                    for i in range(len(centers)) for j in range(i + 1, len(centers))]) - 1e-12
        ])
        if violation <= 1e-13:
            # Feasible: increase budget
            B_current += step
            z = z_optimized
        else:
            # Infeasible: decrease budget
            B_current -= step

        if step < 1e-6:
            break
        if B_current <= 0:
            # Less likely if starting with B=3.0
            break

    return z

def postprocess(z):
    centers, radii = unpack(z)
    
    # Clip and shrink
    eps = 5e-10
    centers = np.clip(centers, eps, 1 - eps)
    radii = np.maximum(radii - eps, 0)
    
    # Post-repair pass
    n = len(centers)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    distances = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            d = distances[i, j]
            r_sum = radii[i] + radii[j]
            overlap = max(0, r_sum - d)
            if overlap > 0:
                # Reduce one of the radii to fix overlap (here, the larger one)
                # We leave it simple and reduce both equally for robustness
                delta = overlap / 2 + eps
                radii[i] = max(0, radii[i] - delta)
                radii[j] = max(0, radii[j] - delta)
                
    sum_r = np.sum(radii)

    return centers, radii, sum_r

def run_packing():
    centers, radii = initial_configuration(26)
    z = pack(centers, radii)
    z_opt = max_feasible_budget(z, B=3.0, step=5e-4, max_iter=40)
    centers, radii, sum_r = postprocess(z_opt)
    return centers, radii, sum_r
