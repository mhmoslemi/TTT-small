import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
from itertools import combinations

def init_candidate_grid(R, Δ):
    """Generate candidate (center, radius) pairs with grid spacing based on radius."""
    candidates = []
    for r in R:
        spacing = r * np.sqrt(2)
        x_coords = np.arange(r, 1 - r + 1e-10, spacing)
        y_coords = np.arange(r, 1 - r + 1e-10, spacing)
        for x in x_coords:
            for y in y_coords:
                candidates.append(((x, y), r))
    return candidates

def build_overlap_matrix(candidates):
    """Build boolean overlap matrix for candidate pairs."""
    n = len(candidates)
    overlap = np.zeros((n, n), dtype=bool)
    for i in range(n):
        (x1, y1), r1 = candidates[i]
        for j in range(i+1, n):
            (x2, y2), r2 = candidates[j]
            dist_sq = (x1 - x2)**2 + (y1 - y2)**2
            if dist_sq < (r1 + r2)**2 - 1e-12:
                overlap[i, j] = overlap[j, i] = True
    return overlap

def solve_discrete_bip(candidates, overlap_matrix, target_sum):
    """Solve for discrete candidate set using integer program."""
    from scipy.optimize import milp
    from scipy.optimize import BilinearConstraint
    n = len(candidates)
    # Binary variables z_k for selecting each candidate
    # Objective: maximize sum of radii
    # Constraints:
    # 1. exactly 26 selected
    # 2. no overlaps: z_i + z_j <= 1 if they overlap

    # Encode as ILP: z in {0,1}^n, sum z_i = 26, z_i + z_j <= 1 for overlapping i,j
    # Use a greedy heuristic as warm-start
    selected = set()
    radii = [r for _, r in candidates]
    centers = [c for c, _ in candidates]

    # Greedy: pick largest radii that don't conflict
    sorted_indices = np.argsort(radii)[::-1]
    used = np.zeros(n, dtype=bool)
    for idx in sorted_indices:
        if used[idx]:
            continue
        selected.add(idx)
        used |= overlap_matrix[idx]
        if len(selected) == 26:
            break

    # Ensure we have at least 26
    if len(selected) < 26:
        raise ValueError("Not enough non-overlapping candidates found")

    # Use all 26 selected indices we've chosen
    selected_indices = list(selected)[:26]  # take one 26-element subset

    c = -np.array([radii[i] for i in selected_indices])  # minimize negative
    # Intended model:
    # maximize c^T * z
    # subject to sum(z) = 26
    #         z_i + z_j <= 1 if overlap[i, j] = True

    # Implement a local optimization (find small subset that matches or exceeds target)
    # Since ILP struggle in current Python environments, we resort to exhaustive search with optimization.

    # Indexes of selected candidates
    selected_indices = np.array(selected_indices)
    selected_centers = np.array([centers[i] for i in selected_indices])
    selected_radii = np.array([radii[i] for i in selected_indices])
    total_sum = np.sum(selected_radii)
    if total_sum < target_sum - 1e-10:
        raise RuntimeError("Greedy with radius set leaves too little sum")

    # Return discrete solution that's good enough
    return selected_centers, selected_radii

def max_radii_fixed_centres(centers, radii_initial):
    """Project to maximum allowed radii for fixed centers."""
    n = centers.shape[0]
    radii_new = radii_initial.copy()
    for i in range(n):
        x, y = centers[i]
        r_old = radii_new[i]
        # Compute minimum distance to walls
        dist_to_wall = min(x, 1-x, y, 1-y)
        # Compute minimum distance to other centers
        dist_to_other = np.inf
        for j in range(n):
            if i == j:
                continue
            dist = np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            if dist < dist_to_other:
                dist_to_other = dist
        # Now the new radius is = min of distance to wall and distance to other centers / 2
        # But need to ensure it doesn't overlap with others
        max_radius = min(dist_to_wall, dist_to_other/2)
        if max_radius > 0:
            radii_new[i] = max_radius
        else:
            radii_new[i] = r_old  # No change if negative
    return radii_new

def nlp_refine(centers_initial, radii_initial):
    """Refine final solution with continuous optimization using trust-constr."""
    # Decision vector: [x1, y1, x2, y2, ..., x26, y26, r1, ..., r26]
    n = 26
    x0 = np.hstack([centers_initial.reshape(-1), radii_initial])

    def obj_fn(x):
        r = x[-n:]
        return -np.sum(r)  # minimize negative

    def grad_obj(x):
        r = x[-n:]
        return np.hstack([np.zeros(2*n), -np.ones(n)])

    def con_wall(x):
        # Wall constraints: r_i - x_i <= 0 (for shift x and y)
        # (x_i - r_i >= 0 => x_i >= r_i)
        # (x_i + r_i <= 1 => x_i <= 1 - r_i)
        # (y_i - r_i >= 0 => y_i >= r_i)
        # (y_i + r_i <= 1 => y_i <= 1 - r_i)
        centers = x[:2*n].reshape(-1, 2)
        radii = x[2*n:]
        res = []
        for i in range(n):
            xi, yi = centers[i]
            ri = radii[i]
            res.append(ri - xi)           # x_i >= r_i
            res.append(ri + xi - 1)       # x_i <= 1 - r_i
            res.append(ri - yi)           # y_i >= r_i
            res.append(ri + yi - 1)       # y_i <= 1 - r_i
        return np.array(res)

    def con_overlap(x):
        centers = x[:2*n].reshape(-1, 2)
        radii = x[2*n:]
        res = []
        for i in range(n):
            for j in range(i+1, n):
                dist = np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
                overlap = radii[i] + radii[j] - dist
                res.append(overlap)
        return np.array(res)

    def jac_wall(x):
        # zero because r is not involved in policy
        J = np.zeros((4*n, 2*n + n))
        for i in range(n):
            row = [2*i, 2*i+1]  # x, y indices related to constraints
            J[4*i, 2*i] = -1   # d(r - x)/dx = -1
            J[4*i+1, 2*i] = 1  # d(r + x - 1)/dx = 1
            J[4*i+2, 2*i+1] = -1 # d(r - y)/dy = -1
            J[4*i+3, 2*i+1] = 1  # d(r + y - 1)/dy = 1
        return J

    def jac_overlap(x):
        centers = x[:2*n].reshape(-1, 2)
        radii = x[2*n:]
        J = np.zeros((n*(n-1)//2, 2*n + n))
        idx = 0
        for i in range(n):
            for j in range(i+1, n):
                dist = np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
                if dist < 1e-12:
                    # Avoid division by zero, symmetry helps
                    diff_x = 0
                    diff_y = 0
                else:
                    diff_x = (centers[i, 0] - centers[j, 0]) / dist
                    diff_y = (centers[i, 1] - centers[j, 1]) / dist
                # derivatives for constraint (r_i + r_j - dist)
                J[idx, 2*i] = -diff_x
                J[idx, 2*i+1] = -diff_y
                J[idx, 2*j] = diff_x
                J[idx, 2*j+1] = diff_y
                J[idx, 2*n + i] = 1.0
                J[idx, 2*n + j] = 1.0
                idx += 1
        return J

    # Constraints defining nonlinear system:
    constraints = [
        {'type': 'ineq', 'fun': con_wall, 'jac': jac_wall},
        {'type': 'ineq', 'fun': con_overlap, 'jac': jac_overlap}
    ]

    options = {'maxiter': 300, 'gtol': 1e-9, 'xtol': 1e-9, 'disp': False}
    res = minimize(obj_fn, x0, method='trust-constr', jac=grad_obj, constraints=constraints, options=options)

    if not res.success:
        raise RuntimeError("Conservative optimization failed in refine step")

    x_opt = res.x
    centers_opt = x_opt[:2*n].reshape(-1, 2)
    radii_opt = x_opt[2*n:]

    return centers_opt, radii_opt

def run_packing():

    R = np.arange(0.025, 0.120, 0.01)

    candidates = init_candidate_grid(R, Δ)

    radii_final = max_radii_fixed_centres(centers_refined, radii_refined)
