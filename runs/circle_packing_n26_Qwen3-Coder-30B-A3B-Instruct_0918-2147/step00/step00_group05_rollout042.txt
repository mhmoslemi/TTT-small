import numpy as np

def hex_lattice(d):
    """Generate a hexagonal lattice within [0,1]x[0,1] with spacing d."""
    sqrt3o2 = np.sqrt(3) / 2
    rows = []
    y = 0
    row = 0
    while y <= 1:
        x = 0
        col = 0
        row_points = []
        while x <= 1:
            row_points.append([x, y])
            x += d
            col += 1
        if row % 2 == 1:
            # offset odd rows
            row_points = [[p[0] - d / 2, p[1]] for p in row_points if p[0] >= 0 and p[0] <= 1]
        else:
            row_points = [p for p in row_points if p[0] >= 0 and p[0] <= 1]
        rows.extend(row_points)
        y += sqrt3o2 * d
        row += 1
    return np.array(rows)

def add_border_circles(core_points, d):
    """Add 6 additional circles near corners and edges."""
    r = d / 4
    border = [
        [r, r], [1 - r, r], [r, 1 - r], [1 - r, 1 - r],
        [0.5, r], [0.5, 1 - r]
    ]
    return np.vstack([core_points, border])

def initial_radii(points):
    """Calculate initial radii ensuring no overlaps and boundary constraints."""
    n = points.shape[0]
    # Wall distances
    wall_distances = np.array([
        points[:, 0], 1 - points[:, 0],
        points[:, 1], 1 - points[:, 1]
    ]).min(axis=0)
    # Min distance to other centers
    distances = np.sqrt(np.sum((points[:, None, :] - points[None, :, :]) ** 2, axis=2))
    np.fill_diagonal(distances, np.inf)
    min_distances = distances.min(axis=1)
    initial_r = np.minimum(wall_distances, min_distances / 2)
    return np.minimum(initial_r, 0.1)

def inflate_radii(centers, radii):
    """Increase radii to their maximum feasible values."""
    n = centers.shape[0]
    # Distance matrix
    D = np.sqrt(np.sum((centers[:, None, :] - centers[None, :, :]) ** 2, axis=2))
    new_radii = radii.copy()
    epsilon = 1e-7
    for i in range(n):
        wall_caps = [
            centers[i, 0], 1 - centers[i, 0],
            centers[i, 1], 1 - centers[i, 1]
        ]
        nearest_caps = D[i, :] - radii
        nearest_caps[i] = np.inf
        radius_upper = min(min(wall_caps), nearest_caps.min())
        if radius_upper > new_radii[i] + epsilon:
            new_radii[i] = radius_upper
    return new_radii

def relax_centers(centers, radii, steps=10):
    """Move centers to optimize spacing using force-based relaxation."""
    n = centers.shape[0]
    kappa = 0.02
    lambda_param = 0.01
    alpha = 0.5
    beta = 0.02
    gamma = 0.02
    new_centers = centers.copy()
    for _ in range(steps):
        forces = np.zeros_like(new_centers)
        for i in range(n):
            for j in range(n):
                if i == j:
                    continue
                diff = new_centers[i] - new_centers[j]
                dist = np.linalg.norm(diff)
                if dist < 1e-10:
                    continue
                # Repulsive force
                if dist < radii[i] + radii[j] + beta:
                    force_magnitude = kappa * max(0, beta - dist)
                    forces[i] += force_magnitude * diff / dist
            # Wall forces
            wall_force_x = 0
            if new_centers[i, 0] - radii[i] < gamma:
                wall_force_x = -lambda_param * max(0, gamma - (new_centers[i, 0] - radii[i]))
            if 1 - new_centers[i, 0] - radii[i] < gamma:
                wall_force_x += lambda_param * max(0, gamma - (1 - new_centers[i, 0] - radii[i]))
            wall_force_y = 0
            if new_centers[i, 1] - radii[i] < gamma:
                wall_force_y = -lambda_param * max(0, gamma - (new_centers[i, 1] - radii[i]))
            if 1 - new_centers[i, 1] - radii[i] < gamma:
                wall_force_y += lambda_param * max(0, gamma - (1 - new_centers[i, 1] - radii[i]))
            forces[i, 0] += wall_force_x
            forces[i, 1] += wall_force_y
        new_centers += alpha * forces
        # Clamp to [r, 1-r]
        new_centers[:, 0] = np.clip(new_centers[:, 0], radii, 1 - radii)
        new_centers[:, 1] = np.clip(new_centers[:, 1], radii, 1 - radii)
    return new_centers

def run_packing():
    d = 0.18
    core_points = hex_lattice(d)
    # Take first 20 points
    core_points = core_points[:20]
    full_points = add_border_circles(core_points, d)
    radii = initial_radii(full_points)
    centers = full_points.copy()
    max_outer = 30
    previous_sum = np.sum(radii)
    for _ in range(max_outer):
        radii = inflate_radii(centers, radii)
        centers = relax_centers(centers, radii)
        current_sum = np.sum(radii)
        if current_sum - previous_sum < 1e-6:
            break
        previous_sum = current_sum
