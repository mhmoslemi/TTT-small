import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial.distance import cdist

def build_lp_model(n):
    # Build LP model with 26 circles
    # Variables: 26*(x,y,r) = 78 total
    # Constraints: 
    # - Wall constraints for each circle (4 per circle)
    # - Axis separation constraints for each unordered pair (4 per pair)
    # Total pairs: 26*25/2 = 325
    # Total constraints: 26*4 + 325*4 = 104 + 1300 = 1404 
    # But we can treat the pair constraints as variable-size for clarity
    
    # Variables indices
    x_indices = list(range(0, 26))
    y_indices = list(range(26, 52))
    r_indices = list(range(52, 78))
    
    # Build A_ub, b_ub, c for optimization problem min -sum(r_i)
    # So maximize sum(r_i)
    
    # Build number of constraints
    # Wall constraints: 26 * 4 (each circle touches one of: x-r, 1-x-r, y-r, 1-y-r)
    # Separation constraints: For each pair of circles, 4 constraints (x+rx <= xj-rj, etc.)
    num_vars = 78
    
    # Each circle has 4 wall constraints
    wall_rows = 26*4
    # Each pair of 26 circles contributes up to 4 axis constraints
    sep_rows = 325*4
    
    total_constraints = wall_rows + sep_rows
    A_ub = np.zeros((total_constraints, num_vars))
    b_ub = np.zeros(total_constraints)
    c = np.zeros(num_vars)
    c[52:78] = -1.0  # maximize sum of radii: minimize -sum(r_i)
    
    # Fill in wall constraints (last 4*26 rows)
    row_idx = 0
    for i in range(26):
        # r_i <= x_i
        A_ub[row_idx, r_indices[i]] = 1.0
        A_ub[row_idx, x_indices[i]] = -1.0
        b_ub[row_idx] = 0.0
        row_idx += 1
        
        # r_i <= 1 - x_i
        A_ub[row_idx, r_indices[i]] = 1.0
        A_ub[row_idx, x_indices[i]] = 1.0
        b_ub[row_idx] = 1.0
        row_idx += 1
        
        # r_i <= y_i
        A_ub[row_idx, r_indices[i]] = 1.0
        A_ub[row_idx, y_indices[i]] = -1.0
        b_ub[row_idx] = 0.0
        row_idx += 1
        
        # r_i <= 1 - y_i
        A_ub[row_idx, r_indices[i]] = 1.0
        A_ub[row_idx, y_indices[i]] = 1.0
        b_ub[row_idx] = 1.0
        row_idx += 1
    
    # Set up separation constraints
    # There are N(N-1)/2 = 325 pairs of circles
    for i in range(26):
        for j in range(i+1, 26):
            # x_i + r_i <= x_j - r_j     (i left of j along x-axis)  
            A_ub[row_idx, x_indices[i]] = 1.0
            A_ub[row_idx, x_indices[j]] = -1.0
            A_ub[row_idx, r_indices[i]] = 1.0
            A_ub[row_idx, r_indices[j]] = 1.0
            b_ub[row_idx] = 0.0
            row_idx += 1
            
            # x_j + r_j <= x_i - r_i     (j left of i along x-axis)
            A_ub[row_idx, x_indices[i]] = -1.0
            A_ub[row_idx, x_indices[j]] = 1.0
            A_ub[row_idx, r_indices[i]] = 1.0
            A_ub[row_idx, r_indices[j]] = 1.0
            b_ub[row_idx] = 0.0
            row_idx += 1
            
            # y_i + r_i <= y_j - r_j     (i below j along y-axis)
            A_ub[row_idx, y_indices[i]] = 1.0
            A_ub[row_idx, y_indices[j]] = -1.0
            A_ub[row_idx, r_indices[i]] = 1.0
            A_ub[row_idx, r_indices[j]] = 1.0
            b_ub[row_idx] = 0.0
            row_idx += 1
            
            # y_j + r_j <= y_i - r_i     (j below i along y-axis)
            A_ub[row_idx, y_indices[i]] = -1.0
            A_ub[row_idx, y_indices[j]] = 1.0
            A_ub[row_idx, r_indices[i]] = 1.0
            A_ub[row_idx, r_indices[j]] = 1.0
            b_ub[row_idx] = 0.0
            row_idx += 1
    
    # Return model and reshape indices
    return A_ub, b_ub, c, x_indices, y_indices, r_indices

def solve_lp(A_ub, b_ub, c):
    # Solve the linear program using scipy.optimize.linprog
    # Using method='highs' for robustness
    
    try:
        result = linprog(
            c=c, 
            A_ub=A_ub, 
            b_ub=b_ub,
            method='highs',
            options={'presolve': True, 'time_limit': 2.0}
        )
        if result.status == 0:
            # Converged successfully
            return result.x, result.status
        else:
            raise RuntimeError(f"LP solve failed with status {result.status}")
    except Exception as e:
        raise RuntimeError(f"Error in LP solve: {repr(e)}")

def extract_active_constraints(x_sol, y_sol, r_sol, A_ub, b_ub, x_indices, y_indices, r_indices):
    # Determine which constraints are active for solution x_sol
    # From scipy's linprog, get the final active set by checking if constraints are tight
    active = []
    tolerance = 1e-9
    for i in range(A_ub.shape[0]):
        lhs = np.dot(A_ub[i, :], x_sol)
        rhs = b_ub[i]
        if abs(lhs - rhs) < tolerance:
            active.append(i)
    return active

def detect_diagonal_conflicts(centers, radii):
    """Detect pairs of circles that overlap in Euclidean distance"""
    n = centers.shape[0]
    conflicts = []
    for i in range(n):
        for j in range(i+1, n):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            d = np.sqrt(dx*dx + dy*dy)
            r_sum = radii[i] + radii[j]
            if d < r_sum - 1e-12:
                conflicts.append((i,j))
    return conflicts

def get_active_coordinate_constraints(centers, radii, x_indices, y_indices, r_indices, active_list):
    """Identify active coordinate and wall constraints from full solution"""
    
    active_pairs = set()
    active_walls = []
    
    for idx in active_list:
        # Determine what kind of constraint this is by examining A_ub row
        # Liblinear constraints have certain identifiable structure
        # The most practical is to do it by looking at constraint index ranges
        # Wall constraints: 0 - 103
        # Separation constraints: 104+  
        if idx < 104:
            # Wall constraints
            circle_idx = idx // 4  # 0 thru 25
            wall_type = idx % 4    # 0=x-r, 1=1-x-r, 2=y-r, 3=1-y-r
            active_walls.append((circle_idx, wall_type))
        else:
            # separator constraints -- detect which circles
            offset = idx - 104
            pair_i = 0
            pair_j = 0
            pair_count = 0
            for i in range(26):
                for j in range(i+1, 26):
                    if pair_count == offset // 4:
                        pair_i = i
                        pair_j = j
                        break
                    pair_count += 1
                if pair_i > 0: break

            if pair_count * 4 + offset % 4 <= 1300:
                # separator constraint from one of the four
                orient = offset % 4
                # from x_i + r_i <= x_j - r_j  -> pair_i left of pair_j
                active_pairs.add((pair_i, pair_j, orient))
            else:
                # This shouldn't happen... but let's guard for consistency
                continue
                
    return active_pairs, active_walls

def resolve_conflicts(centers, radii, active_pairs, conflicts):
    # Create subproblem with active constraints + conflicting pairs
    n = centers.shape[0]
    num_vars = 2*n + 26
    
    # Simple iterative solution
    # For conflicts, compute Euclidean radii adjustments through bounded optimization
    max_it = 50
    for iter in range(max_it):
        changed = False
        
        for pair_i, pair_j in conflicts:
            dx = centers[pair_i][0] - centers[pair_j][0]
            dy = centers[pair_i][1] - centers[pair_j][1]
            dist = np.sqrt(dx*dx + dy*dy)
            gap = radii[pair_i] + radii[pair_j] - dist
            
            if gap > 1e-12:
                # They do overlap; we'll adjust the radii of violators
                
                # Here's a simple adjustment: increase both radii proportionally
                # Should respect the LM estimate
                # But for now, just increase each by small amounts
                
                incr = min(gap * 0.5, 0.02)  # Cap the large radii boost
                radii[pair_i] += incr
                radii[pair_j] += incr
                changed = True
                
        if not changed:
            break
    
    return centers, radii

def polish_global(centers, radii, active_pairs, active_walls, max_iter=200):
    # Run final polishing using trust-constr algorithm  
    # Keep both wall and axis constraints as equality; sum radii as objective
    
    n = centers.shape[0]
    
    # Corresponding to r, x, y variables
    r_vars = np.arange(52, 78)
    x_vars = np.arange(0, 26)
    y_vars = np.arange(26, 52)
    
    # Reduce to active constraints only
    # Set up Jacobian and gradients (very small number of active constraints)
    
    # Use symbolic nonlinear optimization to improve sum radii
    def objective(x_vars_radii):
        # x_vars_radii is 26 + 26 + 26 = 78
        r_values = x_vars_radii[52:78]
        return -np.sum(r_values)
    
    # Manually construct constraints from active_pairs/walls
    cons = []
    
    def wall_eq(circle_idx, type_val):
        # type_val = 0,1,2,3 => [r <= x, r <= 1-x, r <= y, r <= 1-y]
        
        def c(x_vals):
            ps = x_vals[52:78]  # radii
            xs = x_vals[0:26]   # x
            ys = x_vals[26:52]  # y
            
            if type_val == 0: return ps[circle_idx] - xs[circle_idx]  # r <= x
            elif type_val == 1: return ps[circle_idx] - (1 - xs[circle_idx])  # r <= 1-x
            elif type_val == 2: return ps[circle_idx] - ys[circle_idx]  # r <= y
            elif type_val == 3: return ps[circle_idx] - (1 - ys[circle_idx])  # r <= 1-y
            else: return 0
        
        return {'type': 'eq', 'fun': c}
    
    def axis_sep_eq(i, j, orient):
        # orient = 0,1: x-direction; 2,3: y-direction
        def c(x_vals):
            ps = x_vals[52:78]  # radii
            xs = x_vals[0:26]   # x
            ys = x_vals[26:52]  # y
            
            xi, xj = xs[i], xs[j]
            yi, yj = ys[i], ys[j]
            ri, rj = ps[i], ps[j]
            
            if orient == 0: # i left of j (x_i + r_i <= x_j - r_j)
                return xi + ri - xj + rj  # -> make == 0
            elif orient == 1: # j left of i (x_j + r_j <= x_i - r_i)
                return xj + rj - xi + ri  # -> make == 0
            elif orient == 2: # i below j (y_i + r_i <= y_j - r_j)
                return yi + ri - yj + rj
            elif orient == 3: # j below i (y_j + r_j <= y_i - r_i)
                return yj + rj - yi + ri
            else:
                return 0
        return {'type': 'eq', 'fun': c}
    
    # Post-process active sets to find their corresponding active constraints
    # iteration: collect them
    for cir, type_val in active_walls:
        cons.append(wall_eq(cir, type_val))
    
    # Separation constraints
    for i, j, orient in active_pairs:
        cons.append(axis_sep_eq(i, j, orient))
    
    bounds = []
    for i in range(52):
        bounds.append((None, None))  # unbounded
    for i in range(26):
        bounds.append((0, 1))  # 0 <= r <= 1, but will be handled with R_i <= x_i, etc.
        
    # Initialize as center + radii
    x0 = np.zeros(78)
    for i in range(26):
        x0[i] = centers[i][0]
        x0[i+26] = centers[i][1]
        x0[i+52] = radii[i]
    
    # Put margin on final optimization
    try:
        sol = minimize(
            objective,
            x0,
            method='trust-constr',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': max_iter}
        )
        
        # Update solution
        for i in range(26):
            centers[i][0] = sol.x[i]
            centers[i][1] = sol.x[i+26]
            radii[i] = sol.x[i+52]
    except Exception as e:
        pass

    return centers, radii

def max_radius_projection(centers, radii):
    """Do one final analytical projection to extract the maximal radii that respect all constraints"""
    n = centers.shape[0]
    
    # For each point, the radius is min of distance to walls and to other circles
    updated = True
    max_it = 50
    
    # Stop if no change
    for _ in range(max_it):
        updated = False
        new_radii = radii.copy()
        
        for i in range(n):
            # Compute distance from i to walls
            r_min = min(centers[i][0], 1 - centers[i][0], 
                        centers[i][1], 1 - centers[i][1])
            
            # Distance to all other circles
            for j in range(n):
                if i != j:
                    dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                    r_min = min(r_min, dist - radii[j])
            
            if r_min > radii[i] + 1e-12:
                new_radii[i] = r_min
                updated = True
        
        if not updated:
            break
            
        radii[:] = new_radii

    return centers, radii

def run_packing():
    # 1. Set up the LP model
    A_ub, b_ub, c, x_indices, y_indices, r_indices = build_lp_model(26)
    
    # 2. Solve the LP
    solution = solve_lp(A_ub, b_ub, c)
    x_sol, status = solution
    
    # Extract radii
    radii = x_sol[r_indices]
    centers = np.array([[x_sol[x_indices[i]], x_sol[y_indices[i]]] for i in range(26)])
    
    # Confirm no large negative radii
    for r in radii:
        if r < 0:
            raise ValueError(f"Got negative radius {r}")
    
    # 3. Identify active constraints
    active_indices = extract_active_constraints(x_sol, x_sol[y_indices], radii, A_ub, b_ub, x_indices, y_indices, r_indices)
    active_pairs, active_walls = get_active_coordinate_constraints(centers, radii, x_indices, y_indices, r_indices, active_indices)
    
    # 4. Check for diagonal Euclidean overlaps
    conflicts = detect_diagonal_conflicts(centers, radii)
    
    # 5. Resolve diagonal conflicts in small subproblems
    centers, radii = resolve_conflicts(centers, radii, active_pairs, conflicts)

    # 6. Perform full global polishing based on active constraints
    centers, radii = polish_global(centers, radii, active_pairs, active_walls)

    # 7. Final maximal radius projection
    centers, radii = max_radius_projection(centers, radii)

    # 8. Compute final sum
    sum_radii = float(np.sum(radii))
    # The validator must be run directly on the final result
    # since center & radius values might just be slightly out of bounds 
    # due to floating point errors.
    
    # Validate  
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise ValueError(f"Failed validation: {msg}")

    return centers, radii, sum_radii
