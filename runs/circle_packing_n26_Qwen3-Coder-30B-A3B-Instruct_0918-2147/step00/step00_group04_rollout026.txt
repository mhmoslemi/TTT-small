import numpy as np
import cvxpy as cp
from scipy.spatial.distance import cdist

def build_leaf_problem(cell, k, centers=None, radii=None):
    # cell: (x_min, y_min, x_max, y_max)
    x_min, y_min, x_max, y_max = cell
    
    # Variables
    vars_ = []
    for i in range(k):
        vars_.append(cp.Variable(2))  # center (x, y)
        vars_.append(cp.Variable(1, nonneg=True))  # radius
    
    # Variables grouped by circle
    c = [vars_[2*i] for i in range(k)]
    r = [vars_[2*i+1][0] for i in range(k)]
    
    # Constraints
    constr = []
    
    # Boundary constraints (linear)
    for i in range(k):
        constr.append(c[i][0] - r[i] >= x_min)
        constr.append(c[i][0] + r[i] <= x_max)
        constr.append(c[i][1] - r[i] >= y_min)
        constr.append(c[i][1] + r[i] <= y_max)
        
    # Non-overlap constraints (SOC)
    for i in range(k):
        for j in range(i+1, k):
            dist = cp.norm(c[i] - c[j])
            constr.append(dist >= r[i] + r[j])
    
    # Objective - maximize sum of radii
    obj = cp.Maximize(cp.sum(r))
    
    # Create problem
    prob = cp.Problem(obj, constr)
    return prob, c, r

def solve_leaf(cell, k):
    prob, c, r = build_leaf_problem(cell, k)
    prob.solve(solver=cp.OSQP, verbose=False, eps_abs=1e-10, eps_rel=1e-10)
    if prob.status != cp.OPTIMAL:
        raise RuntimeError(f"Leaf SOCP failed for cell {cell} with k={k}: {prob.status}")
    
    centers = np.array([[c[i][0].value, c[i][1].value] for i in range(k)])
    radii = np.array([r[i].value for i in range(k)])
    return centers, radii

def calculate_distance_matrix(centers, radii):
    n = centers.shape[0]
    distances = cdist(centers, centers)
    # Create matrix of min distances between circles (including radii)
    min_distances = np.zeros((n, n))
    for i in range(n):
        for j in range(n):
            if i != j:
                min_distances[i, j] = radii[i] + radii[j]
    
    return distances, min_distances

def project_to_feasible(x, y, r, bounds):
    x_min, y_min, x_max, y_max = bounds
    x = np.clip(x, x_min + r, x_max - r)
    y = np.clip(y, y_min + r, y_max - r)
    return x, y

def size_corrected_centers_radii(centers, radii, cell_bounds):
    new_centers = centers.copy()
    new_radii = radii.copy()
    x_min, y_min, x_max, y_max = cell_bounds
    for i in range(len(new_centers)):
        # clipping based on boundary constraints
        new_centers[i][0] = np.clip(new_centers[i][0], x_min + new_radii[i], x_max - new_radii[i])
        new_centers[i][1] = np.clip(new_centers[i][1], y_min + new_radii[i], y_max - new_radii[i])
    return new_centers, new_radii

def adjust_cluster_for_boundary(centers, radii, bounds):
    if len(centers) == 0:
        return centers, radii
    new_centers = centers.copy()
    new_radii = radii.copy()
    x_min, y_min, x_max, y_max = bounds
    for i in range(len(new_centers)):
        x, y = new_centers[i]
        r = new_radii[i]
        x = np.clip(x, x_min + r, x_max - r)
        y = np.clip(y, y_min + r, y_max - r)
        new_centers[i] = [x, y]
    return new_centers, new_radii

def polish_solution(centers, radii, iterations=3):
    # Convert to numpy arrays
    centers = centers.copy()
    radii = radii.copy()
    n_original = len(centers)
    # Create penalties for walls
    for _ in range(iterations):
        # Forces updated every iteration
        forces = np.zeros_like(centers)
        # Compute repulsion forces
        dist_matrix = cdist(centers, centers) 
        mask = np.eye(n_original, dtype=bool)
        # Need to expand the loop to handle missing matrix values
        for i in range(n_original):

                if not mask[i, j]:
                    dx = centers[i][0] - centers[j][0]
                    dy = centers[i][1] - centers[j][1]
                    dist = max(1e-10, np.sqrt(dx*dx + dy*dy))
                    repulsion_strength = 1.0 / (dist * (dist + radii[i] + radii[j]))
                    forces[i][0] += -dx * repulsion_strength
                    forces[i][1] += -dy * repulsion_strength
        
        # Walls
        for i in range(n_original):
            # Effectively push away from walls
            margin_xl = centers[i][0] - radii[i]
            margin_xr = 1 - centers[i][0] - radii[i]
            margin_yl = centers[i][1] - radii[i]
            margin_yr = 1 - centers[i][1] - radii[i]
            
            # Handle walls

                forces[i][0] -= margin_xl * 1e3
            if margin_xr < 0:
                forces[i][0] += margin_xr * 1e3
            if margin_yl < 0:

    return centers, radii
