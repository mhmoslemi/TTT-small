import numpy as np
from scipy.optimize import linprog

def run_packing():
    # Step 1: Geometry-first layout (deterministic seed)
    n_circles = 26
    # Start with triangular lattice pattern
    spacing = 0.8 / np.sqrt(n_circles)
    rows = int(np.ceil(1.0 / spacing))
    cols = int(np.ceil(1.0 / spacing))
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = i * spacing + (j % 2) * spacing / 2
            y = j * spacing * np.sqrt(3) / 2
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers.append([x, y])
            if len(centers) >= n_circles:
                break
        if len(centers) >= n_circles:
            break
    # Take first n_circles, make sure we don't exceed
    centers = np.array(centers[:n_circles])
    # Add margin to ensure feasibility
    margin = 0.05
    centers[:, 0] = np.clip(centers[:, 0], margin, 1 - margin)
    centers[:, 1] = np.clip(centers[:, 1], margin, 1 - margin)
    # Initial radii
    radii = np.full(n_circles, 0.01)
    
    # Polygonal approximation of forbidden regions - 8-gon
    k = 8
    angles = np.linspace(0, 2*np.pi, k, endpoint=False)
    u_vectors = np.column_stack([np.cos(angles), np.sin(angles)])
    cos_pi_k = np.cos(np.pi / k)
    
    def compute_distance_matrix(centers):
        diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        return np.sqrt(np.sum(diffs**2, axis=2))

    def solve_radius_lp(centers, radii):
        n = centers.shape[0]
        # Build constraints matrix A_ub and vector b_ub as A_ub * r <= b_ub
        # Wall constraints: r_i <= min(distance_from_wall)
        A_rows = []
        b_rows = []
        for i in range(n):
            c = centers[i]
            wall_constraints = [
                [1.0],  # r_i <= c.x
                [-1.0],  # r_i <= 1 - c.x
                [1.0],  # r_i <= c.y
                [-1.0]   # r_i <= 1 - c.y
            ]
            b_vals = [
                c[0],
                1 - c[0],
                c[1],
                1 - c[1]
            ]
            A_rows.extend(wall_constraints)
            b_rows.extend(b_vals)
        
        # Separation constraints
        distances = compute_distance_matrix(centers)
        for i in range(n):
            for j in range(i+1, n):
                # r_i + r_j <= d_ij
                A_rows.append([1.0 if m == i else (1.0 if m == j else 0.0) for m in range(n)])
                b_rows.append(distances[i, j])
        
        A = np.array(A_rows)
        b = np.array(b_rows)
        # Objective vector: maximize 1^T r
        c = np.ones(n)
        # Solve
        res = linprog(c, A_ub=A, b_ub=b, method='highs', options={'tol':1e-10})
        if not res.success:
            return radii
        return np.maximum(0.0, np.minimum(0.5, res.x))

    def solve_center_lp(centers, radii):
        n = centers.shape[0]
        # For each circle i, define its safe region
        # Constraints:
        #   c_i.x - r_i >= s_i  =>  c_i.x - r_i - s_i >= 0
        #   1 - c_i.x - r_i >= s_i  =>  -c_i.x - r_i - s_i >= -1
        #   c_i.y - r_i >= s_i  =>  c_i.y - r_i - s_i >= 0
        #   1 - c_i.y - r_i >= s_i  =>  -c_i.y - r_i - s_i >= -1
        # For pair constraints:
        #   u_l · (c_i - c_j) <= - (r_i + r_j) * cos(π/k)
        #   <=>
        #   u_l · c_i - u_l · c_j <= - (r_i + r_j) * cos(π/k)
        #   u_l · c_i <= u_l · c_j - (r_i + r_j) * cos(π/k)
        #   <=>
        #   -u_l · c_i + u_l · c_j >= (r_i + r_j) * cos(π/k)
        # Add slack variables s_i
        num_constraints = 4 * n + n*(n-1)//2 * k
        A = np.zeros((num_constraints, 2*n + n))
        b = np.zeros(num_constraints)
        
        # Wall constraints
        ind_start = 0
        for i in range(n):
            # c_i.x - r_i - s_i >= 0
            A[ind_start, i*2] = 1.0
            A[ind_start, n*2 + i] = -1.0
            b[ind_start] = radii[i]
            ind_start += 1
            # 1 - c_i.x - r_i - s_i >= -1
            A[ind_start, i*2] = -1.0
            A[ind_start, n*2 + i] = -1.0
            b[ind_start] = -1 + radii[i]
            ind_start += 1
            # c_i.y - r_i - s_i >= 0
            A[ind_start, i*2 + 1] = 1.0
            A[ind_start, n*2 + i] = -1.0
            b[ind_start] = radii[i]
            ind_start += 1
            # 1 - c_i.y - r_i - s_i >= -1
            A[ind_start, i*2 + 1] = -1.0
            A[ind_start, n*2 + i] = -1.0
            b[ind_start] = -1 + radii[i]
            ind_start += 1

        # Separation constraints - polygonal approximation
        distances = compute_distance_matrix(centers)
        for i in range(n):
            for j in range(i+1, n):
                for l in range(k):
                    u = u_vectors[l]
                    A[ind_start, i*2] = -u[0]
                    A[ind_start, j*2] = u[0]
                    A[ind_start, i*2 + 1] = -u[1]
                    A[ind_start, j*2 + 1] = u[1]
                    A[ind_start, n*2 + i] = -1.0
                    A[ind_start, n*2 + j] = -1.0
                    b[ind_start] = (radii[i] + radii[j]) * cos_pi_k
                    ind_start += 1

        # Objective: maximize sum of s_i -> minimize -sum(s_i)
        c = np.zeros(2*n + n)
        c[n*2:] = -1.0
        
        # Solve LP
        res = linprog(c, A_ub=A, b_ub=b, method='highs', options={'tol':1e-10})
        if not res.success:
            return centers
        new_centers = centers.copy()
        new_centers[:] = res.x[0:n*2].reshape(n, 2)
        return new_centers

    # Alternating SLP loop
    max_iter = 30
    eps = 1e-6
    sum_radii_old = 0.0
    for i_iter in range(max_iter):
        radii = solve_radius_lp(centers, radii)
        centers = solve_center_lp(centers, radii)
        sum_radii = np.sum(radii)
        if np.abs(sum_radii - sum_radii_old) < eps:
            break
        sum_radii_old = sum_radii

    # Local inflation improvement
    for _ in range(5):
        improved_any = False
        distances = compute_distance_matrix(centers)
        for i in range(n_circles):
            # Compute exact distances to walls
            exact_gaps = [
                centers[i, 0] - radii[i],           # left
                1.0 - centers[i, 0] - radii[i],     # right
                centers[i, 1] - radii[i],           # bottom
                1.0 - centers[i, 1] - radii[i]      # top
            ]
            min_wall_dist = min(exact_gaps)
            # Neighbours
            max_neighbor_dist = 0.0
            for j in range(n_circles):
                if i == j:
                    continue
                actual_dist = distances[i, j]
                expected_dist = radii[i] + radii[j]
                if actual_dist < expected_dist + 1e-10:
                    max_neighbor_dist = max(max_neighbor_dist, expected_dist - actual_dist)
            # How much we could theoretically add
            gain = max(min_wall_dist, max_neighbor_dist)
            if gain > 1e-5:
                # Try increase radii[i] by 0.4 * gain
                radii[i] = min(0.5, radii[i] + 0.4 * gain)
                improved_any = True
        if not improved_any:
            break
        # Re-solve small radius LP for affected circles
        radii = solve_radius_lp(centers, radii)
        # If needed, call recentering
        centers = solve_center_lp(centers, radii)

    # Validate and clean up numerics
    valid, msg = validate_packing(centers, radii)
    if not valid:
        # If not valid, slightly shrink radii
        for _ in range(5):
            radii *= 0.999999
            valid, msg = validate_packing(centers, radii)
            if valid:
                break
        if not valid:
            # Last resort: rescale down all
            radii *= 0.999999
            valid, msg = validate_packing(centers, radii)
            if not valid:
                radii *= 0.999999
                valid, msg = validate_packing(centers, radii)
                if not valid:
                    raise RuntimeError("Postprocessing fails.")

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
