import numpy as np
from scipy.optimize import minimize

# Constants
NUM_CIRCLES = 26
NUM_RESTARTS = 5
INITIAL_RADIUS = 0.02
BOUNDARY_TOL = 1e-9

# Helper functions
def hex_lattice_points(n_rows, spacing):
    """Generate hexagonal lattice points covering roughly a unit square."""
    points = []
    y = 0.0
    row = 0
    while y <= 1.0:
        x_offset = (row % 2) * spacing / 2
        x = x_offset
        while x <= 1.0:
            points.append([x, y])
            x += spacing
        y += spacing * np.sqrt(3) / 2
        row += 1
        if len(points) >= n_rows:
            break
    return np.array(points[:n_rows])

def initial_guess(seed):
    """Generate initial guess using hexagonal lattice with jitter."""
    np.random.seed(seed)
    spacing = 0.2
    points = hex_lattice_points(NUM_CIRCLES, spacing)
    # Jitter the points slightly
    jitter = np.random.uniform(-0.02, 0.02, size=(NUM_CIRCLES, 2))
    points += jitter
    # Clip to valid range
    points = np.clip(points, 0, 1)
    # Initialize radii
    radii = np.full(NUM_CIRCLES, INITIAL_RADIUS)
    # Concatenate into one vector
    vars0 = np.zeros(3 * NUM_CIRCLES)
    vars0[0::3] = points[:, 0]    # x positions
    vars0[1::3] = points[:, 1]    # y positions
    vars0[2::3] = radii           # radii
    return vars0

def objective(vars):
    """Objective function to minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraint_boundary(vars):
    """Return boundary constraints: [x-r, 1-x-r, y-r, 1-y-r] for each circle."""
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    constraints = []
    constraints.extend(x - r)         # x - r >= 0
    constraints.extend(1 - x - r)     # 1 - x - r >= 0
    constraints.extend(y - r)         # y - r >= 0
    constraints.extend(1 - y - r)     # 1 - y - r >= 0
    return np.array(constraints)

def constraint_nonoverlap(vars):
    """Return non-overlap constraints."""
    n = NUM_CIRCLES
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dist_sq = dx**2 + dy**2
            sum_radii = r[i] + r[j]
            # Constraint: distance >= sum_radii - epsilon
            constraints.append(np.sqrt(dist_sq) - sum_radii)
    return np.array(constraints)

def postprocess(vars):
    """Apply post-processing to fix minor violations."""
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    # Clip centers to unit square while preserving radii
    x = np.clip(x, r, 1.0 - r)
    y = np.clip(y, r, 1.0 - r)
    # Ensure all radii non-negative
    r = np.maximum(r, 0.0)
    
    # Keep the corrected sums
    new_vars = vars.copy()
    new_vars[0::3] = x
    new_vars[1::3] = y
    new_vars[2::3] = r
    return new_vars

def run_packing():
    """Main function to pack 26 circles to maximize sum of radii."""
    best_sum = -np.inf
    best_vars = None

    # Define global bounds (x, y, r) for each circle
    bounds = []
    for _ in range(NUM_CIRCLES):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])

    # Define constraints
    constraints = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_nonoverlap}
    ]

    for seed in range(NUM_RESTARTS):
        # Generate initial guess
        vars0 = initial_guess(seed)
        
        # Optimization
        res = minimize(
            objective,
            vars0,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 2000, 'ftol': 1e-6}
        )
        
        if not res.success:
            continue

        # Apply postprocessing
        vars_corr = postprocess(res.x)
        s = np.sum(vars_corr[2::3])
        
        if s > best_sum:
            best_sum = s
            best_vars = vars_corr.copy()
    
    # Reshape final variables
    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii = best_vars[2::3]
    return centers, radii, float(best_sum)
