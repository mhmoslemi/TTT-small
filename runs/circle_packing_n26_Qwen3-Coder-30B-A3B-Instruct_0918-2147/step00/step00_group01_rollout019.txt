import numpy as np
from scipy.optimize import minimize

def initial_layout():
    # Create a hexagonal grid pattern inside the unit square
    rows = 5
    cols = 5
    spacing = 0.18
    radius = spacing / 2 - 0.01
    
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = (j + 0.5) * spacing
            y = (i + 0.5) * spacing
            # Adjust for square bounds
            x = max(radius, min(1 - radius, x))
            y = max(radius, min(1 - radius, y))
            centers.append([x, y])
    
    # Place one more circle in the center
    centers.append([0.5, 0.5])
    
    centers = np.array(centers)
    radii = np.full(26, radius)
    return centers, radii

def pack_vector(centers, radii):
    z = np.empty(3 * 26)
    z[0:52:3] = centers[:, 0]  # x coords
    z[1:52:3] = centers[:, 1]  # y coords
    z[2:52:3] = radii         # radii
    return z

def unpack_vector(z):
    centers = np.empty((26, 2))
    radii = np.empty(26)
    centers[:, 0] = z[0:52:3]
    centers[:, 1] = z[1:52:3]
    radii = z[2:52:3]
    return centers, radii

def objective(z):
    centers, radii = unpack_vector(z)
    return -np.sum(radii)

def constraint_boundary(z):
    centers, radii = unpack_vector(z)
    constraints = []
    # For each circle i: boundary constraints
    for i in range(26):
        x, y, r = centers[i, 0], centers[i, 1], radii[i]
        constraints.extend([
            x - r,           # x - r >= 0
            1 - x - r,       # 1 - x - r >= 0
            y - r,           # y - r >= 0
            1 - y - r        # 1 - y - r >= 0
        ])
    return np.array(constraints)

def constraint_nonoverlap(z):
    centers, radii = unpack_vector(z)
    constraints = []
    # For each pair of circles
    for i in range(26):
        for j in range(i+1, 26):
            x_i, y_i, r_i = centers[i, 0], centers[i, 1], radii[i]
            x_j, y_j, r_j = centers[j, 0], centers[j, 1], radii[j]
            dist_sq = (x_i - x_j)**2 + (y_i - y_j)**2
            r_sum = r_i + r_j
            constraints.append(dist_sq - r_sum**2)
    return np.array(constraints)

def run_packing():
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    for _ in range(3):
        # Create initial layout
        centers, radii = initial_layout()
        
        # Random perturbation of centers and radii
        centers += np.random.uniform(-0.02, 0.02, (26, 2))
        radii += np.random.uniform(-0.01, 0.01, 26)
        radii = np.maximum(0, radii)  # make sure all radii are nonnegative
        
        # Pack into vector
        z0 = pack_vector(centers, radii)
        
        # Define bounds
        bounds = [(0, 1) for _ in range(52)] + [(0, 0.5) for _ in range(26)]
        
        # Create constraints
        constraints = [
            {'type': 'ineq', 'fun': constraint_boundary},
            {'type': 'ineq', 'fun': constraint_nonoverlap}
        ]
        
        # Optimize
        result = minimize(
            objective,
            z0,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'ftol': 1e-9, 'maxiter': 2000}
        )
        
        # Extract result
        final_centers, final_radii = unpack_vector(result.x)
        
        # Validate
        valid, msg = validate_packing(final_centers, final_radii)
        
        if valid:
            sum_radii = np.sum(final_radii)
            if sum_radii > best_sum:
                best_sum = sum_radii
                best_centers = final_centers.copy()
                best_radii = final_radii.copy()
    
    if best_centers is None:
        raise RuntimeError("No valid packing found")
    
    return best_centers, best_radii, best_sum
