import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
from itertools import combinations

def generate_lattice(s):
    """Generate a hexagonal lattice covering the unit square."""
    sqrt3 = np.sqrt(3)
    rows = int(np.ceil(1.0 / (s * sqrt3 / 2))) + 1
    cols = int(np.ceil(1.0 / s)) + 1
    points = []
    for r in range(rows):
        y = r * s * sqrt3 / 2
        if y > 1 - s / 2:
            break
        for c in range(cols):
            if r % 2 == 0:
                x = c * s
            else:
                x = (c + 0.5) * s
            if x > 1 - s / 2:
                break
            if s / 2 <= x <= 1 - s / 2 and s / 2 <= y <= 1 - s / 2:
                points.append([x, y])
    return np.array(points)

def build_milp(P):
    """Constructs a function that solves the LP relaxation given binary assignments."""
    K = P.shape[0]
    z = cp.Variable(K, boolean=True)
    r = cp.Variable(K, nonneg=True)
    
    # Compute wall distances for each point
    wL = P[:, 0]
    wR = 1 - P[:, 0]
    wB = P[:, 1]
    wT = 1 - P[:, 1]
    w = np.minimum(np.minimum(wL, wR), np.minimum(wB, wT))
    
    # Precompute distance matrix
    dist_matrix = np.zeros((K, K))
    for i in range(K):
        for j in range(i + 1, K):
            d = np.linalg.norm(P[i] - P[j])
            dist_matrix[i, j] = dist_matrix[j, i] = d

    # Build constraints
    constraints = []
    # Cardinality constraint
    constraints.append(cp.sum(z) == 26)
    # Radius limits
    for k in range(K):
        constraints.append(r[k] <= w[k] * z[k])
    # Non-overlap constraints (big-M)
    M = 1.0
    for i, j in combinations(range(K), 2):
        # When both selected, enforce minimum separation
        constraints.append(dist_matrix[i, j] >= r[i] + r[j] - M*(2 - z[i] - z[j]))
    
    # Objective
    objective = cp.Maximize(cp.sum(r))

    def solve_lp(z_fixed=None):
        if z_fixed is not None:
            # Fix some binary variables
            for k in range(K):
                if k in z_fixed:
                    z[k].fix(z_fixed[k])
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            return r.value, prob.value
        else:
            return None, None
    
    return solve_lp

def branch_and_bound(milp_solver, node_limit=20000):
    """Custom branch-and-bound solver."""
    # Initial empty set of binaries
    best_solution = None
    best_objective = -np.inf
    stack = [{}]  # stack of partial assignments
    node_count = 0
    is_complete = lambda x: len(x) == 26
    
    while stack and node_count < node_limit:
        node = stack.pop()  # pick last added (can be changed to priority-based)
        node_count += 1
        
        # Solve LP relaxation
        r_vals, obj = milp_solver(z_fixed=node)
        if obj is None or obj <= best_objective:
            continue
        
        if is_complete(node):
            # All 26 circles selected; this is a feasible integer solution
            rs = np.array(r_vals)
            total = np.sum(rs)
            if total > best_objective:
                best_objective = total
                best_solution = (node, rs)
        else:
            # Branch: choose variable farthest from integer for branching
            candidates = [k for k in range(len(r_vals)) if k not in node]
            if not candidates:
                continue
            # Prefer the one with midpoint nearest to 0.5 in fractional part
            best_k = min(candidates, key=lambda k: abs((node.get(k, 0.5)) - 0.5))
            # Create child nodes: set bit to 0 and to 1
            node0 = dict(node)
            node1 = dict(node)
            node0[best_k] = 0
            node1[best_k] = 1
            stack.append(node1)
            stack.append(node0)

    if best_solution is None:
        raise ValueError("Branch and bound did not find a solution")
    selected_indices, radii = best_solution
    selected = list(selected_indices.keys())
    return selected, radii

def refine_solution(centers, radii):
    """Use local optimization with SLSQP to improve a solution."""
    # Convert to a flat parameter vector (centers + radii)
    def obj_and_grad(x):
        n = len(centers)
        c = x[: 2 * n].reshape((n, 2))
        r = x[2 * n:]

        obj = -np.sum(r)
        # For MIN function, need the gradient of -sum(r)
        grad = np.zeros_like(x)
        grad[2 * n:] = -1.0  # gradient for radii is -1 each
        
        # Add penalties for near-zero radii
        eps = 1e-4
        for i in range(n):
            if r[i] < eps:
                obj += (eps - r[i]) ** 2
                grad[2 * n + i] -= 2 * (eps - r[i])
        return obj, grad

    def constraint_func(x):
        n = len(centers)
        c = x[: 2 * n].reshape((n, 2))
        r = x[2 * n:]
        constraints = []

        # Wall constraints r_i <= dist to wall
        for i in range(n):
            constraints.append(c[i, 0] - r[i])  # left wall
            constraints.append(1 - c[i, 0] - r[i])  # right wall
            constraints.append(c[i, 1] - r[i])  # bottom wall
            constraints.append(1 - c[i, 1] - r[i])  # top wall

        # Non-overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                dist_sq = np.sum((c[i] - c[j]) ** 2)
                constraints.append(dist_sq - (r[i] + r[j]) ** 2)
        return np.array(constraints)

    # Initial guess
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Optimize
    # Use SLSQP with bounds
    bounds = []
    for i in range(len(x0)):
        if i < 2 * len(centers):
            # Center parameter
            bounds.append((0.0, 1.0))
        else:
            # Radius parameter
            bounds.append((0.0, None))

    def constraint_eq(x):
        # return [x[0] - 0.5]  # Example constraint (not needed for equality, but leave format for template)
        return np.array([])

    def constraint_ineq(x):
        return constraint_func(x)

    res = minimize(
        fun=lambda x: obj_and_grad(x)[0],
        x0=x0,
        method='SLSQP',
        jac=lambda x: obj_and_grad(x)[1],
        bounds=bounds,
        constraints={'type': 'ineq', 'fun': constraint_ineq},
        options={'ftol': 1e-9, 'maxiter': 2000}
    )
    if not res.success:
        # Fallback: recreate if not improving much
        print(f"SLSQP refinement failed with message: {res.message}")

    refined_centers = res.x[: 2 * len(centers)].reshape((len(centers), 2))
    refined_radii = res.x[2 * len(centers):]

    # Revalidate and clean up
    sum_radii = np.sum(refined_radii)
    return refined_centers, refined_radii, sum_radii

def run_packing():
    # Generate a fine hexagonal lattice
    lattice_points = generate_lattice(0.125)
    # Build the MILP model
    milp_func = build_milp(lattice_points)
    # Solve with branch-and-bound
    selected_indices, radii = branch_and_bound(milp_func)
    # Extract the selected centers
    centers = lattice_points[selected_indices]
    # Refine the solution
    ref_centers, ref_radii, total = refine_solution(centers, radii)
