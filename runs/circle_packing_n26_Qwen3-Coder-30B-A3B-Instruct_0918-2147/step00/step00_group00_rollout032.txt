import numpy as np
from scipy.optimize import minimize

def make_hex_lattice(n, margin=0.05):
    # Generate a hexagonal lattice pattern that fits in the unit square
    sqrt3 = np.sqrt(3)
    dy = 1.0 / (np.ceil(np.sqrt(n)) + 0.5)
    dx = dy / sqrt3 * 2
    points = []
    i = 0
    while len(points) < n:
        for j in range(int(np.ceil(n / 1.5)) + 2):
            x = i * dx + (j % 2) * dx / 2
            y = j * dy
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append([x, y])
            if len(points) >= n:
                break
        i += 1
    points = np.array(points[:n])
    # Center the pattern
    min_x, max_x = np.min(points[:, 0]), np.max(points[:, 0])
    min_y, max_y = np.min(points[:, 1]), np.max(points[:, 1])
    center_x = (min_x + max_x) / 2
    center_y = (min_y + max_y) / 2
    offset_x = 0.5 - center_x
    offset_y = 0.5 - center_y
    points[:, 0] += offset_x
    points[:, 1] += offset_y
    # Apply margin
    points[:, 0] = np.clip(points[:, 0], margin, 1 - margin)
    points[:, 1] = np.clip(points[:, 1], margin, 1 - margin)
    return points

def objective(z):
    """Total negative sum of radii"""
    return -np.sum(z[2::3])

def boundary_constraints(z):
    """Inequality constraints: each circle must be within the square"""
    n = len(z) // 3
    constraints = []
    for i in range(n):
        x, y, r = z[3*i], z[3*i+1], z[3*i+2]
        constraints.append(x - r)        # x - r >= 0
        constraints.append(1 - x - r)    # 1 - x - r >= 0
        constraints.append(y - r)        # y - r >= 0
        constraints.append(1 - y - r)    # 1 - y - r >= 0
    return np.array(constraints)

def pairwise_constraints(z):
    """Inequality constraints: non-overlapping condition"""
    n = len(z) // 3
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            x1, y1, r1 = z[3*i], z[3*i+1], z[3*i+2]
            x2, y2, r2 = z[3*j], z[3*j+1], z[3*j+2]
            dist_sq = (x1 - x2)**2 + (y1 - y2)**2
            sum_radii_sq = (r1 + r2)**2
            constraints.append(dist_sq - sum_radii_sq)
    return np.array(constraints)

def jacobian_boundary(z):
    """Analytic Jacobian for boundary constraints"""
    n = len(z) // 3
    jac = np.zeros((4 * n, 3 * n))
    for i in range(n):
        jac[4*i, 3*i] = 1.0
        jac[4*i, 3*i+2] = -1.0
        jac[4*i+1, 3*i] = -1.0
        jac[4*i+1, 3*i+2] = -1.0
        jac[4*i+2, 3*i+1] = 1.0
        jac[4*i+2, 3*i+2] = -1.0
        jac[4*i+3, 3*i+1] = -1.0
        jac[4*i+3, 3*i+2] = -1.0
    return jac

def jacobian_pairwise(z):
    """Analytic Jacobian for pairwise constraints"""
    n = len(z) // 3
    jac = np.zeros((n * (n - 1) // 2, 3 * n))
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            x1, y1, r1 = z[3*i], z[3*i+1], z[3*i+2]
            x2, y2, r2 = z[3*j], z[3*j+1], z[3*j+2]
            
            # Gradient of (x1 - x2)^2 + (y1 - y2)^2 - (r1 + r2)^2
            # with respect to (x1, y1, r1, x2, y2, r2)
            grad = np.zeros(6)
            grad[0] = 2*(x1 - x2)
            grad[1] = 2*(y1 - y2)
            grad[2] = -2*(r1 + r2)
            grad[3] = -2*(x1 - x2)
            grad[4] = -2*(y1 - y2)
            grad[5] = -2*(r1 + r2)
            
            jac[idx, 3*i:3*i+3] = grad[:3]
            jac[idx, 3*j:3*j+3] = grad[3:]
            idx += 1
    return jac

def solve_nlp(initial_z):
    bounds = [(0, 1), (0, 1), (0, 0.5)] * 26
    
    cons = [
        {'type': 'ineq', 'fun': boundary_constraints, 'jac': jacobian_boundary},
        {'type': 'ineq', 'fun': pairwise_constraints, 'jac': jacobian_pairwise}
    ]

    result = minimize(
        objective,
        initial_z,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-9, 'maxiter': 1000, 'disp': False}
    )
    return result.x

def nudge_centers(centers, radii, step=0.001):
    """Move circles that are tight against a wall by a small step away from the wall"""
    modified_centers = centers.copy()
    for i in range(26):
        x, y = centers[i]
        r = radii[i]
        
        # Check which walls are tight
        left = x - r <= 1e-10
        right = 1 - x - r <= 1e-10
        bottom = y - r <= 1e-10
        top = 1 - y - r <= 1e-10
        
        # If near a wall, nudge it slightly outward
        if left:
            modified_centers[i, 0] += step
        if right:
            modified_centers[i, 0] -= step
        if bottom:
            modified_centers[i, 1] += step
        if top:
            modified_centers[i, 1] -= step
    return modified_centers

    initial_radii = np.full(26, 0.02)  # Initial small radii

    # Solve optimization problem

    # Extract final centers and radii

        final_centers = final_z[::3].reshape((26, 2))
        final_radii = final_z[2::3]
        new_sum = np.sum(final_radii)
        if abs(new_sum - prev_sum) < 1e-6:
            break
        # Apply nudges
        final_centers = nudge_centers(final_centers, final_radii, 0.001)
