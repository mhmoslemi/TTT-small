import numpy as np
from scipy.optimize import minimize

def init_grid(n_circles=26, spacing=0.07):
    rows = 5
    cols = 6
    sqrt3 = np.sqrt(3)
    y_coords = []
    x_coords = []
    for k in range(rows):
        y = spacing / 2 + k * spacing * sqrt3 / 2
        if k % 2 == 0:
            x_offset = 0
        else:
            x_offset = spacing / 2
        for m in range(cols):
            x = spacing / 2 + m * spacing + x_offset
            if 0 <= x <= 1 and 0 <= y <= 1:
                if x > 0.02 and x < 0.98 and y > 0.02 and y < 0.98:
                    x_coords.append(x)
                    y_coords.append(y)
                    if len(x_coords) == n_circles:
                        break
        if len(x_coords) == n_circles:
            break

    centers = np.column_stack([x_coords, y_coords])
    np.random.seed(0)
    centers += np.random.uniform(-0.001, 0.001, centers.shape)

    radii = np.full(n_circles, 0.001)
    return centers, radii

def compute_forces(centers, radii, k_wall=10., k_pair=5.):
    n = centers.shape[0]
    forces = np.zeros_like(centers)
    wall_dists = np.column_stack([
        centers[:, 0] - radii,
        1 - centers[:, 0] - radii,
        centers[:, 1] - radii,
        1 - centers[:, 1] - radii,
    ])

    F_wall = np.zeros_like(centers)
    for i in range(4):
        dist = wall_dists[:, i]
        F_wall[:, i//2] += np.where(dist < 0, k_wall * (-dist), 0) * (-1)**i * (i % 2 == 0)

    F_pair = np.zeros_like(centers)
    for i in range(n):
        for j in range(i+1, n):
            diff = centers[i] - centers[j]
            dist = np.linalg.norm(diff)
            if dist > 0:
                gap = dist - (radii[i] + radii[j])
                if gap < 0:
                    F_pair[i] += k_pair * (-gap) * diff / dist
                    F_pair[j] -= k_pair * (-gap) * diff / dist

    return F_wall + F_pair

def grow_radii(centers, radii, forces, dt=0.02, alpha=1e-3):
    n = centers.shape[0]
    dot_radii = np.full(n, alpha)
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        hit_wall = (x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1)
        for j in range(n):
            if i != j:
                dist = np.linalg.norm(centers[i] - centers[j])
                if dist < radii[i] + radii[j]:
                    hit_wall = True
                    break
        if hit_wall:
            dot_radii[i] = 0
    radii += dt * dot_radii
    return radii, dot_radii

def fixed_centers_max_radii(centers, radii,tol=1e-9,max_iter=30):
    n = centers.shape[0]
    old_radii = radii.copy()
    for _ in range(max_iter):
        new_radii = np.zeros_like(radii)
        for i in range(n):
            min_dist = np.inf
            for j in range(n):
                if i == j:
                    continue
                dist = np.linalg.norm(centers[i] - centers[j])
                min_dist = min(min_dist, dist - radii[j])
            new_radii[i] = min(centers[i, 0], 1 - centers[i, 0],
                               centers[i, 1], 1 - centers[i, 1], min_dist)
        radii = new_radii
        if np.allclose(old_radii, radii, atol=tol):
            break
        old_radii = radii.copy()
    return radii

def objective_sum_radii(centers, all_radii):
    n = len(centers) // 2
    radii = all_radii[:n]
    return -fixed_centers_max_radii(centers.reshape(n, 2), radii).sum()

def run_packing():
    np.random.seed(0)

    centers, radii = init_grid(26)

    dt = 0.02
    damping_coeff = 1.0
    max_steps = 5000
    min_growth = 1e-6
    min_force = 1e-6

    velocities = np.zeros_like(centers)

    for step in range(max_steps):
        forces = compute_forces(centers, radii)
        # Update velocities
        velocities += dt * (forces - damping_coeff * velocities)
        # Update centers
        centers += dt * velocities
        # Project centers to square
        centers = np.clip(centers, 0, 1)
        # Grow radii
        radii, dot_radii = grow_radii(centers, radii, forces, dt, 1e-3)
        # Check termination conditions
        if np.all(dot_radii < min_growth) and np.max(np.linalg.norm(forces, axis=1)) < min_force:
            break

    radii = fixed_centers_max_radii(centers, radii)

    # Use L-BFGS-B with optimization wrapped around a custom objective
    def opt_obj(x_vec):
        centers_flat = x_vec.reshape(26, 2)
        max_radii = fixed_centers_max_radii(centers_flat, radii)  # r_i should be adapted too?
        return -np.sum(max_radii)

    def opt_constr(x_vec):
        centers_flat = x_vec.reshape(26, 2)
        # barriers on box constraints
        penalties = 0
        for i in range(26):
            x, y = centers_flat[i]
            penalties += max(0, -x)**2 + max(0, x - 1)**2 + max(0, -y)**2 + max(0, y - 1)**2
        return penalties

    # Initial guess
    x0 = centers.flatten()

    # Optimize
    result = minimize(
        fun=opt_obj,
        x0=x0,
        method='L-BFGS-B',
        options={'maxiter': 2000, 'gtol': 1e-9},
    )

    opt_centers = result.x.reshape(26, 2)
    final_radii = fixed_centers_max_radii(opt_centers, radii)

    sum_radii = np.sum(final_radii)
