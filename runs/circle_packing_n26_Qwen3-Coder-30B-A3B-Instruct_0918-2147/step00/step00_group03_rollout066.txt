import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial.distance import cdist

def build_lp_model(n):
    # Variables: [x1,y1,r1,x2,y2,r2,...,x_n,y_n,r_n]
    # Total variables = 3*n
    num_vars = 3 * n
    
    # Objective: maximize sum of radii -> minimize negative sum
    c = np.zeros(num_vars)
    c[2::3] = -1.0  # radii are at indices 2, 5, 8, ...

    # Prepare inequality constraints A_ub * x <= b_ub
    A_ub = []
    b_ub = []

    # Wall constraints (4 per circle): r <= x, r <= 1-x, r <= y, r <= 1-y
    for i in range(n):
        row = np.zeros(num_vars)
        row[3*i + 2] = 1.0  # r_i
        row[3*i] = -1.0     # -x_i
        A_ub.append(row)
        b_ub.append(0.0)

        row = np.zeros(num_vars)
        row[3*i + 2] = 1.0  # r_i
        row[3*i] = 1.0      # x_i
        A_ub.append(row)
        b_ub.append(1.0)

        row = np.zeros(num_vars)
        row[3*i + 2] = 1.0  # r_i
        row[3*i + 1] = -1.0 # -y_i
        A_ub.append(row)
        b_ub.append(0.0)

        row = np.zeros(num_vars)
        row[3*i + 2] = 1.0  # r_i
        row[3*i + 1] = 1.0  # y_i
        A_ub.append(row)
        b_ub.append(1.0)

    # Separation constraints (4 per pair of circles)
    for i in range(n):
        for j in range(i + 1, n):
            # x_i + r_i <= x_j - r_j  ->  x_i - x_j + r_i + r_j <= 0
            row = np.zeros(num_vars)
            row[3*i] = 1.0      # x_i
            row[3*j] = -1.0     # -x_j
            row[3*i + 2] = 1.0  # r_i
            row[3*j + 2] = 1.0  # r_j
            A_ub.append(row)
            b_ub.append(0.0)

            # x_j + r_j <= x_i - r_i  ->  x_j - x_i + r_i + r_j <= 0
            row = np.zeros(num_vars)
            row[3*i] = -1.0     # -x_i
            row[3*j] = 1.0      # x_j
            row[3*i + 2] = 1.0  # r_i
            row[3*j + 2] = 1.0  # r_j
            A_ub.append(row)
            b_ub.append(0.0)

            # y_i + r_i <= y_j - r_j  ->  y_i - y_j + r_i + r_j <= 0
            row = np.zeros(num_vars)
            row[3*i + 1] = 1.0  # y_i
            row[3*j + 1] = -1.0 # -y_j
            row[3*i + 2] = 1.0  # r_i
            row[3*j + 2] = 1.0  # r_j
            A_ub.append(row)
            b_ub.append(0.0)

            # y_j + r_j <= y_i - r_i  ->  y_j - y_i + r_i + r_j <= 0
            row = np.zeros(num_vars)
            row[3*i + 1] = -1.0 # -y_i
            row[3*j + 1] = 1.0  # y_j
            row[3*i + 2] = 1.0  # r_i
            row[3*j + 2] = 1.0  # r_j
            A_ub.append(row)
            b_ub.append(0.0)

    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    return c, A_ub, b_ub

def solve_lp(c, A_ub, b_ub, n):
    # Solve the linear program using HiGHS
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})

    if not res.success:
        raise RuntimeError("Failed to solve LP")

    x = res.x
    centers = np.array([[x[3*i], x[3*i+1]] for i in range(n)])
    radii = np.array([x[3*i+2] for i in range(n)])

    # Determine active constraints using tight tolerance
    active = []

    # Check wall constraints
    for i in range(n):
        r = radii[i]
        x_i, y_i = centers[i]
        # r \le x_i means r - x_i <= 0
        if abs(r - x_i) < 1e-9:
            active.append(('wall', i, 'leftx'))
        if abs(r - (1.0 - x_i)) < 1e-9:
            active.append(('wall', i, 'rightx'))
        if abs(r - y_i) < 1e-9:
            active.append(('wall', i, 'lefty'))
        if abs(r - (1.0 - y_i)) < 1e-9:
            active.append(('wall', i, 'righty'))

    # Check separation constraints
    for i in range(n):
        for j in range(i + 1, n):
            x_i, y_i = centers[i]
            x_j, y_j = centers[j]
            r_i, r_j = radii[i], radii[j]
            # x_i + r_i <= x_j - r_j  ->  x_i - x_j + r_i + r_j <= 0
            if abs((x_i - x_j + r_i + r_j)) < 1e-9:
                active.append(('sepx', i, j))
            # x_j + r_j <= x_i - r_i  ->  x_j - x_i + r_i + r_j <= 0
            if abs((x_j - x_i + r_i + r_j)) < 1e-9:
                active.append(('sepx', j, i))
            # y_i + r_i <= y_j - r_j  ->  y_i - y_j + r_i + r_j <= 0
            if abs((y_i - y_j + r_i + r_j)) < 1e-9:
                active.append(('sepy', i, j))
            # y_j + r_j <= y_i - r_i  ->  y_j - y_i + r_i + r_j <= 0
            if abs((y_j - y_i + r_i + r_j)) < 1e-9:
                active.append(('sepy', j, i))

    return centers, radii, active

def detect_diagonal_conflicts(centers, radii):
    n = len(centers)
    conflicts = []

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            combined_r = radii[i] + radii[j]
            if dist < combined_r - 1e-12:
                conflicts.append((i, j))
    return conflicts

def max_radius_projection(centers, radii):
    # Reconstruct maximal radii inside boundaries and ensuring no overlaps
    changed = True
    n = len(centers)
    max_iter = 50
    for _ in range(max_iter):
        old_radii = radii.copy()
        success = True
        for i in range(n):
            x, y = centers[i]
            # Current radius
            current_r = radii[i]
            # Update radius to match actual boundaries
            r_max = min(x, 1 - x, y, 1 - y)
            # Intersect with pairwise distances to other circles
            for j in range(n):
                if i != j:
                    dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                    r_pair = dist - radii[j]
                    if r_pair > 0:  # Only if permissible
                        r_max = min(r_max, r_pair)

            radii[i] = r_max
            if abs(radii[i] - current_r) > 1e-10:
                success = False
        if success:
            break
    return centers, radii

def run_packing():

    # Resolve diagonal conflicts reliably by few iterations
    conflicts = detect_diagonal_conflicts(centers, radii)
    if len(conflicts) > 0:
        centers, radii = resolve_conflicts(centers, radii, active_set, conflicts)

    # Additional polishing to bump up sum
    # We re-check conflicts after first resolution (valid state post-adjust)
    conflicts = detect_diagonal_conflicts(centers, radii)
    while len(conflicts) > 0:
        print(f"Resolving {len(conflicts)} overlapping pairs...")
        centers, radii = resolve_conflicts(centers, radii, active_set, conflicts)
        conflicts = detect_diagonal_conflicts(centers, radii)

    # Final projection to exact maximal radii within constraints
    centers, radii = max_radius_projection(centers, radii)

    # Sum of radii
    sum_r = float(radii.sum())

    # Validation
    valid, msg = validate_packing(centers, radii)
    if not valid and 'over' in msg:
        # Try collision resolution at high cost versus validation - last resort
        for i in range(n):
            if radii[i] < 1e-9:
                print(f"Excessively small radii detected in circle {i}")
        raise ValueError("Invalid packing despite attempt to resolve overlaps")

# Provided validation function (read-only)
def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
