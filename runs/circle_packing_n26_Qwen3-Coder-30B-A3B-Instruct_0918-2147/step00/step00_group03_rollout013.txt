import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

# Constants
N_CIRCLES = 26
WALL_STIFFNESS = 10.0
PAIRWISE_STIFFNESS = 5.0
DAMPING = 1.0
BASE_GROWTH_RATE = 1e-3
TIME_STEP = 0.02
FORCE_THRESHOLD = 1e-6
MAX_SIM_STEPS = 5000
MAX_REFINE_ITERS = 2000
TOLERANCE = 1e-9
MAX_RADIUS_CONVERGE = 1e-9
Epsilon = 1e-12

def init_grid():
    s = 0.07
    y_coords = []
    x_coords = []
    
    for k in range(10):
        y = s / 2 + k * s * np.sqrt(3) / 2
        if y > 1 - s:
            continue
        y_coords.append(y)
        row_x = []
        for m in range(20):
            x = s / 2 + m * s
            if x > 1 - s:
                break
            row_x.append(x)
        if k % 2 == 1:
            row_x = [x + s / 2 for x in row_x]
        x_coords.extend(row_x)
        if len(x_coords) >= N_CIRCLES:
            break

    xcoords = np.array(x_coords[:N_CIRCLES])
    ycoords = np.array(y_coords[:N_CIRCLES])
    if len(xcoords) < N_CIRCLES:
        # pad with last valid coordinates
        xcoords = np.pad(xcoords, (0, N_CIRCLES - len(xcoords)), constant_values=xcoords[-1])
        ycoords = np.pad(ycoords, (0, N_CIRCLES - len(ycoords)), constant_values=ycoords[-1])
    
    # Create initial centers (centered in grid points)
    centers = np.column_stack([xcoords, ycoords])
    # Add jitter to break symmetry
    np.random.seed(0)
    jitter = np.random.uniform(-0.001, 0.001, size=(N_CIRCLES, 2))
    centers += jitter
    # Clip to [0, 1]
    centers = np.clip(centers, 0, 1)
    
    # Filter out circles that are too close to walls
    keep = []
    for i in range(N_CIRCLES):
        x, y = centers[i]
        if x > 0.02 and x < 0.98 and y > 0.02 and y < 0.98:
            keep.append(i)
    centers = centers[keep]
    radii = np.full(len(keep), 0.001)
    
    if len(keep) < N_CIRCLES:
        # fill with last ones
        n_extra = N_CIRCLES - len(keep)
        extra_centers = centers[-1:] * np.ones((n_extra, 2))
        centers = np.vstack([centers, extra_centers])
        radii = np.append(radii, np.full(n_extra, 0.001))
    return centers, radii

def compute_forces(centers, radii):
    n = centers.shape[0]
    forces = np.zeros((n, 2))
    
    # Wall forces
    dx_l = centers[:, 0] - radii
    dx_r = 1 - (centers[:, 0] + radii)
    dy_b = centers[:, 1] - radii
    dy_t = 1 - (centers[:, 1] + radii)
    
    for i in range(n):
        fl = np.maximum(0., -dx_l[i]) * WALL_STIFFNESS * np.array([-1., 0.])
        fr = np.maximum(0., -dx_r[i]) * WALL_STIFFNESS * np.array([1., 0.])
        fb = np.maximum(0., -dy_b[i]) * WALL_STIFFNESS * np.array([0., -1.])
        ft = np.maximum(0., -dy_t[i]) * WALL_STIFFNESS * np.array([0., 1.])
        forces[i] += fl + fr + fb + ft

    # Pairwise forces
    if n < 2:
        return forces

    # compute pairwise distances
    dist_matrix = cdist(centers, centers)
    for i in range(n):
        for j in range(i+1, n):
            d_ij = dist_matrix[i, j]
            r_sum = radii[i] + radii[j]
            if d_ij < r_sum - Epsilon:
                # overlap
                diff = centers[i] - centers[j]
                mag = np.linalg.norm(diff)
                if mag > Epsilon:
                    force = PAIRWISE_STIFFNESS * (r_sum - d_ij) * diff / mag
                    forces[i] -= force
                    forces[j] += force

    return forces

def grow_radii(centers, radii, forces):
    n = centers.shape[0]
    dot_r = np.zeros(n)
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Check if touching any boundary, or any pair
        touching_wall = (x - r < 0) or (x + r > 1) or (y - r < 0) or (y + r > 1)
        # Compute current pairwise overlaps
        overlap = False
        for j in range(n):
            if i == j:
                continue
            d_ij = np.linalg.norm(centers[i] - centers[j])
            if d_ij < radii[i] + radii[j] - Epsilon:
                overlap = True
                break
        if not touching_wall and not overlap:
            dot_r[i] = BASE_GROWTH_RATE
    return dot_r

def max_radii_fixed_centers(centers, radii):
    n = centers.shape[0]
    new_radii = np.array(radii)
    
    for _ in range(100):  # arbitrary high bound
        old_radii = np.array(new_radii)
        for i in range(n):
            x, y = centers[i]
            # Compute min distance to walls
            d_left = x
            d_right = 1 - x
            d_bottom = y
            d_top = 1 - y
            min_wall_dist = min(d_left, d_right, d_bottom, d_top)

            # Compute min distance to other centers
            min_center_dist = np.inf
            for j in range(n):
                if i == j:
                    continue
                d_ij = np.linalg.norm(centers[i] - centers[j])
                if d_ij > 0:
                    dist = d_ij - radii[j]
                    if dist < min_center_dist:
                        min_center_dist = dist

            new_radii[i] = min(min_wall_dist, min_center_dist)
        if np.allclose(old_radii, new_radii, atol=MAX_RADIUS_CONVERGE):
            break

    return new_radii

def objective_sum_radii(centers_flat):
    n = len(centers_flat) // 2
    centers = centers_flat.reshape(n, 2)
    # Compute radius via max radii for current centers
    radii = max_radii_fixed_centers(centers, np.full(n, 0.001))
    return -np.sum(radii)

def run_packing():
    centers, radii = init_grid()
    vel = np.zeros_like(centers)

    # Simulation loop
    step = 0
    while step < MAX_SIM_STEPS:
        forces = compute_forces(centers, radii)
        dot_r = grow_radii(centers, radii, forces)
        # update velocity
        vel += TIME_STEP * (forces - DAMPING * vel)
        # update positions
        centers += TIME_STEP * vel
        # enforce box constraints (clip)
        centers = np.clip(centers, 0, 1)
        # update radii
        radii += TIME_STEP * dot_r
        # enforce non-negativity
        radii = np.maximum(radii, 0)
        
        # check if done
        if np.max(np.abs(forces)) < FORCE_THRESHOLD:
            if np.all(dot_r <= 1e-12):
                break
        step += 1

    # Compute exact maximal radii for final centres
    radii = max_radii_fixed_centers(centers, radii)

    # Convert centers to flattened format for optimization
    centers_flat = centers.flatten()

    # Optimization step: maximize sum of radii
    # Minimize -sum_radii with bounds [0,1]
    bnds = [(0, 1) for _ in range(len(centers_flat))]

    res = minimize(
        objective_sum_radii,
        centers_flat,
        method='L-BFGS-B',
        bounds=bnds,
        options={'maxiter': MAX_REFINE_ITERS, 'gtol': TOLERANCE}
    )

    # Reconstruct centers
    centers_opt = res.x.reshape(-1, 2)
    # Make sure they are within [0, 1]
    centers_opt = np.clip(centers_opt, 0, 1)
    # Recompute max radii from optimized centers
    radii_opt = max_radii_fixed_centers(centers_opt, np.full(N_CIRCLES, 0.001))

    # Final validation
    _, status = validate_packing(centers_opt, radii_opt)
    if status != "ok":
        raise RuntimeError(f"Validation failed: {status}")

    sum_radii = np.sum(radii_opt)
    return centers_opt, radii_opt, sum_radii
