```python
import numpy as np
from scipy.optimize import minimize

NUM_CIRCLES = 26
NUM_RESTARTS = 5
HEX_ROWS = 5
HEX_COLS = 6
HEX_SPACING = 0.2

def hex_lattice_points(n_rows, spacing):
    points = []
    for i in range(n_rows):
        y = i * spacing * np.sqrt(3) / 2
        for j in range(HEX_COLS):
            x = j * spacing + (i % 2) * spacing / 2
            if x <= 1 and y <= 1:
                points.append([x, y])
    return np.array(points)[:NUM_CIRCLES]

def initial_guess(seed):
    np.random.seed(seed)
    lattice = hex_lattice_points(HEX_ROWS, HEX_SPACING)
    # Add jitter
    jitter = np.random.uniform(-0.02, 0.02, (NUM_CIRCLES, 2))
    centers = np.clip(lattice + jitter, 0, 1)
    radii = np.full(NUM_CIRCLES, 0.02)
    vars0 = np.empty(3 * NUM_CIRCLES)
    vars0[0::3] = centers[:, 0]
    vars0[1::3] = centers[:, 1]
    vars0[2::3] = radii
    return vars0

def objective(vars):
    return -np.sum(vars[2::3])

def constraint_boundary(vars):
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    # x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    # return [x - r, 1 - x - r, y - r, 1 - y - r]
    return np.concatenate([
        x - r,
        1 - x - r,
        y - r,
        1 - y - r
    ])

def constraint_nonoverlap(vars):
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    n = NUM_CIRCLES
    margins = []
    for i in range(n):
        for j in range(i+1, n):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dist = np.sqrt(dx*dx + dy*dy)
            margin = dist - (r[i] + r[j])
            margins.append(margin)
    return np.array(margins)

def postprocess(vars):
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Clip centers to unit square and preserve radii
    x = np.clip(x, r, 1 - r)
    y = np.clip(y, r, 1 - r)
    
    # Enforce non-overlap (very slight correction)
    n = NUM_CIRCLES
    for i in range(n):
        for j in range(i+1, n):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dist = np.sqrt(dx*dx + dy*dy)
            overlap = dist - (r[i] + r[j])
            if overlap < -1e-12:
                # Overlap: move centers apart
                if dist > 1e-12:
                    move = -overlap / 2
                    dx = dx / dist * move
                    dy = dy / dist * move
                    x[i] += dx
                    y[i] += dy
                    x[j] -= dx
                    y[j] -= dy
                    # Reduce radii
                    r[i] -= overlap / 2
                    r[j] -= overlap / 2
                    r[i] = max(0, r[i])
                    r[j] = max(0, r[j])
    
    # Recompute
    new_vars = vars.copy()
    new_vars[0::3] = x
    new_vars[1::3] = y
    new_vars[2::3] = r
    return new_vars

def run_packing():
    best_sum = -np.inf
    best_vars = None
    
    # Global constraints
    bc = {'type': 'ineq', 'fun': constraint_boundary}
    nc = {'type': 'ineq', 'fun': constraint_nonoverlap}
    global_constraints = [bc, nc]
    
    # Bounds (x, y, r) for each circle
    global_bounds = [(0, 1), (0, 1), (0, 0.5)] * NUM_CIRCLES
    
    for seed in range(NUM_RESTARTS):
        vars0 = initial_guess(seed)
        res = minimize(objective, vars0,
                      method='SLSQP',
                      bounds=global_bounds,
                      constraints=global_constraints,
                      options={'maxiter': 3000, 'ftol': 1e-7})
        if not res.success:
            continue
        vars_corr = postprocess(res.x)
        s = np.sum(vars_corr[2::3])
        if s > best_sum:
            best_sum = s
            best_vars = vars_corr
    
    if best_vars is None:
        raise RuntimeError("No valid packing found")
    
    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii = best_vars[2::3]
    return centers, radii, float(best_sum)
```
