import numpy as np
from scipy.stats.qmc import Sobol
from scipy.spatial.distance import cdist

def initialize_seeds(n):
    sobol = Sobol(2, seed=42)
    points = sobol.random(n)
    delta = 0.02
    points = delta + (1 - 2 * delta) * points
    return points

def compute_feasible_radii(centers):
    n = centers.shape[0]
    # Distance matrix
    D = cdist(centers, centers)
    # Nearest neighbor distances (excluding self)
    nn_dists = np.partition(D, 1, axis=1)[:, 1]
    # Wall distances
    wall_dists = np.minimum(centers[:, 0], 1 - centers[:, 0])
    wall_dists = np.minimum(wall_dists, centers[:, 1])
    wall_dists = np.minimum(wall_dists, 1 - centers[:, 1])
    # Feasible radii
    radii = np.minimum(0.5 * nn_dists, wall_dists)
    return radii

def compute_slack(radii, nn_dists):
    # Slack = 0.5 * nn_dist - radius
    slack = 0.5 * nn_dists - radii
    slack = np.maximum(slack, 0.0)
    return slack

def redistribute_radii(radii, slack, alpha):
    total_slack = np.sum(slack)
    if total_slack == 0:
        return radii
    # Amount to increase each wall-limited circle
    delta = alpha * slack
    # Redistribute
    new_radii = radii + delta
    return new_radii

def push_away(centers, radii, beta, eps):
    n = centers.shape[0]
    new_centers = centers.copy()
    D = cdist(centers, centers)
    for i in range(n):
        for j in range(n):
            if i == j:
                continue
            dist = D[i, j]
            if dist < radii[i] + radii[j] + eps:
                # Overlap
                overlap = radii[i] + radii[j] + eps - dist
                # Push away
                v_ij = centers[i] - centers[j]
                v_len = np.sqrt(np.sum(v_ij ** 2))
                if v_len > eps:
                    push = beta * overlap * (v_ij / v_len)
                    new_centers[i] += push
    # Clip to interior
    new_centers = np.clip(new_centers, eps, 1 - eps)
    return new_centers

def iterate(centers, radii, alpha, beta, max_iter):
    for _ in range(max_iter):
        # Compute nearest neighbor distances
        D = cdist(centers, centers)
        nn_dists = np.partition(D, 1, axis=1)[:, 1]
        # Compute slack
        slack = compute_slack(radii, nn_dists)
        total_slack = np.sum(slack)
        if total_slack < 1e-5:
            break
        # Redistribute
        new_radii = redistribute_radii(radii, slack, alpha)
        # Push away
        new_centers = push_away(centers, new_radii, beta, 1e-8)
        # Recompute radii
        centers = new_centers
        radii = compute_feasible_radii(centers)
    return centers, radii

def repair(centers, radii, eps_final):
    # Shrink all radii
    radii = np.maximum(0, radii - eps_final)
    # Clamp centers
    centers = np.clip(centers, eps_final, 1 - eps_final)
    return centers, radii

def run_packing():
    n = 26
    # Step 1: Initialize seeds
    centers = initialize_seeds(n)
    # Step 2: Compute initial feasible radii
    radii = compute_feasible_radii(centers)
    # Step 3: Iterate to improve packing
    centers, radii = iterate(centers, radii, alpha=0.4, beta=0.3, max_iter=200)
    # Step 4: Final repair
    eps_final = 5e-10
    centers, radii = repair(centers, radii, eps_final)
    # Sum of radii
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
