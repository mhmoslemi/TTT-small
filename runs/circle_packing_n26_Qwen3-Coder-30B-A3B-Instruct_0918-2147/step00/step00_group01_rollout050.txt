import numpy as np
from scipy.optimize import differential_evolution
from scipy.spatial.distance import cdist

def compute_radii(centers):
    n = centers.shape[0]
    radii = np.zeros(n)
    
    # Wall distances
    x = centers[:, 0]
    y = centers[:, 1]
    wall_distances = np.minimum.reduce([x, 1 - x, y, 1 - y])
    
    # Pairwise distances
    distances = cdist(centers, centers)
    np.fill_diagonal(distances, np.inf)
    
    # Nearest neighbor distances (half)
    nearest_distances = 0.5 * np.min(distances, axis=1)
    
    # Radii are min of wall and neighbor distances
    radii = np.minimum(wall_distances, nearest_distances)
    
    return radii

def neg_sum_radii(z):
    centers = z.reshape(-1, 2)
    radii = compute_radii(centers)
    return -np.sum(radii)

        maxiter=2000,
        popsize=10,
        tol=1e-7,

    )
    
    # Convert to centers
    centers = result_de.x.reshape(-1, 2)
