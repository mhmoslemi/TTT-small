import numpy as np
from scipy.optimize import minimize
from ortools.linear_solver import pywraplp

def generate_candidates(g, radius_levels):
    candidates = []
    grid_points = np.arange(0, 1 + g/2, g)
    for x in grid_points:
        for y in grid_points:
            for r in radius_levels:
                if x - r >= 0 and x + r <= 1 and y - r >= 0 and y + r <= 1:
                    candidates.append((x, y, r))
    return candidates

def build_milp(candidates):
    solver = pywraplp.Solver.CreateSolver('SCIP')
    if not solver:
        solver = pywraplp.Solver.CreateSolver('CBC')

    n = len(candidates)
    z = [solver.BoolVar(f'z_{i}') for i in range(n)]

    # Each grid point has at most one circle
    grid_dict = {}
    for i, (x, y, r) in enumerate(candidates):
        key = (round(x/g), round(y/g))
        if key not in grid_dict:
            grid_dict[key] = []
        grid_dict[key].append(i)

    for key, indices in grid_dict.items():
        solver.Add(solver.Sum([z[i] for i in indices]) <= 1)

    # Exactly 26 circles
    solver.Add(solver.Sum(z) == 26)

    # Pairwise non-overlap
    for i in range(n):
        for j in range(i+1, n):
            x1, y1, r1 = candidates[i]
            x2, y2, r2 = candidates[j]
            dx = solver.NumVar(0, 1, f'dx_{i}_{j}')
            dy = solver.NumVar(0, 1, f'dy_{i}_{j}')
            
            solver.Add(dx >= x1 - x2)
            solver.Add(dx >= x2 - x1)
            solver.Add(dy >= y1 - y2)
            solver.Add(dy >= y2 - y1)
            
            solver.Add(dx + dy >= r1 + r2)
            solver.Add(dx - dy >= r1 + r2 - 1)
            
            solver.Add(dx <= 1 * (z[i] + z[j]))
            solver.Add(dy <= 1 * (z[i] + z[j]))

    # Objective: maximize sum of radii
    solver.Maximize(solver.Sum([r * z[i] for i, (_, _, r) in enumerate(candidates)]))

    return solver, z

def solve_milp(solver, z):
    status = solver.Solve()
    if status != pywraplp.Solver.OPTIMAL:
        raise RuntimeError("Failed to solve MILP for discrete selection")
    selected = [i for i in range(len(z)) if z[i].solution_value() > 0.5]
    return selected

def extract_initial_solution(selected_indices, candidates):
    centers = np.array([[candidates[i][0], candidates[i][1]] for i in selected_indices])
    radii = np.array([candidates[i][2] for i in selected_indices])
    return centers, radii

def shrink_initial(radii, delta):
    return radii - delta

def pack_variables(centers, radii):
    # Pack [x1,y1,r1,x2,y2,r2,...]
    z = []
    for i in range(len(centers)):
        z.extend([centers[i][0], centers[i][1], radii[i]])
    return np.array(z)

def unpack_solution(z):
    n = len(z) // 3
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i][0] = z[3*i]
        centers[i][1] = z[3*i+1]
        radii[i] = z[3*i+2]
    return centers, radii

def compute_radius_sum(radii):
    return np.sum(radii)

def define_objective(z):
    _, radii = unpack_solution(z)
    return -np.sum(radii)

def define_jacobian(z):
    _, radii = unpack_solution(z)
    n = len(radii)
    jac = np.zeros(3 * n)
    jac[2::3] = -1.0  # gradient of sum wrt radii
    return jac

def define_constraints(z):
    centers, radii = unpack_solution(z)
    constraints = []
    n = len(centers)
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # r <= x -> -r + x >= 0
        def boundary_constraint_r_le_x():
            return r - x
        def boundary_jac_r_le_x():
            return np.array([0, 0, 1, 0, 0, 0])  # Only r element's gradient

        # r <= 1-x -> -r + 1 - x >= 0
        def boundary_constraint_r_le_1_minus_x():
            return r - (1 - x)
        def boundary_jac_r_le_1_minus_x():
            return np.array([0, 0, 1, 0, 0, 0])  # Only r grad

        # r <= y -> -r + y >= 0
        def boundary_constraint_r_le_y():
            return r - y
        def boundary_jac_r_le_y():
            return np.array([0, 0, 1, 0, 0, 0])

        # r <= 1-y -> -r + 1 - y >= 0
        def boundary_constraint_r_le_1_minus_y():
            return r - (1 - y)
        def boundary_jac_r_le_1_minus_y():
            return np.array([0, 0, 1, 0, 0, 0])  # Only r grad

        constraints.append({
            'fun': boundary_constraint_r_le_x,
            'jac': boundary_jac_r_le_x
        })
        constraints.append({
            'fun': boundary_constraint_r_le_1_minus_x,
            'jac': boundary_jac_r_le_1_minus_x
        })
        constraints.append({
            'fun': boundary_constraint_r_le_y,
            'jac': boundary_jac_r_le_y
        })
        constraints.append({
            'fun': boundary_constraint_r_le_1_minus_y,
            'jac': boundary_jac_r_le_1_minus_y
        })

    # Non-overlap constraints
    for i in range(n):
        for j in range(i+1, n):
            x1, y1 = centers[i]
            r1 = radii[i]
            x2, y2 = centers[j]
            r2 = radii[j]

            dx = x1 - x2
            dy = y1 - y2
            dist = np.sqrt(dx*dx + dy*dy)

            def constraint():
                return (r1 + r2) - dist
            def jacobian():
                # grad(g) = partial/(partial x1 y1 r1 x2 y2 r2)
                # Use [dx/dist, dy/dist, 1, -dx/dist, -dy/dist, -1]
                if dist > 1e-12:
                    scale = 1.0 / dist
                    return np.array([
                        scale * dx, scale * dy, 1,
                        -scale * dx, -scale * dy, -1
                    ])
                else:
                    return np.array([0, 0, 0, 0, 0, 0])

            constraints.append({
                'fun': constraint,
                'jac': jacobian
            })

    return constraints

def repair_solution(centers, radii):
    n = len(centers)
    for i in range(n):
        r = radii[i]
        x, y = centers[i]
        r = min(r, x, 1 - x, y, 1 - y)
        radii[i] = r

    # Handle overlaps
    MAX_ITER = 5
    for _ in range(MAX_ITER):
        changed = False
        for i in range(n):
            for j in range(i+1, n):
                x1, y1 = centers[i]
                x2, y2 = centers[j]
                r1 = radii[i]
                r2 = radii[j]
                dx = x1 - x2
                dy = y1 - y2
                dist = np.sqrt(dx*dx + dy*dy)

                overlap = (r1 + r2) - dist
                if overlap > 0:
                    delta = overlap / 2.0
                    if r1 > r2:
                        radii[i] -= delta
                        radii[j] -= delta
                    else:
                        radii[j] -= delta
                        radii[i] -= delta
                    changed = True
        
        if not changed:
            break

    return centers, radii

def validate_and_correct_solution(centers, radii):
    # 1. Ensure valid constraints
    for i in range(len(centers)):
        x, y = centers[i]
        r = radii[i]
        if x < r or x > 1 - r or y < r or y > 1 - r:
            # Failed validation
            raise ValueError("Invalid bounds.")

    # 2. Ensure no overlaps
    for i in range(len(centers)):
        for j in range(i+1, len(centers)):
            x1, y1 = centers[i]
            x2, y2 = centers[j]
            r1 = radii[i]
            r2 = radii[j]
            dx = x1 - x2
            dy = y1 - y2
            dist = np.sqrt(dx*dx + dy*dy)
            if dist < r1 + r2 - 1e-12:
                raise ValueError("Overlapping circles detected.")

    return True

def run_packing():
    g = 0.035
    radius_levels = np.arange(0.02, 0.26, 0.02)

    candidates = generate_candidates(g, radius_levels)
    solver, z = build_milp(candidates)
    selected = solve_milp(solver, z)
    centers0, radii0 = extract_initial_solution(selected, candidates)
    
    # Adjust radii slightly
    radii0 = shrink_initial(radii0, 1e-5)
    
    # Pack for NLP
    z0 = pack_variables(centers0, radii0)
    initial = z0.copy()

    # Optimize using scipy
    def obj_func(z):
        return define_objective(z)
        
    def jac_func(z):
        return define_jacobian(z)
        
    constraints = define_constraints(z0)
    bounds = [(0, 1), (0, 1), (0, 0.5)] * len(centers0)

    # Convert constraints for scipy
    scipy_constraints = []
    for c in constraints:
        def make_func(f):
            return lambda x: f()
        def make_jac(j):
            return lambda x: j()
        scipy_constraints.append({
            'fun': make_func(c['fun']),
            'jac': make_jac(c['jac']),
            'type': 'ineq'
        })
    
    # Run optimization
    result = minimize(
        fun=obj_func,
        x0=z0,
        method='trust-constr',
        jac=jac_func,
        bounds=[(0,1), (0,1), (0,0.5)] * len(centers0),
        constraints=scipy_constraints,
        options={'disp': False}
    )

    if not result.success:
        raise RuntimeError("Optimization failed")

    centers, radii = unpack_solution(result.x)
    centers, radii = repair_solution(centers, radii)
    sum_radii = compute_radius_sum(radii)

    # Final validation
    validate_packing(centers, radii)
    
    return centers, radii, sum_radii
