import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def init_candidate_grid(R, Δ):
    C = []
    for r in R:
        s = r * np.sqrt(2)
        n = int(np.ceil((1 - 2 * r) / s))
        if n <= 0:
            continue
        x_coords = np.linspace(r, 1 - r, n)
        y_coords = np.linspace(r, 1 - r, n)
        for x in x_coords:
            for y in y_coords:
                C.append((x, y, r))
    return C

def build_overlap_matrix(candidates):
    n = len(candidates)
    overlap = np.zeros((n, n), dtype=bool)
    coords = np.array([[c[0], c[1]] for c in candidates])
    radii = np.array([c[2] for c in candidates])
    dists = cdist(coords, coords)
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j]:
                overlap[i, j] = True
                overlap[j, i] = True
    return overlap

def solve_discrete_bip(candidates, overlap_matrix, target_sum):
    from scipy.optimize import milp
    from scipy.optimize import Bounds
    import cvxpy as cp

    n = len(candidates)
    # Binary variables
    z = cp.Variable(n, boolean=True)
    # Objective
    obj = cp.Maximize(cp.sum(cp.multiply([c[2] for c in candidates], z)))
    # Constraints
    constraints = [
        cp.sum(z) == 26,
    ]
    for i in range(n):
        for j in range(i + 1, n):
            if overlap_matrix[i, j]:
                constraints.append(z[i] + z[j] <= 1)
    prob = cp.Problem(obj, constraints)
    
    # Here, instead of full optimisation, use simple greedy initial approach for feasibility
    chosen = []
    chosen_ids = set()
    radii = [c[2] for c in candidates]
    coords = np.array([[c[0], c[1]] for c in candidates])
    current_sum = 0.0
    while len(chosen) < 26 and len(chosen_ids) < n:
        best_idx = -1
        best_r = -1
        for i in range(n):
            if i not in chosen_ids:
                if best_idx == -1 or radii[i] > best_r:
                    best_idx = i
                    best_r = radii[i]
        if best_idx == -1 or current_sum + best_r > target_sum:
            break
        chosen.append(best_idx)
        chosen_ids.add(best_idx)
        current_sum += best_r
        # Remove overlapping candidates
        for j in range(n):
            if j not in chosen_ids and overlap_matrix[best_idx, j]:
                chosen_ids.add(j)
    
    # If not enough, just select top 26 non-overlapping (greedy)
    if len(chosen) < 26:
        chosen = []
        chosen_ids = set()
        current_sum = 0.0
        all_keys = list(range(n))
        all_keys.sort(key=lambda i: -radii[i])
        for i in all_keys:
            if len(chosen) >= 26:
                break
            valid = True
            for j in chosen:
                if overlap_matrix[i, j]:
                    valid = False
                    break
            if valid:
                chosen.append(i)
                chosen_ids.add(i)
                current_sum += radii[i]
    # Reassign from chosen to actual desired pairs
    if len(chosen) < 26:
        raise ValueError("Could not select 26 non-overlapping candidates")
    centers = np.array([[candidates[i][0], candidates[i][1]] for i in chosen])
    radii = np.array([candidates[i][2] for i in chosen])
    return centers, radii

def max_radii_fixed_centres(centers, radii_initial):
    n = centers.shape[0]
    radii = radii_initial.copy()
    
    for i in range(n):
        min_dist = 1.0
        for j in range(n):
            if i != j:
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                min_dist = min(min_dist, dist)
        radii[i] = min(radii[i], min_dist / 2.0)
        # Adjust to stay in bounds
        max_r = min(centers[i][0], centers[i][1], 1 - centers[i][0], 1 - centers[i][1])
        radii[i] = min(radii[i], max_r)
    return radii

def nlp_refine(centers, radii_initial):
    n = centers.shape[0]
    # Vec: (x1,y1,...,xn,yn,r1,...,rn)
    x0 = np.zeros(2 * n + n)
    x0[:2*n:2] = centers[:,0]     # x_i
    x0[1:2*n:2] = centers[:,1]    # y_i
    x0[2*n:] = radii_initial      # r_i

    def objective(v):
        return -np.sum(v[2*n:])

    def jac(v):
        J = np.zeros_like(v)
        J[2*n:] = -1.0
        return J

    def constraints(v):
        x = v[:2*n:2]
        y = v[1:2*n:2]
        r = v[2*n:]
        cons = []
        # Wall constraints
        for i in range(n):
            cons.append(r[i] - x[i])
            cons.append(r[i] + x[i] - 1.0)
            cons.append(r[i] - y[i])
            cons.append(r[i] + y[i] - 1.0)
        # Non-overlap constraints
        for i in range(n):
            for j in range(i+1, n):
                dx = x[i] - x[j]
                dy = y[i] - y[j]
                dist = np.sqrt(dx*dx + dy*dy)
                cons.append(r[i] + r[j] - dist)
        return np.array(cons)

    def jac_constraints(v):
        x = v[:2*n:2]
        y = v[1:2*n:2]
        r = v[2*n:]
        m = 4 * n + n * (n - 1) // 2   # Total constraints
        J = np.zeros((m, 2*n + n))

        # Wall constraint Jacobians (i-th vertex, 4 constraints)
        for i in range(n):
            col_base = 2*i
            J[4*i][col_base] = -1.0       # r[i] - x[i]
            J[4*i+1][col_base] = 1.0      # r[i] + x[i] - 1
            J[4*i][col_base+1] = -1.0     # r[i] - y[i]
            J[4*i+1][col_base+1] = 1.0    # r[i] + y[i] - 1
            J[4*i][2*n+i] = 1.0           # r[i] - x[i]
            J[4*i+1][2*n+i] = 1.0         # r[i] + x[i] - 1
            J[4*i][2*n+i] = 1.0           # r[i] - y[i]
            J[4*i+1][2*n+i] = 1.0         # r[i] + y[i] - 1

        # Non-overlap constraint Jacobians
        con_idx = 4 * n
        for i in range(n):
            for j in range(i+1, n):
                dx = x[i] - x[j]
                dy = y[i] - y[j]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist > 1e-12:
                    ddx = dx / dist
                    ddy = dy / dist
                    J[con_idx][2*i] = -ddx
                    J[con_idx][2*j] = ddx
                    J[con_idx][2*i+1] = -ddy
                    J[con_idx][2*j+1] = ddy
                    J[con_idx][2*n+i] = 1.0
                    J[con_idx][2*n+j] = 1.0
                con_idx += 1
        return J

    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0))   # x
        bounds.append((0.0, 1.0))   # y
    for i in range(n):
        bounds.append((0.0, 1.0))   # r

    options = {
        'maxiter': 300,
        'xtol': 1e-9,
        'gtol': 1e-9,
    }

    try:
        res = minimize(objective, x0, jac=jac, bounds=bounds, method='trust-constr',
                       constraints={'fun': constraints, 'jac': jac_constraints},
                       options=options)
        v_final = res.x
        new_centers = np.column_stack((v_final[:2*n:2], v_final[1:2*n:2]))
        new_radii = v_final[2*n:]
        return new_centers, new_radii
    except Exception as e:
        print("Error in refine:", e)
        return centers, radii_initial
