import numpy as np
from scipy.optimize import differential_evolution
from scipy.spatial.distance import cdist

def compute_radii(centers):
    n = centers.shape[0]
    radii = np.zeros(n)
    
    # Wall distances
    x = centers[:, 0]
    y = centers[:, 1]
    wall_distances = np.array([
        x,                    # left
        1 - x,                # right
        y,                    # bottom
        1 - y                 # top
    ])
    
    # Neighbour distances
    distances = cdist(centers, centers)
    np.fill_diagonal(distances, np.inf)
    
    # Compute radii for each circle
    for i in range(n):
        wall_dist = np.min(wall_distances[:, i])
        neigh_dist = 0.5 * np.min(distances[i, :])
        radii[i] = min(wall_dist, neigh_dist)
    
    return radii

def neg_sum_radii(z):
    centers = z.reshape(-1, 2)
    radii = compute_radii(centers)
    return -np.sum(radii)

def run_packing():
    np.random.seed(42)
    
    # Differential Evolution
    bounds = [(0, 1)] * 52
    result = differential_evolution(
        neg_sum_radii,
        bounds,
        maxiter=2000,
        popsize=10,
        tol=1e-7,
        polish=False,
        strategy='best1bin',
        seed=42
    )
    
    # Extract best centers
    centers = result.x.reshape(-1, 2)
    
    # Local refinement with hill-climbing
    sigma = 0.02
    max_iter = 3000
    best_centers = centers.copy()
    best_sum = -neg_sum_radii(best_centers.flatten())
    
    for _ in range(max_iter):
        # Generate candidate
        candidate = best_centers + sigma * np.random.randn(*best_centers.shape)
        candidate = np.clip(candidate, 0, 1)
        
        # Evaluate
        candidate_sum = -neg_sum_radii(candidate.flatten())
        
        if candidate_sum > best_sum:
            best_centers = candidate
            best_sum = candidate_sum
            sigma = min(1.1 * sigma, 0.5)
        else:
            sigma = max(0.98 * sigma, 1e-5)
