import numpy as np
from scipy.optimize import linprog
from collections import defaultdict

def initial_hex_centers(k, spacing):
    """Generate k initial centers in a hexagonal lattice scaled to unit square."""
    sqrt3 = np.sqrt(3)
    rows = []
    i = 0
    j = 0
    while len(rows) < k:
        x = i * spacing + (j % 2) * spacing / 2
        y = j * spacing * sqrt3 / 2
        if 0 <= x <= 1 and 0 <= y <= 1:
            rows.append([x, y])
        if x > 1:
            j += 1
            i = 0
        else:
            i += 1
    return np.array(rows[:k])

def initial_radii(k, r0):
    """Return array of initial radii."""
    return np.full(k, r0)

def build_lp_constants(centers, radii):
    """Compute wall distances and pairwise distances for LP."""
    n = centers.shape[0]
    d_left = centers[:, 0]
    d_right = 1 - centers[:, 0]
    d_bottom = centers[:, 1]
    d_top = 1 - centers[:, 1]
    d_wall = np.stack([d_left, d_right, d_bottom, d_top], axis=1)
    
    # For each pair, compute the distance
    distances = np.zeros((n, n))
    for i in range(n):
        for j in range(i+1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            distances[i, j] = dist
            distances[j, i] = dist

    return {
        'd_wall': d_wall,
        'distances': distances
    }

def solve_radius_lp(consts, radii, graph, alpha=0.5):
    """Solve LP to maximize sum of radius increments."""
    n = len(radii)
    D = consts['distances']
    d_wall = consts['d_wall']
    
    # Build constraints matrix and vector
    A = []
    b = []
    
    # Wall capacity constraints
    for i in range(n):
        wall_caps = d_wall[i] - radii[i]
        wall_caps[wall_caps < 0] = 0
        bounds = np.min(wall_caps)
        A.append([-1.0] * n)
        b.append(-bounds)
    
    # Pairwise constraints
    for i in range(n):
        for j in range(i+1, n):
            if D[i, j] > radii[i] + radii[j]:
                # Constraint: Δr_i + Δr_j <= D[i,j] - (r_i + r_j)
                A.append([0.] * i + [1., 1.] + [0.] * (n - j - 1))
                b.append(D[i, j] - (radii[i] + radii[j]))
    
    # Optional graph constraints (for transfer limit)
    for i in range(n):
        if len(graph[i]) > 0:
            # Δr_i <= alpha * Σ_{j in N(i)} (D[i,j] - (r_i + r_j))
            limit = 0
            for j in graph[i]:
                if j != i and i < j:
                    limit += (D[i, j] - (radii[i] + radii[j]))
            limit *= alpha
            A.append([-1.] * n)
            A[-1][i] = 1.0
            b.append(limit)
    
    A = np.array(A)
    b = np.array(b)
    
    # Objective: maximize sum of Δr_i
    c = np.ones(n)
    
    # Solve LP
    try:
        res = linprog(-c, A_ub=A, b_ub=b, method='highs', 
                      options={'presolve': True, 'time_limit': 10.0})
    except:
        # If solver fails, default to no growth
        return np.zeros(n)
    
    if res.success and not np.isnan(res.x).any():
        return res.x
    else:
        # Fallback when infeasible
        return np.zeros(n)

def apply_increment(radii, Δr):
    """Apply increment to radii."""
    return radii + Δr

def push_feasibility(centers, radii):
    """Apply geometric pushes to restore feasibility."""
    n = centers.shape[0]
    new_centers = centers.copy()
    new_radii = radii.copy()
    edges = []

    # Wall push
    for i in range(n):
        x, y = new_centers[i]
        r = new_radii[i]
        if x - r < 0:
            new_radii[i] = x
        elif x + r > 1:
            new_radii[i] = 1 - x
        if y - r < 0:
            new_radii[i] = y
        elif y + r > 1:
            new_radii[i] = 1 - y
    
    # Pairwise overlap push
    for i in range(n):
        for j in range(i+1, n):
            dist = np.linalg.norm(new_centers[i] - new_centers[j])
            r_sum = new_radii[i] + new_radii[j]
            if dist < r_sum:
                overlap = r_sum - dist
                dx = new_centers[i] - new_centers[j]
                if np.linalg.norm(dx) > 0:
                    dx = dx / np.linalg.norm(dx)
                    move = (overlap / 2) * dx
                    new_centers[i] -= move
                    new_centers[j] += move
                    edges.append((i, j))
                    
    # Fix potential wall violations after pushes
    for i in range(n):
        x, y = new_centers[i]
        r = new_radii[i]
        new_radii[i] = min(new_radii[i], x, 1-x, y, 1-y)
        
    return new_centers, new_radii, edges

def update_graph(graph, new_edges, wall contacts):
    """Update the contact graph with new edges."""
    for e in new_edges:
        graph[e[0]].add(e[1])
        graph[e[1]].add(e[0])

def run_iterations(centers, radii, max_iter, ε, α):
    """Run the main iterative optimization."""
    graph = defaultdict(set)
    for it in range(max_iter):
        # Build LP constants
        consts = build_lp_constants(centers, radii)
        
        # Solve radii LP
        Δr = solve_radius_lp(consts, radii, graph, α)
        
        # Apply increment
        radii = apply_increment(radii, Δr)
        
        # Check convergence
        if np.sum(Δr) < ε:
            break
            
        # Push feasibility
        centers, radii, edges = push_feasibility(centers, radii)
        
        # Update graph
        update_graph(graph, edges, [])
        
    return centers, radii

def final_refinement(centers, radii):
    """Final refinement with tight LP."""
    consts = build_lp_constants(centers, radii)
    Δr = solve_radius_lp(consts, radii, defaultdict(set), 0.5)
    radii = apply_increment(radii, Δr)
    
    # Clip to ensure walls
    for i in range(len(radii)):
        r = radii[i]
        x, y = centers[i]
        radii[i] = min(r, x, 1-x, y, 1-y)
        
    # Additional small clipping for overlaps
    eps = 1e-13
    n = len(radii)
    for i in range(n):
        for j in range(i+1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            r_sum = radii[i] + radii[j]
            if dist < r_sum:
                delta = r_sum - dist + eps
                if radii[i] >= radii[j]:
                    radii[i] -= delta / 2
                else:
                    radii[j] -= delta / 2
    
    return centers, radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial setup
    n_circles = 26
    spacing = 0.12
    r0 = 0.005
    
    # Step 1: Generate initial hexagonal center positions
    centers = initial_hex_centers(n_circles, spacing)
    radii = initial_radii(n_circles, r0)
    
    # Step 2-3: Run iterations
    centers, radii = run_iterations(centers, radii, max_iter=150, ε=1e-5, α=0.5)
    
    # Step 4: Final refinement
    centers, radii = final_refinement(centers, radii)
    
    # Compute sum of radii
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
