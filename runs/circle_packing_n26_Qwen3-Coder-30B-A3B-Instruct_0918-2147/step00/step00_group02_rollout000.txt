import numpy as np
from scipy.optimize import minimize

def initial_lattice(n=26):
    # Generate hexagonal lattice points
    spacing = 0.2
    sqrt3 = np.sqrt(3)
    rows = int(np.ceil(np.sqrt(n * 2 / sqrt3)))
    cols = int(np.ceil(n / rows))
    
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = j * spacing
            y = i * spacing * sqrt3 / 2
            if i % 2 == 1:
                x += spacing / 2
            if x <= 1 and y <= 1:
                centers.append([x, y])
            if len(centers) >= n:
                break
        if len(centers) >= n:
            break
    
    centers = np.array(centers[:n])
    
    # Compute admissible radii
    radii = []
    for i in range(n):
        x, y = centers[i]
        min_dist = min(x, 1 - x, y, 1 - y)
        for j in range(n):
            if i != j:
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                min_dist = min(min_dist, dist / 2)
        radii.append(min(0.5, min_dist))
    
    return centers, np.array(radii)

def pack(centers, radii):
    z = np.empty(3 * len(centers))
    z[0::3] = centers[:, 0]
    z[1::3] = centers[:, 1]
    z[2::3] = radii
    return z

def unpack(z):
    n = len(z) // 3
    centers = np.empty((n, 2))
    centers[:, 0] = z[0::3]
    centers[:, 1] = z[1::3]
    radii = z[2::3]
    return centers, radii

def objective(z):
    centers, radii = unpack(z)
    return -np.sum(radii)

def constraint_boundary(z):
    centers, radii = unpack(z)
    constraints = []
    for i in range(len(centers)):
        x, y = centers[i]
        r = radii[i]
        # left
        constraints.append(x - r)
        # right
        constraints.append(1 - x - r)
        # bottom
        constraints.append(y - r)
        # top
        constraints.append(1 - y - r)
    return np.array(constraints)

def constraint_nonoverlap(z):
    centers, radii = unpack(z)
    constraints = []
    n = len(centers)
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.hypot(dx, dy) + 1e-9
            r_sum = radii[i] + radii[j]
            constraints.append(r_sum - dist)
    return np.array(constraints)

def repair(centers, radii):
    # Boundary repair
    for i in range(len(centers)):
        x, y = centers[i]
        r = radii[i]
        r_new = min(r, x, 1 - x, y, 1 - y)
        if r_new < r:
            radii[i] = r_new

    # Overlap repair
    for _ in range(10):
        any_repair = False
        n = len(centers)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.hypot(dx, dy)
                r_sum = radii[i] + radii[j]
                if dist < r_sum:
                    overlap = r_sum - dist
                    delta = overlap / 2
                    radii[i] = max(0, radii[i] - delta)
                    radii[j] = max(0, radii[j] - delta)
                    any_repair = True
        if not any_repair:
            break

    return centers, radii

def run_packing():
    # Generate initial layout
    centers, radii = initial_lattice(26)
    z0 = pack(centers, radii)
    
    # Build constraints
    constraints = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_nonoverlap}
    ]
    
    # Bounds
    bounds = []
    for i in range(26):
        bounds.append((0, 1))
        bounds.append((0, 1))
        bounds.append((0, 0.5))
    
    # Optimization
    result = minimize(
        objective,
        z0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False}
    )
    
    # Unpack and repair
    centers, radii = unpack(result.x)
    centers, radii = repair(centers, radii)

    return centers, radii, sum_radii
