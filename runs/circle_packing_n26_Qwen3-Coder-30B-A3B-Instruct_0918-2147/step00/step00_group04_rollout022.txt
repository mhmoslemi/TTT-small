import numpy as np
from scipy.optimize import minimize

NUM_CIRCLES = 26
NUM_RESTARTS = 5
BOUNDARY_SLACK = 1e-9
NONOVERLAP_SLACK = 1e-9

def hex_lattice_points(n_rows, spacing):
    """Generate hexagonal lattice points inside unit square"""
    points = []
    for i in range(n_rows):
        y = i * spacing * np.sqrt(3) / 2
        if y > 1:
            break
        offset = (i % 2) * spacing / 2
        n_cols = int((1 - offset) / spacing) + 1
        for j in range(n_cols):
            x = offset + j * spacing
            if x > 1:
                break
            points.append([x, y])
    return np.array(points)

def initial_guess(seed):
    np.random.seed(seed)
    # Generate hexagonal lattice covering roughly 26 circles
    spacing = 0.2
    points = hex_lattice_points(10, spacing)
    
    # Keep only first 26 points (or as many as available)
    if points.shape[0] < NUM_CIRCLES:
        points = np.pad(points, ((0, NUM_CIRCLES - points.shape[0]), (0, 0)), mode='constant')
    
    points = points[:NUM_CIRCLES]
    
    # Add jitter
    jitter = np.random.uniform(-0.02, 0.02, (NUM_CIRCLES, 2))
    points += jitter
    
    # Ensure they are within bounds
    points = np.clip(points, 0, 1)
    
    # Initialize radii
    radii = np.full(NUM_CIRCLES, 0.01)
    
    # Flatten to variable vector format (x0, y0, r0, x1, y1, r1, ...)
    vars0 = np.empty(NUM_CIRCLES * 3)
    vars0[0::3] = points[:, 0]   # x coordinates
    vars0[1::3] = points[:, 1]   # y coordinates
    vars0[2::3] = radii          # radii
    
    return vars0

def objective(vars):
    # Minimize negative sum of radii
    radii = vars[2::3]
    return -np.sum(radii)

def constraint_boundary(vars):
    # For each circle i, return [x_i-r_i, 1-x_i-r_i, y_i-r_i, 1-y_i-r_i]
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Boundary constraints
    constraints = np.empty(4 * NUM_CIRCLES)
    constraints[0::4] = x - r + BOUNDARY_SLACK  # x - r >= 0
    constraints[1::4] = 1 - x - r + BOUNDARY_SLACK  # 1 - x - r >= 0
    constraints[2::4] = y - r + BOUNDARY_SLACK  # y - r >= 0
    constraints[3::4] = 1 - y - r + BOUNDARY_SLACK  # 1 - y - r >= 0
    
    return constraints

def constraint_nonoverlap(vars):
    # Return pairwise distance constraints
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Generate all pairs of indices
    indices = []
    for i in range(NUM_CIRCLES):
        for j in range(i + 1, NUM_CIRCLES):
            indices.append((i, j))
    
    # Compute pairwise distances and constraints
    constraints = []
    for i, j in indices:
        dx = x[i] - x[j]
        dy = y[i] - y[j]
        dist = np.sqrt(dx*dx + dy*dy)
        # Distance minus sum of radii
        constraints.append(dist - (r[i] + r[j]) + NONOVERLAP_SLACK)
    
    return np.array(constraints)

def postprocess(vars):
    # Make a copy of variables
    vars = vars.copy()
    
    # Adjust boundary constraint violations
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    for i in range(NUM_CIRCLES):
        if x[i] - r[i] < 0:
            x[i] = r[i]
        if 1 - x[i] - r[i] < 0:
            x[i] = 1 - r[i]
        if y[i] - r[i] < 0:
            y[i] = r[i]
        if 1 - y[i] - r[i] < 0:
            y[i] = 1 - r[i]
    
    # Address small overlaps
    for i in range(NUM_CIRCLES):
        for j in range(i + 1, NUM_CIRCLES):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dist = np.sqrt(dx*dx + dy*dy)
            overlap = dist - (r[i] + r[j])
            if overlap < -1e-12:  # Overlap exists
                # Reduce larger radius
                total = r[i] + r[j]
                r[i] = r[i] - overlap/2
                r[j] = r[j] - overlap/2
                # Adjust if negative
                if r[i] < 0:
                    r[j] += r[i]
                    r[i] = 0
                if r[j] < 0:
                    r[i] += r[j]
                    r[j] = 0
                    
    return vars

def run_packing():
    best_sum = -np.inf
    best_vars = None
    
    # Build constraint list for scipy
    constraints = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_nonoverlap}
    ]
    
    # Global bounds: x in [0,1], y in [0,1], r in [0,0.5]
    bounds = [(0, 1), (0, 1), (0, 0.5)] * NUM_CIRCLES
    
    for seed in range(NUM_RESTARTS):
        vars0 = initial_guess(seed)
        res = minimize(objective, vars0,
                       method='SLSQP',
                       bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': 3000, 'ftol': 1e-7})
        if not res.success:
            continue

        vars_corr = postprocess(res.x)
        s = np.sum(vars_corr[2::3])
        if s > best_sum:
            best_sum = s
            best_vars = vars_corr
    
    # Reshape best_vars
    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii = best_vars[2::3]
    
    return centers, radii, float(best_sum)
