import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial.distance import cdist

def build_lp_model(n_circles):
    # Build the LP model for packing n_circles in unit square
    # Variables: [x1, y1, r1, x2, y2, r2, ..., xn, yn, rn]
    n_vars = 3 * n_circles
    c = np.zeros(n_vars)
    c[2::3] = -1.0  # We minimize -sum(r_i) which maximizes sum(r_i)

    # Inequality constraints: A_ub @ x <= b_ub
    A_ub = []
    b_ub = []
    
    # Wall constraints for each circle
    for i in range(n_circles):
        xi_idx = 3 * i
        yi_idx = 3 * i + 1
        ri_idx = 3 * i + 2
        
        # r_i <= x_i => -x_i + r_i <= 0
        row = np.zeros(n_vars)
        row[xi_idx] = -1
        row[ri_idx] = 1
        A_ub.append(row)
        b_ub.append(0)
        
        # r_i <= 1-x_i => x_i + r_i <= 1
        row = np.zeros(n_vars)
        row[xi_idx] = 1
        row[ri_idx] = 1
        A_ub.append(row)
        b_ub.append(1)
        
        # r_i <= y_i => -y_i + r_i <= 0
        row = np.zeros(n_vars)
        row[yi_idx] = -1
        row[ri_idx] = 1
        A_ub.append(row)
        b_ub.append(0)
        
        # r_i <= 1-y_i => y_i + r_i <= 1
        row = np.zeros(n_vars)
        row[yi_idx] = 1
        row[ri_idx] = 1
        A_ub.append(row)
        b_ub.append(1)

    # Axis-wise separation constraints
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            xi_idx = 3 * i
            yi_idx = 3 * i + 1
            ri_idx = 3 * i + 2
            xj_idx = 3 * j
            yj_idx = 3 * j + 1
            rj_idx = 3 * j + 2
            
            # x_i + r_i <= x_j - r_j => -x_i + x_j - r_i - r_j >= 0
            row = np.zeros(n_vars)
            row[xi_idx] = -1
            row[xj_idx] = 1
            row[ri_idx] = -1
            row[rj_idx] = -1
            A_ub.append(row)
            b_ub.append(0)
            
            # x_j + r_j <= x_i - r_i => x_j - x_i - r_i - r_j >= 0
            row = np.zeros(n_vars)
            row[xj_idx] = 1
            row[xi_idx] = -1
            row[ri_idx] = -1
            row[rj_idx] = -1
            A_ub.append(row)
            b_ub.append(0)
            
            # y_i + r_i <= y_j - r_j => -y_i + y_j - r_i - r_j >= 0
            row = np.zeros(n_vars)
            row[yi_idx] = -1
            row[yj_idx] = 1
            row[ri_idx] = -1
            row[rj_idx] = -1
            A_ub.append(row)
            b_ub.append(0)
            
            # y_j + r_j <= y_i - r_i => y_j - y_i - r_i - r_j >= 0
            row = np.zeros(n_vars)
            row[yj_idx] = 1
            row[yi_idx] = -1
            row[ri_idx] = -1
            row[rj_idx] = -1
            A_ub.append(row)
            b_ub.append(0)

    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    return A_ub, b_ub, c

def solve_lp(A_ub, b_ub, c):
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    if not result.success:
        raise RuntimeError("LP solver failed")

    x = result.x
    centers = x[::3].reshape(-1, 1), x[1::3].reshape(-1, 1)
    radii = x[2::3]
    return np.array(centers).T, radii

def max_radius_projection(centers, radii):
    # Return updated radii to match actual maximum possible.
    n = centers.shape[0]
    new_radii = np.copy(radii)
    for _ in range(5):  # Converge in few steps
        updated = False
        for i in range(n):
            r_new = min(
                centers[i, 0],
                1 - centers[i, 0],
                centers[i, 1],
                1 - centers[i, 1]
            )
            for j in range(n):
                if i != j:
                    d = np.linalg.norm(centers[i] - centers[j])
                    r_new = min(r_new, d - radii[j])
            if r_new > new_radii[i] + 1e-10:
                new_radii[i] = r_new
                updated = True
        if not updated:
            break
    return new_radii
