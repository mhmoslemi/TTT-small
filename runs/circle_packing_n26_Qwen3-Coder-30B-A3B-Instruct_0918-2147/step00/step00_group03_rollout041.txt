import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Set seed for reproducibility
    np.random.seed(0)
    
    def init_grid(n_circles=26, spacing=0.07):
        """Generate initial configuration approximately in hexagonal grid."""
        points = []
        k = 0
        y = spacing / 2
        while len(points) < n_circles and k < 100:
            if k % 2 == 0:
                x_start = spacing / 2
                offset = 0
            else:
                x_start = spacing
                offset = spacing / 2
            x = x_start
            while len(points) < n_circles and x <= 1 + spacing:
                if x >= spacing and x <= 1 - spacing and y >= spacing and y <= 1 - spacing:
                    points.append([x, y])
                x += spacing
            y += spacing * np.sqrt(3) / 2
            k += 1
        points = np.array(points[:n_circles])
        # Add jitter to break symmetry
        points += (np.random.rand(n_circles, 2) - 0.5) * 0.002
        return points
    
    def compute_forces(centers, radii, k_wall=10.0, k_pair=5.0):
        n = len(centers)
        forces = np.zeros_like(centers)
        
        # Compute wall distances
        d_left = centers[:, 0] - radii
        d_right = 1 - centers[:, 0] - radii
        d_bottom = centers[:, 1] - radii
        d_top = 1 - centers[:, 1] - radii
        
        # Wall repulsion
        for i in range(n):
            # Left wall
            if d_left[i] < 0:
                forces[i, 0] -= k_wall * (-d_left[i])
            # Right wall
            if d_right[i] < 0:
                forces[i, 0] += k_wall * (-d_right[i])
            # Bottom wall
            if d_bottom[i] < 0:
                forces[i, 1] -= k_wall * (-d_bottom[i])
            # Top wall
            if d_top[i] < 0:
                forces[i, 1] += k_wall * (-d_top[i])
        
        # Pairwise repulsion
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.linalg.norm(diff)
                gap = dist - (radii[i] + radii[j])
                if gap < 0:
                    # Repulsion vector
                    force_magnitude = k_pair * (-gap)
                    force_unit = diff / (dist + 1e-12)
                    force = force_magnitude * force_unit
                    forces[i] += force
                    forces[j] -= force
        return forces
    
    def grow_radii(centers, radii, forces, alpha=1e-3):
        """Update radii based on whether forces are active."""
        n = len(centers)
        dot_radii = np.full(n, alpha, dtype=np.float64)
        for i in range(n):
            # Check if touching any wall or another circle
            d_left = centers[i, 0] - radii[i]
            d_right = 1 - centers[i, 0] - radii[i]
            d_bottom = centers[i, 1] - radii[i]
            d_top = 1 - centers[i, 1] - radii[i]
            touch_wall = (d_left < 0) or (d_right < 0) or (d_bottom < 0) or (d_top < 0)
            
            # For pairwise, we check if forces are active
            # Small tolerance just to ensure not right at the edge of overlap
            touch_pair = False
            for j in range(n):
                if i != j:
                    diff = centers[i] - centers[j]
                    dist = np.linalg.norm(diff)
                    gap = dist - (radii[i] + radii[j])
                    if gap < 1e-6 and abs(dist) > 1e-12:
                        touch_pair = True
                        break
            
            if touch_wall or touch_pair:
                dot_radii[i] = 0
        return dot_radii
    
    # Initialize
    centers = init_grid(26)
    radii = np.full(26, 0.001)
    
    # Simulation loop
    dt = 0.02
    damping = 1.0
    velocity = np.zeros_like(centers)
    
    max_steps = 5000
    has_grown = True  # Flag to trigger convergence check
    
    for step in range(max_steps):
        forces = compute_forces(centers, radii)
        dot_radii = grow_radii(centers, radii, forces)
        
        # Update velocities and positions
        velocity += dt * (forces - damping * velocity)
        centers += dt * velocity
        
        # Apply constraints to positions
        centers[:, 0] = np.clip(centers[:, 0], 0, 1)
        centers[:, 1] = np.clip(centers[:, 1], 0, 1)
        
        # Update radii
        radii += dt * dot_radii
        
        # Clip negative radii
        radii = np.maximum(radii, 0)
        
        # Convergence check
        if not np.any(dot_radii > 0) and np.max(np.abs(forces)) < 1e-6:
            # Check convergence consistency by iterating for a few more steps
            if not has_grown:
                break
            has_grown = False
        
        if np.any(dot_radii > 0):
            has_grown = True
    
    # Compute exact maximal radii for selected configuration
    def max_radii_fixed_centers(centers_, radii_):
        n = len(centers_)
        new_radii = radii_.copy()
        # vectorized maximums
        for _ in range(50):
            converged = True
            for i in range(n):
                r_max = np.inf
                # Wall constraints
                r_max = min(r_max, centers_[i, 0])
                r_max = min(r_max, 1 - centers_[i, 0])
                r_max = min(r_max, centers_[i, 1])
                r_max = min(r_max, 1 - centers_[i, 1])
                # Collision constraints
                for j in range(n):
                    if i != j:
                        dist = np.sqrt(np.sum((centers_[i] - centers_[j]) ** 2))
                        r_max = min(r_max, dist - radii_[j])
                if r_max < new_radii[i]:
                    new_radii[i] = r_max
                    converged = False
            if converged:
                break
        # Clamp radii
        new_radii = np.maximum(new_radii, 0)
        return new_radii
    
    radii = max_radii_fixed_centers(centers, radii)
    
    # Set up function for maximization
    def objective_sum_radii(centers_flat):
        # Reconstruct centers from flat array
        centers_flat = np.asarray(centers_flat)
        n_centers = len(centers_flat) // 2
        centers_ = centers_flat.reshape((n_centers, 2))
        radii_ = max_radii_fixed_centers(centers_, radii)
        return -np.sum(radii_)
    
    def constraint_func(centers_flat):
        centers_ = centers_flat.reshape((len(centers), 2))
        return [c[0] - 0.0, 1.0 - c[0], c[1] - 0.0, 1.0 - c[1] for c in centers_]
    
    # Flatten centers matrix to single dimension
    initial_centers_flat = centers.flatten()
    
    # L-BFGS-B Optimization # Seed the optimization with centers (unchanged radii)
    opt_result = minimize(objective_sum_radii,
                          initial_centers_flat,
                          method='L-BFGS-B',
                          options={'maxiter': 2000},
                          bounds=[(0.0, 1.0)] * len(initial_centers_flat))
    
    # Reconstruct centers from best solution
    best_centers_flat = opt_result.x
    best_centers = best_centers_flat.reshape((26, 2))
    
    # Compute final radii based on optimal centers
    final_radii = max_radii_fixed_centers(best_centers, radii)
    
    # Final check for feasibility
    validation_result, msg = validate_packing(best_centers, final_radii)
    if not validation_result:
        raise RuntimeError("Validation error after optimization: " + msg)
    
    # Return as required
    final_sum_radii = np.sum(final_radii)
    return best_centers, final_radii, final_sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"
    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"
    return True, "ok"
