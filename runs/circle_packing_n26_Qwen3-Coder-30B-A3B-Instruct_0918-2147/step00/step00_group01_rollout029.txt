import numpy as np
from scipy.optimize import minimize

def initial_layout():
    # Hexagonal grid layout for 25 circles, then add one in center
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Grid parameters
    spacing = 0.18
    radius = spacing / 2.0 - 0.01
    
    # Place 5x5 hexagonal grid
    rows = 5
    cols = 5
    x_offset = 0.0
    y_offset = 0.0
    
    count = 0
    for i in range(rows):
        for j in range(cols):
            x = x_offset + j * spacing
            y = y_offset + i * spacing * np.sqrt(3) / 2
            # Adjust for hexagonal pattern
            if i % 2 == 1:
                x += spacing / 2
            x = max(radius, min(1.0 - radius, x))
            y = max(radius, min(1.0 - radius, y))
            centers[count] = [x, y]
            radii[count] = radius
            count += 1
    
    # Place 26th circle in center of square
    centers[count] = [0.5, 0.5]
    radii[count] = 0.05
    
    return centers, radii

def pack_vector(centers, radii):
    # Flatten into a single vector: [x0, y0, r0, x1, y1, r1, ...]
    return np.hstack([centers.flatten(), radii])

def unpack_vector(z):
    # Unflatten vector into centers and radii
    n = len(z) // 3
    centers = z[:2*n].reshape(n, 2)
    radii = z[2*n:]
    return centers, radii

def objective(z):
    # Maximize sum of radii => minimize -sum(radii)
    _, radii = unpack_vector(z)
    return -np.sum(radii)

def constraint_boundary(z):
    # Each circle within [0,1]^2 with radius r
    centers, radii = unpack_vector(z)
    n = len(radii)
    cons = np.zeros(4 * n)
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        cons[4*i] = x - r  # x - r >= 0
        cons[4*i+1] = 1 - x - r  # 1 - x - r >= 0
        cons[4*i+2] = y - r  # y - r >= 0
        cons[4*i+3] = 1 - y - r  # 1 - y - r >= 0
    return cons

def constraint_nonoverlap(z):
    # Pairwise non-overlap constraints: (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
    centers, radii = unpack_vector(z)
    n = len(radii)
    # Generate all pairs
    cons = []
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx * dx + dy * dy
            r_sum = radii[i] + radii[j]
            cons.append(dist_sq - r_sum * r_sum)
    return np.array(cons)

def run_packing():
    # Initial layout
    centers, radii = initial_layout()
    z0 = pack_vector(centers, radii)
    n = len(centers)
    
    # Set up bounds and constraints
    bounds = [(0, 1) for _ in range(2*n)] + [(0, 0.5) for _ in range(n)]
    
    constraints = []
    # Boundary constraints (4 per circle)
    constraints.append({'type': 'ineq', 'fun': constraint_boundary})
    # Non-overlap constraints
    constraints.append({'type': 'ineq', 'fun': constraint_nonoverlap})
    
    # Solve with multiple restarts
    best_sum = -np.inf
    best_z = None
    
    for restart in range(3):
        # Add small random jitter
        z = z0.copy()
        np.random.seed(restart)
        z[:2*n] += np.random.uniform(-0.01, 0.01, 2*n)
        z[2*n:] += np.random.uniform(-0.005, 0.005, n)
        
        res = minimize(objective, z, method='SLSQP', constraints=constraints,
                       bounds=bounds, options={'ftol': 1e-9, 'maxiter': 2000})

            sum_radii = np.sum(radii)
            if sum_radii > best_sum:
                best_sum = sum_radii
                best_z = final_z.copy()
        else:
            # Use initial layout if optimization fails
            if best_sum <= -np.inf:
                best_sum = np.sum(radii)
                best_z = z0.copy()
    
    # Final unpacking
    best_centers, best_radii = unpack_vector(best_z)
