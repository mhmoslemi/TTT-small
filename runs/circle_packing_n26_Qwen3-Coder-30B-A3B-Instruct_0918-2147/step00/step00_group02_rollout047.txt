import numpy as np
from scipy.stats.qmc import Sobol

def initialize_seeds(n):
    sobol = Sobol(d=2, seed=42)
    points = sobol.random(n)
    # Clip to interior with margin
    margin = 0.02
    points = margin + (1 - 2 * margin) * points
    return points

def compute_feasible_radii(centers):
    n = centers.shape[0]
    # Compute pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    # Set diagonal to large value to avoid self-distance
    np.fill_diagonal(dists, np.inf)
    min_nns = np.min(dists, axis=1)
    # Compute wall distances
    wall_dists = np.array([
        centers[:, 0],  # x distances to left and right walls
        1 - centers[:, 0],
        centers[:, 1],  # y distances to bottom and top walls
        1 - centers[:, 1]
    ])
    wall_min = np.min(wall_dists, axis=0)
    # feasible radii
    radii = np.minimum(0.5 * min_nns, wall_min)
    return radii

def compute_slack(radii, centers):
    n = centers.shape[0]
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    min_nns = np.min(dists, axis=1)
    slack = 0.5 * min_nns - radii
    return slack

def redistribute_radii(radii, slack, alpha):
    total_slack = np.sum(slack)
    if total_slack <= 0:
        return radii
    # Increase radii of wall-limited circles
    new_radii = radii.copy()
    increase = alpha * slack
    new_radii += increase
    # Subtract from neighbour-limited circles
    # Use proportional sharing based on current radii
    mask = slack > 1e-15  # wall-limited
    if np.sum(mask) == 0:
        return new_radii
    reduction = np.sum(increase[mask])
    if reduction <= 1e-15:
        return new_radii

    # Proportional reduction
    other_radii = radii[~mask]
    other_radii_sum = np.sum(other_radii)
    if other_radii_sum > 1e-15:
        frac = other_radii / other_radii_sum
        reduction_amount = reduction * frac
        new_radii[~mask] = np.maximum(0, new_radii[~mask] - reduction_amount)
    return new_radii

def push_away(centers, radii, beta, eps):
    n = centers.shape[0]
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    new_centers = centers.copy()
    for i in range(n):
        for j in range(i):
            if dists[i, j] < radii[i] + radii[j] + eps:
                dv = diff[i, j]
                length = np.sqrt(np.sum(dv**2))
                if length < eps:
                    continue
                push_amount = (radii[i] + radii[j] + eps - length) * beta
                dv_norm = dv / length
                new_centers[i] += push_amount * dv_norm
                new_centers[j] -= push_amount * dv_norm
    # Clip to interior
    margin = eps
    new_centers = np.clip(new_centers, margin, 1 - margin)
    return new_centers

def iterate(centers, radii, alpha, beta, max_iter):
    for iteration in range(max_iter):
        # Compute new radii
        old_sum = np.sum(radii)
        # Compute slack
        slack = compute_slack(radii, centers)
        # Redistribute
        radii = redistribute_radii(radii, slack, alpha)
        # Push away
        centers = push_away(centers, radii, beta, 1e-8)
        # Update radii
        radii = compute_feasible_radii(centers)
        new_sum = np.sum(radii)
        if new_sum - old_sum < 1e-5:
            break
    return centers, radii

def repair(centers, radii, eps_final):
    # Shrink radii
    radii = np.maximum(0, radii - eps_final)
    # Clamp centers
    margin = eps_final
    centers = np.clip(centers, margin, 1 - margin)
    return centers, radii

def run_packing():
    # Initial seeding
    centers = initialize_seeds(26)
    radii = compute_feasible_radii(centers)
    
    # Iterative refinement
    centers, radii = iterate(centers, radii, alpha=0.4, beta=0.3, max_iter=200)
    
    # Final safety
    eps_final = 5e-10
    centers, radii = repair(centers, radii, eps_final)
