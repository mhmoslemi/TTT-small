import numpy as np
import cvxpy as cp
from scipy.spatial.distance import cdist

def build_socp_problem(n):
    r = cp.Variable(n, nonneg=True)
    dist_params = []
    constraints = []

    for i in range(n):
        constraints += [r[i] <= cp.minimum(cp.minimum(
            cp.maximum(0, 1 - cp.Variable(1, nonneg=True) - 0.5), 
            cp.maximum(0, 0.5 - cp.Variable(1, nonneg=True) - 0.5)
        ) + 0.5 - 0.5)]
    # Simplified to use boundary constraints directly
    # We'll handle boundary using ateratively updated constraints

    return r, constraints

def solve_radii(centres, prob, r, dist_params, dist_ij):
    # Update the distances in parameters
    for i in range(len(dist_params)):
        dist_params[i].value = dist_ij[i]
    # Solve with high penalty for strict termination
    prob.solve(solver=cp.ECOS, abstol=1e-12, reltol=1e-12)
    radii = r.value
    return radii 

def energy(centres, prob, r, dist_params, n):
    # Compute all pairwise distances
    dist_ij = cdist(centres, centres)
    dist_ij[np.eye(n, dtype=bool)] = np.inf  # All diagonal gaps
    # Flatten pairs to update parameters
    flat_distances = []
    for i in range(n):
        flat_distances.extend(dist_ij[i, i+1:])
    # Update all distances in dist_params
    # For each i<j: dist_params[i*n + j - (i+1)*(i+2)//2] = dist_ij[i][j]
    # Simplified: Assign each pairwise constraint to a unique parameter
    # This line below assumes parameters are pre-created in order: (0,1),(0,2),...
    # Update each parameter
    for i in range(n):
        for j in range(i+1, n):
            idx = i * n + j - (i+1) * (i+2) // 2
            dist_params[idx].value = dist_ij[i][j]
    # Solve SOCP
    prob.solve(solver=cp.ECOS, abstol=1e-12, reltol=1e-12)
    if prob.status in ["optimal", "optimal_inaccurate"]:
        return -np.sum(r.value)
    else:
        return np.inf

def random_move(state, rng, p=0.1):
    new_state = state.copy()
    i = rng.integers(0, len(state)//2)
    # Choose move type
    move_type = rng.choice(["jitter_single", "jitter_cluster", "edge_push", "swap"])
    if move_type == "jitter_single":
        new_state[2*i] += rng.normal(0, 0.02)
        new_state[2*i+1] += rng.normal(0, 0.02)
        new_state[2*i] = np.clip(new_state[2*i], 0, 1)
        new_state[2*i+1] = np.clip(new_state[2*i+1], 0, 1)
    elif move_type == "jitter_cluster":
        # Create cluster of 3-5 nearby centers
        indices = rng.choice(len(state)//2, rng.integers(3, 6), replace=False)
        offset = rng.normal(0, 0.02, 2)
        for idx in indices:
            new_state[2*idx] = np.clip(new_state[2*idx] + offset[0], 0, 1)
            new_state[2*idx+1] = np.clip(new_state[2*idx+1] + offset[1], 0, 1)
    elif move_type == "edge_push":
        if rng.random() < p:
            x, y = new_state[2*i], new_state[2*i+1]
            dx = -np.sign(x - 0.5) * 0.01
            dy = -np.sign(y - 0.5) * 0.01
            new_state[2*i] += dx
            new_state[2*i+1] += dy
            new_state[2*i] = np.clip(new_state[2*i], 0, 1)
            new_state[2*i+1] = np.clip(new_state[2*i+1], 0, 1)
    return new_state

def anneal_chain(seed, n=26):
    rng = np.random.default_rng(seed)
    # Initialize centers randomly
    state = rng.random(n * 2)
    state = np.clip(state, 0, 1)
    current_energy = energy(state.reshape(-1, 2), prob, r, dist_params, n)
    best_state = state.copy()
    best_energy = current_energy
    temp = 0.05
    alpha = 0.995
    count = 0
    max_iter = 150000
    while temp > 1e-6 and count < max_iter:
        new_state = random_move(state, rng)
        new_energy = energy(new_state.reshape(-1, 2), prob, r, dist_params, n)
        delta = new_energy - current_energy
        if delta < 0 or np.exp(-delta / temp) > rng.random():
            state = new_state
            current_energy = new_energy
            if current_energy < best_energy:
                best_state = state.copy()
                best_energy = current_energy
        temp *= alpha
        count += 1
    return best_state.reshape(-1, 2)

def local_refine(centres, prob, r, dist_params, n):
    # Simple smoothing using mirrored constraints
    # Resize critically touching circles
    prev_centres = centres.copy()
    centres_iter = centres.copy()
    max_iter = 10
    for _ in range(max_iter):
        # Recompute based on current configuration
        dist_ij = cdist(centres_iter, centres_iter)
        flat_distances = []
        idx = 0
        for i in range(n):
            for j in range(i+1, n):
                if idx < len(dist_params):
                    dist_params[idx].value = dist_ij[i][j]
                    idx += 1
        prob.solve(solver=cp.ECOS, abstol=1e-12, reltol=1e-12)
        radii = r.value
        new_centers = centres_iter.copy()
        for i in range(n):
            x_i, y_i = centres_iter[i]
            # Adjust position towards boundary or contact limits
            r_i = radii[i]
            if x_i - r_i < 1e-12:
                new_centers[i, 0] += r_i - x_i
            elif 1 - x_i - r_i < 1e-12:
                new_centers[i, 0] -= 1 - x_i - r_i
            if y_i - r_i < 1e-12:
                new_centers[i, 1] += r_i - y_i
            elif 1 - y_i - r_i < 1e-12:
                new_centers[i, 1] -= 1 - y_i - r_i
        centres_iter = new_centers
        if np.max(np.abs(prev_centres - centres_iter)) < 1e-5:
            break
    return centres_iter

def finalise(centres, prob, r, dist_params, n):
    # Extra verification and strict correction
    # Solve with very high precision
    dist_ij = cdist(centres, centres)
    flat_distances = []
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            dist_params[idx].value = dist_ij[i][j]
            idx += 1
    prob.solve(solver=cp.ECOS, abstol=1e-12, reltol=1e-12)
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        raise RuntimeError("Inner SOCP failed to converge in finalise")
    radii = r.value
    # Check for exact boundary violations and adjust
    for i in range(n):
        x_i, y_i = centres[i]
        r_i = radii[i]
        r_min = min(x_i, 1-x_i, y_i, 1-y_i)
        if r_i > r_min + 1e-13:
            radii[i] = r_min
    return centres, radii

def run_packing():
    # Parameters
    n = 26
    # Build problem structure once
    r = cp.Variable(n, nonneg=True)
    # Create parameters for all constraints
    # Pairwise distances: n*(n-1)//2 parameters
    n_params = n * (n - 1) // 2
    dist_params = [cp.Parameter(1, nonneg=True) for _ in range(n_params)]
    constraints = []
    # Add boundary constraints for each circle
    # One global control, use internally via constraints
    # Add pairwise constraints: for each i<j, r[i]+r[j] <= dist_ij
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            # Safe constraint placement using Parameter
            constraints.extend([r[i] + r[j] <= dist_params[idx]])
            idx += 1
    # Add boundary clipping constraints for convenience
    constraints.append(r >= 0)
    # Finalize problem
    prob = cp.Problem(cp.Maximize(cp.sum(r)), constraints)
    # Initialize possible seed layouts
    seeds = []
    # Grid-based seed
    grid_positions = np.mgrid[0.05:0.95:5j, 0.05:0.95:5j].reshape(2, 25).T
    # Flatten grid positions to list format
    positions = np.concatenate([grid_positions, [[0.5,0.5]]], axis=0)  # Add extra in center
    seeds.append(positions)
    # Random seed
    rng = np.random.default_rng(42)
    seeds.append(rng.random((n, 2)))
    # Corner push seed
    corner_positions = np.array([[0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]])[:min(4, n)]
    center_positions = []
    # Fill rest randomly
    if len(corner_positions) < n:
        rest_positions = rng.random((n-len(corner_positions), 2))
        corner_positions = np.vstack([corner_positions, rest_positions])
    seeds.append(corner_positions)
    # Repeat with shuffled grid to make diverse initial configurations
    rng = np.random.default_rng(43)
    grid_pos2 = rng.random((n, 2))
    seeds.append(grid_pos2)
    # Start multiple chains
    results = []
    best_value = np.inf
    best_centers = None 
    best_radii = None
    for seed in seeds:
        # Call coding chain 1:
        current_centers = seed.copy()
        # Try one local refinement to get initial energy
        current_energy = energy(current_centers, prob, r, dist_params, n)
        if np.isnan(current_energy) or current_energy == np.inf:
            continue
        state = current_centers.flatten()
        # Re-solve to get approximation
        result_centers = current_centers
        # Just take current setup
        dist_ij = cdist(result_centers, result_centers)
        idx = 0
        for i in range(n):
            for j in range(i+1, n):
                dist_params[idx].value = dist_ij[i][j]
                idx += 1
        # Anti-infinite loop condition
        try:
            prob.solve(solver=cp.ECOS, abstol=1e-12, reltol=1e-12)
        except:
            continue
        if prob.status not in ["optimal", "optimal_inaccurate"]:
            continue
        final_radii = r.value
        s = np.sum(final_radii)
        results.append((result_centers, final_radii, s))
        if s < best_value:
            best_value = s
            best_centers = result_centers
            best_radii = final_radii
    # Supervisor chain loop over seeds asynchronously (but sequential)
    # Take best from results
    if results:
        best_centers, best_radii, _ = min(results, key=lambda x: -x[2])
        # Further refine computed.
        best_centers = local_refine(best_centers, prob, r, dist_params, n)
        # Re-run high accuracy finalization
        # Prepare final parameters
        best_centers, best_radii = finalise(best_centers, prob, r, dist_params, n)
        sum_radii = np.sum(best_radii)
        return best_centers, best_radii, sum_radii
    else:
        # Return last valid attempt in case of collapse 
        return best_centers, best_radii, best_value
