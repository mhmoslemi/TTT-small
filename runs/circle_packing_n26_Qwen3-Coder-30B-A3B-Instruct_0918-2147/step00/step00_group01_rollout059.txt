import numpy as np
from scipy.optimize import minimize

def initial_layout():
    # Create a hexagonal grid layout with 5 rows and 5 columns
    rows, cols = 5, 5
    spacing = 0.18
    radius = spacing / 2 - 0.01
    
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = (j + 0.5) * spacing
            y = (i + 0.5) * spacing
            # Adjust to stay within bounds
            x = max(radius, min(1 - radius, x))
            y = max(radius, min(1 - radius, y))
            centers.append([x, y])
    
    # Add one more center at the square center
    centers.append([0.5, 0.5])
    centers = np.array(centers)
    radii = np.full(26, radius)
    return centers, radii

def pack_vector(centers, radii):
    # Combine centers and radii into a single vector
    z = np.empty(3 * 26)
    for i in range(26):
        z[3*i] = centers[i, 0]
        z[3*i+1] = centers[i, 1]
        z[3*i+2] = radii[i]
    return z

def unpack_vector(z):
    # Extract centers and radii from the vector
    centers = np.empty((26, 2))
    radii = np.empty(26)
    for i in range(26):
        centers[i, 0] = z[3*i]
        centers[i, 1] = z[3*i+1]
        radii[i] = z[3*i+2]
    return centers, radii

def objective(z):
    centers, radii = unpack_vector(z)
    return -np.sum(radii)

def constraint_boundary(z):
    centers, radii = unpack_vector(z)
    constraints = []
    # For each circle, add four boundary constraints
    for i in range(26):
        x, y, r = centers[i, 0], centers[i, 1], radii[i]
        constraints.append(x - r)  # x - r >= 0
        constraints.append(1 - x - r)  # 1 - x - r >= 0
        constraints.append(y - r)  # y - r >= 0
        constraints.append(1 - y - r)  # 1 - y - r >= 0
    return np.array(constraints)

def constraint_nonoverlap(z):
    centers, radii = unpack_vector(z)
    constraints = []
    for i in range(26):
        for j in range(i+1, 26):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx*dx + dy*dy
            r_sum = radii[i] + radii[j]
            constraints.append(dist_sq - r_sum*r_sum)
    return np.array(constraints)

def run_packing():
    best_sum = -np.inf
    best_centers = None
    best_radii = None

    for _ in range(3):
        centers, radii = initial_layout()
        z0 = pack_vector(centers, radii)

        # Define constraints
        constraints = []
        constraints.append({'type': 'ineq', 'fun': constraint_boundary})
        constraints.append({'type': 'ineq', 'fun': constraint_nonoverlap})

        # Define bounds
        bounds = [(0, 1) for _ in range(26*3)]

        # Optimize
        result = minimize(
            objective,
            z0,
            method='SLSQP',
            constraints=constraints,
            bounds=bounds,
            options={'ftol': 1e-9, 'maxiter': 2000}
        )

        if not result.success:
            continue

        opt_centers, opt_radii = unpack_vector(result.x)
        # Clip any negative radii
        opt_radii = np.maximum(opt_radii, 0)
        sum_radii = np.sum(opt_radii)
        
        # Validate
        is_valid, _ = validate_packing(opt_centers, opt_radii)
        if is_valid and sum_radii > best_sum:
            best_sum = sum_radii
            best_centers = opt_centers.copy()
            best_radii = opt_radii.copy()

    # Boost final sum by doing one more small optimization with few restarts
    best_sum = -np.inf
    for _ in range(2):
        centers, radii = initial_layout()
        # Perturb slightly
        centers += np.random.uniform(-0.01, 0.01, centers.shape)
        radii += np.random.uniform(-0.005, 0.005, radii.shape)
        radii = np.maximum(radii, 0.001)
        z0 = pack_vector(centers, radii)
        
        constraints = []
        constraints.append({'type': 'ineq', 'fun': constraint_boundary})
        constraints.append({'type': 'ineq', 'fun': constraint_nonoverlap})
        bounds = [(0, 1) for _ in range(26*3)]

        result = minimize(
            objective,
            z0,
            method='SLSQP',
            constraints=constraints,
            bounds=bounds,
            options={'ftol': 1e-9, 'maxiter': 1000}
        )

        if not result.success:
            continue

        opt_centers, opt_radii = unpack_vector(result.x)
        opt_radii = np.maximum(opt_radii, 0)
        sum_radii = np.sum(opt_radii)
        
        is_valid, _ = validate_packing(opt_centers, opt_radii)
        if is_valid and sum_radii > best_sum:
            best_sum = sum_radii
            best_centers = opt_centers.copy()
            best_radii = opt_radii.copy()

    if best_sum == -np.inf:
        raise RuntimeError("Failed to find a valid packing.")
        
    return best_centers, best_radii, best_sum
