import numpy as np

def hex_lattice(d):
    """Generate a hexagonal lattice within [0,1]x[0,1] with spacing d."""
    sqrt3 = np.sqrt(3)
    dy = sqrt3 * d / 2
    rows = []
    y = 0
    row_idx = 0
    while y <= 1 + 1e-12:
        x_start = d / 2 if row_idx % 2 == 1 else 0
        x = x_start
        row = []
        while x <= 1 + 1e-12:
            row.append([x, y])
            x += d
        rows.extend(row)
        y += dy
        row_idx += 1
    return np.array(rows)

def add_border_circles(points, d):
    """Append six border-enhanced circles to the points."""
    r = d / 4
    # 4 corners
    corners = [
        [r, r],
        [1 - r, r],
        [r, 1 - r],
        [1 - r, 1 - r]
    ]
    # 2 edges
    edges = [
        [0.5, r],
        [0.5, 1 - r]
    ]
    border_points = np.array(corners + edges)
    return np.vstack([points, border_points])

def initial_radii(points):
    """Compute initial uniform radius for all points."""
    n = points.shape[0]
    # For each point, compute minimum distance to walls
    wall_distances = np.array([
        points[:, 0],  # x
        1 - points[:, 0],  # 1-x
        points[:, 1],  # y
        1 - points[:, 1]  # 1-y
    ]).min(axis=0)
    # Compute pairwise distances
    distances = np.linalg.norm(points[:, None, :] - points[None, :, :], axis=2)
    # For each point, compute minimum distance to others, excluding self
    min_distances = np.array([np.min(distances[i, np.arange(n) != i]) for i in range(n)])
    # Initial radii: min of wall and half the min distance to others
    r0 = np.min(np.minimum(wall_distances, min_distances / 2))
    return np.full(n, r0)

def inflate_radii(centers, radii):
    """Inflate radii greedily respecting constraints."""
    n = centers.shape[0]
    # Compute distance matrix
    D = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    new_radii = radii.copy()
    updated = True
    while updated:
        updated = False
        for i in range(n):
            # Wall constraints
            r_i_max = min(
                centers[i, 0],
                1 - centers[i, 0],
                centers[i, 1],
                1 - centers[i, 1]
            )
            # Pairwise constraints
            pairwise_caps = []
            for j in range(n):
                if i != j:
                    pairwise_caps.append(D[i, j] - radii[j])
            if len(pairwise_caps) > 0:
                r_i_max = min(r_i_max, min(pairwise_caps))
            if r_i_max > new_radii[i] + 1e-6:
                new_radii[i] = r_i_max
                updated = True
    return new_radii

def relax_centers(centers, radii, steps=10):
    """Relax center positions using force-based dynamics."""
    n = centers.shape[0]
    κ = 0.02
    λ = 0.01
    α = 0.5  # damping
    β = 0.02  # repulsion threshold
    γ = 0.02  # wall buffer
    old_centers = centers.copy()
    for _ in range(steps):
        # Force vectors
        forces = np.zeros_like(centers)
        # Compute pairwise distances
        D = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
        # Repulsion forces
        for i in range(n):
            for j in range(n):
                if i == j:
                    continue
                dist = D[i, j]
                δ = radii[i] + radii[j] - dist
                if δ > 0:
                    # Overlap (should not normally occur after inflation)
                    # But we are using a non-zero β to allow minor overlaps that can be fixed
                    pass
                safe_dist = radii[i] + radii[j] + β
                if dist < safe_dist:
                    # Repulsive force
                    diff = centers[i] - centers[j]
                    if dist > 1e-10:
                        force = κ * max(0, safe_dist - dist) * diff / dist
                        forces[i] += force
        # Wall forces
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            force_x = 0
            force_y = 0
            if x - r < γ:
                force_x += λ * max(0, γ - (x - r))
            if 1 - x - r < γ:
                force_x -= λ * max(0, γ - (1 - x - r))
            if y - r < γ:
                force_y += λ * max(0, γ - (y - r))
            if 1 - y - r < γ:
                force_y -= λ * max(0, γ - (1 - y - r))
            forces[i] += [force_x, force_y]
        # Update centers
        centers += α * forces
        # Clamp to boundaries
        centers[:, 0] = np.clip(centers[:, 0], radii, 1 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1 - radii)
    return centers

def run_packing():
    # Initial parameters
    d = 0.18  # hexagonal spacing
    MAX_OUTER = 30

    # 1. Generate hexagon core
    core_points = hex_lattice(d)
    # Keep only first 20
    core_points = core_points[:20]
    # 2. Add border circles
    all_points = add_border_circles(core_points, d)
    n = all_points.shape[0]
    # 3. Compute initial radius
    radii = initial_radii(all_points)
    centers = all_points.copy()
    # 4. Optimize iteratively
    previous_sum = np.sum(radii)
    for _ in range(MAX_OUTER):
        radii = inflate_radii(centers, radii)
        centers = relax_centers(centers, radii)
        current_sum = np.sum(radii)
        if current_sum - previous_sum < 1e-6:
            break
        previous_sum = current_sum
    # Final validation
    is_valid, msg = validate_packing(centers, radii)
    assert is_valid, msg
    return centers, radii, np.sum(radii)
