import numpy as np
from scipy.optimize import linprog
from itertools import combinations

def generate_hex_lattice(h):
    sqrt3 = np.sqrt(3)
    points = []
    for j in range(int(1.0 / (h * sqrt3 / 2)) + 2):
        y = j * h * sqrt3 / 2
        if y > 1:
            break
        for i in range(int(1.0 / h) + 2):
            x = i * h + (j % 2) * (h / 2)
            if x > 1:
                break
            points.append([x, y])
    return np.array(points)


def build_neighbour_list(centers, max_dist):
    edges = []
    n = centers.shape[0]
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < max_dist:
                edges.append((i, j, dist))
    return edges


def solve_radius_lp(selected_indices, R_wall, neighbour_list):
    n_selected = len(selected_indices)
    if n_selected == 0:
        return np.array([]), True

    # Build LP: maximize sum of radii under linear constraints
    c = -np.ones(n_selected)  # We negate because linprog minimizes

    # Build A_ub and b_ub for inequalities -r_i - r_j <= -d_ij
    A_ub = []
    b_ub = []

    for i, j, d_ij in neighbour_list:
        if i in selected_indices and j in selected_indices:
            idx_i = selected_indices.index(i)
            idx_j = selected_indices.index(j)
            row = np.zeros(n_selected)
            row[idx_i] = -1
            row[idx_j] = -1
            A_ub.append(row)
            b_ub.append(-d_ij)

    # Wall constraints: 0 <= r_i <= R_wall[i]
    A_eq = []
    b_eq = []
    bounds = []

    for i, idx in enumerate(selected_indices):
        bounds.append((0, R_wall[idx]))

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if res.success:
        return res.x, True
    else:
        return np.array([]), False


def upper_bound(partial_set, remaining_indices, R_wall):
    bound = 0
    # Add wall bounds for already selected
    for i in partial_set:
        bound += R_wall[i]
    # Add max possible wall radii for remaining slots
    max_wall = [R_wall[i] for i in remaining_indices]
    max_wall.sort(reverse=True)
    for i in range(26 - len(partial_set)):
        bound += max_wall[i]
    return bound


def branch_and_bound(centers, R_wall, neighbour_list):
    best_sum = -1
    best_solution = None

    def search(S_partial, U, lb):
        nonlocal best_sum, best_solution

        if len(S_partial) == 26:
            radii, feasible = solve_radius_lp(S_partial, R_wall, neighbour_list)
            if feasible and np.sum(radii) > lb:
                lb = np.sum(radii)
                best_solution = (np.array(S_partial), radii)
            return lb

        if len(S_partial) + len(U) < 26:
            return lb

        ub = upper_bound(S_partial, U, R_wall)
        if ub <= lb:
            return lb

        # Select candidate with largest R_wall among U
        best_idx = None
        best_r = -1
        for i in U:
            if R_wall[i] > best_r:
                best_r = R_wall[i]
                best_idx = i

        if best_idx is None:
            return lb

        # Exclude
        new_U = U.copy()
        new_U.remove(best_idx)
        new_lb = search(S_partial.copy(), new_U, lb)

        # Include
        new_S = S_partial.copy()
        new_S.append(best_idx)
        new_U = U.copy()
        new_U.remove(best_idx)
        new_lb = search(new_S, new_U, new_lb)

        return new_lb

    initial_U = list(range(len(centers)))
    search([], initial_U, -1)

    if best_solution is None:
        return None, None
    selected_indices, radii = best_solution
    return centers[selected_indices], radii


def local_continuous_refinement(centers, radii):
    # Simple approach: just re-solve the problem starting with these values
    # Small perturbation
    centers = centers.copy()
    radii = radii.copy()
    for i in range(centers.shape[0]):
        centers[i] += np.random.uniform(-0.002, 0.002, 2)
        radii[i] = max(1e-8, radii[i])
    
    # Sequential optimization using trust-constr
    from scipy.optimize import minimize
    
    def objective(x):
        return -np.sum(x[26:])  # negative because we maximize

    def constraint_func(x):
        # x contains [x1, y1, ..., x26, y26, r1, ..., r26]
        n = x.shape[0] // 2
        centers_x = x[:n]
        centers_y = x[n:2*n]
        radii_vals = x[2*n:]
        
        constraints = []
        
        # Wall constraints
        for i in range(26):
            constraints.append(centers_x[i] - radii_vals[i])
            constraints.append(1 - centers_x[i] - radii_vals[i])
            constraints.append(centers_y[i] - radii_vals[i])
            constraints.append(1 - centers_y[i] - radii_vals[i])
        
        # Separation constraints
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers_x[i] - centers_x[j]
                dy = centers_y[i] - centers_y[j]
                dist = np.sqrt(dx**2 + dy**2)
                constraints.append(dist - radii_vals[i] - radii_vals[j])
        
        return np.array(constraints)

    # Flatten initial guess
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Define bounds: x, y in [0,1], radius >= 0
    bounds = []
    for _ in range(26):
        bounds.append((0.0, 1.0))
    for _ in range(26):
        bounds.append((0.0, 1.0))
    for _ in range(26):
        bounds.append((0.0, 1.0))

    constraints = {'type': 'ineq', 'fun': constraint_func}
    
    result = minimize(objective, x0, method='trust-constr', bounds=bounds, constraints=constraints,
                      options={'gtol': 1e-6, 'maxiter': 200})
    
    if result.success:
        final_centers = result.x[:52].reshape((26, 2))
        final_radii = result.x[52:]
        return final_centers, final_radii
    return centers, radii
