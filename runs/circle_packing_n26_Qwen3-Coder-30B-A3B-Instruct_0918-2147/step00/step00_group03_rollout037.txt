import numpy as np
from scipy.optimize import minimize
import scipy.sparse as sp

def init_hex_grid(n=26, r_init=0.04):
    """Generate initial hexagonal grid of circles inside unit square."""
    # Hexagonal lattice parameters
    h = np.sqrt(3) * r_init
    rows = []
    y = r_init
    row_idx = 0
    while len(rows) < n:
        if row_idx % 2 == 0:
            x_start = r_init
            x_step = 2 * r_init
        else:
            x_start = 2 * r_init
            x_step = 2 * r_init
        x = x_start
        row = []
        while len(row) < n and x <= 1 - r_init:
            row.append((x, y))
            x += x_step
        if row:
            rows.extend(row)
        y += h
        row_idx += 1
    # Take first n points
    centers = np.array(rows[:n])
    radii = np.full(n, r_init)
    return centers, radii

def max_radii(centers, radii):
    """Iteratively maximize radii while keeping centers fixed."""
    # Compute distance matrix
    D = np.sqrt(((centers[:, None, :] - centers[None, :, :]) ** 2).sum(-1))

    old_radii = radii.copy()
    max_iter = 100
    for _ in range(max_iter):
        # Wall constraints
        wall = np.minimum(centers, 1 - centers).min(axis=1)

        # Neighbour constraints
        neighbor = D - radii[:, None]
        # Set diagonal to large value to exclude self-distance
        np.fill_diagonal(neighbor, np.inf)
        neighbor_min = np.min(neighbor, axis=1)

        new_radii = np.minimum(wall, neighbor_min)
        new_radii = np.maximum(new_radii, 0.0)

        if np.max(np.abs(new_radii - old_radii)) < 1e-8:
            break
        old_radii = new_radii

    return new_radii

def hill_climb(centers, radii, iterations=5000, step=0.01):
    """Stochastic hill-climbing improvement."""
    D = np.sqrt(((centers[:, None, :] - centers[None, :, :]) ** 2).sum(-1))
    current_sum = np.sum(radii)
    for _ in range(iterations):
        # Randomly select a circle to adjust
        i = np.random.randint(0, len(centers))
        old_center = centers[i].copy()
        old_radius = radii[i]
        # Propose a move
        dx = np.random.uniform(-step, step)
        dy = np.random.uniform(-step, step)

        # Ensure new center is within bounds
        # Note: each circle should at least have its radius in distance from wall
        new_x = np.clip(old_center[0] + dx, old_radius, 1 - old_radius)
        new_y = np.clip(old_center[1] + dy, old_radius, 1 - old_radius)

        centers[i] = [new_x, new_y]

        # Recompute maximal radii (with new center)

def refine_with_scipy(centers, radii):
    """Use scipy.optimize.minimize to refine."""
    n = len(centers)
    # Variable vector [x1,y1,...,xn,yn,r1,...,rn]
    x0 = np.concatenate([centers.flatten(), radii])

    # Define bounds
    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n  # max radius 0.5

    def obj_func(x):
        # x = [x1,y1,...,xn,yn,r1,...,rn]
        c = x[:2 * n].reshape(n, 2)
        r = x[2 * n:]
        return -np.sum(r)

    # State for computing Jacobians, defined once
    D = np.sqrt(((c[:, None, :] - c[None, :, :]) ** 2).sum(-1))

    def wall_constraints(x):
        c = x[:2 * n].reshape(n, 2)
        r = x[2 * n:]
        # Wall distance constraints:
        # r_i <= x_i  =>  r_i - x_i <= 0
        # r_i <= 1-x_i =>  r_i + x_i - 1 <= 0
        # r_i <= y_i  =>  r_i - y_i <= 0
        # r_i <= 1-y_i =>  r_i + y_i - 1 <= 0
        # We return vector of:
        # r_i - x_i, r_i + x_i - 1, r_i - y_i, r_i + y_i - 1 for i=0..25
        constraints = np.hstack([
            r - c[:, 0],   # r_i - x_i
            r + c[:, 0] - 1,  # r_i + x_i - 1
            r - c[:, 1],   # r_i - y_i
            r + c[:, 1] - 1   # r_i + y_i - 1
        ])
        return constraints

    def non_overlap_constraints(x):
        c = x[:2 * n].reshape(n, 2)
        r = x[2 * n:]
        # For i,j: r_i + r_j <= dist(c_i, c_j)
        # Equivalently: r_i + r_j - dist(c_i, c_j) <= 0
        D = np.sqrt(((c[:, None, :] - c[None, :, :]) ** 2).sum(-1))
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                val = r[i] + r[j] - D[i, j]
                constraints.append(val)
        return np.array(constraints)

    def construct_jac(x):
        m = len(x)
        # Initialize an empty list to store the sparse Jacobian in COO format
        rows = []
        cols = []
        data = []

        # Wall Jacobian (4 per circle)
        c = x[:2 * n].reshape(n, 2)
        r = x[2 * n:]
        for i in range(n):
            # wrt x_i: -1 (r_i - x_i)
            rows.extend([4*i,   4*i,   4*i+1, 4*i+1])
            cols.extend([2*i,   2*i,   2*i,   2*i+1])
            data.extend([-1.0, 1.0, 1.0, 1.0])  # Reverse signs when using positive inequality

        # Slight offset to make larger Jacobian loop more obvious
        # Note precision loss for small values
        j = 4 * n
        # Non-overlap constraints Jacobian
        for i in range(n):
            for j in range(i + 1, n):
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist > 1e-12:
                    dx /= dist
                    dy /= dist

                    # wrt x_i
                    rows.extend([j, j, j, j])
                    cols.extend([2*i, 2*i+1, 2*j, 2*j+1])
                    data.extend([dx, dy, -dx, -dy])

                    # wrt r_i
                    rows.append(j)
                    cols.append(2 * n + i)
                    data.append(1.0)
                    # wrt r_j
                    rows.append(j)
                    cols.append(2 * n + j)
                    data.append(1.0)

        row_indices = np.array(rows, dtype=int)
        col_indices = np.array(cols, dtype=int)
        values = np.array(data, dtype=float)
        return sp.coo_matrix((values, (row_indices, col_indices)), shape=(len(rows), m))

    # Define constraints
    # wall constraints = [4*n]
    # overlap constraints = n*(n-1)/2
    # Total constraints: 4*n + n*(n-1)/2 = 4*n + n(n-1)/2 = n(4 + (n-1)/2) = n*(n+7)/2
    # For n = 26: 26 * (26+7)/2 = 26*33/2 = 429

    # Create a custom function that checks all constraints
    def walls_func(x):
        return wall_constraints(x)

    def overlaps_func(x):
        return non_overlap_constraints(x)

    # Nonlinear constraints
    nl_cons = [
        {'type': 'ineq', 'fun': walls_func},
        {'type': 'ineq', 'fun': overlaps_func}
    ]

    # Convert to inequality format for ability to build Jacobians
    # Just one combined constraint, but does not split it explicitly
    # Which is also fine, basic use like below is ok

    result = minimize(
        fun=obj_func,
        x0=x0,
        method='trust-constr',
        bounds=bounds,
        constraints=nl_cons,
        options={'maxiter': 200}
    )

    if result.success:
        updated_x = result.x
        centers = updated_x[:2 * n].reshape(n, 2)
        radii = updated_x[2 * n:]
    return centers, radii

def run_packing():
    centers, radii = init_hex_grid()
    radii = max_radii(centers, radii)
    centers, radii = hill_climb(centers, radii)
    centers, radii = refine_with_scipy(centers, radii)
    sum_r = float(radii.sum())
    return centers, radii, sum_r
