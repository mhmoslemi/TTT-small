import numpy as np
import random

def make_hex_seed(n, margin):
    """Generate initial hexagonal seed with varying radii based on wall distances."""
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))
    
    # Create hexagonal grid
    centers = []
    radii = []
    
    # Offset for hexagonal arrangement
    dy = 1.0 / (rows + 1)
    dx = np.sqrt(3) * dy / 2
    
    for i in range(rows):
        for j in range(cols):
            if len(centers) >= n:
                break
            x = margin + j * 2 * dx + (i % 2) * dx
            y = margin + i * dy
            if x > 1 - margin or y > 1 - margin:
                continue
            centers.append([x, y])
            radii.append(min(x, 1 - x, y, 1 - y) * 0.4)

    centers = np.array(centers[:n])
    radii = np.array(radii[:n])
    
    # Ensure minimum radius
    min_r = 0.015
    scale = min_r / np.min(radii)
    centers *= scale
    radii *= scale
    
    return centers, radii

def max_feasible_radius(i, centers, radii):
    """Compute the maximum feasible radius for circle i given all others."""
    x, y = centers[i]
    r = radii[i]
    
    # Wall distances
    d_left = x
    d_right = 1 - x
    d_bottom = y
    d_top = 1 - y
    
    # Neighbour distances
    d_min = float('inf')
    for k in range(len(centers)):
        if i == k:
            continue
        dx = centers[i][0] - centers[k][0]
        dy = centers[i][1] - centers[k][1]
        dist = np.sqrt(dx*dx + dy*dy)
        d_min = min(d_min, dist - radii[k])
    
    return min(d_left, d_right, d_bottom, d_top, d_min)

def inflate_radii(centers, radii, alpha):
    """Inflate all radii by a global amount and then adjust to feasibility."""
    old_sum = np.sum(radii)
    
    mean_r = np.mean(radii)
    delta = alpha * mean_r
    
    new_radii = radii + delta
    
    for i in range(len(centers)):
        max_rad = max_feasible_radius(i, centers, radii)
        new_radii[i] = min(new_radii[i], max_rad)
    
    new_sum = np.sum(new_radii)
    
    return new_radii, new_sum - old_sum

def anneal_centers(centers, radii, temp0, gamma, sigma0, steps):
    """Simulated annealing to reposition circle centers."""
    current_radii = radii.copy()
    current_centers = centers.copy()
    
    best_sum = np.sum(current_radii)
    best_centers = current_centers.copy()
    best_radii = current_radii.copy()
    
    T = temp0
    sigma = sigma0
    
    for step in range(steps):
        # Occasionally translate all centers
        if step % 50 == 0:
            offset = np.random.normal(0, 1e-5, size=2)
            current_centers += offset
        
        # Pick a random circle to move
        i = np.random.randint(0, len(centers))
        
        # Generate displacement
        disp = np.random.normal(0, sigma, size=2)
        
        # Proposed new center
        new_center = current_centers[i] + disp
        
        # Clip to square
        new_center[0] = np.clip(new_center[0], 0, 1)
        new_center[1] = np.clip(new_center[1], 0, 1)
        
        # Save old values
        old_center = current_centers[i].copy()
        old_radius = current_radii[i]
        
        # Update
        current_centers[i] = new_center
        
        # Compute feasible radius for moved circle
        new_radius = max_feasible_radius(i, current_centers, current_radii)
        
        # Accept or reject based on feasibility and acceptance probability
        if new_radius > old_radius:
            # Accept increase in radius
            current_radii[i] = new_radius
        else:
            # Revert center and compute old feasible radius
            current_centers[i] = old_center
            old_radius = max_feasible_radius(i, current_centers, current_radii)
            # Accept with probability based on Boltzmann probability
            delta = new_radius - old_radius
            if delta < 0 and np.random.rand() > np.exp(delta / T):
                current_centers[i] = old_center
                current_radii[i] = old_radius
        
        # Resulting sum after updating
        new_sum = np.sum(current_radii)
        
        # Update best
        if new_sum > best_sum:
            best_sum = new_sum
            best_centers = current_centers.copy()
            best_radii = current_radii.copy()
        
        # Update annealing
        T *= gamma
        if T < 1e-10:
            break
    
    return best_centers, best_radii, best_sum

def mixed_cycle(centers, radii):
    """Alternate between inflation and repositioning."""
    alpha = 0.04
    epsilon = 1e-6
    stagnation_limit = 8
    sa_steps = 200
    cycle_stop = 5
    
    last_sum = np.sum(radii)
    stagnation_count = 0
    cycle_count = 0
    
    while cycle_count < cycle_stop:
        # Inflation phase (Phase B)
        for i in range(100):  # limit inflation steps
            radii, delta_sum = inflate_radii(centers, radii, alpha)
            if abs(delta_sum) < epsilon:
                stagnation_count += 1
            else:
                stagnation_count = 0
                last_sum += delta_sum
            
            if stagnation_count >= stagnation_limit:
                break
        
        # Check for convergence
        if stagnation_count >= stagnation_limit:
            cycle_count += 1
            if cycle_count >= cycle_stop:
                break
            stagnation_count = 0
        
        # Reposition phase (Phase C)
        temp_start = 0.02
        temp_decay = 0.97
        sigma_start = 0.03
        sa_block_size = sa_steps
        
        centers, radii, new_sum = anneal_centers(centers, radii,
                                                temp_start, temp_decay,
                                                sigma_start, sa_block_size)

        last_sum = new_sum
        
    return centers, radii

def polish_radii(centers, radii, beta, passes):
    """Apply exact polishing to refine radii."""
    for _ in range(passes):
        # Compute exact gaps for each circle
        new_radii = radii.copy()
        changed = True
        
        while changed:
            changed = False
            for i in range(len(centers)):
                x, y = centers[i]
                r = new_radii[i]
                
                # Wall gaps
                g_wall = min(x - r, 1 - x - r, y - r, 1 - y - r)
                g_wall = max(0, g_wall)
                
                # Neighbour gaps
                g_nei = float('inf')
                for j in range(len(centers)):
                    if i == j:
                        continue
                    dx = centers[i][0] - centers[j][0]
                    dy = centers[i][1] - centers[j][1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    gap = dist - (r + new_radii[j])
                    if gap < g_nei:
                        g_nei = gap
                
                g_nei = max(0, g_nei)
                g = min(g_wall, g_nei)
                
                if g > 0:
                    new_radius = r + beta * g
                    max_feasible = max_feasible_radius(i, centers, new_radii)
                    new_radii[i] = min(max_feasible, new_radius)
                    if abs(new_radii[i] - r) > 1e-10:
                        changed = True
                            
        # Overlap resolution
        over = True
        while over:
            over = False
            for i in range(len(centers)):
                for j in range(i + 1, len(centers)):
                    dx = centers[i][0] - centers[j][0]
                    dy = centers[i][1] - centers[j][1]
                    dist = np.sqrt(dx*dx + dy*dy)
                    overlap = dist - (new_radii[i] + new_radii[j])
                    if overlap < 0:
                        # Shrink each by half of overlap
                        reduction = -overlap / 2
                        new_radii[i] = max(1e-10, new_radii[i] - reduction)
                        new_radii[j] = max(1e-10, new_radii[j] - reduction)
                        over = True
                        
        radii = new_radii
        
    return radii
