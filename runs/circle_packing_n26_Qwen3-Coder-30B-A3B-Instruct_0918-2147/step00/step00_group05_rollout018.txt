import numpy as np
from scipy.optimize import linprog, minimize
from math import sqrt, ceil

def build_hex_lattice(d):
    """Build a hexagonal lattice to seed the initial configuration."""
    # Generate hexagonal lattice in the unit square
    # Approximate spacing for 26 circles
    spacing = d / sqrt(3)
    rows = ceil(1 / spacing)
    cols = ceil(1 / (spacing * sqrt(3)))
    
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = (j + 0.5 * (i % 2)) * spacing
            y = i * spacing * sqrt(3) / 2
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers.append([x, y])
            if len(centers) >= 26:
                break
        if len(centers) >= 26:
            break
    
    centers = centers[:26]
    while len(centers) < 26:
        centers.append([0.5, 0.5])  # fallback
    
    return np.array(centers)

def lp_max_radii(centers):
    """Solve a linear program to compute the max radii for fixed centers."""
    N = centers.shape[0]
    c = np.ones(N)
    
    # Wall constraints (4 constraints per circle)
    A_ub = []
    b_ub = []
    
    for i in range(N):
        x, y = centers[i]
        # r <= x, r <= 1 - x, r <= y, r <= 1 - y
        row = [0] * N
        row[i] = -1
        A_ub.append(row)
        b_ub.append(-x)
        A_ub.append(row)
        b_ub.append(-(1 - x))
        A_ub.append(row)
        b_ub.append(-y)
        A_ub.append(row)
        b_ub.append(-(1 - y))
    
    # Pairwise non-overlap constraints (binomial*N constraints)
    for i in range(N):
        for j in range(i + 1, N):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist = sqrt(dx*dx + dy*dy)
            
            row = [0] * N
            row[i] = -1
            row[j] = -1
            A_ub.append(row)
            b_ub.append(-dist)
    
    # Ensure non-negative radii
    bounds = [(0, None) for _ in range(N)]
    
    # Solve LP
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', tol=1e-9)
        if result.success:
            radii = result.x
            total_radius = sum(radii)
            return radii, total_radius
        else:
            return np.zeros(N), 0.0
    except Exception:
        return np.zeros(N), 0.0

def perturb_centers(centers, sigma):
    """Add perturbations to centers."""
    perturbed = centers + np.random.normal(0, sigma, centers.shape)
    perturbed = np.clip(perturbed, 0, 1)
    return perturbed

def anneal_centers(initial_centers):
    """Run simulated annealing to optimize centers."""
    np.random.seed(42)
    current_centers = initial_centers.copy()
    current_radii, current_sum = lp_max_radii(current_centers)
    best_centers, best_radii, best_sum = current_centers.copy(), current_radii.copy(), current_sum
    
    # Annealing schedule
    schedule = [
        (0.02, 0.05, 2000),
        (0.01, 0.03, 4000),
        (0.005, 0.015, 3000),
        (0.002, 0.008, 2000)
    ]
    
    for T, sigma, num_iters in schedule:
        for _ in range(num_iters):
            new_centers = perturb_centers(current_centers, sigma)
            new_radii, new_sum = lp_max_radii(new_centers)
            
            # Acceptance probability
            if new_sum > current_sum or np.random.rand() < np.exp((new_sum - current_sum) / T):
                current_centers = new_centers
                current_radii = new_radii
                current_sum = new_sum
                
                if current_sum > best_sum:
                    best_centers = current_centers.copy()
                    best_radii = current_radii.copy()
                    best_sum = current_sum

    return best_centers, best_radii, best_sum

def local_refine(centers, radii):
    """Refine centers and radii with trust-constr optimization."""
    N = centers.shape[0]
    # Start with current radii as a lower bound
    initial_r = radii.copy()
    
    def fun(x):
        # x is [c1_x, c1_y, ..., cN_x, cN_y, r1, ..., rN]
        coords = x[:2*N].reshape((N, 2))
        rs = x[2*N:]
        return -sum(rs)
    
    def jac(x):
        # Gradient of the objective function
        jac_vec = np.zeros_like(x)
        jac_vec[2*N:] = -1.0
        return jac_vec
    
    def constraint_func(x):
        # x = [c1_x, c1_y, ..., cN_x, cN_y, r1, ..., rN]
        coords = x[:2*N].reshape((N, 2))
        rs = x[2*N:]
        
        result = []
        # Distance constraints (overlap)
        for i in range(N):
            for j in range(i+1, N):
                dx = coords[i, 0] - coords[j, 0]
                dy = coords[i, 1] - coords[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                result.append(dist - (rs[i] + rs[j]))
        
        # Wall constraints (r <= min(x, 1-x, y, 1-y))
        for i in range(N):
            x, y = coords[i]
            r = rs[i]
            result.append(x - r)
            result.append(1 - x - r)
            result.append(y - r)
            result.append(1 - y - r)
        
        # Lower bound radii
        for i in range(N):
            result.append(rs[i] - initial_r[i])  # r_i >= r_i^LP
        
        return np.array(result)
    
    con = {'type': 'ineq', 'fun': constraint_func}
    
    # Initial guess
    initial_x = np.hstack([centers.reshape(-1), initial_r])
    
    bounds = []
    for i in range(N):
        bounds.append((0, 1))     # x coordinate
        bounds.append((0, 1))     # y coordinate
        bounds.append((0, None))  # radius
        
    # Optimize using trust-constr
    try:
        result = minimize(fun, initial_x, method='trust-constr', jac=jac, bounds=bounds, constraints=con)
        if result.success:
            final_coords = result.x[:2*N].reshape((N, 2))
            final_radii = result.x[2*N:]
            new_sum = sum(final_radii)
            return final_coords, final_radii, new_sum
        else:
            # If it fails, use previous good values
            return centers, radii, sum(radii)
    except Exception:
        return centers, radii, sum(radii)
