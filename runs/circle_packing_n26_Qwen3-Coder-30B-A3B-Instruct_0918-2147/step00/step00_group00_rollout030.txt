import numpy as np
import random

def make_hex_seed(n, margin):
    """Generate an initial hexagonal seed layout."""
    # Create hexagonal grid with spacing based on n and margin
    side = np.sqrt(2 * n / (3 * np.sqrt(3))) - margin
    if side <= 0:
        side = 0.1  # fallback
    
    sqrt3 = np.sqrt(3)
    
    rows = int(np.ceil(np.sqrt(n / (sqrt3 / 2))))
    cols = int(np.ceil(n / rows))
    
    centers = []
    radii = []
    
    y = margin
    row = 0
    count = 0
    
    while count < n and row < rows:
        x = margin
        col = 0
        while col < cols and count < n:
            # Add to centers
            centers.append([x, y])
            # Place in grid; adjust column offset per row
            x += side * 1.5
            col += 1
            count += 1

        y += side * sqrt3
        row += 1
        if row % 2 == 1:
            x = (side * 1.5) / 2 + margin
        else:
            x = margin

    centers = np.array(centers[:n])
    
    # Compute wall-distances-based radii
    radius_vals = []
    for i in range(n):
        x, y = centers[i]
        min_dist = min(x, 1 - x, y, 1 - y)
        radius_vals.append(min_dist * 0.45)
        
    radii = np.array(radius_vals)
    
    # Rescale to ensure min radius >= 0.015
    min_radius = np.min(radii)
    if min_radius < 0.015:
        factor = 0.015 / min_radius
        radii *= factor
        centers[:, 0] *= factor
        centers[:, 1] *= factor

    return centers, radii


def max_feasible_radius(i, centers, radii):
    """
    Compute exact max allowable radius for circle i.
    """
    x, y = centers[i]
    r = radii[i]
    # Wall distances
    d_L = x
    d_R = 1 - x
    d_B = y
    d_T = 1 - y
    dist_to_walls = [d_L, d_R, d_B, d_T]

    # Neighbour distances
    min_dist_to_nei = float('inf')
    for j in range(len(centers)):
        if i != j:
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist = np.sqrt(dx*dx + dy*dy)
            dist_to_nei = dist - radii[j]
            if dist_to_nei < min_dist_to_nei:
                min_dist_to_nei = dist_to_nei

    # Max feasible radius
    if min_dist_to_nei == float('inf'):
        min_dist_to_nei = float('inf')
    bounds = dist_to_walls + [min_dist_to_nei]
    r_max = min(bounds)
    return r_max


def inflate_radii(centers, radii, alpha):
    """
    Adds delta = alpha * mean(radii) to each radius, then clamps each
    to its maximal feasible value.
    """
    mean_r = np.mean(radii)
    delta = alpha * mean_r
    new_radii = radii + delta

    total_increase = 0.0
    for i in range(len(centers)):
        max_r = max_feasible_radius(i, centers, radii)
        new_radii[i] = min(new_radii[i], max_r)
        total_increase += new_radii[i] - radii[i]

    return new_radii, total_increase


def anneal_centers(centers, radii, temp0, gamma, sigma0, steps):
    """
    Simulated Annealing on center positions only.
    """
    # Initialize
    current_centers = centers.copy()
    current_radii = radii.copy()
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = np.sum(radii)
    
    # Temperature schedule
    temp = temp0
    
    # Hallucinate best_sum from the start as a good baseline
    for _ in range(5):
        idx = random.randint(0, len(current_centers) - 1)
        dx = np.random.normal(0, sigma0)
        dy = np.random.normal(0, sigma0)
        old_x, old_y = current_centers[idx]
        new_x = old_x + dx
        new_y = old_y + dy
        
        # Ensure it stays in bounds
        if not (0 <= new_x <= 1 and 0 <= new_y <= 1):
            continue
            
        current_centers[idx] = [new_x, new_y]
        
        # Recalculate feasible radius for this one
        new_r = max_feasible_radius(idx, current_centers, current_radii)
        old_r = current_radii[idx]
        current_radii[idx] = new_r
        new_sum = np.sum(current_radii)
        
        # Accept if improvement or by probabilty
        delta = new_sum - best_sum
        if delta > 0 or random.random() < np.exp(delta / temp):
            best_sum = new_sum
            best_centers = current_centers.copy()
            best_radii = current_radii.copy()
        else:
            # Restore
            current_centers[idx] = [old_x, old_y]
            current_radii[idx] = old_r
        
        temp *= gamma
        
        # Possibly global shift
        if _ % 50 == 0 and random.random() < 0.1:
            shift_x = np.random.normal(0, 0.002)
            shift_y = np.random.normal(0, 0.002)
            current_centers[:, 0] += shift_x
            current_centers[:, 1] += shift_y
            # Ensure all in bounds
            current_centers[:, 0] = np.clip(current_centers[:, 0], 0, 1)
            current_centers[:, 1] = np.clip(current_centers[:, 1], 0, 1)
            # Update all radii
            for i in range(len(current_centers)):
                r = max_feasible_radius(i, current_centers, current_radii)
                current_radii[i] = r

    return best_centers, best_radii, best_sum


def mixed_cycle(centers, radii, params):
    """Mix phases B and C."""
    alpha = params['alpha']
    epsilon = params['epsilon']
    K = params['K']
    M = params['M']
    P = params['P']
    T0 = params['T0']
    gamma = params['gamma']
    sigma0 = params['sigma0']

    total_sum = np.sum(radii)
    last_sum = total_sum
    stagnations = 0
    cycles = 0

    while cycles < P:
        # Phase B - inflation
        updated = True
        rounds = 0
        while updated and rounds < 20:
            old_sum = total_sum
            radii, increase = inflate_radii(centers, radii, alpha)
            total_sum += increase
            rounds += 1
            # Check if stagnated
            if abs(total_sum - old_sum) < epsilon:
                stagnations += 1
            else:
                stagnations = 0
            if stagnations >= K:
                updated = False
        
        # Phase C - SA
        new_centers, new_radii, new_sum = anneal_centers(
            centers, radii, T0, gamma, sigma0, M
        )
        total_sum = new_sum
        centers = new_centers
        radii = new_radii

        if abs(total_sum - last_sum) < epsilon:
            cycles += 1
        else:
            cycles = 0
        
        last_sum = total_sum
        # Optionally decrease rate slightly
        alpha *= 0.95

    return centers, radii, total_sum


def polish_radii(centers, radii, beta, passes):
    """Polish final gaps."""
    for _ in range(passes):
        changed = True
        while changed:
            changed = False
            for i in range(len(centers)):
                # Compute local gaps
                x, y = centers[i]
                r = radii[i]
                g_wall = min(x - r, 1 - x - r, y - r, 1 - y - r)
                g_nei = float('inf')
                for j in range(len(centers)):
                    if i != j:
                        dist = np.sqrt((centers[i][0] - centers[j][0])**2 + 
                                       (centers[i][1] - centers[j][1])**2)
                        gap = dist - (r + radii[j])
                        if gap < g_nei:
                            g_nei = gap
                g = min(g_wall, g_nei)
                if g > 0:
                    new_r = r + beta * g
                    if new_r < max_feasible_radius(i, centers, radii):
                        radii[i] = new_r
                        changed = True
            
            # overlapping sweep
            for i in range(len(centers)):
                for j in range(i + 1, len(centers)):
                    dx = centers[i][0] - centers[j][0]
                    dy = centers[i][1] - centers[j][1]
                    dist = np.sqrt(dx*dx + dy*dy)
                    overlap = dist - (radii[i] + radii[j])
                    if overlap < 0:
                        shrink = -overlap / 2.0
                        radii[i] -= shrink
                        radii[j] -= shrink
                        
    return centers, radii


def run_packing():
    # Parameters
    params = {
        'alpha': 0.04,
        'epsilon': 1e-6,
        'K': 8,
        'M': 200,
        'P': 5,
        'T0': 0.02,
        'gamma': 0.97,
        'sigma0': 0.03,
        'beta': 0.6,
        'passes': 3
    }

    # Make initial seed
    centers, radii = make_hex_seed(26, 0.04)
    # Run mixed cycle
    centers, radii, total_sum = mixed_cycle(centers, radii, params)
    # Polish final gaps
    centers, radii = polish_radii(centers, radii, params['beta'], params['passes'])
    # Final validation
    success, msg = validate_packing(centers, radii)
    assert success, msg
    return centers, radii, total_sum
