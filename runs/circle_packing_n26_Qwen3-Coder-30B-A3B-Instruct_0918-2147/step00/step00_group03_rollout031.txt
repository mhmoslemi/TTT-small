import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def init_grid(n_circles=26, spacing=0.07):
    rows = int(np.ceil(np.sqrt(n_circles)))
    cols = int(np.ceil(n_circles / rows))
    grid = []
    for k in range(rows):
        y = spacing / 2 + k * spacing * np.sqrt(3) / 2
        for m in range(cols):
            if k % 2 == 0:
                x = spacing / 2 + m * spacing
            else:
                x = spacing / 2 + m * spacing + spacing / 2
            if 0 <= x <= 1 and 0 <= y <= 1:
                grid.append([x, y])
    grid = np.array(grid[:n_circles])
    np.random.seed(0)
    grid += np.random.uniform(-0.001, 0.001, grid.shape)
    return grid

def compute_forces(centers, radii, k_wall=10.0, k_pair=5.0, gamma=1.0):
    n = centers.shape[0]
    forces = np.zeros_like(centers)
    for i in range(n):
        # Wall repulsion
        dL = centers[i, 0] - radii[i]
        dR = 1 - centers[i, 0] - radii[i]
        dB = centers[i, 1] - radii[i]
        dT = 1 - centers[i, 1] - radii[i]
        forces[i, 0] += k_wall * max(-dL, 0) * (-1.0)
        forces[i, 0] += k_wall * max(-dR, 0) * (1.0)
        forces[i, 1] += k_wall * max(-dB, 0) * (-1.0)
        forces[i, 1] += k_wall * max(-dT, 0) * (1.0)
        # Pairwise repulsion
        for j in range(n):
            if i == j:
                continue
            diff = centers[i] - centers[j]
            dist = np.sqrt(np.sum(diff**2))
            gap = dist - (radii[i] + radii[j])
            if gap < 0:
                forces[i] += k_pair * (-gap) * diff / (dist + 1e-12)
    return forces

def grow_radii(centers, radii, forces, alpha=1e-3):
    n = centers.shape[0]
    dot_r = np.ones(n) * alpha
    for i in range(n):
        # Check if touching walls
        dL = centers[i, 0] - radii[i]
        dR = 1 - centers[i, 0] - radii[i]
        dB = centers[i, 1] - radii[i]
        dT = 1 - centers[i, 1] - radii[i]
        touching = (dL < 0) or (dR < 0) or (dB < 0) or (dT < 0)
        if touching:
            dot_r[i] = 0
        # Check pairwise overlaps
        for j in range(n):
            if i == j:
                continue
            diff = centers[i] - centers[j]
            dist = np.sqrt(np.sum(diff**2))
            gap = dist - (radii[i] + radii[j])
            if gap < 0:
                dot_r[i] = 0
                break
    return dot_r

def max_radii_fixed_centers(centers, radii, max_iter=30, tol=1e-9):
    n = centers.shape[0]
    radii = np.array(radii)
    for _ in range(max_iter):
        new_radii = np.zeros_like(radii)
        for i in range(n):
            r = radii[i]
            x, y = centers[i]
            # Boundaries
            r_max = min(x, 1 - x, y, 1 - y)
            # Neighbors
            distances = np.sqrt(np.sum((centers - centers[i])**2, axis=1))
            gaps = distances - radii
            gaps[i] = np.inf
            if len(gaps) > 0:
                neighbor_r = np.min(gaps)
                if neighbor_r > 0:
                    r_max = min(r_max, neighbor_r)
            new_radii[i] = max(0, r_max)
        if np.allclose(radii, new_radii, atol=tol):
            break
        radii = new_radii
    return radii

def objective_sum_radii(centers_flat, centers, original_radii):
    n = len(centers)
    current_centers = centers_flat.reshape(n, 2)
    current_radii = max_radii_fixed_centers(current_centers, original_radii)
    return -np.sum(current_radii)

def run_packing():
    # Initial setup
    centers = init_grid(26)
    radii = np.full(26, 0.001)
    # Physics simulation
    dt = 0.02
    gamma = 1.0
    alpha = 1e-3
    v = np.zeros_like(centers)
    for step in range(5000):
        forces = compute_forces(centers, radii)
        v += dt * (forces - gamma * v)
        centers += dt * v
        centers = np.clip(centers, 0, 1)
        dot_r = grow_radii(centers, radii, forces, alpha)
        radii += dt * dot_r
        if np.all(dot_r < 1e-10):
            break
    # Refine with exact radii
    radii = max_radii_fixed_centers(centers, radii)
    # Global refinement
    initial_flat = centers.flatten()
    result = minimize(
        objective_sum_radii,
        initial_flat,
        args=(centers, radii),
        method='L-BFGS-B',
        options={'maxiter': 2000, 'gtol': 1e-9}
    )
    final_centers = result.x.reshape(-1, 2)
    final_radii = max_radii_fixed_centers(final_centers, radii)
    sum_radii = np.sum(final_radii)
    # Validate
    validate_packing(final_centers, final_radii)
    return final_centers, final_radii, sum_radii
