import numpy as np

def build_hex_lattice(d):
    # Generate a hexagonal lattice that fits at least d points in [0,1] x [0,1]
    # Spacing should be such that we get at least d points
    spacing = 1.0 / ceil(sqrt(d / (sqrt(3)/2)))
    rows = int(1.0 / spacing) + 1
    cols = int(1.0 / spacing) + 1
    centers = []
    for i in range(rows):
        y = i * spacing
        if y > 1.0:
            continue
        offset = (i % 2) * spacing / 2
        for j in range(cols):
            x = j * spacing + offset
            if x > 1.0:
                continue
            centers.append([x, y])
            if len(centers) >= d:
                break
        if len(centers) >= d:
            break
    return np.array(centers[:d])

def lp_max_radii(centers):
    n = centers.shape[0]
    # Build constraints for LP:
    # Each r_i <= min(x_i, 1-x_i, y_i, 1-y_i) for wall constraints
    # r_i + r_j <= d_ij for non-overlap constraints

    # Wall constraints matrix A_ub and vector b_ub
    A_ub = []
    b_ub = []
    
    # For each circle i, r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
    for i in range(n):
        x, y = centers[i]
        rmax = min(x, 1-x, y, 1-y)
        A_ub.append([-1.0])  # -r_i <= -rmax
        b_ub.append(-rmax)

    # Non-overlap constraints: r_i + r_j <= d_ij
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            d = sqrt(dx*dx + dy*dy)
            A_ub.append([1.0, 1.0])  # r_i + r_j <= d_ij
            b_ub.append(d)
    
    # Variables: [r1, r2, ..., r_n]
    # Objective: maximize sum of r_i -> minimize -sum of r_i
    c = np.array([-1.0] * n)  # -1 for each variable in objective

    result = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'presolve': True, 'tol': 1e-9})

    if not result.success:
        # If infeasible due to numerical issues, return trivial case or small radii
        return np.zeros(n), 0.0
    
    radii = result.x
    if np.isnan(radii).any():
        # fallback case - all zero or some small value
        return np.zeros(n), 0.0
    
    sum_radii = np.sum(radii)
    return radii, sum_radii

def perturb_centers(centers, sigma):
    new_centers = centers + np.random.normal(0, sigma, centers.shape)
    # Clamp to [0,1] boundaries
    new_centers = np.clip(new_centers, 0, 1)
    return new_centers

        (0.005, 0.015, 3000),
        (0.002, 0.008, 2000)
    ]
    
    for T, sigma, n_iter in temperature_schedule:
        for _ in range(n_iter):
            new_centers = perturb_centers(best_centers, sigma)

                best_radii = new_radii
