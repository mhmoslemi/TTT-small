def init_hex_grid(n=26, r_init=0.04):
    # Generate hexagonal grid points
    h = np.sqrt(3) * r_init
    rows = []
    y = r_init
    row_num = 0
    
    while len(rows) < n:
        row = []
        x = r_init if row_num % 2 == 0 else 2 * r_init
        while x <= 1 - r_init and len(rows) < n:
            row.append([x, y])
            x += 2 * r_init
        if row:
            rows.extend(row)
        y += h
        row_num += 1
    
    # Keep only first n points
    centers = np.array(rows[:n])
    radii = np.full(n, r_init)
    return centers, radii

def max_radii(centers, radii):
    max_iter = 100
    tol = 1e-8
    n = centers.shape[0]
    
    for _ in range(max_iter):
        # Compute wall limits
        wall = np.minimum(centers, 1 - centers).min(axis=1)
        
        # Compute neighbour limits
        D = np.sqrt(((centers[:, None, :] - centers[None, :, :]) ** 2).sum(-1))
        np.fill_diagonal(D, np.inf)
        neighbor = D - radii[:, None]
        neighbor = np.maximum(neighbor, 0)
        min_neighbor = np.min(neighbor, axis=1)
        
        # New radii
        new_radii = np.minimum(wall, min_neighbor)
        new_radii = np.clip(new_radii, 0, None)
        
        if np.max(np.abs(new_radii - radii)) < tol:
            break
        radii = new_radii
    
    return radii
