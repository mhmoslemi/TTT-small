import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Constants
    n_circles = 26
    max_iterations = 30
    epsilon = 1e-6
    polygon_sides = 8
    inflate_factor = 0.4

    # Generate hexagonal seed layout
    # Scale to roughly fit 80% of the unit square
    sqrt_n = np.sqrt(n_circles)
    spacing = 0.8 / np.ceil(sqrt_n)
    sqrt3 = np.sqrt(3)

    centers_seed = []
    for i in range(int(np.ceil(1.0 / spacing)) + 1):
        for j in range(int(np.ceil(1.0 / (spacing * sqrt3 / 2))) + 1):
            x = i * spacing + (j % 2) * spacing / 2
            y = j * spacing * sqrt3 / 2
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers_seed.append([x, y])
                if len(centers_seed) == n_circles:
                    break
        if len(centers_seed) == n_circles:
            break

    centers_seed = np.array(centers_seed)
    margin = 0.05
    centers_seed[:, 0] += margin
    centers_seed[:, 1] += margin

    # Initial radii
    radii = np.full(n_circles, 0.01)

    # Precompute direction vectors for polygonal approximation
    theta = np.linspace(0, 2*np.pi, polygon_sides, endpoint=False)
    u_vectors = np.vstack([np.cos(theta), np.sin(theta)]).T  # shape (polygon_sides, 2)
    cos_pi_k = np.cos(np.pi / polygon_sides)

    def solve_radius_lp(centers, radii):
        # For fixed centers, find optimal radii
        # Objective: maximize sum of radii
        # Constraints:
        # 1. Wall constraints: r_i <= min(c_i.x, 1 - c_i.x, c_i.y, 1 - c_i.y)
        # 2. Separation constraints: r_i + r_j <= dist(c_i, c_j) for all j

        n = len(centers)
        # Build constraint matrix and vector
        A_ub = []
        b_ub = []

        # Wall constraints
        for i in range(n):
            x, y = centers[i]
            # Min of 4 wall distances
            r_max = min(x, 1 - x, y, 1 - y)
            # Add constraint: r_i <= r_max => -r_i <= r_max
            A_ub.append([-1.0])
            b_ub.append(r_max)

        # Separation constraints
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                # r_i + r_j <= dist => -r_i - r_j <= -dist
                A_ub.append([-1.0, -1.0])
                b_ub.append(-dist)

        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        c = np.ones(n)
        # Solve the linear program using dual simplex
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'tol': 1e-10})
        if not res.success:
            # Take the best possible result, or fallback to the center radii
            opt_radii = np.minimum(
                np.minimum(centers[:, 0], 1 - centers[:, 0]),
                np.minimum(centers[:, 1], 1 - centers[:, 1])
            )
            for i in range(n):
                for j in range(i + 1, n):
                    dist_ij = np.linalg.norm(centers[i] - centers[j])
                    effective_r = dist_ij - radii[j]
                    if effective_r >= 0:
                        opt_radii[i] = min(opt_radii[i], effective_r)
            return np.clip(opt_radii, 0, 0.5)
        return np.clip(res.x, 0, 0.5)

    def solve_center_lp(centers, radii):
        # For fixed radii, find optimal centers to increase sum of radii (via slack)
        # Objective: maximize sum of slack variables
        # We want to maximize sum of minimum distances to walls and neighbors
        n = len(centers)
        # Create a consistent polytope for feasible centers
        # Each center has 4 wall constraints and 26*(26-1)/2 neighbor constraints
        # Total constraints expected: 4*n + (n*(n-1)/2)*polygon_sides
        # For now, we focus on wall constraints and neighbor proxy

        # Define slack variables
        # s_i = minimum of the 4 wall distances
        # We set up constraints on the center:
        # c_i.x - r_i >= s_i -> -c_i.x + s_i <= - r_i
        # c_i.y - r_i >= s_i -> -c_i.y + s_i <= - r_i
        # 1 - c_i.x - r_i >= s_i -> c_i.x + s_i <= 1 - r_i
        # 1 - c_i.y - r_i >= s_i -> c_i.y + s_i <= 1 - r_i

        num_slack = n
        num_vars = 2 * n + num_slack

        A_eq = []
        b_eq = []

        # Coefficient for each variable
        # Let centers be [c1_x, c1_y, ..., cn_x, cn_y, s1, ..., sn]
        # So each circle has variables (c_i_x, c_i_y, s_i)
        # But for ease of programming, we build the variable vector as [x1,y1,x2,y2,..., sn]

        dim = 2 * n + num_slack
        # Constraint matrix will be large and dense
        # We build it in parts

        # Wall constraints for each circle
        # idx = 0..n-1
        # For circle x_i, y_i, s_i
        #  c_i.x - r_i >= s_i -> -c_i.x + s_i <= - r_i
        #  c_i.y - r_i >= s_i -> -c_i.y + s_i <= - r_i
        #  1 - c_i.x - r_i >= s_i -> c_i.x + s_i <= 1 - r_i
        #  1 - c_i.y - r_i >= s_i -> c_i.y + s_i <= 1 - r_i
        # Build A_ub_mat
        num_constraints = 4 * n

        A_ub_mat = np.zeros((num_constraints, dim))
        b_ub_vec = np.zeros(num_constraints)

        for i in range(n):
            # s_i coefficients
            A_ub_mat[4*i + 0, 2*n + i] = 1.0
            A_ub_mat[4*i + 1, 2*n + i] = 1.0
            A_ub_mat[4*i + 2, 2*n + i] = 1.0
            A_ub_mat[4*i + 3, 2*n + i] = 1.0
            # c_i.x coefficients
            A_ub_mat[4*i + 0, 2*i + 0] = -1.0
            A_ub_mat[4*i + 1, 2*i + 0] = -1.0
            A_ub_mat[4*i + 2, 2*i + 0] = 1.0
            A_ub_mat[4*i + 3, 2*i + 0] = 1.0
            # c_i.y coefficients
            A_ub_mat[4*i + 0, 2*i + 1] = -1.0
            A_ub_mat[4*i + 1, 2*i + 1] = -1.0
            A_ub_mat[4*i + 2, 2*i + 1] = 1.0
            A_ub_mat[4*i + 3, 2*i + 1] = 1.0
            # Right-hand side
            b_ub_vec[4*i + 0] = -radii[i]           # -c_i.x + s_i <= -r_i
            b_ub_vec[4*i + 1] = -radii[i]           # -c_i.y + s_i <= -r_i
            b_ub_vec[4*i + 2] = 1 - radii[i]        # c_i.x + s_i <= 1 - r_i
            b_ub_vec[4*i + 3] = 1 - radii[i]        # c_i.y + s_i <= 1 - r_i

        # Neighbor separation constraints
        # Form = u_l · (c_i - c_j) + (r_i + r_j)*cos(π/8) >= s_i + s_j
        # => -u_l·c_i + u_l·c_j + (r_i + r_j)*cos(π/8) >= s_i + s_j
        # => -u_l·c_i + u_l·c_j + (r_i + r_j)*cos(π/8) - s_i - s_j >= 0
        # => u_l·c_i - u_l·c_j - (r_i + r_j)*cos(π/8) + s_i + s_j <= 0
        # => -u_l·c_i + u_l·c_j - (r_i + r_j)*cos(π/8) + s_i + s_j <= 0
        num_neigh_constraints = n * (n - 1) // 2 * polygon_sides
        start_idx = num_constraints
        A_ub_mat = np.resize(A_ub_mat, (start_idx + num_neigh_constraints, dim))
        b_ub_vec = np.resize(b_ub_vec, start_idx + num_neigh_constraints)

        k = start_idx
        for i in range(n):
            for j in range(i + 1, n):
                # We iterate through all directions of polygon
                for l in range(polygon_sides):
                    u = u_vectors[l]
                    A_ub_mat[k, 2*i + 0] = -u[0]
                    A_ub_mat[k, 2*i + 1] = -u[1]
                    A_ub_mat[k, 2*j + 0] = u[0]
                    A_ub_mat[k, 2*j + 1] = u[1]
                    A_ub_mat[k, 2*n + i] = 1.0
                    A_ub_mat[k, 2*n + j] = 1.0
                    b_ub_vec[k] = - (radii[i] + radii[j]) * cos_pi_k
                    k += 1

        # Objective: maximize sum of slacks
        c = np.zeros(dim)
        c[2*n:] = 1.0  # maximize sum of slack variables

        # Call linprog with method=highs
        try:
            res = linprog(-c, A_ub=A_ub_mat, b_ub=b_ub_vec, method='highs', options={'tol': 1e-10})
        except:
            # fallback to block coordinates with margin
            centers_new = centers.copy()
            # Just move centers to the center of their feasible region
            # Each center is bounded by its radius and walls
            for i in range(n):
                x, y = centers[i]
                r = radii[i]
                # Simple "room to safely move" approach:
                new_x = max(r, min(1 - r, x))
                new_y = max(r, min(1 - r, y))
                centers_new[i, :] = [new_x, new_y]
            return centers_new

        if not res.success:
            centers_new = centers.copy()
            for i in range(n):
                x, y = centers[i]
                r = radii[i]
                new_x = max(r, min(1 - r, x))
                new_y = max(r, min(1 - r, y))
                centers_new[i, :] = [new_x, new_y]
            return centers_new

        # Extract new centers from solution
        centers_new = np.zeros_like(centers)
        for i in range(n):
            centers_new[i, 0] = res.x[2*i]
            centers_new[i, 1] = res.x[2*i + 1]
        return centers_new

    # Alternate between solving for optimal radii and centers
    for iteration in range(max_iterations):
        old_sum = np.sum(radii)
        radii = solve_radius_lp(centers_seed, radii)
        centers_seed = solve_center_lp(centers_seed, radii)
        new_sum = np.sum(radii)
        if abs(new_sum - old_sum) < epsilon:
            break

    # Local inflation to recover any gap from polygonal approximation
    # After the alternating loop, some circles might have a gap that is underused
    for _ in range(5):
        improved = False
        for i in range(n_circles):
            # Exact distance to nearest neighbor
            dist_min = float('inf')
            for j in range(n_circles):
                if i != j:
                    dist_ij = np.linalg.norm(centers_seed[i] - centers_seed[j])
                    if dist_ij < dist_min:
                        dist_min = dist_ij
            # For walls
            own_r = radii[i]
            dist_to_wall = min(
                centers_seed[i, 0],
                1 - centers_seed[i, 0],
                centers_seed[i, 1],
                1 - centers_seed[i, 1]
            )
            if dist_min < float('inf'):
                gap = dist_min - 2 * own_r
                if gap > 1e-8:
                    inc = inflate_factor * gap
                    if inc > 1e-8:
                        radii[i] += inc
                        improved = True
        if not improved:
            break

    # Final adjustment to keep within bounds
    radii = np.clip(radii, 0, 0.5)
    centers_seed = np.clip(centers_seed, 0, 1)

    # Validate
    valid, msg = validate_packing(centers_seed, radii)
    if not valid:
        # Shrink slightly
        radii *= (1 - 1e-7)
        valid, msg = validate_packing(centers_seed, radii)
        if not valid:
            # Repeatedly shrink until valid
            for _ in range(5):
                radii *= (1 - 1e-7)
                valid, msg = validate_packing(centers_seed, radii)
                if valid:
                    break

    sum_radii = np.sum(radii)
    return centers_seed, radii, sum_radii
