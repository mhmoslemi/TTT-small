import numpy as np
from scipy.optimize import minimize

def make_hex_lattice(n, margin=0.05):
    # Generate hexagonal lattice points in the unit square
    dx = 1.0 / (np.ceil(np.sqrt(n)) + 0.5)
    dy = dx * np.sqrt(3) / 2

    points = []
    i = 0
    while len(points) < n:
        for j in range(int(np.ceil(n / 2)) + 5):
            x = i * dx + (j % 2) * dx / 2
            y = j * dy
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append([x, y])
            if len(points) >= n:
                break
        i += 1
    
    points = np.array(points[:n])
    
    # Center the pattern
    x_center = np.mean(points[:, 0])
    y_center = np.mean(points[:, 1])
    
    shift_x = 0.5 - x_center
    shift_y = 0.5 - y_center
    
    points[:, 0] += shift_x
    points[:, 1] += shift_y
    
    # Apply margin
    points[:, 0] = np.clip(points[:, 0], margin, 1 - margin)
    points[:, 1] = np.clip(points[:, 1], margin, 1 - margin)
    
    return points

def objective(z):
    # z is [x0, y0, r0, x1, y1, r1, ...]
    n = len(z) // 3
    radii = z[2::3]
    return -np.sum(radii)

def boundary_constraints(z):
    n = len(z) // 3
    constraints = []
    for i in range(n):
        x, y, r = z[3*i], z[3*i+1], z[3*i+2]
        # x - r >= 0
        constraints.append(x - r)
        # x + r <= 1
        constraints.append(1 - x - r)
        # y - r >= 0
        constraints.append(y - r)
        # y + r <= 1
        constraints.append(1 - y - r)
    return np.array(constraints)

def pairwise_constraints(z):
    n = len(z) // 3
    constraints = []
    for i in range(n):
        for j in range(i+1, n):
            x1, y1, r1 = z[3*i], z[3*i+1], z[3*i+2]
            x2, y2, r2 = z[3*j], z[3*j+1], z[3*j+2]
            dist_sq = (x1 - x2)**2 + (y1 - y2)**2
            sum_radii_sq = (r1 + r2)**2
            constraints.append(dist_sq - sum_radii_sq)
    return np.array(constraints)

def jacobian_boundary(z):
    n = len(z) // 3
    J = np.zeros((4*n, 3*n))
    for i in range(n):
        J[4*i, 3*i] = 1   # d/dx_i (x_i - r_i)
        J[4*i, 3*i+2] = -1  # d/dr_i (x_i - r_i)
        J[4*i+1, 3*i] = -1   # d/dx_i (1 - x_i - r_i)
        J[4*i+1, 3*i+2] = -1  # d/dr_i (1 - x_i - r_i)
        J[4*i+2, 3*i+1] = 1   # d/dy_i (y_i - r_i)
        J[4*i+2, 3*i+2] = -1  # d/dr_i (y_i - r_i)
        J[4*i+3, 3*i+1] = -1   # d/dy_i (1 - y_i - r_i)
        J[4*i+3, 3*i+2] = -1  # d/dr_i (1 - y_i - r_i)
    return J

def jacobian_pairwise(z):
    n = len(z) // 3
    J = np.zeros((n*(n-1)//2, 3*n))
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            x1, y1, r1 = z[3*i], z[3*i+1], z[3*i+2]
            x2, y2, r2 = z[3*j], z[3*j+1], z[3*j+2]
            
            # Gradient of (x1-x2)**2 + (y1-y2)**2 - (r1+r2)**2
            # wrt z[3*i] = x1
            J[idx, 3*i] = 2*(x1 - x2)
            # wrt z[3*i+1] = y1
            J[idx, 3*i+1] = 2*(y1 - y2)
            # wrt z[3*i+2] = r1
            J[idx, 3*i+2] = -2*(r1 + r2)
            # wrt z[3*j] = x2
            J[idx, 3*j] = -2*(x1 - x2)
            # wrt z[3*j+1] = y2
            J[idx, 3*j+1] = -2*(y1 - y2)
            # wrt z[3*j+2] = r2
            J[idx, 3*j+2] = -2*(r1 + r2)
            idx += 1
    return J

def solve_nlp(initial_z):
    n = len(initial_z) // 3
    
    cons = []
    cons.append({'type': 'ineq', 'fun': boundary_constraints, 'jac': jacobian_boundary})
    cons.append({'type': 'ineq', 'fun': pairwise_constraints, 'jac': jacobian_pairwise})
    
    bounds = []
    for i in range(n):
        bounds.append((0, 1))  # x bounds
        bounds.append((0, 1))  # y bounds
        bounds.append((0, 0.5))  # r bounds (saturation constraint)
    
    res = minimize(objective, initial_z, method='SLSQP', bounds=bounds, constraints=cons, 
                   options={'maxiter': 1000, 'ftol': 1e-9}, tol=1e-9)
    
    return res.x if res.success else initial_z

def nudge_centers(centers, radii):
    # Find circles that are tight to a wall
    margin = 1e-6
    nudged_centers = centers.copy()

    for i in range(len(centers)):
        x, y = centers[i]
        r = radii[i]
        
        # Determine which walls are tight
        tight_walls = []
        if abs(x - r) < margin:
            tight_walls.append('left')
        if abs(x + r - 1) < margin:
            tight_walls.append('right')
        if abs(y - r) < margin:
            tight_walls.append('bottom')
        if abs(y + r - 1) < margin:
            tight_walls.append('top')

        # Perturb slightly in normal direction
        if tight_walls:
            # Compute normal vector to the wall (for now, simple nudges)
            shift = 0.001
            if 'left' in tight_walls:
                nudged_centers[i, 0] += shift
            if 'right' in tight_walls:
                nudged_centers[i, 0] -= shift
            if 'bottom' in tight_walls:
                nudged_centers[i, 1] += shift
            if 'top' in tight_walls:
                nudged_centers[i, 1] -= shift
                
    return nudged_centers
