```python
import numpy as np
from scipy.optimize import minimize

NUM_CIRCLES = 26
NUM_RESTARTS = 5
HEX_ROWS = 5
HEX_COLS = 5
HEX_SPACING = 0.2
HEX_RADIUS = 0.02

def hex_lattice_points(n_rows, spacing):
    points = []
    for i in range(n_rows):
        for j in range(n_rows):
            x = (j + 0.5 * (i % 2)) * spacing
            y = i * spacing * np.sqrt(3) / 2
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append([x, y])
    return np.array(points)

def initial_guess(seed):
    np.random.seed(seed)
    points = hex_lattice_points(HEX_ROWS, HEX_SPACING)
    points = points[:NUM_CIRCLES]
    if len(points) < NUM_CIRCLES:
        # Pad with random points if not enough hex points
        extra_points = np.random.rand(NUM_CIRCLES - len(points), 2)
        points = np.vstack([points, extra_points])
    
    # Add jitter
    jitter = np.random.uniform(-0.02, 0.02, (NUM_CIRCLES, 2))
    points += jitter
    points = np.clip(points, 0, 1)
    
    vars0 = np.zeros(NUM_CIRCLES * 3)
    vars0[0::3] = points[:, 0]    # x
    vars0[1::3] = points[:, 1]    # y
    vars0[2::3] = 0.02            # r (initially all equal)
    
    return vars0

def objective(vars):
    return -np.sum(vars[2::3])

def constraint_boundary(vars):
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    # x - r, 1 - x - r, y - r, 1 - y - r
    return np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])

def constraint_nonoverlap(vars):
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    n = len(x)
    margins = []
    for i in range(n):
        for j in range(i + 1, n):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dist = np.sqrt(dx * dx + dy * dy)
            combined_radius = r[i] + r[j]
            margins.append(dist - combined_radius)
    return np.array(margins)

def postprocess(vars):
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    for i in range(len(r)):
        # Ensure circle is within bounds
        x[i] = np.clip(x[i], r[i], 1.0 - r[i])
        y[i] = np.clip(y[i], r[i], 1.0 - r[i])
        
    # Refine overlapping constraints slightly
    # This might be needed due to numerical imprecision from SLSQP
    for i in range(len(r)):
        for j in range(i+1, len(r)):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dist = np.sqrt(dx*dx + dy*dy)
            combined_radius = r[i] + r[j]
            if dist < combined_radius:
                diff = combined_radius - dist
                # Reduce radii proportionally if needed
                r[i] = max(1e-8, r[i] - diff * 0.5)
                r[j] = max(1e-8, r[j] - diff * 0.5)
                
    # Reconstruct vars
    vars_new = np.zeros_like(vars)
    vars_new[0::3] = x
    vars_new[1::3] = y
    vars_new[2::3] = r
    
    return vars_new

def run_packing():
    best_sum = -np.inf
    best_vars = None
    
    constraints = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_nonoverlap}
    ]
    
    bounds = [(0, 1), (0, 1), (0, 0.5)] * NUM_CIRCLES
    
    for seed in range(NUM_RESTARTS):
        vars0 = initial_guess(seed)
        res = minimize(
            objective,
            vars0,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 3000, 'ftol': 1e-7}
        )
        
        if not res.success:
            continue
            
        vars_corr = postprocess(res.x)
        s = np.sum(vars_corr[2::3])
        
        if s > best_sum:
            best_sum = s
            best_vars = vars_corr
    
    if best_vars is None:
        raise RuntimeError("Failed to find a valid configuration.")
    
    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii = best_vars[2::3]
    return centers, radii, float(best_sum)
```
