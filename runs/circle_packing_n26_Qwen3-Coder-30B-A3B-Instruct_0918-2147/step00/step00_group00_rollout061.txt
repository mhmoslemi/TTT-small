import numpy as np
import random

def make_hex_seed(n, margin):
    # Create a hexagonal arrangement with different initial radii based on distance from walls
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))
    
    centers = []
    radii = []
    
    # Generate grid points
    x_positions = np.linspace(margin, 1 - margin, cols)
    y_positions = np.linspace(margin, 1 - margin, rows)
    
    for i, y in enumerate(y_positions):
        offset = 0 if i % 2 == 0 else 0.5 * (x_positions[1] - x_positions[0])
        for j, x in enumerate(x_positions):
            if len(centers) >= n:
                break
            centers.append([x + offset, y])
    
    if len(centers) < n:
        # Add more points to reach n centers
        x_positions = np.linspace(margin, 1 - margin, cols + 1)
        y_positions = np.linspace(margin, 1 - margin, rows + 1)
        for i, y in enumerate(y_positions):
            offset = 0 if i % 2 == 0 else 0.5 * (x_positions[1] - x_positions[0])
            for j, x in enumerate(x_positions):
                if len(centers) >= n:
                    break
                centers.append([x + offset, y])

    centers = np.array(centers[:n])
    
    # Assign radii based on distance from walls and center
    wall_dist = np.minimum(centers[:, 0], np.minimum(1 - centers[:, 0], 
                                                     np.minimum(centers[:, 1], 1 - centers[:, 1])))
    center_dist = np.sqrt(np.sum((centers - [0.5, 0.5])**2, axis=1))
    radii = 0.015 + 0.02 * (1 - center_dist / np.sqrt(0.5)) * (wall_dist > 0.015)
    
    # Normalize so that the smallest radius is 0.015
    min_radius = np.min(radii)
    if min_radius < 0.015:
        scale_factor = 0.015 / min_radius
        radii *= scale_factor
    
    return centers, radii

def max_feasible_radius(i, centers, radii):
    # Calculate maximum possible radius for circle i
    x, y = centers[i]
    r = radii[i]
    
    # Distance to walls
    d_L = x
    d_R = 1 - x
    d_B = y
    d_T = 1 - y
    
    # Distances to other circles
    min_dist = float('inf')
    
    for j in range(len(centers)):
        if i != j:
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            distance = np.sqrt(dx * dx + dy * dy)
            min_dist = min(min_dist, distance - radii[j])
    
    # Return the minimum of all constraints
    return min(d_L, d_R, d_B, d_T, min_dist)

def inflate_radii(centers, radii, alpha):
    old_sum = np.sum(radii)
    new_radii = radii + alpha * np.mean(radii)
    
    # Adjust each radius to be feasible
    for i in range(len(centers)):
        max_r = max_feasible_radius(i, centers, new_radii)
        new_radii[i] = min(new_radii[i], max_r)
    
    new_sum = np.sum(new_radii)
    return new_radii, new_sum - old_sum

def anneal_centers(centers, radii, temp0, gamma, sigma0, steps):
    current_centers = centers.copy()
    current_radii = radii.copy()
    best_sum = np.sum(current_radii)
    best_centers = current_centers.copy()
    best_radii = current_radii.copy()
    
    temp = temp0
    sigma = sigma0
    
    # Approximate number of steps without improvement
    no_improvement_steps = 0
    
    for step in range(steps):
        i = random.randint(0, len(current_centers) - 1)
        
        # Normal displacement with the current sigma
        displacement = np.random.normal(0, sigma, 2)
        new_center = current_centers[i] + displacement
        
        # Ensure new_center is within bounds
        new_center = np.clip(new_center, 0, 1)
        
        # Compute new feasible radius for moved circle
        old_radius = current_radii[i]
        current_centers[i] = new_center
        new_radius = max_feasible_radius(i, current_centers, current_radii)
        
        # Accept based on probability if it doesn't increase radius
        delta_r = new_radius - old_radius
        if delta_r > 0:
            current_radii[i] = new_radius
            current_sum = np.sum(current_radii)
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = current_centers.copy()
                best_radii = current_radii.copy()
            no_improvement_steps = 0
        else:
            if random.random() < np.exp(delta_r / temp):
                current_radii[i] = new_radius
                current_sum = np.sum(current_radii)
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = current_centers.copy()
                    best_radii = current_radii.copy()
                no_improvement_steps = 0
            else:
                current_centers[i] = centers[i]
                no_improvement_steps += 1
        
        if step % 20 == 0:
            current_centers = np.clip(current_centers, 0, 1)
            
        if step % 50 == 0:
            # Global shift
            shift = np.random.normal(0, 0.005, 2)
            current_centers += shift
            current_centers = np.clip(current_centers, 0, 1)
        
        # Update normalization
        temp *= gamma
        sigma = sigma0 * temp / temp0
        
        if no_improvement_steps > 100:
            break
    
    return best_centers, best_radii, best_sum

def mixed_cycle(centers, radii, params):
    alpha = params['alpha']
    epsilon = params['epsilon']
    max_iter = params['max_iter']
    num_sa_steps = params['sa_steps']
    sa_temp = params['sa_temp']
    sa_gamma = params['sa_gamma']
    sa_sigma = params['sa_sigma']
    convergence_stagnation = params['convergence_stagnation']
    
    last_sum = np.sum(radii)
    stable_count = 0
    
    for iteration in range(max_iter):
        # Inflation phase
        new_radii, delta = inflate_radii(centers, radii, alpha)
        
        if delta < epsilon:
            stable_count += 1
        else:
            stable_count = 0
        
        radii = new_radii
        
        # If we're stable for enough iterations, move to SA phase
        if stable_count > convergence_stagnation:
            # Perform one round of simulated annealing
            centers, radii, new_sum = anneal_centers(
                centers, radii, sa_temp, sa_gamma, sa_sigma, num_sa_steps
            )
            
            if new_sum <= last_sum + epsilon:
                break
            last_sum = new_sum
            stable_count = 0
            
            # Reset inflation for next pass
            alpha *= 0.95
            
    return centers, radii

def polish_radii(centers, radii, beta, passes):
    # Performing multiple rounds of exact local improvements
    for _ in range(passes):
        # Calculate for each circle:
        # - how much it can expand due to wall distance
        # - how much due to neighbours
        # then inflate by a factor of beta and recheck
        new_radii = radii.copy()
        for i in range(len(centers)):
            x, y = centers[i]
            r = radii[i]
            
            # Compute gap to walls
            g_wall = min(x, 1 - x, y, 1 - y)
            
            # Compute gap to other circles
            g_nei = float('inf')
            for j in range(len(centers)):
                if i != j:
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    distance = np.sqrt(dx*dx + dy*dy)
                    gap = distance - (r + radii[j])
                    g_nei = min(g_nei, gap)
            
            g = min(g_wall, g_nei) if g_nei != float('inf') else g_wall
            
            r_new = min(r + beta * g, max_feasible_radius(i, centers, new_radii))
            new_radii[i] = r_new
            
        # Handle overlaps by shrinking
        radii = new_radii
        # Sweeping process to avoid overlaps
        for _ in range(3):
            something_shrunk = False
            for i in range(len(centers)):
                for j in range(i + 1, len(centers)):
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    distance = np.sqrt(dx*dx + dy*dy)
                    total_r = radii[i] + radii[j]
                    if distance < total_r:
                        overlap = total_r - distance
                        radii[i] -= overlap / 2
                        radii[j] -= overlap / 2
                        something_shrunk = True
            
            if not something_shrunk:
                break
    
    return radii

def run_packing():
    # Seed parameters
    margin = 0.05
    n = 26
    
    # Initialize seed
    centers, radii = make_hex_seed(n, margin)
    
    # Set up function parameters
    params = {
        'alpha': 0.04,
        'epsilon': 1e-6,
        'max_iter': 100,
        'sa_steps': 200,
        'sa_temp': 0.02,
        'sa_gamma': 0.97,
        'sa_sigma': 0.03,
        'convergence_stagnation': 8
    }
    
    # Run mixed cycle
    centers, radii = mixed_cycle(centers, radii, params)
    
    # Final polishing
    radii = polish_radii(centers, radii, 0.6, 3)

    if not valid:
