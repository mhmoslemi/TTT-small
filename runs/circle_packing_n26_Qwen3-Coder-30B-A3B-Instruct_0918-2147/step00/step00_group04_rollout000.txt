```python
import numpy as np
from scipy.optimize import minimize

NUM_CIRCLES = 26
NUM_RESTARTS = 5
BOUNDARY_TOLERANCE = 1e-9
OVERLAP_TOLERANCE = 1e-9

def hex_lattice_points(n_rows, spacing):
    """Generate hexagonal lattice points within the unit square."""
    points = []
    for i in range(n_rows):
        y = i * spacing * np.sqrt(3) / 2
        if y > 1:
            break
        for j in range(int(1.0 / spacing) + 1):
            x = j * spacing + (i % 2) * spacing / 2
            if x > 1:
                break
            points.append([x, y])
    return np.array(points)

def initial_guess(seed):
    """Create initial variable vector based on a coarse hexagonal lattice with jitter."""
    np.random.seed(seed)
    spacing = 0.2
    points = hex_lattice_points(10, spacing)
    n_points = min(len(points), NUM_CIRCLES)
    points = points[:n_points]
    
    # Add jitter
    jitter = np.random.uniform(-0.02, 0.02, (n_points, 2))
    points = points + jitter
    points = np.clip(points, 0, 1)
    
    # Initialize radii
    radii = np.full(n_points, 0.02)
    
    # Flatten into variable vector
    vars_ = []
    for i in range(n_points):
        vars_.append(points[i][0])
        vars_.append(points[i][1])
        vars_.append(radii[i])
    
    return np.array(vars_)

def objective(vars_):
    """Minimize negative sum of radii."""
    return -np.sum(vars_[2::3])

def constraint_boundary(vars_):
    """Return array of boundary margins for all circles."""
    x = vars_[0::3]
    y = vars_[1::3]
    r = vars_[2::3]
    
    # Each circle must satisfy: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    margin_x_min = x - r
    margin_x_max = 1 - x - r
    margin_y_min = y - r
    margin_y_max = 1 - y - r
    
    return np.concatenate([margin_x_min, margin_x_max, margin_y_min, margin_y_max])

def constraint_nonoverlap(vars_):
    """Return array of non-overlap margins for all pairs of circles."""
    n = NUM_CIRCLES
    x = vars_[0::3]
    y = vars_[1::3]
    r = vars_[2::3]
    margins = []
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dist_sq = dx*dx + dy*dy
            dist = np.sqrt(dist_sq)
            margin = dist - (r[i] + r[j])
            margins.append(margin)
    
    return np.array(margins)

def postprocess(vars_):
    """Post-process for minor constraint violations."""
    # Copy for safety
    vars_copy = vars_.copy()
    x = vars_copy[0::3]
    y = vars_copy[1::3]
    r = vars_copy[2::3]
    
    # Fix boundary constraints with very small margin
    for i in range(NUM_CIRCLES):
        if x[i] - r[i] < -BOUNDARY_TOLERANCE:
            x[i] = r[i] + 1e-10
        if 1 - x[i] - r[i] < -BOUNDARY_TOLERANCE:
            x[i] = 1 - r[i] - 1e-10
        if y[i] - r[i] < -BOUNDARY_TOLERANCE:
            y[i] = r[i] + 1e-10
        if 1 - y[i] - r[i] < -BOUNDARY_TOLERANCE:
            y[i] = 1 - r[i] - 1e-10

    # Enforce non-overlap constraints again if needed
    for i in range(NUM_CIRCLES):
        for j in range(i + 1, NUM_CIRCLES):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dist_sq = dx*dx + dy*dy
            dist = np.sqrt(dist_sq)
            overlap = dist - (r[i] + r[j])
            if overlap < -OVERLAP_TOLERANCE:
                # Reduce the larger of r[i] or r[j]
                if r[i] > r[j]:
                    r[i] += overlap
                else:
                    r[j] += overlap

    return vars_copy

def run_packing():
    """Main function to pack 26 circles in the unit square to maximize sum of radii."""
    best_sum = -np.inf
    best_vars = None
    
    # Set bounds for variables
    bounds = []
    for i in range(NUM_CIRCLES):
        bounds.append((0., 1.))   # x
        bounds.append((0., 1.))   # y
        bounds.append((0., 0.5))  # r

    constraints = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_nonoverlap},
    ]
    
    for seed in range(NUM_RESTARTS):
        vars0 = initial_guess(seed)
        res = minimize(objective, vars0,
                       method='SLSQP',
                       bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': 3000, 'ftol': 1e-7})
        
        if not res.success:
            continue
            
        vars_corr = postprocess(res.x)
        s = np.sum(vars_corr[2::3])
        if s > best_sum:
            best_sum = s
            best_vars = vars_corr

    if best_vars is None:
        raise RuntimeError("Optimization failed to produce any valid solution.")
    
    # Reshape into centers and radii
    centers_x = best_vars[0::3]
    centers_y = best_vars[1::3]
    radii = best_vars[2::3]
    centers = np.column_stack((centers_x, centers_y))
    
    return centers, radii, float(best_sum)
```
