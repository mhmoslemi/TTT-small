import numpy as np
from scipy.optimize import linprog, minimize
from math import sqrt

def build_hex_lattice(d):
    # Generate a hexagonal lattice scaled to fit 26 points
    # Spacing between points based on desired number and area
    spacing = d / sqrt(3)  # ensures 26 points fit inside unit square
    rows = int(1 / spacing + 1)
    cols = int(1 / spacing + 1)
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = (j + 0.5 * (i % 2)) * spacing
            y = i * spacing * sqrt(3) / 2
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers.append([x, y])
                if len(centers) >= 26:
                    break
        if len(centers) >= 26:
            break
    # Trim to exactly 26 if needed
    while len(centers) > 26:
        centers.pop()
    # Ensure exactly 26
    if len(centers) < 26:
        raise ValueError("Did not generate enough centers")
    # Return shape (26, 2)
    return np.array(centers)

def lp_max_radii(centers):
    # Build LP constraints for maximal radii given fixed centers
    n = centers.shape[0]
    # Wall constraints: r <= min(x, 1-x, y, 1-y)
    A_ub = []
    b_ub = []
    # Non-overlap (pairwise) constraints: r_i + r_j <= dist_ij
    for i in range(n):
        for j in range(i+1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = sqrt(dx*dx + dy*dy)
            # Add constraint r_i + r_j <= dist
            row = [0]*n
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)
    # Wall constraints for each circle
    for i in range(n):
        x, y = centers[i]
        wall_inequalities = [
            (1.0, x),   # r <= x
            (1.0, 1-x), # r <= 1-x
            (1.0, y),   # r <= y
            (1.0, 1-y)  # r <= 1-y
        ]
        for a, b in wall_inequalities:
            row = [0]*n
            row[i] = a
            A_ub.append(row)
            b_ub.append(b)
    # Objective: maximize sum of radii
    obj_coeffs = [-1.0] * n  # negative because linprog minimizes
    # Solve LP
    res = linprog(
        c=obj_coeffs,
        A_ub=A_ub,
        b_ub=b_ub,
        method='highs',
        options={'presolve': True, 'disp': False}
    )
    if res.success:
        radii = -res.fun
        return -res.x, radii
    else:
        raise RuntimeError("LP solver failed")

def perturb_centers(centers, sigma):
    # Perturb each center independently with Gaussian noise
    noise = np.random.normal(0, sigma, size=centers.shape)
    perturbed = centers + noise
    # Clamp to [0, 1]
    perturbed = np.clip(perturbed, 0, 1)
    return perturbed

def anneal_centers(initial_centers):
    np.random.seed(42)
    current_centers = initial_centers.copy()
    current_radii, current_sum = lp_max_radii(current_centers)
    best_centers = current_centers.copy()
    best_radii = current_radii.copy()
    best_sum = current_sum
    # Annealing schedule
    stages = [
        {'temp': 0.02, 'sigma': 0.05, 'steps': 2000},
        {'temp': 0.01, 'sigma': 0.03, 'steps': 4000},
        {'temp': 0.005, 'sigma': 0.015, 'steps': 3000},
        {'temp': 0.002, 'sigma': 0.008, 'steps': 2000}
    ]
    for stage in stages:
        temp = stage['temp']
        sigma = stage['sigma']
        steps = stage['steps']
        for _ in range(steps):
            # Perturb centers
            new_centers = perturb_centers(current_centers, sigma)
            try:
                new_radii, new_sum = lp_max_radii(new_centers)
                if new_sum > current_sum:
                    current_centers = new_centers
                    current_radii = new_radii
                    current_sum = new_sum
                    if new_sum > best_sum:
                        best_centers = new_centers
                        best_radii = new_radii
                        best_sum = new_sum
                else:
                    # Accept with probability
                    prob = np.exp((new_sum - current_sum) / temp)
                    if np.random.rand() < prob:
                        current_centers = new_centers
                        current_radii = new_radii
                        current_sum = new_sum
            except Exception:
                # If LP fails, keep the old configuration
                continue
    return best_centers, best_radii, best_sum

def local_refine(centers, radii):
    # Local refinement using trust-constr minimization
    n = centers.shape[0]
    # Lower bound constraint on radii
    def constraint_lower_bound(i):
        def f(x):
            return radii[i] - x[n+i]
        return f
    # Overlap constraint between circles
    def constraint_overlap(i, j):
        def f(x):
            c_i = (x[i], x[n+i])
            c_j = (x[j], x[n+j])
            r_i = x[n+i]
            r_j = x[n+j]
            dist = sqrt((c_i[0]-c_j[0])**2 + (c_i[1]-c_j[1])**2)
            return dist - (r_i + r_j)
        return f
    # Wall constraints
    def constraint_wall(i):
        def f(x):
            c_i = (x[i], x[n+i])
            r_i = x[n+i]
            return min(c_i[0]-r_i, 1-c_i[0]-r_i, c_i[1]-r_i, 1-c_i[1]-r_i)
        return f
    # Build constraints list
    cons = []
    # Lower bounds on radii
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': constraint_lower_bound(i)})
    # Non-overlap constraints
    for i in range(n):
        for j in range(i+1, n):
            cons.append({'type': 'ineq', 'fun': constraint_overlap(i,j)})
    # Wall constraints
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': constraint_wall(i)})
    # Objective: minimize -sum of radii (equivalent to maximizing sum)
    def objective(x):
        # x consists of [x0,y0,x1,y1,...,r0,r1,...]
        return -np.sum(x[n:])
    # Initial values
    x0 = np.concatenate([centers.flatten(), radii])
    # Optimise
    res = minimize(
        objective,
        x0,
        method='trust-constr',
        constraints=cons,
        options={
            'disp': False,
            'maxiter': 1000
        }
    )
    if res.success:
        refined_centers = np.reshape(res.x[:2*n], (n, 2))
        refined_radii = res.x[2*n:]
        total_radius = -res.fun
        return refined_centers, refined_radii, total_radius
    else:
        # Return original if failed
        return centers, radii, np.sum(radii)
