import numpy as np
import cvxpy as cp
from scipy.spatial.distance import cdist

def build_leaf_problem(cell, k):
    # cell: [x_min, x_max, y_min, y_max]
    x_min, x_max, y_min, y_max = cell
    
    # Variables
    centers = cp.Variable((k, 2))
    radii = cp.Variable(k)
    
    # Constraints
    constraints = []
    
    # Boundary constraints
    for i in range(k):
        constraints.append(centers[i, 0] - radii[i] >= x_min)
        constraints.append(centers[i, 0] + radii[i] <= x_max)
        constraints.append(centers[i, 1] - radii[i] >= y_min)
        constraints.append(centers[i, 1] + radii[i] <= y_max)
        
    # Non-overlap constraints (SOC)
    for i in range(k):
        for j in range(i + 1, k):
            dist = cp.norm(centers[i] - centers[j])
            constraints.append(dist >= radii[i] + radii[j])
            
    # Objective: maximize sum of radii
    objective = cp.Maximize(cp.sum(radii))
    
    return cp.Problem(objective, constraints), centers, radii

def solve_leaf(cell, k):
    problem, centers, radii = build_leaf_problem(cell, k)
    problem.solve(solver=cp.ECOS, abstol=1e-12, reltol=1e-12)
    
    if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        centers_val = centers.value
        radii_val = radii.value
        return centers_val, radii_val
    else:
        # If LP is infeasible or unbounded, there's an issue
        raise RuntimeError(f"Leaf solver failed with status: {problem.status}")

def hierarchical_merge(clusters):

    a = 0.25
    # Corner cells
    corners = [

        [1-a, 1, 0, a],      # bottom-right
        [0, a, 1-a, 1],      # top-left

    ]
    # Edge cells
    edges = [
        [a, 1-a, 0, a],      # bottom
        [a, 1-a, 1-a, 1],    # top
        [0, a, a, 1-a],      # left
        [1-a, 1, a, 1-a]     # right
    ]
    # Center
    center = [a, 1-a, a, 1-a]
    
    # Number of circles in each
    distribution = [5, 5, 5, 5, 1, 1, 1, 1, 2]
    
    # Allocate circles manually to reduce complexity
    cell_geometry = corners + edges + [center]
    counts = distribution
    
    # Compute initial leaf solution
    leaves = []
    for i, (geo, count) in enumerate(zip(cell_geometry, counts)):
        if count > 0:
            centers, radii = solve_leaf(geo, count)
            leaves.append({
                "centers": centers,
                "radii": radii,
                "cell_idx": i
            })
    
    # Merge all leaves into one cluster
    if len(leaves) == 1:
        return leaves[0]
    else:
        # Simple greedy merge (concatenate, then re-cluster)
        # For simplicity, combine 2 leaves at a time
        current_leaves = leaves.copy()
        while len(current_leaves) > 1:
            # Pop two and merge
            a = current_leaves.pop()
            b = current_leaves.pop()
            merged_cluster = merge_clusters(a, b, None)
            current_leaves.append(merged_cluster)
        
        return current_leaves[0]

def global_polish(centers, radii):
    # Gradient descent style polishing with barrier terms
    # Keep radii and positions of all circles
    x = centers.flatten().copy()
    x = np.clip(x, 1e-10, 1 - 1e-10)  # Avoid division by zero
    r = radii.copy()
    r = np.clip(r, 1e-10, 0.5)  # To avoid overlaps
    
    mu = 1e-4
    
    def barrier_gradient(x, r):
        n = len(r)
        centers_mat = x.reshape((n, 2))
        
        # Compute all pairwise distances
        D = cdist(centers_mat, centers_mat)
        
        # Initialize gradients
        grad = np.zeros_like(x)
        obj = - np.sum(r)
        
        # Compute barrier terms
        for i in range(n):
            # Boundary barriers
            for j in range(2):  # x, y component
                if j == 0:
                    dist = centers_mat[i, j] - r[i]
                    obj += mu * np.log(max(dist, 1e-10))
                    grad[i*2 + j] -= mu / max(dist, 1e-10)
                
                if j == 1:
                    dist = centers_mat[i, j] - r[i]
                    obj += mu * np.log(max(dist, 1e-10))
                    grad[i*2 + j] -= mu / max(dist, 1e-10)
                
                # Right
                dist = 1 - centers_mat[i, j] - r[i]
                obj += mu * np.log(max(dist, 1e-10))
                grad[i*2 + j] += mu / max(dist, 1e-10)
                
        # Circle-circle barriers
        for i in range(n):
            for j in range(i+1, n):
                dist = D[i, j]
                if dist < r[i] + r[j] + 1e-10:
                    # Violation - no penalty
                    pass
                else:
                    dist = max(dist, 1e-10)
                    obj -= mu * np.log(dist - r[i] - r[j])
                    dr = 1 / (dist - r[i] - r[j])
                    # Gradient contributions
                    dx = (centers_mat[i, 0] - centers_mat[j, 0]) / (dist + 1e-10)
                    dy = (centers_mat[i, 1] - centers_mat[j, 1]) / (dist + 1e-10)
                    grad[i*2] += -dr * dx
                    grad[i*2 + 1] += -dr * dy
                    grad[j*2] += dr * dx
                    grad[j*2 + 1] += dr * dy
        
        # Gradient of objective
        return grad, obj
    
    # Update in small steps
    for _ in range(3):
        grad, _ = barrier_gradient(x, r)
        # We perform gradient ascent with adaptive stepsize
        step = 1e-6
        x += step * grad
        
        # Project to feasible region
        x = np.clip(x, 1e-10, 1 - 1e-10)
        # For radii, we project slightly & avoid exact 0
        r = np.clip(r, 1e-10, 0.5)
        
        # Update centers from flattened x
        centers_new = x.reshape((len(r), 2))
        
        # For final optimization check
        # We return new values after allowed iterations
    
    return centers_new, r

def final_refine(centers, radii):
    # Re-solve the SOCP with high precision
    total = len(radii)
    centers_var = cp.Variable((total, 2))
    radii_var = cp.Variable(total)

    constraints = []
    
    for i in range(total):
        constraints.append(centers_var[i, 0] - radii_var[i] >= 0)
        constraints.append(centers_var[i, 0] + radii_var[i] <= 1)
        constraints.append(centers_var[i, 1] - radii_var[i] >= 0)
        constraints.append(centers_var[i, 1] + radii_var[i] <= 1)
        
    for i in range(total):
        for j in range(i + 1, total):
            dist = cp.norm(centers_var[i] - centers_var[j])
            constraints.append(dist >= radii_var[i] + radii_var[j]) 
    
    objective = cp.Maximize(cp.sum(radii_var))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, abstol=1e-13, reltol=1e-13)
    
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        new_centers = centers_var.value
        new_radii = radii_var.value
        
        # Validate and ensure no violation from numerical error
        for i in range(total):
            if new_radii[i] < 0:
                new_radii[i] = 0
            # Ensure all circles are within bounds
            x_new, y_new = new_centers[i]
            r_new = new_radii[i]
            if x_new - r_new < -1e-12 or x_new + r_new > 1 + 1e-12:
                new_radii[i] = min(x_new, 1 - x_new)
            if y_new - r_new < -1e-12 or y_new + r_new > 1 + 1e-12:
                new_radii[i] = min(y_new, 1 - y_new)
                
        return new_centers, new_radii
    else:
        raise RuntimeError(f"Final refinement failed, status: {prob.status}")
