import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial.distance import cdist

def build_lp_model(n_circles):
    # Build LP model for 26 circles in unit square
    # Variables: 26 centers (x_i, y_i), 26 radii (r_i)
    n_vars = 2 * n_circles + n_circles
    
    # Coefficient matrix A_ub and bias b_ub (all constraints in form Ax <= b)
    # Start with empty constraints list
    A_ub = []
    b_ub = []

    # Wall constraints for each circle
    # r_i <= x_i, r_i <= 1 - x_i, r_i <= y_i, r_i <= 1 - y_i
    for i in range(n_circles):
        # r_i <= x_i => -r_i + x_i >= 0
        row = [0] * n_vars
        row[2*i] = -1.0  # r_i
        row[2*i+1] = 1.0  # x_i
        A_ub.append(row)
        b_ub.append(0.0)
        
        # r_i <= 1 - x_i => r_i + x_i >= 1
        row = [0] * n_vars
        row[2*i] = 1.0   # r_i
        row[2*i+1] = 1.0  # x_i
        A_ub.append(row)
        b_ub.append(1.0)
        
        # r_i <= y_i => -r_i + y_i >= 0
        row = [0] * n_vars
        row[2*i] = -1.0  # r_i
        row[2*i+2] = 1.0  # y_i
        A_ub.append(row)
        b_ub.append(0.0)
        
        # r_i <= 1 - y_i => r_i + y_i >= 1
        row = [0] * n_vars
        row[2*i] = 1.0   # r_i
        row[2*i+2] = 1.0  # y_i
        A_ub.append(row)
        b_ub.append(1.0)
        
    # Pairwise constraints (4 per pair of circles)
    # For circles i and j:
    # x_i + r_i <= x_j - r_j  =>  -x_i + x_j - r_i - r_j >= 0
    # x_j + r_j <= x_i - r_i  =>  x_j - x_i - r_j - r_i >= 0
    # y_i + r_i <= y_j - r_j  =>  -y_i + y_j - r_i - r_j >= 0
    # y_j + r_j <= y_i - r_i  =>  y_j - y_i - r_j - r_i >= 0
    for i in range(n_circles):
        for j in range(i+1, n_circles):
            # Pairwise constraints
            # x_i + r_i <= x_j - r_j
            row = [0] * n_vars
            row[2*i] = -1.0  # r_i
            row[2*j] = 1.0   # r_j
            row[2*i+1] = -1.0  # x_i
            row[2*j+1] = 1.0   # x_j
            A_ub.append(row)
            b_ub.append(0.0)
            
            # x_j + r_j <= x_i - r_i
            row = [0] * n_vars
            row[2*i] = 1.0   # r_i
            row[2*j] = -1.0  # r_j
            row[2*i+1] = 1.0   # x_i
            row[2*j+1] = -1.0  # x_j
            A_ub.append(row)
            b_ub.append(0.0)
            
            # y_i + r_i <= y_j - r_j
            row = [0] * n_vars
            row[2*i] = -1.0  # r_i
            row[2*j] = 1.0   # r_j
            row[2*i+2] = -1.0  # y_i
            row[2*j+2] = 1.0   # y_j
            A_ub.append(row)
            b_ub.append(0.0)
            
            # y_j + r_j <= y_i - r_i
            row = [0] * n_vars
            row[2*i] = 1.0   # r_i
            row[2*j] = -1.0  # r_j
            row[2*i+2] = 1.0   # y_i
            row[2*j+2] = -1.0  # y_j
            A_ub.append(row)
            b_ub.append(0.0)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)

    # Objective: maximize sum of radii (negate for linprog, which minimizes)
    c = np.zeros(n_vars)
    for i in range(n_circles):
        c[2*n_circles + i] = -1.0

    return A_ub, b_ub, c

def solve_lp(A_ub, b_ub, c):
    # Solve the linear program
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    
    if not res.success:
        raise RuntimeError("LP solver did not converge")

    # Extract result
    n_circles = (len(res.x) - 2) // 3
    centers = res.x[:2*n_circles].reshape((n_circles, 2))
    radii = res.x[2*n_circles:]
    
    return centers, radii

def detect_diagonal_conflicts(centers, radii, threshold=1e-12):
    n = centers.shape[0]
    conflicts = []
    
    for i in range(n):
        for j in range(i+1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - threshold:
                conflicts.append((i, j))
    
    return conflicts

def resolve_conflicts(centers, radii, active_set, conflict_pairs):
    # This is a simplified revision assuming all conflicts need handling
    
    # For each conflict pair, remove one to prevent overlap - keeping only one active constraint
    for i, j in conflict_pairs:
        # Slightly reduce radii to avoid overlap, and adjust centers minimally to preserve active constraints
        # Split total overlap and allow small restructuring
        overlap = (radii[i] + radii[j]) - np.linalg.norm(centers[i] - centers[j])
        if overlap > 0.:
            # Simple redistribute slightly
            radii[i] = max(0, radii[i] - overlap/4)
            radii[j] = max(0, radii[j] - overlap/4)
            
    return centers, radii

def polish_global(centers, radii, active_set):
    # Create a local optimization with few active constraints enforced exactly
    n_circles = centers.shape[0]
    
    # For the polish task, temporarily simplify to actual optimization on just active set
    # Keep radii as is, future improvements with `trust-constr` if needed
    
    return centers, radii

def max_radius_projection(centers, radii):
    # This implements the max-radius projection
    # Constrained optimization per circle around its geometric limits
    old_radii = radii.copy()
    for _ in range(20):  # Iterate until convergence or max steps
        converged = True
        new_radii = np.zeros_like(radii)
        
        for i in range(centers.shape[0]):
            # Wall constraints
            r = min(
                centers[i, 0], 
                1 - centers[i, 0], 
                centers[i, 1], 
                1 - centers[i, 1]
            )

            # Pairwise constraints
            for j in range(centers.shape[0]):
                if i != j:
                    dist = np.linalg.norm(centers[i] - centers[j])
                    r = min(r, dist - radii[j])
            
            # Update is valid only if positive and improves current
            if r > 0:
                new_radii[i] = r
                
        if np.allclose(radii, new_radii, atol=1e-10):
            break
            
        radii = new_radii
        converged = False
        
    return centers, radii

def run_packing():
    n_circles = 26
    A_ub, b_ub, c = build_lp_model(n_circles)
    centers, radii = solve_lp(A_ub, b_ub, c)
    
    # Build active constraints list (based on LP solution approximate tolerance)
    active_set = set()

    # Check wall constraints
    for i in range(n_circles):
        if abs(radii[i] - centers[i, 0]) < 1e-9:
            active_set.add((f'wall_x_l_{i}', i))
        if abs(radii[i] - (1 - centers[i, 0])) < 1e-9:
            active_set.add((f'wall_x_r_{i}', i))
        if abs(radii[i] - centers[i, 1]) < 1e-9:
            active_set.add((f'wall_y_b_{i}', i))
        if abs(radii[i] - (1 - centers[i, 1])) < 1e-9:
            active_set.add((f'wall_y_t_{i}', i))
    
    # Check some pairwise constraints (fast approximation)
    for i in range(n_circles):
        for j in range(i+1, n_circles):
            if (
                abs(centers[i, 0] + radii[i] - centers[j, 0] + radii[j]) < 1e-9 or
                abs(centers[j, 0] + radii[j] - centers[i, 0] + radii[i]) < 1e-9 or
                abs(centers[i, 1] + radii[i] - centers[j, 1] + radii[j]) < 1e-9 or
                abs(centers[j, 1] + radii[j] - centers[i, 1] + radii[i]) < 1e-9
            ):
                active_set.add((f'pair_{i}_{j}', i, j))

    # Detect Euclidean overlaps
    conflicts = detect_diagonal_conflicts(centers, radii)
    
    # Resolve conflicts (simplified model)
    if conflicts:
        centers, radii = resolve_conflicts(centers, radii, active_set, conflicts)
    
    # Polish again
    centers, radii = polish_global(centers, radii, active_set)

    # Final projection to ensure feasibility
    centers, radii = max_radius_projection(centers, radii)
    
    # Ensure no radii are negative
    radii = np.maximum(radii, 0)
    sum_r = float(radii.sum())
    
    # Validate
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise RuntimeError(f"Validation failed: {msg}")

    return centers, radii, sum_r
