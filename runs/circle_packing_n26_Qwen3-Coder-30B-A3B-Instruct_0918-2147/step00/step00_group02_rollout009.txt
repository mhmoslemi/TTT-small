import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def initial_configuration(n=26):
    # Generate Sobol sequence using a simple low-discrepancy method
    # For simplicity, use a scrambled Halton-like sequence
    phi = (1 + np.sqrt(5)) / 2  # golden ratio
    indices = np.arange(n)
    x = (indices * 1.0 / n) % 1
    y = (indices * phi / n) % 1
    centers = np.column_stack((x, y))
    # Assign initial radii based on minimum distance to walls and nearest neighbor
    radii = np.full(n, 0.0)
    for i in range(n):
        cx, cy = centers[i]
        r_wall = min(cx, 1 - cx, cy, 1 - cy)
        dists = np.linalg.norm(centers - centers[i], axis=1)
        dists[i] = np.inf
        nearest = np.min(dists)
        r_nearest = nearest * 0.5
        radii[i] = min(r_wall, r_nearest)
    # Ensure all radii are non-negative and reasonable
    radii = np.maximum(0, radii)
    return centers, radii

def unpack(z):
    n = len(z) // 3
    centers = z[:2*n].reshape((n, 2))
    radii = z[2*n:]
    return centers, radii

def pack(centers, radii):
    z = np.concatenate((centers.flat, radii))
    return z

def energy_and_grad(z):
    centers, radii = unpack(z)
    n = centers.shape[0]
    E = 0.0
    grad = np.zeros_like(z)
    
    # Wall repulsion
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        dx = max(0, r - x)
        dy = max(0, r - (1 - x))
        dz = max(0, r - y)
        dw = max(0, r - (1 - y))
        E += dx**2 + dy**2 + dz**2 + dw**2
        
        dr = 2 * (dx + dy + dz + dw)
        grad[2*n + i] += dr
        
        # Gradients w.r.t. center
        grad[i] -= 2 * dx
        grad[n + i] -= 2 * dz

    # Pairwise overlaps
    D = cdist(centers, centers)
    # Add small epsilon to prevent division by zero
    D += np.eye(n) * 1e-12
    for i in range(n):
        for j in range(i+1, n):
            d = D[i, j]
            r_i = radii[i]
            r_j = radii[j]
            overlap = max(0, r_i + r_j - d)
            E += overlap**2
            # Gradients w.r.t. radii
            dr_i = 2 * overlap
            dr_j = 2 * overlap
            grad[2*n + i] += dr_i
            grad[2*n + j] += dr_j
            # Gradients w.r.t. centers
            if overlap > 0:
                dx = (centers[i, 0] - centers[j, 0]) / d
                dy = (centers[i, 1] - centers[j, 1]) / d
                grad[i] -= 2 * overlap * dx
                grad[n + i] -= 2 * overlap * dy
                grad[j] += 2 * overlap * dx
                grad[n + j] += 2 * overlap * dy

    return E, grad

def inner_optimize(z0, B, lambda_budget):
    def obj_and_grad(z):
        E, g = energy_and_grad(z)
        # Budget penalty
        S = np.sum(z[2*len(z)//3:])
        E += lambda_budget * (B - S)**2
        g[2*len(z)//3:] += -2 * lambda_budget * (B - S)
        return E, g

    res = minimize(obj_and_grad, z0, method='L-BFGS-B',
                   bounds=[(0, 1)] * (2 * len(z0) // 3) + [(0, 0.5)] * (len(z0) // 3),
                   jac=True, options={'maxiter': 200})
    return res.x

def max_feasible_budget(z_start, lambda_budget=1000):
    B = 3.0
    delta = 5e-4
    last_feasible = z_start
    max_iter = 40
    for _ in range(max_iter):
        z_opt = inner_optimize(z_start, B, lambda_budget)
        centers, radii = unpack(z_opt)
        violation = 0.0
        # Check wall violations
        for i in range(len(radii)):
            x, y = centers[i]
            r = radii[i]
            v = max(0, r - x, r - (1 - x), r - y, r - (1 - y))
            violation = max(violation, v)

        # Check overlap violations
        D = cdist(centers, centers)
        np.fill_diagonal(D, np.inf)
        for i in range(len(radii)):
            for j in range(i+1, len(radii)):
                if D[i, j] < radii[i] + radii[j]:
                    v = max(0, radii[i] + radii[j] - D[i, j])
                    violation = max(violation, v)
                    
        if violation <= 1e-13:
            last_feasible = z_opt
            B += delta
        else:
            B -= delta
        if delta < 1e-6:
            break
    return last_feasible

def postprocess(z):
    # Shrink radii by epsilon and clamp centers
    centers, radii = unpack(z)
    eps = 5e-10
    # Shrink radii
    radii = np.maximum(0, radii - eps)
    # Clamp centers
    centers = np.clip(centers, eps, 1 - eps)
    # Repair overlaps if any
    D = cdist(centers, centers)
    np.fill_diagonal(D, np.inf)
    for i in range(len(radii)):
        for j in range(i+1, len(radii)):
            overlap = max(0, radii[i] + radii[j] - D[i, j])
            if overlap > 0:
                d = D[i, j]
                dr = overlap / 2 + eps
                radii[i] -= dr
                radii[j] -= dr
    # After repair, ensure non-negative radii
    radii = np.maximum(0, radii)
    return centers, radii, np.sum(radii)

def run_packing():
    centers, radii = initial_configuration(26)
    z = pack(centers, radii)
    z_final = max_feasible_budget(z, lambda_budget=1000)
    centers, radii, sum_r = postprocess(z_final)
    return centers, radii, sum_r
