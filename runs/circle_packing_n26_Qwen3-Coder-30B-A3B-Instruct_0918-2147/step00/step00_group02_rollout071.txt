import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def initial_configuration(n=26):
    # Generate Sobol sequence-like points using a low-discrepancy approach
    phi = (1 + np.sqrt(5)) / 2
    indices = np.arange(n)
    golden_angles = 2 * np.pi * indices / phi
    radii = np.sqrt(indices / (n - 1))
    x = 0.5 + 0.4 * radii * np.cos(golden_angles)
    y = 0.5 + 0.4 * radii * np.sin(golden_angles)
    centers = np.column_stack((x, y))

    # Compute pairwise distances and initial radius estimates
    dist_matrix = cdist(centers, centers)
    np.fill_diagonal(dist_matrix, np.inf)
    min_distances = np.min(dist_matrix, axis=1)
    
    radii = np.minimum(
        centers[:, 0],  # distance to left wall
        np.minimum(
            1 - centers[:, 0],  # right wall
            np.minimum(
                centers[:, 1],  # bottom wall
                1 - centers[:, 1]  # top wall
            )
        )
    )
    
    # Adjust each radius to be half the distance to nearest neighbor
    for i in range(n):
        radii[i] = min(radii[i], 0.5 * min_distances[i])
        
    # Ensure sufficient initial values
    radii = np.maximum(0.001, radii)
    return centers, radii

def unpack(z):
    n = 26
    centers = z[:2*n].reshape((n, 2))
    radii = z[2*n:]
    return centers, radii

def pack(centers, radii):
    return np.concatenate([centers.flatten(), radii])

def energy_and_grad(z):
    centers, radii = unpack(z)
    n = centers.shape[0]
    
    # wall penalities
    wall_pen = np.zeros(n)
    for i in range(n):
        r = radii[i]
        x, y = centers[i]
        wall_pen[i] = (max(0, r - x) ** 2 +
                       max(0, r - (1 - x)) ** 2 +
                       max(0, r - y) ** 2 +
                       max(0, r - (1 - y)) ** 2)
    
    # pair overlaps
    eps = 1e-12
    dist_matrix = cdist(centers, centers)
    np.fill_diagonal(dist_matrix, np.inf)
    overlap_penalty = 0.0
    
    grad_c = np.zeros_like(centers)
    grad_r = np.zeros(n)
    
    for i in range(n):
        r_i = radii[i]
        for j in range(i+1, n):
            d_ij = dist_matrix[i, j]
            if d_ij < r_i + radii[j]:
                overlap = r_i + radii[j] - d_ij
                overlap_penalty += overlap ** 2
                
                # gradient contribution of this pair
                if d_ij > eps:
                    dx = (centers[i] - centers[j]) / d_ij
                    grad_c[i] -= 2 * overlap * dx
                    grad_c[j] += 2 * overlap * dx
                    grad_r[i] += 2 * overlap
                    grad_r[j] += 2 * overlap
        # wall gradients
        x, y = centers[i]
        r = radii[i]
        if r > x:
            grad_r[i] += 2 * (r - x)
            grad_c[i, 0] -= 2 * (r - x)
        if r > 1 - x:
            grad_r[i] += 2 * (r - (1-x))
            grad_c[i, 0] += 2 * (r - (1-x))
        if r > y:
            grad_r[i] += 2 * (r - y)
            grad_c[i, 1] -= 2 * (r - y)
        if r > 1 - y:
            grad_r[i] += 2 * (r - (1-y))
            grad_c[i, 1] += 2 * (r - (1-y))
            
    return np.sum(wall_pen) + overlap_penalty, grad_c.flatten(), grad_r

def inner_optimize(z0, B, lambda_budget):
    def obj_and_grad(z):
        E, g_c, g_r = energy_and_grad(z)
        E += lambda_budget * (B - np.sum(z[52:])) ** 2
        gradient = np.concatenate([g_c, g_r])
        gradient += 2 * lambda_budget * (B - np.sum(z[52:])) * np.concatenate([np.zeros(52), np.ones(26)])
        return E, gradient
    
    bounds = [(0, 1)] * 52 + [(0, 0.5)] * 26
    opt_result = minimize(obj_and_grad, z0, jac=True, method='L-BFGS-B', bounds=bounds)
    return opt_result.x

def max_feasible_budget(z_start, B0=3.0, delta=5e-4, lambda_budget=2000.0, max_iters=40):
    B = B0
    z = z_start.copy()
    
    for _ in range(max_iters):
        z = inner_optimize(z, B, lambda_budget)
        centers, radii = unpack(z)
        
        # Check maximum violation
        # wall violations
        wall_viol = np.max([
            radii[i] - centers[i, 0],
            radii[i] - (1 - centers[i, 0]),
            radii[i] - centers[i, 1],
            radii[i] - (1 - centers[i, 1])
            for i in range(26)
        ])
        
        # overlap violations
        dists = cdist(centers, centers)
        np.fill_diagonal(dists, np.inf)
        overlap_viol = np.max([
            radii[i] + radii[j] - dists[i, j] 
            for i in range(26) for j in range(i+1, 26)
        ])
        
        max_viol = max(wall_viol, overlap_viol)
        max_viol = max(0, max_viol)  # handle negative violation
        
        if max_viol < 1e-13: # feasible
            B += delta
        else:
            B -= delta
        if abs(delta) < 1e-6:
            break
    return z

def postprocess(z):
    centers, radii = unpack(z)
    eps = 5e-10
    
    # Ensure radii are well within constraints
    radii = np.maximum(0, radii - eps)
    centers[:, 0] = np.clip(centers[:, 0], eps, 1 - eps)
    centers[:, 1] = np.clip(centers[:, 1], eps, 1 - eps)
    
    # Repair any overlap
    dists = cdist(centers, centers)
    np.fill_diagonal(dists, np.inf)
    for i in range(26):
        for j in range(i+1, 26):
            d_ij = dists[i, j]
            rsum = radii[i] + radii[j]
            if d_ij < rsum:
                overlap = rsum - d_ij
                radii[i] -= overlap / 2 + eps
                radii[j] -= overlap / 2 + eps
                radii[i] = max(0, radii[i])
                radii[j] = max(0, radii[j])
    
    # Final update
    sum_r = np.sum(radii)
    return centers, radii, sum_r

def run_packing():
    centers, radii = initial_configuration(26)
    z_start = pack(centers, radii)
    z_opt = max_feasible_budget(z_start)
    centers, radii, sum_r = postprocess(z_opt)
    return centers, radii, sum_r
