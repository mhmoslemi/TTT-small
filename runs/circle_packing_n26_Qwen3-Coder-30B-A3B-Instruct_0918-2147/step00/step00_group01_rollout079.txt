import numpy as np
from scipy.spatial.distance import cdist

def penalty(z, alpha=10.0, beta=10.0):
    # z is (26, 3) where each row is (x, y, r)
    centers = z[:, :2]
    radii = z[:, 2]
    
    # Wall penalties
    wall_violations = np.zeros_like(radii)
    wall_violations += np.maximum(0, radii - centers[:, 0])      # left
    wall_violations += np.maximum(0, radii - (1 - centers[:, 0])) # right
    wall_violations += np.maximum(0, radii - centers[:, 1])      # bottom
    wall_violations += np.maximum(0, radii - (1 - centers[:, 1])) # top
    
    # Overlap penalties
    dists = cdist(centers, centers)
    np.fill_diagonal(dists, np.inf)
    overlap_violations = np.zeros_like(dists)
    for i in range(len(radii)):
        for j in range(i + 1, len(radii)):
            dist = dists[i, j]
            overlap = radii[i] + radii[j] - dist
            overlap_violations[i, j] = max(0, -overlap)
    
    total_wall_penalty = np.sum(wall_violations)
    total_overlap_penalty = np.sum(overlap_violations) / 2  # avoid double-counting
    
    return alpha * total_wall_penalty + beta * total_overlap_penalty

def energy(z, lam, alpha=10.0, beta=10.0):
    # z is (26, 3) where each row is (x, y, r)
    sum_radii = np.sum(z[:, 2])
    pen = penalty(z, alpha, beta)
    return -sum_radii + lam * pen

def propose_move(z, sigma_c, sigma_r, temperature):
    new_z = z.copy()
    move_type = np.random.choice(3)
    
    if move_type == 0:
        # center jitter
        i = np.random.randint(0, len(z))
        noise = np.random.normal(0, sigma_c * np.sqrt(temperature), 2)
        new_z[i, :2] += noise
        new_z[i, :2] = np.clip(new_z[i, :2], 0, 1)
        
    elif move_type == 1:
        # radius perturb
        i = np.random.randint(0, len(z))
        noise = np.random.normal(0, sigma_r * np.sqrt(temperature))
        new_z[i, 2] += noise
        new_z[i, 2] = max(0, new_z[i, 2])
        
    else:
        # swap pair
        i, j = np.random.choice(len(z), size=2, replace=False)
        new_z[i, 2], new_z[j, 2] = new_z[j, 2], new_z[i, 2]
    
    return new_z

def project_and_ascent(centers, radii):
    z = np.column_stack([centers, radii])
    eta = 5e-3
    max_iter = 500
    batch_size = 100
    
    for _ in range(max_iter):
        # gradient ascent step: respect constraints
        z_new = z.copy()
        z_new[:, 2] += eta  # gradient ascent direction

        # project onto wall constraints
        for i in range(len(z)):
            max_radius = min(z_new[i, 0], 1 - z_new[i, 0], z_new[i, 1], 1 - z_new[i, 1])
            z_new[i, 2] = min(z_new[i, 2], max_radius)
        
        # resolution of overlaps step
        centers_new = z_new[:, :2]
        radii_new = z_new[:, 2]
        dists = cdist(centers_new, centers_new)
        np.fill_diagonal(dists, np.inf)
        
        update_count = 0
        all_done = True
        for i in range(len(radii_new)):
            for j in range(i + 1, len(radii_new)):
                if dists[i, j] < radii_new[i] + radii_new[j]:
                    all_done = False
                    overlap = radii_new[i] + radii_new[j] - dists[i, j]
                    shrink_amount = overlap / 2
                    radii_new[i] -= shrink_amount
                    radii_new[j] -= shrink_amount
                    update_count += 1
                    
        # wall projection after overlap resolution
        for i in range(len(z)):
            max_radius = min(centers_new[i, 0], 1 - centers_new[i, 0], centers_new[i, 1], 1 - centers_new[i, 1])
            radii_new[i] = min(radii_new[i], max_radius)
            radii_new[i] = max(0, radii_new[i])
            
        z = np.column_stack([centers_new, radii_new])
        eta *= 0.95
        
        if all_done and update_count == 0:
            break
    
    return z

def run_packing():
    # Initial setup
    n_circles = 26
    np.random.seed(42)
    
    # Generate hexagonal grid constrained to the unit square
    grid_spacing = 0.18
    rows = int(1.0 / grid_spacing) + 1
    cols = int(1.0 / grid_spacing) + 1

    centers = []
    for i in range(rows):
        for j in range(cols):
            x = i * grid_spacing
            y = j * grid_spacing
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers.append([x, y])
    
    # Make sure we have at least n_circles by distorting a few points
    while len(centers) < n_circles:
        x = np.random.uniform(0, 1)
        y = np.random.uniform(0, 1)
        centers.append([x, y])

    centers = np.array(centers[:n_circles])
    radii = np.minimum(centers[:, 0], 1 - centers[:, 0], centers[:, 1], 1 - centers[:, 1])
    z = np.column_stack([centers, radii])

    # Annealing parameters
    T0 = 1.0
    c_T = 0.995
    T = T0
    lam = 0.1
    c_lam = 1.03
    iterations = 10000
    sigma_c = 0.03
    sigma_r = 0.02
    
    best_z = z.copy()
    best_energy = energy(z, lam)
    best_energy_feasible = np.inf
    best_feasible_z = z.copy()
    
    # Annealing loop
    for k in range(iterations):
        z_proposed = propose_move(z, sigma_c, sigma_r, T)
        e_proposed = energy(z_proposed, lam)
        
        delta_E = e_proposed - energy(z, lam)
        
        if e_proposed < best_energy:
            best_energy = e_proposed
            best_z = z_proposed.copy()
            
        if delta_E <= 0 or np.random.rand() < np.exp(-delta_E / T):
            z = z_proposed
            
        if penalty(z) < 1e-6:
            if e_proposed < best_energy_feasible:
                best_energy_feasible = e_proposed
                best_feasible_z = z.copy()
                
        T *= c_T
        if T < 1e-5:
            break
        lam *= c_lam

    # Final smooth optimization via projection-based gradient ascent
    refined_z = project_and_ascent(best_feasible_z[:, :2], best_feasible_z[:, 2])
    
    # Extract result
    refined_centers = refined_z[:, :2]
    refined_radii = refined_z[:, 2]
    sum_radii = np.sum(refined_radii)
    
    # final validation
    status, msg = validate_packing(refined_centers, refined_radii)
    if not status:
        # apply tiny uniform shrink to ensure validity for rare cases post projection
        shrink_factor = 1.0 - 1e-7
        refined_radii *= shrink_factor
        status, msg = validate_packing(refined_centers, refined_radii)
        if not status:
            raise RuntimeError("Validation failed even after shrinking")
    
    return refined_centers, refined_radii, sum_radii
