import numpy as np

def hex_lattice(d):
    """Generate a hexagonal lattice of points within the unit square."""
    sqrt3 = np.sqrt(3)
    dx = d
    dy = sqrt3 * d / 2
    rows = []
    y = 0
    row_index = 0
    while y <= 1 + 1e-12:
        x = 0
        if row_index % 2 == 1:
            x = dx / 2
        row = []
        while x <= 1 + 1e-12:
            row.append([x, y])
            x += dx
        rows.extend(row)
        y += dy
        row_index += 1
    return np.array(rows)

def add_border_circles(points, d):
    """Append six border-enhanced circles to the hexagonal core."""
    r = d / 4
    # Corner circles
    corners = [
        [r, r],
        [1 - r, r],
        [r, 1 - r],
        [1 - r, 1 - r]
    ]
    # Edge mid circles
    edges = [
        [0.5, r],
        [0.5, 1 - r]
    ]
    border = np.array(corners + edges)
    return np.vstack([points, border])

def initial_radii(points):
    """Assign initial uniform safe radii to all points."""
    n = points.shape[0]
    dmat = np.linalg.norm(points[:, None, :] - points[None, :, :], axis=2)
    wall_distances = np.minimum(points[:, 0], 1 - points[:, 0])
    wall_distances = np.minimum(wall_distances, points[:, 1])
    wall_distances = np.minimum(wall_distances, 1 - points[:, 1])
    min_dists = np.min(dmat + np.eye(n) * 1e10, axis=1)
    r0 = np.min(np.minimum(wall_distances, min_dists / 2))
    return np.full(n, r0)

def inflate_radii(centers, radii):
    """Inflate radii as much as possible while maintaining feasibility."""
    n = centers.shape[0]
    dmat = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    new_radii = np.zeros(n)
    eps = 1e-12
    for i in range(n):
        # Wall constraints
        wall_r = centers[i, 0]  # left
        wall_r = min(wall_r, 1 - centers[i, 0])  # right
        wall_r = min(wall_r, centers[i, 1])  # bottom
        wall_r = min(wall_r, 1 - centers[i, 1])  # top
        # Neighbour constraints
        max_safe = wall_r
        for j in range(n):
            if i != j:
                dist = dmat[i, j]
                safe = dist - radii[j]
                if safe > max_safe + eps:
                    max_safe = safe
        new_radii[i] = max(max_safe, radii[i])
    return new_radii

def relax_centers(centers, radii, steps=10):
    """Relax center positions using repulsive and wall forces."""
    n = centers.shape[0]
    # Force parameters
    kappa = 0.02
    lam = 0.01
    alpha = 0.5
    beta = 0.02
    gamma = 0.02
    for _ in range(steps):
        forces = np.zeros_like(centers)
        # Compute all pairwise repulsive forces
        dmat = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
        for i in range(n):
            for j in range(n):
                if i != j:
                    dist = dmat[i, j]
                    delta = radii[i] + radii[j] - dist
                    if delta > 0:
                        raise AssertionError("Overlapping circles in relaxation")
                    if dist < beta:
                        f = kappa * max(0, beta - dist) * (centers[i] - centers[j]) / (dist + 1e-12)
                        forces[i] += f
        # Wall forces
        for i in range(n):
            f_x = 0
            f_x += -lam * max(0, gamma - (centers[i, 0] - radii[i]))
            f_x += lam * max(0, gamma - (1 - centers[i, 0] - radii[i]))
            f_y = 0
            f_y += -lam * max(0, gamma - (centers[i, 1] - radii[i]))
            f_y += lam * max(0, gamma - (1 - centers[i, 1] - radii[i]))
            forces[i, 0] += f_x
            forces[i, 1] += f_y
        # Update centers
        new_centers = centers + alpha * forces
        # Clamp centers to unit square respecting radii
        new_centers[:, 0] = np.clip(new_centers[:, 0], radii, 1 - radii)
        new_centers[:, 1] = np.clip(new_centers[:, 1], radii, 1 - radii)
        centers = new_centers
    return centers

def run_packing():

    d = 0.18
    core = hex_lattice(d)
    centers = add_border_circles(core, d)

        centers = relax_centers(centers, radii, steps=10)
