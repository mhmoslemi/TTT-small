import numpy as np
from scipy.optimize import minimize

def init_grid():
    np.random.seed(0)
    s = 0.07
    centers = []
    for k in range(6):
        y = s / 2 + k * s * np.sqrt(3) / 2
        if y > 1 - s:
            break
        for m in range(10):
            if k % 2 == 0:
                x = s / 2 + m * s
            else:
                x = s / 2 + m * s + s / 2
            if x > 1 - s:
                break
            if x >= s and x <= 1 - s and y >= s and y <= 1 - s:
                centers.append([x, y])
            if len(centers) >= 26:
                break
        if len(centers) >= 26:
            break

    centers = np.array(centers[:26])
    centers += np.random.uniform(-0.001, 0.001, (26, 2))
    radii = np.full(26, 0.001)
    return centers, radii

def compute_forces(centers, radii):
    n = centers.shape[0]
    k_w = 10.0
    k_p = 5.0
    forces = np.zeros_like(centers)
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Wall forces
        dL = x - r
        dR = (1 - x) - r
        dB = y - r
        dT = (1 - y) - r
        fL = k_w * max(-dL, 0) * np.array([-1, 0])
        fR = k_w * max(-dR, 0) * np.array([1, 0])
        fB = k_w * max(-dB, 0) * np.array([0, -1])
        fT = k_w * max(-dT, 0) * np.array([0, 1])
        forces[i] += fL + fR + fB + fT

        # Pairwise repulsion
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            gap = dist - (radii[i] + radii[j])
            if gap < 0:
                f = k_p * (-gap) * np.array([dx, dy]) / (dist + 1e-12)
                forces[i] += f
                forces[j] -= f

    return forces

def grow_radii(centers, radii, forces, dt=0.02, alpha=1e-3):
    n = centers.shape[0]
    dot_r = np.ones(n) * alpha
    for i in range(n):
        # Check wall constraints
        x, y = centers[i]
        r = radii[i]
        if x - r <= 1e-12 or x + r >= 1 - 1e-12 or y - r <= 1e-12 or y + r >= 1 - 1e-12:
            dot_r[i] = 0
        # Check overlap constraints
        for j in range(n):
            if i != j:
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    dot_r[i] = 0
                    break
    return radii + dt * dot_r

def max_radii_fixed_centers(centers, radii):
    n = centers.shape[0]
    old_radii = np.copy(radii)
    any_changed = True
    iteration = 0
    while any_changed and iteration < 30:
        any_changed = False
        new_radii = np.zeros(n)
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            min_r = min(x, 1 - x, y, 1 - y)
            for j in range(n):
                if i != j:
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    gap = dist - radii[j]
                    if gap < min_r:
                        min_r = gap
            new_radii[i] = max(0, min_r)
            if abs(new_radii[i] - old_radii[i]) > 1e-9:
                any_changed = True
        radii = np.copy(new_radii)
        old_radii = np.copy(radii)
        iteration += 1
    return radii

def objective_sum_radii(centers_flat):
    n = 26
    centers = centers_flat.reshape((n, 2))
    # Helper function to compute effective radii (force-simulated)
    radii = np.zeros(n)
    for i in range(n):
        x, y = centers[i]
        r = 0.001  # base radius
        min_r = min(x, 1 - x, y, 1 - y)
        for j in range(n):
            if i != j:
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                gap = dist - radii[j]
                if gap < min_r:
                    min_r = gap
        radii[i] = max(0, min_r)
    return -radii.sum()

def run_packing():
    centers, radii = init_grid()
    dt = 0.02
    gamma = 1.0
    v = np.zeros_like(centers)
    for step in range(5000):  # simulation steps
        forces = compute_forces(centers, radii)
        v = v + dt * (forces - gamma * v)
        centers = centers + dt * v
        centers = np.clip(centers, 0, 1)
        radii = grow_radii(centers, radii, forces, dt)
        if np.allclose(radii, np.zeros_like(radii)) and np.linalg.norm(forces) < 1e-6:
            break

    radii = max_radii_fixed_centers(centers, radii)
    # Final refinement using scipy minimize
    def func(x):
        n = 26
        local_centers = x.reshape((n, 2))
        r = np.zeros(n)
        for i in range(n):
            x_i, y_i = local_centers[i]
            min_r = min(x_i, 1 - x_i, y_i, 1 - y_i)
            for j in range(n):
                if i != j:
                    dx = local_centers[i, 0] - local_centers[j, 0]
                    dy = local_centers[i, 1] - local_centers[j, 1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    gap = dist - r[j]
                    if gap < min_r:
                        min_r = gap
            r[i] = max(0, min_r)
        return -r.sum()

    bounds = [(0, 1) for _ in range(52)]
    x0 = centers.flatten()
    result = minimize(func, x0, method='L-BFGS-B', bounds=bounds, options={'maxiter': 2000, 'gtol': 1e-9})
    centers = result.x.reshape((26, 2))
    radii = np.zeros(26)
    for i in range(26):
        x, y = centers[i]
        min_r = min(x, 1 - x, y, 1 - y)
        for j in range(26):
            if i != j:
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                gap = dist - radii[j]
                if gap < min_r:
                    min_r = gap
        radii[i] = max(0, min_r)

    sum_radii = np.sum(radii)
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise Exception(msg)
    return centers, radii, sum_radii
