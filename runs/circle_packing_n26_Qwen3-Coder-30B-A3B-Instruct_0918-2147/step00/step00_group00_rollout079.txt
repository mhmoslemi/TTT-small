import numpy as np
import random

def make_hex_seed(n, margin):
    # Generate a hexagonal grid with margin
    rows = int(np.sqrt(n))
    cols = (n + rows - 1) // rows
    
    # Use spacing to ensure margin
    spacing_x = (1 - 2 * margin) / cols
    spacing_y = (1 - 2 * margin) / rows
    
    centers = []
    radii = []
    
    for i in range(rows):
        for j in range(cols):
            if len(centers) >= n:
                break
            x = margin + (j + 0.5) * spacing_x
            y = margin + (i + 0.5) * spacing_y
            
            # Compute wall distance
            dist_to_walls = [x, 1-x, y, 1-y]
            min_wall_dist = min(dist_to_walls)
            
            # Radius based on distance to walls, clamped
            r = min_wall_dist * 0.7
            
            centers.append([x, y])
            radii.append(r)
            
    return np.array(centers[:n]), np.array(radii[:n])

def max_feasible_radius(i, centers, radii):
    # Get the center and current radius of circle i
    c_i = centers[i]
    r_i = radii[i]
    
    # Wall distances
    d_L = c_i[0]
    d_R = 1 - c_i[0]
    d_B = c_i[1]
    d_T = 1 - c_i[1]
    
    # Neighbour distances
    min_nei_dist = float('inf')
    for j in range(len(centers)):
        if i == j:
            continue
        dist = np.linalg.norm(c_i - centers[j])
        min_nei_dist = min(min_nei_dist, dist - radii[j])
    
    # Take the minimum of all constraints
    feasible_radius = min(d_L, d_R, d_B, d_T, min_nei_dist, r_i)
    
    return max(0, feasible_radius)

def inflate_radii(centers, radii, alpha):
    mean_r = np.mean(radii)
    delta = alpha * mean_r
    
    new_radii = radii + delta
    total_increase = 0
    
    for i in range(len(centers)):
        max_r_i = max_feasible_radius(i, centers, new_radii)
        old_radius = radii[i]
        new_radii[i] = min(new_radii[i], max_r_i)
        total_increase += new_radii[i] - old_radius
    
    return new_radii, total_increase

def anneal_centers(centers, radii, temp0, gamma, sigma0, steps):
    def propose_move(i, centers, radii, sigma):
        current_c = centers[i].copy()
        # Use numpy's random to generate a small normal displacement
        delta = np.random.normal(0, sigma, 2)
        new_c = current_c + delta
        # Clamp within square
        new_c[0] = np.clip(new_c[0], 0, 1)
        new_c[1] = np.clip(new_c[1], 0, 1)
        return new_c
    
    current_radii = radii.copy()
    current_centers = centers.copy()
    best_sum = np.sum(current_radii)
    best_centers = current_centers.copy()
    best_radii = current_radii.copy()
    
    T = temp0
    sigma = sigma0
    
    # Track recent improvements
    recent_improvements = 0
    recent_stall = 0
    max_stall = 1000
    global_shift_freq = 50
    
    for step in range(steps):
        # Occasionally apply global shift to escape local traps
        if step % global_shift_freq == 0:
            shift_vec = np.random.uniform(-0.001, 0.001, 2)
            current_centers += shift_vec
            current_centers[:, 0] = np.clip(current_centers[:, 0], 0, 1)
            current_centers[:, 1] = np.clip(current_centers[:, 1], 0, 1)
        
        # Pick a random circle
        i = random.randint(0, len(centers) - 1)
        current_center = current_centers[i].copy()
        current_radius = current_radii[i]
        
        # Propose new center
        new_center = propose_move(i, current_centers, current_radii, sigma)
        
        # Save original centers
        old_centers = current_centers.copy()
        old_radii = current_radii.copy()
        
        # Replace and recalculate feasible radius
        current_centers[i] = new_center
        new_feasible_radius = max_feasible_radius(i, current_centers, current_radii)
        
        # Accept or reject the move
        delta_radius = new_feasible_radius - current_radius
        if delta_radius > 0:
            current_radii[i] = new_feasible_radius
            # Update best if needed
            current_sum = np.sum(current_radii)
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = current_centers.copy()
                best_radii = current_radii.copy()
            recent_improvements += 1
        else:
            acceptance_prob = np.exp(delta_radius / T)
            if np.random.rand() < acceptance_prob:
                current_radii[i] = new_feasible_radius
                current_sum = np.sum(current_radii)
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = current_centers.copy()
                    best_radii = current_radii.copy()
                recent_improvements += 1
            else:
                # Revert changes
                current_centers = old_centers
                current_radii = old_radii
                
        # Update temperature and sigma
        T *= gamma
        sigma = sigma0 * T / temp0
        
        # Update stall counter
        if delta_radius > 0 or np.random.rand() < 0.1:
            recent_stall = 0
        else:
            recent_stall += 1
            
        if recent_stall > max_stall:
            break
            
    return best_centers, best_radii, best_sum

def mixed_cycle(centers, radii, params):
    alpha = params['alpha']
    stop_tolerance = params['stop_tolerance']
    stagnation_steps = params['stagnation_steps']
    sa_steps = params['sa_steps']
    cycles_before_stop = params['cycles_before_stop']
    temp0 = params['temp0']
    gamma = params['gamma']
    sigma0 = params['sigma0']
    
    current_centers = centers.copy()
    current_radii = radii.copy()
    
    last_sum = np.sum(current_radii)
    stagnation_count = 0
    cycles_without_improvement = 0
    
    while cycles_without_improvement < cycles_before_stop:
        # Phase B: inflation
        for round_num in range(100):
            new_radii, delta_sum = inflate_radii(current_centers, current_radii, alpha)
            current_radii = new_radii
            
            # Calculate relative change
            current_sum = np.sum(current_radii)
            if abs(delta_sum) < stop_tolerance:
                stagnation_count += 1
                if stagnation_count > stagnation_steps:
                    break
            else:
                stagnation_count = 0
            
            # Occasionally update alpha
            if round_num % 10 == 0 and alpha > 0.001:
                alpha *= 0.95
        
        # The inflation has stalled, perform SA
        new_centers, new_radii, sa_sum = anneal_centers(
            current_centers, current_radii, temp0, gamma, sigma0, sa_steps)
        
        current_centers = new_centers
        current_radii = new_radii
        
        current_sum = np.sum(current_radii)
        if abs(current_sum - last_sum) < stop_tolerance:
            cycles_without_improvement += 1
        else:
            cycles_without_improvement = 0
        last_sum = current_sum
        
        # Reset inflation alpha
        alpha = 0.04
    
    return current_centers, current_radii

def polish_radii(centers, radii, beta, passes):
    # Apply local polishing improvement
    def calculate_gaps(i, centers, radii):
        c_i = centers[i]
        r_i = radii[i]
        
        # Wall gaps
        g_wall = min(
            c_i[0] - r_i,
            1 - c_i[0] - r_i,
            c_i[1] - r_i,
            1 - c_i[1] - r_i
        )
        
        # Neighbour gaps
        g_nei = float('inf')
        for j in range(len(centers)):
            if i == j:
                continue
            dist = np.linalg.norm(c_i - centers[j])
            gap = dist - (r_i + radii[j])
            g_nei = min(g_nei, gap)
        
        return min(g_wall, g_nei)
    
    # Simplified exact packing adjustment
    for _ in range(passes):
        new_radii = radii.copy()
        for i in range(len(radii)):
            gap = calculate_gaps(i, centers, radii)
            if gap > 0:
                inc = beta * gap
                max_feasible_r = max_feasible_radius(i, centers, new_radii)
                new_radii[i] = min(new_radii[i] + inc, max_feasible_r)
        
        # Resolve overlaps
        for i in range(len(centers)):
            for j in range(i+1, len(centers)):
                dist = np.linalg.norm(centers[i] - centers[j])
                total_r = new_radii[i] + new_radii[j]
                if dist < total_r:
                    overlap = total_r - dist
                    new_radii[i] -= overlap / 2
                    new_radii[j] -= overlap / 2
        
        # Adjust any negative values
        new_radii = np.maximum(new_radii, 0)
        radii = new_radii
    
    return radii

def run_packing():
    # Set seed for reproducibility
    np.random.seed(42)
    random.seed(42)
    
    # Use hexagonal seed with margin of 0.04
    centers, radii = make_hex_seed(26, 0.04)
    
    # Parameters for the mixed cycle method
    params = {
        'alpha': 0.04,
        'stop_tolerance': 1e-6,
        'stagnation_steps': 20,
        'sa_steps': 2000,
        'cycles_before_stop': 5,
        'temp0': 0.05,
        'gamma': 0.97,
        'sigma0': 0.03
    }
    
    # Run the mixed inflation-SA cycle
    centers, radii = mixed_cycle(centers, radii, params)
    
    # Final polishing
    radii = polish_radii(centers, radii, beta=0.6, passes=3)
    
    # Return final centers, radii, sum
    total_sum = np.sum(radii)
    return centers, radii, total_sum
