import numpy as np
from scipy.stats.qmc import Sobol
from scipy.spatial.distance import cdist

def initialize_seeds(n, seed=0):
    sobol = Sobol(d=2, seed=seed)
    points = sobol.random(n)
    # Clip to interior with margin
    margin = 0.02
    clipped = margin + (1 - 2 * margin) * points
    return clipped

def compute_feasible_radii(centers):
    n = centers.shape[0]
    # Distance matrix
    D = cdist(centers, centers)
    # For each point, find the minimum distance to others (excluding self)
    nn_distances = np.min(D + np.eye(n) * np.inf, axis=0)
    # Wall distances
    wall_distances = np.minimum(centers[:, 0], 1 - centers[:, 0])
    wall_distances = np.minimum(wall_distances, centers[:, 1])
    wall_distances = np.minimum(wall_distances, 1 - centers[:, 1])
    # Feasible radius for each circle
    radii = np.minimum(wall_distances, 0.5 * nn_distances)
    return radii

def redistribute_radii(radii, slack, alpha):
    B = np.sum(slack)
    if B <= 0:
        return radii.copy()
    new_radii = radii.copy()
    # Redistribute the increase
    indices_wall_limited = np.where(slack > 1e-12)[0]
    if len(indices_wall_limited) == 0:
        return new_radii
    total_slack = np.sum(slack[indices_wall_limited])
    delta_radii = np.zeros_like(radii)
    for i in indices_wall_limited:
        delta_radii[i] = alpha * slack[i]
    # Reduce from neighbour-limited circles proportionally
    indices_neighbour_limited = np.where(slack <= 1e-12)[0]
    total_radii = np.sum(radii[indices_neighbour_limited])
    if total_radii <= 1e-12:
        # No redistribution possible, skip
        return new_radii
    for i in indices_neighbour_limited:
        delta = delta_radii[i]
        prop = radii[i] / total_radii if total_radii > 0 else 0.0
        reduction = prop * B * alpha
        new_radii[i] = max(0, new_radii[i] - reduction)
    # Add increase to wall-limited
    for i in indices_wall_limited:
        new_radii[i] += delta_radii[i]
    return new_radii

def push_away(centers, radii, beta, eps):
    n = centers.shape[0]
    new_centers = centers.copy()
    for i in range(n):
        c_i = new_centers[i]
        r_i = radii[i]
        for j in range(n):
            if i == j:
                continue
            c_j = centers[j]
            r_j = radii[j]
            v_ij = c_i - c_j
            dist = np.linalg.norm(v_ij)
            if dist <= r_i + r_j + eps:
                delta = (r_i + r_j + eps - dist)
                move = beta * delta * (v_ij / (dist + eps))
                new_centers[i] += move
        # Project back into [eps, 1-eps]
        new_centers[i] = np.clip(new_centers[i], eps, 1 - eps)
    return new_centers

def iterate(centers, radii, alpha, beta, max_iter):
    n = centers.shape[0]
    for it in range(max_iter):
        old_sum = np.sum(radii)
        # Compute new radii and slack
        new_radii = compute_feasible_radii(centers)
        slack = 0.5 * np.min(cdist(centers, centers), axis=1) - new_radii
        slack = np.maximum(0, slack)
        # Redistribute
        old_radii = new_radii.copy()
        new_radii = redistribute_radii(new_radii, slack, alpha)
        # Push away
        centers = push_away(centers, new_radii, beta, 1e-8)
        # Compute new radii after move
        new_radii = compute_feasible_radii(centers)
        new_sum = np.sum(new_radii)
        if new_sum - old_sum < 1e-5:
            break
    return centers, new_radii

def repair(centers, radii, eps_final):
    # Shrink radii slightly
    new_radii = np.maximum(0, radii - eps_final)
    # Clip centers
    new_centers = np.clip(centers, eps_final, 1 - eps_final)
    return new_centers, new_radii

def run_packing():
    n = 26
    centers = initialize_seeds(n)
    radii = compute_feasible_radii(centers)
    centers, radii = iterate(centers, radii, alpha=0.4, beta=0.3, max_iter=200)
    centers, radii = repair(centers, radii, eps_final=5e-10)
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
