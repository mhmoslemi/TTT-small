import numpy as np
from scipy.optimize import linprog, minimize
from math import sqrt, ceil

def build_hex_lattice(d):
    """Generate a hexagonal lattice of at least d points inside the unit square."""
    # For d=26, aiming for a spacing of around 0.18
    spacing = 0.18
    rows = ceil(sqrt(d) * 1.2)
    cols = ceil(d / rows) + 1
    
    centers = []
    y = spacing / 2
    row = 0
    while len(centers) < d and row < rows:
        x = spacing / 2 if row % 2 == 0 else spacing
        col = 0
        while len(centers) < d and col < cols:
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers.append([x, y])
            x += spacing
            col += 1
        y += spacing * sqrt(3) / 2
        row += 1
    return np.array(centers[:d])

def lp_max_radii(centers):
    """Compute the optimal radii for fixed centers using a linear program."""
    n = centers.shape[0]
    # Build constraints: A_ub @ radii <= b_ub
    # Wall constraints: r <= min(x, 1-x, y, 1-y) for each circle
    # Pairwise non-overlap constraints: r_i + r_j <= dist(c_i, c_j)
    A_ub = []
    b_ub = []
    
    for i in range(n):
        x, y = centers[i]
        r_max = min(x, 1 - x, y, 1 - y)
        # Wall inequality: r_i <= r_max
        A_ub.append([1 if j == i else 0 for j in range(n)])
        b_ub.append(r_max)
        
    # Pairwise overlap constraints
    for i in range(n):
        for j in range(i+1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            A_ub.append([1 if k == i else (1 if k == j else 0) for k in range(n)])
            b_ub.append(dist)
    
    # Objective: maximize sum of radii
    c = [-1.0] * n  # Negative because linprog minimizes
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', tol=1e-9)
        if res.status == 0:
            radii = res.x
            sum_radii = -res.fun
        else:
            # fallback to zero radii if infeasible (should not happen with valid centers)
            radii = np.zeros(n)
            sum_radii = 0.0
    except:
        # Fallback if any exception occurs
        radii = np.zeros(n)
        sum_radii = 0.0
        
    return radii, sum_radii

def perturb_centers(centers, sigma):
    """Perturb centers by Gaussian noise and clamp to [0,1]."""
    perturbed = centers + np.random.normal(0, sigma, centers.shape)
    perturbed = np.clip(perturbed, 0, 1)
    return perturbed

def anneal_centers(initial_centers):
    """Run simulated annealing to find a good center configuration."""
    np.random.seed(42)
    
    current_centers = initial_centers.copy()
    current_radii, current_sum = lp_max_radii(current_centers)
    
    best_centers = current_centers.copy()
    best_radii = current_radii.copy()
    best_sum = current_sum
    
    temperature_schedule = [
        (0.02, 0.05, 2000),
        (0.01, 0.03, 4000),
        (0.005, 0.015, 3000),
        (0.002, 0.008, 2000)
    ]
    
    for temp, sigma, iterations in temperature_schedule:
        for _ in range(iterations):
            new_centers = perturb_centers(current_centers, sigma)
            new_radii, new_sum = lp_max_radii(new_centers)
            
            if new_sum > current_sum or np.random.rand() < np.exp((new_sum - current_sum) / temp):
                current_centers = new_centers
                current_radii = new_radii
                current_sum = new_sum
                
                if current_sum > best_sum:
                    best_centers = current_centers.copy()
                    best_radii = current_radii.copy()
                    best_sum = current_sum
                    
    return best_centers, best_radii, best_sum

def local_refine(centers, radii):
    """Refine the configuration with trust-constr non-linear optimization."""
    n = centers.shape[0]
    # Set up the full constrained optimization problem
    # Minimize -sum(r_i) via scipy.optimize.minimize
    # Constraints:
    # -0 <= x_i <= 1, 0 <= y_i <= 1
    # r_i >= r_i^{LP}(c) (lower bounds on radii)
    # r_i >= 0
    # (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2 for all i < j
    # x_i - r_i >= 0, x_i + r_i <= 1, y_i - r_i >= 0, y_i + r_i <= 1
    
    def obj_and_grad(x_and_r):
        x_and_r = np.array(x_and_r)
        x = x_and_r[:2*n].reshape((n, 2))
        r = x_and_r[2*n:]
        return -np.sum(r), -np.ones_like(r)
    
    # Helper to construct inequality constraints for overlap
    def overlap_constr(j, i):
        def c(x_and_r):
            x = x_and_r[:2*n].reshape((n, 2))
            r = x_and_r[2*n:]
            x_i = x[i]
            x_j = x[j]
            r_i = r[i]
            r_j = r[j]
            dist_sq = np.sum((x_i - x_j)**2)
            return dist_sq - (r_i + r_j)**2
        return c
    
    # Check if the start is acceptable (no duplicate)
    unique, counts = np.unique(np.round(centers, 8), axis=0, return_counts=True)
    duplicates = np.sum(counts > 1)
    if duplicates > 0:
        print("Warning: duplicate initial centers found.")
    
    # Initialize variables: [x_1, y_1, ..., x_n, y_n, r_1, ..., r_n]
    start = np.zeros(2*n + n)
    start[:2*n] = centers.flatten()
    start[2*n:] = radii
    
    # Bounds for variables
    bounds_list = []
    for i in range(n):
        bounds_list.append((0, 1))  # x_i
        bounds_list.append((0, 1))  # y_i
    for i in range(n):
        bounds_list.append((radii[i], None))  # r_i >= LP radius

    # Constraints
    cons_list = []
    
    # Wall constraints: [-r_i + x_i >= 0, x_i + r_i <= 1, -r_i + y_i >= 0, y_i + r_i <= 1]
    for i in range(n):
        def wall_constr_1(x_and_r):
            x = x_and_r[:2*n].reshape((n, 2))
            r = x_and_r[2*n:]
            return x[i, 0] - r[i]
        cons_list.append({'type': 'ineq', 'fun': wall_constr_1})
        
        def wall_constr_2(x_and_r):
            x = x_and_r[:2*n].reshape((n, 2))
            r = x_and_r[2*n:]
            return 1.0 - (x[i, 0] + r[i])
        cons_list.append({'type': 'ineq', 'fun': wall_constr_2})
        
        def wall_constr_3(x_and_r):
            x = x_and_r[:2*n].reshape((n, 2))
            r = x_and_r[2*n:]
            return x[i, 1] - r[i]
        cons_list.append({'type': 'ineq', 'fun': wall_constr_3})
        
        def wall_constr_4(x_and_r):
            x = x_and_r[:2*n].reshape((n, 2))
            r = x_and_r[2*n:]
            return 1.0 - (x[i, 1] + r[i])
        cons_list.append({'type': 'ineq', 'fun': wall_constr_4})
    
    # Non-overlap constraints: (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
    # Create a list of all possible (i,j), i < j pairs
    for i in range(n):
        for j in range(i+1, n):
            cons_list.append({'type': 'ineq', 'fun': overlap_constr(j, i)})
    
    res = minimize(
        obj_and_grad,
        start,
        method="trust-constr",
        bounds=bounds_list,
        constraints=cons_list,
        jac=True,
        options={'maxiter': 500, 'disp': False}
    )
    
    if res.success:
        refined_centers = res.x[:2*n].reshape((n, 2))
        refined_radii = res.x[2*n:]
        refined_sum = -res.fun
    else:
        # Fall back to original
        refined_centers = centers
        refined_radii = radii
        refined_sum = np.sum(radii)
        
    return refined_centers, refined_radii, refined_sum

def validate_and_return(centers, radii):

def run_packing():
    initial_centers = build_hex_lattice(26)
    annealed_centers, annealed_radii, annealed_sum = anneal_centers(initial_centers)
    final_centers, final_radii, final_sum = local_refine(annealed_centers, annealed_radii)
