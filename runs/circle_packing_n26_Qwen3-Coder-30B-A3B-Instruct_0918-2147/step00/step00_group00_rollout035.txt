import numpy as np
from scipy.optimize import linprog

def run_packing():
    # --- 1. Generate initial hexagonal layout ---
    n = 26
    margin = 0.05
    grid_spacing = 0.8 / np.ceil(np.sqrt(n))
    sqrt3 = np.sqrt(3)
    
    centers = []
    j = 0
    while len(centers) < n:
        for i in range(int(1.0 / grid_spacing) + 1):
            x = i * grid_spacing + (j % 2) * grid_spacing / 2
            y = j * grid_spacing * sqrt3 / 2
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers.append([x, y])
        j += 1
    
    centers = np.array(centers[:n])
    
    # Shift so that margin is maintained
    centers[:, 0] += margin - np.min(centers[:, 0])
    centers[:, 1] += margin - np.min(centers[:, 1])
    
    # Ensure all fit into unit square
    centers[:, 0] = np.clip(centers[:, 0], margin, 1 - margin)
    centers[:, 1] = np.clip(centers[:, 1], margin, 1 - margin)
    
    # Initial small radii
    radii = np.full(n, 0.01)
    
    # --- 2. Parameters ---
    max_iterations = 30
    tolerance = 1e-6
    k = 8  # polygonal approximation chunks per circle
    cos_pi_k = np.cos(np.pi / k)
    
    # Precompute direction vectors for polygonal outer linearization
    angles = np.linspace(0, 2 * np.pi, k, endpoint=False)
    directions = np.column_stack((np.cos(angles), np.sin(angles)))
    
    # --- 3. Alternating Linear Programming Loop ---
    last_sum = 0.0
    
    for iteration in range(max_iterations):
        # Current sum of radii
        current_sum = np.sum(radii)
        if abs(current_sum - last_sum) < tolerance:
            break
        last_sum = current_sum
        
        # *** Solve radius LP for fixed centers ***
        # Wall constraints
        A_ub = []
        b_ub = []
        for i in range(n):
            A_ub.append([1, 0, 0])         # r_i <= x_i
            b_ub.append(centers[i, 0])
            A_ub.append([-1, 0, 0])        # r_i <= 1 - x_i
            b_ub.append(1 - centers[i, 0])
            A_ub.append([0, 1, 0])         # r_i <= y_i
            b_ub.append(centers[i, 1])
            A_ub.append([0, -1, 0])        # r_i <= 1 - y_i
            b_ub.append(1 - centers[i, 1])
        
        # Separation constraints
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                A_ub.append([1, 1, 0])      # r_i + r_j <= dist
                b_ub.append(dist)
        
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        
        # Linear program: maximize sum of radii
        c = np.ones(n)
        
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'tol': 1e-10})
        if not result.success:
            # Try to recover
            radii = np.clip(0.001, 0.0, 0.5)
        else:
            radii = np.clip(result.x, 0.0, 0.5)
        
        # *** Solve center LP for fixed radii ***
        # Formulate new LP to push centers outward
        # Variables: [x1,y1,x2,y2,...,x26,y26,s1,...,s26]
        n_vars = 2 * n + n
        
        # Objective: maximize sum of slacks (s1 + ... + s26)
        c_new = np.zeros(n_vars)
        c_new[-n:] = 1  # coefficients for slack variables s_i
        
        # Build constraints
        # Wall constraints for each center (with slack): c_i.x - r_i >= s_i,
        # c_i.y - r_i >= s_i, 1-c_i.x - r_i >= s_i, 1-c_i.y - r_i >= s_i
        
        A_ub_new = []
        b_ub_new = []
        
        for i in range(n):
            # Left wall
            A_row = np.zeros(n_vars)
            A_row[i*2] = 1  # x_i
            A_row[n + i] = -1  # -r_i
            A_row[n + n + i] = -1  # -s_i
            A_ub_new.append(A_row)
            b_ub_new.append(radii[i])
            
            # Right wall
            A_row = np.zeros(n_vars)
            A_row[i*2] = -1
            A_row[n + i] = -radii[i]
            A_row[n + n + i] = -1
            A_ub_new.append(A_row)
            b_ub_new.append(-1 + radii[i])
            
            # Bottom wall
            A_row = np.zeros(n_vars)
            A_row[i*2 + 1] = 1
            A_row[n + i] = -radii[i]
            A_row[n + n + i] = -1
            A_ub_new.append(A_row)
            b_ub_new.append(radii[i])
            
            # Top wall
            A_row = np.zeros(n_vars)
            A_row[i*2 + 1] = -1
            A_row[n + i] = -radii[i]
            A_row[n + n + i] = -1
            A_ub_new.append(A_row)
            b_ub_new.append(-1 + radii[i])
        
        # Pairwise separation constraints (polygonal linearization)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist_ij = np.sqrt(dx**2 + dy**2)
                r_sum_ij = radii[i] + radii[j]
                
                for u in directions:
                    # u^T * (c_i - c_j) <= - r_sum_ij * cos(π/k)
                    A_row = np.zeros(n_vars)
                    A_row[i*2] = u[0]
                    A_row[i*2 + 1] = u[1]
                    A_row[j*2] = -u[0]
                    A_row[j*2 + 1] = -u[1]
                    A_row[n + n] = -1  # s_i
                    A_row[n + n + i] = -1  # s_j
                    A_ub_new.append(A_row)
                    # The offset from paper: - r_sum_ij * cos(π/k)
                    b_ub_new.append(-r_sum_ij * cos_pi_k)
        
        # Convert to arrays
        A_ub_new = np.array(A_ub_new)
        b_ub_new = np.array(b_ub_new)
        
        # Solve the center LP
        try:
            res = linprog(c_new, A_ub=A_ub_new, b_ub=b_ub_new, method='highs', options={'tol': 1e-10})
            if res.success:
                sol = res.x
                # Extract updated centers and mediate overwriting of data
                centers_new = np.zeros((n, 2))
                for i in range(n):
                    centers_new[i, 0] = sol[i * 2]
                    centers_new[i, 1] = sol[i * 2 + 1]
                centers = centers_new
        except:
            # Fallback
            pass
            
        centers = np.clip(centers, 0, 1)
    
    # --- 4. Local Inflation ---
    for _ in range(5):
        converged = True
        updated = False
        for i in range(n):
            # Compute minimal exact gap to each neighbor
            min_gap = float("inf")
            for j in range(n):
                if i == j:
                    continue
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                gap = dist - radii[i] - radii[j]
                if gap < min_gap:
                    min_gap = gap
            
            # Wall gaps
            wall_gaps = [
                centers[i, 0] - radii[i],
                1 - centers[i, 0] - radii[i],
                centers[i, 1] - radii[i],
                1 - centers[i, 1] - radii[i]
            ]
            min_wall = np.min(wall_gaps)
            
            # Take the minimum
            gap = min(min_gap, min_wall)
            
            # If significant extra space, expand radii
            if gap > 1e-5:
                inflate = 0.4 * gap
                if inflate > 1e-5:
                    old_r = radii[i]
                    new_r = old_r + inflate
                    # Clamp at reasonable limit
                    radios[i] = min(new_r, 0.5)
                    updated = True
                    converged = False
        
        # If no updates happened, stop
        if converged:
            break
    
    # --- 5. Final validation ---
    # Verify packing
    valid, msg = validate_packing(centers, radii)
    if not valid:
        # Try to shrink all radii slightly
        shrink = 1e-7
        radii *= (1 - shrink)
        valid, msg = validate_packing(centers, radii)
        if not valid:
            # Make the smallest adjustment
            radii *= 0.999999
            valid, msg = validate_packing(centers, radii)
            
            # If still invalid, raise error
            if not valid:
                raise RuntimeError("Failed validation after shrinking")
    
    total_sum = np.sum(radii)
    return centers, radii, total_sum

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
