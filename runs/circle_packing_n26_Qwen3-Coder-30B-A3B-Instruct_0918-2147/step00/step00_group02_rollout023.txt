import numpy as np
from scipy.optimize import minimize
from ortools.linear_solver import pywraplp

def generate_candidates(g, radius_levels):
    candidates = []
    grid_points = np.arange(0, 1 + g/2, g)
    for x in grid_points:
        for y in grid_points:
            for r in radius_levels:
                if (x - r >= 0 and x + r <= 1 and
                    y - r >= 0 and y + r <= 1):
                    candidates.append((x, y, r))
    return candidates

def build_milp(candidates):
    solver = pywraplp.Solver.CreateSolver('SCIP')
    if not solver:
        solver = pywraplp.Solver.CreateSolver('CBC')

    n = len(candidates)
    z = [solver.IntVar(0, 1, f'z_{i}') for i in range(n)]

    # One circle per grid point
    grid_map = {}
    for i, (x, y, r) in enumerate(candidates):
        grid_key = (round(x/g), round(y/g))
        if grid_key not in grid_map:
            grid_map[grid_key] = []
        grid_map[grid_key].append(i)

    for key, indices in grid_map.items():
        solver.Add(solver.Sum([z[i] for i in indices]) <= 1)

    # Exactly 26 circles
    solver.Add(solver.Sum(z) == 26)

    # Pairwise non-overlap using L∞ constraints
    dx_vars = {}
    dy_vars = {}
    for i in range(n):
        for j in range(i + 1, n):
            ci = candidates[i]
            cj = candidates[j]
            x1, y1, r1 = ci
            x2, y2, r2 = cj

            dx = solver.NumVar(0, 1, f'dx_{i}_{j}')
            dy = solver.NumVar(0, 1, f'dy_{i}_{j}')

            dx_vars[(i, j)] = dx
            dy_vars[(i, j)] = dy

            # dx >= |x1 - x2|, dx >= -(x1 - x2)
            solver.Add(dx >= x1 - x2)
            solver.Add(dx >= x2 - x1)

            # dy >= |y1 - y2|, dy >= -(y1 - y2)
            solver.Add(dy >= y1 - y2)
            solver.Add(dy >= y2 - y1)

            # dx + dy >= r1 + r2
            solver.Add(dx + dy >= r1 + r2)

            # dx - dy >= r1 + r2 (after rotation)
            solver.Add(dx - dy >= r1 + r2 - 1e-3)

            # Big-M linking
            solver.Add(dx <= 1 * (z[i] + z[j]))
            solver.Add(dy <= 1 * (z[i] + z[j]))

    # Objective
    solver.Maximize(solver.Sum([z[i] * candidates[i][2] for i in range(n)]))

    return solver, z

def solve_milp(solver, z):
    status = solver.Solve()
    if status != pywraplp.Solver.OPTIMAL:
        raise Exception("MILP solver did not find an optimal solution")

    selected_indices = [i for i in range(len(z)) if z[i].solution_value() > 0.5]
    return selected_indices

def extract_initial_solution(selected_indices, candidates):
    centers = np.array([[candidates[i][0], candidates[i][1]] for i in selected_indices])
    radii = np.array([candidates[i][2] for i in selected_indices])
    return centers, radii

def shrink_initial(radii, delta=1e-5):
    return np.maximum(radii - delta, 0)

def pack_variables(centers0, radii0):
    return np.hstack([centers0.flatten(), radii0])

def unpack_solution(z_vec, n_circles):
    centers = z_vec[:2*n_circles].reshape(-1, 2)
    radii = z_vec[2*n_circles:]
    return centers, radii

def define_objective(z):
    return -np.sum(z[2*n_circles:])

def define_jacobian(z):
    n_circles = len(z) // 3
    jacobian = np.full(len(z), -1.0)
    return jacobian

def define_constraints(z, centers0, radii0, pairs):
    n_circles = len(z) // 3
    constraints = []
    
    centers, radii = unpack_solution(z, n_circles)
    
    # Boundary constraints
    for i in range(n_circles):
        constraints.append({
            'type': 'ineq',
            'fun': lambda z, i=i: z[2*n_circles+i] - z[i*2] + 1e-15,  # r - x
            'jac': lambda z, i=i: np.concatenate([np.zeros(2*n_circles), np.zeros(i) + 1, np.zeros(n_circles-i-1)])
        })
        constraints.append({
            'type': 'ineq',
            'fun': lambda z, i=i: z[2*n_circles+i] - (1 - z[i*2+1]) + 1e-15,  # r - (1-y)
            'jac': lambda z, i=i: np.concatenate([np.zeros(2*n_circles), np.zeros(i) + 1, np.zeros(n_circles-i-1)])
        })
        constraints.append({
            'type': 'ineq',
            'fun': lambda z, i=i: z[2*n_circles+i] - z[i*2+1] + 1e-15,  # r - y
            'jac': lambda z, i=i: np.concatenate([np.zeros(2*n_circles), np.zeros(i) + 1, np.zeros(n_circles-i-1)])
        })
        constraints.append({
            'type': 'ineq',
            'fun': lambda z, i=i: z[2*n_circles+i] - (1 - z[i*2]) + 1e-15,  # r - (1-x)
            'jac': lambda z, i=i: np.concatenate([np.zeros(2*n_circles), np.zeros(i) + 1, np.zeros(n_circles-i-1)])
        })

    # Non-overlap constraints
    for i, j in pairs:
        constraints.append({
            'type': 'ineq',
            'fun': lambda z, i=i, j=j: (z[2*n_circles+i] + z[2*n_circles+j]) - 
                                       np.sqrt((z[i*2] - z[j*2])**2 + (z[i*2+1] - z[j*2+1])**2) + 1e-15,
            'jac': lambda z, i=i, j=j: np.concatenate([
                np.array([(z[2*n_circles+i] + z[2*n_circles+j])*(z[i*2] - z[j*2]) / 
                          (np.sqrt((z[i*2] - z[j*2])**2 + (z[i*2+1] - z[j*2+1])**2) + 1e-20) - 1, 
                         (z[2*n_circles+i] + z[2*n_circles+j])*(z[i*2+1] - z[j*2+1]) / 
                          (np.sqrt((z[i*2] - z[j*2])**2 + (z[i*2+1] - z[j*2+1])**2) + 1e-20) - 1,
                         (z[2*n_circles+i] + z[2*n_circles+j])*(z[j*2] - z[i*2]) / 
                          (np.sqrt((z[i*2] - z[j*2])**2 + (z[i*2+1] - z[j*2+1])**2) + 1e-20) - 1, 
                         (z[2*n_circles+i] + z[2*n_circles+j])*(z[j*2+1] - z[i*2+1]) / 
                          (np.sqrt((z[i*2] - z[j*2])**2 + (z[i*2+1] - z[j*2+1])**2) + 1e-20) - 1]),
                np.zeros(n_circles) - 1
            ])

    return constraints

def run_packing():
    # Generate candidates with grid spacing 0.03 and radii from 0.02 to 0.25
    g = 0.03
    radius_levels = np.arange(0.02, 0.26, 0.02)
    candidates = generate_candidates(g, radius_levels)

    # Build MILP
    solver, z = build_milp(candidates)

    # Solve MILP
    selected_indices = solve_milp(solver, z)

    # Extract initial solution
    centers0, radii0 = extract_initial_solution(selected_indices, candidates)

    # Shrink initial radii slightly
    radii0 = shrink_initial(radii0)

    # Pack initial solution
    z0 = pack_variables(centers0, radii0)

    # Create pairs for overlap constraints (simplified for bulk use)
    pairs = []
    for i in range(len(centers0)):
        for j in range(i + 1, len(centers0)):
            pairs.append((i, j))

    # Optimize using scipy
    bounds = [(0, 1)] * (2 * len(centers0)) + [(0, 0.5)] * len(radii0)
    
    # Custom function to minimize
    def obj_fn(z):
        return -np.sum(z[2*len(centers0):])
    
    def jac_fn(z):
        return np.full(len(z), -1.0)
    
    constraints = define_constraints(z, centers0, radii0, pairs)
    
    # Run optimizer
    result = minimize(obj_fn, z0, method='trust-constr', jac=jac_fn, bounds=bounds, constraints=constraints, options={'maxiter': 20000})

    # Unpack final solution
    centers, radii = unpack_solution(result.x, len(centers0))

    # Repair at end
    for i in range(len(centers)):
        radii[i] = min(radii[i], centers[i][0], 1 - centers[i][0], centers[i][1], 1 - centers[i][1])

    sum_radii = np.sum(radii)
