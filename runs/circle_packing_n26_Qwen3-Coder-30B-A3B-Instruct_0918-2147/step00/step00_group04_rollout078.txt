import numpy as np
from scipy.optimize import linprog

def initial_hex_centers(k, spacing):
    sqrt3 = np.sqrt(3)
    centers = []
    i = j = 0
    while len(centers) < k:
        x = i * spacing + (j % 2) * spacing / 2
        y = j * spacing * sqrt3 / 2
        if 0 <= x <= 1 and 0 <= y <= 1:
            centers.append([x, y])
        if x + spacing > 1 or y + spacing * sqrt3 / 2 > 1:
            j += 1
            i = -1
        i += 1
    return np.array(centers[:k])

def initial_radii(k, r0):
    return np.full(k, r0)

def build_lp_constants(centers, radii):
    n = centers.shape[0]
    # Wall distances
    d_left = centers[:, 0]
    d_right = 1 - centers[:, 0]
    d_bottom = centers[:, 1]
    d_top = 1 - centers[:, 1]
    wall_distances = np.stack([d_left, d_right, d_bottom, d_top], axis=1)

    # Pairwise distances
    pairwise_dists = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            pairwise_dists[i, j] = dist
            pairwise_dists[j, i] = dist

    return wall_distances, pairwise_dists

def solve_radius_lp(consts, radii, graph, alpha):
    wall_distances, pairwise_dists = consts
    n = len(radii)
    # Constraint matrix for LP
    A_ub = []
    b_ub = []
    # Wall constraints
    for i in range(n):
        max_growth = np.min(wall_distances[i]) - radii[i]
        max_growth = max(0.0, max_growth)
        A_ub.append([-1.0])
        b_ub.append(-max_growth)
    # Pairwise constraints
    for i in range(n):
        for j in range(i + 1, n):
            d_ij = pairwise_dists[i, j]
            max_growth = d_ij - (radii[i] + radii[j])
            if max_growth >= 0:
                A_ub.append([1.0, 1.0])
                b_ub.append(max_growth)
            else:
                # Constraint would be violated, treat as non-binding
                pass

    # Optional graph-based constraints: Δr_i ≤ α * Σ_{j: edge(i,j)} (D_ij - (r_i + r_j))
    # We'll just pool all potential energy in the legend of a single constraint, adding this as a very large penalty
    # since we want the total energy to be maximized, but we also want some control over the growth distribution
    # In practice we'll disable this in most cases to simplify

    # Objective: maximize sum(Δr_i)  -> minimize sum(-Δr_i)
    c = np.full(n, -1.0)

    # Solve
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
        if result.success:
            return result.x
        else:
            return np.zeros(n)
    except Exception:
        return np.zeros(n)

def apply_increment(radii, Δr):
    return radii + Δr

def push_feasibility(centers, radii):
    n = centers.shape[0]
    new_centers = centers.copy()
    new_radii = radii.copy()
    changed = True

    # Keep iterating until no changes
    while changed:
        changed = False
        for i in range(n):
            # Wall checks
            if new_radii[i] >= new_centers[i, 0]:
                new_radii[i] = new_centers[i, 0]
                changed = True
            if new_radii[i] >= 1 - new_centers[i, 0]:
                new_radii[i] = 1 - new_centers[i, 0]
                changed = True
            if new_radii[i] >= new_centers[i, 1]:
                new_radii[i] = new_centers[i, 1]
                changed = True
            if new_radii[i] >= 1 - new_centers[i, 1]:
                new_radii[i] = 1 - new_centers[i, 1]
                changed = True

        # Check pairwise overlaps
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((new_centers[i] - new_centers[j])**2))
                if dist < new_radii[i] + new_radii[j] - 1e-12:
                    overlap = new_radii[i] + new_radii[j] - dist
                    disp = (new_centers[i] - new_centers[j]) * overlap / (2 * dist)
                    new_centers[i] -= disp
                    new_centers[j] += disp
                    changed = True
                    # Clamp to prevent going out of bounds
                    new_centers[i] = np.clip(new_centers[i], [0, 0], [1, 1])
                    new_centers[j] = np.clip(new_centers[j], [0, 0], [1, 1])

    return new_centers, new_radii

def run_iterations(centers, radii, max_iter=100, ε=1e-5, α=0.5):
    n = centers.shape[0]
    graph = {}
    
    for it in range(max_iter):
        # compute LP constants
        consts = build_lp_constants(centers, radii)
        # solve
        Δr = solve_radius_lp(consts, radii, graph, α)
        # update radii
        radii = apply_increment(radii, Δr)
        # push for feasibility
        centers, radii = push_feasibility(centers, radii)
        # Check stopping criteria
        totalΔ = np.sum(Δr)
        if totalΔ < ε:
            break

    return centers, radii

def final_refinement(centers, radii):
    # Re-solve with tight tolerances, then do final clipping/clamping
    consts = build_lp_constants(centers, radii)
    # Solve with stricter parameters
    try:
        Δr = solve_radius_lp(consts, radii, {}, 0.5)
        new_radii = apply_increment(radii, Δr)
        # Final pass over ranges
        bounds = np.stack([new_radii, new_radii, new_radii, new_radii], axis=1)
        bounds[:, 0] = np.minimum(bounds[:, 0], centers[:, 0])
        bounds[:, 1] = np.minimum(bounds[:, 1], 1 - centers[:, 0])
        bounds[:, 2] = np.minimum(bounds[:, 2], centers[:, 1])
        bounds[:, 3] = np.minimum(bounds[:, 3], 1 - centers[:, 1])
        new_radii = np.min(bounds, axis=1)
        # final check
        for i in range(len(new_radii)):
            for j in range(i + 1, len(new_radii)):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < new_radii[i] + new_radii[j] - 1e-12:
                    overlap = new_radii[i] + new_radii[j] - dist
                    new_radii[i] -= overlap / 2 + 1e-13
                    new_radii[j] -= overlap / 2 + 1e-13
        return centers, new_radii
    except Exception:
        return centers, radii
