import numpy as np
from scipy.optimize import minimize
import scipy.sparse as sp

def init_hex_grid(n=26, r_init=0.04):
    # Generate hexagonal grid points
    rows = []
    y = r_init
    row_idx = 0
    while len(rows) < n:
        # Alternate row offset
        x_offset = (row_idx % 2) * r_init
        x = x_offset + r_init
        row = []
        while x <= 1 - r_init and len(rows) < n:
            row.append([x, y])
            x += 2 * r_init
            if len(rows) >= n:
                break
        if row:
            rows.extend(row)
        y += np.sqrt(3) * r_init
        row_idx += 1
        if y > 1 - r_init:
            break
    
    # Take first n points and ensure they're within bounds
    centers = np.array(rows[:n])
    # Ensure all centers are valid
    centers = np.clip(centers, r_init, 1 - r_init)
    radii = np.full(n, r_init)
    return centers, radii

def max_radii(centers, radii):
    # Iteratively improve radii to maximum allowed values
    max_iter = 100
    tol = 1e-8
    for _ in range(max_iter):
        # Wall constraints
        wall = np.minimum(centers, 1 - centers).min(axis=1)
        # Neighbour constraints
        D = np.sqrt(((centers[:,None,:] - centers[None,:,:])**2).sum(-1))
        np.fill_diagonal(D, np.inf)
        neighbor = D - radii[:,None]
        # Min over neighbours
        neighbor_min = np.minimum.reduce(neighbor, axis=1)
        new_radii = np.minimum(wall, neighbor_min)
        new_radii = np.maximum(new_radii, 0)
        if np.max(np.abs(new_radii - radii)) < tol:
            break
        radii = new_radii
    return radii

def hill_climb(centers, radii, iterations=5000, step=0.01):
    # Simple random hill climbing
    for _ in range(iterations):
        i = np.random.randint(0, len(centers))
        old_sum = np.sum(radii)
        old_r = radii[i]
        old_c = centers[i].copy()
        
        # Try a random move
        delta = np.random.uniform(-step, step, size=2)
        new_c = old_c + delta
        
        # Enforce bounds
        new_c = np.clip(new_c, radii[i], 1 - radii[i])
        
        # Accept if improvement in sum
        centers[i] = new_c
        radii = max_radii(centers, radii)
        new_sum = np.sum(radii)
        if new_sum <= old_sum:
            # Reject move
            centers[i] = old_c
            radii[i] = old_r
            radii = max_radii(centers, radii)
    return centers, radii

def refine_with_scipy(centers, radii):

    n = len(centers)
    x0 = np.concatenate([centers.flatten(), radii])
    
    def objective(x):
        return -np.sum(x[n*2:])
    
    # Build constraints
    constraints = []
    
    # Wall constraints: r_i - x_i <= 0, r_i + x_i - 1 <= 0, etc.
    # r_i - x_i <= 0
    def wall_constraints_upper(x):
        r = x[n*2:]
        c = x[:n*2].reshape(-1, 2)
        return r - c[:,0]
    
    def wall_constraints_lower(x):

        c = x[:n*2].reshape(-1, 2)
        return r + c[:,0] - 1
    
    def wall_constraints_upper_y(x):
        r = x[n*2:]
        c = x[:n*2].reshape(-1, 2)
        return r - c[:,1]
    
    def wall_constraints_lower_y(x):

        c = x[:n*2].reshape(-1, 2)
        return r + c[:,1] - 1
    
    # Non-overlap constraints
    def non_overlap(x):

        c = x[:n*2].reshape(-1, 2)

    constraints.append(NonlinearConstraint(wall_constraints_lower, -np.inf, 0))
    constraints.append(NonlinearConstraint(wall_constraints_upper_y, -np.inf, 0))
    constraints.append(NonlinearConstraint(wall_constraints_lower_y, -np.inf, 0))
    
    # Add overlap constraints
    # Extract matrix indices including non-diagonal entries
    overlaps = []
    for i in range(n):
        for j in range(i+1, n):
            overlaps.append((i,j))
    # Try to avoid large data-structures
    # Instead just assert overlap constraints by writing them as vector function
    # This uses sparse construction to reduce size
    def overlap_all(x):
        r = x[n*2:]
        c = x[:n*2].reshape(-1, 2)
        D = np.sqrt(((c[:,None,:] - c[None,:,:])**2).sum(-1))
        D = D + np.eye(n) * 1e10  # Avoid diagonal
        return r[:,None] + r[None,:] - D
    
    # Constraint for ALL pairs: Diagonal not in the matrix (no self-overlap),
    # elements of constraint vector correspond to C[i,j] with i<j
    def overlap_function_all(x):
        r = x[n*2:]
        c = x[:n*2].reshape(-1, 2)
        D = np.sqrt(((c[:,None,:] - c[None,:,:])**2).sum(-1))
        # Layer a big number on diagonal so it's ignored
        np.fill_diagonal(D, 1e10)
        # For any pair (i,j) where i<j, we require r_i + r_j <= D_ij
        # Therefore D_ij - r_i - r_j >= 0 in constraint vector
        # This gives us a vector of size (n)(n-1)/2
        # First convert to upper triangular (since for i<j we want Cij)
        idxs = np.triu_indices(n, k=1)
        pairs = list(zip(idxs[0], idxs[1]))
        vec = []
        for i, j in pairs:
            dist = D[i,j]
            vec.append(dist - r[i] - r[j])
        return np.array(vec)
    
    constraints.append(NonlinearConstraint(overlap_function_all, 0, np.inf))

    # Bounds: x ranges from 0 to 1 for coordinates, 0 to 0.5 for radii
    bounds = [(0, 1)] * (n*2) + [(0, 0.5)] * n
    
    # Minimize using trust-constr
    result = minimize(objective, x0, bounds=bounds, constraints=constraints, 
                      method='trust-constr', options={'maxiter':200})

    # Extract final solution
    final_centers = result.x[:n*2].reshape(-1, 2)
    final_radii = result.x[n*2:]
    
    return final_centers, final_radii

def run_packing():
    # Start with hex grid
    centers, radii = init_hex_grid()
    
    # Maximise radii with fixed centers
    radii = max_radii(centers, radii)
    
    # Attempt stochastic hill climbing
    centers, radii = hill_climb(centers, radii)
    
    # Final global refinement
    centers, radii = refine_with_scipy(centers, radii)
    
    # Validate
    success, msg = validate_packing(centers, radii)
    if not success:
        raise RuntimeError(f"Validation failed: {msg}")
    
    sum_r = float(radii.sum())
    return centers, radii, sum_r

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
