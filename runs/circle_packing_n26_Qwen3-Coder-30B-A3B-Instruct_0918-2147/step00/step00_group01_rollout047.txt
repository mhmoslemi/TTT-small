import numpy as np
from scipy.optimize import minimize

def initial_layout():
    # Hexagonal grid with 5 rows x 5 columns = 25 circles
    # Adjust spacing to allow for more efficient packing
    spacing = 0.18
    radius = spacing / 2 - 0.01
    
    centers = []
    for i in range(5):
        for j in range(5):
            x = (j + 0.5) * spacing
            y = (i + 0.5) * spacing
            # Adjust to be within square if needed
            x = max(radius, min(1 - radius, x))
            y = max(radius, min(1 - radius, y))
            centers.append([x, y])
    
    # Add one more circle in the center
    centers.append([0.5, 0.5])
    centers_arr = np.array(centers)
    radii_arr = np.full(26, radius)
    return centers_arr, radii_arr

def pack_vector(centers, radii):
    # Flatten centers and radii into a single vector
    z = np.zeros(3 * 26)
    for i in range(26):
        z[3*i] = centers[i, 0]
        z[3*i + 1] = centers[i, 1]
        z[3*i + 2] = radii[i]
    return z

def unpack_vector(z):
    # Extract centers and radii from vector
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    for i in range(26):
        centers[i, 0] = z[3*i]
        centers[i, 1] = z[3*i + 1]
        radii[i] = z[3*i + 2]
    return centers, radii

def objective(z):
    centers, radii = unpack_vector(z)
    return -np.sum(radii)

def constraint_boundary(z):
    centers, radii = unpack_vector(z)
    constraints = []
    for i in range(26):
        x, y = centers[i]
        r = radii[i]
        # x - r >= 0
        constraints.append(x - r)
        # 1 - x - r >= 0
        constraints.append(1 - x - r)
        # y - r >= 0
        constraints.append(y - r)
        # 1 - y - r >= 0
        constraints.append(1 - y - r)
    return np.array(constraints)

def constraint_nonoverlap(z):
    # Return array of non-overlap constraints (>= 0)
    centers, radii = unpack_vector(z)
    constraints = []
    for i in range(26):
        for j in range(i + 1, 26):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx**2 + dy**2
            r_sum = radii[i] + radii[j]
            # Non-overlap constraint (square version): dist^2 >= (r_i + r_j)^2
            constraints.append(dist_sq - r_sum**2)
    return np.array(constraints)

def run_packing():
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    for _ in range(3):
        centers, radii = initial_layout()
        # Slightly randomize center positions for better exploration
        for i in range(26):
            centers[i, 0] += (np.random.rand() - 0.5) * 0.02
            centers[i, 1] += (np.random.rand() - 0.5) * 0.02
            centers[i, 0] = np.clip(centers[i, 0], 0, 1)
            centers[i, 1] = np.clip(centers[i, 1], 0, 1)
        
        z0 = pack_vector(centers, radii)
        
        # Define bounds and constraints
        bounds = [(0, 1) for _ in range(26 * 3)]  # x, y, r for each circle
        constraints = [
            {'type': 'ineq', 'fun': constraint_boundary},
            {'type': 'ineq', 'fun': constraint_nonoverlap}
        ]
        
        # Solver
        res = minimize(
            objective,
            z0,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'ftol': 1e-9, 'maxiter': 2000}
        )
        
        if res.success:
            centers_opt, radii_opt = unpack_vector(res.x)
            # Clip any negative radii that may arise from numerical errors
            radii_opt = np.maximum(radii_opt, 0)
            sum_radii = np.sum(radii_opt)
            if sum_radii > best_sum:
                best_sum = sum_radii
                best_centers = centers_opt.copy()
                best_radii = radii_opt.copy()

    if not valid:
        raise ValueError(f"Final packing is invalid: {msg}")
    
    return best_centers, best_radii, best_sum
