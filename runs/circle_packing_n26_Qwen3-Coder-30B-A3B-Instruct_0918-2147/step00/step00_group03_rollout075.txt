import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
from itertools import combinations

def init_candidate_grid(R, Δ):
    """Generate candidate (center, radius) pairs with spacing adjusted to radius."""
    candidates = []
    for r in R:
        # Create a grid but leave margin r from all boundaries
        grid_spacing = r * np.sqrt(2)  # Spacing for non-overlap of same-radius circles
        x_coords = np.arange(r, 1 - r + 1e-12, grid_spacing)
        y_coords = np.arange(r, 1 - r + 1e-12, grid_spacing)
        
        for x in x_coords:
            for y in y_coords:
                candidates.append((x, y, r))
    return candidates

def build_overlap_matrix(candidates):
    """
    Return a boolean matrix indicating which pairs of candidates overlap.
    """
    n = len(candidates)
    overlap = np.zeros((n, n), dtype=bool)
    coords = np.array([[c[0], c[1]] for c in candidates])
    radii = np.array([c[2] for c in candidates])
    
    # Compute pair-wise distances
    dists = cdist(coords, coords)
    
    # Mark overlaps (distance < sum radii)
    for i in range(n):
        for j in range(i+1, n):
            dist = dists[i, j]
            r1 = radii[i]
            r2 = radii[j]
            if dist < r1 + r2:
                overlap[i, j] = True
                overlap[j, i] = True
    return overlap

def solve_discrete_bip(candidates, overlap_matrix, target_sum):
    """
    Use a greedy heuristic to get a good starting solution and then 
    refine further using max sum of core radii guaranteed to meet target.
    """
    import pulp
    
    n = len(candidates)
    
    # Use pulp MILP solver for discrete set packing
    prob = pulp.LpProblem("CirclePacking", pulp.LpMaximize)
    
    # Variable for each candidate (binary)
    z = pulp.LpVariable.dicts("selected", range(n), cat="Binary")
    
    # Objective: maximize sum of radii
    prob += pulp.lpSum([candidates[i][2] * z[i] for i in range(n)])
    
    # Constraint: exactly 26 circles selected
    prob += pulp.lpSum([z[i] for i in range(n)]) == 26
    
    # Overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            if overlap_matrix[i, j]:
                prob += z[i] + z[j] <= 1
    
    # Optional: solve with low tolerance
    prob.solve(pulp.PULP_CBC_CMD(msg=0, timeLimit=10, threads=1))
    
    # Extract selected indices
    selected_indices = []
    for i in range(n):
        if z[i].varValue > 0.5:
            selected_indices.append(i)
    
    # Return only the first 26 if there are more due to duplicates or solver error
    if len(selected_indices) != 26:
        # Fallback: greedy selection
        selected_indices = []
        used_centers = set()
        radii = [c[2] for c in candidates]
        
        # Sort all candidates by radius descending
        sorted_cand = [(i, radius) for i, radius in enumerate(radii)]
        sorted_cand.sort(key=lambda x: x[1], reverse=True)
        
        selected_indices = []
        used_centers = set()
        
        for i, _ in sorted_cand:
            if len(selected_indices) >= 26:
                break
                
            x, y, r = candidates[i]
            is_valid = True
            for (x2, y2, r2) in [candidates[j] for j in selected_indices]:
                dist_sq = (x - x2)**2 + (y - y2)**2
                if dist_sq < (r + r2)**2:
                    is_valid = False
                    break
                    
            if is_valid:
                selected_indices.append(i)
    
    # Ensure exactly 26
    if len(selected_indices) > 26:
        selected_indices = selected_indices[:26]
    elif len(selected_indices) < 26:
        # For safety in case of issues, pick any remaining
        to_add = 26 - len(selected_indices)
        available_indices = [i for i in range(n) if i not in selected_indices]
        selected_indices.extend(available_indices[:to_add])
    
    # Ensure exactly 26 and feasible
    if len(selected_indices) != 26:
        raise RuntimeError("Failed to get exactly 26 circles")
    
    centers = np.array([[candidates[i][0], candidates[i][1]] for i in selected_indices])
    radii = np.array([candidates[i][2] for i in selected_indices])
    return centers, radii

def max_radii_fixed_centres(centers, radii_initial):
    """
    For fixed centers, compute maximal radii that satisfy constraints.
    """
    n = centers.shape[0]
    radii = radii_initial.copy()
    
    # For each circle, compute the maximal allowed radius
    for i in range(n):
        x, y = centers[i]
        r_min = min(x, y, 1 - x, 1 - y)  # Wall margin
        # Now check all other circles
        r_actual = r_min
        for j in range(n):
            if i == j:
                continue
            dist = np.sqrt((x - centers[j, 0])**2 + (y - centers[j, 1])**2)
            r_max = dist - radii_initial[j]  # Must be less than this
            r_actual = min(r_actual, r_max)
        radii[i] = r_actual
    
    # Clip negative values
    radii[radii < 0] = 0.0
    return radii

def nlp_refine(centers_initial, radii_initial):
    """
    Refine using continuous optimization to maximize sum of radii.
    """
    def objective(r):
        return -np.sum(r)
    
    def constraint_func(args):
        centers_flat = args[:-26]
        radii = args[-26:]
        
        centers = centers_flat.reshape(-1, 2)
        n = len(centers)
        
        # Wall constraints: 
        # x_i - r_i <= 0
        # -x_i - r_i + 1 <= 0
        # y_i - r_i <= 0
        # -y_i - r_i + 1 <= 0
        walls = []
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            walls.append( x - r )  # Must be <= 0
            walls.append( -x - r + 1 )  # Must be <= 0
            walls.append( y - r )  # Must be <= 0
            walls.append( -y - r + 1 )  # Must be <= 0
            
        # Overlap constraints:
        overlaps = []
        for i in range(n):
            x1, y1 = centers[i]
            r1 = radii[i]
            for j in range(i+1, n):
                x2, y2 = centers[j]
                r2 = radii[j]
                dist = np.sqrt((x1-x2)**2 + (y1-y2)**2)
                c = r1 + r2 - dist  # <= 0
                overlaps.append(c)
        
        return np.array(walls + overlaps)
    
    # Initial values
    args0 = np.concatenate([centers_initial.flatten(), radii_initial])
    
    # Bounds
    bounds = []
    n = len(centers_initial)
    # (x, y) bounds
    for i in range(n):
        bounds.append((1e-10, 1 - 1e-10))  # min, max for x_i
        bounds.append((1e-10, 1 - 1e-10))  # min, max for y_i
    # radii bounds (ε safe from 0)
    for r in radii_initial:
        bounds.append((1e-10, 1 - 1e-10))  # max r is bounded by square side 1
    
    # Optimize with 'trust-constr'
    res = minimize(objective, args0, bounds=bounds,
                   constraints={'type': 'ineq', 'fun': constraint_func},
                   method='trust-constr',
                   options={'gtol': 1e-9, 'maxiter': 300})
    
    # Extract results
    centers_final_flat = res.x[:-26]
    radii_final = res.x[-26:]
    
    centers_final = centers_final_flat.reshape(-1, 2)
    
    # Clip negative radii
    radii_final = np.clip(radii_final, 0, None)
    
    return centers_final, radii_final
