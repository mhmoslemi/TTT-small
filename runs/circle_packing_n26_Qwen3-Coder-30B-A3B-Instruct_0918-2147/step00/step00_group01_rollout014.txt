import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
from math import sqrt

def generate_lattice(s):
    # Generate hexagonal lattice points inside [s/2, 1-s/2]^2
    sqrt3 = sqrt(3)
    rows = int(1.0 / (s * sqrt3 / 2)) + 2
    cols = int(1.0 / s) + 2
    points = []
    for r in range(rows):
        y = r * s * sqrt3 / 2
        if y < s / 2 or y > 1 - s / 2:
            continue
        for i in range(cols):
            if r % 2 == 0:
                x = i * s
            else:
                x = (i + 0.5) * s
            if x < s / 2 or x > 1 - s / 2:
                continue
            points.append([x, y])
    return np.array(points)

def build_milp(P):
    K = len(P)
    z = cp.Variable(K, boolean=True)
    r = cp.Variable(K, nonneg=True)
    # Wall constraints: r_k <= w_k * z_k where w_k = min of distances to walls
    w = np.minimum(P[:, 0], 1 - P[:, 0])
    w = np.minimum(w, P[:, 1])
    w = np.minimum(w, 1 - P[:, 1])

    constraints = [0 <= r, r <= w * z]
    # Non-overlap constraints
    M = 1.0
    for k in range(K):
        for l in range(k+1, K):
            d_kl = np.linalg.norm(P[k] - P[l])
            constraints.append(d_kl >= r[k] + r[l] - M * (2 - z[k] - z[l]))
    # Exactly 26 circles
    constraints.append(cp.sum(z) == 26)
    # Objective
    objective = cp.Maximize(cp.sum(r))
    prob = cp.Problem(objective, constraints)

    def solve_lp(z_fixed_vals=None):
        if z_fixed_vals is not None:
            for i in range(K):
                z[i].fix(z_fixed_vals[i])
        try:
            prob.solve(solver=cp.ECOS, abstol=1e-9, reltol=1e-9)
            sol_r = r.value
            sol_z = z.value
            bound = prob.value
        except Exception:
            sol_r = np.zeros(K)
            sol_z = np.zeros(K)
            bound = -np.inf
        return sol_r, sol_z, bound

    return solve_lp

def branch_and_bound(milp_solver, node_limit=20000):
    import heapq
    from collections import deque

    heap = []
    best_obj = -np.inf
    best_z = None
    best_r = None
    node_count = 0

    # Initial LP (all variables free)
    initial_r, initial_z, initial_bound = milp_solver()
    if initial_bound > best_obj:
        best_obj = initial_bound
        best_z = np.round(initial_z)
        best_r = initial_r

    heapq.heappush(heap, (-initial_bound, 0, np.zeros(len(initial_z), dtype=np.int32), np.ones(len(initial_z), dtype=np.int32)))
    node_queue = deque([0])

    while heap and node_count < node_limit:
        neg_bound, level, fixed_z, free_mask = heapq.heappop(heap)
        bound = -neg_bound
        if bound <= best_obj - 1e-9:
            continue
        # Find first free binary variable
        free_idx = -1
        for i in range(len(fixed_z)):
            if free_mask[i] == 1:
                free_idx = i
                break
        if free_idx == -1:
            # All binary variables are fixed
            if bound > best_obj:
                best_obj = bound
                best_z = fixed_z.copy()
                best_r = initial_r.copy()
            continue
        # Branch on index `free_idx`
        # Create nodes for z[free_idx] = 0 and 1
        free_mask[free_idx] = 0
        for val in [0, 1]:
            new_z = fixed_z.copy()
            new_z[free_idx] = val
            new_fixed = new_z.astype(bool).astype(np.int32)
            z_vals = new_fixed
            sol_r, sol_z, node_bound = milp_solver(z_vals)
            if node_bound > best_obj:
                best_obj = node_bound
                best_z = sol_z
                best_r = sol_r
            if node_bound >= best_obj - 1e-10:
                new_mask = free_mask.copy()
                heap_entry = (-node_bound, level+1, new_z, new_mask)
                heapq.heappush(heap, heap_entry)
            node_count += 1
            if node_count >= node_limit:
                break
        if node_count >= node_limit:
            break
    return best_z, best_r

def refine_solution(centers, radii):
    # Adjust both centers and radii using local optimization
    n = len(centers)
    x0 = np.concatenate([centers.flatten(), radii])
    bounds = [(0, 1) for _ in range(2 * n)] + [(0, None) for _ in range(n)]

    def objective(x):
        pos = x[:2*n].reshape(n, 2)
        r = x[2*n:]
        return -np.sum(r)

    def constraints_func(x):
        pos = x[:2*n].reshape(n, 2)
        r = x[2*n:]
        con = []
        # Wall constraints
        for i in range(n):
            con.append(pos[i, 0] - r[i])  # Left wall
            con.append(1 - pos[i, 0] - r[i])  # Right wall
            con.append(pos[i, 1] - r[i])  # Bottom wall
            con.append(1 - pos[i, 1] - r[i])  # Top wall

        # Non-overlap constraints
        for i in range(n):
            for j in range(i+1, n):
                dist_sq = np.sum((pos[i] - pos[j])**2)
                con.append(dist_sq - (r[i] + r[j])**2)
        return np.array(con)

    cons = {'type': 'ineq', 'fun': constraints_func}
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                   options={'ftol': 1e-9, 'maxiter': 2000})
    
    refined_centers = res.x[:2*n].reshape(n, 2)
    refined_radii = res.x[2*n:]
    sum_radii = np.sum(refined_radii)
    return refined_centers, refined_radii, sum_radii

def run_packing():
    # Parameters
    spacing = 0.13
    lattice_points = generate_lattice(spacing)
    milp_solver = build_milp(lattice_points)
    selected_z, selected_radii = branch_and_bound(milp_solver, 20000)
    # Get the actual selected centers (lattice points)
    selected_indices = np.where(selected_z > 0.5)[0]
    centers = lattice_points[selected_indices]
    radii = selected_radii[selected_indices]
    refined_centers, refined_radii, sum_radii = refine_solution(centers, radii)
    return refined_centers, refined_radii, sum_radii
