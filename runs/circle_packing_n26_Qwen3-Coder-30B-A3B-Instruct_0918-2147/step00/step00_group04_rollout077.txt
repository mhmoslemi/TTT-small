import numpy as np

def build_leaf_problem(cell, k):
    """Build a SOCP problem for packing k circles in a rectangular cell."""
    x_min, y_min, x_max, y_max = cell
    
    # Variables
    centers = cp.Variable((k, 2))
    radii = cp.Variable(k, nonneg=True)
    
    # Build constraints
    constraints = []
    
    # Boundary constraints
    for i in range(k):
        constraints.extend([
            centers[i, 0] - radii[i] >= x_min,
            x_max - centers[i, 0] >= radii[i],
            centers[i, 1] - radii[i] >= y_min,
            y_max - centers[i, 1] >= radii[i]
        ])
    
    # Non-overlap constraints
    for i in range(k):
        for j in range(i+1, k):
            # Second-order cone constraint: ||c_i - c_j||_2 >= r_i + r_j
            diff = centers[i] - centers[j]
            constraints.append(
                cp.norm(diff, 2) >= radii[i] + radii[j]
            )
    
    # Objective
    objective = cp.Maximize(cp.sum(radii))
    
    # Problem
    prob = cp.Problem(objective, constraints)
    return prob, centers, radii

def solve_leaf(cell, k):
    """Solve a leaf cell with k circles and return centers, radii."""
    prob, centers, radii = build_leaf_problem(cell, k)
    prob.solve(solver=cp.ECOS, abstol=1e-10, reltol=1e-10)
    
    if prob.status != cp.OPTIMAL:
        # Return feasible but suboptimal solution
        raise RuntimeError(f"Failed to solve leaf problem; status: {prob.status}")
    
    centers_val = centers.value
    radii_val = radii.value
    return centers_val, radii_val

def create_cell_partition():
    # Create a partition of the square into 9 boxes
    a = 0.25
    cells = []
    # Four corners
    cells.append((0.0, 0.0, a, a))
    cells.append((1-a, 0, 1, a))
    cells.append((0, 1-a, a, 1))
    cells.append((1-a, 1-a, 1, 1))
    # Four edges
    cells.append((a, 0.0, 1-a, a))
    cells.append((a, 1-a, 1-a, 1))
    cells.append((0.0, a, a, 1-a))
    cells.append((1-a, a, 1, 1-a))
    # Center
    cells.append((a, a, 1-a, 1-a))
    return cells

def global_polish(centers, radii):
    # Perform limited barrier-based gradient ascent

    def progress_step(centers, radii, step_size=0.1, mu=1e-5):
        n = len(radii)
        # Compute gradients of barrier function manually
        grad_r = np.ones(n) - mu * np.sum([1.0 / (np.linalg.norm(centers[i] - centers[j]) + 1e-10 - radii[i] - radii[j]) if i != j else 0 for j in range(n) for i in range(n)], axis=0)
        
        grad_c = np.zeros_like(centers)
        
        for i in range(n):
            for j in range(n):
                if i != j:
                    d = np.linalg.norm(centers[i] - centers[j])
                    if d > (radii[i] + radii[j]):
                        diff = centers[i] - centers[j]
                        grad_c[i] += mu * (diff / max(d, 1e-10)) / (d - radii[i] - radii[j])
        
        # Update
        new_centers = centers - step_size * grad_c
        new_radii = radii + step_size * grad_r
        
        # Maintain constraints
        new_centers = np.clip(new_centers, 0, 1)
        new_radii = np.maximum(new_radii, 0)
        
        return new_centers, new_radii
    
    # Run few iterations
    for _ in range(5):  # 5 iters enough?
        centers, radii = progress_step(centers, radii)
        
    return centers, radii

def final_refine(centers, radii):
    # Run high precision SOCP on final system
    n = len(radii)
    centers_var = cp.Variable((n, 2))
    radii_var = cp.Variable(n, nonneg=True)
    
    constraints = []
    
    for i in range(n):
        constraints.append(centers_var[i, 0] - radii_var[i] >= 0)
        constraints.append(1 - centers_var[i, 0] >= radii_var[i])
        constraints.append(centers_var[i, 1] - radii_var[i] >= 0)
        constraints.append(1 - centers_var[i, 1] >= radii_var[i])
        
    for i in range(n):
        for j in range(i+1, n):
            constraints.append(cp.norm(centers_var[i] - centers_var[j], 2) >= radii_var[i] + radii_var[j])
    
    obj = cp.Maximize(cp.sum(radii_var))
    prob = cp.Problem(obj, constraints)
    
    prob.solve(solver=cp.ECOS, abstol=1e-12, reltol=1e-12)
    
    if prob.status != cp.OPTIMAL:
        raise RuntimeError("Failed post-refinement of high precision")
        
    centers_result = centers_var.value
    radii_result = radii_var.value
    
    # Post-process for numerical slacks
    total_sum = np.sum(radii_result)
    return centers_result, radii_result, total_sum
