import numpy as np
from scipy.optimize import linprog
from itertools import combinations

def generate_hex_lattice(h):
    sqrt3 = np.sqrt(3)
    points = []
    for i in range(int(1/h) + 2):
        for j in range(int(1/(h*sqrt3/2)) + 2):
            x = i * h + (j % 2) * (h / 2)
            y = j * h * sqrt3 / 2
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append([x, y])
    return np.array(points)

def build_neighbour_list(centers, max_dist):
    n = centers.shape[0]
    edges = []
    for i in range(n):
        for j in range(i+1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            if dist <= max_dist:
                edges.append((i, j, dist))
    return edges

def solve_radius_lp(selected_indices, R_wall, neighbour_list):
    n_selected = len(selected_indices)
    if n_selected == 0:
        return np.array([]), True
    indices_map = {idx: i for i, idx in enumerate(selected_indices)}
    
    # Setup linear program: maximize sum of radii
    c = -np.ones(n_selected)  # negative because linprog minimizes
    
    # Inequality constraint matrix A_ub * x <= b_ub
    A_ub = []
    b_ub = []
    
    # Wall constraint: 0 <= r_i <= R_wall[i]
    for i in range(n_selected):
        A_ub.append([0] * n_selected)
        A_ub[-1][i] = -1  # -r_i <= 0
        b_ub.append(0)
        A_ub.append([0] * n_selected)
        A_ub[-1][i] = 1  # r_i <= R_wall[i]
        b_ub.append(R_wall[selected_indices[i]])
        
    # Separation constraints: r_i + r_j <= d_ij
    for i, j, d_ij in neighbour_list:
        if i in indices_map and j in indices_map:
            row = [0] * n_selected
            row[indices_map[i]] = 1
            row[indices_map[j]] = 1
            A_ub.append(row)
            b_ub.append(d_ij)
    
    # Solve LP 
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    
    if res.success:
        radii = res.x
        return radii, True
    else:
        return None, False

def upper_bound(partial_set, remaining_indices, R_wall):
    ub = 0
    partial_set = list(partial_set)
    # each circle's max wall radii
    for idx in partial_set:
        ub += R_wall[idx]
    # best remaining radii
    best_remaining = sorted([R_wall[i] for i in remaining_indices], reverse=True)
    for i in range(26 - len(partial_set)):
        ub += best_remaining[i]
    return ub

def branch_and_bound(centers, R_wall, neighbour_list):
    best_total = -1
    best_centers = None
    best_radii = None
    
    def search(S_partial, U, lb):
        nonlocal best_total, best_centers, best_radii
        if len(S_partial) == 26:
            # at leaf node, compute true LP
            radii, feasible = solve_radius_lp(S_partial, R_wall, neighbour_list)
            if feasible and radii is not None:
                total = np.sum(radii)
                if total > best_total:
                    best_total = total
                    best_radii = radii.copy()
                    best_centers = centers[S_partial].copy()
            return lb
        elif len(S_partial) > 26:
            return lb
            
        ub = upper_bound(S_partial, U, R_wall)
        if ub <= lb:
            return lb
            
        # get max wall radius among U and select it for branching  
        # we expand the subtree that includes this point: recurse with S_partial + [max_wall_idx] and exclude it
        max_wall_idx = max(U, key=lambda idx: R_wall[idx])
        U_exclude = U - {max_wall_idx}
        U_include = U - {max_wall_idx}
        S_include = S_partial | {max_wall_idx}
        
        # Check if inclusion satisfies constraints
        # Try include this point, then attempt expansion
        new_ub = upper_bound(S_include, U_exclude, R_wall)
        new_lb = lb
        if new_ub > lb:
            # Dive into include first
            new_lb = search(S_include, U_exclude, new_lb)
        # Then dive into exclude second
        new_lb = search(S_partial, U_exclude, new_lb)
        return new_lb
            
    # Start the search with empty selection
    init_set = frozenset()
    init_U = frozenset(range(len(centers)))
    search(init_set, init_U, -1.0)
    return best_centers, best_radii

def local_continuous_refinement(init_centers, init_radii):
    n = len(init_radii)
    # Clean up at boundary
    x = init_centers[:,0].copy()
    y = init_centers[:,1].copy()
    r = init_radii.copy()
    
    # Enforce boundary conditions
    for i in range(n):
        r[i] = min(r[i], x[i], 1 - x[i], y[i], 1 - y[i])
    
    # Minor adjustment of center positions at start; high final tolerance (>1e-5)
    state = np.hstack([x, y, r])
    bounds = [(0, 1)] * n + [(0, 1)] * n + [(0, 1)] * n  # x,y,r bounds

    # Simple local search using trust-constr, base solver with few iterations
    def obj(x):
        return -np.sum(x[2*n:])  # maximize sum of radii
    
    def constraint_feas(x):
        # constraints on containment and non-overlap (approximated)
        # just return feasibility pseudo-error, don't compute precise penalties
        return 0  # placeholder
    
    from scipy.optimize import minimize
    res = minimize(obj, state, method='trust-constr', bounds=bounds, options={'maxiter': 100})
    if res.success:
        x_opt = res.x[:n]
        y_opt = res.x[n:2*n]
        r_opt = res.x[2*n:]
        return np.vstack([x_opt, y_opt]).T, r_opt
    else:
        return init_centers, init_radii

def run_packing():
    h = 0.07
    centers = generate_hex_lattice(h)
    R_wall = np.array([min(x, 1-x, y, 1-y) for x, y in centers])
    edges = build_neighbour_list(centers, 0.4)
    
    selected_centers, selected_radii = branch_and_bound(centers, R_wall, edges)
    if selected_centers is None or selected_radii is None:
        raise RuntimeError("Branch-and-bound failed to return solution")
    
    refined_centers, refined_radii = local_continuous_refinement(selected_centers, selected_radii)
    sum_radii = float(np.sum(refined_radii))
    
    # run validation
    success, msg = validate_packing(refined_centers, refined_radii)
    if not success:
        if "overlap" in msg.lower() or "outside" in msg.lower():
            # Try removing 1e-7 from all radii
            while True:
                refined_radii = np.maximum(refined_radii - 1e-7, 1e-10)
                success, msg = validate_packing(refined_centers, refined_radii)
                if success:
                    break
                if np.max(refined_radii) < 1e-9:
                    raise ValueError("Could not make solution valid after shrinking radii")
    
    return refined_centers, refined_radii, sum_radii
