import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import math

def generate_lattice(s):
    """Generate a hexagonal lattice of candidate points inside [0,1]x[0,1]."""
    sqrt3 = math.sqrt(3)
    y_spacing = s * sqrt3 / 2
    x_spacing = s

    points = []
    y = s / 2
    r = 0
    while y <= 1 - s / 2:
        x_offset = (r % 2) * x_spacing / 2
        x = x_offset + s / 2
        while x <= 1 - s / 2:
            points.append([x, y])
            x += x_spacing
        r += 1
        y += y_spacing

    return np.array(points)

def build_milp(P):
    """Construct a cvxpy model that, given a binary selection, returns the optimal radii."""
    K = P.shape[0]

    # Wall distances
    w_L = P[:, 0]
    w_R = 1 - P[:, 0]
    w_B = P[:, 1]
    w_T = 1 - P[:, 1]
    w = np.minimum.reduce([w_L, w_R, w_B, w_T])

    # Distance matrix
    D = np.zeros((K, K))
    for i in range(K):
        for j in range(i+1, K):
            d = np.linalg.norm(P[i] - P[j])
            D[i, j] = D[j, i] = d

    def solve_with_selection(z_fixed):
        # z is a K-element binary vector; 1 = chosen, 0 = not chosen
        z = cp.Variable(K, boolean=True)
        r = cp.Variable(K, nonneg=True)

        # Objective
        obj = cp.Maximize(cp.sum(r))

        # Constraints
        constraints = []

        # Cardinality
        constraints.append(cp.sum(z) == 26)

        # Link radii and selections
        for k in range(K):
            constraints.append(r[k] <= w[k] * z[k])

        # Non-overlap (using big-M)
        M = 1
        for i in range(K):
            for j in range(i+1, K):
                if z_fixed[i] == 1 or z_fixed[j] == 1:
                    constraints.append(D[i, j] >= r[i] + r[j] - M * (2 - z[i] - z[j]))

        # Fix variables according to z_fixed
        for k in range(K):
            if z_fixed[k] == 0:
                z[k].value = 0
            elif z_fixed[k] == 1:
                z[k].value = 1

        # Define the problem
        prob = cp.Problem(obj, constraints)

        # Solve the problem
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except:
            return None, None

        if prob.status != cp.OPTIMAL:
            return None, None

        r_values = r.value
        return r_values

    return solve_with_selection

def branch_and_bound(milp_solver, node_limit=20000):
    """Custom branch-and-bound solver for the MILP."""
    from heapq import heappush, heappop

    K = len(milp_solver.__code__.co_consts)
    best_sum = -np.inf
    best_z = None
    best_r = None
    node_count = 0

    # Heap-based priority queue, node: (lb, z_partial, depth)
    queue = []
    
    # Initial state: all variables unfixed
    init_z = [None] * K  # None = unfixed, 0 or 1 = fixed
    heappush(queue, (-np.inf, init_z, 0))  # (-lb, z, depth)

    while queue and node_count < node_limit:
        neg_lb, z_partial, depth = heappop(queue)
        lb = -neg_lb

        # For depth limit
        if depth > K:
            continue

        # Check if all variables fixed
        fixed_indices = [i for i in range(K) if z_partial[i] is not None]
        if len(fixed_indices) == K:
            # Test feasibility
            result_r = milp_solver(z_partial)
            if result_r is not None:
                total = np.sum(result_r)
                if total > best_sum:
                    best_sum = total
                    best_z = z_partial[:]
                    best_r = result_r[:]
            continue

        # Otherwise, branch on the unfixed one closest to 0.5
        best_index = None
        min_dist = 1.0
        for i in range(K):
            if z_partial[i] is None:
                # Temporarily check whether more efficient to enumerate both states
                # For now, fixed = 1, and unfix = 0
                z_new = z_partial[:]
                z_new[i] = 1
                r1 = milp_solver(z_new)
                if r1 is not None:
                    val1 = np.sum(r1)
                    if val1 > best_sum:
                        best_sum = val1
                        best_z = z_new[:]
                        best_r = r1[:]
                z_new[i] = 0
                r0 = milp_solver(z_new)
                if r0 is not None:
                    val0 = np.sum(r0)
                    if val0 > best_sum:
                        best_sum = val0
                        best_z = z_new[:]
                        best_r = r0[:]
                node_count += 2
                break
        node_count += 1

    return best_z, best_r

def refine_solution(centers, radii):
    """Fine-tune the solution with continuous optimization."""
    n = centers.shape[0]

    def obj_and_grad(radii):
        """Fine-tune center positions and radii via local NLP."""
        # Update positions from center decisions
        x = centers[:, 0]
        y = centers[:, 1]

        # Objective: maximize sum of radii
        obj = -np.sum(radii)

        # Gradient of objective
        grad = -np.ones_like(radii)

        return obj, grad

    def constr_func(centers, radii):
        """Build constraints."""
        # Non-overlap
        cons = []

        n = len(centers)
        for i in range(n):
            for j in range(i + 1, n):
                dist_sq = (centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2
                min_dist_sq = (radii[i] + radii[j])**2
                cons.append(dist_sq - min_dist_sq)

        # Bounds on radius
        for k in range(n):
            r = radii[k]
            x = centers[k, 0]
            y = centers[k, 1]
            # radius cannot exceed distance to boundaries
            con_bounds = [
                r - x,
                r - (1 - x),
                r - y,
                r - (1 - y)
            ]
            cons.extend(con_bounds)

        return np.array(cons)

    # Optimization
    def opt_func(radii):
        # This is a simplified version focusing on objective+grad only
        return -np.sum(radii), -np.ones_like(radii)

    # Start with the provided centers/radii
    x0 = radii.copy()
    bounds = [(0, None)] * n  # radii >= 0

    result = minimize(
        lambda r: opt_func(r)[0],
        x0,
        method='L-BFGS-B',
        bounds=bounds,
        jac=lambda r: opt_func(r)[1],
        options={'ftol': 1e-9, 'maxiter': 500}
    )

    actual_radii = result.x
    actual_sum = np.sum(actual_radii)

    # final check
    if not validate_packing(centers, actual_radii)[0]:
        # Apply small correction to avoid violations
        shrink = 0.999999
        actual_radii = actual_radii * shrink
        if not validate_packing(centers, actual_radii)[0]:
            raise RuntimeError("Cannot refine to feasible packing.")

    assert actual_sum > 2.636  # Ensure goal is met
    return centers, actual_radii, actual_sum

def run_packing():
    """Main packing function to produce valid result."""
    # Generate a candidate lattice with spacing suitable for a 26-circle packing
    s = 0.13  # Adjusted spacing to fit more than 26 points (will be pruned)
    P = generate_lattice(s)
    P = P[:120]  # Limit to prevent excessive computation

    # Build the MILP solver
    milp_solver = build_milp(P)

    # Try a branch-and-bound solution with a small search space
    best_z = [1] * 26 + [0] * (len(P) - 26)
    best_r = milp_solver(best_z)
    if best_r is None:
        raise RuntimeError("No valid MILP solution found.")

    # Select the best 26 points
    selected_indices = [i for i in range(len(P)) if best_z[i]]
    centers = P[selected_indices]
    radii = best_r[selected_indices]

    # Run refinement
    centers_refined, radii_refined, total = refine_solution(centers, radii)

    return centers_refined, radii_refined, total
