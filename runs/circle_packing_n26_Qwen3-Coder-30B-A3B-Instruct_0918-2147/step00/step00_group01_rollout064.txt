import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import math

def generate_lattice(s):
    """
    Generate a hexagonal lattice of candidate center points.
    """
    sqrt3 = math.sqrt(3)
    rows = int(1.0 / (s * sqrt3 / 2)) + 2
    cols = int(1.0 / s) + 2
    points = []
    for r in range(rows):
        y = r * s * sqrt3 / 2
        if y < s / 2 or y > 1 - s / 2:
            continue
        start = 0.5 if r % 2 == 1 else 0.0
        for i in range(cols):
            x = (i + start) * s
            if x < s / 2 or x > 1 - s / 2:
                continue
            points.append([x, y])
    return np.array(points)

def build_milp(P):
    """
    Builds a MILP model for the circle packing problem.
    Returns a function that, given a partial binary assignment, 
    computes the optimal radii and objective.
    """
    K = len(P)
    # Precompute pairwise distances
    dists = np.zeros((K, K))
    for i in range(K):
        for j in range(i + 1, K):
            d = np.linalg.norm(P[i] - P[j])
            dists[i, j] = d
            dists[j, i] = d

    # Wall distances
    wall_dists = np.zeros((K, 4))
    for i in range(K):
        x, y = P[i]
        wall_dists[i, 0] = x  # left
        wall_dists[i, 1] = 1 - x  # right
        wall_dists[i, 2] = y  # bottom
        wall_dists[i, 3] = 1 - y  # top
    wall_limits = np.min(wall_dists, axis=1)

    # Variables
    z = cp.Variable(K, boolean=True)
    r = cp.Variable(K, nonneg=True)

    def solve_fixed_z(z_fixed):
        # Create a new problem with fixed z values
        prob = cp.Problem(
            cp.Maximize(cp.sum(r)),
            [
                cp.sum(z) == 26,
                r <= wall_limits * z,
                # Non-overlap constraints
                *[dists[i, j] >= r[i] + r[j] - 1e-12 * (2 - z[i] - z[j]) 
                  for i in range(K) for j in range(i+1, K) 
                  if z_fixed[i] == 1 and z_fixed[j] == 1]
            ]
        )
        try:
            prob.solve(solver=cp.OSQP, verbose=False, eps_abs=1e-9, eps_rel=1e-9)
            return r.value.copy() if r.value is not None else np.zeros(K)
        except:
            return np.zeros(K)

    return solve_fixed_z

            obj = np.sum(r_vals)
            if obj > best_obj:

                best_z = z_actual

        fallback_z = np.zeros(26, dtype=int)
        fallback_z[:26] = 1
        fallback_r = milp_solver(fallback_z)
        return fallback_z, fallback_r

def refine_solution(centers, radii):
    """
    Refines the solution with continuous optimisation.
    Uses scipy.optimize.minimize.
    """
    n = len(centers)
    # Initial values
    x0 = np.empty(2*n + n)  # x1,y1,r1,...,xn,yn,rn
    for i in range(n):
        x0[2*i] = centers[i][0]
        x0[2*i+1] = centers[i][1]
        x0[2*n+i] = radii[i]

    def obj_func(x):
        r_values = x[2*n:]
        return -np.sum(r_values)

    def constraint_func(x):
        n = len(centers)
        centers_c = x[:2*n].reshape((n, 2))
        r_values = x[2*n:]
        constraints = []

        # Wall constraints
        for i in range(n):
            x_i, y_i = centers_c[i]
            r_i = r_values[i]
            constraints.append(x_i - r_i)  # >= 0
            constraints.append(1 - x_i - r_i)  # >= 0
            constraints.append(y_i - r_i)  # >= 0
            constraints.append(1 - y_i - r_i)  # >= 0

        # Non-overlap constraints
        for i in range(n):
            for j in range(i+1, n):
                dist2 = np.sum((centers_c[i] - centers_c[j])**2)
                r_sum = r_values[i] + r_values[j]
                constraints.append(dist2 - r_sum**2)  # >= 0

        return np.array(constraints)

    # Jacobian of constraints cannot be expressed cleanly due to nonlinear nature
    # Let's use finite differences, or just do it numerically
    bounds = []
    for i in range(n):
        bounds.append((1e-5, 1-1e-5))  # x's
        bounds.append((1e-5, 1-1e-5))  # y's
        bounds.append((1e-12, 0.5))    # r's - cap at 0.5 to allow movement to walls

    try:
        result = minimize(
            obj_func,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'ineq', 'fun': constraint_func},
            options={'ftol': 1e-9, 'maxiter': 1000},
            tol=1e-9
        )
    except Exception:
        # Fallback for numerical issues
        return centers, radii, np.sum(radii)

    # Extract results
    updated_centers = result.x[:2*n].reshape((n,2))
    updated_radii = result.x[2*n:]
    total = np.sum(updated_radii)

    return updated_centers, updated_radii, total

def run_packing():
    """
    Main function to run the packing.
    """
    # Generate a sufficiently dense lattice
    P = generate_lattice(0.13)  # spacing gives ~120 candidate points
    print(f"Generated {len(P)} candidate points")

    # Build the MILP model
    milp_solver = build_milp(P)

    # Run branch and bound to get initial discrete selection
    print("Starting branch and bound...")
    best_z, best_r = branch_and_bound(milp_solver, node_limit=20000)
    selected_indices = np.where(best_z == 1)[0]
    print(f"Selected {len(selected_indices)} centers")

    # Extract selected centers
    centers_packed = P[selected_indices]
    radii_packed = best_r[selected_indices]

    # Refine with continuous optimisation
    refined_centers, refined_radii, total = refine_solution(centers_packed, radii_packed)
