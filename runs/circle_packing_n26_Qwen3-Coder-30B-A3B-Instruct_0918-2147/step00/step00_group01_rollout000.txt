import numpy as np
from scipy.optimize import minimize

def initial_layout():
    # Place circles in a hexagonal grid pattern
    n = 26
    rows = 5
    cols = 5
    spacing = 0.18
    radius = spacing / 2 - 0.01
    
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = j * spacing + 0.05
            y = i * spacing + 0.05
            # Correct positions that are outside the unit square
            x = max(radius, min(1 - radius, x))
            y = max(radius, min(1 - radius, y))
            centers.append([x, y])
    
    # Add one more circle at center
    centers.append([0.5, 0.5])
    
    # Adjust to be exactly 26
    centers = centers[:26]
    radii = [radius] * 26
    
    return np.array(centers), np.array(radii)

def pack_vector(centers, radii):
    z = np.zeros(3 * len(centers))
    for i in range(len(centers)):
        z[3*i] = centers[i][0]
        z[3*i+1] = centers[i][1]
        z[3*i+2] = radii[i]
    return z

def unpack_vector(z):
    n = len(z) // 3
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i][0] = z[3*i]
        centers[i][1] = z[3*i+1]
        radii[i] = z[3*i+2]
    return centers, radii

def objective(z):
    _, radii = unpack_vector(z)
    return -np.sum(radii)

def constraint_boundary(z):
    centers, radii = unpack_vector(z)
    constr = []
    for i in range(len(centers)):
        x, y = centers[i]
        r = radii[i]
        # x - r >= 0
        constr.append(x - r)
        # 1 - x - r >= 0
        constr.append(1 - x - r)
        # y - r >= 0
        constr.append(y - r)
        # 1 - y - r >= 0
        constr.append(1 - y - r)
    return np.array(constr)

def constraint_nonoverlap(z):
    centers, radii = unpack_vector(z)
    constr = []
    n = len(centers)
    for i in range(n):
        for j in range(i+1, n):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist_sq = dx*dx + dy*dy
            r_sum = radii[i] + radii[j]
            # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
            constr.append(dist_sq - r_sum * r_sum)
    return np.array(constr)

    best_centers = None

    for _ in range(5):

            centers[i][1] += np.random.uniform(-0.02, 0.02)
            # Keep within bounds
            centers[i][0] = np.clip(centers[i][0], 0.01, 0.99)
            centers[i][1] = np.clip(centers[i][1], 0.01, 0.99)
            
            # Add some variation in radii
            radii[i] *= (1 + np.random.uniform(-0.05, 0.05))
            radii[i] = max(0.01, radii[i])
        
        z0 = pack_vector(centers, radii)
        
        # Define bounds for each variable
        bounds = []
        for i in range(len(z0)):
            if i % 3 == 2:  # radius variable
                bounds.append((0, 0.5))
            else:  # x or y variable
                bounds.append((0, 1))
        
        # Define constraints
        constraints = []
        
        # Boundary constraints
        constraint_func = lambda z: constraint_boundary(z)
        constraints.append({'type': 'ineq', 'fun': constraint_func})
        
        # Non-overlap constraints
        constraint_func = lambda z: constraint_nonoverlap(z)
        constraints.append({'type': 'ineq', 'fun': constraint_func})
        
        # Optimize
        result = minimize(objective, z0, method='SLSQP', bounds=bounds,
                         constraints=constraints, options={'ftol': 1e-9, 'maxiter': 2000})
        
        if result.success:
            final_centers, final_radii = unpack_vector(result.x)
