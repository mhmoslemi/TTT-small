import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial.distance import cdist

def build_lp_model(n_circles):
    # Variables: [x1, y1, r1, x2, y2, r2, ..., xn, yn, rn]
    # Each circle: 3 variables (x, y, r)
    n_vars = 3 * n_circles

    # Objective: maximize sum of radii
    c = np.zeros(n_vars)
    c[2::3] = -1.0  # Negative because linprog minimizes

    # Inequality constraints: A_ub @ x <= b_ub
    A_ub = []
    b_ub = []

    # Wall constraints for each circle
    for i in range(n_circles):
        # r <= x, r <= 1-x, r <= y, r <= 1-y
        row1 = np.zeros(n_vars)
        row1[3*i + 2] = 1.0  # r_i
        row1[3*i + 0] = -1.0  # -x_i
        A_ub.append(row1)
        b_ub.append(0.0)

        row2 = np.zeros(n_vars)
        row2[3*i + 2] = 1.0  # r_i
        row2[3*i + 0] = 1.0   # x_i
        A_ub.append(row2)
        b_ub.append(1.0)

        row3 = np.zeros(n_vars)
        row3[3*i + 2] = 1.0  # r_i
        row3[3*i + 1] = -1.0  # -y_i
        A_ub.append(row3)
        b_ub.append(0.0)

        row4 = np.zeros(n_vars)
        row4[3*i + 2] = 1.0  # r_i
        row4[3*i + 1] = 1.0   # y_i
        A_ub.append(row4)
        b_ub.append(1.0)

    # Rectangle separation constraints for all pairs
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            # x_i + r_i <= x_j - r_j  ||  x_j + r_j <= x_i - r_i
            # Inequality 1: x_i + r_i - x_j + r_j <= 0
            row1 = np.zeros(n_vars)
            row1[3*i + 0] = 1.0  # x_i
            row1[3*i + 2] = 1.0  # r_i
            row1[3*j + 0] = -1.0  # -x_j
            row1[3*j + 2] = 1.0  # r_j
            A_ub.append(row1)
            b_ub.append(0.0)

            # Inequality 2: x_j + r_j - x_i + r_i <= 0
            row2 = np.zeros(n_vars)
            row2[3*i + 0] = -1.0  # -x_i
            row2[3*i + 2] = 1.0  # r_i
            row2[3*j + 0] = 1.0   # x_j
            row2[3*j + 2] = 1.0  # r_j
            A_ub.append(row2)
            b_ub.append(0.0)

            # y_i + r_i <= y_j - r_j  ||  y_j + r_j <= y_i - r_i
            # Inequality 3: y_i + r_i - y_j + r_j <= 0
            row3 = np.zeros(n_vars)
            row3[3*i + 1] = 1.0  # y_i
            row3[3*i + 2] = 1.0  # r_i
            row3[3*j + 1] = -1.0  # -y_j
            row3[3*j + 2] = 1.0  # r_j
            A_ub.append(row3)
            b_ub.append(0.0)

            # Inequality 4: y_j + r_j - y_i + r_i <= 0
            row4 = np.zeros(n_vars)
            row4[3*i + 1] = -1.0  # -y_i
            row4[3*i + 2] = 1.0  # r_i
            row4[3*j + 1] = 1.0   # y_j
            row4[3*j + 2] = 1.0  # r_j
            A_ub.append(row4)
            b_ub.append(0.0)

    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)

    return A_ub, b_ub, c


def solve_lp(A_ub, b_ub, c):
    # Solve: minimize -sum of radii (or maximize sum of radii)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    if not res.success:
        raise RuntimeError("LP solver failed")

    # Extract solution
    x_sol = res.x
    centers = x_sol[0::3], x_sol[1::3]
    radii = x_sol[2::3]
    return centers, radii

def resolve_conflicts(centers, radii, active_set, conflict_pairs):
    # Simplified: only increase radii and move in simple fashion
    # We treat this in a hand-coded way with conservative updates
    # For simplicity, we just do one step of improvement ensuring no overlaps,
    # not full nonlinear optimization
    centers = np.array(centers).copy()
    radii = np.array(radii).copy()

    # For each conflict pair, shrink one or both slightly
    for i, j in conflict_pairs:
        r_i = radii[i]
        r_j = radii[j]
        x_i, y_i = centers[i]
        x_j, y_j = centers[j]
        dx = x_j - x_i
        dy = y_j - y_i
        dist = np.sqrt(dx**2 + dy**2)
        if dist < r_i + r_j - 1e-12:
            # Reduce both radii slightly to allow margin
            overlap = (r_i + r_j - dist)
            if overlap > 0:
                adjustment = min(0.5 * overlap, 0.01)
                radii[i] -= adjustment
                radii[j] -= adjustment
                # Try to move them apart gently
                if dist > 1e-10:
                    unit_dx = dx / dist
                    unit_dy = dy / dist
                    centers[i, 0] -= adjustment * unit_dx * 0.5
                    centers[i, 1] -= adjustment * unit_dy * 0.5
                    centers[j, 0] += adjustment * unit_dx * 0.5
                    centers[j, 1] += adjustment * unit_dy * 0.5

    # Clamp negative radii
    radii = np.maximum(radii, 0.)

    # Enforce boundary constraints
    for i in range(len(centers)):
        r = radii[i]
        x, y = centers[i]
        # Correct any slight boundary violations
        x = np.clip(x, r, 1 - r)
        y = np.clip(y, r, 1 - r)
        centers[i] = [x, y]

    return centers, radii

def max_radius_projection(centers, radii):
    # Reproject each circle to its maximal possible radius
    centers = np.array(centers)
    radii = np.array(radii)
    n = len(centers)
    updated = True
    max_iter = 100
    for _ in range(max_iter):
        prev_radii = radii.copy()
        # For each circle: tightest bound from walls and others
        for i in range(n):
            x_i, y_i = centers[i]
            r_bound = np.array([
                x_i,               # left
                1 - x_i,           # right
                y_i,               # down
                1 - y_i            # up
            ])
            min_r_euclidean = float('inf')
            for j in range(n):
                if i == j:
                    continue
                x_j, y_j = centers[j]
                d_ij = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)
                min_r_euclidean = min(min_r_euclidean, d_ij - radii[j])
            if min_r_euclidean != float('inf'):
                r_bound = np.append(r_bound, [min_r_euclidean])
            radii[i] = np.min(r_bound)
        # If no change, stop early
        if np.allclose(prev_radii, radii, atol=1e-9):
            break
    return centers, radii
