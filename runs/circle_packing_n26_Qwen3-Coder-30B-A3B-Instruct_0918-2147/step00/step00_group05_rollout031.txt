import numpy as np

def hex_lattice(d):
    """Generate hexagonal lattice points within unit square."""
    sqrt3 = np.sqrt(3)
    dx = d
    dy = sqrt3 * d / 2
    points = []
    y = 0
    row = 0
    while y <= 1:
        x = 0
        offset = (row % 2) * dx / 2
        while x <= 1:
            points.append([x + offset, y])
            x += dx
        y += dy
        row += 1
    return np.array(points)

def add_border_circles(points, d):
    """Add 6 border-enhanced circles to the existing core points."""
    r = d / 4
    corners = [
        [r, r],
        [1 - r, r],
        [r, 1 - r],
        [1 - r, 1 - r]
    ]
    edges = [
        [0.5, r],
        [0.5, 1 - r]
    ]
    border_points = corners + edges
    return np.vstack([points, border_points])

def initial_radii(points):
    """Compute the initial safe uniform radius for all points."""
    n = points.shape[0]
    # Min distance to walls
    wall_distances = np.minimum(points[:, 0], 1 - points[:, 0])
    wall_distances = np.minimum(wall_distances, points[:, 1])
    wall_distances = np.minimum(wall_distances, 1 - points[:, 1])
    # Min distance to other centers
    dist_matrix = np.linalg.norm(points[:, None, :] - points[None, :, :], axis=2)
    np.fill_diagonal(dist_matrix, np.inf)
    min_distances = np.min(dist_matrix, axis=1)
    min_distances = min_distances / 2
    # Use the safer of the two limits
    initial_radii = np.minimum(wall_distances, min_distances)
    # Ensure all are nonnegative
    initial_radii = np.maximum(initial_radii, 0)
    return np.min(initial_radii)

def inflate_radii(centers, radii):
    """Inflate radii to maximum possible values while avoiding overlaps."""
    n = centers.shape[0]
    new_radii = radii.copy()
    epsilon = 1e-7
    # Precompute all pairwise distances
    dist_matrix = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    # For each circle, compute the max possible radius
    for i in range(n):
        # Wall limits
        x, y = centers[i]
        wall_limits = [
            x,  # left wall
            1 - x,  # right wall
            y,  # bottom wall
            1 - y  # top wall
        ]
        # Pairwise limits
        pairwise_limits = []
        for j in range(n):
            if i != j:
                pairwise_limits.append(dist_matrix[i, j] - radii[j])
        # Compute minimum of all limits
        max_radius = min(wall_limits + pairwise_limits)
        # Update only if increase is significant
        if max_radius > new_radii[i] + epsilon:
            new_radii[i] = max_radius
    return new_radii

def relax_centers(centers, radii, steps=10):
    """Apply force-based relaxation to centers."""
    # Parameters
    kappa = 0.02
    beta = 0.02
    lambda_force = 0.01
    gamma = 0.02
    alpha = 0.5
    
    n = centers.shape[0]
    old_centers = centers.copy()
    
    # Precompute all pairwise distances

    for step in range(steps):
        forces = np.zeros_like(centers)
        
        # Compute repulsive forces between all pairs
        for i in range(n):
            for j in range(n):
                if i != j:
                    dir_vec = centers[i] - centers[j]
                    dist = np.linalg.norm(dir_vec)
                    if dist > 1e-12:
                        normal = dir_vec / dist
                        # Distance threshold
                        threshold = radii[i] + radii[j] + beta
                        force_magnitude = kappa * max(0, threshold - dist)
                        forces[i] += force_magnitude * normal
        
        # Wall forces
        for i in range(n):
            x, y = centers[i]
            fx, fy = 0.0, 0.0
            # Left wall
            dist_left = x - radii[i]

            # Right wall

            # Top wall

            if dist_top < gamma:
                fy -= lambda_force * max(0, gamma - dist_top)
            forces[i] += [fx, fy]
        
        # Update centers
        new_centers = centers + alpha * forces
        
        # Clamp to within bounds
        for i in range(n):
            # Clamp x and y to [r_i, 1 - r_i]
            new_centers[i, 0] = np.clip(new_centers[i, 0], radii[i], 1 - radii[i])

        if min_dist < 1e-8:
            # Overlap detected, revert to old centers and break early
            centers = old_centers.copy()
            break
    
    return centers
