import numpy as np
from scipy.spatial.distance import cdist

def penalty(z, alpha=10.0, beta=10.0):
    n = 26
    centers = z[:2*n].reshape(n, 2)
    radii = z[2*n:]
    
    # Wall penalties
    wall_pen = 0.0
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        dists = [x, 1-x, y, 1-y]
        for d in dists:
            if d < r:
                wall_pen += alpha * (r - d)
    
    # Overlap penalties
    overlap_pen = 0.0
    for i in range(n):
        for j in range(i+1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            r_sum = radii[i] + radii[j]
            if dist < r_sum:
                overlap_pen += beta * (r_sum - dist)
    
    return wall_pen + overlap_pen

def energy(z, lam, alpha=10.0, beta=10.0):
    sum_radii = np.sum(z[2*26:])
    pen = penalty(z, alpha, beta)
    return -sum_radii + lam * pen

def propose_move(z, sigma_c, sigma_r):
    z_new = z.copy()
    n = 26
    # Choose move type: 0=center jitter, 1=radius perturb, 2=swap pair
    move_type = np.random.randint(0, 3)
    
    if move_type == 0:
        # Center jitter
        i = np.random.randint(0, n)
        z_new[2*i] += np.random.normal(0, sigma_c)
        z_new[2*i+1] += np.random.normal(0, sigma_c)
        # Clip to unit square
        z_new[2*i] = np.clip(z_new[2*i], 0, 1)
        z_new[2*i+1] = np.clip(z_new[2*i+1], 0, 1)
        
    elif move_type == 1:
        # Radius perturb
        i = np.random.randint(0, n)
        z_new[2*n + i] += np.random.normal(0, sigma_r)
        # Clip to nonnegative
        z_new[2*n + i] = np.maximum(z_new[2*n + i], 0)
        
    else:
        # Swap pair
        i, j = np.random.choice(n, 2, replace=False)
        r_i = z_new[2*n + i]
        z_new[2*n + i] = z_new[2*n + j]
        z_new[2*n + j] = r_i
        
    return z_new

def anneal():
    np.random.seed(42)
    n = 26
    
    # Initial configuration: hexagonal grid
    spacing = 0.18
    rows = int(np.ceil(1.0 / spacing))
    cols = int(np.ceil(1.0 / spacing))
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = (j + 0.5) * spacing
            y = (i + 0.5) * spacing
            if x <= 1.0 and y <= 1.0:
                centers.append([x, y])
                
    # Take first 26 points, or repeat if needed
    while len(centers) < n:
        centers.append(centers[len(centers) % len(centers)])
    centers = np.array(centers[:n])
    
    # Assign initial radii based on wall distances
    radii = []
    for i in range(n):
        x, y = centers[i]
        r = min(x, 1-x, y, 1-y)
        radii.append(r)
    radii = np.array(radii)
    
    # Form full vector z = [x0,y0,x1,y1,...,r0,r1,...]
    z = np.concatenate([centers.flatten(), radii])
    
    # Annealing parameters
    T0 = 1.0
    c_T = 0.995
    T = T0
    lam0 = 0.1
    c_lam = 1.03
    lam = lam0
    sigma_c = 0.03
    sigma_r = 0.02
    
    # Track best feasible solution
    best_feasible_z = z.copy()
    best_feasible_sum = np.sum(radii)
    
    # Outer iterations
    for step in range(10000):
        # Propose move
        z_prop = propose_move(z, sigma_c * np.sqrt(T/T0), sigma_r * np.sqrt(T/T0))
        
        # Compute energy change
        E_current = energy(z, lam)
        E_proposed = energy(z_prop, lam)
        delta_E = E_proposed - E_current
        
        # Accept or reject move
        if delta_E <= 0 or np.random.rand() < np.exp(-delta_E / T):
            z = z_prop
            
        # Update temperature and penalty weight
        T *= c_T
        if T < 1e-5:
            break
        lam *= c_lam
        
        # Check if current solution is feasible
        if penalty(z) < 1e-6:
            current_sum = np.sum(z[2*n:])
            if current_sum > best_feasible_sum:
                best_feasible_z = z.copy()
                best_feasible_sum = current_sum
    
    # Return best feasible solution
    centers = best_feasible_z[:2*n].reshape(n, 2)
    radii = best_feasible_z[2*n:]
    return centers, radii

def project_and_ascent(centers, radii, max_iter=500, eta=5e-3, eta_decay=0.95):
    n = centers.shape[0]
    z = np.concatenate([centers.flatten(), radii])
    eta_current = eta
    
    for _ in range(max_iter):
        # Gradient ascent step: move radially outward from each center
        step_size = eta_current
        step = np.zeros_like(z)
        step[2*n:] = 1.0  # Gradient w.r.t. radii (we want larger radii)
        
        # Update z
        z_new = z + step_size * step
        
        # Clamp radii to feasible region
        for i in range(n):
            x, y = z_new[2*i], z_new[2*i+1]
            r_max = min(x, 1-x, y, 1-y)
            z_new[2*n+i] = np.clip(z_new[2*n+i], 0, r_max)
        
        # Overlap resolution
        centers_new = z_new[:2*n].reshape(n, 2)
        radii_new = z_new[2*n:]
        
        for i in range(n):
            for j in range(i+1, n):
                dx = centers_new[i, 0] - centers_new[j, 0]
                dy = centers_new[i, 1] - centers_new[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                r_sum = radii_new[i] + radii_new[j]
                
                if dist < r_sum:
                    overlap = r_sum - dist
                    # Shrink both radii equally
                    radii_new[i] -= overlap / 2
                    radii_new[j] -= overlap / 2
                    
                    # Ensure non-negativity and wall constraints
                    x_i, y_i = centers_new[i]
                    x_j, y_j = centers_new[j]
                    r_i = radii_new[i]
                    r_j = radii_new[j]
                    r_i = max(0, min(r_i, x_i, 1-x_i, y_i, 1-y_i))
                    r_j = max(0, min(r_j, x_j, 1-x_j, y_j, 1-y_j))
                    radii_new[i] = r_i
                    radii_new[j] = r_j
        
        z_new = np.concatenate([centers_new.flatten(), radii_new])
        
        # Step size decay
        eta_current *= eta_decay
        
        # Update z
        z = z_new.copy()
    
    final_centers = z[:2*n].reshape(n, 2)
    final_radii = z[2*n:]
    return final_centers, final_radii
