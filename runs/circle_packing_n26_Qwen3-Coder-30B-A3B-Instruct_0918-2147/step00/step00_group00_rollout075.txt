    def generate_hexagonal_seed(n_circles=26, margin=0.05):

        sqrt3 = np.sqrt(3)
        spacing = 0.8 / np.floor(np.sqrt(n_circles))
        cols = int(np.ceil(1.0 / spacing))
        rows = int(np.ceil(1.0 / (spacing * sqrt3 / 2)))
        
        centers = []
        for i in range(rows):
            for j in range(cols):
                x = i * spacing + (j % 2) * spacing / 2
                y = j * spacing * sqrt3 / 2
                if 0 <= x <= 1 and 0 <= y <= 1:
                    centers.append([x, y])
                if len(centers) >= n_circles:
                    break
            if len(centers) >= n_circles:
                break
        
        centers = np.array(centers[:n_circles])
        
        # Apply margin
        min_x = np.min(centers[:, 0])
        min_y = np.min(centers[:, 1])
        shift_x = margin - min_x
        shift_y = margin - min_y
        centers[:, 0] += shift_x
        centers[:, 1] += shift_y

        return centers

    def compute_distance_matrix(centers):
        n = centers.shape[0]

        for i in range(n):
            for j in range(i+1, n):
                d = np.linalg.norm(centers[i] - centers[j])
                dist_matrix[i, j] = d
                dist_matrix[j, i] = d
        return dist_matrix

            A_ub.append([1, 0])

            A_ub.append([0, 1])
            b_ub.append(1 - centers[i, 1])

                dist = dist_matrix[i, j]
                A_ub.append([1, 1])  # r_i + r_j <= dist
                b_ub.append(dist)

        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)

            row[2*n + i] = -1
