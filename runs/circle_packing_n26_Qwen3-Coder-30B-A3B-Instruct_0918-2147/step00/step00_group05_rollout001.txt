import numpy as np
from scipy.optimize import minimize

def hex_lattice_points(n, d):
    """Generate at least n points in a hexagonal lattice."""
    rows = []
    y = 0.0
    while len(rows) < n:
        row = []
        x = 0.0
        while x <= 1.0 + d:
            row.append((x, y))
            x += d
        rows.append(row)
        y += d * np.sqrt(3) / 2
        if len(rows) % 2 == 0:
            x = d / 2
        else:
            x = 0.0
    points = []
    for row in rows:
        for x, y in row:
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append((x, y))
                if len(points) >= n:
                    break
        if len(points) >= n:
            break
    return np.array(points[:n])

def initial_guess():

    # Use a hexagonal lattice with spacing to fit 26 circles
    # Start with a spacing that places the centers near the edges
    d = 0.25  # Initial guess for spacing
    centers = hex_lattice_points(26, d)
    # Offset to center in the square, make sure all circles are inside
    x_mean = np.mean(centers[:, 0])
    y_mean = np.mean(centers[:, 1])
    centers[:, 0] -= x_mean - 0.5
    centers[:, 1] -= y_mean - 0.5
    # Ensure at least a small buffer from walls
    min_dist = 1.0
    for i in range(26):
        x, y = centers[i]
        min_dist = min(min_dist, x, 1-x, y, 1-y)
    # Set a safe uniform radius
    r = min_dist / 2.0
    if r <= 0:
        r = 0.05  # fallback if too small
    # Create the full list of variables [x0, y0, r0, x1, y1, r1, ...]
    vars = []
    for i in range(26):
        vars.append(centers[i, 0])
        vars.append(centers[i, 1])
        vars.append(r)
    return np.array(vars)

def objective(vars):
    # We want to maximize sum of radii, which is to minimize negative sum
    # vars = [x0, y0, r0, x1, y1, r1, ...]
    radii = vars[2::3]
    return -np.sum(radii)

def wall_constraints(vars):
    # , 0 <= x_i - r_i, x_i + r_i <= 1
    # ->    r_i - x_i <= 0,
    # ->    x_i + r_i - 1 <= 0
    n_circles = 26
    result = []
    for i in range(n_circles):
        x = vars[3*i]
        y = vars[3*i + 1]
        r = vars[3*i + 2]
        result.append(r - x) # r_i - x_i <= 0
        result.append(x + r - 1) # x_i + r_i - 1 <= 0
        result.append(r - y) # r_i - y_i <= 0
        result.append(y + r - 1) # y_i + r_i - 1 <= 0
    return np.array(result)

def nonoverlap_constraints(vars):
    n_circles = 26
    result = []
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            x_i = vars[3*i]
            y_i = vars[3*i + 1]
            r_i = vars[3*i + 2]

            x_j = vars[3*j]
            y_j = vars[3*j + 1]
            r_j = vars[3*j + 2]
            
            # Distance squared
            dist_sq = (x_i - x_j)**2 + (y_i - y_j)**2
            # Required separation squared
            sep_sq = (r_i + r_j)**2
            # For non-overlap: dist_sq >= sep_sq -> sep_sq - dist_sq <= 0
            result.append(sep_sq - dist_sq)
    return np.array(result)
