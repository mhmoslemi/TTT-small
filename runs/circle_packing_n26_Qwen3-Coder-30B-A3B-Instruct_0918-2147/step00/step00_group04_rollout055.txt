import numpy as np
from scipy.optimize import minimize

def hex_lattice_points(n_rows, spacing):
    """Generate hexagonal lattice points."""
    points = []
    for i in range(n_rows):
        y = i * spacing * np.sqrt(3) / 2
        if y > 1:
            break
        offset = (i % 2) * spacing / 2
        n_cols = int((1 - offset) / spacing) + 1
        for j in range(n_cols):
            x = offset + j * spacing
            if x > 1:
                break
            points.append([x, y])
    return np.array(points)

def initial_guess(seed=0):
    """Create initial guess with hexagonal lattice and small radii."""
    np.random.seed(seed)
    spacing = 0.15
    points = hex_lattice_points(10, spacing)
    points = points[:26]
    while len(points) < 26:
        points = np.vstack([points, np.array([[0.5, 0.5]])])
    points = points[:26]
    
    # Add jitter
    jitter = np.random.uniform(-0.02, 0.02, (26, 2))
    points += jitter
    
    # Keep within bounds
    points[:, 0] = np.clip(points[:, 0], 0, 1)
    points[:, 1] = np.clip(points[:, 1], 0, 1)
    
    # Initialize small radii
    radii = np.full(26, 0.02)
    
    # Flatten into variables (x, y, r) for each circle
    vars0 = np.zeros(78)
    vars0[0::3] = points[:, 0]
    vars0[1::3] = points[:, 1]
    vars0[2::3] = radii
    return vars0

def objective(vars):
    """Objective: maximize the sum of radii. So minimize the negative."""
    return -np.sum(vars[2::3])

def constraint_boundary(vars):
    """Constraint function for boundaries."""
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    # Each circle must stay within [0,1]x[0,1] with radius r
    constraints = []
    # x - r >= 0
    constraints.append(x - r)
    # 1 - x - r >= 0
    constraints.append(1 - x - r)
    # y - r >= 0
    constraints.append(y - r)
    # 1 - y - r >= 0
    constraints.append(1 - y - r)
    return np.concatenate(constraints)

def constraint_nonoverlap(vars):
    """Constraint function for non-overlapping."""
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    n = len(x)
    constraints = []
    for i in range(n):
        for j in range(i+1, n):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dist_sq = dx*dx + dy*dy
            dist = np.sqrt(dist_sq)
            r_sum = r[i] + r[j]
            # dist - (r[i] + r[j]) >= 0 for non-overlap
            constraints.append(dist - r_sum)
    return np.array(constraints)

def postprocess(vars):
    """Post-process final solution."""
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Ensure all centers are inside square and radii are positive
    x = np.clip(x, r, 1 - r)
    y = np.clip(y, r, 1 - r)
    r = np.maximum(r, 0.0)
    
    # Update vars
    vars = np.zeros_like(vars)
    vars[0::3] = x
    vars[1::3] = y
    vars[2::3] = r
    
    return vars

def run_packing():
    """Run the full packing algorithm."""
    best_sum = -np.inf
    best_vars = None
    
    NUM_RESTARTS = 5
    for seed in range(NUM_RESTARTS):
        vars0 = initial_guess(seed)
        
        # Create bounds for variables
        bounds = []
        for i in range(26):
            bounds.append((0, 1))  # x bounds
            bounds.append((0, 1))  # y bounds
            bounds.append((0, 0.5))  # r bounds: max radius possible for a single circle in a 1x1 square is 0.5
        
        # Set up constraints
        constraints = [
            {'type': 'ineq', 'fun': constraint_boundary},
            {'type': 'ineq', 'fun': constraint_nonoverlap}
        ]
        
        # Optimize
        res = minimize(objective, vars0,
                      method='SLSQP',
                      bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': 3000, 'ftol': 1e-7})

        if not res.success:
            continue
            
        vars_corr = postprocess(res.x)
        s = np.sum(vars_corr[2::3])
        if s > best_sum:
            best_sum = s
            best_vars = vars_corr

    # Reshape best_vars into centers and radii
    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii = best_vars[2::3]
    
    return centers, radii, float(best_sum)
