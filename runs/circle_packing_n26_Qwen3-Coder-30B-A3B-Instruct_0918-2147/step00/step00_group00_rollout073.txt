import numpy as np
from scipy.optimize import minimize

def make_hex_lattice(n, margin=0.05):
    # Create a hexagonal lattice that fits in [margin, 1-margin] x [margin, 1-margin]
    width = 1 - 2 * margin
    height = 1 - 2 * margin
    dx = width / np.ceil(np.sqrt(n))
    dy = dx * np.sqrt(3) / 2
    points = []
    i = 0
    j = 0
    while len(points) < n:
        x = margin + (i + (j % 2) * 0.5) * dx
        y = margin + j * dy
        if 0 <= x <= 1 and 0 <= y <= 1:
            points.append((x, y))
        # Move to next row/column
        if x + dx > 1 or len(points) >= n:
            j += 1
            i = 0
        else:
            i += 1
    return np.array(points[:n])

def objective(z):
    # z[2*i:2*i+2] = center (x,y), z[2*i+2] = radius
    # objective to maximize sum of radii = -sum(radii)
    radii = z[2::3]
    return -np.sum(radii)

def boundary_constraints(z):
    # x - r >= 0, x + r <= 1, y - r >= 0, y + r <= 1
    # for each circle: 4 constraints
    # compute all 4*26
    result = []
    n_circles = 26
    for i in range(n_circles):
        x = z[2*i]
        y = z[2*i+1]
        r = z[2*i+2]
        result.extend([
            x - r,        # x - r >= 0
            1 - x - r,    # x + r <= 1
            y - r,        # y - r >= 0
            1 - y - r     # y + r <= 1
        ])
    return np.array(result)

def pairwise_constraints(z):
    # Separation constraints between all pairs: (x1-x2)^2 + (y1-y2)^2 - (r1+r2)^2 >= 0
    result = []
    n_circles = 26
    for i in range(n_circles):
        for j in range(i+1, n_circles):
            x1, y1, r1 = z[2*i], z[2*i+1], z[2*i+2]
            x2, y2, r2 = z[2*j], z[2*j+1], z[2*j+2]
            dist_sq = (x1 - x2)**2 + (y1 - y2)**2
            sum_radii_sq = (r1 + r2)**2
            result.append(dist_sq - sum_radii_sq)
    return np.array(result)

def solve_nlp(initial_z):
    # Returns z that minimizes -sum(radii) s.t. constraints are satisfied
    # Assemble constraints: boundary and pairwise
    bounds = []
    n = 26
    for i in range(n):
        bounds.extend([(0,1), (0,1), (0, 0.5)])
    bounds = tuple(bounds)
    constraints = [
        {'type':'ineq', 'fun': boundary_constraints},
        {'type':'ineq', 'fun': pairwise_constraints}
    ]
    # Access user functions through globals
    res = minimize(objective, initial_z, method='SLSQP', bounds=bounds,
                   constraints=constraints, options={'ftol': 1e-9, 'maxiter': 1000})
    return res.x

def nudge_centers(centers, radii):
    # For each circle that's nearly touching a wall, slightly move it away from that wall
    nudged_centers = centers.copy()
    for i in range(26):
        x, y = centers[i]
        r = radii[i]
        # Compute actual distance to all walls and if too small, nudge slightly
        dist_to_left = x
        dist_to_right = 1 - x
        dist_to_bottom = y
        dist_to_top = 1 - y
        for d, move_dir in [
            (dist_to_left, np.array([1, 0])),
            (dist_to_right, np.array([-1, 0])),
            (dist_to_bottom, np.array([0, 1])),
            (dist_to_top, np.array([0, -1]))
        ]:
            if d < 1e-3 and d + r < 1e-2:
                nudged_centers[i] += 0.001 * move_dir
                break
    return nudged_centers

def run_packing():
    initial_centers = make_hex_lattice(26)
    initial_radii = np.full(26, 0.02)
    initial_z = np.concatenate([initial_centers.flatten(), initial_radii])
    final_z = solve_nlp(initial_z)
    final_centers = final_z[::3].reshape((26,2))
    final_radii = final_z[2::3]
    # Iterative refinement
    for _ in range(5):
        old_sum = np.sum(final_radii)
        final_z = solve_nlp(final_z)
        final_centers = final_z[::3].reshape((26,2))
        final_radii = final_z[2::3]
        new_sum = np.sum(final_radii)
        if new_sum - old_sum < 1e-5:
            break
        # Nudge any wall-touching circles
        final_centers = nudge_centers(final_centers, final_radii)
        # Recompute z from new centers and old radii to feed to next solve
        final_z = np.concatenate([final_centers.flatten(), final_radii])
    # Final validation
    valid, msg = validate_packing(final_centers, final_radii)
    if not valid:

            final_radii = final_radii * 0.999999
            valid, msg = validate_packing(final_centers, final_radii)

        else:

    sum_radii = float(np.sum(final_radii))
