import numpy as np
from scipy.optimize import linprog

def initial_hex_centers(k, spacing):
    sqrt3 = np.sqrt(3)
    centers = []
    i = j = 0
    while len(centers) < k:
        x = i * spacing + (j % 2) * spacing / 2
        y = j * spacing * sqrt3 / 2
        if 0 <= x <= 1 and 0 <= y <= 1:
            centers.append([x, y])
        if x > 1:
            j += 1
            i = -1
        i += 1
    return np.array(centers[:k])

def initial_radii(k, r0):
    return np.full(k, r0)

def build_lp_constants(centers, radii):
    n = centers.shape[0]
    # Compute wall distances for each circle
    d_left = centers[:, 0]
    d_right = 1 - centers[:, 0]
    d_bottom = centers[:, 1]
    d_top = 1 - centers[:, 1]
    wall_distances = np.stack([d_left, d_right, d_bottom, d_top], axis=1)

    # Compute pairwise distances
    D = np.zeros((n, n))
    for i in range(n):
        for j in range(i+1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            D[i, j] = D[j, i] = dist
    return wall_distances, D

def solve_radius_lp(consts, radii, graph, alpha):
    wall_distances, D = consts
    n = len(radii)
    # Constraints: A_ub @ x <= b_ub
    # Variables Δr_i with bounds (0, inf)
    # FinalRadii = oldRadii + Δr

    # Wall capacity constraints
    A_ub_wall = []
    b_ub_wall = []
    for i in range(n):
        wall_dist = wall_distances[i]
        max_growth = np.min(wall_dist) - radii[i]
        if max_growth < 0:
            max_growth = 0
        A_ub_wall.append([-1.0])  # Δr_i <= max_growth
        b_ub_wall.append(-max_growth)

    # Pairwise coverage constraints (Δr_i + Δr_j ≤ D_ij - (r_i + r_j))
    A_ub_pair = []
    b_ub_pair = []

    # Graph edges: if an edge exists, we must not allow Δr_i + Δr_j to increase.
    # But in a sense, they can grow by up to D_ij - (r_i + r_j) - D_ij'.
    # If pairwise are touching at start, their current sum of Δr <= 0.
    # We soft use their contact status later.

    for i in range(n):
        for j in range(i+1, n):
            if D[i, j] > radii[i] + radii[j]:  # Only active if could grow further
                # Δr_i + Δr_j <= D_ij - (r_i + r_j)
                A = [0.0] * n
                A[i] = 1.0
                A[j] = 1.0
                A_ub_pair.append(A)
                b_ub_pair.append(D[i, j] - (radii[i] + radii[j]))

    # Graph-based transfer limiter (α-based)
    # Optional constraint:
    # Δr_i ≤ α * Σ_{j in neighbors of i} (D_ij - (r_i + r_j))
    # Only add constraints for circles with neighbors
    A_ub_graph = []
    b_ub_graph = []
    if graph is not None:
        for i in range(n):
            neighbors = graph.get(i, [])
            if neighbors:
                # Build the RHS
                total = 0
                for j in neighbors:
                    total += D[i, j] - (radii[i] + radii[j])
                if total > 0:
                    A = [0.0] * n
                    A[i] = -1.0
                    A_ub_graph.append(A)
                    b_ub_graph.append(-alpha * total)

    # Combine all constraints
    A_ub = A_ub_wall.copy()
    b_ub = b_ub_wall.copy()
    A_ub.extend(A_ub_pair)
    b_ub.extend(b_ub_pair)
    A_ub.extend(A_ub_graph)
    b_ub.extend(b_ub_graph)

    # Objective: maximize sum(Δr_i) => minimize -sum(Δr_i)
    c = np.full(n, -1.0)

    # Bounds
    bounds = [(0, None) for _ in range(n)]

    if len(A_ub) > 0:
        # Solve LP
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', tol=1e-12)
        if not res.success:
            # Fallback: return zeros
            Δr = np.zeros(n)
        else:
            Δr = res.x
    else:
        Δr = np.zeros(n)

    return Δr

def apply_increment(radii, Δr):
    return radii + Δr

def push_feasibility(centers, radii):
    n = centers.shape[0]
    # Keep copies for modification
    new_centers = centers.copy()
    new_radii = radii.copy()

    # Wall clipping first
    for i in range(n):
        x, y = new_centers[i]
        r = new_radii[i]
        # Clamp to square
        d_left = x - r
        d_right = 1 - x - r
        d_bottom = y - r
        d_top = 1 - y - r
        if d_left < 0:
            new_radii[i] -= d_left
        if d_right < 0:
            new_radii[i] -= d_right
        if d_bottom < 0:
            new_radii[i] -= d_bottom
        if d_top < 0:
            new_radii[i] -= d_top

    # Clean any negative radii
    new_radii = np.maximum(new_radii, 0.0)

    # Pairwise overlap check and push
    for i in range(n):
        for j in range(i+1, n):
            dist = np.sqrt(np.sum((new_centers[i] - new_centers[j]) ** 2))
            sum_r = new_radii[i] + new_radii[j]
            if dist < sum_r - 1e-12:
                overlap = sum_r - dist
                # Move them apart
                if dist > 1e-12:
                    norm = (new_centers[i] - new_centers[j]) / dist
                    delta = overlap / 2
                    new_centers[i] -= delta * norm
                    new_centers[j] += delta * norm
                else:
                    # If they're nearly overlapping, push in a random direction
                    # (randomness avoids cycle but keep it deterministic)
                    new_centers[i] += np.array([delta, 0])
                    new_centers[j] -= np.array([delta, 0])
    # Make it actually safe now, one more pass to clamp to walls
    for i in range(n):
        x, y = new_centers[i]
        r = new_radii[i]
        if x - r < 0:
            new_radii[i] = x
        if x + r > 1:
            new_radii[i] = 1 - x
        if y - r < 0:
            new_radii[i] = y
        if y + r > 1:
            new_radii[i] = 1 - y

    return new_centers, new_radii

def update_graph(graph, new_edges, wall_contacts):
    if graph is None:
        graph = {}
    for i, j in new_edges:
        if i not in graph:
            graph[i] = set()
        if j not in graph:
            graph[j] = set()
        graph[i].add(j)
        graph[j].add(i)
    return graph

def run_iterations(centers, radii, max_iter, ε, α):
    graph = {}
    for iter_count in range(max_iter):
        consts = build_lp_constants(centers, radii)
        Δr = solve_radius_lp(consts, radii, graph, α)
        old_sum = np.sum(radii)
        radii = apply_increment(radii, Δr)
        centers, radii = push_feasibility(centers, radii)
        sum_radii = np.sum(radii)
        # Update graph by detecting new edges
        new_edges = []
        new_wall_contacts = []
        for i in range(len(centers)):
            x, y = centers[i]
            r = radii[i]
            if x - r <= 1e-12:
                new_wall_contacts.append((i, 'left'))
            if x + r >= 1 - 1e-12:
                new_wall_contacts.append((i, 'right'))
            if y - r <= 1e-12:
                new_wall_contacts.append((i, 'bottom'))
            if y + r >= 1 - 1e-12:
                new_wall_contacts.append((i, 'top'))
            for j in range(i+1, len(centers)):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if abs(dist - (radii[i] + radii[j])) <= 1e-12:
                    new_edges.append((i, j))
        graph = update_graph(graph, new_edges, new_wall_contacts)
        if sum_radii - old_sum < ε:
            break
    return centers, radii

def final_refinement(centers, radii):
    # One final high-precision LP
    consts = build_lp_constants(centers, radii)
    α = 0.0  # not used in final refinement
    Δr = solve_radius_lp(consts, radii, None, α)
    radii = apply_increment(radii, Δr)
    # Do one final push for hard constraint satisfaction
    centers, radii = push_feasibility(centers, radii)
    return centers, radii

def run_packing():
    n_circles = 26
    spacing = 0.12
    r0 = 0.005
    centers = initial_hex_centers(n_circles, spacing)
    radii = initial_radii(n_circles, r0)
    ε = 1e-5
    max_iter = 200
    α = 0.5
    centers, radii = run_iterations(centers, radii, max_iter, ε, α)
    centers, radii = final_refinement(centers, radii)
    sum_radii = float(np.sum(radii))
    return centers, radii, sum_radii
