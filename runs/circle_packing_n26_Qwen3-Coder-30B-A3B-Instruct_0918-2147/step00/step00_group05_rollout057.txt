import numpy as np
from scipy.optimize import linprog, minimize
from math import sqrt, ceil

def build_hex_lattice(d):
    """Generate a deterministic hexagonal grid of at least d points inside unit square."""
    # Use a hexagonal lattice with spacing s such that we fit d circles
    # Consider spacing s = 2*r where r is the radius of the circle.
    # For hex tiling, horizontal spacing = s, vertical spacing = s*sqrt(3)/2

    # For 26 circles, let's try different spacing values to find a good fit
    # Let initially try s = 0.18 to see how many circles we get

    s = 0.18
    sqrt3_2 = sqrt(3) / 2.0
    rows = ceil(1.0 / (s * sqrt3_2))
    cols = ceil(1.0 / s)
    centers = []

    for i in range(rows):
        y = s * sqrt3_2 * i
        if y > 1.0:
            break
        offset = s * 0.5 if i % 2 == 1 else 0.0
        for j in range(cols):
            x = s * j + offset
            if x > 1.0:
                break
            centers.append([x, y])
            if len(centers) >= d:
                break
        if len(centers) >= d:
            break

    # If not enough points, adjust spacing and recompute
    if len(centers) < d:
        s = 0.2
        sqrt3_2 = sqrt(3) / 2.0
        rows = ceil(1.0 / (s * sqrt3_2))
        cols = ceil(1.0 / s)
        centers = []

        for i in range(rows):
            y = s * sqrt3_2 * i
            if y > 1.0:
                break
            offset = s * 0.5 if i % 2 == 1 else 0.0
            for j in range(cols):
                x = s * j + offset
                if x > 1.0:
                    break
                centers.append([x, y])
                if len(centers) >= d:
                    break
            if len(centers) >= d:
                break
    
    # Ensure we have exactly d centers by sampling/adding more if needed
    # But since our hex pattern is ordered and should be large enough, just clip to 26
    if len(centers) < d:
        # Use a simple square-like pattern to fill remaining
        remaining = d - len(centers)
        base_x = 1.0 - 0.1
        base_y = 1.0 - 0.1
        inc_step = 0.05
        
        for i in range(remaining):
            x = base_x - i * inc_step
            y = base_y - i * inc_step
            if x < 0 or y < 0:
                x = 0.1 + (i % 5) * 0.05
                y = 0.1 + (i // 5) * 0.05
            centers.append([x, y])
    
    centers = np.array(centers[:d])
    return centers

def lp_max_radii(centers):
    """Compute optimal radii given fixed center positions using a linear program."""
    n = centers.shape[0]
    
    # Precompute distances for all pairs
    # For each pair (i, j), constraint is r_i + r_j <= d_ij
    # Use integer indices for efficiency
    # Build A_ub * r <= b_ub
    # Wall constraints per circle
    # r_i <= x_i, x_i, y_i, 1-y_i for all i
    
    # First, prepare wall constraints (4 per circle) and pairwise
    num_constraints = 4 * n + n * (n - 1) // 2
    # So number of rows = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_constraints, n))  # Rows = constraints
    b_ub = np.zeros(num_constraints)

    # Wall constraints (linear)
    for i in range(n):
        # r_i <= x_i   -->  -r_i >= -x_i
        A_ub[4*i, i] = -1.0
        b_ub[4*i] = -centers[i][0]

        # r_i <= 1 - x_i  -->  -r_i >= x_i - 1
        A_ub[4*i+1, i] = -1.0
        b_ub[4*i+1] = centers[i][0] - 1.0

        # r_i <= y_i   -->  -r_i >= -y_i
        A_ub[4*i+2, i] = -1.0
        b_ub[4*i+2] = -centers[i][1]

        # r_i <= 1 - y_i  -->  -r_i >= y_i - 1
        A_ub[4*i+3, i] = -1.0
        b_ub[4*i+3] = centers[i][1] - 1.0

    # Pairwise non-overlapping constraints
    idx = 4 * n
    for i in range(n):
        for j in range(i+1, n):
            d_ij = np.linalg.norm(centers[i] - centers[j])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d_ij
            idx += 1

    c = -np.ones(n)  # Because linprog expects min objective

    # Variable bounds: r >= 0
    bounds = [(0, None) for _ in range(n)]

    # Run linear program
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={"presolve": True, "tol": 1e-9})
        if res.success:
            radii = res.x
            return radii, np.sum(radii)
        else:
            # Fallback to small radii to prevent crash
            radii = np.zeros(n)
            return radii, 0.0

    except Exception:
        # If even a fallback fails return zeros, but we should not reach here in normal
        # Ifs we raise from outside due to non-zero tolerance in validation
        return np.zeros(n), 0.0

def perturb_centers(centers, sigma):
    """Add Gaussian noise to all centers and clamp to [0,1]."""
    new_centers = centers + np.random.normal(0, sigma, centers.shape)
    # Clamp
    new_centers = np.clip(new_centers, 0, 1)
    return new_centers

def anneal_centers(initial_centers):
    """Simulated Annealing for finding good center configuration."""
    np.random.seed(42)

    # Set up temperature schedule
    temps = [0.02, 0.01, 0.005, 0.002]
    sigmas = [0.05, 0.03, 0.015, 0.008]
    depths = [2000, 4000, 3000, 2000]

    current_centers = initial_centers.copy()
    current_radii, current_sum = lp_max_radii(current_centers)

    best_centers = current_centers.copy()
    best_radii = current_radii.copy()
    best_sum = current_sum

    # For early stopping detection
    last_best = best_sum
    streak = 0

    # Run multiple stages
    for stage, (T, sigma, depth) in enumerate(zip(temps, sigmas, depths)):
        for iteration in range(depth):
            # Perturb
            proposed_centers = perturb_centers(current_centers, sigma)
            proposed_radii, proposed_sum = lp_max_radii(proposed_centers)

            # Accept or reject
            if proposed_sum >= current_sum:
                current_sum = proposed_sum
                current_centers = proposed_centers
                current_radii = proposed_radii
                if proposed_sum > best_sum:
                    best_sum = proposed_sum
                    best_centers = proposed_centers.copy()
                    best_radii = proposed_radii.copy()
                    # Reset early stopping counter on improvement
                    last_best = best_sum
                    streak = 0
            else:
                # Accept with probability e^((proposed - current) / T)
                delta = proposed_sum - current_sum
                acc_prob = np.exp(delta / T)
                if np.random.rand() < acc_prob:
                    current_sum = proposed_sum
                    current_centers = proposed_centers
                    current_radii = proposed_radii

            streak += 1
            if streak >= 1000 and abs(last_best - best_sum) < 1e-6:
                break

    return best_centers, best_radii

def local_refine(centers, radii):
    """Refine final configuration using trust-constr with non-linear optimization."""
    n = len(centers)
    # Define the nonlinear program:

    # Minimize -sum(r_i) to maximize sum(r_i)
    def objective(x):
        return -np.sum(x[n:])  # x contains [c1_x, c1_y, ..., cn_x, cn_y, r1, r2, ..., rn]
    
    # Gradient (analytical)
    def gradient(x):
        # [0,...,0, -1,...,-1]
        grad = np.zeros_like(x)
        grad[n*2:] = -1.0
        return grad

    # Constraint functions: h_i(x) = 0 for equality, g_i(x) <= 0 for inequality
    # Non-overlapping
    def constraint_nonoverlap(i, j, x):
        # Circle i: (i*2, i*2+1) (x_i, y_i)
        # Circle j: (j*2, j*2+1) (x_j, y_j)
        # Constraint: (x_i-x_j)^2 + (y_i-y_j)^2 >= (r_i+r_j)^2
        # So make: (x_i-x_j)^2 + (y_i-y_j)^2 - (r_i+r_j)^2 >= 0
        return (x[i*2] - x[j*2])**2 + (x[i*2+1] - x[j*2+1])**2 - (x[n*2+i] + x[n*2+j])**2

    # Wall constraints
    def constraint_wall(i, x):
        # r_i <= x_i, 1-x_i, y_i, 1-y_i
        # So r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
        xs = x[i*2]
        ys = x[i*2+1]
        r = x[n*2+i]
        max_radius = min(xs, 1-xs, ys, 1-ys)
        return max_radius - r  # > 0 if satisfied

    # Helper to build constraints
    def eval_constraints(x):
        cons = []
        # Non-overlap constraints
        for i in range(n):
            for j in range(i+1, n):
                cons.append(constraint_nonoverlap(i, j, x))
        # Wall constraints
        for i in range(n):
            cons.append(constraint_wall(i, x))
        return np.array(cons)

    def Jacobian(x):
        # We compute an approximate Jacobian for constraints based on idx.d. of fixed structure
        # Total variables = 2n + n
        j = np.zeros((len(x), 2*n + n))  # Number of constraints x Total variables = ???
        return j  # Placeholder

    # Bounds (i.e., x in [0, 1] for x,y; r >= 0)
    bounds = []
    for i in range(n):
        bounds.append((0, 1)) # x_i
        bounds.append((0, 1)) # y_i
        bounds.append((0, None)) # r_i

    # Start from current values
    x0 = np.concatenate([centers.flatten(), radii])

    # Linear constraints on lower bound for radii: r_i >= radii[i] (LP-optimality!)
    # This is hard to enforce exactly in non-linear solver, so we can just take the LP solution as a backup
    # And run optimization from that or slightly modify

    # Alternative: Start with a slightly more robust setup.

    # Use scipy.optimize.minimize to perform a constrained optimization
    # Passing the constraint functions as inequalities

    # Inequality constraints (It is better to restrict to valid range)
    constraints = []
    
    # For each circle:
    # r_i >= radii[i]
    def lower_bound_r_i(i):
        def fun(x):
            return x[n*2 + i] - radii[i]
        return {"type": "ineq", "fun": fun}
    
    # Circle i must be in the square
    def in_square_bound(i):
        def fun(x):
            return min(
                x[i*2],              # x <= 1 - 0 = 1
                1 - x[i*2],          # x >= 0
                x[i*2+1],            # y <= 1
                1 - x[i*2+1]         # y >= 0
            )  # >=0 if all bounds holds
        return {"type": "ineq", "fun": fun}   # No idea too check this args matching
    
    # We move in the variable space and then make sure constraints are met.
    # So just reform upper-level and trust that it finds good solution and recheck consistency.

    # Add lower bound constraints
    for i in range(n):
        constraints.append(lower_bound_r_i(i))
        # In square?
        constraints.append(in_square_bound(i))

    # Run optimization
    try:
        result = minimize(
            objective,
            x0,
            method='trust-constr',
            bounds=bounds,
            constraints=constraints,
            jac=gradient,
            options={'disp': False, 'maxiter': 2000}
        )
        if result.success:
            x_final = result.x
            new_centers = x_final[:n*2].reshape(n, 2)
            new_radii = x_final[n*2:]
            new_sum = np.sum(new_radii)
            return new_centers, new_radii, new_sum
        else:
            # If optimization fails, fallback to LP result for safety
            return centers.copy(), radii.copy(), np.sum(radii)
    except:
        return centers.copy(), radii.copy(), np.sum(radii)

def validate_and_return(centers, radii):

def run_packing():
    """Main entry point."""
    # Step 1: Initialize with hexagonal lattice
    initial_centers = build_hex_lattice(26)  # This gives a good starting structure

    # Step 2: Annealing to improve the center positions
    final_centers, final_radii = anneal_centers(initial_centers)

    # Step 3: Local refinement
    refined_centers, refined_radii, refined_sum = local_refine(final_centers, final_radii)

    # Step 4: Validation
    return validate_and_return(refined_centers, refined_radii)
