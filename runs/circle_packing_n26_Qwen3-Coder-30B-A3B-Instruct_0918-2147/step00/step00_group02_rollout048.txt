import numpy as np
from scipy.stats.qmc import Sobol

def initialize_seeds(n, seed=42):
    sobol = Sobol(d=2, seed=seed)
    points = sobol.random(n)
    margin = 0.02
    return np.clip(margin + (1 - 2 * margin) * points, margin, 1 - margin)

def compute_feasible_radii(centers):
    n = centers.shape[0]
    # Compute pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_matrix = np.sqrt(np.sum(diff**2, axis=2))
    # For each point, find the minimum distance to any other point
    # Set diagonal to infinity to ignore self-distance
    np.fill_diagonal(dist_matrix, np.inf)
    min_distances = np.min(dist_matrix, axis=1)
    
    # Compute wall distances
    wall_distances = np.array([
        centers[:, 0],           # left
        1 - centers[:, 0],       # right
        centers[:, 1],           # bottom
        1 - centers[:, 1]        # top
    ])
    wall_dist_min = np.min(wall_distances, axis=0)
    
    # Feasible radius is minimum of wall distance and half of nearest neighbor distance
    feasible_radii = np.minimum(wall_dist_min, 0.5 * min_distances)
    
    return feasible_radii

def redistribute_radii(radii, slack, alpha):
    total_slack = np.sum(slack)
    if total_slack <= 1e-12:
        return radii.copy()
    
    # Redistribute radius from slack circles to others
    # Increase radii of circles with slack, decrease radii of others proportionally
    new_radii = radii.copy()
    
    # Add to wall-limited circles
    add_amount = alpha * slack
    new_radii += add_amount
    
    # Compute how much we need to subtract from other circles
    total_add = np.sum(add_amount)
    # Distribute the reduction proportionally to current radii
    if np.sum(radii) > 1e-12:
        scale = total_add / np.sum(radii)
        for i in range(len(radii)):
            if slack[i] == 0:  # already neighbor-limited
                reduction = scale * radii[i]
                new_radii[i] = max(0, new_radii[i] - reduction)
    
    return new_radii

def push_away(centers, radii, beta, eps):
    n = centers.shape[0]
    new_centers = centers.copy()
    
    for i in range(n):
        ci = centers[i]
        ri = radii[i]
        for j in range(n):
            if i == j:
                continue
            cj = centers[j]
            rj = radii[j]
            diff = ci - cj
            dist = np.sqrt(np.sum(diff**2))
            
            if dist < ri + rj + eps:
                # Push away from j
                delta = (ri + rj + eps - dist)
                push_vec = diff / (dist + eps) * beta * delta
                new_centers[i] += push_vec
    
    # Apply a small margin
    eps_safe = 1e-8
    new_centers = np.clip(new_centers, eps_safe, 1 - eps_safe)
    return new_centers

def iterate(centers, radii, alpha, beta, max_iter):
    for _ in range(max_iter):
        # Compute slack before redistribution
        n = centers.shape[0]
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dist_matrix = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dist_matrix, np.inf)
        min_distances = np.min(dist_matrix, axis=1)
        wall_distances = np.array([
            centers[:, 0],
            1 - centers[:, 0],
            centers[:, 1],
            1 - centers[:, 1]
        ])
        wall_dist_min = np.min(wall_distances, axis=0)
        feasible_radii = np.minimum(wall_dist_min, 0.5 * min_distances)

        slack = 0.5 * min_distances - radii
        slack = np.maximum(slack, 0)  # Ensure non-negative

        # Redistribute radius
        radii = redistribute_radii(radii, slack, alpha)
        # Recalculate radii after redistribution
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dist_matrix = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dist_matrix, np.inf)
        min_distances = np.min(dist_matrix, axis=1)
        wall_distances = np.array([
            centers[:, 0],
            1 - centers[:, 0],
            centers[:, 1],
            1 - centers[:, 1]
        ])
        wall_dist_min = np.min(wall_distances, axis=0)
        radii = np.minimum(wall_dist_min, 0.5 * min_distances)

        # Push away
        centers = push_away(centers, radii, beta, 1e-8)

        # Recalculate after push
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dist_matrix = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dist_matrix, np.inf)
        min_distances = np.min(dist_matrix, axis=1)
        wall_distances = np.array([
            centers[:, 0],
            1 - centers[:, 0],
            centers[:, 1],
            1 - centers[:, 1]
        ])
        wall_dist_min = np.min(wall_distances, axis=0)
        new_radii = np.minimum(wall_dist_min, 0.5 * min_distances)
        
        if np.abs(np.sum(new_radii) - np.sum(radii)) < 1e-5:
            break
        radii = new_radii

    return centers, radii

def repair(centers, radii, eps_final):
    # Shrink radii
    radii = np.maximum(0, radii - eps_final)
    # Clip centers
    centers = np.clip(centers, eps_final, 1 - eps_final)
    return centers, radii

def run_packing():
    n = 26
    # Initialize
    centers = initialize_seeds(n)
    radii = compute_feasible_radii(centers)
    
    # Iterate
    centers, radii = iterate(centers, radii, alpha=0.4, beta=0.3, max_iter=200)

    # Final repair
    eps_final = 5e-10
    centers, radii = repair(centers, radii, eps_final)
    
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
