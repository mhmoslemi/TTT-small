import numpy as np
import cvxpy as cp
from itertools import combinations

def build_socp_problem(n):
    r = cp.Variable(n, nonneg=True)
    # Create parameter for pairwise distances
    dist_params = []
    for i in range(n):
        row = []
        for j in range(i+1, n):
            param = cp.Parameter(nonneg=True)
            row.append(param)
        dist_params.append(row)
    # Objective
    objective = cp.Maximize(cp.sum(r))
    # Constraints
    constraints = []
    # Add boundary constraints
    for i in range(n):
        constraints.append(r[i] <= 1e-12)  # Initialize with small value to avoid None
        constraints.append(r[i] <= 1)  # Initialize with large enough value
    # Build pairwise distance constraints (wij <= dist_ij for all i<j)
    for i in range(n):
        for j in range(i+1, n):
            constraints.append(r[i] + r[j] <= dist_params[i][j-i-1])
    prob = cp.Problem(objective, constraints)
    return prob, r, dist_params

def solve_radii(centres, prob, r, dist_params):
    # compute all pairwise distances and update parameters
    n = len(centres)
    # set up duals array
    duals = np.zeros((n, n))
    # Prepare distance constraints
    idx = 0  # index in dist_params
    for i in range(n):
        for j in range(i+1, n):
            dist = np.linalg.norm(centres[i] - centres[j])
            dist_params[i][j-i-1].value = dist
    # Solve SOCP
    try:
        prob.solve(solver=cp.ECOS, verbose=False, abstol=1e-12, reltol=1e-12)
    except:
        # fallback to less strict
        prob.solve(solver=cp.ECOS, verbose=False, abstol=1e-8, reltol=1e-8)
    if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
        radii = r.value
        # get dual values for non-overlap constraints
        # Build duals matrix from dual values of constraints
        dual_idx = 0
        for i in range(n):
            for j in range(i+1, n):
                duals[i][j] = prob.constraints[dual_idx].dual_value
                duals[j][i] = duals[i][j]
                dual_idx += 1
        return radii, duals
    else:
        # Fallback to handle infeasibility
        full_radii = np.zeros(n)
        for i in range(n):
            cx, cy = centres[i]
            rmax = min(cx, 1-cx, cy, 1-cy)
            full_radii[i] = rmax
        return full_radii, np.zeros((n,n))

def transform_state(state):
    """Transform vector of positions back into centers array."""
    n = len(state) // 2
    return np.array(state).reshape((n, 2))

def energy(centres, prob, r, dist_params):
    radii, _ = solve_radii(centres, prob, r, dist_params)
    return -np.sum(radii)

def random_move(state, rng):
    n = len(state) // 2
    # Transform to centers
    centers = transform_state(state)
    new_centers = centers.copy()
    move_type = rng.choice(['single', 'cluster', 'edge', 'swap'], p=[0.6, 0.25, 0.1, 0.05])
    
    if move_type == 'single':
        idx = rng.integers(n)
        dx = rng.normal(0, 0.02)
        dy = rng.normal(0, 0.02)
        new_centers[idx][0] += dx
        new_centers[idx][1] += dy
        new_centers[idx][0] = np.clip(new_centers[idx][0], 0, 1)
        new_centers[idx][1] = np.clip(new_centers[idx][1], 0, 1)
    elif move_type == 'cluster':
        num_circles = rng.integers(3, min(6, n+1))
        selected = rng.choice(n, num_circles, replace=False)
        # Compute centroid
        cm = np.mean(centers[selected], axis=0)
        # Apply correlated perturbation
        dx = rng.normal(0, 0.02)
        dy = rng.normal(0, 0.02)
        new_centers[selected, 0] += dx
        new_centers[selected, 1] += dy
        new_centers[:, 0] = np.clip(new_centers[:, 0], 0, 1)
        new_centers[:, 1] = np.clip(new_centers[:, 1], 0, 1)
    elif move_type == 'edge':
        idx = rng.integers(n)
        x, y = centers[idx]
        d = 0.005 if r < 0.005 else r * 0.1
        if x < 0.5:
            new_centers[idx][0] -= d
        else:
            new_centers[idx][0] += d
        if y < 0.5:
            new_centers[idx][1] -= d
        else:
            new_centers[idx][1] += d
        new_centers[idx][0] = np.clip(new_centers[idx][0], 0, 1)
        new_centers[idx][1] = np.clip(new_centers[idx][1], 0, 1)
    elif move_type == 'swap':
        # Randomly swap two centers in state representation
        idx1 = rng.integers(n)
        idx2 = rng.integers(n)
        i1, j1 = 2*idx1, 2*idx1+1
        i2, j2 = 2*idx2, 2*idx2+1
        new_centers[idx1] = centers[idx2]
        new_centers[idx2] = centers[idx1]
        # Reconstruct new_state adjusting indices s.t. they become permuted center positions
        new_state = centers.flatten().copy()
        new_state[i1], new_state[i2] = new_state[i2], new_state[i1]
        new_state[j1], new_state[j2] = new_state[j2], new_state[j1]
        return new_state.tolist()

    flat = new_centers.flatten()
    return flat.tolist()

def anneal_chain(seed, prob, r, dist_params):
    rng = np.random.default_rng(seed)
    # Initialize centers
    centers = np.random.rand(26, 2)
    # Or use some predefined patterns like grid, hex, etc. For randomness we'll skip complexity and use random
    times = 0
    niter = 150000
    T = 0.05
    alpha = 0.995
    T_min = 1e-6
    
    state = centers.flatten().tolist()
    best_energy = energy(centers, prob, r, dist_params)
    best_state = state.copy()
    best_centres = centers.copy()
    
    for it in range(niter):
        new_state = random_move(state, rng)
        new_centers = transform_state(new_state)
        new_energy = energy(new_centers, prob, r, dist_params)
        delta = new_energy - best_energy
        if delta < 0:
            accept = True
        else:
            try:
                accept = np.random.rand() < np.exp(-delta / T)
            except:
                accept = False
        if accept:
            state = new_state
            best_energy = new_energy
            if new_energy < energy(best_centres, prob, r, dist_params):
                best_centres = new_centers.copy()
        if it % 1000 == 0:  # Normalized updates
            T *= alpha
        if T < T_min:
            break
    return best_centres, best_energy

def local_refine(centres, prob, r, dist_params):
    max_iter = 10
    for _ in range(max_iter):
        radii, duals = solve_radii(centres, prob, r, dist_params)
        # Find active constraints: dual > threshold
        active_pairs = []
        thresh = 1e-6
        for i in range(len(centres)):
            for j in range(i+1, len(centres)):
                if duals[i][j] > thresh:
                    active_pairs.append((i, j))
        if not active_pairs:
            break
        target_count = len(active_pairs)
        # Calculate required update for each center to meet active constraints
        # We'll stabilize with least squares adjustment of center positions
        # Using Gauss-Newton method to solve: Ax = b where A ~ Jacobian of constraints, b ~ residuals
        dx = np.zeros((len(centres), 2))
        for i, (k, l) in enumerate(active_pairs):
            # One constraint like ||c_k - c_l|| = r_k + r_l => (cx_k-cx_l)^2 + (cy_k - cy_l)^2 - (r_k + r_l)^2 = 0
            # Gradient for k:     2 (cx_k - cx_l) - 2 (r_k + r_l) * ( d(r_k + r_l)/dc_k ) = 0
            # d(r_k + r_l)/dc_k = 0, but we're holding radii fixed in inner problem
            # So we have linear constraint A*dx = b 
            # But better approach: fix r_k and r_l, then explicitly solve for centers to make ||c_k - c_l|| = r_k + r_l
            c1 = centres[k]
            c2 = centres[l]
            vec = c1 - c2
            dist = np.linalg.norm(vec)
            if dist < 1e-10 or abs(dist - (radii[k] + radii[l])) < 1e-12:
                continue
            grad = vec / dist
            delta = (radii[k] + radii[l]) - dist
            scale = 0.5 * delta / (dist + 1e-12)  # Scale factor to adjust to make right-hand side happen
            dx[k] += scale * grad
            dx[l] -= scale * grad
        if np.allclose(dx, 0, atol=1e-12):
            break
        # Update center positions slightly
        centres += dx * 0.1
        # Keep within bounds
        centres = np.clip(centres, 0, 1)
    return centres

def finalise(centres, prob, r, dist_params):
    # Final high precision solve to avoid small violations
    radii, _ = solve_radii(centres, prob, r, dist_params)
    # Clamp invalid radii
    for i in range(len(centres)):
        cx, cy = centres[i]
        rmax = min(cx, 1-cx, cy, 1-cy)
        if radii[i] > rmax + 1e-12:
            radii[i] = rmax
    return centres, radii

def run_packing():
    N = 26
    prob, r, dist_params = build_socp_problem(N)
    # Create multiple chains
    rng = np.random.default_rng(42)  # Fixed seed for reproducibility but multiple runs
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    best_total = 0
    seeds = np.arange(8)
    for seed in seeds:
        # Run one chain
        mean_centers = np.random.rand(N, 2)
        best_centres, _ = anneal_chain(seed, prob, r, dist_params)
        refined = local_refine(best_centres, prob, r, dist_params)
        final_centers, final_radii = finalise(refined, prob, r, dist_params)
        total = np.sum(final_radii)
        if total > best_sum:
            best_sum = total
            best_centers = final_centers
            best_radii = final_radii
            best_total = total
    assert best_centers is not None
    assert best_radii is not None
    return best_centers, best_radii, best_total
