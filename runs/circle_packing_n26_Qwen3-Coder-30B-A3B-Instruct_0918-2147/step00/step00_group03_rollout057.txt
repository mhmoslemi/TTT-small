import numpy as np
from scipy.optimize import linprog, minimize

def build_lp_model(n_circles):
    # Build the LP model for circle packing
    # Variables: [x1, y1, r1, x2, y2, r2, ..., xn, yn, rn]
    n_vars = 3 * n_circles
    c = np.zeros(n_vars)
    c[2::3] = -1.0  # Coefficients for radii (negative because we maximize sum of radii)

    # Build A_ub, b_ub (linear inequality constraints)
    A_ub = []
    b_ub = []

    # Wall constraints (r <= x, r <= 1-x, r <= y, r <= 1-y)
    for i in range(n_circles):
        row = np.zeros(n_vars)
        row[3*i + 2] = -1  # -r_i
        row[3*i + 0] = 1   # x_i
        A_ub.append(row)
        b_ub.append(0)

        row = np.zeros(n_vars)
        row[3*i + 2] = -1  # -r_i
        row[3*i + 0] = -1  # -x_i
        A_ub.append(row)
        b_ub.append(-1)

        row = np.zeros(n_vars)
        row[3*i + 2] = -1  # -r_i
        row[3*i + 1] = 1   # y_i
        A_ub.append(row)
        b_ub.append(0)

        row = np.zeros(n_vars)
        row[3*i + 2] = -1  # -r_i
        row[3*i + 1] = -1  # -y_i
        A_ub.append(row)
        b_ub.append(-1)

    # Separation constraints (axis-aligned)
    for i in range(n_circles):
        for j in range(i+1, n_circles):
            # x_i + r_i <= x_j - r_j  (i to the left of j)
            row = np.zeros(n_vars)
            row[3*i + 0] = 1   # x_i
            row[3*i + 2] = 1   # r_i
            row[3*j + 0] = -1  # -x_j
            row[3*j + 2] = 1   # r_j
            A_ub.append(row)
            b_ub.append(0)

            # x_j + r_j <= x_i - r_i  (j to the left of i)
            row = np.zeros(n_vars)
            row[3*j + 0] = 1   # x_j
            row[3*j + 2] = 1   # r_j
            row[3*i + 0] = -1  # -x_i
            row[3*i + 2] = 1   # r_i
            A_ub.append(row)
            b_ub.append(0)

            # y_i + r_i <= y_j - r_j  (i below j)
            row = np.zeros(n_vars)
            row[3*i + 1] = 1   # y_i
            row[3*i + 2] = 1   # r_i
            row[3*j + 1] = -1  # -y_j
            row[3*j + 2] = 1   # r_j
            A_ub.append(row)
            b_ub.append(0)

            # y_j + r_j <= y_i - r_i  (j below i)
            row = np.zeros(n_vars)
            row[3*j + 1] = 1   # y_j
            row[3*j + 2] = 1   # r_j
            row[3*i + 1] = -1  # -y_i
            row[3*i + 2] = 1   # r_i
            A_ub.append(row)
            b_ub.append(0)

    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)

    return c, A_ub, b_ub

def solve_lp(c, A_ub, b_ub):
    # Solve the LP using high performance solver
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, method="highs")
    assert result.success, "LP solver failed to find an optimal solution"
    vars_sol = result.x
    centers = vars_sol[::3][:26].reshape(-1, 2)
    radii = vars_sol[2::3][:26]
    return centers, radii

def detect_diagonal_conflicts(centers, radii):
    n = len(centers)
    conflicts = []
    # Find pairs that have no active axis constraint but may overlap
    # We'll check all pairs and compute Euclidean distance for both directions
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            if dist < radii[i] + radii[j]:
                conflicts.append((i, j))
    return conflicts
