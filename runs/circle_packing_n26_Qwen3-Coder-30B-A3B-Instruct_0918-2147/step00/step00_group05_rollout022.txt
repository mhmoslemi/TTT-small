import numpy as np

def hex_lattice(d):
    """Generate hexagonal lattice points in unit square."""
    rows = []
    y = 0.0
    while y <= 1.0:
        row = []
        x = 0.0
        dy = d * np.sqrt(3) / 2
        while x <= 1.0:
            row.append([x, y])
            x += d
        rows.append(row)
        y += dy
        if len(rows) % 2 == 1:
            # Shift odd rows
            rows[-1] = [[p[0] + d/2, p[1]] for p in rows[-1]]
    # Flatten and take first 20
    points = []
    for row in rows:
        points.extend(row)
        if len(points) >= 20:
            break
    points = points[:20]
    # Sort by (y, x)
    points.sort(key=lambda p: (p[1], p[0]))
    return np.array(points)

def add_border_circles(points, d):
    """Add 6 border-enhanced circles to core."""
    r = d / 4
    border = [
        [r, r], [1-r, r], [r, 1-r], [1-r, 1-r],
        [0.5, r], [0.5, 1-r]
    ]
    return np.vstack([points, border])

def initial_radii(points):
    """Compute uniform initial radius for all circles."""
    n = points.shape[0]
    # Wall distances
    wall_distances = np.array([
        points[:, 0], 1 - points[:, 0],
        points[:, 1], 1 - points[:, 1]
    ]).min(axis=0)
    # Closest neighbor distances (vectorized)
    dist_matrix = np.linalg.norm(
        points[:, np.newaxis, :] - points[np.newaxis, :, :], axis=2
    )
    np.fill_diagonal(dist_matrix, np.inf)
    nearest_distances = dist_matrix.min(axis=1)
    # Initial uniform radius
    r0 = min(wall_distances.min(), (nearest_distances / 2).min())
    return np.full(n, r0)

def inflate_radii(centers, radii):
    """Inflate radii to maximum feasible values."""
    n = centers.shape[0]
    new_radii = radii.copy()
    # Distance matrix
    dist_matrix = np.linalg.norm(
        centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=2
    )
    # For each circle
    for i in range(n):
        # Wall constraints
        wall_caps = [
            centers[i, 0], 1 - centers[i, 0],
            centers[i, 1], 1 - centers[i, 1]
        ]
        # Pairwise constraints
        pairwise_caps = []
        for j in range(n):
            if i != j:
                pairwise_caps.append(dist_matrix[i, j] - radii[j])
        # All caps
        all_caps = wall_caps + pairwise_caps
        # Minimal upper bound
        cap = min(all_caps)
        if cap > new_radii[i] + 1e-7:
            new_radii[i] = cap
    return new_radii

def relax_centers(centers, radii, steps=10):
    """Relax centers to improve packing by force-based method."""
    n = centers.shape[0]
    # Constants
    kappa = 0.02
    lambda_wall = 0.01
    gamma = 0.02
    beta = 0.02
    alpha = 0.5
    # Store old
    old_centers = centers.copy()
    for _ in range(steps):
        # Compute forces
        forces = np.zeros_like(centers)
        # Pairwise repulsion
        dist_matrix = np.linalg.norm(
            centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=2
        )
        for i in range(n):
            for j in range(n):
                if i != j:
                    dij = dist_matrix[i, j]
                    if dij < radii[i] + radii[j] + beta:
                        direction = centers[i] - centers[j]
                        dist = np.linalg.norm(direction)
                        if dist > 1e-10:
                            force_magnitude = kappa * max(0, beta - dij)
                            forces[i] += force_magnitude * direction / dist
        # Wall forces
        for i in range(n):
            x, y = centers[i]
            fx, fy = 0.0, 0.0
            if x - radii[i] < gamma:
                fx += lambda_wall * max(0, gamma - (x - radii[i]))
            if 1 - x - radii[i] < gamma:
                fx -= lambda_wall * max(0, gamma - (1 - x - radii[i]))
            if y - radii[i] < gamma:
                fy += lambda_wall * max(0, gamma - (y - radii[i]))
            if 1 - y - radii[i] < gamma:
                fy -= lambda_wall * max(0, gamma - (1 - y - radii[i]))
            forces[i] += [fx, fy]
        # Update positions
        centers += alpha * forces
        # Clamp to legal region
        centers[:, 0] = np.clip(centers[:, 0], radii, 1 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1 - radii)
        # Check feasibility
        dist_matrix = np.linalg.norm(
            centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=2
        )
        overlap = dist_matrix < radii[:, np.newaxis] + radii[np.newaxis, :]
        np.fill_diagonal(overlap, False)
        if overlap.any():
            # Revert
            centers[:] = old_centers
            break
    return centers

def run_packing():
    # 1. Hexagonal core
    d = 0.18
    core_points = hex_lattice(d)
    # 2. Add border circles
    centers = add_border_circles(core_points, d)
    # 3. Initial radii
    radii = initial_radii(centers)
    # 4. Main optimization loop
    MAX_OUTER = 30
    previous_sum = np.sum(radii)
    for _ in range(MAX_OUTER):
        radii = inflate_radii(centers, radii)
        centers = relax_centers(centers, radii)
