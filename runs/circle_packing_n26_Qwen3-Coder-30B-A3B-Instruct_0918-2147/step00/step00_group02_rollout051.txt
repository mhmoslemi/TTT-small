import numpy as np
from scipy.optimize import minimize
from sobol_sequence import i4_sobol_generate


def initial_configuration(n=26):
    # Generate Sobol sequence points in [0,1]^2
    points = i4_sobol_generate(2, n)
    centers = points.copy()

    # Compute initial radii
    radii = np.zeros(n)
    for i in range(n):
        x, y = centers[i]
        r_wall = min(x, 1 - x, y, 1 - y)
        # Find nearest neighbor in terms of distance
        min_dist = np.inf
        for j in range(n):
            if i != j:
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < min_dist:
                    min_dist = dist
        r_nearest = 0.5 * min_dist if min_dist > 0 else 0.5
        radii[i] = min(r_wall, r_nearest)

    return centers, radii


def unpack(z):
    n = len(z) // 3
    centers = z[:2*n].reshape((n, 2))
    radii = z[2*n:]
    return centers, radii


def pack(centers, radii):
    return np.concatenate([centers.flatten(), radii])


def energy_and_grad(z):
    centers, radii = unpack(z)
    n = centers.shape[0]

    # Distance matrix
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))

    # Wall interactions
    wall_pen = np.zeros(n)
    dxs = np.array([radii - centers[:, 0], radii - (1 - centers[:, 0]),
                    radii - centers[:, 1], radii - (1 - centers[:, 1])])
    wall_pen = np.sum(np.maximum(0, dxs)**2, axis=0)

    # Pair interactions
    sums = radii[:, None] + radii[None, :]
    pairs = np.maximum(0, sums - dists)
    pair_pen = np.sum(pairs**2) / 2  # Divided by 2 to prevent double-counting

    # Budget penalty
    S = np.sum(radii)
    B = 3.0  # Initial budget (will be adjusted)
    lambda_budget = 100.0
    budget_pen = lambda_budget * (B - S)**2

    # Total energy
    energy = np.sum(wall_pen) + pair_pen + budget_pen

    # Gradients
    grad = np.zeros_like(z)

    # Gradients wrt centers
    grad_centers = np.zeros_like(centers)

    # Wall terms
    wall_grads = np.zeros((n, 2))
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # x component
        dx = 2 * (r - x) * (r > x) if r > x else 0.
        dx += 2 * (r - (1 - x)) * (r > 1 - x) if r > 1 - x else 0.
        # y component
        dy = 2 * (r - y) * (r > y) if r > y else 0.
        dy += 2 * (r - (1 - y)) * (r > 1 - y) if r > 1 - y else 0.
        wall_grads[i] = np.array([dx, dy])

    # Pair terms
    pair_grads = np.zeros((n, 2))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] > 0 and (radii[i] + radii[j] > dists[i, j]):
                dx = -(radii[i] + radii[j] - dists[i, j]) * diff[i, j, 0] / dists[i, j]
                dy = -(radii[i] + radii[j] - dists[i, j]) * diff[i, j, 1] / dists[i, j]
                pair_grads[i] += np.array([dx, dy])
                pair_grads[j] -= np.array([dx, dy])

    # Total gradient
    grad_centers = wall_grads + pair_grads

    # Gradient wrt radii
    grad_radii = np.zeros_like(radii)
    # wall terms
    walls_vals = np.array([radii - centers[:, 0], radii - (1 - centers[:, 0]),
                           radii - centers[:, 1], radii - (1 - centers[:, 1])])
    for i in range(n):
        walls = walls_vals[:, i]
        for k, w in enumerate(walls):
            if w > 0:
                grad_radii[i] += 2 * w
    # pair terms
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] > 0 and (radii[i] + radii[j] > dists[i, j]):
                grad_radii[i] += 2 * (radii[i] + radii[j] - dists[i, j])
                grad_radii[j] += 2 * (radii[i] + radii[j] - dists[i, j])
    # budget term
    grad_radii -= 2 * lambda_budget * (B - S)

    grad[:2*n] = grad_centers.flatten()
    grad[2*n:] = grad_radii

    return energy, grad


def inner_optimize(z0, B, lambda_budget):
    def func_and_grad(z):
        return energy_and_grad(z)

    bounds = [(0, 1)] * (2*len(z0)//3) + [(0, 0.5)] * (len(z0)//3)
    opt = minimize(func_and_grad, z0, method='L-BFGS-B', bounds=bounds,
                   jac=True, options={'maxiter': 300})
    return opt.x


def max_feasible_budget(z_start):
    B = 3.0
    delta = 5e-4
    ll = 1e-13  # tolerance for feasibility

    while delta > 1e-6:
        # Try to increase B slightly
        try:
            z_sol = inner_optimize(z_start, B, 100.0)
            centers, radii = unpack(z_sol)
            max_violation = 0
            n = len(centers)
            for i in range(n):
                x, y = centers[i]
                r = radii[i]
                max_violation = max(max_violation,
                                   max(0, r - x),
                                   max(0, r - (1 - x)),
                                   max(0, r - y),
                                   max(0, r - (1 - y)))
            for i in range(n):
                for j in range(i + 1, n):
                    d = np.sqrt(np.sum((centers[i] - centers[j])**2))
                    overlap = (radii[i] + radii[j]) - d
                    if overlap > max_violation:
                        max_violation = overlap

            if max_violation <= ll:
                # feasible
                B += delta
            else:
                B -= delta
        except:
            # If optimization fails, decrease B
            B -= delta
        delta *= 0.5
    return z_sol


def postprocess(z):
    centers, radii = unpack(z)
    eps = 5e-10

    # Shrink radii
    radii = np.maximum(0, radii - eps)
    # Clamp centers
    centers = np.clip(centers, eps, 1.0 - eps)

    # Optional repair pass: try to fix overlaps
    n = centers.shape[0]
    MAX_REPAIR_ITERS = 5
    for _ in range(MAX_REPAIR_ITERS):
        any_fixed = False
        for i in range(n):
            for j in range(i + 1, n):
                d = np.sqrt(np.sum((centers[i] - centers[j])**2))
                expected = radii[i] + radii[j]
                if d < expected:
                    # Overlap exists
                    overlap = expected - d
                    # Reduce both radii slightly
                    delta = overlap / 2 + eps
                    radii[i] = max(0, radii[i] - delta)
                    radii[j] = max(0, radii[j] - delta)
                    any_fixed = True
        if not any_fixed:
            break

    # Final sum
    sum_r = np.sum(radii)

    return centers, radii, sum_r


def run_packing():
    # Initial configuration
    centers, radii = initial_configuration(26)
    z0 = pack(centers, radii)
    
    # Global optimization
    z_final = max_feasible_budget(z0)

    # Post-process
    centers, radii, sum_r = postprocess(z_final)

    # Validate final result
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise RuntimeError(f"Validation failed: {msg}")

    return centers, radii, sum_r
