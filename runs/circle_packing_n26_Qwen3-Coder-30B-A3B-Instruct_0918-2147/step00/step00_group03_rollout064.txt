import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
from itertools import combinations

def init_candidate_grid(radii_list, grid_spacing_factor=1.0):
    # Generate candidate center-radius pairs
    candidates = []
    for r in radii_list:
        # Grid spacing ensures no overlap for same-radius circles
        spacing = r * np.sqrt(2) * grid_spacing_factor
        x_coords = np.arange(r, 1 - r + spacing / 2, spacing)
        y_coords = np.arange(r, 1 - r + spacing / 2, spacing)
        for x in x_coords:
            for y in y_coords:
                candidates.append((x, y, r))
    return candidates

def build_overlap_matrix(candidates):
    # Build overlap matrix
    n = len(candidates)
    overlap = np.zeros((n, n), dtype=bool)
    centers = np.array([[c[0], c[1]] for c in candidates])
    radii = np.array([c[2] for c in candidates])
    distances = cdist(centers, centers)
    # For each pair, check if they would overlap
    for i in range(n):
        for j in range(i + 1, n):
            if distances[i, j] < radii[i] + radii[j]:
                overlap[i, j] = True
                overlap[j, i] = True
    return overlap

def solve_discrete_bip(candidates, overlap_matrix, target_sum):
    n = len(candidates)
    x0 = np.zeros(n, dtype=int)
    # Greedy initial solution (bounded by target)
    selected = []
    used = set()
    indices = list(range(n))
    # Score each by radius
    scores = [c[2] for c in candidates]
    
    # Sort by score descending
    sorted_indices = sorted(range(n), key=lambda i: scores[i], reverse=True)
    max_sum = 0
    
    for i in sorted_indices:
        if i in used:
            continue
        # Check overlap with already selected
        valid = True
        for j in selected:
            if overlap_matrix[i, j]:
                valid = False
                break
        if valid:
            selected.append(i)
            used.add(i)
            max_sum += scores[i]
            if max_sum >= target_sum:
                break
    # Reconstruct selection
    selected_ids = np.array(selected)
    centers = np.array([[candidates[i][0], candidates[i][1]] for i in selected])
    radii = np.array([candidates[i][2] for i in selected])
    return centers, radii

def max_radii_fixed_centres(centers, radii_initial):
    # Fixed centers, maximize radii
    n = len(centers)
    radii = radii_initial.copy()
    
    # Stop if we have enough precision
    for _ in range(50): # Empirical limit
        old_radii = radii.copy()
        for i in range(n):
            # Compute distance to all others
            center_i = centers[i]
            dists = np.sqrt(np.sum((centers - center_i)**2, axis=1))
            # Max allowed radius is min of (distance - radius of other) for all others
            # We do all pairs (excluding i)
            allowed = 1.0 - radii[i]
            for j in range(n):
                if i == j:
                    continue
                min_radius = dists[j] - radii[j]
                if min_radius < allowed:
                    allowed = min_radius
            # Wall constraints
            wall_constr = np.array([
                radii[i] - (1 - centers[i][0]),  # right
                radii[i] - centers[i][0],        # left
                radii[i] - (1 - centers[i][1]),  # top
                radii[i] - centers[i][1]         # bottom
            ])
            wall_constr = np.min(wall_constr)
            if wall_constr < allowed:
                allowed = wall_constr
            # Set to the maximum valid value, clamped to be non-negative
            radii[i] = max(0.0, allowed)
        if np.allclose(old_radii, radii, rtol=1e-12):
            break
    return radii

def nlp_refine(centers, radii_initial):
    n = len(centers)
    # Variables: [x0,y0, x1,y1, ..., x25,y25, r0,r1, ..., r25]
    # Initial guess
    initial = np.zeros(2*n + n)
    initial[0::2] = centers[:, 0]
    initial[1::2] = centers[:, 1]
    initial[2*n:] = radii_initial
    
    def obj(x):
        return -np.sum(x[2*n:])
    
    def jac(x):
        # [-1,-1,...,-1] for radii part
        j = np.zeros_like(x)
        j[2*n:] = -1.0
        return j
    
    # Constraints
    cons = []
    
    # Wall constraints: for each circle i
    #  r_i - x_i <= 0
    #  r_i + x_i - 1 <= 0
    #  r_i - y_i <= 0
    #  r_i + y_i - 1 <= 0
    # These are lower bounds on 1e-12
    for i in range(n):
        r_idx = 2*n + i
        x_idx = 2*i
        y_idx = 2*i + 1
        
        # r_i - x_i <= 0
        cons.append(lambda x, i=i, r_idx=r_idx, x_idx=x_idx: x[r_idx] - x[x_idx])
        # r_i + x_i - 1 <= 0
        cons.append(lambda x, i=i, r_idx=r_idx, x_idx=x_idx: x[r_idx] + x[x_idx] - 1)
        # r_i - y_i <= 0
        cons.append(lambda x, i=i, r_idx=r_idx, y_idx=y_idx: x[r_idx] - x[y_idx])
        # r_i + y_i - 1 <= 0
        cons.append(lambda x, i=i, r_idx=r_idx, y_idx=y_idx: x[r_idx] + x[y_idx] - 1)
        
    # Non-overlap:
    for i, j in combinations(range(n), 2):
        r_idx1 = 2*n + i
        r_idx2 = 2*n + j
        x_idx1 = 2*i
        x_idx2 = 2*j
        y_idx1 = 2*i + 1
        y_idx2 = 2*j + 1
        
        def overlap_cons(x, i=i, j=j, r_idx1=r_idx1, r_idx2=r_idx2, 
                         x_idx1=x_idx1, x_idx2=x_idx2, y_idx1=y_idx1, y_idx2=y_idx2):
            dx = x[x_idx1] - x[x_idx2]
            dy = x[y_idx1] - x[y_idx2]
            dist = np.sqrt(dx*dx + dy*dy)
            return x[r_idx1] + x[r_idx2] - dist
        
        cons.append(overlap_cons)
        
    bounds = []
    for i in range(2*n):
        if i % 2 == 0:
            bounds.append((0, 1))  # x coordinate
        else:
            bounds.append((0, 1))  # y coordinate
    
    for i in range(n):
        bounds.append((0, 1))
    
    # Solve
    res = minimize(obj, initial, method='trust-constr', 
                   jac=jac, constraints=cons, bounds=bounds,
                   options={'maxiter': 300, 'xtol': 1e-9, 'gtol': 1e-9})
    
    # Extract solution
    centers_refined = np.zeros((n, 2))
    radii_refined = np.zeros(n)
    
    centers_refined[:, 0] = res.x[0::2]
    centers_refined[:, 1] = res.x[1::2]
    radii_refined[:] = res.x[2*n:]
    
    return centers_refined, radii_refined

def run_packing():
    # Define radius discretization
    radii_list = [0.025, 0.035, 0.045, 0.055, 0.065, 0.075, 0.085, 0.095, 0.105, 0.115]
    # Get candidate grid
    candidates = init_candidate_grid(radii_list)
    # Create overlap matrix
    overlap_matrix = build_overlap_matrix(candidates)
    # Solve the discrete BIP
    centers, radii = solve_discrete_bip(candidates, overlap_matrix, 2.636)
    # Refine radii with fixed centers
    radii = max_radii_fixed_centres(centers, radii)
    # Use trust-constr to refine positions and radii
    centers_refined, radii_refined = nlp_refine(centers, radii)
    # Final projection
    radii_final = max_radii_fixed_centres(centers_refined, radii_refined)
    # Final result
    return centers_refined, radii_final, np.sum(radii_final)
