import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def init_grid(n_circles=26, spacing=0.07, jitter=0.001):
    """Generate initial configuration with perturbed hexagonal grid"""
    sqrt3 = np.sqrt(3)
    y_coords = []
    x_coords = []
    k = 0
    while len(x_coords) < n_circles:
        y_pos = spacing/2 + k * spacing * sqrt3/2
        if y_pos > 1 - spacing/2:
            break
        offset = (k % 2) * spacing/2
        m = 0
        while len(x_coords) < n_circles:
            x_pos = spacing/2 + m * spacing + offset
            if x_pos > 1 - spacing/2:
                break
            if (x_pos > spacing/2 + 0.02 and x_pos < 1 - spacing/2 - 0.02 and
                y_pos > spacing/2 + 0.02 and y_pos < 1 - spacing/2 - 0.02):
                x_coords.append(x_pos)
                y_coords.append(y_pos)
            m += 1
        k += 1
    centers = np.column_stack([x_coords[:n_circles], y_coords[:n_circles]])
    np.random.seed(0)
    centers += np.random.uniform(-jitter, jitter, centers.shape)
    centers = np.clip(centers, 0, 1)
    return centers

def compute_forces(centers, radii, wall_stiff=10.0, pair_stiff=5.0):
    n = centers.shape[0]
    forces = np.zeros_like(centers)
    
    # Wall forces
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        dd = np.array([
            x - r,   # left
            1 - x - r, # right
            y - r,   # bottom
            1 - y - r  # top
        ])
        wall_forces = np.array([
            [-1.0, 0], [1.0, 0], [0, -1.0], [0, 1.0]
        ])
        for k in range(4):
            if dd[k] < 0:
                forces[i] += wall_stiff * dd[k] * wall_forces[k]
    
    # Pair forces
    dist_mat = cdist(centers, centers)
    for i in range(n):
        for j in range(i+1, n):
            dist = dist_mat[i, j]
            gap = dist - (radii[i] + radii[j])
            if gap < 0:
                diff = centers[i] - centers[j]
                if dist > 1e-12:
                    dir_vec = diff / dist
                    force = pair_stiff * (-gap) * dir_vec
                    forces[i] += force
                    forces[j] -= force
    return forces

def grow_radii(centers, radii, forces, dt=0.02, alpha=0.001):
    # Determine which circles can grow
    can_grow = np.ones(len(radii), dtype=bool)
    for i in range(len(radii)):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
            can_grow[i] = False
        for j in range(len(radii)):
            if i == j:
                continue
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j]:
                can_grow[i] = False
                break

    dot_r = np.where(can_grow, alpha, 0.0)
    radii_new = radii + dt * dot_r
    return radii_new, dot_r

def max_radii_fixed_centers(centers, radii, max_iter=50):
    """Compute maximal radii for fixed centers"""
    n = centers.shape[0]
    r_new = radii.copy()
    for _ in range(max_iter):
        r_old = r_new.copy()
        for i in range(n):
            x, y = centers[i]
            r = r_new[i]
            max_r = min(x, 1-x, y, 1-y)
            for j in range(n):
                if i == j:
                    continue
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                max_r = min(max_r, dist - r_new[j])
            r_new[i] = max(0, max_r)
        if np.allclose(r_old, r_new, rtol=1e-9):
            break
    return r_new

def objective_sum_radii(centers):
    """Evaluate sum of radii as function of centers"""
    n = len(centers) // 2
    centers_mat = centers.reshape(n, 2)
    radii = max_radii_fixed_centers(centers_mat, np.ones(n) * 0.001)
    return -np.sum(radii)

def run_packing():
    np.random.seed(0)
    centers = init_grid()
    radii = np.ones(len(centers)) * 0.001
    v = np.zeros_like(centers)
    dt = 0.02
    gamma = 1.0
    max_steps = 5000
    last_grown = np.zeros(len(centers), dtype=bool)
    
    for step in range(max_steps):
        forces = compute_forces(centers, radii)
        v = v + dt * (forces - gamma * v)
        centers = centers + dt * v
        centers = np.clip(centers, 0, 1)
        radii, dot_r = grow_radii(centers, radii, forces, dt, alpha=0.001)
        if not np.any(dot_r > 1e-12) and np.max(np.linalg.norm(forces, axis=1)) < 1e-6:
            break
        
    # Compute exact max radii for frozen centers
    radii = max_radii_fixed_centers(centers, radii)
    
    # Refine using L-BFGS-B
    centers_flat = centers.flatten()
    result = minimize(objective_sum_radii, centers_flat, method='L-BFGS-B', options={'maxiter': 2000, 'gtol': 1e-9})
    
    centers = result.x.reshape(-1, 2)
    radii = max_radii_fixed_centers(centers, np.ones(centers.shape[0]) * 0.001)
    sum_r = np.sum(radii)
    
    # Validate
    valid, msg = validate_packing(centers, radii)
    if not valid:

    return centers, radii, sum_r
