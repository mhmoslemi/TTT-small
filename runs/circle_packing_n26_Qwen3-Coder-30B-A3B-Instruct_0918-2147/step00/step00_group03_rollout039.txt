import numpy as np
from scipy.optimize import minimize
import scipy.sparse as sp

def init_hex_grid(n=26, r_init=0.04):
    # Generate hexagonal grid points
    sqrt3 = np.sqrt(3)
    h = sqrt3 * r_init
    w = 2 * r_init
    
    points = []
    y = r_init
    row = 0
    
    while len(points) < n:
        if row % 2 == 0:
            x = r_init
        else:
            x = 1.5 * r_init
        
        while x <= 1 + r_init and len(points) < n:
            points.append([x, y])
            x += w
            
        y += h
        row += 1
        
        if y > 1 + r_init:
            break
    
    # Use first n points
    centers = np.array(points[:n])
    radii = np.full(n, r_init)
    
    return centers, radii

def max_radii(centers, radii):
    # Fix point: increase radii to maximum allowable values
    max_iter = 100
    eps = 1e-12
    
    for _ in range(max_iter):
        # Compute wall constraints: each circle's max radius based on boundaries
        wall = np.minimum(centers, 1 - centers).min(axis=1)
        
        # Compute nearest neighbor constraints (excluding self)
        D = np.sqrt(((centers[:, None, :] - centers[None, :, :]) ** 2).sum(-1))
        np.fill_diagonal(D, np.inf)  # Ignore self-distance
        neighbor = D - radii[:, None]
        neighbor = np.minimum(neighbor, np.inf)  # Ensure no negative values
        
        # Minimum radius allowed by constraints
        new_radii = np.minimum(wall, neighbor.min(axis=1)).clip(min=0)
        
        # Check for convergence
        if np.max(np.abs(new_radii - radii)) < 1e-8:
            break
            
        radii = new_radii
    return radii

def hill_climb(centers, radii, iterations=5000, step=0.01):
    # Stochastic hill-climbing to optimize total radius sum
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = float(best_radii.sum())
    
    for _ in range(iterations):
        # Randomly shuffle circles
        indices = np.random.permutation(len(centers))
        
        for i in indices:
            # Store current state
            old_center = centers[i].copy()
            old_radius = radii[i]
            
            # Try small random displacement
            delta = (np.random.rand(2) - 0.5) * 2 * step
            new_center = old_center + delta
            
            # Ensure it stays inside bounds given radius
            if (new_center[0] - radii[i] < -1e-12 or 
                new_center[0] + radii[i] > 1 + 1e-12 or
                new_center[1] - radii[i] < -1e-12 or
                new_center[1] + radii[i] > 1 + 1e-12):
                continue  # discard invalid move
            
            # Update center temporarily
            centers[i] = new_center
            
            # Re-optimize radii
            new_radii = max_radii(centers, radii.copy())
            
            # Check validity and update if beneficial
            if validate_packing(centers, new_radii)[0]:
                new_sum = float(new_radii.sum())
                if new_sum > best_sum + 1e-12:
                    best_centers = centers.copy()
                    best_radii = new_radii
                    best_sum = new_sum
            else:
                # Revert change if invalid
                centers[i] = old_center
                new_radii = radii.copy()
                
        # Optional: if still improving, stop early
        if _ > 0 and _ % 1000 == 0:
            if abs(best_sum - float(max_radii(best_centers, best_radii).sum())) < 1e-6:
                break
                
    return best_centers, best_radii

def refine_with_scipy(centers, radii):
    # Refine with scipy's trust-constr optimization
    # Flatten inputs for scipy
    n = len(centers)
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Define constraints
    def wall_constraints(v):
        # Wall constraints: r_i - x_i <= 0
        # Floor: r_i - x_i <= 0  
        # Ceiling: r_i + x_i - 1 <= 0
        # Left: r_i - y_i <= 0
        # Right: r_i + y_i - 1 <= 0
        centers_flat = v[:2*n].reshape(n, 2)
        radii_flat = v[2*n:]
        
        res = np.zeros(4*n)
        # Left wall
        res[0*n:1*n] = radii_flat - centers_flat[:, 0]
        # Right wall  
        res[1*n:2*n] = radii_flat + centers_flat[:, 0] - 1
        # Bottom wall
        res[2*n:3*n] = radii_flat - centers_flat[:, 1]
        # Top wall
        res[3*n:4*n] = radii_flat + centers_flat[:, 1] - 1
        return res
        
    def overlap_constraints(v):
        # Overlap constraints: r_i + r_j - d_ij <= 0 for all neighbors
        centers_flat = v[:2*n].reshape(n, 2)
        radii_flat = v[2*n:]
        
        # Compute inter-point distances
        D = np.sqrt(((centers_flat[:, None, :] - centers_flat[None, :, :]) ** 2).sum(-1))
        # Compute overlap inequalities: r_i + r_j - d_ij <= 0
        res = []
        for i in range(n):
            for j in range(i + 1, n):
                d = D[i, j]
                inequality = radii_flat[i] + radii_flat[j] - d
                res.append(inequality)
        return np.array(res)
    
    # Vectorized version without lambda
    def wall_constraints_vec(v):
        return wall_constraints(v)
        
    def overlap_constraints_vec(v):
        return overlap_constraints(v)

    overlap_con = NonlinearConstraint(overlap_constraints_vec, -np.inf, 0, jac='2-point')
    
    # Objective to minimize (negative sum)
    def obj(v):
        return -v[2*n:].sum()
    
    # Bound constraints
    bounds = []
    for i in range(2*n):
        bounds.append((0, 1))  # x_i, y_i in [0,1]
    for i in range(n):
        bounds.append((0, 0.5))  # r_i in [0, 0.5] (rough upper bound)
    
    # Optimize
    res = minimize(obj, x0, method='trust-constr', 
                   constraints=(wall_con, overlap_con),
                   bounds=bounds,
                   options={'maxiter': 200})
    
    # Extract results
    result_centers = res.x[:2*n].reshape(n, 2)
    result_radii = res.x[2*n:]
    
    # Ensure still valid
    if not validate_packing(result_centers, result_radii)[0]:
        # Fallback to previous best if this fails
        # This should usually not happen in well-converged cases
        return centers, radii
    
    return result_centers, result_radii

def run_packing():
    # Initial deterministic layout
    centers, radii = init_hex_grid()
    
    # Maximize radii iteratively given current positions
    radii = max_radii(centers, radii)
    
    # Try stochastic placement optimization
    centers, radii = hill_climb(centers, radii)
    
    # Final refinement with scipy trust-constr
    centers, radii = refine_with_scipy(centers, radii)
    
    # Final validation
    assert validate_packing(centers, radii)[0], "Final packing not valid"
    
    sum_r = float(radii.sum())
    
    return centers, radii, sum_r

# This is a standalone validation function as provided
def validate_packing(centers, radii):
    n = centers.shape[0]
    
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"
    
    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"
    
    return True, "ok"
