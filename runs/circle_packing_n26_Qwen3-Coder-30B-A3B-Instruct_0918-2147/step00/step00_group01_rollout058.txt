import numpy as np
from scipy.optimize import minimize

def initial_layout():
    # Create a hexagonal grid with 5 rows and 5 columns (25 circles), plus one in center
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Hexagonal grid parameters
    spacing = 0.18
    radius0 = spacing / 2 - 0.01  # Initial radius with margin
    
    # Generate hexagonal grid
    rows = 5
    cols = 5
    y_offset = 0.0
    for i in range(rows):
        x_offset = 0.0
        if i % 2 == 1:
            x_offset = spacing / 2
        for j in range(cols):
            x = x_offset + j * spacing
            y = y_offset + i * spacing * np.sqrt(3) / 2
            # Push inside bounds
            x = max(radius0, min(1 - radius0, x))
            y = max(radius0, min(1 - radius0, y))
            centers[i * cols + j] = [x, y]
            radii[i * cols + j] = radius0
    
    # Add one in center
    centers[-1] = [0.5, 0.5]
    radii[-1] = radius0
    
    return centers, radii

def pack_vector(centers, radii):
    z = np.zeros(3 * len(centers))
    for i in range(len(centers)):
        z[3*i] = centers[i, 0]
        z[3*i+1] = centers[i, 1]
        z[3*i+2] = radii[i]
    return z

def unpack_vector(z):
    n = len(z) // 3
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = z[3*i]
        centers[i, 1] = z[3*i+1]
        radii[i] = z[3*i+2]
    return centers, radii

def objective(z):
    centers, radii = unpack_vector(z)
    return -np.sum(radii)

def constraint_boundary(z):
    centers, radii = unpack_vector(z)
    n = len(centers)
    constraints = []
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # x - r >= 0
        constraints.append(x - r)
        # 1 - x - r >= 0
        constraints.append(1 - x - r)
        # y - r >= 0
        constraints.append(y - r)
        # 1 - y - r >= 0
        constraints.append(1 - y - r)
    return np.array(constraints)

def constraint_nonoverlap(z):
    centers, radii = unpack_vector(z)
    n = len(centers)
    constraints = []
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx*dx + dy*dy
            r_sum = radii[i] + radii[j]
            # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
            constraints.append(dist_sq - r_sum * r_sum)
    return np.array(constraints)

            radii = np.clip(radii, 0, 0.5)
