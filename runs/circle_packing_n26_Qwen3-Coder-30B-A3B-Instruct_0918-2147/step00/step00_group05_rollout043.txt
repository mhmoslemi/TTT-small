def build_hex_lattice(spacing):
    """Generate hexagonal lattice points inside unit square."""
    points = []
    y = spacing / 2
    row = 0
    while y <= 1 - spacing / 2:
        x = spacing / 2 if row % 2 == 0 else spacing
        while x <= 1 - spacing / 2:
            points.append([x, y])
            x += spacing
        y += spacing * sqrt(3) / 2
        row += 1
    return np.array(points)

def augment_with_extra_points(lattice):
    """Add two extra points in the center and one edge midpoint."""
    points = lattice.tolist()

    points.append([0.5, 0])       # bottom mid
    return np.array(points)

def local_optimize(initial_centers, initial_radii):
    """
    Run a local optimization to improve initial solution.
    Uses scipy trust-constr optimizer to maximize sum of radii.
    """
    centers = initial_centers.copy()
    radii = initial_radii.copy()

    # Flatten for optimization
    x0 = np.concatenate((centers.flatten(),radii))

    def obj(x):
        n = len(centers) # number of circles
        cx = x[:2*n].reshape(n,2)
        r = x[2*n:]
        return -sum(r)  # negative because we minimize

    def jac(x):
        n = len(centers)
        grad = np.zeros_like(x)
        grad[2*n:] = -1.0 # d/dr sum = -1
        return grad

    def cons(x):
        # List of all constraint violations (if you want to be shaped for trust-constr)
        # For this implement, just return an empty vector since we want to pass the constraints along
        return np.array([])

    def hess(x):
        n = len(centers)
        h = np.zeros((len(x),len(x)))
        # Only second derivatives of radii part (zeros)
        return h

    # Bounds: center x in [r, 1-r], center y in [r, 1-r], radius >= 0
    bounds = []
    for i in range(len(centers)):
        bounds.append((initial_radii[i], 1 - initial_radii[i]))  # x
        bounds.append((initial_radii[i], 1 - initial_radii[i]))  # y
    for i in range(len(initial_radii)):
        bounds.append((1e-12, None))  # radii >= 0 (but small lower bound for stability)

    # For simplicity, beam and limit constrain functions
    opts = {'maxiter': 500, 'disp': False}
    result = minimize(obj, x0, method='trust-constr', jac=jac, hess=hess,
                      bounds=bounds, constraints=cons, options=opts)

    # Return updated centers and radii
    final_centers = result.x[:2*len(centers)].reshape(len(centers), 2)
    final_radii = result.x[2*len(centers):]
    return final_centers, final_radii

def perturb_centers(centers, sigma, radii):
    """
    Perturb the centers slightly while keeping them inside feasible bounds.
    """
    perturbed = []
    for i in range(len(centers)):
        new_c = centers[i].copy()
        new_c[0] += np.random.normal(0, sigma)
        new_c[1] += np.random.normal(0, sigma)
        # Project to [r, 1-r] valid ranges
        new_c[0] = np.clip(new_c[0], radii[i], 1 - radii[i])
        new_c[1] = np.clip(new_c[1], radii[i], 1 - radii[i])
        perturbed.append(new_c)
    return np.array(perturbed)

def run_packing():
    # 1. Build hexagonal lattice
    spacing = 0.18
    lattice = build_hex_lattice(spacing)
    # 2. Add extra circles at center and edge midpoint
    centers0 = augment_with_extra_points(lattice)
    # 3. Generate some plausible contact graphs
    graphs = generate_contact_graphs(centers0)
    # 4. Solve and refine
    best = (None, None, -1.0)
    for i, g in enumerate(graphs):
        try:
            radii = solve_contact_system(g)
            if radii is None:
                continue
            if len(radii) != len(centers0):
                continue
            centers_refine, radii_refine = local_optimize(centers0.copy(), np.array(radii))
            s = sum(radii_refine)
            if s > best[2]:
                best = (centers_refine, radii_refine, s)
        except Exception:
            continue

    # 5. Random restarts if needed
    if best[2] < 2.636:
        for _ in range(12):
            try:
                perturbed = perturb_centers(best[0], 0.01, best[1])
                centers_refine, radii_refine = local_optimize(perturbed, best[1])
                s = sum(radii_refine)
                if s > best[2]:
                    best = (centers_refine, radii_refine, s)
            except Exception:
                continue

    # Final validity check and return
    assert validate_packing(best[0], best[1])[0], "Invalid packing after optimization"
    return best[0], best[1], best[2]

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
