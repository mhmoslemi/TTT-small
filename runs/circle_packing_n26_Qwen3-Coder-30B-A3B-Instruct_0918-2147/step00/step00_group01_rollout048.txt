import numpy as np
from scipy.optimize import minimize

def initial_layout():
    # Create a hexagonal layout with 25 circles in a 5x5 grid
    spacing = 0.18
    rows = 5
    cols = 5
    centers = []
    
    for i in range(rows):
        for j in range(cols):
            x = j * spacing + 0.01
            y = i * spacing + 0.01
            
            # Adjust if out of bounds
            if x > 1 - spacing / 2:
                x = 1 - spacing / 2
            if y > 1 - spacing / 2:
                y = 1 - spacing / 2
                
            centers.append([x, y])
            
    # Add one more circle in the center
    centers.append([0.5, 0.5])
    
    centers = np.array(centers)
    
    # Assign initial radii
    radii = np.full(26, spacing / 2 - 0.01)
    
    return centers, radii

def pack_vector(centers, radii):
    z = np.zeros(3 * 26)
    for i in range(26):
        z[3 * i] = centers[i][0]
        z[3 * i + 1] = centers[i][1]
        z[3 * i + 2] = radii[i]
    return z

def unpack_vector(z):
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    for i in range(26):
        centers[i][0] = z[3 * i]
        centers[i][1] = z[3 * i + 1]
        radii[i] = z[3 * i + 2]
    return centers, radii

def objective(z):
    centers, radii = unpack_vector(z)
    return -np.sum(radii)

def constraint_boundary(z):
    centers, radii = unpack_vector(z)

        cons.append(1 - x - r)  # 1 - x - r >= 0
        cons.append(y - r)  # y - r >= 0
        cons.append(1 - y - r)  # 1 - y - r >= 0
        
    return np.array(cons)

def constraint_nonoverlap(z):

            dist_sq = dx * dx + dy * dy
            r_sum = radii[i] + radii[j]
            cons.append(dist_sq - r_sum * r_sum)
            
    return np.array(cons)
