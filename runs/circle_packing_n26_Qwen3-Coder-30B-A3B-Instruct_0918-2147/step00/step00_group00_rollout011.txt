import numpy as np
import random

def make_hex_seed(n, margin):
    # Generate hexagonal pattern with margin
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))
    
    centers = []
    radii = []
    
    # Grid offset for hexagonal packing
    row_offset = margin
    col_offset = margin
    row_spacing = 1 - 2 * margin
    col_spacing = 1 - 2 * margin
    
    # Build hexagonal pattern
    for i in range(rows):
        row_y = margin + i * row_spacing / (rows - 1) if rows > 1 else 0.5
        for j in range(cols):
            col_x = margin + j * col_spacing / (cols - 1) if cols > 1 else 0.5
            # Offset odd rows
            if i % 2 == 1:
                col_x += 0.5 * col_spacing / (cols - 1) if cols > 1 else 0.25
            
            if len(centers) < n:
                centers.append([col_x, row_y])
    
    # Shrink pattern to ensure no overlap initially
    centers = np.array(centers[:n])
    
    # Compute initial radii based on distances to walls
    min_radius = 0.015
    for i in range(n):
        x, y = centers[i]
        r = min(x, 1 - x, y, 1 - y) * 0.5
        if r < min_radius:
            r = min_radius
        radii.append(r)
    
    # Rescale all radii so that the smallest is min_radius
    min_r = min(radii)
    scale = min_radius / min_r
    radii = [r * scale for r in radii]
    
    return centers, np.array(radii)

def max_feasible_radius(i, centers, radii):
    x, y = centers[i]
    r = radii[i]
    
    # Distance to walls
    d_L = x
    d_R = 1 - x
    d_B = y
    d_T = 1 - y
    
    d_near = min(d_L, d_R, d_B, d_T)
    
    # Distance to other circles
    for j in range(centers.shape[0]):
        if i != j:
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            d = dist - radii[j]
            if d < d_near:
                d_near = d
    
    return d_near

def inflate_radii(centers, radii, alpha):
    old_sum = np.sum(radii)
    n = centers.shape[0]
    
    # Add global inflation
    delta = alpha * np.mean(radii)
    new_radii = radii + delta
    
    # Clamp each radius to its maximal feasible value
    for i in range(n):
        max_r = max_feasible_radius(i, centers, new_radii)
        new_radii[i] = min(new_radii[i], max_r)
    
    new_sum = np.sum(new_radii)
    return new_radii, (new_sum - old_sum)

def anneal_centers(centers, radii, temp0, gamma, sigma0, steps):
    n = centers.shape[0]
    current_sum = np.sum(radii)
    best_sum = current_sum
    best_centers = centers.copy()
    best_radii = radii.copy()
    
    temp = temp0
    sigma = sigma0
    
    for step in range(steps):
        # Pick random circle
        i = random.randint(0, n - 1)
        
        # Store old values
        old_x, old_y = centers[i, 0], centers[i, 1]
        old_r = radii[i]
        
        # Propose move
        dx, dy = np.random.normal(0, sigma, size=2)
        centers[i, 0] += dx
        centers[i, 1] += dy
        
        # Ensure within bounds
        centers[i, 0] = max(0, min(1, centers[i, 0]))
        centers[i, 1] = max(0, min(1, centers[i, 1]))
        
        # Compute new feasible radius for moved circle
        new_r = max_feasible_radius(i, centers, radii)
        
        # Accept or reject
        delta_r = new_r - old_r
        if delta_r >= 0 or random.random() < np.exp(delta_r / temp):
            radii[i] = new_r
        else:
            # Revert move
            centers[i, 0] = old_x
            centers[i, 1] = old_y

        # Update sum
        current_sum = np.sum(radii)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
        
        # Occasionally global shift
        if step % 50 == 0:
            shift = np.random.normal(0, sigma / 10, size=2)
            centers += shift
            centers = np.clip(centers, 0, 1)
        
        # Update temperature and sigma
        temp *= gamma
        sigma *= gamma
    
    return best_centers, best_radii, best_sum

def mixed_cycle(centers, radii, params):
    alpha = params['alpha']
    K = params['K']
    M = params['M']
    P = params['P']
    gamma_SA = params['gamma_SA']
    T0 = params['T0']
    sigma0 = params['sigma0']
    
    last_sum = np.sum(radii)
    stagnations = 0
    full_cycles = 0
    
    # Phase B: Adaptive Inflation Loop
    while stagnations < K:
        new_radii, delta = inflate_radii(centers, radii, alpha)
        radii = new_radii
        
        if delta < 1e-6:
            stagnations += 1
        else:
            stagnations = 0
        last_sum = np.sum(radii)
        alpha *= 0.95
    
    # Phase C: Stochastic Centre Re-optimisation
    while full_cycles < P:
        centers, radii, new_sum = anneal_centers(
            centers, radii, T0, gamma_SA, sigma0, M
        )
        
        # Check convergence
        # After the SA step, perform a small number of inflation rounds
        for _ in range(3):
            radii, _ = inflate_radii(centers, radii, 0.2 * alpha)
        
        last_sum = new_sum
        full_cycles += 1
    
    return centers, radii

def polish_radii(centers, radii, beta, passes):
    n = centers.shape[0]
    
    for _ in range(passes):
        still_change = True
        iter_count = 0
        while still_change and iter_count < 10:
            still_change = False
            for i in range(n):
                x, y = centers[i]
                r = radii[i]
                
                # Wall gaps
                g_wall = min(x, 1 - x, y, 1 - y)
                
                # Neighbourhood gaps
                g_nei = float('inf')
                for j in range(n):
                    if i != j:
                        dx = centers[i, 0] - centers[j, 0]
                        dy = centers[i, 1] - centers[j, 1]
                        dist = np.sqrt(dx * dx + dy * dy)
                        gap = dist - (r + radii[j])
                        if gap < g_nei:
                            g_nei = gap
                
                g = min(g_wall, g_nei)
                
                if g > 0:
                    new_r = r + beta * g
                    max_r = max_feasible_radius(i, centers, radii)
                    if new_r > max_r:
                        new_r = max_r
                    if new_r > r:
                        radii[i] = new_r
                        still_change = True
            
            iter_count += 1
        
        # Resolve overlaps if any
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                needed = radii[i] + radii[j]
                if dist < needed:
                    overlap = needed - dist
                    new_r_i = radii[i] - overlap/2
                    new_r_j = radii[j] - overlap/2
                    
                    # Ensure non-negative
                    if new_r_i < 0:
                        new_r_i = 0
                    if new_r_j < 0:
                        new_r_j = 0
                        
                    radii[i] = new_r_i
                    radii[j] = new_r_j
        
        # Re-inflate after overlap resolution
        for i in range(n):
            max_r = max_feasible_radius(i, centers, radii)
            radii[i] = min(radii[i], max_r)
    
    return centers, radii

# Main entry point
def run_packing():
    # Set random seed for reproducibility
    np.random.seed(42)
    random.seed(42)
    
    # Parameters for algorithm
    params = {
        'alpha': 0.04,
        'K': 8,
        'M': 200,
        'P': 5,
        'gamma_SA': 0.97,
        'T0': 0.02,
        'sigma0': 0.03,
        'beta': 0.6,
        'passes': 3
    }
    
    # Phase A: Structured Initialisation
    centers, radii = make_hex_seed(26, 0.04)
    
    # Phase B, C, D: Mixed Inflation and Re-optimisation
    centers, radii = mixed_cycle(centers, radii, params)
    
    # Phase E: Final Polishing
    centers, radii = polish_radii(centers, radii, params['beta'], params['passes'])
    
    # Summary stats
    total_radius = np.sum(radii)
    
    return centers, radii, total_radius
