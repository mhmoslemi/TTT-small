import numpy as np

def hex_lattice(d):
    """Generate hexagonal lattice points in [0,1]x[0,1]"""
    sqrt3 = np.sqrt(3)
    dy = sqrt3 * d / 2
    points = []
    y = 0
    row_offset = 0
    while y <= 1:
        x = row_offset
        while x <= 1:
            points.append([x, y])
            x += d
        y += dy
        row_offset = (row_offset + d / 2) % d
    return np.array(points)

def add_border_circles(points, d):
    """Append 6 border-enhanced circles to the lattice points."""
    r = d / 4
    border_points = [
        [r, r], [1 - r, r], [r, 1 - r], [1 - r, 1 - r],
        [0.5, r], [0.5, 1 - r]
    ]
    return np.vstack([points, border_points])

def initial_radii(points):
    """Compute initial uniform safe radius for all points."""
    n = points.shape[0]
    min_dist = np.inf
    for i in range(n):
        x, y = points[i]
        wall_dist = min(x, 1 - x, y, 1 - y)
        for j in range(n):
            if i != j:
                dist = np.linalg.norm(points[i] - points[j])
                min_dist = min(min_dist, dist)
        min_dist = min(min_dist, wall_dist)
    return min_dist / 2

def inflate_radii(centers, radii):
    """Inflate radii to maximum feasible values."""
    n = centers.shape[0]
    new_radii = radii.copy()
    # Compute distance matrix
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.linalg.norm(diff, axis=2)
    for i in range(n):
        # Wall constraints
        x, y = centers[i]
        walls = [x, 1 - x, y, 1 - y]
        # Pairwise constraints
        min_pairwise = np.inf
        for j in range(n):
            if i != j:
                min_pairwise = min(min_pairwise, dists[i, j] - radii[j])
        new_r = min(walls + [min_pairwise])
        if new_r > radii[i] + 1e-6:
            new_radii[i] = new_r
    return new_radii

def relax_centers(centers, radii, steps=10):
    """Relax centers to optimize spacing (force-based)."""
    # Soft repulsion and wall forces
    n = centers.shape[0]
    sq = np.sqrt(2)
    beta = 0.02
    kappa = 0.02
    lambda_w = 0.01
    gamma = 0.02
    alpha = 0.5
    center_copy = centers.copy()
    for _ in range(steps):
        forces = np.zeros_like(centers)
        for i in range(n):
            x_i, y_i = centers[i]
            # Wall force
            dx = 0
            dy = 0
            if x_i - radii[i] < gamma:
                dx += lambda_w * max(0, gamma - (x_i - radii[i]))
            if 1 - x_i - radii[i] < gamma:
                dx -= lambda_w * max(0, gamma - (1 - x_i - radii[i]))
            if y_i - radii[i] < gamma:
                dy += lambda_w * max(0, gamma - (y_i - radii[i]))
            if 1 - y_i - radii[i] < gamma:
                dy -= lambda_w * max(0, gamma - (1 - y_i - radii[i]))
            forces[i] += [dx, dy]
            # Pairwise repulsion
            for j in range(n):
                if i == j:
                    continue
                diff_vec = centers[i] - centers[j]
                dist = np.linalg.norm(diff_vec)
                delta = radii[i] + radii[j] - dist
                if delta > 0:
                    # Slight repulsion
                    force_mag = kappa * max(0, beta - dist) / max(dist, 1e-10)
                    forces[i] += force_mag * diff_vec
        # Apply forces
        new_centers = centers + alpha * forces
        # Clamp centers
        for i in range(n):
            new_centers[i, 0] = np.clip(new_centers[i, 0], radii[i], 1 - radii[i])
            new_centers[i, 1] = np.clip(new_centers[i, 1], radii[i], 1 - radii[i])
        centers = new_centers
    return centers

def run_packing():
