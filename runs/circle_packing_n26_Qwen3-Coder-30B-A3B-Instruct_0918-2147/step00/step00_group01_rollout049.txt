import numpy as np
from scipy.optimize import differential_evolution
from scipy.spatial.distance import cdist

def compute_radii(centers):
    n = centers.shape[0]
    radii = np.zeros(n)
    
    # Wall distances
    x = centers[:, 0]
    y = centers[:, 1]
    wall_distances = np.minimum.reduce([x, 1 - x, y, 1 - y])
    
    # Pairwise distances
    distances = cdist(centers, centers)
    np.fill_diagonal(distances, np.inf)
    
    # Neighbour distances (half of minimum distance to other centers)
    neighbour_distances = 0.5 * np.min(distances, axis=1)
    
    # Radius is minimum of wall and neighbour distances
    radii = np.minimum(wall_distances, neighbour_distances)
    
    return radii

def neg_sum_radii(z):
    centers = z.reshape(-1, 2)
    radii = compute_radii(centers)
    return -np.sum(radii)

def run_packing():

    # Differential Evolution
    bounds = [(0, 1)] * 52
    result = differential_evolution(
        neg_sum_radii,
        bounds,
        maxiter=2000,
        popsize=10,
        tol=1e-7,
        polish=False,
        strategy='best1bin',
        seed=42
    )
    
    best_centers_flat = result.x
    centers = best_centers_flat.reshape(-1, 2)
    
    # Hill-climbing refinement
    sigma = 0.02
    max_iter = 3000
    best_centers = centers.copy()
    best_sum = -neg_sum_radii(best_centers_flat)
    
    for _ in range(max_iter):
        # Generate perturbation
        noise = np.random.normal(0, sigma, best_centers.shape)
        new_centers = best_centers + noise
        new_centers = np.clip(new_centers, 0, 1)
        
        # Evaluate

            best_sum = new_sum
