import numpy as np
from scipy.optimize import differential_evolution
from scipy.spatial.distance import cdist

def compute_radii(centers):
    """Compute maximal feasible radii for given center positions."""
    n = centers.shape[0]
    # Wall distances
    wall_dists = np.minimum.reduce([
        centers[:, 0],           # left
        1 - centers[:, 0],       # right
        centers[:, 1],           # bottom
        1 - centers[:, 1]        # top
    ])
    
    # Pairwise distances
    distances = cdist(centers, centers)
    np.fill_diagonal(distances, np.inf)
    
    # Neighbour distances
    min_neighbor_dists = 0.5 * np.min(distances, axis=1)
    
    # Radii are min of wall and neighbor distances
    radii = np.minimum(wall_dists, min_neighbor_dists)
    return radii

def neg_sum_radii(z):
    """Objective: negative sum of radii for given center coordinates."""
    centers = z.reshape(-1, 2)
    radii = compute_radii(centers)
    return -np.sum(radii)

        tol=1e-7,
        polish=False,

    )
    
    # Extract best centers from DE

    # Hill-climbing refinement
