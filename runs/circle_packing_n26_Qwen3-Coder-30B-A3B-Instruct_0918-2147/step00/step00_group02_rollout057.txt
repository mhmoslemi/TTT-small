import numpy as np
from scipy.optimize import minimize
from ortools.linear_solver import pywraplp

def generate_candidates(g, radius_levels):
    candidates = []
    grid_points = np.arange(0, 1 + g/2, g)
    for x in grid_points:
        for y in grid_points:
            for r in radius_levels:
                if x - r >= 0 and x + r <= 1 and y - r >= 0 and y + r <= 1:
                    candidates.append((x, y, r))
    return candidates

def build_milp(candidates):
    solver = pywraplp.Solver.CreateSolver('SCIP')
    if not solver:
        raise Exception("Could not create solver")
    
    n = len(candidates)
    
    # Binary variables
    z = [solver.BoolVar(f'z_{i}') for i in range(n)]
    
    # Grid point handling
    grid_map = {}
    for i, (x, y, r) in enumerate(candidates):
        grid_key = (round(x / 0.04), round(y / 0.04))
        if grid_key not in grid_map:
            grid_map[grid_key] = []
        grid_map[grid_key].append(i)
    
    # At most one circle per grid point
    for grid_key, indices in grid_map.items():
        solver.Add(sum(z[i] for i in indices) <= 1)
    
    # Exactly 26 circles
    solver.Add(sum(z[i] for i in range(n)) == 26)
    
    # Non-overlap constraints with L-infinity approximation
    big_m = 1.0
    dx_vars = {}
    dy_vars = {}
    
    for i in range(n):
        for j in range(i+1, n):
            xi, yi, ri = candidates[i]
            xj, yj, rj = candidates[j]
            
            dx = solver.NumVar(0.0, 1.0, f'dx_{i}_{j}')
            dy = solver.NumVar(0.0, 1.0, f'dy_{i}_{j}')
            
            # Link dx to |xi - xj|
            solver.Add(dx >= xi - xj)
            solver.Add(dx >= xj - xi)
            
            # Link dy to |yi - yj|
            solver.Add(dy >= yi - yj)
            solver.Add(dy >= yj - yi)
            
            # Add constraint for non-overlap: dx + dy >= ri + rj
            solver.Add(dx + dy >= ri + rj - big_m * (1 - z[i] - z[j]))
            solver.Add(dx - dy >= ri + rj - big_m * (1 - z[i] - z[j]))
            solver.Add(dy - dx >= ri + rj - big_m * (1 - z[i] - z[j]))
            
            dx_vars[(i,j)] = dx
            dy_vars[(i,j)] = dy
    
    # Objective: maximize sum of radii
    solver.Maximize(sum(candidates[i][2] * z[i] for i in range(n)))
    
    return solver, z, candidates

def solve_milp(solver, z, candidates):
    status = solver.Solve()
    selected_indices = []
    if status == pywraplp.Solver.OPTIMAL:
        for i in range(len(candidates)):
            if z[i].solution_value() > 0.5:
                selected_indices.append(i)
    return selected_indices

def extract_initial_solution(selected_indices, candidates):
    centers0 = np.array([[candidates[i][0], candidates[i][1]] for i in selected_indices])
    radii0 = np.array([candidates[i][2] for i in selected_indices])
    return centers0, radii0

def shrink_initial(radii0, delta):
    return radii0 - delta

def pack_variables(centers0, radii0):
    z0 = np.zeros(3 * len(centers0))
    for i in range(len(centers0)):
        z0[3*i] = centers0[i][0]
        z0[3*i+1] = centers0[i][1]
        z0[3*i+2] = radii0[i]
    return z0

def compute_radius_sum(radii):
    return np.sum(radii)

def define_objective(z):
    # unpack
    n = len(z) // 3
    r = z[2::3]
    return -np.sum(r)

def within_bounds(z):
    n = len(z) // 3
    x = z[0::3]
    y = z[1::3]
    r = z[2::3]
    # Check boundary constraints
    for i in range(n):
        if r[i] > x[i] or r[i] > 1-x[i] or r[i] > y[i] or r[i] > 1-y[i]:
            return False
    return True

def define_jacobian(z):
    n = len(z) // 3
    grad = np.zeros(len(z))
    grad[2::3] = -1.0  # gradient of -sum(r)
    return grad

def define_constraints(z, centers0, radii0, candidates):
    # Non-overlap and boundary constraints
    constraints = []
    n = len(z) // 3
    
    # Boundary constraints
    for i in range(n):
        x, y, r = z[3*i], z[3*i+1], z[3*i+2]
        # r <= x  =>  x - r >= 0
        constraints.append({
            'fun': lambda z, i=i: z[3*i] - z[3*i+2],
            'jac': lambda z, i=i: np.array([1.0 if j == 3*i else 0.0 for j in range(len(z))])
        })
        # r <= 1 - x
        constraints.append({
            'fun': lambda z, i=i: 1 - z[3*i] - z[3*i+2],
            'jac': lambda z, i=i: np.array([-1.0 if j == 3*i else 0.0 for j in range(len(z))])
        })
        # r <= y
        constraints.append({
            'fun': lambda z, i=i: z[3*i+1] - z[3*i+2],
            'jac': lambda z, i=i: np.array([0.0 if j != 3*i+1 and j != 3*i+2 else (1.0 if j == 3*i+1 else -1.0) for j in range(len(z))])
        })
        # r <= 1 - y
        constraints.append({
            'fun': lambda z, i=i: 1 - z[3*i+1] - z[3*i+2],
            'jac': lambda z, i=i: np.array([0.0 if j != 3*i+1 and j != 3*i+2 else (-1.0 if j == 3*i+1 else -1.0) for j in range(len(z))])
        })

    # Non-overlap constraints (at least for active pairs between initial and refined set)
    for i in range(n):
        for j in range(i+1, n):
            # {x_i, y_i, r_i} and {x_j, y_j, r_j}
            def fun_wrap(i, j, z_i, z_j, z):
                dx = z_z[3*i] - z_z[3*j]
                dy = z_z[3*i+1] - z_z[3*j+1]
                return (z_z[3*i+2] + z_z[3*j+2]) ** 2 - (dx**2 + dy**2)
            constraints.append({
                'fun': lambda z, i=i, j=j: (z[3*i+2] + z[3*j+2])**2 - ((z[3*i] - z[3*j])**2 + (z[3*i+1] - z[3*j+1])**2),
                'jac': lambda z, i=i, j=j: np.array([
                    0 if j != 3*i and j != 3*i+1 and j != 3*i+2 and j != 3*j and j != 3*j+1 and j != 3*j+2 else
                    0.0 if (j == 3*i+2 or j == 3*j+2) and j == 3*i+2 else (
                        2*(z[3*i+2] + z[3*j+2]) if j == 3*i+2 else -2*(z[3*j+2] + z[3*i+2]) if j == 3*j+2 else
                        0.0 if j == 3*i     else 0.0 if j == 3*j else
                        0.0 if j == 3*i+1  else 0.0 if j == 3*j+1 else 0.0 if j == 3*i+2 else 0.0 if j == 3*i+2 else
                        0.0
                    )
                    for j in range(len(z))
                ])
            })
    
    return constraints

def run_packing():
    # Generate candidates
    radius_levels = np.arange(0.02, 0.251, 0.02)
    g = 0.04
    candidates = generate_candidates(g, radius_levels)
    
    # Solve discrete problem
    solver, z_vars, orig_candidates = build_milp(candidates)
    selected_indices = solve_milp(solver, z_vars, candidates)
    
    # Extract initial configuration
    centers0, radii0 = extract_initial_solution(selected_indices, candidates)
    
    # Slight shrink for stability
    radii0 = shrink_initial(radii0, 1e-5)
    
    # Initialize with continuous variables
    z0 = pack_variables(centers0, radii0)
    
    # Define optimized variables
    n_circles = len(centers0)
    bounds = [(0, 1) for _ in range(n_circles * 3)]  # bounds for x, y, r
    
    def objfunc(z):
        return define_objective(z)
    
    def jacfunc(z):
        return define_jacobian(z)

    def constr_func(z):
        # Wrapper for constraint evaluation. Return list of (value, Jacobian)
        # Note: Frivolous to compute each time in such a way.
        # However, we proceed assuming reduction by negation (so minimize) and constraints are <= 0
        # So return a list of dicts, each with 'fun': function and 'jac': Jacobian function
        constraints = define_constraints(z, centers0, radii0, candidates)
        result = []
        for c in constraints:
            result.append({'fun': c['fun'](z), 'jac': c['jac'](z)})
        return result
    
    # Prepare the problem for scipy.optimize.minimize
    bounds_list = bounds
    cons_list = constr_func(z0)  # Call to set up constraints
    
    # Run continuous optimization
    result = minimize(
        objfunc,
        z0,
        method='trust-constr',
        jac=jacfunc,
        constraints=cons_list,
        bounds=bounds_list,
        options={'disp': False}
    )
    
    if not result.success:
        raise Exception("Continuous optimization failed")
    
    # Unpack the solution
    centers = np.zeros((n_circles, 2))
    radii = np.zeros(n_circles)
    for i in range(n_circles):
        centers[i][0] = result.x[3*i]
        centers[i][1] = result.x[3*i+1]
        radii[i] = result.x[3*i+2]
    
    # Repair if needed
    for i in range(n_circles):
        radii[i] = min(radii[i], centers[i][0], 1 - centers[i][0], centers[i][1], 1 - centers[i][1])
    
    # Apply validator
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise Exception(f"Invalid packing after optimization: {msg}")
    
    sum_radii = compute_radius_sum(radii)
    return centers, radii, sum_radii
