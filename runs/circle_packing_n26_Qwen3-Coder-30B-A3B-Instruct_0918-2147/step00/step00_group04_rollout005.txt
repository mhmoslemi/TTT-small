import numpy as np
import cvxpy as cp
from scipy.spatial.distance import pdist, squareform

def build_socp_problem(n):
    r = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(r)), [])
    dist_params = []
    for i in range(n):
        for j in range(i + 1, n):
            param = cp.Parameter(nonneg=True)
            dist_params.append(param)
    return prob, r, dist_params

def solve_radii(centres, prob, r, dist_params):
    n = centres.shape[0]
    constraints = []
    
    # Boundary constraints
    for i in range(n):
        x, y = centres[i]
        constraints.append(r[i] <= x)
        constraints.append(r[i] <= 1 - x)
        constraints.append(r[i] <= y)
        constraints.append(r[i] <= 1 - y)
    
    # Non-overlap constraints
    distances = squareform(pdist(centres, 'euclidean'))
    param_idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dist = distances[i, j]
            param = dist_params[param_idx]
            param.value = dist
            constraints.append(r[i] + r[j] <= dist)
            param_idx += 1
    
    prob.constraints = constraints
    prob.solve(solver=cp.ECOS, abstol=1e-12, reltol=1e-12)
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        radii = r.value
        return radii, None  # Placeholder for duals if needed
    else:
        raise RuntimeError("SOCP solver failed")

def energy(centres, prob, r, dist_params):
    radii, _ = solve_radii(centres, prob, r, dist_params)
    return - np.sum(radii)

def random_move(state, rng, move_type_prob=0.1):
    new_state = state.copy()
    n = len(new_state) // 2
    if rng.random() < move_type_prob:
        # Edge push
        idx = rng.integers(n)
        x = new_state[2 * idx]
        y = new_state[2 * idx + 1]
        dx = -np.sign(x - 0.5) * 0.02
        dy = -np.sign(y - 0.5) * 0.02
        new_state[2 * idx] = np.clip(x + dx, 0, 1)
        new_state[2 * idx + 1] = np.clip(y + dy, 0, 1)
    elif rng.random() < 0.5:
        # Single center jitter
        idx = rng.integers(n)
        dx = rng.normal(0, 0.02)
        dy = rng.normal(0, 0.02)
        new_state[2 * idx] = np.clip(new_state[2 * idx] + dx, 0, 1)
        new_state[2 * idx + 1] = np.clip(new_state[2 * idx + 1] + dy, 0, 1)
    else:
        # Cluster jitter
        num_centers = rng.integers(3, 6)
        indices = rng.choice(n, size=num_centers, replace=False)
        center_of_mass = np.mean([new_state[2*i:2*i+2] for i in indices], axis=0)
        offset = rng.normal(0, 0.02, 2)
        for i in indices:
            new_state[2*i:2*i+2] = np.clip(new_state[2*i:2*i+2] + offset, 0, 1)
    
    return new_state

def anneal_chain(seed, prob, r, dist_params, max_iter=150000, T0=0.05, alpha=0.995, rng_seed=None):
    if rng_seed is not None:
        rng = np.random.default_rng(rng_seed)
    else:
        rng = np.random.default_rng(seed)
    
    # Initialize with some seed configuration
    if seed % 5 == 0:
        # Hexagonal pattern
        centres = np.empty((26, 2))
        row_idx = []
        col_idx = []
        count = 0
        for row in range(5):
            for col in range(6):
                if count >= 26:
                    break
                x = 0.1 + col * 0.15
                if row % 2 == 1:
                    x += 0.075
                y = 0.1 + row * 0.15
                centres[count] = [x, y]
                count += 1
                if count >= 26:
                    break
    elif seed % 5 == 1:
        # Grid pattern
        centres = np.empty((26, 2))
        count = 0
        for i in range(5):
            for j in range(5):
                if count >= 26:
                    break
                x = 0.1 + i * 0.2
                y = 0.1 + j * 0.2
                centres[count] = [x, y]
                count += 1
                if count >= 26:
                    break
    elif seed % 5 == 2:
        # Corner-heavy pattern
        centres = np.empty((26, 2))
        # Place four circles in corners with small radii
        corners = [[0.05, 0.05], [0.05, 0.95], [0.95, 0.05], [0.95, 0.95]]
        for i in range(4):
            centres[i] = corners[i]
        # Place remaining in grid pattern
        count = 4
        for i in range(5):
            for j in range(5):
                if count >= 26:
                    break
                x = 0.2 + i * 0.15
                y = 0.2 + j * 0.15
                centres[count] = [x, y]
                count += 1
                if count >= 26:
                    break
    elif seed % 5 == 3:
        # Random uniform
        centres = rng.random((26, 2))
    else:
        # Regular pattern + small noise
        centres = np.empty((26, 2))
        count = 0
        for i in range(5):
            for j in range(5):
                if count >= 26:
                    break
                x = 0.1 + i * 0.2 + (rng.random() - 0.5) * 0.05
                y = 0.1 + j * 0.2 + (rng.random() - 0.5) * 0.05
                centres[count] = [x, y]
                count += 1
                if count >= 26:
                    break

    # Translate to flat state vector
    state = centres.flatten()
    current_energy = energy(centres, prob, r, dist_params)
    best_energy = current_energy
    best_state = state.copy()
    T = T0
    
    accepted = 0
    while accepted < max_iter:
        new_state = random_move(state, rng)
        new_centres = new_state.reshape(-1, 2)
        new_energy = energy(new_centres, prob, r, dist_params)
        delta = new_energy - current_energy
        if delta < 0 or rng.random() < np.exp(-delta / T):
            state = new_state
            current_energy = new_energy
            if current_energy < best_energy:
                best_energy = current_energy
                best_state = state.copy()
            accepted += 1
        T = max(T * alpha, 1e-6)
    
    best_centres = best_state.reshape(-1, 2)
    best_radii, _ = solve_radii(best_centres, prob, r, dist_params)
    
    return best_centres, best_radii, -best_energy

def local_refine(centres, radii, prob, r, dist_params, max_iter=10):
    # Identify active constraints, subject to a threshold for practicality
    delta = 1e-5
    final_centres = centres.copy()
    final_radii = radii.copy()
    
    for _ in range(max_iter):
        # Solve for current config
        _, _ = solve_radii(final_centres, prob, r, dist_params)
        new_radii, _ = solve_radii(final_centres, prob, r, dist_params)
        
        if np.sum(np.abs(new_radii - final_radii)) < 1e-5:
            break
        final_radii = new_radii
        
    return final_centres, final_radii

def finalise(centres, radii, prob, r, dist_params):
    # High precision final solve
    final_radii, _ = solve_radii(centres, prob, r, dist_params)
    # Ensure boundary compliance
    for i in range(len(centres)):
        x, y = centres[i]
        r_i = final_radii[i]
        r_i = min(r_i, x, 1 - x, y, 1 - y)
        final_radii[i] = r_i
    
    return centres, final_radii

def run_packing():
    n = 26
    
    # Build problem once
    prob, r, dist_params = build_socp_problem(n)
    
    best_sum = -float('inf')
    best_centers = None
    best_radii = None
    
    # Run several annealing chains
    for seed in range(8):
        print(f"Starting chain {seed}")
        centres, radii, sum_rad = anneal_chain(seed, prob, r, dist_params, 
                                              max_iter=50000, rng_seed=seed)
        print(f"Chain {seed} finished with sum {sum_rad:.6f}")
        
        # Refine
        refined_centers, refined_radii = local_refine(centres, radii, prob, r, dist_params)
        print(f"After refinement: {np.sum(refined_radii):.6f}")
        
        # Final check
        validated_centers, validated_radii = finalise(refined_centers, refined_radii, prob, r, dist_params)
        new_sum = np.sum(validated_radii)
        
        if new_sum > best_sum:
            best_sum = new_sum
            best_centers = validated_centers.copy()
            best_radii = validated_radii.copy()
            
    return best_centers, best_radii, best_sum
