import numpy as np
from scipy.optimize import differential_evolution
from scipy.spatial.distance import cdist

def compute_radii(centers):
    n = centers.shape[0]
    radii = np.zeros(n)
    
    # Wall distances
    x = centers[:, 0]
    y = centers[:, 1]
    wall_distances = np.array([
        x,                   # left
        1 - x,               # right
        y,                   # bottom
        1 - y                # top
    ])
    
    # Pairwise distances
    distances = cdist(centers, centers)
    np.fill_diagonal(distances, np.inf)
    
    # For each circle, compute the minimum of wall distances and half of neighbor distances
    for i in range(n):
        wall_min = np.min(wall_distances[:, i])
        neighbor_min = 0.5 * np.min(distances[i, :])
        radii[i] = min(wall_min, neighbor_min)
    
    return radii

def neg_sum_radii(z):
    centers = z.reshape(-1, 2)
    radii = compute_radii(centers)
    return -np.sum(radii)

def run_packing():

    # Differential Evolution
    bounds = [(0, 1)] * 52
    result = differential_evolution(
        neg_sum_radii,
        bounds,
        strategy='best1bin',
        maxiter=2000,
        popsize=10,
        tol=1e-7,
        polish=False,
        seed=42
    )
    
    # Convert best result to centers
    centers = result.x.reshape(-1, 2)
    
    # Hill climbing refinement
    sigma = 0.02
    max_iter = 3000
    best_centers = centers.copy()
    best_sum = -neg_sum_radii(best_centers.flatten())
    
    for _ in range(max_iter):
        # Generate candidate

        candidate = best_centers + noise
        candidate = np.clip(candidate, 0, 1)
        
        # Evaluate

        if candidate_sum > best_sum:
            best_centers = candidate
            best_sum = candidate_sum
            sigma = min(1.1 * sigma, 0.5)  # Increase step size
        else:
            sigma = max(0.98 * sigma, 1e-5)  # Decrease step size
