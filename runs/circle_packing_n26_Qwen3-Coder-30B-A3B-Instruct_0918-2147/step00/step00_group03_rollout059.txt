import numpy as np
from scipy.optimize import minimize
import scipy.sparse as sp

def init_hex_grid(n=26, r_init=0.04):
    # Generate points on a hexagonal grid
    rows = []
    y = r_init
    row_num = 0
    while len(rows) < n:
        if row_num % 2 == 0:
            x_start = r_init
        else:
            x_start = 2 * r_init
        x = x_start
        while x <= 1 - r_init and len(rows) < n:
            rows.append((x, y))
            x += 2 * r_init
        y += np.sqrt(3) * r_init
        row_num += 1
    centers = np.array(rows[:n])
    radii = np.full(n, r_init)
    return centers, radii

def max_radii(centers, radii, max_iter=100, tolerance=1e-8):
    # Iteratively update radii to their maximum allowed values
    old_radii = radii.copy()
    for _ in range(max_iter):
        # Wall constraints
        wall = np.minimum(np.minimum(centers[:, 0], 1 - centers[:, 0]),
                          np.minimum(centers[:, 1], 1 - centers[:, 1]))
        
        # Neighbour constraints
        D = np.sqrt(((centers[:, None, :] - centers[None, :, :]) ** 2).sum(-1))
        np.fill_diagonal(D, np.inf)
        neighbor = D - radii[None, :]
        neighbor = np.minimum.reduce(neighbor, axis=1)
        
        # Combine constraints
        new_radii = np.minimum(wall, neighbor)
        new_radii = np.maximum(new_radii, 0)
        
        if np.max(np.abs(new_radii - old_radii)) < tolerance:
            break
        old_radii = new_radii
    return new_radii

def hill_climb(centers, radii, iterations=5000, step=0.01):
    # Stochastic hill climbing to improve layout
    for _ in range(iterations):
        i = np.random.randint(0, len(centers))
        old_r = radii[i]
        old_c = centers[i].copy()
        
        # Try small random displacement
        delta = (np.random.rand(2) - 0.5) * 2 * step
        new_c = old_c + delta
        
        # Ensure new center stays inside (r, 1-r) bounds
        if (new_c[0] >= radii[i] and new_c[0] <= 1 - radii[i] and
            new_c[1] >= radii[i] and new_c[1] <= 1 - radii[i]):
            
            centers[i] = new_c
            new_radii = max_radii(centers, radii)
            
            # Accept move if total sum improves
            if np.sum(new_radii) > np.sum(radii):
                radii[:] = new_radii
        else:
            # Move outside bounds, revert to old
            centers[i] = old_c
    return centers, radii
