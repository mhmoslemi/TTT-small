import numpy as np
from scipy.stats.qmc import Sobol

def initialize_seeds(n):
    sobol = Sobol(d=2, seed=42)
    points = sobol.random(n)
    margin = 0.02
    points = margin + (1 - 2 * margin) * points
    return points

def compute_feasible_radii(centers):
    n = centers.shape[0]
    # Compute distance matrix
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_matrix = np.sqrt(np.sum(diff ** 2, axis=2))
    # Nearest neighbour distances (excluding self)
    nn = np.min(dist_matrix + np.eye(n) * np.inf, axis=1)
    # Wall distances
    wall_distances = np.array([
        centers[:, 0],               # x edge distances
        1 - centers[:, 0],           # 1-x edge distances
        centers[:, 1],               # y edge distances
        1 - centers[:, 1]            # 1-y edge distances
    ]).min(axis=0)
    # Compute feasible radii
    radii = np.minimum(0.5 * nn, wall_distances)
    return radii

def redistribute_radii(radii, slack, alpha):
    total_slack = np.sum(slack)
    if total_slack <= 1e-12:
        return radii
    # Increase wall-limited circles
    increased = np.zeros_like(radii)
    wall_limited = slack > 1e-12
    if np.any(wall_limited):
        increase_amount = alpha * slack
        increased = increase_amount
        radii = np.clip(radii + increased, 0, None)
    return radii

def push_away(centers, radii, beta, eps):
    n = centers.shape[0]
    new_centers = centers.copy()
    for i in range(n):
        c_i = new_centers[i]
        r_i = radii[i]
        for j in range(n):
            if i == j:
                continue
            c_j = centers[j]
            r_j = radii[j]
            diff = c_i - c_j
            dist = np.sqrt(np.sum(diff ** 2))
            if dist < r_i + r_j + eps:
                overlap = r_i + r_j + eps - dist
                delta = beta * overlap * (diff / (dist + eps))
                new_centers[i] = c_i + delta
    # Project back into interior
    new_centers = np.clip(new_centers, eps, 1 - eps)
    return new_centers

def iterate(centers, radii, alpha, beta, max_iter):
    for _ in range(max_iter):
        old_sum = np.sum(radii)
        # Compute slack
        slack = 0.5 * np.min(
            np.linalg.norm(centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=2) + np.eye(len(centers)) * np.inf,
            axis=1) - radii
        slack = np.maximum(slack, 0.0)
        # Redistribute
        radii = redistribute_radii(radii, slack, alpha)
        # Push away
        centers = push_away(centers, radii, beta, 1e-8)
        # Re-evaluate radii
        radii = compute_feasible_radii(centers)
        # Check convergence
        new_sum = np.sum(radii)
        if new_sum - old_sum < 1e-5:
            break
    return centers, radii

def repair(centers, radii, eps_final):
    radii = np.maximum(0, radii - eps_final)
    centers = np.clip(centers, eps_final, 1 - eps_final)
    # Optional overlap correction
    n = centers.shape[0]
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff ** 2, axis=2))
    for i in range(n):
        for j in range(i + 1, n):
            if dist[i, j] < radii[i] + radii[j] - 1e-12:
                overlap = radii[i] + radii[j] - dist[i, j]
                radii[i] -= overlap / 2 + eps_final
                radii[j] -= overlap / 2 + eps_final
    radii = np.clip(radii, 0, None)
    centers = np.clip(centers, eps_final, 1 - eps_final)
    return centers, radii

def run_packing():
    # Parameters
    n_circles = 26
    initial_centers = initialize_seeds(n_circles)
    alpha = 0.4
    beta = 0.3
    max_iter = 200
    eps_final = 5e-10

    # Initial feasible radii
    initial_radii = compute_feasible_radii(initial_centers)

    # Iterate
    final_centers, final_radii = iterate(initial_centers, initial_radii, alpha, beta, max_iter)

    # Final repair
    final_centers, final_radii = repair(final_centers, final_radii, eps_final)

    # Ensure valid packing
    valid, _ = validate_packing(final_centers, final_radii)
    assert valid, "Final packing is invalid"

    total_radius = np.sum(final_radii)
    return final_centers, final_radii, total_radius
