import numpy as np
from scipy.optimize import minimize

def initial_layout():
    # Create a hexagonal grid pattern with 5 rows and 5 columns
    # to get 25 circles, then add a 26th one in the center
    rows, cols = 5, 5
    spacing = 0.18
    radius = spacing / 2.0 - 0.01
    
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = (j + 0.5) * spacing
            y = (i + 0.5) * spacing
            # Adjust to keep within bounds
            x = max(radius, min(1 - radius, x))
            y = max(radius, min(1 - radius, y))
            centers.append([x, y])
    
    # Add a 26th circle in the center
    centers.append([0.5, 0.5])
    
    centers = np.array(centers)
    radii = np.full(26, radius)
    return centers, radii

def pack_vector(centers, radii):
    z = np.empty(3 * 26)
    for i in range(26):
        z[3*i] = centers[i, 0]
        z[3*i+1] = centers[i, 1]
        z[3*i+2] = radii[i]
    return z

def unpack_vector(z):
    centers = np.empty((26, 2))
    radii = np.empty(26)
    for i in range(26):
        centers[i, 0] = z[3*i]
        centers[i, 1] = z[3*i+1]
        radii[i] = z[3*i+2]
    return centers, radii

def objective(z):
    centers, radii = unpack_vector(z)
    return -np.sum(radii)

def constraint_boundary(z):
    centers, radii = unpack_vector(z)
    result = []
    for i in range(26):
        x, y, r = centers[i, 0], centers[i, 1], radii[i]
        # x - r >= 0
        result.append(x - r)
        # 1 - x - r >= 0
        result.append(1 - x - r)
        # y - r >= 0
        result.append(y - r)
        # 1 - y - r >= 0
        result.append(1 - y - r)
    return np.array(result)

def constraint_nonoverlap(z):
    centers, radii = unpack_vector(z)
    result = []
    for i in range(26):
        for j in range(i+1, 26):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx*dx + dy*dy
            r_sum = radii[i] + radii[j]
            # (x_i-x_j)^2 + (y_i-y_j)^2 - (r_i + r_j)^2 >= 0
            result.append(dist_sq - r_sum * r_sum)
    return np.array(result)

def run_packing():
    best_sum = -1
    best_centers = None
    best_radii = None
    
    for trial in range(5):
        # Get initial layout
        centers, radii = initial_layout()
        
        # Jitter centers slightly
        if trial > 0:
            jitter = (np.random.rand(26, 2) - 0.5) * 0.02
            centers += jitter
        
        # Start optimization with the same seed
        np.random.seed(trial * 12345)
        z0 = pack_vector(centers, radii)
        
        # Define bounds
        bounds = [(0, 1), (0, 1), (0, 0.5)] * 26
        
        # Define constraints
        constraints = []
        constraints.append({'type': 'ineq', 'fun': constraint_boundary})
        constraints.append({'type': 'ineq', 'fun': constraint_nonoverlap})
        
        # Optimize
        res = minimize(
            objective,
            z0,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'ftol': 1e-9, 'maxiter': 2000}
        )
        
        if not res.success:
            continue
            
        centers_opt, radii_opt = unpack_vector(res.x)
        
        # Ensure non-negative radii
        radii_opt = np.maximum(radii_opt, 0)

        if not valid:

            # Shrink all radii slightly
            radii_opt = np.maximum(radii_opt * 0.999999, 0)
            valid, msg = validate_packing(centers_opt, radii_opt)
            if not valid:
                continue
        
        current_sum = np.sum(radii_opt)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers_opt.copy()
            best_radii = radii_opt.copy()
    
    if best_centers is None:
        raise RuntimeError("Failed to find a valid packing")
