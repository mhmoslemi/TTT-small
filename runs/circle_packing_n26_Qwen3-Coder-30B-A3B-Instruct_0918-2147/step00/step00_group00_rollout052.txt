import numpy as np
import random

def make_hex_seed(n, margin):
    # Generate a hexagonal packing with margin from boundaries
    rows = int(np.ceil(np.sqrt(n / (np.sqrt(3)/2))))
    cols = int(np.ceil(n / rows))
    
    # Hexagon packing parameters
    width = 1 - 2*margin
    height = 1 - 2*margin
    
    # Inner radius of hexagon (distance between centres)
    r = 0.5
    dx = r * 2
    dy = r * np.sqrt(3)
    
    # Determine number of rows/cols that can actually fit
    actual_rows = int(height / dy)
    actual_cols = int(width / dx)
    
    if actual_rows * actual_cols < n:
        # If not enough space, scale down
        dx = width / actual_cols
        dy = height / actual_rows
        r = dx / 2
        actual_rows = int(height / dy)
        actual_cols = int(width / dx)
    
    # Calculate actual scale and shift to center
    scale = min(width / (actual_cols * dx), height / (actual_rows * dy))
    
    dx *= scale
    dy *= scale
    r *= scale
    
    centers = []
    radii = []
    
    row_offset = (1 - width) / 2 + margin
    col_offset = (1 - height) / 2 + margin
    
    for i in range(actual_rows):
        y = col_offset + i * dy
        for j in range(actual_cols):
            x = row_offset + j * dx
            if i % 2 == 1:
                x += dx / 2
            
            # Check if we've added enough circles
            if len(centers) >= n:
                break
            
            # Compute minimum distance to boundaries
            min_dist_to_edge = min(x, 1 - x, y, 1 - y)
            # Compute radius as a function of distance to wall, trying to make order of magnitude 0.01
            rad = max(min_dist_to_edge * 0.8, 0.005)
            if rad > 0.1:
                rad = 0.1
            centers.append([x, y])
            radii.append(rad)
        
        if len(centers) >= n:
            break
    
    # Trim if needed
    centers = centers[:n]
    radii = radii[:n]
    
    # Ensure at least one circle is not too small
    if len(radii) > 0 and min(radii) < 0.015:
        for i in range(len(radii)):
            radii[i] = max(radii[i], 0.015)
    
    # Return in numpy arrays
    return np.array(centers), np.array(radii)


def max_feasible_radius(i, centers, radii):
    # Computes the max feasible radius for circle i given current layout
    x, y = centers[i]
    r_current = radii[i]
    
    # Distance to four walls
    d_left = x
    d_right = 1 - x
    d_bottom = y
    d_top = 1 - y
    
    d_min = min(d_left, d_right, d_bottom, d_top)
    
    # Distance to all other circles
    for j in range(centers.shape[0]):
        if i == j:
            continue
        dist_ij = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
        d_nei = dist_ij - radii[j]
        d_min = min(d_min, d_nei)
    
    # Take minimum with current radius to avoid shrinking
    r_max = min(d_min, r_current)
    return r_max


def inflate_radii(centers, radii, alpha):
    # Adds a global step to all radii and then clamps based on feasibility
    total = np.sum(radii)
    mean_r = total / len(radii)
    delta = alpha * mean_r
    
    # Update all radii by delta
    new_radii = radii + delta
    
    # Clamp each one to its feasible maximum
    total_increase = 0
    for i in range(len(new_radii)):
        max_feas = max_feasible_radius(i, centers, new_radii) 
        new_radii[i] = min(new_radii[i], max_feas)
        total_increase += new_radii[i] - radii[i]
    
    return new_radii, total_increase


def anneal_centers(centers, radii, temp0, gamma, sigma0, steps):
    # Performs simulated annealing on center positions
    old_total = np.sum(radii)
    best_total = old_total
    best_centers = centers.copy()
    best_radii = radii.copy()
    
    current_centers = centers.copy()
    current_radii = radii.copy()
    
    temp = temp0
    sigma = sigma0
    
    global_steps = 0
    
    for step in range(steps):
        # Occasionally shift entire configuration to avoid symmetry traps
        if step % 50 == 0:
            shift = 0.001 * (np.random.rand(2) - 0.5)
            current_centers += shift
        
        # Pick a random circle to move
        i = np.random.randint(0, len(current_centers))
        
        # Step size according to current temperature
        step_size = sigma * temp / temp0
        
        # Sample displacement
        dx, dy = np.random.normal(0, step_size, 2)
        
        # Save original position and new one
        old_pos = current_centers[i].copy()
        new_pos = old_pos + np.array([dx, dy])
        
        # Clip to unit square
        new_pos = np.clip(new_pos, 0, 1)
        
        # Set new position
        current_centers[i] = new_pos
        
        # Compute new feasible max radius for this circle
        new_max_r = max_feasible_radius(i, current_centers, current_radii)
        
        # Apply it to this circle's radius
        old_r = current_radii[i]
        current_radii[i] = new_max_r
        
        # Compute updated sum
        new_total = np.sum(current_radii)
        
        # Accept or reject the change
        delta = new_total - old_total
        
        if delta > 0:
            old_total = new_total
        elif np.random.rand() < np.exp(delta / temp):
            old_total = new_total
        else:
            # Reject: revert position
            current_centers[i] = old_pos
            current_radii[i] = old_r
            continue  # This step did not improve
            
        # Update best
        if old_total > best_total:
            best_total = old_total
            best_centers = current_centers.copy()
            best_radii = current_radii.copy()
            
        # Update temperature
        temp *= gamma
        
        global_steps += 1
    
    return best_centers, best_radii, best_total


def mixed_cycle(centers, radii, params):
    alpha = params['alpha_zero']
    gamma = params['gamma']
    temp0 = params['temp0']
    sigma0 = params['sigma0']
    sa_steps = params['sa_steps']
    inflate_steps = params['inflate_steps']
    stagnation_thresh = params['stagnation_thresh']
    max_cycles = params['max_cycles']
    
    total = np.sum(radii)
    last_total = total
    last_iters = 0
    
    num_cycles = 0
    
    while num_cycles < max_cycles:
        # Phase B: Inflation
        for inflate_iter in range(inflate_steps):
            new_radii, increase = inflate_radii(centers, radii, alpha)
            radii = new_radii
            total += increase
            
            # Check convergence
            if abs(increase) < 1e-6:
                last_iters += 1
            else:
                last_iters = 0
            
            if last_iters > stagnation_thresh:
                break
            
        # Update alpha for next inflation round
        alpha *= gamma
        
        # Phase C: SA on centers
        centers, radii, total = anneal_centers(centers, radii, temp0, gamma, sigma0, sa_steps)
        
        # Measurement
        if total - last_total < 1e-6:
            num_cycles += 1
        else:
            num_cycles = 0
    
    return centers, radii, total


def polish_radii(centers, radii, beta, passes):
    # Applies exact local inflation with overlap resolution
    for p in range(passes):
        old_sum = np.sum(radii)
        
        # Compute actual gaps for all circles
        updated = False
        for i in range(len(radii)):
            x, y = centers[i]
            r = radii[i]
            
            # Wall gaps
            g_wall = min(x, 1 - x, y, 1 - y)
            
            # Neighbour gaps
            g_nei = float('inf')
            for j in range(len(centers)):
                if i == j:
                    continue
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                gap = dist - (r + radii[j])
                if gap < g_nei:
                    g_nei = gap
            
            g = min(g_wall, g_nei)
            
            if g > 0:
                # Increase radius by factor
                inc = beta * g
                new_r = r + inc
                max_feas = max_feasible_radius(i, centers, radii)
                if new_r <= max_feas:
                    radii[i] = new_r
                    updated = True
        
        # Now we may have overlaps, resolve them
        for _ in range(3):  # Try a few times
            any_overlap = False
            for i in range(len(centers)):
                for j in range(i + 1, len(centers)):
                    dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                    total_r = radii[i] + radii[j]
                    overlap = total_r - dist
                    
                    if overlap > 0:
                        any_overlap = True
                        reduction = overlap / 2
                        
                        radii[i] = max(0, radii[i] - reduction)
                        radii[j] = max(0, radii[j] - reduction)
            
            if not any_overlap:
                break
            
        # If the polish didn't improve, we're done
        if np.sum(radii) <= old_sum + 1e-8:
            break
    
    return centers, radii


def run_packing():
    # Set random seed for reproducibility
    np.random.seed(42)
    random.seed(42)
    
    # Default parameters
    params = {
        'alpha_zero': 0.04,
        'gamma': 0.95,
        'temp0': 0.02,
        'sigma0': 0.03,
        'sa_steps': 200,
        'inflate_steps': 50,
        'stagnation_thresh': 8,
        'max_cycles': 5
    }
    
    # Phase A: Initial Hexagon Seed
    centers, radii = make_hex_seed(26, 0.04)
    
    # Phase B, C, D: Mixed Inflation and SA
    centers, radii, total = mixed_cycle(centers, radii, params)
    
    # Phase E: Polishing
    centers, radii = polish_radii(centers, radii, 0.6, 3)
    
    # Final validation
    _, msg = validate_packing(centers, radii)
    if not msg == "ok":
        raise RuntimeError("Final packing failed validation: " + msg)
    
    return centers, radii, np.sum(radii)
