import numpy as np
from scipy.spatial.distance import cdist

def penalty(z, alpha, beta):
    centers = z.reshape(-1, 3)[:, :2]
    radii = z.reshape(-1, 3)[:, 2]
    
    # Wall penalties
    wall_distances = np.array([
        centers[:, 0],
        1 - centers[:, 0],
        centers[:, 1],
        1 - centers[:, 1]
    ])
    wall_penalties = np.maximum(0, radii[:, None] - wall_distances.T).sum(axis=1)
    wall_penalty = alpha * wall_penalties.sum()
    
    # Overlap penalties
    pairwise_distances = cdist(centers, centers)
    overlap_penalties = np.maximum(0, radii[:, None] + radii[None, :] - pairwise_distances)
    np.fill_diagonal(overlap_penalties, 0)
    overlap_penalty = beta * overlap_penalties.sum() / 2
    
    return wall_penalty + overlap_penalty

def energy(z, lam, alpha, beta):
    sum_radii = np.sum(z.reshape(-1, 3)[:, 2])
    return -sum_radii + lam * penalty(z, alpha, beta)

def propose_move(z, sigma_c, sigma_r, T):
    new_z = z.copy()
    move_type = np.random.choice(3)
    
    if move_type == 0:  # Center jitter
        idx = np.random.randint(len(z) // 3)
        new_z[idx*3:idx*3+2] += np.random.normal(0, sigma_c * np.sqrt(T), 2)
        new_z[idx*3:idx*3+2] = np.clip(new_z[idx*3:idx*3+2], 0, 1)
        
    elif move_type == 1:  # Radius perturb
        idx = np.random.randint(len(z) // 3)
        new_z[idx*3+2] += np.random.normal(0, sigma_r * np.sqrt(T))
        new_z[idx*3+2] = np.maximum(0, new_z[idx*3+2])
        
    else:  # Swap pair
        idx1, idx2 = np.random.choice(len(z) // 3, 2, replace=False)
        new_z[idx1*3+2], new_z[idx2*3+2] = new_z[idx2*3+2], new_z[idx1*3+2]
    
    return new_z

def anneal():
    n_circles = 26
    alpha, beta = 10, 10
    T0, cT = 1.0, 0.995
    lam0, cl = 0.1, 1.03
    sigma_c, sigma_r = 0.03, 0.02
    max_iterations = 10000
    max_proposals = 200000
    
    # Initialize centers on hexagonal grid
    spacing = 0.18
    rows = int(np.ceil(1 / spacing))
    x_vals = np.linspace(spacing/2, 1 - spacing/2, rows)
    y_vals = np.linspace(spacing/2, 1 - spacing/2, rows)
    centers = []
    for i, x in enumerate(x_vals):
        for j, y in enumerate(y_vals):
            if len(centers) < n_circles:
                centers.append([x, y])
            else:
                break
        if len(centers) >= n_circles:
            break
    
    # Fill with any remaining positions
    if len(centers) < n_circles:
        for _ in range(n_circles - len(centers)):
            centers.append([np.random.uniform(0, 1), np.random.uniform(0, 1)])
    
    centers = np.array(centers[:n_circles])
    
    # Initial radii are wall limiting
    radii = np.minimum(np.minimum(centers[:, 0], 1 - centers[:, 0]),
                       np.minimum(centers[:, 1], 1 - centers[:, 1]))
    
    # Initial state
    z = np.hstack([centers.flatten(), radii])
    
    # Annealing loop
    T = T0
    lam = lam0
    best_feasible_z = z.copy()
    best_feasible_sum = np.sum(radii)
    best_e = energy(z, lam, alpha, beta)
    
    proposals = 0
    for iter in range(max_iterations):
        if T < 1e-5:
            break
            
        # Propose move
        z_prop = propose_move(z, sigma_c, sigma_r, T)
        proposals += 1
        
        # Compute energy
        e_prop = energy(z_prop, lam, alpha, beta)
        
        # Accept or reject
        delta_e = e_prop - energy(z, lam, alpha, beta)
        if delta_e <= 0 or np.random.rand() < np.exp(-delta_e / T):
            z = z_prop
        
        # Update best feasible
        if penalty(z, alpha, beta) < 1e-5:
            current_radii = z.reshape(-1, 3)[:, 2]
            current_sum = np.sum(current_radii)
            if current_sum > best_feasible_sum:
                best_feasible_sum = current_sum
                best_feasible_z = z.copy()
                best_e = e_prop
                
        # Cooling and penalty weight update
        T *= cT
        lam *= cl
        
        # Early stopping of annealing if already saturated
        if proposals > max_proposals:
            break
    
    return best_feasible_z

def project_and_ascent(centers, radii, max_iter=500, eta=5e-3, decay=0.95, tol=1e-7):
    z = np.hstack([centers.flatten(), radii])
    n = len(radii)
    
    for step in range(max_iter):
        z_copy = z.copy()
        # Gradient w.r.t. radii (only)
        grad = np.zeros_like(z)
        grad[2::3] = 1
        
        # Gradient step
        z = z + eta * grad
        
        # Projection step
        z = z.reshape(-1, 3)
        centers_proj = z[:, :2]
        radii_proj = z[:, 2]
        
        # Wall projection
        for i in range(n):
            r_i = radii_proj[i]
            x_i, y_i = centers_proj[i]
            max_r = min(x_i, 1-x_i, y_i, 1-y_i)
            radii_proj[i] = max(0, min(r_i, max_r))
        
        # Overlap resolution
        centers_proj = z[:, :2]
        radii_proj = z[:, 2]
        while True:
            pairwise_distances = cdist(centers_proj, centers_proj)
            overlap_matrix = radii_proj[:, None] + radii_proj[None, :] - pairwise_distances
            np.fill_diagonal(overlap_matrix, 0)
            
            max_overlap = np.max(overlap_matrix)
            if max_overlap <= 0:
                break
                
            # Find overlapping pair
            overlap_matrix[overlap_matrix <= 0] = np.inf
            pair_idx = np.unravel_index(np.argmin(overlap_matrix), overlap_matrix.shape)
            
            i, j = pair_idx
            o_ij = overlap_matrix[i, j]
            if o_ij > 0:
                radii_proj[i] -= o_ij / 2
                radii_proj[j] -= o_ij / 2
                radii_proj[i] = max(0, radii_proj[i])
                radii_proj[j] = max(0, radii_proj[j])
            
            # Reclamp
            for idx in range(n):
                r_i = radii_proj[idx]
                x_i, y_i = centers_proj[idx]
                max_r = min(x_i, 1-x_i, y_i, 1-y_i)
                radii_proj[idx] = max(0, min(r_i, max_r))
        
        z = np.hstack([centers_proj.flatten(), radii_proj])
        z = z.reshape(-1)
        
        # Check for convergence
        if np.linalg.norm(z - z_copy) < tol:
            break
            
        eta *= decay
        
    centers = z.reshape(-1, 3)[:, :2]
    radii = z.reshape(-1, 3)[:, 2]
    return centers, radii

def run_packing():
    # Anneal to get a good initial configuration
    z = anneal()
    
    # Extract centers and radii from z
    centers = z.reshape(-1, 3)[:, :2]
    radii = z.reshape(-1, 3)[:, 2]
    
    # Perform local refinement using projected gradient ascent
    centers, radii = project_and_ascent(centers, radii)
    
    # Validate final solution
    _, dummy = validate_packing(centers, radii)
    
    # Ensure valid packing
    sum_radii = np.sum(radii)
    
    # Check if necessary adjustments needed
