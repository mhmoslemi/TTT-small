import numpy as np
import cvxpy as cp
from scipy.spatial.distance import cdist

def build_socp_problem(n):
    r = cp.Variable(n, nonneg=True)
    dist_params = []
    constraints = []
    
    # Boundary constraints
    x = cp.Parameter(n)
    y = cp.Parameter(n)
    constraints += [r <= x, r <= 1-x, r <= y, r <= 1-y]
    
    # Non-overlap constraints
    for i in range(n):
        for j in range(i+1, n):
            dist_param = cp.Parameter(nonneg=True)
            dist_params.append(dist_param)
            constraints.append(r[i] + r[j] <= dist_param)
    
    prob = cp.Problem(cp.Maximize(cp.sum(r)), constraints)
    return prob, r, dist_params

def solve_radii(centres, prob, r, dist_params):
    n = centres.shape[0]
    x = centres[:, 0]
    y = centres[:, 1]
    
    # Update boundary parameters
    prob.parameters()[0].value = x
    prob.parameters()[1].value = y
    
    # Update distance parameters
    dists = cdist(centres, centres)
    for i, param in enumerate(dist_params):
        param.value = dists.flat[i]
    
    prob.solve(solver=cp.ECOS, abstol=1e-10, reltol=1e-10)
    radii = r.value
    if radii is None:
        raise RuntimeError("SOCP solver failed")
    return radii

def energy(centres, prob, r, dist_params):
    radii = solve_radii(centres, prob, r, dist_params)
    return -np.sum(radii)

def random_move(state, rng):
    new_state = state.copy()
    n = len(state) // 2
    move_type = rng.choice(['single', 'cluster', 'edge', 'swap'])
    
    if move_type == 'single':
        idx = rng.integers(n)
        dx = rng.normal(0, 0.02)
        dy = rng.normal(0, 0.02)
        new_state[2*idx] = np.clip(new_state[2*idx] + dx, 0, 1)
        new_state[2*idx+1] = np.clip(new_state[2*idx+1] + dy, 0, 1)
    elif move_type == 'cluster':
        indices = rng.choice(n, size=rng.integers(3, 6), replace=False)
        offset_x = rng.normal(0, 0.02)
        offset_y = rng.normal(0, 0.02)
        for i in indices:
            new_state[2*i] = np.clip(new_state[2*i] + offset_x, 0, 1)
            new_state[2*i+1] = np.clip(new_state[2*i+1] + offset_y, 0, 1)
    elif move_type == 'edge':
        idx = rng.integers(n)
        x, y = new_state[2*idx], new_state[2*idx+1]
        delta = 0.01
        if x < 0.5:
            new_state[2*idx] = np.clip(new_state[2*idx] - delta, 0, 1)
        else:
            new_state[2*idx] = np.clip(new_state[2*idx] + delta, 0, 1)
        if y < 0.5:
            new_state[2*idx+1] = np.clip(new_state[2*idx+1] - delta, 0, 1)
        else:
            new_state[2*idx+1] = np.clip(new_state[2*idx+1] + delta, 0, 1)
    else:  # swap
        pass  # no swap in this implementation
    
    return new_state

def anneal_chain(seed, prob, r, dist_params):
    rng = np.random.default_rng(seed)
    # Initialize with random positions
    state = rng.random(52)
    best_state = state.copy()
    best_energy = energy(state.reshape(-1, 2), prob, r, dist_params)
    
    # Annealing schedule
    T0 = 0.05
    alpha = 0.995
    T = T0
    iter_count = 0
    max_iter = 150000
    
    while T > 1e-6 and iter_count < max_iter:
        new_state = random_move(state, rng)
        new_energy = energy(new_state.reshape(-1, 2), prob, r, dist_params)
        
        delta = new_energy - best_energy
        if delta < 0 or rng.random() < np.exp(-delta / T):
            state = new_state
            if new_energy < best_energy:
                best_energy = new_energy
                best_state = state.copy()
            iter_count += 1
        
        T *= alpha
    
    centres = best_state.reshape(-1, 2)
    radii = solve_radii(centres, prob, r, dist_params)
    return centres, radii, -best_energy

def local_refine(centres, radii, prob, r, dist_params):
    n = len(centres)
    # Converge to near-optimal configuration by active set Newton approach
    for _ in range(10):
        # Identify active constraints
        new_radii = solve_radii(centres, prob, r, dist_params)
        if np.allclose(new_radii, radii, atol=1e-5):
            break
        radii = new_radii
    
    return centres, radii

def finalise(centres, radii):
    prob, r, dist_params = build_socp_problem(len(centres))
    x = centres[:, 0]
    y = centres[:, 1]
    prob.parameters()[0].value = x
    prob.parameters()[1].value = y
    dists = cdist(centres, centres)
    for i, param in enumerate(dist_params):
        param.value = dists.flat[i]
    prob.solve(solver=cp.ECOS, abstol=1e-12, reltol=1e-12)
    radii = r.value
    if radii is None:
        raise RuntimeError("Final SOCP solve failed")
    return centres, radii

def run_packing():
    prob, r, dist_params = build_socp_problem(26)
    best_sum = 0
    best_centers = None
    best_radii = None
    
    # Multiple chains with different random seeds
    for seed in range(8):
        try:
            centres, radii, sum_radii = anneal_chain(seed, prob, r, dist_params)
            if sum_radii > best_sum:
                best_sum = sum_radii
                best_centers = centres
                best_radii = radii
        except Exception as e:
            continue
    
    # Refine best solution
    refined_centers, refined_radii = local_refine(best_centers, best_radii, prob, r, dist_params)
    
    # Final validation
    final_centers, final_radii = finalise(refined_centers, refined_radii)
    
    # Validate placement
    valid, msg = validate_packing(final_centers, final_radii)
    if not valid:
        raise RuntimeError(f"Final packing invalid: {msg}")
        
    return final_centers, final_radii, np.sum(final_radii)
