import numpy as np
from scipy.optimize import minimize

def hex_lattice_points(n, d):
    """Generate n points in a hexagonal (triangular) lattice within [0,1]^2."""
    # Hexagonal lattice spacing
    dy = d * np.sqrt(3) / 2
    rows = int(np.ceil(np.sqrt(n / (np.sqrt(3) / 2))))
    points = []
    for i in range(rows):
        offset = (i % 2) * d / 2
        for j in range(int(np.ceil(n / rows)) + 1):
            x = offset + j * d
            y = i * dy
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append([x, y])
                if len(points) >= n:
                    break
        if len(points) >= n:
            break
    return np.array(points[:n])

def initial_guess():
    """Generate an initial feasible configuration for 26 circles."""
    # Start with a reasonably dense hexagonal lattice
    d = 0.2  # spacing parameter
    points = hex_lattice_points(26, d)
    
    # Determine maximum radius based on distance to walls
    r_wall = min(
        np.min(points[:, 0]), np.min(1 - points[:, 0]),
        np.min(points[:, 1]), np.min(1 - points[:, 1])
    )
    
    # Set all radii to minimal of wall distance and spacing/2
    r = min(r_wall, d / 2)
    
    # Build initial guess vector [x0, y0, r0, x1, y1, r1, ...]
    vars_init = np.empty(3 * 26)
    for i in range(26):
        vars_init[3*i] = points[i, 0]  # x
        vars_init[3*i + 1] = points[i, 1]  # y
        vars_init[3*i + 2] = r  # r
    return vars_init

def objective(vars):
    """Minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def wall_constraints(vars):
    """Generate wall constraint values for SLSQP. Return list of inequalities."""
    n = 26
    g = []
    for i in range(n):
        x = vars[3*i]
        y = vars[3*i + 1]
        r = vars[3*i + 2]
        # x - r >= 0  == r - x <= 0
        g.append(r - x)
        # x + r <= 1  == x + r - 1 <= 0
        g.append(x + r - 1)
        # y - r >= 0  == r - y <= 0
        g.append(r - y)
        # y + r <= 1  == y + r - 1 <= 0
        g.append(y + r - 1)
    return g

def nonoverlap_constraints(vars):
    """Generate all non-overlap constraint values."""
    n = 26
    g = []
    for i in range(n):
        for j in range(i + 1, n):
            x1 = vars[3*i]
            y1 = vars[3*i + 1]
            r1 = vars[3*i + 2]
            x2 = vars[3*j]
            y2 = vars[3*j + 1]
            r2 = vars[3*j + 2]
            # (x1 - x2)^2 + (y1 - y2)^2 >= (r1 + r2)^2
            dist_sq = (x1 - x2)**2 + (y1 - y2)**2
            sum_radii_sq = (r1 + r2)**2
            g.append(dist_sq - sum_radii_sq)
    return g

def run_packing():
    """Packing 26 circles in unit square to maximize sum of radii."""
    # Get initial guess
    x0 = initial_guess()
    
    # Set bounds for each variable
    bounds = []
    for i in range(26):
        bounds.append((0, 1))      # x_i
        bounds.append((0, 1))      # y_i
        bounds.append((0, 0.5))    # r_i
    
    # Collect all constraints
    constraints = []
    constraints.append({'type': 'ineq', 'fun': lambda v: wall_constraints(v)})
    constraints.append({'type': 'ineq', 'fun': lambda v: nonoverlap_constraints(v)})
    
    # Optimize
    res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                   constraints=constraints, options={'maxiter': 2000, 'ftol': 1e-10})
