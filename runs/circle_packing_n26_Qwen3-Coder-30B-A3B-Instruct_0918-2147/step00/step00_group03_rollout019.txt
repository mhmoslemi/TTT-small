import numpy as np
from scipy.optimize import minimize
import scipy.sparse as sp

def init_hex_grid(n=26, r_init=0.04):
    # Generate hexagonal grid points and keep first n inside [0,1]^2
    h = np.sqrt(3) * r_init
    rows = []
    y = r_init
    row_idx = 0
    while len(rows) < n:
        if row_idx % 2 == 0:
            x_start = r_init
        else:
            x_start = 2 * r_init
        x = x_start
        while len(rows) < n and x <= 1:
            rows.append((x, y))
            x += 2 * r_init
        y += h
        row_idx += 1
    centers = np.array(rows[:n])
    radii = np.full(n, r_init)
    return centers, radii

def max_radii(centers, radii):
    # Iteratively compute maximum radii for each circle given fixed centers
    max_iter = 100
    eps = 1e-12
    for _ in range(max_iter):
        D = np.sqrt(((centers[:, None, :] - centers[None, :, :]) ** 2).sum(-1))
        # Set diagonal to large value to avoid self-interaction
        D[np.arange(len(D)), np.arange(len(D))] = 1e12
        # Compute wall constraints
        wall = np.minimum(centers, 1 - centers).min(axis=1)
        # Compute neighbor constraints
        neighbor = D - radii[:, None]
        # The minimum of all constraints
        min_constraints = np.minimum(wall, neighbor.min(axis=1))
        new_radii = np.maximum(min_constraints, 0)
        if np.max(np.abs(new_radii - radii)) < eps:
            break
        radii = new_radii
    return radii

def hill_climb(centers, radii, iterations=5000, step=0.01):
    # Stochastic hill-climbing to improve sum of radii
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = np.sum(best_radii)
    for _ in range(iterations):
        i = np.random.randint(0, len(centers))
        old_center = centers[i].copy()
        old_radius = radii[i]
        # Propose small random displacement
        new_center = old_center + np.random.uniform(-step, step, 2)
        # Ensure new center is valid
        if (new_center[0] - old_radius >= 0 and new_center[0] + old_radius <= 1 and
            new_center[1] - old_radius >= 0 and new_center[1] + old_radius <= 1):
            centers[i] = new_center
            # Recompute maximum radii
            radii = max_radii(centers, radii)
            new_sum = np.sum(radii)
            if new_sum > best_sum:
                # Accept the move
                best_centers = centers.copy()
                best_radii = radii.copy()
                best_sum = new_sum
            else:
                # Reject the move
                centers[i] = old_center
                radii[i] = old_radius
    return best_centers, best_radii
