import numpy as np
from scipy.optimize import minimize


def hex_lattice_points(n, d):
    """Generate at least n points on a hexagonal lattice with spacing d."""
    rows = []
    y = 0.0
    row_idx = 0
    while len(rows) < n:
        row = []
        x = 0.0 if row_idx % 2 == 0 else d / 2
        while x <= 1.0 + d:
            if 0.0 <= y <= 1.0:
                row.append((x, y))
            x += d
        rows.extend(row)
        y += d * np.sqrt(3) / 2
        row_idx += 1
    return np.array(rows[:n])

    y_min, y_max = points[:, 1].min(), points[:, 1].max()
    
    center_x = (x_min + x_max) / 2
    center_y = (y_min + y_max) / 2
    
    points[:, 0] -= center_x - 0.5
    points[:, 1] -= center_y - 0.5
    
    # Compute radii bounded by walls
    r_wall = np.inf
    for x, y in points:
        r = min(x, 1 - x, y, 1 - y)
        r_wall = min(r_wall, r)
    
    # For safety, use a slightly smaller radius
    r = 0.95 * r_wall
    
    # Pack into variable vector
    vars = []
    for x, y in points:
        vars.extend([x, y, r])
    return np.array(vars)


def objective(vars):
    """Minimize negative sum of radii."""
    radii = vars[2::3]
    return -np.sum(radii)


def wall_constraints(vars):

        # y - r ≥ 0
        result.append(r - y)
        # y + r ≤ 1
        result.append(y + r - 1)

def nonoverlap_constraints(vars):
