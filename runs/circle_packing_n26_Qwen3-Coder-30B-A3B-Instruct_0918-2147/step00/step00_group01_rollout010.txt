import numpy as np
from scipy.optimize import minimize

def initial_layout():
    # Create a hexagonal grid pattern
    n_rows = 5
    n_cols = 5
    spacing = 0.18
    radius = spacing / 2 - 0.01
    
    centers = []
    for i in range(n_rows):
        for j in range(n_cols):
            x = (j + 0.5) * spacing
            y = (i + 0.5) * spacing
            # Adjust position to keep within square
            x = max(radius, min(1.0 - radius, x))
            y = max(radius, min(1.0 - radius, y))
            centers.append([x, y])
    
    # Add a 26th circle in the center
    centers.append([0.5, 0.5])
    
    # Convert to numpy arrays
    centers = np.array(centers)
    radii = np.full(26, radius)
    
    # Ensure we have exactly 26 circles
    assert centers.shape[0] == 26
    assert radii.shape[0] == 26
    
    return centers, radii

def pack_vector(centers, radii):
    return np.concatenate([centers.flatten(), radii])

def unpack_vector(z):
    n = len(z) // 3
    centers = z[:2*n].reshape(n, 2)
    radii = z[2*n:]
    return centers, radii

def objective(z):
    _, radii = unpack_vector(z)
    return -np.sum(radii)  # Negative because we minimize

def constraint_boundary(z):
    centers, radii = unpack_vector(z)
    constraints = []
    # For each circle, add 4 boundary constraints
    for i in range(len(centers)):
        x, y = centers[i]
        r = radii[i]
        # x - r >= 0
        constraints.append(x - r)
        # 1 - x - r >= 0
        constraints.append(1 - x - r)
        # y - r >= 0
        constraints.append(y - r)
        # 1 - y - r >= 0
        constraints.append(1 - y - r)
    return np.array(constraints)

def constraint_nonoverlap(z):
    centers, radii = unpack_vector(z)
    n = len(centers)
    constraints = []
    
    # For each pair of circles
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist_sq = dx*dx + dy*dy
            r_sum = radii[i] + radii[j]
            # We want dist_sq >= (r_sum)^2 => (dist_sq - r_sum^2) >= 0
            constraints.append(dist_sq - r_sum * r_sum)
            
    return np.array(constraints)

        radii = np.clip(radii, 0, 0.5)

        # Define the bounds (x, y, r) for all circles
        bounds = []

            bounds.append((0, 1))  # y bounds
            bounds.append((0, 0.5))  # r bounds

        constraints = []
        constraints.append({'type': 'ineq', 'fun': constraint_boundary})
        constraints.append({'type': 'ineq', 'fun': constraint_nonoverlap})
        
        # Minimize using SLSQP
        res = minimize(objective, z0, method='SLSQP', bounds=bounds, constraints=constraints,

            final_centers, final_radii = unpack_vector(res.x)
