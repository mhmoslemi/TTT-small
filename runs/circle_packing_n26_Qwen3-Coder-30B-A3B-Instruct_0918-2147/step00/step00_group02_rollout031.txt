import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def initial_configuration(n=26):
    # Generate Sobol points for initial centres
    # Using a deterministic low-discrepancy sequence approach
    # This is a placeholder for a proper Sobol sequence generator if needed
    # We use a pseudo-random pattern that's well distributed for this small n
    np.random.seed(42)  # For reproducibility
    centers = np.random.rand(n, 2)
    
    # Set initial radii
    radii = np.zeros(n)
    for i in range(n):
        x, y = centers[i]
        min_dist_to_wall = min(x, 1-x, y, 1-y)
        # For each point, find distance to nearest neighbor
        dists = np.sqrt(np.sum((centers - centers[i])**2, axis=1))
        dists[i] = np.inf
        min_dist_to_nearest = np.min(dists)
        radii[i] = min(min_dist_to_wall, min_dist_to_nearest / 2.0)
    
    # Ensure all radii are valid (non-negative)
    radii = np.maximum(radii, 0.0)
    
    return centers, radii

def pack(centers, radii):
    # Pack into 1D array
    n = centers.shape[0]
    z = np.zeros(2 * n + n)
    z[0:n] = centers[:, 0]      # x coords
    z[n:2*n] = centers[:, 1]    # y coords
    z[2*n:2*n+n] = radii        # radii
    return z

def unpack(z):
    n = (len(z) - 26) // 3
    centers = np.zeros((n, 2))
    centers[:, 0] = z[0:n]
    centers[:, 1] = z[n:2*n]
    radii = z[2*n:2*n+n]
    return centers, radii

def max_feasible_budget(z_start):
    B = 3.0         # Initial generous budget B0
    lambda_budget = 100.0
    delta = 5e-4
    max_iter = 40
    eps = 1e-13
    B_history = [B]
    
    z_current = z_start.copy()
    
    for iteration in range(max_iter):
        # Inner optimization: minimize energy
        res = inner_optimize(z_current, B, lambda_budget)
        z_opt = res.x
        
        # Check feasibility
        centers_opt, radii_opt = unpack(z_opt)
        violations = max_violation(centers_opt, radii_opt)
        
        feasibility = violations <= eps
        
        # Adjust budget
        if feasibility:
            B_new = B + delta
            # Ensure it does not exceed 3.0
            B_new = min(B_new, 3.0)
            B = B_new
            z_current = z_opt
        else:
            B_new = B - delta
            B_new = max(B_new, 0.0)
            B = B_new
            z_current = z_start  # Once it fails to be feasible, restart
            
        B_history.append(B)
        
        # Stopping criteria
        if abs(delta) < 1e-6:
            break
            
        # Update delta based on number of iterations for binary search-like convergence
        if iteration > 0:
            delta *= 0.5
            
        if B >= 2.0 and B <= 3.0:
            # Try to tighten at least once
            pass
            
    return z_opt

def energy_and_grad(z):
    centers, radii = unpack(z)
    n = centers.shape[0]
    
    # Distance matrix for pairs
    D = cdist(centers, centers)
    
    # Wall energies
    E_wall = 0.0
    grad_wall = np.zeros_like(z)
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Every x is in [0,1], y in [0,1]
        # Radius must be at most x, 1-x, y, 1-y
        wall_viol = np.array([
            max(0, r - x),
            max(0, r - (1 - x)),
            max(0, r - y),
            max(0, r - (1 - y))
        ])
        E_wall += np.sum(wall_viol**2)
        
        # Gradient w.r.t. (x, y, r)
        d_x = 2 * wall_viol[0] if r > x else 0
        d_x += 2 * wall_viol[1] if r > (1 - x) else 0
        d_y = 2 * wall_viol[2] if r > y else 0
        d_y += 2 * wall_viol[3] if r > (1 - y) else 0
        d_r = 2 * wall_viol[0] if r > x else 0
        d_r += 2 * wall_viol[1] if r > (1 - x) else 0
        d_r += 2 * wall_viol[2] if r > y else 0
        d_r += 2 * wall_viol[3] if r > (1 - y) else 0
        
        grad_wall[i] = -d_x
        grad_wall[n + i] = -d_y
        grad_wall[2 * n + i] = d_r
    
    # Pairwise overlap energies (possibly less than 1e-12)
    E_pair = 0.0
    grad_pair = np.zeros_like(z)
    
    # For numerical stability of gradients, add a small value if needed
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            if dist < 1e-12:  # Avoid division by zero
                dist = 1e-12
            
            overlap = max(0.0, radii[i] + radii[j] - dist)
            E_pair += overlap**2
            
            # Non-zero only if overlapping
            if overlap > 0:
                grad_i = -2 * overlap * dx / dist
                grad_j = 2 * overlap * dx / dist
                
                # Gradient component for centers
                grad_pair[i] += grad_i
                grad_pair[n + i] += -2 * overlap * dy / dist
                grad_pair[j] += grad_j
                grad_pair[n + j] += 2 * overlap * dy / dist
                
                # For radius
                grad_pair[2 * n + i] += 2 * overlap
                grad_pair[2 * n + j] += 2 * overlap
    
    E_budget = lambda_budget * (B - np.sum(radii))**2
    grad_budget = -2 * lambda_budget * (B - np.sum(radii)) * np.ones(26) * 1.0
    
    Energy = E_wall + E_pair + E_budget
    grad = grad_wall + grad_pair
    
    # Implement budget term
    grad[2 * n : 2 * n + 26] += grad_budget
    
    return Energy, grad

def inner_optimize(z0, target_B, lambda_budget):

    B = target_B
    
    # Reduce to 78 variables
    z = z0.copy()
    
    def obj_and_grad(z):
        E, g = energy_and_grad(z)
        return E, g
    
    bounds = []
    n = 26
    for i in range(n):
        bounds.append((0, 1))         # x
        bounds.append((0, 1))         # y
        bounds.append((0, 0.5))       # r
    
    # Convert bounds to tuple of tuples
    bounds_tuple = tuple(bounds)
    
    # Minimize using L-BFGS-B
    result = minimize(
        obj_and_grad,
        z,
        method='L-BFGS-B',
        bounds=bounds_tuple,
        jac=True,
        options={'maxiter': 500, 'disp': False}
    )
    
    return result

def max_violation(centers, radii):
    n = centers.shape[0]
    # Constraint: radii <= distance to nearest wall.
    violations = []
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if r > 1e-12:
            violations.append(
                max(
                    r - x,
                    r - (1 - x),
                    r - y,
                    r - (1 - y)
                )
            )
    
    # Now check overlaps
    D = cdist(centers, centers)
    for i in range(n):
        for j in range(i + 1, n):
            d_ij = D[i, j]
            sum_r = radii[i] + radii[j]
            if sum_r > d_ij:
                violations.append(sum_r - d_ij)
    
    if len(violations) == 0:
        return 0.0
    else:
        return max(violations)

    # Ensure sufficient margin for validation

    # Clamp centers to valid region [eps, 1-eps] to avoid touching boundaries
    centers[:, 0] = np.clip(centers[:, 0], eps, 1 - eps)
    centers[:, 1] = np.clip(centers[:, 1], eps, 1 - eps)

    # Repair small overlaps carefully (if any remain due to floating point)
    D = cdist(centers, centers)
    
    for i in range(26):
        for j in range(i + 1, 26):
            d_ij = D[i, j]
            sum_r = radii[i] + radii[j]
            if sum_r > d_ij:

                # Reduce both radii

                radii[j] -= 0.5 * overlap
    
    # Ensure non-negative
    radii = np.maximum(0.0, radii)
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
