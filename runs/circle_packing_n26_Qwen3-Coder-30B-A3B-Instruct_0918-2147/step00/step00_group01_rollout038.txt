import numpy as np
from scipy.optimize import minimize

def initial_layout():
    # Hexagonal grid with 5 rows and 5 columns, plus one extra in center
    rows, cols = 5, 5
    spacing = 0.18
    radius = spacing / 2 - 0.01  # initial radius with small margin
    
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = (j + 0.5) * spacing
            y = (i + 0.5) * spacing
            # Adjust to fit inside square
            x = max(radius, min(1 - radius, x))
            y = max(radius, min(1 - radius, y))
            centers.append([x, y])
    
    # Add central circle
    centers.append([0.5, 0.5])
    
    # Make sure we have 26 circles
    centers = centers[:26]
    
    radii = [radius] * 26
    return np.array(centers), np.array(radii)

def pack_vector(centers, radii):
    n = centers.shape[0]
    z = np.zeros(3 * n)
    for i in range(n):
        z[3*i] = centers[i, 0]
        z[3*i+1] = centers[i, 1]
        z[3*i+2] = radii[i]
    return z

def unpack_vector(z):
    n = z.shape[0] // 3
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = z[3*i]
        centers[i, 1] = z[3*i+1]
        radii[i] = z[3*i+2]
    return centers, radii

def objective(z):
    _, radii = unpack_vector(z)
    return -np.sum(radii)

def constraint_boundary(z):
    centers, radii = unpack_vector(z)
    n = centers.shape[0]
    constraints = []
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        constraints.extend([
            x - r,          # x - r >= 0
            1 - x - r,      # 1 - x - r >= 0
            y - r,          # y - r >= 0
            1 - y - r       # 1 - y - r >= 0
        ])
    return np.array(constraints)

def constraint_nonoverlap(z):
    centers, radii = unpack_vector(z)
    n = centers.shape[0]
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            r_i = radii[i]
            r_j = radii[j]
            # Constraint: (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
            dist_sq = dx*dx + dy*dy
            sum_radii_sq = (r_i + r_j)*(r_i + r_j)
            constraints.append(dist_sq - sum_radii_sq)
    return np.array(constraints)

def run_packing():
    best_sum = 0
    best_centers = None
    best_radii = None
    
    for trial in range(5):
        # Generate initial layout
        centers, radii = initial_layout()
        
        # Slightly randomize for better restarts
        if trial > 0:
            np.random.seed(trial)
            centers += np.random.normal(0, 0.01, centers.shape)
            radii += np.random.normal(0, 0.005, radii.shape)
        
        # Ensure validity of initial guess
        centers = np.clip(centers, 0, 1)
        radii = np.maximum(radii, 0)
        
        # Pack into vector
        z0 = pack_vector(centers, radii)
        
        # Define bounds for all variables
        bounds = []
        n = centers.shape[0]
        for i in range(n):
            bounds.append((0, 1))  # x_i
            bounds.append((0, 1))  # y_i
            bounds.append((0, 0.5))  # r_i
        
        # Setup constraints
        constraints = []
        
        # Boundary constraints
        constraints.append({
            'type': 'ineq',
            'fun': constraint_boundary
        })
        
        # Non-overlap constraints
        constraints.append({
            'type': 'ineq',
            'fun': constraint_nonoverlap
        })
        
        # Run optimization
        result = minimize(
            objective,
            z0,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'ftol': 1e-9, 'maxiter': 2000},
            tol=1e-9
        )
        
        if not result.success:
            continue
            
        # Extract solution
        final_centers, final_radii = unpack_vector(result.x)
        
        # Fix any numerical issues
        final_radii = np.maximum(final_radii, 0)
        
        # Test validity
        is_valid, msg = validate_packing(final_centers, final_radii)
        if is_valid:
            total_sum = np.sum(final_radii)
            if total_sum > best_sum:
                best_sum = total_sum
                best_centers = final_centers.copy()
                best_radii = final_radii.copy()
    
    # If no valid solution found, try one final optimization with best known starting configuration
    if best_centers is None:
        centers, radii = initial_layout()
        z0 = pack_vector(centers, radii)
        
        bounds = []
        for i in range(n):
            bounds.append((0, 1))  # x_i
            bounds.append((0, 1))  # y_i
            bounds.append((0, 0.5))  # r_i
        
        constraints = [
            {'type': 'ineq', 'fun': constraint_boundary},
            {'type': 'ineq', 'fun': constraint_nonoverlap}
        ]
        
        result = minimize(
            objective,
            z0,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'ftol': 1e-9, 'maxiter': 2000},
            tol=1e-9
        )
        
        if result.success:
            final_centers, final_radii = unpack_vector(result.x)
            final_radii = np.maximum(final_radii, 0)
            if validate_packing(final_centers, final_radii)[0]:
                best_sum = np.sum(final_radii)
                best_centers = final_centers
                best_radii = final_radii
    
    if best_centers is not None:
        return (best_centers, best_radii, best_sum)
    else:
        # Fallback
        centers, radii = initial_layout()
        return (centers, radii, np.sum(radii))
