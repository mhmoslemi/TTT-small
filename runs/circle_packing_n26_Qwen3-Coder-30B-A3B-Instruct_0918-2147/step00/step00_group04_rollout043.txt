import numpy as np
from scipy.optimize import minimize

NUM_CIRCLES = 26
NUM_RESTARTS = 5

def hex_lattice_points(n_rows, spacing):
    # Generate points for a hexagonal lattice
    points = []
    for i in range(n_rows):
        y = i * spacing * np.sqrt(3) / 2
        if y > 1:
            break
        offset = (i % 2) * spacing / 2
        n_cols = int((1.0 + spacing / 2) / spacing)
        for j in range(n_cols):
            x = offset + j * spacing
            if x > 1:
                break
            points.append([x, y])
    return np.array(points)

def initial_guess(seed):
    np.random.seed(seed)
    # Use a hexagonal lattice with spacing that fits ~26 circles
    spacing = 0.2
    points = hex_lattice_points(10, spacing)
    # Trim to exactly 26 points (or pad)
    if len(points) < NUM_CIRCLES:
        # Add some random points to ensure 26
        extra_points = NUM_CIRCLES - len(points)
        points = np.vstack([points, np.random.rand(extra_points, 2)])
    else:
        points = points[:NUM_CIRCLES]

    # Add jitter
    jitter = np.random.uniform(-0.02, 0.02, (NUM_CIRCLES, 2))
    points += jitter

    # Clip to valid positions
    points = np.clip(points, 0, 1)

    # Initial small radii
    radii = np.full(NUM_CIRCLES, 0.02)

    # Pack into variables
    vars0 = np.zeros(3 * NUM_CIRCLES)
    vars0[0::3] = points[:, 0]
    vars0[1::3] = points[:, 1]
    vars0[2::3] = radii
    return vars0

def objective(vars):
    return -np.sum(vars[2::3])

def constraint_boundary(vars):
    # For each circle, return [x-r, 1-x-r, y-r, 1-y-r]
    num_circles = NUM_CIRCLES
    result = np.zeros(num_circles * 4)
    for i in range(num_circles):
        x, y, r = vars[3*i], vars[3*i+1], vars[3*i+2]
        result[4*i] = x - r   # x - r >= 0
        result[4*i+1] = 1 - x - r  # 1 - x - r >= 0
        result[4*i+2] = y - r   # y - r >= 0
        result[4*i+3] = 1 - y - r  # 1 - y - r >= 0
    return result

def constraint_nonoverlap(vars):
    # Compute pairwise non-overlap margin for all i < j
    # Return [dist(i,j) - (r_i + r_j)]
    num_circles = NUM_CIRCLES
    # All distances and radii
    centers = vars[0::3].reshape(-1, 1)  # (n,1)
    centers_y = vars[1::3].reshape(-1, 1)
    radii = vars[2::3].reshape(-1, 1)

    # Compute pairwise differences in x & y
    dx = centers.T - centers  # (n,n)
    dy = centers_y.T - centers_y
    # Compute distances
    dist = np.sqrt(dx**2 + dy**2)
    # Radii sums
    sum_radii = radii.T + radii  # (n,n)
    # Margins
    margins = dist - sum_radii

    # Extract upper triangular part (i < j)
    triu_indices = np.triu_indices(num_circles, k=1)
    return margins[triu_indices]

def postprocess(vars):
    # Apply tight boundary correction for boundary violations and small overlaps
    # Reconstruct centers and radii
    centers = np.column_stack((vars[0::3], vars[1::3]))
    radii = vars[2::3]

    # Enforce boundary constraints
    for i in range(NUM_CIRCLES):
        x, y, r = centers[i][0], centers[i][1], radii[i]
        # Shift center to nearest valid position if needed
        x = np.clip(x, r, 1 - r)
        y = np.clip(y, r, 1 - r)
        centers[i] = [x, y]

    # Deal with overlap violations by touching tops and whatnot
    for i in range(NUM_CIRCLES):
        for j in range(i+1, NUM_CIRCLES):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist = np.sqrt(dx**2 + dy**2)
            r_i, r_j = radii[i], radii[j]
            if dist < r_i + r_j - 1e-12:
                # Adjust radii to prevent overlap
                overlap = (r_i + r_j) - dist
                # Reduce larger radius
                if r_i >= r_j:
                    radii[i] -= overlap / 2
                else:
                    radii[j] -= overlap / 2
                # Re-adjust extra radius
                if radii[i] < 0:
                    radii[i] = 0
                if radii[j] < 0:
                    radii[j] = 0

    # Return corrected variables
    vars_corr = np.zeros(3 * NUM_CIRCLES)
    vars_corr[0::3] = centers[:, 0]  # x
    vars_corr[1::3] = centers[:, 1]  # y
    vars_corr[2::3] = radii  # r
    return vars_corr

def run_packing():
    best_sum = -np.inf
    best_vars = None

    # Set bounds: [0,1] for x,y and [0,0.5] for r
    bounds = []
    for i in range(NUM_CIRCLES):
        bounds.extend([(0, 1), (0, 1), (0, 0.5)])
    
    constraints = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_nonoverlap}
    ]

    for seed in range(NUM_RESTARTS):
        vars0 = initial_guess(seed)

        # Optimize
        res = minimize(
            objective,
            vars0,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 2000, 'ftol': 1e-6}
        )

        if not res.success:
            continue

        # Postprocess
        vars_corr = postprocess(res.x)
        s = np.sum(vars_corr[2::3])

        if s > best_sum:
            best_sum = s
            best_vars = vars_corr.copy()

    if best_vars is None:
        raise RuntimeError("No valid configuration found.")

    # Reconstruct centers and radii
    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii = best_vars[2::3]
    return centers, radii, float(best_sum)
