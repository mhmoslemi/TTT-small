import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
from sobol_seq import i4_sobol_generate

def initial_configuration(n=26):
    # Generate Sobol sequence for initial centres
    seeds = i4_sobol_generate(2, n)
    
    # Ensure points are within the unit square
    seeds = np.clip(seeds, 0, 1)
    
    # Compute distances for initial radius estimation
    dists = cdist(seeds, seeds)
    np.fill_diagonal(dists, np.inf)
    min_dists = np.min(dists, axis=1)
    
    # Compute initial radii
    radii = np.minimum(
        seeds[:, 0], 1 - seeds[:, 0]
    )
    radii = np.minimum(
        radii, seeds[:, 1]
    )
    radii = np.minimum(
        radii, 1 - seeds[:, 1]
    )
    radii = np.minimum(radii, min_dists / 2.0)
    
    # Clip radii to reasonable limits
    radii = np.maximum(radii, 1e-6)
    radii = np.minimum(radii, 0.5)
    
    return seeds, radii

def pack(centers, radii):
    # Flatten centers and radii into a single array
    return np.hstack((centers.flatten(), radii))

def unpack(z):
    # Unpack from flat array to centers and radii
    n = len(z) // 3
    centers = z[:2*n].reshape((n, 2))
    radii = z[2*n:]
    return centers, radii

def energy_and_grad(z):
    centers, radii = unpack(z)
    
    # Precompute distance matrix
    D = cdist(centers, centers)
    # To prevent division by zero in gradient
    D = np.maximum(D, 1e-12)
    
    # Wall energy components
    wall_energy = 0.0
    wall_grad = np.zeros_like(z)
    
    # Wall penalties (x, y coordinates)
    for i in range(len(centers)):
        x, y = centers[i]
        r = radii[i]
        
        # Wall corrections
        dx_l = max(0, r - x)
        dx_r = max(0, r - (1 - x))
        dy_b = max(0, r - y)
        dy_t = max(0, r - (1 - y))
        
        wall_energy += dx_l**2 + dx_r**2 + dy_b**2 + dy_t**2
        
        # Gradients
        wall_grad[2*i] += -2 * dx_l + 2 * dx_r  # x coord
        wall_grad[2*i + 1] += -2 * dy_b + 2 * dy_t  # y coord
        wall_grad[26 + i] += 2 * dx_l + 2 * dx_r + 2 * dy_b + 2 * dy_t  # r coord
    
    # Pairwise overlap energy
    pair_energy = 0.0
    pair_grad = np.zeros_like(z)
    
    lambda_pair = 100.0
    bc = len(radii)
    
    for i in range(bc):
        for j in range(i+1, bc):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            d = np.sqrt(dx * dx + dy * dy)
            
            overlap = max(0, radii[i] + radii[j] - d)
            
            pair_energy += overlap**2
            
            if overlap > 0:
                # Gradients
                grad = 2 * overlap
                
                # d_r_i = grad * (1)
                # d_r_j = grad * (1)
                pair_grad[26 + i] += grad
                pair_grad[26 + j] += grad
                
                # d_c_i = grad * (c_i - c_j) / d
                if d > 1e-12:
                    dx /= d
                    dy /= d
                    pair_grad[2*i]   -= grad * dx
                    pair_grad[2*i+1] -= grad * dy
                    pair_grad[2*j]   += grad * dx
                    pair_grad[2*j+1] += grad * dy
    
    # Budget constraint
    total_radius = np.sum(radii)
    budget = 3.0  # Start with generous budget
    lambda_budget = 1000.0
    budget_energy = lambda_budget * (budget - total_radius)**2
    budget_grad = np.zeros_like(z)
    for i in range(len(radii)):
        budget_grad[26 + i] -= 2 * lambda_budget * (budget - total_radius)

    # Total energy and gradient
    total_energy = wall_energy + lambda_pair * pair_energy + budget_energy
    total_grad = wall_grad + lambda_pair * pair_grad + budget_grad
    
    return total_energy, total_grad

def inner_optimize(z0, B, lambda_budget):
    res = minimize(
        fun=lambda z: energy_and_grad(z)[0],
        x0=z0,
        method='L-BFGS-B',
        jac=lambda z: energy_and_grad(z)[1],
        bounds=[(0, 1)] * 52 + [(0, 0.5)] * 26,
        options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12}
    )
    
    # Ensure optimized radius is within limits
    centers, radii = unpack(res.x)
    radii = np.clip(radii, 0, 0.5)
    z_opt = pack(centers, radii)
    return z_opt

def max_feasible_budget(z_start):
    z = z_start.copy()
    
    # Budget adaptation parameters
    B = 3.0  # initial guess
    lambda_budget = 1000.0
    step_size = 5e-4
    min_step = 1e-6
    max_iterations = 40
    
    last_feasible_z = z.copy()
    last_B = B
    for _ in range(max_iterations):
        z_test = inner_optimize(z, B, lambda_budget)
        centers, radii = unpack(z_test)
        max_overlap = 0.0
        for i in range(len(centers)):
            for j in range(i+1, len(centers)):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                d = np.sqrt(dx * dx + dy * dy)
                overlap = max(0, radii[i] + radii[j] - d)
                max_overlap = max(max_overlap, overlap)
        max_wall_violation = np.max([
            np.max([radii[i]-centers[i][0],
                    radii[i]-(1-centers[i][0]),
                    radii[i]-centers[i][1],
                    radii[i]-(1-centers[i][1])])
            for i in range(len(centers))])
        
        # Still slightly infeasible
        if max_overlap > 1e-13 or max_wall_violation > 1e-13:
            B -= step_size
            if B < 0.0:
                break
        else:
            last_feasible_z = z_test.copy()
            last_B = B
            B += step_size
            if step_size < min_step:
                break
    return last_feasible_z

def postprocess(z):
    eps = 5e-10
    centers, radii = unpack(z)
    
    # Shrink radii
    radii = np.maximum(0, radii - eps)
    
    # Clamp centers
    centers = np.clip(centers, eps, 1 - eps)
    
    # Repair overlaps
    D = cdist(centers, centers)
    changed = True
    while changed:
        changed = False
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                d = D[i, j]
                if d < radii[i] + radii[j] - 2 * eps:
                    delta = (radii[i] + radii[j] - d) / 2.0
                    radii[i] = max(0, radii[i] - delta - eps)
                    radii[j] = max(0, radii[j] - delta - eps)
                    changed = True
    
    # Ensure final clipping
    radii = np.minimum(radii, 0.5)
    # Final sum
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii

def run_packing():
    # Initial configuration
    centers, radii = initial_configuration(26)
    z_start = pack(centers, radii)
    
    # Find feasible configuration with max budget
    z_feasible = max_feasible_budget(z_start)
    
    # Clean up and post-process
    centers, radii, sum_radii = postprocess(z_feasible)
    
    return centers, radii, sum_radii
