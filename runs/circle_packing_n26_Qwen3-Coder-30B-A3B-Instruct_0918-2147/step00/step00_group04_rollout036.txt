import numpy as np
import cvxpy as cp
from scipy.spatial.distance import pdist, squareform

def build_socp_problem(n):
    r = cp.Variable(n, nonneg=True)
    # Parameters for distances
    dist_params = []
    for i in range(n):
        row = []
        for j in range(i+1, n):
            param = cp.Parameter(nonneg=True)
            row.append(param)
        dist_params.append(row)
    
    # Constraints
    constraints = []
    # Boundary constraints
    for i in range(n):
        constraints.append(r[i] <= 1)  # just to make constraint matrix match
        constraints.append(r[i] <= 1)  # just to make constraint matrix match
    
    # Non-overlap constraints
    for i in range(n):
        for j in range(i+1, n):
            idx = i * (n-1) - (i * (i+1)) // 2 + (j - i - 1)
            constraints.append(r[i] + r[j] <= dist_params[i][j-i-1])
    
    # Objective
    objective = cp.Maximize(cp.sum(r))
    
    # Create problem
    prob = cp.Problem(objective, constraints)
    return prob, r, dist_params

def solve_radii(centres, prob, r, dist_params):
    n = centres.shape[0]
    
    # Compute pairwise distances
    distances = squareform(pdist(centres, metric='euclidean'))
    
    # Update parameters
    for i in range(n):
        for j in range(i+1, n):
            dist_params[i][j-i-1].value = distances[i, j]
    
    # Solve
    try:
        prob.solve(solver=cp.ECOS, abstol=1e-10, reltol=1e-10)
    except:
        prob.solve(solver=cp.ECOS, abstol=1e-8, reltol=1e-8)
    
    radii = r.value
    if radii is None:
        raise RuntimeError("Failed to compute radii")
    
    return radii, None

def energy(centres, prob, r, dist_params):
    radii, _ = solve_radii(centres, prob, r, dist_params)
    return -np.sum(radii)

def random_move(state, rng, boundary_bias=True):
    new_state = state.copy()
    n = len(state) // 2
    move_type = rng.choice([0, 1, 2, 3], p=[0.6, 0.2, 0.15, 0.05])
    
    if move_type == 0:  # single-centre jitter
        idx = rng.integers(n)
        dx = rng.normal(0, 0.02)
        dy = rng.normal(0, 0.02)
        new_state[2*idx] = np.clip(new_state[2*idx] + dx, 0, 1)
        new_state[2*idx+1] = np.clip(new_state[2*idx+1] + dy, 0, 1)
    
    elif move_type == 1:  # cluster jitter
        # pick some nearby centers based on current distances
        centers = state.reshape(-1, 2)
        dists = squareform(pdist(centers, metric='euclidean'))
        # make faster by using upper triangle
        indices = np.arange(n)
        candidates = rng.choice(indices, min(5, n), replace=False)
        chosen_dist = np.mean(dists[np.ix_(candidates, candidates)])
        use_to = []
        for i in candidates:
            for j in candidates:
                if i < j and dists[i, j] < chosen_dist:
                    use_to += [i, j]
        use_to = list(set(use_to))
        if len(use_to) < 2:
            use_to = candidates[:2]
            
        dx = rng.normal(0, 0.03)
        dy = rng.normal(0, 0.03)
        for idx in use_to:
            new_state[2*idx] = np.clip(new_state[2*idx] + dx, 0, 1)
            new_state[2*idx+1] = np.clip(new_state[2*idx+1] + dy, 0, 1)
            
    elif move_type == 2:  # edge push
        idx = rng.integers(n)
        x = new_state[2*idx]
        y = new_state[2*idx+1]
        # move towards the nearest side
        dx = -np.sign(x - 0.5)*0.02
        dy = -np.sign(y - 0.5)*0.02
        new_state[2*idx] = np.clip(x + dx, 0, 1)
        new_state[2*idx+1] = np.clip(y + dy, 0, 1)
        
    else:  # swap order
        # permute order of centers, but use same centers
        # just reshuffling the ones
        order = np.arange(n)
        rng.shuffle(order)
        new_state = np.zeros_like(new_state)
        for i, old_idx in enumerate(order):
            new_state[2*i] = state[2*old_idx]
            new_state[2*i+1] = state[2*old_idx+1]
        
    return new_state

def anneal_chain(seed, prob, r, dist_params, rng):
    # Initialize with seed configuration
    centers = np.random.default_rng(seed).random((26, 2))
    centres_flat = centers.flatten()
    
    best_energy = energy(centers, prob, r, dist_params)
    best_state = centres_flat.copy()
    
    T = 0.05
    alpha = 0.995
    N_iter = 150000
    
    for i in range(N_iter):
        new_state = random_move(centres_flat, rng)
        new_centers = new_state.reshape(-1, 2)
        new_energy = energy(new_centers, prob, r, dist_params)
        
        delta = new_energy - best_energy
        
        if delta < 0 or np.exp(-delta / T) > rng.random():
            centres_flat = new_state.copy()
            if new_energy < best_energy:
                best_energy = new_energy
                best_state = centres_flat.copy()
                
        if i % 1000 == 0:
            T *= alpha
                
    final_centers = best_state.reshape(-1, 2)
    return final_centers

def finalise(centres, prob, r, dist_params):

    try:
        radii, _ = solve_radii(centres, prob, r, dist_params)
