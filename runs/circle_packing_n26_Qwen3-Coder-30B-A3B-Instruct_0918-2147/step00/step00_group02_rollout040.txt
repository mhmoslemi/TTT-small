import numpy as np
from scipy.stats.qmc import Sobol
from scipy.spatial.distance import cdist

def initialize_seeds(n):
    sobol = Sobol(d=2, seed=42)
    points = sobol.random(n)
    margin = 0.02
    points = margin + (1 - 2*margin) * points
    return points

def compute_feasible_radii(centers):
    n = centers.shape[0]
    # Compute pairwise distances
    D = cdist(centers, centers)
    # Set diagonal to large value so it's not selected as minimum
    np.fill_diagonal(D, np.inf)
    # Nearest neighbor distances
    nn = np.min(D, axis=1)
    # Wall distances
    wall_dists = np.array([
        centers[:, 0],
        1 - centers[:, 0],
        centers[:, 1],
        1 - centers[:, 1]
    ]).T
    min_wall = np.min(wall_dists, axis=1)
    # Feasible radii: minimum of wall distance and half nearest neighbor distance
    radii = np.minimum(min_wall, 0.5 * nn)
    return radii

def redistribute_radii(radii, slack, alpha):
    total_slack = np.sum(slack)
    # Increase wall-limited radii
    new_radii = radii.copy()
    wall_limited = slack > 1e-12
    if np.sum(wall_limited) == 0:
        return new_radii
    # Add alpha * slack to wall-limited radii
    delta = alpha * slack
    new_radii[wall_limited] += delta[wall_limited]
    # Subtract from neighbour-limited radii proportionally to current radii
    # To avoid negative, normalize subtraction by total current non-zero radii
    nei_limited = ~wall_limited
    if np.sum(nei_limited) > 0:
        total_nonzero = np.sum(radii[nei_limited])
        if total_nonzero > 1e-12:
            reduction_ratio = delta[nei_limited] / total_nonzero
            new_radii[nei_limited] -= reduction_ratio * radii[nei_limited]
            new_radii[new_radii < 0] = 0.0
    return new_radii

def push_away(centers, radii, beta, eps):
    n = centers.shape[0]
    new_centers = centers.copy()
    # Compute pairwise distances
    D = cdist(centers, centers)
    for i in range(n):
        for j in range(n):
            if i == j:
                continue
            dist = D[i, j]
            r_i = radii[i]
            r_j = radii[j]
            if dist < r_i + r_j + eps:
                delta = (r_i + r_j + eps - dist)
                v_ij = centers[i] - centers[j]
                v_ij_norm = np.linalg.norm(v_ij)
                if v_ij_norm > eps:
                    push_amount = beta * delta * v_ij / v_ij_norm
                    new_centers[i] += push_amount
    # Clamp to valid region
    new_centers[:, 0] = np.clip(new_centers[:, 0], eps, 1.0 - eps)
    new_centers[:, 1] = np.clip(new_centers[:, 1], eps, 1.0 - eps)
    return new_centers

def iterate(centers, radii, alpha, beta, max_iter):
    for _ in range(max_iter):
        old_sum = np.sum(radii)
        # Compute slack
        D = cdist(centers, centers)
        np.fill_diagonal(D, np.inf)
        nn = np.min(D, axis=1)
        wall_dists = np.array([
            centers[:, 0],
            1 - centers[:, 0],
            centers[:, 1],
            1 - centers[:, 1]
        ]).T
        min_wall = np.min(wall_dists, axis=1)
        # Update radii with feasible constraint
        rad = np.minimum(min_wall, 0.5 * nn)
        slack = 0.5 * nn - rad
        # Redistribution
        new_radii = redistribute_radii(radii, slack, alpha)
        # Push away
        new_centers = push_away(centers, new_radii, beta, 1e-8)
        # Re-compute radii after push
        centers = new_centers
        radii = compute_feasible_radii(centers)
        new_sum = np.sum(radii)
        if new_sum - old_sum < 1e-5:
            break
    return centers, radii

def repair(centers, radii, eps_final):
    # Shrink radii slightly
    radii = np.maximum(0, radii - eps_final)
    # Re-clamp centers
    centers = np.clip(centers, eps_final, 1.0 - eps_final)
    # Optional repair loop
    # Find nearest neighbors and resolve overlaps
    D = cdist(centers, centers)
    np.fill_diagonal(D, np.inf)
    for i in range(centers.shape[0]):
        for j in range(i+1, centers.shape[0]):
            dist = D[i, j]
            r_i = radii[i]
            r_j = radii[j]
            if dist < r_i + r_j:
                # Reduce larger radius
                overlap = r_i + r_j - dist
                reduce_by = overlap/2 + eps_final
                if radii[i] >= radii[j]:
                    radii[i] -= reduce_by
                else:
                    radii[j] -= reduce_by
    return centers, radii

def run_packing():
    # Initialize
    centers = initialize_seeds(26)
    radii = compute_feasible_radii(centers)
    # Iterate to improve packing
    centers, radii = iterate(centers, radii, alpha=0.4, beta=0.3, max_iter=200)
    # Final repair
    eps_final = 5e-10
    centers, radii = repair(centers, radii, eps_final)
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
