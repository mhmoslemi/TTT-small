import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def initial_configuration(n=26):
    # Generate Sobol sequence using low discrepancy sequence (first 26 points)
    # We use a simple Halton-like approach for 2D with base 2 and 3
    def halton_sequence(dim, n):
        result = np.zeros((n, dim))
        primes = [2, 3] if dim >= 2 else [2]
        for i in range(n):
            for d in range(dim):
                base = primes[d] if d < len(primes) else primes[-1]
                f = 1.0
                result[i, d] = 0.0
                while i > 0:
                    f /= base
                    result[i, d] += (i % base) * f
                    i //= base
        return result

    points = halton_sequence(2, n)
    centers = points
    radii = np.zeros(n)
    for i in range(n):
        x, y = centers[i]
        min_dist_wall = min(x, 1-x, y, 1-y)
        dist_to_neighbors = []
        for j in range(n):
            if i != j:
                d = np.sqrt((centers[i,0]-centers[j,0])**2 + (centers[i,1]-centers[j,1])**2)
                dist_to_neighbors.append(d)
        if len(dist_to_neighbors) > 0:
            min_dist_neighbor = min(dist_to_neighbors)
            radii[i] = min(min_dist_wall, min_dist_neighbor / 2.0)
        else:
            radii[i] = min_dist_wall
    return centers, radii

def unpack(z):
    n = len(z) // 3
    centers = z[:2*n].reshape((n, 2))
    radii = z[2*n:]
    return centers, radii

def pack(centers, radii):
    return np.concatenate([centers.ravel(), radii])

def energy_and_grad(z, B=2.636, lambda_budget=1e3):
    n = len(z) // 3
    centers, radii = unpack(z)
    
    # Wall energies and gradients
    E_wall = 0
    dE_dr_wall = np.zeros(n)
    dE_dc_wall = np.zeros((n, 2))
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        dxl = max(0, r - x)
        dxr = max(0, r - (1 - x))
        dyl = max(0, r - y)
        dyr = max(0, r - (1 - y))
        E_wall += dxl**2 + dxr**2 + dyl**2 + dyr**2
        dE_dr_wall[i] += 2*dxl - 2*dxr + 2*dyl - 2*dyr
        dE_dc_wall[i, 0] += 2*dxl - 2*dxr
        dE_dc_wall[i, 1] += 2*dyl - 2*dyr

    # Pairwise energies and gradients
    dist_matrix = cdist(centers, centers)
    E_pair = 0
    dE_dr_pair = np.zeros(n)
    dE_dc_pair = np.zeros((n, 2))

    for i in range(n):
        for j in range(i+1, n):
            d = dist_matrix[i,j]
            r_sum = radii[i] + radii[j]
            if r_sum > d:
                overlap = r_sum - d
                E_pair += overlap**2
                
                # Gradients
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                # Normalized direction vector
                if d > 1e-12:
                    ux = dx / d
                    uy = dy / d
                else:
                    ux = 0
                    uy = 0
                dE_dr_pair[i] += 2*overlap
                dE_dr_pair[j] += 2*overlap
                dE_dc_pair[i, 0] -= 2*overlap * ux
                dE_dc_pair[i, 1] -= 2*overlap * uy
                dE_dc_pair[j, 0] += 2*overlap * ux
                dE_dc_pair[j, 1] += 2*overlap * uy

    # Budget penalty
    total_r = np.sum(radii)
    E_budget = lambda_budget * (B - total_r)**2
    dE_dr_budget = -2 * lambda_budget * (B - total_r)

    E = E_wall + E_pair + E_budget
    dE_dr = dE_dr_wall + dE_dr_pair + dE_dr_budget

    # Full gradient
    grad = np.zeros_like(z)
    grad[:2*n] = dE_dc_wall.ravel()
    grad[2*n:] = dE_dr

    return E, grad

def inner_optimize(z0, B, lambda_budget=1e3):
    res = minimize(
        lambda z: energy_and_grad(z, B, lambda_budget)[0],
        z0,
        method='L-BFGS-B',
        jac=lambda z: energy_and_grad(z, B, lambda_budget)[1],
        bounds=[(0, 1)] * (2*len(z0)//3) + [(0, 0.5)] * (len(z0)//3),
        options={'maxiter': 300}
    )
    return res.x

def max_feasible_budget(z_start, B0=3.0, step=5e-4, min_step=1e-6):
    B = B0
    while step > min_step:
        z = inner_optimize(z_start, B)
        centers, radii = unpack(z)
        total_r = np.sum(radii)
        
        # Check if configuration is feasible by:
        # 1. radii being within [0, 0.5]
        # 2. Distances from wall are >= radii
        # 3. No pairwise overlaps

        feasible = True
        for i in range(len(radii)):
            x, y = centers[i]
            r = radii[i]
            if (r > x or r > 1-x or r > y or r > 1-y):
                feasible = False
                break
        
        if feasible:
            # Verify that no overlaps exist
            dist_m = cdist(centers, centers)
            for i in range(len(radii)):
                for j in range(i+1, len(radii)):
                    if dist_m[i,j] < radii[i] + radii[j]:
                        feasible = False
                        break
                if not feasible:
                    break
        
        # Adapt B
        if feasible:
            B += step
            z_start = z.copy()  # advance to new feasible state
        else:
            B -= step
    return z_start, B

def postprocess(z):
    centers, radii = unpack(z)
    
    # Shrink radii by a small amount for safety
    eps = 5e-10
    radii = np.clip(radii - eps, 0, None)
    centers[:, 0] = np.clip(centers[:, 0], eps, 1-eps)
    centers[:, 1] = np.clip(centers[:, 1], eps, 1-eps)
    
    # One pass repair (in case of marginal violations)
    dist_m = cdist(centers, centers)
    max_iterations = 10
    for _ in range(max_iterations):
        overlap_violated = False
        for i in range(len(centers)):
            for j in range(i+1, len(centers)):
                d = dist_m[i,j]
                if d < radii[i] + radii[j]:
                    overlap_violated = True
                    r_total = radii[i] + radii[j]
                    shortfall = r_total - d
                    # Reduce larger radii
                    if radii[i] > radii[j]:
                        radii[i] -= shortfall/2 + eps
                    else:
                        radii[j] -= shortfall/2 + eps
        if not overlap_violated:
            break

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii

def run_packing():
    centers, radii = initial_configuration(26)
    z_start = pack(centers, radii)
    z_opt, final_B = max_feasible_budget(z_start)
    centers_final, radii_final, sum_radii = postprocess(z_opt)
    return centers_final, radii_final, sum_radii
