import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import math

def generate_lattice(s):
    """Generate hexagonal lattice of candidate center points inside unit square."""
    sqrt3 = math.sqrt(3)
    # Rows (even and odd spaced)
    rows = []
    y = s / 2
    r = 0
    while y < 1 - s / 2:
        row = []
        x = s / 2
        if r % 2 == 1:
            x += s / 2
        while x < 1 - s / 2:
            row.append([x, y])
            x += s
        if len(row) > 0:
            rows.append(np.array(row))
        y += s * sqrt3 / 2
        r += 1

    points = []
    for i, row in enumerate(rows):
        if i % 2 == 0:
            points.extend(row)
        else:
            points.extend(row)
    return np.array(points)

def build_milp(P):
    """Build the MILP model for circle packing with 26 circles."""
    K = len(P)
    # Variables
    z = cp.Variable(K, boolean=True)
    r = cp.Variable(K, nonneg=True)
    # wall distances
    wL = P[:, 0]
    wR = 1 - P[:, 0]
    wB = P[:, 1]
    wT = 1 - P[:, 1]
    w = np.minimum(np.minimum(wL, wR), np.minimum(wB, wT))
    # distance matrix (precomputed)
    D = np.zeros((K, K))
    for i in range(K):
        for j in range(i+1, K):
            dist = np.linalg.norm(P[i] - P[j])
            D[i, j] = D[j, i] = dist

    # Constraints
    constraints = [
        cp.sum(z) == 26,
    ]

    # Link to wall boundaries
    for k in range(K):
        constraints.append(r[k] <= w[k] * z[k])

    # Non-overlap constraints (Big-M)
    M = 1.0  # max possible distance
    for i in range(K):
        for j in range(i + 1, K):
            if D[i, j] > 0:  # valid pair
                # d_ij >= r_i + r_j - M*(2 - z_i - z_j)
                # or r_i + r_j <= d_ij + M*(2 - z_i - z_j)
                # The big-M formulation simplifies to:
                # if both selected: d_ij >= r_i + r_j
                # if one not selected: 0 >= r_i + r_j - M (fake "if" by forcing r=0)
                constraints.append(r[i] + r[j] <= D[i, j] + M * (2 - z[i] - z[j]))

    obj = cp.Maximize(cp.sum(r))

    def solve_fixed_z(z_fixed):
        # Set fixed binary values
        z.value = z_fixed
        # Solve LP relaxation
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            return None, None
        if prob.status == cp.OPTIMAL:
            return z.value, r.value
        return None, None

    return solve_fixed_z

def branch_and_bound(milp_builder, node_limit=20000):
    """Custom branch-and-bound solver for the MILP."""
    import heapq
    best_obj = -np.inf
    best_z = None
    best_r = None
    heap = [(0, 0, np.zeros(120, dtype=int), np.zeros(120))]  # (-bound, ID, z_fixed, r_fixed)
    node_id = 0
    nodes_explored = 0

    # While loop limit
    while heap and nodes_explored < node_limit:
        neg_bound, _, z_current, _ = heapq.heappop(heap)
        bound = -neg_bound
        if bound <= best_obj:
            continue
        # Attempt to fix a free variable
        free_indices = np.where(z_current == -1)[0]
        if len(free_indices) == 0:
            # All fixed
            z_fixed = z_current.astype(bool)
            _, r_sol = milp_builder(z_fixed)
            if r_sol is not None:
                obj = np.sum(r_sol)
                if obj > best_obj:
                    best_obj = obj
                    best_r = r_sol
                    best_z = z_fixed
            # We can stop early since nodes are ordered by bound, and root node is processed already
            continue
        # Pick the index most away from 0.5 (furthest from fractional)
        val = np.array([0.5 if z_current[i] == -1 else z_current[i] for i in free_indices])
        idx = np.argmin(np.abs(val - 0.5))
        i = free_indices[idx]
        # Branch: fix to 0 and 1
        z_partial = z_current.copy()
        z_partial[z_partial == -1] = 0  # unfixed to 0
        z_partial[i] = 0
        z_partial2 = z_current.copy()
        z_partial2[z_partial2 == -1] = 0
        z_partial2[i] = 1
        # Check how many ones are already set
        ones_count1 = np.sum(z_partial)
        ones_count2 = np.sum(z_partial2)
        # If the count is beyond 26 or below 26, might be unfair to check feasibility now.
        # Recheck after inference
        for toggle in [0, 1]:
            z_try = z_partial if toggle == 0 else z_partial2
            _, r_try = milp_builder(z_try.astype(bool))
            if r_try is not None:
                obj = np.sum(r_try)
                if obj > best_obj:
                    best_obj = obj
                    best_r = r_try
                    best_z = z_try.astype(bool)
                neg_bound_try = -obj
                heapq.heappush(heap, (neg_bound_try, node_id, z_try.astype(int), r_try))
                node_id += 1
        nodes_explored += 2

    # Fallback / base case if B&B found nothing:
    if best_z is None:
        # Assign to any arbitrary 26 points, all with radii 0.1
        sel_indices = np.arange(26)
        z = np.zeros(120, dtype=bool)
        z[sel_indices] = 1
        r = np.full(120, 0.1)
        return z, r

    return best_z, best_r

def refine_solution(centers, radii):
    """Apply local continuous refinement using SLSQP."""
    n = len(centers)
    def objective(x):
        # x is [c1_x, c1_y, r1, c2_x, c2_y, r2, ..., c_n_x, c_n_y, r_n]
        total = 0
        for i in range(n):
            total += x[3 * i + 2]  # radius
        return -total  # maximize by minimizing negative

    def constraint_eq(x):
        # No equality constraints
        return np.array([])

    def constraint_ineq(x):
        # Non-overlap and boundary constraints
        ineq = []
        # Keep each radii > 0 by a small epsilon
        for i in range(n):
            ineq.append(x[3 * i + 2])  # radius must be positive

        for i in range(n):
            ci = x[3 * i: 3 * i + 2]
            ri = x[3 * i + 2]
            # Wall constraints
            ineq.append(ci[0] - ri)  # left
            ineq.append(1 - ci[0] - ri)  # right
            ineq.append(ci[1] - ri)  # bottom
            ineq.append(1 - ci[1] - ri)  # top

            for j in range(i + 1, n):
                # distance >= ri + rj
                cij = x[3 * j: 3 * j + 2]
                rij = x[3 * j + 2]
                dist_sq = np.sum((ci - cij) ** 2)
                ineq.append(ri + rij - np.sqrt(dist_sq))
        return np.array(ineq)  # be non-negative

    # Initial guess
    x0 = []
    for i in range(n):
        x0.append(centers[i, 0])
        x0.append(centers[i, 1])
        x0.append(radii[i])
    x0 = np.array(x0)

    # Bound each parameter
    bounds = []
    for i in range(n):
        bounds.extend([(0, 1), (0, 1), (0, 1)])  # bounds for x, y, r

    # Optimize
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': constraint_ineq},
                   options={'ftol': 1e-9, 'maxiter': 2000})

    if not res.success:
        # Try fallback solution
        x0 = x0[:3*n].reshape(n, 3)
        new_c = x0[:, :2]
        new_r = np.clip(x0[:, 2], 0, 1)
        return new_c, new_r, np.sum(new_r)

    x_opt = res.x.reshape(n, 3)
    new_centers = x_opt[:, :2]
    new_radii = np.clip(x_opt[:, 2], 0, 1)
    sum_radii = np.sum(new_radii)

    return new_centers, new_radii, sum_radii

def run_packing():
    # Parameters
    s = 0.13
    P = generate_lattice(s)
    # Build the model
    milp_solver = build_milp(P)
    # Branch and Bound optimization
    selected_z, selected_r = branch_and_bound(milp_solver, node_limit=20000)
    # Select centers and radii
    centers = P[selected_z]
    radii = selected_r[selected_z]

    # Refine with continuous optimization
    centers_ref, radii_ref, total = refine_solution(centers, radii)

    return centers_ref, radii_ref, total
