def init_candidate_grid(R, Δ):
    """Generate candidate (center, radius) pairs."""
    candidates = []
    for r in R:
        spacing = r * np.sqrt(2)
        x_coords = np.arange(r, 1 - r + spacing/2, spacing)
        y_coords = np.arange(r, 1 - r + spacing/2, spacing)
        for x in x_coords:
            for y in y_coords:
                candidates.append((x, y, r))
    return candidates

def build_overlap_matrix(candidates):
    """Build overlap matrix of size (n, n)"""
    n = len(candidates)
    overlaps = np.zeros((n, n), dtype=bool)
    centers = np.array([[c[0], c[1]] for c in candidates])
    radii = np.array([c[2] for c in candidates])
    distances = cdist(centers, centers)
    for i in range(n):
        for j in range(i+1, n):
            if distances[i, j] < radii[i] + radii[j]:
                overlaps[i, j] = True
                overlaps[j, i] = True
    return overlaps

def max_radii_fixed_centres(centers, radii_initial):
    """Project radii to maximum feasible without changing centers."""
    n = centers.shape[0]
    radii = radii_initial.copy()

    for i in range(n):
        x, y = centers[i]
        r_max = min(x, 1-x, y, 1-y)

        # Try to find the maximum radius possible
        # We'll do binary search but very simple now
        r_lower = 0.0
        r_upper = r_max
        step = 1e-6
        while r_upper - r_lower > step:
            r_mid = (r_lower + r_upper) / 2
            # Check if possible with this radius
            valid = True
            for j in range(n):
                if i == j:
                    continue
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < r_mid + radii[j]:
                    valid = False
                    break
            if valid:
                r_lower = r_mid
            else:
                r_upper = r_mid
        radii[i] = r_lower

    return radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
