import numpy as np
import cvxpy as cp
from scipy.spatial.distance import cdist

def build_socp_problem(n):
    r = cp.Variable(n, nonneg=True)
    dist_params = [cp.Parameter(1, value=0.0) for _ in range(n * (n - 1) // 2)]
    
    # Build constraints
    constraints = []
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append(r[i] + r[j] <= dist_params[idx])
            idx += 1
    
    # Boundary constraints
    # These will be added in solve_radii
    prob = cp.Problem(cp.Maximize(cp.sum(r)), constraints)
    return prob, r, dist_params

def solve_radii(centres, prob, r, dist_params):
    n = centres.shape[0]
    # Compute pairwise distances
    D = cdist(centres, centres)
    
    # Update parameter values
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dist_params[idx].value = D[i, j]
            idx += 1
    
    # Solve the problem
    prob.solve(solver=cp.ECOS, abstol=1e-12, reltol=1e-12)
    radii = r.value

    # Get dual values
    duals = []
    for c in prob.constraints:
        if hasattr(c, 'dual_value') and c.dual_value is not None:
            duals.append(c.dual_value)
        else:
            duals.append(0.0)

    return radii, duals

def energy(centres, prob, r, dist_params):
    radii, _ = solve_radii(centres, prob, r, dist_params)
    return -np.sum(radii)

def random_move(state, rng, temperature):
    new_state = state.copy()
    n = len(new_state) // 2
    
    # Select move type
    move_type = rng.choice([0, 1, 2, 3], p=[0.7, 0.05, 0.05, 0.2])
    
    # Single centre jitter
    if move_type == 0:
        i = rng.integers(0, n)
        dx = rng.normal(0, 0.02)
        dy = rng.normal(0, 0.02)
        new_state[2*i] = np.clip(new_state[2*i] + dx, 0, 1)
        new_state[2*i + 1] = np.clip(new_state[2*i + 1] + dy, 0, 1)
    
    # Cluster jitter
    elif move_type == 1:
        num = rng.integers(3, 6)
        idxs = rng.choice(n, num, replace=False)
        dx = rng.normal(0, 0.02)
        dy = rng.normal(0, 0.02)
        for i in idxs:
            new_state[2*i] = np.clip(new_state[2*i] + dx, 0, 1)
            new_state[2*i + 1] = np.clip(new_state[2*i + 1] + dy, 0, 1)
    
    # Edge push
    elif move_type == 2:
        i = rng.integers(0, n)
        x, y = new_state[2*i], new_state[2*i + 1]
        delta = 0.01
        if x < 0.5:
            new_state[2*i] = np.clip(new_state[2*i] - delta, 0, 1)
        else:
            new_state[2*i] = np.clip(new_state[2*i] + delta, 0, 1)
        if y < 0.5:
            new_state[2*i + 1] = np.clip(new_state[2*i + 1] - delta, 0, 1)
        else:
            new_state[2*i + 1] = np.clip(new_state[2*i + 1] + delta, 0, 1)
    
    # Swap order
    elif move_type == 3:
        i, j = rng.choice(n, 2, replace=False)
        new_state[2*i], new_state[2*j] = new_state[2*j], new_state[2*i]
        new_state[2*i+1], new_state[2*j+1] = new_state[2*j+1], new_state[2*i+1]
    
    return new_state

def anneal_chain(seed, prob, r, dist_params):
    rng = np.random.default_rng(seed)
    n = 26
    
    # Initialize with a random valid configuration
    centres = rng.random((n, 2))
    
    # Get initial energy
    e_current = energy(centres, prob, r, dist_params)
    best_centres = centres.copy()
    best_energy = e_current
    e_history = [e_current]
    
    # Annealing
    T = 0.05
    k = 0
    max_iter = 150000
    max_accepted = 0
    accepted = 0
    
    while k < max_iter and T > 1e-6:
        # Generate new state
        new_state = random_move(centres.flatten(), rng, T)
        new_centres = new_state.reshape(-1, 2)
        
        # Calculate new energy
        e_new = energy(new_centres, prob, r, dist_params)
        delta = e_new - e_current
        if delta < 0 or rng.random() < np.exp(-delta / T):
            centres = new_centres
            e_current = e_new
            if e_current < best_energy:
                best_energy = e_current
                best_centres = centres.copy()
            accepted += 1
            max_accepted = max(max_accepted, accepted)
        
        # Cooling
        T = 0.995 * T
        k += 1
        if k % 5000 == 0:
            e_history.append(e_current)
    
    return best_centres, -best_energy

def local_refine(centres, radii, prob, r, dist_params):
    n = len(centres)
    
    # Define tolerance for active constraints
    tol = 1e-6
    for _ in range(10):
        # Solve inner SOCP to get duals
        sol_radii, duals = solve_radii(centres, prob, r, dist_params)
        
        # Find active constraints (non-overlapping, boundary)
        active_pairs = []
        idx = 0
        for i in range(n):
            for j in range(i+1, n):
                if duals[idx] > tol:
                    active_pairs.append((i, j))
                idx += 1
        
        # Actively enforce active contacts while keeping radii fixed
        # Create a least-squares problem based on active constraints
        if not active_pairs:
            break
        
        A = np.zeros((len(active_pairs), 2*n))
        b = np.zeros(len(active_pairs))
        
        idx = 0
        for i, j in active_pairs:
            # Constraint: ||c_i - c_j||_2 == r_i + r_j
            diff = centres[i] - centres[j]
            norm = np.linalg.norm(diff)
            unit = diff / (norm + 1e-14)
            # We are trying to minimize: ||c_i - c_j||_2 - (r_i + r_j) = 0
            # So we linearize in terms of drift
            A[idx, 2*i:2*i + 2] = -unit
            A[idx, 2*j:2*j + 2] = unit
            b[idx] = norm - (sol_radii[i] + sol_radii[j])
            idx += 1
            
        # Solve A * delta_c = b
        try:
            deltas = np.linalg.lstsq(A, b, rcond=None)[0]
        except np.linalg.LinAlgError:
            break  # If singular, stop
            
        # Update centers
        for i in range(n):
            centres[i] += deltas[2*i:2*i + 2]
            
        # Ensure bounds
        centres = np.clip(centres, 0, 1)
        
    return centres, sol_radii

def finalise(centres, prob, r, dist_params):
    # Re-solve with high precision
    radii, _ = solve_radii(centres, prob, r, dist_params)
    
    # Clamp radii that may have slightly violated boundary
    for i in range(len(centres)):
        x, y = centres[i]
        r_max = min(x, 1-x, y, 1-y)
        if radii[i] > r_max + 1e-13:
            radii[i] = r_max
    
    return centres, radii

def run_packing():
    n = 26
    prob, r, dist_params = build_socp_problem(n)
    
    # Seeds for multiple chains
    seeds = [100 + i for i in range(8)]
    
    best_sum = -np.inf
    best_result = None
    
    for seed in seeds:
        centres, sum_radii = anneal_chain(seed, prob, r, dist_params)
        centres, radii = local_refine(centres, radii, prob, r, dist_params)
        centres, radii = finalise(centres, prob, r, dist_params)
        if sum_radii > best_sum:
            best_sum = sum_radii
            best_result = (centres, radii, sum_radii)
    
    assert best_result is not None
    return best_result
