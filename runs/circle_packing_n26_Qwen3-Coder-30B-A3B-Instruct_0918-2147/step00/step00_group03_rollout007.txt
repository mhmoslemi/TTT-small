import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial.distance import cdist

def build_lp_model(n):
    # Prepare coefficient matrices for the LP
    # Variables: [x1, y1, r1, x2, y2, r2, ..., x_n, y_n, r_n]
    num_vars = 3 * n
    A_eq = []
    b_eq = []
    A_ub = []
    b_ub = []
    c = -np.ones(num_vars)  # Minimize -sum of radii = maximize sum of radii

    # Pairwise constraints (axis-wise separation)
    for i in range(n):
        for j in range(i + 1, n):
            # x_i + r_i <= x_j - r_j     =>   x_i - x_j + r_i + r_j <= 0
            row = np.zeros(num_vars)
            row[3*i] = 1.0
            row[3*j] = -1.0
            row[3*i + 2] = 1.0
            row[3*j + 2] = 1.0
            A_ub.append(row)
            b_ub.append(0.0)

            # x_j + r_j <= x_i - r_i     =>   x_j - x_i + r_i + r_j <= 0
            row = np.zeros(num_vars)
            row[3*i] = -1.0
            row[3*j] = 1.0
            row[3*i + 2] = 1.0
            row[3*j + 2] = 1.0
            A_ub.append(row)
            b_ub.append(0.0)

            # y_i + r_i <= y_j - r_j     =>   y_i - y_j + r_i + r_j <= 0
            row = np.zeros(num_vars)
            row[3*i + 1] = 1.0
            row[3*j + 1] = -1.0
            row[3*i + 2] = 1.0
            row[3*j + 2] = 1.0
            A_ub.append(row)
            b_ub.append(0.0)

            # y_j + r_j <= y_i - r_i     =>   y_j - y_i + r_i + r_j <= 0
            row = np.zeros(num_vars)
            row[3*i + 1] = -1.0
            row[3*j + 1] = 1.0
            row[3*i + 2] = 1.0
            row[3*j + 2] = 1.0
            A_ub.append(row)
            b_ub.append(0.0)

    # Wall constraints for each circle
    for i in range(n):
        # r_i <= x_i
        row = np.zeros(num_vars)
        row[3*i] = -1.0
        row[3*i + 2] = 1.0
        A_ub.append(row)
        b_ub.append(0.0)
        # r_i <= 1 - x_i
        row = np.zeros(num_vars)
        row[3*i] = 1.0
        row[3*i + 2] = 1.0
        A_ub.append(row)
        b_ub.append(1.0)
        # r_i <= y_i
        row = np.zeros(num_vars)
        row[3*i + 1] = -1.0
        row[3*i + 2] = 1.0
        A_ub.append(row)
        b_ub.append(0.0)
        # r_i <= 1 - y_i
        row = np.zeros(num_vars)
        row[3*i + 1] = 1.0
        row[3*i + 2] = 1.0
        A_ub.append(row)
        b_ub.append(1.0)

    return A_ub, b_ub, c, num_vars

def solve_lp(A_ub, b_ub, c):
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'presolve': True})
    if not result.success:
        raise RuntimeError("LP solver failed")
    
    # Extract solution
    y = result.x
    n = len(y) // 3
    centers = y[::3].reshape(n, 2)
    radii = y[2::3]
    return centers, radii, result

def detect_diagonal_conflicts(centers, radii):
    n = len(centers)
    conflicts = []
    dist_matrix = cdist(centers, centers)
    for i in range(n):
        for j in range(i + 1, n):
            # Check if not separated in x or y directions (axis-wise)
            dx = abs(centers[i][0] - centers[j][0])
            dy = abs(centers[i][1] - centers[j][1])
            sum_radius = radii[i] + radii[j]
            if dx >= sum_radius or dy >= sum_radius:
                # If not separated in x or y, check Euclidean distance
                euclidean_dist = dist_matrix[i][j]
                if euclidean_dist < sum_radius:
                    conflicts.append((i, j))
    return conflicts

def polish_global(centers, radii, active_set):

    # plus the LP active constraints encoded as fixed walls and axis separation
    # This is a trust-constr problem with
    # - Equality constraints for active constraints from the LP
    # - Inequality constraints for others

    # Fix all active LP constraints and optimize others
    # Simplified approach:
    # Repeat until convergence
    updated_centers = centers.copy()
    updated_radii = radii.copy()

        centers_flat = vars[:2*n]
        radii_vec = vars[2*n:]
        return -np.sum(radii_vec)

        bounds.append((0.0, 1.0))

        bounds.append((1e-10, 0.5))  # Should be positive and reasonable

                    if dist > 0:
