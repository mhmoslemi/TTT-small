import numpy as np
from scipy.optimize import linprog, minimize


def build_hex_lattice(d):
    # Generate a hexagonal lattice that fits at least d points in [0,1]^2
    # Spacing derived from wanting ~26 points in unit square
    spacing = 1.0 / np.sqrt(d / (np.sqrt(3)/2))
    # Adjust spacing slightly to ensure enough points fit
    spacing = 0.18

    rows = int(1.0 / spacing) + 2
    cols = int(1.0 / spacing) + 2

    centers = []
    for i in range(rows):
        y = i * spacing
        if y > 1.0:
            break
        for j in range(cols):
            x = j * spacing
            if x > 1.0:
                break
            # Offset odd rows
            if i % 2 == 1:
                x += spacing / 2.0
                if x > 1.0:
                    continue
            centers.append([x, y])

    centers = np.array(centers[:d])
    return centers


def lp_max_radii(centers):
    n = centers.shape[0]
    # Wall constraints: r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
    # Non-overlap constraints: r_i + r_j <= dist_ij
    # This gives the set of linear constraints:
    #   A_ub * r <= b_ub

    # Start with wall constraints:
    # For each center i: r_i <= x_i, 1-x_i, y_i, 1-y_i
    A_ub = []
    b_ub = []

    for i in range(n):
        x, y = centers[i]
        # constraint: r_i <= x_i
        row = [0]*n
        row[i] = -1
        A_ub.append(row)
        b_ub.append(-x)

        # constraint: r_i <= 1-x_i
        row = [0]*n
        row[i] = -1
        A_ub.append(row)
        b_ub.append(x - 1)

        # constraint: r_i <= y_i
        row = [0]*n
        row[i] = -1
        A_ub.append(row)
        b_ub.append(-y)

        # constraint: r_i <= 1-y_i
        row = [0]*n
        row[i] = -1
        A_ub.append(row)
        b_ub.append(y - 1)

    # Add pairwise non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            row = [0]*n
            row[i] = -1
            row[j] = -1
            A_ub.append(row)
            b_ub.append(-dist)

    # Constraints are now in A_ub * r <= b_ub
    c = np.ones(n)
    # Use the 'highs' method for numerical robustness
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'presolve': True, 'tol': 1e-9})
    if not result.success:
        # If infeasible (unlikely), return zero radii
        return np.zeros(n), 0.0

    r = result.x
    return r, np.sum(r)


def perturb_centers(centers, sigma):
    # Add Gaussian noise and clamp to [0,1]
    new_centers = centers + np.random.normal(0, sigma, centers.shape)
    new_centers = np.clip(new_centers, 0, 1)
    return new_centers


def anneal_centers(initial_centers):
    # Fixed random seed for reproducibility
    np.random.seed(42)

    # Start with a copy to avoid modifying input
    current_centers = initial_centers.copy()
    current_radii, current_sum = lp_max_radii(current_centers)
    best_centers = current_centers.copy()
    best_radii = current_radii.copy()
    best_sum = current_sum

    # Annealing schedule
    schedule = [
        (0.02, 0.05, 2000),
        (0.01, 0.03, 4000),
        (0.005, 0.015, 3000),
        (0.002, 0.008, 2000)
    ]

    for temp, sigma, steps in schedule:
        for _ in range(steps):
            new_centers = perturb_centers(current_centers, sigma)
            new_radii, new_sum = lp_max_radii(new_centers)
            if new_sum > current_sum or np.random.rand() < np.exp((new_sum - current_sum) / temp):
                current_centers = new_centers
                current_radii = new_radii
                current_sum = new_sum
                if current_sum > best_sum:
                    best_centers = current_centers.copy()
                    best_radii = current_radii.copy()
                    best_sum = current_sum
    return best_centers, best_radii


def local_refine(centers, radii):
    n = centers.shape[0]
    # Start from current solution
    c0 = centers.copy()
    r0 = radii.copy()

    # Formulate the NLP with trust-constr
    # Variables are x_i, y_i, r_i for i in 0..n-1

    # Helper: compute squared distances between centers
    def dist2(i, j):
        dx = c0[i, 0] - c0[j, 0]
        dy = c0[i, 1] - c0[j, 1]
        return dx*dx + dy*dy

    # Define constraints for optimization
    def cons(constr_type, i, j=None):
        if constr_type == 'nonoverlap':
            def fun(vars):
                # vars = [x0, y0, r0, x1, y1, r1, ...]
                x_i = vars[i*3]
                y_i = vars[i*3 + 1]
                r_i = vars[i*3 + 2]
                if j is None:
                    # Wall constraint
                    out = np.empty(4)
                    out[0] = r_i - x_i
                    out[1] = r_i - (1 - x_i)
                    out[2] = r_i - y_i
                    out[3] = r_i - (1 - y_i)
                    return out
                else:
                    x_j = vars[j*3]
                    y_j = vars[j*3 + 1]
                    r_j = vars[j*3 + 2]
                    dist_sq = (x_i - x_j)**2 + (y_i - y_j)**2
                    overlap = dist_sq - (r_i + r_j)**2
                    return np.array([overlap])
        elif constr_type == 'min_r':
            def fun(vars):
                return np.array([r0[i] - vars[i*3 + 2]])
        return fun

    # Variables: center_x, center_y, radius for each circle
    x0 = np.empty(3*n)
    for i in range(n):
        x0[3*i] = c0[i, 0]
        x0[3*i + 1] = c0[i, 1]
        x0[3*i + 2] = r0[i]

    # L-BFGS-B accepts bounds, no need for explicit bounds on [0,1]
    bounds = [(0, 1) for _ in range(3*n)]  # (min, max) for each variable

    # Keep a copy of the LP-radii to ensure we don't reduce
    min_r_lb = r0.copy()

    # Build constraints
    constraints = []

    # Wall constraints: r_i <= x_i, 1-x_i, y_i, 1-y_i
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': cons('nonoverlap', i, None)})

    # Non-overlap constraints
    for i in range(n):
        for j in range(i+1, n):
            constraints.append({'type': 'ineq', 'fun': cons('nonoverlap', i, j)})

    # Radius lower bound from LP solution
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': cons('min_r', i)})

    # Objective is to maximize sum of radii (minimize -sum)
    def obj(vars):
        total = 0.0
        for i in range(n):
            total += vars[i*3 + 2]  # radius
        return -total

    # Solve with trust-constr
    res = minimize(obj, x0, method='trust-constr', bounds=bounds, constraints=constraints,
                   options={'disp': False, 'maxiter': 1000})

    # Extract final values
    if res.success:
        x_final = res.x
        c_final = np.empty((n, 2))
        r_final = np.empty(n)
        for i in range(n):
            c_final[i, 0] = x_final[i*3]
            c_final[i, 1] = x_final[i*3 + 1]
            r_final[i] = x_final[i*3 + 2]
    else:
        # If optimization failed, stick with the previous LP return
        c_final = c0.copy()
        r_final = r0.copy()

    sum_final = np.sum(r_final)
    return c_final, r_final, sum_final

    initial_centers = build_hex_lattice(26)
    
    # Step 2: Anneal to find better center layout
    centers, radii = anneal_centers(initial_centers)
