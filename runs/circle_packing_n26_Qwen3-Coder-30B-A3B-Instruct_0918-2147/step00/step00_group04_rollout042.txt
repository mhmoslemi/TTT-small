import numpy as np
from scipy.optimize import linprog

def initial_hex_centers(k, spacing):
    sqrt3 = np.sqrt(3)
    centers = []
    i, j = 0, 0
    while len(centers) < k:
        x = i * spacing + (j % 2) * spacing / 2
        y = j * spacing * sqrt3 / 2
        if 0 <= x <= 1 and 0 <= y <= 1:
            centers.append([x, y])
        if x > 1:
            j += 1
            i = 0
        else:
            i += 1
    return np.array(centers[:k])

def initial_radii(k, r0):
    return np.full(k, r0)

def build_lp_constants(centers, radii):
    n = centers.shape[0]
    d_left = centers[:, 0]
    d_right = 1 - centers[:, 0]
    d_bottom = centers[:, 1]
    d_top = 1 - centers[:, 1]
    wall_distances = np.stack([d_left, d_right, d_bottom, d_top], axis=1)

    # Pairwise distances
    dists = []
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            dists.append((i, j, d))
    return wall_distances, dists

def solve_radius_lp(consts, radii, graph, alpha):
    n = len(radii)
    wall_distances, pairs = consts
    # Objective to maximize sum of Δr
    c = np.ones(n)

    # Constraints matrix A_ub * x <= b_ub
    A_ub = []
    b_ub = []

    # Wall capacity constraints: Δr_i <= min(d_i_left, d_i_right, d_i_bottom, d_i_top) - r_i
    wall_caps = np.min(wall_distances, axis=1)
    for i in range(n):
        A_ub.append([-1 if j == i else 0 for j in range(n)])
        b_ub.append(wall_caps[i] - radii[i])

    # Pairwise capacity constraints: Δr_i + Δr_j <= D_ij - (r_i + r_j)
    for i, j, d_ij in pairs:
        A_ub.append([-1 if k == i else (-1 if k == j else 0) for k in range(n)])
        b_ub.append(d_ij - (radii[i] + radii[j]))

    # Graph-based transfer limits (optional)
    if graph is not None and graph.edges:
        for i, j in graph.edges:
            if i < n and j < n:
                A_ub.append([1 if k == i else 0 for k in range(n)])
                b_ub.append(alpha * (pairs[(i,j)] - radii[i] - radii[j]))

    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'presolve': True, 'disp': False})
    if res.success:
        return res.x
    else:
        # In case of failure, return zeros to continue iteration gracefully
        return np.zeros(n)

def apply_increment(radii, Δr):
    return radii + Δr

def push_feasibility(centers, radii):
    n = len(radii)
    # Wall push - no overlap with walls
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            radii[i] = x
        elif x + r > 1:
            radii[i] = 1 - x
        if y - r < 0:
            radii[i] = min(radii[i], y)
        elif y + r > 1:
            radii[i] = min(radii[i], 1 - y)

    # Pairwise push
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            r_sum = radii[i] + radii[j]
            if dist < r_sum:
                delta = r_sum - dist
                dx = (centers[i][0] - centers[j][0]) / dist
                dy = (centers[i][1] - centers[j][1]) / dist
                centers[i][0] += 0.5 * delta * dx
                centers[i][1] += 0.5 * delta * dy
                centers[j][0] -= 0.5 * delta * dx
                centers[j][1] -= 0.5 * delta * dy
                # Ensure boundedness
                centers[i][0] = np.clip(centers[i][0], radii[i], 1 - radii[i])
                centers[i][1] = np.clip(centers[i][1], radii[i], 1 - radii[i])
                centers[j][0] = np.clip(centers[j][0], radii[j], 1 - radii[j])
                centers[j][1] = np.clip(centers[j][1], radii[j], 1 - radii[j])

    return centers, radii

class Graph:
    def __init__(self, n):
        self.n = n
        self.edges = set()
        self.wall_contacts = {}  # vertex id -> set of walls {0,1,2,3} representing left right bottom top
    
    def add_edge(self, i, j):
        self.edges.add((min(i,j), max(i,j)))
        
    def touch_wall(self, i, wall_id):
        if i not in self.wall_contacts:
            self.wall_contacts[i] = set()
        self.wall_contacts[i].add(wall_id)

def run_iterations(centers, radii, max_iter, epsilon, alpha):
    graph = Graph(len(radii))
    for idx in range(len(centers)):
        x, y = centers[idx]
        r = radii[idx]
        if x - r <= 0:
            graph.touch_wall(idx, 0)
        if x + r >= 1:
            graph.touch_wall(idx, 1)
        if y - r <= 0:
            graph.touch_wall(idx, 2)
        if y + r >= 1:
            graph.touch_wall(idx, 3)

    for _ in range(max_iter):
        wall_dists, pairs = build_lp_constants(centers, radii)
        const_tuples = (wall_dists, pairs)
        Δr = solve_radius_lp(const_tuples, radii, graph, alpha)
        
        total_incr = np.sum(Δr)
        if total_incr < epsilon:
            break
            
        radii = apply_increment(radii, Δr)
        centers, radii = push_feasibility(centers, radii)

        # Update graph edges based on new distances
        graph.edges.clear()
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                r_sum = radii[i] + radii[j]
                if abs(dist - r_sum) < 1e-12:
                    graph.add_edge(i, j)
                    
    return centers, radii

def final_refinement(centers, radii):
    wall_dists, pairs = build_lp_constants(centers, radii)
    const_tuples = (wall_dists, pairs)
    graph = Graph(len(radii))
    # Final rigorous high precision sequence
    Δr = solve_radius_lp(const_tuples, radii, graph, 0.5)
    radii = apply_increment(radii, Δr)
    
    # Further clipping
    for i in range(len(radii)):
        x, y = centers[i]
        r = radii[i]
        radii[i] = min(r, x, 1 - x, y, 1 - y)
        
    # Pairwise correction
    for i in range(len(radii)):
        for j in range(i + 1, len(radii)):
            dist_ij = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            r_sum_i_j = radii[i] + radii[j]
            if dist_ij < r_sum_i_j:
                diff = r_sum_i_j - dist_ij
                radii[i] -= diff * 0.5
                radii[j] -= diff * 0.5
                
    return centers, radii

def run_packing():
    n_circles = 26
    spacing = 0.12
    r0 = 0.005
    max_iter = 200
    epsilon = 1e-5
    alpha = 0.5
    
    centers = initial_hex_centers(n_circles, spacing)
    radii = initial_radii(n_circles, r0)
    centers, radii = run_iterations(centers, radii, max_iter, epsilon, alpha)
    centers, radii = final_refinement(centers, radii)
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
