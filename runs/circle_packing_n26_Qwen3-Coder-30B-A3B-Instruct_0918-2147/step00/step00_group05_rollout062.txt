            x_start = spacing * (row % 2) / 2
            x = x_start
            while x <= 1.0:
                if 0 <= x <= 1.0 and 0 <= y <= 1.0:
                    points.append((x, y))
                x += spacing
            y += spacing * sqrt(3) / 2
            row += 1
