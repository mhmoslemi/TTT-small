import numpy as np
from scipy.optimize import minimize
import scipy.sparse as sp

def init_hex_grid(n=26, r_init=0.04):
    """Initialize 26 circles on a hexagonal grid inside unit square."""
    h = np.sqrt(3) * r_init
    rows = []
    y = r_init
    row_num = 0
    while len(rows) < n:
        if row_num % 2 == 0:
            x_start = r_init
            step = 2 * r_init
        else:
            x_start = 2 * r_init
            step = 2 * r_init
        x = x_start
        while x <= 1 - r_init and len(rows) < n:
            rows.append((x, y))
            x += step
        y += h
        row_num += 1
    centers = np.array(rows[:n])
    radii = np.full(n, r_init)
    return centers, radii

def max_radii(centers, radii):
    """Iteratively increase each circle's radius to its maximum allowed value."""
    max_iter = 100
    eps = 1e-12
    for _ in range(max_iter):
        # Wall limits
        wall = np.minimum(centers, 1 - centers).min(axis=1)
        # Neighbour limits
        D = np.sqrt(((centers[:, None, :] - centers[None, :, :]) ** 2).sum(-1))
        np.fill_diagonal(D, np.inf)
        neighbor = D - radii[:, None]
        neighbor = np.maximum(neighbor, 0)  # Ensure non-negative
        neighbor_min = neighbor.min(axis=1)
        new_radii = np.minimum(wall, neighbor_min)
        new_radii = np.maximum(new_radii, 0)  # Ensure non-negative
        if np.max(np.abs(new_radii - radii)) < 1e-8:
            break
        radii = new_radii
    return radii

def hill_climb(centers, radii, iterations=5000, step=0.01):
    """Stochastic hill-climbing to improve packing by adjusting centers."""
    eps = 1e-12
    old_sum = np.sum(radii)
    for _ in range(iterations):
        # Pick a random circle
        i = np.random.randint(0, len(centers))
        # Save old values
        old_center = centers[i].copy()
        old_radius = radii[i]
        # Propose move
        delta = np.random.uniform(-step, step, 2)
        new_center = old_center + delta
        # Enforce boundary constraints
        new_center = np.clip(new_center, radii[i], 1 - radii[i])
        # Set temporarily to try
        centers[i] = new_center
        # Re-compute available radii
        new_radii = max_radii(centers, radii.copy())
        # Check if improvement
        new_sum = np.sum(new_radii)
        if new_sum > old_sum + 1e-12:
            # Accept change
            radii[:] = new_radii
            old_sum = new_sum
        else:
            # Restore old values
            centers[i] = old_center
    return centers, radii

def refine_with_scipy(centers, radii):
    """Use scipy.optimize.minimize to refine using the trust-constr method."""
    n = len(centers)
    # Construct constraint functions
    def wall_constraints(v):
        # v = [x1, y1, x2, y2, ..., x26, y26, r1, r2, ..., r26]
        x = v[:n]
        y = v[n:2*n]
        r = v[2*n:]
        # Wall constraints
        constraints = []
        constraints.extend(r - x)  # r <= x
        constraints.extend(r + x - 1)  # r + x <= 1
        constraints.extend(r - y)  # r <= y
        constraints.extend(r + y - 1)  # r + y <= 1
        return np.array(constraints)

    def overlap_constraints(v):
        # v = [x1, y1, x2, y2, ..., x26, y26, r1, r2, ..., r26]
        x = v[:n]
        y = v[n:2*n]
        r = v[2*n:]
        centers_view = np.stack([x, y], axis=1)
        D = np.sqrt(((centers_view[:, None, :] - centers_view[None, :, :]) ** 2).sum(-1))
        # Extract the constraint values
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                dist = D[i, j]
                constraints.append(r[i] + r[j] - dist)
        return np.array(constraints)

    def make_constraint_func(constraint_type):
        if constraint_type == "wall":
            func = wall_constraints
            lb = np.full(n * 4, -np.inf)
            ub = np.zeros(n * 4)
        elif constraint_type == "overlap":
            func = overlap_constraints
            lb = np.full(n * (n - 1) // 2, -np.inf)
            ub = np.zeros(n * (n - 1) // 2)
        else:
            raise ValueError("Unknown constraint type")
        return func, lb, ub

    # Create variable array (initial guess)
    x0 = np.concatenate([centers.flat, radii.flat])

    # Define constraints and bounds
    wall_func, lb_wall, ub_wall = make_constraint_func("wall")
    overlap_func, lb_overlap, ub_overlap = make_constraint_func("overlap")

    cons = []
    cons.append({'type': 'ineq', 'fun': lambda v: ub_wall - lb_wall - wall_func(v)})
    cons.append({'type': 'ineq', 'fun': lambda v: ub_overlap - lb_overlap - overlap_func(v)})

    # Bounds: 0 <= x_i, y_i <= 1; 0 <= r_i <= 0.5
    bounds = []
    for i in range(n):
        bounds.append((0, 1))  # x_i
        bounds.append((0, 1))  # y_i
    for i in range(n):
        bounds.append((0, 0.5))  # r_i

    # Minimize negative sum of radii, equivalent to maximizing sum of radii
    def objective(v):
        return -np.sum(v[2 * n:])

    # Call scipy minimize
    res = minimize(objective,
                   x0,
                   method='trust-constr',
                   jac='2-point',
                   hess='BFGS',
                   bounds=bounds,
                   constraints=cons,
                   options={'maxiter': 500})

    if not res.success:
        raise RuntimeError(f"Optimization failed with status: {res.message}")
    
    # Extract result
    x_opt = res.x[:n]
    y_opt = res.x[n:2*n]
    r_opt = res.x[2*n:]
    centers = np.column_stack([x_opt, y_opt])
    radii = r_opt
    
    # Final refinement loop to ensure feasibility
    radii = max_radii(centers, radii)
    return centers, radii

def run_packing():
    centers, radii = init_hex_grid()
    radii = max_radii(centers, radii)
    centers, radii = hill_climb(centers, radii)
    centers, radii = refine_with_scipy(centers, radii)
    sum_r = float(radii.sum())
    return centers, radii, sum_r
