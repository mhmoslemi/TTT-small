import numpy as np
from scipy.optimize import minimize

def initial_layout():
    # Create a hexagonal grid layout
    rows, cols = 5, 5
    spacing = 0.18
    radius0 = spacing / 2 - 0.01

    centers = []
    radii = []

    for i in range(rows):
        for j in range(cols):
            x = (j + 0.5) * spacing
            y = (i + 0.5) * spacing
            # Adjust positions to be within the unit square
            x = max(radius0, min(1 - radius0, x))
            y = max(radius0, min(1 - radius0, y))
            centers.append([x, y])
            radii.append(radius0)

    # Add one more circle in the center
    centers.append([0.5, 0.5])
    radii.append(0.05)
    
    return np.array(centers), np.array(radii)

def pack_vector(centers, radii):
    z = np.empty(3 * centers.shape[0])
    z[0::3] = centers[:, 0]
    z[1::3] = centers[:, 1]
    z[2::3] = radii
    return z

def unpack_vector(z):
    n = z.shape[0] // 3
    centers = np.empty((n, 2))
    centers[:, 0] = z[0::3]
    centers[:, 1] = z[1::3]
    radii = z[2::3]
    return centers, radii

def objective(z):
    centers, radii = unpack_vector(z)
    return -np.sum(radii)

def constraint_boundary(z):
    centers, radii = unpack_vector(z)
    n = centers.shape[0]
    constraints = []

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # x - r >= 0
        constraints.append(x - r)
        # 1 - x - r >= 0
        constraints.append(1 - x - r)
        # y - r >= 0
        constraints.append(y - r)
        # 1 - y - r >= 0
        constraints.append(1 - y - r)

    return np.array(constraints)

def constraint_nonoverlap(z):
    centers, radii = unpack_vector(z)
    n = centers.shape[0]
    constraints = []

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx * dx + dy * dy
            sum_radii = radii[i] + radii[j]
            # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
            constraints.append(dist_sq - sum_radii * sum_radii)

    return np.array(constraints)
