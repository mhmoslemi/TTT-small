import numpy as np
from scipy.stats.qmc import Sobol

def initialize_seeds(n):
    sobol = Sobol(d=2, seed=42)
    points = sobol.random(n)
    margin = 0.02
    return np.clip(margin + (1 - 2 * margin) * points, margin, 1 - margin)

def compute_feasible_radii(centers):
    n = centers.shape[0]
    # Compute pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_matrix = np.sqrt(np.sum(diff**2, axis=2))
    
    # Nearest neighbor distances (excluding self)
    nn_dist = np.min(dist_matrix + np.eye(n) * 1e10, axis=1)
    
    # Wall distances
    wall_distances = np.array([
        centers[:, 0],  # x distance to left
        1 - centers[:, 0], # x distance to right
        centers[:, 1],  # y distance to bottom
        1 - centers[:, 1]  # y distance to top
    ])
    wall_min = np.min(wall_distances, axis=0)
    
    # Maximal feasible radii
    radii = np.minimum(wall_min, 0.5 * nn_dist)
    return radii

def redistribute_radii(radii, slack, alpha):
    total_slack = np.sum(slack)
    if total_slack <= 1e-15:
        return radii

    # Increase wall-limited circles
    wall_limited = slack > 1e-15
    increase = np.zeros_like(radii)
    increase[wall_limited] = alpha * slack[wall_limited]

    # Reduce neighbour-limited circles proportionally
    neighbor_limited = ~wall_limited
    if np.sum(neighbor_limited) > 0:
        total_radii = np.sum(radii[neighbor_limited])
        if total_radii > 1e-15:
            decrease = (alpha * total_slack) * (radii[neighbor_limited] / total_radii)
        else:
            decrease = np.zeros_like(radii[neighbor_limited])
        radii[neighbor_limited] -= decrease
        radii[neighbor_limited] = np.maximum(0.0, radii[neighbor_limited])

    radii += increase
    return radii

def push_away(centers, radii, beta, eps):
    n = centers.shape[0]
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_matrix = np.sqrt(np.sum(diff**2, axis=2))
    new_centers = centers.copy()

    for i in range(n):
        for j in range(n):
            if i == j:
                continue
            d = dist_matrix[i, j]
            r_i = radii[i]
            r_j = radii[j]
            if d < r_i + r_j + eps:
                # Push away
                delta = (r_i + r_j + eps - d)
                norm_vec = diff[i, j] / (d + eps)
                new_centers[i] += beta * delta * norm_vec

    # Clamp to interior
    new_centers = np.clip(new_centers, eps, 1 - eps)
    return new_centers

def iterate(centers, radii, alpha, beta, max_iter=200):
    for _ in range(max_iter):
        # Compute current radii
        radii = compute_feasible_radii(centers)
        # Compute slack
        nn_dist = np.min(
            np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :])**2, axis=2)) 
            + np.eye(centers.shape[0]) * 1e10,
            axis=1
        )
        slack = 0.5 * nn_dist - radii
        # Redistribution
        radii = redistribute_radii(radii, slack, alpha)
        # Push away
        centers = push_away(centers, radii, beta, 1e-8)
        # Recompute radii
        radii = compute_feasible_radii(centers)
        # Check convergence (optional for final iteration)
    return centers, radii

def repair(centers, radii, eps_final=5e-10):
    radii = np.maximum(0.0, radii - eps_final)
    centers = np.clip(centers, eps_final, 1 - eps_final)
    return centers, radii

def run_packing():
    # Initialize
    centers = initialize_seeds(26)
    # Iterate
    centers, radii = iterate(centers, compute_feasible_radii(centers), alpha=0.4, beta=0.3, max_iter=100)
