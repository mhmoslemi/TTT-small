def build_hex_lattice(spacing):
    # Generate a hexagonal lattice inside [0,1]x[0,1]
    x_coords = []
    y_coords = []
    y = 0.0
    row = 0
    while y <= 1.0:
        x_offset = (row % 2) * spacing / 2.0
        x = x_offset
        while x <= 1.0:
            if (x >= 0.0 and x <= 1.0 and y >= 0.0 and y <= 1.0):
                x_coords.append(x)
                y_coords.append(y)
            x += spacing
        y += spacing * np.sqrt(3) / 2.0
        row += 1
    return np.array(x_coords), np.array(y_coords)
