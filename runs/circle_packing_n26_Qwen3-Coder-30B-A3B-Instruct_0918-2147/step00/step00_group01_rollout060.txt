import numpy as np
from scipy.optimize import differential_evolution
from scipy.spatial.distance import cdist

def compute_radii(centers):
    n = centers.shape[0]
    radii = np.zeros(n)
    
    # Wall distances
    x = centers[:, 0]
    y = centers[:, 1]
    wall_distances = np.stack([x, 1 - x, y, 1 - y]).T  # shape (n, 4)
    wall_min = np.min(wall_distances, axis=1)  # shape (n,)
    
    # Pairwise distances
    distances = cdist(centers, centers)
    np.fill_diagonal(distances, np.inf)
    neighbor_distances = 0.5 * np.min(distances, axis=1)  # shape (n,)
    
    # Final radii
    radii = np.minimum(wall_min, neighbor_distances)
    return radii

def neg_sum_radii(z):
    centers = z.reshape(-1, 2)
    radii = compute_radii(centers)
    return -np.sum(radii)

    np.random.seed(42)

    result = differential_evolution(

        bounds,
        strategy='best1bin',
        maxiter=2000,
        popsize=10,
        tol=1e-7,
        polish=False,
        seed=42
    )
    
    best_centers = result.x.reshape(-1, 2)
    
    # Local refinement with adaptive step size
    sigma = 0.02
    max_iter = 3000
    best_centers_copy = best_centers.copy()

        # Generate perturbation
        perturb = np.random.normal(0, sigma, best_centers_copy.shape)
        new_centers = best_centers_copy + perturb

        new_sum = np.sum(new_radii)

            sigma = min(1.1 * sigma, 0.1)  # increase step size, but cap it
