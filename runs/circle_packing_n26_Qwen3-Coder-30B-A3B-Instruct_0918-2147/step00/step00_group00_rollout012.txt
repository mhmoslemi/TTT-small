import numpy as np
from scipy.optimize import minimize

def make_hex_lattice(n, margin=0.05):
    # Generate a hexagonal lattice scaled to fit in [margin, 1-margin] x [margin, 1-margin]
    # This should be good for dense packing and allow space for radii
    size = int(np.ceil(np.sqrt(n)))
    dx = (1 - 2*margin) / (size + 0.5)
    dy = dx * np.sqrt(3) / 2

    pts = []
    for i in range(size):
        for j in range(size):
            x = margin + i * dx + (j % 2) * dx / 2
            y = margin + j * dy
            if len(pts) < n and 0 <= x <= 1 and 0 <= y <= 1:
                pts.append([x, y])
    # always take first n points
    return np.array(pts[:n])

def objective(z):
    # z is array of [x0,y0,r0,x1,y1,r1...]
    radii = z[2::3]
    return -np.sum(radii)

def boundary_constraints(z):
    # z = [x0,y0,r0,...]
    centers = z[0::3].reshape(-1, 2)
    radii = z[2::3]
    # for each circle, 4 constraints (x-r >= 0, x+r <= 1, y-r >= 0, y+r <= 1)
    # we're checking inequalities of the form fun <= 0
    # so we compute -constraint for +value for inequality
    constraints = []
    for i in range(len(radii)):
        x, y = centers[i]
        r = radii[i]
        constraints.extend([
            x - r,               # x - r >= 0 => -(x - r) <= 0
            -(x + r - 1),        # x + r <= 1 => -(x + r - 1) <= 0
            y - r,               # y - r >= 0 => -(y - r) <= 0
            -(y + r - 1)         # y + r <= 1 => -(y + r - 1) <= 0
        ])
    return np.array(constraints)

def pairwise_constraints(z):
    centers = z[0::3].reshape(-1, 2)
    radii = z[2::3]
    n = len(centers)
    constraints = []
    # pairwise distance constraints
    for i in range(n):
        for j in range(i+1, n):
            dx = centers[i,0] - centers[j,0]
            dy = centers[i,1] - centers[j,1]
            dist_sq = dx*dx + dy*dy
            r_sum = radii[i] + radii[j]
            # we want (dist)^2 - (r1+r2)^2 >= 0
            # this means fun <= 0 where fun = dist^2 - (r1+r2)^2
            constraints.append(dist_sq - r_sum*r_sum)
    return np.array(constraints)

def solve_nlp(initial_z):
    # stack constraints
    cons = []
    # boundary constraints
    cons.append({'type': 'ineq', 'fun': boundary_constraints})
    # pairwise constraints
    cons.append({'type': 'ineq', 'fun': pairwise_constraints})

    # bounds
    # +2 for x,y, -1 for which x,y are limited to [0,1] and radii [0, 0.5]
    bounds = []
    for i in range(len(initial_z)):
        if i % 3 == 0 or i % 3 == 1:  # x or y
            bounds.append((0.0, 1.0))
        else:  # r
            bounds.append((0.0, 0.5))

    res = minimize(
        objective, initial_z,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 1000, 'ftol': 1e-9}
    )
    return res

def nudge_centers(centers, radii):
    # nudge the centers of circles close to boundaries
    # distance from each circle to any of the 4 walls
    dists = np.array([
        [radii[i], 1 - centers[i,0], radii[i], 1 - centers[i,1]]
        for i in range(len(radii))
    ])
    # find margins less than 1e-5 or something like that
    # nudge circles with any margin small enough
    indices_to_nudge = np.where(np.min(dists, axis=1) < 1e-5)[0]
    nudge_dist = 0.001
    new_centers = centers.copy()
    for i in indices_to_nudge:
        # nudge closest wall
        min_dist = np.min(dists[i])
        idx = np.argmin(dists[i])
        # 0-left, 1-right, 2-bottom, 3-top
        if idx == 0: # left
            new_centers[i, 0] += nudge_dist
        elif idx == 1: # right
            new_centers[i, 0] -= nudge_dist
        elif idx == 2: # bottom
            new_centers[i, 1] += nudge_dist
        else: # top
            new_centers[i, 1] -= nudge_dist
    return new_centers

    radii = np.full(n_circles, 0.02)

    # First solve

    # Retrieve result
