import numpy as np
from scipy.optimize import linprog, minimize
from math import sqrt, ceil

def build_hex_lattice(d):
    """Generate a hexagonal lattice of at least d points in [0,1] x [0,1]"""
    # Hexagonal lattice spacing for d points
    # For a square of side 1, we want to fit d points with spacing s
    # About sqrt(d) rows and columns needed
    rows = int(ceil(sqrt(d)))
    cols = int(ceil(d / rows))
    
    # Spacing for a hexagonal lattice
    # We'll use the area of a hexagon to determine spacing
    # Area = (3*sqrt(3)/2)*s^2, where s is distance between centers
    # For d points unit square area = 1
    # Each point uses an area of ~1/d
    # For hexagons this area is ~1.5*sqrt(3)*s^2
    # Therefore s = sqrt(1/(1.5*sqrt(3)*d))  
    # Let's just make it simple and use spacing of 1/sqrt(d)
    
    s = 1.0 / sqrt(max(1, d))
    h = s * sqrt(3) / 2  # Vertical spacing of hexagon
    # Allow some padding at edges: use [0.02, 1-0.02] range
    x_start = 0.02
    y_start = 0.02
    x_end = 0.98
    y_end = 0.98
    x_range = x_end - x_start
    y_range = y_end - y_start

    centers = []
    i = 0
    for row in range(ceil(y_range/h)+2):
        y = y_start + row * h
        if y > 0.98:
            break
        offset = (row % 2) * s / 2
        for col in range(ceil(x_range/s)+2):
            x = x_start + col * s + offset
            if x > 0.98:
                break
            if i >= d:
                break
            centers.append((x, y))
            i += 1
        if i >= d:
            break

    # Adjust so we have exactly d points and they're within [0,1]
    centers = centers[:d]
    return np.array(centers)

def lp_max_radii(centers):
    """Given center coordinates, solve linear program to get max radii"""
    n = len(centers)
    if n == 0:
        return np.array([]), 0.0

    # Build LP constraints
    # Wall constraints: r <= min(x, 1-x, y, 1-y) for each circle
    # Pairwise constraints: r_i + r_j <= d_ij

    # Objective: maximize sum of radii
    c = np.ones(n)

    # Build A_ub * radii <= b_ub
    # Wall constraints first
    A_ub_list = []
    b_ub_list = []
    
    # For each circle
    for i in range(n):
        x, y = centers[i]
        r_wall = min(x, 1-x, y, 1-y)
        # r_i <= r_wall (-> -r_i <= r_wall)
        A_ub_list.append([-1.0])  # -r_i <= r_wall
        b_ub_list.append(r_wall)

    # Pairwise constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            # r_i + r_j <= dist (-> -r_i - r_j >= -dist)
            A_ub = np.zeros(n)
            A_ub[i] = -1.0
            A_ub[j] = -1.0
            A_ub_list.append(A_ub)
            b_ub_list.append(-dist)

    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    
    # Solve LP
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', tol=1e-9)
        if res.success:
            radii = res.x
            sum_radii = res.fun
        else:
            # Fallback: try with dummy solution
            radii = np.zeros(n)
            sum_radii = 0.0
    except Exception:
        # Should not happen, but just in case
        radii = np.zeros(n)
        sum_radii = 0.0

    return radii, sum_radii

def perturb_centers(centers, sigma):
    """Add Gaussian noise to centers and clamp to unit square"""
    new_centers = centers + np.random.normal(0, sigma, centers.shape)
    new_centers[:, 0] = np.clip(new_centers[:, 0], 0, 1)
    new_centers[:, 1] = np.clip(new_centers[:, 1], 0, 1)
    return new_centers

def anneal_centers(initial_centers):
    """Run simulated annealing"""
    np.random.seed(42)  # For reproducibility
    best_centers = initial_centers.copy()
    best_radii, best_sum = lp_max_radii(best_centers)
    current_centers = initial_centers.copy()
    current_radii, current_sum = best_sum, best_sum
    T_schedule = [
        (0.02, 0.05, 2000),
        (0.01, 0.03, 4000),
        (0.005, 0.015, 3000),
        (0.002, 0.008, 2000)
    ]
    for temp, sigma, iterations in T_schedule:
        for _ in range(iterations):
            new_centers = perturb_centers(current_centers, sigma)
            new_radii, new_sum = lp_max_radii(new_centers)

            # Acceptance probability
            if new_sum >= current_sum:
                accepted = True
            else:
                # Chance of acceptance
                delta = new_sum - current_sum
                prob = np.exp(delta / temp)
                accepted = np.random.rand() < prob

            if accepted:
                current_centers = new_centers.copy()
                current_radii = new_radii.copy()
                current_sum = new_sum
            
            # Update best if improved
            if current_sum > best_sum:
                best_centers = current_centers.copy()
                best_radii = current_radii.copy()
                best_sum = current_sum

    return best_centers, best_radii, best_sum

def local_refine(centers, radii):
    """Refine the centers and radii using NL optimization with trust constrained solver"""
    n = len(centers)
    # Convert to flattened
    vars_init = np.column_stack([centers, radii])
    vars_init = vars_init.flatten()
    
    # Reconstruct the current centers and radii
    current_centers = vars_init[:2*n].reshape(n,2)
    current_radii = vars_init[2*n:]
    
    # Build fixed bounds and constraints for optimization
    def obj(x):
        return -np.sum(x[2*n:])  # Negative because we minimize

    def jac(x):
        return np.concatenate([np.zeros(2*n), np.ones(n)])

    # The actual nonlinear program to solve
    # Yields new centers and radii
    def constraints_func(x):
        # x is [c1_x,c1_y,...,c26_x,c26_y,r1,r2,...,r26]
        n = len(centers)
        c = x[:2*n].reshape(n,2)
        r = x[2*n:]
        
        # Geometric constraints
        constraints = []
        # Wall constraints
        for i in range(n):
            constraints.append(c[i][0] - r[i])  # x - r >= 0
            constraints.append(1 - c[i][0] - r[i])  # 1 - x - r >= 0
            constraints.append(c[i][1] - r[i])  # y - r >= 0
            constraints.append(1 - c[i][1] - r[i])  # 1 - y - r >= 0

        # Pairwise non-overlap
        for i in range(n):
            for j in range(i+1, n):
                dist = np.linalg.norm(c[i] - c[j])
                constraints.append(dist - r[i] - r[j])  # dist - r_i - r_j >= 0
            
        return constraints

    # Bounds [0,1] for centers, [r_LP, inf] for radii
    bounds = []
    for i in range(n):
        bounds.append((0, 1))  # x_i in [0, 1]
        bounds.append((0, 1))  # y_i in [0, 1]
    # Radii bound, just make a lower bound.
    for i in range(n):
        bounds.append((radii[i], None))  # r_i >= r_LP

    try:
        res = minimize(obj, vars_init, method='trust-constr', 
                       bounds=bounds, 
                       constraints={'type': 'ineq', 'fun': lambda x: constraints_func(x)},
                       jac=jac, 
                       options={'maxiter': 300, 'gtol': 1e-6})
        
        if res.success:
            final_vars = res.x
            final_centers = final_vars[:2*n].reshape(n,2)
            final_radii = final_vars[2*n:]
            sum_radii = np.sum(final_radii)
        else:
            # Return the original if non-linear optimization failed
            final_centers = centers
            final_radii = radii
            sum_radii = np.sum(radii)
    except Exception:
        # Just in case return the original
        final_centers = centers
        final_radii = radii
        sum_radii = np.sum(radii)

    return final_centers, final_radii, sum_radii
