import numpy as np
from scipy.optimize import minimize
import scipy.sparse as sp

def init_hex_grid(n=26, r_init=0.04):
    # Generate hexagonal grid points in the unit square
    h = np.sqrt(3) * r_init
    row_height = h
    col_width = 2 * r_init
    
    centers = []
    y = r_init
    
    row = 0
    while len(centers) < n:
        if row % 2 == 0:
            x_start = r_init
        else:
            x_start = r_init + col_width / 2
            
        x = x_start
        while x <= 1 - r_init and len(centers) < n:
            centers.append([x, y])
            x += col_width
        y += row_height
        row += 1
        
    centers = np.array(centers[:n])
    radii = np.full(n, r_init)
    return centers, radii

def max_radii(centers, radii):
    # compute max radii with fixed centers
    max_iter = 100
    eps = 1e-8
    
    for _ in range(max_iter):
        # wall constraints
        wall_limits = np.minimum(centers, 1 - centers).min(axis=1)
        
        # neighbor constraints
        D = np.sqrt(((centers[:,None,:] - centers[None,:,:])**2).sum(-1))
        # set diagonal to large value to ignore self
        D += np.eye(len(D)) * 1e10
        neighbor_limits = (D - radii[:,None]).min(axis=1)
        
        new_radii = np.minimum(wall_limits, neighbor_limits)
        new_radii = np.clip(new_radii, 0, None)
        
        if np.max(np.abs(new_radii - radii)) < eps:
            break
            
        radii = new_radii
        
    return radii

def hill_climb(centers, radii, iterations=5000, step=0.01):
    # stochastic hill climbing with random moves
    n = len(centers)
    
    for _ in range(iterations):
        # random order
        order = np.random.permutation(n)
        old_sum = np.sum(radii)
        
        for i in order:
            # propose move
            old_center = centers[i].copy()
            new_center = old_center + np.random.uniform(-step, step, 2)
            
            # ensure new center is valid
            new_center[0] = np.clip(new_center[0], radii[i], 1 - radii[i])
            new_center[1] = np.clip(new_center[1], radii[i], 1 - radii[i])
            
            # temporarily set new center
            centers[i] = new_center
            
            # recompute max radii with new center
            old_radii = radii.copy()
            radii = max_radii(centers, radii)
            
            # accept if improvement
            new_sum = np.sum(radii)
            if new_sum <= old_sum:
                # reject
                centers[i] = old_center
                radii = old_radii
            else:
                old_sum = new_sum
                
    return centers, radii

def refine_with_scipy(centers, radii):
    # convert to flat vector
    n = len(centers)
    v0 = np.concatenate([centers.flatten(), radii])
    
    def obj(v):
        return -np.sum(v[2*n:])
    
    def wall_constraints(v):
        centers_v = v[:2*n].reshape((n, 2))
        radii_v = v[2*n:]
        out = np.zeros(4 * n)
        out[0*n:1*n] = radii_v - centers_v[:, 0]  # r - x
        out[1*n:2*n] = radii_v + centers_v[:, 0] - 1  # r + x - 1
        out[2*n:3*n] = radii_v - centers_v[:, 1]  # r - y
        out[3*n:4*n] = radii_v + centers_v[:, 1] - 1  # r + y - 1
        return out
    
    def overlap_constraints(v):
        centers_v = v[:2*n].reshape((n, 2))
        radii_v = v[2*n:]
        out = []
        for i in range(n):
            for j in range(i+1, n):
                dist = np.sqrt(np.sum((centers_v[i] - centers_v[j]) ** 2))
                out.append(radii_v[i] + radii_v[j] - dist)
        return np.array(out)
    
    # jacobian for wall constraints
    def jac_wall(v):
        n = len(centers)
        rows = []
        cols = []
        data = []
        for i in range(n):
            # r - x
            rows.extend([i, i + n, i + 2*n, i + 3*n])
            cols.extend([i + 2*n, i*2, i + 2*n, i*2 + 1])
            data.extend([1, -1, 1, -1])
        J = sp.coo_matrix((data, (rows, cols)), shape=(4*n, 2*n + n))
        return J
    
    # jacobian for overlap constraints 
    def jac_overlap(v):
        centers_v = v[:2*n].reshape((n, 2))
        radii_v = v[2*n:]
        m = n * (n - 1) // 2  # number of overlap constraints
        rows = []
        cols = []
        data = []
        idx = 0
        for i in range(n):
            for j in range(i+1, n):
                dx = centers_v[i, 0] - centers_v[j, 0]
                dy = centers_v[i, 1] - centers_v[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist > 0:
                    dr = 1 / dist
                    ddx = -dx * dr
                    ddy = -dy * dr
                else:
                    ddx = ddy = 0
                    
                rows.extend([idx, idx, idx])
                cols.extend([i + 2*n, j + 2*n, i*2, i*2+1, j*2, j*2+1])
                data.extend([1, 1, ddx, ddy, -ddx, -ddy])
                idx += 1
        J = sp.coo_matrix((data, (rows, cols)), shape=(m, 2*n + n))
        return J
    
    # constraints
    wall_cnstr = NonlinearConstraint(wall_constraints, -np.inf, 0, jac=jac_wall)
    overlap_cnstr = NonlinearConstraint(overlap_constraints, -np.inf, 0, jac=jac_overlap)
    
    bounds = [(0, 1)] * (2*n) + [(0, 0.5)] * n
    
    res = minimize(obj, v0, method='trust-constr', 
                   constraints=[wall_cnstr, overlap_cnstr],
                   bounds=bounds,
                   options={'maxiter': 200, 'disp': False})
    
    if res.success:
        new_centers = res.x[:2*n].reshape((n, 2))
        new_radii = res.x[2*n:]
        return new_centers, new_radii
    else:
        # fallback: return original
        return centers, radii

def run_packing():
    centers, radii = init_hex_grid()
    radii = max_radii(centers, radii)
    centers, radii = hill_climb(centers, radii)
    centers, radii = refine_with_scipy(centers, radii)
    sum_r = float(radii.sum())
    return centers, radii, sum_r
