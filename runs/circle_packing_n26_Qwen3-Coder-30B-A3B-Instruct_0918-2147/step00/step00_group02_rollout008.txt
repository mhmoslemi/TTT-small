import numpy as np
from scipy.stats.qmc import Sobol
from scipy.spatial.distance import cdist

def initialize_seeds(n):
    sobol = Sobol(d=2, seed=0)
    points = sobol.random(n)
    # Clip to interior with margin
    delta = 0.02
    points = delta + (1 - 2 * delta) * points
    return points

def compute_feasible_radii(centers):
    n = centers.shape[0]
    # Distance to walls
    wall_distances = np.array([
        centers[:, 0],  # x
        1 - centers[:, 0],  # 1-x
        centers[:, 1],  # y
        1 - centers[:, 1]   # 1-y
    ]).min(axis=0)
    
    # Nearest neighbor distances
    D = cdist(centers, centers)
    np.fill_diagonal(D, np.inf)
    nn_distances = D.min(axis=1)
    
    # Compute radii as min of wall and neighbor limiting distances
    radii = np.minimum(wall_distances, 0.5 * nn_distances)
    return radii

def redistribute_radii(radii, slack, alpha):
    total_slack = slack.sum()
    if total_slack == 0:
        return radii.copy()
    
    new_radii = radii.copy()
    # Add to wall-limited circles (slack > 0)
    wall_limited = slack > 0
    if wall_limited.any():
        add_amount = alpha * slack[wall_limited]
        new_radii[wall_limited] += add_amount
    
    # Subtract from neighbor-limited circles (slack = 0)
    neighbor_limited = slack == 0
    if neighbor_limited.any():
        total_to_subtract = add_amount.sum() if wall_limited.any() else 0
        if total_to_subtract > 0:
            weights = radii[neighbor_limited]
            weights = weights / weights.sum()
            subtract_amounts = total_to_subtract * weights
            new_radii[neighbor_limited] -= subtract_amounts
            # Ensure no negative radii
            new_radii[new_radii < 0] = 0
    return new_radii

def push_away(centers, radii, beta, eps):
    n = centers.shape[0]
    new_centers = centers.copy()
    for i in range(n):
        c_i = new_centers[i]
        r_i = radii[i]
        for j in range(n):
            if i != j:
                c_j = centers[j]
                r_j = radii[j]
                v_ij = c_i - c_j
                d_ij = np.sqrt(np.sum(v_ij ** 2))
                if d_ij < r_i + r_j + eps:
                    delta = (r_i + r_j + eps - d_ij)
                    new_centers[i] += beta * delta * (v_ij / (d_ij + eps))
    
    # Clamp to interior
    new_centers = np.clip(new_centers, eps, 1 - eps)
    return new_centers

def iterate(centers, radii, alpha, beta, max_iter):
    for _ in range(max_iter):
        old_sum = radii.sum()
        # Compute slack
        D = cdist(centers, centers)
        np.fill_diagonal(D, np.inf)
        nn_distances = D.min(axis=1)
        wall_distances = np.array([
            centers[:, 0], 1 - centers[:, 0], centers[:, 1], 1 - centers[:, 1]
        ]).min(axis=0)
        r_max = np.minimum(wall_distances, 0.5 * nn_distances)
        slack = 0.5 * nn_distances - radii
        
        # Redistribution
        new_radii = redistribute_radii(radii, slack, alpha)
        
        # Push away
        new_centers = push_away(centers, new_radii, beta, 1e-8)
        
        # Recalculate radii
        radii = compute_feasible_radii(new_centers)
        sum_radii = radii.sum()
        
        # Convergence check
        if sum_radii - old_sum < 1e-5:
            break
        centers = new_centers
        
    return centers, radii

def repair(centers, radii, eps_final):
    # Slight shrink
    radii = np.maximum(0, radii - eps_final)
    # Clip centers
    centers = np.clip(centers, eps_final, 1 - eps_final)
    # Ensure no overlaps
    D = cdist(centers, centers)
    np.fill_diagonal(D, np.inf)
    for i in range(centers.shape[0]):
        for j in range(i + 1, centers.shape[0]):
            if D[i, j] < radii[i] + radii[j] + 1e-12:
                overlap = (radii[i] + radii[j] - D[i, j]) / 2 + eps_final
                radii[i] -= overlap
                radii[j] -= overlap
    # Final clamping
    radii = np.maximum(0, radii)
    centers = np.clip(centers, 1e-9, 1 - 1e-9)
    return centers, radii

def run_packing():
    n = 26
    alpha = 0.4
    beta = 0.3
    max_iter = 200
    eps_final = 5e-10
    
    centers = initialize_seeds(n)
    radii = compute_feasible_radii(centers)
    
    # Iterative improvement
    centers, radii = iterate(centers, radii, alpha, beta, max_iter)
    
    # Final repair
    centers, radii = repair(centers, radii, eps_final)
    
    sum_radii = radii.sum()
    return centers, radii, sum_radii
