import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Initial layout using hexagonal lattice
    centers, radii = initial_lattice(26)
    
    # Flatten for optimization
    z0 = pack(centers, radii)
    
    # Define bounds: [x0,y0,r0,x1,y1,r1,...]
    bounds = []
    for i in range(26):
        bounds.append((0, 1))  # x bounds
        bounds.append((0, 1))  # y bounds
        bounds.append((0, 0.5))  # r bounds
    
    # Define constraints
    constraints = []
    
    # Boundary constraints
    constraints.extend(constraint_boundary(z0))
    
    # Non-overlap constraints
    constraints.extend(constraint_nonoverlap(z0))
    
    # Objective: minimize negative sum of radii
    def objective(z):
        _, radii = unpack(z)
        return -np.sum(radii)
    
    # Run optimization
    result = minimize(objective, z0, method='SLSQP', bounds=bounds, 
                      constraints=constraints, options={'maxiter': 500, 'ftol': 1e-9}, 
                      disp=False)
    
    # Unpack the result
    centers, radii = unpack(result.x)
    
    # Repair if needed
    centers, radii = repair(centers, radii)
    
    # Final check
    is_valid, _ = validate_packing(centers, radii)
    if not is_valid:
        raise RuntimeError("Final packing is not valid after repair")
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii

def initial_lattice(n_circles):
    # Generate hexagonal lattice points
    # We'll use a triangular lattice with spacing s
    s = 0.25  # Initial guess for lattice spacing
    phi = np.sqrt(3) / 2.0
    pts = []
    for i in range(100):
        for j in range(100):
            x = i * s
            y = j * s * phi
            if i % 2 == 1:
                y += s * phi / 2
            if 0 <= x <= 1 and 0 <= y <= 1:
                pts.append([x, y])
            if len(pts) >= n_circles:
                break
        if len(pts) >= n_circles:
            break
    
    pts = np.array(pts[:n_circles])
    
    # Compute admissible radii
    radii = []
    for i in range(len(pts)):
        x, y = pts[i]
        min_dist_to_boundary = min(x, 1-x, y, 1-y)
        
        min_dist_to_others = float('inf')
        for j in range(len(pts)):
            if i != j:
                dist = np.sqrt((pts[i][0] - pts[j][0])**2 + (pts[i][1] - pts[j][1])**2)
                min_dist_to_others = min(min_dist_to_others, dist)
        
        r = min(min_dist_to_boundary, min_dist_to_others / 2.0)
        r = max(0, min(0.5, r))  # No negative, no oversize
        radii.append(r)
    
    return pts, np.array(radii)

def unpack(z):
    n = len(z) // 3
    centers = z[0:2*n].reshape(n, 2)
    radii = z[2*n:3*n]
    return centers, radii

def pack(centers, radii):
    z = np.zeros(3 * len(centers))
    z[0:2*len(centers)] = centers.flatten()
    z[2*len(centers):3*len(centers)] = radii
    return z

def constraint_boundary(z):
    centers, radii = unpack(z)
    constraints = []
    for i in range(len(centers)):
        x, y = centers[i]
        r = radii[i]
        # Left boundary: r <= x
        constraints.append(lambda z, i=i: z[2*i] - z[2*i + 2])  # r - x <= 0
        constraints.append(lambda z, i=i: 1 - z[2*i] - z[2*i + 2])  # 1 - x - r <= 0
        constraints.append(lambda z, i=i: z[2*i + 1] - z[2*i + 2])  # r - y <= 0
        constraints.append(lambda z, i=i: 1 - z[2*i + 1] - z[2*i + 2])  # 1 - y - r <= 0
    return constraints

def constraint_nonoverlap(z):
    centers, radii = unpack(z)
    constraints = []
    n = len(centers)
    for i in range(n):
        for j in range(i+1, n):
            # Define a function that computes: (r_i + r_j) - distance
            def nonoverlap_lambda(i, j):
                def c(z):
                    x1, y1 = z[2*i], z[2*i + 1]
                    x2, y2 = z[2*j], z[2*j + 1]
                    r1 = z[2*n + i]
                    r2 = z[2*n + j]
                    dist = np.hypot(x1 - x2, y1 - y2)
                    return r1 + r2 - dist
                return c
            constraints.append(nonoverlap_lambda(i, j))
    return constraints

def repair(centers, radii):
    n = len(centers)
    
    # Apply boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        r = min(r, x, 1 - x, y, 1 - y)
        radii[i] = max(0, r)
    
    # Resolve overlaps iteratively
    max_iter = 10
    for _ in range(max_iter):
        any_overlap = False
        for i in range(n):
            for j in range(i+1, n):
                dist = np.sqrt((centers[i][0] - centers[j][0])**2 + (centers[i][1] - centers[j][1])**2)
                r_sum = radii[i] + radii[j]
                if dist < r_sum:
                    overlap = r_sum - dist
                    shrink = overlap / 2.0
                    radii[i] = max(0, radii[i] - shrink)
                    radii[j] = max(0, radii[j] - shrink)
                    any_overlap = True
        if not any_overlap:
            break
    
    # Final boundary correction
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        r = min(r, x, 1 - x, y, 1 - y)
        radii[i] = max(0, r)
    
    return centers, radii
