import numpy as np
from scipy.optimize import minimize

def hex_lattice_points(n, d):
    """Generate at least n points in a hexagonal lattice within [0,1]x[0,1]."""
    sqrt3 = np.sqrt(3)
    rows = []
    y = 0.0
    while len(rows) < n:
        row = []
        x = 0.0
        while x <= 1.0:
            row.append((x, y))
            x += d
        rows.append(row)
        y += d * sqrt3 / 2
        if len(rows) >= 2:
            # Offset every other row
            if len(rows) % 2 == 0:
                for i in range(len(rows[-1])):
                    rows[-1][i] = (rows[-1][i][0] + d/2, rows[-1][i][1])
    points = []
    for row in rows:
        for x, y in row:
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append((x, y))
        if len(points) >= n:
            break
    return np.array(points[:n])

def initial_guess():
    """Generate an initial configuration that fits 26 circles."""
    # Hexagonal lattice
    d = 0.2  # initial spacing
    points = hex_lattice_points(26, d)

    # Compute minimal safe radius
    min_radius = 1.0
    for x, y in points:
        r_wall = min(x, 1-x, y, 1-y)
        min_radius = min(min_radius, r_wall)
    
    # Set all radii to a uniform safe value, with some buffer to avoid issues
    r0 = min_radius * 0.95

    # Create the initial parameter vector [x0, y0, r0, x1, y1, r1, ...]
    x_vals = points[:, 0]
    y_vals = points[:, 1]
    r_vals = np.full(26, r0)
    
    # Pack into a 1D array
    vars0 = np.empty(3 * 26)
    vars0[0::3] = x_vals
    vars0[1::3] = y_vals
    vars0[2::3] = r_vals

    return vars0

def objective(vars):
    # We're minimizing the negative sum of radii
    return -np.sum(vars[2::3])

def wall_constraints(vars):
    """Ensure all circles are inside the unit square."""
    constraints = []
    n = len(vars) // 3
    for i in range(n):
        x = vars[3*i + 0]
        y = vars[3*i + 1]
        r = vars[3*i + 2]
        # x - r >= 0  --> r - x <= 0
        constraints.append(r - x)
        # x + r <= 1  --> x + r - 1 <= 0
        constraints.append(x + r - 1)
        # y - r >= 0  --> r - y <= 0
        constraints.append(r - y)
        # y + r <= 1  --> y + r - 1 <= 0
        constraints.append(y + r - 1)
    return np.array(constraints)


def nonoverlap_constraints(vars):
    """Non-overlap constraints."""
    constraints = []
    n = len(vars) // 3
    for i in range(n):
        for j in range(i + 1, n):
            x_i = vars[3*i + 0]
            y_i = vars[3*i + 1]
            r_i = vars[3*i + 2]
            x_j = vars[3*j + 0]
            y_j = vars[3*j + 1]
            r_j = vars[3*j + 2]
            # Distance squared - (radii sum) squared
            dist_sq = (x_i - x_j)**2 + (y_i - y_j)**2
            rad_sum_sq = (r_i + r_j)**2
            constraints.append(dist_sq - rad_sum_sq)
    return np.array(constraints)

def run_packing():
    """Run the packing optimization and return centers, radii, and sum."""
    # Get initial guess
    vars0 = initial_guess()
    n = 26

    # Define bounds: x in [0,1], y in [0,1], r in [0,0.5]
    bounds = []
    for i in range(n):
        bounds.append((0, 1))  # x
        bounds.append((0, 1))  # y
        bounds.append((0, 0.5))  # r

    # Define constraints
    constr_wall = {'type': 'ineq', 'fun': wall_constraints}
    constr_overlap = {'type': 'ineq', 'fun': nonoverlap_constraints}

    # Optimize
    result = minimize(objective, vars0, method='SLSQP',
                      bounds=bounds, constraints=[constr_wall, constr_overlap],
                      options={'maxiter': 2000, 'ftol': 1e-10})

    if not result.success:
        # Fall back to initial guess
        vars = vars0
    else:
        vars = result.x

    # Extract centers and radii
    centers = np.empty((n, 2))
    centers[:, 0] = vars[0::3]
    centers[:, 1] = vars[1::3]
    radii = vars[2::3]
