import numpy as np
from scipy.optimize import linprog
from math import sqrt

def initial_hex_centers(k, spacing):
    """Generate first k points from hexagonal lattice scaled to unit square."""
    sqrt3 = sqrt(3)
    y_step = spacing * sqrt3 / 2
    x_step = spacing
    centers = []
    i = 0
    j = 0
    while len(centers) < k:
        x = i * x_step + (j % 2) * x_step / 2
        y = j * y_step
        if 0 <= x <= 1 and 0 <= y <= 1:
            centers.append([x, y])
        if x > 1:
            j += 1
            i = 0
        else:
            i += 1
        if j > 20:  # Safety check
            break
    return np.array(centers[:k])

def initial_radii(k, r0):
    """Return a length-k array filled with r0."""
    return np.full(k, r0)

def build_lp_constants(centers, radii):
    """Build wall and pairwise distances."""
    n = centers.shape[0]
    # Wall distances
    d_left = centers[:, 0]
    d_right = 1 - centers[:, 0]
    d_bottom = centers[:, 1]
    d_top = 1 - centers[:, 1]
    # Pairwise distances
    D_ij = np.zeros((n, n))
    for i in range(n):
        for j in range(i+1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            D_ij[i, j] = dist
            D_ij[j, i] = dist
    
    return {
        'd_left': d_left,
        'd_right': d_right,
        'd_bottom': d_bottom,
        'd_top': d_top,
        'D_ij': D_ij
    }

def solve_radius_lp(consts, radii, graph, alpha=0.5):
    """Solve the LP using scipy linprog."""
    n = len(radii)
    # Variables: Δr_i
    c = np.ones(n)  # maximize sum of Δr_i
    # All constraints are of type A_ub @ x <= b_ub
    
    A_ub = []
    b_ub = []
    
    # Wall constraints
    for i in range(n):
        max_delta = min(consts['d_left'][i], consts['d_right'][i],
                        consts['d_bottom'][i], consts['d_top'][i]) - radii[i]
        if max_delta < 0:
            max_delta = 0
        A_ub.append([-1.0])  # -Δr_i <= -max_delta => -Δr_i <= b_ub[i]
        b_ub.append(-max_delta)
    
    # Pairwise constraints
    for i in range(n):
        for j in range(i+1, n):
            bound = consts['D_ij'][i, j] - (radii[i] + radii[j])
            if bound > 0:
                A_ub.append([-1.0, -1.0])  # -Δr_i - Δr_j <= -bound
                b_ub.append(-bound)
    
    # Optional graph-based constraints
    for i in range(n):
        neighbors = graph.get(i, [])
        if neighbors:
            # Graph-based constraint
            total_capacity = 0
            for j in neighbors:
                total_capacity += consts['D_ij'][i, j] - (radii[i] + radii[j])
            total_capacity *= alpha
            A_ub.append([-1.0])  # -Δr_i <= -total_capacity
            b_ub.append(-total_capacity)
    
    # Solve
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'presolve': True})
        if res.success:
            return res.x
        else:
            # Infeasible, return all zeros (leave as is)
            return np.zeros(n)
    except:
        return np.zeros(n)

def apply_increment(radii, Δr):
    """Add Δr to radii."""
    return radii + Δr

def push_feasibility(centers, radii):
    """A deterministic, exact push to restore feasibility."""
    n = len(radii)
    # Ensure no wall violations by explicitly clamping the radii
    for i in range(n):
        max_r = min(centers[i][0], 1 - centers[i][0],
                    centers[i][1], 1 - centers[i][1])
        if radii[i] > max_r:
            radii[i] = max_r
    
    # Now correct pairwise overlaps: each simple push of center positions
    # along the normal direction to push them apart.
    changed = True
    while changed:
        changed = False
        for i in range(n):
            for j in range(i+1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                r_sum = radii[i] + radii[j]
                if dist < r_sum:
                    overlap = r_sum - dist
                    # Small safety margin
                    overlap += 1e-12
                    if overlap > 0:
                        changed = True
                        # Move center i away from j, and j away from i
                        dx = centers[i][0] - centers[j][0]
                        dy = centers[i][1] - centers[j][1]
                        norm = sqrt(dx*dx + dy*dy)
                        if norm > 1e-12:
                            dx /= norm
                            dy /= norm
                            # Push each by half the overlap
                            push_amount = overlap / 2
                            centers[i] += np.array([dx, dy]) * push_amount
                            centers[j] -= np.array([dx, dy]) * push_amount
                            # Re-check wall constraints
                            for k in range(n):
                                max_r = min(centers[k][0], 1 - centers[k][0],
                                            centers[k][1], 1 - centers[k][1])
                                if radii[k] > max_r:
                                    radii[k] = max_r
    
    # Return updated centers and radii
    return centers, radii

def update_graph(graph, new_edges, wall_contacts):
    """Update graph with new edges, preserves wall contact information."""
    for i, j in new_edges:
        if i not in graph:
            graph[i] = []
        if j not in graph:
            graph[j] = []
        if j not in graph[i]:
            graph[i].append(j)
        if i not in graph[j]:
            graph[j].append(i)
    for i, wall in wall_contacts.items():
        if i not in graph:
            graph[i] = []
        graph[i] = list(set(graph[i]) | set(wall))
    return graph

def run_iterations(centers, radii, max_iter=200, ε=1e-5, α=0.5):
    """Main loop of the iterative scanning algorithm."""
    graph = {}
    for step in range(max_iter):
        # Build constants for this iteration
        consts = build_lp_constants(centers, radii)
        # Solve new Δr
        Δr = solve_radius_lp(consts, radii, graph, α)
        # Update radii
        old_radii = radii.copy()
        radii = apply_increment(radii, Δr)
        # Push feasibility
        centers, radii = push_feasibility(centers, radii)
        
        # Check change in sum
        delta_sum = np.sum(Δr)
        if delta_sum < ε:
            break
    return centers, radii

def final_refinement(centers, radii):
    """Final iteration: high precision LP and correct final bounds."""
    # High precision solve
    try:
        consts = build_lp_constants(centers, radii)
        n = len(radii)
        c = np.ones(n)
        A_ub = []
        b_ub = []
        # Wall constraints
        for i in range(n):
            max_delta = min(consts['d_left'][i], consts['d_right'][i],
                            consts['d_bottom'][i], consts['d_top'][i]) - radii[i]
            if max_delta < 0:
                max_delta = 0
            A_ub.append([-1.0])
            b_ub.append(-max_delta)
        # Pairwise
        for i in range(n):
            for j in range(i+1, n):
                bound = consts['D_ij'][i, j] - (radii[i] + radii[j])
                if bound > 0:
                    A_ub.append([-1.0, -1.0])
                    b_ub.append(-bound)
        # Solve
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', 
                      options={'presolve': True, 'time_limit': 10.0})
        if res.success:
            Δr = res.x
            new_radii = apply_increment(radii, Δr)
            # Final deterministic step
            for i in range(len(new_radii)):
                new_radii[i] = min(new_radii[i],
                                   centers[i][0],
                                   1 - centers[i][0],
                                   centers[i][1],
                                   1 - centers[i][1])
            # Final collision checks
            for i in range(len(new_radii)):
                for j in range(i+1, len(new_radii)):
                    dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                    r_sum = new_radii[i] + new_radii[j]
                    if dist < r_sum:
                        # Adjust the larger one
                        if new_radii[i] >= new_radii[j]:
                            new_radii[i] -= (r_sum - dist) + 1e-13
                        else:
                            new_radii[j] -= (r_sum - dist) + 1e-13
            return centers, new_radii
        else:
            # fallback to original
            return centers, radii
    except:
        return centers, radii
