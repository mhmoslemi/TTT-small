import numpy as np
from scipy.optimize import linprog

def initial_hex_centers(k, spacing):
    sqrt3 = np.sqrt(3)
    centers = []
    i = j = 0
    while len(centers) < k:
        x = i * spacing + (j % 2) * spacing / 2
        y = j * spacing * sqrt3 / 2
        if 0 <= x <= 1 and 0 <= y <= 1:
            centers.append([x, y])
        if x + spacing > 1:
            j += 1
            i = 0
        else:
            i += 1
    return np.array(centers[:k])

def initial_radii(k, r0):
    return np.full(k, r0)

def build_lp_constants(centers, radii):
    n = centers.shape[0]
    d_left = centers[:, 0]
    d_right = 1 - centers[:, 0]
    d_bottom = centers[:, 1]
    d_top = 1 - centers[:, 1]
    d_wall = np.stack([d_left, d_right, d_bottom, d_top], axis=1)
    d_wall_min = np.min(d_wall, axis=1)
    # Compute pairwise distances
    D = np.zeros((n, n))
    for i in range(n):
        for j in range(i+1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            D[i, j] = D[j, i] = dist
    return {
        'd_wall_min': d_wall_min,
        'D': D,
        'radii': radii
    }

def solve_radius_lp(consts, radii, graph, alpha):
    n = len(radii)
    # Build constraints matrix and bounds
    # Variables: [Δr_0, ..., Δr_{n-1}]
    # Wall constraints
    A_eq = []
    b_eq = []
    # Total number of constraints: 4*n (walls) + n*(n-1)/2 (pairs)
    # Wall capacity
    for i in range(n):
        Ai = np.zeros(n)
        Ai[i] = 1
        A_eq.append(Ai)
        b_eq.append(consts['d_wall_min'][i] - radii[i])
    # Pairwise constraints: Δr_i + Δr_j <= D_ij - (r_i + r_j)
    for i in range(n):
        for j in range(i+1, n):
            if consts['D'][i, j] > radii[i] + radii[j] + 1e-12:
                Ai = np.zeros(n)
                Ai[i] = 1
                Ai[j] = 1
                A_eq.append(Ai)
                b_eq.append(consts['D'][i, j] - (radii[i] + radii[j]))
    A_eq = np.array(A_eq)
    b_eq = np.array(b_eq)
    c = np.ones(n)  # maximize sum of Δr_i
    try:
        res = linprog(-c, A_ub=-A_eq, b_ub=-b_eq, method='highs')
        if res.success:
            Δr = res.x
        else:
            Δr = np.zeros(n)
    except:
        Δr = np.zeros(n)
    return Δr

def apply_increment(radii, Δr):
    return radii + Δr

def push_feasibility(centers, radii):
    # Wall constraints
    pushed_centers = centers.copy()
    pushed_radii = radii.copy()
    # handle wall contacts (exact, always safe)
    for i in range(centers.shape[0]):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            pushed_radii[i] = x
        elif x + r > 1:
            pushed_radii[i] = 1 - x
        if y - r < 0:
            pushed_radii[i] = min(pushed_radii[i], y)
        elif y + r > 1:
            pushed_radii[i] = min(pushed_radii[i], 1 - y)

    # pairwise overlaps
    for i in range(centers.shape[0]):
        for j in range(i + 1, centers.shape[0]):
            r_i = pushed_radii[i]
            r_j = pushed_radii[j]
            c_i = pushed_centers[i]
            c_j = pushed_centers[j]
            dist = np.sqrt(np.sum((c_i - c_j) ** 2))
            if dist < r_i + r_j - 1e-12:
                delta = r_i + r_j - dist
                #-direction vector
                dx = c_i[0] - c_j[0]
                dy = c_i[1] - c_j[1]
                dist_direct = np.sqrt(dx*dx + dy*dy)
                if dist_direct > 1e-14:  # avoid division by zero
                    d_x = dx / dist_direct
                    d_y = dy / dist_direct
                    # move each circle away from the other by half of overlap
                    moved_i = [c_i[0] - d_x * delta/2, c_i[1] - d_y * delta/2]
                    moved_j = [c_j[0] + d_x * delta/2, c_j[1] + d_y * delta/2]
                    pushed_centers[i] = np.array(moved_i)
                    pushed_centers[j] = np.array(moved_j)

    # Wall check again after potential moves
    for i in range(centers.shape[0]):
        x = pushed_centers[i][0]
        y = pushed_centers[i][1]
        r = pushed_radii[i]
        if x - r < 0:
            pushed_radii[i] = x
        elif x + r > 1:
            pushed_radii[i] = 1 - x
        if y - r < 0:
            pushed_radii[i] = min(pushed_radii[i], y)
        elif y + r > 1:
            pushed_radii[i] = min(pushed_radii[i], 1 - y)

    return pushed_centers, pushed_radii

def run_iterations(centers, radii, max_iter=200, ε=1e-5, alpha=0.5):
    # initialization
    graph = {}  # holds edges
    for i in range(len(radii)):
        graph[i] = []

    for iteration in range(max_iter):
        consts = build_lp_constants(centers, radii)
        Δr = solve_radius_lp(consts, radii, graph, alpha)
        old_sum = np.sum(radii)
        radii = apply_increment(radii, Δr)
        new_sum = np.sum(radii)
        if new_sum - old_sum < ε:
            break
        # geometric push of circles
        centers, radii = push_feasibility(centers, radii)

    return centers, radii

def final_refinement(centers, radii):
    consts = build_lp_constants(centers, radii)
    Δr = solve_radius_lp(consts, radii, {}, 0.5)
    radii = apply_increment(radii, Δr)
    centers, radii = push_feasibility(centers, radii)
    # Final tight check.
    # for each circle, clamp radii to wall distance
    for i in range(len(radii)):
        x, y = centers[i]
        r = radii[i]
        r = min(r, x, 1 - x, y, 1 - y)
        # Adjust other radii if needed
        radii[i] = r
    return centers, radii

def run_packing():
    # Initial seed
    spacing = 0.12
    k = 26
    centers = initial_hex_centers(k, spacing)
    radii = initial_radii(k, 0.005)
    # Run iterations
    centers, radii = run_iterations(centers, radii, max_iter=100)
    # Final refinement
    centers, radii = final_refinement(centers, radii)
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
