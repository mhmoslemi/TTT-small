import numpy as np
from scipy.optimize import minimize

def hex_lattice_points(n, d):
    """Generate at least n points in a hexagonal lattice with spacing d."""
    rows = []
    y = 0.0
    row_idx = 0
    while len(rows) < n:
        # Alternate offset
        x_offset = (row_idx % 2) * d / 2
        x = x_offset
        while x <= 1.0 + d:
            rows.append((x, y))
            x += d
        y += d * np.sqrt(3) / 2
        row_idx += 1
    return np.array(rows[:n])

def initial_guess():
    """Generate initial layout using hexagonal lattice and uniform radii."""
    d = 0.2  # initial lattice spacing
    points = hex_lattice_points(26, d)
    
    # Center the lattice
    x_center = np.mean(points[:, 0])
    y_center = np.mean(points[:, 1])
    points[:, 0] -= x_center
    points[:, 1] -= y_center
    points[:, 0] += 0.5
    points[:, 1] += 0.5

    # Compute max radius such that all circles fit in unit square
    min_dist_to_wall = min(
        np.min(points[:, 0]), np.min(1 - points[:, 0]),
        np.min(points[:, 1]), np.min(1 - points[:, 1])
    )
    r0 = min_dist_to_wall * 0.9  # give small margin

    # Add margin to ensure valid placement
    r0 = min(r0, 0.5)  # cannot exceed 0.5 (max possible radius)

    # Create initial variable vector [x0, y0, r0, x1, y1, r1, ...]
    vars = []
    for x, y in points:
        vars.extend([x, y, r0])
    return np.array(vars)

def objective(vars):
    """Minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def wall_constraints(vars):
    """Return wall constraint violations (should be non-negative)."""
    n = len(vars) // 3
    result = []
    for i in range(n):
        x = vars[3*i]
        y = vars[3*i+1]
        r = vars[3*i+2]
        # r - x <= 0   (-x + r <= 0)
        result.append(-x + r)
        # x + r - 1 <= 0
        result.append(x + r - 1)
        # r - y <= 0
        result.append(-y + r)
        # y + r - 1 <= 0
        result.append(y + r - 1)
    return np.array(result)

def nonoverlap_constraints(vars):
    """Return non-overlap constraint violations (should be non-negative)."""
    n = len(vars) // 3
    result = []
    for i in range(n):
        for j in range(i + 1, n):
            x_i = vars[3*i]
            y_i = vars[3*i+1]
            r_i = vars[3*i+2]
            x_j = vars[3*j]
            y_j = vars[3*j+1]
            r_j = vars[3*j+2]
            dist_sq = (x_i - x_j)**2 + (y_i - y_j)**2
            radii_sum_sq = (r_i + r_j)**2
            # (xi - xj)^2 + (yi - yj)^2 - (ri + rj)^2 >= 0
            result.append(dist_sq - radii_sum_sq)
    return np.array(result)

def run_packing():
    initial = initial_guess()
    bounds = []
    n = 26

    # Set bounds
    for i in range(n):
        bounds.append((0.0, 1.0))  # x
        bounds.append((0.0, 1.0))  # y
        bounds.append((0.0, 0.5))  # r (max radius is 0.5)
    
    # Prepare constraints
    cons = [
        {'type': 'ineq', 'fun': wall_constraints},
        {'type': 'ineq', 'fun': nonoverlap_constraints}
    ]

    # Minimize using SLSQP
    res = minimize(
        objective,
        initial,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 2000, 'ftol': 1e-10}
    )

    if not res.success:
        raise RuntimeError("Optimization failed, check initial layout")

    # Extract final centers and radii
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    for i in range(26):
        centers[i, 0] = res.x[3*i]
        centers[i, 1] = res.x[3*i+1]
        radii[i] = res.x[3*i+2]
        
    # Clip negative radii to 0 (numerical errors)
    radii = np.clip(radii, 0, None)

    # Validate the packing
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise AssertionError(f"Validation failed: {msg}")

    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
