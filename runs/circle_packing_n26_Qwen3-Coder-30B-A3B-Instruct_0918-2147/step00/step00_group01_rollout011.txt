import numpy as np
from scipy.optimize import differential_evolution
from scipy.spatial.distance import cdist

def compute_radii(centers):
    n = centers.shape[0]
    radii = np.zeros(n)
    
    # Wall distances
    x, y = centers[:, 0], centers[:, 1]
    wall_distances = np.array([
        x, 1 - x, y, 1 - y
    ])
    wall_min = np.min(wall_distances, axis=0)
    
    # Pairwise distances
    distances = cdist(centers, centers)
    np.fill_diagonal(distances, np.inf)
    nearest_distances = np.min(distances, axis=0)
    neighbor_distances = nearest_distances / 2
    
    # Radii are min of wall and neighbor distances
    radii = np.minimum(wall_min, neighbor_distances)
    
    return radii

def neg_sum_radii(z):
    centers = z.reshape(-1, 2)
    radii = compute_radii(centers)
    return -np.sum(radii)

def run_packing():
    np.random.seed(42)
    dim = 26 * 2
    bounds = [(0, 1)] * dim
    
    # Differential Evolution
    result_de = differential_evolution(
        neg_sum_radii,
        bounds,
        maxiter=2000,
        popsize=10,
        tol=1e-7,
        polish=False,
        strategy='best1bin',
        seed=42
    )
    
    # Refine with hill-climbing
    centers = result_de.x.reshape(-1, 2)
    best_centers = centers.copy()
    best_sum = -neg_sum_radii(result_de.x)
    sigma = 0.02
    min_sigma = 1e-5
    max_iter = 3000
    
    for _ in range(max_iter):
        candidate = best_centers + sigma * np.random.randn(*best_centers.shape)
        candidate = np.clip(candidate, 0, 1)
        candidate_z = candidate.flatten()
        candidate_sum = -neg_sum_radii(candidate_z)
        
        if candidate_sum > best_sum:
            best_centers = candidate
            best_sum = candidate_sum
            sigma = min(1.1 * sigma, 0.5)
        else:
            sigma = max(0.98 * sigma, min_sigma)
    
    # Final radii computation
    final_centers = best_centers
    radii = compute_radii(final_centers)
    sum_radii = np.sum(radii)
