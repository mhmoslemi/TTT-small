import numpy as np
from scipy.optimize import minimize

NUM_CIRCLES = 26
NUM_RESTARTS = 5
HEX_ROWS = 5
HEX_COLS = 6
HEX_SPACING = 0.2
HEX_OFFSET = 0.1

def hex_lattice_points(n_rows, spacing):
    points = []
    for i in range(n_rows):
        y = i * spacing * np.sqrt(3) / 2
        for j in range(HEX_COLS):
            if i % 2 == 0:
                x = j * spacing
            else:
                x = (j + 0.5) * spacing
            points.append([x, y])
    return np.array(points)[:NUM_CIRCLES]

def initial_guess(seed):
    np.random.seed(seed)
    points = hex_lattice_points(HEX_ROWS, HEX_SPACING)
    jitter = np.random.uniform(-0.02, 0.02, (NUM_CIRCLES, 2))
    centers = np.clip(points + jitter, 0, 1)
    radii = np.full(NUM_CIRCLES, 0.02)
    vars0 = np.zeros(NUM_CIRCLES * 3)
    vars0[0::3] = centers[:, 0]
    vars0[1::3] = centers[:, 1]
    vars0[2::3] = radii
    return vars0

def objective(vars):
    return -np.sum(vars[2::3])

def constraint_boundary(vars):
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    boundaries = np.array([
        x - r,
        1 - x - r,
        y - r,
        1 - y - r
    ])
    return boundaries.T.flatten()

def constraint_nonoverlap(vars):
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    n = NUM_CIRCLES
    margins = []
    for i in range(n):
        for j in range(i+1, n):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dist = np.sqrt(dx*dx + dy*dy)
            overlap = dist - (r[i] + r[j])
            margins.append(overlap)
    return np.array(margins)

def postprocess(vars):
    # Correct numerical violations
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Fix boundary violations
    for i in range(NUM_CIRCLES):
        if x[i] - r[i] < -1e-9:
            x[i] = r[i]
        if x[i] + r[i] > 1 + 1e-9:
            x[i] = 1 - r[i]
        if y[i] - r[i] < -1e-9:
            y[i] = r[i]
        if y[i] + r[i] > 1 + 1e-9:
            y[i] = 1 - r[i]
    
    # Resolve overlaps through small adjustments
    for i in range(NUM_CIRCLES):
        for j in range(i+1, NUM_CIRCLES):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dist = np.sqrt(dx*dx + dy*dy)
            overlap = dist - (r[i] + r[j])
            if overlap < -1e-9:  # overlap exists
                correction = -overlap / 2
                if r[i] > r[j]:
                    r[i] -= correction
                else:
                    r[j] -= correction
    return vars

def run_packing():
    best_sum = -np.inf
    best_vars = None
    bounds = []
    for i in range(NUM_CIRCLES):
        bounds.append((0, 1))  # x
        bounds.append((0, 1))  # y
        bounds.append((0, 0.5))  # r
    
    constraints = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_nonoverlap}
    ]
    
    for seed in range(NUM_RESTARTS):
        vars0 = initial_guess(seed)
        res = minimize(
            objective, 
            vars0,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 3000, 'ftol': 1e-7}
        )
        if res.success:
            vars_corr = postprocess(res.x)
            s = np.sum(vars_corr[2::3])
            if s > best_sum:
                best_sum = s
                best_vars = vars_corr.copy()

    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii = best_vars[2::3]
    return centers, radii, float(best_sum)
