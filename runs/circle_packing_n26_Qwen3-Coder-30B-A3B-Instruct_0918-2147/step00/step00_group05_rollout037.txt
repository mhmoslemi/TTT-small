import numpy as np
from scipy.optimize import minimize

def hex_lattice_points(n, d):
    """Generate at least n points in a hexagonal lattice with spacing d."""
    # Hexagonal lattice: rows offset by d/2 horizontally
    rows = []
    y = 0.0
    while len(rows) < n:
        row = []
        x = 0.0
        while x <= 1.0 + d:
            row.append((x, y))
            x += d
        rows.append(row)
        y += d * np.sqrt(3) / 2
        if len(rows) % 2 == 1:
            # Offset every other row
            for i in range(len(rows[-1])):
                rows[-1][i] = (rows[-1][i][0] + d/2, rows[-1][i][1])
        if len(rows) == 1:
            continue
        if len(rows) > 1 and len(rows[-1]) == 0:
            # last row had no points due to offset -> remove
            rows = rows[:-1]
    # flatten list of list of (x,y) to list of (x,y) 
    points = [p for row in rows for p in row]
    # Keep only those inside [0,1]^2
    points = [(x, y) for x, y in points if 0 <= x <= 1 and 0 <= y <= 1]
    return np.array(points[:n])

def initial_guess():
    """Generate a good initial starting point."""
    # Try different spacings to see how many points we get
    spacing = 0.15
    initial_points = hex_lattice_points(30, spacing)
    if initial_points.shape[0] < 26:
        # Need more points, adjust spacing
        spacing = 0.1
        initial_points = hex_lattice_points(30, spacing)
    
    # Trim to 26 points
    points = initial_points[:26]

    # Center the lattice
    min_x, max_x = points[:, 0].min(), points[:, 0].max()
    min_y, max_y = points[:, 1].min(), points[:, 1].max()
    center_x = (min_x + max_x) / 2
    center_y = (min_y + max_y) / 2
    shift_x = 0.5 - center_x
    shift_y = 0.5 - center_y
    points[:, 0] += shift_x
    points[:, 1] += shift_y

    # Now compute shadow radius r based on minimum to all borders
    min_r = np.inf
    for x, y in points:
        r_to_edge = min(x, 1-x, y, 1-y)
        if r_to_edge < min_r:
            min_r = r_to_edge
    # A safe uniform radius
    r0 = min_r * 0.99
    # Make radii all equal
    radii = np.full(26, r0)
    # Pack into 1D array [x0, y0, r0, x1, y1, r1, ...]
    vars = np.empty(3*26)
    vars[0::3] = points[:, 0]  # x
    vars[1::3] = points[:, 1]  # y
    vars[2::3] = radii         # r
    return vars

def objective(vars):
    return -np.sum(vars[2::3])

def wall_constraints(vars):
    """Wall constraints: x - r >= 0 and x + r <= 1, similar for y"""
    out = []
    n = 26
    for i in range(n):
        x, y, r = vars[3*i], vars[3*i+1], vars[3*i+2]
        out.append(r - x)  # x - r <= 0
        out.append(x + r - 1)  # x + r - 1 <= 0
        out.append(r - y)  # y - r <= 0
        out.append(y + r - 1)  # y + r - 1 <= 0
    return np.array(out)  # length = 4*n

def nonoverlap_constraints(vars):
    """Non-overlap constraints: (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2."""
    out = []
    n = 26
    for i in range(n):
        for j in range(i + 1, n):
            x_i, y_i, r_i = vars[3*i], vars[3*i+1], vars[3*i+2]
            x_j, y_j, r_j = vars[3*j], vars[3*j+1], vars[3*j+2]
            dist_sq = (x_i - x_j)**2 + (y_i - y_j)**2
            sum_radii_sq = (r_i + r_j)**2
            out.append(sum_radii_sq - dist_sq)
    return np.array(out)  # length = n*(n-1)/2

def run_packing():
    # Initial guess: hexagonal lattice + equal radii
    x0 = initial_guess()
    n = 26
    # Build bounds
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0))  # x
        bounds.append((0.0, 1.0))  # y
        bounds.append((0.0, 0.5))  # r (max possible in unit square)
    
    # Create constraints
    cons = []
    # Wall constraints
    cons.append({'type': 'ineq', 'fun': wall_constraints})
    # Non-overlap constraints
    cons.append({'type': 'ineq', 'fun': nonoverlap_constraints})
    
    # Optimization with SLSQP
    result = minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={
            'maxiter': 2000,
            'ftol': 1e-10,
            'disp': False
        }
    )
    # Extract centers and radii
    centers = np.column_stack((result.x[0::3], result.x[1::3]))
    radii = result.x[2::3]
    # If radii are slightly negative, clip to zero
    radii = np.maximum(radii, 0.0)
    # Final sum
    sum_radii = np.sum(radii)
    # Ensure validity
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise AssertionError(f"Post-optimization validation failed: {msg}")
    return (centers, radii, sum_radii)
