import numpy as np
import cvxpy as cp
from typing import Tuple, List
import itertools

def build_socp_problem(n: int) -> Tuple[cp.Problem, cp.Variable, List[cp.Parameter]]:
    """Builds a second-order cone program for maximum sum of radii."""
    r = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(r)), [])
    
    # Parameters for distances (to be updated each time)
    dist_params = []
    for i in range(n):
        for j in range(i+1, n):
            param = cp.Parameter(nonneg=True)
            dist_params.append(param)
    
    return prob, r, dist_params

def solve_radii(centres: np.ndarray, prob: cp.Problem, r: cp.Variable, 
                dist_params: List[cp.Parameter]) -> Tuple[np.ndarray, np.ndarray]:
    """Solves the SOCP for given centres to get optimal radii and duals."""
    n = centres.shape[0]
    
    # Compute pairwise distances and update params
    param_idx = 0
    constraints = []
    for i in range(n):
        x, y = centres[i]
        # Boundary constraints
        constraints.append(r[i] <= x)
        constraints.append(r[i] <= 1 - x)
        constraints.append(r[i] <= y)
        constraints.append(r[i] <= 1 - y)
        
        for j in range(i+1, n):
            diff = centres[i] - centres[j]
            dist = np.linalg.norm(diff)
            dist_params[param_idx].value = dist
            param_idx += 1
            constraints.append(r[i] + r[j] <= dist)
    
    prob.constraints = constraints
    prob.solve(solver=cp.ECOS, abstol=1e-12, reltol=1e-12)
    
    # Get the solution vector and duals
    radii = r.value
    duals = []
    # Extract dual variables for non-overlap constraints only
    if len(prob.constraints) > 4*n:
        for i, c in enumerate(prob.constraints[4*n:]):
            if hasattr(c, "dual_value") and c.dual_value is not None:
                duals.append(c.dual_value)
            else:
                duals.append(0.0)
    else:
        duals = [0.0] * len(dist_params)
    
    return radii, np.array(duals)

def energy(centres: np.ndarray, prob: cp.Problem, r: cp.Variable,
           dist_params: List[cp.Parameter]) -> float:
    """Computes the energy (negative sum of radii) for given centres."""
    radii, _ = solve_radii(centres, prob, r, dist_params)
    return -np.sum(radii)

def random_move(state: np.ndarray, rng: np.random.Generator,
                move_type_probs: List[float] = None) -> np.ndarray:
    """Applies one random move to a state (x0, y0, x1, y1 ...)."""
    if move_type_probs is None:
        move_type_probs = [0.7, 0.1, 0.1, 0.1]
    
    n = len(state) // 2
    new_state = state.copy()
    
    move_type = rng.choice(4, p=move_type_probs)
    
    if move_type == 0: # Single centre jitter
        idx = rng.integers(n)
        dx = rng.normal(0, 0.02)
        dy = rng.normal(0, 0.02)
        new_state[2*idx] = np.clip(new_state[2*idx] + dx, 0, 1)
        new_state[2*idx+1] = np.clip(new_state[2*idx+1] + dy, 0, 1)
    
    elif move_type == 1: # Cluster jitter
        num_centres = rng.integers(3, 6)
        indices = rng.choice(n, size=num_centres, replace=False)
        offset_x = rng.normal(0, 0.02)
        offset_y = rng.normal(0, 0.02)
        for idx in indices:
            new_state[2*idx] = np.clip(new_state[2*idx] + offset_x, 0, 1)
            new_state[2*idx+1] = np.clip(new_state[2*idx+1] + offset_y, 0, 1)
    
    elif move_type == 2: # Edge push (with low probability)
        idx = rng.integers(n)
        x, y = new_state[2*idx], new_state[2*idx+1]
        dx = 0.01 * (-1 if x > 0.5 else 1)
        dy = 0.01 * (-1 if y > 0.5 else 1)
        new_state[2*idx] = np.clip(new_state[2*idx] + dx, 0, 1)
        new_state[2*idx+1] = np.clip(new_state[2*idx+1] + dy, 0, 1)
        
    elif move_type == 3: # Swap order (any chance of directional stability)
        # Instead of swapping centers, we shift their ordering to reduce a bias in solver
        shake_factor = 0.01
        indices = rng.choice(n, size=n, replace=False)
        new_order = indices.copy()
        if rng.random() < 0.1: # Occasionally shuffle
            new_order = []
            for i in range(n):
                new_order.append(indices[i])
        next_indices = list(indices)
        for i in range(n):
            new_state[2*i] = state[2*next_indices[i]]
            new_state[2*i+1] = state[2*next_indices[i]+1]
    
    return new_state

def anneal_chain(seed: int, prob: cp.Problem, r: cp.Variable,
                 dist_params: List[cp.Parameter]) -> Tuple[np.ndarray, np.ndarray, float]:
    """Runs a single simulated annealing chain."""
    rng = np.random.default_rng(seed)
    
    # Start with a good seed configuration, e.g., hexagonal
    n = 26
    centres = np.zeros((n, 2))
    
    # Use a hexagonal packing with enough space
    k = 0
    rows = 6
    cols = 5
    pad_left = 0.05
    pad_bottom = 0.05
    cell_width = (1 - 2*pad_left)/cols
    cell_height = (1 - 2*pad_bottom)/rows
    
    for i in range(rows):
        y = pad_bottom + i * cell_height
        for j in range(cols):
            if k >= n:
                break
            x = pad_left + j * cell_width
            # Slight offset for even rows
            if i % 2 == 1:
                x = pad_left + 0.5 * cell_width + j * cell_width
            centres[k] = [x, y]
            k += 1
        if k >= n:
            break
    
    # Also try a corner start if possible
    # Place 4 circles in corners first, then others
    if n >= 4:
        centres[0] = [0.05, 0.05]      # bottom-left
        centres[1] = [0.95, 0.05]      # bottom-right
        centres[2] = [0.05, 0.95]      # top-left
        centres[3] = [0.95, 0.95]      # top-right
        # Place others in hexagonal layout
        k = 4
        rows = 6
        cols = 5
        pad_left = 0.05
        pad_bottom = 0.05
        cell_width = (1 - 2*pad_left)/cols
        cell_height = (1 - 2*pad_bottom)/rows
        for i in range(rows):
            y = pad_bottom + i * cell_height
            for j in range(cols):
                if k >= n:
                    break
                x = pad_left + j * cell_width
                if i % 2 == 1:
                    x = pad_left + 0.5 * cell_width + j * cell_width
                centres[k] = [x, y]
                k += 1
                if k >= n:
                    break
            if k >= n:
                break
    
    # Perturb the seed slightly
    for i in range(n):
        centres[i] += rng.normal(0, 0.005, 2)
        centres[i] = np.clip(centres[i], 0, 1)
    
    # Annealing parameters
    T0 = 0.05
    alpha = 0.995
    t = T0
    N_iter = 150000
    
    current_energy = energy(centres, prob, r, dist_params)
    
    best_centres = centres.copy()
    best_energy = current_energy
    
    for _ in range(N_iter):
        # Generate a new proposal state
        new_state = random_move(centres.flatten(), rng)
        new_centres = new_state.reshape(-1, 2)
        
        new_energy = energy(new_centres, prob, r, dist_params)
        delta = new_energy - current_energy
        
        # Accept or reject according to Metropolis criterion
        if delta < 0 or np.exp(-delta / t) > rng.random():
            centres = new_centres.copy()
            current_energy = new_energy
        
        # Update best found
        if current_energy < best_energy:
            best_energy = current_energy
            best_centres = centres.copy()
            
        # Cool down
        t *= alpha
        
        # Occasionally make large jumps for thorough search
        if rng.random() < 0.005:
            for i in range(0, len(new_state), 2):
                new_state[i] += rng.normal(0, 0.08)
                new_state[i + 1] += rng.normal(0, 0.08)
                new_state[i] = np.clip(new_state[i], 0, 1)
                new_state[i + 1] = np.clip(new_state[i + 1], 0, 1)
            
    radii, _ = solve_radii(best_centres, prob, r, dist_params)
    return best_centres, radii, -best_energy

def local_refine(centres: np.ndarray, radii: np.ndarray,

    # Start with current solution

            if abs(r_i - (1 - x)) < 1e-6:  # Right wall
                active_constraints.append(('boundary', i, 'right', r_i))
            if abs(r_i - y) < 1e-6:  # Bottom wall
                active_constraints.append(('boundary', i, 'bottom', r_i))
            if abs(r_i - (1 - y)) < 1e-6:  # Top wall
                active_constraints.append(('boundary', i, 'top', r_i))
                
            for j in range(i+1, n):
                # Non-overlapping constraints
                if abs(radii[i] + radii[j] - np.linalg.norm(centres[i] - centres[j])) < 1e-6:
                    active_constraints.append(('pairwise', i, j))
                    
        # If low number of active constraints, break early
        if len(active_constraints) < 2:
            break

        b_rows = []

def finalise(centres: np.ndarray, radii: np.ndarray,
            prob: cp.Problem, r: cp.Variable,
            dist_params: List[cp.Parameter]) -> Tuple[np.ndarray, np.ndarray]:
    """Final polishing and guarantee of feasibility."""
    # Final resolution with tight tolerances
    _, _ = solve_radii(centres, prob, r, dist_params)
    final_radii = r.value
    
    # Ensure feasibility: check boundary violations
    n = centres.shape[0]
    for i in range(n):
        x, y = centres[i]
        r_i = final_radii[i]
        # Check that radii are within boundaries (with tolerance)
        max_r_allowed = min(x, 1 - x, y, 1 - y)
        if r_i > max_r_allowed + 1e-13:
            # Clamp radius
            final_radii[i] = max_r_allowed
            
    # Final validation step if necessary:
    return centres, final_radii

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    """Main entry point of the algorithm."""
    n = 26
    
    # Build overall structure once
    prob, r, dist_params = build_socp_problem(n)
    
    # Try several annealing chains with different seeds
    best_result = None
    best_sum = -np.inf
    
    # Seed configurations  
    seeds = [100, 200, 300, 400, 500, 600, 700, 800]
    for seed in seeds:
        try:
            centres, radii, sum_rad = anneal_chain(seed, prob, r, dist_params) 
            centres_refined, radii_refined = local_refine(centres, radii, prob, r, dist_params)
            centres_final, radii_final = finalise(centres_refined, radii_refined, prob, r, dist_params)
            
            if np.sum(radii_final) > best_sum:
                best_sum = np.sum(radii_final)
                best_result = (centres_final, radii_final)
        except Exception as e:
            # Optionally log failure, but continue with next chain
            continue
    
    if best_result is None:
        raise RuntimeError("All chains failed to compute valid packing")
    
    centers, radii = best_result
    sum_radii = np.sum(radii)
    
    # Validate result
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise RuntimeError(f"Validated packing failed: {msg}")
    
    return centers, radii, sum_radii
