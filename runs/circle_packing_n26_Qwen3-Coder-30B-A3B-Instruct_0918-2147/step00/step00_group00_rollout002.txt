import numpy as np
from scipy.optimize import minimize

def make_hex_lattice(n, margin=0.05):
    # Generate a hexagonal lattice that fits within a square with margin
    dx = 1 / (np.ceil(np.sqrt(n)) + 0.5)
    dy = dx * np.sqrt(3) / 2

    points = []
    for j in range(int(np.ceil(n / (1 / dx)))):
        for i in range(int(np.ceil(1 / dx))):
            x = i * dx + (j % 2) * dx / 2
            y = j * dy
            if x - margin >= 0 and x + margin <= 1 and y - margin >= 0 and y + margin <= 1:
                points.append([x, y])
            if len(points) >= n:
                break
        if len(points) >= n:
            break

    points = np.array(points[:n])
    # Center the pattern
    center = np.mean(points, axis=0)
    offset = 0.5 - center
    points += offset
    return points

def objective(z):
    # z is [x0, y0, r0, x1, y1, r1, ...]
    radii = z[2::3]
    return -np.sum(radii)

def boundary_constraints(z):
    n = len(z) // 3
    constraints = []
    for i in range(n):
        x, y, r = z[3*i], z[3*i+1], z[3*i+2]
        constraints.extend([
            x - r,             # x - r >= 0
            1 - x - r,         # 1 - x - r >= 0
            y - r,             # y - r >= 0
            1 - y - r          # 1 - y - r >= 0
        ])
    return np.array(constraints)

def pairwise_constraints(z):
    n = len(z) // 3
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            x1, y1, r1 = z[3*i], z[3*i+1], z[3*i+2]
            x2, y2, r2 = z[3*j], z[3*j+1], z[3*j+2]
            dist_sq = (x1 - x2)**2 + (y1 - y2)**2
            sum_radii_sq = (r1 + r2)**2
            constraints.append(dist_sq - sum_radii_sq)
    return np.array(constraints)

def solve_nlp(initial_z):
    n = len(initial_z) // 3
    # Define constraints
    bnds = []
    for i in range(n):
        bnds.extend([(0, 1), (0, 1), (0, 0.5)])
    bounds = tuple(bnds)
    
    cons = []
    # Boundary constraints
    cons.append({'type': 'ineq', 'fun': boundary_constraints})
    # Pairwise separation constraints
    cons.append({'type': 'ineq', 'fun': pairwise_constraints})
    
    result = minimize(objective, initial_z, method='SLSQP', bounds=bounds, constraints=cons, 
                      options={'maxiter': 1000, 'ftol': 1e-9})
    return result.x

def nudge_centers(centers, radii, step=0.001):
    new_centers = centers.copy()
    for i in range(len(centers)):
        x, y = centers[i]
        r = radii[i]
        # Check if touching any wall
        if abs(x - r) < 1e-6:  # Left wall
            new_centers[i, 0] += step
        elif abs(1 - x - r) < 1e-6:  # Right wall
            new_centers[i, 0] -= step
        if abs(y - r) < 1e-6:  # Bottom wall
            new_centers[i, 1] += step
        elif abs(1 - y - r) < 1e-6:  # Top wall
            new_centers[i, 1] -= step
    return new_centers

def run_packing():
    # Step 1: Create initial hexagonal lattice
    centers = make_hex_lattice(26)
    radii = np.full(26, 0.02)
    
    # Step 2: Initial guess for z = [x0, y0, r0, x1, y1, r1, ...]
    initial_z = np.zeros(3 * 26)
    initial_z[0::3] = centers[:, 0]    # x coordinates
    initial_z[1::3] = centers[:, 1]    # y coordinates
    initial_z[2::3] = radii            # radii

    # Step 3: Solve optimization
    final_z = solve_nlp(initial_z)
    
    # Step 4: Extract final centers and radii
    final_centers = np.column_stack((final_z[0::3], final_z[1::3]))
    final_radii = final_z[2::3]

    # Step 5: Iterative refinement
    for _ in range(5):
        # Nudge centers that are probably tight against walls
        final_centers = nudge_centers(final_centers, final_radii)
        # Re-solve with nudged centers
        reinit_z = np.zeros(3 * 26)
        reinit_z[0::3] = final_centers[:, 0]
        reinit_z[1::3] = final_centers[:, 1]
        reinit_z[2::3] = final_radii

        final_z = solve_nlp(reinit_z)
        final_centers = np.column_stack((final_z[0::3], final_z[1::3]))
        final_radii = final_z[2::3]
