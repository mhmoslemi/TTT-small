import numpy as np
from scipy.optimize import linprog
from scipy.spatial.distance import cdist

def generate_hex_lattice(h):
    sqrt3 = np.sqrt(3)
    points = []
    y = 0
    while y <= 1 + 1e-10:
        xstart = (y % (h * sqrt3)) / sqrt3 * h
        x = xstart
        while x <= 1 + 1e-10:
            points.append([x, y])
            x += h
        y += h * sqrt3 / 2
    return np.array(points)

def build_neighbour_list(centers, max_dist):
    dist = cdist(centers, centers)
    pairs = []
    n = centers.shape[0]
    for i in range(n):
        for j in range(i + 1, n):
            d = dist[i, j]
            if d <= max_dist:
                pairs.append((i, j, d))
    return pairs

def solve_radius_lp(selected_indices, R_wall, neighbour_list):
    # LP: maximize sum of radii
    # Variables: r_i for i in selected_indices
    # Constraints: r_i <= R_wall[i]
    #            : r_i + r_j <= d_ij if (i,j,d_ij) in neighbour_list and both in selected_indices
    
    n = len(selected_indices)
    if n == 0:
        return np.array([]), True

    # Variable indices mapping
    var_map = {i: idx for idx, i in enumerate(selected_indices)}
    
    # Objective: maximize sum of radii (minimize negative sum)
    c = -np.ones(n)
    
    # Bounds: 0 <= r_i <= R_wall[i]
    bounds = [(0, R_wall[i]) for i in selected_indices]
    
    # Inequality constraints A_ub @ x <= b_ub
    A_ub = []
    b_ub = []
    
    # Wall constraints
    for i in range(n):
        row = [0] * n
        row[i] = -1
        A_ub.append(row)
        b_ub.append(-R_wall[selected_indices[i]])
    
    for i, j, d in neighbour_list:
        if i in var_map and j in var_map:
            row = [0] * n
            row[var_map[i]] = 1
            row[var_map[j]] = 1
            A_ub.append(row)
            b_ub.append(d)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            return None, False
        return res.x, True
    except:
        return None, False

def local_continuous_refinement(centers, radii):
    # Refinement: improve on discrete match with continuous post-process
    
    # This is needed to polish solution and encourage continuity fit
    # Use scipy minimize with trust-constr + bounds to improve geometry
    
    # Define the problem
    # Variables: x_i, y_i, r_i (3*26 total) 
    # We want to maximize sum of r_i
    # subject to:
    # - 0 <= x_i <= 1
    # - 0 <= y_i <= 1  
    # - 0 <= r_i <= min(x_i, 1 - x_i, y_i, 1 - y_i)
    # - (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2 for i < j
    
    n = 26
    # Prepare all constraints and variables
    x0 = []
    bounds = []
    
    for i in range(n):
        x0.append(centers[i][0])
        x0.append(centers[i][1])
        x0.append(radii[i])
        # bounds for x,y coordinates  
        bounds.append((0, 1))
        bounds.append((0, 1))
        bounds.append((0, min(centers[i][0], 1 - centers[i][0], centers[i][1], 1 - centers[i][1]) ))
    
    # Define objective (minimize negative sum of radii)
    def objective(vars_):
        res = 0.0
        for i in range(n):
            res += vars_[3*i + 2]
        return -res # because we want to maximize
    
    # Constraints
    def con_wall(vars_):
        # r_i <= R_wall[i] for all 
        result = []
        for i in range(n):
            r_i = vars_[3*i + 2]
            x = vars_[3*i]
            y = vars_[3*i + 1]
            max_r = min(x, 1 - x, y, 1 - y)
            result.append(max_r - r_i)  # >= 0
        return np.array(result)
    
    def con_overlap(vars_):
        # (xi-xj)^2 + (yi-yj)^2 >= (ri + rj)^2 
        result = []
        for i in range(n):
            for j in range(i+1, n):
                xi, yi, ri = vars_[3*i], vars_[3*i+1], vars_[3*i+2]
                xj, yj, rj = vars_[3*j], vars_[3*j+1], vars_[3*j+2]
                d_sq = (xi - xj)**2 + (yi - yj)**2 
                s_sq = (ri + rj)**2
                result.append(d_sq - s_sq)  # >= 0
        return np.array(result)
    
    cons = [
        {'type': 'ineq', 'fun': con_wall},
        {'type': 'ineq', 'fun': con_overlap}
    ]
    
    # Solve minimization

        objective,
        x0,
        method='trust-constr',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 200, 'disp': False}
