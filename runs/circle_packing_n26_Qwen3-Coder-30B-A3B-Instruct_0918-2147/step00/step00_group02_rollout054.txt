import numpy as np
from scipy.optimize import minimize

def initial_lattice(n=26):
    # Generate hexagonal lattice points within unit square
    # Spacing chosen to allow at least n points
    spacing = 0.2
    sqrt3 = np.sqrt(3)
    rows = int(np.ceil(1.0 / (spacing * sqrt3 / 2)))
    cols = int(np.ceil(1.0 / spacing))
    centers = []
    for i in range(rows):
        y = i * spacing * sqrt3 / 2
        if y > 1.0:
            break
        offset = (i % 2) * spacing / 2
        for j in range(cols):
            x = j * spacing + offset
            if x > 1.0:
                break
            centers.append([x, y])
            if len(centers) >= n:
                break
        if len(centers) >= n:
            break
    centers = np.array(centers[:n])
    
    # Compute admissible radii
    radii = np.zeros(n)
    for i in range(n):
        x, y = centers[i]
        min_dist = min(x, 1-x, y, 1-y)
        for j in range(n):
            if i != j:
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                min_dist = min(min_dist, dist / 2)
        radii[i] = min(min_dist, 0.5)
    
    return centers, radii

def pack(centers, radii):
    z = np.empty(3 * len(centers))
    z[0::3] = centers[:, 0]
    z[1::3] = centers[:, 1]
    z[2::3] = radii
    return z

def unpack(z):
    n = len(z) // 3
    centers = z[0::3].reshape((n, 1)), z[1::3].reshape((n, 1))
    centers = np.hstack(centers)
    radii = z[2::3]
    return centers, radii

def objective(z):
    _, radii = unpack(z)
    return -np.sum(radii)

def constraint_boundary(z):
    centers, radii = unpack(z)
    c = []
    for i in range(len(centers)):
        x, y = centers[i]
        r = radii[i]
        # left boundary: r - x <= 0
        c.append(r - x)
        # right boundary: r + x - 1 <= 0
        c.append(r + x - 1)
        # bottom boundary: r - y <= 0
        c.append(r - y)
        # top boundary: r + y - 1 <= 0
        c.append(r + y - 1)
    return np.array(c)

def constraint_nonoverlap(z):
    centers, radii = unpack(z)
    c = []
    n = len(centers)
    for i in range(n):
        for j in range(i+1, n):
            x1, y1 = centers[i]
            x2, y2 = centers[j]
            r1 = radii[i]
            r2 = radii[j]
            dist = np.hypot(x1 - x2, y1 - y2)
            # non-overlap: r1 + r2 - dist <= 0
            c.append(r1 + r2 - dist)
    return np.array(c)

def repair(centers, radii):
    # Simple iterative repair
    n = len(centers)
    max_iter = 10
    for _ in range(max_iter):
        changed = False
        # Check boundaries
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            r_new = min(r, x, 1 - x, y, 1 - y)
            if r_new < r:
                radii[i] = r_new
                changed = True
        
        # Check overlaps
        for i in range(n):
            for j in range(i+1, n):
                x1, y1 = centers[i]
                x2, y2 = centers[j]
                r1 = radii[i]
                r2 = radii[j]
                dist = np.hypot(x1 - x2, y1 - y2)
                overlap = dist - (r1 + r2)
                if overlap < 0:
                    # Shrink circles by half of overlap
                    shrink = -overlap / 2
                    if r1 - shrink > 0 and r2 - shrink > 0:
                        radii[i] -= shrink
                        radii[j] -= shrink
                        changed = True
        if not changed:
            break
    return centers, radii

def run_packing():
    # First, generate a good initial placement
    centers, radii = initial_lattice(26)
    z0 = pack(centers, radii)
    
    # Set up constraints
    bounds = []
    for i in range(26):
        bounds.append((0.0, 1.0))  # x bounds
        bounds.append((0.0, 1.0))  # y bounds
        bounds.append((0.0, 0.5))  # r bounds
    
    constraints = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_nonoverlap}
    ]
    
    # Optimize
    sol = minimize(objective, z0, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter':500, 'ftol':1e-9, 'disp':False})
    
    final_centers, final_radii = unpack(sol.x)
    
    # Repair any constraint violations
    final_centers, final_radii = repair(final_centers, final_radii)
    
    # Validate
    valid, msg = validate_packing(final_centers, final_radii)
    if not valid:
        raise ValueError(f"Validation failed: {msg}")
    
    sum_radii = np.sum(final_radii)
    
    # Ensure we meet the required goal
    if sum_radii < 2.636:
        raise ValueError(f"Sum of radii {sum_radii} is below target 2.636")
    
    return final_centers, final_radii, sum_radii
