import numpy as np
from scipy.optimize import minimize

def initial_lattice(n=26):
    # Generate hexagonal lattice
    sqrt3 = np.sqrt(3)
    spacing = 1.0
    while True:
        rows = int(np.ceil(1.0 / (spacing * sqrt3 / 2)))
        cols = int(np.ceil(1.0 / spacing))
        total = rows * cols
        if total >= n:
            break
        spacing *= 0.99

    centers = []
    for i in range(rows):
        y = i * spacing * sqrt3 / 2
        if y > 1.0:
            break
        offset = spacing / 2 if i % 2 == 1 else 0
        for j in range(cols):
            x = j * spacing + offset
            if x > 1.0:
                break
            centers.append([x, y])

    centers = np.array(centers[:n])
    radii = np.full(n, 0.5)
    for i in range(n):
        cx, cy = centers[i]
        r = min(cx, 1 - cx, cy, 1 - cy)
        for j in range(n):
            if i != j:
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                r = min(r, dist / 2)
        radii[i] = max(0, r)
    
    return centers, radii

def objective(z):
    # z is flattened [x0, y0, r0, x1, y1, r1, ...]
    radii = z[2::3]
    return -np.sum(radii)

def pack(centers, radii):
    z = np.zeros(3 * centers.shape[0])
    z[0::3] = centers[:, 0]
    z[1::3] = centers[:, 1]
    z[2::3] = radii
    return z

def unpack(z):
    n = z.shape[0] // 3
    centers = np.zeros((n, 2))
    centers[:, 0] = z[0::3]
    centers[:, 1] = z[1::3]
    radii = z[2::3]
    return centers, radii

def constraint_boundary(z):
    centers, radii = unpack(z)
    constraints = []
    for i in range(centers.shape[0]):
        x, y = centers[i]
        r = radii[i]
        # left constraint
        constraints.append(x - r)
        # right constraint
        constraints.append(1 - x - r)
        # bottom constraint
        constraints.append(y - r)
        # top constraint
        constraints.append(1 - y - r)
    return np.array(constraints)

def constraint_nonoverlap(z):
    centers, radii = unpack(z)
    constraints = []
    n = centers.shape[0]
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            # We want dist >= radii[i] + radii[j] to avoid overlap
            # So constraint is dist - radii[i] - radii[j] >= 0
            constraints.append(dist - radii[i] - radii[j])
    return np.array(constraints)

def repair(centers, radii):
    # Repair constraints that might have been violated due to numerical errors
    n = centers.shape[0]
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        r = min(r, x, 1 - x, y, 1 - y)
        x = np.clip(x, r, 1 - r)
        y = np.clip(y, r, 1 - r)
        centers[i] = [x, y]
        radii[i] = r

    # Iteratively shrink overlapping circles
    iterations = 0
    while iterations < 10:
        changed = False
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                r_sum = radii[i] + radii[j]
                if dist < r_sum:
                    overlap = r_sum - dist
                    shrink = overlap / 2
                    radii[i] = max(0, radii[i] - shrink)
                    radii[j] = max(0, radii[j] - shrink)
                    changed = True
        if not changed:
            break
        iterations += 1
    return centers, radii

def run_packing():
    centers, radii = initial_lattice(26)
    z0 = pack(centers, radii)

    bounds = [(0, 1), (0, 1), (0, 0.5)] * 26
    constraints = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_nonoverlap}
    ]
    result = minimize(
        objective,
        z0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 500, 'ftol': 1e-9, 'disp': False}
    )

    centers, radii = unpack(result.x)
    centers, radii = repair(centers, radii)

    # Ensure final result is valid
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise ValueError(f"Packing not valid after optimization: {msg}")

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
