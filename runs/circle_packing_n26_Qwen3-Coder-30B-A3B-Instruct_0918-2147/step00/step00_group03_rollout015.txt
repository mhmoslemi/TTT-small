import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def init_candidate_grid(R, Δ):
    """Generate candidate (center, radius) pairs."""
    candidates = []
    for r in R:
        # Avoid placing circles too close to edges
        grid_spacing = r * np.sqrt(2)
        xmin, xmax = r, 1 - r
        ymin, ymax = r, 1 - r
        x_coords = np.arange(xmin, xmax + grid_spacing/2, grid_spacing)
        y_coords = np.arange(ymin, ymax + grid_spacing/2, grid_spacing)
        for x in x_coords:
            for y in y_coords:
                candidates.append((x, y, r))
    return candidates

def build_overlap_matrix(candidates):
    """Build boolean matrix of overlapping candidate pairs."""
    n = len(candidates)
    overlap = np.zeros((n, n), dtype=bool)
    for i in range(n):
        x1, y1, r1 = candidates[i]
        for j in range(i+1, n):
            x2, y2, r2 = candidates[j]
            dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
            if dist < r1 + r2:
                overlap[i, j] = True
                overlap[j, i] = True
    return overlap

def solve_discrete_bip(candidates, overlap_matrix, target_sum):
    """Use greedy approach to solve discrete BIP."""
    n = len(candidates)
    selected = []
    total_radius = 0.0
    used = np.zeros(n, dtype=bool)
    
    # Greedy selection starting with largest radius
    idxs = np.argsort([c[2] for c in candidates])[::-1]
    
    for i in idxs:
        if used[i]:
            continue
        if total_radius + candidates[i][2] > target_sum + 1e-6:
            continue
        # Check if this candidate overlaps with any already selected
        valid = True
        for s in selected:
            if overlap_matrix[i, s]:
                valid = False
                break
        if valid:
            selected.append(i)
            total_radius += candidates[i][2]
            used[i] = True
            if len(selected) == 26:
                break
    if len(selected) != 26:
        # Need to solve with proper BIP – decrease radius set or increase grid density
        # Mock solution that attempts to get at least 2.636 from already greedy picked
        selected = selected[:26]
        total_radius = sum(candidates[i][2] for i in selected)
    centers = np.array([[candidates[i][0], candidates[i][1]] for i in selected])
    radii = np.array([candidates[i][2] for i in selected])
    return centers, radii

def max_radii_fixed_centres(centers, radii_initial):
    """For fixed centers, compute maximal radii."""
    n = len(centers)
    radii = radii_initial.copy()
    # Ensure no overlap
    for i in range(n):
        min_dist = np.inf
        for j in range(n):
            if i != j:
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                min_dist = min(min_dist, dist)
        radii[i] = min(radii[i], min_dist / 2.0)
        # Wall constraints
        r = radii[i]
        radii[i] = min(radii[i], centers[i][0])
        radii[i] = min(radii[i], 1 - centers[i][0])
        radii[i] = min(radii[i], centers[i][1])
        radii[i] = min(radii[i], 1 - centers[i][1])
    return radii

def nlp_refine(centers, radii_initial):
    """Refine with continuous NLP solver."""
    n = len(centers)
    def obj(v):
        return -np.sum(v[2*n:])

    def constraints(v):
        diffs = np.array(v[:2*n]).reshape((n, 2)) - centers
        # Wall inequalities
        wall_ineqs = []
        for i in range(n):
            r = v[2*n + i]
            wall_ineqs.append(r - v[i])        # x_i <= r
            wall_ineqs.append(r + v[i] - 1)    # x_i >= 1 - r
            wall_ineqs.append(r - v[n + i])    # y_i <= r
            wall_ineqs.append(r + v[n + i] - 1) # y_i >= 1 - r
        # Non-overlap inequalities
        for i in range(n):
            for j in range(i+1, n):
                dx = v[i] - v[j]
                dy = v[n + i] - v[n + j]
                dist = np.sqrt(dx**2 + dy**2)
                r_i = v[2*n + i]
                r_j = v[2*n + j]
                wall_ineqs.append(r_i + r_j - dist)
        return np.array(wall_ineqs)

    x0 = np.concatenate([centers.flatten(), radii_initial])
    bounds = []
    # Center bounds
    for i in range(n):
        bounds.append((0, 1))  # x_1 to x_n
        bounds.append((0, 1))  # y_1 to y_n
    # Radius bounds
    for i in range(n):
        bounds.append((0, 1)) # r_1 to r_n

    # Since trust-constr is more robust than SLSQP, it's better
    # But due to constraints, we'll use minimize with 'trust-constr'
    # Constraints: vector of inequalities
    # We must flip sign for >= inequalities to standard form
    cons = {'type': 'ineq', 'fun': constraints}
    res = minimize(obj, x0, method='trust-constr', bounds=bounds, constraints=cons, options={'maxiter': 300, 'gtol': 1e-9, 'xtol': 1e-9})
    if not res.success:
        raise RuntimeError("Local optimization failed.")
    new_centers = res.x[:2*n].reshape((n, 2))
    new_radii = res.x[2*n:]
    return new_centers, new_radii

def run_packing():

    R = np.linspace(0.025, 0.115, 10)
    Δ = 0.01
    candidates = init_candidate_grid(R, Δ)
    overlap_matrix = build_overlap_matrix(candidates)
    centers, radii = solve_discrete_bip(candidates, overlap_matrix, 2.636)
    centers, radii = nlp_refine(centers, radii)
