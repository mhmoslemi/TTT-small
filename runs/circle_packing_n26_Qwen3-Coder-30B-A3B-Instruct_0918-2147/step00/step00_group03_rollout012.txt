import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    np.random.seed(0)
    
    # 1. Initial grid setup
    s = 0.07
    rows = 5
    cols = 6
    pts = []
    for k in range(rows):
        y = s / 2 + k * s * np.sqrt(3) / 2
        for m in range(cols):
            x = s / 2 + m * s
            if k % 2 == 1:
                x += s / 2
            if 0 <= x <= 1 and 0 <= y <= 1:
                if x > 0.02 and x < 0.98 and y > 0.02 and y < 0.98:
                    pts.append([x, y])
    
    # take first 26 points
    centers = np.array(pts[:26])
    # add jitter
    centers += np.random.uniform(-0.001, 0.001, centers.shape)
    # set initial radii
    radii = np.full(26, 0.001)
    
    # 2. Physics simulation
    def compute_forces(c, r):
        F = np.zeros_like(c)
        k_w = 10.0
        k_p = 5.0
        # wall forces
        for i in range(len(c)):
            x, y = c[i]
            rx, ry = r[i]
            dL = x - rx
            dR = (1 - x) - rx
            dB = y - ry
            dT = (1 - y) - ry

            F[i, 0] += k_w * max(-dL, 0) * (-1 if dL < 0 else 0)
            F[i, 0] += k_w * max(-dR, 0) * (1 if dR < 0 else 0)
            F[i, 1] += k_w * max(-dB, 0) * (-1 if dB < 0 else 0)
            F[i, 1] += k_w * max(-dT, 0) * (1 if dT < 0 else 0)

        # pairwise repulsion
        dists = cdist(c, c)
        for i in range(len(c)):
            for j in range(i+1, len(c)):
                d = dists[i, j]
                gap = d - (r[i] + r[j])
                if gap < 0:
                    dx = c[i, 0] - c[j, 0]
                    dy = c[i, 1] - c[j, 1]
                    if d > 1e-12:
                        f = -k_p * gap / d
                        F[i, 0] += f * dx
                        F[i, 1] += f * dy
                        F[j, 0] -= f * dx
                        F[j, 1] -= f * dy
        return F

    def grow_radii(c, r, F):
        dr = np.full(len(r), 0.001)  # base rate
        # Stop growth if overlapping
        dists = cdist(c, c)
        for i in range(len(c)):
            for j in range(i+1, len(c)):
                d = dists[i, j] - (r[i] + r[j])
                if d < 0:
                    dr[i] = 0
                    dr[j] = 0
                    break
        # Stop growth if touching wall
        for i in range(len(c)):
            x, y = c[i]
            rx, ry = r[i]
            if x - rx <= 1e-12 or x + rx >= 1 - 1e-12 or \
               y - ry <= 1e-12 or y + ry >= 1 - 1e-12:
                dr[i] = 0
                
        return dr

    # Simulation loop
    vel = np.zeros_like(centers)
    dt = 0.02
    gamma = 1.0
    max_force = 1e-6
    max_iter = 5000
    
    for _ in range(max_iter):
        F = compute_forces(centers, radii)
        # damping
        vel += dt * (F - gamma * vel)
        centers += dt * vel
        # clip to box
        centers = np.clip(centers, 0, 1)
        # grow radii
        dr = grow_radii(centers, radii, F)
        radii += dt * dr
        # stop if no growth and small forces
        is_stable = np.all(dr < 1e-7) and np.max(np.linalg.norm(F, axis=1)) < max_force
        if is_stable:
            break

    # 3. Fixed centers maximal radii update
    def max_radii_fixed_centers(c, r):
        r_new = np.copy(r)
        eps = 1e-9
        for _ in range(30):
            old_r = np.copy(r_new)
            # compute min distance to walls
            dists_to_walls = np.array([
                c[:, 0],             # left
                1 - c[:, 0],         # right
                c[:, 1],             # bottom
                1 - c[:, 1]          # top
            ])
            min_wall_dist = np.min(dists_to_walls, axis=0)

            # compute min distance to other nodes
            distances = cdist(c, c)
            min_dist = np.min(distances + np.eye(len(c)) * 1e20, axis=1)
            # each circle i gets to go inside min_distance - r_i
            r_new = np.minimum(min_wall_dist, min_dist - r + eps)
            r_new = np.maximum(r_new, 0.0)
            # check differences
            diff = np.max(np.abs(r_new - old_r))
            if diff < 1e-9:
                break
        return r_new

    radii = max_radii_fixed_centers(centers, radii)

    # 4. L-BFGS-B refinement (center coordinates only)
    def objective_sum_radii(centers_flat):
        centers = centers_flat.reshape(-1, 2)
        return -np.sum(max_radii_fixed_centers(centers, radii))

    bounds = [(0, 1) for _ in range(52)]

    # Initial guess from current state
    x0 = centers.flatten()
    
    # To discourage large penalties, add soft barrier
    def penalized_objective(x):
        c = x.reshape(-1, 2)
        val = objective_sum_radii(x)
        # barrier term
        penalty = 0
        for i in range(len(c)):
            penalty += np.sum(np.maximum(0, -c[i])**2)
            penalty += np.sum(np.maximum(0, c[i] - 1)**2)
        lambda_barrier = 1e-4
        return val + lambda_barrier * penalty
    
    res = minimize(penalized_objective, x0, method='L-BFGS-B', bounds=bounds, 
                   options={'maxiter': 2000, 'gtol': 1e-9})

    # Final update to centers and radii
    centers = res.x.reshape(-1, 2)
    radii = max_radii_fixed_centers(centers, radii)

    # 5. Final validation
    sum_radii = np.sum(radii)
    assert validate_packing(centers, radii)[0], "Packing validation failed"
    return centers, radii, sum_radii
