import numpy as np
from scipy.optimize import linprog
from math import sqrt

def initial_hex_centers(k, spacing):
    """Generate k centers from a hexagonal lattice within [0,1]x[0,1]."""
    sqrt3 = sqrt(3)
    s = spacing
    centers = []
    i = j = 0
    while len(centers) < k:
        x = i * s + (j % 2) * s / 2
        y = j * s * sqrt3 / 2
        if 0 <= x <= 1 and 0 <= y <= 1:
            centers.append([x, y])
        i += 1
        if i * s >= 1:
            i = 0
            j += 1
    return np.array(centers[:k])

def initial_radii(k, r0):
    """Initialize all radii to r0."""
    return np.full(k, r0)

def build_lp_constants(centers, radii):
    """Compute distances for LP constraints."""
    n = centers.shape[0]
    wall_dist_left = centers[:, 0]
    wall_dist_right = 1 - centers[:, 0]
    wall_dist_bottom = centers[:, 1]
    wall_dist_top = 1 - centers[:, 1]
    wall_dists = np.stack([wall_dist_left, wall_dist_right, wall_dist_bottom, wall_dist_top], axis=1)
    dist_matrix = np.zeros((n, n))
    for i in range(n):
        for j in range(i+1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            dist_matrix[i, j] = dist
            dist_matrix[j, i] = dist
    return {
        'wall_dists': wall_dists,
        'dist_matrix': dist_matrix,
        'radii': radii
    }

def solve_radius_lp(consts, radii, graph, alpha=0.5):
    """Solve the radius redistribution LP with given setup and graph."""
    n = len(radii)
    c = np.ones(n)
    A_ub = []
    b_ub = []
    # wall capacity constraints
    for i in range(n):
        max_r = np.min(consts['wall_dists'][i]) - radii[i]
        A_ub.append(-np.eye(1, n, i)[0])  # -Δr_i
        b_ub.append(-max_r)
    # pairwise capacity constraints
    D = consts['dist_matrix']
    for i in range(n):
        for j in range(i+1, n):
            max_r = D[i, j] - (radii[i] + radii[j])
            row = np.zeros(n)
            row[i] = -1
            row[j] = -1
            A_ub.append(row)
            b_ub.append(-max_r)
    # graph-based transfer limits (optional)
    for i in range(n):
        edges = graph.get(i, set())
        if edges:
            row = np.zeros(n)
            row[i] = -1
            contrib = 0
            for j in edges:
                contrib += D[i, j] - (radii[i] + radii[j])
            A_ub.append(row)
            b_ub.append(-alpha * contrib)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', 
                  options={'presolve': True, 'time_limit': 5})
    if res.success:
        return res.x
    else:  # fallback to same solution
        return np.zeros(n)

def apply_increment(radii, delta_r):
    """Add delta_r to current radii."""
    return radii + delta_r

def push_feasibility(centers, radii):
    """Apply geometric push to restore feasibility."""
    n = centers.shape[0]
    centers = centers.copy()
    radii = radii.copy()
    # first, clamp radii to wall bounds; slightly stricter
    for i in range(n):
        r = radii[i]
        x, y = centers[i]
        to_left = x
        to_right = 1 - x
        to_bottom = y
        to_top = 1 - y
        max_r = min(to_left, to_right, to_bottom, to_top)
        if r > max_r - 1e-12:
            radii[i] = max_r - 1e-12
            if radii[i] < 0:
                radii[i] = 0
    # pairwise overlaps
    for i in range(n):
        for j in range(i+1, n):
            dist_ij = np.linalg.norm(centers[i] - centers[j])
            sum_radii = radii[i] + radii[j]
            if dist_ij < sum_radii - 1e-12:
                overlap = sum_radii - dist_ij
                # move i towards j to restore tangency
                dir_vec = (centers[i] - centers[j])
                mag = np.linalg.norm(dir_vec)
                if mag > 1e-15:
                    dir_unit = dir_vec / mag
                    shift = overlap / 2
                    centers[i] -= shift * dir_unit
                    centers[j] += shift * dir_unit
    # final clamping due to movement
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 1e-12:
            radii[i] = x - 1e-12
        if x + r > 1 - 1e-12:
            radii[i] = 1 - x - 1e-12
        if y - r < 1e-12:
            radii[i] = y - 1e-12
        if y + r > 1 - 1e-12:
            radii[i] = 1 - y - 1e-12
    return centers, radii

def run_iterations(centers, radii, max_iter=200, epsilon=1e-5, alpha=0.5):
    """Iteratively optimize radii while maintaining feasibility."""
    graph = {}
    for it in range(max_iter):
        consts = build_lp_constants(centers, radii)
        delta_r = solve_radius_lp(consts, radii, graph, alpha)
        new_radii = apply_increment(radii, delta_r)
        if np.sum(delta_r) < epsilon:
            break
        # push to remove overlaps & enforce bounds
        old_centers = centers.copy()
        centers, radii = push_feasibility(centers, new_radii)
        # update graph: record new edges
        new_graph = {}
        for i in range(len(centers)):
            new_graph[i] = set()
        for i in range(len(centers)):
            for j in range(i+1, len(centers)):
                dist_ij = np.linalg.norm(centers[i] - centers[j])
                sum_radii = radii[i] + radii[j]
                if abs(dist_ij - sum_radii) < 1e-12:
                    new_graph[i].add(j)
                    new_graph[j].add(i)
        graph = new_graph
    return centers, radii

def final_refinement(centers, radii):
    """Do one final high-precision LP and cleanup."""
    consts = build_lp_constants(centers, radii)
    delta_r = solve_radius_lp(consts, radii, {}, alpha=0.5)
    new_radii = apply_increment(radii, delta_r)
    # Final clean-up passes
    new_centers, new_radii = push_feasibility(centers, new_radii)
    for i in range(len(new_radii)):
        r = new_radii[i]
        x, y = new_centers[i]
        to_wall = [x, 1-x, y, 1-y]
        new_radii[i] = min(r, *to_wall)
    for i in range(len(new_centers)):
        for j in range(i+1, len(new_centers)):
            dist_ij = np.linalg.norm(new_centers[i] - new_centers[j])
            sum_radii = new_radii[i] + new_radii[j]
            if dist_ij < sum_radii:
                overlap = sum_radii - dist_ij
                # reduce the larger of the two
                if new_radii[i] >= new_radii[j]:
                    new_radii[i] -= (overlap + 1e-13)
                else:
                    new_radii[j] -= (overlap + 1e-13)
    return new_centers, new_radii

def run_packing():

    init_centers = initial_hex_centers(26, 0.12)
    init_radii = initial_radii(26, 0.005)
    # Do iterations
    refined_centers, refined_radii = run_iterations(init_centers, init_radii, max_iter=100)
