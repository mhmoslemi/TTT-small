import numpy as np
from scipy.optimize import minimize

def initial_lattice(n=26):
    # Generate hexagonal lattice points
    # Each row offset alternates by 0.5 in x direction
    sqrt3 = np.sqrt(3)
    spacing = 0.2  # Initial spacing; will adjust to fit n points
    rows = int(np.ceil(np.sqrt(n * 2 / sqrt3)))
    cols = int(np.ceil(n / rows))
    
    centers = []
    for i in range(rows):
        y = i * spacing * sqrt3 / 2
        if y + spacing > 1:
            break
        for j in range(cols):
            x = j * spacing + (i % 2) * spacing / 2
            if x + spacing > 1:
                break
            centers.append([x, y])
            if len(centers) >= n:
                break
        if len(centers) >= n:
            break
            
    centers = np.array(centers[:n])
    
    # Compute safe radii for each center
    radii = []
    for i in range(n):
        x, y = centers[i]
        r = min(x, 1 - x, y, 1 - y)
        
        # Check distance to other centers
        for j in range(i):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist = np.sqrt(dx*dx + dy*dy)
            r = min(r, dist / 2)
        
        # Cap the radius to ensure it fits in the square
        r = min(r, 0.5)
        if r < 0:
            r = 0.001
        radii.append(r)
    
    return centers, np.array(radii)

def pack(centers, radii):
    z = np.empty(3 * len(centers))
    z[0::3] = centers[:, 0]
    z[1::3] = centers[:, 1]
    z[2::3] = radii
    return z

def unpack(z):
    n = len(z) // 3
    centers = np.empty((n, 2))
    centers[:, 0] = z[0::3]
    centers[:, 1] = z[1::3]
    radii = z[2::3]
    return centers, radii

def objective(z):
    _, radii = unpack(z)
    return -np.sum(radii)

def constraint_boundary(z):
    centers, radii = unpack(z)
    constraints = []
    n = len(centers)
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Left boundary: x - r >= 0
        constraints.append(x - r)
        # Right boundary: 1 - x - r >= 0
        constraints.append(1 - x - r)
        # Bottom boundary: y - r >= 0
        constraints.append(y - r)
        # Top boundary: 1 - y - r >= 0
        constraints.append(1 - y - r)
    return np.array(constraints)

def constraint_nonoverlap(z):
    centers, radii = unpack(z)
    constraints = []
    n = len(centers)
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            r_sum = radii[i] + radii[j]
            # Non-overlap: dist >= r_sum
            # Constraint function: dist - r_sum >= 0
            constraints.append(dist - r_sum)
    return np.array(constraints)

def repair(centers, radii):
    n = len(centers)
    perturbed = True
    max_iter = 10
    
    # Apply small dampening to prevent wild perturbations
    for _ in range(max_iter):
        if not perturbed:
            break
        perturbed = False
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            old_r = r
            
            # Adjust radius if it violates the boundary
            r = min(r, x, 1 - x, y, 1 - y)
            
            # Adjust for overlaps with other circles
            for j in range(n):
                if i == j:
                    continue
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.hypot(dx, dy) + 1e-9  # Avoid NaN
                overlap = dist - (radii[i] + radii[j])
                if overlap < 0:
                    delta = -overlap / 2
                    if i < j:
                        radii[i] -= delta
                        radii[j] -= delta
                    else:
                        radii[j] -= delta
                        radii[i] -= delta
                    perturbed = True

            # Ensure non-negativity and update if changed
            r = max(0, r)
            if r < radii[i]:
                radii[i] = r
                perturbed = True
            elif not perturbed and old_r != radii[i]:
                perturbed = True
    
    return centers, radii

def run_packing():
    # Generate initial hexagonal packing
    centers, radii = initial_lattice(26)
    
    # Optimize
    z0 = pack(centers, radii)
    
    # Combine constraints
    constraints = []
    constraints.append({'type': 'ineq', 'fun': constraint_boundary})
    constraints.append({'type': 'ineq', 'fun': constraint_nonoverlap})
    
    # Bounds for (x, y, r)
    bounds = [(0, 1), (0, 1), (0, 0.5)] * 26
    
    # Run optimization
    res = minimize(
        objective, z0, method='SLSQP',
        bounds=bounds, constraints=constraints,
        options={'ftol': 1e-10, 'maxiter': 1000, 'disp': False}
    )
    
    if not res.success:
        print("Optimization failed:", res.message)
