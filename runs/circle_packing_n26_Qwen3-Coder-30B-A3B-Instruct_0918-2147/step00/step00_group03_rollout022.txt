import numpy as np
from scipy.optimize import minimize
from math import sqrt

def run_packing():
    np.random.seed(0)
    
    # 1. Initial setup
    n_circles = 26
    initial_radius = 0.001
    dt = 0.02
    k_wall = 10.0
    k_pair = 5.0
    damping = 1.0
    alpha = 1e-3
    tol_force = 1e-6
    tol_radius = 1e-9
    max_steps = 5000
    
    # Generate initial centres (perturbed hex grid)
    def generate_initial_centers():
        spacing = 0.07
        rows = int(np.ceil(1.0 / (spacing * sqrt(3) / 2)))
        centers = []
        for k in range(rows):
            y = spacing / 2 + k * spacing * sqrt(3) / 2
            if y > 1.0 - spacing:
                continue
            if k % 2 == 0:
                x_start = spacing / 2
                step = spacing
            else:
                x_start = spacing
                step = spacing
                
            x = x_start
            while x < 1.0:
                if x < spacing or x > 1.0 - spacing or y < spacing or y > 1.0 - spacing:
                    x += step
                    continue
                centers.append([x, y])
                if len(centers) >= n_circles:
                    break
                x += step
            if len(centers) >= n_circles:
                break
        centers = np.array(centers[:n_circles])
        # Add jitter
        jitter = (np.random.rand(n_circles, 2) - 0.5) * 0.002
        return np.clip(centers + jitter, 0, 1)
    
    # 2. Physics simulation
    def compute_forces(centers, radii):
        forces = np.zeros_like(centers)
        n = centers.shape[0]
        for i in range(n):
            # Wall repulsion
            dist_left = centers[i, 0] - radii[i]
            dist_right = 1.0 - centers[i, 0] - radii[i]
            dist_bottom = centers[i, 1] - radii[i]
            dist_top = 1.0 - centers[i, 1] - radii[i]
            
            f_left = k_wall * max(-dist_left, 0) * np.array([-1.0, 0.0])
            f_right = k_wall * max(-dist_right, 0) * np.array([1.0, 0.0])
            f_bottom = k_wall * max(-dist_bottom, 0) * np.array([0.0, -1.0])
            f_top = k_wall * max(-dist_top, 0) * np.array([0.0, 1.0])
            
            forces[i] += f_left + f_right + f_bottom + f_top
            
            # Pairwise repulsion
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.linalg.norm(diff)
                gap = dist - (radii[i] + radii[j])
                if gap < 0:
                    force = k_pair * (-gap) * diff / (dist + 1e-12)
                    forces[i] += force
                    forces[j] -= force
                    
        return forces
    
    def grow_radii(centers, radii, forces):
        should_grow = np.ones_like(radii)
        n = centers.shape[0]
        for i in range(n):
            # Check if at wall
            dist_left = centers[i, 0] - radii[i]
            dist_right = 1.0 - centers[i, 0] - radii[i]
            dist_bottom = centers[i, 1] - radii[i]
            dist_top = 1.0 - centers[i, 1] - radii[i]
            
            if dist_left < 0 or dist_right < 0 or dist_bottom < 0 or dist_top < 0:
                should_grow[i] = 0.
            
            # Check overlap
            for j in range(n):
                if i == j:
                    continue
                diff = centers[i] - centers[j]
                dist = np.linalg.norm(diff)
                if dist < radii[i] + radii[j] + 1e-12:
                    should_grow[i] = 0.
                    break
                    
        dot_r = alpha * should_grow
        new_radii = radii + dt * dot_r
        return new_radii, should_grow
    
    centers = generate_initial_centers()
    radii = np.full(n_circles, initial_radius)
    velocities = np.zeros_like(centers)
    
    steps = 0
    while steps < max_steps:
        forces = compute_forces(centers, radii)
        # Update velocities
        velocities += dt * (forces - damping * velocities)
        # Update centers
        centers += dt * velocities
        # Clip to [0,1]
        centers = np.clip(centers, 0.0, 1.0)
        # Grow radii
        radii, should_grow = grow_radii(centers, radii, forces)
        # Check termination
        max_force = np.max(np.linalg.norm(forces, axis=1))
        if np.all(should_grow == 0) and max_force < tol_force:
            break
        steps += 1
        
    # 3. Exact maximal radii computation
    def compute_maximal_radii_fixed_centers(centers, radii):
        max_r = np.zeros_like(radii)
        for i in range(len(centers)):
            # Wall distances
            d_left = centers[i, 0]
            d_right = 1.0 - centers[i, 0]
            d_bottom = centers[i, 1]
            d_top = 1.0 - centers[i, 1]
            
            # Nearest neighbor distances
            min_dist = float('inf')
            for j in range(len(centers)):
                if i == j:
                    continue
                diff = centers[i] - centers[j]
                d = np.linalg.norm(diff)
                min_dist = min(min_dist, d - radii[j])
            
            max_r[i] = min(d_left, d_right, d_bottom, d_top, min_dist)
            if max_r[i] < 0:
                max_r[i] = 0
            
        return max_r
    
    # Iterate the max radii until convergence
    for _ in range(100):
        new_radii = compute_maximal_radii_fixed_centers(centers, radii)
        if np.allclose(radii, new_radii, atol=tol_radius):
            radii = new_radii
            break
        radii = new_radii
        
    # 4. Global optimization
    def objective_sum_radii(centers_flat):
        centers = centers_flat.reshape(-1, 2)
        radii = compute_maximal_radii_fixed_centers(centers, np.zeros_like(centers[:, 0]))
        return -np.sum(radii)
    
    def constraint_box(x):
        centers = x.reshape(-1, 2)
        infeasible = np.zeros(2 * len(centers))
        for i in range(len(centers)):
            infeasible[2*i] = -centers[i, 0]  # x_i >= 0
            infeasible[2*i+1] = centers[i, 0] - 1.0  # x_i <= 1
            infeasible[2*i+2] = -centers[i, 1]  # y_i >= 0
            infeasible[2*i+3] = centers[i, 1] - 1.0  # y_i <= 1
        return infeasible
    
    bounds = [(-1e-10, 1.0)] * (2*n_circles)
    
    result = minimize(objective_sum_radii, centers.flatten(), method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': 2000, 'gtol': 1e-9},
                      constraints={'type': 'ineq', 'fun': constraint_box})
    if not result.success:
        raise RuntimeError("Optimization failed: " + result.message)
        
    final_centers = result.x.reshape(-1, 2)
    final_radii = compute_maximal_radii_fixed_centers(final_centers, np.zeros_like(final_centers[:, 0]))
    
    sum_radii = np.sum(final_radii)
    
    # Validate
    assert validate_packing(final_centers, final_radii)[0], "Validation failed"
    
    return final_centers, final_radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
