import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial.distance import cdist

def build_lp_model(n):
    # Build LP constraints for 26 circles
    # Variables needed: 26 centers (x_i, y_i) and 26 radii (r_i)
    # Total variables = 26*2 + 26 = 78
    num_vars = 26 * 3  # x_i, y_i, r_i for all i
    
    # Objective: maximize sum of radii
    c = np.zeros(num_vars)
    c[2::3] = -1  # We want to minimize -sum(r_i)
    
    # Constraint matrix A and vector b
    # Each constraint is of form A @ x <= b
    
    # Precompute up to 4 constraints per pair
    # Max 4 * C(26, 2) = 4 * 325 = 1300 additional constraints
    # Include wall constraints, but block off-the-shelf
    # Fill in manually
    
    # Build all pairwise separation constraints
    # Each pair i,j contributes 4 inequalities
    A_ub = []
    b_ub = []

    # Wall constraints: r_i <= x_i, 1-x_i, y_i, 1-y_i
    # That is: 
    # -x_i + r_i <= 0
    # x_i + r_i <= 1
    # -y_i + r_i <= 0  
    # y_i + r_i <= 1
    for i in range(26):
        idx = [i*3, i*3+1, i*3+2]  # [x_i, y_i, r_i]
        
        row = np.zeros(num_vars)
        row[idx[0]] = -1  # x_i
        row[idx[2]] = 1   # r_i
        A_ub.append(row)
        b_ub.append(0)  # 0 <= x_i - r_i

        row = np.zeros(num_vars)  
        row[idx[0]] = 1   # x_i  
        row[idx[2]] = 1   # r_i
        A_ub.append(row)
        b_ub.append(1)  # x_i + r_i <= 1

        row = np.zeros(num_vars)
        row[idx[1]] = -1  # y_i  
        row[idx[2]] = 1   # r_i
        A_ub.append(row)  
        b_ub.append(0)  # 0 <= y_i - r_i

        row = np.zeros(num_vars)
        row[idx[1]] = 1   # y_i
        row[idx[2]] = 1   # r_i
        A_ub.append(row)
        b_ub.append(1)  # y_i + r_i <= 1
    
    # For every pair of circles i,j, add 4 inequalities (constrained axis)
    for i in range(26):
        for j in range(i + 1, 26):
            idx_i = [i*3, i*3+1, i*3+2]  # [x_i, y_i, r_i]
            idx_j = [j*3, j*3+1, j*3+2]  # [x_j, y_j, r_j]

            # x_i <= x_j - r_i - r_j
            row = np.zeros(num_vars)
            row[idx_i[0]] = 1         # x_i
            row[idx_j[0]] = -1        # -x_j  
            row[idx_i[2]] = 1         # r_i
            row[idx_j[2]] = 1         # r_j
            A_ub.append(row)
            b_ub.append(0) # 0 <= x_i - x_j + r_i + r_j

            # x_j <= x_i - r_i - r_j  
            row = np.zeros(num_vars)
            row[idx_i[0]] = -1        # -x_i
            row[idx_j[0]] = 1         # x_j
            row[idx_i[2]] = -1        # -r_i  
            row[idx_j[2]] = -1        # -r_j
            A_ub.append(row)
            b_ub.append(0) # 0 <= x_j - x_i + r_i + r_j

            # y_i <= y_j - r_i - r_j
            row = np.zeros(num_vars)
            row[idx_i[1]] = 1         # y_i
            row[idx_j[1]] = -1        # -y_j
            row[idx_i[2]] = 1         # r_i
            row[idx_j[2]] = 1         # r_j
            A_ub.append(row)
            b_ub.append(0) # 0 <= y_i - y_j + r_i + r_j

            # y_j <= y_i - r_i - r_j
            row = np.zeros(num_vars)
            row[idx_i[1]] = -1        # -y_i
            row[idx_j[1]] = 1         # y_j
            row[idx_i[2]] = -1        # -r_i
            row[idx_j[2]] = -1        # -r_j
            A_ub.append(row)
            b_ub.append(0) # 0 <= y_j - y_i + r_i + r_j

    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    bounds = [(None, None)] * 26 * 2 + [(0, None)] * 26
    
    return A_ub, b_ub, c, bounds

def solve_lp():
    A_ub, b_ub, c, bounds = build_lp_model(26)
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not result.success:
        raise RuntimeError("LP solver failed")
    
    x = result.x
    centers = x[::3].reshape(-1, 1), x[1::3]
    radii = x[2::3]
    return centers, radii

def detect_diagonal_conflicts(centers, radii):
    conflict_pairs = []
    nr = len(centers[0])  # num circles
    centers = np.column_stack(centers)
    
    # For each pair i,j, check if Euclidean distance < sum of radii
    for i in range(nr):
        for j in range(i+1, nr):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            if dist < radii[i] + radii[j] - 1e-12:
                conflict_pairs.append((i, j))
    
    return conflict_pairs

def max_radius_projection(centers, radii):
    # Clip radii to be consistent with wall constraints
    centers = np.array(centers)  # Shape (26, 2)
    radii = np.array(radii)  # Shape (26,)
    
    updated = True
    while updated:
        updated = False
        for i in range(26):
            # Maximize r_i subject to r_i <= x_i, 1-x_i, y_i, 1-y_i
            # and r_i + r_j >= d(i,j) for nearby circles
            min_shift = min(
                centers[i, 0],
                1 - centers[i, 0],
                centers[i, 1],
                1 - centers[i, 1]
            )
            for j in range(26):
                if i != j and np.sqrt(np.sum((centers[i] - centers[j])**2)) < 10:
                    min_shift = min(
                        min_shift,
                        np.sqrt(np.sum((centers[i] - centers[j])**2)) - radii[j]
                    )
            if min_shift < radii[i] and min_shift > 1e-14:
                radii[i] = min_shift
                updated = True
    
    return centers, radii
