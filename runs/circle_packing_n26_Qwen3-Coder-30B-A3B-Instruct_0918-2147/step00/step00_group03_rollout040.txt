import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def init_grid(n_circles=26, spacing=0.07):
    sqrt3 = np.sqrt(3)
    rows = int(np.ceil(np.sqrt(n_circles) * 2 / sqrt3))
    centers = []
    for k in range(rows):
        y = spacing / 2 + k * spacing * sqrt3 / 2
        if k % 2 == 0:
            xs = [spacing / 2 + m * spacing for m in range(int(np.ceil(n_circles / (rows * 2))) + 1)]
        else:
            xs = [spacing / 2 + m * spacing + spacing / 2 for m in range(int(np.ceil(n_circles / (rows * 2))) + 1)]
        for x in xs:
            if len(centers) >= n_circles:
                break
            if 0 <= x <= 1 and 0 <= y <= 1:
                if abs(x) > 0.02 and abs(x - 1) > 0.02 and abs(y) > 0.02 and abs(y - 1) > 0.02:
                    centers.append([x, y])
        if len(centers) >= n_circles:
            break
    centers = np.array(centers[:n_circles])
    noise = np.random.uniform(-0.001, 0.001, centers.shape)
    centers += noise
    return centers

def compute_forces(centers, radii, wall_stiff=10.0, pair_stiff=5.0, damping=1.0):
    n = centers.shape[0]
    forces = np.zeros_like(centers)
    for i in range(n):
        # Wall forces
        x, y = centers[i]
        r = radii[i]
        dL = x - r
        dR = (1 - x) - r
        dB = y - r
        dT = (1 - y) - r
        F_L = wall_stiff * max(-dL, 0) * np.array([-1, 0])
        F_R = wall_stiff * max(-dR, 0) * np.array([1, 0])
        F_B = wall_stiff * max(-dB, 0) * np.array([0, -1])
        F_T = wall_stiff * max(-dT, 0) * np.array([0, 1])
        forces[i] += F_L + F_R + F_B + F_T

        # Pairwise forces
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2, axis=-1))
            gap = dist - (radii[i] + radii[j])
            if gap < 0:
                f = pair_stiff * (-gap) * (centers[i] - centers[j]) / (dist + 1e-12)
                forces[i] += f
                forces[j] -= f
    return forces

def grow_radii(centers, radii, forces, dt=0.02, alpha=0.001):
    n = centers.shape[0]
    dot_radii = np.ones(n) * alpha
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        dL = x - r
        dR = (1 - x) - r
        dB = y - r
        dT = (1 - y) - r
        if dL <= 0 or dR <= 0 or dB <= 0 or dT <= 0:
            dot_radii[i] = 0
        for j in range(n):
            if i == j:
                continue
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2, axis=-1))
            gap = dist - (radii[i] + radii[j])
            if gap <= 0:
                dot_radii[i] = 0
                break
    return dot_radii

def max_radii_fixed_centers(centers, radii, tol=1e-9, max_iter=30):
    n = centers.shape[0]
    r_old = np.array(radii)
    for _ in range(max_iter):
        r_new = np.zeros(n)
        for i in range(n):
            x, y = centers[i]
            r = r_old[i]
            r_new[i] = min(
                x,
                1 - x,
                y,
                1 - y
            )
            for j in range(n):
                if i == j:
                    continue
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2, axis=-1))
                r_new[i] = min(r_new[i], dist - r_old[j])
            r_new[i] = max(0, r_new[i])
        if np.allclose(r_old, r_new, rtol=tol):
            break
        r_old = r_new
    return r_new

def objective_sum_radii(centers_flattened, centers, radii_fixed):
    n = len(centers)
    centers = centers_flattened.reshape(n, 2)
    r_vals = max_radii_fixed_centers(centers, radii_fixed)
    return -np.sum(r_vals)

def run_packing():
    np.random.seed(0)
    num_circles = 26
    centers = init_grid(num_circles)
    radii = np.full(num_circles, 0.001)
    velocities = np.zeros_like(centers)

    # Physics simulation
    dt = 0.02
    k_w = 10.0
    k_p = 5.0
    gamma = 1.0
    alpha = 0.001

    max_no_growth_steps = 10000
    no_growth_steps = 0
    min_force_norm = 1e-6

    for step in range(max_no_growth_steps):
        forces = compute_forces(centers, radii, wall_stiff=k_w, pair_stiff=k_p, damping=gamma)
        force_norm = np.linalg.norm(forces, axis=1).max()
        if force_norm < min_force_norm:
            dot_radii = grow_radii(centers, radii, forces, dt, alpha)
            if np.allclose(dot_radii, 0):
                no_growth_steps += 1
                if no_growth_steps > 100:
                    break
            else:
                no_growth_steps = 0
        else:
            no_growth_steps = 0

        dot_radii = grow_radii(centers, radii, forces, dt, alpha)
        velocities += dt * (forces - gamma * velocities)
        centers += dt * velocities
        centers = np.clip(centers, 0, 1)
        radii += dt * dot_radii
        radii = np.clip(radii, 0, None)

    # Refine with fixed center maximal radius
    radii = max_radii_fixed_centers(centers, radii)

    # L-BFGS-B optimization
    centers_flat = centers.flatten()
    res = minimize(
        objective_sum_radii,
        centers_flat,
        args=(centers, radii),
        method='L-BFGS-B',
        options={'maxiter': 2000, 'gtol': 1e-9},
        bounds=[(0, 1) for _ in range(2 * num_circles)]
    )

    centers_opt = res.x.reshape(num_circles, 2)
    radii_opt = max_radii_fixed_centers(centers_opt, radii)

    sum_radii = np.sum(radii_opt)
    validate_packing(centers_opt, radii_opt)
    return centers_opt, radii_opt, sum_radii
