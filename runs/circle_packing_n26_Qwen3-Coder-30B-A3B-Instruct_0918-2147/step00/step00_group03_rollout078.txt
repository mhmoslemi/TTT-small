import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial.distance import cdist

def build_lp_model(n_circles):
    # Build linear programming model for packing 26 circles
    # Variables: [x1, y1, r1, x2, y2, r2, ..., x_n, y_n, r_n]
    n_vars = 3 * n_circles
    c = np.zeros(n_vars)
    c[2::3] = -1  # Maximize sum of radii (negative because linprog minimizes)

    # Inequality constraints A_ub @ x <= b_ub
    # Each circle has 4 wall constraints: r <= x, r <= 1-x, r <= y, r <= 1-y
    A_ub = []
    b_ub = []

    # Wall constraints
    for i in range(n_circles):
        row = np.zeros(n_vars)
        row[3*i + 2] = 1   # r_i
        row[3*i] = -1      # x_i
        A_ub.append(row)
        b_ub.append(0)
        
        row = np.zeros(n_vars)
        row[3*i + 2] = 1   # r_i
        row[3*i] = 1       # -x_i
        A_ub.append(row)
        b_ub.append(1)
        
        row = np.zeros(n_vars)
        row[3*i + 2] = 1   # r_i
        row[3*i + 1] = -1  # y_i
        A_ub.append(row)
        b_ub.append(0)
        
        row = np.zeros(n_vars)
        row[3*i + 2] = 1   # r_i
        row[3*i + 1] = 1   # -y_i
        A_ub.append(row)
        b_ub.append(1)

    # Pairwise axis-aligned separation constraints  
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            # x-axis separation
            row = np.zeros(n_vars)
            row[3*i] = 1       # x_i
            row[3*i + 2] = 1   # r_i
            row[3*j] = -1      # -x_j
            row[3*j + 2] = 1   # r_j
            A_ub.append(row)
            b_ub.append(0)
            
            row = np.zeros(n_vars)
            row[3*j] = 1       # x_j
            row[3*j + 2] = 1   # r_j
            row[3*i] = -1      # -x_i
            row[3*i + 2] = 1   # r_i
            A_ub.append(row)
            b_ub.append(0)
            
            # y-axis separation
            row = np.zeros(n_vars)
            row[3*i + 1] = 1   # y_i
            row[3*i + 2] = 1   # r_i
            row[3*j + 1] = -1  # -y_j
            row[3*j + 2] = 1   # r_j
            A_ub.append(row)
            b_ub.append(0)
            
            row = np.zeros(n_vars)
            row[3*j + 1] = 1   # y_j
            row[3*j + 2] = 1   # r_j
            row[3*i + 1] = -1  # -y_i
            row[3*i + 2] = 1   # r_i
            A_ub.append(row)
            b_ub.append(0)

    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    return c, A_ub, b_ub

def solve_lp(c, A_ub, b_ub, n_circles):
    # Solve LP
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    if not res.success:
        raise RuntimeError('LP solver failed')
        
    # Extract solution
    sol = res.x
    
    centers = sol[::3][:n_circles].reshape(-1, 1), sol[1::3][:n_circles].reshape(-1, 1)
    radii = sol[2::3][:n_circles]
    
    # Identify active constraints
    # The solver is relatively reliable, but let's use a conservative 1e-9 indicator
    active_set = []
    
    # Check wall constraints  
    for i in range(n_circles):
        x, y, r = centers[0][i], centers[1][i], radii[i]
        if abs(r - x) < 1e-9 or abs(r - (1 - x)) < 1e-9 or abs(r - y) < 1e-9 or abs(r - (1 - y)) < 1e-9:
            active_set.append(('wall', i))
    
    # Pairwise constraints
    # Since it's hard to probabilistically determine all active constraints exactly,
    # we'll store them with a looser tolerance here providing some benefit for polishing
    n_constraints = A_ub.shape[0]
    for i in range(n_constraints):
        dual_val = res.slack[i]
        if dual_val < 1e-9:
            active_set.append(('pair', i))
        
    return np.array(centers).T, radii, active_set

def detect_diagonal_conflicts(centers, radii):
    n = centers.shape[0]
    conflicts = []
    
    # Compute Euclidean distances
    dist_matrix = cdist(centers, centers)
    for i in range(n):
        for j in range(i + 1, n):
            d_ij = dist_matrix[i, j]
            if d_ij < radii[i] + radii[j] - 1e-12:
                conflicts.append((i, j))
                
    return conflicts

def resolve_conflicts(centers, radii, active_set, conflict_pairs):
    # This function resolves conflicts in subproblem manner
    # In practice, using a simpler fixed perturbation pattern
    # here for strategic sake
    step_size = 0.005
    
    for i, j in conflict_pairs:
        if radii[i] + radii[j] > 0:
            d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            diff = (radii[i] + radii[j]) - d
            if diff > 0:
                # Repel the circles
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                if dx == 0 and dy == 0:
                    dx, dy = 1, 0  # Avoid division by zero
                norm = np.sqrt(dx**2 + dy**2)
                dx, dy = dx/norm, dy/norm
                
                # Move each circle away from the other
                move_amount = diff / 2.0
                centers[i] += np.array([dx * move_amount, dy * move_amount])
                centers[j] -= np.array([dx * move_amount, dy * move_amount])
                
                # Adjust radii to be approximately preserved but guarantee non-overlap
                # New radii based on geometry
                radii[i] = max(1e-6, radii[i] - 0.5 * diff) 
                radii[j] = max(1e-6, radii[j] - 0.5 * diff)
    
    return centers, radii

def polish_global(centers, radii, active_set):
    # A full non-linear optimization that keeps active constraints exact
    # For now approximate by simple bounds propagation and repositioning
    
    # Heuristic processing of active set to maintain network structure
    # While this simulated helmholtz approach doesn't require complex logic, 
    # we double-check our non-overlap conditions and adjust overall radii
    # We'll run a few passes of radial-restraint propagation

    n = centers.shape[0]
    for _ in range(5):
        # Realign each circle to have maximum possible radius within constraints
        # Check wall and pairwise constraints, adjust inversely for conflicts
        for i in range(n):
            r = radii[i]
            
            # Wall boundaries
            r_x = min(centers[i][0], 1 - centers[i][0])
            r_y = min(centers[i][1], 1 - centers[i][1])
            r = min(r, r_x, r_y)
            
            # Non-overlap with others
            for j in range(n):
                if i != j:
                    d_ij = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                    r = min(r, d_ij - radii[j])
            
            if r > 0:
                radii[i] = min(r, radii[i] * 1.01)
            else:
                radii[i] = max(1e-8, radii[i] * 0.99)  # Small decay if shrinking
    
    return centers, radii

            r = min( // wall bounds
                centers[i][0],

            )
            
            # Check pairwise distances
            for j in range(n):
                if i != j:
                    dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                    r = min(r, dist - radii[j])
            
            if r > 0:
                radii_updated.append(r)
            else:
                radii_updated.append(1e-10)
            
        new_radii = np.array(radii_updated)
        if np.allclose(new_radii, old_radii, atol=1e-12):
            break
        old_radii = new_radii.copy()
        radii = new_radii
    
    return radii
