import numpy as np

def build_lp_model(n_circles):
    # Build linear programming model for circle packing
    # Variables: [x1, y1, r1, x2, y2, r2, ..., x_n, y_n, r_n]
    n_vars = 3 * n_circles
    A_eq = []
    b_eq = []
    A_ub = []
    b_ub = []
    c = [-1.0] * n_vars  # coefficients for maximization of sum of radii

    # Wall constraints for each circle: r <= x, r <= 1-x, r <= y, r <= 1-y
    for i in range(n_circles):
        # r_i <= x_i => -x_i + r_i <= 0
        row = [0] * n_vars
        row[3*i] = -1
        row[3*i + 2] = 1
        A_ub.append(row)
        b_ub.append(0)
        # r_i <= 1 - x_i => x_i + r_i <= 1
        row = [0] * n_vars
        row[3*i] = 1
        row[3*i + 2] = 1
        A_ub.append(row)
        b_ub.append(1)
        # r_i <= y_i => -y_i + r_i <= 0
        row = [0] * n_vars
        row[3*i + 1] = -1
        row[3*i + 2] = 1
        A_ub.append(row)
        b_ub.append(0)
        # r_i <= 1 - y_i => y_i + r_i <= 1
        row = [0] * n_vars
        row[3*i + 1] = 1
        row[3*i + 2] = 1
        A_ub.append(row)
        b_ub.append(1)

    # Separation constraints for all pairs of circles
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            # x_j - x_i >= r_i + r_j -> -x_i + x_j - r_i - r_j >= 0
            row = [0] * n_vars
            row[3*i] = -1
            row[3*j] = 1
            row[3*i + 2] = -1
            row[3*j + 2] = -1
            A_ub.append(row)
            b_ub.append(0)

            # x_i - x_j >= r_i + r_j -> -x_j + x_i - r_i - r_j >= 0
            row = [0] * n_vars
            row[3*i] = 1
            row[3*j] = -1
            row[3*i + 2] = -1
            row[3*j + 2] = -1
            A_ub.append(row)
            b_ub.append(0)

            # y_j - y_i >= r_i + r_j -> -y_i + y_j - r_i - r_j >= 0
            row = [0] * n_vars
            row[3*i + 1] = -1
            row[3*j + 1] = 1
            row[3*i + 2] = -1
            row[3*j + 2] = -1
            A_ub.append(row)
            b_ub.append(0)

            # y_i - y_j >= r_i + r_j -> -y_j + y_i - r_i - r_j >= 0
            row = [0] * n_vars
            row[3*i + 1] = 1
            row[3*j + 1] = -1
            row[3*i + 2] = -1
            row[3*j + 2] = -1
            A_ub.append(row)
            b_ub.append(0)

    return A_ub, b_ub, c

def solve_lp(n_circles):
    A_ub, b_ub, c = build_lp_model(n_circles)
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    if not result.success:
        raise RuntimeError("LP solver failed to find an optimal solution.")

    solution = result.x
    centers = solution[::3][:n_circles], solution[1::3][:n_circles]
    radii = solution[2::3][:n_circles]
    return np.array(centers).T, radii, result

def detect_diagonal_conflicts(centers, radii):
    n = len(centers)
    pairs = []
    for i in range(n):
        for j in range(i+1,n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j]:
                pairs.append((i,j))
    return pairs

    new_radii = radii.copy()

    # Apply simple boundary constraint fixes
    for i in range(n):
        x, y = new_centers[i]
        r = new_radii[i]
        if x + r > 1:
            new_centers[i][0] -= (x + r - 1)
        if x - r < 0:
            new_centers[i][0] += (r - x)
        if y + r > 1:
            new_centers[i][1] -= (y + r - 1)
        if y - r < 0:
            new_centers[i][1] += (r - y)

    return new_centers, new_radii

def max_radius_projection(centers, radii):
    # Adjust radii to be the exact maximum allowed for each circle
    n = len(centers)
    for _ in range(10):  # iteration limit

                    min_dist = min(min_dist, dist - radii[j])

            else:
                new_radii[i] = radii[i]
        if np.allclose(radii, new_radii, rtol=1e-9):
            break
        radii = new_radii
    return radii
