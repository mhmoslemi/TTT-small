import numpy as np
from scipy.optimize import minimize

def initial_lattice(n=26):
    # Generate hexagonal lattice points
    sqrt3 = np.sqrt(3)
    spacing = 1.0
    while True:
        # Determine how many rows and cols we would need
        cols = int(1.0 / spacing) + 1
        rows = int(1.0 / (spacing * sqrt3 / 2)) + 1
        total = 0
        points = []
        for i in range(rows):
            offset = 0.5 * (i % 2)
            for j in range(cols):
                x = j * spacing + offset
                y = i * spacing * sqrt3 / 2
                if x <= 1 and y <= 1:
                    total += 1
                    points.append((x, y))
                if total >= n:
                    break
            if total >= n:
                break
        if total >= n:
            points = points[:n]
            break
        spacing *= 0.99

    centers = np.array(points)
    radii = np.full(n, 0.5 * spacing)

    # Compute actual admissible radii
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Distance to boundaries
        distances = [x, 1-x, y, 1-y]
        min_dist = min(distances)
        # Distance to other centers
        for j in range(n):
            if i != j:
                dist = np.sqrt((centers[i,0] - centers[j,0])**2 + (centers[i,1] - centers[j,1])**2)
                if dist < 2 * r:
                    # We need: dist >= 2 * r
                    max_r = dist / 2
                    min_dist = min(min_dist, max_r)
        radii[i] = min(r, min_dist)
    
    # Ensure no radii exceed 0.5
    radii = np.minimum(radii, 0.5)
    return centers, radii

def pack(centers, radii):
    z = np.empty(3 * len(centers))
    z[0::3] = centers[:,0]
    z[1::3] = centers[:,1]
    z[2::3] = radii
    return z

def unpack(z):
    n = len(z) // 3
    centers = np.empty((n, 2))
    centers[:,0] = z[0::3]
    centers[:,1] = z[1::3]
    radii = z[2::3]
    return centers, radii

def objective(z):
    _, radii = unpack(z)
    return -np.sum(radii)

def constraint_boundary(z):
    centers, radii = unpack(z)
    constraints = []
    for i in range(len(centers)):
        x, y = centers[i]
        r = radii[i]
        # left
        constraints.append(x - r)
        # right
        constraints.append(1 - x - r)
        # bottom
        constraints.append(y - r)
        # top
        constraints.append(1 - y - r)
    return constraints

def constraint_nonoverlap(z):
    centers, radii = unpack(z)
    constraints = []
    n = len(centers)
    for i in range(n):
        for j in range(i+1, n):
            dx = centers[i,0] - centers[j,0]
            dy = centers[i,1] - centers[j,1]
            dist = np.hypot(dx, dy)
            r_sum = radii[i] + radii[j]
            constraints.append(dist - r_sum)
    return constraints

def repair(centers, radii):
    # Copy inputs
    centers = centers.copy()
    radii = radii.copy()
    # Pass 1: fix boundary violations
    for i in range(len(centers)):
        x, y = centers[i]
        r = radii[i]
        # boundary checks with tolerance
        r = min(r, x, 1-x, y, 1-y)
        radii[i] = r
    # Pass 2: fix overlaps
    changed = True
    iter = 0
    max_iter = 10
    while changed and iter < max_iter:
        iter += 1
        changed = False
        for i in range(len(centers)):
            for j in range(i+1, len(centers)):
                dx = centers[i,0] - centers[j,0]
                dy = centers[i,1] - centers[j,1]
                dist = np.hypot(dx, dy)
                r_sum = radii[i] + radii[j]
                if dist < r_sum:
                    overlap = r_sum - dist
                    dr = overlap / 2
                    if radii[i] >= dr and radii[j] >= dr:
                        radii[i] -= dr
                        radii[j] -= dr
                        changed = True
                    else:
                        # Shrink the larger of the two
                        if radii[i] >= radii[j]:
                            dr_i = min(dr, radii[i])
                            radii[i] -= dr_i
                            radii[j] = max(0, radii[j] - (overlap - dr_i))
                        else:
                            dr_j = min(dr, radii[j])
                            radii[j] -= dr_j
                            radii[i] = max(0, radii[i] - (overlap - dr_j))
                        changed = True
    # bounded final repair
    radii = np.maximum(0, radii)
    for i in range(len(centers)):
        x, y = centers[i]
        r = radii[i]
        r = min(r, x, 1-x, y, 1-y)
        radii[i] = r
    return centers, radii

def run_packing():
    # Initial layout with hexagonal lattice
    centers, radii = initial_lattice(26)
    z0 = pack(centers, radii)
    # Define bounds and constraints
    bounds = []
    for i in range(26):
        bounds.append((0, 1))  # x_i
        bounds.append((0, 1))  # y_i
        bounds.append((0, 0.5))  # r_i
    cons = []
    cons.extend([{'type': 'ineq', 'fun': lambda z, i=i: constraint_boundary(z)[i]} for i in range(104)])
    cons.extend([{'type': 'ineq', 'fun': lambda z, i=i: constraint_nonoverlap(z)[i]} for i in range(325)])
    
    # Optimize
    result = minimize(objective, z0, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 500, 'ftol': 1e-9, 'disp': False})
    if not result.success:
        # Backup initialization: try again
        centers, radii = initial_lattice(26)
        # Slightly randomize
        np.random.seed(100)
        centers += np.random.uniform(-0.02, 0.02, size=centers.shape)
        # Keep inside unit square
        centers = np.clip(centers, 0, 1)
        z0 = pack(centers, radii)
        result = minimize(objective, z0, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 500, 'ftol': 1e-9, 'disp': False})
    
    centers, radii = unpack(result.x)
    centers, radii = repair(centers, radii)
    sum_radii = np.sum(radii)
    # Validate
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise ValueError(f"Invalid packing after optimization: {msg}")
    # Ensure we don't go below target
    if sum_radii < 2.636:
        raise ValueError(f"Sum of radii {sum_radii} is below target 2.636")
    return centers, radii, sum_radii
