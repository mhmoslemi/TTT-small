import numpy as np
from scipy.optimize import minimize
import scipy.sparse as sp

def init_hex_grid(n=26, r_init=0.04):
    # Generate hexagonal grid points
    sqrt3 = np.sqrt(3)
    h = sqrt3 * r_init
    rows = []
    y = r_init
    row_num = 0
    while len(rows) < n:
        if row_num % 2 == 0:
            x_start = r_init
            step = 2 * r_init
        else:
            x_start = 2 * r_init
            step = 2 * r_init
        x = x_start
        row = []
        while x <= 1 - r_init and len(rows) < n:
            row.append([x, y])
            x += step
            if len(rows) == n:
                break
        rows.extend(row)
        y += h
        row_num += 1
        if y > 1 - r_init:
            break
    # Ensure we have exactly n points
    centers = np.array(rows[:n])
    radii = np.full(n, r_init)
    return centers, radii

def max_radii(centers, radii):
    MAX_ITER = 100
    eps = 1e-12
    for _ in range(MAX_ITER):
        old_radii = radii.copy()
        # Wall constraints
        wall = np.minimum(centers, 1 - centers).min(axis=1)
        # Neighbor constraints
        D = np.sqrt(((centers[:, None, :] - centers[None, :, :]) ** 2).sum(-1))
        np.fill_diagonal(D, np.inf)
        neighbor = D - radii[:, None]
        neighbor = np.maximum(neighbor, 0)  # Ensure non-negative
        neighbor = neighbor.min(axis=1)
        radii = np.minimum(wall, neighbor)
        # Keep radii non-negative
        radii = np.maximum(radii, 0)
        # Check convergence
        if np.max(np.abs(radii - old_radii)) < 1e-8:
            break
    return radii

def hill_climb(centers, radii, iterations=5000, step=0.01):
    eps = 1e-12
    current_sum = np.sum(radii)
    for _ in range(iterations):
        # Propose move for one circle at a time
        idx = np.random.randint(0, len(centers))
        old_center = centers[idx].copy()
        old_radius = radii[idx]
        # Generate a small random displacement
        delta = np.random.uniform(-step, step, 2)
        new_center = old_center + delta
        # Check if the new center is valid
        if (new_center[0] - old_radius < -eps or 
            new_center[0] + old_radius > 1 + eps or
            new_center[1] - old_radius < -eps or 
            new_center[1] + old_radius > 1 + eps):
            continue
        # Update the center temporarily
        centers[idx] = new_center
        # Re-compute max radii for all circles
        new_radii = max_radii(centers, radii)
        new_sum = np.sum(new_radii)
        if new_sum > current_sum + 1e-12:
            # Accept the move and update the current sum
            radii[:] = new_radii
            current_sum = new_sum
        else:
            # Reject the move
            centers[idx] = old_center
    return centers, radii

def refine_with_scipy(centers, radii):
    n = centers.shape[0]
    # Define bounds and initial guess
    x0 = np.hstack((centers.flatten(), radii))
    bounds = []
    for i in range(n):
        bounds.append((0, 1))
        bounds.append((0, 1))
    for i in range(n):
        bounds.append((0, 0.5))
    
    def wall_constraints(x):
        res = []
        for i in range(n):
            r = x[2*n + i]
            cx = x[i]
            cy = x[n + i]
            res.append(r - cx)
            res.append(r + cx - 1)
            res.append(r - cy)
            res.append(r + cy - 1)
        return np.array(res)
    
    def overlap_constraints(x):
        res = []
        for i in range(n):
            for j in range(i + 1, n):
                r_i = x[2*n + i]
                r_j = x[2*n + j]
                cx_i = x[i]
                cy_i = x[n + i]
                cx_j = x[j]
                cy_j = x[n + j]
                dist = np.sqrt((cx_i - cx_j)**2 + (cy_i - cy_j)**2)
                res.append(r_i + r_j - dist)
        return np.array(res)
    
    # Build sparse Jacobians for constraints
    def wall_jac(x):
        rows = []
        cols = []
        data = []
        for i in range(n):
            # For r - x_i
            rows.append(i * 4 + 0)
            cols.append(i)
            data.append(-1)
            # For r_i + x_i - 1
            rows.append(i * 4 + 1)
            cols.append(i)
            data.append(1)
            # For r - y_i
            rows.append(i * 4 + 2)
            cols.append(n + i)
            data.append(-1)
            # For r_i + y_i - 1
            rows.append(i * 4 + 3)
            cols.append(n + i)
            data.append(1)
        j = sp.coo_matrix((data, (rows, cols)), shape=(4 * n, 2 * n + n))
        return j.tocsr()

    def overlap_jac(x):
        rows = []
        cols = []
        data = []
        for i in range(n):
            for j in range(i + 1, n):
                r_i = x[2*n + i]
                r_j = x[2*n + j]
                cx_i = x[i]
                cy_i = x[n + i]
                cx_j = x[j]
                cy_j = x[n + j]
                dx = cx_i - cx_j
                dy = cy_i - cy_j
                dist = np.sqrt(dx**2 + dy**2)
                if dist > 1e-12:
                    d = dx / dist
                    rows.append(i * (n - 1) + (j - i - 1))
                    cols.append(i)
                    data.append(-d)
                    rows.append(i * (n - 1) + (j - i - 1))
                    cols.append(j)
                    data.append(d)
                    rows.append(i * (n - 1) + (j - i - 1))
                    cols.append(2*n + i)
                    data.append(1)
                    rows.append(i * (n - 1) + (j - i - 1))
                    cols.append(2*n + j)
                    data.append(1)
        j = sp.coo_matrix((data, (rows, cols)), shape=(n * (n - 1) // 2, 2 * n + n))
        return j.tocsr()
    
    constraints = [
        NonlinearConstraint(wall_constraints, -np.inf, 0, jac=wall_jac),
        NonlinearConstraint(overlap_constraints, -np.inf, 0, jac=overlap_jac)
    ]
    
    # Minimize negative sum of radii
    def objective(x):
        return -np.sum(x[2*n:])
    
    # Run optimization
    result = minimize(objective, x0, method='trust-constr', bounds=bounds, 
                      constraints=constraints, options={'maxiter': 200})
    
    if not result.success:
        # Revert to previous best if optimization fails
        pass
        
    # Extract final values
    final_centers = result.x[:2*n].reshape(n, 2)
    final_radii = result.x[2*n:]
    
    return final_centers, final_radii

def run_packing():
    centers, radii = init_hex_grid()
    radii = max_radii(centers, radii)
    centers, radii = hill_climb(centers, radii)
    centers, radii = refine_with_scipy(centers, radii)
    # Make sure the result is valid
    flag, msg = validate_packing(centers, radii)
    if not flag:
        raise RuntimeError(f"Packing validation failed: {msg}")
    sum_r = float(radii.sum())
    return centers, radii, sum_r

# Ensure that validate_packing is defined in the same module
def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
