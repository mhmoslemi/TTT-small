import numpy as np
import cvxpy as cp
from scipy.spatial.distance import cdist
from copy import deepcopy

def build_socp_problem(n):
    """Builds a reusable SOCP problem for maximum sum of radii."""
    r = cp.Variable(n, nonneg=True)
    dist_params = []
    constraints = []
    
    # Boundary constraints
    x = cp.Variable(n)
    y = cp.Variable(n)
    constraints += [x <= 1, x >= 0, y <= 1, y >= 0]
    
    # Creating pairs of indices for constraints
    pairs = [(i, j) for i in range(n) for j in range(i+1, n)]
    pairs = sorted(pairs, key=lambda x: x[0] * 1000 + x[1])

    # Create parameterized constants for pairwise distances
    distance_params = [cp.Parameter((), name=f'dist_{i}_{j}') for i, j in pairs]
    dist_params.extend(distance_params)
    
    # Non-overlap constraints using SOC
    for k, (i, j) in enumerate(pairs):
        # r[i] + r[j] <= dist_ij
        constraints += [r[i] + r[j] <= distance_params[k]]
    
    # Objective: maximize sum of radii
    objective = cp.Maximize(cp.sum(r))
    
    # Create problem with constraints
    prob = cp.Problem(objective, constraints)
    
    return prob, r, distance_params, pairs

def solve_radii(centres, prob, r, dist_params, pairs):
    """Solves the inner SOCP for given centres to get maximum radii."""
    # Calculate pairwise distances once
    distances = cdist(centres, centres)
    
    # Update distance parameters with actual values
    for k, (i, j) in enumerate(pairs):
        dist_params[k].value = distances[i, j]
    
    # Solve the problem
    prob.solve(solver=cp.ECOS, verbose=False, abstol=1e-12, reltol=1e-12)
    
    if prob.status in ["infeasible", "unbounded"]:
        # Return zero radii on failure    
        return np.zeros(len(centres)), None

    radii = r.value
    duals = []
    # Get duals of non-overlap constraints
    for i in range(len(prob.constraints)):
        c = prob.constraints[i]
        if isinstance(c, cp.constraints.SOC) or isinstance(c, cp.constraints.Equality):
            try:
                dual_value = c.dual_value
                duals.append(dual_value)
            except:
                duals.append(0.)

    return radii, duals

# Build SOCP problem exactly once
SOC_PROB, SOC_R, SOC_DIST_PARAMS, SOC_PAIRS = build_socp_problem(26)

def energy(centres):
    """Computes energy (negative sum of radii) for given centres."""
    radii, _ = solve_radii(centres, SOC_PROB, SOC_R, SOC_DIST_PARAMS, SOC_PAIRS)
    if radii is None:
        return float('inf')
    return -np.sum(radii)

def random_move(state, rng, duals=None):
    """Generate a random move."""
    new_state = state.copy()
    n = len(state) // 2

    move_type = rng.choice(["jitter", "cluster", "edge", "swap"])

    if move_type == "jitter":
        # Single centre jitter
        idx = rng.integers(n)
        dx = rng.normal(0, 0.02)
        dy = rng.normal(0, 0.02)
        new_state[2*idx] = np.clip(new_state[2*idx] + dx, 0, 1)
        new_state[2*idx+1] = np.clip(new_state[2*idx+1] + dy, 0, 1)
    elif move_type == "cluster":
        # Cluster jitter: select around 3-5 centres
        idxs = rng.choice(n, size=rng.integers(3,6), replace=False)
        # Use a correlated offset
        dx = rng.normal(0, 0.01)
        dy = rng.normal(0, 0.01) 
        for i in idxs:
            new_state[2*i] = np.clip(new_state[2*i] + dx, 0, 1)
            new_state[2*i+1] = np.clip(new_state[2*i+1] + dy, 0, 1)
    elif move_type == "edge":
        # Move towards a wall
        idx = rng.integers(n)
        px, py = new_state[2*idx], new_state[2*idx+1]
        delta = 0.01
        if px < 0.5:
            new_state[2*idx] = np.clip(new_state[2*idx] - delta, 0, 1)
        else:
            new_state[2*idx] = np.clip(new_state[2*idx] + delta, 0, 1)
        if py < 0.5:
            new_state[2*idx+1] = np.clip(new_state[2*idx+1] - delta, 0, 1)
        else:
            new_state[2*idx+1] = np.clip(new_state[2*idx+1] + delta, 0, 1)
    else:
        # Swap order: just shuffle
        idxs = rng.permutation(n)
        for k, i in enumerate(idxs):
            new_state[2*k] = state[2*i]
            new_state[2*k+1] = state[2*i+1]

    return new_state

def anneal_chain(seed_state, rng):
    """Run a single simulated annealing chain."""
    temp = 0.05
    cooling_rate = 0.995
    max_iter = 150000
    curr_state = seed_state.copy()
    curr_energy = energy(curr_state)
    best_state = curr_state.copy()
    best_energy = curr_energy
    
    # Counter for accepted moves
    accepted = 0
    
    # Ancestral marking to store last valid state for reuse
    last_valid_state = curr_state.copy()
    last_valid_energy = curr_energy

    while accepted < max_iter and temp > 1e-6:
        new_state = random_move(curr_state, rng)
        new_energy = energy(new_state)
        
        # Accept or reject based on Metropolis
        delta = new_energy - curr_energy
        if delta < 0 or rng.uniform() < np.exp(-delta / temp):
            curr_state = new_state
            curr_energy = new_energy
            if curr_energy < best_energy:
                best_state = curr_state.copy()
                best_energy = curr_energy
                
            if curr_energy < last_valid_energy:
                last_valid_state = curr_state.copy()
                last_valid_energy = curr_energy
        
        accepted += 1
        temp *= cooling_rate

    return best_state, -best_energy

def local_refine(centres, radii, prob, r, dist_params, pairs):
    """Apply deterministic local refinement to polish result."""
    # Get dual values from previous computation
    duals = []
    for i, c in enumerate(prob.constraints):
        try:
            if hasattr(c, 'dual_value') and c.dual_value is not None:
                duals.append(c.dual_value)
            else:
                duals.append(0.)
        except:
                duals.append(0.)
    
    # Collect active constraints with strong duals 
    active_pairs = []
    threshold = 1e-6
    for k, (i, j) in enumerate(pairs):
        if k < len(duals) and duals[k] > threshold:
            active_pairs.append((i, j))

    # Find boundary contacts
    boundary_contacts = []
    for i in range(len(centres)):
        x, y = centres[i]
        r_i = radii[i]
        if abs(x - r_i) < 1e-10 or abs(x - 1 + r_i) < 1e-10 or \
           abs(y - r_i) < 1e-10 or abs(y - 1 + r_i) < 1e-10:
            boundary_contacts.append(i)

    # Iteratively project centres to satisfy active constraints
    max_iter = 10
    for _ in range(max_iter):
        improved = False
        # Need to reset from scratch if duals are stale
        
        # Even faster: don't recompute inner problem for refinement 
        # Just run solve_radii again to get updated info
        
        new_radii, new_duals = solve_radii(centres, prob, r, dist_params, pairs)
        if new_radii is None:
            break
        old_sum = np.sum(radii)
        new_sum = np.sum(new_radii)
        if new_sum - old_sum < 1e-5:
            break
        radii = new_radii.copy()
        
    return centres, radii

def finalise(centres, radii):
    """Run high-precision SOCP solve and apply tight clamping to satisfy validator."""
    final_radii, _ = solve_radii(centres, SOC_PROB, SOC_R, SOC_DIST_PARAMS, SOC_PAIRS)
    if final_radii is None:
        # Fallback
        final_radii = np.zeros_like(radii)
    
    # Clamp radii to allowable limits
    for i in range(len(centres)):
        x, y = centres[i]
        r_i = final_radii[i]
        max_r = min(x, 1-x, y, 1-y)
        final_radii[i] = min(r_i, max_r)

    # Verify validity for solver precision
    _, duals = solve_radii(centres, SOC_PROB, SOC_R, SOC_DIST_PARAMS, SOC_PAIRS)
    
    # Adjust small overlaps (probabilistic)
    for i in range(len(centres)):
        for j in range(i+1, len(centres)):
            dist = np.linalg.norm(centres[i] - centres[j])
            if dist < final_radii[i] + final_radii[j]:
                # Overlap exists; reduce one radius
                overlap = (final_radii[i] + final_radii[j]) - dist
                margin = 1e-13
                if overlap > margin:
                    # Reduce both radii slightly
                    diff = overlap * 0.5
                    final_radii[i] = max(0, final_radii[i] - diff)
                    final_radii[j] = max(0, final_radii[j] - diff)
    
    return centres, final_radii

def run_packing():
    """Main packing function that returns (centers, radii, sum_radii)."""
    n = 26
    rng = np.random.default_rng(42)
    
    # Seeds for starting configurations  
    seeds = [
        # 1. Hexagonal lattice
        np.zeros((n, 2)),
        # 2. Regular grid
        np.zeros((n, 2)),
        # 3. Corner-heavy
        np.zeros((n, 2)),
        # 4. Uniform random
        np.random.rand(n, 2),
    ] 
    
    # Fill seeds
    seed1 = np.zeros((n, 2))
    k = 0
    for i in range(5):
        for j in range(5):
            if k < 26:
                seed1[k, 0] = (j + 0.5)/5.0
                seed1[k, 1] = (i + 0.5)/5.0
                k += 1
    seeds[0] = seed1
    
    seed2 = np.zeros((n, 2))
    k = 0
    for i in range(5):
        for j in range(5):
            if k < 26:
                px, py = (j + 0.5)*0.19 + 0.01, (i + 0.5)*0.19 + 0.01
                seed2[k, 0] = px
                seed2[k, 1] = py
                k += 1
    seeds[1] = seed2
    
    seed3 = np.zeros((n, 2))
    for i in range(26):
        if i < 4 and i < 26:
            px, py = 0.1 if i % 2 == 0 else 0.9, 0.1 if i // 2 == 0 else 0.9
            seed3[i, 0] = px
            seed3[i, 1] = py
        else:
            seed3[i, 0] = np.random.uniform(0.2, 0.8)
            seed3[i, 1] = np.random.uniform(0.2, 0.8)
    seeds[2] = seed3

    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    for seed_num, seed in enumerate(seeds):
        seed_state = seed.flatten()
        # Run chain
        result_state, result_sum = anneal_chain(seed_state, rng)
        # Convert to centres
        result_centres = np.reshape(result_state, (26, 2))
        # Refine and finalize
        refined_c, refined_r = local_refine(result_centres, np.zeros(26), SOC_PROB, SOC_R, SOC_DIST_PARAMS, SOC_PAIRS)
        final_c, final_r = finalise(refined_c, refined_r)
        
        if result_sum > best_sum:
            best_sum = result_sum
            best_centers = final_c.copy()
            best_radii = final_r.copy()

    return best_centers, best_radii, best_sum
