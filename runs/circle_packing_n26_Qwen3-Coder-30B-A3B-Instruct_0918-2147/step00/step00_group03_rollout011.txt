import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    np.random.seed(0)
    
    # 1. Generate initial placement (perturbed hexagonal grid)
    def init_grid(n_circles=26, spacing=0.07):
        rows = int(np.ceil(np.sqrt(n_circles) * 2))
        cols = int(np.ceil(n_circles / rows))
        y_coords = []
        x_coords = []
        for k in range(rows):
            y = spacing / 2 + k * spacing * np.sqrt(3) / 2
            if k % 2 == 0:
                x_offset = 0
            else:
                x_offset = spacing / 2
            row_x = [spacing / 2 + m * spacing + x_offset for m in range(cols)]
            for x in row_x:
                if 0 <= x <= 1 and 0 <= y <= 1:
                    if min(x, 1 - x, y, 1 - y) >= 0.02:
                        x_coords.append(x)
                        y_coords.append(y)
                if len(x_coords) >= n_circles:
                    break
            if len(x_coords) >= n_circles:
                break
        centers = np.stack([x_coords[:n_circles], y_coords[:n_circles]], axis=1)
        centers += np.random.uniform(-0.001, 0.001, size=centers.shape)
        radii = np.full(n_circles, 0.001)
        return centers, radii
    
    centers, radii = init_grid()
    
    # 2. Physics simulation
    def compute_forces(centers, radii, k_w=10.0, k_p=5.0):
        n = centers.shape[0]
        forces = np.zeros_like(centers)
        # Wall forces
        d_L = centers[:, 0] - radii
        d_R = 1.0 - centers[:, 0] - radii
        d_B = centers[:, 1] - radii
        d_T = 1.0 - centers[:, 1] - radii
        forces[:, 0] += k_w * np.maximum(-d_L, 0.0)
        forces[:, 0] -= k_w * np.maximum(-d_R, 0.0)
        forces[:, 1] += k_w * np.maximum(-d_B, 0.0)
        forces[:, 1] -= k_w * np.maximum(-d_T, 0.0)
        
        # Pairwise forces
        dist_matrix = cdist(centers, centers)
        for i in range(n):
            for j in range(i+1, n):
                d_ij = dist_matrix[i, j]
                gap = d_ij - (radii[i] + radii[j])
                if gap < 0:
                    delta = centers[i] - centers[j]
                    if d_ij > 1e-12:
                        force_magnitude = k_p * (-gap) / d_ij
                        force = force_magnitude * delta
                        forces[i] += force
                        forces[j] -= force
        return forces

    def grow_radii(centers, radii, forces, alpha=1e-3):
        n = centers.shape[0]
        dot_r = np.ones(n) * alpha
        dist_matrix = cdist(centers, centers)
        for i in range(n):
            # Check if touching walls
            d_L = centers[i, 0] - radii[i]
            d_R = 1.0 - centers[i, 0] - radii[i]
            d_B = centers[i, 1] - radii[i]
            d_T = 1.0 - centers[i, 1] - radii[i]
            if d_L <= 1e-12 or d_R <= 1e-12 or d_B <= 1e-12 or d_T <= 1e-12:
                dot_r[i] = 0.0
            # Check overlap with others
            for j in range(n):
                if i != j:
                    d_ij = dist_matrix[i, j]
                    gap = d_ij - (radii[i] + radii[j])
                    if gap < -1e-12:
                        dot_r[i] = 0.0
                        break
        return dot_r

    # Euler integration step
    def step(centers, radii, v, forces, dt=0.02, gamma=1.0):
        v += dt * (forces - gamma * v)
        centers += dt * v
        centers = np.clip(centers, 0.0, 1.0)
        return centers, v

    # Simulation loop
    v = np.zeros_like(centers)
    for _ in range(5000):
        forces = compute_forces(centers, radii)
        dot_r = grow_radii(centers, radii, forces)
        centers, v = step(centers, radii, v, forces)
        radii += dt * dot_r
        if np.allclose(dot_r, 0.0, atol=1e-12) and np.max(np.linalg.norm(forces, axis=1)) < 1e-6:
            break
    
    # 3. Compute exact maximal radii for fixed centers
    def max_radii_fixed_centers(centers, radii, max_iter=30):
        old_radii = radii.copy()
        n = centers.shape[0]
        for _ in range(max_iter):
            new_radii = np.full_like(radii, np.inf)
            # Distance to walls
            new_radii = np.minimum(new_radii, centers[:, 0])
            new_radii = np.minimum(new_radii, 1.0 - centers[:, 0])
            new_radii = np.minimum(new_radii, centers[:, 1])
            new_radii = np.minimum(new_radii, 1.0 - centers[:, 1])
            # Distance to other circles
            for i in range(n):
                distances = np.linalg.norm(centers[i] - centers, axis=1)
                new_radii[i] = min(new_radii[i], np.min(distances - radii))
            new_radii = np.maximum(new_radii, 0.0)
            if np.allclose(new_radii, old_radii, rtol=1e-9):
                break
            old_radii = new_radii
        return new_radii

    radii = max_radii_fixed_centers(centers, radii)
    
    # 4. Refine with L-BFGS-B
    def objective_sum_radii(centers_flat, centers, radii):
        n = centers.shape[0]
        new_centers = centers_flat.reshape((n, 2))
        new_radii = max_radii_fixed_centers(new_centers, radii)
        return -new_radii.sum()
    
    def constraint_bounds(x):
        return np.hstack([x, 1 - x])
    
    centers_flat = centers.flatten()
    bounds = [(0, 1)] * len(centers_flat)
    result = minimize(objective_sum_radii, centers_flat, args=(centers, radii), 
                      method='L-BFGS-B', bounds=bounds, options={'maxiter': 2000, 'gtol': 1e-9})
    refined_centers = result.x.reshape((26, 2))
    
    # 5. Final maximal radii computation
    final_radii = max_radii_fixed_centers(refined_centers, radii)
    
    # 6. Final sum
    sum_radii = final_radii.sum()
    
    # 7. Validation
    assert validate_packing(refined_centers, final_radii)[0], "Invalid packing"
    
    return refined_centers, final_radii, sum_radii
