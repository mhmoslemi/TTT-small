import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def initial_lattice(n=26):
    # Generate a hexagonal lattice, scaled to fit in [0,1] x [0,1]
    # Try to find a spacing that gives at least n points
    spacing = 0.2
    while True:
        # Generate hexagonal lattice points
        rows = int(1.0 / (spacing * np.sqrt(3) / 2)) + 2
        cols = int(1.0 / spacing) + 2
        centers = []
        for i in range(rows):
            y = i * spacing * np.sqrt(3) / 2
            if y > 1 + 1e-12:
                break
            for j in range(cols):
                x = j * spacing + (i % 2) * spacing / 2
                if x > 1 + 1e-12:
                    break
                centers.append([x, y])
        centers = np.array(centers)
        if centers.shape[0] >= n:
            centers = centers[:n]
            break
        spacing *= 0.95

    radii = []
    for i, center in enumerate(centers):
        x, y = center
        # Distance to boundaries
        dist_to_boundaries = [x, 1 - x, y, 1 - y]
        min_dist_to_bound = min(dist_to_boundaries)
        
        # Distance to other centers
        dists_to_others = []
        for j in range(centers.shape[0]):
            if i != j:
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                dists_to_others.append(dist)
        
        min_dist_to_others = min(dists_to_others) if dists_to_others else float('inf')
        max_radius = min(min_dist_to_bound, min_dist_to_others / 2.0)
        
        # Ensure reasonable starting point
        radii.append(min(max_radius, 0.5))

    return centers, np.array(radii)

def pack(centers, radii):
    z = np.empty(3 * centers.shape[0])
    z[0::3] = centers[:, 0]
    z[1::3] = centers[:, 1]
    z[2::3] = radii
    return z

def unpack(z):
    n = z.shape[0] // 3
    centers = np.empty((n, 2))
    centers[:, 0] = z[0::3]
    centers[:, 1] = z[1::3]
    radii = z[2::3]
    return centers, radii

def objective(z):
    _, radii = unpack(z)
    return -np.sum(radii)

def constraint_boundary(z):
    centers, radii = unpack(z)
    n = centers.shape[0]
    constraints = []
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # x - r >= 0
        constraints.append(x - r)
        # 1 - x - r >= 0
        constraints.append(1 - x - r)
        # y - r >= 0
        constraints.append(y - r)
        # 1 - y - r >= 0
        constraints.append(1 - y - r)
    return np.array(constraints)

def constraint_nonoverlap(z):
    centers, radii = unpack(z)
    n = centers.shape[0]
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.hypot(dx, dy)
            r_sum = radii[i] + radii[j]
            # dist >= r_i + r_j
            # -> dist - (r_i + r_j) >= 0
            constraints.append(dist - r_sum)
    return np.array(constraints)

def repair(centers, radii):
    # Apply boundary fix
    for i in range(centers.shape[0]):
        x, y = centers[i]
        r = radii[i]
        new_r = min(r, x, 1 - x, y, 1 - y)
        if new_r < r:
            radii[i] = new_r
    
    # Apply overlap fix
    for iter in range(10):
        any_changed = False
        for i in range(centers.shape[0]):
            for j in range(i+1, centers.shape[0]):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.hypot(dx, dy)
                r_sum = radii[i] + radii[j]
                overlap = r_sum - dist
                if overlap > 0:
                    shrink = overlap / 2.0
                    if radii[i] > shrink and radii[j] > shrink:
                        radii[i] -= shrink
                        radii[j] -= shrink
                        any_changed = True
        if not any_changed:
            break
    
    return centers, radii

def random_perturb(centers, radii, amount=0.02):
    # Add small random perturbation to centers
    centres_new = centers + np.random.uniform(-amount, amount, centers.shape)
    # Clip to unit square
    centres_new = np.clip(centres_new, 0, 1)
    return centres_new, radii

def run_packing():
    centers, radii = initial_lattice()
    z0 = pack(centers, radii)
    
    # Define bounds for each variable
    bounds = []
    n = centers.shape[0]
    for i in range(n):
        bounds.append((0, 1))  # x coordinate
        bounds.append((0, 1))  # y coordinate
        bounds.append((0, 0.5))  # radius only up to 0.5
    
    # Define constraints
    constraints = []
    # Boundary constraints
    constraints.append({'type': 'ineq', 'fun': constraint_boundary})
    # Non-overlap constraints
    constraints.append({'type': 'ineq', 'fun': constraint_nonoverlap})

    # Optimize
    res = minimize(
        objective,
        z0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'ftol': 1e-9, 'maxiter': 1000, 'disp': False}
    )
    
    if not res.success:
        raise RuntimeError("Optimization failed with message: " + res.message)
    
    centers_opt, radii_opt = unpack(res.x)
    centers_opt, radii_opt = repair(centers_opt, radii_opt)
    
    # Sum of radii
    sum_radii = np.sum(radii_opt)
    
    # Run validation
    is_valid, msg = validate_packing(centers_opt, radii_opt)
    if not is_valid:
        raise RuntimeError(f"Validation failed after optimization: {msg}")
    
    # Try improving result with random perturbation
    centers_pert, radii_pert = random_perturb(centers_opt, radii_opt)
    z_pert = pack(centers_pert, radii_pert)
    res_pert = minimize(
        objective,
        z_pert,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'ftol': 1e-9, 'maxiter': 1000, 'disp': False}
    )
    
    if res_pert.success:
        centers_pert2, radii_pert2 = unpack(res_pert.x)
        centers_pert2, radii_pert2 = repair(centers_pert2, radii_pert2)
        sum_radii_pert = np.sum(radii_pert2)
        
        if sum_radii_pert > sum_radii:
            centers_opt, radii_opt = centers_pert2, radii_pert2
            sum_radii = sum_radii_pert
            is_valid, msg = validate_packing(centers_opt, radii_opt)
            if not is_valid:

    if sum_radii < 2.636:
        raise RuntimeError(f"Sum of radii {sum_radii} below target 2.636")
