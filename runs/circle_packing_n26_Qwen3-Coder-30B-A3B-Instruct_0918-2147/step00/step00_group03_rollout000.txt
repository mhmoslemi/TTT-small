import numpy as np
from scipy.optimize import minimize
import scipy.sparse as sp

def init_hex_grid(n=26, r_init=0.04):
    # Generate points in a hexagonal grid
    h = np.sqrt(3) * r_init
    rows = []
    y = r_init
    row_index = 0
    while len(rows) < n:
        if row_index % 2 == 0:
            x_start = r_init
        else:
            x_start = 2 * r_init
        x = x_start
        row = []
        while len(row) < n and x <= 1 - r_init:
            row.append((x, y))
            x += 2 * r_init
        if len(row) > 0:
            rows.extend(row)
        y += h
        row_index += 1
        if y > 1 - r_init:
            break

    centers = np.array(rows[:n])
    radii = np.full(n, r_init)
    return centers, radii

def max_radii(centers, radii):
    # Iteratively compute maximal radii
    max_iter = 100
    tol = 1e-8
    for _ in range(max_iter):
        # Wall constraints
        wall = np.minimum(centers, 1 - centers).min(axis=1)
        # Neighbour constraints
        D = np.sqrt(((centers[:, None, :] - centers[None, :, :]) ** 2).sum(-1))
        neighbor = D - radii[:, None]
        np.fill_diagonal(neighbor, np.inf)
        neighbor = np.maximum(neighbor.min(axis=1), 0)
        new_radii = np.minimum(wall, neighbor)
        if np.max(np.abs(new_radii - radii)) < tol:
            break
        radii = new_radii
    return radii

def hill_climb(centers, radii, iterations=5000, step=0.01):
    # Random perturbations to improve sum of radii
    for _ in range(iterations):
        i = np.random.randint(0, len(centers))
        # propose move
        delta = np.random.uniform(-step, step, 2)
        new_center = centers[i] + delta
        # check if center is still valid
        if (new_center[0] - radii[i] >= 0 and new_center[0] + radii[i] <= 1 and
            new_center[1] - radii[i] >= 0 and new_center[1] + radii[i] <= 1):
            # check feasibility of move
            old_radius = radii[i]
            centers[i] = new_center
            radii = max_radii(centers, radii)
            # if sum decreased, revert
            if np.sum(radii) < np.sum(radii) - old_radius + radii[i]:
                centers[i] = centers[i] - delta
                radii[i] = old_radius
                radii = max_radii(centers, radii)
    return centers, radii

def refine_with_scipy(centers, radii):
    # Multiplier for overlaps
    n = len(centers)
    # Flatten centers and radii to a single vector
    initial = np.concatenate([centers.flatten(), radii])
    bounds = []
    for i in range(n):
        bounds.append((0, 1))  # x_i
        bounds.append((0, 1))  # y_i
        bounds.append((0, 0.5))  # r_i
    bounds = bounds[:2*n] + bounds[2*n:]
    # Define constraints
    def wall_constraints(x):
        res = []
        for i in range(n):
            # r_i <= x_i, r_i <= 1 - x_i, r_i <= y_i, r_i <= 1 - y_i
            xi = x[2*i]
            yi = x[2*i + 1]
            ri = x[2*n + i]
            res.append(ri - xi)
            res.append(ri - (1 - xi))
            res.append(ri - yi)
            res.append(ri - (1 - yi))
        return np.array(res)
    
    def overlap_constraints(x):
        res = []
        for i in range(n):
            for j in range(i + 1, n):
                xi = x[2*i]
                yi = x[2*i + 1]
                ri = x[2*n + i]
                xj = x[2*j]
                yj = x[2*j + 1]
                rj = x[2*n + j]
                dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
                res.append(ri + rj - dist)
        return np.array(res)
    
    def jacobian(x):
        # Construct the sparse Jacobian manually
        m_wall = 4 * n
        m_overlap = n * (n - 1) // 2
        m_total = m_wall + m_overlap
        rows = np.arange(m_wall)
        cols = np.arange(m_wall)
        data = np.ones(m_wall)
        for i in range(n):
            rows = np.concatenate([rows, [m_wall + i * (n - 1) - (i * (i - 1)) // 2 + j for j in range(i)]])
            cols = np.concatenate([cols, [2 * n + i, 2 * n + j] for j in range(i)])
            data = np.concatenate([data, np.ones(2 * i)])  # Row for overlap constraint
            rows = np.concatenate([rows, [m_wall + i * (n - 1) - (i * (i - 1)) // 2 + j for j in range(i)]])
            cols = np.concatenate([cols, [2 * i, 2 * j] for j in range(i)])
            data = np.concatenate([data, np.ones(2 * i)])
        jacobian = sp.coo_matrix((data, (rows, cols)), shape=(m_total, 2 * n + n))
        return jacobian
    
    constraints = [
        NonlinearConstraint(wall_constraints, -np.inf, 0),
        NonlinearConstraint(overlap_constraints, -np.inf, 0)
    ]
    # Objective: minimize (-sum(radii))
    def objective(x):
        return -np.sum(x[2*n:])
    
    res = minimize(objective, initial, method='trust-constr', constraints=constraints, bounds=bounds, maxiter=200)
    # Extract final values
    final_centers = res.x[0:2*n].reshape((n, 2))
    final_radii = res.x[2*n:]
    return final_centers, final_radii

def run_packing():
    centers, radii = init_hex_grid()
    radii = max_radii(centers, radii)
    centers, radii = hill_climb(centers, radii)
    centers, radii = refine_with_scipy(centers, radii)
    # Validate packing
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise RuntimeError(f"Validation failed: {msg}")
    sum_r = float(radii.sum())
    return centers, radii, sum_r

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
