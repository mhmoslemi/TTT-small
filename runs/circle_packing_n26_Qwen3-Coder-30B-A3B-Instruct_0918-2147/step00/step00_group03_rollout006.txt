import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial.distance import cdist

def build_lp_model(n_circles):
    """Build LP model for circle packing with axis-aligned constraints."""
    # Variables: [x1, y1, r1, x2, y2, r2, ..., xn, yn, rn]
    n_vars = 3 * n_circles
    A_ub = []
    b_ub = []
    
    # Pairwise separation constraints (axis-aligned)
    for i in range(n_circles):
        for j in range(i+1, n_circles):
            # x_i + r_i <= x_j - r_j  (i left of j)
            row = np.zeros(n_vars)
            row[3*i] = 1.0   # x_i
            row[3*i+2] = 1.0 # r_i
            row[3*j] = -1.0  # x_j
            row[3*j+2] = 1.0 # r_j
            A_ub.append(row)
            b_ub.append(0.0)
            
            # x_j + r_j <= x_i - r_i  (j left of i)
            row = np.zeros(n_vars)
            row[3*j] = 1.0   # x_j
            row[3*j+2] = 1.0 # r_j
            row[3*i] = -1.0  # x_i
            row[3*i+2] = 1.0 # r_i
            A_ub.append(row)
            b_ub.append(0.0)
            
            # y_i + r_i <= y_j - r_j  (i below j)
            row = np.zeros(n_vars)
            row[3*i+1] = 1.0 # y_i
            row[3*i+2] = 1.0 # r_i
            row[3*j+1] = -1.0 # y_j
            row[3*j+2] = 1.0 # r_j
            A_ub.append(row)
            b_ub.append(0.0)
            
            # y_j + r_j <= y_i - r_i  (j below i)
            row = np.zeros(n_vars)
            row[3*j+1] = 1.0 # y_j
            row[3*j+2] = 1.0 # r_j
            row[3*i+1] = -1.0 # y_i
            row[3*i+2] = 1.0 # r_i
            A_ub.append(row)
            b_ub.append(0.0)

    # Wall constraints: r_i <= x_i, r_i <= 1 - x_i, r_i <= y_i, r_i <= 1 - y_i
    for i in range(n_circles):
        # r_i <= x_i
        row = np.zeros(n_vars)
        row[3*i] = -1.0  # -x_i
        row[3*i+2] = 1.0 # r_i
        A_ub.append(row)
        b_ub.append(0.0)
        
        # r_i <= 1 - x_i
        row = np.zeros(n_vars)
        row[3*i] = 1.0   # x_i
        row[3*i+2] = 1.0 # r_i
        A_ub.append(row)
        b_ub.append(1.0)
        
        # r_i <= y_i
        row = np.zeros(n_vars)
        row[3*i+1] = -1.0  # -y_i
        row[3*i+2] = 1.0   # r_i
        A_ub.append(row)
        b_ub.append(0.0)
        
        # r_i <= 1 - y_i
        row = np.zeros(n_vars)
        row[3*i+1] = 1.0   # y_i
        row[3*i+2] = 1.0   # r_i
        A_ub.append(row)
        b_ub.append(1.0)

    # Objective: maximize sum of radii
    c = np.zeros(n_vars)
    for i in range(n_circles):
        c[3*i+2] = -1.0  # negative because linprog minimizes

    return np.array(A_ub), np.array(b_ub), c

def solve_lp(A_ub, b_ub, c, n_circles):
    """Solve LP model using scipy.optimize.linprog."""
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    if not res.success:
        raise RuntimeError("LP solver failed to find a solution")
    
    # Extract centers and radii from solution
    centers = np.zeros((n_circles, 2))
    radii = np.zeros(n_circles)
    for i in range(n_circles):
        centers[i] = [res.x[3*i], res.x[3*i+1]]
        radii[i] = res.x[3*i+2]
    
    # Detect active constraints
    active_constraints = []
    for i in range(len(b_ub)):
        # Constraint is active if lhs ~ rhs
        lhs = np.dot(A_ub[i], res.x)
        if abs(lhs - b_ub[i]) < 1e-9:
            active_constraints.append(i)
    
    return centers, radii, active_constraints

def detect_diagonal_conflicts(centers, radii):
    """Check for euclidean overlaps."""
    n = centers.shape[0]
    conflicts = []
    dist_matrix = cdist(centers, centers)
    for i in range(n):
        for j in range(i+1, n):
            if dist_matrix[i, j] < radii[i] + radii[j] - 1e-12:
                conflicts.append((i, j))
    return conflicts

def resolve_conflicts(centers, radii, active_set, conflict_pairs, A_ub, b_ub, c):
    """Resolved conflicts using small local optimization."""
    for _ in range(100):  # limit iter
        changed = False
        for i, j in conflict_pairs:
            off_x = centers[i, 0] - centers[j, 0]
            off_y = centers[i, 1] - centers[j, 1]
            dist_sq = off_x**2 + off_y**2
            min_dist = radii[i] + radii[j]
            if dist_sq < min_dist**2:
                scale = (min_dist - np.sqrt(dist_sq)) / (2 * np.sqrt(dist_sq))
                centers[i, 0] += scale * off_x
                centers[i, 1] += scale * off_y
                centers[j, 0] -= scale * off_x
                centers[j, 1] -= scale * off_y
                changed = True
        if not changed:
            break
        # Re-check
        conflict_pairs = detect_diagonal_conflicts(centers, radii)
        if len(conflict_pairs) == 0:
            break
    return centers, radii

def polish_global(centers, radii, active_set):
    """Global polishing using trust-constr with exact constraints."""
    # Explicitly write constraints for feasibility
    n = centers.shape[0]
    
    def obj(x):
        return -np.sum(x[2*n:])
    
    def cons_hess(x):
        return np.zeros((len(x), len(x)))

    def cons_jac(x):
        j = np.zeros((len(active_set), len(x)))
        # For wall and axis constraints we know they were exactly maintained
        return j
    
    # For large scale, continue with local update
    # For simplicity here, we update periodically and project back
    for _ in range(10):
        radii = np.minimum(radii, centers[:, 0])
        radii = np.minimum(radii, 1 - centers[:, 0])
        radii = np.minimum(radii, centers[:, 1])
        radii = np.minimum(radii, 1 - centers[:, 1])
        for i in range(n):
            min_dist = np.inf
            for j in range(n):
                if i != j:
                    dist = np.sqrt((centers[i,0] - centers[j,0])**2 + 
                                   (centers[i,1] - centers[j,1])**2)
                    min_dist = min(min_dist, dist - radii[j])
            radii[i] = min(radii[i], min_dist)
    return centers, radii

def max_radius_projection(centers, radii):
    """Final projection to exact maximal radii."""
    n = centers.shape[0]
    last_radii = np.copy(radii)
    for _ in range(20):
        new_radii = np.copy(radii)
        for i in range(n):
            # nearest wall
            new_radii[i] = min(
                centers[i, 0],
                1 - centers[i, 0],
                centers[i, 1],
                1 - centers[i, 1]
            )
            # nearest circle
            for j in range(n):
                if i != j:
                    dist = np.sqrt((centers[i,0]-centers[j,0])**2 + 
                                   (centers[i,1]-centers[j,1])**2)
                    new_radii[i] = min(new_radii[i], dist - radii[j])
        radii = np.maximum(new_radii, 0)
        if np.allclose(radii, last_radii):
            break
        last_radii = np.copy(radii)
    return centers, radii

    A_ub, b_ub, c = build_lp_model(n_circles)
    centers, radii, active_set = solve_lp(A_ub, b_ub, c, n_circles)
    
    # Fast Conflict Detection and Reduce Velocity

    # Global polishing with softening
