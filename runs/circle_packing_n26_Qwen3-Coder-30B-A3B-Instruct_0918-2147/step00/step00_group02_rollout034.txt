import numpy as np
from scipy.optimize import minimize

def initial_lattice(n=26):
    # Generate hexagonal lattice points inside unit square
    # We'll use a spacing that gives us at least n points
    sqrt3 = np.sqrt(3)
    spacing = 1.0
    rows = int(np.ceil(np.sqrt(n / (sqrt3 / 2))))
    cols = int(np.ceil(n / rows))
    
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = j * spacing
            y = i * spacing * sqrt3 / 2
            if i % 2 == 1:
                x += spacing / 2
            if x <= 1 and y <= 1:
                centers.append([x, y])
            if len(centers) >= n:
                break
        if len(centers) >= n:
            break
    
    # Keep only the first n centers
    centers = np.array(centers[:n])
    
    # Compute radii based on neighbors and boundaries
    radii = np.zeros(n)
    for i in range(n):
        x, y = centers[i]
        min_dist = min(x, 1 - x, y, 1 - y)
        for j in range(n):
            if i != j:
                dist = np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
                min_dist = min(min_dist, dist / 2.0)
        radii[i] = min_dist
    
    return centers, radii

def pack(centers, radii):
    z = np.zeros(3 * len(centers))
    for i in range(len(centers)):
        z[3*i] = centers[i, 0]
        z[3*i+1] = centers[i, 1]
        z[3*i+2] = radii[i]
    return z

def unpack(z):
    n = len(z) // 3
    centers = z[::3].reshape(n, 2)
    radii = z[2::3]
    return centers, radii

def objective(z):
    _, radii = unpack(z)
    return -np.sum(radii)

def constraint_boundary(z):
    centers, radii = unpack(z)
    n = len(centers)
    c = []
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Left boundary: x - r >= 0
        c.append(x - r)
        # Right boundary: 1 - x - r >= 0
        c.append(1 - x - r)
        # Bottom boundary: y - r >= 0
        c.append(y - r)
        # Top boundary: 1 - y - r >= 0
        c.append(1 - y - r)
    return np.array(c)

def constraint_nonoverlap(z):
    centers, radii = unpack(z)
    n = len(centers)
    c = []
    for i in range(n):
        for j in range(i+1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            r_sum = radii[i] + radii[j]
            # Non-overlap constraint: dist >= r_sum 
            c.append(r_sum - dist)
    return np.array(c)

def repair(centers, radii):
    n = len(centers)
    # Ensure each circle is inside the square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        r = min(r, x, 1-x, y, 1-y)
        radii[i] = r
    
    # Fix overlaps by shrinking intersecting circles
    max_iter = 10
    for _ in range(max_iter):
        changed = False
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                r_sum = radii[i] + radii[j]
                if dist < r_sum:
                    overlap = r_sum - dist
                    # Reduce radii by half the overlap amount
                    delta = overlap / 2
                    if radii[i] >= delta and radii[j] >= delta:
                        radii[i] -= delta
                        radii[j] -= delta
                        changed = True
        if not changed:
            break

    return centers, radii

def run_packing():
    # Initial packing using hexagonal lattice
    centers, radii = initial_lattice(26)

    # Flatten for optimization
    z0 = pack(centers, radii)

    # Set up bounds and constraints
    bounds = []
    n = len(centers)
    # Bounds for x and y: [0, 1]
    for i in range(n):
        bounds.append((0, 1))  # x bounds
        bounds.append((0, 1))  # y bounds
    # Bounds for radii: [0, 0.5]
    for i in range(n):
        bounds.append((0, 0.5))  # r bounds

    constraints = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_nonoverlap}
    ]

    # Optimization with SLSQP
    options = {'maxiter': 500, 'ftol': 1e-9, 'disp': False}
    result = minimize(objective, z0, method='SLSQP', bounds=bounds, constraints=constraints, options=options)

    # Unpack the result
    centers, radii = unpack(result.x)
