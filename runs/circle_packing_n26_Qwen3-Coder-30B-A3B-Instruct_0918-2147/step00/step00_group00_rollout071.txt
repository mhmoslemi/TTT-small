import numpy as np
import random

def make_hex_seed(n, margin):
    """
    Generate a hexagonal seed packing with varying radii based on distance to walls.
    """
    rows = int(np.ceil(np.sqrt(n / (np.sqrt(3) / 2))))
    cols = int(np.ceil(n / rows))
    
    centers = []
    radii = []

    y_spacing = np.sqrt(3) / 2
    x_spacing = 1.0
    
    for i in range(rows):
        y = margin + i * y_spacing
        for j in range(cols):
            x = margin + j * x_spacing + (i % 2) * x_spacing / 2
            if len(centers) >= n:
                break
            if x > 1 - margin or y > 1 - margin:
                continue
            centers.append([x, y])
            radii.append(0.0)
    
    centers = np.array(centers[:n])
    radii = np.array(radii[:n])
    
    # Set initial radii based on distance to nearest wall
    for i in range(n):
        x, y = centers[i]
        r = min(x, 1 - x, y, 1 - y)
        # Adjust radius based on distance to center for more uniform distribution
        center_dist = np.sqrt((x - 0.5)**2 + (y - 0.5)**2)
        r = min(r, 0.2 * (1 - center_dist))
        if r < 0.015:
            r = 0.015
        radii[i] = r
    
    # Normalize to make sure all radii are at least 0.015
    scale_factor = 0.015 / np.min(radii)
    radii *= scale_factor
    return centers, radii

def max_feasible_radius(i, centers, radii):
    """
    Compute the maximum feasible radius for circle i given current centers and radii.
    """
    x, y = centers[i]
    r_current = radii[i]
    
    # Distance to walls
    d_left = x
    d_right = 1 - x
    d_bottom = y
    d_top = 1 - y
    
    min_dist = min(d_left, d_right, d_bottom, d_top)
    
    # Distance to other circles
    for j in range(len(centers)):
        if i == j:
            continue
        dx = centers[i][0] - centers[j][0]
        dy = centers[i][1] - centers[j][1]
        dist = np.sqrt(dx*dx + dy*dy)
        min_dist = min(min_dist, dist - radii[j])
    
    return min(min_dist, r_current)

def inflate_radii(centers, radii, alpha):
    """
    Uniformly increase all radii by added_delta, then shrink based on feasibility.
    Returns updated radii and sum increase.
    """
    old_sum = np.sum(radii)
    mean_r = np.mean(radii)
    added_delta = alpha * mean_r
    
    new_radii = radii + added_delta
    
    # Now clamp each radius to its maximum allowabe size
    for i in range(len(centers)):
        max_r = max_feasible_radius(i, centers, radii)
        new_radii[i] = min(new_radii[i], max_r)
    
    new_sum = np.sum(new_radii)
    return new_radii, new_sum - old_sum

def anneal_centers(centers, radii, temp0, gamma, sigma0, steps):
    """
    Perform simulated annealing on circle centers only, keeping radii constant.
    """
    current_radii = radii.copy()
    current_centers = centers.copy()
    best_centers = current_centers.copy()
    best_radii = current_radii.copy()
    best_sum = np.sum(best_radii)
    
    temp = temp0
    sigma = sigma0
    for step in range(steps):
        # On every 50 steps, do a small global translation to escape traps
        if step % 50 == 0:
            translate = np.random.normal(0, 0.005, 2)
            current_centers += translate
            
        # Pick a random circle and perturb its center
        i = np.random.randint(0, len(current_centers))
        
        # Generate a random displacement
        displacement = np.random.normal(0, sigma, 2)
        old_center = current_centers[i].copy()
        new_center = old_center + displacement
        
        # Ensure new center stays within bounds
        new_center[0] = np.clip(new_center[0], 0, 1)
        new_center[1] = np.clip(new_center[1], 0, 1)
        
        # Update current centers temporarily
        old_center_i = current_centers[i].copy()
        current_centers[i] = new_center
        
        # Compute new feasible radius for this circle
        new_feasible_radius = max_feasible_radius(i, current_centers, current_radii)
        old_radius_i = current_radii[i]
        
        # Decide whether to accept
        delta_radius = new_feasible_radius - old_radius_i
        if delta_radius > 0:
            # Always accept if radius increases
            current_radii[i] = new_feasible_radius
            if new_feasible_radius > old_radius_i:
                current_radii[i] = new_feasible_radius
                
        else:
            # Accept with probability exp(delta_radius / temp)
            if np.random.rand() < np.exp(delta_radius / temp):
                current_radii[i] = new_feasible_radius
            else:
                # Revert change
                current_centers[i] = old_center_i
        
        # Update best if needed
        current_sum = np.sum(current_radii)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = current_centers.copy()
            best_radii = current_radii.copy()
        
        # Update temp and sigma
        temp *= gamma
        sigma = sigma0 * temp / temp0
        
    return best_centers, best_radii, best_sum

def mixed_cycle(centers, radii, params):
    """
    Run alternating inflation and SA cycles.
    """
    alpha = params['alpha']
    T0 = params['T0']
    gamma = params['gamma']
    sigma0 = params['sigma0']
    steps_SA = params['steps_SA']
    K = params['K']
    M = params['M']
    P = params['P']
    
    last_improvement = 0
    cycle_count = 0
    old_sum = np.sum(radii)
    
    while cycle_count < P:
        # Phase B: Inflation
        stagnation = 0
        while stagnation < K:
            new_radii, improvement = inflate_radii(centers, radii, alpha)
            delta_sum = np.sum(new_radii) - old_sum
            radii = new_radii
            if delta_sum < 1e-6:
                stagnation += 1
            else:
                stagnation = 0
            old_sum = np.sum(radii)
        
        # Phase C: SA
        centers, radii, sa_sum = anneal_centers(centers, radii, T0, gamma, sigma0, steps_SA)
        
        if abs(sa_sum - old_sum) < 1e-5:
            cycle_count += 1
        else:
            cycle_count = 0
        old_sum = sa_sum
        
        # Reduce α and increase T (reset)
        alpha *= 0.95
    
    return centers, radii

def polish_radii(centers, radii, beta, passes):
    """
    Local polishing via exact gap measurement and inflation.
    """
    for _ in range(passes):
        # Compute gaps for all circles
        new_radii = radii.copy()
        for i in range(len(centers)):
            x, y = centers[i]
            r = radii[i]
            # Wall distances
            g_wall = min(x - r, 1 - x - r, y - r, 1 - y - r)
            # Neighbour distances
            g_nei = float('inf')
            for j in range(len(centers)):
                if i == j:
                    continue
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx*dx + dy*dy)
                gap = dist - (r + radii[j])
                if gap < g_nei:
                    g_nei = gap
            g = min(g_wall, g_nei)
            if g > 0:
                new_r = r + beta * g
                new_radii[i] = min(new_r, max_feasible_radius(i, centers, new_radii))
                
        # Resolve overlaps
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx*dx + dy*dy)
                r_sum = new_radii[i] + new_radii[j]
                overlap = r_sum - dist
                if overlap > 0:
                    if dist > 0:
                        adjust = overlap / 2
                        new_radii[i] = max(0, new_radii[i] - adjust)
                        new_radii[j] = max(0, new_radii[j] - adjust)
                    else:
                        new_radii[i] = new_radii[i] / 2
                        new_radii[j] = new_radii[j] / 2
                        
        radii = new_radii
    return radii
