import numpy as np
from scipy.optimize import minimize
from ortools.linear_solver import pywraplp
from itertools import combinations

def generate_candidates(g, radius_levels):
    candidates = []
    grid_points = np.arange(0, 1 + g/2, g)
    for x in grid_points:
        for y in grid_points:
            for r in radius_levels:
                if (x - r >= -1e-12 and x + r <= 1 + 1e-12 and
                    y - r >= -1e-12 and y + r <= 1 + 1e-12):
                    candidates.append((x, y, r))
    return candidates

def build_milp(candidates):
    solver = pywraplp.Solver.CreateSolver('SCIP')
    if not solver:
        raise RuntimeError("SCIP solver not available")
    
    n = len(candidates)
    z = [solver.IntVar(0, 1, f'z_{i}') for i in range(n)]

    # One circle per grid point constraint
    grid_dict = {}
    for i, (x, y, r) in enumerate(candidates):
        grid_key = (round(x/g), round(y/g))
        if grid_key not in grid_dict:
            grid_dict[grid_key] = []
        grid_dict[grid_key].append(i)
    
    for points in grid_dict.values():
        solver.Add(sum(z[i] for i in points) <= 1)
    
    # Exactly 26 circles
    solver.Add(sum(z) == 26)
    
    # Non-overlap constraints via L-infinity approximation
    M = 1.0
    for i, j in combinations(range(n), 2):
        dx = solver.NumVar(0, M, f'dx_{i}_{j}')
        dy = solver.NumVar(0, M, f'dy_{i}_{j}')
        solver.Add(dx >= candidates[i][0] - candidates[j][0])
        solver.Add(dx >= candidates[j][0] - candidates[i][0])
        solver.Add(dy >= candidates[i][1] - candidates[j][1])
        solver.Add(dy >= candidates[j][1] - candidates[i][1])
        solver.Add(dx + dy >= candidates[i][2] + candidates[j][2])
        solver.Add(dx - dy >= candidates[i][2] + candidates[j][2] - M * (1 - z[i] - z[j]))
        solver.Add(dy - dx >= candidates[i][2] + candidates[j][2] - M * (1 - z[i] - z[j]))
    
    # Objective
    solver.Maximize(sum(candidates[i][2] * z[i] for i in range(n)))

    return solver, z, candidates

def solve_milp(solver, z_vars):
    status = solver.Solve()
    if status != pywraplp.Solver.OPTIMAL:
        raise RuntimeError("MILP solver did not find optimal solution")
    
    selected_indices = [i for i, var in enumerate(z_vars) if var.solution_value() > 0.5]
    return selected_indices

def extract_initial_solution(selected_indices, candidates):
    centers = np.array([[candidates[i][0], candidates[i][1]] for i in selected_indices])
    radii = np.array([candidates[i][2] for i in selected_indices])
    return centers, radii

def shrink_initial(radii, delta):
    return radii - delta

def pack_variables(centers, radii):
    return np.hstack([centers.flatten(), radii])

def unpack_solution(z):
    n = 26
    centers = z[:2*n].reshape((n, 2))
    radii = z[2*n:]
    return centers, radii

def define_constraints(z):
    n = 26
    constraints = []
    centers, radii = unpack_solution(z)
    
    # Boundary constraints
    for i in range(n):
        constraints.append({
            'type': 'ineq',
            'fun': lambda z, i=i: z[2*n+i] - z[i*2] + 1e-12,
            'jac': lambda z, i=i: np.concatenate([np.zeros(2*n), np.zeros(i) + 1, np.zeros(n-i-1)])
        })
        constraints.append({
            'type': 'ineq',
            'fun': lambda z, i=i: z[2*n+i] - (1 - z[i*2+1]) + 1e-12,
            'jac': lambda z, i=i: np.concatenate([np.zeros(2*n), np.zeros(i) + 1, np.zeros(n-i-1)])
        })
    
    # Non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            def overlap_constraint(z, i=i, j=j):
                center_i = z[i*2:i*2+2]
                center_j = z[j*2:j*2+2]
                r_i = z[2*n+i]
                r_j = z[2*n+j]
                dist_sq = np.sum((center_i - center_j)**2)
                safe_dist_sq = (r_i + r_j)**2
                return safe_dist_sq - dist_sq + 1e-12
            def overlap_jac(z, i=i, j=j):
                center_i = z[i*2:i*2+2]
                center_j = z[j*2:j*2+2]
                r_i = z[2*n+i]
                r_j = z[2*n+j]
                dx = center_i[0] - center_j[0]
                dy = center_i[1] - center_j[1]
                denominator = np.sqrt(dx*dx + dy*dy) + 1e-12
                grad = np.zeros(2*n)
                grad[i*2] = -dx / denominator
                grad[i*2+1] = -dy / denominator
                grad[j*2] = dx / denominator
                grad[j*2+1] = dy / denominator
                grad[2*n+i] = -(r_i + r_j)
                grad[2*n+j] = -(r_i + r_j)
                return grad
            constraints.append({
                'type': 'ineq',
                'fun': overlap_constraint,
                'jac': overlap_jac
            })
    
    return constraints

def run_packing():
    g = 0.03
    radius_levels = np.arange(0.02, 0.25, 0.02)
    candidates = generate_candidates(g, radius_levels)
    solver, z_vars, candidates = build_milp(candidates)
    selected_indices = solve_milp(solver, z_vars)
    centers0, radii0 = extract_initial_solution(selected_indices, candidates)

    # Apply small shrinkage to initial radii
    radii0 = shrink_initial(radii0, 1e-5)
    
    # Pack initial solution
    z0 = pack_variables(centers0, radii0)
    
    # Define objective function
    def objective(z):
        return -np.sum(z[2*26:])

    def jacobian(z):
        grad = np.zeros_like(z)
        grad[2*26:] = -1.0
        return grad
    
    # Construct constraints cleanly
    constraints = define_constraints(z0)
    
    # Bounds for variables [x0,y0,...,x25,y25,r0,...,r25]
    bounds = [(0, 1) for _ in range(2*26)] + [(0, 0.5) for _ in range(26)]
    
    # Run optimization
    result = minimize(
        objective,
        z0,
        method='trust-constr',
        jac=jacobian,
        constraints=constraints,
        bounds=bounds,
        options={'gtol': 1e-9, 'ftol': 1e-9}
    )
    
    if not result.success:
        raise RuntimeError(f'Optimization failed: {result.message}')
    
    # Unpack and repair the solution
    centers, radii = unpack_solution(result.x)
    
    # Repair: ensure all constraints are violated
    best_radii = radii.copy()
    best_centers = centers.copy()
    
    for i in range(26):
        best_radii[i] = min(best_radii[i], best_centers[i][0], 1-best_centers[i][0],
                           best_centers[i][1], 1-best_centers[i][1])
    
    # Non-overlap repair
    for i in range(26):
        for j in range(i+1, 26):
            dx = best_centers[i][0] - best_centers[j][0]
            dy = best_centers[i][1] - best_centers[j][1]
            dist = np.sqrt(dx*dx + dy*dy)
            overlap = dist - (best_radii[i] + best_radii[j])
            if overlap < 0:
                diff = -overlap
                best_radii[i] -= diff / 2
                best_radii[j] -= diff / 2
    
    # Apply final repairs ensuring constraints are satisfied
    for i in range(26):
        best_radii[i] = min(best_radii[i], 1e-12, 1-best_centers[i][0], 1-best_centers[i][1],
                           best_centers[i][0], best_centers[i][1])
        best_radii[i] = max(best_radii[i], 1e-12)
    
    # Validate final solution
    _, msg = validate_packing(best_centers, best_radii)
    if not msg == "ok":
        raise RuntimeError(f'Validation failed: {msg}')
    
    # Return final result
    return best_centers, best_radii, np.sum(best_radii)
