import numpy as np

from scipy.spatial.distance import cdist

def build_leaf_problem(cell, k):
    x_min, y_min, x_max, y_max = cell
    # Create variables
    centers = cp.Variable((k, 2))
    radii = cp.Variable(k)
    
    # Add constraints for cell boundaries
    constraints = []
    for i in range(k):
        constraints.append(centers[i, 0] - radii[i] >= x_min)
        constraints.append(centers[i, 0] + radii[i] <= x_max)
        constraints.append(centers[i, 1] - radii[i] >= y_min)
        constraints.append(centers[i, 1] + radii[i] <= y_max)
    
    # Add non-overlap constraints (SOC)
    for i in range(k):
        for j in range(i + 1, k):
            diff = centers[i] - centers[j]
            dist = cp.norm(diff, 2)
            constraints.append(dist >= radii[i] + radii[j])
    
    # Maximize sum of radii
    objective = cp.Maximize(cp.sum(radii))
    
    prob = cp.Problem(objective, constraints)
    return prob, centers, radii


def solve_leaf(cell, k):
    prob, centers, radii = build_leaf_problem(cell, k)
    prob.solve(solver=cp.ECOS, verbose=False, abstol=1e-12, reltol=1e-12)
    
    if prob.status != cp.OPTIMAL:
        raise ValueError("Leaf problem is not optimal")
    return centers.value, radii.value
