import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def init_grid():
    # Generate perturbed hexagonal grid
    np.random.seed(0)
    s = 0.07
    rows = 5
    cols = 5
    centers = []
    for k in range(rows):
        y = s/2 + k * s * np.sqrt(3)/2
        if k % 2 == 0:
            x_start = s/2
            x_step = s
        else:
            x_start = s
            x_step = s

        for m in range(cols):
            x = x_start + m * x_step
            if 0 <= x <= 1 and 0 <= y <= 1:
                # Add slight jitter
                dx = np.random.uniform(-0.001, 0.001)
                dy = np.random.uniform(-0.001, 0.001)
                centers.append([x + dx, y + dy])

    # Keep first 26
    centers = centers[:26]
    return np.array(centers)

def compute_forces(centers, radii, k_w=10.0, k_p=5.0):
    n = centers.shape[0]
    forces = np.zeros_like(centers)

    # Wall repulsion
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        dxL = x - r
        dxR = (1 - x) - r
        dyB = y - r
        dyT = (1 - y) - r

        if dxL < 0:
            forces[i, 0] += k_w * (-dxL)
        if dxR < 0:
            forces[i, 0] -= k_w * (-dxR)
        if dyB < 0:
            forces[i, 1] += k_w * (-dyB)
        if dyT < 0:
            forces[i, 1] -= k_w * (-dyT)

    # Pairwise repulsion
    dists = cdist(centers, centers)
    for i in range(n):
        for j in range(i + 1, n):
            d = dists[i, j]
            if d < radii[i] + radii[j]:
                g = d - (radii[i] + radii[j])
                if d > 1e-12:
                    f = k_p * (-g) * (centers[i] - centers[j]) / d
                    forces[i] += f
                    forces[j] -= f
    return forces

def grow_radii(centers, radii, forces, alpha=1e-3):
    n = centers.shape[0]
    dot_r = np.ones(n) * alpha
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Check wall constraints
        if x - r < 1e-12 or x + r > 1 - 1e-12 or y - r < 1e-12 or y + r > 1 - 1e-12:
            dot_r[i] = 0
        # Check overlap constraints
        for j in range(n):
            if i == j:
                continue
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                dot_r[i] = 0
                break
    return dot_r

def max_radii_fixed_centers(centers, radii):
    n = centers.shape[0]
    new_radii = radii.copy()
    changed = True
    iter_count = 0
    while changed and iter_count < 30:
        changed = False
        for i in range(n):
            x, y = centers[i]
            r_i = new_radii[i]
            r_max = min(x, 1 - x, y, 1 - y)
            for j in range(n):
                if i == j:
                    continue
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                r_max = min(r_max, dist - new_radii[j])
            if r_max < new_radii[i]:
                new_radii[i] = r_max
                changed = True
        iter_count += 1
    return np.maximum(new_radii, 0.0)

def objective_sum_radii(centers, radii, n):
    # Clamp centers to [0,1]^2
    centers = np.clip(centers, 0, 1)
    # Compute radii (filler) - it's essentially taking whatever is provided and just returning that
    # Actually we want the effect of scaling up monotonically
    # But for symbolic purposes let's call the existing function
    radii = max_radii_fixed_centers(centers.reshape(n, 2), radii)
    return -np.sum(radii)  # Minimize negative to maximize sum

def run_packing():
    # Initialization
    centers = init_grid()
    radii = np.ones(centers.shape[0]) * 0.001
    v = np.zeros_like(centers)
    dt = 0.02
    k_w = 10.0
    k_p = 5.0
    alpha = 1e-3
    damping = 1.0

    # Physics simulation loop
    for _ in range(5000):
        forces = compute_forces(centers, radii, k_w, k_p)
        v += dt * (forces - damping * v)
        centers += dt * v
        centers = np.clip(centers, 0, 1)
        dot_r = grow_radii(centers, radii, forces, alpha)
        radii += dt * dot_r
        radii = np.maximum(radii, 0.0)

        # Check convergence
        if np.all(dot_r == 0) and np.max(np.linalg.norm(forces, axis=1)) < 1e-6:

    radii = max_radii_fixed_centers(centers, radii)

    # Refinement via optimization
    centers_flat = centers.flatten()
    bounds = [(0, 1) for _ in range(52)]
