import numpy as np
from scipy.optimize import minimize

def initial_layout():
    # Create a hexagonal grid pattern
    rows = 5
    cols = 5
    spacing = 0.18
    radius_base = spacing / 2 - 0.01
    
    centers = []
    radii = []
    
    # Generate hexagonal grid
    for i in range(rows):
        for j in range(cols):
            x = j * spacing + 0.05
            y = i * spacing * np.sqrt(3) / 2 + 0.05
            # Adjust for hexagonal pattern
            if i % 2 == 1:
                x += spacing / 2
            centers.append([x, y])
            radii.append(radius_base)
    
    # Place the 26th circle in the center
    centers.append([0.5, 0.5])
    radii.append(radius_base)
    
    # Adjust centers to fit in the unit square
    for i in range(len(centers)):
        x, y = centers[i]
        r = radii[i]
        # Clip to unit square considering the radius
        x = np.clip(x, r, 1 - r)
        y = np.clip(y, r, 1 - r)
        centers[i] = [x, y]
    
    return np.array(centers), np.array(radii)

def pack_vector(centers, radii):
    """Pack centers and radii into a single vector"""
    z = np.zeros(3 * len(centers))
    for i in range(len(centers)):
        z[3*i] = centers[i][0]
        z[3*i + 1] = centers[i][1]
        z[3*i + 2] = radii[i]
    return z

def unpack_vector(z):
    """Unpack the vector into centers and radii"""
    n = len(z) // 3
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i][0] = z[3*i]
        centers[i][1] = z[3*i + 1]
        radii[i] = z[3*i + 2]
    return centers, radii

def objective(z):
    centers, radii = unpack_vector(z)
    return -np.sum(radii)

def constraint_boundary(z):
    centers, radii = unpack_vector(z)
    con = []
    n = len(centers)
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # x - r >= 0  ==>  x - r - 0 >= 0
        con.append(x - r)
        # 1 - x - r >= 0  ==>  1 - x - r - 0 >= 0
        con.append(1 - x - r)
        # y - r >= 0  ==>  y - r - 0 >= 0
        con.append(y - r)
        # 1 - y - r >= 0  ==>  1 - y - r - 0 >= 0
        con.append(1 - y - r)
    return np.array(con)

def constraint_nonoverlap(z):
    centers, radii = unpack_vector(z)
    con = []
    n = len(centers)
    for i in range(n):
        for j in range(i+1, n):
            x1, y1 = centers[i]
            x2, y2 = centers[j]
            r1 = radii[i]
            r2 = radii[j]
            # (x1 - x2)^2 + (y1 - y2)^2 - (r1 + r2)^2 >= 0
            dist_sq = (x1 - x2)**2 + (y1 - y2)**2
            sum_radii_sq = (r1 + r2)**2
            con.append(dist_sq - sum_radii_sq)
    return np.array(con)
