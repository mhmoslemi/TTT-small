import numpy as np
from scipy.optimize import minimize
import scipy.sparse as sp

def init_hex_grid(n=26, r_init=0.04):
    # Generate hexagonal grid points
    h = np.sqrt(3) * r_init
    rows = []
    y = r_init
    row_num = 0
    while len(rows) < n:
        row = []
        x = r_init if row_num % 2 == 0 else 2 * r_init
        while len(row) < 6:  # Approximate
            if row_num % 2 == 0:
                row.append([x, y])
            else:
                row.append([x, y])
            x += 2 * r_init
            if x > 1 - r_init:
                break
        if len(row) > 0:
            rows.extend(row)
        y += h
        if y > 1 - r_init:
            break
        row_num += 1
    rows = np.array(rows[:n])
    return rows, np.full(n, r_init)

def max_radii(centers, radii):
    # Repeatedly compute maximal radii for all circles
    old_radii = np.zeros_like(radii)
    max_iter = 100
    for _ in range(max_iter):
        if np.max(np.abs(old_radii - radii)) < 1e-8:
            break
        old_radii = radii.copy()

        # Wall bounds
        wall_bounds = np.minimum(centers, 1 - centers).min(axis=1)

        # neighbour bounds
        d = np.sqrt(np.sum((centers[:, None, :] - centers[None, :, :]) ** 2, axis=2))
        np.fill_diagonal(d, np.inf)
        neighbor_bounds = np.min(d - radii[:, None], axis=1)
        
        # Update radii
        radii = np.minimum(wall_bounds, neighbor_bounds)
        radii = np.maximum(radii, 0)
    return radii

def hill_climb(centers, radii, iterations=5000, step=0.01):
    # Stochastic hill climbing
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = np.sum(best_radii)
    
    for _ in range(iterations):
        idx = np.random.randint(0, len(centers))
        old_center = centers[idx].copy()
        old_radius = radii[idx]
        
        # Propose new center
        new_center = old_center + np.random.uniform(-step, step, 2)
        
        # Clamp to stay within bounds
        new_center = np.clip(new_center, [old_radius, old_radius], [1-old_radius, 1-old_radius])
        
        centers[idx] = new_center
        new_radii = max_radii(centers, radii)
        new_sum = np.sum(new_radii)
        
        if new_sum > best_sum:
            best_centers = centers.copy()
            best_radii = new_radii
            best_sum = new_sum
        else:
            centers[idx] = old_center
            radii[idx] = old_radius
    
    return best_centers, best_radii

def refine_with_scipy(centers, radii):
    # Refine using scipy.optimize.minimize with trust-constr
    n = len(centers)
    # Variable vector: [x1,y1,...,x26,y26,r1,...,r26]
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Objective: minimize -sum(radii), i.e. maximize sum(radii)
    def obj(x):
        return -np.sum(x[2*n:])
    
    # Constraints:
    # 1. Wall bounds: r_i - x_i <= 0
    #    r_i + x_i - 1 <= 0
    #    r_i - y_i <= 0
    #    r_i + y_i - 1 <= 0
    def wall_cons(x):
        centers_flat = x[:2*n].reshape(-1, 2)
        radii = x[2*n:]
        constraints = []
        for i in range(n):
            constraints.extend([
                radii[i] - centers_flat[i, 0],  # r - x <= 0
                radii[i] + centers_flat[i, 0] - 1,  # r + x - 1 <= 0
                radii[i] - centers_flat[i, 1],  # r - y <= 0
                radii[i] + centers_flat[i, 1] - 1   # r + y - 1 <= 0
            ])
        return np.array(constraints)
    
    # Non-overlap: r_i + r_j - norm(c_i - c_j) <= 0
    def overlap_cons(x):
        centers_flat = x[:2*n].reshape(-1, 2)
        radii = x[2*n:]
        constraints = []
        for i in range(n):
            for j in range(i+1, n):
                dist = np.sqrt(np.sum((centers_flat[i] - centers_flat[j])**2))
                constraints.append(radii[i] + radii[j] - dist)
        return np.array(constraints)

    def make_jac_wall(n):
        # Jacobian for wall constraints: 4*n rows
        rows = []
        cols = []
        data = []
        for i in range(n):
            # For r_i - x_i <= 0:
            rows.append(4*i)
            cols.append(2*i)  # x_i component
            data.append(-1.0)
            rows.append(4*i)
            cols.append(2*n + i)  # r_i
            data.append(1.0)
            # For r_i + x_i - 1 <= 0:
            rows.append(4*i + 1)
            cols.append(2*i)
            data.append(1.0)
            rows.append(4*i + 1)
            cols.append(2*n + i)
            data.append(1.0)
            # For r_i - y_i <= 0:
            rows.append(4*i + 2)
            cols.append(2*i + 1)  # y_i
            data.append(-1.0)
            rows.append(4*i + 2)
            cols.append(2*n + i)
            data.append(1.0)
            # For r_i + y_i - 1 <= 0:
            rows.append(4*i + 3)
            cols.append(2*i + 1)
            data.append(1.0)
            rows.append(4*i + 3)
            cols.append(2*n + i)
            data.append(1.0)
        return sp.coo_matrix((data, (rows, cols)), shape=(4*n, 2*n + n))
    
    def make_jac_overlap(n):
        # Jacobian for overlap constraints: n*(n-1)/2 rows
        rows = []
        cols = []
        data = []
        idx = 0
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist > 0:
                    dx /= dist
                    dy /= dist
                    
                # Grad wrt x_i: dx, -dx
                rows.append(idx)
                cols.append(2*i)
                data.append(dx)
                rows.append(idx)
                cols.append(2*j)
                data.append(-dx)
                rows.append(idx)
                cols.append(2*n + i)
                data.append(1.0)
                rows.append(idx)
                cols.append(2*n + j)
                data.append(1.0)
                idx += 1
        return sp.coo_matrix((data, (rows, cols)), shape=(n*(n-1)//2, 2*n + n))
    
    # Build constraints
    wall_jac = make_jac_wall(n)
    overlap_jac = make_jac_overlap(n)
    
    cons = [
        NonlinearConstraint(wall_cons, -np.inf, 0, jac=wall_jac),
        NonlinearConstraint(overlap_cons, -np.inf, 0, jac=overlap_jac)
    ]
    
    # Bounds
    bounds = [(0, 1) for _ in range(2*n)] + [(0, 0.5) for _ in range(n)]
    
    # Optimize
    res = minimize(obj, x0, method='trust-constr', constraints=cons, bounds=bounds, options={'maxiter': 200})
    
    # Reconstruct result
    final_centers = res.x[:2*n].reshape(-1, 2)
    final_radii = res.x[2*n:]
    
    return final_centers, final_radii

def run_packing():
    centers, radii = init_hex_grid()
    radii = max_radii(centers, radii)
    centers, radii = hill_climb(centers, radii)
    centers, radii = refine_with_scipy(centers, radii)
    sum_r = float(radii.sum())
    return centers, radii, sum_r
