import numpy as np
from scipy.optimize import minimize

NUM_CIRCLES = 26
NUM_RESTARTS = 5
HEX_RADIUS = 0.15
BOUNDARY_MARGIN = 1e-9

def hex_lattice_points(n_rows, spacing):
    # Generate a hexagonal lattice in the unit square
    # spacing = 2*r where r is the radius of circles that can be placed
    # n_rows = 5 (as 5 rows give ~26 circles in the hex pattern)
    points = []
    for i in range(n_rows):
        # Alternate row offset
        offset = (i % 2) * spacing / 2
        for j in range(int(1.0 / spacing) + 1):
            x = offset + j * spacing
            y = i * spacing * np.sqrt(3) / 2
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append((x, y))
    return np.array(points[:NUM_CIRCLES])

def initial_guess(seed):
    # Generate initial configuration from hexagonal packing with jitter
    np.random.seed(seed)
    grid_points = hex_lattice_points(5, 0.2)
    # Add slight jitter
    jitter = np.random.uniform(-0.02, 0.02, size=(NUM_CIRCLES, 2))
    centers = grid_points + jitter
    
    # Ensure all centers are within bounds
    centers = np.clip(centers, 0, 1)
    
    # Initial radii (small to ensure feasibility)
    radii = np.full(NUM_CIRCLES, 0.05)
    
    # Flatten the array: [x0,y0,r0,x1,y1,r1,...]
    vars0 = np.zeros(3 * NUM_CIRCLES)
    vars0[0::3] = centers[:, 0]
    vars0[1::3] = centers[:, 1]
    vars0[2::3] = radii
    return vars0

def objective(vars):
    # Return negative sum of radii (minimize negative to maximize sum)
    return -np.sum(vars[2::3])

def constraint_boundary(vars):
    # Ensure each circle is within unit square with margin
    # Returns array of [x-r, 1-x-r, y-r, 1-y-r] for all 26 circles
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    constraints = np.empty((4, NUM_CIRCLES))
    constraints[0] = x - r  # x-r >= 0
    constraints[1] = 1 - x - r  # 1-x-r >= 0
    constraints[2] = y - r  # y-r >= 0
    constraints[3] = 1 - y - r  # 1-y-r >= 0
    return constraints.flatten()

def constraint_nonoverlap(vars):
    # Return array of [dist(i,j) - (r_i + r_j)] for all pairs i<j
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    all_constraints = []
    
    for i in range(NUM_CIRCLES):
        for j in range(i + 1, NUM_CIRCLES):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            distance = np.sqrt(dx * dx + dy * dy)
            overlap = distance - (r[i] + r[j])
            all_constraints.append(overlap)
    return np.array(all_constraints)

def postprocess(vars):
    # Make final corrections to ensure hard constraints
    vars = vars.copy()
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Fix boundary violations
    for i in range(NUM_CIRCLES):
        # Clip centers
        x[i] = np.clip(x[i], r[i], 1 - r[i])
        y[i] = np.clip(y[i], r[i], 1 - r[i])
    
    # Ensure no overlaps
    for i in range(NUM_CIRCLES):
        for j in range(i + 1, NUM_CIRCLES):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dist = np.sqrt(dx * dx + dy * dy)
            overlap = dist - (r[i] + r[j])
            if overlap < -1e-12:
                # Reduce larger radius
                if r[i] > r[j]:
                    r[i] += overlap / 2
                else:
                    r[j] += overlap / 2
    
    return vars

def run_packing():
    best_sum = -np.inf
    best_vars = None
    global_constraints = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_nonoverlap}
    ]
    global_bounds = [(0, 1), (0, 1), (0, 0.5)] * 3 * NUM_CIRCLES

    for seed in range(NUM_RESTARTS):
        vars0 = initial_guess(seed)
        res = minimize(objective, vars0,
                      method='SLSQP',
                      bounds=global_bounds,
                      constraints=global_constraints,
                      options={'maxiter': 5000, 'ftol': 1e-9})
        if not res.success:
            continue
        vars_corr = postprocess(res.x)
        s = np.sum(vars_corr[2::3])
        if s > best_sum:
            best_sum = s
            best_vars = vars_corr
    
    # Reshape best_vars back to centers and radii
    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii = best_vars[2::3]
    return centers, radii, float(best_sum)
