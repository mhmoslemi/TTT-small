import numpy as np
import random
from math import sqrt

def make_hex_seed(n, margin):
    # Create a hexagonal arrangement with slightly varying radii
    rows = int(sqrt(n))
    cols = (n + rows - 1) // rows
    centers = []
    radii = []

    # Generate a hexagonal pattern
    spacing_x = (1 - 2 * margin) / cols
    spacing_y = (1 - 2 * margin) / rows
    offset_y = spacing_y * 0.5

    for i in range(rows):
        for j in range(cols):
            if len(centers) >= n:
                break
            x = margin + (j + 0.5 * (i % 2)) * spacing_x
            y = margin + i * spacing_y + offset_y * (i % 2)
            centers.append([x, y])
            # Radius depends on distance to closest wall
            wall_distances = [x, 1-x, y, 1-y]
            min_wall = min(wall_distances)
            # Make larger radii in center, smaller near corners
            # Base radius is proportional to min_wall, but ensure minimum
            base_r = min_wall * 0.6
            radius = max(0.015, base_r)
            radii.append(radius)

    # Trim to n circles
    centers = centers[:n]
    radii = radii[:n]
    
    # Ensure minimum radius enforced
    min_r = min(radii)
    if min_r < 0.015:
        scale = 0.015 / min_r
        for i in range(n):
            radii[i] *= scale
    return np.array(centers), np.array(radii)

def max_feasible_radius(i, centers, radii):
    # Compute max feasible radius for circle i
    x, y = centers[i]
    r_i = radii[i]
    
    # Distance to walls
    d_left = x
    d_right = 1 - x
    d_bottom = y
    d_top = 1 - y
    
    # Distance to all other circles
    min_dist = float('inf')
    for j in range(len(centers)):
        if i != j:
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist = sqrt(dx*dx + dy*dy)
            min_dist = min(min_dist, dist - radii[j])
    
    # Return the minimum of all constraints
    feasible = min(d_left, d_right, d_bottom, d_top, min_dist)
    return max(0, feasible)

def inflate_radii(centers, radii, alpha):
    # Global inflation step
    n = len(radii)
    mean_r = np.mean(radii)
    delta = alpha * mean_r
    
    # Add delta to all radii
    new_radii = np.copy(radii) + delta
    
    # For each circle, clamp by max feasible radius
    total_increase = 0
    for i in range(n):
        max_r = max_feasible_radius(i, centers, new_radii)
        new_radii[i] = min(new_radii[i], max_r)
        if new_radii[i] > radii[i]:
            total_increase += new_radii[i] - radii[i]
    
    return new_radii, total_increase

def anneal_centers(centers, radii, temp0, gamma, sigma0, steps):
    n = len(centers)
    current_sum = np.sum(radii)
    best_sum = current_sum
    best_centers = np.copy(centers)
    best_radii = np.copy(radii)
    
    temp = temp0
    sigma = sigma0
    
    for step in range(steps):
        # Every 50 steps, shift everything to escape local traps
        if step % 50 == 0:
            delta_shift = np.random.normal(scale=0.001, size=2)
            centers += delta_shift
        
        # Pick a random circle
        i = random.randint(0, n - 1)
        # Displace
        dx = np.random.normal(scale=sigma)
        dy = np.random.normal(scale=sigma)
        old_center = np.copy(centers[i])
        centers[i] += [dx, dy]
        
        # Clamp if out-of-bound
        centers[i] = np.clip(centers[i], 0, 1)
        
        # Compute old and new feasible radius
        old_r = radii[i]
        new_r = max_feasible_radius(i, centers, radii)
        
        delta_r = new_r - old_r
        delta_sum = delta_r
        
        if delta_r > 0 or random.random() < np.exp(delta_r / temp):
            # Accept the move
            radii[i] = new_r
            current_sum += delta_sum
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = np.copy(centers)
                best_radii = np.copy(radii)
        else:
            # Revert
            centers[i] = old_center
        
        # Update temperature and sigma
        if temp > 0.0001:
            temp *= gamma
        if sigma > 0.0001:
            sigma = sigma0 * (temp / temp0)
    
    return best_centers, best_radii, best_sum

def mixed_cycle(centers, radii, params):
    # params: alpha, T0, gamma, sigma0, K, M, P
    alpha, T0, gamma, sigma0, K, M, P = params
    
    old_sum = np.sum(radii)
    stagnation_count = 0
    cycle_count = 0
    n = len(radii)
    
    while True:
        # Phase B: Adaptive inflation
        for _ in range(K):
            new_radii, delta_sum = inflate_radii(centers, radii, alpha)
            radii = new_radii
            if abs(delta_sum) < 1e-6:
                stagnation_count += 1
            else:
                stagnation_count = 0
            if stagnation_count >= K:
                break
        sum_r = np.sum(radii)
        
        if stagnation_count >= K or sum_r <= old_sum + 1e-6:
            # Done with inflation phase
            break

        old_sum = sum_r
        
        # Phase C: Surface annealing
        centers, radii, current_sum = anneal_centers(centers, radii, T0, gamma, sigma0, M)
        if current_sum > old_sum:
            old_sum = current_sum
        else:
            # Revert to last good state
            centers = np.copy(best_centers)
            radii = np.copy(best_radii)
            old_sum = best_sum
        
        cycle_count += 1
        if cycle_count >= P:
            break
            
        # Reset alpha slightly
        alpha *= 0.95
    
    return centers, radii

def polish_radii(centers, radii, beta, passes):
    n = len(radii)
    # Sweep through all circles to slightly increase radii
    for _ in range(passes):
        any_change = True
        iterations = 0
        while any_change and iterations < 5:
            any_change = False
            for i in range(n):
                x, y = centers[i]
                r = radii[i]
                
                # Compute gaps
                gap_wall = min(x, 1-x, y, 1-y)
                gap_nei = float('inf')
                for j in range(n):
                    if i != j:
                        dx = centers[i][0] - centers[j][0]
                        dy = centers[i][1] - centers[j][1]
                        dist = sqrt(dx*dx + dy*dy)
                        gap = dist - (r + radii[j])
                        gap_nei = min(gap_nei, gap)
                
                gap = min(gap_wall, gap_nei)
                
                if gap > 0:
                    new_r = r + beta * gap
                    max_r = max_feasible_radius(i, centers, radii)
                    if new_r <= max_r:
                        radii[i] = new_r
                        any_change = True
            iterations += 1
        
        # Resolve overlaps if any
        # Sweep to correct overlaps one more time
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = sqrt(dx*dx + dy*dy)
                gap = dist - (radii[i] + radii[j])
                if gap < 0:
                    # Correct overlap by reducing both radii
                    overlap = -gap
                    delta = overlap / 2.0
                    radii[i] = max(0, radii[i] - delta)
                    radii[j] = max(0, radii[j] - delta)
