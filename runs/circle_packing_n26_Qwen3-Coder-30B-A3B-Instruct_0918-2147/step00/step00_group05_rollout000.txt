import numpy as np
from scipy.optimize import minimize

def hex_lattice_points(n, d):
    # Generate hexagonal lattice points with spacing d
    rows = []
    y = 0.0
    row = 0
    while len(rows) < n:
        x = 0.0
        if row % 2 == 1:
            x = d / 2
        row_points = []
        while len(row_points) < n and x <= 1.0:
            row_points.append((x, y))
            x += d
        if row_points:
            rows.extend(row_points)
        y += d * np.sqrt(3) / 2
        row += 1
    return np.array(rows[:n])

    d = 0.2  # Initial spacing

    x_min, y_min = points.min(axis=0)
    x_max, y_max = points.max(axis=0)
    x_center = (x_min + x_max) / 2
    y_center = (y_min + y_max) / 2
    points[:, 0] -= x_center
    points[:, 1] -= y_center
    points[:, 0] += 0.5
    points[:, 1] += 0.5
    # Ensure points are within unit square
    points = np.clip(points, 0, 1)
    # Compute safe radii
    r_safe = 1.0
    for i in range(26):
        x, y = points[i]
        r = min(x, 1-x, y, 1-y)
        r_safe = min(r_safe, r)
    # Now adjust all radii to be safe
    radii = np.full(26, r_safe)
    # Flatten for optimization: [x0, y0, r0, x1, y1, r1, ...]
    vars = []
    for i in range(26):
        vars.extend([points[i, 0], points[i, 1], radii[i]])
    return np.array(vars)

def objective(vars):
    # We want to maximize sum of radii -> minimize negative of that
    r_vals = vars[2::3]
    return -np.sum(r_vals)

def wall_constraints(vars):
    # Each circle has 4 wall constraints:
    # x - r >= 0
    # x + r <= 1
    # y - r >= 0
    # y + r <= 1

        res.append(x + r - 1)  # x + r <= 1
        res.append(r - y)  # y - r >= 0
        res.append(y + r - 1)  # y + r <= 1
    return np.array(res)

def nonoverlap_constraints(vars):
    # Non-overlap constraints: (xi - xj)^2 + (yi - yj)^2 >= (ri + rj)^2
    # We return -(dist^2 - (ri + rj)^2) <= 0
    res = []
    n = 26
    for i in range(n):
        for j in range(i + 1, n):
            xi = vars[3*i + 0]
            yi = vars[3*i + 1]
            ri = vars[3*i + 2]
            xj = vars[3*j + 0]
            yj = vars[3*j + 1]
            rj = vars[3*j + 2]
            dx = xi - xj
            dy = yi - yj
            dist_sq = dx*dx + dy*dy
            sum_r_sq = (ri + rj)*(ri + rj)
            res.append(-(dist_sq - sum_r_sq))  # <= 0
    return np.array(res)
