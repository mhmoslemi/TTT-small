import numpy as np
import random

def make_hex_seed(n, margin):
    """Generate a hexagonal seed layout for n circles."""
    sqrt3 = np.sqrt(3)
    rows = int(np.ceil(np.sqrt(n / (sqrt3 / 2))))
    cols = int(np.ceil(n / rows))
    
    spacing_x = (1 - 2 * margin) / (cols - 1) if cols > 1 else 1
    spacing_y = (1 - 2 * margin) / (rows - 1) if rows > 1 else 1
    
    centers = []
    radii = []
    
    for i in range(rows):
        for j in range(cols):
            if len(centers) >= n:
                break
            x = margin + j * spacing_x
            y = margin + i * spacing_y
            if i % 2 == 1:
                x += spacing_x / 2
            centers.append([x, y])
            # Initial radii inversely related to distance to nearest wall
            dist_to_wall = min(x, 1 - x, y, 1 - y)
            r = max(0.015, dist_to_wall * 0.4)
            radii.append(r)
        if len(centers) >= n:
            break
    
    return np.array(centers[:n]), np.array(radii[:n])

def max_feasible_radius(i, centers, radii):
    """Compute the maximum feasible radius for circle i given current layout."""
    x, y = centers[i]
    r_i = radii[i]
    
    # Distance to walls
    d_L = x
    d_R = 1 - x
    d_B = y
    d_T = 1 - y
    
    # Distance to other circles
    min_dist_to_others = float('inf')
    for j in range(len(centers)):
        if i == j:
            continue
        dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
        min_dist_to_others = min(min_dist_to_others, dist - radii[j])
    
    return min(d_L, d_R, d_B, d_T, min_dist_to_others, r_i)

def inflate_radii(centers, radii, alpha):
    """Inflate all radii by a factor and then clamp individually."""
    mean_r = np.mean(radii)
    delta = alpha * mean_r
    new_radii = radii + delta
    
    sum_increase = 0
    for i in range(len(radii)):
        max_r = max_feasible_radius(i, centers, new_radii)
        old_r = radii[i]
        new_radii[i] = min(new_radii[i], max_r)
        sum_increase += (new_radii[i] - old_r)
    
    return new_radii, sum_increase

def anneal_centers(centers, radii, temp0, gamma, sigma0, steps):
    """Perform simulated annealing on centers only."""
    n = len(centers)
    current_sum = np.sum(radii)
    best_sum = current_sum
    best_centers = centers.copy()
    best_radii = radii.copy()
    
    temp = temp0
    sigma = sigma0
    
    for step in range(steps):
        # Occasionally do a global shift to escape deadlock
        if step % 50 == 0 and step > 0:
            shift = np.random.normal(0, 0.001, 2)
            centers += shift
        
        # Proposal: choose a random circle and slightly move it
        i = random.randint(0, n - 1)
        noise = np.random.normal(0, sigma, 2)
        old_center = centers[i].copy()
        centers[i] += noise
        
        # Compute new feasible radius for this circle
        new_r = max_feasible_radius(i, centers, radii)
        old_r = radii[i]
        
        # Accept if increase or with probability based on change
        delta_r = new_r - old_r
        if delta_r >= 0 or np.random.rand() < np.exp(delta_r / temp):
            radii[i] = new_r
            current_sum += delta_r
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = centers.copy()
                best_radii = radii.copy()
        else:
            # Revert change
            centers[i] = old_center
    
        # Update temperature and sigma
        temp *= gamma
        sigma = sigma0 * (temp / temp0)
    
    return best_centers, best_radii, best_sum

def mixed_cycle(centers, radii, params):
    """Run alternating inflation and SA until convergence."""
    alpha = params['alpha']
    epsilon = params['epsilon']
    k = params['k']
    m = params['m']
    p = params['p']
    temp0 = params['temp0']
    gamma = params['gamma']
    sigma0 = params['sigma0']
    
    last_improvement = 0
    cycles_without_improvement = 0
    
    while cycles_without_improvement < p:
        # Phase B: Inflation
        last_total = np.sum(radii)
        stagnated = False
        for i in range(k):
            new_radii, incr = inflate_radii(centers, radii, alpha)
            radii = new_radii
            if incr < epsilon:
                stagnated = True
                break
        
        if stagnated:
            # Phase C: SA
            temp = temp0
            sa_centers, sa_radii, sa_sum = anneal_centers(centers, radii, temp0, gamma, sigma0, m)
            centers = sa_centers
            radii = sa_radii
            last_total = sa_sum
            
            # Re-inflate slightly after SA
            _, incr = inflate_radii(centers, radii, alpha * 0.5)
            if incr < epsilon:
                cycles_without_improvement += 1
            else:
                cycles_without_improvement = 0
        else:
            # Continue with another inflation round
            cycles_without_improvement = 0
            
        alpha *= 0.95  # Decay inflation factor
        
        # Check convergence
        this_total = np.sum(radii)
        if abs(this_total - last_total) < epsilon:
            cycles_without_improvement += 1
        else:
            cycles_without_improvement = 0
    
    return centers, radii

def polish_radii(centers, radii, beta, passes):
    """Apply final polishing to maximize radii to use all available space."""
    for _ in range(passes):
        old_sum = np.sum(radii)
        updated = False
        
        # Try to increase each radius slightly
        for i in range(len(centers)):
            x, y = centers[i]
            r = radii[i]
            
            # Compute local gaps
            g_wall = min(x, 1 - x, y, 1 - y)
            g_nei = float('inf')
            for j in range(len(centers)):
                if i == j:
                    continue
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                g_nei = min(g_nei, dist - radii[j])
            
            g = min(g_wall, g_nei)
            
            if g > 0:
                new_r = r + beta * g
                max_feasible_r = max_feasible_radius(i, centers, radii)
                new_r = min(new_r, max_feasible_r)
                
                if new_r > r + 1e-10:
                    radii[i] = new_r
                    updated = True
        
        # Resolve overlaps if any
        for i in range(len(centers)):
            for j in range(i+1, len(centers)):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                sum_radii = radii[i] + radii[j]
                
                if dist < sum_radii:
                    overlap = sum_radii - dist
                    radii[i] -= overlap / 2
                    radii[j] -= overlap / 2
        
        if not updated:
            break
    
    return centers, radii
