import numpy as np
from scipy.optimize import minimize
import scipy.sparse as sp

def init_hex_grid(n=26, r_init=0.04):
    # Generate hexagonal lattice points and keep first n inside unit square
    h = np.sqrt(3) * r_init
    row_height = h
    col_width = 2 * r_init
    rows = []
    y = r_init
    row_idx = 0
    while len(rows) < n:
        if row_idx % 2 == 0:
            # Even row: full width alignment
            x_start = r_init
        else:
            # Odd row: half width shift
            x_start = r_init + col_width / 2

        x = x_start
        while x <= 1 - r_init and len(rows) < n:
            rows.append((x, y))
            x += col_width
        y += row_height
        row_idx += 1

    centers = np.array(rows[:n])
    radii = np.full(n, r_init)
    return centers, radii

def max_radii(centers, radii):
    max_iter = 100
    for _ in range(max_iter):
        # Compute wall constraints
        wall = np.minimum(centers, 1 - centers).min(axis=1)

        # Compute neighbour constraints
        D = np.sqrt(((centers[:, None, :] - centers[None, :, :]) ** 2).sum(-1))
        np.fill_diagonal(D, np.inf)
        neighbor = D - radii[:, None]
        # Set small negative values to zero to handle numerical errors
        neighbor = np.maximum(neighbor, 0)

        new_radii = np.minimum(wall, neighbor.min(axis=1))

        if np.max(np.abs(new_radii - radii)) < 1e-8:
            break
        radii = new_radii.clip(min=0)

    return radii

def hill_climb(centers, radii, iterations=5000, step=0.01):
    # Iterate N times with random displacements
    best_centers, best_radii = centers.copy(), radii.copy()
    best_sum = float(best_radii.sum())

    for i in range(iterations):
        # Pick a random circle
        idx = np.random.randint(0, len(centers))
        # Propose new center
        old_center = centers[idx].copy()
        new_center = old_center + np.random.uniform(-step, step, 2)
        # Enforce box constraints for new center
        new_center[0] = np.clip(new_center[0], radii[idx], 1 - radii[idx])
        new_center[1] = np.clip(new_center[1], radii[idx], 1 - radii[idx])

        centers[idx] = new_center
        # Re-maximize radii
        new_radii = max_radii(centers, radii.copy())
        # Accept move if it improves the sum
        new_sum = float(new_radii.sum())
        if new_sum > best_sum:
            best_centers, best_radii = centers.copy(), new_radii.copy()
            best_sum = new_sum
        else:
            # Revert center
            centers[idx] = old_center

    return best_centers, best_radii

    # Combine all variables into one vector [x0 y0 x1 y1 ... x25 y25 r0 ... r25]
    # Variables: x0, y0, x1, y1, ..., x25, y25, r0, r1, ..., r25

    r0 = radii
    v = np.concatenate([x0, r0])
    # Constants for constraints
    lb = np.zeros_like(v)
    ub = np.ones_like(v)
    lb[n*2:] = 0
    ub[n*2:] = 0.5
    # Define constraints

        results = []
        for i in range(n):
            x, y = v[2*i], v[2*i + 1]
            r = v[n*2 + i]
            results.extend([r - x, r + x - 1, r - y, r + y - 1])
        return np.array(results)

    def overlap_constraints(v):

        # Non-overlap for all pairs

                xj, yj = v[2*j], v[2*j + 1]
                ri = v[n*2 + i]
                rj = v[n*2 + j]
                dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
                results.append(ri + rj - dist)
        return np.array(results)

        return -np.sum(v[n*2:])  # Minimize negative sum

            # r + x - 1 <= 0

            cols.extend([2*i, 2*i + 1])
            data.extend([1, 1])
            # ind x: 1, y: 1, r: -1 for first

            cols.extend([n*2 + i, n*2 + i])
            data.extend([-1, 1])

        rows = np.array(rows)
        cols = np.array(cols)
        data = np.array(data)
        j = sp.coo_matrix((data, (rows, cols)), shape=(4*n, 2*n + n))
        return j.tocsr()

        cols = []
        data = []
        # Non-overlap for all pairs
        for i in range(n):
            for j in range(i+1, n):

                xj, yj = v[2*j], v[2*j + 1]
                ri = v[n*2 + i]
                rj = v[n*2 + j]
                dist = np.sqrt((xi - xj)**2 + (yi - yj)**2) + 1e-12  # Avoid zero
                dx = (xi - xj) / dist
                dy = (yi - yj) / dist
                # ri + rj - dist <= 0
                rows.extend([len(rows), len(rows), len(rows) + 1, len(rows) + 1])
                cols.extend([2*i, 2*i + 1, 2*j, 2*j + 1])
                data.extend([1, 1, 1, 1])
                # Gradients w.r.t. ri, rj
                rows.extend([len(rows), len(rows)])
                cols.extend([n*2 + i, n*2 + j])
                data.extend([-1, -1])
        rows = np.array(rows)
        cols = np.array(cols)
        data = np.array(data)
        j = sp.coo_matrix((data, (rows, cols)), shape=(n*(n-1)//2, 2*n + n))
        return j.tocsr()

    c2 = NonlinearConstraint(overlap_constraints, -np.inf, 0, jac=overlap_grad)
    res = minimize(obj, v, method='trust-constr', bounds=list(zip(lb, ub)),
                    constraints=[c1, c2], options={'maxiter': 200})

    if res.success:
        new_x = res.x[:n*2]
        new_r = res.x[n*2:]
        centers = new_x.reshape(n, 2)
        radii = new_r
    else:
        # If optimisation failed, use previous best values

def run_packing():

    radii = max_radii(centers, radii)
