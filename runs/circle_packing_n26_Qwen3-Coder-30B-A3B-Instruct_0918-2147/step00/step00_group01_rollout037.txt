import numpy as np
from scipy.optimize import minimize

def initial_layout():
    # Create a hexagonal grid in the unit square
    # Approximate 5 rows x 5 columns for 25 circles
    spacing = 0.18
    rows = 5
    cols = 5
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = (j + 0.5) * spacing
            y = (i + 0.5) * spacing
            if x > 1.0 or y > 1.0:
                continue
            centers.append([x, y])
    
    # If fewer than 25 points, add more spacing near the center
    if len(centers) < 25:
        centers.append([0.5, 0.5])
    
    # Add one more at the center
    if len(centers) < 26:
        centers.append([0.5, 0.5])
    
    # Build them
    centers = np.array(centers[:26])
    radii = np.full(26, spacing / 2 - 0.01)
    
    # Ensure valid spacing
    for i in range(len(centers)):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            x = r + 1e-12
        if x + r > 1:
            x = 1 - r - 1e-12
        if y - r < 0:
            y = r + 1e-12
        if y + r > 1:
            y = 1 - r - 1e-12
        centers[i] = [x, y]
        
    return centers, radii

def pack_vector(centers, radii):
    return np.hstack([centers.ravel(), radii])

def unpack_vector(z):
    n = len(z) // 3
    centers = z[:2*n].reshape((n, 2))
    radii = z[2*n:]
    return centers, radii

def objective(z):
    _, radii = unpack_vector(z)
    return -np.sum(radii)

def constraint_boundary(z):
    centers, radii = unpack_vector(z)
    n = len(radii)
    constraints = []
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # x - r >= 0
        constraints.append(x - r)
        # 1 - x - r >= 0
        constraints.append(1 - x - r)
        # y - r >= 0
        constraints.append(y - r)
        # 1 - y - r >= 0
        constraints.append(1 - y - r)
    return np.array(constraints)

def constraint_nonoverlap(z):
    centers, radii = unpack_vector(z)
    n = len(radii)
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx*dx + dy*dy
            r_sum = radii[i] + radii[j]
            # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
            constraints.append(dist_sq - r_sum*r_sum)
    return np.array(constraints)

def run_packing():
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    for run in range(3):  # Multiple restarts
        centers_init, radii_init = initial_layout()
        
        # Add random jitter to initial layout for variety
        centers_init += np.random.uniform(-0.02, 0.02, centers_init.shape)
        radii_init += np.random.uniform(-0.005, 0.005, radii_init.shape)
        radii_init = np.maximum(radii_init, 1e-6)  # Non-negative radii
        
        z0 = pack_vector(centers_init, radii_init)
        
        # Prepare constraints
        bnds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * 26
        constraint_list = [
            {'type': 'ineq', 'fun': constraint_boundary},
            {'type': 'ineq', 'fun': constraint_nonoverlap}
        ]
        
        # Optimize
        result = minimize(
            objective,
            z0,
            method='SLSQP',
            bounds=bnds,
            constraints=constraint_list,
            options={'ftol': 1e-9, 'maxiter': 2000}
        )
        
        if result.success:
            centers, radii = unpack_vector(result.x)
            # Clip tiny negative radii
            radii = np.maximum(radii, 0.0)
            total_sum = np.sum(radii)
            
            if total_sum > best_sum:
                best_sum = total_sum
                best_centers = centers.copy()
                best_radii = radii.copy()
        else:
