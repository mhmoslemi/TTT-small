import numpy as np
from scipy.stats.qmc import Sobol

def initialize_seeds(n):
    sobol = Sobol(d=2, scramble=False)
    points = sobol.random(n)
    # Clip to interior with margin
    margin = 0.02
    points = margin + (1 - 2*margin) * points
    return points

def compute_feasible_radii(centers):
    n = centers.shape[0]
    # Compute pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    # For each point, find nearest neighbor distance (excluding self)
    np.fill_diagonal(dists, np.inf)
    nn_dists = np.min(dists, axis=1)
    # Wall distances
    wall_dists = np.array([
        centers[:, 0],   # x left
        1 - centers[:, 0],  # x right
        centers[:, 1],   # y bottom
        1 - centers[:, 1]  # y top
    ]).min(axis=0)
    # Feasible radii
    radii = np.minimum(wall_dists, 0.5 * nn_dists)
    return radii

def redistribute_radii(radii, slack, alpha):
    total_slack = np.sum(slack)
    if total_slack <= 1e-12:
        return radii.copy()
    # Increase wall-limited circles
    wall_limited = slack > 1e-12
    if not np.any(wall_limited):
        return radii.copy()
    # Redistribute
    delta = alpha * slack
    new_radii = radii + delta
    # Reduce neighbor-limited circles proportionally
    neighbor_limited = ~wall_limited
    to_reduce = np.sum(delta[wall_limited])
    if to_reduce > 1e-12 and np.any(neighbor_limited):
        frac = radii[neighbor_limited] / np.sum(radii[neighbor_limited])
        reduction = to_reduce * frac
        new_radii[neighbor_limited] = np.maximum(0, radii[neighbor_limited] - reduction)
    return new_radii

def push_away(centers, radii, beta, eps):
    n = centers.shape[0]
    new_centers = centers.copy()
    for i in range(n):
        ci = centers[i]
        ri = radii[i]
        for j in range(n):
            if i == j:
                continue
            cj = centers[j]
            rj = radii[j]
            dist = np.sqrt(np.sum((ci - cj)**2))
            overlap = ri + rj + eps - dist
            if overlap > 0:
                direction = (ci - cj) / (dist + eps)
                new_centers[i] += beta * overlap * direction
    # Clip to interior
    new_centers = np.clip(new_centers, eps, 1 - eps)
    return new_centers

def iterate(centers, radii, alpha=0.4, beta=0.3, max_iter=200):
    for _ in range(max_iter):
        # Compute slack
        nn_dists = np.full(radii.shape, np.inf)
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dists, np.inf)
        nn_dists = np.min(dists, axis=1)
        slack = 0.5 * nn_dists - radii
        # Redistribute
        new_radii = redistribute_radii(radii, slack, alpha)
        # Push away
        new_centers = push_away(centers, new_radii, beta, 1e-8)
        # Recompute radii
        centers = new_centers
        radii = compute_feasible_radii(centers)
        # Check convergence
        sum_change = np.sum(radii) - np.sum(new_radii)
        if abs(sum_change) < 1e-5:
            break
    return centers, radii

def repair(centers, radii, eps_final=5e-10):
    # Shrink radii
    radii = np.maximum(0, radii - eps_final)
    # Clip centers
    centers = np.clip(centers, eps_final, 1 - eps_final)
    # Final repair loop
    n = centers.shape[0]
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i+1, n):
            if dists[i, j] < radii[i] + radii[j] - 1e-12:
                overlap = (radii[i] + radii[j] - dists[i, j])
                change = overlap / 2 + eps_final
                # Reduce larger radius
                if radii[i] >= radii[j]:
                    radii[i] = max(0, radii[i] - change)
                else:
                    radii[j] = max(0, radii[j] - change)
    return centers, radii

def run_packing():
    centers = initialize_seeds(26)
    radii = compute_feasible_radii(centers)
    centers, radii = iterate(centers, radii)
    centers, radii = repair(centers, radii)
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
