import numpy as np
from scipy.optimize import linprog
from scipy.spatial.distance import cdist

def generate_hex_lattice(h):
    sqrt3 = np.sqrt(3)
    points = []
    y_step = h * sqrt3 / 2
    x_step = h
    
    for j in range(int(1 / y_step) + 2):
        y = j * y_step
        if y > 1 + 1e-12:
            break
        for i in range(int(1 / x_step) + 2):
            x = i * x_step + (j % 2) * (x_step / 2)
            if x > 1 + 1e-12:
                break
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append([x, y])
    return np.array(points)

def build_neighbour_list(centers, max_dist=0.4):
    distances = cdist(centers, centers)
    edges = []
    n = centers.shape[0]
    for i in range(n):
        for j in range(i+1, n):
            d = distances[i, j]
            if d < max_dist:
                edges.append((i, j, d))
    return edges

def solve_radius_lp(selected_indices, R_wall, neighbour_list):
    n = len(selected_indices)
    if n == 0:
        return np.array([]), True

    A_eq = []
    b_eq = []
    A_ub = []
    b_ub = []
    c = [-1.0] * n  # Maximize sum of radii (negate for minimization)

    # Wall constraints: 0 <= r_i <= R_wall[i]
    for i, idx in enumerate(selected_indices):
        A_ub.append([0] * i + [-1.0] + [0] * (n - i - 1))  # r_i >= 0
        b_ub.append(0.0)
        A_ub.append([0] * i + [1.0] + [0] * (n - i - 1))  # r_i <= R_wall[idx]
        b_ub.append(R_wall[idx])

    # Neighbour separation constraints
    selected_set = set(selected_indices)
    for i, j, d_ij in neighbour_list:
        if i in selected_set and j in selected_set:
            idx_i = selected_indices.index(i)
            idx_j = selected_indices.index(j)
            A_ub.append([0] * idx_i + [1.0] + [0] * (idx_j - idx_i - 1) + [1.0] + [0] * (n - idx_j - 1))
            b_ub.append(d_ij)

    # Optimization
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, method='highs')
        if res.success:
            radii = res.x
            return radii, True
        else:
            return np.array([]), False
    except Exception:
        return np.array([]), False

def upper_bound(partial_set, remaining_indices, R_wall):
    # Add up the wall radii of remaining points for best-case estimate
    ub_est = sum(R_wall[i] for i in partial_set)
    for i in remaining_indices:
        ub_est += R_wall[i]
    return ub_est

def branch_and_bound(centers, R_wall, neighbour_list, num_circles=26):
    best_sum = -np.inf
    best_solution = None

    # Simple recursive backtracking search
    def search(S_partial, U, lb):
        nonlocal best_sum, best_solution

        if len(S_partial) == num_circles:
            radii, feasible = solve_radius_lp(S_partial, R_wall, neighbour_list)
            if not feasible:
                return lb
            total = np.sum(radii)
            if total > best_sum:
                best_sum = total
                best_solution = (S_partial.copy(), radii.copy())
            return total

        # Pruning step: compute optimistic bound
        ub = upper_bound(S_partial, U, R_wall)
        if ub <= lb:
            return lb

        # Select next candidate: largest wall radius among remaining
        if not U:
            return lb
        i = max(U, key=lambda x: R_wall[x])
        U_new = U - {i}

        # Explore inclusion of i
        S_with = S_partial + [i]
        lb = max(lb, search(S_with, U_new, lb))

        # Explore exclusion of i
        lb = max(lb, search(S_partial, U_new, lb))

        return lb

    # Start search
    U = set(range(len(centers)))
    search([], U, -np.inf)

    if best_solution is None:
        raise RuntimeError("No valid solution found in branch-and-bound.")

    selected_indices, radii_solution = best_solution
    final_centers = centers[selected_indices]
    return final_centers, radii_solution

def local_continuous_refinement(centers, radii):
    # Refinement using SLSQP
    def objective(x):
        # x is [cx1, cy1, r1, cx2, cy2, r2, ..., cx26, cy26, r26]
        radii_vals = x[2::3]
        return -np.sum(radii_vals)

    def constraint_func(x):
        # Ensure each circle is within bounds
        for i in range(26):
            cx = x[3*i]
            cy = x[3*i + 1]
            r = x[3*i + 2]
            if r < 0 or cx - r < 0 or cx + r > 1 or cy - r < 0 or cy + r > 1:
                return 1e12  # Invalid
        # Ensure no overlaps
        for i in range(26):
            for j in range(i+1, 26):
                cx_i = x[3*i]
                cy_i = x[3*i + 1]
                r_i = x[3*i + 2]
                cx_j = x[3*j]
                cy_j = x[3*j + 1]
                r_j = x[3*j + 2]
                dist_sq = (cx_i - cx_j)**2 + (cy_i - cy_j)**2
                if dist_sq < (r_i + r_j)**2:
                    return 1e12  # Overlapping
        return 0

    # Initial values
    x0 = np.zeros(3*26)
    for i in range(26):
        x0[3*i] = centers[i][0]
        x0[3*i + 1] = centers[i][1]
        x0[3*i + 2] = radii[i]

    from scipy.optimize import minimize
    res = minimize(objective, x0, method='SLSQP', constraints={'type': 'eq', 'fun': constraint_func}, options={'maxiter': 100})
    if not res.success:
        # Fall back to initial if refinement fails
        pass
    else:
        # Extract refined values
        refined_centers = np.zeros((26, 2))
        refined_radii = np.zeros(26)
        for i in range(26):
            refined_centers[i][0] = res.x[3*i]
            refined_centers[i][1] = res.x[3*i + 1]
            refined_radii[i] = res.x[3*i + 2]

        centers, radii = refined_centers, refined_radii

    return centers, radii


def run_packing():
    # Generate candidate lattice points
    h = 0.07
    centers_lattice = generate_hex_lattice(h)
    R_wall = np.array([min(p[0], 1-p[0], p[1], 1-p[1]) for p in centers_lattice])
    edge_list = build_neighbour_list(centers_lattice, max_dist=0.4)

    # Branch and bound
    final_centers, radii = branch_and_bound(centers_lattice, R_wall, edge_list, 26)

    # Local refinement
    final_centers, radii = local_continuous_refinement(final_centers, radii)

    # Final validation
    is_valid, msg = validate_packing(final_centers, radii)
    if not is_valid:
