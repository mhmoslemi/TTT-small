import numpy as np
from scipy.optimize import minimize

def initial_layout():
    # Create a hexagonal grid inside the unit square
    n_cols = 5
    n_rows = 5
    spacing = 0.18
    radius = spacing / 2 - 0.01
    
    centers = []
    for i in range(n_rows):
        for j in range(n_cols):
            x = (j + 0.5) * spacing
            y = (i + 0.5) * spacing
            # Push back into square if needed
            x = max(radius, min(1-radius, x))
            y = max(radius, min(1-radius, y))
            centers.append([x, y])
    
    # Add a 26th circle in the center if it fits
    centers.append([0.5, 0.5])
    
    centers = np.array(centers)
    radii = np.full(26, radius)
    
    return centers, radii

def pack_vector(centers, radii):
    z = np.zeros(3 * 26)
    for i in range(26):
        z[3*i] = centers[i, 0]
        z[3*i+1] = centers[i, 1]
        z[3*i+2] = radii[i]
    return z

def unpack_vector(z):
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    for i in range(26):
        centers[i, 0] = z[3*i]
        centers[i, 1] = z[3*i+1]
        radii[i] = z[3*i+2]
    return centers, radii

def objective(z):
    _, radii = unpack_vector(z)
    return -np.sum(radii)

def constraint_boundary(z):
    centers, radii = unpack_vector(z)
    constraints = []
    for i in range(26):
        x, y = centers[i]
        r = radii[i]
        constraints.append(x - r)         # x - r >= 0
        constraints.append(1 - x - r)     # 1 - x - r >= 0
        constraints.append(y - r)         # y - r >= 0
        constraints.append(1 - y - r)     # 1 - y - r >= 0
    return np.array(constraints)

def constraint_nonoverlap(z):
    centers, radii = unpack_vector(z)
    constraints = []

            dist_sq = dx*dx + dy*dy
            r_sum = radii[i] + radii[j]
            constraints.append(dist_sq - r_sum*r_sum)
    return np.array(constraints)
