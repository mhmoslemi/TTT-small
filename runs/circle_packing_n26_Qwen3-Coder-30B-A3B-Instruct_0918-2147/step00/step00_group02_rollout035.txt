import numpy as np
from scipy.optimize import minimize
from ortools.linear_solver import pywraplp
from itertools import combinations

def generate_candidates(g, radius_levels):
    candidates = []
    grid_points = np.arange(0, 1 + g/2, g)
    for x in grid_points:
        for y in grid_points:
            for r in radius_levels:
                if (x - r >= 0 and x + r <= 1 and 
                    y - r >= 0 and y + r <= 1):
                    candidates.append((x, y, r))
    return candidates

def build_milp(candidates):
    solver = pywraplp.Solver.CreateSolver('SCIP')
    if not solver:
        solver = pywraplp.Solver.CreateSolver('CBC')
    n = len(candidates)
    
    # Binary variables for selecting candidates
    z = [solver.IntVar(0, 1, f'z_{i}') for i in range(n)]
    
    # Grid point constraints
    grid_dict = {}
    for i, (x, y, r) in enumerate(candidates):
        grid_key = (round(x/g), round(y/g))
        if grid_key not in grid_dict:
            grid_dict[grid_key] = []
        grid_dict[grid_key].append(i)
    
    for key, indices in grid_dict.items():
        solver.Add(sum(z[i] for i in indices) <= 1)
    
    # Exactly 26 circles selected
    solver.Add(sum(z) == 26)
    
    # Non-overlap constraints (L-infinity linearization)
    big_m = 1.0
    dx_vars = {}
    dy_vars = {}
    
    for i, j in combinations(range(n), 2):
        x1, y1, r1 = candidates[i]
        x2, y2, r2 = candidates[j]
        
        # Create auxiliary variables
        dx_ij = solver.NumVar(0, 1, f'dx_{i}_{j}')
        dy_ij = solver.NumVar(0, 1, f'dy_{i}_{j}')
        
        # Add constraints 
        solver.Add(dx_ij >= x1 - x2)
        solver.Add(dx_ij >= x2 - x1)
        solver.Add(dy_ij >= y1 - y2)
        solver.Add(dy_ij >= y2 - y1)
        solver.Add(dx_ij + dy_ij >= r1 + r2)
        solver.Add(dx_ij - dy_ij >= r1 + r2 - 2*big_m*(1 - z[i] - z[j]))
        solver.Add(dx_ij + dy_ij >= r1 + r2 - 2*big_m*(1 - z[i] - z[j]))
        
        dx_vars[(i,j)] = dx_ij
        dy_vars[(i,j)] = dy_ij
    
    # Objective: maximize sum of radii
    solver.Maximize(sum(candidates[i][2] * z[i] for i in range(n)))
    
    return solver, z, candidates

def solve_milp(solver, z, candidates):
    status = solver.Solve()
    if status in [pywraplp.Solver.OPTIMAL, pywraplp.Solver.FEASIBLE]:
        selected_indices = [i for i in range(len(candidates)) if z[i].solution_value() > 0.5]
        return selected_indices
    else:
        raise RuntimeError("MILP solver did not find a feasible solution")

def extract_initial_solution(selected_indices, candidates):
    centers = np.array([[candidates[i][0], candidates[i][1]] for i in selected_indices])
    radii = np.array([candidates[i][2] for i in selected_indices])
    return centers, radii

def shrink_initial(radii, delta):
    return radii - delta

def pack_variables(centers, radii):
    z = np.column_stack([centers, radii])
    return z.flatten()

def unpack_solution(z):
    n = len(z) // 3
    centers = z[:2*n].reshape(n, 2)
    radii = z[2*n:]
    return centers, radii

def compute_bounds(centers, radii):
    bounds = []
    for i in range(len(radii)):
        bounds.extend([[0, 1], [0, 1], [0, 0.5]])  # x, y, r bounds
    return bounds

def define_objective(z):
    return -np.sum(z[2::3])  # Minimize negative of sum of radii

def define_jacobian(z):
    n = len(z) // 3
    grad = np.zeros(len(z))
    grad[2::3] = -1.0  # Gradient w.r.t. radii
    return grad

def define_constraints(z):
    n = len(z) // 3
    centers = z[:2*n].reshape(n, 2)
    radii = z[2*n:]
    
    constraints = []
    
    for i in range(n):
        # Boundary constraints
        x, y = centers[i]
        r = radii[i]
        constraints.append({
            'fun': lambda z, i=i: r - x,
            'jac': lambda z, i=i: np.array([0]*2*i + [-1.0, 0.0] + [0]*2*(n-i-1) + [-1.0] + [0]*n)
        })
        constraints.append({
            'fun': lambda z, i=i: r - (1 - x),
            'jac': lambda z, i=i: np.array([0]*2*i + [1.0, 0.0] + [0]*2*(n-i-1) + [-1.0] + [0]*n)
        })
        constraints.append({
            'fun': lambda z, i=i: r - y,
            'jac': lambda z, i=i: np.array([0]*2*i + [0.0, -1.0] + [0]*2*(n-i-1) + [-1.0] + [0]*n)
        })
        constraints.append({
            'fun': lambda z, i=i: r - (1 - y),
            'jac': lambda z, i=i: np.array([0]*2*i + [0.0, 1.0] + [0]*2*(n-i-1) + [-1.0] + [0]*n)
        })
        
        # Inter-circle constraints
        for j in range(i+1, n):
            x1, y1 = centers[i]
            x2, y2 = centers[j]
            r1 = radii[i]
            r2 = radii[j]
            
            def constraint_fun(z, i=i, j=j):
                dx = x1 - x2
                dy = y1 - y2
                dist = np.sqrt(dx*dx + dy*dy)
                return (r1 + r2) - dist
            
            def constraint_jac(z, i=i, j=j):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx*dx + dy*dy) + 1e-12
                grad = np.zeros(len(z))
                grad[2*i] = (dx / dist) - 1.0  # Partial w.r.t. x1
                grad[2*i+1] = (dy / dist) - 1.0  # Partial w.r.t. y1
                grad[2*j] = -(dx / dist) - 1.0  # Partial w.r.t. x2
                grad[2*j+1] = -(dy / dist) - 1.0  # Partial w.r.t. y2
                grad[2*n+i] = -1.0  # Partial w.r.t. r1
                grad[2*n+j] = -1.0  # Partial w.r.t. r2
                return grad
                
            constraints.append({
                'fun': constraint_fun,
                'jac': constraint_jac
            })
    
    return constraints

def repair_solution(centers, radii):
    n = len(centers)
    # Ensure radii fit inside boundaries
    for i in range(n):
        max_radius = min(centers[i][0], 1 - centers[i][0], 
                         centers[i][1], 1 - centers[i][1])
        radii[i] = min(radii[i], max_radius)
    
    # Repair overlapping circles
    for iter in range(5):
        any_change = False
        for i, j in combinations(range(n), 2):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist = np.sqrt(dx*dx + dy*dy)
            overlap = (radii[i] + radii[j]) - dist
            if overlap > 0:
                reduction = overlap / 2
                radii[i] = max(1e-10, radii[i] - reduction)
                radii[j] = max(1e-10, radii[j] - reduction)
                any_change = True
        if not any_change:
            break
    
    return centers, radii

def run_packing():
    g = 0.035
    radius_levels = np.arange(0.02, 0.26, 0.02)
    
    candidates = generate_candidates(g, radius_levels)
    solver, z_vars, candidates = build_milp(candidates)
    selected_indices = solve_milp(solver, z_vars, candidates)
    
    centers, radii = extract_initial_solution(selected_indices, candidates)
    radii = shrink_initial(radii, 1e-5)
    
    # Pack variables for continuous optimization
    z0 = pack_variables(centers, radii)
    
    # Perform trust-constr optimization
    bounds = compute_bounds(centers, radii)
    constraints = define_constraints(z0)
    
    sol = minimize(
        fun=define_objective,
        x0=z0,
        method='trust-constr',
        jac=define_jacobian,
        constraints=constraints,
        bounds=bounds,
        options={'maxiter': 1000, 'disp': False}
    )
    
    # Extract final solution
    centers, radii = unpack_solution(sol.x)
    
    # Repair final solution
    centers, radii = repair_solution(centers, radii)
    sum_radii = np.sum(radii)
    
    # Validate
    success, message = validate_packing(centers, radii)
    if not success:
        raise RuntimeError(f"Validation failed: {message}")
    
    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
