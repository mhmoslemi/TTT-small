import numpy as np
import random

def make_hex_seed(n, margin):
    # Generate a hexagonal seed arrangement for n circles
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))
    
    centers = []
    radii = []
    
    spacing_x = (1 - 2*margin) / (cols - 1) if cols > 1 else 1
    spacing_y = (1 - 2*margin) / (rows - 1) if rows > 1 else 1
    
    for i in range(rows):
        for j in range(cols):
            x = margin + j * spacing_x
            y = margin + i * spacing_y
            
            if len(centers) >= n:
                break
                
            # Distance to walls for a conservative estimate of initial radius
            d_left = x
            d_right = 1 - x
            d_bottom = y
            d_top = 1 - y
            
            # Take minimum distance to a wall as initial estimated radius
            r = min(d_left, d_right, d_bottom, d_top)
            
            # Reduce based on position: prefer larger radii in center
            dist_to_center = np.sqrt((x - 0.5)**2 + (y - 0.5)**2)
            r = min(r, 0.15 * (1 - dist_to_center))
            
            # Ensure we have a non-zero radius
            r = max(r, 0.015)
            
            centers.append([x, y])
            radii.append(r)
            
            if len(centers) >= n:
                break
                
    return np.array(centers[:n]), np.array(radii[:n])

def max_feasible_radius(i, centers, radii):
    # Compute the maximum feasible radius for circle i
    x, y = centers[i]
    r_current = radii[i]
    
    # Distance to walls
    d_left = x
    d_right = 1 - x
    d_bottom = y
    d_top = 1 - y
    
    # Distance to other circles
    d_neighbours = []
    for j in range(centers.shape[0]):
        if i != j:
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            d_neighbours.append(dist - radii[j])
            
    min_d = min(d_left, d_right, d_bottom, d_top)
    if d_neighbours:
        min_d = min(min_d, min(d_neighbours))
        
    return min(min_d, r_current)

def inflate_radii(centers, radii, alpha):
    # Perform global inflation and then shrink violating radii
    old_sum = np.sum(radii)
    mean_r = np.mean(radii)
    delta = alpha * mean_r
    
    # Increase all radii
    radii_new = radii + delta
    
    # Clamp each radius to max feasible
    for i in range(len(radii_new)):
        r_max = max_feasible_radius(i, centers, radii_new)
        radii_new[i] = min(radii_new[i], r_max)
        
    new_sum = np.sum(radii_new)
    return radii_new, (new_sum - old_sum)

def anneal_centers(centers, radii, temp0, gamma, sigma0, steps):
    # Perform simulated annealing on center positions
    n = centers.shape[0]
    current_radii = radii.copy()
    best_centers = centers.copy()
    best_radii = current_radii.copy()
    best_sum = np.sum(best_radii)
    
    temp = temp0
    
    for step in range(steps):
        # Perturb a random circle
        i = random.randint(0, n - 1)
        sigma = sigma0 * temp / temp0
        
        # Sample a displacement
        dx, dy = np.random.normal(0, sigma, 2)
        
        # Save old center
        old_center = centers[i].copy()
        old_radius = current_radii[i]
        
        # Update center
        centers[i] += [dx, dy]
        
        # Ensure the circle is within boundaries
        centers[i] = np.clip(centers[i], 0, 1)
        
        # Compute feasible radius for moved circle
        r_max = max_feasible_radius(i, centers, current_radii)
        
        if r_max > old_radius:
            current_radii[i] = r_max
            new_sum = np.sum(current_radii)
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = centers.copy()
                best_radii = current_radii.copy()
        else:
            # Accept with probability based on temperature
            delta = r_max - old_radius
            if random.random() < np.exp(delta / temp):
                current_radii[i] = r_max
                new_sum = np.sum(current_radii)
                if new_sum > best_sum:
                    best_sum = new_sum
                    best_centers = centers.copy()
                    best_radii = current_radii.copy()
            else:
                # Put back
                centers[i] = old_center
                current_radii[i] = old_radius
                
        # Periodically shift entire configuration to escape local traps
        if step % 50 == 0 and step > 0:
            shift = np.random.uniform(-0.001, 0.001, 2)
            centers += shift
        
        temp *= gamma
        
    return best_centers, best_radii, best_sum

def mixed_cycle(centers, radii, params):
    # Run the inflating and repositioning cycle
    alpha = params['alpha_start']
    epsilon = params['epsilon']
    k_stagnation = params['k_stagnation']
    m_sa_steps = params['m_sa_steps']
    p_cycles = params['p_cycles']
    temp0 = params['temp0']
    gamma = params['gamma']
    sigma0 = params['sigma0']
    
    last_improvement = 0
    convergence_cycles = 0
    
    while convergence_cycles < p_cycles:
        # Phases B (inflation) and C (SA reposition)
        stagnated = 0
        while stagnated < k_stagnation:
            old_sum = np.sum(radii)
            radii, improvement = inflate_radii(centers, radii, alpha)
            new_sum = np.sum(radii)
            
            if improvement < epsilon:
                stagnated += 1
            else:
                stagnated = 0
                
            if stagnated < k_stagnation:
                # Do a small number of inflation steps to re-adjust
                for _ in range(3):
                    _, improvement = inflate_radii(centers, radii, alpha)
                    if improvement < epsilon:
                        break
            else:
                break
                
        # Phase C (stochastic center re-optimization)
        centers, radii, best_sum = anneal_centers(centers, radii, temp0, gamma, sigma0, m_sa_steps)
        
        # Try to improve if total sum seems to stall
        if last_improvement > 0:
            if abs(best_sum - last_improvement) < epsilon:
                convergence_cycles += 1
        else:
            convergence_cycles = 0
            
        last_improvement = best_sum
        alpha *= 0.95  # Decay inflation parameter
        
    return centers, radii

def polish_radii(centers, radii, beta, passes):
    # Apply final polishing to extract small gains
    n = centers.shape[0]
    
    for _ in range(passes):
        improved = True
        while improved:
            improved = False
            for i in range(n):
                x, y = centers[i]
                r = radii[i]
                
                # Compute gap to nearest wall and neighbour
                g_wall = min(x, 1 - x, y, 1 - y)
                g_nei = float('inf')
                for j in range(n):
                    if i != j:
                        dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                        g = dist - (r + radii[j])
                        if g < g_nei:
                            g_nei = g
                            
                g = min(g_wall, g_nei)
                if g > 0:
                    delta = beta * g
                    r_new = r + delta
                    r_max = max_feasible_radius(i, centers, radii)
                    r_new = min(r_new, r_max)
                    
                    if r_new > r:
                        radii[i] = r_new
                        improved = True
                        
        # Resolve any overlap
        for _ in range(3):
            any_changed = False
            for i in range(n):
                for j in range(i + 1, n):
                    dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                    overlap = dist - (radii[i] + radii[j])
                    
                    if overlap < 0:
                        delta = -overlap / 2
                        radii[i] = max(0, radii[i] - delta)
                        radii[j] = max(0, radii[j] - delta)
                        any_changed = True
                        
            if not any_changed:
                break
    return centers, radii

def run_packing():
    # Set parameters
    params = {
        'alpha_start': 0.04,
        'epsilon': 1e-6,
        'k_stagnation': 8,
        'm_sa_steps': 200,
        'p_cycles': 5,
        'temp0': 0.02,
        'gamma': 0.97,
        'sigma0': 0.03,
        'beta': 0.6,
        'polish_passes': 3
    }
    
    # Initialize
    centers, radii = make_hex_seed(26, 0.04)
    
    # Phase B-C mixed cycle
    centers, radii = mixed_cycle(centers, radii, params)
    
    # Phase E polishing
    centers, radii = polish_radii(centers, radii, params['beta'], params['polish_passes'])
    
    # Final validation
    success, message = validate_packing(centers, radii)
    if not success:
        raise RuntimeError("Final packing is not valid: " + message)
        
    return centers, radii, np.sum(radii)
