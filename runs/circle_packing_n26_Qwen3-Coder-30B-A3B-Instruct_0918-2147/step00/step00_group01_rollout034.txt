import numpy as np
from scipy.spatial.distance import cdist

def penalty(z, alpha=10.0, beta=10.0):
    # z: (26, 3) array with [x, y, r] for each circle
    centers = z[:, :2]
    radii = z[:, 2]
    
    # Wall penalties
    wall_distances = np.array([
        centers[:, 0],           # x
        1.0 - centers[:, 0],     # 1 - x
        centers[:, 1],           # y
        1.0 - centers[:, 1]      # 1 - y
    ])
    
    wall_violations = np.maximum(0.0, radii - wall_distances.T).sum(axis=1)
    wall_penalty = alpha * wall_violations.sum()
    
    # Overlap penalties
    distances = cdist(centers, centers)
    rsums = radii.reshape(-1, 1) + radii.reshape(1, -1)
    overlap = np.maximum(0.0, rsums - distances)
    overlap.fill_diagonal(0.0)  # No self-overlap
    overlap_penalty = beta * overlap.sum() / 2.0
    
    return wall_penalty + overlap_penalty

def energy(z, lam, alpha=10.0, beta=10.0):
    total_radius = z[:, 2].sum()
    return -total_radius + lam * penalty(z, alpha, beta)

def propose_move(z, sigma_c, sigma_r, temp_scale=1.0):
    new_z = z.copy()
    move_type = np.random.choice([0, 1, 2])
    
    if move_type == 0:  # Center jitter
        idx = np.random.randint(0, 26)
        new_z[idx, 0] += np.random.normal(0, sigma_c * temp_scale)
        new_z[idx, 1] += np.random.normal(0, sigma_c * temp_scale)
        new_z[idx, 0] = np.clip(new_z[idx, 0], 0.0, 1.0)
        new_z[idx, 1] = np.clip(new_z[idx, 1], 0.0, 1.0)
    
    elif move_type == 1:  # Radius perturb
        idx = np.random.randint(0, 26)
        new_z[idx, 2] += np.random.normal(0, sigma_r * temp_scale)
        new_z[idx, 2] = np.maximum(new_z[idx, 2], 0.0)
    
    else:  # Swap pair
        i, j = np.random.choice(26, size=2, replace=False)
        new_z[i, 2], new_z[j, 2] = new_z[j, 2], new_z[i, 2]
    
    return new_z

def anneal():
    np.random.seed(42)  # For reproducible results
    
    # Initialize 26 circles in a hexagonal grid
    grid_spacing = 0.18
    rows = int(1.0 / grid_spacing) + 1
    cols = int(1.0 / grid_spacing) + 1
    centers = np.array([
        (i * grid_spacing + 0.01, j * grid_spacing + 0.01)
        for i in range(cols) for j in range(rows)
        if i * grid_spacing < 1.0 and j * grid_spacing < 1.0
    ])
    centers = centers[:26]  # Take up to 26
    
    # Assign initial radii (max allowed by boundaries)
    radii = np.array([
        min(x, 1 - x, y, 1 - y) for x, y in centers
    ])
    
    z = np.column_stack([centers, radii])
    
    # Annealing parameters
    T0 = 1.0
    T = T0
    c_T = 0.995
    lam = 0.1
    c_lam = 1.03
    sigma_c = 0.03
    sigma_r = 0.02
    max_iter = 10000
    temp_scale = 1.0
    
    best_feasible_z = z.copy()
    best_feasible_sum = z[:, 2].sum()
    best_energy = energy(z, lam)
    
    # Keep track of previous steps for regularization
    last_z = z.copy()
    energy_history = []
    
    for iteration in range(max_iter):
        # Annealing step size controls
        temp_scale = np.sqrt(T / T0)
        z_proposed = propose_move(z, sigma_c, sigma_r, temp_scale)
        
        E_old = energy(z, lam)
        E_new = energy(z_proposed, lam)
        delta_E = E_new - E_old
        
        if delta_E <= 0 or np.random.rand() < np.exp(-delta_E / T):
            z = z_proposed
        
        # Update T and lam
        T *= c_T
        if T < 1e-5:
            break
        lam *= c_lam
        
        # Update best feasible
        p = penalty(z)
        if p < 1e-8:
            z_sum = z[:, 2].sum()
            if z_sum > best_feasible_sum:
                best_feasible_sum = z_sum
                best_feasible_z = z.copy()
        
        energy_history.append(E_new)
    
    return best_feasible_z

def project_and_ascent(centers, radii):
    # z := (26, 3)
    z = np.column_stack([centers, radii])
    eta = 5e-3
    max_iter = 500
    tol = 1e-7
    
    for i in range(max_iter):
        # We want to increase sum of radii, ∂O/∂r = 1
        
        # Improvement step (gradient ascent without constraint)
        z[:, 2] += eta  # This is a very simplified ascent (step size too large but early stop will catch it)
        
        # Project to ensure non-negativity and wall limits
        walls = np.array([
            z[:, 0], 1.0 - z[:, 0], z[:, 1], 1.0 - z[:, 1]
        ])
        max_radii = np.min(walls, axis=0)
        z[:, 2] = np.minimum(z[:, 2], max_radii)
        z[:, 2] = np.maximum(z[:, 2], 0.0)
        
        # Resolve overlaps by shrinking radii
        centers = z[:, :2]
        radii = z[:, 2]
        
        dist = cdist(centers, centers)
        rsums = radii.reshape(-1, 1) + radii.reshape(1, -1)
        overlaps = np.maximum(0.0, rsums - dist)
        overlaps.fill_diagonal(0.0)
        
        total_change = 0.0
        for j in range(26):
            if np.any(overlaps[j] > 1e-12):
                # Reduce radii to resolve overlaps with all neighbors
                to_shrink = overlaps[j]
                shrink = to_shrink / 2.0
                sanitize = np.minimum(shrink, z[j, 2])
                z[j, 2] -= sanitize
                total_change += sanitize
                # Apply reduction to neighbors too
                for k in range(26):
                    if k != j and overlaps[j,k] > 1e-12:
                        reduce_amount = overlaps[j,k] / 2.0
                        reduce_amount = np.minimum(reduce_amount, z[k, 2])
                        if reduce_amount > 1e-12:
                            z[k, 2] -= reduce_amount
                            total_change += reduce_amount
            z[:, 2] = np.maximum(z[:, 2], 0.0)
            z[:, 2] = np.minimum(z[:, 2], np.array([
                z[i, 0], 1.0 - z[i, 0], z[i, 1], 1.0 - z[i, 1]
            ]).min(axis=0))
            
        if total_change < tol or i > max_iter - 50:
            break

        eta *= 0.95
        
        if i % 50 == 0:
            # Check if valid for early exit
            if penalty(z) < 1e-8:
                pass  # continue polishing
    
    # Final cleanup
    z[:, 2] = np.minimum(z[:, 2], np.array([
        z[i, 0], 1.0 - z[i, 0], z[i, 1], 1.0 - z[i, 1]
    ]).min(axis=0))
    z[:, 2] = np.maximum(z[:, 2], 0.0)
    
    return z[:, :2], z[:, 2]

def run_packing():
    # Run annealing
    z = anneal()
    
    # Extract centers and radii
    centers = z[:, :2]
    radii = z[:, 2]
    sum_radii = radii.sum()
    
    # Run local refinement
    centers, radii = project_and_ascent(centers, radii)
    
    # Final validation
    is_valid, msg = validate_packing(centers, radii)
    
    if not is_valid:
        # Apply uniform shrink of 1e-7 if necessary
        radii *= (1 - 1e-7)
        is_valid, msg = validate_packing(centers, radii)
        if not is_valid:
            raise RuntimeError(f"Post-refinement validation failed: {msg}")
    
    sum_radii = radii.sum()
    
    return centers, radii, sum_radii
