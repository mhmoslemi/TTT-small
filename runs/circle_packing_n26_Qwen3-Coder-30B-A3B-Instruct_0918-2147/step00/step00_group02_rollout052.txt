import numpy as np
from scipy.optimize import minimize
from scipy.stats import qmc

def run_packing():
    n = 26
    B0 = 3.0
    delta = 5e-4
    lambda_budget = 100.0
    max_outer_iter = 40
    eps = 5e-10

    # Initial configuration using Sobol sequence
    sampler = qmc.Sobol(d=2, scramble=True)
    points = sampler.random(n)
    centers, radii = initial_configuration(points)

    # Pack into flat vector
    z_start = pack(centers, radii)

    # Outer optimization loop with budget adaptation
    B = B0
    z_best = z_start.copy()
    sum_r_best = 0.0

    for outer_iter in range(max_outer_iter):
        z = inner_optimize(z_start, B, lambda_budget)
        sum_r = np.sum(unpack(z)[1])
        if sum_r > sum_r_best:
            sum_r_best = sum_r
            z_best = z.copy()

        # Compute maximum violation
        c, r = unpack(z)
        v_max = max_feasible_violation(c, r)
        if v_max <= 1e-13:
            B += delta
        else:
            B -= delta
        if abs(delta) < 1e-6:
            break
        z_start = z.copy()

    # Final processing
    c, r = unpack(z_best)
    c, r = postprocess(c, r, eps)
    sum_r = np.sum(r)

    return c, r, sum_r

def initial_configuration(points):
    n = points.shape[0]
    centers = points.copy()
    radii = np.zeros(n)
    
    # Compute wall distances and nearest neighbor distances
    for i in range(n):
        x, y = points[i]
        r_wall = min(x, 1-x, y, 1-y)
        
        min_dist = 1e10
        for j in range(n):
            if i != j:
                dx = x - points[j, 0]
                dy = y - points[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                min_dist = min(min_dist, dist)
        
        r_nn = 0.5 * min_dist if min_dist > 0 else 0.0
        
        radii[i] = min(r_wall, r_nn)
    
    # Ensure all radii are feasible
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Clamp to wall constraints
        r_max = min(x, 1-x, y, 1-y)
        if r > r_max:
            radii[i] = r_max
        if radii[i] < 0:
            radii[i] = 0.0

    return centers, radii

def pack(centers, radii):
    n = centers.shape[0]
    z = np.empty(2*n + n)
    z[:n] = centers[:, 0]
    z[n:2*n] = centers[:, 1]
    z[2*n:] = radii
    return z

def unpack(z):
    n = (z.shape[0] - 26) // 3
    centers = np.empty((n, 2))
    centers[:, 0] = z[:n]
    centers[:, 1] = z[n:2*n]
    radii = z[2*n:]
    return centers, radii

def energy_and_grad(z):
    c, r = unpack(z)
    n = c.shape[0]
    E = 0.0
    grad = np.zeros_like(z)
    
    # Wall repulsion energy and gradient
    for i in range(n):
        x, y = c[i]
        ri = r[i]
        # Distance to boundaries
        dxl = ri - x
        dxr = ri - (1 - x)
        dyl = ri - y
        dyr = ri - (1 - y)
        # Energy contribution  
        E += dxl**2 * (dxl > 0) + dxr**2 * (dxr > 0) + dyl**2 * (dyl > 0) + dyr**2 * (dyr > 0)
        # Gradient component
        grad[i] += 2 * dxl * (dxl > 0) - 2 * dxr * (dxr > 0)
        grad[n + i] += 2 * dyl * (dyl > 0) - 2 * dyr * (dyr > 0)
        grad[2*n + i] += 2 * (dxl * (dxl > 0) + dxr * (dxr > 0) + dyl * (dyl > 0) + dyr * (dyr > 0))

    # Pairwise overlap energy and gradient
    for i in range(n):
        for j in range(i + 1, n):
            dx = c[i, 0] - c[j, 0]
            dy = c[i, 1] - c[j, 1]
            d = np.sqrt(dx*dx + dy*dy)
            sum_r = r[i] + r[j]
            diff = sum_r - d
            if diff > 0:
                E += diff**2
                ratio = 2 * diff / (d + 1e-12)
                grad[i] -= ratio * dx
                grad[n + i] -= ratio * dy
                grad[j] += ratio * dx
                grad[n + j] += ratio * dy
                grad[2*n + i] += diff
                grad[2*n + j] += diff

    # Budget constraint
    sum_r = np.sum(r)
    E += 100 * (1.0 - sum_r)**2
    penalty = 200 * (1.0 - sum_r)
    grad[2*n:] += penalty

    return E, grad

def inner_optimize(z0, B, lambda_budget):
    # Set bounds for coordinates and radii
    bounds = []
    for i in range(26):
        bounds.append((0.0, 1.0))  # x coordinates  
    for i in range(26):
        bounds.append((0.0, 1.0))  # y coordinates
    for i in range(26):
        bounds.append((0.0, 0.5))  # radii

    # Perform optimization
    res = minimize(
        energy_and_grad,
        z0,
        method='L-BFGS-B',
        bounds=bounds,
        jac=True,
        options={'maxiter': 200, 'ftol': 1e-12}
    )

    if not res.success:
        raise RuntimeError(f"Optimization failed: {res.message}")

    return res.x

def max_feasible_violation(centers, radii):
    n = centers.shape[0]
    v_max = 0.0

    # Check wall violation (maximum extra radius beyond wall constraints)
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        dists = [x, 1-x, y, 1-y]
        for dist in dists:
            if r > dist:
                v_max = max(v_max, r - dist)

    # Check overlap violation
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = np.sqrt(dx*dx + dy*dy)
            sum_r = radii[i] + radii[j]
            diff = sum_r - d
            if diff < 0:
                v_max = max(v_max, -diff)
                
    return v_max

def postprocess(centers, radii, eps):
    # Clamp centers to avoid edge issues
    centers = np.clip(centers, eps, 1-eps)
    
    # Shrink radii
    radii = np.maximum(radii - eps, 0.0)
    
    # Optional repair pass
    n = centers.shape[0]
    fixed = True
    while fixed:
        fixed = False
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                d = np.sqrt(dx*dx + dy*dy)
                sum_r = radii[i] + radii[j]
                overlap = sum_r - d
                
                if overlap > 0:
                    # Fix overlap by reducing larger radius
                    r1, r2 = radii[i], radii[j]
                    if r1 > r2:
                        radii[i] = r1 - overlap/2 - eps
                    else:
                        radii[j] = r2 - overlap/2 - eps
                    fixed = True
                    
    radii = np.maximum(radii, 0.0)
    return centers, radii
