import numpy as np
import cvxpy as cp
from scipy.spatial.distance import pdist, squareform

def build_socp_problem(n):
    """Build a reusable SOCP problem for computing radii given centres."""
    r = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(r)), [])
    dist_params = []
    return prob, r, dist_params

def solve_radii(centres, prob, r, dist_params):
    """Solve SOCP to get optimal radii given fixed centres."""
    n = centres.shape[0]
    # Compute pairwise distances
    dist_matrix = squareform(pdist(centres, metric='euclidean'))
    # Update parameters and solve
    for i in range(len(dist_params)):
        dist_params[i].value = dist_matrix[i//n, i%n]
    prob.solve(solver=cp.ECOS, abstol=1e-12, reltol=1e-12)
    radii = r.value
    duals = [c.dual_value for c in prob.constraints if hasattr(c, 'dual_value')]
    return radii, duals

def energy(centres, prob, r, dist_params):
    """Return -sum of radii for given centres."""
    radii, _ = solve_radii(centres, prob, r, dist_params)
    return -np.sum(radii)

def random_move(state, rng):
    """Generate a random move to perturb the state."""
    new_state = state.copy()
    move_type = rng.choice([0, 1, 2, 3])
    if move_type == 0: # single jitter
        idx = rng.integers(0, len(state)//2)
        new_state[2*idx:2*idx+2] += rng.normal(0, 0.02, 2)
        new_state[2*idx:2*idx+2] = np.clip(new_state[2*idx:2*idx+2], 0, 1)
    elif move_type == 1: # cluster jitter
        num = rng.integers(3, 6)
        indices = rng.choice(len(state)//2, size=num, replace=False)
        offset = rng.normal(0, 0.02, 2)
        for i in indices:
            new_state[2*i:2*i+2] += offset
            new_state[2*i:2*i+2] = np.clip(new_state[2*i:2*i+2], 0, 1)
    elif move_type == 2: # edge push
        idx = rng.integers(0, len(state)//2)
        dx = -np.sign(new_state[2*idx] - 0.5) * 0.01
        dy = -np.sign(new_state[2*idx+1] - 0.5) * 0.01
        new_state[2*idx] += dx
        new_state[2*idx+1] += dy
        new_state[2*idx:2*idx+2] = np.clip(new_state[2*idx:2*idx+2], 0, 1)
    else:  # swap order
        new_state = new_state.copy()
    return new_state

def anneal_chain(seed, prob, r, dist_params, rng):
    """Run a single simulated annealing chain."""
    n = 26
    initial_centers = np.random.default_rng(seed).uniform(0, 1, (n, 2))
    state = initial_centers.flatten()
    current_energy = energy(initial_centers, prob, r, dist_params)
    best_state = state.copy()
    best_energy = current_energy
    T0 = 0.05
    T = T0
    alpha = 0.995
    N_iter = 150000
    accepted = 0

    for i in range(N_iter):
        new_state = random_move(state, rng)
        new_centers = new_state.reshape((n, 2))
        new_energy = energy(new_centers, prob, r, dist_params)
        delta = new_energy - current_energy
        if delta < 0 or rng.random() < np.exp(-delta / T):
            current_energy = new_energy
            state = new_state.copy()
            if current_energy < best_energy:
                best_energy = current_energy
                best_state = state.copy()
        T = T0 * (alpha ** i)
        if T < 1e-6:
            break
        if accepted >= N_iter:
            break
    best_centers = best_state.reshape((n, 2))
    best_radii, _ = solve_radii(best_centers, prob, r, dist_params)
    return best_centers, best_radii, -best_energy

def local_refine(centres, radii, prob, r, dist_params, rng):
    """Refine centres using active constraints."""
    current_sum = np.sum(radii)
    for iteration in range(10):
        new_centres = centres.copy()
        # Find active constraints
        _, duals = solve_radii(centres, prob, r, dist_params)
        # All non-boundary constraints should be enforced with zero slack
        tol = 1e-6
        active_pairs = [(i, j) for i in range(len(duals)) for j in range(i+1, len(duals)) if duals[i + len(duals)*j] > tol]
        if not active_pairs:
            break
        # Simple refinement: move points in direction of current normals (approx)
        # Might do better with exact least-square over active constraints
        moved = 0
        for i, (idx1, idx2) in enumerate(active_pairs):
            dist = np.linalg.norm(centres[idx1] - centres[idx2])
            if abs(dist - (radii[idx1] + radii[idx2])) < 1e-4:
                dir_vec = centres[idx1] - centres[idx2]
                if np.linalg.norm(dir_vec) > 1e-9:
                    dir_vec = dir_vec / np.linalg.norm(dir_vec)
                    move = (radii[idx1] + radii[idx2] - dist)*0.1 * dir_vec
                    new_centres[idx1] += move
                    new_centres[idx2] -= move
                    moved = 1
        if not moved:
            break
        centres = new_centres
        radii, _ = solve_radii(centres, prob, r, dist_params)
        new_sum = np.sum(radii)
        if abs(new_sum - current_sum) < 1e-5:
            break
        current_sum = new_sum
    return centres, radii

def finalise(centres, radii):
    """Final verification and tightening."""
    n = centres.shape[0]
    # Re-solve with stricter tolerances
    r = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(r)), [])
    # Constraints for boundary
    r_bounds = [
        r <= centres[:, 0],
        r <= 1 - centres[:, 0],
        r <= centres[:, 1],
        r <= 1 - centres[:, 1]
    ]
    # Add distance constraints
    dist_matrix = squareform(pdist(centres, metric='euclidean'))
    distance_constraints = []
    for i in range(n):
        for j in range(i+1, n):
            distance_constraints.append(r[i] + r[j] <= dist_matrix[i, j])
    prob.constraints = r_bounds + distance_constraints
    prob.solve(solver=cp.ECOS, abstol=1e-12, reltol=1e-12)
    radii = r.value
    # Tighten any that're close to bounds
    x = centres[:, 0]
    y = centres[:, 1]
    for i in range(n):
        min_r = min(x[i], 1-x[i], y[i], 1-y[i])
        if radii[i] > min_r - 1e-13:
            radii[i] = min_r
    return centres, radii

def run_packing():
    """Main packing function returning centers, radii, and sum."""
    n = 26
    # Setup tolerances for reuse
    # Build outer SOCP problem
    r = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(r)), [])
    all_dist_params = []
    
    # Constraint generation for bounds
    r_left = r <= cp.Parameter(n, value=[0]*n)  # placeholder values
    r_right = r <= cp.Parameter(n, value=[0]*n)
    r_bottom = r <= cp.Parameter(n, value=[0]*n)
    r_top = r <= cp.Parameter(n, value=[0]*n)
    
    # Non-overlap constraints
    dist_params = []
    for i in range(n):
        for j in range(i+1, n):
            param = cp.Parameter(value=0.0)
            constraint = r[i] + r[j] <= param
            prob.constraints.append(constraint)
            dist_params.append(param)

    # Actual entry point
    # Precompute pairwise distances for later use
    rng = np.random.default_rng(42)
    results = []
    seed_configs = [
        "hex",
        "grid",
        "corner",
        "random"
    ]
    temp_centers = []
    temp_radii = []
    temp_sums = []
    
    num_chains = 4  # Less CPU, more manageable runtime
    for i in range(num_chains):
        seed_offset = i*100
        if seed_configs[i % 4] == "hex":
            # Make a hexagonal lattice pattern
            centers = []
            rows = 6
            cols = 5
            for row in range(rows):
                for col in range(cols):
                    x = (col + (row % 2)*0.5) / (cols - 0.5)
                    y = row / (rows - 1)
                    if x >= 0 and x <= 1 and y >= 0 and y <= 1:
                        centers.append([x, y])
                    if len(centers) == n:
                        break
                if len(centers) == n:
                    break
            while len(centers) < n:
                centers.append([0.02 + 0.01*i, 0.02 + 0.01*i])
            centers = np.array(centers[:n])
        elif seed_configs[i % 4] == "grid":
            arr = np.linspace(0.05, 0.95, max(1, int(np.ceil(np.sqrt(n)))))
            centers = np.array([[x, y] for y in arr for x in arr])
            centers = centers[:n]
            if len(centers) < n:
                extra = np.random.rand(n - len(centers), 2)
                centers = np.vstack([centers, extra])
        elif seed_configs[i % 4] == "corner":
            # Place 4 circles in corners, rest in middle
            centers = [
                [0.02, 0.02],
                [0.98, 0.02],
                [0.02, 0.98],
                [0.98, 0.98]
            ]
            while len(centers) < n:
                centers.append([0.5, 0.5])
            centers = np.array(centers[:n])
        else:  # random
            centers = rng.uniform(0, 1, (n, 2))
        # Just use one run per seed
        try:
            centers = centers
            radii, _ = solve_radii(centers, prob, r, dist_params)
            sum_radii = np.sum(radii)
            results.append((centers, radii, sum_radii))
        except:
            continue

    best_result = max(results, key=lambda x: x[2])
    refined_centers, refined_radii = local_refine(best_result[0], best_result[1], prob, r, dist_params, rng)
    final_centers, final_radii = finalise(refined_centers, refined_radii)
    final_sum = np.sum(final_radii)

    return final_centers, final_radii, final_sum
