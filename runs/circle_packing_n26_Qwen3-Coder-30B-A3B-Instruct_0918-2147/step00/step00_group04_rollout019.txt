import numpy as np
from scipy.optimize import linprog

def initial_hex_centers(k, spacing):
    """Generate k centres from a hexagonal lattice within [0,1]^2."""
    sqrt3 = np.sqrt(3)
    s = spacing
    cols = int(np.ceil(1.0 / s))
    rows = int(np.ceil(1.0 / (s * sqrt3 / 2)))
    centers = []
    for i in range(cols):
        for j in range(rows):
            x = i * s + (j % 2) * s / 2
            y = j * s * sqrt3 / 2
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers.append([x, y])
                if len(centers) == k:
                    break
        if len(centers) == k:
            break
    return np.array(centers[:k])

def initial_radii(k, r0):
    """Initialize all radii to r0."""
    return np.full(k, r0)

def build_lp_constants(centers, radii):
    """Build constants needed for LP solver."""
    n = centers.shape[0]
    # Wall distances
    d_left = centers[:, 0]
    d_right = 1.0 - centers[:, 0]
    d_bottom = centers[:, 1]
    d_top = 1.0 - centers[:, 1]
    # Pairwise distances
    D = np.zeros((n, n))
    for i in range(n):
        for j in range(i+1, n):
            diff = centers[i] - centers[j]
            dist = np.sqrt(np.sum(diff ** 2))
            D[i, j] = dist
            D[j, i] = dist
    return d_left, d_right, d_bottom, d_top, D

def solve_radius_lp(consts, radii, graph, alpha):
    """
    Solve the linear prog to compute Δr.
    Return the change in radii.
    """
    d_left, d_right, d_bottom, d_top, D = consts
    n = len(radii)
    # Build constraint matrix A_ub and vector b_ub
    # Wall capacity constraints
    rows = 0
    rows += n  # 26 wall constraints
    rows += n * (n - 1) // 2  # pairwise constraints
    # Optional: graph-based transfer limits (not yet used)
    A_ub = np.zeros((rows, n))
    b_ub = np.zeros(rows)
    idx = 0
    # Wall constraints
    wall_caps = np.array([
        np.minimum(np.minimum(d_left, d_right),
                   np.minimum(d_bottom, d_top)) - radii
    ]).T
    # Clamp negative values to 0
    wall_caps = np.maximum(wall_caps, 0)
    A_ub[idx:idx+n, :] = -np.eye(n)
    b_ub[idx:idx+n] = -wall_caps.flatten()
    idx += n
    # Pairwise constraints
    for i in range(n):
        for j in range(i+1, n):
            # Δr_i + Δr_j <= D_ij - (r_i + r_j)
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = D[i, j] - (radii[i] + radii[j])
            idx += 1
    # Objective = maximize sum of Δr
    c = np.ones(n)
    # Solve
    res = linprog(-c, A_ub=A_ub, b_ub=b_ub, method="highs")
    if not res.success:
        # Debug print about infeasibility if needed
        delta_r = np.zeros(n)
    else:
        delta_r = res.x
    return delta_r

def apply_increment(radii, delta_r):
    """Add Δr to current radii."""
    return radii + delta_r

def push_feasibility(centers, radii):
    """Geometrically push centers to restore feasibility."""
    n = centers.shape[0]
    # Functorial command-style transformations (i.e. in place)
    updated_centers = centers.copy()
    updated_radii = radii.copy()
    # Wall constraints
    on_wall = set()
    for i in range(n):
        x, y = updated_centers[i]
        r = updated_radii[i]
        if x - r < 1e-15:
            updated_radii[i] = x
            on_wall.add((i, "left"))
        if x + r > 1 - 1e-15:
            updated_radii[i] = 1 - x
            on_wall.add((i, "right"))
        if y - r < 1e-15:
            updated_radii[i] = y
            on_wall.add((i, "bottom"))
        if y + r > 1 - 1e-15:
            updated_radii[i] = 1 - y
            on_wall.add((i, "top"))
    # Pairwise overlaps
    new_edges = []
    for i in range(n):
        for j in range(i+1, n):
            diff = updated_centers[i] - updated_centers[j]
            dist = np.sqrt(np.sum(diff ** 2))
            r_sum = updated_radii[i] + updated_radii[j]
            if dist < r_sum - 1e-12:
                overlap = r_sum - dist
                u = diff / dist  # unit vector from j to i
                move = (overlap / 2.0) * u
                updated_centers[i] -= move
                updated_centers[j] += move
                # Correct if wall bounds violated
                for idx, dir in [(i, "left"), (i, "right"), (i, "bottom"), (i, "top")]:
                    r = updated_radii[idx]
                    x, y = updated_centers[idx]
                    if x - r < 1e-15:
                        updated_radii[idx] = x
                    elif x + r > 1 - 1e-15:
                        updated_radii[idx] = 1 - x
                    if y - r < 1e-15:
                        updated_radii[idx] = y
                    elif y + r > 1 - 1e-15:
                        updated_radii[idx] = 1 - y
                new_edges.append((i, j))
    return updated_centers, updated_radii, new_edges

def update_graph(graph, new_edges, wall_contacts):
    """Update the contact graph G."""
    # Add edges to graph
    for e in new_edges:
        if e not in graph and (e[1], e[0]) not in graph:
            graph.add(e)
    return graph

def run_iterations(centers, radii, max_iter, ε, α):
    """Main iterative loop."""
    graph = set()  # set of tuples of indices
    for _ in range(max_iter):
        # 1. Build LP constants
        consts = build_lp_constants(centers, radii)
        # 2. Solve LP for Δr
        delta_r = solve_radius_lp(consts, radii, graph, α)
        # 3. Apply Δr
        radii = apply_increment(radii, delta_r)
        # 4. Push to fix overlaps and bound violations
        old_centers = centers.copy()
        centers, radii, new_edges = push_feasibility(centers, radii)
        # Update the graph
        graph = update_graph(graph, new_edges, set())
        # 5. Convergence check
        total_del_r = np.sum(delta_r)
        if total_del_r < ε:
            break
    return centers, radii, graph

def final_refinement(centers, radii):
    """Perform high-precision final refinement."""
    # Rebuild LP constants using final centers
    consts = build_lp_constants(centers, radii)
    # Try to improve final result
    delta_r = solve_radius_lp(consts, radii, set(), 0.5)
    radii = apply_increment(radii, delta_r)
    # Apply final clamping
    for i in range(len(radii)):
        x, y = centers[i]
        radii[i] = min(radii[i], x, 1-x, y, 1-y)
    # Final pairwise fixes
    n = len(centers)
    for i in range(n):
        for j in range(i+1, n):
            diff = centers[i] - centers[j]
            dist = np.sqrt(np.sum(diff ** 2))
            r_sum = radii[i] + radii[j]
            if dist < r_sum - 1e-13:
                delta = r_sum - dist
                radii[i] -= delta / 2
                radii[j] -= delta / 2
    return centers, radii
