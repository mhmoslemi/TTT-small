import numpy as np
from scipy.optimize import linprog

def run_packing():
    # Parameters
    n_circles = 26
    k = 8  # number of sides for polygonal approximation
    cos_pi_k = np.cos(np.pi / k)
    epsilon = 1e-6
    max_iter = 30
    inflation_max_passes = 5
    shrink_factor = 1e-7

    # Step 1: Geometry-first layout
    # Generate triangular lattice pattern inside [0, 0.8] x [0, 0.8]
    spacing = 0.8 / np.sqrt(n_circles)
    cols = int(np.ceil(0.8 / spacing))
    rows = int(np.ceil(0.8 / (spacing * np.sqrt(3) / 2)))
    
    centers = []
    for j in range(rows):
        y = j * spacing * np.sqrt(3) / 2
        if y > 0.8: continue
        for i in range(cols):
            x = i * spacing + (j % 2) * spacing / 2
            if x > 0.8: continue
            centers.append([x, y])
            if len(centers) == n_circles:
                break
        if len(centers) == n_circles:
            break
    
    centers = np.array(centers, dtype=np.float64)
    
    # Translate to ensure margin
    margin = 0.05
    centers[:, 0] += margin
    centers[:, 1] += margin

    # Initial radii
    radii = np.full(n_circles, 0.01, dtype=np.float64)
    
    # Pre-compute unit vectors for polygonal constraint approximation
    angles = np.linspace(0, 2*np.pi, k, endpoint=False)
    unit_vectors = np.column_stack((np.cos(angles), np.sin(angles)))
    
    # Step 2 & 3: Alternating linear programming loop
    for iteration in range(max_iter):
        # Solve for optimal radii for current centers
        # Variable: r = [r1, r2, ..., r26]
        # Constraints:
        # 1. Wall constraints
        # r_i <= c_i.x
        # r_i <= 1 - c_i.x
        # r_i <= c_i.y
        # r_i <= 1 - c_i.y
        # 2. Separation constraints
        # r_i + r_j <= distance(c_i,c_j)
        A_ub = []
        b_ub = []
        A_ub.append(-np.eye(n_circles))
        b_ub.append(np.zeros(n_circles))
        
        # Wall constraints
        for i in range(n_circles):
            # r_i <= c_i.x
            row = np.zeros(n_circles)
            row[i] = -1
            A_ub.append(row)
            b_ub.append(-centers[i, 0])  # -c_i.x
            # r_i <= 1 - c_i.x
            row = np.zeros(n_circles)
            row[i] = -1
            A_ub.append(row)
            b_ub.append(cents[i, 0] - 1)  # c_i.x - 1
            # r_i <= c_i.y
            row = np.zeros(n_circles)
            row[i] = -1
            A_ub.append(row)
            b_ub.append(-centers[i, 1])  # -c_i.y
            # r_i <= 1 - c_i.y
            row = np.zeros(n_circles)
            row[i] = -1
            A_ub.append(row)
            b_ub.append(centers[i, 1] - 1)  # c_i.y - 1
        
        # Separation constraints
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                distance = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                row = np.zeros(n_circles)
                row[i] = 1
                row[j] = 1
                A_ub.append(row)
                b_ub.append(distance)
                
        A_ub = np.array(A_ub, dtype=np.float64)
        b_ub = np.array(b_ub, dtype=np.float64)
        
        # Solve LP
        result = linprog(
            c=np.ones(n_circles),
            A_ub=A_ub,
            b_ub=b_ub,
            method='highs-ds',
            options={'tol': 1e-10}
        )
        if not result.success:
            break
        new_radii = result.x
        
        # Apply clipping to prevent numerical issues
        new_radii = np.clip(new_radii, 0, 0.5)
        
        # Solve center movement LP
        # Linearized separation constraints approximated with k-gon
        # Let us compute the slack cylinders to control center positions
        A_ub_cent = []
        b_ub_cent = []
        
        # Add constraints to keep centers inside square with radii margin
        for i in range(n_circles):
            # c_i.x - r_i >= s_i  =>  c_i.x >= r_i + s_i
            row = np.zeros(n_circles + n_circles)  # for 'c' and 's'
            row[i * 2 + 0] = 1  # c_i.x
            row[n_circles + i] = -1  # -s_i
            A_ub_cent.append(row)
            b_ub_cent.append(radii[i])
            # 1 - c_i.x - r_i >= s_i  =>  c_i.x <= 1 - r_i - s_i
            row = np.zeros(n_circles + n_circles)
            row[i * 2 + 0] = -1
            row[n_circles + i] = -1
            A_ub_cent.append(row)
            b_ub_cent.append(-1 + radii[i])
            # c_i.y - r_i >= s_i
            row = np.zeros(n_circles + n_circles)
            row[i * 2 + 1] = 1  # c_i.y
            row[n_circles + i] = -1  # -s_i
            A_ub_cent.append(row)
            b_ub_cent.append(radii[i])
            # 1 - c_i.y - r_i >= s_i
            row = np.zeros(n_circles + n_circles)
            row[i * 2 + 1] = -1
            row[n_circles + i] = -1
            A_ub_cent.append(row)
            b_ub_cent.append(-1 + radii[i])
        
        # Add polygonal exclusion constraints:
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                # Euclidean distance between centers
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist > 0:
                    # For each direction, compute support function of congruent prohibitive set
                    # Represent angular region in terms of a set of halfplanes
                    r_sum = radii[i] + radii[j]
                    for vec in unit_vectors:
                        # residual dist cutoff for this polygonal approximation
                        cutoff_dist = r_sum * cos_pi_k
                        # u·(c_i - c_j) <= -cutoff_dist
                        row = np.zeros(n_circles + n_circles)
                        row[i*2  ] = vec[0]  # u·c_i.x
                        row[j*2  ] = -vec[0]  # -u·c_j.x
                        row[i*2+1] = vec[1]  # u·c_i.y
                        row[j*2+1] = -vec[1]  # -u·c_j.y
                        row[n_circles+i] = -1  # -s_i
                        row[n_circles+j] = -1  # -s_j
                        A_ub_cent.append(row)
                        b_ub_cent.append(-cutoff_dist)
                    
        A_ub_cent = np.array(A_ub_cent, dtype=np.float64)
        b_ub_cent = np.array(b_ub_cent, dtype=np.float64)
        
        # Maximize sum of slack variables (multiplied by -1 for linprog)
        c_cent = np.zeros(n_circles + n_circles, dtype=np.float64)
        c_cent[n_circles:] = -1  # Maximize sum of s_i (i from n_circles to 2*n_circles - 1)
        
        try:
            result_c = linprog(
                c=c_cent,
                A_ub=A_ub_cent,
                b_ub=b_ub_cent,
                method='highs-ds',
                options={'tol': 1e-10}
            )
            if not result_c.success:
                break
            # Extract new centers from result
            new_centers = np.zeros_like(centers)
            for i in range(n_circles):
                new_centers[i, 0] = result_c.x[i * 2]
                new_centers[i, 1] = result_c.x[i * 2 + 1]
        except Exception:
            break

        # Update centers and radii
        centers = new_centers
        radii = new_radii
        
        # Break if no significant improvement
        if iteration > 0 and np.sum(radii) - np.sum(old_radii) < epsilon:
            break
        old_radii = radii
    
    # Step 4: Post-optimization with local inflation
    for _ in range(inflation_max_passes):
        improved = False
        distances = []
        for i in range(n_circles):
            v = centers[i]
            min_dist = float('inf')
            # Wall distances
            min_dist = min(min_dist, v[0])  # left
            min_dist = min(min_dist, 1 - v[0])  # right
            min_dist = min(min_dist, v[1])  # bottom
            min_dist = min(min_dist, 1 - v[1])  # top
            
            for j in range(n_circles):
                if i != j:
                    d = np.sqrt(np.sum((v - centers[j]) ** 2))
                    dist_sum = radii[i] + radii[j]
                    if d < dist_sum:
                        distance = d
                    else:
                        distance = d - dist_sum
                    if distance <= min_dist and distance > 0:
                        min_dist = distance
            
            distances.append(min_dist)
        
        if not any(d < 0.001 for d in distances):
            break
            
        # Try to increase radii based on local slack
        new_radii = radii.copy()
        for i in range(n_circles):
            slack = distances[i]
            if slack > 0 and slack > 1e-5:
                inc = min(0.4 * slack, 0.02)  # limit individual increments
                new_radii[i] = min(radii[i] + inc, 0.5)
                
        # Re-solve LP with new radii (limited to only affected circles)
        # Simple heuristics for local refinement
        A_ub = []
        b_ub = []
        A_ub.append(-np.eye(n_circles))
        b_ub.append(np.zeros(n_circles))
        
        # Wall constraints again
        for i in range(n_circles):
            row = np.zeros(n_circles)
            row[i] = -1
            A_ub.append(row)
            b_ub.append(-centers[i, 0])
            row = np.zeros(n_circles)
            row[i] = -1
            A_ub.append(row)
            b_ub.append(centers[i, 0] - 1)
            row = np.zeros(n_circles)
            row[i] = -1
            A_ub.append(row)
            b_ub.append(-centers[i, 1])
            row = np.zeros(n_circles)
            row[i] = -1
            A_ub.append(row)
            b_ub.append(centers[i, 1] - 1)
        
        # Separation constraints: check which ones need updated radii
        # Aggressive but safe conservative recheck
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                row = np.zeros(n_circles)
                row[i] = 1
                row[j] = 1
                A_ub.append(row)
                b_ub.append(dist)
                
        A_ub = np.array(A_ub, dtype=np.float64)
        b_ub = np.array(b_ub, dtype=np.float64)
        
        try:
            result = linprog(
                c=np.ones(n_circles),
                A_ub=A_ub,
                b_ub=b_ub,
                method='highs-ds',
                options={'tol': 1e-10}
            )
            if result.success:
                final_radii = result.x
                final_radii = np.clip(final_radii, 0, 0.5)
                # Only update if beneficial
                if np.sum(final_radii) > np.sum(radii):
                    radii = final_radii
                    improved = True
        except Exception:
            pass
            
        if not improved:
            break
            
    # Precomputed array for cleaning small issues

    radii = np.clip(radii, 0, 0.5)

            break
        else:

            to_shrink = radii > 0
            radii[to_shrink] *= (1.0 - shrink_factor)
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
