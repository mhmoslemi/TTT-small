import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def init_candidate_grid(R, Δ):
    # Precompute candidate centers and radii
    candidates = []
    for r in R:
        # Spacing to ensure no overlap for same radius
        s = r * np.sqrt(2)
        # Grid within [r, 1-r] to respect boundary conditions
        x_vals = np.arange(r, 1 - r + Δ / 10, s)
        y_vals = np.arange(r, 1 - r + Δ / 10, s)
        for x in x_vals:
            for y in y_vals:
                candidates.append((x, y, r))
    return candidates

def build_overlap_matrix(candidates):
    n = len(candidates)
    overlap = np.zeros((n, n), dtype=bool)
    centers = np.array([[c[0], c[1]] for c in candidates])
    radii = np.array([c[2] for c in candidates])
    
    # Compute pairwise distances
    dists = cdist(centers, centers)
    
    # Mark overlapping pairs
    for i in range(n):
        for j in range(i+1, n):
            if dists[i,j] < radii[i] + radii[j]:
                overlap[i,j] = True
                overlap[j,i] = True
    return overlap

def solve_discrete_bip(candidates, overlap_matrix, target_sum):

    # Simplified greedy search, clos to minimal knapsack variant
    n = len(candidates)
    # Greedy initial to ensure feasibility (infeasible outputs not handled)
    # Sorted by radius descent
    indices = np.argsort([c[2] for c in candidates])[::-1]
    selected = []
    sum_rad = 0.0
    used = np.zeros(n, dtype=bool)
    
    for i in indices:
        if used[i]: continue
        overlap_with_selected = np.any(overlap_matrix[i, used])
        if not overlap_with_selected:
            if sum_rad + candidates[i][2] > target_sum:
                break
            selected.append(i)
            sum_rad += candidates[i][2]
            used[i] = True
            # Mark all overlapping candidates as blocked
            used |= overlap_matrix[i]
            if len(selected) == 26:

    if len(selected) < 26:

    final_indices = selected[:26]

    return final_centers, final_radii

def max_radii_fixed_centres(centers, radii_initial):
    n = centers.shape[0]
    radii = radii_initial.copy()
    
    for i in range(n):
        # Compute minimal distance to other circles
        min_dist = np.inf
        for j in range(n):
            if i != j:
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                min_dist = min(min_dist, dist)
        
        # Maximal radius constrained by wall and other circles
        r_max = min(centers[i][0], 1 - centers[i][0],
                    centers[i][1], 1 - centers[i][1],
                    min_dist / 2)
        radii[i] = min(radii[i], r_max)
    
    return radii

def nlp_refine(centers_initial, radii_initial):
    n = centers_initial.shape[0]
    # Flatten coordinates and radii
    x0 = np.hstack([centers_initial.flatten(), radii_initial])

    def objective(x):
        # Return negative of sum of radii (we minimize)
        radii = x[2*n:]
        return -np.sum(radii)
    
    def constraints(x):
        # Return array of constraint values (<=0)
        centers = x[:2*n].reshape((n, 2))
        radii = x[2*n:]
        
        # Wall constraints (each radii should be <= distance to wall)
        wall_constraints = []
        for i in range(n):
            wall_constraints.extend([
                radii[i] - centers[i][0],       # x - r >= 0
                radii[i] + centers[i][0] - 1,   # x + r <= 1
                radii[i] - centers[i][1],       # y - r >= 0
                radii[i] + centers[i][1] - 1    # y + r <= 1
            ])
        
        # Overlap constraints: (r_i + r_j)^2 >= (x_i-x_j)^2 + (y_i-y_j)^2
        overlap_constraints = []
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist_sq = dx*dx + dy*dy
                overlap_constraints.append((radii[i] + radii[j])**2 - dist_sq)
        
        return np.array(wall_constraints + overlap_constraints)
    
    def jacobian(x):
        # Compute Jacobian matrices for constraints
        centers = x[:2*n].reshape((n, 2))
        radii = x[2*n:]
        J = np.zeros((len(constraints(x)), len(x)))
        
        # Wall Jacobian
        for i in range(n):
            J[i*4][2*i] = 1  # d/dx_i (r_i - x_i)
            J[i*4+1][2*i] = 1  # d/dx_i (r_i + x_i - 1)
            J[i*4+2][2*i+1] = 1  # d/dy_i (r_i - y_i)
            J[i*4+3][2*i+1] = 1  # d/dy_i (r_i + y_i - 1)
            J[i*4][2*n+i] = 1  # d/dr_i (r_i - x_i)
            J[i*4+1][2*n+i] = 1  # d/dr_i (r_i + x_i - 1)
            J[i*4+2][2*n+i] = 1  # d/dr_i (r_i - y_i)
            J[i*4+3][2*n+i] = 1  # d/dr_i (r_i + y_i - 1)
        
        # Overlap Jacobian (4 types of derivatives)
        coef = 0
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist_sq = dx*dx + dy*dy
                dist = np.sqrt(dist_sq)
                if dist_sq > 0:
                    d_dr_i_dxi = (radii[i] + radii[j]) / dist
                    d_dr_j_dxi = (radii[i] + radii[j]) / dist
                    d_dr_i_dyi = d_dr_i_dxi
                    d_dr_j_dyi = d_dr_j_dxi
                    
                    # For i
                    J[4*n + coef][2*i] = -dx * d_dr_i_dxi
                    J[4*n + coef][2*i+1] = -dy * d_dr_i_dxi
                    J[4*n + coef][2*n+i] = d_dr_i_dxi
                    
                    # For j
                    J[4*n + coef][2*j] = dx * d_dr_j_dxi
                    J[4*n + coef][2*j+1] = dy * d_dr_j_dxi
                    J[4*n + coef][2*n+j] = d_dr_j_dxi
                else:
                    # Assume no derivative if directly on top (could add small epsilon if needed)
                    J[4*n + coef][2*i] = 0
                    J[4*n + coef][2*i+1] = 0
                    J[4*n + coef][2*n+i] = 0
                    J[4*n + coef][2*j] = 0
                    J[4*n + coef][2*j+1] = 0
                    J[4*n + coef][2*n+j] = 0
                coef += 1
        
        return J
    
    # Bounds: all variables (centers and radii) must be feasible
    lower_bounds = np.zeros_like(x0)
    upper_bounds = np.ones_like(x0)
    for i in range(n):
        lower_bounds[2*i] = 0  # also valid check
        lower_bounds[2*i + 1] = 0
        lower_bounds[2*n + i] = 0
        upper_bounds[2*i] = 1
        upper_bounds[2*i+1] = 1
        upper_bounds[2*n+i] = 1  # assuming maximum radius is 1
    
    # Compile constraint bounds
    con = {'type': 'ineq', 'fun': constraints}
    # We pass derivative information, but I don't have it implemented because theoretical one vs numeric differs
    # For now we use the default method as our priority is described functionality
    res = minimize(objective, x0, method='trust-constr', constraints=con, 
                   bounds=list(zip(lower_bounds, upper_bounds)),
                   options={'maxiter': 300, 'xtol': 1e-9, 'gtol': 1e-9})
    
    if not res.success:
        print("NLP Optimization failed:", res.message)
        raise RuntimeError("NLP optimization failed for refining solution.")
    
    # Extract refined values
    refined_centers = res.x[:2*n].reshape((n, 2))
    refined_radii = res.x[2*n:]
    
    return refined_centers, refined_radii
