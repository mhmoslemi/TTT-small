import numpy as np
from scipy.optimize import minimize

def hex_lattice_points(n, d):
    """Generate at least n points in a hexagonal lattice within [0,1]^2."""
    y_spacing = np.sqrt(3) / 2 * d
    rows = int(np.ceil(np.sqrt(n / (np.sqrt(3)/2))))
    points = []
    for i in range(rows):
        y = i * y_spacing
        if y > 1:
            break
        offset = d/2 if i % 2 == 1 else 0
        for j in range(int(np.ceil(1 / d)) + 1):
            x = j * d + offset
            if x > 1:
                break
            points.append([x, y])
            if len(points) >= n:
                break
        if len(points) >= n:
            break
    return np.array(points[:n])

def initial_guess():

    d = 0.2  # lattice spacing
    points = hex_lattice_points(26, d)
    # Center the points
    center = np.mean(points, axis=0)
    points = points - center + 0.5
    # Compute safe radius (distance to closest wall)
    r_wall = np.min([np.min(points[:, 0]), np.min(1 - points[:, 0]),
                     np.min(points[:, 1]), np.min(1 - points[:, 1])])
    # Ensure circles don't overlap (limit radius based on neighbor distance)
    r_neighbor = d / 2  # adjacent circles just touch
    r = min(r_wall, r_neighbor)
    # Assign same radius to all
    radii = np.full(26, r)
    # Pack variables as [x0, y0, r0, x1, y1, r1, ...]
    vars = np.zeros(3 * 26)
    vars[::3] = points[:, 0]
    vars[1::3] = points[:, 1]
    vars[2::3] = radii
    return vars

def objective(vars):
    """Objective: minimize negative sum of radii."""
    radii = vars[2::3]
    return -np.sum(radii)

def wall_constraints(vars):
    """Inequality constraints for walls."""
    result = []
    n = 26

        y = vars[3*i+1]

        result.append(x + r - 1)
        # y - r >= 0
        result.append(r - y)
        # y + r <= 1
        result.append(y + r - 1)
    return np.array(result)

def nonoverlap_constraints(vars):
    """Inequality constraints for non-overlapping."""
    result = []
    n = 26
    for i in range(n):
        for j in range(i + 1, n):

            y_j = vars[3*j+1]
            r_j = vars[3*j+2]

            radii_sum_sq = (r_i + r_j)**2
            result.append(dist_sq - radii_sum_sq)
    return np.array(result)
