import numpy as np
from scipy.stats.qmc import Sobol

def initialize_seeds(n):
    sobol = Sobol(d=2, seed=42)
    points = sobol.random(n)
    # Scale to interior with margin
    delta = 0.02
    return delta + (1 - 2 * delta) * points

def compute_feasible_radii(centers):
    n = centers.shape[0]
    # Compute pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    distances = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(distances, np.inf)
    
    # Find nearest neighbor distance for each point
    nn_distances = np.min(distances, axis=1)
    
    # Compute wall distances
    wall_distances = np.array([
        centers[:, 0],        # x distance to left
        1 - centers[:, 0],    # x distance to right
        centers[:, 1],        # y distance to bottom
        1 - centers[:, 1]     # y distance to top
    ])
    wall_min = np.min(wall_distances, axis=0)
    
    # Compute feasible radii (half of nearest neighbor or wall distance)
    radii = np.minimum(0.5 * nn_distances, wall_min)
    return radii

def redistribute_radii(radii, slack, alpha):
    total_slack = np.sum(slack)
    if total_slack <= 1e-15:
        return radii.copy()
    
    # Increase wall-limited circles
    wall_limited = slack > 1e-15
    radii_new = radii.copy()
    if np.any(wall_limited):
        increases = alpha * slack
        radii_new[wall_limited] += increases[wall_limited]
    
    # Decrease neighbor-limited circles proportionally to their current radii
    neighbor_limited = ~wall_limited
    if np.any(neighbor_limited):
        total_reduction = alpha * total_slack
        
        # Normalize current radii of neighbor-limited
        current_radii = radii[neighbor_limited]
        total_current = np.sum(current_radii)
        
        if total_current > 1e-15:
            fractions = current_radii / total_current
            reductions = total_reduction * fractions
            radii_new[neighbor_limited] -= reductions
            # Clamp to non-negative (occurs only when very tiny reductions)
            radii_new[neighbor_limited] = np.maximum(radii_new[neighbor_limited], 0)
    
    return radii_new

def push_away(centers, radii, beta, eps):
    n = centers.shape[0]
    centers_new = centers.copy()
    
    # Compute pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    distances = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(n):
            if i == j:
                continue
            
            d = distances[i, j]
            if d < eps:
                continue
                
            r_i, r_j = radii[i], radii[j]
            safety = 1e-8
            
            if d < r_i + r_j + safety:
                # Push away
                overlap = r_i + r_j + safety - d
                direction = diff[i, j] / d
                centers_new[i] += beta * overlap * direction
    
    # Clamp to interior
    epsilon = 1e-10
    centers_new = np.clip(centers_new, epsilon, 1 - epsilon)
    
    return centers_new

def iterate(centers, radii, alpha, beta, max_iter):
    for _ in range(max_iter):
        # Compute new radii
        radii_new = compute_feasible_radii(centers)
        
        # Compute slack
        slack = 0.5 * np.min(
            np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :])**2, axis=2)),
            axis=1
        ) - radii_new
        
        slack[np.isinf(slack)] = 0
        slack[slack < 0] = 0
        
        # Redistribute radii
        radii_new = redistribute_radii(radii_new, slack, alpha)
        
        # Update centers (push-aways)
        centers = push_away(centers, radii_new, beta, 1e-8)
        
        # Compute new radii again
        radii = compute_feasible_radii(centers)
        
        # Check convergence
        delta_sum = np.sum(radii) - np.sum(radii_new)
        if delta_sum < 1e-5:
            break
    
    return centers, radii

def repair(centers, radii, epsilon=5e-10):
    radii = np.maximum(0, radii - epsilon)
    centers = np.clip(centers, epsilon, 1 - epsilon)
    return centers, radii

def run_packing():
    n = 26
    # Initial seeding
    centers = initialize_seeds(n)
    centers = np.clip(centers, 0.02, 0.98)
    
    # Compute initial radii
    radii = compute_feasible_radii(centers)
    
    # Iterate to improve sum of radii
    centers, radii = iterate(centers, radii, alpha=0.4, beta=0.3, max_iter=200)
    
    # Final repair
    centers, radii = repair(centers, radii)
    
    # Final validation (via the provided function)
    # Explicit validation not needed; guaranteed by process
    
    total_sum = np.sum(radii)
    return centers, radii, total_sum
