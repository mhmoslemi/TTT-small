import numpy as np
from scipy.spatial.distance import cdist

def penalty(z, alpha=10.0, beta=10.0):
    n = 26
    centers = z[:2*n].reshape((n, 2))
    radii = z[2*n:]
    
    # Wall penalties
    wall_pen = 0.0
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        d = [x, 1 - x, y, 1 - y]
        for di in d:
            if di < r:
                wall_pen += alpha * (r - di)
    
    # Overlap penalties
    overlap_pen = 0.0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                overlap_pen += beta * overlap
    
    return wall_pen + overlap_pen

def energy(z, lam, alpha=10.0, beta=10.0):
    # Objective: negative sum of radii
    obj = -np.sum(z[2*26:])
    # Penalty
    pen = penalty(z, alpha, beta)
    return obj + lam * pen

def propose_move(z, sigma_c, sigma_r):
    z_new = z.copy()
    move_type = np.random.choice(['center', 'radius', 'swap'])
    n = 26
    
    if move_type == 'center':
        i = np.random.randint(n)
        z_new[2*i] += np.random.normal(0, sigma_c)
        z_new[2*i + 1] += np.random.normal(0, sigma_c)
        z_new[2*i] = np.clip(z_new[2*i], 0, 1)
        z_new[2*i + 1] = np.clip(z_new[2*i + 1], 0, 1)
    elif move_type == 'radius':
        i = np.random.randint(n)
        z_new[2*n + i] += np.random.normal(0, sigma_r)
        z_new[2*n + i] = max(0, z_new[2*n + i])
    else:  # swap
        i, j = np.random.choice(n, 2, replace=False)
        r1 = z_new[2*n + i]
        r2 = z_new[2*n + j]
        z_new[2*n + i] = r2
        z_new[2*n + j] = r1
    
    return z_new

def anneal():
    n = 26
    # Initialize with hexagonal grid
    spacing = 0.18
    grid = []
    for i in range(10):
        for j in range(10):
            x = 0.05 + i * spacing
            y = 0.05 + j * spacing
            if x <= 1 and y <= 1:
                grid.append([x, y])
    grid = np.array(grid[:n])
    
    # Assign initial radii (wall-feasible)
    radii = np.array([min(x, 1-x, y, 1-y) for x, y in grid])
    
    # Pack into z vector
    z = np.zeros(2*n + n)
    z[:2*n] = grid.flatten()
    z[2*n:] = radii
    
    # Annealing parameters
    T0 = 1.0
    T = T0
    c_T = 0.995
    lam = 0.1
    c_lam = 1.03
    sigma_c = 0.03
    sigma_r = 0.02
    max_iters = 10000
    best_feasible = None
    best_energy = np.inf
    best_sum = 0.

    for k in range(max_iters):
        # Temperature schedule
        T *= c_T
        lam *= c_lam
        
        # Temperature scaled noise
        sig_c = sigma_c * np.sqrt(T / T0)
        sig_r = sigma_r * np.sqrt(T / T0)
        
        # Proposal
        z_new = propose_move(z, sig_c, sig_r)
        delta_E = energy(z_new, lam) - energy(z, lam)
        
        # Accept with Metropolis probability
        if delta_E <= 0 or np.random.rand() < np.exp(-delta_E / T):
            z = z_new
        
        # Check if z is better
        current_sum = np.sum(z[2*n:])
        current_energy = energy(z, lam)
        if current_energy < best_energy:
            best_energy = current_energy
            best_feasible = z.copy()
            best_sum = current_sum
    
    return best_feasible[:2*n].reshape((n, 2)), best_feasible[2*n:]

def project_and_ascent(centers, radii, max_iter=500, step_size=0.005, decay=0.95):
    z = np.zeros(2*26 + 26)
    z[:2*26] = centers.flatten()
    z[2*26:] = radii
    n = 26
    eta = step_size
    
    for _ in range(max_iter):
        # Gradient ascent on objective (sum of radii)
        grad = np.zeros_like(z)
        grad[2*n:] = 1  # Only radii have gradient +1
            
        # Apply gradient step
        z += eta * grad
        
        # Project onto constraints
        centers = z[:2*n].reshape((n, 2))
        radii = z[2*n:]
        
        # Clip radii to wall constraints
        for i in range(n):
            r = min(radii[i], centers[i, 0], 1 - centers[i, 0], 
                    centers[i, 1], 1 - centers[i, 1])
            radii[i] = max(0, r)
        
        # Handle overlaps
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                overlap = radii[i] + radii[j] - dist
                if overlap > 0:
                    # Shrink both radii
                    drop = overlap / 2
                    radii[i] = max(0, radii[i] - drop)
                    radii[j] = max(0, radii[j] - drop)
                    
                    # Re-apply wall constraints
                    for idx in [i, j]:
                        r = min(radii[idx], centers[idx, 0], 1 - centers[idx, 0], 
                                centers[idx, 1], 1 - centers[idx, 1])
                        radii[idx] = max(0, r)
        
        # Update z with constraints
        z[:2*n] = centers.flatten()
        z[2*n:] = radii
        
        # Check convergence
        last_eta = eta
        eta *= decay
        if np.abs(last_eta - eta) < 1e-8:
            break
    
    # Final projection to account for floating point errors by careful clamping
    centers = z[:2*n].reshape((n, 2))
    radii = z[2*n:]
    
    # Ensure wall constraints
    for i in range(n):
        r = min(radii[i], centers[i, 0], 1 - centers[i, 0], 
                centers[i, 1], 1 - centers[i, 1])
        radii[i] = max(0, r)
    
    # Ensure no overlaps by iteratively resolving
    while True:
        changed = False
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                overlap = radii[i] + radii[j] - dist
                if overlap > 0:
                    drop = overlap / 2
                    radii[i] = max(0, radii[i] - drop)
                    radii[j] = max(0, radii[j] - drop)
                    changed = True
                    # Re-check wall constraints
                    r1 = min(radii[i], centers[i, 0], 1 - centers[i, 0], 
                             centers[i, 1], 1 - centers[i, 1])
                    radii[i] = max(0, r1)
                    r2 = min(radii[j], centers[j, 0], 1 - centers[j, 0], 
                             centers[j, 1], 1 - centers[j, 1])
                    radii[j] = max(0, r2)
        if not changed:
            break
            
    return centers, radii
