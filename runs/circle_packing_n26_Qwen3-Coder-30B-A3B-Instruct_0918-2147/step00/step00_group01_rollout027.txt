import numpy as np
from scipy.spatial.distance import cdist

def penalty(z, alpha=10.0, beta=10.0):
    """Compute penalty term P(z) = alpha * wall penalties + beta * overlap penalties."""
    n = 26
    centers = z[:2*n].reshape(n, 2)
    radii = z[2*n:]
    
    # Wall penalties
    w_penalty = 0.0
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        wall_distances = [x, 1-x, y, 1-y]
        for d in wall_distances:
            if r > d:
                w_penalty += alpha * (r - d)
    
    # Overlap penalties
    o_penalty = 0.0
    for i in range(n):
        for j in range(i+1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            r_sum = radii[i] + radii[j]
            if dist < r_sum:
                o_penalty += beta * (r_sum - dist)
    
    return w_penalty + o_penalty

def energy(z, lam, alpha=10.0, beta=10.0):
    """Compute energy function E(z) = -sum(r) + lam * P(z)."""
    sum_radii = np.sum(z[2*26:])
    penalty_val = penalty(z, alpha, beta)
    return -sum_radii + lam * penalty_val

def propose_move(z, sigma_c, sigma_r):
    """Propose a new configuration by applying one of three moves."""
    # Pick move type
    move_type = np.random.choice([0, 1, 2])
    
    z_new = z.copy()
    n = 26
    
    if move_type == 0:  # Center jitter
        i = np.random.randint(n)
        x, y = z_new[2*i], z_new[2*i+1]
        # Add noise
        x += np.random.normal(0.0, sigma_c)
        y += np.random.normal(0.0, sigma_c)
        # Clip to bounds
        x = np.clip(x, 0.0, 1.0)
        y = np.clip(y, 0.0, 1.0)
        z_new[2*i] = x
        z_new[2*i+1] = y
    elif move_type == 1:  # Radius perturb
        i = np.random.randint(n)
        r = z_new[2*n + i]
        r += np.random.normal(0.0, sigma_r)
        r = max(0.0, r)
        z_new[2*n + i] = r
    else:  # Swap pair
        i, j = np.random.choice(n, 2, replace=False)
        r_i = z_new[2*n + i]
        r_j = z_new[2*n + j]
        z_new[2*n + i] = r_j
        z_new[2*n + j] = r_i
    
    return z_new

def anneal():
    """Run simulated annealing."""
    np.random.seed(0)  # For reproducibility
    n = 26
    T0 = 1.0
    c_T = 0.995
    T_min = 1e-5
    lambda0 = 0.1
    c_lambda = 1.03
    sigma_c = 0.03
    sigma_r = 0.02
    
    # Initialize on hexagonal grid
    spacing = 0.18
    grid_x = np.arange(0.0, 1.0, spacing)
    grid_y = np.arange(0.0, 1.0, spacing)
    centers_init = []
    for y in grid_y:
        for x in grid_x:
            if len(centers_init) < n:
                centers_init.append([x, y])
            if len(centers_init) >= n:
                break
        if len(centers_init) >= n:
            break
    # Use first n centers, and in case we don't have enough, pad by cloning
    while len(centers_init) < n:
        centers_init.append(centers_init[len(centers_init) % len(centers_init)])
    
    centers_init = np.array(centers_init[:n])
    
    # Set initial radii to maximum allowed
    radii_init = []
    for x, y in centers_init:
        r = min(x, 1-x, y, 1-y)
        radii_init.append(r)
    radii_init = np.array(radii_init)
    
    # Concatenate into z vector
    z = np.concatenate([centers_init.flatten(), radii_init])
    
    T = T0
    lam = lambda0
    best_feasible = None
    best_feasible_sum = -np.inf
    current_z = z.copy()
    current_energy = energy(current_z, lam)
    num_proposals = 0
    max_proposals = 200_000
    
    while T > T_min and num_proposals < max_proposals:
        z_prop = propose_move(current_z, sigma_c * np.sqrt(T/T0), sigma_r * np.sqrt(T/T0))
        energy_prop = energy(z_prop, lam)
        delta_E = energy_prop - current_energy
        
        accept = False
        if delta_E <= 0:
            accept = True
        else:
            if np.random.rand() < np.exp(-delta_E / T):
                accept = True
        
        if accept:
            current_z = z_prop
            current_energy = energy_prop
            
        # Check if feasible
        if penalty(current_z) < 1e-12:
            sum_radii = np.sum(current_z[2*n:])
            if sum_radii > best_feasible_sum:
                best_feasible_sum = sum_radii
                best_feasible = current_z.copy()
        
        num_proposals += 1
        T *= c_T
        if num_proposals % 50 == 0:
            lam *= c_lambda
    
    if best_feasible is None:
        raise RuntimeError("No feasible packing found.")
    
    centers_final = best_feasible[:2*n].reshape(n, 2)
    radii_final = best_feasible[2*n:]
    return centers_final, radii_final

def project_and_ascent(centers, radii, max_iter=500, eta=5e-3, decay_rate=0.95, eps=1e-7):
    """Perform projected gradient ascent with overlap resolution."""
    n = len(centers)
    z = np.concatenate([centers.flatten(), radii])
    eta_t = eta
    
    for _ in range(max_iter):
        # Gradient ascent: increase radii only
        grad = np.zeros_like(z)
        grad[2*n:] = 1.0  # dO/dr_i = 1
        
        # Take step
        z_new = z + eta_t * grad
        
        # Clamp radii to valid range [0, min(x,1-x, y,1-y)]
        centers_new = z_new[:2*n].reshape(n, 2)
        radii_new = z_new[2*n:]
        for i in range(n):
            x, y = centers_new[i]
            r = radii_new[i]
            max_r = min(x, 1-x, y, 1-y)
            radii_new[i] = np.clip(r, 0.0, max_r)
        
        # Resolve overlaps
        while True:
            changed = False
            for i in range(n):
                for j in range(i+1, n):
                    dist = np.sqrt(np.sum((centers_new[i] - centers_new[j]) ** 2))
                    r_sum = radii_new[i] + radii_new[j]
                    if dist < r_sum:
                        overlap = r_sum - dist
                        delta = overlap / 2.0
                        if radii_new[i] >= delta and radii_new[j] >= delta:
                            radii_new[i] -= delta
                            radii_new[j] -= delta
                            # Re-enforce wall constraints
                            x, y = centers_new[i]
                            max_r = min(x, 1-x, y, 1-y)
                            radii_new[i] = min(radii_new[i], max_r)
                            x, y = centers_new[j]
                            max_r = min(x, 1-x, y, 1-y)
                            radii_new[j] = min(radii_new[j], max_r)
                            changed = True
                        elif radii_new[i] >= delta:
                            radii_new[i] -= delta
                            x, y = centers_new[i]
                            max_r = min(x, 1-x, y, 1-y)
                            radii_new[i] = min(radii_new[i], max_r)
                            changed = True
                        elif radii_new[j] >= delta:
                            radii_new[j] -= delta
                            x, y = centers_new[j]
                            max_r = min(x, 1-x, y, 1-y)
                            radii_new[j] = min(radii_new[j], max_r)
                            changed = True
            if not changed:
                break
        
        # Clamp radii again in case shrinkage violated bounds
        for i in range(n):
            x, y = centers_new[i]
            max_r = min(x, 1-x, y, 1-y)
            radii_new[i] = max(0.0, min(radii_new[i], max_r))
        
        z = np.concatenate([centers_new.flatten(), radii_new])
        eta_t *= decay_rate
        if eta_t < 1e-7:
            break
    
    final_centers = z[:2*n].reshape(n, 2)
    final_radii = z[2*n:]
    return final_centers, final_radii

    centers, radii = anneal()
