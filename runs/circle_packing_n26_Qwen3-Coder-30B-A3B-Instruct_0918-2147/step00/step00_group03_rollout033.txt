import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
from itertools import combinations

def init_candidate_grid(R, Δ):
    candidates = []
    for r in R:
        spacing = r * np.sqrt(2)
        x_range = np.arange(r, 1 - r + 1e-10, spacing)
        y_range = np.arange(r, 1 - r + 1e-10, spacing)
        for x in x_range:
            for y in y_range:
                candidates.append((x, y, r))
    return candidates

def build_overlap_matrix(candidates):
    n = len(candidates)
    overlap = np.zeros((n, n), dtype=bool)
    for i in range(n):
        x1, y1, r1 = candidates[i]
        for j in range(i+1, n):
            x2, y2, r2 = candidates[j]
            dist = np.sqrt((x1-x2)**2 + (y1-y2)**2)
            if dist < r1 + r2:
                overlap[i, j] = True
                overlap[j, i] = True
    return overlap

def solve_discrete_bip(candidates, overlap_matrix, target_sum):
    n = len(candidates)
    # Initialize using a greedy approach
    selected = []
    used_indices = set()
    available = list(range(n))
    
    # Sort by radius descending
    sorted_indices = np.argsort([c[2] for c in candidates])[::-1]
    
    for i in sorted_indices:
        if i in used_indices:
            continue
        # Check if this candidate is compatible with selected set
        valid = True
        for j in selected:
            if overlap_matrix[i, j]:
                valid = False
                break
        if valid:
            selected.append(i)
            used_indices.add(i)
            if len(selected) == 26:
                break
    
    # If we didn't get 26, select the remaining ones greedily
    if len(selected) < 26:
        for i in sorted_indices:
            if len(selected) == 26:
                break
            if i in used_indices:
                continue
            is_valid = True
            for j in selected:
                if overlap_matrix[i, j]:
                    is_valid = False
                    break
            if is_valid:
                selected.append(i)
                used_indices.add(i)
                
    # Extract the corresponding centers and radii
    centers = np.array([[candidates[i][0], candidates[i][1]] for i in selected])
    radii = np.array([candidates[i][2] for i in selected])
    
    # Ensure total radius sum is at least target
    total = np.sum(radii)
    if total < target_sum:
        return None  # Fail to meet the target sum
    
    return centers, radii

def max_radii_fixed_centres(centers, radii_initial):
    n = len(centers)
    radii = radii_initial.copy()
    
    for i in range(n):
        # Compute minimum distance to all other circles (normalized)
        min_dist = 1e10
        for j in range(n):
            if i != j:
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                min_dist = min(min_dist, dist)
        
        # Compute free wall distances
        wall_dist = min(
            centers[i][0],
            1 - centers[i][0],
            centers[i][1],
            1 - centers[i][1]
        )
        
        # Radius is limited by wall and non-overlap
        new_radius = min(min_dist - 1e-12, wall_dist - 1e-12)
        if new_radius >= 0:
            radii[i] = new_radius
    
    # Trim to match initial radii or decrease
    for i in range(n):
        radii[i] = min(radii[i], radii_initial[i])
        
    return radii

def nlp_refine(centers, radii_initial):
    n = len(centers)
    # Define decision variables [x1,y1, ..., xn,yn, r1, ..., rn]
    x0 = np.concatenate([centers.ravel(), radii_initial])
    
    def objective(x):
        radii = x[-n:]
        return -np.sum(radii)
    
    # Compute pairwise distances between centers
    def wall_constraints(x):
        centers_flat = x[:2*n].reshape((n, 2))
        radii = x[2*n:]
        constraints = []
        for i in range(n):
            x_i, y_i = centers_flat[i]
            r_i = radii[i]
            # r_i - x_i <= 0
            constraints.append(r_i - x_i)
            # r_i + x_i - 1 <= 0
            constraints.append(r_i + x_i - 1)
            # r_i - y_i <= 0
            constraints.append(r_i - y_i)
            # r_i + y_i - 1 <= 0
            constraints.append(r_i + y_i - 1)
        return np.array(constraints)
    
    def nonoverlap_constraints(x):
        centers_flat = x[:2*n].reshape((n, 2))
        radii = x[2*n:]
        constraints = []
        for i in range(n):
            for j in range(i+1, n):
                x_i, y_i = centers_flat[i]
                x_j, y_j = centers_flat[j]
                r_i = radii[i]
                r_j = radii[j]
                dist_sq = (x_i - x_j)**2 + (y_i - y_j)**2
                dist = np.sqrt(dist_sq)
                # r_i + r_j - dist <= 0
                constraints.append(r_i + r_j - dist)
        return np.array(constraints)
    
    # Boundary constraints
    bounds = []
    # For centers (x_i, y_i) and radii r_i, each has range [0,1] for center and [0, 1] for radius
    for i in range(n):
        bounds.append((1e-6, 1-1e-6))  # x_i in (0,1)
        bounds.append((1e-6, 1-1e-6))  # y_i in (0,1)
    for i in range(n):
        bounds.append((1e-6, 1-1e-6))  # r_i in (0,1)

    # Equality bounds for warmstart
    for i in range(2 * n):
        bounds[i] = (x0[i], x0[i])
    for i in range(n):
        bounds[2*n + i] = (x0[2*n + i], x0[2*n + i])
    
    # Compute constraint bounds for wall and overlap relaxation
    wall_lb = _wall_bounds(centers, radii_initial)
    nonoverlap_lb = _nonoverlap_bounds(centers, radii_initial)
    cons = [
        {'type': 'ineq', 'fun': wall_constraints, 'lb': wall_lb},
        {'type': 'ineq', 'fun': nonoverlap_constraints, 'lb': nonoverlap_lb}
    ]

    res = minimize(
        objective,
        x0,
        method='trust-constr',
        bounds=bounds,
        constraints=cons,
        options={
            'maxiter': 3000,
            'xtol': 1e-9,
            'gtol': 1e-9
        }
    )

    centers_final = res.x[:2*n].reshape((n, 2))
    radii_final = res.x[2*n:]
    
    # Bring in bounds for radii
    for i in range(n):
        if radii_final[i] > 1 - 1e-12:
            radii_final[i] = 1 - 1e-12
        if radii_final[i] < 0:
            radii_final[i] = 0
        # Constraint violation
        if (
            centers_final[i][0] - radii_final[i] < -1e-12
            or centers_final[i][0] + radii_final[i] > 1 + 1e-12
            or centers_final[i][1] - radii_final[i] < -1e-12
            or centers_final[i][1] + radii_final[i] > 1 + 1e-12
        ):
            return None, None
    
    return centers_final, radii_final

def _wall_bounds(centers, radii):
    n = len(centers)
    bounds = []
    for i in range(n):
        x_i, y_i = centers[i]
        r_i = radii[i]
        bounds.append(r_i - x_i)  # <= 0
        bounds.append(r_i - 1 + x_i) # <= 0
        bounds.append(r_i - y_i)  # <= 0
        bounds.append(r_i - 1 + y_i)  # <= 0
    return np.array(bounds)

def _nonoverlap_bounds(centers, radii):
    n = len(centers)
    bounds = []
    for i in range(n):
        for j in range(i+1, n):
            x_i, y_i = centers[i]
            x_j, y_j = centers[j]
            r_i = radii[i]
            r_j = radii[j]
            dist = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)
            # r_i + r_j - dist <= 0
            bounds.append(r_i + r_j - dist)
    return np.array(bounds)

def run_packing():
    # Parameters
    r_values = np.arange(0.025, 0.12, 0.01)
    target_sum = 2.636
    
    # Phase 1: Discrete approach
    candidates = init_candidate_grid(r_values.tolist(), 0.01)
    overlap_matrix = build_overlap_matrix(candidates)
    res = solve_discrete_bip(candidates, overlap_matrix, target_sum)
    if res is None:
        raise ValueError("Failed to find a valid discrete solution")
    centers_initial, radii_initial = res
    
    # Remove any circles where we end up with a very small or zero radius
    valid_idx = radii_initial > 1e-10
    centers_initial = centers_initial[valid_idx]
    radii_initial = radii_initial[valid_idx]
    
    # Ensure we have exactly 26
    if len(centers_initial) < 26:
        indices = np.argsort(radii_initial)[::-1][:26 - len(centers_initial)]
        new_candidates = np.array(candidates)[indices]
        if len(new_candidates) > 0:
            new_centers = np.array([[c[0], c[1]] for c in new_candidates])
            new_radii = np.array([c[2] for c in new_candidates])
            centers_initial = np.vstack((centers_initial, new_centers))
            radii_initial = np.concatenate((radii_initial, new_radii))
    
    # If more than 26, limit to 26 by largest radii
    if len(centers_initial) > 26:
        idx = np.argsort(radii_initial)[::-1][:26]
        centers_initial = centers_initial[idx]
        radii_initial = radii_initial[idx]

    # Phase 2: Continuous refinement
    centers_final, radii_final = nlp_refine(centers_initial, radii_initial)
    if centers_final is None:
        raise RuntimeError("NLP refinement failed")

    # Final correction step
    radii_adjusted = max_radii_fixed_centres(centers_final, radii_final)
    centers_final = centers_final.copy()
    radii_final = radii_adjusted

    # Final check
    total_sum = np.sum(radii_final)
