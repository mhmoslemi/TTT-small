import numpy as np
import cvxpy as cp
from scipy.spatial.distance import pdist, squareform

def build_socp_problem(n):
    r = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(r)), [])
    dist_params = []
    return prob, r, dist_params

def solve_radii(centres, prob, r, dist_params):
    n = centres.shape[0]
    # Compute pairwise distances
    dist_matrix = squareform(pdist(centres, 'euclidean'))
    # Update the distance parameters
    for i, param in enumerate(dist_params):
        param.value = dist_matrix[i, i+1:].tolist()
    # Set up constraints
    constraints = []
    # Boundary constraints
    x = centres[:, 0]
    y = centres[:, 1]
    constraints += [r <= x]
    constraints += [r <= 1 - x]
    constraints += [r <= y]
    constraints += [r <= 1 - y]
    # Non-overlap constraints
    for i in range(n):
        for j in range(i+1, n):
            if i < len(dist_params) and j < len(dist_params[i]):
                constraints += [r[i] + r[j] <= dist_params[i][j-i-1]]
    prob.constraints = constraints
    prob.solve(solver=cp.ECOS, abstol=1e-12, reltol=1e-12)
    radii = r.value
    if prob.status == cp.OPTIMAL_INACCURATE:
        # If solution is inaccurate, solve with higher precision
        prob.solve(solver=cp.ECOS, abstol=1e-13, reltol=1e-13)
        radii = r.value
    return radii, prob

def energy(centres, prob, r, dist_params):
    radii, _ = solve_radii(centres, prob, r, dist_params)
    return -np.sum(radii)

def random_move(state, rng):
    # Reshape state into (n, 2) array
    n = len(state) // 2
    centres = state.reshape((n, 2))
    # Pick a move type
    move_types = ['single', 'cluster', 'edge_pull', 'swap']
    move_type = rng.choice(move_types, p=[0.7, 0.15, 0.1, 0.05])
    perturbed = centres.copy()
    
    if move_type == 'single':
        idx = rng.integers(n)
        delta = rng.normal(0, 0.02, 2)
        new_pos = centres[idx] + delta
        # Clamp to [0,1]
        new_pos = np.clip(new_pos, 0, 1)
        perturbed[idx] = new_pos
    elif move_type == 'cluster':
        indices = rng.choice(n, size=rng.integers(3, 6), replace=False)
        # Compute mean
        mean_pos = np.mean(centres[indices], axis=0)
        # Add correlated offset to all in the cluster
        delta = rng.normal(0, 0.02, 2)
        new_positions = centres[indices] + delta
        # Clamp to [0,1]
        new_positions = np.clip(new_positions, 0, 1)
        perturbed[indices] = new_positions
    elif move_type == 'edge_pull':
        idx = rng.integers(n)
        delta = 0.01
        # Move towards nearest wall with probability 0.1
        x, y = centres[idx]
        dx = -np.sign(x - 0.5) * delta
        dy = -np.sign(y - 0.5) * delta
        new_pos = centres[idx] + [dx, dy]
        # Clamp to [0,1]
        new_pos = np.clip(new_pos, 0, 1)
        perturbed[idx] = new_pos
    else:  # swap (do nothing)
        pass
    
    # Return as flat array
    return perturbed.flatten()

def anneal_chain(centres_seed, rng, prob, r, dist_params):
    state = centres_seed.flatten()
    current_energy = energy(state.reshape((-1,2)), prob, r, dist_params)
    current_state = state.copy()
    best_energy = current_energy
    best_state = current_state.copy()
    T = 0.05
    alpha = 0.995
    max_iter = 150000
    accepted_moves = 0
    while accepted_moves < max_iter and T > 1e-6:
        new_state = random_move(state, rng)
        new_energy = energy(new_state.reshape((-1,2)), prob, r, dist_params)
        delta = new_energy - current_energy
        if delta < 0 or rng.random() < np.exp(-delta / T):
            state = new_state
            current_energy = new_energy
            accepted_moves += 1
            if current_energy < best_energy:
                best_energy = current_energy
                best_state = state.copy()
        T *= alpha
    return best_state.reshape((-1,2))

def local_refine(centres, radii, prob, r, dist_params):
    # Use the dual information from the last solve
    # Find active constraints (~those with large duals)
    # In this simplified version, we do a few iterations where we move
    # centres to satisfy boundary and pairwise constraints
    for _ in range(10):
        new_centres = centres.copy()
        # Simple gradient-free refinement - enforce hard constraints
        for i in range(len(centres)):
            x, y = centres[i]
            r_i = radii[i]
            # Boundary constraints
            if x <= r_i + 1e-13:
                new_centres[i, 0] = r_i
            elif x >= 1 - r_i - 1e-13:
                new_centres[i, 0] = 1 - r_i
            if y <= r_i + 1e-13:
                new_centres[i, 1] = r_i
            elif y >= 1 - r_i - 1e-13:
                new_centres[i, 1] = 1 - r_i
        # Re-solve the SOCP to get updated radii
        _, prob = solve_radii(new_centres, prob, r, dist_params)
        new_radii, prob = solve_radii(new_centres, prob, r, dist_params)
        if np.abs(np.sum(new_radii) - np.sum(radii)) < 1e-5:
            break
        radii = new_radii
        centres = new_centres
    return centres, radii

def finalise(centres, prob, r, dist_params):
    # High precision solve
    radii, _ = solve_radii(centres, prob, r, dist_params)
    # Clamp radii that violate boundary by small margin
    x = centres[:, 0]
    y = centres[:, 1]
    for i in range(len(radii)):
        min_radius = min(x[i], 1-x[i], y[i], 1-y[i])
        if radii[i] > min_radius:
            radii[i] = min_radius

def run_packing():
    n = 26
    # Seeds for different configurations
    seeds = []
    # Regular grid seed
    grid_points = np.array([(i, j) for i in range(5) for j in range(5) if i*5+j < 26])
    grid_points = grid_points / 4.0 + 0.1
    seeds.append(grid_points)
    # Hexagonal seed (approximately)
    hex_points = []
    row = 0
    for i in range(26):
        if i % 5 == 0:
            row += 1
        x = (i % 5) * 0.2 + 0.1
        y = (row - 1) * 0.17 + 0.1 + (row % 2) * 0.085
        hex_points.append([x, y])
    seeds.append(np.array(hex_points))
    # Corner-biased seed
    corner_points = np.array([[0.05, 0.05], [0.95, 0.05], [0.05, 0.95], [0.95, 0.95]] + 
                            [[0.1, 0.1], [0.1, 0.9], [0.9, 0.1], [0.9, 0.9]] +
                            [[0.2, 0.2], [0.2, 0.8], [0.8, 0.2], [0.8, 0.8]] +
                            [[0.3, 0.05], [0.3, 0.95], [0.05, 0.3], [0.95, 0.3]] +
                            [[0.7, 0.05], [0.7, 0.95], [0.05, 0.7], [0.95, 0.7]] +
                            [[0.3, 0.5], [0.7, 0.5], [0.5, 0.3], [0.5, 0.7]])
    seeds.append(corner_points[:26])
    # Random uniform seed
    random_points = np.random.rand(26, 2)
    seeds.append(random_points)

    # All procedures should be deterministic
    # Create problem once
    prob, r, dist_params = build_socp_problem(n)

    best_sum = -np.inf
    best_centers = None
    best_radii = None

    for i, seed in enumerate(seeds):
        # Set random seed for reproducibility
        rng = np.random.default_rng(i)
        # Use the first run to initialize the SOCP problem structure
        if i == 0:
            # Initialize the dist_params
            dist_matrix = squareform(pdist(seed, 'euclidean'))
            dist_params = []
            for i in range(n):
                params = [cp.Parameter(n-i-1) for _ in range(n-i-1)]
                dist_params.append(params)
        # Perform annealing
        final_centers = anneal_chain(seed, rng, prob, r, dist_params)
        # Refine it
        initi_rad, _ = solve_radii(final_centers, prob, r, dist_params)
        refined_centers, refined_radii = local_refine(final_centers, initi_rad, prob, r, dist_params)
        # Final check
        final_centers, final_radii = finalise(refined_centers, prob, r, dist_params)
        sum_rad = np.sum(final_radii)
        if sum_rad > best_sum:
            best_sum = sum_rad
            best_centers = final_centers
            best_radii = final_radii

    return best_centers, best_radii, best_sum
