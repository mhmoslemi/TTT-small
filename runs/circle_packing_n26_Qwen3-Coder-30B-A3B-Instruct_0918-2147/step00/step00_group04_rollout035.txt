import numpy as np
import cvxpy as cp
from scipy.spatial.distance import pdist, squareform

def build_socp_problem(n):
    r = cp.Variable(n, nonneg=True)
    dist_params = []
    constraints = []

    # Boundary constraints
    x_coords = cp.Parameter(n)
    y_coords = cp.Parameter(n)
    constraints += [r <= x_coords]
    constraints += [r <= 1 - x_coords]
    constraints += [r <= y_coords]
    constraints += [r <= 1 - y_coords]

    # Non-overlap constraints
    for i in range(n):
        for j in range(i+1, n):
            dist_ij = cp.Parameter(nonneg=True)
            dist_params.append((i, j, dist_ij))
            constraints.append(r[i] + r[j] <= dist_ij)

    objective = cp.Maximize(cp.sum(r))
    prob = cp.Problem(objective, constraints)
    return prob, r, dist_params

def solve_radii(centres, prob, r, dist_params):
    n = centres.shape[0]
    x_coords = centres[:, 0]
    y_coords = centres[:, 1]
    
    # Update boundary parameters
    prob.parameters[0].value = x_coords
    prob.parameters[1].value = y_coords
    prob.parameters[2].value = 1 - x_coords
    prob.parameters[3].value = 1 - y_coords

    # Compute pairwise distances for non-overlap parameters
    distances = squareform(pdist(centres))
    for i, j, param in dist_params:
        param.value = distances[i, j]

    # Solve
    prob.solve(solver=cp.ECOS, abstol=1e-9, reltol=1e-9)
    
    if prob.status != cp.OPTIMAL:
        raise RuntimeError(f"SOCP solver failed: {prob.status}")

    radii = r.value
    return radii

def energy(centres, prob, r, dist_params):
    try:
        radii = solve_radii(centres, prob, r, dist_params)
        return -np.sum(radii)
    except:
        return np.inf

def random_move(state, rng):
    new_state = state.copy()
    n = len(state) // 2

    move_type = rng.choice(['single', 'cluster', 'edge', 'swap'])
    if move_type == 'single':
        idx = rng.integers(n)
        delta = rng.normal(0, 0.02, 2)
        new_state[2*idx:2*idx+2] += delta
        new_state[2*idx:2*idx+2] = np.clip(new_state[2*idx:2*idx+2], 0, 1)
    elif move_type == 'cluster':
        num = rng.integers(3, 6)
        indices = rng.choice(n, num, replace=False)
        delta = rng.normal(0, 0.02, 2)
        for idx in indices:
            new_state[2*idx:2*idx+2] += delta
            new_state[2*idx:2*idx+2] = np.clip(new_state[2*idx:2*idx+2], 0, 1)
    elif move_type == 'edge':
        if rng.random() < 0.1:
            idx = rng.integers(n)
            x, y = new_state[2*idx], new_state[2*idx+1]
            dx = -np.sign(x - 0.5) * 0.01
            dy = -np.sign(y - 0.5) * 0.01
            new_state[2*idx] += dx
            new_state[2*idx+1] += dy
            new_state[2*idx] = np.clip(new_state[2*idx], 0, 1)
            new_state[2*idx+1] = np.clip(new_state[2*idx+1], 0, 1)
    elif move_type == 'swap':
        # No swapping done; just a dummy fallback
        pass
    return new_state

def anneal_chain(seed, prob, r, dist_params):
    rng = np.random.default_rng(seed)
    n = 26
    state = rng.random(2 * n)
    state = np.clip(state, 0, 1)
    current_energy = energy(state.reshape(n, 2), prob, r, dist_params)
    best_energy = current_energy
    best_state = state.copy()
    T = 0.05
    alpha = 0.995
    iterations = 0
    max_iterations = 150000

    while T > 1e-6 and iterations < max_iterations:
        new_state = random_move(state, rng)
        new_energy = energy(new_state.reshape(n, 2), prob, r, dist_params)
        delta = new_energy - current_energy
        if delta < 0 or rng.random() < np.exp(-delta / T):
            state = new_state
            current_energy = new_energy
            if current_energy < best_energy:
                best_energy = current_energy
                best_state = state.copy()
        T *= alpha
        iterations += 1
    best_centers = best_state.reshape(n, 2)
    best_radii = solve_radii(best_centers, prob, r, dist_params)
    return best_centers, best_radii, -best_energy

def local_refine(centres, radii, prob, r, dist_params):
    n = len(centres)
    for _ in range(10):
        try:
            old_r = radii.copy()
            # Re-solve with same centres
            new_radii = solve_radii(centres, prob, r, dist_params)
            diff = np.sum(np.abs(new_radii - old_r))
            if diff < 1e-5:
                break
            radii = new_radii
        except Exception as e:
            break
    return centres, radii

def finalise(centres, radii, prob, r, dist_params):
    try:
        # Solve one more time with high precision
        final_radii = solve_radii(centres, prob, r, dist_params)
        # Clamp radii to boundaries to ensure constraints
        for i in range(len(centres)):
            x, y = centres[i]
            r_max = min(x, 1-x, y, 1-y)
            final_radii[i] = min(final_radii[i], r_max)
        return centres, final_radii
    except:
        return centres, radii

def run_packing():
    n = 26
    prob, r, dist_params = build_socp_problem(n)
    
    # Seeds for annealing chains
    seeds = [1, 2, 3, 4, 5, 6, 7, 8]
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    for seed in seeds:
        centers, radii, sum_radii = anneal_chain(seed, prob, r, dist_params)
        refined_centers, refined_radii = local_refine(centers, radii, prob, r, dist_params)
        final_centers, final_radii = finalise(refined_centers, refined_radii, prob, r, dist_params)
        final_sum = np.sum(final_radii)
        if final_sum > best_sum:
            best_sum = final_sum
            best_centers = final_centers
            best_radii = final_radii

    return best_centers, best_radii, best_sum
