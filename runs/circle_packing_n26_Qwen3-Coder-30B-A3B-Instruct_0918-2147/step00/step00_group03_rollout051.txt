import numpy as np
from scipy.optimize import minimize

    # Set seed for deterministic results
    np.random.seed(0)
    
    # Parameters
    num_circles = 26
    dt = 0.02
    wall_stiffness = 10.0
    pairwise_stiffness = 5.0
    damping = 1.0

    max_iter = 5000
    force_tolerance = 1e-6

    centers, radii = init_hexagonal_grid(num_circles)

    centers, radii = simulate_physics(centers, radii, dt, wall_stiffness, pairwise_stiffness, damping, base_growth_rate, max_iter, force_tolerance)

    radii = max_radii_fixed_centers(centers, radii, max_radius_update_iter)
    
    # Step 6: Sum of radii
    sum_radii = np.sum(radii)
    
    # Step 7: Validate
    valid, msg = validate_packing(centers, radii)
    if not valid:

def init_hexagonal_grid(n):
    """Generate initial grid of centers, perturbed from a hexagonal lattice."""
    spacing = 0.07
    sqrt3 = np.sqrt(3)
    cols = 8
    rows = 6
    points = []

    for k in range(rows):
        y = spacing / 2 + k * spacing * sqrt3 / 2
        if k % 2 == 0:
            start_x = spacing / 2
            step_x = spacing
        else:
            start_x = spacing
            step_x = spacing

        for m in range(cols):
            x = start_x + m * step_x
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append([x, y])
            if len(points) >= n:
                break
        if len(points) >= n:
            break

    points = points[:n]
    centers = np.array(points, dtype=np.float64)
    
    # Add jitter
    centers += np.random.uniform(-0.001, 0.001, size=centers.shape)
    
    # Clip to square
    centers = np.clip(centers, 0, 1)
    
    # Set initial small radii
    radii = np.full(n, 0.001, dtype=np.float64)
    
    return centers, radii

def compute_forces(centers, radii, wall_stiffness, pairwise_stiffness):
    """Compute forces acting on each circle."""
    n = centers.shape[0]
    forces = np.zeros_like(centers)
    
    # Wall forces
    dx_left = centers[:, 0] - radii
    dx_right = (1 - centers[:, 0]) - radii
    dy_bottom = centers[:, 1] - radii
    dy_top = (1 - centers[:, 1]) - radii
    
    for i in range(n):
        f = np.zeros(2)
        # Left wall
        if dx_left[i] < 0:
            f[0] += wall_stiffness * (-dx_left[i])
        # Right wall
        if dx_right[i] < 0:
            f[0] += wall_stiffness * (dx_right[i])
        # Bottom wall
        if dy_bottom[i] < 0:
            f[1] += wall_stiffness * (-dy_bottom[i])
        # Top wall
        if dy_top[i] < 0:
            f[1] += wall_stiffness * (dy_top[i])
        forces[i] += f

    # Pairwise forces
    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            dist = np.linalg.norm(diff)
            gap = dist - (radii[i] + radii[j])
            if gap < 0:
                force_magnitude = pairwise_stiffness * (-gap)
                if dist > 1e-12:
                    force = force_magnitude * diff / dist
                    forces[i] += force
                    forces[j] -= force

    return forces

def simulate_physics(centers, radii, dt, wall_stiffness, pairwise_stiffness, damping, base_growth_rate, max_iter, force_tolerance):
    """Run the physics simulation."""
    velocities = np.zeros_like(centers)
    n = centers.shape[0]
    
    for _ in range(max_iter):
        # Compute forces
        forces = compute_forces(centers, radii, wall_stiffness, pairwise_stiffness)
        
        # Update velocities
        velocities += dt * (forces - damping * velocities)
        
        # Update centers
        centers += dt * velocities
        
        # Clip centers to [0,1]^2
        centers = np.clip(centers, 0, 1)
        
        # Update radii based on whether circle is fully supported
        dot_r = np.ones(n)
        for i in range(n):
            # Check constraints
            d_left = centers[i, 0] - radii[i]
            d_right = (1 - centers[i, 0]) - radii[i]
            d_bottom = centers[i, 1] - radii[i]
            d_top = (1 - centers[i, 1]) - radii[i]
            valid_constraints = (d_left >= 0) and (d_right >= 0) and (d_bottom >= 0) and (d_top >= 0)
            
            # Check overlap with other circles
            valid_overlap = True
            for j in range(n):
                if i == j:
                    continue
                diff = centers[i] - centers[j]
                dist = np.linalg.norm(diff)
                gap = dist - (radii[i] + radii[j])
                if gap < 0:
                    valid_overlap = False
                    break
            
            if not valid_constraints or not valid_overlap:
                dot_r[i] = 0
            else:
                dot_r[i] = 1

        # Grow radii
        radii += dt * base_growth_rate * dot_r
        
        # Ensure radii are non-negative
        radii = np.maximum(radii, 0)
        
        # Compute max force magnitude
        max_force = np.max(np.linalg.norm(forces, axis=1))
        if max_force < force_tolerance:
            break
    return centers, radii

def max_radii_fixed_centers(centers, radii, max_iter):
    """Compute maximal radii for fixed center positions by iterative update."""
    n = centers.shape[0]
    for _ in range(max_iter):
        new_radii = np.zeros_like(radii)
        converged = True
        for i in range(n):
            # Wall distances
            d_left = centers[i, 0] - radii[i]
            d_right = (1 - centers[i, 0]) - radii[i]
            d_bottom = centers[i, 1] - radii[i]
            d_top = (1 - centers[i, 1]) - radii[i]
            
            # Min overlap distance to other circles
            min_gap = np.inf
            for j in range(n):
                if i == j:
                    continue
                diff = centers[i] - centers[j]
                dist = np.linalg.norm(diff)
                gap = dist - (radii[j] + radii[i])
                min_gap = min(min_gap, gap)
            
            # Compute new radius as minimum of valid bounding constraints
            safe_radius = np.min([d_left, d_right, d_bottom, d_top, min_gap])
            new_radii[i] = safe_radius
            if abs(radii[i] - new_radii[i]) > 1e-9:
                converged = False
        
        radii = new_radii
        if converged:
            break
    return radii

def objective_sum_radii(centers_flat, centers, radii):
    """Objective function for optimization: compute sum of maximal radii."""
    # Reshape
    centers = centers_flat.reshape(-1, 2)
    n = centers.shape[0]
    
    # Compute max radii based on new centers
    new_radii = np.zeros(n)
    for i in range(n):
        # Wall distances
        d_left = centers[i, 0] - radii[i]
        d_right = (1 - centers[i, 0]) - radii[i]
        d_bottom = centers[i, 1] - radii[i]
        d_top = (1 - centers[i, 1]) - radii[i]
        
        # Min overlap distance to other circles
        min_gap = np.inf
        for j in range(n):
            if i == j:
                continue
            diff = centers[i] - centers[j]
            dist = np.linalg.norm(diff)
            gap = dist - (radii[j] + radii[i])
            min_gap = min(min_gap, gap)
        
        safe_radius = np.min([d_left, d_right, d_bottom, d_top, min_gap])
        new_radii[i] = safe_radius
    
    return -np.sum(new_radii)

def refine_centers(centers, radii):
    """Refine centers with global optimization to maximize sum of radii."""
    # Flatten centers for scipy
    centers_flat = centers.flatten()
    
    # Compute barrier weights
    lambda_barrier = 1e-4
    
    def objective_with_barrier(x_flat):
        centers_local = x_flat.reshape(-1, 2)
        n = centers_local.shape[0]
        # Compute barrier
        barrier = 0.0
        for i in range(n):
            x, y = centers_local[i]
            barrier += np.maximum(0, -x)**2 + np.maximum(0, x - 1)**2 + np.maximum(0, -y)**2 + np.maximum(0, y - 1)**2
        
        # Objective
        centers_current = x_flat.reshape(-1, 2)
        n = centers_current.shape[0]
        new_radii = np.zeros(n)
        for i in range(n):
            d_left = centers_current[i, 0] - radii[i]
            d_right = (1 - centers_current[i, 0]) - radii[i]
            d_bottom = centers_current[i, 1] - radii[i]
            d_top = (1 - centers_current[i, 1]) - radii[i]
            
            min_gap = np.inf
            for j in range(n):
                if i == j:
                    continue
                diff = centers_current[i] - centers_current[j]
                dist = np.linalg.norm(diff)
                gap = dist - (radii[j] + radii[i])
                min_gap = min(min_gap, gap)
            
            safe_radius = np.min([d_left, d_right, d_bottom, d_top, min_gap])
            new_radii[i] = safe_radius
        
        return -np.sum(new_radii) + lambda_barrier * barrier
    
    # Optimize
    result = minimize(objective_with_barrier, centers_flat, method='L-BFGS-B',
                      bounds=[(0, 1) for _ in range(2 * centers.shape[0])],
                      options={'maxiter': 2000, 'gtol': 1e-9})
    
    # Unflatten and apply to new centers
    optimized_centers = result.x.reshape(-1, 2)
    return optimized_centers, radii
