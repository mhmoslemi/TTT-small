import numpy as np
from scipy.optimize import minimize
from ortools.linear_solver import pywraplp

def generate_candidates(g, radius_levels):
    candidates = []
    grid_points = np.arange(0, 1 + g/2, g)
    for x in grid_points:
        for y in grid_points:
            for r in radius_levels:
                if (x - r >= 0 and x + r <= 1 and y - r >= 0 and y + r <= 1):
                    candidates.append((x, y, r))
    return candidates

def build_milp(candidates):
    solver = pywraplp.Solver.CreateSolver('SCIP')
    if not solver:
        raise RuntimeError('SCIP could not be loaded')
    
    n = len(candidates)
    
    # Binary variables for selecting candidates
    z = [solver.BoolVar(f'z_{i}') for i in range(n)]
    
    # Constraint: At most one circle per grid point
    grid_map = {}
    for i, (x, y, r) in enumerate(candidates):
        grid_key = (round(x/g)*g, round(y/g)*g)
        if grid_key not in grid_map:
            grid_map[grid_key] = []
        grid_map[grid_key].append(i)
    
    for key, indices in grid_map.items():
        solver.Add(solver.Sum([z[i] for i in indices]) <= 1)
    
    # Constraint: Exactly 26 circles
    solver.Add(solver.Sum(z) == 26)
    
    # Constraints for pairwise non-overlap using L_infinity approximation
    # For each unordered pair (i,j) with i < j:
    for i in range(n):
        for j in range(i + 1, n):
            x_i, y_i, r_i = candidates[i]
            x_j, y_j, r_j = candidates[j]
            
            # Auxiliary variables
            dx = solver.NumVar(0, 1, f'dx_{i}_{j}')
            dy = solver.NumVar(0, 1, f'dy_{i}_{j}')
            
            # dx >= |x_i - x_j|
            solver.Add(dx >= x_i - x_j)
            solver.Add(dx >= x_j - x_i)
            
            # dy >= |y_i - y_j|
            solver.Add(dy >= y_i - y_j)
            solver.Add(dy >= y_j - y_i)
            
            # dx + dy >= r_i + r_j
            solver.Add(dx + dy >= r_i + r_j)
            
            # dx - dy >= r_i + r_j
            solver.Add(dx - dy >= r_i + r_j)
            
            # Enforce that dx, dy are only active when both selected
            M = 1.0
            solver.Add(dx <= M * (z[i] + z[j]))
            solver.Add(dy <= M * (z[i] + z[j]))
    
    # Objective: maximize sum of radii
    solver.Maximize(solver.Sum([r * z[i] for i, (_, _, r) in enumerate(candidates)]))
    
    return solver, z

def solve_milp(solver, z):
    status = solver.Solve()
    if status != pywraplp.Solver.OPTIMAL:
        raise RuntimeError('MILP solver did not find an optimal solution')
    
    selected_indices = [i for i in range(len(z)) if z[i].solution_value() > 0.5]
    return selected_indices

def extract_initial_solution(selected_indices, candidates):
    centers = np.array([[candidates[i][0], candidates[i][1]] for i in selected_indices])
    radii = np.array([candidates[i][2] for i in selected_indices])
    return centers, radii

def shrink_initial(radii, delta=1e-5):
    return np.maximum(0, radii - delta)

def pack_variables(centers, radii):
    z = np.zeros(3 * len(centers))
    for i in range(len(centers)):
        z[3*i] = centers[i][0]
        z[3*i+1] = centers[i][1]
        z[3*i+2] = radii[i]
    return z

def unpack_solution(z):
    n = len(z) // 3
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i][0] = z[3*i]
        centers[i][1] = z[3*i+1]
        radii[i] = z[3*i+2]
    return centers, radii

def define_objective(z):
    return -np.sum(z[2::3])  # Negative for minimization

def define_jacobian(z):
    n = len(z) // 3
    grad = np.zeros_like(z)
    grad[2::3] = -1.0
    return grad

def define_constraints(z):
    centers, radii = unpack_solution(z)
    n = len(centers)
    constraints = []
    
    # Boundary constraints
    for i in range(n):
        def boundary_func(i):
            def fun(z):
                x, y, r = z[3*i], z[3*i+1], z[3*i+2]
                return np.array([
                    -r + x, -r + (1-x), -r + y, -r + (1-y)
                ])
            return fun
    
    # Non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            def overlap_func(i, j):
                def fun(z):
                    x_i, y_i, r_i = z[3*i], z[3*i+1], z[3*i+2]
                    x_j, y_j, r_j = z[3*j], z[3*j+1], z[3*j+2]
                    dx = x_i - x_j
                    dy = y_i - y_j
                    dist = np.sqrt(dx**2 + dy**2 + 1e-12)
                    return (r_i + r_j) - dist
                return fun
            
            # Approximate JACOBIAN (we can do this better)
            def jac_func(i, j):
                def jac(z):
                    x_i, y_i, r_i = z[3*i], z[3*i+1], z[3*i+2]
                    x_j, y_j, r_j = z[3*j], z[3*j+1], z[3*j+2]
                    dx = x_i - x_j
                    dy = y_i - y_j
                    dist = np.sqrt(dx**2 + dy**2 + 1e-12)
                    grad = np.zeros_like(z)
                    # Derivative w.r.t. i
                    grad[3*i] = (dx / dist) - 1.0
                    grad[3*i+1] = (dy / dist) - 1.0
                    grad[3*i+2] = 1.0  # r_i
                    # Derivative w.r.t. j
                    grad[3*j] = -(dx / dist) - 1.0
                    grad[3*j+1] = -(dy / dist) - 1.0
                    grad[3*j+2] = 1.0  # r_j
                    return grad
                return jac
            
            constraints.append({
                'fun': overlap_func(i, j),
                'jac': jac_func(i, j),
                'type': 'ineq',
            })
            
    return constraints

def run_packing():
    g = 0.04
    radius_levels = [0.02 * i for i in range(1, 13)]

    candidates = generate_candidates(g, radius_levels)
    solver, z = build_milp(candidates)
    selected_indices = solve_milp(solver, z)
    centers0, radii0 = extract_initial_solution(selected_indices, candidates)
    radii0 = shrink_initial(radii0)

    z0 = pack_variables(centers0, radii0)

    bounds = []
    n = len(z0) // 3
    for i in range(n):
        bounds.append((1e-6, 1-1e-6))  # x bounds
        bounds.append((1e-6, 1-1e-6))  # y bounds
        bounds.append((1e-6, 0.5))     # r bounds

    constraints = define_constraints(z0)

    def objective(z):
        return define_objective(z)

    def jac(z):
        return define_jacobian(z)

    result = minimize(
        objective,
        z0,
        method='trust-constr',
        jac=jac,
        bounds=bounds,
        constraints=constraints,
        options={'disp': False}
    )

    if not result.success:
        raise RuntimeError("Continuous optimization failed")

    centers, radii = unpack_solution(result.x)
 
    # Repair any constraint violations
    for i in range(len(centers)):
        radii[i] = min(radii[i], centers[i][0], 1 - centers[i][0], 
                       centers[i][1], 1 - centers[i][1])
        radii[i] = max(0, radii[i])

    # Refine further if needed
    for _ in range(5):
        any_change = False
        for i in range(len(centers)):
            for j in range(i+1, len(centers)):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2 + 1e-12)
                rsum = radii[i] + radii[j]
                if dist < rsum:
                    overlap = rsum - dist
                    # Adjust both radii
                    radii[i] -= overlap/2
                    radii[j] -= overlap/2
                    any_change = True
        if not any_change:
            break

    sum_radii = np.sum(radii)
    
    # Validate final packing
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise RuntimeError(f"Final packing validation failed: {msg}")

    return centers, radii, sum_radii
