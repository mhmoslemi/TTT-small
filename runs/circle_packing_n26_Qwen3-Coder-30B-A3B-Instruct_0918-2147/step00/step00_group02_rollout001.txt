import numpy as np
from scipy.optimize import minimize

def initial_lattice(n=26):
    # Generate hexagonal lattice points scaled to fit in [0,1]x[0,1]
    # With spacing to accommodate n circles
    # Try different spacings to determine how many fit in the unit square
    spacing = 0.1  # Should be refined, start conservative
    while True:
        rows = int(np.ceil(1.0 / spacing))
        cols = int(np.ceil(1.0 / (spacing * np.sqrt(3) / 2)))
        total_required = rows * cols
        if total_required >= n:
            break
        spacing *= 0.99
    # Generate first 'n' points
    points = []
    for i in range(rows):
        for j in range(cols):
            x = j * spacing
            y = i * spacing * np.sqrt(3) / 2
            if i % 2 == 1:
                x += spacing/2
            if x <= 1 and y <= 1:
                points.append([x, y])
            if len(points) >= n:
                break
        if len(points) >= n:
            break
    # Trim to exactly n
    points = points[:n]
    centers = np.array(points)
    # Initial radii: minimum of distance to boundary and half nearest neighbor distance
    radii = np.zeros(n)
    for i in range(n):
        x, y = centers[i]
        r_limit = min(x, 1 - x, y, 1 - y)
        min_dist = float('inf')
        for j in range(n):
            if i == j:
                continue
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.hypot(dx, dy)
            if dist < min_dist:
                min_dist = dist
        r_neighbor = min_dist / 2.0
        radii[i] = min(r_limit, r_neighbor)
    # Ensure that no radii are > 0.5
    radii = np.clip(radii, 0, 0.5)
    return centers, radii

def pack(centers, radii):
    z = np.empty(3 * centers.shape[0])
    for i in range(centers.shape[0]):
        z[3*i] = centers[i,0]
        z[3*i + 1] = centers[i,1]
        z[3*i + 2] = radii[i]
    return z

def unpack(z):
    n = z.shape[0] // 3
    centers = np.empty((n,2))
    radii = np.empty(n)
    for i in range(n):
        centers[i, 0] = z[3*i]
        centers[i, 1] = z[3*i + 1]
        radii[i] = z[3*i + 2]
    return centers, radii

def objective(z):
    _, radii = unpack(z)
    return -np.sum(radii)

def constraint_boundary(z):
    centers, radii = unpack(z)
    c = []
    n = centers.shape[0]
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Left boundary: r - x <= 0
        c.append(r - x)
        # Right boundary: r + x - 1 <= 0
        c.append(r + x - 1)
        # Bottom boundary: r - y <= 0
        c.append(r - y)
        # Top boundary: r + y - 1 <= 0
        c.append(r + y - 1)
    return np.array(c)

def constraint_nonoverlap(z):
    centers, radii = unpack(z)
    c = []
    n = centers.shape[0]
    for i in range(n):
        for j in range(i+1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.hypot(dx, dy)
            r_sum = radii[i] + radii[j]
            # Non-overlap: dist >= r_sum
            # So we define constraint as: r_sum - dist <= 0
            c.append(r_sum - dist)
    return np.array(c)

def repair(centers, radii):
    # Make a copy to avoid modifying during iteration
    centers_new = centers.copy()
    radii_new = radii.copy()
    # Make sure radii are non-negative and respect boundary
    max_iter = 10
    for _ in range(max_iter):
        changed = False
        for i in range(centers_new.shape[0]):
            x, y = centers_new[i]
            r = radii_new[i]
            # Correct boundary violations first
            r_corr = min(r, x, 1-x, y, 1-y)
            if r_corr < r:
                radii_new[i] = r_corr
                changed = True
        # Then fix overlaps between pairs
        for i in range(centers_new.shape[0]):
            for j in range(i+1, centers_new.shape[0]):
                dx = centers_new[i, 0] - centers_new[j, 0]
                dy = centers_new[i, 1] - centers_new[j, 1]
                dist = np.hypot(dx, dy)
                r_sum = radii_new[i] + radii_new[j]
                if dist < r_sum:
                    overlap = r_sum - dist
                    # Reduce both radii by half the overlap
                    delta = overlap / 2
                    if radii_new[i] > delta:
                        radii_new[i] -= delta
                        changed = True
                    else:
                        radii_new[i] = 0.0
                    if radii_new[j] > delta:
                        radii_new[j] -= delta
                        changed = True
                    else:
                        radii_new[j] = 0.0
        if not changed:
            break
    # Snap to valid ranges for final safety
    radii_new = np.clip(radii_new, 0, 0.5)
    return centers_new, radii_new
