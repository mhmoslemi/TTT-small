import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def init_candidate_grid(R, Δ):
    candidates = []
    for r in R:
        spacing = r * np.sqrt(2)
        x_coords = np.arange(r, 1 - r + spacing/2, spacing)
        y_coords = np.arange(r, 1 - r + spacing/2, spacing)
        for x in x_coords:
            for y in y_coords:
                candidates.append((x, y, r))
    return candidates

def build_overlap_matrix(candidates):
    n = len(candidates)
    overlap = np.zeros((n, n), dtype=bool)
    for i in range(n):
        xi, yi, ri = candidates[i]
        for j in range(i+1, n):
            xj, yj, rj = candidates[j]
            dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
            if dist < ri + rj:
                overlap[i, j] = True
                overlap[j, i] = True
    return overlap

def max_radii_fixed_centres(centers, radii_initial):
    n = centers.shape[0]
    radii = radii_initial.copy()
    # For each circle, compute the maximum allowed radius without overlap
    for i in range(n):
        x_i, y_i = centers[i]
        r_i = radii[i]
        # Check boundaries
        r_max = min(
            x_i,
            1 - x_i,
            y_i,
            1 - y_i
        )
        # Check overlaps with all others
        for j in range(n):
            if i != j:
                x_j, y_j = centers[j]
                dist_ij = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)
                r_max = min(r_max, dist_ij - radii[j])
        # Note: if r_max is not positive, we must use a more reliable technique
        # but for now we clamp it conservatively
        if r_max < 0.001:
            r_max = 0.001
        radii[i] = r_max
    return radii

def nlp_refine(centers, radii_initial):
    n = len(centers)
    # We solve: minimize -sum(r) s.t. constraints
    def obj(v):
        # v = [x1, y1, ..., xn, yn, r1, ..., rn]
        centers_v = v[:2*n].reshape((n, 2))
        radii_v = v[2*n:]
        return -np.sum(radii_v)

    def jac(v):
        n = len(centers)
        return np.concatenate([np.zeros(2*n), -np.ones(n)])

    def cons(v):
        # Wall + pairwise overlap constraints
        centers_v = v[:2*n].reshape((n, 2))
        radii_v = v[2*n:]
        con = []
        # Wall constraints (each of the 4 per circle)
        for i in range(n):
            x, y = centers_v[i]
            r = radii_v[i]
            con.append(r - x)      # r >= x
            con.append(r + x - 1)  # r >= 1 - x
            con.append(r - y)      # r >= y
            con.append(r + y - 1)  # r >= 1 - y
        # Overlap constraints
        for i in range(n):
            for j in range(i+1, n):
                x_i, y_i = centers_v[i]
                x_j, y_j = centers_v[j]
                r_i = radii_v[i]
                r_j = radii_v[j]
                dist = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)
                con.append(r_i + r_j - dist)
        return np.array(con)

    def jac_cons(v):
        n = len(centers)
        centers_v = v[:2*n].reshape((n, 2))
        radii_v = v[2*n:]
        J = []
        # Jacobian of wall constraints
        for i in range(n):
            x, y = centers_v[i]
            r = radii_v[i]
            row = np.zeros(2*n + n)
            # d(r - x) / dx
            row[2*i] = -1.0
            # d(r - x) / dr
            row[2*n + i] = 1.0
            J.append(row)
            row = np.zeros(2*n + n)
            # d(r + x - 1) / dx
            row[2*i] = 1.0
            # d(r + x - 1) / dr
            row[2*n + i] = 1.0
            J.append(row)
            row = np.zeros(2*n + n)
            # d(r - y) / dy
            row[2*i + 1] = -1.0
            # d(r - y) / dr
            row[2*n + i] = 1.0
            J.append(row)
            row = np.zeros(2*n + n)
            # d(r + y - 1) / dy
            row[2*i + 1] = 1.0
            # d(r + y - 1) / dr
            row[2*n + i] = 1.0
            J.append(row)
        # Overlap constraints
        for i in range(n):
            for j in range(i+1, n):
                x_i, y_i = centers_v[i]
                x_j, y_j = centers_v[j]
                r_i = radii_v[i]
                r_j = radii_v[j]
                dist = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)
                if dist > 0:
                    # dx_i/dx_i = 1, dy_i/dy_i = 1, dx_i/dr_i = 1, ...
                    row = np.zeros(2*n + n)
                    # deriv of (r_i + r_j - dist) w.r.t. x_i
                    row[2*i] = - (x_i - x_j) / dist
                    # deriv of (r_i + r_j - dist) w.r.t. x_j
                    row[2*j] = + (x_i - x_j) / dist
                    # deriv of (r_i + r_j - dist) w.r.t. y_i
                    row[2*i + 1] = - (y_i - y_j) / dist
                    # deriv of (r_i + r_j - dist) w.r.t. y_j
                    row[2*j + 1] = + (y_i - y_j) / dist
                    # deriv of (r_i + r_j - dist) w.r.t. r_i
                    row[2*n + i] = 1.0
                    # deriv of (r_i + r_j - dist) w.r.t. r_j
                    row[2*n + j] = 1.0
                    J.append(row)
        return np.array(J)

    # Initial guess
    v0 = np.concatenate([centers.flatten(), radii_initial])
    bounds = []
    for i in range(2*n):
        bounds.append((0.0001, 0.9999))
    for i in range(n):
        bounds.append((0.0001, 0.9999))

    # Solve
    options = {'gtol': 1e-9, 'maxiter': 300}
    result = minimize(obj, v0, method='trust-constr', jac=jac, hess=None,
                      bounds=bounds, constraints={'type': 'ineq', 'fun': cons, 'jac': jac_cons},
                      options=options)
    if not result.success:
        # Retry with the initial value
        print(f"Note: NLP solver failed with message: {result.message}")
    centers_refined = result.x[:2*n].reshape((n, 2))
    radii_refined = result.x[2*n:]
    return centers_refined, radii_refined

def run_packing():
    # Define discretization grid
    Δ = 0.01
    R = [0.025 + i*Δ for i in range(10, 116)]  # up to 1.15
    candidates = init_candidate_grid(R, Δ)
    overlap_matrix = build_overlap_matrix(candidates)

    centers_initial, radii_initial = solve_discrete_bip(candidates, overlap_matrix, 2.636)
    radii_initial = max_radii_fixed_centres(centers_initial, radii_initial)
    centers_refined, radii_refined = nlp_refine(centers_initial, radii_initial)
    # Final adjustment
    radii_final = max_radii_fixed_centres(centers_refined, radii_refined)

    # Final sum
    sum_radii = np.sum(radii_final)
    # Validate
    assert validate_packing(centers_refined, radii_final)[0], "Validation failed"
    
    return centers_refined, radii_final, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
