import numpy as np
from scipy.stats.qmc import Sobol

def initialize_seeds(n):
    sobol = Sobol(d=2, scramble=False)
    points = sobol.random(n)
    # Clip to interior with margin δ = 0.02 to ensure positive initial radii
    margin = 0.02
    clipped_points = margin + (1 - 2*margin) * points
    return clipped_points

def compute_feasible_radii(centers):
    n = centers.shape[0]
    # Compute pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_matrix = np.sqrt(np.sum(diff**2, axis=2))
    
    # For each circle, find the minimum distance to any other center (excluding itself)
    np.fill_diagonal(dist_matrix, np.inf)
    nn_distances = np.min(dist_matrix, axis=1)
    
    # For each circle, find the minimum distance to the edges of the square
    wall_distances = np.array([
        centers[:, 0],  # distance to left edge
        1 - centers[:, 0],  # distance to right edge
        centers[:, 1],  # distance to bottom edge
        1 - centers[:, 1]  # distance to top edge
    ]).min(axis=0)
    
    # Feasible radius is half the nearest neighbor distance or wall distance, whichever is smaller
    radii = np.minimum(0.5 * nn_distances, wall_distances)
    return radii

def redistribute_radii(radii, slack, alpha):
    # Compute total slack
    B = np.sum(slack)
    
    # Increase radii for wall-limited circles (slack > 0)
    new_radii = radii.copy()
    increase = alpha * slack
    new_radii += increase
    
    # Reduce radii for neighbor-limited circles proportionally to current radii
    neighbor_limited = slack <= 1e-12
    if np.any(neighbor_limited):
        total_reduction = alpha * B
        total_current = np.sum(radii[neighbor_limited])
        if total_current > 1e-12:
            reduction_factors = radii[neighbor_limited] / total_current
            reductions = total_reduction * reduction_factors
            new_radii[neighbor_limited] -= reductions
            # Ensure no negative radii
            new_radii[neighbor_limited] = np.maximum(0, new_radii[neighbor_limited])

    return new_radii

def push_away(centers, radii, beta, eps):
    n = centers.shape[0]
    new_centers = centers.copy()
    
    # For each pair, check for overlap
    for i in range(n):
        for j in range(i + 1, n):
            diff = new_centers[i] - new_centers[j]
            dist = np.sqrt(np.sum(diff**2))
            r_sum = radii[i] + radii[j] + eps
            
            if dist < r_sum:
                # Push circles apart along the vector from j to i
                delta = r_sum - dist
                direction = diff / (dist + eps)
                push_distance = beta * delta
                new_centers[i] += push_distance * direction
                new_centers[j] -= push_distance * direction
    
    # Clip to interior with small margin to prevent boundary violations
    margin = eps
    new_centers = np.clip(new_centers, margin, 1 - margin)
    
    return new_centers

def iterate(centers, radii, alpha=0.4, beta=0.3, max_iter=200):
    for _ in range(max_iter):
        # Compute current slack
        slack = 0.5 * np.min(
            np.delete(np.linalg.norm(centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=2), 
                      np.arange(len(centers)), axis=1),
            axis=1) - radii
        
        # Redistribute radii
        new_radii = redistribute_radii(radii, slack, alpha)
        
        # Compute new centers by pushing away
        new_centers = push_away(centers, new_radii, beta, 1e-8)
        
        # Recompute radii
        radii = compute_feasible_radii(new_centers)
        
        # Check for convergence
        if np.sum(np.abs(new_radii - radii)) < 1e-5:
            break
        
        centers = new_centers
        radii = new_radii
        
    return centers, radii

def repair(centers, radii, eps_final=5e-10):
    # Shrink radii slightly to satisfy strict validation
    radii = np.maximum(0, radii - eps_final)
    
    # Clip centers
    centers = np.clip(centers, eps_final, 1 - eps_final)
    
    # Optional final overlap correction
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_matrix = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dist_matrix, np.inf)
    
    for i in range(centers.shape[0]):
        for j in range(i + 1, centers.shape[0]):
            if dist_matrix[i, j] < radii[i] + radii[j]:
                overlap = radii[i] + radii[j] - dist_matrix[i, j]
                correction = overlap / 2 + eps_final
                radii[i] = max(0, radii[i] - correction)
                radii[j] = max(0, radii[j] - correction)
    
    return centers, radii

def run_packing():
    # Seed initialization
    centers = initialize_seeds(26)
    
    # Compute initial feasible radii
    radii = compute_feasible_radii(centers)
    
    # Iterative improvement
    centers, radii = iterate(centers, radii, alpha=0.4, beta=0.3, max_iter=150)
    
    # Final repair
    centers, radii = repair(centers, radii)
    
    # Compute sum of radii
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
