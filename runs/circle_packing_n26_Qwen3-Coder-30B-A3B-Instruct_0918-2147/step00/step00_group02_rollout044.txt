import numpy as np
from scipy.optimize import minimize
from math import sqrt

def initial_lattice(n=26):
    # Generate hexagonal lattice points
    spacing = 0.2  # Start with spacing that'll likely give enough points
    sqrt3o2 = sqrt(3) / 2.0
    points = []
    
    # Use a grid that's larger than 1x1 to capture enough points
    for i in range(20):
        for j in range(20):
            x = i * spacing
            y = j * spacing * sqrt3o2
            if j % 2 == 1:
                x += spacing / 2.0
            if x <= 1 and y <= 1:
                points.append((x, y))
    
    # Keep only the first n points
    centers = np.array(points[:n])
    
    # Compute maximum allowable radii based on boundaries and neighbors
    radii = []
    for i, (x, y) in enumerate(centers):
        # Distance to boundaries        
        r_bound = min(x, 1-x, y, 1-y)
        
        # Distance to other centers
        r_neighbor = 1.0
        for j, (x2, y2) in enumerate(centers):
            if i != j:
                dist = sqrt((x - x2)**2 + (y - y2)**2)
                r_candidate = dist / 2.0
                r_neighbor = min(r_neighbor, r_candidate)
        
        r = min(r_bound, r_neighbor)
        r = max(0, r)
        radii.append(r)        
    return centers, np.array(radii)

def pack(centers, radii):
    z = []
    for i in range(len(centers)):
        z.extend([centers[i][0], centers[i][1], radii[i]])
    return np.array(z)

def unpack(z):
    n = len(z) // 3
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i] = [z[3*i], z[3*i+1]]
        radii[i] = z[3*i+2]
    return centers, radii

def objective(z):
    _, radii = unpack(z)
    return -np.sum(radii)

def constraint_boundary(z):
    centers, radii = unpack(z)
    c = []
    for i in range(len(centers)):
        x, y = centers[i]
        r = radii[i]
        # Left boundary: x - r >= 0
        c.append(x - r)
        # Right boundary: 1 - x - r >= 0
        c.append(1 - x - r)
        # Bottom boundary: y - r >= 0
        c.append(y - r)
        # Top boundary: 1 - y - r >= 0
        c.append(1 - y - r)
    return np.array(c)

def constraint_nonoverlap(z):
    centers, radii = unpack(z)
    c = []
    n = len(centers)
    for i in range(n):
        for j in range(i + 1, n):
            x1, y1 = centers[i]
            x2, y2 = centers[j]
            r1 = radii[i]
            r2 = radii[j]
            # Non-overlap: dist >= r1 + r2
            dist = sqrt((x1 - x2)**2 + (y1 - y2)**2)
            c.append(dist - r1 - r2)
    return np.array(c)

def repair(centers, radii):
    n = len(centers)
    # Ensure all boundaries respected
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        maxr = min(x, 1-x, y, 1-y)
        radii[i] = min(r, maxr)
        
    # Resolve overlaps
    max_iters = 10
    for _ in range(max_iters):
        overlaps = []
        for i in range(n):
            for j in range(i + 1, n):
                x1, y1 = centers[i]
                x2, y2 = centers[j]
                r1 = radii[i]
                r2 = radii[j]
                dist = sqrt((x1 - x2)**2 + (y1 - y2)**2)
                overlap = (r1 + r2) - dist
                if overlap > 1e-12:
                    overlaps.append((i, j, overlap))
        
        if not overlaps:
            break
            
        for i, j, overlap in overlaps:
            # Reduce both radii by half the overlap amount
            delta = overlap / 2.0
            radii[i] = max(0, radii[i] - delta)
            radii[j] = max(0, radii[j] - delta) 
            
    return centers, radii

def run_packing():
    # Initial layout
    centers, radii = initial_lattice(26)
    z0 = pack(centers, radii)
    
    # Prepare constraints
    constraints = []
    constraints.append({'type': 'ineq', 'fun': constraint_boundary})
    constraints.append({'type': 'ineq', 'fun': constraint_nonoverlap})
    
    # Bounds
    bounds = []
    for i in range(len(z0)):
        if i % 3 == 2:  # Radius
            bounds.append((0, 0.5))  # Max radius 0.5
        else:  # x or y
            bounds.append((0, 1))
    
    # Optimization
    result = minimize(objective, z0, method='SLSQP', bounds=bounds, 
                      constraints=constraints, 
                      options={'maxiter': 500, 'ftol': 1e-9}, 
                      disp=False)
    
    # Unpack results
    centers_opt, radii_opt = unpack(result.x)
     
    # Repair
    centers_opt, radii_opt = repair(centers_opt, radii_opt)
