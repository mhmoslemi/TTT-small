import numpy as np
from scipy.optimize import linprog

def initial_hex_centers(k, spacing):
    sqrt3 = np.sqrt(3)
    centers = []
    i = 0
    j = 0
    while len(centers) < k:
        x = i * spacing + (j % 2) * spacing / 2
        y = j * spacing * sqrt3 / 2
        if 0 <= x <= 1 and 0 <= y <= 1:
            centers.append([x, y])
        if x + spacing > 1:
            j += 1
            i = -1
        i += 1
    return np.array(centers[:k])

def initial_radii(k, r0):
    return np.full(k, r0)

def build_lp_constants(centers, radii):
    n = centers.shape[0]
    d_left = centers[:, 0]
    d_right = 1.0 - centers[:, 0]
    d_bottom = centers[:, 1]
    d_top = 1.0 - centers[:, 1]
    wall_distances = np.stack([d_left, d_right, d_bottom, d_top], axis=1)
    
    D = np.zeros((n, n))
    for i in range(n):
        for j in range(i+1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            D[i, j] = D[j, i] = dist
    
    return wall_distances, D

def solve_radius_lp(consts, radii, graph, alpha=0.5):
    wall_distances, D = consts
    n = len(radii)
    c = np.ones(n)
    
    A_ub = []
    b_ub = []
    
    # Wall constraints
    for i in range(n):
        wall_caps = wall_distances[i]
        max_radius = np.min(wall_caps) - radii[i]
        A_ub.append([-1.0] * n)
        b_ub.append(-max_radius)
    
    # Pairwise constraints
    for i in range(n):
        for j in range(i+1, n):
            cap = D[i, j] - (radii[i] + radii[j])
            if cap >= 0:
                row = [0] * n
                row[i] = -1.0
                row[j] = -1.0
                A_ub.append(row)
                b_ub.append(-cap)
                
    # Graph-based transfer limits (optional)
    if len(graph.edges) > 0:
        for i in range(n):
            neighbors = [j for j in range(n) if (i, j) in graph.edges or (j, i) in graph.edges]
            if neighbors:
                total_capacity = sum(D[i, j] - (radii[i] + radii[j]) for j in neighbors if D[i, j] > radii[i] + radii[j])
                if total_capacity > 0:
                    row = [0] * n
                    row[i] = -1.0
                    A_ub.append(row)
                    b_ub.append(-alpha * total_capacity)
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'presolve': True})
    
    if not result.success:
        return np.zeros(n)
    return result.x

def apply_increment(radii, Δr):
    return radii + Δr

def push_feasibility(centers, radii):
    n = len(radii)
    original_centers = centers.copy()
    pushed_centers = centers.copy()
    pushed_radii = radii.copy()
    
    # Wall corrections
    for i in range(n):
        x, y = pushed_centers[i]
        r = pushed_radii[i]
        if x - r < 0:
            pushed_radii[i] = x
            pushed_centers[i] = [x, y]
        if x + r > 1:
            pushed_radii[i] = 1 - x
            pushed_centers[i] = [x, y]
        if y - r < 0:
            pushed_radii[i] = y
            pushed_centers[i] = [x, y]
        if y + r > 1:
            pushed_radii[i] = 1 - y
            pushed_centers[i] = [x, y]
    
    # Overlap corrections
    overlaps = []
    for i in range(n):
        for j in range(i+1, n):
            dist = np.linalg.norm(pushed_centers[i] - pushed_centers[j])
            r_sum = pushed_radii[i] + pushed_radii[j]
            if dist < r_sum:
                overlaps.append((i, j, r_sum - dist))
    
    for i, j, δ in overlaps:
        # Move away from each other along line of centers
        if i == j: continue
        vec = pushed_centers[i] - pushed_centers[j]
        norm_vec = np.linalg.norm(vec)
        if norm_vec > 0:
            direction = vec / norm_vec
            shift_by = δ / 2
            pushed_centers[i] -= shift_by * direction
            pushed_centers[j] += shift_by * direction
            # Clamp to bounds if necessary
            if pushed_centers[i][0] - pushed_radii[i] < 0:
                pushed_radii[i] = pushed_centers[i][0]
            if pushed_centers[i][0] + pushed_radii[i] > 1:
                pushed_radii[i] = 1 - pushed_centers[i][0]
            if pushed_centers[i][1] - pushed_radii[i] < 0:
                pushed_radii[i] = pushed_centers[i][1]
            if pushed_centers[i][1] + pushed_radii[i] > 1:
                pushed_radii[i] = 1 - pushed_centers[i][1]
                
            if pushed_centers[j][0] - pushed_radii[j] < 0:
                pushed_radii[j] = pushed_centers[j][0]
            if pushed_centers[j][0] + pushed_radii[j] > 1:
                pushed_radii[j] = 1 - pushed_centers[j][0]
            if pushed_centers[j][1] - pushed_radii[j] < 0:
                pushed_radii[j] = pushed_centers[j][1]
            if pushed_centers[j][1] + pushed_radii[j] > 1:
                pushed_radii[j] = 1 - pushed_centers[j][1]
    
    # Ensure absolute validity after pushes
    valid_centers = pushed_centers.copy()
    valid_radii = pushed_radii.copy()
    for i in range(n):
        x, y = valid_centers[i]
        r = valid_radii[i]
        # Clip to boundaries
        r = min(r, x, 1-x, y, 1-y)
        valid_radii[i] = r
    return valid_centers, valid_radii

def run_iterations(centers, radii, max_iter=100, ε=1e-5, α=0.5):
    # Simple graph based on adjacency
    class Graph:
        def __init__(self):
            self.edges = set()
        
        def add_edge(self, i, j):
            if i != j:
                self.edges.add((i, j)) 
                self.edges.add((j, i))

    graph = Graph()
    old_sum = np.sum(radii)
    
    for iteration in range(max_iter):
        # Build constraints
        consts = build_lp_constants(centers, radii)
        
        # Solve LP
        Δr = solve_radius_lp(consts, radii, graph, α)
        
        # Apply increment
        new_radii = apply_increment(radii, Δr)
        
        # Get the new sum
        new_sum = np.sum(new_radii)
        if new_sum - old_sum < ε:
            break
        old_sum = new_sum
        
        # Update radii (will be used by next iteration)
        radii = new_radii
        
        # Apply geometric push to restore feasibility
        centers, radii = push_feasibility(centers, radii)
    
    return centers, radii

def final_refinement(centers, radii):
    # Run one more high precision LP
    consts = build_lp_constants(centers, radii)
    Δr = solve_radius_lp(consts, radii, None, 0.5)
    final_radii = apply_increment(radii, Δr)
    
    # Final clipping pass
    for i in range(len(final_radii)):
        final_radii[i] = min(final_radii[i], centers[i][0], 1-centers[i][0], 
                             centers[i][1], 1-centers[i][1])
    
    # Now small overlaps are fixed
    n = len(final_radii)
    for i in range(n):
        for j in range(i+1, n):
            dist_ij = np.linalg.norm(centers[i] - centers[j])
            if dist_ij < final_radii[i] + final_radii[j] - 1e-13:
                delta = final_radii[i] + final_radii[j] - dist_ij
                if final_radii[i] > final_radii[j]:
                    final_radii[i] -= delta / 2.0 + 1e-13
                else:
                    final_radii[j] -= delta / 2.0 + 1e-13
    return centers, final_radii

def run_packing():
    # Initial hexagonal lattice with spacing for spacing=0.12
    initial_centers = initial_hex_centers(26, 0.12)
    initial_radii_array = initial_radii(26, 0.005)
    
    # Run iterative solver
    final_centers, final_radii = run_iterations(initial_centers, initial_radii_array)
    
    # Final refinement
    final_centers, final_radii = final_refinement(final_centers, final_radii)
    
    # Return centers, radii, and sum
    total_sum = np.sum(final_radii)
    return final_centers, final_radii, total_sum
