import numpy as np
from scipy.optimize import minimize

def make_hex_lattice(n, margin=0.05):
    """Generate a hexagonal lattice of n points within [margin, 1-margin]^2"""
    # Estimate spacing to fit n points
    sqrt_n = np.ceil(np.sqrt(n))
    dx = (1 - 2*margin) / (sqrt_n + 0.5)
    dy = dx * np.sqrt(3) / 2
    points = []
    for j in range(int(np.ceil(n / (sqrt_n + 1)) + 1)):
        for i in range(int(sqrt_n + 1)):
            x = margin + i * dx + (j % 2) * dx / 2
            y = margin + j * dy
            if x >= 1 - margin or y >= 1 - margin:
                continue
            if len(points) >= n:
                break
            points.append([x, y])
        if len(points) >= n:
            break
    return np.array(points[:n])

def objective(z):
    """Minimize negative sum of radii"""
    radii = z[2::3]
    return -np.sum(radii)

def boundary_constraints(z):
    """Compute boundary constraint violations (>= 0 when feasible)"""
    n = len(z) // 3
    result = []
    for i in range(n):
        x, y, r = z[3*i], z[3*i+1], z[3*i+2]
        result.extend([
            x - r,                  # x - r >= 0
            1 - x - r,              # 1 - x - r >= 0
            y - r,                  # y - r >= 0
            1 - y - r               # 1 - y - r >= 0
        ])
    return np.array(result)

def pairwise_constraints(z):
    """Compute separation constraint violations (>= 0 when feasible)"""
    n = len(z) // 3
    result = []
    for i in range(n):
        x_i, y_i, r_i = z[3*i], z[3*i+1], z[3*i+2]
        for j in range(i+1, n):
            x_j, y_j, r_j = z[3*j], z[3*j+1], z[3*j+2]
            dist_sq = (x_i - x_j)**2 + (y_i - y_j)**2
            sep_sq = (r_i + r_j)**2
            result.append(sep_sq - dist_sq)
    return np.array(result)

def solve_nlp(initial_z, maxiter=1000):
    """Solve the NLP using SLSQP"""
    cons = [
        {'type': 'ineq', 'fun': boundary_constraints},
        {'type': 'ineq', 'fun': pairwise_constraints}
    ]

    bounds = []
    n = len(initial_z) // 3
    for i in range(n):
        bounds.extend([(0, 1), (0, 1), (0, 0.5)])  # x, y in [0,1], r in [0, 0.5]
    
    res = minimize(objective, initial_z, method='SLSQP',
                   bounds=bounds, constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-9})
    if not res.success:
        print("Optimization failed")
    return res.x

def nudge_centers(centers, radii):
    """Nudge circles that are tightly constrained to walls"""
    nudge_amount = 0.001
    new_centers = centers.copy()
    for i in range(len(centers)):
        x, y = centers[i]
        r = radii[i]
        # Check which wall is tight
        tight_walls = []
        if abs(x - r) < 1e-8: tight_walls.append('left')
        if abs(1 - x - r) < 1e-8: tight_walls.append('right')
        if abs(y - r) < 1e-8: tight_walls.append('bottom')
        if abs(1 - y - r) < 1e-8: tight_walls.append('top')
        if tight_walls:
            # Nudge away from tight wall
            dx, dy = 0., 0.
            if 'left' in tight_walls:
                dx += nudge_amount
            if 'right' in tight_walls:
                dx -= nudge_amount
            if 'bottom' in tight_walls:
                dy += nudge_amount
            if 'top' in tight_walls:
                dy -= nudge_amount
            # Normalize and apply
            norm = np.sqrt(dx**2 + dy**2)
            if norm > 0:
                dx, dy = dx/norm * nudge_amount, dy/norm * nudge_amount
            new_centers[i] += [dx, dy]
    return new_centers
