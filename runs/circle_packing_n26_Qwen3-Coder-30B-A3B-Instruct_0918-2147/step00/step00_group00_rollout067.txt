import numpy as np
from scipy.optimize import linprog

def run_packing():
    # Configuration
    n = 26
    max_iter = 30
    tol = 1e-6
    k = 8  # number of sides for polygonal approximation
    cos_pi_k = np.cos(np.pi / k)
    eps = 1e-12

    # 1. Geometry-first layout - triangular lattice seed
    s = 0.8 / np.floor(np.sqrt(n))
    points = []
    i = 0
    while len(points) < n:
        for j in range(int(np.ceil(n / 1.5))):
            x = i * s + (j % 2) * s / 2
            y = j * s * np.sqrt(3) / 2
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append([x, y])
            if len(points) >= n:
                break
        i += 1

    # take first n points, rank row-major
    centers = np.array(points[:n])
    
    # shift to avoid boundary
    margin = 0.05
    centers[:, 0] = np.clip(centers[:, 0], margin, 1 - margin)
    centers[:, 1] = np.clip(centers[:, 1], margin, 1 - margin)

    # initial small radii
    radii = np.full(n, 0.01)

    # 2. Helper function to compute L2 distance matrix
    def distance_matrix(C):
        diff = C[:, np.newaxis, :] - C[np.newaxis, :, :]
        return np.sqrt(np.sum(diff**2, axis=2))

    # 3. Helper function to build A_ub and b_ub for LP
    def build_radius_lp_constraints(C, r):
        n = len(C)
        # Wall constraints: each r_i <= coord
        A = []
        b = []
        for i in range(n):
            # left
            A.append([0]*i + [-1] + [0]*(n-i-1))
            b.append(-C[i, 0])
            # right
            A.append([0]*i + [-1] + [0]*(n-i-1))
            b.append(-(1 - C[i, 0]))
            # bottom
            A.append([0]*i + [-1] + [0]*(n-i-1))
            b.append(-C[i, 1])
            # top
            A.append([0]*i + [-1] + [0]*(n-i-1))
            b.append(-(1 - C[i, 1]))
            
        # Separation constraints now
        D = distance_matrix(C)
        for i in range(n):
            for j in range(i+1, n):
                dist = D[i, j]
                # r_i + r_j <= dist
                A_row = [0]*n
                A_row[i] = 1
                A_row[j] = 1
                A.append(A_row)
                b.append(dist)
        
        # Convert to np arrays to avoid issues from list of lists
        A = np.array(A)
        b = np.array(b)        
        return A, b

    def build_center_lp_constraints(C, r):
        n = len(C)
        # Build polygonal approximation of non-overlap constraints
        # Upper bound for s is size of polytope
        A_ub = []
        b_ub = []
        A_eq = []
        b_eq = []
        
        # Distance vectors for polygonal approximation
        thetas = np.linspace(0, 2*np.pi, k, endpoint=False)
        u = np.array([[np.cos(t), np.sin(t)] for t in thetas])
        usepsilon = 1e-10

        # Wall constraints: x in [r_i, 1-r_i], y in [r_i, 1-r_i]
        # Slack variables s_i for each circle
        # Add 4 structural constraints for wall
        for i in range(n):
            # left: c_i.x - r_i >= s_i
            # right: 1 - c_i.x - r_i >= s_i
            # bottom: c_i.y - r_i >= s_i
            # top: 1 - c_i.y - r_i >= s_i
            
            # these become:  
            #  -c_i.x + s_i <= -r_i  
            #   c_i.x - s_i <= 1 - r_i  
            #  -c_i.y + s_i <= -r_i  
            #   c_i.y - s_i <= 1 - r_i  
      
            A_row_i = np.zeros(2*n + n)  # 2 centers + n slacks
            A_row_i[2*i] = -1
            A_row_i[2*n + i] = 1
            A_ub.append(A_row_i)
            b_ub.append(-r[i])

            A_row_i = np.zeros(2*n + n)
            A_row_i[2*i] = 1
            A_row_i[2*n + i] = -1
            A_ub.append(A_row_i)
            b_ub.append(1 - r[i])

            A_row_i = np.zeros(2*n + n)
            A_row_i[2*i+1] = -1
            A_row_i[2*n + i] = 1
            A_ub.append(A_row_i)
            b_ub.append(-r[i])

            A_row_i = np.zeros(2*n + n)
            A_row_i[2*i+1] = 1
            A_row_i[2*n + i] = -1
            A_ub.append(A_row_i)
            b_ub.append(1 - r[i])


        # Inter-circle constraints using polygonal approx
        D = distance_matrix(C)
        for i in range(n):
            for j in range(i+1, n):
                dist_ij = D[i, j]
                if dist_ij > 0:
                    # Compute the normal sign vector for a unit vector
                    # along (c_i - c_j)
                    v_ij = C[i] - C[j]
                    v_ij_unit = v_ij / (np.linalg.norm(v_ij) + usepsilon)
                    # construct set of 8 halfspaces approximating the
                    # forbidden zone: u_l . (c_i - c_j) <= -(r_i + r_j) * cos(pi/k)
                    for l in range(k):
                        u_l = u[l]
                        dot = np.dot(u_l, v_ij_unit)
                        if abs(dot) < 1e-12:
                            continue  # Skip near-zero contributions
                        A_row = np.zeros(2*n + n)
                        # contribute to c_i and c_j
                        # left side coefficient for c_i from dot product
                        A_row[2*i] += dot
                        A_row[2*i+1] += dot
                        # right side coefficient for c_j  
                        A_row[2*j] -= dot
                        A_row[2*j+1] -= dot
                        
                        # global penalty for s_i + s_j for this constraint
                        A_row[2*n + i] += 1
                        A_row[2*n + j] += 1
                        
                        b = -(r[i] + r[j]) * cos_pi_k * dot / dot
                        if not np.isnan(b):
                            A_ub.append(A_row)
                            b_ub.append(b)
                else:
                    # QQ
                    A_row = np.zeros(2*n + n)
                    A_row[2*i] = 1
                    A_row[2*i+1] = 1
                    A_row[2*n + i] = 1
                    
                    A_row2 = np.zeros(2*n + n)
                    A_row2[2*j] = 1
                    A_row2[2*j+1] = 1
                    A_row2[2*n + j] = 1
                    A_ub.append(A_row)
                    b_ub.append(1e5)  # ensure avoid current overlaps, but slight feasible
                    A_ub.append(A_row2)
                    b_ub.append(1e5)

        # Objective is maximizing sum of s_i for each circle
        c = np.zeros(2*n + n)  # for W, X, S respectively
        c[2*n:] = 1  # sum in slack variables

        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        return A_ub, b_ub, c

    # Starting value of objective
    current_sum = np.sum(radii)
    prev_sum = -1e10
    
    # Main alternating loop
    for iteration in range(max_iter):
        prev_sum = current_sum
        
        # Fix centers, compute optimal radii (radius LP)
        A, b = build_radius_lp_constraints(centers, radii)
        # maximize 1^T r, so minimize -1^T r
        c = -np.ones(n)
        result = linprog(c, A_ub=A, b_ub=b, method='highs', tol=1e-10)
        if not result.success:
            print("Failed to solve radius LP")
            print("Summary:", result)
            # Fallback
            # try to fix by restart with slightly larger r
            radii = np.maximum(np.minimum(radii + 0.001, 0.5), 0)
            continue
        radii = np.maximum(result.x, 0)
        
        # Fix radii, move centers to increase slack (center LP)
        A_ub, b_ub, c = build_center_LP_constraints(centers, radii)
        # Solve this LP
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', tol=1e-10)
        if not result.success:
            print("Failed to solve center LP")
            print("Summary:", result)
            radii = np.maximum(np.minimum(radii + 0.001, 0.5), 0)
            continue
        # Extract new centers:
        updated_centers = result.x[:2*n].reshape((-1, 2))
        centers = np.clip(updated_centers, 0, 1)
        
        current_sum = np.sum(radii)
        if abs(current_sum - prev_sum) < tol:
            break
            
    # Local inflation
    def calc_gap(i, C, r):
        # Return minimum gap to all neighbors and walls
        min_dist = float('inf')
        ci = C[i]
        ri = r[i]
        
        # walls
        min_dist = min(min_dist, ci[0] - ri, 1 - ci[0] - ri, ci[1] - ri, 1 - ci[1] - ri)
        
        # neighbors
        for j in range(len(C)):
            if i == j:
                continue
            dist = np.linalg.norm(ci - C[j])
            min_dist = min(min_dist, dist - (ri + r[j]))
        return min_dist
    
    num_inflations = 0
    for inflate_pass in range(5):
        improved = False
        for i in range(n):
            gap = calc_gap(i, centers, radii)
            if gap > 1e-5:
                delta = min(0.4 * gap, 0.01)
                radii[i] += delta
                # Re-solve radius LP only with this circle
                A, b = build_radius_lp_constraints(centers, radii)
                c = -np.ones(n)
                # Handle learned radii constraints
                idx = [i]  # modify only the single one that changed 
                result = linprog(c, A_ub=A, b_ub=b, method='highs')
                if result.success:
                    new_radii = np.maximum(result.x, 0)
                    if new_radii[i] > radii[i] + 1e-8:
                        radii = new_radii
                        improved = True
        if not improved:
            break

    # One final validation step
    success, msg = validate_packing(centers, radii)
    if not success:
        # try to shrink radii slightly and revalidate
        shrink_factor = 1e-7
        for tries in range(5):
            radii = radii * (1 - shrink_factor)
            success, msg = validate_packing(centers, radii)
            if success:
                break

    # return final values
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
