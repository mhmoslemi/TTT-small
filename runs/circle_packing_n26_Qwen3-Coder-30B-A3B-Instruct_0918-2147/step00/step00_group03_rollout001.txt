import numpy as np
from scipy.optimize import minimize
import scipy.sparse as sp

def init_hex_grid(n=26, r_init=0.04):
    """Generate initial hexagonal grid placement."""
    rows = []
    y = r_init
    row_idx = 0
    while len(rows) < n:
        if row_idx % 2 == 0:  # even rows: aligned
            x_start = r_init
            step = 2 * r_init
        else:  # odd rows: offset
            x_start = 1.5 * r_init
            step = 2 * r_init

        x = x_start
        row = []
        while x <= 1 - r_init and len(rows) < n:
            row.append([x, y])
            x += step
            if len(rows) >= n:
                break
        rows.extend(row)
        y += np.sqrt(3) * r_init
        row_idx += 1
        if y > 1 - r_init:
            break
    
    centers = np.array(rows[:n])
    radii = np.full(n, r_init)
    return centers, radii

def max_radii(centers, radii):
    """Iteratively grow radii to their maximum allowed values."""
    old_radii = radii.copy()
    max_iter = 100
    for _ in range(max_iter):
        # Wall constraints
        wall_limit = np.minimum(centers[:, 0], 1 - centers[:, 0])
        wall_limit = np.minimum(wall_limit, centers[:, 1])
        wall_limit = np.minimum(wall_limit, 1 - centers[:, 1])
        
        # Neighbour constraints
        D = np.sqrt(((centers[:, None, :] - centers[None, :, :]) ** 2).sum(-1))
        # Set diagonal to large value to ignore self-distance
        np.fill_diagonal(D, 1e12)
        neighbor_limit = D - radii[None, :]
        neighbor_limit = np.min(neighbor_limit, axis=1)

        # Take minimum of all constraints
        new_radii = np.minimum(wall_limit, neighbor_limit)
        new_radii = np.maximum(new_radii, 0)
        
        if np.max(np.abs(new_radii - old_radii)) < 1e-8:
            break
        old_radii = new_radii
    
    return new_radii

def hill_climb(centers, radii, iterations=5000, step=0.01):
    """Stochastic hill-climbing to improve the packing."""
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = float(best_radii.sum())
    
    for _ in range(iterations):
        idx = np.random.randint(0, centers.shape[0])
        center = best_centers[idx].copy()
        r = best_radii[idx]
        
        # Propose a move
        delta = np.random.uniform(-step, step, 2)
        new_center = center + delta
        
        # Ensure it's valid to move
        if (new_center[0] - r >= 0 and new_center[0] + r <= 1 and
            new_center[1] - r >= 0 and new_center[1] + r <= 1):
            # Temporarily update center
            temp_centers = best_centers.copy()
            temp_centers[idx] = new_center
            
            # Recalculate radii to max values
            temp_radii = max_radii(temp_centers, best_radii)
            
            # Check if better
            new_sum = float(temp_radii.sum())
            if new_sum > best_sum:
                best_centers = temp_centers
                best_radii = temp_radii
                best_sum = new_sum
    
    return best_centers, best_radii

def refine_with_scipy(centers, radii):
    """Use scipy optimizer to refine the final packing."""
    n = centers.shape[0]
    # Form variable vector v = [x1,y1,x2,y2,...,xn,yn,r1,r2,...,rn]
    v0 = np.concatenate([centers.flatten(), radii])
    
    # Build constraints
    def wall_constraints(v):
        x = v[0::3]
        y = v[1::3]
        r = v[2::3]
        # r - x <= 0,  r + x -1 <= 0, r - y <= 0, r + y -1 <= 0
        result = np.zeros(4 * n)
        result[0::4] = r - x  # r - x <= 0
        result[1::4] = r + x - 1  # r + x - 1 <= 0
        result[2::4] = r - y  # r - y <= 0
        result[3::4] = r + y - 1  # r + y - 1 <= 0
        return result
    
    def overlap_constraints(v):
        x = v[0::3]
        y = v[1::3]
        r = v[2::3]
        # r_i + r_j - sqrt((x_i-x_j)^2 + (y_i-y_j)^2) <= 0
        result = []
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
                result.append(r[i] + r[j] - dist)
        return np.array(result)
    
    def jacobian_wall(v):
        x = v[0::3]
        y = v[1::3]
        r = v[2::3]
        row = []
        col = []
        data = []
        
        # wall constraints: 4 per circle
        for i in range(n):
            row.extend([4*i, 4*i+1, 4*i+2, 4*i+3])
            col.extend([3*i, 3*i, 3*i+1, 3*i+1, 3*i+2, 3*i+2, 3*i+3, 3*i+3])
            data.extend([-1, 1, -1, 1, -1, 1, -1, 1])  # gradient w.r.t. x_i and y_i and r_i
        
        return sp.coo_matrix((data, (row, col)), shape=(4*n, 3*n+26)).tocsr()
    
    def jacobian_overlap(v):
        x = v[0::3]
        y = v[1::3]
        r = v[2::3]
        row = []
        col = []
        data = []
        idx = 0
        for i in range(n):
            for j in range(i + 1, n):
                di = x[i] - x[j]
                dj = y[i] - y[j]
                dist = np.sqrt(di**2 + dj**2)
                if dist > 0:
                    dx = di / dist
                    dy = dj / dist
                    row.extend([idx, idx, idx, idx])
                    col.extend([3*i, 3*i+1, 3*j, 3*j+1])
                    data.extend([1, 1, -1, -1])  # +1 is coeff of r_i, -1 for r_j
                else:
                    row.extend([idx, idx, idx, idx])
                    col.extend([3*i, 3*i+1, 3*j, 3*j+1])
                    data.extend([1, 1, -1, -1])
                idx += 1
        return sp.coo_matrix((data, (row, col)), shape=(n*(n-1)//2, 3*n+26)).tocsr()
    
    # Bounds
    bounds = [(0, 1)] * (2*n) + [(0, 0.5)] * n  # x, y, r bounds
    
    # Constraints
    wall_cons = NonlinearConstraint(wall_constraints, -1e12, 0, jac=jacobian_wall)
    overlap_cons = NonlinearConstraint(overlap_constraints, -1e12, 0, jac=jacobian_overlap)
    
    def objective(v):
        return -np.sum(v[2*n:])  # maximize sum of radii
    
    # Optimize
    result = minimize(objective, v0, bounds=bounds, constraints=[wall_cons, overlap_cons],
                      method='trust-constr', options={'maxiter': 50})
    
    final_v = result.x
    final_centers = final_v[:2*n].reshape(-1, 2)
    final_radii = final_v[2*n:]
    
    return final_centers, final_radii

def run_packing():
    centers, radii = init_hex_grid()
    radii = max_radii(centers, radii)
    centers, radii = hill_climb(centers, radii)
    centers, radii = refine_with_scipy(centers, radii)
    sum_r = float(radii.sum())
    return centers, radii, sum_r
