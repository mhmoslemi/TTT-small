import numpy as np
from scipy.optimize import differential_evolution
from scipy.spatial.distance import cdist

def compute_radii(centers):
    n = centers.shape[0]
    radii = np.zeros(n)
    
    # Wall distances
    x = centers[:, 0]
    y = centers[:, 1]
    wall_distances = np.array([
        x,                  # left
        1 - x,              # right
        y,                  # bottom
        1 - y               # top
    ])
    
    # Pairwise distances
    distances = cdist(centers, centers)
    np.fill_diagonal(distances, np.inf)
    
    # For each circle, compute minimum of wall distances and half of closest neighbor distance
    for i in range(n):
        wall_min = np.min(wall_distances[:, i])
        neighbor_min = 0.5 * np.min(distances[i])
        radii[i] = min(wall_min, neighbor_min)
    
    return radii

def neg_sum_radii(z):
    centers = z.reshape(-1, 2)
    radii = compute_radii(centers)
    return -np.sum(radii)

def run_packing():
    np.random.seed(42)
    n_circles = 26
    dim = 2 * n_circles
    bounds = [(0, 1)] * dim
    
    # Differential Evolution
    result = differential_evolution(
        neg_sum_radii,
        bounds,
        maxiter=2000,
        popsize=10,
        tol=1e-7,
        polish=False,
        strategy='best1bin',
        seed=42
    )
    
    # Convert best centers
    centers = result.x.reshape(-1, 2)
    
    # Local hill-climbing refinement
    sigma = 0.02
    max_iter = 3000
    best_centers = centers.copy()
    best_radius_sum = -neg_sum_radii(result.x)
    
    for k in range(max_iter):
        # Generate perturbation
        perturbation = np.random.normal(0, sigma, size=centers.shape)
        new_centers = centers + perturbation
        new_centers = np.clip(new_centers, 0, 1)
        
        # Evaluate
        new_radius_sum = -neg_sum_radii(new_centers.reshape(-1))
        
        if new_radius_sum > best_radius_sum:
            best_centers = new_centers.copy()
            best_radius_sum = new_radius_sum
            sigma = min(1.1 * sigma, 0.5)
        else:
            sigma = max(0.98 * sigma, 1e-5)
        
        centers = new_centers.copy()
    
    # Final computation of radii with exact computation
