import numpy as np
from scipy.optimize import linprog
from itertools import combinations

def generate_hex_lattice(h):
    """Generate points on a hexagonal lattice with spacing h."""
    sqrt3 = np.sqrt(3)
    points = []
    j_max = int(np.ceil(1 / (h * sqrt3 / 2))) + 1
    i_max = int(np.ceil(1 / h)) + 1
    for j in range(j_max):
        y = j * h * sqrt3 / 2
        if y > 1 + 1e-12:
            break
        for i in range(i_max):
            x = i * h + (j % 2) * h / 2
            if x > 1 + 1e-12:
                break
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append([x, y])
    return np.array(points)

def build_neighbour_list(centers, max_dist=0.4):
    """Build list of neighbour pairs with distances."""
    pairs = []
    n = centers.shape[0]
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = np.sqrt(dx*dx + dy*dy)
            if d < max_dist:
                pairs.append((i, j, d))
    return pairs

def solve_radius_lp(selected_indices, R_wall, neighbour_list):
    """Solve LP for optimal radii given selected centers."""
    k = len(selected_indices)

    # Set up the LP in standard form: minimize -c^T * x
    c = -np.ones(k)
    A_eq = []
    b_eq = []
    A_ub = []
    b_ub = []

    # Wall constraints (0 <= r_i <= R_wall[i])
    for i in range(k):
        idx = selected_indices[i]
        A_ub.append([0]*k)
        A_ub[-1][i] = -1  # -r_i <= 0
        b_ub.append(0)
        A_ub.append([0]*k)
        A_ub[-1][i] = 1   # r_i <= R_wall[i]
        b_ub.append(R_wall[idx])

    # Separation constraints for neighbors
    sel_set = set(selected_indices)
    for i in range(len(neighbour_list)):
        a, b, d_ab = neighbour_list[i]
        if a in sel_set and b in sel_set:
            i_a = selected_indices.index(a)
            i_b = selected_indices.index(b)
            A_ub.append([0]*k)
            A_ub[-1][i_a] = 1
            A_ub[-1][i_b] = 1
            b_ub.append(d_ab)

    # Solve the LP
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'presolve': True})
        if res.success:
            radii = res.x
            return radii, True
        else:
            return None, False
    except:
        return None, False

def upper_bound(partial_set, remaining_indices, R_wall):
    """Estimate upper bound on total radius."""
    ub = sum(R_wall[i] for i in partial_set)
    k = 26 - len(partial_set)
    if k == 0:
        return ub
    max_r = max(R_wall[i] for i in remaining_indices)
    ub += k * max_r
    return ub

def branch_and_bound(centers, R_wall, neighbour_list, n_circles=26):
    """Find a set of 26 centers with maximum total radius."""
    best_solution = None
    best_sum = -np.inf
    stack = [(set(), set(range(len(centers))), -np.inf)]

    while stack:
        S_partial, U, lb = stack.pop()
        if len(S_partial) == n_circles:
            # Leaf node: try to solve LP
            S_list = list(S_partial)
            radii, feasible = solve_radius_lp(S_list, R_wall, neighbour_list)
            if feasible:
                total = np.sum(radii)
                if total > best_sum:
                    best_sum = total
                    best_solution = (S_list, radii)
            continue

        if len(S_partial) > n_circles:
            continue

        # Compute optimistic bound
        ub = upper_bound(S_partial, U, R_wall)
        if ub <= best_sum:
            continue

        # Greedy branching: try to include the one with highest wall radius
        best_i = max(U, key=lambda i: R_wall[i])
        U_new = U.copy()
        U_new.remove(best_i)

        # Include best_i
        S_new = S_partial.copy()
        S_new.add(best_i)
        if len(S_new) <= n_circles:
            stack.append((S_new, U_new, lb))  # This can get updated later

        # Exclude best_i
        stack.append((S_partial, U_new, lb))

    # Re-sort by index so result is deterministic
    final_set = list(best_solution[0])
    final_set.sort()
    final_radii = best_solution[1]
    return np.array(final_set), final_radii

        return -np.sum(xyr[2*n:3*n])

    # Define constraints (separations and boundaries)
    def separation_constraint(i, j):
        def c(xyr):
            x_i, y_i, r_i = xyr[i], xyr[n+i], xyr[2*n+i]
            x_j, y_j, r_j = xyr[j], xyr[n+j], xyr[2*n+j]
            dist_sq = (x_i - x_j)**2 + (y_i - y_j)**2
            return r_i + r_j - np.sqrt(dist_sq)
        return c

    def wall_constraints():
        def c(xyr):
            x = xyr[:n]
            y = xyr[n:2*n]
            r = xyr[2*n:]
            # Ensure circles are in the square and don't exceed wall radius
            lower = np.minimum(x, y)
            upper = np.minimum(1 - x, 1 - y)
            return np.concatenate([r - lower, r - upper])  # All negative or zero
        return c

        return obj(xyr)

        result = []

            result.append({'type': 'ineq', 'fun': lambda xyr, i=i: xyr[2*n+i]})  # r_i >= 0
            result.append({'type': 'ineq', 'fun': lambda xyr, i=i: 1 - xyr[i]})   # x_i <= 1
            result.append({'type': 'ineq', 'fun': lambda xyr, i=i: xyr[i]})      # x_i >= 0
            result.append({'type': 'ineq', 'fun': lambda xyr, i=i: 1 - xyr[n+i]}) # y_i <= 1
            result.append({'type': 'ineq', 'fun': lambda xyr, i=i: xyr[n+i]})     # y_i >= 0

        # Add separation constraints
        for i in range(n):
            for j in range(i+1, n):
                result.append({'type': 'ineq', 'fun': separation_constraint(i, j)})

        # Wall constraints
        wall = wall_constraints()
        for i in range(2*n):
            result.append({'type': 'ineq', 'fun': lambda xyr, i=i: wall(xyr)[i]})

        options={'maxiter': 100, 'disp': False})

        pass

    refined_x = res.x[:n]
    refined_y = res.x[n:2*n]
    refined_r = res.x[2*n:]
    
    # Final validation - ensure non-negativity and boundary constraints
    final_centers = np.column_stack([refined_x, refined_y])
    final_radii = np.maximum(refined_r, 1e-10)
    
    # Reconstruct for good measure
    return final_centers, final_radii
