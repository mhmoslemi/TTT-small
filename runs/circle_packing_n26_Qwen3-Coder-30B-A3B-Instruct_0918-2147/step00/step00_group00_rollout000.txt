import numpy as np
from scipy.optimize import minimize

def make_hex_lattice(n, margin=0.05):
    # Generate a hexagonal lattice inside the unit square
    # spacing = dx = dy * 2 / sqrt(3)
    dx = 1.0 / (np.ceil(np.sqrt(n)) + 0.5)
    dy = dx * np.sqrt(3) / 2
    
    centers = []
    j = 0
    while len(centers) < n:
        for i in range(int(np.ceil(1.0 / dx)) + 1):
            x = i * dx + (j % 2) * dx / 2
            y = j * dy
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers.append([x, y])
            if len(centers) >= n:
                break
        j += 1
    centers = np.array(centers[:n])
    
    # Center the lattice to leave margin from edges
    x_min, x_max = centers[:, 0].min(), centers[:, 0].max()
    y_min, y_max = centers[:, 1].min(), centers[:, 1].max()
    cx = (x_min + x_max) / 2
    cy = (y_min + y_max) / 2
    shift_x = 0.5 - cx
    shift_y = 0.5 - cy
    
    centers[:, 0] += shift_x
    centers[:, 1] += shift_y
    
    # Adjust so that points are inside with margin
    # We'll then calculate a safe initial radius
    return centers

def objective(z):
    # z = [x0, y0, r0, x1, y1, r1, ...]
    n = len(z) // 3
    return -np.sum(z[2::3])

def boundary_constraints(z):
    n = len(z) // 3
    constraints = []
    for i in range(n):
        x, y, r = z[3*i], z[3*i+1], z[3*i+2]
        constraints.extend([
            x - r,  # x - r >= 0
            1 - x - r,  # 1 - x - r >= 0
            y - r,  # y - r >= 0
            1 - y - r  # 1 - y - r >= 0
        ])
    return np.array(constraints)

def pairwise_constraints(z):
    n = len(z) // 3
    constraints = []
    for i in range(n):
        for j in range(i+1, n):
            x0, y0, r0 = z[3*i], z[3*i+1], z[3*i+2]
            x1, y1, r1 = z[3*j], z[3*j+1], z[3*j+2]
            dist_sq = (x0 - x1)**2 + (y0 - y1)**2
            sum_radii = r0 + r1
            constraints.append(dist_sq - sum_radii**2)
    return np.array(constraints)

def jacobian_boundary(z):
    n = len(z) // 3
    jac = np.zeros((4*n, 3*n))
    for i in range(n):
        jac[4*i, 3*i] = 1.0   # d(x - r)
        jac[4*i, 3*i+2] = -1.0
        jac[4*i+1, 3*i] = -1.0   # d(1 - x - r)
        jac[4*i+1, 3*i+2] = -1.0
        jac[4*i+2, 3*i+1] = 1.0   # d(y - r)
        jac[4*i+2, 3*i+2] = -1.0
        jac[4*i+3, 3*i+1] = -1.0   # d(1 - y - r)
        jac[4*i+3, 3*i+2] = -1.0
    return jac

def jacobian_pairwise(z):
    n = len(z) // 3
    jac = np.zeros((n*(n-1)//2, 3*n))
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            x0, y0, r0 = z[3*i], z[3*i+1], z[3*i+2]
            x1, y1, r1 = z[3*j], z[3*j+1], z[3*j+2]
            d = 2 * ((x0 - x1)*(x0 - x1) + (y0 - y1)*(y0 - y1))
            jac[idx, 3*i] = 2 * (x0 - x1) - 2 * (x0 - x1) * (r0 + r1)  # d/dx0
            jac[idx, 3*i+1] = 2 * (y0 - y1) - 2 * (y0 - y1) * (r0 + r1)  # d/dy0
            jac[idx, 3*i+2] = 2 * (r0 + r1) - 2 * (r0 + r1)**2  # d/dr0
            jac[idx, 3*j] = -2 * (x0 - x1) - 2 * (x0 - x1) * (r0 + r1)  # d/dx1
            jac[idx, 3*j+1] = -2 * (y0 - y1) - 2 * (y0 - y1) * (r0 + r1)  # d/dy1
            jac[idx, 3*j+2] = 2 * (r0 + r1) - 2 * (r0 + r1)**2  # d/dr1
            idx += 1
    return jac

def solve_nlp(initial_z):
    bounds = [(0,1), (0,1), (0,0.5)] * 26
    constraints = [
        {'type': 'ineq', 'fun': boundary_constraints, 'jac': jacobian_boundary},
        {'type': 'ineq', 'fun': pairwise_constraints, 'jac': jacobian_pairwise}
    ]
    res = minimize(
        objective,
        initial_z,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 1000, 'ftol': 1e-9}
    )
    return res.x

def run_packing():
    # step 1: initial configuration
    init_centers = make_hex_lattice(26)
    initial_radii = np.full(26, 0.02)
    initial_z = np.concatenate([init_centers.flatten(), initial_radii])

    # step 2: solve NLP
    optimized_z = solve_nlp(initial_z)

    # step 3: extract 2D array of centers
    centers = optimized_z[::3].reshape(26, 2)
    radii = optimized_z[2::3]

    # step 4: iterative refinement
    sum_radii_prev = np.sum(radii)
    for iter_idx in range(5):
        # nudge tight circles
        new_centers = centers.copy()
        new_radii = radii.copy()
        # Identify active constraints near boundaries
        for i in range(26):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            # Near walls, nudge slightly outwards
            dist_to_walls = [x - r, 1 - x - r, y - r, 1 - y - r]
            min_dist = min(dist_to_walls)
            if min_dist < 1e-4:
                # find which wall is active and nudge away
                axis_and_sign = np.argmin(dist_to_walls)
                nudge = 0.001
                if axis_and_sign == 0:
                    new_centers[i, 0] += nudge
                elif axis_and_sign == 1:
                    new_centers[i, 0] -= nudge
                elif axis_and_sign == 2:
                    new_centers[i, 1] += nudge
                else:
                    new_centers[i, 1] -= nudge
        # re-solve with nudged centers
        combined = np.concatenate([new_centers.flatten(), new_radii])
        reoptimized_z = solve_nlp(combined)
        centers = reoptimized_z[::3].reshape(26, 2)
        radii = reoptimized_z[2::3]
        sum_radii = np.sum(radii)
        if sum_radii - sum_radii_prev < 1e-5:
            break
        sum_radii_prev = sum_radii

    # step 5: validation and fine adjustment
    valid, msg = validate_packing(centers, radii)
    if not valid:
        # reduce all radii to make valid
        count = 0
        while not valid and count < 100:
            radii -= 1e-6
            if np.any(radii < 0):
                radii = np.maximum(radii, 0)
            valid, msg = validate_packing(centers, radii)
            count += 1
        if not valid:
            raise RuntimeError(f"Failed to get valid configuration: {msg}")

    return centers, radii, float(np.sum(radii))
