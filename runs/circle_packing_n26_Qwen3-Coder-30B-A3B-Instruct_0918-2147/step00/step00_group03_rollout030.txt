import numpy as np
from scipy.optimize import minimize
from math import sqrt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(0)
    
    def init_grid():
        s = 0.07
        centers = []
        for k in range(6):
            y = s / 2 + k * s * sqrt(3) / 2
            if y > 1 - s:
                break
            for m in range(10):
                if k % 2 == 0:
                    x = s / 2 + m * s
                else:
                    x = s / 2 + m * s + s / 2
                if x > 1 - s:
                    break
                if x >= s and x <= 1 - s and y >= s and y <= 1 - s:
                    centers.append([x, y])
                if len(centers) >= 26:
                    break
            if len(centers) >= 26:
                break
        centers = np.array(centers[:26])
        centers += np.random.uniform(-0.001, 0.001, (26, 2))
        radii = np.full(26, 0.001)
        return centers, radii

    def compute_forces(centers, radii, k_w=10.0, k_p=5.0):
        n = centers.shape[0]
        forces = np.zeros_like(centers)
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            
            # wall forces
            dL = x - r
            dR = (1 - x) - r
            dB = y - r
            dT = (1 - y) - r
            if dL < 0:
                forces[i, 0] += k_w * (-dL)
            if dR < 0:
                forces[i, 0] += k_w * (dR)
            if dB < 0:
                forces[i, 1] += k_w * (-dB)
            if dT < 0:
                forces[i, 1] += k_w * (dT)
            
            # pairwise forces
            for j in range(i+1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = sqrt(dx*dx + dy*dy)
                gap = dist - (radii[i] + radii[j])
                if gap < 0:
                    force_mag = k_p * (-gap)
                    forces[i, 0] += force_mag * dx / dist
                    forces[i, 1] += force_mag * dy / dist
                    forces[j, 0] -= force_mag * dx / dist
                    forces[j, 1] -= force_mag * dy / dist
        return forces

    def grow_radii(centers, radii, forces, dt=0.02, alpha=0.001):
        dot_radii = np.ones_like(radii) * alpha
        n = centers.shape[0]
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            dL = x - r
            dR = (1 - x) - r
            dB = y - r
            dT = (1 - y) - r
            if dL < 1e-12 or dR < 1e-12 or dB < 1e-12 or dT < 1e-12:
                dot_radii[i] = 0
            for j in range(n):
                if i == j:
                    continue
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = sqrt(dx*dx + dy*dy)
                gap = dist - (radii[i] + radii[j])
                if gap < 1e-12:
                    dot_radii[i] = 0
        return dot_radii

    def simulate(centers, radii, dt=0.02, gamma=1.0, tol=1e-6, max_iter=5000):
        velocities = np.zeros_like(centers)
        for iteration in range(max_iter):
            forces = compute_forces(centers, radii)
            accelerations = forces - gamma * velocities
            velocities += dt * accelerations
            centers += dt * velocities
            centers = np.clip(centers, 0, 1)
            dot_radii = grow_radii(centers, radii, forces, dt, alpha=0.001)
            radii += dt * dot_radii
            if np.all(dot_radii < 1e-12) and np.max(np.linalg.norm(forces, axis=1)) < tol:
                break
        return centers, radii

    def max_radii_fixed_centers(centers, radii, max_iters=50):
        n = centers.shape[0]
        old_radii = radii.copy()
        for _ in range(max_iters):
            new_radii = np.zeros_like(radii)
            for i in range(n):
                x, y = centers[i]
                # boundary constraints
                r_bound = min(x, 1-x, y, 1-y)
                r_min = r_bound
                for j in range(n):
                    if i == j:
                        continue
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = sqrt(dx*dx + dy*dy) - radii[j]
                    if dist > 0:
                        r_min = min(r_min, dist)
                new_radii[i] = max(0, r_min)
            if np.allclose(new_radii, old_radii, rtol=1e-9):
                break
            old_radii = new_radii
        return new_radii

    def objective_sum_radii(centers_flat):
        centers = centers_flat.reshape((-1, 2))
        radii = max_radii_fixed_centers(centers, np.full(26, 0.001))
        return -np.sum(radii)

    def objective_sum_radii_with_barrier(centers_flat, barrier_weight=1e-4):
        centers = centers_flat.reshape((-1, 2))
        radii = max_radii_fixed_centers(centers, np.full(26, 0.001))
        obj = -np.sum(radii)
        barrier = 0
        for i in range(26):
            x, y = centers[i, 0], centers[i, 1]
            barrier += (max(0, -x)**2 + max(0, x-1)**2 + max(0, -y)**2 + max(0, y-1)**2)
        return obj + barrier_weight * barrier

    # main
    centers, radii = init_grid()
    centers, radii = simulate(centers, radii)
    radii = max_radii_fixed_centers(centers, radii)
    
    # L-BFGS-B refinement on centers
    centers_flat = centers.flatten()
    res = minimize(objective_sum_radii_with_barrier, centers_flat, method='L-BFGS-B', 
                   options={'maxiter': 2000, 'ftol': 1e-9, 'gtol': 1e-9})
    refined_centers_flat = res.x
    refined_centers = refined_centers_flat.reshape((-1, 2))
    refined_radii = max_radii_fixed_centers(refined_centers, np.full(26, 0.001))
    
    sum_radii = np.sum(refined_radii)
    
    # validation
    validate_packing(refined_centers, refined_radii)
    
    return refined_centers, refined_radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
