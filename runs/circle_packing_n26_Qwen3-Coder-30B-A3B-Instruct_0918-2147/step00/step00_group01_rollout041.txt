import numpy as np
from scipy.optimize import differential_evolution
from scipy.spatial.distance import cdist

def compute_radii(centers):
    n = centers.shape[0]
    # Wall distances
    x = centers[:, 0]
    y = centers[:, 1]
    wall_distances = np.minimum.reduce([
        x, 1 - x, y, 1 - y
    ])
    
    # Pairwise distances
    distances = cdist(centers, centers)
    np.fill_diagonal(distances, np.inf)
    
    # Neighbour distances (half of minimum)
    neighbour_distances = 0.5 * np.min(distances, axis=1)
    
    # Radii are the minimum of wall and neighbour distances
    radii = np.minimum(wall_distances, neighbour_distances)
    return radii

def neg_sum_radii(z):
    centers = z.reshape(-1, 2)
    radii = compute_radii(centers)
    return -np.sum(radii)
