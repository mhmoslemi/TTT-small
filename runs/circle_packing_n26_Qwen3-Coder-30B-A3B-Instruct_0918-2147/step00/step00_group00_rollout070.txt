import numpy as np
import random

def make_hex_seed(n, margin):
    # Generate hexagonal packing with margin
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))
    
    centers = []
    radii = []
    
    spacing_x = (1 - 2 * margin) / cols
    spacing_y = (1 - 2 * margin) / rows
    half_spacing_x = spacing_x / 2.0
    
    for i in range(rows):
        for j in range(cols):
            if len(centers) >= n:
                break
            x = margin + (j + (i % 2) * 0.5) * spacing_x
            y = margin + i * spacing_y
            centers.append([x, y])
            
            # Compute initial radius based on wall distances
            r = min(x, 1 - x, y, 1 - y)
            radii.append(r * 0.4)  # Scale radius

    centers = np.array(centers[:n])
    radii = np.array(radii[:n])
    
    # Ensure the minimum radius is 0.015
    min_radius = np.min(radii)
    if min_radius < 0.015:
        scale = 0.015 / min_radius
        radii *= scale
        
    return centers, radii

def max_feasible_radius(i, centers, radii):
    c_i = centers[i]
    r_i = radii[i]
    
    # Wall distances
    d_L = c_i[0]  # Left distance
    d_R = 1.0 - c_i[0]  # Right distance
    d_B = c_i[1]  # Bottom distance
    d_T = 1.0 - c_i[1]  # Top distance
    
    # Neighbour distances
    min_d_nei = float('inf')
    for j in range(len(centers)):
        if i == j:
            continue
        c_j = centers[j]
        d_ij = np.linalg.norm(c_i - c_j) - radii[j]
        if d_ij < min_d_nei:
            min_d_nei = d_ij

    # Return the minimum of all constraints
    feasible_radius = min(d_L, d_R, d_B, d_T, min_d_nei)
    
    if feasible_radius < 0:
        feasible_radius = 0.0
    return feasible_radius

def inflate_radii(centers, radii, alpha):
    old_sum = np.sum(radii)
    mean_r = np.mean(radii)
    delta = alpha * mean_r
    radii_new = radii + delta
    
    # Clamp radii to their maximum feasible value
    for i in range(len(radii_new)):
        max_r = max_feasible_radius(i, centers, radii_new)
        radii_new[i] = min(radii_new[i], max_r)
        
    new_sum = np.sum(radii_new)
    return radii_new, new_sum - old_sum

def anneal_centers(centers, radii, temp0, gamma, sigma0, steps):
    n = len(centers)
    old_radii = radii.copy()
    max_sum = np.sum(radii)
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = max_sum
    
    # Assign random seeds for reproducibility
    np.random.seed(1234)
    random.seed(1234)
    
    temp = temp0
    sigma = sigma0
    
    for iter_num in range(steps):
        # Occasionally shift everything globally to escape local traps
        if iter_num % 50 == 0:
            shift = np.random.normal(0, 0.001, 2)
            centers += shift
            
        # Pick a random circle to move
        i = np.random.randint(0, n)
        
        # Store old position and radius
        old_center = centers[i].copy()
        old_radius = max_feasible_radius(i, centers, old_radii)
        
        # Compute new centre
        displacement = np.random.normal(0, sigma, 2)
        centers[i] += displacement
        
        # Ensure it stays in the unit square
        centers[i][0] = max(0, min(1, centers[i][0]))
        centers[i][1] = max(0, min(1, centers[i][1]))
        
        # Recompute new radius
        new_radius = max_feasible_radius(i, centers, old_radii)
        
        delta_radius = new_radius - old_radius
        if delta_radius > 0:
            # Accept immediately
            old_radii[i] = new_radius
        else:
            # Accept with probability
            try:
                p_accept = np.exp(delta_radius / temp)
                if np.random.random() < p_accept:
                    old_radii[i] = new_radius
                else:
                    # Revert the change
                    centers[i] = old_center
            except:
                # If exp overflows, accept with probability 0, i.e., revert
                centers[i] = old_center
        
        # Update best if needed
        current_sum = np.sum(old_radii)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = old_radii.copy()
            
        # Cool down
        temp *= gamma
        sigma *= gamma

    return best_centers, best_radii, best_sum

def mixed_cycle(centers, radii, params):
    alpha = params['alpha']
    temp0 = params['temp0']
    gamma = params['gamma']
    sigma0 = params['sigma0']
    steps_sa = params['steps_sa']
    steps_inflate = params['steps_inflate']
    stagnation_tolerance = params['stagnation_tolerance']
    max_cycles = params['max_cycles']
    
    old_sum = np.sum(radii)
    n = len(centers)
    best_sum = old_sum
    
    cycles_without_improvement = 0
    cycle = 0

    while cycle < max_cycles:
        prev_sum = np.sum(radii)

        # Inflate loop
        for _ in range(steps_inflate):
            new_radii, delta_sum = inflate_radii(centers, radii, alpha)
            radii = new_radii
            if delta_sum < 1e-6:
                break
        
        current_sum = np.sum(radii)
        
        # Perform SA on centres
        centers, radii, sa_sum = anneal_centers(centers, radii, temp0, gamma, sigma0, steps_sa)

        # Check convergence
        if abs(current_sum - prev_sum) < stagnation_tolerance and abs(current_sum - sa_sum) < stagnation_tolerance:
            cycles_without_improvement += 1
        else:
            cycles_without_improvement = 0
        
        if cycles_without_improvement > 3:
            break
            
        cycle += 1

    return centers, radii

def polish_radii(centers, radii, beta, passes):
    n = len(centers)
    for _ in range(passes):
        updated = True
        while updated:
            updated = False
            for i in range(n):
                # Compute exact gaps
                c_i = centers[i]
                r_i = radii[i]
                g_wall = min(c_i[0], 1 - c_i[0], c_i[1], 1 - c_i[1])
                g_nei = float('inf')
                for j in range(n):
                    if i == j:
                        continue
                    dist = np.linalg.norm(c_i - centers[j])
                    gap = dist - (r_i + radii[j])
                    if gap < g_nei:
                        g_nei = gap
                g = min(g_wall, g_nei)
                if g > 0:
                    # Try to increase radius
                    dr = beta * g
                    new_r = min(r_i + dr, max_feasible_radius(i, centers, radii))
                    if new_r > r_i:
                        radii[i] = new_r
                        updated = True
                        
        # Resolve overlaps
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                r_sum = radii[i] + radii[j]
                if dist < r_sum:
                    overlap = r_sum - dist
                    radii[i] -= overlap / 2.0
                    radii[j] -= overlap / 2.0
                    # Ensure non-negativity
                    radii[i] = max(0.0, radii[i])
                    radii[j] = max(0.0, radii[j])

    return centers, radii

def run_packing():
    np.random.seed(1234)
    random.seed(1234)
    
    # Configuration parameters
    n = 26
    margin = 0.03
    params = {
        'alpha': 0.04,
        'temp0': 0.02,
        'gamma': 0.97,
        'sigma0': 0.03,
        'steps_sa': 200,
        'steps_inflate': 20,
        'stagnation_tolerance': 1e-6,
        'max_cycles': 5
    }
    beta = 0.6
    polish_passes = 3

    # Step 1: Generate initial seeding
    centers, radii = make_hex_seed(n, margin)
    
    # Step 2: Mixed inflation + centre reoptimization
    centers, radii = mixed_cycle(centers, radii, params)
    
    # Step 3: Final polishing
    centers, radii = polish_radii(centers, radii, beta, polish_passes)
    
    # Final validation
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise Exception(f"Invalid packing: {msg}")
        
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
