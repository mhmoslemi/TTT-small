import numpy as np
from scipy.optimize import linprog, minimize
from math import sqrt

def build_hex_lattice(d):
    """Generate a hexagonal lattice of at least d points in [0,1] x [0,1]"""
    # Hexagonal lattice with spacing s
    s = 0.18  # spacing
    rows = int(1.0 / s) + 2
    cols = int(1.0 / s) + 2
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = (j + (i % 2) * 0.5) * s
            y = i * s * sqrt(3) / 2
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers.append([x, y])
    # Keep at least d points
    while len(centers) < d:
        centers.append([0.5, 0.5])  # fallback if needed
    return np.array(centers[:d])

def lp_max_radii(centers):
    """Solve LP to find radii that maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    if n == 0:
        return np.array([]), 0.0

    # Build constraint matrix
    # Wall constraints: each r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
    A_ub = []
    b_ub = []
    
    for i in range(n):
        x, y = centers[i]
        r_max = min(x, 1-x, y, 1-y)
        # Add four wall inequalities
        # r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
        # This implies: -r_i <= -x_i, -r_i <= x_i - 1, -r_i <= -y_i, -r_i <= y_i - 1
        A_ub.append([0.] * i + [-1.] + [0.] * (n - i - 1))  # -r_i <= -x_i
        b_ub.append(-x)
        
        A_ub.append([0.] * i + [-1.] + [0.] * (n - i - 1))  # -r_i <= x_i - 1
        b_ub.append(x - 1)
        
        A_ub.append([0.] * i + [-1.] + [0.] * (n - i - 1))  # -r_i <= -y_i
        b_ub.append(-y)
        
        A_ub.append([0.] * i + [-1.] + [0.] * (n - i - 1))  # -r_i <= y_i - 1
        b_ub.append(y - 1)

    # Pairwise constraints: r_i + r_j <= dist(c_i, c_j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            # Two inequalities:
            # r_i + r_j <= dist
            A_ub.append([0.] * i + [1.] + [0.] * (j - i - 1) + [1.] + [0.] * (n - j - 1))
            b_ub.append(dist)
    
    A_ub = np.array(A_ub)  # shape (4*n + n*(n-1)/2, n)
    b_ub = np.array(b_ub)  # shape (4*n + n*(n-1)/2,)

    # Objective: max sum of r_i -> minimize -sum r_i
    c = np.array([-1.0] * n)
    
    # Solve the LP
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', tol=1e-9)
    
    if res.status != 0:
        # Fallback to small uniform radii when infeasible
        radii = np.zeros(n)
        return radii, 0.0

    radii = res.x
    sum_radii = -res.fun
    return radii, sum_radii

def perturb_centers(centers, sigma):
    """Add Gaussian noise to each center and clamp to [0,1]"""
    new_centers = centers.copy() + np.random.normal(0, sigma, centers.shape)
    new_centers[:, 0] = np.clip(new_centers[:, 0], 0, 1)
    new_centers[:, 1] = np.clip(new_centers[:, 1], 0, 1)
    return new_centers

def anneal_centers(initial_centers, seed=42):
    """Use simulated annealing to optimize center positions"""
    np.random.seed(seed)
    
    # Define annealing schedule
    import itertools
    schedule = [
        {'T': 0.02, 'sigma': 0.05, 'steps': 2000},
        {'T': 0.01, 'sigma': 0.03, 'steps': 4000},
        {'T': 0.005, 'sigma': 0.015, 'steps': 3000},
        {'T': 0.002, 'sigma': 0.008, 'steps': 2000}
    ]
    
    best_centers = initial_centers.copy()
    best_radii, best_sum = lp_max_radii(best_centers)
    best_sum = float(best_sum)
    
    current_centers = initial_centers.copy()
    current_radii, current_sum = lp_max_radii(current_centers)
    current_sum = float(current_sum)
    
    for stage in schedule:
        T = stage['T']
        sigma = stage['sigma']
        steps = stage['steps']
        
        for _ in range(steps):
            # Propose new centers
            proposed_centers = perturb_centers(current_centers, sigma)
            # Get new radii
            new_radii, new_sum = lp_max_radii(proposed_centers)
            new_sum = float(new_sum)
            
            # Acceptance criterion
            if new_sum >= current_sum or np.random.rand() < np.exp((new_sum - current_sum) / T):
                current_centers = proposed_centers.copy()
                current_radii = new_radii.copy()
                current_sum = new_sum
                
                if current_sum > best_sum:
                    best_centers = current_centers.copy()
                    best_radii = current_radii.copy()
                    best_sum = current_sum

    return best_centers, best_radii, best_sum

def local_refine(centers, radii):

    def objective(x):
        # x contains centers (26*2) + radii (26)
        centers_arr = x[:52].reshape(-1, 2)
        radii_arr = x[52:]
        return -np.sum(radii_arr)

    def constraint_func(x):
        centers_arr = x[:52].reshape(-1, 2)
        radii_arr = x[52:]
        
        # Constraints:
        # wall bounds: x - r > 0, 1 - x > 0, y - r > 0, 1 - y > 0
        out = []
        for i in range(26):
            x_i, y_i = centers_arr[i]
            r_i = radii_arr[i]
            out.extend([x_i - r_i, 1 - x_i - r_i, y_i - r_i, 1 - y_i - r_i])
        
        # minimum radii from LP: radii_arr >= radii_here
        out.extend(radii_arr - radii)
        
        # pairwise constraints
        for i in range(26):
            for j in range(i+1, 26):
                x_i, y_i = centers_arr[i]
                x_j, y_j = centers_arr[j]
                r_i = radii_arr[i]
                r_j = radii_arr[j]
                dist2 = (x_i-x_j)**2 + (y_i-y_j)**2
                out.append(dist2 - (r_i+r_j)**2)
                
        return np.array(out)

    # Initialize with current values
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Define constraints
    # Variable bounds: x_i in [0,1], y_i in [0,1], r_i in [r_lp_i, +inf]
    bounds = []
    for i in range(26):
        bounds.append((0, 1))  # x_i
        bounds.append((0, 1))  # y_i
    for i in range(26):
        bounds.append((radii[i], None))  # r_i
    bounds = tuple(bounds)

    # Inequality constraints
    con = {'type': 'ineq', 'fun': constraint_func}

    # Solve with trust-constr
    res = minimize(objective, x0, method='trust-constr', bounds=bounds, constraints=con, options={'maxiter': 200})
    
    if not res.success:
        # Fallback: preserve LP solution
        centers = centers.copy()
        radii = radii.copy()
        sum_radii = np.sum(radii)
        return centers, radii, sum_radii

    # Extract new values
    new_centers = res.x[:52].reshape(-1, 2)
    new_radii = res.x[52:]
    sum_radii = -res.fun

    # Clamp centers to within bounds (fix numerical drift)
    for i in range(26):
        new_centers[i, 0] = np.clip(new_centers[i, 0], new_radii[i], 1 - new_radii[i]) 
        new_centers[i, 1] = np.clip(new_centers[i, 1], new_radii[i], 1 - new_radii[i])
        
    # Check if final placement satisfies all constraints properly
    # Ensure all radii are valid
    for i in range(26):
        new_radii[i] = max(new_radii[i], 1e-12)  # Avoid near-zero or negative
    
    # Re-optimize with potentially clamped centers if needed (small fix)
    final_radii, final_sum = lp_max_radii(new_centers)
    
    return new_centers, final_radii, final_sum
