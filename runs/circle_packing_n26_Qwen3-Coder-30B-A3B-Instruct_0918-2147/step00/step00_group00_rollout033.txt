import numpy as np
from scipy.optimize import minimize

def make_hex_lattice(n, margin=0.05):
    """Generate a hexagonal lattice of n points in [margin, 1-margin]^2."""
    dx = 1.0 / (np.ceil(np.sqrt(n)) + 0.5)
    dy = dx * np.sqrt(3) / 2
    points = []
    i = 0
    while len(points) < n:
        for j in range(int(np.ceil(n / 0.866) + 1)):
            x = i * dx + (j % 2) * dx / 2
            y = j * dy
            if margin <= x <= 1 - margin and margin <= y <= 1 - margin:
                points.append([x, y])
                if len(points) >= n:
                    break
        i += 1
    return np.array(points[:n])

def objective(z):
    """Objective function: negative sum of radii."""
    return -np.sum(z[2::3])

def boundary_constraints(z):
    """Boundary constraints: x - r >= 0, x + r <= 1, y - r >= 0, y + r <= 1."""
    n = len(z) // 3
    result = np.zeros(4 * n)
    for i in range(n):
        x, y, r = z[3*i], z[3*i+1], z[3*i+2]
        result[4*i] = x - r  # x - r >= 0
        result[4*i+1] = 1 - x - r  # x + r <= 1
        result[4*i+2] = y - r  # y - r >= 0
        result[4*i+3] = 1 - y - r  # y + r <= 1
    return result

def pairwise_constraints(z):
    """Pairwise non-overlap constraints."""
    n = len(z) // 3
    result = []
    for i in range(n):
        for j in range(i + 1, n):
            x1, y1, r1 = z[3*i], z[3*i+1], z[3*i+2]
            x2, y2, r2 = z[3*j], z[3*j+1], z[3*j+2]
            dist_sq = (x1 - x2)**2 + (y1 - y2)**2
            sum_radii = r1 + r2
            result.append(dist_sq - sum_radii**2)
    return np.array(result)

def solve_nlp(initial_z):
    """Solve the NLP problem using scipy.optimize.minimize."""
    n = len(initial_z) // 3
    
    # Objective
    obj = lambda z: objective(z)
    
    # Constraints
    cons = [
        {'type': 'ineq', 'fun': lambda z: boundary_constraints(z)},
        {'type': 'ineq', 'fun': lambda z: pairwise_constraints(z)}
    ]
    
    # Bounds
    bounds = []
    for i in range(n):
        bounds.extend([(0, 1), (0, 1), (0, 0.5)])  # x, y, r bounds
    
    # Solve
    res = minimize(
        obj,
        initial_z,
        method='SLSQP',
        constraints=cons,
        bounds=bounds,
        options={
            'maxiter': 1000,
            'ftol': 1e-9,
            'disp': False
        }
    )
    return res.x

def nudge_centers(centers, radii):
    """Nudge centers that are tight against walls."""
    margin = 1e-6
    nudged_centers = centers.copy()
    for i in range(len(centers)):
        x, y = centers[i]
        r = radii[i]
        # Check if near any wall
        if x - r < margin or x + r > 1 - margin or y - r < margin or y + r > 1 - margin:
            # Move slightly away from the wall (normal direction)
            dx, dy = 0.0, 0.0
            if x - r < margin:
                dx = 0.001
            elif x + r > 1 - margin:
                dx = -0.001
            if y - r < margin:
                dy = 0.001
            elif y + r > 1 - margin:
                dy = -0.001
            # Add small push opposite to boundary direction
            nudged_centers[i][0] += dx
            nudged_centers[i][1] += dy
    return nudged_centers

def run_packing():
    """Main function to pack 26 circles maximizing sum of radii."""
    n = 26
    # Initial setup
    centers = make_hex_lattice(n, margin=0.05)
    radii = np.full(n, 0.02)
    # Combine into one array
    initial_z = np.concatenate([centers.flatten(), radii])

    # Solve multiple times with nudging
    for _ in range(5):
        new_z = solve_nlp(initial_z)
        new_centers = new_z[::3].reshape(n, 2)
        new_radii = new_z[2::3]
        if np.sum(new_radii) - np.sum(radii) < 1e-5:
            break
        # Apply nudging
        nudge_centers_array = nudge_centers(new_centers, new_radii)
        initial_z = np.concatenate([nudge_centers_array.flatten(), new_radii])

    final_centers = initial_z[::3].reshape(n, 2)
    final_radii = initial_z[2::3]
    
    # Validate, fix if needed

            if valid:
                break
        if not valid:
            raise RuntimeError(f"Cannot produce a valid packing: {msg}")
    
    sum_radii = float(np.sum(final_radii))
    return final_centers, final_radii, sum_radii
