import numpy as np
from scipy.optimize import linprog

def initial_hex_centers(k, spacing):
    sqrt3 = np.sqrt(3)
    points = []
    i = 0
    j = 0
    while len(points) < k:
        x = i * spacing + (j % 2) * spacing / 2
        y = j * spacing * sqrt3 / 2
        if 0 <= x <= 1 and 0 <= y <= 1:
            points.append([x, y])
        if x > 1:
            j += 1
            i = 0
        else:
            i += 1
    return np.array(points[:k])

def initial_radii(k, r0):
    return np.full(k, r0)

def build_lp_constants(centers, radii):
    n = centers.shape[0]
    d_left = centers[:, 0]
    d_right = 1 - centers[:, 0]
    d_bottom = centers[:, 1]
    d_top = 1 - centers[:, 1]
    wall_dists = np.stack([d_left, d_right, d_bottom, d_top], axis=1)
    pair_dists = np.linalg.norm(centers[:, None] - centers[None, :], axis=2)
    return wall_dists, pair_dists

def solve_radius_lp(consts, radii, graph, alpha):
    wall_dists, pair_dists = consts
    n = len(radii)
    # Variables: delta_r for each circle

    # Wall capacity constraints
    A_eq = []
    b_eq = []

    # Pairwise constraints
    A_ub = []
    b_ub = []

    # Graph-based transfer limits (optional)
    graph_edges = []
    if graph is not None:
        for i in range(n):
            for j in range(i+1, n):
                if (i, j) in graph:
                    graph_edges.append((i, j))

    # Build constraint matrix and RHS
    # Wall constraints for each circle
    for i in range(n):
        max_delta = np.min(wall_dists[i]) - radii[i]
        # If max_delta < 0, we treat this as a constraint for zero growth
        max_delta = max(0, max_delta)

        A_ub.append([1 if j == i else 0 for j in range(n)])
        b_ub.append(max_delta)

    # Pairwise constraints
    for i in range(n):
        for j in range(i+1, n):
            # We only add a constraint if they aren't already adjacent in the graph
            # unless we don't have a graph, then add all of them
            if graph is None or (i, j) not in graph:
                # Create constraint: delta_r[i] + delta_r[j] <= d_ij - (r_i + r_j)
                delta = 1 if i == j else 0
                constraint = [0] * n
                constraint[i] = 1
                constraint[j] = 1
                A_ub.append(constraint)
                b_ub.append(pair_dists[i, j] - (radii[i] + radii[j]))

    # Graph-based capacity constraint: delta_r[i] <= α * sum of capacities of neighbors
    if graph is not None and alpha > 0:
        for i in range(n):
            neighbors = [j for j in range(n) if (i, j) in graph]
            if neighbors:
                # Compute sum of capacities for neighbors
                cap_sum = 0
                for j in neighbors:
                    if j < i:
                        cap_sum += pair_dists[i, j] - (radii[i] + radii[j])
                # constraint: delta_r[i] <= alpha * cap_sum
                constraint = [0] * n
                constraint[i] = 1
                A_ub.append(constraint)
                b_ub.append(alpha * cap_sum)

    # Objective: maximize sum of delta_r
    c = [-1.0] * n

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
        if result.success:
            return result.x
        else:
            return np.zeros(n)
    except:
        # Fallback - if linprog fails, return zeros
        return np.zeros(n)

def apply_increment(radii, delta_radii):
    return radii + delta_radii

def push_feasibility(centers, radii):
    n = centers.shape[0]
    # First handle wall collisions
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            radii[i] = x
        elif x + r > 1:
            radii[i] = 1 - x
        if y - r < 0:
            radii[i] = y
        elif y + r > 1:
            radii[i] = 1 - y

    # Then resolve pairwise overlaps
    # Handle overlap resolution more carefully
    changed = True
    while changed:
        changed = False
        for i in range(n):
            for j in range(i + 1, n):
                delta = centers[i] - centers[j]
                distance = np.linalg.norm(delta)
                r_sum = radii[i] + radii[j]
                if distance < r_sum:
                    # Calculate overlap and resolve
                    overlap = r_sum - distance
                    move_dist = overlap / 2.0
                    if distance > 0:
                        unit_dir = delta / distance
                        centers[i] -= move_dist * unit_dir
                        centers[j] += move_dist * unit_dir
                    else:
                        # If centers overlap perfectly, we just move one arbitrarily
                        centers[i][0] -= move_dist / 2.0
                        centers[j][0] += move_dist / 2.0
                    changed = True
                    break  # restart loop over overlaps
            if changed:
                break

    # Handle out-of-bounds again after movement
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            radii[i] = x
            centers[i][0] = r
        elif x + r > 1:
            radii[i] = 1 - x
            centers[i][0] = 1 - r
        if y - r < 0:
            radii[i] = y
            centers[i][1] = r
        elif y + r > 1:
            radii[i] = 1 - y
            centers[i][1] = 1 - r

    return centers, radii

def update_graph(graph, new_edges, wall_contacts):
    if graph is None:
        graph = set()
    for edge in new_edges:
        graph.add(edge)
    return graph

def run_iterations(centers, radii, max_iter, epsilon, alpha):
    graph = set()
    total_delta_sum = 0.0
    for _ in range(max_iter):
        constants = build_lp_constants(centers, radii)
        delta_radii = solve_radius_lp(constants, radii, graph, alpha)
        total_delta = np.sum(delta_radii)
        if total_delta < epsilon:
            break
        total_delta_sum += total_delta
        radii = apply_increment(radii, delta_radii)
        centers, radii = push_feasibility(centers, radii)
        # Update graph with new tangent edges
        newedges = []
        for i in range(len(centers)):
            for j in range(i+1, len(centers)):
                dist = np.linalg.norm(centers[i] - centers[j])
                sum_r = radii[i] + radii[j]
                if abs(dist - sum_r) < 1e-12:
                    newedges.append((i, j))
        graph.update(newedges)
    return centers, radii, graph

def final_refinement(centers, radii):
    # Do final LP with tighter constraints
    # Proceed as before
    constants = build_lp_constants(centers, radii)
    delta_radii = solve_radius_lp(constants, radii, None, 0.0)
    radii = apply_increment(radii, delta_radii)

    # Do final clipping to make sure everything works within 1e-12
    n = len(radii)
    for i in range(n):
        # Clamp radii to fit in the unit square
        r = min(radii[i], centers[i][0], 1 - centers[i][0],
                centers[i][1], 1 - centers[i][1])
        radii[i] = r

    # Final pairwise check
    for i in range(n):
        for j in range(i+1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            r_sum = radii[i] + radii[j]
            if dist < r_sum:
                diff = r_sum - dist
                # Reduce max of the two radii by a small amount
                max_r = max(radii[i], radii[j])
                if max_r > 1e-15:
                    # We reduce the one which is larger
                    if radii[i] >= radii[j]:
                        radii[i] = max(1e-15, radii[i] - (diff + 1e-13))
                    else:
                        radii[j] = max(1e-15, radii[j] - (diff + 1e-13))

    return centers, radii

def run_packing():
    # Initial configuration
    initial_spacing = 0.12
    num_circles = 26
    initial_radius = 0.005
    max_iterations = 200
    epsilon = 1e-5
    alpha = 0.5

    centers = initial_hex_centers(num_circles, initial_spacing)
    radii = initial_radii(num_circles, initial_radius)

    centers, radii, graph = run_iterations(centers, radii, max_iterations, epsilon, alpha)
    centers, radii = final_refinement(centers, radii)

    total_sum = np.sum(radii)
    return centers, radii, total_sum
