import numpy as np
from scipy.optimize import minimize

def initial_lattice(n=26):
    # Generate hexagonal lattice points
    sqrt3 = np.sqrt(3)
    spacing = 1.0
    while True:
        # Compute how many rows and cols we need
        rows = int(np.ceil(1.0 / (spacing * sqrt3 / 2)))
        cols = int(np.ceil(1.0 / spacing))
        total = rows * cols
        if total >= n:
            break
        spacing *= 0.99
    # Generate the lattice points
    points = []
    for i in range(rows):
        offset = (i % 2) * spacing / 2
        for j in range(cols):
            x = j * spacing + offset
            y = i * spacing * sqrt3 / 2
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append((x, y))
                if len(points) == n:
                    break
        if len(points) == n:
            break
    centers = np.array(points)
    radii = np.zeros(n)
    for i in range(n):
        x, y = centers[i]
        r = min(x, 1 - x, y, 1 - y)
        for j in range(n):
            if i != j:
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.hypot(dx, dy)
                r = min(r, dist / 2)
        radii[i] = max(0, r)
    return centers, radii

def pack(centers, radii):
    # Flatten the arrays
    z = np.zeros(3 * len(centers))
    for i in range(len(centers)):
        z[3*i] = centers[i, 0]
        z[3*i+1] = centers[i, 1]
        z[3*i+2] = radii[i]
    return z

def unpack(z):
    n = len(z) // 3
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = z[3*i]
        centers[i, 1] = z[3*i+1]
        radii[i] = z[3*i+2]
    return centers, radii

def objective(z):
    centers, radii = unpack(z)
    return -np.sum(radii)

def constraint_boundary(z):
    centers, radii = unpack(z)
    n = len(radii)
    cons = []
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Left boundary: x - r >= 0
        cons.append(x - r)
        # Right boundary: 1 - x - r >= 0
        cons.append(1 - x - r)
        # Bottom boundary: y - r >= 0
        cons.append(y - r)
        # Top boundary: 1 - y - r >= 0
        cons.append(1 - y - r)
    return cons

def constraint_nonoverlap(z):
    centers, radii = unpack(z)
    n = len(radii)
    cons = []
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.hypot(dx, dy)
            r_sum = radii[i] + radii[j]
            # Non-overlap constraint: dist - (r_i + r_j) >= 0
            cons.append(dist - r_sum)
    return cons

def repair(centers, radii):
    n = len(radii)
    # Make a copy to avoid modifying during iteration
    new_centers = centers.copy()
    new_radii = radii.copy()
    # First apply boundary constraints
    for i in range(n):
        x, y = new_centers[i]
        r = new_radii[i]
        r_new = min(r, x, 1 - x, y, 1 - y)
        new_radii[i] = r_new
    # Then check for overlaps
    max_iter = 10
    for _ in range(max_iter):
        any_changed = False
        for i in range(n):
            for j in range(i + 1, n):
                dx = new_centers[i, 0] - new_centers[j, 0]
                dy = new_centers[i, 1] - new_centers[j, 1]
                dist = np.hypot(dx, dy)
                r_sum = new_radii[i] + new_radii[j]
                if dist < r_sum:
                    overlap = r_sum - dist
                    # Reduce both radii
                    delta = overlap / 2
                    r_i = new_radii[i]
                    r_j = new_radii[j]
                    new_radii[i] = max(0, r_i - delta)
                    new_radii[j] = max(0, r_j - delta)
                    any_changed = True
        if not any_changed:
            break
    return new_centers, new_radii

def run_packing():
    # Initial layout
    centers, radii = initial_lattice(26)
    # Pack into flat vector
    z0 = pack(centers, radii)
    # Set up bounds: x, y in [0,1], r in [0, 0.5]
    bounds = []
    for i in range(26):
        bounds.append((0, 1))  # x bounds
        bounds.append((0, 1))  # y bounds
        bounds.append((0, 0.5))  # r bounds
    # Set up constraints
    cons = []
    # Boundary constraints
    cons.append({'type': 'ineq', 'fun': lambda z: np.array(constraint_boundary(z))})
    # Non-overlap constraints
    cons.append({'type': 'ineq', 'fun': lambda z: np.array(constraint_nonoverlap(z))})
    # Run optimization
    result = minimize(
        objective,
        z0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 500, 'ftol': 1e-9, 'disp': False}
    )
    # Unpack result
    centers, radii = unpack(result.x)
    # Apply repair
    centers, radii = repair(centers, radii)

    if not success:
        raise Exception(f"Validation failed after optimization: {msg}")
    # Compute sum
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
