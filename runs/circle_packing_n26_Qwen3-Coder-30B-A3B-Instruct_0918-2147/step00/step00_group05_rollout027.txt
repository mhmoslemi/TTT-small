import numpy as np

def build_hex_lattice(d):
    # Generate a hexagonal lattice with spacing d inside [0,1]x[0,1]
    # Number of points in a hexagonal grid to fit at least 26 points
    # There are ~26 points in a hexagonal grid with spacing 0.185
    rows = int(np.ceil(1.0 / d))
    cols = int(np.ceil(1.0 / d))
    y = np.linspace(d/2, 1-d/2, rows)
    x = np.linspace(d/2, 1-d/2, cols)
    centers = []
    for i, yi in enumerate(y):
        offset = d/2 if i % 2 == 0 else 0
        for j, xi in enumerate(x):
            if j * d + offset < 1.0 and xi + offset < 1.0:
                centers.append([xi + offset, yi])
    return np.array(centers[:26])

def lp_max_radii(centers):
    n = centers.shape[0]
    # Build the LP constraints
    # Wall constraints: r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
    A_ub = []
    b_ub = []
    
    # Row for each wall constraint for each circle
    for i in range(n):
        x, y = centers[i]
        r_bound = min(x, 1 - x, y, 1 - y)
        # Create a row for this constraint (r_i <= r_bound)
        row = [0]*n
        row[i] = 1
        A_ub.append(row)
        b_ub.append(r_bound)
    
    # Pairwise non-overlap constraints
    # For every pair (i,j): r_i + r_j <= ||c_i - c_j||
    for i in range(n):
        for j in range(i+1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            row = [0]*n
            row[i] = 1
            row[j] = 1
            A_ub.append(row)
            b_ub.append(dist)
    
    # Objective: maximize sum of radii
    c = np.ones(n)
    
    # Solve the LP
    res = linprog(-c, A_ub=A_ub, b_ub=b_ub, method='highs', tol=1e-9)
    if res.success:
        radii = res.x
        sum_radii = np.sum(radii)
        return radii, sum_radii
    else:
        # Fallback: return zero radii if unable to solve (shouldn't happen for well-separated points)
        return np.zeros(n), 0.0

def perturb_centers(centers, sigma):
    # Add Gaussian noise with std = sigma and clamp to [0, 1]
    centers = centers + np.random.normal(0, sigma, centers.shape)
    # Clamp to [0, 1]
    centers = np.clip(centers, 0, 1)
    return centers

    best_radii = current_radii.copy()
    best_sum = current_sum
    
    for temp, sigma, iter_count in zip(temperatures, sigmas, iterations):

            # Perturb the centers
            proposed_centers = perturb_centers(current_centers, sigma)
            # Compute new radii for the proposed centers
            proposed_radii, proposed_sum = lp_max_radii(proposed_centers)
            
            # Acceptance probability
            if proposed_sum >= current_sum:
                accept = True
            else:
                accept = np.random.rand() < np.exp((proposed_sum - current_sum) / temp)
