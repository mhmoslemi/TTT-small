import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
from itertools import combinations

def generate_lattice(s):
    """Generate hexagonal lattice of candidate center points."""
    sqrt3 = np.sqrt(3)
    rows = int(np.ceil(1.0 / (s * sqrt3 / 2))) + 1
    cols = int(np.ceil(1.0 / s)) + 1
    points = []
    for r in range(rows):
        y = r * s * sqrt3 / 2
        if y > 1 - s / 2:
            break
        for c in range(cols):
            if r % 2 == 0:
                x = c * s
            else:
                x = (c + 0.5) * s
            if x < s / 2 or x > 1 - s / 2 or y < s / 2 or y > 1 - s / 2:
                continue
            points.append([x, y])
    return np.array(points)

def build_milp(P):
    """Build MILP problem using cvxpy."""
    K = P.shape[0]
    z = cp.Variable(K, boolean=True)
    r = cp.Variable(K, nonneg=True)

    # Wall distances
    w = np.minimum.reduce([
        P[:, 0],           # left distance
        1 - P[:, 0],       # right distance
        P[:, 1],           # bottom distance
        1 - P[:, 1]        # top distance
    ])

    # Precompute distances between lattice points
    d = np.zeros((K, K))
    for i in range(K):
        for j in range(i+1, K):
            d[i, j] = d[j, i] = np.linalg.norm(P[i] - P[j])

    # Constraints
    constraints = [
        cp.sum(z) == 26,
    ]

    # Radius limit by walls
    constraints += [r[i] <= w[i] * z[i] for i in range(K)]

    # Non-overlap constraints
    M = 1.0
    for i, j in combinations(range(K), 2):
        # d_ij >= r_i + r_j - M*(2 - z_i - z_j)
        constraints.append(d[i, j] >= r[i] + r[j] - M * (2 - z[i] - z[j]))

    # Objective
    objective = cp.Maximize(cp.sum(r))

    # Define the problem
    prob = cp.Problem(objective, constraints)

    def solve_lp(z_fixed=None):
        # Fix binary variables if given
        if z_fixed is not None:
            for i in range(K):
                z[i].value = z_fixed[i]
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                return True, r.value
        except:
            pass
        return False, np.zeros(K)

    return solve_lp

def branch_and_bound(solve_lp_func, node_limit=20000):
    """Custom branch-and-bound implementation."""
    best_obj = -np.inf
    best_z = None
    best_r = None

    # Priority queue for B&B nodes
    from heapq import heappush, heappop
    pq = []
    heappush(pq, (-np.inf, 0, []))  # (negated bound, depth, indices)
    node_count = 0

    while pq and node_count < node_limit:
        neg_bound, depth, fixed = heappop(pq)
        bound = -neg_bound

        if bound <= best_obj:
            continue

        num_circles = sum(fixed)
        free_vars = [i for i in range(120) if i not in fixed]
        if num_circles == 26:
            # All independent binary variables are fixed
            # Check if we have a feasible solution
            z_val = np.zeros(120)
            for i, v in enumerate(fixed):
                z_val[i] = v
            found, r_vals = solve_lp_func(z_val)
            if found:
                obj = np.sum(r_vals)
                if obj > best_obj:
                    best_obj = obj
                    best_z = z_val.copy()
                    best_r = r_vals.copy()
            continue
        elif num_circles > 26:
            continue

        # Select next variable to branch
        if free_vars:
            # Pick the one furthest from integer (0.5) in the current relaxation
            max_diff = 0
            chosen = -1
            for i in free_vars:
                z_val = np.zeros(120)
                for ii in fixed:
                    z_val[ii] = 1
                z_val[i] = 0.5
                found, _ = solve_lp_func(z_val)
                if found:
                    # Use LP relaxation
                    z_var = cp.Variable(120)
                    z_var[i].value = 0.5
                    for ii in fixed:
                        z_var[ii].value = 1
                    # Compute how far it is from integer
                    diff = abs(0.5 - 0.5)
                    if diff > max_diff:
                        max_diff = diff
                        chosen = i
            else:
                chosen = free_vars[0]

        # Branch on chosen variable
        for val in [0, 1]:
            new_fixed = fixed + [chosen]
            new_fixed.sort()
            # Solve LP relaxation with free variables
            z_vals = np.zeros(120)
            for i in new_fixed:
                z_vals[i] = 1
            found, r_vals = solve_lp_func(z_vals)
            if found:
                obj = np.sum(r_vals)
                if obj > best_obj and len(new_fixed) == 26:
                    best_obj = obj
                    best_z = z_vals.copy()
                    best_r = r_vals.copy()
                neg_bound_lp = -obj
                heappush(pq, (neg_bound_lp, depth + 1, new_fixed))
            node_count += 1

    return best_z, best_r

def refine_solution(centers, radii):
    """Perform local continuous optimization."""
    # Initial assignment
    init_centers = centers.copy()
    init_radii = radii.copy()

    def objective(r):
        return -np.sum(r)

    def constraint_func(r, c):
        constraints = []
        # Add radii <= wall distances
        for i in range(len(c)):
            x, y = c[i]
            r_i = r[i]
            constraints.append(r_i - x)
            constraints.append(r_i - (1 - x))
            constraints.append(r_i - y)
            constraints.append(r_i - (1 - y))
        # Add non-overlap constraints
        for i in range(len(c)):
            for j in range(i + 1, len(c)):
                dx = c[i][0] - c[j][0]
                dy = c[i][1] - c[j][1]
                dist_sq = dx*dx + dy*dy
                r_sum = r[i] + r[j]
                constraints.append(dist_sq - r_sum*r_sum)
        return np.array(constraints)

    # Use SLSQP to optimize both centers and radii
    def opt_func(x):
        n = len(centers)
        c = x[:2*n].reshape(n, 2)
        r = x[2*n:]
        return -np.sum(r)

    def opt_conditions(x):
        n = len(centers)
        c = x[:2*n].reshape(n, 2)
        r = x[2*n:]
        g = []
        # Wall constraints
        for i in range(n):
            g.extend([
                r[i] - c[i][0],           # r <= x
                r[i] - (1 - c[i][0]),     # r <= 1 - x
                r[i] - c[i][1],           # r <= y
                r[i] - (1 - c[i][1])      # r <= 1 - y
            ])
        # Non-overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                dx = c[i][0] - c[j][0]
                dy = c[i][1] - c[j][1]
                dist_sq = dx*dx + dy*dy
                r_sum = r[i] + r[j]
                g.append(dist_sq - r_sum*r_sum)
        return np.array(g)

    # Vectorized optimization
    n = len(centers)
    x0 = np.zeros(2*n + n)
    x0[:2*n] = centers.flatten()
    x0[2*n:] = radii

    bounds = [(0, 1) for _ in range(2*n)] + [(0, None) for _ in range(n)]
    eq_constraints = {'type': 'eq', 'fun': lambda x: np.array([26 - sum(x[2*n:])])}

    # Use minimize to get better convergence
    res = minimize(
        fun=opt_func,
        x0=x0,
        method='SLSQP',
        jac=True,  # Let it use numerical gradient
        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': opt_conditions}],
        options={'ftol': 1e-9, 'maxiter': 2000}
    )

    if res.success:
        final_c = res.x[:2*n].reshape(n, 2)
        final_r = res.x[2*n:]
        total = np.sum(final_r)
        return final_c, final_r, total
    else:
        return init_centers, init_radii, np.sum(init_radii)

def run_packing():
    """Main function to run the packing."""
    # Generate candidate center lattice
    s = 0.13
    P = generate_lattice(s)

    # Build MILP
    solve_lp_func = build_milp(P)

    # Solve using custom branch and bound

        s_z, s_r = branch_and_bound(solve_lp_func)
        if s_z is None:
            # Fallback

            s_z[:26] = 1
            _, s_r = solve_lp_func(s_z)

        s_z[:26] = 1
        _, s_r = solve_lp_func(s_z)

    # Select centers according to solution
    selected_indices = np.where(s_z > 0.5)[0]

    selected_radii = s_r[selected_indices]
