import numpy as np
from scipy.optimize import differential_evolution
from scipy.spatial.distance import cdist

def compute_radii(centers):
    n = centers.shape[0]
    radii = np.zeros(n)
    
    # Wall distances
    x = centers[:, 0]
    y = centers[:, 1]
    wall_distances = np.minimum.reduce([x, 1 - x, y, 1 - y])
    
    # Pairwise distances
    distances = cdist(centers, centers)
    np.fill_diagonal(distances, np.inf)
    
    # Neighbour distances (half of minimum distance to other centers)
    neighbour_distances = 0.5 * np.min(distances, axis=1)
    
    # Radii = min(wall_distance, neighbour_distance)
    radii = np.minimum(wall_distances, neighbour_distances)
    
    return radii

def neg_sum_radii(z):
    centers = z.reshape(-1, 2)
    radii = compute_radii(centers)
    return -np.sum(radii)

def run_packing():
    np.random.seed(42)
    
    # Differential Evolution
    bounds = [(0, 1)] * 52
    result = differential_evolution(
        neg_sum_radii,
        bounds,
        maxiter=2000,
        popsize=10,
        tol=1e-7,
        polish=False,
        strategy='best1bin',
        seed=42
    )
    
    # Extract best centers from DE
    centers_de = result.x.reshape(-1, 2)
    
    # Hill-climbing refinement
    centers = centers_de.copy()
    sigma = 0.02
    max_iterations = 3000
    best_sum = -neg_sum_radii(centers.flatten())
    
    for _ in range(max_iterations):
        # Mutate
        mutation = np.random.normal(0, sigma, centers.shape)
        new_centers = centers + mutation
        new_centers = np.clip(new_centers, 0, 1)
        
        # Evaluate

        if new_sum > best_sum:
            centers = new_centers
            best_sum = new_sum
            sigma = min(1.0, 1.1 * sigma)
