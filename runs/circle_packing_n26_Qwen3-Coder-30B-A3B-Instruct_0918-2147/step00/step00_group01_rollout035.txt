import numpy as np
from scipy.spatial.distance import cdist

def penalty(z, alpha=10.0, beta=10.0):
    n = 26
    centers = z[:2*n].reshape((n, 2))
    radii = z[2*n:]
    
    # wall penalties
    walls = np.array([centers[:, 0], 1 - centers[:, 0], centers[:, 1], 1 - centers[:, 1]]).T
    wall_violations = np.maximum(0.0, radii - walls).sum()
    
    # overlap penalties
    distances = cdist(centers, centers)
    overlap_violations = 0.0
    for i in range(n):
        for j in range(i + 1, n):
            d = distances[i, j]
            r_sum = radii[i] + radii[j]
            if d < r_sum:
                overlap_violations += r_sum - d
    
    return alpha * wall_violations + beta * overlap_violations

def energy(z, lam, alpha=10.0, beta=10.0):
    sum_radii = np.sum(z[2*26:])
    penalty_val = penalty(z, alpha, beta)
    return -sum_radii + lam * penalty_val

def propose_move(z, sigma_c, sigma_r):
    z_new = z.copy()
    n = 26
    i = np.random.randint(n)
    move_type = np.random.choice(['center', 'radius', 'swap'])
    
    if move_type == 'center':
        z_new[2*i] += np.random.normal(0, sigma_c)
        z_new[2*i + 1] += np.random.normal(0, sigma_c)
        z_new[2*i] = np.clip(z_new[2*i], 0, 1)
        z_new[2*i + 1] = np.clip(z_new[2*i + 1], 0, 1)
    
    elif move_type == 'radius':
        z_new[2*n + i] += np.random.normal(0, sigma_r)
        z_new[2*n + i] = np.maximum(0, z_new[2*n + i])
    
    else:  # swap
        j = np.random.randint(n)
        if i != j:
            z_new[2*n + i], z_new[2*n + j] = z_new[2*n + j], z_new[2*n + i]
    
    return z_new

    centers = np.array([[x, y] for y in y_coords for x in x_coords])
    centers = centers[:n]
    if len(centers) < n:
        # Pad with random points
        pad = n - len(centers)
        extra = np.random.rand(pad, 2)
        centers = np.vstack([centers, extra])
    
    radii = np.minimum(centers[:, 0], 1 - centers[:, 0])
    radii = np.minimum(radii, centers[:, 1])
    radii = np.minimum(radii, 1 - centers[:, 1])
    
    z = np.concatenate([centers.flatten(), radii])
    
    T = 1.0
    lam = 0.1
    c_T = 0.995
    c_lam = 1.03
    n_iter = 10000
    sigma_c = 0.03
    sigma_r = 0.02
    
    best_feasible = z.copy()
    best_sum = np.sum(radii)
    best_energy = energy(z, lam)
    
    for k in range(n_iter):
        T *= c_T
        lam *= c_lam
        
        # Scale move sizes with temperature
        sc = sigma_c * np.sqrt(T)
        sr = sigma_r * np.sqrt(T)
        
        z_prop = propose_move(z, sc, sr)
        e_prop = energy(z_prop, lam)
        e_curr = energy(z, lam)
        
        if e_prop < e_curr or np.random.rand() < np.exp(-(e_prop - e_curr) / T):
            z = z_prop
        
        # Check if the current solution is feasible
        p = penalty(z, alpha, beta)
        if p == 0:
            sum_radii = np.sum(z[2*n:])
            if sum_radii > best_sum:
                best_sum = sum_radii
                best_feasible = z.copy()
                best_energy = e_prop
    
    return best_feasible[:2*n].reshape((n, 2)), best_feasible[2*n:], best_sum

def project_and_ascent(centers, radii, max_iter=500, eta=0.005, decay=0.95):
    n = centers.shape[0]
    z = np.concatenate([centers.flatten(), radii])
    
    for _ in range(max_iter):
        # Gradient ascent step: increase radii
        grad = np.zeros_like(z)
        grad[2*n:] = 1.0
        z += eta * grad
        
        # Project to satisfy constraints
        centers = z[:2*n].reshape((n, 2))
        radii = z[2*n:]
        
        # Clamp to wall constraints
        for i in range(n):
            r = radii[i]
            x, y = centers[i]
            max_r = min(x, 1 - x, y, 1 - y)
            radii[i] = min(r, max_r)
        
        # Resolve overlaps
        distances = cdist(centers, centers)
        max_overlap_passes = 10
        for _ in range(max_overlap_passes):
            converged = True
            for i in range(n):
                for j in range(i + 1, n):
                    d = distances[i, j]
                    r_sum = radii[i] + radii[j]
                    if d < r_sum:
                        overlap = r_sum - d
                        # shrink radii
                        radii[i] -= overlap / 2
                        radii[j] -= overlap / 2
                        converged = False
            if converged:
                break
            
            # Enforce wall constraints after each overlap resolution
            for i in range(n):
                r = radii[i]
                x, y = centers[i]
                max_r = min(x, 1 - x, y, 1 - y)
                radii[i] = min(r, max_r)
        
        z = np.concatenate([centers.flatten(), radii])
        eta *= decay
        
        # Early stopping if improvement is small
        if np.sqrt(np.sum(grad**2)) < 1e-7:
            break
    
    centers = z[:2*n].reshape((n, 2))
    radii = z[2*n:]
    return centers, radii
