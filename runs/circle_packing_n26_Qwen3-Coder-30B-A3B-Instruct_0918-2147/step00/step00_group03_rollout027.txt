import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial.distance import cdist

def build_lp_model(n_circles):
    # Build constraint matrices for the LP relaxation
    n_vars = 2 * n_circles + n_circles  # x_i, y_i, r_i for each circle
    A_eq = []
    b_eq = []
    A_ub = []
    b_ub = []

    # Variable indices
    var_indices = {}
    for i in range(n_circles):
        var_indices[(i, 'x')] = 2 * i
        var_indices[(i, 'y')] = 2 * i + 1
        var_indices[(i, 'r')] = 2 * n_circles + i

    # Wall constraints r_i ≤ x_i, r_i ≤ 1 - x_i, r_i ≤ y_i, r_i ≤ 1 - y_i
    for i in range(n_circles):
        x_idx = var_indices[(i, 'x')]
        y_idx = var_indices[(i, 'y')]
        r_idx = var_indices[(i, 'r')]
        A_ub.append([0] * x_idx + [-1] + [0] * (y_idx - x_idx - 1) + [1] + [0] * (n_vars - y_idx - 1))  # -x_i + r_i <= 0
        b_ub.append(0)
        A_ub.append([0] * x_idx + [1] + [0] * (y_idx - x_idx - 1) + [-1] + [0] * (n_vars - y_idx - 1))  # x_i - r_i - 1 <= 0
        b_ub.append(-1)
        A_ub.append([0] * y_idx + [-1] + [0] * (r_idx - y_idx - 1) + [1] + [0] * (n_vars - r_idx - 1))  # -y_i + r_i <= 0
        b_ub.append(0)
        A_ub.append([0] * y_idx + [1] + [0] * (r_idx - y_idx - 1) + [-1] + [0] * (n_vars - r_idx - 1))  # y_i - r_i - 1 <= 0
        b_ub.append(-1)

    # Pairwise separation constraints
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            # x_i + r_i <= x_j - r_j (left of)
            A_ub.append([0] * var_indices[(i, 'x')] + [1] + [0] * (var_indices[(j, 'x')] - var_indices[(i, 'x')] - 1) +
                        [-1] + [0] * (var_indices[(i, 'r')] - var_indices[(j, 'x')] - 1) + [1] +
                        [0] * (var_indices[(j, 'r')] - var_indices[(i, 'r')] - 1) + [-1] + [0] * (n_vars - var_indices[(j, 'r')] - 1))
            b_ub.append(0)
            # x_j + r_j <= x_i - r_i (right of)
            A_ub.append([0] * var_indices[(j, 'x')] + [1] + [0] * (var_indices[(i, 'x')] - var_indices[(j, 'x')] - 1) +
                        [-1] + [0] * (var_indices[(j, 'r')] - var_indices[(i, 'x')] - 1) + [1] +
                        [0] * (var_indices[(i, 'r')] - var_indices[(j, 'r')] - 1) + [-1] + [0] * (n_vars - var_indices[(i, 'r')] - 1))
            b_ub.append(0)
            # y_i + r_i <= y_j - r_j (below)
            A_ub.append([0] * var_indices[(i, 'y')] + [1] + [0] * (var_indices[(j, 'y')] - var_indices[(i, 'y')] - 1) +
                        [-1] + [0] * (var_indices[(i, 'r')] - var_indices[(j, 'y')] - 1) + [1] +
                        [0] * (var_indices[(j, 'r')] - var_indices[(i, 'r')] - 1) + [-1] + [0] * (n_vars - var_indices[(j, 'r')] - 1))
            b_ub.append(0)
            # y_j + r_j <= y_i - r_i (above)
            A_ub.append([0] * var_indices[(j, 'y')] + [1] + [0] * (var_indices[(i, 'y')] - var_indices[(j, 'y')] - 1) +
                        [-1] + [0] * (var_indices[(j, 'r')] - var_indices[(i, 'y')] - 1) + [1] +
                        [0] * (var_indices[(i, 'r')] - var_indices[(j, 'r')] - 1) + [-1] + [0] * (n_vars - var_indices[(i, 'r')] - 1))
            b_ub.append(0)

    # Objective: minimize sum of -r_i
    c = [0] * (2 * n_circles)
    c.extend([-1] * n_circles)
    return c, A_ub, b_ub, A_eq, b_eq, var_indices

def solve_lp(n_circles, verbose=False):
    c, A_ub, b_ub, A_eq, b_eq, var_indices = build_lp_model(n_circles)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, method='highs', options={'presolve': True})
    if verbose:
        print("LP solution status:", res.status, res.message)
    if not res.success:
        raise RuntimeError("LP solver failed to find a solution")
    # Extract centers and radii
    centers = np.zeros((n_circles, 2))
    radii = np.zeros(n_circles)
    for i in range(n_circles):
        centers[i, 0] = res.x[var_indices[(i, 'x')]]
        centers[i, 1] = res.x[var_indices[(i, 'y')]]
        radii[i] = res.x[var_indices[(i, 'r')]]
    return centers, radii

def detect_diagonal_conflicts(centers, radii):
    n = centers.shape[0]
    edges = []
    distances = cdist(centers, centers)
    for i in range(n):
        for j in range(i + 1, n):
            d = distances[i, j]
            if d < radii[i] + radii[j] - 1e-12:
                edges.append((i, j))
    return edges

    new_radii = radii.copy()
    converged = False
    max_iter = 100
    for _ in range(max_iter):
        old_radii = new_radii.copy()
        for i in range(n):
            min_dist = float('inf')
            r_i = new_radii[i]
            c_i = centers[i]
            for j in range(n):
                if i == j:
                    continue
                r_j = new_radii[j]
                c_ij = centers[j]
                d = np.linalg.norm(c_i - c_ij)
                min_dist = min(min_dist, d - r_j)
            wall_dist = min(c_i[0], 1 - c_i[0], c_i[1], 1 - c_i[1])
            new_radii[i] = min(wall_dist, min_dist)
        if np.allclose(new_radii, old_radii, rtol=1e-12):
            break
    return centers, new_radii
