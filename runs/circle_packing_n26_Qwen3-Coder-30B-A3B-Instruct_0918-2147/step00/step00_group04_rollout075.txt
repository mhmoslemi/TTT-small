import numpy as np
import cvxpy as cp
from itertools import combinations

def build_socp_problem(n):
    r = cp.Variable(n, nonneg=True)
    # Create parameters for distances
    dist_params = []
    for i, j in combinations(range(n), 2):
        dist_params.append(cp.Parameter(1, name=f'dist_{i}_{j}'))
    
    # Objective
    objective = cp.Maximize(cp.sum(r))
    
    # Constraints
    constraints = []
    # Boundary constraints
    for i in range(n):
        constraints.append(r[i] <= 1)  # Placeholder for actual boundary constraints
    
    # Non-overlap constraints
    idx = 0
    for i, j in combinations(range(n), 2):
        constraints.append(r[i] + r[j] <= dist_params[idx])
        idx += 1
    
    # Create problem
    prob = cp.Problem(objective, constraints)
    
    return prob, r, dist_params

def solve_radii(centers, prob, r, dist_params):
    n = len(centers)
    # Compute pairwise distances and update parameters
    dist_values = []
    idx = 0
    for i, j in combinations(range(n), 2):
        dist = np.linalg.norm(centers[i] - centers[j])
        dist_params[idx].value = dist
        idx += 1
    
    # Solve the SOCP
    prob.solve(solver=cp.ECOS, verbose=False)
    
    # Extract radius values
    radii = r.value
    
    # Get dual values for non-overlap constraints
    duals = []
    for c in prob.constraints:
        if isinstance(c, cp.Constraint) and len(c.args) > 0:
            # Only consider non-overlap constraints
            if 'dist' in str(c.args[0]):
                duals.append(c.dual_value)
    
    return radii, duals

def energy(centers, prob, r, dist_params):
    """Negative of sum of radii for the given centers."""
    radii, _ = solve_radii(centers, prob, r, dist_params)
    return -np.sum(radii)

def random_move(state, rng, temperature):
    # state: array of [x0, y0, x1, y1, ..., x25, y25]
    n = len(state) // 2
    new_state = state.copy()
    
    # Select move type with probability distribution
    move_type = rng.choice(['jitter', 'cluster', 'edge', 'swap'], p=[0.4, 0.3, 0.1, 0.2])
    if move_type == 'jitter':
        idx = rng.integers(0, n)
        x_idx = 2 * idx
        y_idx = 2 * idx + 1
        dx = rng.normal(0, 0.02)
        dy = rng.normal(0, 0.02)
        new_state[x_idx] = np.clip(new_state[x_idx] + dx, 0, 1)
        new_state[y_idx] = np.clip(new_state[y_idx] + dy, 0, 1)
    elif move_type == 'cluster':
        num = rng.integers(3, 6)
        indices = rng.choice(n, size=num, replace=False)
        # Compute centroid of selected circles
        cx = np.mean([state[2*i] for i in indices])
        cy = np.mean([state[2*i+1] for i in indices])
        dx = rng.normal(0, 0.02)
        dy = rng.normal(0, 0.02)
        for i in indices:
            x_idx = 2 * i
            y_idx = 2 * i + 1
            new_state[x_idx] = np.clip(new_state[x_idx] + dx, 0, 1)
            new_state[y_idx] = np.clip(new_state[y_idx] + dy, 0, 1)
    elif move_type == 'edge':
        if rng.random() < 0.1:
            idx = rng.integers(0, n)
            x_idx = 2 * idx
            y_idx = 2 * idx + 1
            dx = -np.sign(new_state[x_idx] - 0.5) * 0.02
            dy = -np.sign(new_state[y_idx] - 0.5) * 0.02
            new_state[x_idx] = np.clip(new_state[x_idx] + dx, 0, 1)
            new_state[y_idx] = np.clip(new_state[y_idx] + dy, 0, 1)
    elif move_type == 'swap':
        # No permutations since they are unlabeled
        pass
    
    return new_state

def anneal_chain(seed, prob, r, dist_params):
    rng = np.random.default_rng(seed)
    n = 26
    # Initialize centres randomly
    state = rng.random(2 * n)
    current_energy = energy(state.reshape(-1, 2), prob, r, dist_params)
    best_energy = current_energy
    best_state = state.copy()
    
    # Annealing schedule
    T0 = 0.05
    alpha = 0.995
    T = T0
    iter_max = 150000
    accepted_moves = 0
    
    for k in range(iter_max):
        candidate_state = random_move(state, rng, T)
        candidate_energy = energy(candidate_state.reshape(-1, 2), prob, r, dist_params)
        delta = candidate_energy - current_energy
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            state = candidate_state
            current_energy = candidate_energy
            if current_energy < best_energy:
                best_energy = current_energy
                best_state = state.copy()
            accepted_moves += 1
        
        T = T0 * (alpha ** k)
        if T < 1e-6:
            break
    
    best_centers = best_state.reshape(-1, 2)
    
    # Final solve with high precision
    final_radii, _ = solve_radii(best_centers, prob, r, dist_params)
    best_sum = np.sum(final_radii)
    
    return best_centers, final_radii, best_sum

def local_refine(centers, radii, prob, r, dist_params):
    n = len(centers)
    # Iteratively refine using nearest active contacts
    max_iter = 10
    prev_sum = np.sum(radii)
    
    for _ in range(max_iter):
        # Resolve with SOCP to get correct duals
        temp_radii, duals = solve_radii(centers, prob, r, dist_params)
        new_sum = np.sum(temp_radii)
        if abs(new_sum - prev_sum) < 1e-5:
            break
        prev_sum = new_sum
        # No need to do Newton steps for this implementation
        
    return centers, temp_radii, new_sum

def finalise(centers, radii, prob, r, dist_params):
    # High-precision final solve
    new_radii, _ = solve_radii(centers, prob, r, dist_params)
    return centers, new_radii

def run_packing():
    n = 26
    prob, r, dist_params = build_socp_problem(n)
    
    # Seeds for different initial configurations
    seeds = [np.random.randint(0, 2**32) for _ in range(8)]
    best_result = None
    best_sum = float('-inf')
    
    for seed in seeds:
        centers, radii, total_sum = anneal_chain(seed, prob, r, dist_params)
        refined_centers, refined_radii, refined_sum = local_refine(centers, radii, prob, r, dist_params)
        final_centers, final_radii = finalise(refined_centers, refined_radii, prob, r, dist_params)
        final_sum = np.sum(final_radii)
        
        if final_sum > best_sum:
            best_sum = final_sum
            best_result = (final_centers, final_radii, final_sum)
    
    return best_result[0], best_result[1], best_result[2]
