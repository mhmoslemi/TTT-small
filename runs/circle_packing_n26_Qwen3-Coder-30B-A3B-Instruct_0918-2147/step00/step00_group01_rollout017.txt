import numpy as np
from scipy.spatial.distance import cdist

def penalty(z, alpha=10.0, beta=10.0):
    n = 26
    centers = z[:2*n].reshape(n, 2)
    radii = z[2*n:]
    penalties = 0.0

    # Wall penalties
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        wall_distances = [x, 1-x, y, 1-y]
        for d in wall_distances:
            if r > d:
                penalties += alpha * (r - d)

    # Overlap penalties
    for i in range(n):
        for j in range(i+1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                penalties += beta * overlap

    return penalties

def energy(z, lam, alpha=10.0, beta=10.0):
    n = 26
    radii = z[2*n:]
    obj = np.sum(radii)
    penalty_val = penalty(z, alpha, beta)
    return -obj + lam * penalty_val

def propose_move(z, sigma_c, sigma_r):
    n = 26
    z_new = z.copy()
    move_type = np.random.randint(0, 3)

    if move_type == 0:  # Center jitter
        i = np.random.randint(0, n)
        x, y = z_new[2*i], z_new[2*i+1]
        z_new[2*i] = np.clip(x + np.random.normal(0, sigma_c), 0, 1)
        z_new[2*i+1] = np.clip(y + np.random.normal(0, sigma_c), 0, 1)
    elif move_type == 1:  # Radius perturb
        i = np.random.randint(0, n)
        r = z_new[2*n+i]
        z_new[2*n+i] = max(0, r + np.random.normal(0, sigma_r))
    else:  # Swap pair
        i, j = np.random.choice(n, 2, replace=False)
        r_i = z_new[2*n+i]
        r_j = z_new[2*n+j]
        z_new[2*n+i] = r_j
        z_new[2*n+j] = r_i

    return z_new

def anneal():
    np.random.seed(42)
    n = 26
    T0 = 1.0
    c_T = 0.995
    T_min = 1e-5
    lam0 = 0.1
    c_lam = 1.03
    max_iter = 10000
    sigma_c = 0.03
    sigma_r = 0.02

    # Hexagonal initial placement (approx)
    grid_x = np.linspace(0.18, 0.82, 5)
    grid_y = np.linspace(0.18, 0.82, 5)
    x_vals, y_vals = np.meshgrid(grid_x, grid_y)
    centers = np.array([x_vals.flatten(), y_vals.flatten()])[:26].T
    radii = np.min(centers, axis=1)
    radii = np.minimum(radii, 1 - centers[:, 0])
    radii = np.minimum(radii, centers[:, 1])
    radii = np.minimum(radii, 1 - centers[:, 1])

    z = np.concatenate([centers.flatten(), radii])
    T = T0
    lam = lam0

    best_energy = energy(z, lam)
    best_z = z.copy()
    best_feasible_z = z.copy()
    best_feasible_energy = energy(z, lam)

    for i in range(max_iter):
        z_prop = propose_move(z, sigma_c * np.sqrt(T / T0), sigma_r * np.sqrt(T / T0))
        delta_E = energy(z_prop, lam) - energy(z, lam)
        if delta_E <= 0 or np.random.rand() < np.exp(-delta_E / T):
            z = z_prop

        if penalty(z) <= 1e-10 and energy(z, lam) < best_feasible_energy:
            best_feasible_z = z.copy()
            best_feasible_energy = energy(z, lam)

        if energy(z, lam) < best_energy:
            best_z = z.copy()
            best_energy = energy(z, lam)

        T *= c_T
        lam *= c_lam
        if T < T_min:
            break

    return best_feasible_z

def project_and_ascent(centers, radii):
    n = 26
    eta = 5e-3
    max_iter = 500
    tol = 1e-7

    for _ in range(max_iter):
        # Compute overlaps
        centers_arr = centers.copy()
        radii_arr = radii.copy()
        all_overlaps = []
        for i in range(n):
            for j in range(i+1, n):
                d = np.sqrt(np.sum((centers_arr[i] - centers_arr[j]) ** 2))
                overlap = radii_arr[i] + radii_arr[j] - d
                if overlap > 0:
                    all_overlaps.append((i, j, overlap))

        if not all_overlaps:
            # No overlaps, take a gradient step
            grad = np.zeros_like(radii_arr)
            grad[:] = 1.0
            radii_arr += eta * grad
            # Clamp to wall constraints
            for i in range(n):
                r = radii_arr[i]
                x, y = centers_arr[i]
                r_max = min(x, 1-x, y, 1-y)
                radii_arr[i] = min(r, r_max)
            # Now check for any overlaps again
            flag = False
            for i in range(n):
                for j in range(i+1, n):
                    d = np.sqrt(np.sum((centers_arr[i] - centers_arr[j]) ** 2))
                    overlap = radii_arr[i] + radii_arr[j] - d
                    if overlap > 0:
                        flag = True
                        break
                if flag:
                    break
            if not flag:
                # Update radii and cumulative sum
                radii = radii_arr.copy()
                continue
            else:
                # Some overlaps: decrease them
                for i, j, overlap in all_overlaps:
                    r_old_i, r_old_j = radii_arr[i], radii_arr[j]
                    s = overlap / 2.0
                    radii_arr[i] = max(0.0, r_old_i - s)
                    radii_arr[j] = max(0.0, r_old_j - s)
                # After applying reduction, clamp again
                for i in range(n):
                    r = radii_arr[i]
                    x, y = centers_arr[i]
                    r_max = min(x, 1-x, y, 1-y)
                    radii_arr[i] = min(r, r_max)
                radii = radii_arr.copy()
        else:
            # Resolve overlaps step-by-step
            radii_arr = radii.copy()
            for i, j, overlap in all_overlaps:
                r_old_i, r_old_j = radii_arr[i], radii_arr[j]
                s = overlap / 2.0
                radii_arr[i] = max(0.0, r_old_i - s)
                radii_arr[j] = max(0.0, r_old_j - s)
            # Clamp to boundary
            for i in range(n):
                r = radii_arr[i]
                x, y = centers_arr[i]
                r_max = min(x, 1-x, y, 1-y)
                radii_arr[i] = min(r, r_max)
            radii = radii_arr.copy()

        eta *= 0.95
        if eta < 1e-6:
            break

    return centers, radii
