import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import math

def generate_lattice(s):
    # Generate a hexagonal lattice inside unit square
    # spacing s, with a margin to avoid boundary issues
    margin = s / 2
    rows = int(np.ceil(1 / (s * np.sqrt(3) / 2))) + 1
    cols = int(np.ceil(1 / s)) + 1
    
    points = []
    for r in range(rows):
        y = r * s * np.sqrt(3) / 2
        if y < margin or y > 1 - margin:
            continue
        for c in range(cols):
            if r % 2 == 0:
                x = c * s
            else:
                x = (c + 0.5) * s
            if x < margin or x > 1 - margin:
                continue
            points.append([x, y])
    return np.array(points)

def build_milp(P):
    # Build the MILP model
    K = P.shape[0]
    z = cp.Variable(K, boolean=True)
    r = cp.Variable(K, nonneg=True)
    
    # Precompute wall distances
    wL = P[:, 0]
    wR = 1 - P[:, 0]
    wB = P[:, 1]
    wT = 1 - P[:, 1]
    w = np.minimum(np.minimum(wL, wR), np.minimum(wB, wT))
    
    # Precompute distances
    D = np.zeros((K, K))
    for i in range(K):
        for j in range(i + 1, K):
            dist = np.linalg.norm(P[i] - P[j])
            D[i, j] = dist
            D[j, i] = dist
    
    # Constraints
    constraints = []
    
    # Exactly 26 circles
    constraints.append(cp.sum(z) == 26)
    
    # Link radii to selection with wall limits
    for i in range(K):
        constraints.append(r[i] <= w[i] * z[i])
    
    # Non-overlapping constraints with big-M
    M = 1
    for i in range(K):
        for j in range(i + 1, K):
            constraints.append(D[i, j] >= r[i] + r[j] - M * (2 - z[i] - z[j]))
    
    # Objective
    objective = cp.Maximize(cp.sum(r))
    
    # Build the problem
    prob = cp.Problem(objective, constraints)
    
    def solve_node(z_fixed_vals):
        # Set the fixed values
        for i, val in enumerate(z_fixed_vals):
            z[i].fixed_value = val
        
        try:
            prob.solve(solver=cp.OSQP, eps_abs=1e-9, eps_rel=1e-9)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                return True, prob.value, r.value
            else:
                return False, None, None
        except:
            return False, None, None
    
    return solve_node

def branch_and_bound(milp_solver, node_limit=20000):
    # Custom branch and bound solver
    best_result = None
    best_value = -np.inf
    
    # Priority queue: (bound, node_index, z_vals)
    import heapq
    pq = []
    
    # Initial node: all variables free
    z_initial = np.full(120, -1, dtype=int)  # -1 means free
    
    # Calculate initial bound
    feasible, bound, sol = milp_solver(z_initial)
    if feasible and bound > best_value:
        best_value = bound
        best_result = (z_initial.copy(), sol.copy())
    
    # Start branching
    node_count = 0
    heapq.heappush(pq, (-bound, node_count, z_initial))
    node_count += 1
    
    while pq and node_count < node_limit:
        neg_bound, node_id, z_vals = heapq.heappop(pq)
        bound = -neg_bound
        
        # Check if we already have a better solution
        if bound <= best_value:
            continue
            
        # Find the first free variable
        free_idx = -1
        for i in range(len(z_vals)):
            if z_vals[i] == -1:
                free_idx = i
                break
        
        if free_idx == -1:
            # All fixed, this is a solution
            if not np.isclose(bound, 0):
                if bound > best_value:
                    best_value = bound
                    # Get the actual solution
                    feasible, final_bound, sol = milp_solver(z_vals)
                    if feasible:
                        best_result = (z_vals.copy(), sol.copy())
        else:
            # Branch on this variable
            z0 = z_vals.copy()
            z0[free_idx] = 0
            z1 = z_vals.copy()
            z1[free_idx] = 1
            
            for z_new in [z0, z1]:
                feasible, new_bound, sol = milp_solver(z_new)
                if feasible:
                    if new_bound > best_value:
                        best_value = new_bound
                        best_result = (z_new.copy(), sol.copy())
                    if new_bound > best_value or not np.isclose(new_bound, 0):
                        heapq.heappush(pq, (-new_bound, node_count, z_new))
                        node_count += 1
                        
    if best_result is None:
        raise RuntimeError("No solution found in branch and bound")
    
    return best_result

def refine_solution(centers, radii):
    # Refine with SLSQP
    n = centers.shape[0]
    # Prepare variables: [x1, y1, r1, x2, y2, r2, ...]
    
    def obj(x):
        total = 0
        for i in range(n):
            total += x[i * 3 + 2]
        return -total  # maximize by minimizing negative
    
    def constraintfun(x):
        # Non-overlap constraints
        cons = []
        for i in range(n):
            # Wall constraints
            r = x[i * 3 + 2]
            x_pos = x[i * 3]
            y_pos = x[i * 3 + 1]
            cons.append(x_pos - r)  # r <= x_pos
            cons.append(y_pos - r)  # r <= y_pos
            cons.append(1 - x_pos - r)  # r <= 1 - x_pos
            cons.append(1 - y_pos - r)  # r <= 1 - y_pos
            
            for j in range(i + 1, n):
                dx = x[i * 3] - x[j * 3]
                dy = x[i * 3 + 1] - x[j * 3 + 1]
                dist_sq = dx * dx + dy * dy
                r_sum = x[i * 3 + 2] + x[j * 3 + 2]
                # penalty if this is violated: (dist_sq - r_sum^2) >= 0
                cons.append(dist_sq - r_sum * r_sum)  # dist >= r1 + r2
        return np.array(cons)
    
    # Build initial guess
    x0 = np.zeros(n * 3)
    for i in range(n):
        x0[i * 3] = centers[i, 0]
        x0[i * 3 + 1] = centers[i, 1]
        x0[i * 3 + 2] = radii[i]
    
    # Bounds
    bounds = []
    for i in range(n):
        bounds.append((0, 1))  # x
        bounds.append((0, 1))  # y
        bounds.append((0, 1))  # r
    
    # Constraints
    cons = {'type': 'ineq', 'fun': constraintfun}
    res = minimize(obj, x0, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 2000, 'ftol': 1e-9, 'disp': False})
    
    if res.success:
        refined_centers = np.zeros((n, 2))
        refined_radii = np.zeros(n)
        for i in range(n):
            refined_centers[i, 0] = res.x[i * 3]
            refined_centers[i, 1] = res.x[i * 3 + 1]
            refined_radii[i] = res.x[i * 3 + 2]
            
        # Validate
        status, msg = validate_packing(refined_centers, refined_radii)
        if not status:
            # Try to fix small violations by scaling down
            scale_factor = 1.0
            # Reduce the corresponding values slighty to ensure validity
            shrink = 1e-7
            while scale_factor > 0:
                scale_factor -= shrink
                fixed_centers = refined_centers.copy()
                fixed_radii = refined_radii * scale_factor
                status, msg = validate_packing(fixed_centers, fixed_radii)
                if status:
                    break
            if not status:
                raise RuntimeError("Could not refine solution to valid configuration")
            refined_radii = fixed_radii
            
        return refined_centers, refined_radii, np.sum(refined_radii)
    else:
        raise RuntimeError(f"SLSQP refinement failed: {res.message}")

def run_packing():
    # 1. Generate lattice
    P = generate_lattice(s=0.13)
    
    # 2. Build MILP model
    milp_solver = build_milp(P)
    
    # 3. Branch and bound
    z_vals, r_vals = branch_and_bound(milp_solver, node_limit=20000)
    
    # 4. Get the selected points
    selected_indices = np.where(z_vals == 1)[0]
    selected_centers = P[selected_indices]
    selected_radii = r_vals[selected_indices]
    
    # 5. Refine solution
    centers_refined, radii_refined, total_sum = refine_solution(selected_centers, selected_radii)
    
    return centers_refined, radii_refined, total_sum
