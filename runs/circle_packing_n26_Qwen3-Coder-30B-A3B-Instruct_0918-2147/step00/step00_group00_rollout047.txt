import numpy as np
from scipy.optimize import linprog
from scipy.spatial.distance import cdist

def generate_hex_lattice(h):
    sqrt3 = np.sqrt(3)
    points = []
    for j in range(int(1.0 / h) + 2):
        y = j * h * sqrt3 / 2
        if y > 1.0:
            break
        for i in range(int(1.0 / h) + 2):
            x = i * h + (j % 2) * h / 2
            if x > 1.0:
                break
            points.append([x, y])
    points = np.array(points)
    R_wall = np.minimum(points[:, 0], 1 - points[:, 0])
    R_wall = np.minimum(R_wall, points[:, 1])
    R_wall = np.minimum(R_wall, 1 - points[:, 1])
    return points, R_wall

def build_neighbour_list(centers, max_dist):
    distances = cdist(centers, centers)
    mask = (distances > 0) & (distances < max_dist)
    edges = []
    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):
            if mask[i, j]:
                edges.append((i, j, distances[i, j]))
    return edges

def solve_radius_lp(selected_indices, R_wall, neighbour_list):
    n = len(selected_indices)
    if n == 0:
        return np.array([]), True

    # Build indices map
    idx_map = {selected_indices[i]: i for i in range(n)}

    # Constraints for wall limits
    A_ub = np.zeros((n, n))
    b_ub = np.zeros(n)
    for i, s_idx in enumerate(selected_indices):
        A_ub[i, i] = 1
        b_ub[i] = R_wall[s_idx]

    # Separation constraints
    A_separation = []
    b_separation = []
    for i, j, d_ij in neighbour_list:
        if i in idx_map and j in idx_map:
            idx_i = idx_map[i]
            idx_j = idx_map[j]
            A_row = np.zeros(n)
            A_row[idx_i] = 1
            A_row[idx_j] = 1
            A_separation.append(A_row)
            b_separation.append(d_ij)

    if A_separation:
        A_ub = np.vstack([A_ub] + A_separation)
        b_ub = np.hstack([b_ub] + b_separation)

    # Objective
    c = np.ones(n)

    # Solve LP
    res = linprog(-c, A_ub=A_ub, b_ub=b_ub, method='highs', bounds=[(0, None)] * n)
    if res.success:
        return res.x, True
    else:
        return res.x, False

def upper_bound(partial_set, remaining_indices, R_wall):
    ub = sum(R_wall[i] for i in partial_set)
    # Add largest wall radii for rest
    sorted_rest = sorted(remaining_indices, key=lambda i: R_wall[i], reverse=True)
    needed = 26 - len(partial_set)
    for i in range(needed):
        if i < len(sorted_rest):
            ub += R_wall[sorted_rest[i]]
    return ub

def branch_and_bound(centers, R_wall, neighbour_list):
    M = len(centers)
    best_total = -float('inf')
    best_centers = None
    best_radii = None

    def backtrack(S_partial, U, lb):
        nonlocal best_total, best_centers, best_radii

        # If we have 26 circles, we must check feasibility
        if len(S_partial) == 26:
            radii, feasible = solve_radius_lp(S_partial, R_wall, neighbour_list)
            if feasible:
                total = float(np.sum(radii))
                if total > best_total:
                    best_total = total
                    best_radii = radii.copy()
                    best_centers = centers[S_partial].copy()
            return

        # Pruning
        ub = upper_bound(S_partial, U, R_wall)
        if ub <= lb:
            return

        # If there are fewer than 26 circles, go further
        if len(U) == 0:
            return

        # Pick the next one based on largest wall radius
        c = max(U, key=lambda i: R_wall[i])
        U_new = U - {c}
        S_partial_new = S_partial | {c}

        # Propagation of the exact lower bound (feasibility)
        # Try including c
        topo = list(S_partial_new)
        radii, feasible = solve_radius_lp(topo, R_wall, neighbour_list)
        if feasible:
            total = float(np.sum(radii))
            if total > lb:
                lb = total
            backtrack(S_partial_new, U_new, lb)
        else:
            # even with c included, it is infeasible
            pass

        # Try not including c
        backtrack(S_partial, U_new, lb)

    backtrack(set(), set(range(M)), -float('inf'))
    return best_centers, best_radii

def local_continuous_refinement(initial_centers, initial_radii):
    from scipy.optimize import minimize

    n = len(initial_centers)
    # Flatten variables into [x1, y1, r1, x2, y2, r2, ...]
    def objective(vars):
        return -np.sum(vars[2::3])  # Negative because we maximize

    def constraint(vars):
        # Bounds for center and radii
        bounds_list = []
        for i in range(n):
            x, y, r = vars[3*i], vars[3*i+1], vars[3*i+2]
            bounds_list.append(x - r)       # x - r >= 0
            bounds_list.append(1 - x - r)   # 1 - x - r >= 0
            bounds_list.append(y - r)       # y - r >= 0
            bounds_list.append(1 - y - r)   # 1 - y - r >= 0
            bounds_list.append(r)           # r >= 0
        # Circle separation
        for i in range(n):
            for j in range(i+1, n):
                dx = vars[3*i] - vars[3*j]
                dy = vars[3*i+1] - vars[3*j+1]
                r_i = vars[3*i+2]
                r_j = vars[3*j+2]
                dist_sq = dx*dx + dy*dy
                separation = dist_sq - (r_i + r_j)**2
                bounds_list.append(separation)  # distance^2 - (r_i + r_j)^2 >= 0
        return np.array(bounds_list)

    # initial guess
    initial_guess = np.zeros(3 * n)
    for i in range(n):
        initial_guess[3*i] = initial_centers[i, 0]
        initial_guess[3*i+1] = initial_centers[i, 1]
        initial_guess[3*i+2] = initial_radii[i]

    # bounds
    bounds = [(0, 1)] * (2*n) + [(0, 1)] * n  # x, y, r bounds

    # constraints
    cons = {'type': 'ineq', 'fun': constraint}

    result = minimize(objective, initial_guess, method='trust-constr',
                      bounds=bounds, constraints=cons, options={'maxiter': 200})

    if result.success:
        refined_centers = np.zeros((n, 2))
        refined_radii = np.zeros(n)
        for i in range(n):
            refined_centers[i, 0] = result.x[3*i]
            refined_centers[i, 1] = result.x[3*i+1]
            refined_radii[i] = result.x[3*i+2]
        return refined_centers, refined_radii
    else:
        return initial_centers, initial_radii

def run_packing():
    # Generate lattice
    h = 0.07
    centers, R_wall = generate_hex_lattice(h)

    # Build neighbor list
    max_dist = 0.4
    neighbour_list = build_neighbour_list(centers, max_dist)

    # Run branch and bound
    final_centers, final_radii = branch_and_bound(centers, R_wall, neighbour_list)

    # Local continuous refinement
    refined_centers, refined_radii = local_continuous_refinement(final_centers, final_radii)

    # Validation
    success, msg = validate_packing(refined_centers, refined_radii)
    if not success:
        raise ValueError("Validation failed after refinement: " + msg)

    # Final sum
    sum_radii = float(np.sum(refined_radii))

    return refined_centers, refined_radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
