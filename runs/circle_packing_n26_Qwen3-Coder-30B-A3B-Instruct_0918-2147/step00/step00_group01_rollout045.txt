import numpy as np
from scipy.spatial.distance import cdist

def penalty(z, alpha, beta):
    n = 26
    centers = z[:2*n].reshape((n, 2))
    radii = z[2*n:]
    
    # Wall penalties
    wall_pen = 0.0
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        dist_to_walls = [x, 1 - x, y, 1 - y]
        for d in dist_to_walls:
            if d < r:
                wall_pen += alpha * (r - d)
    
    # Overlap penalties
    overlap_pen = 0.0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                overlap_pen += beta * overlap
    
    return wall_pen + overlap_pen


def energy(z, lam, alpha, beta):
    n = 26
    sum_radii = np.sum(z[2*n:])
    return -sum_radii + lam * penalty(z, alpha, beta)


def propose_move(z, sigma_c, sigma_r, temp_scale):
    z_new = z.copy()
    n = 26
    centers = z_new[:2*n].reshape((n, 2))
    radii = z_new[2*n:]
    
    move_type = np.random.choice(['jitter', 'radius', 'swap'])
    
    if move_type == 'jitter':
        i = np.random.randint(n)
        dx = np.random.normal(0, sigma_c * temp_scale)
        dy = np.random.normal(0, sigma_c * temp_scale)
        centers[i, 0] += dx
        centers[i, 1] += dy
        centers[i, 0] = np.clip(centers[i, 0], 0, 1)
        centers[i, 1] = np.clip(centers[i, 1], 0, 1)
    
    elif move_type == 'radius':
        i = np.random.randint(n)
        dr = np.random.normal(0, sigma_r * temp_scale)
        radii[i] += dr
        radii[i] = np.maximum(radii[i], 0)
    
    else:  # swap
        i, j = np.random.choice(n, size=2, replace=False)
        radii[i], radii[j] = radii[j], radii[i]
    
    return z_new


def anneal():
    np.random.seed(42)
    n = 26
    alpha, beta = 10.0, 10.0  # penalty weights
    sigma_c = 0.03
    sigma_r = 0.02
    
    # Generate initial centers on a hexagonal grid
    spacing = 0.18
    x_coords = np.arange(0.05, 1.0, spacing)
    y_coords = np.arange(0.05, 1.0, spacing)
    centers_init = []
    for y in y_coords:
        for x in x_coords:
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers_init.append([x, y])
    
    # Pad to 26 points
    if len(centers_init) < n:
        centers_init.extend([[0.01, 0.01], [0.99, 0.99]], 
                            [(np.random.rand(), np.random.rand()) for _ in range(n - len(centers_init))])
    elif len(centers_init) > n:
        centers_init = centers_init[:n]
    
    centers = np.array(centers_init)
    radii = np.array([min(x, 1 - x, y, 1 - y) for x, y in centers])
    z = np.hstack((centers.flatten(), radii))
    
    # Annealing parameters
    T0 = 1.0
    c_T = 0.995
    lam_0 = 0.1
    c_lam = 1.03
    max_steps = 10000
    
    T = T0
    lam = lam_0
    best_z = z.copy()
    best_energy = energy(z, lam, alpha, beta)
    best_feasible = z.copy()
    best_feasible_energy = energy(z, lam, alpha, beta)
    
    i_step = 0
    while T > 1e-5 and i_step < max_steps:
        # propose new state
        z_prop = propose_move(z, sigma_c, sigma_r, np.sqrt(T / T0))
        delta_E = energy(z_prop, lam, alpha, beta) - energy(z, lam, alpha, beta)
        
        if delta_E <= 0 or np.random.rand() < np.exp(-delta_E / T):
            z = z_prop
        
        # update best
        current_energy = energy(z, lam, alpha, beta)
        if current_energy < best_energy:
            best_energy = current_energy
            best_z = z.copy()
        
        # check feasibility
        current_pen = penalty(z, alpha, beta)
        if current_pen < 1e-6:
            if current_energy < best_feasible_energy:
                best_feasible_energy = current_energy
                best_feasible = z.copy()
        
        T *= c_T
        lam *= c_lam
        i_step += 1
        
    return best_feasible


def project_and_ascent(centers, radii):
    n = 26
    eta = 5e-3
    max_iter = 500
    converged = False
    
    for _ in range(max_iter):
        # compute gradients (write manually)
        grad = np.zeros(2 * n + n)
        grad[2 * n:] = 1.0  # only radius gradient is 1.0
        grad[:2 * n] = 0.0  # center gradient is 0
        
        # take step
        centers_flat = centers.flatten()
        radii_flat = radii.copy()
        centers_flat += eta * grad[:2 * n]
        radii_flat += eta * grad[2 * n:]
        
        # Project to [0,1] for centers
        centers_flat = np.clip(centers_flat, 0, 1)
        centers = centers_flat.reshape((n, 2))
        
        # Project radii to wall limits
        for i in range(n):
            x, y = centers[i]
            r_max = min(x, 1 - x, y, 1 - y)
            radii_flat[i] = min(radii_flat[i], r_max)
            radii_flat[i] = np.maximum(radii_flat[i], 0)
        
        # resolve overlaps
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                r_sum = radii_flat[i] + radii_flat[j]
                if dist < r_sum:
                    overlap = r_sum - dist
                    radii_flat[i] -= overlap / 2
                    radii_flat[j] -= overlap / 2
                    # enforce non-negativity after overlap reduction
                    radii_flat[i] = np.maximum(radii_flat[i], 0)
                    radii_flat[j] = np.maximum(radii_flat[j], 0)
                    # re-projection to wall limits
                    x, y = centers[i]
                    r_max = min(x, 1 - x, y, 1 - y)
                    radii_flat[i] = min(radii_flat[i], r_max)
                    x, y = centers[j]
                    r_max = min(x, 1 - x, y, 1 - y)
                    radii_flat[j] = min(radii_flat[j], r_max)
        
        # Re-allocate radii to centers
        radii = radii_flat
        
        # Check convergence
        idx = (n + 1) if n > 1 else 1
        if idx >= n:
            break
        # Stop if improvement is small
        prev_sum = np.sum(radii)
        new_sum = np.sum(radii)
        if abs(new_sum - prev_sum) < 1e-7:
            break
            
        eta *= 0.95
        if eta < 1e-6:
            break
            
    return centers, radii
