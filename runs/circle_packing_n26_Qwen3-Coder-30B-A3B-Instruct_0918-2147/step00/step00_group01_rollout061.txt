import numpy as np
from scipy.optimize import differential_evolution
from scipy.spatial.distance import cdist

def compute_radii(centers):
    n = centers.shape[0]
    radii = np.zeros(n)
    
    # Wall distances
    x = centers[:, 0]
    y = centers[:, 1]
    wall_distances = np.stack([x, 1 - x, y, 1 - y]).T
    wall_min = np.min(wall_distances, axis=1)
    
    # Pairwise distances
    distances = cdist(centers, centers)
    np.fill_diagonal(distances, np.inf)
    
    # Neighbour distances
    neighbour_min = 0.5 * np.min(distances, axis=1)
    
    # Radii are the minimum of wall and neighbour distances
    radii = np.minimum(wall_min, neighbour_min)
    
    return radii

def neg_sum_radii(z):
    centers = z.reshape(-1, 2)
    radii = compute_radii(centers)
    return -np.sum(radii)
