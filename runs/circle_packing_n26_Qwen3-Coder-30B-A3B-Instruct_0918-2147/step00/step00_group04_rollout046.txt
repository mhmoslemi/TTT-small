import numpy as np
import cvxpy as cp
from scipy.spatial.distance import pdist, squareform

def build_socp_problem(n):
    r = cp.Variable(n, nonneg=True)
    # Create parameter for pairwise distances
    dist_params = [cp.Parameter((), value=0.0) for _ in range(n*(n-1)//2)]
    # Build the constraints
    constraints = []
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            constraints.append(r[i] + r[j] <= dist_params[idx])
            idx += 1
    # Boundary constraints
    # we will pass the center coordinates to this function so constraints can be defined
    prob = cp.Problem(cp.Maximize(cp.sum(r)), constraints)
    return prob, r, dist_params

def solve_radii(centres, prob, r, dist_params):
    n = centres.shape[0]
    # Compute all pairwise distances (these are constants in the SOCP)
    # Ensure there is no numerical issue with small numbers
    dists = squareform(pdist(centres, metric='euclidean'))
    # Update dist_params with latest values
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            dist_params[idx].value = dists[i, j]
            idx += 1
    # Solve
    prob.solve(solver=cp.ECOS, abstol=1e-10, reltol=1e-10)
    return r.value, [con.dual_value for con in prob.constraints]

def energy(centres, prob, r, dist_params):
    radii, _ = solve_radii(centres, prob, r, dist_params)
    return -np.sum(radii)

def random_move(state, rng):
    # state is 2n array [x0, y0, x1, y1, ..., x25, y25]
    new_state = state.copy()
    move_type = rng.choice([0, 1, 2, 3], p=[0.7, 0.15, 0.1, 0.05])
    if move_type == 0:  # Single-center jitter
        idx = rng.integers(0, 26)
        dx = rng.normal(0, 0.02)
        dy = rng.normal(0, 0.02)
        new_state[idx*2] = np.clip(new_state[idx*2] + dx, 0, 1)
        new_state[idx*2+1] = np.clip(new_state[idx*2+1] + dy, 0, 1)
    elif move_type == 1:  # Cluster jitter
        # select 3-5 centers
        n_select = rng.integers(3, 6)
        indices = rng.choice(26, size=n_select, replace=False)
        # Get indices for x and y coords
        idxs = indices * 2
        idxs_y = idxs + 1
        # compute center of mass
        cx = np.mean(new_state[idxs])
        cy = np.mean(new_state[idxs_y])
        # move cluster with correlated offsets (cos/sin a common angle)
        angle = rng.uniform(0, 2*np.pi)
        dx = rng.normal(0, 0.03)
        dy = rng.normal(0, 0.03)
        new_state[idxs] = np.clip(new_state[idxs] + dx, 0, 1)
        new_state[idxs_y] = np.clip(new_state[idxs_y] + dy, 0, 1)
    elif move_type == 2:  # Edge push
        idx = rng.integers(0, 26)
        x = new_state[idx*2]
        y = new_state[idx*2+1]
        # nudging near edges
        ax = np.sign(0.5 - x) * 0.01
        ay = np.sign(0.5 - y) * 0.01
        new_state[idx*2] = np.clip(x + ax, 0, 1)
        new_state[idx*2+1] = np.clip(y + ay, 0, 1)
    else:  # Swap ordering of two points (so they don't conflict in refiner)
        idx1, idx2 = rng.choice(26, size=2, replace=False)
        x1 = new_state[idx1*2]
        y1 = new_state[idx1*2+1]
        x2 = new_state[idx2*2]
        y2 = new_state[idx2*2+1]
        new_state[idx1*2], new_state[idx2*2] = x2, x1
        new_state[idx1*2+1], new_state[idx2*2+1] = y2, y1
    return new_state

def anneal_chain(seed, prob, r, dist_params):
    rng = np.random.default_rng(seed)
    # Start with a non-ordered configuration
    state = rng.random(52)  # x0,y0,...,x25,y25
    current_energy = energy(state.reshape((26, 2)), prob, r, dist_params)
    best_energy = current_energy
    best_state = state.copy()
    T = 0.05
    alpha = 0.995
    max_iter = 50000
    acc = 0
    for _ in range(max_iter):
        new_state = random_move(state, rng)
        new_energy = energy(new_state.reshape((26, 2)), prob, r, dist_params)
        delta = new_energy - current_energy
        if delta < 0 or rng.random() < np.exp(-delta / T):
            current_energy = new_energy
            state = new_state
            if current_energy < best_energy:
                best_energy = current_energy
                best_state = state.copy()
        T *= alpha
        if T < 1e-6:
            break
        if acc > 0 and acc % 1000 == 0:
            # some teardown here but the solver can handle wins...
            pass
        acc += 1
    return best_state.reshape((26, 2)), -best_energy

def local_refine(centres, radii, prob, r, dist_params):
    update_count = 0
    max_iter = 10
    for _ in range(max_iter):
        new_radii, duals = solve_radii(centres, prob, r, dist_params)
        # Find active constraints
        active_constraints = []
        idx = 0
        n = centres.shape[0]
        for i in range(n):
            for j in range(i+1, n):
                if duals[idx] > 1e-6:
                    active_constraints.append((i, j))
                idx += 1
        # If no active constraints, we are done
        if not active_constraints:
            break
        # Now create the system of equations to solve
        # We want to find a small displacement to centres such that
        # 1. Pairwise distance equals sum of radii
        # 2. Boundary constraints also hold
        # For now, only work on active pair constraints
        n_act = len(active_constraints)
        if n_act == 0:
            break
        A = np.zeros((n_act, 52))
        b = np.zeros(n_act)
        idx = 0
        for i, j in active_constraints:
            # lambda, solve for new centers: ||c_i + delta_i - (c_j + delta_j)|| = r_i + r_j
            # =>  ||(c_i - c_j) + (delta_i - delta_j)|| = r_i + r_j
            diff = centres[i] - centres[j]
            norm_diff = np.linalg.norm(diff)
            # Solve  a * delta_i + b * delta_j = c
            # currently hardcoding implementation to make a tractable update
            # Basic idea: shift away from each other.
            if norm_diff > 0:
                # Direct update toward border
                d = diff / norm_diff
                A[idx, 2*i] = d[0]
                A[idx, 2*i+1] = d[1]
                A[idx, 2*j] = -d[0]
                A[idx, 2*j+1] = -d[1]
                b[idx] = (radii[i] + radii[j]) - norm_diff
                idx += 1
        if idx == 0:
            break
        try:
            # Solve least squares update (get solution that satisfies all active equations exactly)
            # NOTE: A must be such that it defines suitable constraints.
            # Update centres
            updates = np.linalg.lstsq(A[:idx], b[:idx], rcond=None)[0]
            # apply it
            for k in range(26):
                centres[k, 0] += updates[2*k]
                centres[k, 1] += updates[2*k+1]
            centres[:, 0] = np.clip(centres[:, 0], 0, 1)
            centres[:, 1] = np.clip(centres[:, 1], 0, 1)
            # recompute
            old_radii = radii.copy()
            radii, _ = solve_radii(centres, prob, r, dist_params)
            if np.sum(radii) - np.sum(old_radii) < 1e-5:
                break
            update_count += 1
        except:
            break
    return centres, radii

def finalise(centres, prob, r, dist_params):
    # Final fine tune
    radii, _ = solve_radii(centres, prob, r, dist_params)
    # Ensure feasibility by exact solve
    precise_radii, _ = solve_radii(centres, prob, r, dist_params)
    # Clamp any tiny violation
    rad_bound = np.minimum(np.minimum(centres[:, 0], 1 - centres[:, 0]),
                           np.minimum(centres[:, 1], 1 - centres[:, 1]))
    clipped_radii = np.minimum(precise_radii, rad_bound)
    # Check for overlap
    n = centres.shape[0]
    for i in range(n):
        for j in range(i+1, n):
            dist = np.linalg.norm(centres[i] - centres[j])
            if dist - (clipped_radii[i] + clipped_radii[j]) < -1e-12:
                # Adjust radii
                max_sum = dist - 1e-12
                # redistribute
                scale = max_sum / (clipped_radii[i] + clipped_radii[j])
                clipped_radii[i] *= scale
                clipped_radii[j] *= scale
    return centres, clipped_radii

def run_packing():
    n = 26
    # Build the SOCP once
    prob, r, dist_params = build_socp_problem(n)
    # Try several independent chains with different seeds
    best_centers = None

    best_sum = -np.inf
    # Try 4 chains with different seeds
    seeds = [12345, 67890, 54321, 98765]
