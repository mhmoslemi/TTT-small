import numpy as np
from scipy.optimize import minimize
import scipy.sparse as sp

def init_hex_grid(n=26, r_init=0.04):
    # Generate hexagonal grid points and select first n inside unit square
    sqrt3 = np.sqrt(3)
    h = sqrt3 * r_init  # vertical spacing
    rows = []
    y = r_init
    row_count = 0
    while len(rows) < n:
        if row_count % 2 == 0:
            # Even rows: aligned
            x = r_init
        else:
            # Odd rows: offset
            x = 2 * r_init
        row = []
        while len(row) < n and x <= 1:
            row.append((x, y))
            x += 2 * r_init
        if row:
            rows.extend(row)
        y += h
        row_count += 1
    centers = np.array(rows[:n])
    radii = np.full(n, r_init)
    return centers, radii

def max_radii(centers, radii):
    # Maximize radii given fixed centers
    n = len(centers)
    old_radii = np.zeros_like(radii)
    max_iter = 100
    for _ in range(max_iter):
        if np.allclose(old_radii, radii, rtol=1e-8, atol=1e-8):
            break
        old_radii = radii.copy()
        # Wall limits
        wall = np.minimum(centers, 1 - centers).min(axis=1)
        # Pairwise distances
        D = np.sqrt(((centers[:,None,:] - centers[None,:,:])**2).sum(-1))
        # Neighbour limits
        neighbor = D - radii[:,None]
        np.fill_diagonal(neighbor, np.inf)
        neighbor = neighbor.min(axis=1)
        # New radii
        radii = np.minimum(wall, neighbor)
        radii = np.maximum(radii, 0)  # Ensure non-negative
    return radii

def hill_climb(centers, radii, iterations=5000, step=0.01):
    # Simple stochastic hill climbing
    n = len(centers)
    current_sum = float(radii.sum())
    for _ in range(iterations):
        i = np.random.randint(0, n)
        old_x, old_y = centers[i]
        old_r = radii[i]
        # Try small random move
        dx = np.random.uniform(-step, step)
        dy = np.random.uniform(-step, step)
        new_x = old_x + dx
        new_y = old_y + dy
        # Check bounds
        if new_x - old_r < 0 or new_x + old_r > 1 or new_y - old_r < 0 or new_y + old_r > 1:
            continue
        # Update center temporarily
        centers[i, 0] = new_x
        centers[i, 1] = new_y
        # Re-maximize radii
        new_radii = max_radii(centers, radii)
        new_sum = float(new_radii.sum())
        if new_sum > current_sum + 1e-12:
            # Accept
            radii[:] = new_radii
            current_sum = new_sum
        else:
            # Reject
            centers[i, 0] = old_x
            centers[i, 1] = old_y
    return centers, radii

def refine_with_scipy(centers, radii):
    # Optimization using scipy
    n = len(centers)
    # Variables: [x1, y1, ..., xn, yn, r1, ..., rn]
    x0 = np.concatenate([centers.flatten(), radii])
    bounds = []
    for i in range(n):
        bounds.append((0, 1))  # x_i
        bounds.append((0, 1))  # y_i
    for i in range(n):
        bounds.append((0, 0.5))  # r_i
    # Constraints
    def wall_constraints(vars):
        x = vars[:n]
        y = vars[n:2*n]
        r = vars[2*n:]
        # All of these should be <= 0
        result = []
        for i in range(n):
            result.append(r[i] - x[i])    # r <= x
            result.append(r[i] + x[i] - 1)  # r + x <= 1
            result.append(r[i] - y[i])    # r <= y
            result.append(r[i] + y[i] - 1)  # r + y <= 1
        return np.array(result)

    def overlap_constraints(vars):
        x = vars[:n]
        y = vars[n:2*n]
        r = vars[2*n:]
        dists = []
        for i in range(n):
            for j in range(i):
                diff = np.array([x[i] - x[j], y[i] - y[j]])
                dist = np.sqrt(np.sum(diff ** 2))
                # r_i + r_j - dist <= 0
                dists.append(r[i] + r[j] - dist)
        return np.array(dists)

    # Sparse Jacobians
    def wall_jac(vars):
        x = vars[:n]
        y = vars[n:2*n]
        r = vars[2*n:]
        # Derivative for each constraint wrt each variable
        data = []
        rows = []
        cols = []
        idx = 0
        for i in range(n):
            # 4 wall constraints for each circle
            data.extend([1.0, -1.0, 1.0, -1.0])
            rows.extend([idx, idx, idx+1, idx+1])
            cols.extend([n+i, i, n+i, i])
            idx += 2
        return sp.coo_matrix((data, (rows, cols)), shape=(4*n, 3*n))

    def overlap_jac(vars):
        x = vars[:n]
        y = vars[n:2*n]
        r = vars[2*n:]
        # For each pair (i,j), we have constraint r_i + r_j - dist <= 0
        # Its gradient is (df/dx_i, df/dy_i, df/dr_i, df/dx_j, df/dy_j, df/dr_j)
        # Where df/dr_i = 1, df/dr_j = 1
        # df/dx_i = (x_i - x_j)/dist, df/dy_i = (y_i - y_j)/dist
        # df/dx_j = (x_j - x_i)/dist, df/dy_j = (y_j - y_i)/dist
        data = []
        rows = []
        cols = []
        idx = 0
        for i in range(n):
            for j in range(i):
                diff = np.array([x[i] - x[j], y[i] - y[j]])
                dist = np.sqrt(np.sum(diff ** 2))
                if dist < 1e-12:
                    continue
                dx = -diff[0] / dist
                dy = -diff[1] / dist
                # corresponding to [x_i, y_i, r_i, x_j, y_j, r_j]
                data.extend([dx, dy, 1.0, -dx, -dy, 1.0])
                rows.extend([idx]*6)
                cols.extend([i, n+i, 2*n+i, j, n+j, 2*n+j])
                idx += 1
        return sp.coo_matrix((data, (rows, cols)), shape=(idx, 3*n))

    # Build constraints
    wall_cons = NonlinearConstraint(wall_constraints, -np.inf, 0, jac=wall_jac)
    overlap_cons = NonlinearConstraint(overlap_constraints, -np.inf, 0, jac=overlap_jac)
    # Minimze negative of sum
    def obj_func(vars):
        return -np.sum(vars[2*n:])
    
    # Call optimizer
    res = minimize(
        obj_func, x0, method='trust-constr',
        bounds=bounds, constraints=[wall_cons, overlap_cons],
        options={'maxiter': 200}
    )
    if not res.success:
        # If optimization failed, keep current solution
        return centers, radii
    else:
        new_centers = res.x[:2*n].reshape((n, 2))
        new_radii = res.x[2*n:]
        # Recompute to make sure it's still valid.
        # Final max radii step
        return new_centers, max_radii(new_centers, new_radii)

def run_packing():
    # Initial guess: hexagonal grid
    centers, radii = init_hex_grid(26, 0.04)
    # Max radii with fixed centers
    radii = max_radii(centers, radii)
    # Stochastic hill climbing
    centers, radii = hill_climb(centers, radii, 5000, 0.01)
    # Refine with scipy
    centers, radii = refine_with_scipy(centers, radii)
    # Final validation
    success, msg = validate_packing(centers, radii)
    if not success:
        raise ValueError(f"Packaging failed: {msg}")
    sum_r = float(radii.sum())
    return centers, radii, sum_r

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
