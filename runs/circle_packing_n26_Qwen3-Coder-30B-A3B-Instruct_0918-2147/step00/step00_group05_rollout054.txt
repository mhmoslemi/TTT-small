def build_hex_lattice(spacing):
    # Create a hexagonal lattice inside the unit square
    lattice = []
    y = spacing / 2
    row = 0
    while y < 1 - spacing / 2:
        x = spacing / 2 if row % 2 == 0 else spacing
        while x < 1 - spacing / 2:
            lattice.append([x, y])
            x += spacing
        y += spacing * sqrt(3) / 2
        row += 1
    return np.array(lattice)

        grad[:26] = -1.0
        return grad

    def cons(radii_and_centers):
        n = len(radii_and_centers)
        rs = radii_and_centers[:26]
        cs = radii_and_centers[26:].reshape(26, 2)

        # differentiable constraints:
        # 1. Wall constraints (inactive end points > r_i)
        wall_cons = []
        for i in range(26):
            x, y = cs[i]
            r = rs[i]
            wall_cons.extend([
                x - r,           # left
                1 - x - r,      # right
                y - r,          # bottom
                1 - y - r       # top
            ])
            
        # 2. Non-overlap constraints
        non_overlap_cons = []
        for i in range(26):
            for j in range(i + 1, 26):
                dx, dy = cs[i] - cs[j]
                dist_sq = dx**2 + dy**2
                sum_radii = rs[i] + rs[j]
                # penalty if <= 0, so we are minimizing when it's >= 0
                non_overlap_cons.append(dist_sq - sum_radii**2)

        return np.array(wall_cons + non_overlap_cons)

    init_vars = np.concatenate([radii, centers.flatten()])
    bounds = [(1e-7, 1 - 1e-7)] * 26 + [(0 + 1e-7, 1 - 1e-7)] * 52

    def penalty_const(con): return max(0, -con)

        'type': 'ineq',
        'fun': lambda v: cons(v),
        'jac': lambda v: np.zeros((len(cons(v)), len(v)))  # we are approximating simplicitly in bounds check
    }

    # Optimization
    result = minimize(
