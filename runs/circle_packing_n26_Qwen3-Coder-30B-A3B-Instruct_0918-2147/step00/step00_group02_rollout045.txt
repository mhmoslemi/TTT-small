import numpy as np
from scipy.optimize import minimize
from ortools.linear_solver import pywraplp

def generate_candidates(g, radius_levels):
    candidates = []
    grid_points = np.arange(0, 1 + g/2, g)
    for x in grid_points:
        for y in grid_points:
            for r in radius_levels:
                if (x - r >= 0 and x + r <= 1 and y - r >= 0 and y + r <= 1):
                    candidates.append((x, y, r))
    return candidates

def build_milp(candidates):
    solver = pywraplp.Solver.CreateSolver('SCIP')
    if not solver:
        raise RuntimeError("Could not create solver")

    n = len(candidates)
    z = [solver.IntVar(0, 1, f"z_{i}") for i in range(n)]
    
    # Grid point constraint
    grid_dict = {}
    for i, (x, y, r) in enumerate(candidates):
        grid_pos = (round(x/g), round(y/g))
        if grid_pos not in grid_dict:
            grid_dict[grid_pos] = []
        grid_dict[grid_pos].append(i)

    for indices in grid_dict.values():
        solver.Add(sum(z[i] for i in indices) <= 1)

    # Exactly 26 circles
    solver.Add(sum(z) == 26)

    # Non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            xi, yi, ri = candidates[i]
            xj, yj, rj = candidates[j]
            dx = solver.NumVar(0.0, 1.0, f"dx_{i}_{j}")
            dy = solver.NumVar(0.0, 1.0, f"dy_{i}_{j}")
            
            solver.Add(dx >= xi - xj)
            solver.Add(dx >= xj - xi)
            solver.Add(dy >= yi - yj)
            solver.Add(dy >= yj - yi)

            solver.Add(dx + dy >= ri + rj)
            solver.Add(dx - dy >= ri + rj)
            
            solver.Add(dx <= (1 + 1e-6) * (z[i] + z[j]))
            solver.Add(dy <= (1 + 1e-6) * (z[i] + z[j]))

    # Objective: maximize sum of radii
    solver.Maximize(sum(candidates[i][2] * z[i] for i in range(n)))
    return solver, z

def solve_milp(solver, z_vars):
    status = solver.Solve()
    if status != pywraplp.Solver.OPTIMAL:
        raise RuntimeError(f"MILP solver did not find optimal solution. Status: {status}")
    selected_indices = [i for i, var in enumerate(z_vars) if var.solution_value() > 0.5]
    return selected_indices

def extract_initial_solution(selected_indices, candidates):
    centers = np.array([candidates[i][:2] for i in selected_indices])
    radii = np.array([candidates[i][2] for i in selected_indices])
    return centers, radii

def shrink_initial(radii, delta=1e-5):
    return radii - delta

def pack_variables(centers, radii):
    return np.hstack([centers.flatten(), radii])

def unpack_solution(z):
    n = len(z) // 3
    centers = z[:2*n].reshape(-1, 2)
    radii = z[2*n:]
    return centers, radii

def define_objective(z):
    _, radii = unpack_solution(z)
    return -np.sum(radii)

def define_jacobian(z):
    _, radii = unpack_solution(z)
    jacobian = np.zeros_like(z)
    n = len(z) // 3
    jacobian[2*n:] = -1.0
    return jacobian

def define_constraints(z):
    centers, radii = unpack_solution(z)
    n = len(centers)
    constraints = []

    def boundary_constraint(z_vec):
        centers, radii = unpack_solution(z_vec)
        result = []
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            result.extend([x - r, 1 - x - r, y - r, 1 - y - r])
        return np.array(result)

    def boundary_jacobian(z_vec):
        centers, radii = unpack_solution(z_vec)
        n = len(centers)
        J = np.zeros((4*n, 3*n))
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            idx = i * 3
            J[i*4, idx] = 1.0
            J[i*4, idx+1] = 0.0
            J[i*4, idx+2] = -1.0  # dx
            J[i*4+1, idx] = -1.0
            J[i*4+1, idx+1] = 0.0
            J[i*4+1, idx+2] = -1.0  # d(1-x-r)
            J[i*4+2, idx] = 0.0
            J[i*4+2, idx+1] = 1.0
            J[i*4+2, idx+2] = -1.0  # dy
            J[i*4+3, idx] = 0.0
            J[i*4+3, idx+1] = -1.0
            J[i*4+3, idx+2] = -1.0  # d(1-y-r)
        return J

    constraints.append({
        'type': 'ineq',
        'fun': boundary_constraint,
        'jac': boundary_jacobian
    })

    def non_overlap_constraint(z_vec):
        centers, radii = unpack_solution(z_vec)
        result = []
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                overlap = (radii[i] + radii[j]) - dist
                result.append(-overlap)
        return np.array(result)

    def non_overlap_jacobian(z_vec):
        centers, radii = unpack_solution(z_vec)
        n = len(centers)
        result = []
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy + 1e-15)
                dxdd = dx/dist
                dydd = dy/dist
                result.append(np.array([
                    dxdd, dydd, 1.0,
                    -dxdd, -dydd, 1.0
                ]))
        return np.array(result)

    constraints.append({
        'type': 'ineq',
        'fun': non_overlap_constraint,
        'jac': non_overlap_jacobian
    })

    return constraints

def repair_solution(centers, radii):
    n = len(centers)
    radii = np.clip(radii, 0, None)
    for i in range(n):
        radii[i] = min(radii[i], centers[i, 0], 1-centers[i, 0], centers[i, 1], 1-centers[i, 1])
    for _ in range(5):
        changed = False
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                overlap = (radii[i] + radii[j]) - dist
                if overlap > 0:
                    frac = overlap / 2
                    radii[i] -= frac
                    radii[j] -= frac
                    changed = True
        if not changed:
            break
    radii = np.clip(radii, 0, None)
    return centers, radii

def run_packing():
    # Stage 1: Discrete global search
    g = 0.04  
    radius_levels = [0.02*i for i in range(1, 26)]  # 0.02 to 0.50
    
    candidates = generate_candidates(g, radius_levels)
    solver, z_vars = build_milp(candidates)
    selected_indices = solve_milp(solver, z_vars)
    centers0, radii0 = extract_initial_solution(selected_indices, candidates)
    radii0 = shrink_initial(radii0, 1e-5)
    
    # Stage 2: Continuous local optimization
    z0 = pack_variables(centers0, radii0)
    
    def obj_func(z):
        return define_objective(z)
    def jac_func(z):
        return define_jacobian(z)
    def cons_func(z):
        return define_constraints(z)

    bounds = [(0, 1)] * (len(z0))  # bounds for x, y, r 
    result = minimize(
        obj_func,
        z0,
        method='trust-constr',
        jac=jac_func,
        bounds=bounds,
        constraints=cons_func(),
        options={'disp': False, 'gtol': 1e-9, 'xtol': 1e-9}
    )
    
    centers, radii = unpack_solution(result.x)
    centers, radii = repair_solution(centers, radii)
    
    sum_radii = np.sum(radii)
    assert validate_packing(centers, radii)[0], "Final packing is invalid"
    
    return (centers, radii, sum_radii)
