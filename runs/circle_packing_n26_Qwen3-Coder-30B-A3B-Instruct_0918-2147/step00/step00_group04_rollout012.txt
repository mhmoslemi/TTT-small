import numpy as np
from scipy.optimize import minimize

NUM_CIRCLES = 26
NUM_RESTARTS = 5
HEX_SPACING = 0.2
INITIAL_RADIUS = 0.02
BOUNDARY_TOLERANCE = 1e-9

def hex_lattice_points(n_rows, spacing):
    """Generate hexagonal lattice points within unit square."""
    points = []
    y = 0.0
    row = 0
    while y <= 1.0 + 1e-9:
        x_offset = (row % 2) * spacing / 2.0
        x = x_offset
        while x <= 1.0 + 1e-9:
            points.append([x, y])
            x += spacing
        row += 1
        y += spacing * np.sqrt(3) / 2.0
    return np.array(points[:n_rows])

def initial_guess(seed):
    """Create initial guess with jitter."""
    np.random.seed(seed)
    points = hex_lattice_points(NUM_CIRCLES, HEX_SPACING)
    # Ensure we have exactly NUM_CIRCLES points
    if len(points) < NUM_CIRCLES:
        points = np.vstack([points, np.random.rand(NUM_CIRCLES - len(points), 2)])
    else:
        points = points[:NUM_CIRCLES]
    # Add jitter
    jitter = np.random.uniform(-0.02, 0.02, (NUM_CIRCLES, 2))
    points += jitter
    # Clip to [0,1]
    points = np.clip(points, 0, 1)
    # Initialize radii
    radii = np.full(NUM_CIRCLES, INITIAL_RADIUS)
    # Concatenate into a flat vector: [x0, y0, r0, x1, y1, r1, ...]
    vars0 = np.empty(NUM_CIRCLES * 3)
    vars0[0::3] = points[:, 0]  # x coordinates
    vars0[1::3] = points[:, 1]  # y coordinates
    vars0[2::3] = radii         # radii
    return vars0

def objective(vars):
    """Minimize negative sum of radii."""
    radii = vars[2::3]
    return -np.sum(radii)

def constraint_boundary(vars):
    """Ensure all circles are within square and have non-negative radii."""
    x_coords = vars[0::3]
    y_coords = vars[1::3]
    radii = vars[2::3]
    
    # Distance to boundaries
    left_dist = x_coords - radii
    right_dist = 1 - x_coords - radii
    bottom_dist = y_coords - radii
    top_dist = 1 - y_coords - radii
    
    # Return concatenated constraint values: [left, right, bottom, top] for each circle
    return np.concatenate([left_dist, right_dist, bottom_dist, top_dist])

def constraint_nonoverlap(vars):
    """Ensure no two circles overlap."""
    n = NUM_CIRCLES
    x_coords = vars[0::3]
    y_coords = vars[1::3]
    radii = vars[2::3]
    constraints = []

    # Compute all pairwise distances and differences
    for i in range(n):
        for j in range(i + 1, n):
            dx = x_coords[i] - x_coords[j]
            dy = y_coords[i] - y_coords[j]
            dist_sq = dx * dx + dy * dy
            dist = np.sqrt(dist_sq)
            r_sum = radii[i] + radii[j]
            # Non-overlap means: distance >= r_sum
            # So we want: dist - r_sum >= 0
            constraints.append(dist - r_sum)
    return np.array(constraints)

def postprocess(vars):
    """Perform final adjustments to satisfy strict constraints."""
    x_coords = vars[0::3].copy()
    y_coords = vars[1::3].copy()
    radii = vars[2::3].copy()

    # Ensure circles lie within boundaries and radii are non-negative
    for i in range(NUM_CIRCLES):
        # Clamp centers to square interior
        x_coords[i] = np.clip(x_coords[i], radii[i], 1.0 - radii[i])
        y_coords[i] = np.clip(y_coords[i], radii[i], 1.0 - radii[i])
        
        # Ensure radius is non-negative
        radii[i] = max(radii[i], 0.0)

    # Reconstruct flattened variable array
    new_vars = np.empty_like(vars)
    new_vars[0::3] = x_coords
    new_vars[1::3] = y_coords
    new_vars[2::3] = radii

    # Resolved overlaps
    for i in range(NUM_CIRCLES):
        for j in range(i + 1, NUM_CIRCLES):
            # Compute distance
            dx = new_vars[0::3][j] - new_vars[0::3][i]
            dy = new_vars[1::3][j] - new_vars[1::3][i]
            dist = np.sqrt(dx*dx + dy*dy)
            r_sum = new_vars[2::3][i] + new_vars[2::3][j]
            
            # Overlap
            if dist < r_sum - 1e-12:
                overlap = r_sum - dist
                # Reduce larger radius
                r1 = new_vars[2::3][i]
                r2 = new_vars[2::3][j]
                if r1 > r2:
                    new_vars[2::3][i] = r1 - overlap * 0.5
                    new_vars[2::3][j] = max(r2, 1e-10)
                else:
                    new_vars[2::3][i] = max(r1, 1e-10)
                    new_vars[2::3][j] = r2 - overlap * 0.5

    return new_vars

def run_packing():
    best_sum = -np.inf
    best_vars = None

    # Define global bounds
    bounds = []
    for i in range(NUM_CIRCLES):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    global_bounds = bounds

    # Define constraints
    global_constraints = [
        {'type': 'ineq', 'fun': constraint_boundary},
    ]

    for seed in range(NUM_RESTARTS):
        # Start with an initial configuration
        vars0 = initial_guess(seed)
        
        # Add overlap constraint function
        def constr_func(vars):
            return constraint_nonoverlap(vars)
        
        # Add the actual non-overlap constraint
        global_constraints.append({'type': 'ineq', 'fun': constr_func})

        # Optimize
        res = minimize(
            objective,
            vars0,
            method='SLSQP',
            bounds=global_bounds,
            constraints=global_constraints,
            options={'maxiter': 3000, 'ftol': 1e-7}
        )

        global_constraints.pop()  # Remove the last nonoverlap constraint for next iteration

        if res.success:
            vars_corr = postprocess(res.x)
            s = np.sum(vars_corr[2::3])
            if s > best_sum:
                best_sum = s
                best_vars = vars_corr.copy()

    # Construct centers and radii from best_vars
    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii = best_vars[2::3]
    return centers, radii, float(best_sum)
