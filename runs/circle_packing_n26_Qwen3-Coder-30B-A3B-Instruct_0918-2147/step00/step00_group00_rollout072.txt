import numpy as np
from scipy.optimize import minimize

def make_hex_lattice(n, margin=0.05):
    """Generate a hexagonal lattice of n points inside [margin, 1-margin]^2"""
    # Find the spacing that fits n circles in the unit square
    # Hexagonal lattice: dx = dy * sqrt(3)/2
    # We'll use a grid and then trim to exactly n points
    
    # Start with a rough estimate
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))
    
    # Calculate spacing to fit nicely
    dx = (1 - 2*margin) / cols
    dy = dx * np.sqrt(3) / 2
    
    # Generate grid points
    points = []
    for i in range(cols):
        for j in range(rows):
            x = margin + i * dx + (j % 2) * dx / 2
            y = margin + j * dy
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append([x, y])
    
    # Trim to exactly n points
    points = points[:n]
    
    # Center the pattern
    if points:
        center_x = sum(p[0] for p in points) / len(points)
        center_y = sum(p[1] for p in points) / len(points)
        shift_x = 0.5 - center_x
        shift_y = 0.5 - center_y
        points = [[p[0] + shift_x, p[1] + shift_y] for p in points]
    
    return np.array(points)

def objective(z):
    """Sum of radii (negative for minimization)"""
    n = len(z) // 3
    return -sum(z[2*i+2] for i in range(n))

def boundary_constraints(z):
    """Returns array of values for: x-r >= 0, x+r <= 1, y-r >= 0, y+r <= 1 for each circle"""
    n = len(z) // 3
    c = []
    for i in range(n):
        x, y, r = z[3*i], z[3*i+1], z[3*i+2]
        c.extend([x - r, 1 - x - r, y - r, 1 - y - r])
    return np.array(c)

def pairwise_constraints(z):
    """Returns array of separation constraints: (x_i-x_j)^2 + (y_i-y_j)^2 - (r_i+r_j)^2 >= 0"""
    n = len(z) // 3
    c = []
    for i in range(n):
        for j in range(i+1, n):
            x1, y1, r1 = z[3*i], z[3*i+1], z[3*i+2]
            x2, y2, r2 = z[3*j], z[3*j+1], z[3*j+2]
            dist_sq = (x1 - x2)**2 + (y1 - y2)**2
            sep_sq = (r1 + r2)**2
            c.append(dist_sq - sep_sq)
    return np.array(c)

def solve_nlp(initial_z):
    """Solve the NLP problem using SLSQP"""
    n = len(initial_z) // 3
    # Define bounds
    bounds = [(0, 1), (0, 1), (0, 0.5)] * n
    
    # Define constraints
    cons = [
        {'type': 'ineq', 'fun': boundary_constraints},
        {'type': 'ineq', 'fun': pairwise_constraints}
    ]
    
    result = minimize(
        fun=objective,
        x0=initial_z,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-9, 'maxiter': 1000}
    )
    
    return result.x

def nudge_centers(centers, radii, eps=0.001):
    """Move circles away from walls if they're too tight"""
    n = centers.shape[0]
    new_centers = centers.copy()
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Check if circle is near any boundary
        near_left = abs(x - r) < 1e-6
        near_right = abs(1 - x - r) < 1e-6
        near_bottom = abs(y - r) < 1e-6
        near_top = abs(1 - y - r) < 1e-6
        
        # Adjust if very tight
        if near_left:
            new_centers[i, 0] += eps
        elif near_right:
            new_centers[i, 0] -= eps
        if near_bottom:
            new_centers[i, 1] += eps
        elif near_top:
            new_centers[i, 1] -= eps
            
    return new_centers
