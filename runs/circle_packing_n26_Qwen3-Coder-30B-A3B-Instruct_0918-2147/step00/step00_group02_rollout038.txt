import numpy as np
from scipy.stats.qmc import Sobol

def initialize_seeds(n):
    sobol = Sobol(d=2, seed=42)
    samples = sobol.random(n)
    # Clip to interior to avoid boundary issues
    margin = 0.02
    return np.clip(margin + (1 - 2 * margin) * samples, margin, 1 - margin)

def compute_feasible_radii(centers):
    n = centers.shape[0]
    # Compute pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_matrix = np.sqrt(np.sum(diff**2, axis=2))
    # Set diagonal to a large number so we don't choose self as nearest
    np.fill_diagonal(dist_matrix, np.inf)
    # Distance to walls
    wall_dists = np.array([
        centers[:, 0],  # x distance to left
        1 - centers[:, 0],  # x distance to right
        centers[:, 1],  # y distance to bottom
        1 - centers[:, 1]  # y distance to top
    ]).T
    # Find minimum nearest neighbor distance for each
    nn_dists = np.min(dist_matrix, axis=1)
    # Compute minimal feasible radius
    radii = np.minimum(wall_dists.min(axis=1), 0.5 * nn_dists)
    return radii

def redistribute_radii(radii, slack, alpha):
    total_slack = np.sum(slack)
    if total_slack <= 1e-15:
        return radii
    # For all circles that are wall-limited
    wall_limited = slack > 1e-12
    # Redistribution delta
    delta = alpha * slack
    new_radii = radii + delta
    # Ensure no radius becomes negative due to redistribution
    # Total sum to decrease due to redistribution of wall limited to neighbor limited
    # But we do not want to decrease neighbor-limited radii below zero
    overage = np.sum(np.maximum(0, delta[wall_limited]) - np.maximum(0, delta[~wall_limited]))
    # If total of increases > decreases, scale the decreases to maintain sum
    if overage > 1e-12:
        # Decrease neighbor-limited radii proportionally to their current sizes
        neighbor_limited = ~wall_limited
        valid_neighbor_radii = radii[neighbor_limited]
        total_neighbor_radius = np.sum(valid_neighbor_radii)
        if total_neighbor_radius > 1e-12:
            scale_factor = min(1.0, overage / total_neighbor_radius)
            delta[neighbor_limited] = -scale_factor * valid_neighbor_radii
        else:
            delta[neighbor_limited] = -delta[neighbor_limited]  # just make a very small reduction, if no neighbor radii
    new_radii = radii + delta
    new_radii = np.maximum(0, new_radii)
    return new_radii

def push_away(centers, radii, beta, eps):
    n = centers.shape[0]
    new_centers = centers.copy()
    diff = new_centers[:, np.newaxis, :] - new_centers[np.newaxis, :, :]
    dist_matrix = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(n):
            if i == j:
                continue
            d = dist_matrix[i, j]
            if d < radii[i] + radii[j] + eps:
                delta = (radii[i] + radii[j] + eps - d)
                direction = diff[i, j] / (d + eps)
                new_centers[i] += beta * delta * direction
    # Project back into interior
    new_centers = np.clip(new_centers, eps, 1 - eps)
    return new_centers

def iterate(centers, radii, alpha, beta, max_iter):
    for _ in range(max_iter):
        # Compute slack for each circle
        # Find nearest neighbor distance
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dist_matrix = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dist_matrix, np.inf)
        nn_dists = np.min(dist_matrix, axis=1)
        # Compute distance to walls
        wall_dists = np.array([
            centers[:, 0],
            1 - centers[:, 0],
            centers[:, 1],
            1 - centers[:, 1]
        ]).T
        wall_min = np.min(wall_dists, axis=1)
        # Slack = 0.5 * nearest neighbor - current radius
        slack = 0.5 * nn_dists - radii
        # Check early termination
        slack_sum = np.sum(slack)
        if slack_sum < 1e-5:
            break
        # Redistribute
        radii = redistribute_radii(radii, slack, alpha)
        # Push away
        centers = push_away(centers, radii, beta, 1e-8)
        # Recompute feasible radii
        new_radii = compute_feasible_radii(centers)
        sum_difference = np.sum(new_radii) - np.sum(radii)
        if sum_difference < 1e-6:
            break
        radii = new_radii
    return centers, radii

def repair(centers, radii, eps_final):
    # Apply small shrink to all radii
    radii = np.maximum(0, radii - eps_final)
    # Clip centers
    centers = np.clip(centers, eps_final, 1 - eps_final)
    # Repair overlaps that could slip through
    n = centers.shape[0]
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_matrix = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i+1, n):
            d = dist_matrix[i, j]
            r_sum = radii[i] + radii[j]
            if d < r_sum:
                # Reduce the larger radius
                if radii[i] > radii[j]:
                    radii[i] -= (r_sum - d) / 2 + eps_final
                else:
                    radii[j] -= (r_sum - d) / 2 + eps_final
    radii = np.maximum(0, radii)
    return centers, radii

def run_packing():
    centers = initialize_seeds(26)
    radii = compute_feasible_radii(centers)
    centers, radii = iterate(centers, radii, alpha=0.4, beta=0.3, max_iter=100)
    centers, radii = repair(centers, radii, 5e-10)
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
