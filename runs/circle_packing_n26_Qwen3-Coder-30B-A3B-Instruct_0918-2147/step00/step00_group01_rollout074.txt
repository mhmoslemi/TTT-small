import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
from typing import Tuple, List

def generate_lattice(s: float) -> np.ndarray:
    """Generate a hexagonal lattice of candidate points inside unit square."""
    sqrt3 = np.sqrt(3)
    y_spacing = s * sqrt3 / 2
    x_spacing = s
    
    points = []
    row = 0
    y = s / 2
    while y < 1 - s / 2:
        x_start = s / 2 if row % 2 == 0 else s / 2 + x_spacing / 2
        x = x_start
        while x < 1 - s / 2:
            points.append([x, y])
            x += x_spacing
        y += y_spacing
        row += 1
    return np.array(points)

def build_milp(P: np.ndarray):
    """Build a MILP model for the circle packing problem."""
    K = P.shape[0]
    
    # Wall distances
    wL = P[:, 0]
    wR = 1 - P[:, 0]
    wB = P[:, 1]
    wT = 1 - P[:, 1]
    w = np.minimum.reduce([wL, wR, wB, wT])
    
    # Compute pairwise distances
    d = np.zeros((K, K))
    for i in range(K):
        for j in range(i+1, K):
            dist = np.linalg.norm(P[i] - P[j])
            d[i, j] = d[j, i] = dist
    
    # Binary selection variables
    z = cp.Variable(K, boolean=True)
    # Radii variables
    r = cp.Variable(K, nonneg=True)
    
    # Objective
    objective = cp.Maximize(cp.sum(r))
    
    # Constraints
    constraints = [
        cp.sum(z) == 26,
    ]
    
    for k in range(K):
        # Link radius to selection
        constraints.append(r[k] <= w[k] * z[k])
        constraints.append(r[k] >= 0)
    
    # Non-overlap constraints
    M = 1.0  # Big-M
    for k in range(K):
        for l in range(k + 1, K):
            # When either z is 0, constraint becomes redundant.
            # If both are 1, require d[k,l] >= r[k] + r[l]
            constraints.append(d[k, l] >= r[k] + r[l] - M * (2 - z[k] - z[l]))
    
    # Create a reusable function for solving the LP relaxation
    def solve_partial(z_fixed=None):
        if z_fixed is not None:
            # Fix selected variables
            old_z_val = z.value
            for i, val in enumerate(z_fixed):
                z[i].value = val
            # Solve
            problem = cp.Problem(objective, constraints)
            problem.solve(solver=cp.OSQP, warm_start=True, verbose=False, 
                          eps_abs=1e-9, eps_rel=1e-9)
            # Restore previous values
            if old_z_val is not None:
                for i, val in enumerate(old_z_val):
                    z[i].value = val
            else:
                for i in range(K):
                    z[i].value = None
        else:
            problem = cp.Problem(objective, constraints)
            problem.solve(solver=cp.OSQP, verbose=False, 
                          eps_abs=1e-9, eps_rel=1e-9)
        
        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            rh = r.value
            zh = z.value
            assert zh is not None
            return rh, zh
        else:
            return None, None
            
    return solve_partial, r, z, w, d

def branch_and_bound(milp_solver, node_limit: int = 20000):
    """
    Custom branch-and-bound to solve the MILP.
    Returns k_set (indices of selected centers) and r_values.
    """
    # Priority queue of nodes
    import heapq
    best_objective = -np.inf
    best_k_set = None
    best_r_values = None
    
    # Priority queue: (-bound, node_id, z_fixed, num_fixed)
    pq = []
    
    # Initial node with all variables free
    node_id = 0
    heapq.heappush(pq, (0.0, node_id, [], 0))
    node_id += 1
    nodes_processed = 0
    
    while pq and nodes_processed < node_limit:
        neg_bound, _, z_fixed, num_fixed = heapq.heappop(pq)
        bound = -neg_bound
        
        # Pruning: bound worse than current best
        if bound <= best_objective:
            continue
            
        # Solve the LP relaxation for this partial assignment
        r_vals, z_vals = milp_solver(z_fixed)
        if r_vals is None:
            continue
            
        objective_val = np.sum(r_vals)
        
        # Check if all z variables are integer
        if num_fixed == 26:
            # Check if feasible
            z_int = np.round(z_vals).astype(int)
            if np.sum(z_int) == 26 and np.all(z_int >= 0) and np.all(z_int <= 1):
                # Possibly integer solution
                if objective_val > best_objective:
                    best_objective = objective_val
                    best_k_set = np.where(z_vals > 0.5)[0]
                    best_r_values = r_vals
        else:
            # Branch on the variable that's closest to 0.5
            # This is a cheap substitute for a variable selection strategy.
            # In practice, you'd want to consider branching height.
            # But with a small problem size, this works well enough.
            z_non_int = []
            for i in range(len(z_vals)):
                if z_vals[i] > 1e-6 and z_vals[i] < 1 - 1e-6:
                    z_non_int.append(( -abs(z_vals[i] - 0.5), i))
            z_non_int.sort()
            if z_non_int:
                branch_var = z_non_int[0][1]
                # Branch variables
                # Left: z[branch_var] = 0
                z_left = z_fixed + [(branch_var, 0)]
                # Right: z[branch_var] = 1
                z_right = z_fixed + [(branch_var, 1)]
                
                # Try right first (keep bound)  
                # For right, we must set z[branch_var] = 1
                new_z_fixed = z_left  # setting the right branch
                # Note: solving from scratch isn't efficient here, thus brutal approximation.
                # In production this should reuse or intelligently cut.
                # As proxy, we'll check bounds at later involves.
                # For speed, we'll just directly compute.
                # Use full solve for an improved bound to guide.
                # Always solve for bound.
                z_l_set = [v for v, _ in z_left]
                sol_r_l, sol_z_l = milp_solver(z_left)
                if sol_z_l is not None:
                    bound_left = np.sum(sol_r_l)
                    if bound_left > best_objective:
                        heapq.heappush(pq, (-bound_left, node_id, z_left, num_fixed + 1))
                        node_id += 1
                        
                z_r_set = [v for v, _ in z_right]
                sol_r_r, sol_z_r = milp_solver(z_right)
                if sol_z_r is not None:
                    bound_right = np.sum(sol_r_r)
                    if bound_right > best_objective:
                        heapq.heappush(pq, (-bound_right, node_id, z_right, num_fixed + 1))
                        node_id += 1
                        
        nodes_processed += 1

    if best_k_set is None:
        # Bring it down to AT LEAST one good solution by solving the exact thing with no restriction on branching
    
        # Introduce a scam to guess a good solution:
        # Use a local search, fix 0/1 by hand if needed
        # This is a stop-gap to get something workable for refinement
        # Roughly 120 possibilities, empty chaining of search
        r_start, z_start = milp_solver([])
        if r_start is not None:
            z_start_int = np.round(z_start).astype(int)
            indices = np.where(z_start_int > 0)[0]
            if len(indices) > 26:
                # Filter to 26 largest radii by selection
                radii_sorted = [(r_start[i], i) for i in indices]
                radii_sorted.sort(reverse=True)
                selected = [idx for _, idx in radii_sorted[:26]]
                final_z = np.zeros_like(z_start)
                for idx in selected:
                    final_z[idx] = 1
                # Re-solve with all selected
                r_final, z_final = milp_solver(
                    [(i, 1) for i in selected])
                if r_final is not None:
                    best_objective = np.sum(r_final)
                    best_k_set = np.array(selected)
                    best_r_values = r_final

    return best_k_set, best_r_values


def refine_solution(centers: np.ndarray, radii: np.ndarray) -> Tuple[np.ndarray, np.ndarray, float]:
    """
    Perform local continuous refinement of centers/radii.
    """
    c0 = centers.copy()
    r0 = radii.copy()
    n = len(c0)
    x = np.concatenate([c0.reshape(-1), r0])
    
    # Objective (negative because we minimize)
    def obj(x_flat):
        c_flat = x_flat[:2*n]
        r_flat = x_flat[2*n:]
        return -np.sum(r_flat)
    
    # Constraints
    # 1. Wall constraints
    def wall_constraints(x_flat):
        c_flat = x_flat[:2*n]
        r_flat = x_flat[2*n:]
        cons = []
        for i in range(n):
            x_i, y_i = c_flat[2*i], c_flat[2*i+1]
            r_i = r_flat[i]
            cons.append(r_i - x_i)  # r <= x
            cons.append(r_i - (1 - x_i))  # r <= 1-x
            cons.append(r_i - y_i)  # r <= y
            cons.append(r_i - (1 - y_i))  # r <= 1-y
        return np.array(cons)
        
    # 2. Non-overlap constraints
    def overlap_constraints(x_flat):
        c_flat = x_flat[:2*n]
        r_flat = x_flat[2*n:]
        cons = []
        for i in range(n):
            for j in range(i+1, n):
                x_i, y_i = c_flat[2*i], c_flat[2*i+1]
                x_j, y_j = c_flat[2*j], c_flat[2*j+1]
                r_i, r_j = r_flat[i], r_flat[j]
                # (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
                dist_sq = (x_i - x_j)**2 + (y_i - y_j)**2
                cons.append(dist_sq - (r_i + r_j)**2)
        return np.array(cons)
    
    # Bounds
    bounds = []
    for i in range(2*n):
        bounds.append((0, 1))
    for i in range(n):
        bounds.append((0, 1))
    
    # Jacobian of constraints (can be implemented if needed)
    # For simplicity, use numerical Jacobians
    
    # Optimization
    result = minimize(obj, x, method='SLSQP', bounds=bounds, 
                      constraints=[
                        {'type': 'ineq', 'fun': lambda x: wall_constraints(x) + 1e-12},
                        {'type': 'ineq', 'fun': lambda x: overlap_constraints(x) + 1e-12}
                      ], 
                      options={'ftol': 1e-9, 'maxiter': 1000}, 
                      tol=1e-9, 
                      callback=None)
    
    # After finalize
    x_new = result.x
    centers_new = x_new[:2*n].reshape(n, 2)
    radii_new = x_new[2*n:]
    
    # Clip any small negatives
    radii_new = np.clip(radii_new, 0, np.inf)
    
    # Sum of radii
    total = np.sum(radii_new)
    
    # Validation: if it's broken, we let it fail loudly
    return centers_new, radii_new, total

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # 1. Generate lattice
    spacing = 0.13
    candidates = generate_lattice(spacing)
    
    # 2. Build MILP model
    milp_func, r_vars, z_vars, wall_dists, distances = build_milp(candidates)
    
    # 3. Solve with branch-and-bound
    indices, radii_values = branch_and_bound(milp_func, node_limit=20000)
    
    # 4. Refine solution
    selected_centers = candidates[indices]
    refined_centers, refined_radii, sum_radii = refine_solution(selected_centers, radii_values)
    
    return refined_centers, refined_radii, sum_radii
