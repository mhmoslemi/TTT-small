import numpy as np
from scipy.optimize import minimize

NUM_CIRCLES = 26
NUM_RESTARTS = 5

def hex_lattice_points(n_rows, spacing):
    """Generate hexagonal lattice points in [0,1] x [0,1]"""
    points = []
    for i in range(n_rows):
        y = i * spacing * np.sqrt(3) / 2
        if y > 1:
            break
        offset = (i % 2) * spacing / 2
        j = 0
        while True:
            x = offset + j * spacing
            if x > 1:
                break
            points.append([x, y])
            j += 1
    return np.array(points)

def initial_guess(seed):
    """Generate an initial guess with jitter"""
    np.random.seed(seed)
    spacing = 0.2
    points = hex_lattice_points(10, spacing)
    # Select 26 closest points to center
    center = np.array([0.5, 0.5])
    dists = np.sqrt(np.sum((points - center)**2, axis=1))
    indices = np.argsort(dists)[:NUM_CIRCLES]
    selected_points = points[indices]

    # Add jitter to centers
    jitter = np.random.uniform(-0.02, 0.02, (NUM_CIRCLES, 2))
    centers = selected_points + jitter
    # Ensure centers are within unit square
    centers = np.clip(centers, 0, 1)

    # Initial uniform small radii
    radii = np.full(NUM_CIRCLES, 0.02)

    # Flatten into 78 dimensional array: [x0, y0, r0, x1, y1, r1, ...]
    vars0 = np.empty(3 * NUM_CIRCLES)
    vars0[0::3] = centers[:, 0]
    vars0[1::3] = centers[:, 1]
    vars0[2::3] = radii
    return vars0

def objective(vars):
    """Minimize negative sum of radii"""
    return -np.sum(vars[2::3])

def constraint_boundary(vars):
    """Ensure each circle is within the unit square"""
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    # Distance from boundary minus radius
    margins = np.array([
        x - r,              # left
        1 - x - r,          # right
        y - r,              # bottom
        1 - y - r           # top
    ])
    return margins.flatten()

def constraint_nonoverlap(vars):
    """Ensure no overlap between circles"""
    n = NUM_CIRCLES
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Vectorized pairwise distances
    # Compute distance matrix using broadcasting
    diff_x = x[:, np.newaxis] - x[np.newaxis, :]
    diff_y = y[:, np.newaxis] - y[np.newaxis, :]
    dists = np.sqrt(diff_x**2 + diff_y**2)
    
    # Non-overlap margin matrix (positive = no overlap)
    overlaps = dists - (r[:, np.newaxis] + r[np.newaxis, :])
    # Only upper triangle (avoid doubles and self-overlaps)
    overlaps = np.triu(overlaps, k=1)
    # Return only positive margins (where circles don't overlap)
    return overlaps[overlaps > 0]

def postprocess(vars):
    """Refine final result to satisfy constraints tightly."""
    x = vars[0::3].copy()
    y = vars[1::3].copy()
    r = vars[2::3].copy()

    # Clamp centers to unit square ensuring circles are inside
    x = np.clip(x, r, 1 - r)
    y = np.clip(y, r, 1 - r)

    # Resolve overlaps between pairs
    for i in range(NUM_CIRCLES):
        for j in range(i+1, NUM_CIRCLES):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dist = np.sqrt(dx*dx + dy*dy)
            overlap = dist - (r[i] + r[j])
            if overlap < -1e-9:
                # Shrink larger circle
                if r[i] > r[j]:
                    r[i] += overlap / 2
                else:
                    r[j] += overlap / 2

    # Reconstruct vars
    vars = np.empty(3 * NUM_CIRCLES)
    vars[0::3] = x
    vars[1::3] = y
    vars[2::3] = r
    return vars

def run_packing():
    best_sum = -np.inf
    best_vars = None

    for seed in range(NUM_RESTARTS):
        vars0 = initial_guess(seed)
        # Create bounds
        bounds = [(0,1) for i in range(3*NUM_CIRCLES)]
        bounds[2::3] = [(0, 0.5) for i in range(NUM_CIRCLES)]  # radius <= 0.5

        # Constraints
        cons = [
            {'type': 'ineq', 'fun': constraint_boundary},
            {'type': 'ineq', 'fun': constraint_nonoverlap}
        ]

        res = minimize(
            objective,
            vars0,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-7, 'maxiter': 3000}
        )
        if not res.success:
            continue
        vars_corr = postprocess(res.x)
        s = np.sum(vars_corr[2::3])
        if s > best_sum:
            best_sum = s
            best_vars = vars_corr

    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii = best_vars[2::3]
    return centers, radii, float(best_sum)
