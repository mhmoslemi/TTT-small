import numpy as np
from scipy.optimize import minimize
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(0)
    
    def init_grid():
        s = 0.07
        centers = []
        for k in range(10):
            y = s/2 + k * s * math.sqrt(3)/2
            if y > 1 - s:
                break
            for m in range(10):
                if k % 2 == 0:
                    x = s/2 + m * s
                else:
                    x = s/2 + m * s + s/2
                if x > 1 - s:
                    break
                if x >= s and x <= 1 - s and y >= s and y <= 1 - s:
                    centers.append([x, y])
        centers = np.array(centers)
        # Add jitter
        centers += np.random.uniform(-0.001, 0.001, centers.shape)
        centers = centers[:26]
        radii = np.full(26, 0.001)
        return centers, radii
    
    def compute_forces(centers, radii, k_w=10.0, k_p=5.0):
        n = centers.shape[0]
        forces = np.zeros_like(centers)
        for i in range(n):
            # Wall forces
            x, y = centers[i]
            r = radii[i]
            dL = x - r
            dR = (1 - x) - r
            dB = y - r
            dT = (1 - y) - r
            
            if dL < 0:
                forces[i, 0] += k_w * (-dL) * (-1)
            if dR < 0:
                forces[i, 0] += k_w * (-dR) * 1
            if dB < 0:
                forces[i, 1] += k_w * (-dB) * (-1)
            if dT < 0:
                forces[i, 1] += k_w * (-dT) * 1
                
            # Pairwise forces
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = math.sqrt(dx*dx + dy*dy)
                gap = dist - (radii[i] + radii[j])
                if gap < 0:
                    f = k_p * (-gap) / (dist + 1e-12)
                    forces[i, 0] += f * dx
                    forces[i, 1] += f * dy
                    forces[j, 0] -= f * dx
                    forces[j, 1] -= f * dy
                    
        return forces
    
    def grow_radii(centers, radii, forces, dt=0.02, alpha=1e-3):
        n = centers.shape[0]
        dot_radii = np.zeros(n)
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            # Check if touching wall
            touch_wall = (x - r < 1e-12 or x + r > 1 - 1e-12 or
                          y - r < 1e-12 or y + r > 1 - 1e-12)
            # Check if any overlaps
            no_overlaps = True
            for j in range(n):
                if i != j:
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = math.sqrt(dx*dx + dy*dy)
                    if dist <= radii[i] + radii[j] - 1e-12:
                        no_overlaps = False
                        break
                        
            if not touch_wall and no_overlaps:
                dot_radii[i] = alpha
                
        return dot_radii
    
    def simulate(centers, radii, dt=0.02, gamma=1.0, max_iter=5000):
        velocities = np.zeros_like(centers)
        for _ in range(max_iter):
            forces = compute_forces(centers, radii)
            # Update velocity
            velocities = velocities + dt * (forces - gamma * velocities)
            # Update position
            centers = centers + dt * velocities
            # Clip positions
            centers[:, 0] = np.clip(centers[:, 0], 0, 1)
            centers[:, 1] = np.clip(centers[:, 1], 0, 1)
            # Grow radii
            dot_radii = grow_radii(centers, radii, forces, dt=dt)
            radii = radii + dt * dot_radii
            
            # Early termination: no radius growth, and forces low
            if np.all(dot_radii == 0) and np.linalg.norm(forces) < 1e-6:
                break
                
        return centers, radii
    
    def max_radii_fixed_centers(centers, radii):
        n = centers.shape[0]
        max_iter = 30
        for _ in range(max_iter):
            new_radii = np.zeros(n)
            converged = True
            for i in range(n):
                r = radii[i]
                x = centers[i, 0]
                y = centers[i, 1]
                # Wall constraints
                r_max = min(x, 1-x, y, 1-y)
                # Pairwise constraints
                for j in range(n):
                    if i != j:
                        dx = centers[i, 0] - centers[j, 0]
                        dy = centers[i, 1] - centers[j, 1]
                        dist = math.sqrt(dx*dx + dy*dy)
                        r_max = min(r_max, dist - radii[j])
                if r_max * 0.99 < new_radii[i]:
                    converged = False
                new_radii[i] = r_max
            radii = new_radii
            if converged:
                break
        return radii
    
    def objective_sum_radii(centers_flat, centers):
        n = centers.shape[0]
        centers = centers_flat.reshape(n, 2)
        # Compute max radii for fixed centers 
        radii = np.zeros(n)
        for i in range(n):
            r = 0.0
            x = centers[i, 0]
            y = centers[i, 1]
            r = min(x, 1-x, y, 1-y)
            for j in range(n):
                if i != j:
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = math.sqrt(dx*dx + dy*dy)
                    r = min(r, dist - radii[j])
            radii[i] = r
        # Return negative sum because we will minimize 
        return -radii.sum()
    
    # Initial setup
    centers, radii = init_grid()
    
    # Physics simulation
    centers, radii = simulate(centers, radii)
    
    # Run fixed-center max radii iteration
    radii = max_radii_fixed_centers(centers, radii)
    
    # Refine via optimization
    centers_flat = centers.flatten()
    result = minimize(objective_sum_radii, centers_flat, args=(centers), method='L-BFGS-B', options={'maxiter': 2000, 'gtol': 1e-9})
    
    centers = result.x.reshape(26, 2)
    radii = max_radii_fixed_centers(centers, radii) # Recompute max radii after optimization
    
    sum_radii = np.sum(radii)
    
    # Try valid packing
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise Exception(f"Packing not valid: {msg}")
        
    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
