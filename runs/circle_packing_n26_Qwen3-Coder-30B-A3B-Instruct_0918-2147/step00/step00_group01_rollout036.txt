import numpy as np
from scipy.spatial.distance import cdist

def penalty(z, alpha=10.0, beta=10.0):
    """Compute total penalty for a configuration z = [x0, y0, r0, ..., x25, y25, r25]"""
    n = 26
    centers = z.reshape(n, 3)[:, :2]
    radii = z.reshape(n, 3)[:, 2]
    
    # Wall penalties
    wall_distances = np.array([
        centers[:, 0],              # x_i
        1 - centers[:, 0],          # 1 - x_i
        centers[:, 1],              # y_i
        1 - centers[:, 1]           # 1 - y_i
    ])
    wall_violations = np.maximum(0, radii - wall_distances.T).sum(axis=1)
    
    # Overlap penalties  
    dist_matrix = cdist(centers, centers)
    r_matrix = radii.reshape(-1, 1)
    overlap_matrix = (r_matrix + r_matrix.T) - dist_matrix
    overlap_violations = np.maximum(0, -overlap_matrix).sum() / 2  # avoid double-counting
    
    return alpha * wall_violations.sum() + beta * overlap_violations

def energy(z, lam, alpha=10.0, beta=10.0):
    """Energy function = -sum(radii) + lam * penalty"""
    n = 26
    radii = z.reshape(n, 3)[:, 2]
    total_radius = radii.sum()
    return -total_radius + lam * penalty(z, alpha, beta)

def propose_move(z, sigma_c, sigma_r, temp):
    """Return a new z after a random valid move"""
    z_new = z.copy()
    n = 26
    # Gaussian standard deviations
    sig_c = sigma_c * np.sqrt(temp)
    sig_r = sigma_r * np.sqrt(temp)
    
    move_type = np.random.choice(3)
    
    if move_type == 0:  # Center jitter
        idx = np.random.randint(n)
        z_new[idx * 3] += np.random.normal(0, sig_c)
        z_new[idx * 3 + 1] += np.random.normal(0, sig_c)
        # clip
        z_new[idx * 3] = np.clip(z_new[idx * 3], 0, 1)
        z_new[idx * 3 + 1] = np.clip(z_new[idx * 3 + 1], 0, 1)
    elif move_type == 1:  # Radius perturb
        idx = np.random.randint(n)
        z_new[idx * 3 + 2] += np.random.normal(0, sig_r)
        # clip
        z_new[idx * 3 + 2] = max(0, z_new[idx * 3 + 2])
    else:  # Swap pair
        idx1, idx2 = np.random.choice(n, 2, replace=False)
        r1 = z_new[idx1 * 3 + 2]
        z_new[idx1 * 3 + 2] = z_new[idx2 * 3 + 2]
        z_new[idx2 * 3 + 2] = r1
        
    return z_new

def anneal():
    """Perform simulated annealing to get a good initial configuration"""
    np.random.seed(42)
    n = 26
    alpha, beta = 10.0, 10.0
    T0 = 1.0
    c_T = 0.995
    c_lambda = 1.03
    T_min = 1e-5
    steps = 10000
    
    # Initial configuration – hexagonal grid with wall radius
    # Generate a coarse hexagonal lattice
    spacing = 0.18
    x_coords = np.arange(0.1, 1.0, spacing)
    y_coords = np.arange(0.1, 1.0, spacing * np.sqrt(3) / 2)
    centers = [(x, y) for y in y_coords for x in x_coords if x < 1 and y < 1]
    
    # truncate to 26 if needed
    if len(centers) < 26:
        # sampling with small jitter
        while len(centers) < 26:
            x = np.random.uniform(0.05, 0.95)
            y = np.random.uniform(0.05, 0.95)
            centers.append((x, y))
    elif len(centers) > 26:
        centers = centers[:26]

    # Set initial radii
    z = np.zeros(n * 3)
    for i, (x, y) in enumerate(centers):
        z[i * 3] = x
        z[i * 3 + 1] = y
        r = min(x, 1 - x, y, 1 - y)
        z[i * 3 + 2] = r

    # Annealing dynamics
    T = T0
    lam = 0.1
    best_feasible = None
    best_feasible_energy = np.inf
    best_total_radius = 0.0

    # stats for progress tracking
    total_accepted = 0
    last_improved = 0
    
    for k in range(steps):
        if T < T_min:
            break
            
        z_prop = propose_move(z, 0.03, 0.02, T)
        en_new = energy(z_prop, lam, alpha, beta)
        en_old = energy(z, lam, alpha, beta)
        de = en_new - en_old

        if de <= 0 or np.random.rand() < np.exp(-de / T):
            z = z_prop
            total_accepted += 1
            if de <= 0 and penalty(z, alpha, beta) == 0.0:
                total_radius = z.reshape(n, 3)[:, 2].sum()
                if total_radius > best_total_radius:
                    best_total_radius = total_radius
                    best_feasible = z.copy()
                    last_improved = k
        
        # decay temperature & penalty weight
        T *= c_T
        lam *= c_lambda

        if k - last_improved >= 500:
            # hack: reinitialize a few circles if getting stuck
            idxes = np.random.choice(n, 3, replace=False)
            for i in idxes:
                x = np.random.uniform(0.1, 0.9)
                y = np.random.uniform(0.1, 0.9)
                z[i * 3] = x
                z[i * 3 + 1] = y
                z[i * 3 + 2] = min(x, 1 - x, y, 1 - y)

    # Finalize best feasible solution
    if best_feasible is None:
        # fallback to current best if no feasible solution was seen
        best_feasible = z.copy()

    return best_feasible

def project_and_ascent(centers, radii):
    """Refine using projected gradient ascent"""
    n = 26
    # combine into single 78-dim vector
    z = np.zeros(n * 3)
    z[0::3] = centers[:, 0]
    z[1::3] = centers[:, 1]
    z[2::3] = radii
    
    eta = 5e-3
    for epoch in range(500):
        z_before = z.copy()
        # Compute gradients of the objective: only radii
        grad = np.zeros_like(z)
        grad[2::3] = 1.0
        
        # Do projected gradient ascent (fake gradient step)
        z += eta * grad
        
        # Step 2: Projection: 
        # 1. clamp each radius to wall limits
        centers = z.reshape(n, 3)[:, :2]
        radii = z.reshape(n, 3)[:, 2]
        
        # Wall clamping
        max_radius = np.minimum(centers[:, 0], 1 - centers[:, 0], 
                               centers[:, 1], 1 - centers[:, 1])
        radii = np.clip(radii, 0, max_radius)
        
        # 2. Resolve overlaps:
        dist_matrix = cdist(centers, centers)
        r_matrix = radii.reshape(-1, 1)
        overlap_matrix = (r_matrix + r_matrix.T) - dist_matrix
        
        # repeat until no overlaps
        loop_count = 0
        while True:
            loop_count += 1
            if loop_count > 100:  # safety check
                break
            
            max_overlap = np.max(overlap_matrix)
            if max_overlap <= 1e-12:
                break
            for i in range(n):
                for j in range(i + 1, n):
                    if overlap_matrix[i, j] > 1e-12:
                        # reduce both radii
                        r_i, r_j = radii[i], radii[j]
                        o = overlap_matrix[i, j]
                        # keep in proportion to current size (or use equal split...)
                        total = r_i + r_j
                        if total > 1e-12:
                            r_i_new = max(0, r_i - o * r_i / total)
                            r_j_new = max(0, r_j - o * r_j / total)
                        else:
                            r_i_new = r_j_new = max(0, r_i - o / 2)
                            
                        radii[i], radii[j] = r_i_new, r_j_new
                        
                        # Also recompute for the next iteration
                        r_matrix = radii.reshape(-1, 1)
                        overlap_matrix = (r_matrix + r_matrix.T) - dist_matrix
                        
        # Update back to z
        z[2::3] = radii
        
        # test convergence
        delta = np.linalg.norm(z - z_before)
        if delta < 1e-7:
            break
        
        # decrease learning rate
        eta *= 0.95
    
    # Reshape back
    centers = z.reshape(n, 3)[:, :2]
    radii = z.reshape(n, 3)[:, 2]
    return centers, radii

def run_packing():
    """Main function to return centers, radii, sum_radii tuple"""
    z = anneal()
    
    # reshape
    n = 26
    centers = z.reshape(n, 3)[:, :2]
    radii = z.reshape(n, 3)[:, 2]
    
    # run refinement
    centers, radii = project_and_ascent(centers, radii)
    
    # final validation

    if not msg.startswith("ok"):
        # if overlap, scale down slightly
        scale_factor = 1.0 - 1e-7

        # re-validate one more time
        validate_packing(centers, radii)
    
    sum_radii = radii.sum()
    
    return centers, radii, sum_radii
