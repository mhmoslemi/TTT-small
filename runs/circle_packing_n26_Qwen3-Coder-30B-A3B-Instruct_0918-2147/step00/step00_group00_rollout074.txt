import numpy as np
from scipy.optimize import linprog

def run_packing():
    # === 1. Geometry-first layout ===
    # Seed: triangular lattice pattern with hexagonal symmetry
    
    # Generate triangular lattice points for 26 circles
    n_circles = 26
    spacing = 0.8 / np.ceil(np.sqrt(n_circles))
    sqrt3 = np.sqrt(3)
    
    points = []
    i = 0
    while len(points) < n_circles:
        for j in range(int(np.ceil(1.0 / spacing))):
            x = i * spacing + (j % 2) * spacing/2
            y = j * spacing * sqrt3 / 2
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append([x, y])
        i += 1
    
    # Take the first 26 points in row-major order
    centers = np.array(points[:n_circles])
    
    # add margin to ensure non-negative radius at boundary
    margin = 0.05
    centers[:, 0] = np.clip(centers[:, 0], margin, 1 - margin)
    centers[:, 1] = np.clip(centers[:, 1], margin, 1 - margin)
    
    # Initial small radii
    radii = np.full(n_circles, 0.01)
    
    # === 2. Helper functions ===
    def compute_distances(centers):
        n = centers.shape[0]
        d = np.zeros((n, n))
        for i in range(n):
            diff = centers[i] - centers
            d[i] = np.sqrt(np.sum(diff**2, axis=1))
        return d
    
    def build_radius_lp_constraints(centers, radii):
        n = centers.shape[0]
        # Wall constraints
        A1 = np.zeros((n*4, n))
        b1 = np.zeros(n*4)
        for i in range(n):
            A1[i, i] = -1
            b1[i] = -centers[i, 0]  # x <= 1 - r  ->  r <= 1 - x
            A1[i+n, i] = -1
            b1[i+n] = centers[i, 0]  # x >= r ->  r <= x
            A1[i+2*n, i] = -1
            b1[i+2*n] = -centers[i, 1]  # y <= 1 - r
            A1[i+3*n, i] = -1
            b1[i+3*n] = centers[i, 1]  # y >= r
        A_wall = A1
        b_wall = b1
        
        # Separation constraints
        dists = compute_distances(centers)
        A2 = np.zeros((n*(n-1)//2, n))
        b2 = np.zeros(n*(n-1)//2)
        idx = 0
        for i in range(n):
            for j in range(i+1, n):
                A2[idx, i] = -1
                A2[idx, j] = -1
                b2[idx] = -dists[i, j]
                idx += 1
        A_sep = A2
        b_sep = b2
        
        A = np.vstack([A_wall, A_sep])
        b = np.hstack([b_wall, b_sep])
        return A, b
    
    def solve_radius_lp(centers, radii):
        A, b = build_radius_lp_constraints(centers, radii)
        # solve max 1^T * r   s.t. A * r <= b, r >= 0
        res = linprog(c=-np.ones(len(radii)), A_ub=A, b_ub=b, method='highs')
        if res.success:
            r_sol = res.x.copy()
            r_sol = np.clip(r_sol, 0, 0.5)  # safe bound
            return r_sol
        else:
            # fallback: just return current radii
            return radii.copy()
    
    def build_center_lp_constraints(centers, radii):
        n = centers.shape[0]
        # wall bounds
        A1 = np.zeros((n*4, n*2))
        b1 = np.zeros(n*4)
        for i in range(n):
            A1[i, 2*i] = -1  # x
            b1[i] = -radii[i]
            A1[i+n, 2*i] = 1  # x
            b1[i+n] = 1 - radii[i]
            A1[i+2*n, 2*i+1] = -1  # y
            b1[i+2*n] = -radii[i]
            A1[i+3*n, 2*i+1] = 1  # y
            b1[i+3*n] = 1 - radii[i]
        
        # we'll build the non-overlap constraints with 8-gon approx
        A2 = np.zeros((0, n*2))
        b2 = np.zeros(0)
        
        # Direction vectors for 8-gon approximation: cos(π/8) ≈ 0.92388
        k = 8
        cos_pi8 = np.cos(np.pi / 8)
        vs = [np.array([np.cos(i * np.pi / 4), np.sin(i * np.pi / 4)]) for i in range(k)]
        
        for i in range(n):
            for j in range(i+1, n):
                # constraint for each direction vector v_l
                for v in vs:
                    # v ⋅ (c_i - c_j) ≤ - (r_i + r_j) * cos(π/k)
                    # rewrite to A*x <= b form:
                    row = np.zeros(n*2)
                    row[2*i] = v[0]
                    row[2*i+1] = v[1]
                    row[2*j] = -v[0]
                    row[2*j+1] = -v[1]
                    bval = - (radii[i] + radii[j]) * cos_pi8
                    A2 = np.vstack([A2, row])
                    b2 = np.hstack([b2, bval])
        
        A = np.vstack([A1, A2])
        b = np.hstack([b1, b2])
        return A, b
    
    def solve_center_lp(centers, radii):
        A, b = build_center_lp_constraints(centers, radii)
        
        # we'll maximize slack (sum of s_i): capture slack from boundaries.
        # For simplicity, we simplify by solving a modified LP
        # that tries to increase effective radius by moving centers
        
        # Manually built LP: maximize sum of distance from walls (promotion of slack)
        n = centers.shape[0]
        c = np.zeros(n * 2 + n)  # [x1,...,xn, y1,...,yn, s1,...,sn]
        c[-n:] = 1  # we maximize sum of s_i (slack variables)
        
        # Submatrix for slack constraints (we'll do these reasonably by hand)
        A_eq = np.zeros((0, 2*n + n))
        b_eq = np.zeros(0)
        
        # Approximate squared-distance from wall constraints in terms of slacks
        # s_i <= x_i - r_i
        # s_i <= 1 - x_i - r_i
        # s_i <= y_i - r_i
        # s_i <= 1 - y_i - r_i
        
        # Now, in terms of the indexing [x1,...,xn, y1,...,yn, s1,...,sn]
        # These are linear cross-bound constraints:
        # x_i - s_i >= r_i  ->   x_i - s_i - r_i >= 0
        # 1 - x_i - s_i >= r_i ->   1 - x_i - s_i - r_i >= 0
        # y_i - s_i >= r_i ->   y_i - s_i - r_i >= 0
        # 1 - y_i - s_i >= r_i ->    1 - y_i - s_i - r_i >= 0
        
        # Modify A to reflect:
        A_ineq = np.zeros((n*4, 2*n + n))
        b_ineq = np.zeros(n*4)
        for i in range(n):
            # x_i - s_i - r_i >= 0
            A_ineq[i, 2*i] = 1.0
            A_ineq[i, 2*n + i] = -1.0
            b_ineq[i] = radii[i]
            # 1 - x_i - s_i - r_i >= 0
            A_ineq[i + n, 2*i] = -1.0
            A_ineq[i + n, 2*n + i] = -1.0
            b_ineq[i + n] = -1 + radii[i]
            # y_i - s_i - r_i >= 0
            A_ineq[i + 2*n, 2*i+1] = 1.0
            A_ineq[i + 2*n, 2*n + i] = -1.0
            b_ineq[i + 2*n] = radii[i]
            # 1 - y_i - s_i - r_i >= 0
            A_ineq[i + 3*n, 2*i+1] = -1.0
            A_ineq[i + 3*n, 2*n + i] = -1.0
            b_ineq[i + 3*n] = -1 + radii[i]
        
        # Process the earlier subset of manipulated matrices
        A_full = np.vstack([A, A_ineq])
        b_full = np.hstack([b, b_ineq])
        
        # Linear program to maximize sum of slack
        res = linprog(c=c, A_ub=A_full, b_ub=b_full, method='highs')
        if not res.success:
            return centers.copy()
        xsol = res.x[:2*n].reshape((n, 2))
        # clip to valid positions
        xsol[:, 0] = np.clip(xsol[:, 0], radii, 1 - radii)
        xsol[:, 1] = np.clip(xsol[:, 1], radii, 1 - radii)
        return xsol
    
    def local_inflation(centers, radii, max_iter=5):
        for _ in range(max_iter):
            improved = False
            for i in range(len(centers)):
                min_dist = np.inf
                for j in range(len(centers)):
                    if i == j:
                        continue
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    d = np.sqrt(dx*dx + dy*dy)
                    min_dist = min(min_dist, d)
                
                # check wall distance
                wall_dist = min(centers[i, 0], 1 - centers[i, 0], 
                               centers[i, 1], 1 - centers[i, 1])
                gap = min(min_dist, wall_dist) - (radii[i] * 2)
                if gap > 0.0001:
                    # Inflation factor
                    incr = min(0.4 * gap, 0.05)
                    new_r = min(0.5, radii[i] + incr)
                    if new_r > radii[i]:
                        radii[i] = new_r
                        improved = True
            if not improved:
                break
        return radii
    
    # === 3. Alternating LP Loop ===
    prev_sum = 0
    max_iters = 30
    eps = 1e-6
    
    for i in range(max_iters):
        # Solve for optimal radii at current centers
        radii = solve_radius_lp(centers, radii)
        
        # Solve for new center positions
        centers = solve_center_lp(centers, radii)
        
        # Check improvement
        curr_sum = np.sum(radii)
        if abs(curr_sum - prev_sum) < eps:
            break
        prev_sum = curr_sum

    # === 4. Local Inflation ===
    radii = local_inflation(centers, radii)

    # === 5. Final validation ===
    success, msg = validate_packing(centers, radii)
    if not success:
        # FALSE NEGATIVE?
        # we try shrinking a little
        shrink_factor = 1e-7
        rem = 5
        while rem > 0 and not success:
            radii *= (1 - shrink_factor)
            success, msg = validate_packing(centers, radii)
            rem -= 1
        if not success:
            raise RuntimeError("Final packing validation failed after shrinks", msg)
    
    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
