import numpy as np
from scipy.optimize import minimize

def initial_layout():
    # Create a hexagonal grid with 5 rows and 5 columns (25 circles)
    # Then add one in the center
    rows, cols = 5, 5
    spacing = 0.18
    r0 = spacing / 2 - 0.01
    
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = (j + 0.5) * spacing
            y = (i + 0.5) * spacing
            # Adjust to keep inside bounds
            x = max(r0, min(1 - r0, x))
            y = max(r0, min(1 - r0, y))
            centers.append([x, y])
    
    # Add a center circle in the middle
    centers.append([0.5, 0.5])
    
    # Adjust for reasonable initial radii
    radii = [r0] * 26
    
    return np.array(centers), np.array(radii)

def pack_vector(centers, radii):
    z = np.empty(3 * len(centers))
    for i in range(len(centers)):
        z[3*i] = centers[i, 0]
        z[3*i+1] = centers[i, 1]
        z[3*i+2] = radii[i]
    return z

def unpack_vector(z):
    n = len(z) // 3
    centers = np.empty((n, 2))
    radii = np.empty(n)
    for i in range(n):
        centers[i, 0] = z[3*i]
        centers[i, 1] = z[3*i+1]
        radii[i] = z[3*i+2]
    return centers, radii

def objective(z):
    _, radii = unpack_vector(z)
    return -np.sum(radii)

def constraint_boundary(z):
    centers, radii = unpack_vector(z)
    n = len(centers)
    constraints = []
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # x - r >= 0
        constraints.append(x - r)
        # 1 - x - r >= 0
        constraints.append(1 - x - r)
        # y - r >= 0
        constraints.append(y - r)
        # 1 - y - r >= 0
        constraints.append(1 - y - r)
    return np.array(constraints)

def constraint_nonoverlap(z):
    centers, radii = unpack_vector(z)
    n = len(centers)
    constraints = []
    for i in range(n):
        for j in range(i+1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx*dx + dy*dy
            r_sum = radii[i] + radii[j]
            # Constraint: dist >= r_sum => dist_sq >= r_sum^2
            # equivalently: dist_sq - r_sum^2 >= 0
            constraints.append(dist_sq - r_sum*r_sum)
    return np.array(constraints)

def run_packing():
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    # Multi-start with 3 attempts
    for attempt in range(3):
        centers, radii = initial_layout()
        
        # Slightly jitter centers and radii
        if attempt > 0:
            perturb_factor = 0.02
            centers += np.random.uniform(-perturb_factor, perturb_factor, centers.shape)
            # Clip to [0,1]
            centers = np.clip(centers, 0, 1)
            # Slightly modify radii
            radii += np.random.uniform(-0.005, 0.005, radii.shape)
            radii = np.clip(radii, 0, 0.5)
        
        # Pack into vector
        z0 = pack_vector(centers, radii)
        
        # Set up bounds
        bounds = [(0, 1) for _ in range(26)] + [(0, 1) for _ in range(26)] + [(0, 0.5) for _ in range(26)]
        
        # Set up constraints
        cons = []
        # Boundary constraints
        cons.append({'type': 'ineq', 'fun': constraint_boundary})
        # Non-overlap constraints
        cons.append({'type': 'ineq', 'fun': constraint_nonoverlap})
        
        # Optimize
        res = minimize(objective, z0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'ftol': 1e-9, 'maxiter': 2000})
        
        if not res.success:
            continue
            
        # Unpack result
        centers_final, radii_final = unpack_vector(res.x)
        # Clip negative radii due to numerical tolerance
        radii_final = np.maximum(radii_final, 0)
        sum_radii = np.sum(radii_final)
        
        # Verify with validation function
        valid, msg = validate_packing(centers_final, radii_final)
        if not valid:
            # Try small shrink if it's within acceptable error
            if 'overlap' in msg and abs(sum_radii - best_sum) > 1e-10:
                # 1e-6 shrink factor is very small
                radii_final *= 0.999999
                valid, _ = validate_packing(centers_final, radii_final)
                
        if valid and sum_radii > best_sum:
            best_sum = sum_radii
            best_centers = centers_final.copy()
            best_radii = radii_final.copy()
    
    if best_centers is None:
        # If nothing worked, raise error
        raise RuntimeError("No valid packing found")
        
    return best_centers, best_radii, best_sum
