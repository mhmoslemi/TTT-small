import numpy as np
from scipy.optimize import minimize

def hex_lattice_points(n, d):
    """Generate at least n points in a hexagonal lattice."""
    rows = []
    y = 0.0
    row = 0
    while len(rows) < n:
        x_start = (row % 2) * d / 2.0
        x = x_start
        while x <= 1.0 + d:
            rows.append((x, y))
            x += d
        y += d * np.sqrt(3) / 2.0
        row += 1
    return np.array(rows[:n])

def initial_guess():
    """Generate initial packing and return flattened parameter vector."""
    d = 0.2  # initial spacing to get >26 points
    points = hex_lattice_points(30, d)
    # Center and trim to within bounds
    bbox_min = np.min(points, axis=0)
    bbox_max = np.max(points, axis=0)
    center = (bbox_min + bbox_max) / 2.0
    points = points - center + 0.5
    points = points[(points[:, 0] >= 0) & (points[:, 0] <= 1) &
                    (points[:, 1] >= 0) & (points[:, 1] <= 1)]
    # Trim to exactly 26
    points = points[:26]

    # Compute max radius allowed by nearest wall
    r_max = np.min([points[:, 0], 1 - points[:, 0], points[:, 1], 1 - points[:, 1]])
    # But also ensure it does not exceed the spacing
    r_wall = r_max
    r = np.full(26, r_wall / 2.0)

    # Flatten to [x0, y0, r0, x1, y1, r1, ...]
    vars = np.empty(26 * 3)
    vars[0::3] = points[:, 0]
    vars[1::3] = points[:, 1]
    vars[2::3] = r
    return vars

def objective(vars):
    """Minimize negative of sum of radii."""
    return -np.sum(vars[2::3])

def wall_constraints(vars):
    """Return constraint values (should be ≥ 0)."""
    n = len(vars) // 3
    constraints = []
    for i in range(n):
        x = vars[3*i]
        y = vars[3*i+1]
        r = vars[3*i+2]
        # x - r >= 0  -> r - x <= 0
        constraints.append(r - x)
        # x + r <= 1  -> x + r - 1 <= 0
        constraints.append(x + r - 1.0)
        # y - r >= 0  -> r - y <= 0
        constraints.append(r - y)
        # y + r <= 1  -> y + r - 1 <= 0
        constraints.append(y + r - 1.0)
    return np.array(constraints)

def nonoverlap_constraints(vars):
    """Return constraint values (should be ≥ 0)."""
    n = len(vars) // 3
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            x1, y1, r1 = vars[3*i], vars[3*i+1], vars[3*i+2]
            x2, y2, r2 = vars[3*j], vars[3*j+1], vars[3*j+2]
            dist_sq = (x1 - x2)**2 + (y1 - y2)**2
            sum_r_sq = (r1 + r2)**2
            constraints.append(dist_sq - sum_r_sq)
    return np.array(constraints)

def run_packing():
    """Run the packing algorithm."""
    vars0 = initial_guess()
    n = len(vars0) // 3
    
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0))   # x
        bounds.append((0.0, 1.0))   # y
        bounds.append((0.0, 0.5))   # r

    constraints = [
        {'type': 'ineq', 'fun': wall_constraints},
        {'type': 'ineq', 'fun': nonoverlap_constraints}
    ]
    
    result = minimize(
        objective,
        vars0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 2000, 'ftol': 1e-10},
        callback=None
    )

    if not result.success:
        raise RuntimeError(f"Optimization failed: {result.message}")

    centers = np.empty((26, 2))
    radii = np.empty(26)
    centers[:, 0] = result.x[0::3]
    centers[:, 1] = result.x[1::3]
    radii = result.x[2::3]
    
    # clip tiny negative values
    radii[radii < 1e-12] = 0.0
    sum_radii = np.sum(radii)
    
    # Validate final packing
    success, msg = validate_packing(centers, radii)
    if not success:
        raise AssertionError("Validation failed after optimization: " + msg)

    return centers, radii, sum_radii
