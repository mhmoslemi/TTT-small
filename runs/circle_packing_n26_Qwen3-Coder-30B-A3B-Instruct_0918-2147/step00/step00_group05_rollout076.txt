def build_hex_lattice(spacing=0.18):
    """Build a hexagonal lattice inside the unit square for 24 points."""
    # Generate points in a hexagonal pattern around a central area
    rows = ceil(1.0 / spacing)
    cols = ceil(1.0 / (spacing * sqrt(3) / 2))
    points = []
    for i in range(rows):
        offset = spacing * (i % 2) / 2
        for j in range(cols):
            x = offset + j * spacing
            y = i * spacing * sqrt(3) / 2
            if x <= 1.0 and y <= 1.0:
                points.append((x, y))
    # Take only up to 24 points near the center (to avoid violations from far edges)
    points = sorted(points, key=lambda p: (p[0]-0.5)**2 + (p[1]-0.5)**2)[:24]
    return np.array(points)

def is_valid_wall_mask(mask):
