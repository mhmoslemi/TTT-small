import numpy as np
from scipy.optimize import linprog, minimize
from math import sqrt

def build_hex_lattice(d):
    # Generate a hexagonal lattice of at least d points within [0,1]^2
    # Spacing based on circle packing
    spacing = 1.0 / (1.0 + sqrt(3) / 2.0)
    rows = int(1.0 / spacing) + 1
    cols = int(1.0 / (spacing * sqrt(3) / 2.0)) + 1
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = j * spacing * sqrt(3) / 2.0
            y = i * spacing + (j % 2) * spacing / 2.0
            if x <= 1 and y <= 1:
                centers.append([x, y])
    # Select first d points
    centers = centers[:d]
    return np.array(centers)

def lp_max_radii(centers):
    n = centers.shape[0]
    if n == 0:
        return np.array([]), 0.0

    # Build LP constraint matrix
    # Linear inequalities: A_ub @ radii <= b_ub
    # Starting with wall constraints:
    # r_i <= x_i,  r_i <= 1 - x_i,  r_i <= y_i,  r_i <= 1 - y_i
    A_ub = []
    b_ub = []

    for i in range(n):
        x, y = centers[i]
        # r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
        A_ub.append([1 if j == i else 0 for j in range(n)] + [0])
        b_ub.append(x)
        A_ub.append([1 if j == i else 0 for j in range(n)] + [0])
        b_ub.append(1 - x)
        A_ub.append([1 if j == i else 0 for j in range(n)] + [0])
        b_ub.append(y)
        A_ub.append([1 if j == i else 0 for j in range(n)] + [0])
        b_ub.append(1 - y)

    # Pairwise non-overlap constraints:
    # r_i + r_j <= dist_ij
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            distance = sqrt(dx**2 + dy**2)
            row = [1 if k == i else (1 if k == j else 0) for k in range(n)] + [-1]
            A_ub.append(row)
            b_ub.append(distance)

    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)

    # Objective: maximize sum of radii
    c = np.array([-1.0 for _ in range(n)]) + [0.0]  # Negated because linprog minimizes
    bounds = [(0, None) for _ in range(n)] + [(0, None)]  # All vars free? No, radii > 0

    # Use high accuracy solver
    sol = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'presolve': True, 'time_limit': 10.0})
    if sol.status != 0:
        # Fallback to small radii, shouldn't happen for valid inputs
        radii = np.zeros(n)
    else:
        radii = sol.x[:n]

    total_sum = -sol.fun  # Negate again to get actual sum
    return radii, total_sum

def perturb_centers(centers, sigma):
    # Add Gaussian noise to each c, then clamp to [0,1]
    perturbed = centers + np.random.normal(0, sigma, centers.shape)
    perturbed = np.clip(perturbed, 0, 1)
    return perturbed

def anneal_centers(initial_centers):
    np.random.seed(42)  # For reproducibility
    best_centers = initial_centers.copy()
    best_radii, best_sum = lp_max_radii(best_centers)
    T_schedule = [
        {'T': 0.02, 'sigma': 0.05, 'iterations': 2000},
        {'T': 0.01, 'sigma': 0.03, 'iterations': 4000},
        {'T': 0.005, 'sigma': 0.015, 'iterations': 3000},
        {'T': 0.002, 'sigma': 0.008, 'iterations': 2000}
    ]

    current_centers = initial_centers.copy()
    current_radii, current_sum = best_sum, best_sum

    for stage in T_schedule:
        T = stage['T']
        sigma = stage['sigma']
        iterations = stage['iterations']
        for _ in range(iterations):
            new_centers = perturb_centers(current_centers, sigma)
            new_radii, new_sum = lp_max_radii(new_centers)
            if new_sum > current_sum or np.random.rand() < np.exp((new_sum - current_sum) / T):
                current_centers = new_centers
                current_radii = new_radii
                current_sum = new_sum
            if current_sum > best_sum:
                best_centers = current_centers
                best_radii = current_radii
                best_sum = current_sum

    return best_centers, best_radii, best_sum
