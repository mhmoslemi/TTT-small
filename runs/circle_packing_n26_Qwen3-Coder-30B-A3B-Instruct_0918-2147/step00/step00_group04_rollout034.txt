```python
import numpy as np
from scipy.optimize import minimize

NUM_CIRCLES = 26
NUM_RESTARTS = 5
INITIAL_RADIUS = 0.01
BOUNDARY_MARGIN = 1e-9

def hex_lattice_points(n_rows, spacing):
    points = []
    for i in range(n_rows):
        y = i * spacing * np.sqrt(3) / 2
        if y > 1:
            break
        offset = (i % 2) * spacing / 2
        for j in range(int(1 / spacing) + 1):
            x = j * spacing + offset
            if x > 1:
                break
            points.append([x, y])
    return np.array(points)

def initial_guess(seed):
    np.random.seed(seed)
    lattice_points = hex_lattice_points(6, 0.15)
    selected_points = lattice_points[:NUM_CIRCLES]
    if len(selected_points) < NUM_CIRCLES:
        # Fill with random points
        padding = NUM_CIRCLES - len(selected_points)
        extra = np.random.rand(padding, 2)
        selected_points = np.vstack([selected_points, extra])
    
    # Add jitter to initial centers
    jitter = np.random.uniform(-0.02, 0.02, selected_points.shape)
    selected_points += jitter
    
    # Ensure points are within the unit square
    selected_points = np.clip(selected_points, 0, 1)
    
    # Initialize radii
    radii = np.full(NUM_CIRCLES, INITIAL_RADIUS)
    
    # Generate variables array (x, y, r) for all circles
    vars_init = []
    for i in range(NUM_CIRCLES):
        vars_init.extend([selected_points[i, 0], selected_points[i, 1], radii[i]])
    return np.array(vars_init, dtype=np.float64)

def objective(vars):
    # Return negative sum of radii to minimize
    return -np.sum(vars[2::3])

def constraint_boundary(vars):
    # Ensure circles stay inside unit square
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    boundaries = np.column_stack([
        x - r,
        1 - x - r,
        y - r,
        1 - y - r
    ])
    # Return only min boundary slack for each circle
    return np.min(boundaries, axis=1)

def constraint_nonoverlap(vars):
    # Ensure no two circles overlap
    n = NUM_CIRCLES
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    slack_list = []
    for i in range(n):
        for j in range(i+1, n):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dist_sq = dx*dx + dy*dy
            radii_sum = r[i] + r[j]
            slack = radii_sum - np.sqrt(dist_sq) + BOUNDARY_MARGIN
            slack_list.append(slack)
    return np.array(slack_list, dtype=np.float64)

def postprocess(vars):
    # Fix numerical issues after optimization
    x = vars[0::3].copy()
    y = vars[1::3].copy()
    r = vars[2::3].copy()
    
    # Ensure circles are within unit square
    for i in range(NUM_CIRCLES):
        x[i] = np.clip(x[i], r[i], 1 - r[i])
        y[i] = np.clip(y[i], r[i], 1 - r[i])
    
    # Resolve overlaps
    for i in range(NUM_CIRCLES):
        for j in range(i + 1, NUM_CIRCLES):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dist = np.sqrt(dx*dx + dy*dy)
            radii_sum = r[i] + r[j]
            if dist < radii_sum:
                overlap = radii_sum - dist
                # Reduce both radii by half
                r[i] -= overlap / 2
                r[j] -= overlap / 2
                if r[i] < 0 or r[j] < 0:
                    # Should not happen, but safety
                    r[i] = max(r[i], 0)
                    r[j] = max(r[j], 0)
    
    # Reconstruct vars after fixing
    new_vars = np.empty_like(vars)
    new_vars[0::3] = x
    new_vars[1::3] = y
    new_vars[2::3] = r
    return new_vars

def run_packing():
    best_sum = -np.inf
    best_vars = None
    
    # Create global constraints
    boundary_constraint = {
        'type': 'ineq',
        'fun': constraint_boundary
    }
    
    # Flatten the non-overlap constraints
    nonoverlap_constraint = {
        'type': 'ineq',
        'fun': constraint_nonoverlap
    }
    
    # Bounds for x, y, r
    bounds = []
    for i in range(NUM_CIRCLES):
        bounds.extend([
            (0, 1),  # x
            (0, 1),  # y
            (0, 0.5) # r (maximum plausible radius)
        ])
    
    for seed in range(NUM_RESTARTS):
        vars0 = initial_guess(seed)
        res = minimize(objective, vars0,
                     method='SLSQP',
                     bounds=bounds,
                     constraints=[boundary_constraint, nonoverlap_constraint],
                     options={'maxiter': 3000, 'ftol': 1e-7})
        
        if not res.success:
            continue
            
        vars_corr = postprocess(res.x)
        s = np.sum(vars_corr[2::3])
        if s > best_sum:
            best_sum = s
            best_vars = vars_corr
    
    # Reshape best variables
    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii = best_vars[2::3]
    return centers, radii, float(best_sum)
```
