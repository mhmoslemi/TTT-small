import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
from scipy.optimize import Bounds

def init_candidate_grid(radii_set, grid_spacing_factor=1.0):
    """
    Generate a dense lattice of candidate center positions for each radius.
    """
    candidates = []
    for r in radii_set:
        # Grid spacing is r * sqrt(2) to ensure non-overlap for same radii
        spacing = r * np.sqrt(2) * grid_spacing_factor
        x_coords = np.arange(r, 1 - r + spacing / 2, spacing)
        y_coords = np.arange(r, 1 - r + spacing / 2, spacing)
        for x in x_coords:
            for y in y_coords:
                candidates.append((x, y, r))
    return candidates

def build_overlap_matrix(candidates):
    """
    Build a binary matrix indicating which pairs of candidates would overlap.
    Returns an N x N boolean matrix.
    """
    N = len(candidates)
    overlap_matrix = np.zeros((N, N), dtype=bool)

    for i in range(N):
        cx, cy, r1 = candidates[i]
        for j in range(i + 1, N):
            cx2, cy2, r2 = candidates[j]
            dist = np.sqrt((cx - cx2)**2 + (cy - cy2)**2)
            if dist < r1 + r2:
                overlap_matrix[i, j] = True
                overlap_matrix[j, i] = True

    return overlap_matrix

def solve_discrete_bip(candidates, overlap_matrix, target_sum):
    """
    Use a greedy approach to get an initial feasible solution.
    We select the 26 largest available candidates that don't overlap.
    """
    # Sort by radius descending
    indexed_candidates = [(r, i) for i, (_, _, r) in enumerate(candidates)]
    indexed_candidates.sort(reverse=True)
    
    selected_indices = []
    sum_r = 0.0
    
    selected = np.zeros(len(candidates), dtype=bool)
    for radius, idx in indexed_candidates:
        if len(selected_indices) >= 26:
            break
        if selected[idx]:
            continue
        # Check if overlaps with already selected
        overlap = False
        for prev_idx in selected_indices:
            if overlap_matrix[prev_idx, idx]:
                overlap = True
                break
        if not overlap:
            selected_indices.append(idx)
            selected[idx] = True
            sum_r += radius

    if len(selected_indices) < 26:
        # Not enough; use a heuristic: add random valid ones until 26
        to_add = 26 - len(selected_indices)
        available = []
        for idx in range(len(candidates)):
            if not selected[idx]:
                overlap = False
                for prev_idx in selected_indices:
                    if overlap_matrix[prev_idx, idx]:
                        overlap = True
                        break
                if not overlap:
                    available.append(idx)
        if len(available) < to_add:
            raise ValueError("Could not find enough non-overlapping candidates")
        selected_indices.extend(available[:to_add])
        for idx in available[:to_add]:
            sum_r += candidates[idx][2]
            
    # Extract final centers and radii
    centers = np.array([(candidates[i][0], candidates[i][1]) for i in selected_indices])
    radii = np.array([candidates[i][2] for i in selected_indices])
    
    return centers, radii

def max_radii_fixed_centres(centers, radii_initial):
    """
    For fixed centers, compute the maximum possible radii such that:
    - They are within the unit square
    - They don't overlap each other
    """
    n = centers.shape[0]
    radii = radii_initial.copy()

    # For each circle, compute its maximum allowable radius
    for i in range(n):
        # Wall constraints
        r_max_wall = min(
            centers[i][0],
            1.0 - centers[i][0],
            centers[i][1],
            1.0 - centers[i][1]
        )
        
        # Overlap constraints with other circles
        r_max_overlap = np.inf
        
        for j in range(n):
            if i != j:
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                r_max_overlap = min(r_max_overlap, dist - radii[j])
                
        radii[i] = min(r_max_wall, r_max_overlap)
        
    return radii

def nlp_refine(centers, radii_initial):
    """
    Refine the solution with continuous optimization using trust-constr.
    """
    n = centers.shape[0]
    
    # Create bounds
    # All variables: x1, y1, ..., x26, y26, r1, ..., r26
    lb = np.zeros(2 * n + n)
    ub = np.ones(2 * n + n)
    
    # Bounds for x and y: [r, 1-r]  (we're clipping based on actual radii at runtime)
    for i in range(n):
        lb[i] = 0
        ub[i] = 1
        lb[n + i] = 0
        ub[n + i] = 1
    # Radii bounds: [0, 1]
    for i in range(2 * n, 2 * n + n):
        lb[i] = 0
        ub[i] = 1

    # Initial guess vector
    x0 = np.concatenate([centers.flatten(), radii_initial])
    
    def obj(x):
        # x is [x1,y1,...,x26,y26,r1,...,r26]
        radii = x[2*n:]
        return -np.sum(radii)  # we minimize the negative sum
    
    def jac_obj(x):
        radii = x[2*n:]
        J = np.zeros_like(x)
        J[2*n:] = -1.0
        return J
    
    # Constraints:
    constraints = []

    # Wall bounds
    def wall_constraint(x):
        # x = [x1,y1,...,x26,y26,r1,...,r26]
        res = []
        for i in range(n):
            x_i = x[i]
            y_i = x[n + i]
            r_i = x[2*n + i]
            res.extend([
                r_i - x_i,            # r_i - x_i <= 0  
                r_i - (1 - x_i),      # r_i + x_i - 1 <= 0
                r_i - y_i,            # r_i - y_i <= 0
                r_i - (1 - y_i)       # r_i + y_i - 1 <= 0
            ])
        return np.array(res)
    
    def jacobian_wall_constraint(x):
        J = np.zeros((4*n, 3*n))
        for i in range(n):
            # Derivative of each wall constraint w.r.t. x[i], y[i], r[i]
            J[4*i, i] = -1.0
            J[4*i+1, i] = 1.0
            J[4*i, n+i] = -1.0
            J[4*i+1, n+i] = 1.0
            J[4*i+2, 2*n+i] = -1.0
            J[4*i+3, 2*n+i] = -1.0
            J[4*i+3, n+i] = 1.0
        return J
    
    constraints.append({'type': 'ineq', 'fun': wall_constraint, 'jac': jacobian_wall_constraint})

    # Non-overlap constraints
    def overlap_constraint(x):
        res = []
        for i in range(n):
            for j in range(i+1, n):
                xi = x[i]
                yi = x[n+i]
                ri = x[2*n+i]
                xj = x[j]
                yj = x[n+j]
                rj = x[2*n+j]
                dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
                # dist >= ri + rj => ri + rj - dist <= 0
                res.append(ri + rj - dist)
        return np.array(res)

    def jacobian_overlap_constraint(x):
        n_constraints = n * (n-1) // 2
        J = np.zeros((n_constraints, 3*n))
        idx = 0
        for i in range(n):
            for j in range(i+1, n):
                xi = x[i]
                yi = x[n+i]
                ri = x[2*n+i]
                xj = x[j]
                yj = x[n+j]
                rj = x[2*n+j]
                dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
                if abs(dist) < 1e-12:
                    # Avoid numerical issues at exact point
                    # Just assume it's a small distance
                    dist = 1e-12
                
                # Derivative of ri + rj - dist w.r.t. variables
                # w.r.t. xi
                J[idx, i] = (xi - xj) / dist
                # w.r.t. yi
                J[idx, n+i] = (yi - yj) / dist
                # w.r.t. ri
                J[idx, 2*n+i] = 1.0
                # w.r.t. xj
                J[idx, j] = -(xi - xj) / dist
                # w.r.t. yj
                J[idx, n+j] = -(yi - yj) / dist
                # w.r.t. rj
                J[idx, 2*n+j] = 1.0
                
                idx += 1
        return J

    constraints.append({'type': 'ineq', 'fun': overlap_constraint,
                        'jac': jacobian_overlap_constraint})

    # Use trust-constr solver
    res = minimize(obj, x0, method='trust-constr', jac=jac_obj, constraints=constraints,
                   bounds=Bounds(lb, ub), options={'maxiter': 300, 'xtol': 1e-9, 'gtol': 1e-9})
    
    if not res.success:
        pass  # Can raise or just proceed with x of given bounds
        
    refined_centers = res.x[:2*n].reshape((-1, 2))
    refined_radii = res.x[2*n:]
    
    return refined_centers, refined_radii


def run_packing():
    # Discrete phase parameters
    radii_set = np.round(np.arange(0.025, 0.12, 0.01), 3)
    grid_spacing_factor = 1.0
    target_sum = 2.636

    # Generate candidate grid
    candidates = init_candidate_grid(radii_set, grid_spacing_factor)
    
    # Build overlap matrix
    overlap_matrix = build_overlap_matrix(candidates)
    
    # Get discrete solution (this may be an under-estimate)
    centers, radii = solve_discrete_bip(candidates, overlap_matrix, target_sum)
    
    # Run local refinement
    centers_refined, radii_refined = nlp_refine(centers, radii)
    
    # Final fix of radii for fixed centers
    final_radii = max_radii_fixed_centres(centers_refined, radii_refined)
    
    # Ensure all constraints are satisfied
    success, msg = validate_packing(centers_refined, final_radii)
    if not success:
        raise Exception(f"Validation failed: {msg}")
        
    return centers_refined, final_radii, np.sum(final_radii)

# Provided validator function here (copy from task description)
def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
