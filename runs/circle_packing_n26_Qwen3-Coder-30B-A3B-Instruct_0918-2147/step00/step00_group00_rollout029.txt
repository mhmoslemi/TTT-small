import numpy as np
import random

def make_hex_seed(n, margin):
    # Create a hexagonal pattern of n circles
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))
    spacing = 1.0 / (cols + 1)
    centers = []
    radii = []
    idx = 0
    for i in range(rows):
        for j in range(cols):
            if idx >= n:
                break
            x = (j + 1) * spacing
            y = (i + 1) * spacing
            if i % 2 == 1:
                x += spacing / 2
            centers.append([x, y])
            idx += 1
    centers = np.array(centers[:n])
    # Compute radii based on distance to nearest boundary
    radii = []
    for i in range(n):
        x, y = centers[i]
        r = min(x, 1 - x, y, 1 - y) * 0.7
        radii.append(max(r, 0.015))  # Minimum radius to avoid numerical issues
    radii = np.array(radii)
    return centers, radii

def max_feasible_radius(i, centers, radii):
    x, y = centers[i]
    r_current = radii[i]
    # Distance to walls
    d_left = x
    d_right = 1 - x
    d_bottom = y
    d_top = 1 - y
    d_walls = [d_left, d_right, d_bottom, d_top]
    # Distance to other circles
    d_circles = []
    for j in range(centers.shape[0]):
        if i != j:
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            d_circles.append(dist - radii[j])
    all_distances = d_walls + d_circles
    return min(all_distances)

def inflate_radii(centers, radii, alpha):
    old_sum = np.sum(radii)
    mean_radius = np.mean(radii)
    delta = alpha * mean_radius
    new_radii = radii + delta
    # For each circle, clamp to its maximum feasible value
    for i in range(centers.shape[0]):
        max_r = max_feasible_radius(i, centers, radii)
        new_radii[i] = min(new_radii[i], max_r)
    new_sum = np.sum(new_radii)
    return new_radii, (new_sum - old_sum)

def anneal_centers(centers, radii, temp0, gamma, sigma0, steps):
    current_sum = np.sum(radii)
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = current_sum
    temp = temp0
    sigma = sigma0
    last_improvement_step = 0
    global_shift_freq = 50
    for step in range(steps):
        # Occasionally perform global shift to escape deadlock
        if step > 0 and step % global_shift_freq == 0:
            shift = np.random.normal(0, 0.001, 2)
            centers += shift
            centers = np.clip(centers, 0, 1)
            # Recompute radii based on shifted centers
            new_radii = radii.copy()
            for i in range(centers.shape[0]):
                max_r = max_feasible_radius(i, centers, new_radii)
                new_radii[i] = min(new_radii[i], max_r)
            current_sum = np.sum(new_radii)
            if current_sum > best_sum:
                best_centers = centers.copy()
                best_radii = new_radii.copy()
                best_sum = current_sum
            continue

        # Pick a random circle to move
        i = random.randint(0, centers.shape[0] - 1)
        old_center = centers[i].copy()
        old_radius = radii[i]
        # Generate displacement
        displacement = np.random.normal(0, sigma, 2)
        new_center = old_center + displacement
        # Clamp to unit square
        new_center = np.clip(new_center, 0, 1)
        # Update center
        centers[i] = new_center
        # Compute new max feasible radius for this circle
        max_r = max_feasible_radius(i, centers, radii)
        new_radius = min(max_r, radii[i])  # It might shrink due to constraints
        # Accept or reject the move based on Metropolis criterion
        delta_sum = new_radius - old_radius
        if delta_sum > 0 or random.random() < np.exp(delta_sum / temp):
            radii[i] = new_radius
            # Check if this improves the total sum
            new_sum = np.sum(radii)
            if new_sum > best_sum:
                best_centers = centers.copy()
                best_radii = radii.copy()
                best_sum = new_sum
        else:
            # Revert move
            centers[i] = old_center
    return best_centers, best_radii, best_sum

def mixed_cycle(centers, radii, params):
    alpha_start = params['alpha']
    gamma = params['gamma']
    alpha = alpha_start
    patience = params['patience']
    sa_steps = params['sa_steps']
    sa_temp_start = params['sa_temp_start']
    sa_sigma_start = params['sa_sigma_start']
    min_radius = params['min_radius']
    stop_after_cycles = params['stop_after_cycles']
    converged = False
    no_improvement_cycles = 0
    last_total_sum = np.sum(radii)
    iteration = 0
    while not converged:
        # Phase B: inflate radii
        stagnated = 0
        while stagnated < patience:
            iteration += 1
            new_radii, delta_sum = inflate_radii(centers, radii, alpha)
            radii = new_radii
            current_sum = np.sum(radii)
            if delta_sum < 1e-6:
                stagnated += 1
            else:
                stagnated = 0
            if current_sum > last_total_sum:
                last_total_sum = current_sum
            else:
                pass  # Possibly update alpha or stop criterion
        # Scale down inflation factor
        alpha *= gamma
        # Phase C: SA the centers
        new_centers, new_radii, new_sum = anneal_centers(
            centers, radii, sa_temp_start, gamma, sa_sigma_start, sa_steps)
        centers = new_centers
        radii = new_radii
        if new_sum > last_total_sum:
            last_total_sum = new_sum
            no_improvement_cycles = 0
        else:
            no_improvement_cycles += 1
        # Check for convergence
        if no_improvement_cycles >= stop_after_cycles:
            break
    return centers, radii

def polish_radii(centers, radii, beta, passes):
    for _ in range(passes):
        updated = True
        iteration = 0
        while updated and iteration < 5:
            updated = False
            for i in range(centers.shape[0]):
                x, y = centers[i]
                r_current = radii[i]
                # Compute gaps
                g_wall = min(x, 1 - x, y, 1 - y)
                g_nei = float('inf')
                for j in range(centers.shape[0]):
                    if i != j:
                        dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                        gap = dist - (r_current + radii[j])
                        if gap < g_nei:
                            g_nei = gap
                g = min(g_wall, g_nei)
                if g > 0:
                    new_r = r_current + beta * g
                    # Compute max feasible r for i
                    max_r = max_feasible_radius(i, centers, radii)
                    new_r = min(new_r, max_r)
                    if new_r > r_current:
                        radii[i] = new_r
                        updated = True
                        break
            iteration += 1
        # Resolve overlaps if any
        resolved = 0
        for _ in range(3):
            for i in range(centers.shape[0]):
                for j in range(i + 1, centers.shape[0]):
                    dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                    overlap = (radii[i] + radii[j]) - dist
                    if overlap > 0:
                        # Reduce each by half the overlap
                        delta = overlap / 2
                        radii[i] = max(0, radii[i] - delta)
                        radii[j] = max(0, radii[j] - delta)
                        resolved = 1
            if not resolved:
                break
    return centers, radii

def run_packing():
    random.seed(42)
    np.random.seed(42)
    # Phase A: Initial setup
    margin = 0.04
    centers, radii = make_hex_seed(26, margin)
    # Phase B-C-D: Mixed inflation and repositioning
    params = {
        'alpha': 0.04,
        'gamma': 0.95,
        'patience': 8,
        'sa_steps': 200,
        'sa_temp_start': 0.02,
        'sa_sigma_start': 0.03,
        'min_radius': 0.015,
        'stop_after_cycles': 5
    }
    centers, radii = mixed_cycle(centers, radii, params)
    # Phase E: Final polishing
    centers, radii = polish_radii(centers, radii, 0.6, 3)
    # Final validation
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise RuntimeError("Packing failed validation: " + msg)
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
