import numpy as np
from scipy.spatial.distance import cdist

def penalty(z, alpha=10.0, beta=10.0):
    n = 26
    centers = z[:2*n].reshape(n, 2)
    radii = z[2*n:]
    
    # Wall penalties
    wall_pen = 0.0
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        wall_dist = np.array([x, 1-x, y, 1-y])
        violations = np.maximum(0.0, r - wall_dist)
        wall_pen += np.sum(violations)
    
    # Overlap penalties
    overlap_pen = 0.0
    for i in range(n):
        for j in range(i+1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            r_sum = radii[i] + radii[j]
            overlap = np.maximum(0.0, r_sum - dist)
            overlap_pen += overlap
    
    return alpha * wall_pen + beta * overlap_pen

def energy(z, lam, alpha=10.0, beta=10.0):
    sum_radii = np.sum(z[2*26:])
    return -sum_radii + lam * penalty(z, alpha, beta)

def propose_move(z, sigma_c, sigma_r, T):
    # Scale noise with temperature
    sigma_c = sigma_c * np.sqrt(T)
    sigma_r = sigma_r * np.sqrt(T)
    
    z_new = z.copy()
    n = 26
    centers = z_new[:2*n].reshape(n, 2)
    radii = z_new[2*n:]
    
    move_type = np.random.choice(['center', 'radius', 'swap'])
    
    if move_type == 'center':
        i = np.random.randint(n)
        centers[i] += np.random.normal(0, sigma_c, 2)
        centers[i] = np.clip(centers[i], 0, 1)
        
    elif move_type == 'radius':
        i = np.random.randint(n)
        radii[i] += np.random.normal(0, sigma_r)
        radii[i] = np.maximum(0, radii[i])
    
    else:  # swap
        i, j = np.random.choice(n, 2, replace=False)
        radii[i], radii[j] = radii[j], radii[i]
        
    return z_new

def anneal():
    n = 26
    # Initial hexagonal grid placement
    grid_size = 0.18
    x_coords = np.arange(grid_size, 1.0, grid_size)
    y_coords = np.arange(grid_size, 1.0, grid_size)
    
    # Create coarse hexagonal-like grid
    centers = []
    for i, x in enumerate(x_coords):
        for j, y in enumerate(y_coords):
            if len(centers) < n:
                centers.append([x, y])
            if len(centers) >= n:
                break
        if len(centers) >= n:
            break
    
    # If fewer than n centers, pad
    while len(centers) < n:
        centers.append([np.random.random(), np.random.random()])
        
    centers = np.array(centers[:n])
    
    # Assign initial radii (max infeasible)
    radii = np.minimum(centers[:, 0], 1 - centers[:, 0])
    radii = np.minimum(radii, centers[:, 1])
    radii = np.minimum(radii, 1 - centers[:, 1])
    
    z = np.concatenate([centers.flatten(), radii])
    
    # Initial state
    T0 = 1.0
    T = T0
    c_T = 0.995
    lam0 = 0.1
    lam = lam0
    c_lam = 1.03
    sigma_c = 0.03
    sigma_r = 0.02
    
    # Annealing loop
    best_z = z.copy()
    best_energy = energy(z, lam)
    best_feasible_z = z.copy()
    best_feasible_energy = energy(z, lam)
    best_feasible_sum = np.sum(radii)
    
    current_energy = energy(z, lam)
    iter_count = 0
    max_iterations = 10000
    
    while T > 1e-5 and iter_count < max_iterations:
        for _ in range(100):  # Num proposals per outer iteration
            z_prop = propose_move(z, sigma_c, sigma_r, T)
            e_prop = energy(z_prop, lam)
            delta_e = e_prop - current_energy
            
            if delta_e <= 0 or np.random.random() < np.exp(-delta_e / T):
                z = z_prop
                current_energy = e_prop
                
        # Update
        T *= c_T
        lam *= c_lam
        iter_count += 1
        
        # Update best feasible
        pen = penalty(z, 10.0, 10.0)
        if pen < 1e-8:  # Feasible
            sum_r = np.sum(z[2*n:])
            if sum_r > best_feasible_sum:
                best_feasible_sum = sum_r
                best_feasible_z = z.copy()
                best_feasible_energy = energy(z, lam)
                
        # Update global best
        if current_energy < best_energy:
            best_energy = current_energy
            best_z = z.copy()
    
    return best_feasible_z if penalty(best_feasible_z) < 1e-8 else best_z

def project_and_ascent(centers, radii):
    n = len(centers)
    
    # Copy input
    c = centers.copy()
    r = radii.copy()
    
    eta = 5e-3
    max_iter = 500
    eps = 1e-7
    
    # Reconstruct full vector
    z = np.concatenate([c.flatten(), r])
    
    for _ in range(max_iter):
        # Evaluate current state (likely infeasible)
        current_sum = np.sum(r)
        
        # Gradient step: direction to increase sum of radii
        grad = np.zeros_like(z)
        grad[2*n:] = 1.0  # Only radii are optimized
        
        # Step
        z_new = z + eta * grad
        
        # Project
        c_new = z_new[:2*n].reshape(n, 2)
        r_new = z_new[2*n:]
        
        # Clamp radii to wall limits
        for i in range(n):
            x, y = c_new[i]
            max_r = min(x, 1-x, y, 1-y)
            r_new[i] = max(0.0, min(r_new[i], max_r))
        
        # Resolve overlaps
        changed = True
        iterations = 0
        while changed and iterations < 50:
            changed = False
            pairwise_dists = cdist(c_new, c_new)
            for i in range(n):
                for j in range(i+1, n):
                    d = pairwise_dists[i, j]
                    r_sum = r_new[i] + r_new[j]
                    if d < r_sum:
                        # Shrink radii
                        overlap = r_sum - d
                        delta = overlap / 2.0
                        # Clamp to zero and adjust
                        r_new[i] = max(0.0, r_new[i] - delta)
                        r_new[j] = max(0.0, r_new[j] - delta)
                        changed = True
            iterations += 1
        
        # Re-apply wall clamping
        for i in range(n):
            x, y = c_new[i]
            max_r = min(x, 1-x, y, 1-y)
            r_new[i] = max(0.0, min(r_new[i], max_r))
        
        # Check if change in sum is small
        new_sum = np.sum(r_new)
        if abs(new_sum - current_sum) < eps:
            break
        
        # Update values
        z = np.concatenate([c_new.flatten(), r_new])
        c = c_new
        r = r_new
        eta *= 0.95
        
    return c, r
