import numpy as np
from scipy.spatial.distance import cdist

def penalty(z, alpha, beta):
    n = 26
    centers = z[:2*n].reshape(n, 2)
    radii = z[2*n:]

    # Wall penalties
    wall_dists = np.array([
        centers[:, 0],       # x
        1 - centers[:, 0],   # 1-x
        centers[:, 1],       # y
        1 - centers[:, 1]    # 1-y
    ])
    wall_violations = np.maximum(0, radii - wall_dists.T).sum(axis=1)
    wall_penalty = alpha * wall_violations.sum()

    # Overlap penalties
    distances = cdist(centers, centers)
    radii_matrix = np.expand_dims(radii, axis=1)
    sum_radii = radii_matrix + radii_matrix.T
    overlap_matrix = sum_radii - distances
    overlap_violations = np.maximum(0, -overlap_matrix)
    np.fill_diagonal(overlap_violations, 0)
    overlap_penalty = beta * overlap_violations.sum() / 2

    return wall_penalty + overlap_penalty

def energy(z, lam, alpha, beta):
    return -np.sum(z[2*26:]) + lam * penalty(z, alpha, beta)

def propose_move(z, sigma_c, sigma_r, T):
    n = 26
    centers = z[:2*n].reshape(n, 2)
    radii = z[2*n:]

    # Temperature-scaled noise
    sc = sigma_c * np.sqrt(T)
    sr = sigma_r * np.sqrt(T)

    move_type = np.random.choice([0, 1, 2])
    if move_type == 0:  # Center jitter
        i = np.random.randint(n)
        centers[i] += np.random.normal(0, sc, 2)
        centers[i] = np.clip(centers[i], 0, 1)
    elif move_type == 1:  # Radius perturb
        i = np.random.randint(n)
        radii[i] += np.random.normal(0, sr)
        radii[i] = np.maximum(0, radii[i])
    else:  # Swap pair
        i, j = np.random.choice(n, 2, replace=False)
        radii[i], radii[j] = radii[j], radii[i]

    # Reconstruct z
    z_new = np.zeros_like(z)
    z_new[:2*n] = centers.flatten()
    z_new[2*n:] = radii
    return z_new

def anneal():
    np.random.seed(0)
    n = 26
    sigma_c = 0.03
    sigma_r = 0.02
    alpha = 10.0
    beta = 10.0
    T0 = 1.0
    cT = 0.995
    cL = 1.03

    # Initial hexagonal grid
    spacing = 0.18
    rows = int(1.0 / spacing) + 1
    cols = int(1.0 / spacing) + 1
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = (i + 0.5 * (j % 2)) * spacing
            y = j * spacing
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers.append([x, y])
                if len(centers) >= n:
                    break
        if len(centers) >= n:
            break
    centers = np.array(centers[:n])

    # Initial radii from wall constraints
    radii = np.minimum(centers[:, 0], 1 - centers[:, 0])
    radii = np.minimum(radii, centers[:, 1])
    radii = np.minimum(radii, 1 - centers[:, 1])

    # Initial unpacked
    z = np.zeros(2*n + n)
    z[:2*n] = centers.flatten()
    z[2*n:] = radii

    # Annealing
    T = T0
    lam = 0.1
    current_energy = energy(z, lam, alpha, beta)
    best_feasible = z.copy()
    best_feasible_energy = current_energy
    accept_count = 0
    max_iter = 10000

    for iter in range(max_iter):
        z_prop = propose_move(z, sigma_c, sigma_r, T)
        prop_energy = energy(z_prop, lam, alpha, beta)
        de = prop_energy - current_energy
        if de <= 0 or np.random.rand() < np.exp(-de / T):
            z = z_prop
            current_energy = prop_energy
            accept_count += 1

        # Update penalty weight
        if iter % 100 == 0:
            lam *= cL
            if lam > 1000:
                lam = 1000

        # Update temperature
        T *= cT
        if T < 1e-5:
            break

        # Check for feasibility
        if penalty(z, alpha, beta) == 0 and prop_energy < best_feasible_energy:
            best_feasible = z.copy()
            best_feasible_energy = prop_energy

    return best_feasible[:2*n].reshape(n, 2), best_feasible[2*n:]

def project_and_ascent(centers, radii, max_iter=500, eta=5e-3, decay=0.95):
    n = centers.shape[0]
    eta = eta
    iter = 0
    while iter < max_iter:
        # Compute overlap and adjust
        distances = cdist(centers, centers)
        radii_matrix = np.expand_dims(radii, axis=1)
        sum_radii = radii_matrix + radii_matrix.T
        overlap_matrix = sum_radii - distances

        # Iterate over pairs to resolve overlaps
        resolved = False
        for i in range(n):
            for j in range(i+1, n):
                overlap = overlap_matrix[i, j]
                if overlap < 0:  # Overlap
                    r_add = -overlap / 2
                    new_radii = [radii[i], radii[j]]
                    new_radii[0] = max(0, new_radii[0] - r_add)
                    new_radii[1] = max(0, new_radii[1] - r_add)
                    radii[i], radii[j] = new_radii[0], new_radii[1]
                    resolved = True

        # Wall projections
        for i in range(n):
            r_max = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
            if radii[i] > r_max:
                radii[i] = r_max

        # Is it about to zigzag?
        if not resolved:
            break

        iter += 1
        eta *= decay

    return centers, radii

def run_packing():
    centers, radii = anneal()
    centers, radii = project_and_ascent(centers, radii)
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
