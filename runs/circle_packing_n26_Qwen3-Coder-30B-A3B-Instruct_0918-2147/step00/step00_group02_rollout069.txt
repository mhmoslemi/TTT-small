import numpy as np
from scipy.stats.qmc import Sobol

def initialize_seeds(n):
    sobol = Sobol(d=2, seed=42)
    sample = sobol.random(n)
    margin = 0.02
    return margin + (1 - 2 * margin) * sample

def compute_feasible_radii(centers):
    n = centers.shape[0]
    # Compute pairwise distances
    diff = centers[:, None, :] - centers[None, :, :]
    dist_matrix = np.sqrt(np.sum(diff**2, axis=2))
    # Set diagonals to infinity to exclude self-distances
    np.fill_diagonal(dist_matrix, np.inf)
    # Nearest neighbor distances
    nn = np.min(dist_matrix, axis=1)
    # Wall distances
    wall_distances = np.array([
        centers[:, 0],              # x left
        1 - centers[:, 0],          # x right
        centers[:, 1],              # y bottom
        1 - centers[:, 1]           # y top
    ]).T
    min_wall = np.min(wall_distances, axis=1)
    # Feasible radii are half the minimum of nearest neighbor distance and wall distance
    radii = np.minimum(0.5 * nn, min_wall)
    return radii

def redistribute_radii(radii, slack, alpha):
    # Total slack
    total_slack = np.sum(slack)
    # Increase wall-limited radii by alpha * slack
    delta = alpha * slack
    new_radii = radii + delta
    # Now reduce neighbour-limited radii proportionally to their current radii
    # First identify which are neighbour-limited (slack = 0)
    neighbour_limited = (slack < 1e-12)
    if np.any(neighbour_limited):
        # Compute reduction from neighbour-limited radii proportionally
        total_radii = np.sum(radii[neighbour_limited])
        if total_radii > 1e-12:
            reduction = alpha * total_slack
            weights = radii[neighbour_limited] / total_radii
            reductions = weights * reduction
            # Clamp reductions to prevent radii going negative
            clamped_reductions = np.minimum(reductions, radii[neighbour_limited])
            new_radii[neighbour_limited] -= clamped_reductions
    return new_radii

def push_away(centers, radii, beta, eps):
    n = centers.shape[0]
    new_centers = centers.copy()
    for i in range(n):
        ci = centers[i]
        ri = radii[i]
        for j in range(n):
            if i == j:
                continue
            cj = centers[j]
            rj = radii[j]
            diff = ci - cj
            dist = np.sqrt(np.sum(diff**2))
            if dist < ri + rj + eps:
                # Calculate overlap
                overlap = ri + rj + eps - dist
                unit_dir = diff / (dist + eps)
                delta = beta * overlap * unit_dir
                new_centers[i] = new_centers[i] + delta
    # Project back into interior
    new_centers = np.clip(new_centers, eps, 1 - eps)
    return new_centers

def iterate(centers, radii, alpha=0.4, beta=0.3, max_iter=200):
    for _ in range(max_iter):
        old_sum = np.sum(radii)
        # Compute slack
        diff = centers[:, None, :] - centers[None, :, :]
        dist_matrix = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dist_matrix, np.inf)
        nn = np.min(dist_matrix, axis=1)
        wall_distances = np.array([
            centers[:, 0],              # x left
            1 - centers[:, 0],          # x right
            centers[:, 1],              # y bottom
            1 - centers[:, 1]           # y top
        ]).T
        min_wall = np.min(wall_distances, axis=1)
        radii_feasible = np.minimum(0.5 * nn, min_wall)
        slack = 0.5 * nn - radii
        # Redistribution step
        radii = redistribute_radii(radii, slack, alpha)
        # Push away step
        centers = push_away(centers, radii, beta, 1e-8)
        # Re-evaluate radii with new centers
        radii = compute_feasible_radii(centers)
        new_sum = np.sum(radii)
        if new_sum - old_sum < 1e-5:
            break
    return centers, radii

def repair(centers, radii, eps_final=5e-10):
    # Shrink all radii
    radii = np.maximum(0, radii - eps_final)
    # Clamp centers
    centers = np.clip(centers, eps_final, 1 - eps_final)
    # Try single pass to resolve overlaps – not often triggered but defensive
    diff = centers[:, None, :] - centers[None, :, :]
    dist_matrix = np.sqrt(np.sum(diff**2, axis=2))
    overlap_mask = dist_matrix < (radii[:, None] + radii[None, :] + 1e-12)
    np.fill_diagonal(overlap_mask, False)
    for i in range(len(radii)):
        for j in range(i + 1, len(radii)):
            if overlap_mask[i, j]:
                overlap = radii[i] + radii[j] - dist_matrix[i, j]
                reduction = overlap / 2 + eps_final
                if radii[i] >= radii[j]:
                    radii[i] = max(0, radii[i] - reduction)
                else:
                    radii[j] = max(0, radii[j] - reduction)
    return centers, radii

def run_packing():
    n = 26
    centers = initialize_seeds(n)
    radii = compute_feasible_radii(centers)
    centers, radii = iterate(centers, radii, alpha=0.4, beta=0.3, max_iter=200)
    centers, radii = repair(centers, radii)
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
