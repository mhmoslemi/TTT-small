import numpy as np
from scipy.optimize import minimize

def make_hex_lattice(n, margin=0.05):
    """Generate a hexagonal lattice of n points within [margin, 1-margin]^2."""
    sqrt3 = np.sqrt(3)
    dx = 1.0 / (np.ceil(np.sqrt(n)) + 0.5)
    dy = dx * sqrt3 / 2.0

    points = []
    i = 0
    while len(points) < n:
        j = 0
        while len(points) < n:
            x = i * dx + (j % 2) * dx / 2
            y = j * dy
            if (margin <= x <= 1 - margin) and (margin <= y <= 1 - margin):
                points.append([x, y])
            j += 1
            if j > 30:
                break  # Safety to avoid infinite loops if n too large
        i += 1
        if i > 30:
            break
    return np.array(points[:n])

def objective(z):
    """Negate sum of radii for minimization."""
    return -np.sum(z[2::3])

def boundary_constraints(z):
    """Compute the boundary constraint residuals.
    Each circle i has four constraints:
      x_i - r_i >= 0
      x_i + r_i <= 1
      y_i - r_i >= 0
      y_i + r_i <= 1
    Return a flat array of residuals."""
    n = len(z) // 3
    residuals = np.empty(4 * n)
    # Extract centers and radii
    x = z[0::3]
    y = z[1::3]
    r = z[2::3]
    
    residuals[0::4] = x - r  # x - r >= 0      =>   -(x - r) <= 0
    residuals[1::4] = 1 - (x + r)  # x + r <= 1
    residuals[2::4] = y - r  # y - r >= 0
    residuals[3::4] = 1 - (y + r)  # y + r <= 1
    return residuals

def pairwise_constraints(z):
    """Compute pairwise non-overlap residuals.
    For each pair (i, j) with i < j:
        (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    Returns array of residuals."""
    n = len(z) // 3
    count = n * (n - 1) // 2
    residuals = np.empty(count)
    x = z[0::3]
    y = z[1::3]
    r = z[2::3]
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            sum_r = r[i] + r[j]
            residuals[idx] = dx * dx + dy * dy - sum_r * sum_r
            idx += 1
    return residuals

def solve_nlp(initial_z):
    """Solve the NLP."""
    n = len(initial_z) // 3
    cons = [
        {'type': 'ineq', 'fun': boundary_constraints},
        {'type': 'ineq', 'fun': pairwise_constraints}
    ]

    # Variable bounds
    bounds = []
    for _ in range(n):
        bounds.extend([(0, 1), (0, 1), (0, 0.5)])  # x, y, r

    result = minimize(
        objective,
        initial_z,
        method='SLSQP',
        constraints=cons,
        bounds=bounds,
        options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False}
    )
    return result.x

def nudge_centers(centers, radii):
    """Nudge circles near the wall to improve local packing."""
    nudged = centers.copy()
    # Find tight constraints (circle within 1e-6 of a wall)
    for i in range(len(centers)):
        x, y = centers[i]
        r = radii[i]
        dx_min = x - r
        dx_max = 1 - (x + r)
        dy_min = y - r
        dy_max = 1 - (y + r)
        MIN_DIST = 1e-6
        if dx_min < MIN_DIST or dx_max < MIN_DIST or dy_min < MIN_DIST or dy_max < MIN_DIST:
            # Nudge normal to the wall
            delta_x = 0.; delta_y = 0.
            if dx_min < dx_max and dx_min < MIN_DIST:
                delta_x = 0.001  # away from left wall
            elif dx_max < MIN_DIST:
                delta_x = -0.001  # away from right wall
            elif dy_min < dy_max and dy_min < MIN_DIST:
                delta_y = 0.001  # away from bottom wall
            elif dy_max < MIN_DIST:
                delta_y = -0.001  # away from top wall
            nudged[i, 0] += delta_x
            nudged[i, 1] += delta_y
    return nudged

    radii = np.full(26, 0.02)
    # Pack initial guess
    initial_z = np.empty(3 * 26)

    for _ in range(5):
        z = solve_nlp(initial_z)
        new_centers = z[0::3].reshape((26, 1))
        new_centers = np.hstack((new_centers, z[1::3].reshape((26, 1))))
        new_radii = z[2::3]
        new_sum = np.sum(new_radii)
        if new_sum - old_sum < 1e-5:

        old_sum = new_sum
