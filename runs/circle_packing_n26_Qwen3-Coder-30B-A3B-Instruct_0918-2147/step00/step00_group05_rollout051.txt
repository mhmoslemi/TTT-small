import numpy as np

def hex_lattice(d):
    """Generate hexagonal lattice points within [0,1]x[0,1]"""
    sqrt3 = np.sqrt(3)
    dy = sqrt3 * d / 2
    rows = []
    y = 0
    row_idx = 0
    while y <= 1:
        row = []
        x = 0
        x_offset = d / 2 if row_idx % 2 == 1 else 0
        while x <= 1:
            row.append([x + x_offset, y])
            x += d
        rows.extend(row)
        y += dy
        row_idx += 1
    return np.array(rows)

def add_border_circles(lattice_points, d):
    """Add 6 border circles to the hexagonal core"""
    r = d / 4
    border_points = [
        [r, r], [1 - r, r], [r, 1 - r], [1 - r, 1 - r],
        [0.5, r], [0.5, 1 - r]
    ]
    return np.vstack([lattice_points, border_points])

def initial_radii(points):
    """Compute largest uniform safe initial radius"""
    n = points.shape[0]
    # Wall distances
    dist_to_walls = np.minimum(points[:, 0], 1 - points[:, 0])
    dist_to_walls = np.minimum(dist_to_walls, points[:, 1])
    dist_to_walls = np.minimum(dist_to_walls, 1 - points[:, 1])
    # Pairwise distances
    min_dist = np.inf
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(points[i] - points[j])
            if dist < min_dist:
                min_dist = dist
    r0 = min(np.min(dist_to_walls), min_dist / 2)
    return np.full(n, r0)

def inflate_radii(centers, radii):
    """Inflate radii while maintaining feasibility"""
    n = centers.shape[0]
    new_radii = radii.copy()
    D = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    epsilon = 1e-7
    for i in range(n):
        wall_caps = [
            centers[i, 0],
            1 - centers[i, 0],
            centers[i, 1],
            1 - centers[i, 1]
        ]
        neighbor_caps = []
        for j in range(n):
            if i != j:
                neighbor_caps.append(D[i, j] - radii[j])
        all_caps = wall_caps + neighbor_caps
        new_radius = min(all_caps)
        if new_radius > radii[i] + epsilon:
            new_radii[i] = new_radius
    return new_radii

def relax_centers(centers, radii, steps=10):
    """Apply force-based relaxation to centers"""
    n = centers.shape[0]
    alpha = 0.5
    kappa = 0.02
    lambda_wall = 0.01
    beta = 0.02
    gamma = 0.02
    new_centers = centers.copy()
    for _ in range(steps):
        forces = np.zeros_like(new_centers)
        # Pairwise forces
        D = np.linalg.norm(new_centers[:, None, :] - new_centers[None, :, :], axis=2)
        for i in range(n):
            for j in range(n):
                if i != j:
                    dist = D[i, j]
                    delta = radii[i] + radii[j] - dist
                    if delta < 0:
                        force_magnitude = kappa * max(0, beta - dist) / max(dist, 1e-12)
                        dir_vec = (new_centers[i] - new_centers[j]) / max(dist, 1e-12)
                        forces[i] += force_magnitude * dir_vec
        # Wall forces
        for i in range(n):
            fx = 0.0
            fy = 0.0
            if new_centers[i, 0] - radii[i] < gamma:
                fx += lambda_wall * max(0, gamma - (new_centers[i, 0] - radii[i]))
            if 1 - new_centers[i, 0] - radii[i] < gamma:
                fx -= lambda_wall * max(0, gamma - (1 - new_centers[i, 0] - radii[i]))
            if new_centers[i, 1] - radii[i] < gamma:
                fy += lambda_wall * max(0, gamma - (new_centers[i, 1] - radii[i]))
            if 1 - new_centers[i, 1] - radii[i] < gamma:
                fy -= lambda_wall * max(0, gamma - (1 - new_centers[i, 1] - radii[i]))
            forces[i] += [fx, fy]
        # Update centers
        new_centers += alpha * forces
        # Clamp to valid region
        new_centers[:, 0] = np.clip(new_centers[:, 0], radii, 1 - radii)
        new_centers[:, 1] = np.clip(new_centers[:, 1], radii, 1 - radii)
        # Validate
        D_check = np.linalg.norm(new_centers[:, None, :] - new_centers[None, :, :], axis=2)
        if np.any(D_check < radii[:, None] + radii[None, :] - 1e-8) and np.any(D_check < radii[:, None] + radii[None, :] - 1e-8):
            # Revert to previous if overlap
            new_centers = centers.copy()
            break
        centers = new_centers.copy()
    return new_centers

def run_packing():
    """Generate and optimize 26 circle packing in unit square"""
    d = 0.18  # hex spacing
    points = hex_lattice(d)
    points = add_border_circles(points, d)
    radii = initial_radii(points)
    centers = points.copy()
    MAX_OUTER = 30
    previous_sum = np.sum(radii)
    for _ in range(MAX_OUTER):
        radii = inflate_radii(centers, radii)
        centers = relax_centers(centers, radii)
        current_sum = np.sum(radii)
        if current_sum - previous_sum < 1e-6:
            break
        previous_sum = current_sum
    # Final validation
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise AssertionError(f"Packing failed validation: {msg}")
    return centers, radii, np.sum(radii)
