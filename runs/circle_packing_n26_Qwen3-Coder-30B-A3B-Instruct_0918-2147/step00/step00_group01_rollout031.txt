import numpy as np
from scipy.optimize import differential_evolution
from scipy.spatial.distance import cdist

def compute_radii(centers):
    n = centers.shape[0]
    radii = np.zeros(n)
    
    # Wall distances
    x = centers[:, 0]
    y = centers[:, 1]
    wall_distances = np.array([
        x,                    # left
        1 - x,                # right
        y,                    # bottom
        1 - y                 # top
    ])
    
    # Neighbour distances
    distances = cdist(centers, centers)
    np.fill_diagonal(distances, np.inf)
    
    for i in range(n):
        wall_min = np.min(wall_distances[:, i])
        neigh_min = 0.5 * np.min(distances[i])
        radii[i] = min(wall_min, neigh_min)
        
    return radii

def neg_sum_radii(z):
    centers = z.reshape((-1, 2))
    radii = compute_radii(centers)
    return -np.sum(radii)

def run_packing():
    np.random.seed(42)
    dim = 26 * 2
    bounds = [(0, 1)] * dim

    result_de = differential_evolution(
        neg_sum_radii,
        bounds,
        maxiter=2000,
        popsize=10,
        tol=1e-7,
        polish=False,
        strategy='best1bin',
        seed=42
    )
    
    # Convert to centers
    centers = result_de.x.reshape((-1, 2))

    sigma = 0.02

    for _ in range(max_iter):

        step = np.random.normal(0, sigma, size=best_centers.shape)

        new_centers = np.clip(new_centers, 0, 1)
        
        # Evaluate
        new_sum = -neg_sum_radii(new_centers.flatten())
        
        if new_sum > best_sum:
            best_centers = new_centers
            best_sum = new_sum
            sigma = min(1.0, 1.1 * sigma)  # Increase step size

            ]).T,

                np.min(cdist(z.reshape((-1, 2)), z.reshape((-1, 2)))[i][np.arange(26) != i])
                for i in range(26)
            ])
        ).min(axis=1)
    )

        centers_s = z_shifted.reshape((-1, 2))
        x = centers_s[:, 0]
        y = centers_s[:, 1]
        wall_dists = np.minimum.reduce([
            x, 1-x, y, 1-y
        ])
        distances = cdist(centers_s, centers_s)
        np.fill_diagonal(distances, np.inf)
        min_dists = 0.5 * np.min(distances, axis=1)
        smooth_radii = np.minimum(wall_dists, min_dists)
        return -np.sum(smooth_radii)
    
    # Refine centers a bit with smooth objective to pull things together
    refined_centers = best_centers.copy()
    for _ in range(50):
        z_flat = refined_centers.flatten()

        candidate = z_flat + delta
        candidate = np.clip(candidate, 0, 1)
        if smooth_objective(candidate) < smooth_objective(z_flat):
            refined_centers = candidate.reshape((-1, 2))

    final_radii = compute_radii(refined_centers)
    sum_radii = np.sum(final_radii)

        sum_radii = np.sum(final_radii)
