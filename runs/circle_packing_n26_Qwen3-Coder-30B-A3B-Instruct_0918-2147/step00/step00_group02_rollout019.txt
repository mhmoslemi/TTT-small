import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def initial_configuration(n=26):
    # Use Sobol-like sequence for good distribution
    # Generate low-discrepancy points using a simple scrambled Halton-like approach
    # Using the first two primes 2 and 3 for generating a pseudo-Sobol sequence
    phi = (1 + np.sqrt(5)) / 2  # Golden ratio
    
    points = []
    for i in range(n):
        x = (i * phi) % 1
        y = (i * phi * phi) % 1
        points.append([x, y])
    centers = np.array(points)
    
    # Compute initial radii as the min of distance to walls and to nearest neighbor
    radii = []
    for i in range(n):
        x, y = centers[i]
        r = min(x, 1-x, y, 1-y)
        distances = np.linalg.norm(centers[i] - centers, axis=1)
        distances[i] = np.inf  # Exclude self-distance
        r = min(r, 0.5 * np.min(distances))
        radii.append(max(0, r))
    
    radii = np.array(radii)
    return centers, radii

def pack(centers, radii):
    # Flatten into 1D array: centers first, then radii
    return np.concatenate([centers.flatten(), radii])

def unpack(z):
    # Unpack from 1D array back to centers and radii
    n = len(z) // 3  # 26 * 2 + 26
    centers = z[:2*n].reshape(-1, 2)
    radii = z[2*n:]
    return centers, radii

def energy_and_grad(z):
    # Unpack
    centers, radii = unpack(z)
    
    # Number of circles
    n = len(centers)
    
    # Precompute distance matrix
    D = cdist(centers, centers)
    
    # Wall energy
    E_wall = 0.0
    grad_wall = np.zeros_like(centers)
    grad_r_wall = np.zeros(n)
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Compute wall penalties (non-negative if r exceeds wall distance)
        dx_left = r - x
        dx_right = r - (1 - x)
        dy_bottom = r - y
        dy_top = r - (1 - y)
        
        # Wall terms: positive when violation
        wall_terms = [dx_left, dx_right, dy_bottom, dy_top]
        for j, term in enumerate(wall_terms):
            if term > 0:
                E_wall += term**2
                # Gradient w.r.t. r
                grad_r_wall[i] += 2 * term
                # Gradient w.r.t. position
                if j == 0:  # dx_left
                    grad_wall[i, 0] += -2 * term  # gradient w.r.t. x
                elif j == 1:  # dx_right
                    grad_wall[i, 0] += 2 * term   # gradient w.r.t. x
                elif j == 2:  # dy_bottom
                    grad_wall[i, 1] += -2 * term  # gradient w.r.t. y
                elif j == 3:  # dy_top
                    grad_wall[i, 1] += 2 * term   # gradient w.r.t. y
    
    # Interaction energy
    E_pair = 0.0
    grad_pair = np.zeros_like(centers)
    grad_r_pair = np.zeros(n)
    
    # Loop over pairs
    for i in range(n):
        for j in range(i + 1, n):
            d = D[i, j]
            r_i = radii[i]
            r_j = radii[j]
            overlap = r_i + r_j - d
            
            if overlap > 0:
                E_pair += overlap**2
                
                # Compute gradient w.r.t. positions c_i, c_j
                if d > 1e-12:
                    df_dx = (centers[i, 0] - centers[j, 0]) / d
                    df_dy = (centers[i, 1] - centers[j, 1]) / d
                    
                    grad_pair[i, 0] += -2 * overlap * df_dx
                    grad_pair[i, 1] += -2 * overlap * df_dy
                    grad_pair[j, 0] += 2 * overlap * df_dx
                    grad_pair[j, 1] += 2 * overlap * df_dy
                
                # Gradient w.r.t. r_i:
                grad_r_pair[i] += 2 * overlap
                # Gradient w.r.t. r_j:
                grad_r_pair[j] += 2 * overlap
    
    # Budget penalty
    lambda_budget = 1000.0
    total_radius = np.sum(radii)
    E_budget = lambda_budget * (total_radius - B)**2
    grad_r_budget = 2 * lambda_budget * (total_radius - B)
    
    # Total energy
    E = E_wall + E_pair + E_budget
    
    # Gradient
    grad = np.zeros_like(z)
    grad[:2*n] = (grad_wall + grad_pair).flatten()
    grad[2*n:] = (grad_r_wall + grad_r_pair + grad_r_budget)
    
    return E, grad

def inner_optimize(z0, B_inner, lambda_budget=1000.0):
    # Keep a copy of z0 before optimisation, in case fails catastrophically
    try:
        res = minimize(
            fun=lambda z: energy_and_grad(z)[0],
            x0=z0,
            method='L-BFGS-B',
            jac=lambda z: energy_and_grad(z)[1],
            bounds=[(0, 1)] * (26*2) + [(0, 0.5)] * 26,
            options={'gtol': 1e-12, 'ftol': 1e-12}
        )
        return res.x
    except:
        # If minimize fails (should not be normal), fallback to original
        return z0

def max_feasible_budget(z_start):
    B = 3.0  # Start with an overestimate
    delta = 5e-4  # Step size for binary search
    max_iter = 40
    prev_B = B
    
    # The search loop
    for iteration in range(max_iter):
        # Inner optimize
        z_opt = inner_optimize(z_start, B, lambda_budget=1000.0)
        centers, radii = unpack(z_opt)
        
        # Check feasibility with max violation threshold
        violations = []
        for i in range(len(centers)):
            x, y = centers[i]
            r = radii[i]
            violations.append(max(0, r - x, r - (1-x), r - y, r - (1-y)))
        for i in range(len(centers)):
            for j in range(i+1, len(centers)):
                d = np.linalg.norm(centers[i] - centers[j])
                overlap =radii[i] + radii[j] - d
                if overlap > 0:
                    violations.append(overlap)

        max_violation = max(violations) if violations else 0
        
        if max_violation <= 1e-13:
            # Feasible, try to go higher
            B += delta
        else:
            # Infeasible, go lower
            B -= delta
            
        if abs(B - prev_B) < 1e-6:
            break
        prev_B = B
        
    # Return final best z that's feasible
    z_final = inner_optimize(z_start, B, lambda_budget=1000.0)
    centers, radii = unpack(z_final)
    
    return z_final, B

def postprocess(z):
    centers, radii = unpack(z)
    eps = 5e-10
    # Shrink radii
    radii = np.maximum(0, radii - eps)
    # Clamp centers
    centers = np.clip(centers, eps, 1 - eps)
    # Final repair pass
    D = cdist(centers, centers)
    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):
            d = D[i, j]
            r_i = radii[i]
            r_j = radii[j]
            if r_i + r_j > d:
                overlap = r_i + r_j - d
                # Reduce over-lapping radii
                radii[i] -= 0.5 * overlap + eps
                radii[j] -= 0.5 * overlap + eps
                radii[i] = max(0, radii[i])
                radii[j] = max(0, radii[j])
    # Ensure radii do not exceed 0.5
    radii = np.minimum(radii, 0.5)
    # Final safety margin even if somehow overlapping remains
    sum_r = np.sum(radii)
    return centers, radii, sum_r

def run_packing():
    global B
    centers, radii = initial_configuration(26)
    z = pack(centers, radii)
    z_opt, B = max_feasible_budget(z)
    centers, radii, sum_r = postprocess(z_opt)
    return centers, radii, sum_r
