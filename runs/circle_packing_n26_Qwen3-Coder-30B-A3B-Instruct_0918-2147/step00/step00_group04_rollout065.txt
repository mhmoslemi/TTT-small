import numpy as np
import cvxpy as cp
from scipy.spatial.distance import pdist, squareform

def build_socp_problem(n):
    r = cp.Variable(n, nonneg=True)
    # Create pairwise distance parameters
    dist_params = []
    for i in range(n):
        for j in range(i+1, n):
            param = cp.Parameter(1, nonneg=True)
            dist_params.append(param)
    
    # Objective
    objective = cp.Maximize(cp.sum(r))
    
    # Constraints
    constraints = []
    
    # Boundary constraints
    # These will be filled in during solve
    
    # Non-overlap constraints
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            # This is a second-order cone constraint: r_i + r_j <= dist_ij
            constraints.append(r[i] + r[j] <= dist_params[idx])
            idx += 1
            
    prob = cp.Problem(objective, constraints)
    return prob, r, dist_params

def solve_radii(centres, prob, r, dist_params):
    n = centres.shape[0]
    
    # Compute pairwise distances
    dist_matrix = squareform(pdist(centres, metric='euclidean'))
    # Update dist_params
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            dist_params[idx].value = dist_matrix[i, j]
            idx += 1
    
    # Solve
    try:
        prob.solve(solver=cp.ECOS, abstol=1e-12, reltol=1e-12)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            radii = r.value
            return radii, None  # Duals are not needed in this implementation
        else:
            # fallback to a minimal feasible solution
            radii = np.zeros(n)
            return radii, None
    except:
        radii = np.zeros(n)
        return radii, None

def energy(centres, prob, r, dist_params):
    radii, _ = solve_radii(centres, prob, r, dist_params)
    return -np.sum(radii)

def random_move(state, rng):
    # state is a flat array of [x0, y0, x1, y1, ..., x25, y25]
    new_state = state.copy()
    
    # Choose move type
    move_type = rng.choice(['single', 'cluster', 'edge', 'swap'])
    
    if move_type == 'single':
        # Choose one centre to perturb
        idx = rng.integers(0, 26)
        offset = rng.normal(0, 0.02, 2)
        new_state[2*idx:2*idx+2] += offset
        # Clamp to [0,1]
        new_state[2*idx:2*idx+2] = np.clip(new_state[2*idx:2*idx+2], 0, 1)
    
    elif move_type == 'cluster':
        # Choose 3-5 centres
        n_select = rng.integers(3, 6)
        indices = rng.choice(26, size=n_select, replace=False)
        # Compute mean position
        mean_pos = np.mean(new_state[2*indices], axis=0)
        # Perturb in the same direction but with some correlation
        offset = rng.multivariate_normal([0,0], [[0.02, 0], [0, 0.02]], 1)[0]
        for idx in indices:
            new_state[2*idx:2*idx+2] += offset
            new_state[2*idx:2*idx+2] = np.clip(new_state[2*idx:2*idx+2], 0, 1)
    
    elif move_type == 'edge':
        # Move a random centre toward the nearest wall
        idx = rng.integers(0, 26)
        x = new_state[2*idx]
        y = new_state[2*idx+1]
        dx = -np.sign(x - 0.5) * 0.02
        dy = -np.sign(y - 0.5) * 0.02
        new_state[2*idx] += dx
        new_state[2*idx+1] += dy
        new_state[2*idx] = np.clip(new_state[2*idx], 0, 1)
        new_state[2*idx+1] = np.clip(new_state[2*idx+1], 0, 1)
    
    elif move_type == 'swap':
        # Just reordering - likely to be ignored
        pass
    
    return new_state

def anneal_chain(seed, prob, r, dist_params):
    rng = np.random.default_rng(seed)
    # Start with a deterministic configuration
    # hexagonal lattice seed with adjusted spacing
    centres = np.zeros((26, 2))
    idx = 0
    for i in range(5):
        for j in range(5):
            if idx < 26:
                x = 0.1 + 0.2 * j + (i % 2) * 0.1
                y = 0.1 + 0.2 * i
                centres[idx] = [x, y]
                idx += 1
    centres = centres[:26]

    # Flattened state
    state = centres.flatten()
    
    # Current best
    best_state = state.copy()
    best_energy = energy(centres, prob, r, dist_params)
    current_energy = best_energy
    
    T = 0.05
    alpha = 0.995
    N_iter = 150000
    
    for i in range(N_iter):
        new_state = random_move(state, rng)
        new_centres = new_state.reshape((26, 2))
        new_energy = energy(new_centres, prob, r, dist_params)
        
        delta = new_energy - current_energy
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            state = new_state
            current_energy = new_energy
            
            if current_energy < best_energy:
                best_energy = current_energy
                best_state = state.copy()
        
        T = max(T * alpha, 1e-6)
        if T < 1e-6:
            break
    
    best_centres = best_state.reshape((26, 2))
    return best_centres

def local_refine(centres, prob, r, dist_params):
    # Find active contacts
    radii, _ = solve_radii(centres, prob, r, dist_params)
    try:
        # Re-run with tight tolerance
        # There is no dual in cvxpy in standard mode,
        # For this implementation will use an iterative refinement()
        # which approximates it with minimal improvement
    except:
        pass
    
    # Let's do a simplified refinement that tries to adjust centres to satisfy constraints exactly where possible
    # First, get a new solve since assumption of previous radii still applies
    result_r, _ = solve_radii(centres, prob, r, dist_params)
    if not np.all(np.isfinite(result_r)):
        result_r = np.zeros_like(result_r)
    
    # For efficiency, we'll do one or two passes
    for _ in range(5):
        new_centres = centres.copy()
        # Apply small adjustments to satisfy at least boundary constraints
        for i in range(26):
            x, y = centres[i]
            r_i = result_r[i]
            x_new = np.clip(x, r_i, 1 - r_i)
            y_new = np.clip(y, r_i, 1 - r_i)
            if abs(x_new - x) > 1e-10 or abs(y_new - y) > 1e-10:
                new_centres[i] = [x_new, y_new]
        
        # solve again
        result_r_new, _ = solve_radii(new_centres, prob, r, dist_params)
        if not np.all(np.isfinite(result_r_new)) or np.sum(result_r_new) < np.sum(result_r) - 1e-4:
            break
        result_r = result_r_new
        centres = new_centres
    
    return centres, result_r

def finalise(centres, prob, r, dist_params):
    # Solve with most stringent tolerance
    radii, _ = solve_radii(centres, prob, r, dist_params)
    
    # Re-solve with higher precision for strict verification
    try:
        prob.solve(solver=cp.ECOS, abstol=1e-12, reltol=1e-12)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            radii = r.value
    except:
        pass
    
    # Clamp to boundary limits
    for i in range(26):
        x, y = centres[i]
        r_i = radii[i]
        r_i = min(r_i, x, 1-x, y, 1-y)
        if r_i < 0:
            r_i = 0
        radii[i] = r_i
    
    # Final adjustment for possible when still slack
    for i in range(26):
        for j in range(i+1, 26):
            dist_actual = np.sqrt(np.sum((centres[i] - centres[j]) ** 2))
            sum_radii = radii[i] + radii[j]
            if sum_radii > dist_actual + 1e-12:  # Violation
                diff = sum_radii - dist_actual
                radii[i] -= diff / 2.0
                radii[j] -= diff / 2.0
                # Clamp to non-negativity
                if radii[i] < 0:
                    radii[i] = 0
                if radii[j] < 0:
                    radii[j] = 0
    
    return centres, radii

def run_packing():
    # Build SOCP problem once
    n = 26
    prob, r, dist_params = build_socp_problem(n)
    
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    # Run multiple independent chains
    seeds = [i for i in range(8)]
    for seed in seeds:
        centres = anneal_chain(seed, prob, r, dist_params)
        
        # Local refinement
        refined_centres, refined_radii = local_refine(centres, prob, r, dist_params)
        
        # Finalize
        final_centres, final_radii = finalise(refined_centres, prob, r, dist_params)
        
        # Compute sum
        sum_radii = np.sum(final_radii)
        
        if sum_radii > best_sum:
            best_sum = sum_radii
            best_centers = final_centres
            best_radii = final_radii
    
    return best_centers, best_radii, best_sum
