```python
import numpy as np
from scipy.optimize import minimize

NUM_CIRCLES = 26
NUM_RESTARTS = 5
INITIAL_RADIUS = 0.02
BOUNDARY_MARGIN = 1e-9

def hex_lattice_points(n_rows, spacing):
    points = []
    for i in range(n_rows):
        y = i * spacing * np.sqrt(3) / 2
        if y > 1:
            break
        offset = (i % 2) * spacing / 2
        n_cols = int(np.ceil(1.0 / spacing))
        for j in range(n_cols):
            x = offset + j * spacing
            if x > 1:
                break
            points.append([x, y])
    return np.array(points)

def initial_guess(seed):
    np.random.seed(seed)
    # Create hexagonal lattice
    spacing = 0.25
    points = hex_lattice_points(10, spacing)
    # If not enough points, make a grid
    if len(points) < NUM_CIRCLES:
        x = np.linspace(0.1, 0.9, int(np.ceil(np.sqrt(NUM_CIRCLES))))
        y = np.linspace(0.1, 0.9, int(np.ceil(np.sqrt(NUM_CIRCLES))))
        X, Y = np.meshgrid(x, y)
        points = np.column_stack((X.ravel(), Y.ravel()))[:NUM_CIRCLES]
    
    # Trim down to exactly NUM_CIRCLES
    points = points[:NUM_CIRCLES]
    
    # Add jitter to prevent symmetry issues
    jitter = np.random.uniform(-0.02, 0.02, (NUM_CIRCLES, 2))
    points = points + jitter
    
    # Clip to unit square
    points = np.clip(points, 0, 1)
    
    # Initialize radii
    radii = np.full(NUM_CIRCLES, INITIAL_RADIUS)
    
    # Flatten into variable vector (x, y, r) for each circle
    vars0 = np.zeros(3 * NUM_CIRCLES)
    vars0[0::3] = points[:, 0]   # x coordinates
    vars0[1::3] = points[:, 1]   # y coordinates
    vars0[2::3] = radii          # r coordinates
    
    return vars0

def objective(vars):
    return -np.sum(vars[2::3])

def constraint_boundary(vars):
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    # Ensure each circle is inside unit square
    # x - r >= 0 -> x - r + BOUNDARY_MARGIN >= BOUNDARY_MARGIN
    # 1 - x - r >= 0 -> (1 - x - r) + BOUNDARY_MARGIN >= BOUNDARY_MARGIN
    # Same for y
    constraints = np.zeros(4 * NUM_CIRCLES)
    constraints[0::4] = x - r + BOUNDARY_MARGIN
    constraints[1::4] = 1 - x - r + BOUNDARY_MARGIN
    constraints[2::4] = y - r + BOUNDARY_MARGIN
    constraints[3::4] = 1 - y - r + BOUNDARY_MARGIN
    return constraints

def constraint_nonoverlap(vars):
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    constraints = []
    
    for i in range(NUM_CIRCLES):
        for j in range(i + 1, NUM_CIRCLES):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dist_sq = dx * dx + dy * dy
            dist = np.sqrt(dist_sq)
            sum_radii = r[i] + r[j]
            # Non-overlap: dist >= sum_radii - BOUNDARY_MARGIN
            # i.e., dist - sum_radii + BOUNDARY_MARGIN >= 0
            constraints.append(dist - sum_radii + BOUNDARY_MARGIN)
    return np.array(constraints)

def postprocess(vars):
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    # Ensure all centers are within bounds
    x = np.clip(x, r, 1 - r)
    y = np.clip(y, r, 1 - r)
    # Recompute
    result = np.zeros_like(vars)
    result[0::3] = x
    result[1::3] = y
    result[2::3] = r
    return result

def run_packing():
    best_sum = -np.inf
    best_vars = None

    # Create constraints
    constr_boundary = {
        'type': 'ineq',
        'fun': constraint_boundary
    }
    constr_overlap = {
        'type': 'ineq',
        'fun': constraint_nonoverlap
    }
    global_constraints = [constr_boundary, constr_overlap]
    
    # Set bounds (x, y, r for each circle)
    bounds = []
    for _ in range(NUM_CIRCLES):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    
    for seed in range(NUM_RESTARTS):
        vars0 = initial_guess(seed)
        res = minimize(objective, vars0,
                       method='SLSQP',
                       bounds=bounds,
                       constraints=global_constraints,
                       options={'maxiter': 3000, 'ftol':1e-7})
        if not res.success:
            continue
        vars_corr = postprocess(res.x)
        s = np.sum(vars_corr[2::3])
        if s > best_sum:
            best_sum = s
            best_vars = vars_corr

    # Reshape for final output
    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii = best_vars[2::3]
    return centers, radii, float(best_sum)
```
