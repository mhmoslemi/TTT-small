import numpy as np
import scipy.optimize as optimize
import math

def run_packing():

    n = 26

    # Generate a hexagonal grid of points within the unit square
    # Hexagonal grid uses a staggered arrangement of points
    # We start by generating a grid of points in a hexagonal pattern

    # Define spacing and offset for hexagonal packing
    spacing = 0.2
    offset = spacing / 2  # Offset for staggered rows

    # Generate points in a hexagonal grid
    points = []
    for row in range(14):
        for col in range(14):
            x = col * spacing
            y = row * spacing
            if row % 2 == 1:
                x += offset
            points.append((x, y))

    # Trim the points to 26
    points = points[:26]

    # Convert to numpy array
    centers = np.array(points)

    # Ensure centers are within unit square
    centers = np.clip(centers, 0, 1)

    # Initial estimate for radii
    initial_radii = np.full(n, 0.05)

    # Optimization function
    def objective(params):
        # params are [centers_x, centers_y, radii]
        centers = params[:n*2].reshape(n, 2)
        radii = params[n*2:]

        # Check if centers are valid
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if not (0 - 1e-12 <= x - r <= 1 + 1e-12) and not (0 - 1e-12 <= y - r <= 1 + 1e-12):
                return np.inf

        # Check if circles are within the unit square
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 - 1e-12 or x + r > 1 + 1e-12 or y - r < 0 - 1e-12 or y + r > 1 + 1e-12:

        # Check for overlaps
        for i in range(n):

                if dist_sq < (radii[i] + radii[j]) ** 2 - 1e-12:

        return -np.sum(radii)  # Maximize sum of radii

    # Initial guess
    params = np.concatenate([centers.flatten(), initial_radii])

    # Run optimization with SLSQP for better handling of constraints
    result = optimize.minimize(

        method='SLSQP',

    return centers, radii, np.sum(radii)
