import scipy.optimize as optimize

def run_packing():

    initial_positions = np.random.uniform(low=0.1, high=0.9, size=(n, 2))

    # Flatten initial positions and radii to form the initial guess for optimization
    params_initial = np.concatenate([initial_positions.flatten(), initial_radii])

    # Objective function: maximize sum of radii
    def objective(params):
        positions = params[:n*2].reshape(n, 2)
        radii = params[n*2:]

        # Check for negative radii
        for i in range(n):
            if radii[i] < 0:
                return np.inf

        for i in range(n):

            r = radii[i]
            if not (0 - 1e-12 <= x - r <= 1 + 1e-12) or not (0 - 1e-12 <= y - r <= 1 + 1e-12):
                return np.inf

        # Check for overlaps between circles
        for i in range(n):

                dx = positions[i, 0] - positions[j, 0]

                dist_sq = dx * dx + dy * dy

        # Return negative sum of radii to maximize it
        return -np.sum(radii)

    # Perform optimization using SLSQP
    result = optimize.minimize(

        params_initial,
        method='SLSQP',

    def validate_packing(centers, radii):
