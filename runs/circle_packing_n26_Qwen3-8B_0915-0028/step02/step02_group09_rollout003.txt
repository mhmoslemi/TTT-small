import scipy.optimize as optimize

    max_cols = 5
    max_rows = 6

    # Create a hexagonal grid
    centers = []

            x = 0.1 + i * 0.15
            y = 0.1 + j * 0.15
            if i % 2 == 1:
                y += 0.075
            centers.append((x, y))
    # Use the first 26 points
    centers = np.array(centers)[:26]

    # Define initial radii based on the grid spacing
    initial_radii = np.full(26, 0.08)

    def objective(params):

        radii = params[26*2:]
        # Ensure all circles stay within the unit square
        for i in range(26):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:

        # Return negative of sum to maximize

    # Initial parameters
    initial_params = np.concatenate([centers.flatten(), initial_radii])

    result = optimize.minimize(
