import scipy.optimize as optimize

def run_packing():

    for i in range(6):

                y += 0.075

        (0.05, 0.05),

        (0.05, 0.5),
        (0.95, 0.5),
        (0.5, 0.05),
        (0.5, 0.95),

        (0.75, 0.25),
        (0.25, 0.75),
        (0.75, 0.75)

    initial_centers = all_centers

    initial_radii = np.full(26, 0.05)

    initial_params = np.concatenate([initial_centers.flatten(), initial_radii])

            return np.inf

    result = optimize.minimize(

        initial_params,
        method='Nelder-Mead',

    return centers, radii, np.sum(radii)
