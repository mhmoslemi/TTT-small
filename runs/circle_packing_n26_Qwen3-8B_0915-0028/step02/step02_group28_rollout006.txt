    def generate_hexagonal_positions(n):

            x = (i % 4) / 4.0
            y = (i // 4) / 4.0
            positions.append((x, y))
        return np.array(positions)

    def objective(params):
        centers = params[:26*2].reshape(26, 2)
        radii = params[26*2:]

        for i in range(26):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12:

                    return np.inf

        return -np.sum(radii)  # Maximize sum of radii

    # Initial guess
    params = np.concatenate([all_centers.flatten(), initial_radii])
