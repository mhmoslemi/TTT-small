import numpy as np
import scipy.optimize as optimize

def run_packing():

    # Even rows are shifted by 0.5 in x to form hexagonal pattern
    # All circles will have the same initial radius and be spaced at least 2r apart

    # Initial estimate for circle radius based on 5x5 grid in 1x1 square
    initial_radius = 0.15  # Will be optimized

    # Create initial grid positions: 5 rows, 5 columns
    # Coordinate system: (x, y) in [0, 1]^2

    # Width of square = 1
    # To maximize density, we need to place 5 rows, with spacing between rows = sqrt(3)/2 * r
    # Total vertical space = 2 * r * (sqrt(3)/2) + 2r = r * (sqrt(3) + 2)

    # Let's define initial positions
    row_height = np.sqrt(3) / 2 * initial_radius
    total_vertical_space = 2 * initial_radius + 2 * row_height
    vertical_padding = (1 - total_vertical_space) / 2

    # Generate initial grid
    positions = []
    for i in range(5):
        if i % 2 == 0:
            x_offsets = np.linspace(0 + 0.25 * initial_radius, 1 - 0.25 * initial_radius, 5)
        else:
            x_offsets = np.linspace(0.25 * initial_radius, 1 - 0.25 * initial_radius, 5)
        y_offsets = np.linspace(vertical_padding, 1 - vertical_padding, 5)
        for x in x_offsets:
            for y in y_offsets:
                positions.append([x, y])
    positions = np.array(positions)

    # Initial radius for all circles
    radii = np.full(len(positions), initial_radius)

    # Define constraints for optimization
    def objective(radii):
        return -np.sum(radii)  # Maximize sum of radii

    def constraint_overlap(radii, centers):
        n = len(radii)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                distance = np.sqrt(dx ** 2 + dy ** 2)
                if distance < radii[i] + radii[j] - 1e-8:
                    return 1.0  # Violation
        return 0.0

    def constraint_boundary(radii, centers):
        for i in range(len(radii)):
            x, y = centers[i]
            if x - radii[i] < -1e-8 or x + radii[i] > 1 + 1e-8:
                return 1.0
            if y - radii[i] < -1e-8 or y + radii[i] > 1 + 1e8:
                return 1.0
        return 0.0

    # Define optimization bounds: radius must be nonnegative
    bounds = [(0, 1)] * len(radii)

    # Run optimization with constraints
    result = optimize.minimize(
        fun=lambda x: objective(x),
        x0=radii,
        bounds=bounds,
        method='SLSQP',
        constraints=[
            {
                'type': 'ineq',
                'fun': lambda x: 1.0 - constraint_overlap(x, positions),
            },
            {
                'type': 'ineq',
                'fun': lambda x: 1.0 - constraint_boundary(x, positions),
            },
        ],
    )

    # Extract final results
    centers = positions
    radii_result = result.x

    center_x = 0.5

    # Final check for uniqueness

    return centers, radii_result, np.sum(radii_result)
