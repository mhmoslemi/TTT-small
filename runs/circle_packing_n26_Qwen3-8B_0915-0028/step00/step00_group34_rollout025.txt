import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Grid positions with some left out to make 26 circles
    # I will use a 4-row, 7-column grid where one row is popped out
    # This helps avoid outlining the grid clearly

    # Define initial grid points
    grid = np.array([
        [0.1, 0.1], [0.3, 0.1], [0.5, 0.1], [0.7, 0.1], [0.9, 0.1],
        [0.1, 0.3], [0.3, 0.3], [0.5, 0.3], [0.7, 0.3], [0.9, 0.3],
        [0.1, 0.5], [0.3, 0.5], [0.5, 0.5], [0.7, 0.5], [0.9, 0.5],
        [0.1, 0.7], [0.3, 0.7], [0.5, 0.7], [0.7, 0.7], [0.9, 0.7],
        [0.1, 0.9], [0.3, 0.9], [0.5, 0.9], [0.7, 0.9], [0.9, 0.9]
    ])
    
    # Randomly perturb the grid positions to prevent collisions
    np.random.seed(0)
    grid += np.random.uniform(-0.05, 0.05, grid.shape)
    # Ensure all points are inside the unit square
    grid = np.clip(grid, 0, 1)

    # Define the number of circles
    n = grid.shape[0]

    # Initialize radii as small values
    radii = np.random.uniform(0.01, 0.03, n)

    # Objective function to maximize sum of radii
    # We use -sum(radii) for minimization
    def objective(x):
        # x[0:n] contains the radii
        # x[n:] contains perturbations to grid positions
        perturbed_centers = grid + x[n:]
        radii = x[:n]
        return -np.sum(radii)

    # Define constraints
    def constraint1(x):
        # Circle centers must be within [0,1] x [0,1]
        perturbed_centers = grid + x[n:]
        return np.all((perturbed_centers >= 0) & (perturbed_centers <= 1))

    def constraint2(x):
        # Circle must not go beyond the square
        perturbed_centers = grid + x[n:]
        radii = x[:n]
        return np.all(perturbed_centers - radii >= 0) & np.all(perturbed_centers + radii <= 1)

    def constraint3(x):
        # Circle centers must be within [0,1]
        perturbed_centers = grid + x[n:]
        return np.all((perturbed_centers >= 0) & (perturbed_centers <= 1))

    def constraint4(x):
        # Distance between circles must be >= sum of radii
        # x[0:n] contains radii
        # x[n:] contains perturbations to grid positions
        perturbed_centers = grid + x[n:]
        radii = x[:n]
        for i in range(n):
            for j in range(i+1, n):
                dx = perturbed_centers[i, 0] - perturbed_centers[j, 0]
                dy = perturbed_centers[i, 1] - perturbed_centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -1
        return 0

    # Bounds: Radii must be >=0 and perturbations must be in [-0.1, 0.1]
    bounds = [(0, 1) for _ in range(n)] + [(-0.1, 0.1) for _ in range(grid.shape[0])]

    # Set up the optimization
    result = optimize.minimize(
        objective,
        x0=np.concatenate((radii, np.zeros_like(grid).flatten())),
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint1},
            {'type': 'ineq', 'fun': constraint2},
            {'type': 'ineq', 'fun': constraint3},
            {'type': 'ineq', 'fun': constraint4},

        method='SLSQP',

    # Extract results

    radii = result.x[:n]

    # Check if the result is valid
    valid, message = validate_packing(centers, radii)
    if not valid:
        raise ValueError(f"Invalid packing: {message}")

    return centers, radii, np.sum(radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
