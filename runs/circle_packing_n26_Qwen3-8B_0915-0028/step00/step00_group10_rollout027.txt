from scipy.optimize import minimize
import warnings
warnings.filterwarnings("ignore")

def run_packing():
    # Initial grid configuration (5x5 grid with one circle removed)
    n_circles = 26
    centers = []
    for i in range(5):
        for j in range(5):
            if i * 5 + j < n_circles - 1:
                x = i / 4
                y = j / 4
                centers.append([x, y])
    centers = np.array(centers)
    
    # Initial radii as function of distance to nearest boundary
    def get_initial_radius(x, y):
        return min(x, 1 - x, y, 1 - y)

    radii = np.array([get_initial_radius(x, y) for x, y in centers])

    def objective(radii, centers):
        return -np.sum(radii)  # Minimize negative sum to maximize

    # Define constraints (non-overlapping circles and boundaries)
    def constraint_overlap(radii, centers):
        n = len(centers)
        dists = np.zeros((n, n))
        for i in range(n):
            for j in range(i + 1, n):
                dists[i, j] = np.sqrt(np.sum((centers[i] - centers[j])**2))
        return dists - (radii[:, np.newaxis] + radii[np.newaxis, :]) + 1e-8

    bounds = [(0, 1) for _ in range(n_circles)]
    result = minimize(
        fun=objective,
        x0=radii,
        args=(centers,),
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': constraint_overlap, 'args': (centers,)},

    if not valid:

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
