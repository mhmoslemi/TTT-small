import numpy as np
import scipy.optimize as optimize

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def initial_placement():
    """Generate initial configuration: 5x5 grid with central circle."""
    centers = []
    radii = []
    for i in range(5):
        for j in range(5):
            x = j / 4.0
            y = i / 4.0
            centers.append([x, y])
            radii.append(0.1)  # Initial radius

    centers = np.array(centers)
    radii = np.array(radii)

    # Add one more circle to make total 26
    # Place a small circle in the top right corner
    centers = np.vstack((centers, [0.9, 0.9]))
    radii = np.append(radii, 0.05)

    return centers, radii

def objective(params, centers, radii):
    """Optimization objective: maximize sum of radii."""
    # This is a trick to make the optimization problem maximization of sum(radii)
    # By minimizing -sum(radii)
    return -np.sum(radii)

def constraint(params, centers, radii):
    """Constraint function for optimization."""
    # Reconstruct the centers and radii from the parameters
    # params: flattened array of parameters for centers and radii
    n = len(centers)
    params_flat = params.reshape((n * 2,))  # (x1, y1, x2, y2, ..., xn, yn, r1, r2, ..., rn)
    centers_flat = params_flat[:n * 2].reshape((n, 2))
    radii_flat = params_flat[n * 2:].reshape((n,))

    # Check if all circles are within the square
    for i in range(n):
        x, y = centers_flat[i]
        r = radii_flat[i]
        if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
            return -1e6  # Penalize if out of bounds

    # Check for overlap between circles
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers_flat[i, 0] - centers_flat[j, 0]
            dy = centers_flat[i, 1] - centers_flat[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < radii_flat[i] + radii_flat[j] - 1e-8:
                return -1e6  # Penalize if overlapping

    return 0

def run_packing():
    # Step 1: Initial placement
    centers, radii = initial_placement()

    # Step 2: Define parameters for optimization (flattened)
    n = len(centers)
    params_flat = np.concatenate((centers.flatten(), radii))

    # Step 3: Define bounds for parameters
    # For x and y of each circle, bounds are [0, 1]
    # For radius, bounds are [0, 1] (but realistically, much smaller)
    bounds = []
    for i in range(n):
        bounds.extend([(0, 1), (0, 1)])  # x and y for each circle
        bounds.append((0, 1))  # radius for each circle

    # Step 4: Define optimization problem
    optimize_result = optimize.minimize(
        fun=lambda p: objective(p, centers, radii),
        x0=params_flat,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': lambda p: constraint(p, centers, radii)}
        ],
        method='L-BFGS-B',
        options={'maxiter': 1000, 'ftol': 1e-8}
    )

    # Extract optimal parameters
    optimal_params = optimize_result.x

    radii_opt = optimal_params[n * 2:].reshape((n,))

    # Check if the packing is valid
    valid, message = validate_packing(centers_opt, radii_opt)
    if not valid:
        raise ValueError(f"Invalid packing: {message}")

    return centers_opt, radii_opt, np.sum(radii_opt)
