import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Constant values
    n_circles = 26
    center_circle = np.array([0.5, 0.5])
    corner_centers = np.array([
        [0.1, 0.1], [0.9, 0.1], [0.9, 0.9], [0.1, 0.9]
    ])
    hex_grid_centers = []
    offset = 0.2
    for i in range(4):
        for j in range(4):
            if (i + j) % 2 == 0:
                hex_grid_centers.append([0.2 + i * offset, 0.2 + j * offset])
            else:
                hex_grid_centers.append([0.2 + i * offset, 0.2 + j * offset + offset / 2])
    for i in range(4):
        for j in range(4):
            hex_grid_centers.append([0.2 + i * offset, 0.2 + j * offset + offset / 2])
    remaining_centers = [
        [0.5, 0.2], [0.5, 0.8], [0.2, 0.5], [0.8, 0.5], [0.5, 0.5]
    ]

    # Initial center guesses
    centers = np.array([
        center_circle,
        corner_centers[0],
        corner_centers[1],
        corner_centers[2],
        corner_centers[3],
        *hex_grid_centers,
        *remaining_centers
    ])

    # Initial radii (simplified guess)
    initial_radii = np.array([
        0.1,  # Central circle
        0.1,  # Top-left
        0.1,  # Top-right
        0.1,  # Bottom-right
        0.1,  # Bottom-left
        *([0.1] * len(hex_grid_centers)),  # Hexagonal grid
        *([0.05] * len(remaining_centers))   # Remaining circles
    ])

    def objective(x):
        return -np.sum(x[5:])  # Maximize sum of radii

    def constraints(x):
        # Non-overlapping constraint between all pairs
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dx = x[2 * i] - x[2 * j]
                dy = x[2 * i + 1] - x[2 * i + 1 + 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < x[5 + i] + x[5 + j] - 1e-8:
                    return np.inf

        # Bound constraints
        for i in range(n_circles):
            if x[2 * i] < 0 or x[2 * i] > 1 or x[2 * i + 1] < 0 or x[2 * i + 1] > 1:
                return np.inf

        # Circle must be inside square
        for i in range(n_circles):
            if x[2 * i] - x[5 + i] < -1e-8 or x[2 * i] + x[5 + i] > 1 + 1e-8:
                return np.inf
            if x[2 * i + 1] - x[5 + i] < -1e-8 or x[2 * i + 1] + x[5 + i] > 1 + 1e-8:
                return np.inf

        return 0

    # Flatten centers and radii for optimization
    bounds = []
    for i in range(n_circles):
        bounds.extend([(0, 1), (0, 1)])  # x and y
        bounds.extend([(0, 1)])         # radius

    variables = np.concatenate([centers.flatten(), initial_radii])

    result = optimize.minimize(
        objective,
        variables,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}
    )

    # Extract final centers and radii
    centers = result.x[:2 * n_circles].reshape(-1, 2)
    radii = result.x[2 * n_circles:]

    # Validate the result
    is_valid, msg = validate_packing(centers, radii)
    if not is_valid:
        raise ValueError(msg)

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
