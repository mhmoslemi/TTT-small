import numpy as np
import cvxpy as cp

def run_packing():
    # Define variables for centers (x, y) and radii
    num_circles = 26
    centers = cp.Variable((num_circles, 2))
    radii = cp.Variable(num_circles)

    # Objective: Maximize the total radius sum
    obj = cp.Maximize(cp.sum(radii))

    # Constraints
    constraints = []

    # Each circle must lie within the unit square
    for i in range(num_circles):
        x, y = centers[i]
        r = radii[i]
        constraints.append(x - r >= -1e-12)
        constraints.append(x + r <= 1 + 1e-12)
        constraints.append(y - r >= -1e-12)
        constraints.append(y + r <= 1 + 1e-12)

    # Each circle must have non-negative radius
    constraints.append(cp.SOC(radii >= 0))

    # Ensure circles do not overlap
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            distance = cp.sqrt(dx**2 + dy**2)
            constraints.append(distance >= radii[i] + radii[j] - 1e-12)

    # Solve the optimization problem
    problem = cp.Problem(obj, constraints)
    problem.solve(solver=cp.SCS, eps=1e-6)

    # Extract results
    centers_val = centers.value
    radii_val = radii.value
    total_radius_sum = np.sum(radii_val)

    return (centers_val, radii_val, total_radius_sum)
