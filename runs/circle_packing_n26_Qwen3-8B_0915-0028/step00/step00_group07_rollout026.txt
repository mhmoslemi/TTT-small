import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
import math

def compute_pairwise_distances(centers):
    return cdist(centers, centers)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def initialize_greedy_config(n):
    grid = np.linspace(0, 1, int(np.sqrt(n)) + 1)
    points = []
    for x in grid:
        for y in grid:
            if x < 1 and y < 1:
                points.append([x, y])
    points = np.array(points)
    return points[:n]

def run_packing():
    n = 26
    centers = initialize_greedy_config(n)
    radii = np.ones(n)

    # Define bounds for positions and radii
    bounds = []
    for i in range(n):
        bounds.append((0, 1))  # x-coordinate
        bounds.append((0, 1))  # y-coordinate
        bounds.append((0, 1 - np.sqrt(2)))  # radius must fit in square
    bounds = np.array(bounds).reshape(n, 3)

    # Objective function: maximize sum of radii
    def objective(x):
        flat = x.reshape(n, 3)
        centers = flat[:, :2]
        radii = flat[:, 2]
        return -np.sum(radii)

    # Constraint: circles must not overlap with each other

        flat = x.reshape(n, 3)
        centers = flat[:, :2]
        radii = flat[:, 2]

        overlaps = (distances < radii[:, np.newaxis] + radii[np.newaxis, :]) * 1
        return np.sum(overlaps)

    valid, msg = validate_packing(final_centers, final_radii)
    if not valid:
        raise RuntimeError("Invalid packing: " + msg)

    return final_centers, final_radii, np.sum(final_radii)
