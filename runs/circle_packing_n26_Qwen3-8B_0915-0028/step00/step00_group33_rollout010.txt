import numpy as np
import cvxpy as cp

def run_packing():
    # Define total number of circles
    n = 26

    # Set up optimization variables
    centers = cp.Variable((n, 2))
    radii = cp.Variable(n)

    # Objective: Maximize sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints
    constraints = []

    # Boundary constraint: circles must be inside [0, 1]^2
    for i in range(n):
        constraints.append(centers[i, 0] - radii[i] >= 0)
        constraints.append(centers[i, 0] + radii[i] <= 1)
        constraints.append(centers[i, 1] - radii[i] >= 0)
        constraints.append(centers[i, 1] + radii[i] <= 1)

    # Non-overlap constraint between all pairs of circles
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            constraints.append(dist >= radii[i] + radii[j] - 1e-8)

    # Form the problem
    problem = cp.Problem(objective, constraints)

    # Solve the problem
    problem.solve(solver=cp.SCS)

    # Extract the optimal centers and radii
    centers_opt = centers.value
    radii_opt = radii.value

    # Sum of radii
    sum_radii = np.sum(radii_opt)

    return (centers_opt, radii_opt, sum_radii)
