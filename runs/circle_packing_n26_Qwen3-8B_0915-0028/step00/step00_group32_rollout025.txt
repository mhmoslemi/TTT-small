def run_packing():

    n = 26

    grid_rows = 4
    grid_cols = 5
    grid_spacing = 0.25  # Spacing between circles

    main_grid = []
    main_radii = []
    for i in range(grid_rows):
        for j in range(grid_cols):
            x = j * grid_spacing + grid_offset
            y = i * grid_spacing + grid_offset
            main_grid.append([x, y])
            main_radii.append(grid_spacing - 0.02)  # Small spacing between circles

    corner_positions = [
        [0.1, 0.1],

    centers = np.array(main_grid + corner_positions)

    def validate_packing(centers, radii):

        if np.isnan(centers).any() or np.isnan(radii).any():
            return False, "NaN values present"

            if radii[i] < 0:
                return False, f"Circle {i} has negative radius {radii[i]}"

        for i in range(n):

            r = radii[i]

                if dist < radii[i] + radii[j] - 1e-12:

    is_valid, msg = validate_packing(centers, radii)

    return centers, radii, np.sum(radii)
