import numpy as np
import scipy.optimize as optimize

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
            or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    # Initial guess for centers (hexagonal grid of 25 circles)
    N = 25
    centers = np.zeros((N, 2))
    for i in range(N):
        row = i // 5
        col = i % 5
        centers[i, 0] = 0.2 + col * 0.22
        centers[i, 1] = 0.2 + row * 0.22

    # Add one more circle near the center
    centers = np.vstack((centers, [0.5, 0.5]))
    radii = np.ones(26) * 0.05  # Initial guess for radii

    # Objective function to minimize: total weighted distance from center
    def objective(x, centers, radii):
        x = np.reshape(x, (len(radii), 2))
        return np.sum(np.sum((x - centers) ** 2, axis=1) * radii)

    # Constraints: circles must be in the square and not overlap
    def constraint_centers(x):
        x = np.reshape(x, (len(radii), 2))
        return x

    def constraint_boundary(x):
        x = np.reshape(x, (len(radii), 2))
        for i in range(len(radii)):
            if x[i, 0] - radii[i] < -1e-12 or x[i, 0] + radii[i] > 1 + 1e-12:
                return -1
            if x[i, 1] - radii[i] < -1e-12 or x[i, 1] + radii[i] > 1 + 1e-12:
                return -1
        return 0

    def constraint_overlap(x):
        x = np.reshape(x, (len(radii), 2))
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                dx = x[i, 0] - x[j, 0]
                dy = x[i, 1] - x[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return -1
        return 0

    # Combine all constraints
    def constraint(x):
        c1 = constraint_centers(x)
        c2 = constraint_boundary(x)
        c3 = constraint_overlap(x)
        return np.array([c1, c2, c3])

    # Initial guess
    x0 = np.concatenate((centers.flatten(), radii))

    # Optimization
    result = optimize.minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=None,
        constraints=[{'type': 'ineq', 'fun': constraint}]

    # Extract result
    centers_opt = np.reshape(result.x[:2*26], (26, 2))

    valid, message = validate_packing(centers_opt, radii_opt)
    if not valid:
        raise Exception(f"Validation failed: {message}")

    sum_radii = np.sum(radii_opt)
    return centers_opt, radii_opt, sum_radii
