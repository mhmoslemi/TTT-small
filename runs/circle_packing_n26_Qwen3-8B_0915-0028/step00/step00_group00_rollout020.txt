import numpy as np
import scipy.optimize as opt
from scipy.spatial.distance import cdist

def run_packing():
    # Initial grid-based packing for 25 circles
    num_circles = 26
    centers = []
    radii = []

    # 5x5 grid with spacing
    grid_spacing = 1 / 5
    for i in range(5):
        for j in range(5):
            x = i * grid_spacing
            y = j * grid_spacing
            centers.append([x, y])
            radii.append(grid_spacing / 2)

    # Add the 26th circle in the corner (bottom left) for better radius
    centers.append([0, 0])
    radii.append(0.2)

    # Optimization parameters
    max_iter = 100
    tol = 1e-4

    # Define the objective function to maximize the sum of radii
    def objective(x):
        # x contains all variables: [x0, y0, r0, x1, y1, r1, ...]
        n = num_circles
        centers_opt = x[:2*n].reshape(n, 2)
        radii_opt = x[2*n:].reshape(n)

        total_sum = np.sum(radii_opt)

        # Constraint: All circles within the unit square
        for i in range(n):
            x_i, y_i = centers_opt[i]
            r_i = radii_opt[i]
            if x_i - r_i < 0 or x_i + r_i > 1 or y_i - r_i < 0 or y_i + r_i > 1:
                return -np.inf  # Penalize out-of-bound circles

        # Constraint: No overlap between circles

            for j in range(i + 1, n):
                dx = centers_opt[i, 0] - centers_opt[j, 0]
                dy = centers_opt[i, 1] - centers_opt[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii_opt[i] + radii_opt[j] - 1e-8:
                    return -np.inf  # Penalize overlapping circles

        return total_sum

    # Flatten centers and radii for optimization
    center_flat = np.array(centers).flatten()
    radii_flat = np.array(radii).flatten()
    x0 = np.concatenate([center_flat, radii_flat])

    result = opt.minimize(

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
