import numpy as np
import random
from scipy.optimize import minimize

def run_packing():
    def generate_initial_population(n_circles=26):
        # Generate an initial population of circle positions and radii
        centers = np.random.rand(n_circles, 2)
        radii = np.random.rand(n_circles) * 0.1
        return centers, radii

    def mutate(solution, mutation_rate=0.05, mutation_strength=0.1):
        # Mutate a solution by slightly changing positions and radii
        centers, radii = solution
        for i in range(len(centers)):
            if random.random() < mutation_rate:
                centers[i] += np.random.normal(0, mutation_strength, 2)
                centers[i] = np.clip(centers[i], 0, 1)
            if random.random() < mutation_rate:
                radii[i] += np.random.normal(0, mutation_strength)
                radii[i] = np.clip(radii[i], 0, 1)
        return centers, radii

    def crossover(parent1, parent2):
        # Crossover between two solutions
        centers1, radii1 = parent1
        centers2, radii2 = parent2
        centers = np.zeros((len(centers1), 2))
        radii = np.zeros(len(radii1))
        for i in range(len(centers1)):
            if random.random() < 0.5:
                centers[i] = centers1[i]
                radii[i] = radii1[i]
            else:
                centers[i] = centers2[i]
                radii[i] = radii2[i]
        return centers, radii

    def fitness(solution):
        # Fitness function that evaluates the total sum of radii under constraints
        # This is a helper for the GA
        centers, radii = solution
        if validate_packing(centers, radii)[0]:
            return np.sum(radii)
        else:
            return 0  # Penalize invalid solutions

            lambda x: -np.sum(x[1:]),  # Minimize negative sum of radii (maximize sum)
            x0=np.concatenate((centers.flatten(), radii)),
            bounds=[(0, 1) for _ in range(centers.shape[0] * 2)] + [(0, 1) for _ in range(len(radii))],
            constraints=[
                {'type': 'ineq', 'fun': lambda x: np.min(np.sqrt(np.sum((x[:len(x)//2].reshape(-1, 2) - x[len(x)//2: len(x)//2 + len(radii)]**2)**2)) - np.sum(x[len(x)//2: len(x)//2 + len(radii)])) - 1e-8}
            ]
        )
        # Reconstruct centers and radii from the optimized result
        optimal = result.x
        centers = optimal[:len(optimal) // 2].reshape(-1, 2)
        radii = optimal[len(optimal) // 2:]
        return centers, radii

    def validate_packing(centers, radii):
        n = centers.shape[0]
        if np.isnan(centers).any() or np.isnan(radii).any():
            return False, "NaN values present"
        for i in range(n):
            if radii[i] < 0:
                return False, f"Circle {i} has negative radius {radii[i]}"
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                    or y - r < -1e-12 or y + r > 1 + 1e-12):
                return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, f"Circles {i} and {j} overlap"
        return True, "ok"

    # Generate an initial population
    centers, radii = generate_initial_population()

    # Run genetic algorithm
    population_size = 100
    population = [ (np.random.rand(26, 2), np.random.rand(26)*0.1) for _ in range(population_size) ]

        fitness_scores = [fitness(sol) for sol in population]
