import numpy as np
import scipy.optimize as opt

def run_packing():

    rows = 5
    cols = 5
    radius = 0.05  # Initial estimate
    
    # Define positions for hexagonal grid
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = j * (1 + radius) / rows
            y = i * (np.sqrt(3) / 2) * (1 + radius) / rows
            centers.append((x, y))

    # Add boundary circles
    boundary_circles = [
        (0.05, 0.05), (0.95, 0.05), (0.05, 0.95), (0.95, 0.95),
        (0.05, 0.5), (0.95, 0.5), (0.5, 0.05), (0.5, 0.95)
    ]
    centers.extend(boundary_circles)

    # Convert to NumPy array
    centers = np.array(centers)

    # Optimization problem: maximize total radius sum subject to:
    # - Boundary constraints
    # - Non-overlap constraints

    # Objective function: maximize sum of radii (maximize total area)
    # But we will use a constrained optimization method to find the optimal radii
    # while satisfying all constraints

    # Packing function

    def objective(radii):
        return -np.sum(radii)  # We minimize negative total to maximize total

    def constraints(radii):
        # Boundary constraints: all circles must be within [0,1]^2
        for i in range(len(radii)):
            if centers[i, 0] - radii[i] < -1e-8 or centers[i, 0] + radii[i] > 1 + 1e-8:
                return False
            if centers[i, 1] - radii[i] < -1e-8 or centers[i, 1] + radii[i] > 1 + 1e-8:

        return True

    # Optimization settings
    bounds = [(0.0, 1.0) for _ in range(len(centers))]

    result = opt.minimize(
        fun=objective,
        x0=np.full(len(centers), 0.05),
