import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Step 1: Define initial grid positions for circle centers
    n = 26
    # Divide unit square into a grid of 5 rows and 5 columns
    grid_size = 5
    grid_width = 1.0 / grid_size
    grid_height = 1.0 / grid_size

    # Generate initial grid centers
    grid_centers = []
    for i in range(grid_size):
        for j in range(grid_size):
            x = i * grid_width + grid_width / 2
            y = j * grid_height + grid_height / 2
            grid_centers.append([x, y])
    grid_centers = np.array(grid_centers)

    # Initial radii based on grid spacing
    initial_radii = np.full(n, grid_width / 2)

    # Optimization parameters
    bounds = [([0, 1], [0, 1]) for _ in range(n)]  # bounds for (x, y) of each circle
    radius_bounds = [(0, 1) for _ in range(n)]  # bounds for radius

    # Use Nelder-Mead method to optimize radii and center positions
    def objective(x):
        # x has shape (2 * n, ), first n are x-coordinates, next n are y-coordinates
        centers = x[:n].reshape(n, 2)
        radii = x[n:].reshape(n)
        
        # Step 1: Enforce boundary constraints
        for i in range(n):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if x_i - r_i < 0 or x_i + r_i > 1 or y_i - r_i < 0 or y_i + r_i > 1:
                return float('inf')  # Large penalty for out-of-bounds

        # Step 2: Enforce non-overlapping constraints
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return float('inf')

        # Step 3: Compute sum of radii
        return -np.sum(radii)  # Negative for minimization

    # Initial guess: grid-based centers and equal radii
    x0_centers = grid_centers.flatten()
    x0_radii = initial_radii
    x0 = np.concatenate([x0_centers, x0_radii])

    # Run optimization with Nelder-Mead
    result = optimize.minimize(
        objective,
        x0,
        bounds=[(0, 1) for _ in range(2 * n)] + radius_bounds,
        method='Nelder-Mead',
        options={'xatol': 1e-6, 'fatol': 1e-6, 'maxiter': 1000}
    )

    # Extract final configuration from result
    centers = result.x[:n].reshape(n, 2)
    radii = result.x[n:].reshape(n)
    sum_radii = np.sum(radii)

    # Validate
    valid, message = validate_packing(centers, radii)
    if not valid:
        print(f"Validation failed: {message}")
        return np.array([[0, 0]]), np.array([0.0]), 0.0

    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
