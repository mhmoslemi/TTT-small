import numpy as np
import scipy.optimize as optimize
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    def constraint_overlaps(radii, centers):
        n = len(radii)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return dist - (radii[i] + radii[j])
        return 0.0

    def constraint_boundary(radii, centers):
        for i in range(len(radii)):

            if x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12:
                return x + radii[i] - 1.0

                return y + radii[i] - 1.0
        return 0.0

    # Initialize with a grid-like arrangement
    centers = np.zeros((26, 2))
    for i in range(26):
        row, col = divmod(i, 5)

        centers[i, 1] = row / 4.5 + 0.05

    x0 = np.ones(26) * 0.05

    cons = [
