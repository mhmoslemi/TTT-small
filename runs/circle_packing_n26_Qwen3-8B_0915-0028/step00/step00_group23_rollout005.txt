from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def _initial_hexagonal_packing(n: int, size: float = 1.0) -> np.ndarray:
    """
    Generate initial hexagonal pattern for n circles in [0, 1] x [0, 1].
    """
    max_rows = int(np.sqrt(n))
    remaining = n - max_rows * max_rows
    rows = max_rows + (1 if remaining > 0 else 0)
    cols = max_rows if remaining <= max_rows else max_rows + 1

    angles = np.linspace(0, 2 * np.pi, rows, endpoint=False)
    radii = np.linspace(0.5 / rows, 0.5, rows)
    initial_centers = []

    for i in range(rows):
        r = radii[i]
        angle = angles[i]
        for j in range(cols):
            x = 0.5 * (1 - j) + r * np.cos(angle)
            y = 0.5 * (1 - i) + r * np.sin(angle)
            initial_centers.append((x, y))
    
    # Convert to numpy array and ensure all are within [0, 1] x [0, 1]
    centers = np.array(initial_centers)
    centers = np.clip(centers, 0, 1)
    return centers

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
