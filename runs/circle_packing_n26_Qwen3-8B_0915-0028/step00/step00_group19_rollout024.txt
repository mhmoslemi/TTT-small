from scipy.optimize import minimize
from scipy.spatial import distance

def compute_hexagonal_grid(n):

    grid = []
    cols = int(np.sqrt(n))
    rows = (n + cols - 1) // cols

    for i in range(rows):
        for j in range(cols):
            x = j + 0.5 * i
            y = i
            if x < cols and y < rows:
                grid.append([x, y])
    return np.array(grid)
