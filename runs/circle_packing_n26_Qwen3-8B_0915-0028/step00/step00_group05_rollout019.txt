import numpy as np
import scipy.optimize as opt

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12

            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    # Initial grid-based hexagonal packing
    def generate_grid(num_circles):
        num_rows = int(np.sqrt(num_circles))
        num_cols = num_rows + 1 if num_rows * num_rows < num_circles else num_rows

        grid = []
        for i in range(num_rows):
            for j in range(num_cols):
                x = j / (num_cols - 1) if num_cols > 1 else 0.5
                y = i / (num_rows - 1) if num_rows > 1 else 0.5
                grid.append([x, y])
        return np.array(grid)

    # Initialize with a grid-based hexagonal packing
    initial_centers = generate_grid(26)
    initial_radii = np.full(26, 0.08)

        positions = x[:2*26].reshape(26, 2)

        return -np.sum(radii)  # We maximize the sum, so minimize negative sum

    def constraints(x):
        positions = x[:2*26].reshape(26, 2)
        radii = x[2*26:]
        
        # Ensure no circle extends outside the square
        for i in range(26):
            x_i, y_i = positions[i]

            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12:

            if y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:

        # Ensure no overlap between circles

                if dist < radii[i] + radii[j] - 1e-12:
                    return np.inf

        return 0

    # Prepare initial guess
    x_initial = np.concatenate((initial_centers.flatten(), initial_radii))

    # Optimize using Sequential Least Squares Programming (SLSQP)
    result = opt.minimize(
        objective,
        x_initial,
        bounds=[(-1, 1) for _ in range(2*26)] + [(0, 1) for _ in range(26)],
