import scipy.optimize as optimize
import cvxpy as cp

def validate_packing(centers, radii):

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

        x, y = centers[i]
        r = radii[i]

            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):

            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    n = 26
    # Initial grid-like placement
    initial_centers = np.zeros((n, 2))
    radius_guess = np.ones(n) * 0.05  # Initial guess for radius

    # First pass: grid-like arrangement with hexagonal packing
    col, row = np.meshgrid(np.arange(0, n), np.arange(0, n))
    idx = np.ravel_multi_index((row, col), (n, n))
    idx = idx[:n]
    initial_centers[idx] = np.stack((col[idx], row[idx])) / n

    # Define optimization variables
    centers = cp.Variable((n, 2))
    radii = cp.Variable(n)

    # Objective function: Maximize sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints
    constraints = []

    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        constraints.append(x - r >= -1e-12)
        constraints.append(x + r <= 1 + 1e-12)
        constraints.append(y - r >= -1e-12)
        constraints.append(y + r <= 1 + 1e-12)

    # Non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            constraints.append(dist >= radii[i] + radii[j] - 1e-12)

    # Initial values for CVXPY
    initial_values = {}
    for i in range(n):
        initial_values[centers[i, 0]] = initial_centers[i, 0]
        initial_values[centers[i, 1]] = initial_centers[i, 1]
        initial_values[radii[i]] = radius_guess[i]

    # Create and solve problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver='SCS', eps=1e-5, max_iter=1000, verbose=False, init_val=initial_values)
