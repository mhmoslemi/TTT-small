from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():

        return np.minimum(center[0] - radius, 1 - radius)  # Ensure circle within [0,1]^2

    bounds = Bounds(np.zeros(num_circles * 2), np.full(num_circles * 2, 1.0))
