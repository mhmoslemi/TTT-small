import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds

def run_packing():
    # Helper: Hexagonal tiling function
    def hexagonal_tiling(n, spacing):
        centers = []
        for i in range(n):
            row = i // spacing
            col = i % spacing
            x = col + 0.5 * (row % 2)
            y = row * np.sqrt(3) / 2
            centers.append([x, y])
        return np.array(centers)
    
    # Start with a hexagonal grid-based configuration
    spacing = 4
    n = 26
    centers = hexagonal_tiling(n, spacing)
    
    # Manually add circles near corners for better packing
    manual_centers = np.array([
        [0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9], [0.5, 0.5]
    ])
    centers = np.vstack([centers, manual_centers])
    
    # Optimization bounds (initial guess for radii is 0.01)
    r0 = np.full(centers.shape[0], 0.01)
    
    # Cost function: minimize -sum_radii (maximize sum_radii), with constraints
    def objective(x):
        r = x[:len(r0)]  # Extract radii from the state vector
        centers = x[len(r0):]  # Extract centers from the state vector
        sum_r = np.sum(r)
        
        # Apply boundary constraints: radii cannot exceed the distance to the nearest boundary
        r_bound = np.minimum(
            np.minimum(centers[:, 0], 1 - centers[:, 0]),
            np.minimum(centers[:, 1], 1 - centers[:, 1])
        )
        r = np.clip(r, 0, r_bound)
        
        # Pairwise distance constraints: circles cannot overlap
        for i in range(len(r)):
            for j in range(i+1, len(r)):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < r[i] + r[j] - 1e-10:
                    return np.inf  # Penalize overlap
        return -sum_r  # Minimize negative sum_r to maximize sum_r
    
    # Combine bounds: radii must be non-negative, centers must be in [0,1]^2
    r_bounds = Bounds(0, 1)
    center_bounds = Bounds(0, 1)
    
    # Optimize with L-BFGS-B
    x = np.hstack([r0, centers.flatten()])
    result = opt.minimize(
        objective,
        x,
        bounds=[
            r_bounds,
            center_bounds
        ],
        method='L-BFGS-B',
        options={'maxiter': 100}
    )
    
    # Extract final centers and radii
    r = result.x[:len(r0)]
    centers = result.x[len(r0):].reshape(-1, 2)
    
    # Final validation
    valid, msg = validate_packing(centers, r)
    if not valid:
        raise ValueError(f"Validation failed: {msg}")
    
    return (centers, r, np.sum(r))

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
