import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds

def run_packing():
    # Number of circles
    n = 26

    # Define a grid for the starting positions (hexagonal packing)
    # This is a 5-row hexagonal packing inside the unit square
    rows = 5
    cols = 5
    centers = []
    radii = []

    # Heuristically define starting positions for hexagonal packing
    for i in range(rows):
        for j in range(cols):
            x = 0.2 + j * (1 - 2 * rows / (rows + 1)) / (cols - 1)
            y = 0.1 + i * (1 - 2 * rows / (rows + 1)) / (rows - 1)
            centers.append([x, y])

    # The starting radius for each circle is 0.05
    radii = [0.05] * n

    # Now perform the optimization
    def total_radius_sum(x):
        # x is a flat array of x1, y1, r1, x2, y2, r2, ..., xn, yn, rn
        # Split into centers and radii
        x_flat = x.reshape(-1, 2)
        r_flat = x[2 * n :]

        # Check if any circle is outside the unit square
        for i in range(n):
            x_i, y_i = x_flat[i]
            r_i = r_flat[i]
            if (x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12
                or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12):
                return -np.inf  # Penalize

        # Check for overlaps
        for i in range(n):
            for j in range(i + 1, n):
                dx = x_flat[i][0] - x_flat[j][0]
                dy = x_flat[i][1] - x_flat[j][1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                if dist < r_flat[i] + r_flat[j] - 1e-12:
                    return -np.inf  # Penalize overlaps

        # Compute total radius sum
        return np.sum(r_flat)

    # Initial guess
    initial_guess = np.array([center for center in centers] + [0.05] * n)

    # Define bounds for x and y coordinates
    bounds = Bounds(np.array([0.0] * 2 * n), np.array([1.0] * 2 * n))

    # Optimization parameters
    result = opt.minimize(
        fun=total_radius_sum,
        x0=initial_guess,
        bounds=bounds,
        method='SLSQP',
        options={'maxiter': 500, 'ftol': 1e-8, 'disp': False}

    # Extract the optimal centers and radii
    centers = result.x[:2 * n].reshape(n, 2)
    radii = result.x[2 * n:]

    if np.isnan(centers).any() or np.isnan(radii).any():

    # Compute sum of radii
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
