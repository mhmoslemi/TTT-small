import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Initial placement: hexagonal packing with 5 rows
    num_circles = 26
    rows = 5
    cols = 5  # last row has 6 circles

    # Initial positions
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Horizontal spacing for hexagonal packing
    x_spacing = 0.5  # Start from 0.25 to keep within square, spread 26 circles
    y_spacing = 0.5

    for i in range(rows):
        for j in range(cols if i % 2 == 0 else cols - 1):
            col_index = j
            row_index = i
            x = 0.25 + col_index * x_spacing
            y = 0.25 + row_index * y_spacing
            centers[i * cols + j] = [x, y]
            radii[i * cols + j] = 0.05  # Initial guess

    # Define the objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize

    # Define constraints: non-overlapping and within square
    def constraint_non_overlapping(radii, centers):
        n = len(radii)
        for i in range(n):
            for j in range(i + 1, n):
                x1, y1 = centers[i]
                x2, y2 = centers[j]
                dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        return True

    # Optimization
    bounds = [(0, 0.5) for _ in range(num_circles)]
    res = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints={
            'type': 'ineq',
            'fun': lambda r: constraint_non_overlapping(r, centers),
            'jac': None
        }
    )

    radii = res.x
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
