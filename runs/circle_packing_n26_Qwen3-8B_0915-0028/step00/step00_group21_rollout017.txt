import numpy as np

def run_packing():
    # Grid parameters
    grid_rows = 5
    grid_cols = 5
    spacing = 0.25  # diameter
    start_x = 0.1
    start_y = 0.1

    # Generate positions for first 25 circles in a grid
    centers = []
    for i in range(grid_rows):
        for j in range(grid_cols):
            x = start_x + j * spacing
            y = start_y + i * spacing
            centers.append([x, y])

    # Add the 26th circle at center of the square
    centers.append([0.5, 0.5])

    # Now, calculate the maximum radius such that all circles are within the square
    # We can find the minimal distance from any circle to the boundary of the square
    # and to the other circles to find the maximum possible radius

    # First, compute the minimal distance from any circle to the square boundary
    distances_to_boundary = []
    for x, y in centers:
        dist_to_left = x
        dist_to_right = 1 - x
        dist_to_bottom = y
        dist_to_top = 1 - y
        min_dist_to_boundary = min(dist_to_left, dist_to_right, dist_to_bottom, dist_to_top)
        distances_to_boundary.append(min_dist_to_boundary)

    # The maximum radius possible is the minimum of these distances
    max_possible_radius = min(distances_to_boundary)

    # Now, check if any circles are overlapping with the (0.5, 0.5) circle
    # If so, we will adjust the radius to ensure non-overlapping condition

    # Compute the distance between the (0.5, 0.5) circle and all others
    dist_to_center = []
    for x, y in centers[:-1]:  # Exclude the center circle
        dist = np.sqrt((x - 0.5) ** 2 + (y - 0.5) ** 2)
        dist_to_center.append(dist)

    # The maximum radius such that none of the circles overlap with the center circle
    max_radius_for_center = min(dist_to_center) - 1e-12

    # Therefore, the maximum radius we can use is the minimum between the boundary and center constraints
    max_radius = min(max_possible_radius, max_radius_for_center)

    # Now, create the final centers and radii
    centers = np.array(centers)
    radii = np.full(len(centers), max_radius)

    # Sum of radii
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
