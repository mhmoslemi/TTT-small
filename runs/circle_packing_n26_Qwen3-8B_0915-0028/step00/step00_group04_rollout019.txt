import numpy as np
import scipy.optimize as optimize

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    # Initial grid-like pattern for 26 circles
    # Arranged in a hybrid layout: 5 rows with 5 and 6 circles in different rows
    # Start with 5 rows: first row with 5 circles, followed by 4 rows with 5 circles, then 1 row with 6 circles
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    
    # First row horizontally (6 circles, evenly spaced)
    x = np.linspace(0.1, 0.9, 6)
    for i in range(6):
        centers[i, 0] = x[i]
        centers[i, 1] = 0.1
        radii[i] = 0.1
    
    # Second to fifth rows (5 circles each)
    for row_idx in range(1, 5):
        y = 0.3 + row_idx * 0.2
        x = np.linspace(0.1, 0.9, 5)
        for i in range(5):
            centers[6 + row_idx * 5 + i, 0] = x[i]
            centers[6 + row_idx * 5 + i, 1] = y
            radii[6 + row_idx * 5 + i] = 0.15
    
    # Sixth row (last row with 6 circles)
    y = 0.3 + 5 * 0.2
    x = np.linspace(0.1, 0.9, 6)
    for i in range(6):
        centers[6 + 5 * 5 + i, 0] = x[i]
        centers[6 + 5 * 5 + i, 1] = y
        radii[6 + 5 * 5 + i] = 0.15

    # Define function to compute the total sum of radii and constraints
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize total

    def constraint_center_bounds(radii):
        return radii >= 0.0

    def constraint_overlap(radii):
        n = len(radii)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return -1.0
        return 0.0

    # Optimization
    bounds = [(0.0, 1.0) for _ in range(26)]
    constraints = [
        {'type': 'ineq', 'fun': constraint_center_bounds},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]

    result = optimize.minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP'
    )

    if not result.success:
        raise RuntimeError(f"Optimization failed: {result.message}")

    radii = result.x
    centers = np.array([[x, y] for x, y in centers])

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
