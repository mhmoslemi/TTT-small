import numpy as np
import cvxpy as cp

def run_packing():
    # Define variables for the centers and radii of the circles
    n = 26
    centers = cp.Variable((n, 2), name="centers")
    radii = cp.Variable(n, name="radii", non_negative=True)

    # Define the optimization problem
    objective = cp.Maximize(cp.sum(radii))

    # Constraints
    constraints = []

    # Each circle must be inside the unit square [0, 1] x [0, 1]
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        constraints.append(x - r <= 1e-12)
        constraints.append(x + r <= 1 + 1e-12)
        constraints.append(y - r <= 1e-12)
        constraints.append(y + r <= 1 + 1e-12)

    # No two circles may overlap
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist_sq = dx**2 + dy**2
            constraints.append(dist_sq >= (radii[i] + radii[j])**2 - 1e-12)

    # Solve the optimization problem
    problem = cp.Problem(objective, constraints)
    problem.solve()

    # Extract results
    centers_val = centers.value
    radii_val = radii.value
    sum_radii = np.sum(radii_val)

    return (centers_val, radii_val, sum_radii)
