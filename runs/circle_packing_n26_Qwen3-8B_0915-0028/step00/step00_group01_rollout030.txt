import numpy as np
from scipy.optimize import minimize
import random

def run_packing():
    # Generate initial positions using a hexagonal grid
    n = 26

    # Initial configuration: Hexagonal grid with 4 rows (up to 7 circles per row)
    # The rows alternate in offset to mimic hexagonal packing
    positions = []
    r_initial = 0.05  # Initial small uniform radius

    # Create a hexagonal grid with rows
    rows = 4
    max_circles_in_row = 7

    for row_idx in range(rows):
        y = row_idx * np.sqrt(3) * r_initial  # Vertical spacing
        if row_idx % 2 == 0:
            # Even rows align with the x-axis
            for col_idx in range(max_circles_in_row):
                x = col_idx * 2 * r_initial
                positions.append((x, y))
        else:
            # Odd rows are offset
            for col_idx in range(max_circles_in_row - 1):
                x = (col_idx + 0.5) * 2 * r_initial
                positions.append((x, y))

    # Ensure positions stay within [0,1]^2 (normalize to unit square)
    positions = np.array(positions)
    positions /= np.max(positions, axis=0)  # Normalize to [0,1]^2

    # Start with equal radii
    radii = np.full(n, r_initial)

    # Concatenate positions and radii into a flat vector for optimization
    x0 = np.concatenate([positions.ravel(), radii])

    # Objective function: Maximize total radius (negative for minimization)
    def objective(x):
        radii = x[-n:]
        return -np.sum(radii)  # Negative because we are minimizing

    # Constraints: Each circle must stay within the unit square
    def constraint_square(x):
        pos = x[:-n].reshape(n, 2)
        rad = x[-n:]
        r = rad
        for i in range(n):
            x_i, y_i = pos[i]
            if x_i - r[i] < 0 or x_i + r[i] > 1 or y_i - r[i] < 0 or y_i + r[i] > 1:
                return -1
        return 0

    # Constraints: Circles must not overlap
    def constraint_overlap(x):
        pos = x[:-n].reshape(n, 2)
        rad = x[-n:]
        for i in range(n):
            for j in range(i + 1, n):
                dx = pos[i, 0] - pos[j, 0]
                dy = pos[i, 1] - pos[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < rad[i] + rad[j] - 1e-12:
                    return -1
        return 0

    # Set up optimization problem
    cons = [
        {'type': 'ineq', 'fun': constraint_square},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]

    # Run optimization
    result = minimize(
        objective,
        x0,
        method='SLSQP',
        constraints=cons,
        bounds=[(0, 1) for _ in range(2 * n)] + [(0, 1) for _ in range(n)]
    )

    # Extract results
    x_opt = result.x
    positions_opt = x_opt[:-n].reshape(n, 2)
    radii_opt = x_opt[-n:]
    sum_radii = np.sum(radii_opt)

    return positions_opt, radii_opt, sum_radii
