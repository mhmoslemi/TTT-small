import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    # Initial layout: Hexagonal packing
    rows = 5
    cols = 5
    radii = np.full(26, 0.04)
    centers = np.zeros((26, 2))
    idx = 0
    for r in range(rows):
        for c in range(cols):
            x = c + 0.5 * (r % 2)
            y = r
            centers[idx] = [x, y]
            idx += 1

    # Refine with optimization
    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        # Ensure non-overlapping
        for i in range(26):
            for j in range(i+1, 26):
                dist = cdist([centers[i]], [centers[j]], 'euclidean')[0]
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        # Ensure circles are within square
        for i in range(26):
            x, y = centers[i]
            if x - radii[i] < 0 or x + radii[i] > 1 or y - radii[i] < 0 or y + radii[i] > 1:
                return False
        return True

    bounds = [(0, 0.1)] * 26

    result = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'ftol': 1e-8, 'maxiter': 100}
    )

    # Extract final centers and radii
    centers_final = centers
    radii_final = result.x

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
