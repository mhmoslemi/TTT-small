import numpy as np
import scipy.optimize as opt

def run_packing():
    """
    Generate a packing of 26 circles within a unit square [0,1]x[0,1],
    maximizing the sum of the radii while ensuring no overlaps and staying within bounds.
    """

    # Initial grid-based hexagonal packing
    num_circles = 26
    grid_rows = 5
    grid_cols = 6
    circle_radius = 0.05  # Initial estimate based on unit square
    row_height = np.sqrt(3) / 2 * circle_radius  # spacing between rows in hexagonal packing

    # Efficiently generate hexagonal grid
    centers = []
    for i in range(grid_rows):
        for j in range(grid_cols):
            x = j * (1 + circle_radius) / (grid_cols - 1)
            y = i * row_height / (grid_rows - 1)
            centers.append([x, y])
    centers = np.array(centers)

    # Objective function: maximize sum of (radius^2) for non-overlapping circles
    def objective(radii):
        return -np.sum(radii)  # Minimize negative to maximize sum

    # Constraints:
    # 1. No overlap between circles: dist(i,j) >= r_i + r_j - 1e-8
    # 2. All circles must be within [0,1]x[0,1]

    def constraint_overlap(radii):
        n = len(radii)
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                constraints.append(dist - (radii[i] + radii[j]) >= -1e-8)
        return constraints

    def constraint_boundary(radii):
        n = len(radii)
        constraints = []
        for i in range(n):
            x, y = centers[i]
            constraints.append(x - radii[i] >= -1e-8)
            constraints.append(x + radii[i] <= 1 + 1e-8)
            constraints.append(y - radii[i] >= -1e-8)
            constraints.append(y + radii[i] <= 1 + 1e-8)
        return constraints

    # Check that centers are within square
    if np.any(centers < -1e-8) or np.any(centers > 1 + 1e-8):
        raise ValueError("Initial centers are outside unit square.")

    # Initial guess for radii
    radii_initial = np.full(num_circles, circle_radius)

    # Optimization problem
    bounds = [(0.0, 1.0)] * num_circles
    constraints = constraint_overlap(radii_initial) + constraint_boundary(radii_initial)

    # Use gradient-based optimization
    result = opt.minimize(
        objective,
        radii_initial,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'ftol': 1e-9, 'maxiter': 1000}
    )

    if not result.success:
        raise RuntimeError(f"Optimization failed: {result.message}")

    radii = result.x
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
