import numpy as np
import math
from scipy.optimize import minimize


def run_packing():
    # Generate a grid of points in a hexagonal packing
    def get_hexagonal_grid(n_rows, n_cols, r):
        centers = []
        for row in range(n_rows):
            for col in range(n_cols):
                x = r + col * 2 * r
                y = r + row * r * math.sqrt(3)
                if row % 2 == 1:
                    x += r
                centers.append((x, y))
        return np.array(centers)

    # Check if point is inside the unit square
    def is_inside(point, r):
        x, y = point
        return (x - r >= 0) and (x + r <= 1) and (y - r >= 0) and (y + r <= 1)

    # Starting configuration: 4 rows of 6 circles each, with radius 0.05
    n_rows = 4
    n_cols = 6
    initial_r = 0.05
    centers = get_hexagonal_grid(n_rows, n_cols, initial_r)
    radii = np.full(n_rows * n_cols, initial_r)

    # Define the objective function to maximize total radius sum
    def objective(radii):
        return -np.sum(radii)  # Minimize negative of sum

    # Define constraints:
    def constraint1(radii):
        # Ensure circles stay within the square
        for i in range(len(radii)):
            x, y = centers[i]
            if not is_inside((x, y), radii[i]):
                return False
        return True

    def constraint2(radii):
        # Ensure circles do not overlap
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                dx = centers[i][0] - centers[j][0]
