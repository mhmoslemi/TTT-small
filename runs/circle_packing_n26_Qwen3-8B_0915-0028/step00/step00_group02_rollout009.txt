import scipy.optimize as optimize

def run_packing():

    # Create an initial grid with 5 rows of 5 circles, with some extra circles to reach 26
    # Grid layout:
    # Row 0: 4 circles
    # Row 1: 5 circles
    # Row 2: 5 circles
    # Row 3: 5 circles
    # Row 4: 4 circles
    # Total: 23 circles, so add 3 more in a slight offset

    def initial_centers():
        centers = np.zeros((26, 2))
        idx = 0
        for i in range(5):
            for j in range(5):
                x = j / 4.0
                y = i / 4.0
                if i % 2 == 0:
                    centers[idx] = [x, y]
                else:
                    centers[idx] = [x + 0.25, y]
                idx += 1

        centers[23] = [0.25, 0.75]
        centers[24] = [0.75, 0.75]

    centers = initial_centers()
    radii = np.ones(26) * 0.05  # Initial guess for radii

    def objective(radii, centers):

        return -np.sum(radii)

    def constraints(radii, centers):
        # Constraints: circles do not overlap and stay within the unit square
        n = len(radii)
        for i in range(n):
            x, y = centers[i]
            if x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12 or y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12:

                dx = centers[i, 0] - centers[j, 0]

                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False

        fun=lambda x: objective(x, centers),
