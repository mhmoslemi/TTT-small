from scipy.optimize import minimize

        for i in range(num_rows):
            for j in range(num_cols):

                y = i + 0.5 * (i % 2)
                centers.append([x, y])
        return np.array(centers)

    # Parameters

    def pairwise_distances(c1, c2):
        return np.sqrt(np.sum((c1 - c2) ** 2))

    # Objective function: Maximize sum of radii, with penalties for overlap and boundary violation
    def objective(x):
        centers = x[:2 * num_circles].reshape(num_circles, 2)
        radii = x[2 * num_circles:]

        # Check for out-of-bounds
        for i in range(num_circles):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return float("-inf")  # Penalize out-of-bounds placement

        # Compute total radius sum
        total = np.sum(radii)

        # Add penalty for circle overlaps
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dist = pairwise_distances(centers[i], centers[j])
                if dist < radii[i] + radii[j] - 1e-12:
                    return total - 1000  # Heavy penalty for overlap

        # Add penalty for circles being too close to edges (to encourage boundary use)
        edge_penalty = 0

            edge_penalty += max(0, (x_i - r_i) + 1e-12) + max(0, (x_i + r_i) - 1 + 1e-12) + \

        return total - 100 * edge_penalty  # Penalize circles staying away from edges

    # Run optimization
    result = minimize(
