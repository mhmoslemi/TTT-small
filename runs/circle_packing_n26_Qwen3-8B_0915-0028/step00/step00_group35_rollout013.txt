import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize

def run_packing():
    # Define the number of circles
    n = 26

    # Define the hexagonal grid layout
    # 5 rows of 5 circles, and 1 row of 6 circles
    rows = 5
    col1 = 5
    col2 = 6

    # Initialize centers and radii
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Build the initial grid
    def build_grid():
        i = 0
        x = np.linspace(0, 1, col1 + 1)[1:-1]
        y = np.linspace(0, 1, rows + 1)[1:-1]
        for row in range(rows):
            for col in range(col1):
                centers[i, 0] = x[col]
                centers[i, 1] = y[row]
                i += 1
        x = np.linspace(0, 1, col2 + 1)[1:-1]
        y = np.linspace(0, 1, rows + 1)[1:-1]
        for row in range(rows):
            for col in range(col2):
                centers[i, 0] = x[col]
                centers[i, 1] = y[row]
                i += 1

    build_grid()

    # First-pass radii: based on area per circle
    # For hexagonal packing, the area per circle is about (sqrt(3)/2) * r^2
    # Total area = 1, so total area per circle = 1/n
    # => (sqrt(3)/2) * r^2 = 1/n => r = sqrt(2/(n*sqrt(3)))
    initial_radius = np.sqrt(2 / (n * np.sqrt(3)))
    radii = np.full(n, initial_radius)

    # Resize to fit boundaries
    for i in range(n):
        x, y = centers[i]
        if x - radii[i] < 0 or x + radii[i] > 1:
            radii[i] = np.minimum(radii[i], 1 - x)
            radii[i] = np.minimum(radii[i], x)
        if y - radii[i] < 0 or y + radii[i] > 1:
            radii[i] = np.minimum(radii[i], 1 - y)
            radii[i] = np.minimum(radii[i], y)

    # Optimization
    def objective(radii):
        return -np.sum(radii)  # Maximize sum of radii

    def constraints(radii):
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        return True

    # Initial guess
    x0 = radii

    # Run optimization
    result = minimize(
        objective,
        x0,
        bounds=[(0, 1)] * n,
        constraints=[{"type": "ineq", "fun": constraints}],
        method="L-BFGS-B"
    )

    centers = np.array(centers)
    radii = result.x

    # Final check
    assert not np.isnan(radii).any(), "NaN in radii"
    assert not np.isnan(centers).any(), "NaN in centers"

    # Final sum
    sum_radii = np.sum(radii)

    return (centers, radii, sum_radii)
