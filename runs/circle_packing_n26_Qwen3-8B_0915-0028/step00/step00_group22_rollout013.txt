```python
import numpy as np
from scipy.optimize import minimize
import matplotlib.pyplot as plt

def run_packing():
    # Initial guess: 13 circles on top and 13 on bottom, equally spaced
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    
    # Initial positions for the circles
    for i in range(26):
        row = i // 13
        col = i % 13
        x = col / 12
        y = 0.5 if row == 0 else 0.5 + 0.5 / (13 - 1)
        centers[i] = [x, y]
    
    # Initial radii for the top and bottom rows
    # We'll start with equal radii for each row
    r_top = 0.15
    r_bottom = 0.15
    for i in range(26):
        if i < 13:
            radii[i] = r_top
        else:
            radii[i] = r_bottom
    
    # Objective function: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)
    
    # Constraints: non-overlapping and within bounds
    def constraint_non_overlap(radii):
        # Check non-overlapping between circles
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -1  # Violation
        return 0
    
    # Create a bounds array for radii
    bounds = [(0, 1) for _ in range(26)]
    
    # Optimization
    result = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint_non_overlap}
        ],
        options={'disp': False}
    )
    
    # Extract solution
    radii_opt = result.x
    sum_radii = np.sum(radii_opt)
    
    return centers, radii_opt, sum_radii
```
