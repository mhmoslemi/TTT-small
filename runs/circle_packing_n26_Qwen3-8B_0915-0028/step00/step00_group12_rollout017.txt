import scipy.optimize as optimize

def run_packing():
    # Base grid of 26 circles in a hexagonal grid in a unit square
    # This is an initial approximate layout
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # This is a suboptimal initial configuration but enough to start the optimization
    grid = []
    for row in range(3):
        for col in range(9):
            grid.append((col / 8, row / 2))

        for col in range(8):
            grid.append((col / 7, row / 2))
    centers = np.array(grid)
    radii = np.full(len(centers), 0.05)

    def energy(radii, centers):
        energy = 0.0
        for i in range(len(centers)):

                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-6:
                    energy += (radii[i] + radii[j] - dist) ** 2
        return energy

    def optimization_function(x):
        # x contains all variables: first half for radii, second for centers
        x_radii = x[:num_circles]
        x_centers = x[num_circles:]
        centers = np.reshape(x_centers, (num_circles, 2))
        radii = x_radii

            if centers[i, 0] - radii[i] < -1e-6 or centers[i, 0] + radii[i] > 1 + 1e-6:
                return float('inf')

                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-6:
                    return float('inf')

        return energy(radii, centers)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
