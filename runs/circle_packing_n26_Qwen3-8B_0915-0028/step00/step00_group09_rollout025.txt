import numpy as np
import cvxpy as cp

def run_packing():
    # Define initial hexagonal grid
    num_circles = 26
    grid_rows = 5
    grid_cols = 6

    # Define the spacing for hexagonal packing
    spacing = 0.18  # approximate spacing to fit in the unit square

    # Generate initial positions in a hexagonal grid
    centers = []
    for row in range(grid_rows):
        for col in range(grid_cols):
            x = col * spacing
            y = row * (np.sqrt(3) / 2) * spacing
            centers.append([x, y])

    centers = np.array(centers)

    # Define variables
    radii = cp.Variable(num_circles)
    obj = cp.Minimize(-cp.sum(radii))  # Maximize sum of radii

    # Constraints
    constraints = []
    for i in range(num_circles):
        # Ensure each circle fits inside the square
        x, y = centers[i]
        constraints.append(x - radii[i] >= -1e-8)
        constraints.append(x + radii[i] <= 1 + 1e-8)
        constraints.append(y - radii[i] >= -1e-8)
        constraints.append(y + radii[i] <= 1 + 1e-8)

    # Ensure no two circles overlap
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            dist = np.linalg.norm(centers[i] - centers[j])
            constraints.append(dist >= radii[i] + radii[j] - 1e-8)

    # Solve the optimization problem
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.SCS, verbose=False)

    if not prob.status == "optimal":
        raise RuntimeError(f"Optimization failed with status: {prob.status}")

    radii_vals = radii.value
    centers_vals = centers

    # Calculate sum of radii
    sum_radii = np.sum(radii_vals)

    return (centers_vals, radii_vals, sum_radii)
