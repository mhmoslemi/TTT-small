from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():

    n = 26
    rows = int(np.sqrt(n))

    initial_centers = []

        for j in range(cols):
            x = j / (cols - 1) if cols > 1 else 0.5
            y = i / (rows - 1) if rows > 1 else 0.5

    radii = np.full(n, 0.1)

    def objective(x):

        centers_flat = x[:n*2].reshape(n, 2)
        radii_flat = x[n*2:]

        # 2. Distances between centers must be at least sum of radii
        # 3. Radii must be non-negative
        # 4. We use constraint violation to scale the objective
        
        # Violation of boundary constraints
        boundary_violation = 0
        for i in range(n):
            x, y = centers_flat[i]
            r = radii_flat[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:

        return -np.sum(radii_flat) + penalty
