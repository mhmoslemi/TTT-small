import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from typing import Tuple, List

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    def generate_initial_hexagonal(n: int) -> np.ndarray:
        """Generate initial hexagonal packing of n circles in a square."""
        centers = []
        row_count = int(np.sqrt(n))
        col_count = int(np.ceil(n / row_count))
        
        for i in range(row_count):
            for j in range(col_count):
                x = j + 0.5 * (i % 2)
                y = i * (np.sqrt(3)/2) * (1 / (row_count - 1)) if i < row_count - 1 else 0.5
                centers.append([x, y])
        centers = np.array(centers)
        return centers

    def objective(radii: np.ndarray) -> float:
        """Objective function to maximize sum of radii."""
        return -np.sum(radii)  # Minimize negative sum

    def constraints(centers: np.ndarray, radii: np.ndarray) -> List:
        """Return constraints for optimization."""
        constraints = []
        n = len(centers)
        for i in range(n):
            # Constraint: Circle i must be inside the unit square
            constraints.append((centers[i, 0] - radii[i], 0.0, 1.0 + 1e-12))
            constraints.append((centers[i, 1] - radii[i], 0.0, 1.0 + 1e-12))
            constraints.append((radii[i] - centers[i, 0], -1.0 - 1e-12, 0.0))
            constraints.append((radii[i] - centers[i, 1], -1.0 - 1e-12, 0.0))

        for i in range(n):
            for j in range(i + 1, n):
                # Distance between centers i and j must be >= radii[i] + radii[j]
                constraints.append((np.sqrt(np.sum((centers[i] - centers[j])**2)) - radii[i] - radii[j], 1e-12, np.inf))

        return constraints

    def optimize_grid(centers: np.ndarray) -> np.ndarray:
        """Run local optimization to maximize sum of radii with constraints."""
        bounds = Bounds([0.0] * len(centers), [np.inf] * len(centers))
        res = minimize(
            objective,
            bounds=bounds,
            method="SLSQP",
            constraints=constraints(centers, radii=np.zeros(len(centers))),
            options={"ftol": 1e-10, "maxiter": 100}
        )
        return res.x

    best_sum = 0.0
    best_centers = None
    best_radii = None

    # Generate initial hexagonal grid with 26 circles
    trial_centers = generate_initial_hexagonal(26)
    trial_radii = np.ones(len(trial_centers)) * 0.05  # Initial small radii

    # Perform local optimization
    optimized_radii = optimize_grid(trial_centers)
    sum_radii = np.sum(optimized_radii)
    best_centers = trial_centers
    best_radii = optimized_radii

    # Set best_sum
    best_sum = sum_radii

    return best_centers, best_radii, best_sum
