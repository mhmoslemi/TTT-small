import scipy.optimize as optimize
import scipy.spatial.distance as distance

def run_packing():

    n = 26
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.05  # Initial radius guess

    # Step 1: Place circles in a grid-like pattern
    grid_size = int(np.ceil(n ** 0.5))
    for i in range(n):
        row = i // grid_size
        col = i % grid_size
        centers[i] = [col / (grid_size - 1), row / (grid_size - 1)]
        radii[i] = 0.05

    # Define the objective function to maximize the sum of the radii (minimize negative sum)
    def objective(x):
        # x is a flattened array of [x0, y0, r0, x1, y1, r1, ...]

        # Ensure centers are within [0,1]^2

            if centers_flat[i, 0] < 0 or centers_flat[i, 0] > 1 or \
               centers_flat[i, 1] < 0 or centers_flat[i, 1] > 1:
                return -np.inf

        # Ensure radii are non-negative
        for i in range(n):
            if radii_flat[i] < 0:
                return -np.inf

        # Ensure no overlaps
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii_flat[i] + radii_flat[j] - 1e-8:

        # Return the negative of the total radius to maximize it
