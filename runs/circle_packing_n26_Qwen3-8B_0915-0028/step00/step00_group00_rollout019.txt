import numpy as np

from scipy.optimize import minimize

def run_packing():
    def initial_config(n_circles):
        """
        Generate an initial approximate hexagonal packing with n_circles.
        """
        rows = int(np.sqrt(n_circles))
        cols = n_circles // rows
        if n_circles % rows != 0:
            cols += 1

        centers = []
        for i in range(rows):
            for j in range(cols):
                x = (j + 0.5) * (1 / (cols - 1)) if cols > 1 else 0.5
                y = (i + 0.5) * (1 / (rows - 1)) if rows > 1 else 0.5
                centers.append((x, y))
        centers = np.array(centers)
        return centers

    n_circles = 26
    centers = initial_config(n_circles)

    def obj_func(radii):
        """
        Objective function to minimize (since we are maximizing sum of radii),
        but we are minimizing the negative of the sum of radii.
        """
        return -np.sum(radii)

    def constraint_overlap(radii):
        """
        Constraint: distance between centers >= sum of radii for each pair.
        """
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        return True

    # Define bounds for each radius: must be non-negative
    bounds = [(0, 1)] * n_circles

    # Use Sequential Least Squares Programming (SLSQP) for constrained optimization
    result = minimize(
        obj_func,
        x0=np.ones(n_circles),
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint_overlap},
        ],
        method='SLSQP'
    )

    if not result.success:

    radii = result.x
    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
