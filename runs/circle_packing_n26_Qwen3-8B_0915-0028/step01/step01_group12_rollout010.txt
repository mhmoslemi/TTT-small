import numpy as np
import scipy.optimize as optimize

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    # Generate a hexagonal packing with 26 circles
    def generate_hexagonal_packing(n_circles):
        # Arrange in hexagonal pattern
        centers = []
        radii = []
        for i in range(n_circles):
            row = i // 6
            col = i % 6
            x = 0.1 + col * 0.15
            y = 0.1 + row * 0.15
            # Adjust for density and square boundaries
            if row % 2 == 1:
                x += 0.075
            centers.append([x, y])
            radii.append(0.05)
        return np.array(centers), np.array(radii)

    centers, radii = generate_hexagonal_packing(26)

    # Optimization function to maximize sum of radii
    def objective(x):
        new_radii = x[:26]
        dx, dy = x[26], x[27]

        # Adjust centers based on dx and dy
        new_centers = centers + np.array([dx, dy])

        # Constraint: all circles must stay within [0, 1]^2
        # Constraint: distances between centers must be at least sum of radii
        if not validate_packing(new_centers, new_radii)[0]:
            return np.inf

        # Goal: maximize sum of radii, so we minimize negative sum
        return -np.sum(new_radii)

    # Initial guess: same radii as in the original, with small shifts
    x0 = np.concatenate((radii, [0.01, 0.01]))

    # Bounds: radii between 0 and 0.2, dx and dy between -0.1 and 0.1
    bounds = [(0, 0.2) for _ in range(26)] + [(-0.1, 0.1), (-0.1, 0.1)]

    # Optimize
    result = optimize.minimize(objective, x0, bounds=bounds, method='L-BFGS-B')

    if not result.success:

    # Extract the result
    best_radii = result.x[:26]

    best_centers = centers + np.array([dx, dy])
