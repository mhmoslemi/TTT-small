import numpy as np
import random
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import cdist

def run_packing():
    n = 26
    centers = []
    radii = []

    # Step 1: Random initial placement for optimization
    for _ in range(n):
        x = np.random.uniform(0.1, 0.9)
        y = np.random.uniform(0.1, 0.9)
        r = np.random.uniform(0.01, 0.1)
        centers.append([x, y])
        radii.append(r)

    # Convert to numpy arrays
    centers = np.array(centers)
    radii = np.array(radii)

    # Objective function: Maximize total area (sum of pi * r^2)
    def objective(x):
        x = x.reshape((n, 2))
        r = x[:, 2]
        return -np.sum(np.pi * r ** 2)  # Minimize negative of total area

    # Constraint 1: Ensure each circle is inside the unit square
    def square_constraint(x):
        x = x.reshape((n, 2))
        r = x[:, 2]
        return np.array([x[i, 0] - r[i] >= -1e-12 and x[i, 0] + r[i] <= 1 + 1e-12 and x[i, 1] - r[i] >= -1e-12 and x[i, 1] + r[i] <= 1 + 1e-12 for i in range(n)])

    # Constraint 2: No overlapping between circles
    def overlap_constraint(x):
        x = x.reshape((n, 2))
        r = x[:, 2]
        dist = cdist(x[:, :2], x[:, :2])
        overlap = np.array([dist[i, j] < r[i] + r[j] - 1e-12 for i in range(n) for j in range(i + 1, n)])
        return np.array([not overlap[i] for i in range(len(overlap))])

    # Flatten centers and radii for optimization
    initial_guess = np.concatenate([centers.reshape(n, 2), radii.reshape(n, 1)], axis=1).flatten()
    bounds = Bounds(np.array([0.0, 0.0, 0.0] * n), np.array([1.0, 1.0, 1.0] * n))

    # Run optimization
    result = minimize(
        objective,
        initial_guess,
        bounds=bounds,
        constraints=[

        ],
        method='SLSQP',
        options={'maxiter': 500, 'ftol': 1e-12}

    return (centers_opt, radii_opt.flatten(), sum_radii)
