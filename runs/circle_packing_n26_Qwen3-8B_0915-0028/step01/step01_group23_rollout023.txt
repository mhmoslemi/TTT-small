import numpy as np
import scipy.optimize as optimize

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():

    # This uses a more optimized layout with variable radii and edge/corner optimization

    # Hexagonal layout: 6 rows with decreasing number of circles
    num_rows = 6
    num_cols = [5, 5, 5, 5, 5, 5]  # For 6 rows
    center_x = np.linspace(0.1, 0.9, 5)  # 5 columns in x direction
    spacing = 0.21

    centers = []
    for r in range(num_rows):
        if r % 2 == 0:
            # Even rows (left-aligned)
            for c in range(num_cols[r]):

                y = 0.1 + r * spacing
                centers.append([x, y])

            for c in range(num_cols[r]):

                centers.append([x, y])

    centers = np.array(centers)

    radii_initial = np.array([0.05] * len(centers))

    # Optimization parameters
    def objective(x):
        # x[:len(centers)] = radii
        # x[len(centers):] = shift in x and y directions
        new_radii = x[:len(centers)]
        dx, dy = x[len(centers)], x[len(centers) + 1]

        new_centers = centers + np.array([dx, dy])

        return -np.sum(new_radii)

    # Initial guess: same radii as in the original, with small shifts
    x0 = np.concatenate((radii_initial, [0.0, 0.0]))
