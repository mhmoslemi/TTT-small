import numpy as np
import scipy.optimize as optimize

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    # Create a hexagonal close packing of 26 circles
    n = 26
    centers = []
    
    # First row
    row = []
    for i in range(13):
        x = i / 12
        y = 0.05
        row.append([x, y])
    
    # Second row
    row2 = []
    for i in range(12):
        x = i / 12 + 0.5 / 12
        y = 0.3
        row2.append([x, y])
    
    # Third row
    row3 = []
    for i in range(12):
        x = i / 12 + 0.5 / 12
        y = 0.5
        row3.append([x, y])
    
    # Fourth row
    row4 = []
    for i in range(11):
        x = i / 11 + 0.5 / 11
        y = 0.7
        row4.append([x, y])
    
    # Fifth row
    row5 = []
    for i in range(10):
        x = i / 10 + 0.5 / 10
        y = 0.85
        row5.append([x, y])
    
    centers = row + row2 + row3 + row4 + row5

    # Create the initial radii guess
    radii = [0.1] * n

    # Optimization bounds
    bounds = [(0, 0.2)] * n

    # Objective function to maximize sum of radii
    def objective(x):
        # x[0:n] = radii
        new_radii = x[:n]
        # Constraint: all circles must stay within [0, 1]^2
        # Constraint: distances between centers must be at least sum of radii
        if not validate_packing(np.array(centers), new_radii)[0]:
            return np.inf  # Penalize invalid configurations

        # Our goal is to maximize sum of radii
        return -np.sum(new_radii)

    # Perform the optimization
    result = optimize.minimize(
        objective,
        x0=radii,
        bounds=bounds,
        method='SLSQP',
        options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-10}
    )

    if not result.success:
        raise ValueError(f"Optimization failed: {result.message}")

    best_radii = result.x
    best_centers = np.array(centers)
    sum_radii = np.sum(best_radii)
    return best_centers, best_radii, sum_radii
