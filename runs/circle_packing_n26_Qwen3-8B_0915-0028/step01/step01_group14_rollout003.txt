import numpy as np
import scipy.optimize as optimize

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    # Define a hexagonal packing with 5 rows of 5 or 6 circles (approx 26 circles)
    num_rows = 5
    num_cols = 5
    offset = 0.5  # Offset for staggered rows in hexagonal packing

    # Generate centers for a hexagonal packing
    centers = []
    for row in range(num_rows):
        for col in range(num_cols):
            if row % 2 == 0:
                x = col * (1.0 / (num_cols - 1)) + 0.5 / (num_cols - 1)
            else:
                x = col * (1.0 / (num_cols - 1)) + 0.5 / (num_cols - 1) + offset
            y = row / (num_rows - 1)
            centers.append((x, y))

    # Add the missing circle for the odd row
    if num_rows % 2 == 1:
        centers.append((0.5, 0.5))

    # Convert centers to numpy array and shuffle to ensure a better initial guess
    centers = np.array(centers)
    np.random.shuffle(centers)

    # Initial guess: all circles have the same radius
    radius_guess = 0.15
    radii = np.full(centers.shape[0], radius_guess)

    # Function to compute the objective: maximize sum of radii
    def objective(x):
        new_radii = x
        if not validate_packing(centers, new_radii)[0]:
            return np.inf
        return -np.sum(new_radii)

    # Optimization bounds: radii between 0 and 0.25 (safeguard against going out of bounds)
    bounds = [(0, 0.25) for _ in range(centers.shape[0])]

    # We optimize the radii of the circles
    x0 = radii
    result = optimize.minimize(objective, x0, bounds=bounds, method='L-BFGS-B')

    if not result.success:
        raise ValueError(f"Optimization failed: {result.message}")

    # Extract the result
    best_radii = result.x
    best_centers = centers
    sum_radii = np.sum(best_radii)

    return best_centers, best_radii, sum_radii
