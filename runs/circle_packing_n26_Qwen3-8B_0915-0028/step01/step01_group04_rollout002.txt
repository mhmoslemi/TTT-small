import random
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():

    n = 26

    # Initial grid-based placement of circles in a square grid
    grid_size = int(np.ceil(n**0.5))
    grid = []
    radii = []

    # Generate a grid of positions
    for i in range(grid_size):
        for j in range(grid_size):
            if i * grid_size + j < n:
                x = j / (grid_size - 1)
                y = i / (grid_size - 1)
                grid.append([x, y])

    # Now, pass this initial grid to the optimization process

    # Initial guess for radii: all 0.05
    initial_radius_guess = np.full(n, 0.05)

    # Set up bounds for radii: non-negative
    radius_bounds = Bounds(0, 1)

    def objective_function(radii_guess):
        # Objective is to maximize the sum of radii
        return -np.sum(radii_guess)

    def constraints(radii_guess):
        # Enforce that the circles do not overlap
        # And fit within the unit square
        centers = np.array(grid)
        n = len(centers)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                if dist < radii_guess[i] + radii_guess[j] - 1e-12:
                    return False
        for i in range(n):
            x, y = centers[i]
            r = radii_guess[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:

        return True

    # Optimization with constraints
    result = minimize(
