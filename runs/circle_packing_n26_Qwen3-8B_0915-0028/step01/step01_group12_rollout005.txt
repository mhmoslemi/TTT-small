import cvxpy as cp

def run_packing():

    r = cp.Variable(n, non_negative=True)

    objective = cp.Maximize(cp.sum(r))

    # Constraints:
    # 1. Circle centers must be within the unit square
    #    This allows for some flexibility for maximum packing, hence we don't impose strict boundaries
    # 2. Circle boundaries must not overlap: distance between centers >= r1 + r2

    #    This ensures circles staying in square
