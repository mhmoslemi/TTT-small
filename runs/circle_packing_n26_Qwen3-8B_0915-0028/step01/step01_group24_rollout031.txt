import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Define a hexagonal packing with a dense grid of 26 circles
    side_length = 0.2  # Initial spacing
    grid_start = 0.15
    grid_end = 0.85
    num_rows = 5
    num_cols = 6  # Total 26 circles

    # Generate hexagonal grid
    x = np.linspace(grid_start, grid_end, num_cols)
    y = np.linspace(grid_start, grid_end, num_rows)
    centers = []
    for i in range(num_rows):
        for j in range(num_cols):
            if i % 2 == 0:
                x_i = x[j]
            else:
                x_i = x[j] + (x[1] - x[0]) / 2
            y_i = y[i]
            centers.append([x_i, y_i])
    centers = np.array(centers[:26])

    # Initial radii based on spacing
    initial_radii = np.full(26, side_length / 2)

    # Optimization function
    def objective(params):
        centers = params[:26*2].reshape(26, 2)
        radii = params[26*2:]
        valid = True

        # Check boundary constraints
        for i in range(26):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                    or y - r < -1e-12 or y + r > 1 + 1e-12):
                valid = False
                break

        if not valid:
            return np.inf

        # Check non-overlapping
        for i in range(26):
            for j in range(i+1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist_sq = dx**2 + dy**2
                if dist_sq < (radii[i] + radii[j])**2 - 1e-12:
                    return np.inf

        return -np.sum(radii)  # Maximize sum of radii

    # Initial guess
    params = np.concatenate([centers.flatten(), initial_radii])

    # Define bounds: circle centers must stay within [0, 1]^2, radii >= 0
    bounds = []
    for i in range(26):
        bounds.append((0, 1))  # x-coordinate
        bounds.append((0, 1))  # y-coordinate
    for i in range(26):
        bounds.append((0, 1 - 1e-12))  # radius

    # Optimization with constraints
    result = optimize.minimize(
        objective,
        params,
        bounds=bounds,
        method='SLSQP',
        options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-8}
    )

    if not result.success:
        raise ValueError(f"Optimization failed: {result.message}")

    centers = result.x[:26*2].reshape(26, 2)
    radii = result.x[26*2:]

    return centers, radii, np.sum(radii)
