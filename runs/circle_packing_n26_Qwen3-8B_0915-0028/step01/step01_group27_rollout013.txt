import scipy.optimize as optimize

    bounds = []

        valid = np.all((centers[:, 0] >= 0) & (centers[:, 0] <= 1) &
                      (centers[:, 1] >= 0) & (centers[:, 1] <= 1) &
                      (radii >= 0))

            return np.inf
