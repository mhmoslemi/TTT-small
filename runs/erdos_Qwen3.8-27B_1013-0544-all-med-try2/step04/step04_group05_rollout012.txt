import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -400, 400)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def slsqp_optimize(h, target, dx, beta, max_iter):
    n = len(h)
    bounds = Bounds(np.zeros(n), np.ones(n))
    def obj_fn(h_in):
        c = np.correlate(h_in, 1.0 - h_in, mode="full")
        return smooth_max(c, beta) * dx
    def con_fn(h_in):
        return h_in.sum() - target
    def jac_fn(h_in):
        return np.ones(n)
    try:
        res = minimize(obj_fn, h, method='SLSQP', bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': con_fn, 'jac': jac_fn}],
                      options={'maxiter': max_iter, 'ftol': 1e-18})
        h_new = project_h(res.x, target)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, compute_c5(h)

def subgrad_compute(best, m_act, n):
    corr = np.correlate(best, 1.0 - best, mode="full")
    top = np.argsort(corr)[::-1][:m_act]
    lags = [int(i) - (n - 1) for i in top]
    g = np.zeros(n)
    for lag in lags:
        if lag == 0:
            g += (1.0 - 2.0 * best)
        elif lag > 0:
            g[lag:] += (1.0 - best[:-lag])
            g[:n - lag] -= best[lag:]
        else:
            m = -lag
            g[:n - m] += (1.0 - best[m:])
            g[m:] -= best[:n - m]
    return g, corr

def nesterov_subgrad(h, target, time_fn, max_iters, m_act, lr_scale):
    n = len(h)
    x = h.copy()
    y = h.copy()
    best = h.copy()
    best_c5 = compute_c5(best)
    no_improve = 0
    t_k = 1.0
    base_lr = lr_scale
    for it in range(max_iters):
        if time_fn() < 2:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        beta_n = (t_k - 1.0) / t_new
        y = x + beta_n * (x - y)
        y_clip = np.clip(y, 0.0, 1.0)
        g, _ = subgrad_compute(y_clip, m_act, n)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        lr = base_lr / np.sqrt(t_k)
        x_new = project_h(y_clip - lr * (g / gn), target)
        c5_new = compute_c5(x_new)
        if c5_new < best_c5 - 1e-15:
            best = x_new.copy()
            best_c5 = c5_new
            no_improve = 0
        else:
            no_improve += 1
        x = x_new
        t_k = t_new
        if no_improve > 8000:
            break
    return best, best_c5

def momentum_subgrad(h, target, time_fn, max_iters, m_act, momentum, init_step):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    vel = np.zeros(n)
    no_improve = 0
    step = init_step
    for it in range(max_iters):
        if time_fn() < 2:
            break
        g, _ = subgrad_compute(best, m_act, n)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        vel = momentum * vel + (1.0 - momentum) * (g / gn)
        vn = np.linalg.norm(vel)
        if vn < 1e-14:
            break
        d = -vel / vn
        found = False
        s = step
        for _ in range(12):
            ht = project_h(best + s * d, target)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-15:
                best = ht
                best_c5 = c5t
                no_improve = 0
                found = True
                step = min(s * 1.15, 0.03)
                break
            s *= 0.65
        if not found:
            no_improve += 1
            step *= 0.7
            if no_improve == 400:
                vel *= 0.2
                step = init_step * 0.6
                no_improve = 0
            if no_improve >= 1200 and it % 400 == 399:
                p = np.random.randn(n) * 0.004
                ht = project_h(best + p, target)
                c5t = compute_c5(ht)
                if c5t < best_c5:
                    best = ht
                    best_c5 = c5t
                    vel = np.zeros(n)
                    no_improve = 0
            if no_improve > 4000:
                break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def rem():
        return budget_s - (time.time() - t0)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(200, 0.5)
    best_c5 = compute_c5(best_h)
    n0 = len(best_h)
    target0 = n0 / 2.0
    dx0 = 2.0 / n0
    # Phase 1: SLSQP warmup
    for b in [50, 200, 800]:
        if rem() < 15:
            break
        best_h, best_c5 = slsqp_optimize(best_h, target0, dx0, b, 2500)
    # Phase 2: Nesterov subgradient
    if rem() > 30:
        best_h, best_c5 = nesterov_subgrad(
            best_h, target0, rem, max_iters=60000, m_act=30, lr_scale=0.8)
    # Phase 3: Momentum subgradient (fine-tune)
    if rem() > 20:
        best_h, best_c5 = momentum_subgrad(
            best_h, target0, rem, max_iters=40000, m_act=25, momentum=0.86, init_step=0.008)
    # Phase 4: SLSQP polish at n0
    if rem() > 40:
        for b in [3000, 10000]:
            if rem() < 15:
                break
            best_h, best_c5 = slsqp_optimize(best_h, target0, dx0, b, 4000)
    # Phase 5: Resolution doubling
    if rem() > 100:
        n2 = n0 * 2
        target2 = n2 / 2.0
        dx2 = 2.0 / n2
        x_old = np.linspace(0, 2, n0)
        x_new = np.linspace(0, 2, n2)
        h2 = project_h(np.interp(x_new, x_old, best_h), target2)
        c5_2 = compute_c5(h2)
        if rem() > 30:
            h2, c5_2 = slsqp_optimize(h2, target2, dx2, 400, 2000)
        if rem() > 40:
            h2, c5_2 = nesterov_subgrad(
                h2, target2, rem, max_iters=35000, m_act=35, lr_scale=0.6)
        if rem() > 25:
            h2, c5_2 = momentum_subgrad(
                h2, target2, rem, max_iters=25000, m_act=30, momentum=0.87, init_step=0.006)
        if rem() > 20:
            h2, c5_2 = slsqp_optimize(h2, target2, dx2, 8000, 3500)
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2
    # Phase 6: Final polish
    n_f = len(best_h)
    target_f = n_f / 2.0
    dx_f = 2.0 / n_f
    if rem() > 15:
        best_h, best_c5 = slsqp_optimize(best_h, target_f, dx_f, 20000, 5000)
    if rem() > 8:
        best_h, best_c5 = momentum_subgrad(
            best_h, target_f, rem, max_iters=8000, m_act=20, momentum=0.9, init_step=0.003)
    # Final projection
    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
