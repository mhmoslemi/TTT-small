import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_exact(p, S):
    lo = p.min() - 2.0
    hi = p.max() + 2.0
    for _ in range(200):
        tau = 0.5 * (lo + hi)
        h = np.clip(p - tau, 0.0, 1.0)
        if h.sum() > S:
            lo = tau
        else:
            hi = tau
    h = np.clip(p - 0.5 * (lo + hi), 0.0, 1.0)
    diff = S - h.sum()
    if abs(diff) > 1e-15:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_lag(h, lag, n):
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            idx = np.arange(m)
            grad[idx] += (1.0 - h[idx + lag])
            grad[idx + lag] -= h[idx]
    else:
        la = -lag
        m = n - la
        if m > 0:
            idx = np.arange(m)
            grad[idx + la] += (1.0 - h[idx])
            grad[idx] -= h[idx + la]
    return grad

def smooth_max_obj(h, dx, beta):
    corr = get_corr(h)
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

def con_eq(h, ts):
    return h.sum() - ts

def con_eq_jac(h, ts):
    return np.ones_like(h)

def slsqp_polish(h, n, deadline, seed, betas=None, perturb_scale=0.005):
    if betas is None:
        betas = [50, 200, 1000, 5000, 20000, 100000, 500000]
    np.random.seed(seed)
    dx = 2.0 / n
    ts = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac, 'args': (ts,)}]
    c5 = compute_c5(h)
    for beta in betas:
        if time.time() > deadline:
            break
        for p in [0.0, perturb_scale]:
            if time.time() > deadline:
                break
            x0 = h.copy()
            if p > 0:
                x0 = project_exact(x0 + np.random.randn(n) * p, ts)
            try:
                res = minimize(smooth_max_obj, x0, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 10000, 'ftol': 1e-15})
                h_new = project_exact(res.x, ts)
                c5_new = compute_c5(h_new)
                if c5_new < c5:
                    h = h_new
                    c5 = c5_new
            except Exception:
                pass
    return h, c5

def active_set_opt(h, n, deadline, seed, tol_frac=0.005):
    np.random.seed(seed)
    dx = 2.0 / n
    ts = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac, 'args': (ts,)}]
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    for outer in range(100):
        if time.time() > deadline:
            break
        corr = get_corr(h)
        max_corr = corr.max()
        active = np.where(corr > max_corr * (1.0 - tol_frac))[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        active_list = list(active)

        def active_obj(x, act=active_list):
            c = get_corr(x)

        try:

                best_c5 = c5

        except Exception:

def proj_subgrad(h, n, deadline, seed, top_k=10):

            x = x_new if c5_new < c5 else x
