def binary_round(h, n):
    """Round to binary and fix sum to exactly n/2."""

def simulated_annealing(h, n, deadline, t_start=1.0, t_end=0.01):
    """Simulated annealing on binary sequences."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    n_ones = len(ones)
    
    n_total = max(1, int((deadline - time.time()) * 1000))
    for it in range(n_total):
        if time.time() > deadline:
            break
        t = t_start * (t_end / t_start) ** (it / n_total)
        # Random swap
        i = np.random.randint(n_ones)
        j = np.random.randint(n - n_ones)
        oi = ones[i]
        zj = zeros[j]
        
        h[oi] = 0
        h[zj] = 1
        c = compute_c5(h, n)
        delta = c - cur_c
        
        if delta < 0 or np.random.random() < np.exp(-delta / (t * 0.01)):
            cur_c = c
            ones[i] = zj
            zeros[j] = oi
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            h[oi] = 1
            h[zj] = 0
    
    return best_h, best_c

    return h
