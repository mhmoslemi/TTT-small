    best = h.copy()
    best_c5 = compute_c5(best)
    corr = np.correlate(best, 1.0 - best, mode="full").copy()
    best_max = corr.max()
    deltas = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    
    no_improve = 0
    for sweep in range(max_sweeps):
        if time_fn() < 5:
            break
        improved = False
        order = np.random.permutation(n)
        
        for p in order:
            if best[p] < 1e-8 or best[p] > 1.0 - 1e-8:
                continue
            eff_p = compute_dec_effect(p, best, n)
            
            qs = []
            qs.append((p + 1) % n)
            qs.append((p - 1) % n)
            qs.append((p + n // 2) % n)
            while len(qs) < n_trials:
                q = np.random.randint(n)
                if q not in qs:
                    qs.append(q)
            
            for q in qs:
                if q == p:
                    continue
                eff_q = compute_dec_effect(q, best, n)
                diff_eff = eff_p - eff_q
                
                for delta in deltas:
                    # Try h[p] -= delta, h[q] += delta
                    if best[p] - delta < -1e-12 or best[q] + delta > 1.0 + 1e-12:
                        continue
                    new_corr = corr + delta * diff_eff
                    new_max = new_corr.max()
                    if new_max < best_max - 1e-13:
                        best[p] -= delta
                        best[q] += delta
                        corr = new_corr
                        best_max = new_max
                        best_c5 = best_max * dx
                        improved = True
                        eff_p = compute_dec_effect(p, best, n)
                        break
                    
                    # Try h[p] += delta, h[q] -= delta
                    if best[p] + delta > 1.0 + 1e-12 or best[q] - delta < -1e-12:
                        continue
                    new_corr = corr - delta * diff_eff
                    new_max = new_corr.max()
                    if new_max < best_max - 1e-13:
                        best[p] += delta
                        best[q] -= delta
                        corr = new_corr
                        best_max = new_max
                        best_c5 = best_max * dx
                        improved = True
                        eff_p = compute_dec_effect(p, best, n)
                        break
    
        if not improved:
            no_improve += 1
            if no_improve >= 4:
                break
        else:
            no_improve = 0
    
    return best, best_c5

def subgrad_polish(h, target, time_fn, max_iters=20000):
    n = len(h)
    dx = 2.0 / n
    best = h.copy()
    best_c5 = compute_c5(best)
    vel = np.zeros(n)
    momentum = 0.85
    step_list = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    no_improve = 0
    
    for it in range(max_iters):
        if time_fn() < 3:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        thresh = corr.max() - 0.01
        active = np.where(corr > thresh)[0]
        if len(active) == 0:
            active = np.argsort(corr)[::-1][:5]
        
        grad = np.zeros(n)
        shifts = np.arange(-(n-1), n)
        for k_idx in active:
            k = int(k_idx) - (n - 1)
            if k == 0:
                grad += (1.0 - 2.0 * best)
            elif k > 0:
                grad[k:] += (1.0 - best[:-k])
                grad[:n-k] -= best[k:]
            else:
                m = -k
                grad[:n-m] += (1.0 - best[m:])
                grad[m:] -= best[:n-m]
        
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        vel = momentum * vel + (1.0 - momentum) * (grad / gn)
        vn = np.linalg.norm(vel)
        if vn < 1e-14:
            break
        d = -vel / vn
        found = False
        for step in step_list:
            ht = project_h(best + step * d, target)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-14:
                best = ht
                best_c5 = c5t
                no_improve = 0
                found = True
                break
        if not found:
            no_improve += 1
            if no_improve > 500:
                p = np.random.randn(n) * 0.005
                ht = project_h(best + p, target)
                c5t = compute_c5(ht)
                if c5t < best_c5:
                    best = ht
                    best_c5 = c5t
                    no_improve = 0
                if no_improve > 2000:
                    break
    return best, best_c5

def upscale(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    target = n_new / 2.0
    return project_h(np.interp(x_new, x_old, h), target)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(200, 0.5)

    best_c5 = compute_c5(best_h)
    n0 = len(best_h)
    target0 = n0 / 2.0

    # Phase 1: SLSQP at initial resolution
    best_h, best_c5 = slsqp_opt(best_h, target0, rem,
                                betas=[50, 200, 800, 3000, 10000], max_iter=4000)

    # Phase 2: Local search at initial resolution
    best_h, best_c5 = local_search(best_h, target0, rem, max_sweeps=30, n_trials=10)

    # Phase 3: Subgradient polish
    best_h, best_c5 = subgrad_polish(best_h, target0, rem, max_iters=15000)

    # Phase 4: Second local search after subgrad
    best_h, best_c5 = local_search(best_h, target0, rem, max_sweeps=20, n_trials=6)

    # Phase 5: Try upscaling to 2x
    if rem() > 120:
        n2 = n0 * 2
        h2 = upscale(best_h, n2)
        target2 = n2 / 2.0
        c5_2 = compute_c5(h2)

        if rem() > 80:
            h2, c5_2 = slsqp_opt(h2, target2, rem,
                                 betas=[50, 200, 800, 3000, 10000, 30000], max_iter=3000)
        if rem() > 40:
            h2, c5_2 = local_search(h2, target2, rem, max_sweeps=20, n_trials=6)
        if rem() > 15:
            h2, c5_2 = subgrad_polish(h2, target2, rem, max_iters=10000)
        if rem() > 10:
            h2, c5_2 = local_search(h2, target2, rem, max_sweeps=15, n_trials=6)

        if c5_2 < best_c5:
            best_h = h2
            best_c5 = c5_2

    # Phase 6: Try 1.5x if we're still at base and have time
    n_cur = len(best_h)
    if n_cur == n0 and rem() > 80:
        n3 = int(n0 * 1.5)
        if n3 % 2 != 0:
            n3 += 1
        h3 = upscale(best_h, n3)
        target3 = n3 / 2.0
        c5_3 = compute_c5(h3)
        if rem() > 40:
            h3, c5_3 = slsqp_opt(h3, target3, rem,
                                 betas=[100, 500, 2000, 8000], max_iter=3000)
        if rem() > 20:
            h3, c5_3 = local_search(h3, target3, rem, max_sweeps=15, n_trials=6)
        if c5_3 < best_c5:
            best_h = h3
            best_c5 = c5_3

    # Phase 7: Final local search polish
    nf = len(best_h)
    tf = nf / 2.0
    if rem() > 10:
        best_h, best_c5 = local_search(best_h, tf, rem, max_sweeps=20, n_trials=8)

    # Phase 8: Tiny SLSQP polish
    if rem() > 15:
        nf2 = len(best_h)
        tf2 = nf2 / 2.0
        best_h, best_c5 = slsqp_opt(best_h, tf2, rem,
                                     betas=[20000, 50000, 100000], max_iter=2000)

    # Final local search
    if rem() > 5:
        nf3 = len(best_h)
        tf3 = nf3 / 2.0
        best_h, best_c5 = local_search(best_h, tf3, rem, max_sweeps=10, n_trials=5)

    # Ensure feasibility
    nf = len(best_h)
    best_h = project_h(best_h, nf / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
