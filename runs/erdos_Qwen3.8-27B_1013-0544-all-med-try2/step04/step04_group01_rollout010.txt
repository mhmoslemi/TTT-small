    dx = 2.0 / n
    g = np.zeros(n)
    for sh in shifts:

        if sh == 0:
            grad[:] = dx * (1 - 2*h)

            grad[sh:] += dx * (1 - h[:n-sh])

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-12:
            break
        if d > 0:
            r = 1.0 - h
            t = r.sum()
            if t < 1e-15:
                break
            h += (d / t) * r
        else:
            t = h.sum()
            if t < 1e-15:
                break
            h += (d / t) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    sm = s.max()
    return float(m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha)

def _sc(x, n):
    return float(x.sum() - n / 2.0)

def _scj(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter=4000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sc, 'jac': _scj, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def _shift_vec(h, s):
    n = len(h)
    out = np.zeros(n)
    if s == 0:
        out[:] = h
    elif s > 0:
        out[:n - s] = h[s:]
    else:
        out[-s:] = h[:n + s]
    return out

def subgradient_search(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    dx = 2.0 / n
    eta = 0.005
    tol = 0.001
    stalled = 0
    for it in range(200000):
        if time.time() > deadline:
            break
        c_full = np.correlate(h, 1 - h, mode='full') * dx
        c_max = float(c_full.max())
        if c_max < best_c - 1e-12:
            best_c = c_max
            best_h = h.copy()
            stalled = 0
        else:
            stalled += 1
            if stalled > 5000:
                eta *= 0.7
                stalled = 0
        active = c_full >= (c_max - tol)
        active_idx = np.where(active)[0]
        if len(active_idx) == 0:
            active_idx = np.argsort(c_full)[-3:]
        shifts = active_idx - (n - 1)
        g = np.zeros(n)
        for sh in shifts:
            g += dx * ((1.0 - _shift_vec(h, sh)) - _shift_vec(h, -sh))
        g /= len(shifts)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        h_new = h - eta * g
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < c_max - 1e-12:
            h = h_new
            eta = min(eta * 1.02, 0.05)
        else:
            eta *= 0.92
    return best_h, best_c

def pair_search(h, n, deadline, deltas):
    best = compute_c5(h, n)
    max_passes = 80
    for d in deltas:
        for _p in range(max_passes):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                hi = h[i] + d
                if hi > 1:
                    continue
                hi2 = h[i] - d
                if hi2 < 0:
                    continue
                for j in range(i + 1, n):
                    if time.time() > deadline:
                        return h, best
                    if hi <= 1 and h[j] - d >= 0:
                        h[i], h[j] = hi, h[j] - d
                        c = compute_c5(h, n)
                        if c < best - 1e-14:
                            best = c
                            improved = True
                            hi2 = h[i] - d
                        else:
                            h[i], h[j] = h[i] - d, h[j] + d
                            hi2 = h[i] - d
                    if hi2 >= 0 and h[j] + d <= 1:
                        h[i], h[j] = hi2, h[j] + d
                        c = compute_c5(h, n)
                        if c < best - 1e-14:
                            best = c
                            improved = True
                        else:
                            h[i], h[j] = h[i] + d, h[j] - d
            if not improved:
                break
    return h, best

def optimize_block(h, n, deadline, rng):
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    alphas = [5000, 50000, 200000, 1000000]
    for a in alphas:
        if time.time() > deadline - 10:
            break
        h_new = slsqp_opt(best_h, n, a, maxiter=3000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new
    if time.time() < deadline - 5:
        best_h, best_c = subgradient_search(best_h, n, deadline - 5)
        if time.time() < deadline - 3:
            h_new = slsqp_opt(best_h, n, 500000, maxiter=2000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()

    if h_init is None:

    best_h = project_feasible(h_init.copy(), n_init)
    best_c = compute_c5(best_h, n_init)
    n = n_init
    alloc1 = (deadline - t0) * 0.40
    best_h, best_c = optimize_block(best_h, n, t0 + alloc1, rng)
    if time.time() < deadline - 40:
        for n_try in [n_init + 50, n_init - 40, n_init + 100]:
            if n_try < 100 or time.time() > deadline - 25:
                continue
            x_old = np.linspace(0, 2, n)
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            alloc = min(50, (deadline - time.time() - 20) * 0.3)
            if alloc < 10:
                break
            h_opt, c_opt = optimize_block(h_interp, n_try, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
                n = n_try
    if time.time() < deadline - 30:
        n_restarts = 4
        for r in range(n_restarts):
            if time.time() > deadline - 15:
                break
            scale = 0.01 + 0.04 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(40, (deadline - time.time() - 12) * 0.35)
            if alloc < 8:

            h_opt, c_opt = optimize_block(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    if time.time() < deadline - 10:
        best_h, best_c = pair_search(best_h, n, deadline - 3,
                                     [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002])
    if time.time() < deadline - 5:
        best_h, best_c = subgradient_search(best_h, n, deadline - 2)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
