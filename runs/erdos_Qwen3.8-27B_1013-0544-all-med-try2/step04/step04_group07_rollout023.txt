import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def sum_eq(x, t):
    return np.sum(x) - t

def sum_eq_jac(x, t):
    return np.ones(x.shape)

def lse_obj(x, beta, dx):
    corr = compute_corr(x)
    m = np.max(corr)
    e = np.exp(np.clip(beta * (corr - m), -500, 500))
    return (m + np.log(np.sum(e) / len(corr)) / beta) * dx

def lse_grad(x, beta, dx):
    n = len(x)
    corr = compute_corr(x)
    m = np.max(corr)
    e = np.exp(np.clip(beta * (corr - m), -500, 500))
    p = e / np.sum(e)
    g = np.zeros(n)
    for k in range(len(corr)):
        if p[k] < 1e-12:
            continue
        la = abs(k - (n - 1))
        if la == 0:
            g[:] += p[k] * (1.0 - x)
            g[:] -= p[k] * x
        else:
            g[:n - la] -= p[k] * x[la:]
            g[la:] += p[k] * (1.0 - x[:n - la])
    return g * dx

def slsqp_solve(h0, n, target, beta, bounds, maxiter):
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    dx = 2.0 / n
    try:
        r = minimize(lse_obj, h0, args=(beta, dx), method='SLSQP',
                     bounds=bounds, constraints=cons, jac=lambda x: lse_grad(x, beta, dx),
                     options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_h(r.x, target)
    except Exception:
        return h0

def nesterov_attack(h0, n, target, n_iter, t0, budget):
    h = h0.copy()
    h_prev = h0.copy()
    y = h0.copy()
    best_h, best_c5 = h.copy(), compute_c5(h)
    step = 0.04
    stall = 0
    dx = 2.0 / n
    for it in range(n_iter):
        if time.time() - t0 > budget - 5:
            break
        corr = compute_corr(y)
        m = np.max(corr)
        e = np.exp(np.clip(5000.0 * (corr - m), -500, 500))
        p = e / np.sum(e)
        g = np.zeros(n)
        nk = len(corr)
        for k in range(nk):
            if p[k] < 1e-12:
                continue
            la = abs(k - (n - 1))
            if la == 0:
                g[:] += p[k] * (1.0 - y)
                g[:] -= p[k] * y
            else:
                g[:n - la] -= p[k] * y[la:]
                g[la:] += p[k] * (1.0 - y[:n - la])
        g *= dx
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        h_new = project_h(y - step * g, target)
        mu = (1.0 - np.sqrt(1.0 + 4.0 / ((it + 1) ** 2))) / 2.0
        y = np.clip(h_new + mu * (h_new - h_prev), 0, 1)
        h_prev, h = h, h_new
        c = compute_c5(h)
        if c < best_c5 - 1e-15:
            best_h, best_c5, stall = h.copy(), c, 0
        else:
            stall += 1
            if stall % 50 == 0:
                step *= 0.75
    return best_h, best_c5

def bottleneck_attack(h0, n, target, n_iter, t0, budget):
    h = h0.copy()
    best_h, best_c5 = h.copy(), compute_c5(h)
    stall = 0
    for it in range(n_iter):
        if time.time() - t0 > budget - 3:
            break
        corr = compute_corr(h)
        m = np.max(corr)
        thr = m - 0.005 * m
        active = np.where(corr >= thr)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        improved = False
        for k_idx in active[:20]:
            if time.time() - t0 > budget - 2:
                break
            la = abs(k_idx - (n - 1))
            g = np.zeros(n)
            if la == 0:
                g[:] = (1.0 - h) - h
            else:
                g[:n - la] -= h[la:]
                g[la:] += (1.0 - h[:n - la])
            gn = np.linalg.norm(g)

            g /= gn
            for alpha in (0.005, 0.002, 0.001, 0.0005):

                    improved = True
                    break
        if not improved:

                break
        else:

    return best_h, best_c5

        h_try, c_try = nesterov_attack(best_h, n, target, 6000, t0, budget_s)
        if c_try < best_c5:
            best_h, best_c5 = h_try, c_try

    # Phase 3: Bottleneck attack (single-lag gradient)
    rem = budget_s - (time.time() - t0)
    if rem > 60:
        h_try, c_try = bottleneck_attack(best_h, n, target, 3000, t0, budget_s)
        if c_try < best_c5:
            best_h, best_c5 = h_try, c_try

    # Phase 4: SLSQP polish at very high beta
    rem = budget_s - (time.time() - t0)
    if rem > 40:
        h_try = slsqp_solve(best_h, n, target, 100000, bounds, 20000)
        c_try = compute_c5(h_try)
        if c_try < best_c5:
            best_h, best_c5 = h_try, c_try

    # Phase 5: Pair local search
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        h_try, c_try = pair_search(best_h, n, target, 3000, t0, budget_s)
        if c_try < best_c5:
            best_h, best_c5 = h_try, c_try

    # Phase 6: Coordinate search
    rem = budget_s - (time.time() - t0)
    if rem > 20:
        h_try, c_try = coord_search(best_h, n, target, 20, t0, budget_s)
        if c_try < best_c5:
            best_h, best_c5 = h_try, c_try

    # Phase 7: Another SLSQP + Nesterov cycle
    rem = budget_s - (time.time() - t0)
    if rem > 40:
        h_try = slsqp_solve(best_h, n, target, 80000, bounds, 15000)
        c_try = compute_c5(h_try)
        if c_try < best_c5:
            best_h, best_c5 = h_try, c_try
        rem = budget_s - (time.time() - t0)
        if rem > 30:
            h_try, c_try = nesterov_attack(best_h, n, target, 3000, t0, budget_s)
            if c_try < best_c5:
                best_h, best_c5 = h_try, c_try

    # Phase 8: Try higher resolution
    rem = budget_s - (time.time() - t0)
    if rem > 80:
        n2 = n + 32
        target2 = n2 / 2.0
        bounds2 = Bounds(np.zeros(n2), np.ones(n2))
        h2 = project_h(resample_h(best_h, n2), target2)
        for beta in [2000, 10000]:
            if time.time() - t0 > budget_s - 30:
                break
            h2 = slsqp_solve(h2, n2, target2, beta, bounds2, 10000)
        c2 = compute_c5(h2)
        if c2 < best_c5:
            best_h, best_c5, n = h2, c2, n2
            target = n2 / 2.0

    # Final: ensure constraints
    best_h = project_h(best_h, len(best_h) / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
