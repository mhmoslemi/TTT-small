from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / len(h)
        else:
            h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def sum_eq(x, target):
    return np.sum(x) - target

def sum_eq_jac(x, target):
    return np.ones(x.shape)

def lag_obj(x, lag, n):
    if 0 <= lag < n:
        m = n - lag
        return np.sum(x[:m] * (1.0 - x[lag:]))
    else:
        la = -lag
        m = n - la
        return np.sum(x[la:] * (1.0 - x[:m]))

def lag_grad(x, lag, n):
    g = np.zeros(n)
    if 0 <= lag < n:
        m = n - lag
        g[:m] += (1.0 - x[lag:])
        g[lag:] -= x[:m]
    else:
        la = -lag
        m = n - la
        g[la:] += (1.0 - x[:m])
        g[:m] -= x[la:]
    return g

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / len(h)
        else:
            h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def sum_eq(x, target):
    return np.sum(x) - target

def sum_eq_jac(x, target):
    return np.ones(x.shape)

def lag_obj(x, lag, n):
    if 0 <= lag < n:
        m = n - lag
        return np.sum(x[:m] * (1.0 - x[lag:]))
    else:
        la = -lag
        m = n - la
        return np.sum(x[la:] * (1.0 - x[:m]))

def lag_grad(x, lag, n):
    g = np.zeros(n)
    if 0 <= lag < n:
        m = n - lag
        g[:m] += (1.0 - x[lag:])
        g[lag:] -= x[:m]
    else:
        la = -lag
        m = n - la
        g[la:] += (1.0 - x[:m])
        g[:m] -= x[la:]
    return g

def smooth_max(x, beta):
    m = np.max(x)
    xs = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(xs)) / len(x)) / beta

def sg_obj(x, beta, n):
    c = np.correlate(x, 1.0 - x, mode="full")
    return smooth_max(c, beta) * (2.0 / n)

def optimize_lag_slsqp(h, lag, n, target, maxiter=5000):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        r = minimize(lag_obj, h, args=(lag, n), jac=lag_grad,
                     method='SLSQP', bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-17})
        return project_h(r.x, target)
    except Exception:
        return h

def optimize_smooth_slsqp(h, n, target, beta, maxiter=5000):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        r = minimize(sg_obj, h, args=(beta, n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-17})
        return project_h(r.x, target)
    except Exception:
        return h

def lag_cycling(h, n, target, t0, budget, max_rounds=50, top_k=3, maxiter=3000):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    for rnd in range(max_rounds):
        if budget - (time.time() - t0) < 5:
            break
        lags = compute_lags(cur)
        top_idx = np.argsort(lags)[::-1][:top_k]
        improved = False
        for idx in top_idx:
            if budget - (time.time() - t0) < 3:
                break
            lag = int(idx) - (n - 1)
            new_h = optimize_lag_slsqp(cur, lag, n, target, maxiter=maxiter)
            c_try = compute_c5(new_h)
            if c_try < best_c5 - 1e-15:
                cur = new_h
                best_c5 = c_try
                improved = True
        if not improved:
            break
    return cur, best_c5

def binary_anneal(h, n, t0, budget, n_iter=400000):
    rng = np.random.default_rng(999)
    h_bin = np.zeros(n)
    idx = np.argsort(h)[::-1][:n // 2]
    h_bin[idx] = 1.0
    lags = compute_lags(h_bin)
    cur_max = float(np.max(lags))
    best_h = h_bin.copy()
    best_c5 = cur_max
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = ones[rng.integers(len(ones))]
        b = zeros[rng.integers(len(zeros))]
        h_bin[a] = 0.0
        h_bin[b] = 1.0
        new_lags = compute_lags(h_bin)
        new_max = float(np.max(new_lags))
        if new_max < cur_max:
            cur_max = new_max
            lags = new_lags
            if new_max < best_c5:
                best_c5 = new_max
                best_h = h_bin.copy()
        else:
            frac = it / max(n_iter, 1)
            temp = 0.4 * (0.001) ** frac
            if rng.random() < np.exp(-(new_max - cur_max) / max(temp, 1e-15)):
                cur_max = new_max
                lags = new_lags
            else:
                h_bin[a] = 1.0
                h_bin[b] = 0.0
    return best_h, best_c5

def coord_descent(h, target, n, max_sweeps, t0, budget):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    deltas = [0.05, -0.05, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005]
    for sw in range(max_sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 1:
                break
            for d in deltas:
                trial = cur.copy()
                trial[j] = np.clip(trial[j] + d, 0.0, 1.0)
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-14:
                    cur = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved:
            break
    return cur, best_c5

def push_binary(h, target, n, thresh=0.3):
    cur = h.copy()
    cur[cur < thresh] = 0.0
    cur[cur > 1.0 - thresh] = 1.0
    return project_h(cur, target)

    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)
