import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_wide(h, n, thr_frac=0.02, beta_w=300):
    corr = get_corr(h)
    mc = np.max(corr)
    thr = mc - thr_frac * (mc + 1e-10)
    active = corr >= thr
    if not np.any(active):
        active = np.zeros_like(corr, dtype=bool)
        active[np.argmax(corr)] = True
    aidx = np.where(active)[0]
    rw = np.exp(beta_w * (corr[aidx] - mc))
    w = rw / rw.sum()
    g = np.zeros(n)
    for idx in range(len(aidx)):
        ai = int(aidx[idx])
        lag = ai - (n - 1)
        wi = w[idx]
        if lag >= 0:
            g[:n - lag] += wi * (1.0 - h[lag:])
            g[lag:] -= wi * h[:n - lag]
        else:
            la = -lag
            g[la:] += wi * (1.0 - h[:n - la])
            g[:n - la] -= wi * h[la:]
    return g, corr

def nesterov_bgd(h0, n, target, n_iter, step0, t0, t_end):
    h = h0.copy()
    h_prev = h0.copy()
    y = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = step0
    stall = 0
    for it in range(n_iter):
        if time.time() > t_end:
            break
        g, _ = subgrad_wide(y, n, thr_frac=0.02, beta_w=300)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        h_new = project_h(y - step * g, target)
        mu = (1.0 - np.sqrt(1.0 + 4.0 / ((it + 1) ** 2))) / 2.0
        y = h_new + mu * (h_new - h_prev)
        y = np.clip(y, 0, 1)
        h_prev, h = h, h_new
        c = compute_c5(h)
        if c < best_c5 - 1e-15:
            best_h = h.copy()
            best_c5 = c
            stall = 0
        else:
            stall += 1
            if stall % 80 == 0:
                step *= 0.82
    return best_h, best_c5

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(np.clip(beta * (x - m), -500, 500))) / len(x)) / beta

def sml_obj(h_in, beta, dx):
    return smooth_max(np.correlate(h_in, 1.0 - h_in, mode="full"), beta) * dx

def sum_eq(x, t):
    return np.sum(x) - t

def sum_eq_jac(x, t):
    return np.ones(x.shape)

def do_slsqp(h0, n, target, beta, bounds, maxiter, ftol):
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        r = minimize(sml_obj, h0, args=(beta, 2.0 / n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def perturb_optimize(h, n, target, t0, t_end, seed, sigma):
    rng = np.random.RandomState(seed)
    h_p = project_h(h + sigma * rng.randn(n), target)
    h_opt, c5_opt = nesterov_bgd(h_p, n, target, 50000, 0.035, t0, t_end)
    return h_opt, c5_opt

def coord_descent(cur, target, n, max_sweeps, t0, t_end):
    best_c5 = compute_c5(cur)
    deltas = (0.04, -0.04, 0.015, -0.015, 0.007, -0.007)
    for sw in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for j in range(n):
            for d in deltas:
                trial = cur.copy()
                trial[j] += d
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best_c5 - 1e-14:
                    cur, best_c5, improved = trial, c, True
                    break
        if not improved:
            break
    return cur, best_c5

def frank_wolfe(h0, n, target, n_iter, t0, t_end):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stall = 0
    for it in range(n_iter):
        if time.time() > t_end:
            break
        g, _ = subgrad_wide(h, n, thr_frac=0.03, beta_w=200)
        if np.linalg.norm(g) < 1e-14:
            break
        order = np.argsort(g)
        z = np.zeros(n)
        k_full = int(target)
        z[order[:k_full]] = 1.0
        diff = target - z.sum()
        if diff > 1e-10:
            z[order[k_full]] = diff
        elif diff < -1e-10:
            z[order[k_full - 1]] = 1.0 + diff
        alpha = 2.0 / (it + 2)
        h_new = np.clip((1 - alpha) * h + alpha * z, 0, 1)
        h_new = project_h(h_new, target)
        c = compute_c5(h_new)
        if c < best_c5 - 1e-15:
            best_h = h_new.copy()
            best_c5 = c
            stall = 0
        else:
            stall += 1
        h = h_new
        if stall > 400:
            break
    return best_h, best_c5

    t_end4 = t0 + 0.90 * budget_s
