        v_hat = v_v / (1 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, ts)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best = h_new.copy()
            best_c5 = c5_new
        h = h_new
        
        # Adaptive lr decay
        if t % 1000 == 0:
            lr *= 0.95
    
    return best, best_c5

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_smooth_max(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    exp_v = np.exp(shifted)
    w = exp_v / exp_v.sum()
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-12:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            m_idx = n - lag
            grad[:m_idx] += w[k] * (1.0 - h[lag:])
            grad[lag:] -= w[k] * h[:m_idx]
        else:
            la = -lag
            m_idx = n - la
            grad[la:] += w[k] * (1.0 - h[:m_idx])
            grad[:m_idx] -= w[k] * h[la:]
    return grad * dx

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def con_fun(h, ts):
    return h.sum() - ts

def con_jac(h, ts):
    return np.ones_like(h)

def slsqp_step(h, n, beta, maxiter=2000, ftol=1e-13):
    dx = 2.0 / n
    ts = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts,)}]
    try:
        res = minimize(obj_smooth, h, args=(dx, beta), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, ts)
    except Exception:
        return h

def adam_subgradient(h, n, deadline, seed, beta_init=100.0, beta_final=5000.0,
                     lr=0.002, pen=50.0, top_k=30):
    np.random.seed(seed)
    ts = n / 2.0
    dx = 2.0 / n
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    mv = np.zeros(n)
    vv = np.zeros(n)
    t = 0
    stagnation = 0
    total_iter = 50000

    while time.time() < deadline and t < total_iter:
        t += 1
        frac = t / total_iter
        beta = beta_init * (beta_final / beta_init) ** frac
        lr_t = lr * (1.0 - frac) ** 0.5

        # Compute weighted gradient using top-k lags
        corr = np.correlate(h, 1.0 - h, mode='full')
        if top_k < len(corr):
            top_idx = np.argpartition(corr, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(corr))
        top_corr = corr[top_idx]
        shifted = np.clip(beta * (top_corr - top_corr.max()), -700.0, 0.0)
        w = np.exp(shifted)
        w /= w.sum()

        grad_main = np.zeros(n)
        for j in range(len(top_idx)):
            if w[j] < 1e-12:
                continue
            lag = top_idx[j] - (n - 1)
            if lag >= 0:
                m_idx = n - lag
                grad_main[:m_idx] += w[j] * (1.0 - h[lag:])
                grad_main[lag:] -= w[j] * h[:m_idx]
            else:
                la = -lag
                m_idx = n - la
                grad_main[la:] += w[j] * (1.0 - h[:m_idx])
                grad_main[:m_idx] -= w[j] * h[la:]
        grad_main *= dx

        # Penalty gradient
        pen_grad = 2.0 * pen * (h.sum() - ts)
        grad_total = grad_main + pen_grad

        # Adam update
        mv = 0.9 * mv + 0.1 * grad_total
        vv = 0.999 * vv + 0.001 * grad_total ** 2
        mv_hat = mv / (1.0 - 0.9 ** t)
        vv_hat = vv / (1.0 - 0.999 ** t)
        h_new = h - lr_t * mv_hat / (np.sqrt(vv_hat) + 1e-8)
        h_new = project_feasible(h_new, ts)

        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_h = h_new.copy()
            best_c5 = c5_new
            stagnation = 0
        else:
            stagnation += 1
            if stagnation > 800:
                h_new = project_feasible(best_h + np.random.randn(n) * 0.008, ts)
                mv *= 0.1
                stagnation = 0

        h = h_new

        # Periodic SLSQP polish
        if t % 3000 == 0:
            h_slq = slsqp_step(best_h.copy(), n, beta=300, maxiter=1000, ftol=1e-13)
            c_slq = compute_c5(h_slq)
            if c_slq < best_c5:
                best_h = h_slq
                best_c5 = c_slq
                h = h_slq.copy()
                mv = np.zeros(n)
                vv = np.zeros(n)
                t = 0
                stagnation = 0

    return best_h, best_c5

def subgradient_momentum(h, n, deadline, seed, k_top=50, beta_sm=8.0):
    np.random.seed(seed)
    ts = n / 2.0
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    velocity = np.zeros(n)
    momentum = 0.85
    alpha = 0.0008
    stagnation = 0
    prev_h = h.copy()
    prev_g = None

    for it in range(30000):
        if time.time() > deadline:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-k_top:]
        top_corr = corr[top_idx]
        shifted = np.clip(beta_sm * (top_corr - top_corr.max()), -700.0, 0.0)
        w = np.exp(shifted)
        w /= w.sum()

        grad = np.zeros(n)
        for j in range(k_top):
            if w[j] < 1e-12:
                continue
            lag = top_idx[j] - (n - 1)
            if lag >= 0:
                m_idx = n - lag
                grad[:m_idx] += w[j] * (1.0 - h[lag:])
                grad[lag:] -= w[j] * h[:m_idx]
            else:
                la = -lag
                m_idx = n - la
                grad[la:] += w[j] * (1.0 - h[:m_idx])
                grad[:m_idx] -= w[j] * h[la:]
        grad *= (2.0 / n)

        # BB step size
        s = h - prev_h
        y = grad - prev_g if prev_g is not None else None
        prev_h = h.copy()
        prev_g = grad.copy()
        if y is not None:
            yy = np.dot(y, y)
            sy = np.dot(s, y)
            if yy > 1e-20 and sy > 1e-12:
                alpha = np.clip(sy / yy, 1e-6, 0.01)

        velocity = momentum * velocity - alpha * grad
        h_new = project_feasible(h + velocity, ts)
        c_new = compute_c5(h_new)

        if c_new < best_c5 - 1e-12:
            h = h_new
            best_h = h_new.copy()
            best_c5 = c_new
            stagnation = 0
            momentum = min(momentum + 0.002, 0.95)
        else:
            h = h_new
            stagnation += 1
            if stagnation % 100 == 0:
                momentum = max(momentum - 0.03, 0.6)
                alpha *= 0.8
            if stagnation > 600:
                h = project_feasible(best_h + np.random.randn(n) * 0.01, ts)
                velocity *= 0.1
                momentum = 0.85
                alpha = 0.0008
                stagnation = 0

        if it % 2000 == 1999:
            h_slq = slsqp_step(best_h.copy(), n, beta=200, maxiter=800)
            c_slq = compute_c5(h_slq)
            if c_slq < best_c5:
                best_h = h_slq
                best_c5 = c_slq
                h = h_slq.copy()
                velocity = np.zeros(n)
                prev_h = h.copy()
                prev_g = None
                stagnation = 0

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    parent = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float).copy()
    if parent is None:
        parent = np.full(200, 0.5)
    pn = len(parent)

    # Phase 1: Adam subgradient at parent resolution (35%)
    d1 = t0 + budget_s * 0.35
    h1, c1 = adam_subgradient(parent.copy(), pn, d1, seed)
    best_h, best_c5, best_n = h1.copy(), c1, pn

    # Phase 2: SLSQP polish (15%)
    d2 = t0 + budget_s * 0.50
    for beta_val in [200, 500, 2000, 10000]:
        if time.time() > d2:
            break
        h_s = slsqp_step(best_h.copy(), best_n, beta=beta_val, maxiter=3000, ftol=1e-14)
        c_s = compute_c5(h_s)
        if c_s < best_c5:
            best_h, best_c5 = h_s, c_s

    # Phase 3: Momentum subgradient with different params (20%)
    d3 = t0 + budget_s * 0.70
    h3, c3 = subgradient_momentum(best_h.copy(), best_n, d3, seed + 200, k_top=60, beta_sm=6.0)
    if c3 < best_c5:
        best_h, best_c5 = h3, c3

    # Another Adam pass with different hyperparams
    d3b = t0 + budget_s * 0.80
    if time.time() < d3b:
        h4, c4 = adam_subgradient(best_h.copy(), best_n, d3b, seed + 500,
                                   beta_init=50.0, beta_final=2000.0, lr=0.001, pen=100.0, top_k=40)
        if c4 < best_c5:
            best_h, best_c5 = h4, c4

    # Phase 4: Try different resolutions (15%)
    d4 = t0 + budget_s * 0.95
    res_list = sorted(set([best_n - 20, best_n + 20, best_n - 50, best_n + 50,
                           best_n - 10, best_n + 10]))
    res_list = [r for r in res_list if 100 <= r <= 600 and r != best_n]
    for n_try in res_list:
        if time.time() > d4:
            break
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h0 = np.interp(x_new, x_old, best_h)
        h0 = project_feasible(h0, n_try / 2.0)
        h_r, c_r = adam_subgradient(h0, n_try, d4, seed + n_try,
                                     beta_init=100.0, beta_final=3000.0, lr=0.0015, pen=80.0, top_k=30)
        if c_r < best_c5:
            best_h, best_c5, best_n = h_r, c_r, n_try
        if time.time() < d4:
            h_s = slsqp_step(best_h.copy(), best_n, beta=1000, maxiter=2000)
            c_s = compute_c5(h_s)
            if c_s < best_c5:
                best_h, best_c5 = h_s, c_s

    # Final SLSQP polish
    if time.time() < t0 + budget_s * 0.98:
        h_f = slsqp_step(best_h.copy(), best_n, beta=20000, maxiter=5000, ftol=1e-15)
        c_f = compute_c5(h_f)
        if c_f < best_c5:
            best_h, best_c5 = h_f, c_f

    if best_h is None:
        best_h = np.full(100, 0.5)
        best_n = 100
    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
