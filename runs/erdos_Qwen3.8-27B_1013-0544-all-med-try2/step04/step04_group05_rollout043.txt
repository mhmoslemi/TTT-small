- It decreases until the slope crosses 0, then increases
- We can find the optimal point by checking where the active set changes

- If min active s_pair < 0 < max active s_pair, we're at the minimum
- If all active s_pair > 0, increase isn't helpful, try decreasing δ
- If all active s_pair < 0, try increasing δ

def find_optimal_delta(corr, s_pair, lo, hi, n_eval=30):
    """Minimize max(corr + delta * s_pair) for delta in [lo, hi] using golden section."""
    if lo >= hi:
        return lo
    # Golden section search for unimodal function
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618
    
    def obj(d):
        return np.max(corr + d * s_pair)
    
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = obj(c)
    fd = obj(d)
    
    for _ in range(n_eval):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = obj(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = obj(d)
    
    return (a + b) / 2

    n = len(h)
    s = np.zeros(2 * n - 1)

    # h[i] appears as:
    #   h[j] with j=i: need 0<=i+k<n → term is h[i]*(1-h[i+k]), deriv = (1-h[i+k])
    #   h[j+k] with j+k=i, j=i-k: need 0<=i-k<n → term is h[i-k]*(1-h[i]), deriv = -h[i-k]

        val = 0.0
        if 0 <= i + k < n:
            val += (1.0 - h[i + k])
        if 0 <= i - k < n:
            val += (-h[i - k])
        s[idx] = val
    return s

    """Find delta in [lo,hi] minimizing max(corr + delta * s_pair)"""
    if hi - lo < 1e-14:
        return lo
    # Golden section search (convex function)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = np.max(corr + c * s_pair)

            d = c

            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = np.max(corr + d * s_pair)

    """Pair-based coordinate descent maintaining sum constraint."""
    n = len(h)
    best = h.copy()
    best_corr = corr.copy()
    best_c5 = float(np.max(best_corr) * dx)
    sens = [sensitivity(best, i) for i in range(n)]
    
    no_improve = 0

            break
        if it % 5000 == 0 and it > 0:
            # Periodically recompute sensitivities (they change as h changes)
            sens = [sensitivity(best, i) for i in range(n)]

        j = np.random.randint(n)

                best[j] -= delta
                best_corr += delta * s_pair
                best_c5 = float(new_max * dx)
                no_improve = 0
                # Update sensitivities for i and j
                sens[i] = sensitivity(best, i)
                sens[j] = sensitivity(best, j)
        
        if no_improve > 100000:
            break
    
    # Final projection
    best = project_h(best, target)
    best_corr = compute_corr(best)
    best_c5 = float(np.max(best_corr) * dx)
    return best, best_c5, best_corr

def slsqp(h, target, dx, betas, max_iter, time_fn):
    n = len(h)
    bounds = Bounds(np.zeros(n), np.ones(n))
    best = h.copy()
    best_c5 = compute_c5(best)
    for beta in betas:
        if time_fn() < 15:
            break
        def obj(h_in):
            return smooth_max(compute_corr(h_in), beta) * dx
        def con(h_in):
            return h_in.sum() - target
        try:
            res = minimize(obj, best, method='SLSQP', bounds=bounds,
                          constraints=[{'type':'eq','fun':con,'jac':lambda x: np.ones(n)}],
                          options={'maxiter': max_iter, 'ftol': 1e-15})
            hn = project_h(res.x, target)
            c5n = compute_c5(hn)
            if c5n < best_c5:
                best = hn
                best_c5 = c5n
        except Exception:
            pass
    return best, best_c5
