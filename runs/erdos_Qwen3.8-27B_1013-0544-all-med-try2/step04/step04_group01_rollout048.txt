import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def coord_descent(h, n, deadline, n_sweeps=500):
    """Coordinate descent: optimize one coordinate at a time via bisection."""
    h = h.copy()
    best_c = compute_c5(h, n)
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            old = h[i]
            lo, hi = 0.0, 1.0
            # Start bisection around current value
            lo = max(0.0, old - 0.1)
            hi = min(1.0, old + 0.1)
            for _ in range(20):
                mid = (lo + hi) * 0.5
                h[i] = mid + 1e-8
                c_plus = compute_c5(h, n)
                h[i] = mid - 1e-8
                c_minus = compute_c5(h, n)
                if c_plus > c_minus:
                    hi = mid
                else:
                    lo = mid
            h[i] = (lo + hi) * 0.5
            c = compute_c5(h, n)
            if c < best_c - 1e-14:
                best_c = c
                improved = True
        # Project to maintain sum constraint
        h = project_feasible(h, n)
        best_c = compute_c5(h, n)
        if not improved:
            break
    return h, best_c

def pair_search(h, n, deadline, deltas, radius=None):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(200):
            if time.time() > deadline:

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def compute_dc(h, i, delta, n):
    """Compute the change in correlation vector when h[i] changes by delta."""
    dc = np.zeros(2 * n - 1)
    m1 = np.arange(i, i + n)
    j1 = i + n - 1 - m1
    dc[m1] += delta * (1.0 - h[j1])
    m2 = np.arange(n - 1 - i, 2 * n - 1 - i)
    j2 = i + m2 - n + 1
    dc[m2] -= delta * h[j2]
    return dc
