                best = h.copy()
        else:
            h[j] = 0.0
            h[i] = 1.0
            ones[i_idx] = i
            zeros[j_idx] = j
    
    return best, best_c5

def slsqp_refine(h, time_fn, betas=None, max_iter=5000):
    n = len(h)
    target = n // 2.0
    dx = 2.0 / n
    best = h.copy()
    best_c5 = compute_c5(best)
    bounds = Bounds(np.zeros(n), np.ones(n))
    if betas is None:
        betas = [100, 400, 1500, 5000]
    for beta in betas:
        if time_fn() < 10:
            break
        def obj_fn(h_in):
            c = np.correlate(h_in, 1.0 - h_in, mode="full")
            m = np.max(c)
            shifted = np.clip(beta * (c - m), -300, 300)
            return (m + np.log(np.sum(np.exp(shifted)) / len(c)) / beta) * dx
        def con_fn(h_in):
            return h_in.sum() - target
        def jac_fn(h_in):
            return np.ones(n)
        try:
            res = minimize(obj_fn, best, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': con_fn, 'jac': jac_fn}],
                          options={'maxiter': max_iter, 'ftol': 1e-16})
            h_new = project_sum(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best = h_new
                best_c5 = c5_new
        except Exception:
            pass
    return best, best_c5

def guided_local_search(h, time_fn, n_iters=5000, seed_rng=None):
    n = len(h)
    target = n // 2
    dx = 2.0 / n
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    if seed_rng is None:
        seed_rng = np.random
    
    ones = list(np.where(cur == 1)[0])
    zeros = list(np.where(cur == 0)[0])
    
    for it in range(n_iters):
        if time_fn() < 3:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        worst_lag = int(np.argmax(corr)) - (n - 1)
        
        improved = False
        candidates = []
        if worst_lag > 0:
            for i in ones:
                if i + worst_lag < n and cur[i + worst_lag] == 0:
                    for j in zeros:
                        if j - worst_lag >= 0 and cur[j - worst_lag] == 1:
                            candidates.append((i, j))
        elif worst_lag < 0:
            m = -worst_lag
            for i in ones:
                if i - m >= 0 and cur[i - m] == 0:
                    for j in zeros:
                        if j + m < n and cur[j + m] == 1:
                            candidates.append((i, j))
        
        if not candidates:
            i = ones[seed_rng.integers(len(ones))]
            j = zeros[seed_rng.integers(len(zeros))]
            candidates = [(i, j)]
        
        best_swap = None
        best_swap_c5 = cur_c5
        sample = candidates[:20] if len(candidates) > 20 else candidates
        for (i, j) in sample:
            cur[i] = 0.0
            cur[j] = 1.0
            c = compute_c5(cur)
            if c < best_swap_c5:
                best_swap_c5 = c
                best_swap = (i, j)
            cur[i] = 1.0
            cur[j] = 0.0
        
        if best_swap is not None:
            i, j = best_swap
            cur[i] = 0.0
            cur[j] = 1.0
            ones[ones.index(i) if i in ones else seed_rng.integers(len(ones))] = j
            zeros[zeros.index(j) if j in zeros else seed_rng.integers(len(zeros))] = i
            cur_c5 = best_swap_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best = cur.copy()
            improved = True
        else:
            i = ones[seed_rng.integers(len(ones))]
            j = zeros[seed_rng.integers(len(zeros))]
            cur[i] = 0.0
            cur[j] = 1.0
            c = compute_c5(cur)
            if c < cur_c5:
                cur_c5 = c
                ones[ones.index(i)] = j
                zeros[zeros.index(j)] = i
                if c < best_c5:
                    best_c5 = c
                    best = cur.copy()
    
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    t0 = time.time()
    
    def rem():
        return budget_s - (time.time() - t0)
    
    # Get initial
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = golden_init(400)
    
    best_c5 = compute_c5(best_h)
    n0 = len(best_h)
    
    # Phase 1: SLSQP polish on continuous
    if rem() > 50:
        best_h, best_c5 = slsqp_refine(best_h, rem, [200, 800, 3000], 4000)
    
    # Phase 2: SA on 0-1 at current resolution
    h_bin = project_to_01(best_h, n0)
    c5_bin = compute_c5(h_bin)
    
    # Also try golden ratio start
    h_gold = golden_init(n0)
    c5_gold = compute_c5(h_gold)
    sa_start = h_gold if c5_gold < c5_bin else h_bin
    
    if rem() > 100:
        sa_budget = min(200, rem() - 100)
        h_sa, c5_sa = sa_01(sa_start.copy(), rem, sa_budget, rng)
        if c5_sa < best_c5:
            best_h = h_sa.copy()
            best_c5 = c5_sa
    
    # Phase 3: Guided local search from best 0-1
    if rem() > 60:
        gls_start = project_to_01(best_h, n0)
        h_gls, c5_gls = guided_local_search(gls_start, rem, 8000, rng)
        if c5_gls < best_c5:
            best_h = h_gls.copy()
            best_c5 = c5_gls
    
    # Phase 4: SLSQP refinement
    if rem() > 30:
        best_h, best_c5 = slsqp_refine(best_h, rem, [200, 1000, 4000, 8000], 6000)
    
    # Phase 5: Try different resolutions
    for n_try in [300, 500, 600]:
        if rem() < 80:
            break
        if n_try == n0:
            continue
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_try = project_sum(np.interp(x_new, x_old, best_h), n_try / 2.0)
        c5_try = compute_c5(h_try)
        
        h_bin_try = project_to_01(h_try, n_try)
        sa_b = min(60, rem() - 40)
        if sa_b > 10:
            h_sa_try, c5_sa_try = sa_01(h_bin_try, rem, sa_b, rng)
            if c5_sa_try < c5_try:
                h_try = h_sa_try.copy()
                c5_try = c5_sa_try
        
        if rem() > 20:
            h_try, c5_try = slsqp_refine(h_try, rem, [300, 1000], 3000)
        
        if c5_try < best_c5:
            best_h = h_try.copy()
            best_c5 = c5_try
            n0 = n_try
    
    # Phase 6: Final SA + polish on best resolution
    if rem() > 40:
        h_fin = project_to_01(best_h, len(best_h))
        sa_b = min(rem() - 20, 40)
        h_sa_fin, c5_sa_fin = sa_01(h_fin, rem, sa_b, rng)
        if c5_sa_fin < best_c5:
            best_h = h_sa_fin.copy()
            best_c5 = c5_sa_fin
    
    if rem() > 15:
        best_h, best_c5 = slsqp_refine(best_h, rem, [500, 2000, 6000], 4000)
    
    # Final check
    best_h = project_sum(best_h, len(best_h) // 2.0)
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, len(best_h)
