import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target):
    h = h.copy()
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        np.clip(h, 0.0, 1.0, out=h)
    return h

def project_fast(h, target):
    h = h.copy()
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        free = (h > 0) & (h < 1)
        if free.sum() > 0:
            h[free] += diff / free.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(np.clip(beta * (x - m), -500, 500))) / len(x)) / beta

def sml_obj(h, beta, dx):

def do_slsqp(h0, n, target, beta, bounds, maxiter, ftol):

        r = minimize(sml_obj, h0, args=(beta, 2.0 / n), method='SLSQP',
                     bounds=bounds, constraints=cons,

        return project_h(r.x, target)

def sa_optimize(h0, n, target, t0, budget, rng, step_scale=1.0):
    h = h0.copy()
    cur_c5 = compute_c5(h)
    best_h = h0.copy()
    best_c5 = cur_c5
    T = 0.003 * step_scale
    T_floor = 1e-6
    cooling = 0.99992
    stall = 0
    stall_threshold = 50000

    while time.time() - t0 < budget - 3:
        j = rng.randint(n)
        k = rng.randint(n)
        if j == k:
            continue
        delta = rng.uniform(0.003, 0.06) * step_scale * rng.choice([1, -1])
        trial = h.copy()
        trial[j] -= delta
        trial[k] += delta
        trial = project_fast(trial, target)
        c = compute_c5(trial)

        if c < best_c5 - 1e-14:
            best_h = trial.copy()
            best_c5 = c
            stall = 0
        else:
            stall += 1

        if c <= cur_c5 or rng.rand() < np.exp(-(c - cur_c5) / max(T, 1e-12)):
            h = trial
            cur_c5 = c
        T /= cooling
        if T < T_floor:
            T = T_floor
        if stall > stall_threshold:
            T = 0.003 * step_scale
            stall = 0
            step_scale *= 0.7
            cooling = 0.99995

    return best_h, best_c5

def structured_sa(h0, n, target, t0, budget, rng):
    h = h0.copy()
    cur_c5 = compute_c5(h)
    best_h = h0.copy()

        return h.copy()
    x_old = np.linspace(0, 2, len(h))
    x_new = np.linspace(0, 2, n_new)
