```python
            new_corr = np.correlate(h, 1.0 - h, mode="full") * dx
            new_max = new_corr.max()
            if new_max < best_new_max - 1e-12:
                best_new_max = new_max
                best_flip = i
                best_i = i
            h[i] = old_val
        
        if best_i >= 0:
            h[best_i] = 1.0 - h[best_i]
            cur_max = best_new_max
            if cur_max < best_c5:
                best_c5 = cur_max
                best_h = h.copy()
        else:
            break
    
    return best_h, best_c5
```

```python
def flip_delta(h, i, n):
    """Compute the delta vector for correlation when flipping h[i] from 1 to 0."""
    # h[i] = 1, flipping to 0
    # For each k' = 0, ..., n-1:
    #   delta at lag +k' (index n-1+k' in full corr):
    #     -(1-h[i+k']) if i+k' < n else 0
    #     + h[i-k'] if i >= k' else 0
    #   delta at lag -k' (index n-1-k' in full corr):
    #     -(1-h[i-k']) if i >= k' else 0
    #     + h[i+k'] if i+k' < n else 0
    
    delta = np.zeros(2*n-1)
    # k' from 0 to n-1
    kp = np.arange(n)  # k' values
    
    # For lag +k' (index n-1+k'):
    valid_plus = kp <= (n-1-i)  # i+k' < n
    valid_minus = kp <= i  # i >= k'
    
    delta[n-1+kp[valid_plus]] -= (1 - h[i+kp[valid_plus]])
    delta[n-1+kp[valid_minus]] += h[i-kp[valid_minus]]
    
    # For lag -k' (index n-1-k'):
    delta[n-1-kp[valid_minus]] -= (1 - h[i-kp[valid_minus]])
    delta[n-1-kp[valid_plus]] += h[i+kp[valid_plus]]
    
    return delta
```
