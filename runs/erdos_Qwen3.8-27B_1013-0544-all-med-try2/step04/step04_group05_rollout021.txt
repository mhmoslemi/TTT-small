        # Threshold near boundaries
        h[np.abs(h) < threshold_eps] = 0.0
        h[np.abs(h - 1.0) < threshold_eps] = 1.0
        h = project_h(h, target)
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    return best_h, best_c5

def slsqp_polish(h, target, rem_fn, max_iter=8000):
    n = len(h)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    bounds = Bounds(np.zeros(n), np.ones(n))
    betas = [200, 800, 3000, 10000]
    for beta in betas:
        if rem_fn() < 10:
            break
        def obj_fn(h_in):
            c = np.correlate(h_in, 1.0 - h_in, mode="full")
            return smooth_max(c, beta) * dx
        def con_fn(h_in):
            return h_in.sum() - target
        def jac_fn(h_in):
            return np.ones(n)
        try:
            res = minimize(obj_fn, h, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': con_fn, 'jac': jac_fn}],
                          options={'maxiter': max_iter, 'ftol': 1e-16})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                h = h_new
                best_c5 = c5_new
        except Exception:
            pass
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = np.full(400, 0.5)

    n = len(h)
    target = n / 2.0
    h = project_h(h, target)
    best_c5 = compute_c5(h)
    best_h = h.copy()

    # Phase 1: Coordinate descent (main workhorse)
    h, c5_cd = coord_descend(h, target, rem, n_sweeps=5000, grid_size=48)
    if c5_cd < best_c5:
        best_c5 = c5_cd
        best_h = h.copy()

    # Phase 2: More coordinate descent with different params
    if rem() > 60:
        h2 = best_h.copy()
        h2, c5_cd2 = coord_descend(h2, target, rem, n_sweeps=3000, grid_size=64, threshold_eps=0.001)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h2.copy()

    # Phase 3: SLSQP polish
    if rem() > 30:
        h3, c5_sl = slsqp_polish(best_h, target, rem)
        if c5_sl < best_c5:
            best_c5 = c5_sl
            best_h = h3.copy()

    # Phase 4: Try higher resolution
    if rem() > 120:
        n2 = n * 2
        target2 = n2 / 2.0
        x_old = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, n2)
        h4 = project_h(np.interp(x_new, x_old, best_h), target2)
        c5_4 = compute_c5(h4)
        h4, c5_4 = coord_descend(h4, target2, rem, n_sweeps=2000, grid_size=40)
        if rem() > 30:
            h4, c5_4 = slsqp_polish(h4, target2, rem, max_iter=4000)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4
            n = n2
            target = target2

    return best_h, best_c5, len(best_h)
