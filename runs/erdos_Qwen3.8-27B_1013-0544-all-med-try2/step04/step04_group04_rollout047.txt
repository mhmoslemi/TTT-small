    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    for _ in range(n_steps):
        if time_fn() < time_limit:
            break

                c5_cur = c5_try
                improved = True

        m = np.max(corr)
        s = np.sum(np.exp(beta * (corr - m)))
        return (m + np.log(s) / beta) * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _lag_grad(grad, h, lag, w, n):
    if lag == 0:
        grad += w * (1.0 - 2.0 * h)
    elif 0 < lag < n:
        grad[lag:] += w * (1.0 - h[:-lag])
        grad[:n - lag] -= w * h[lag:]
    elif lag < 0 and abs(lag) < n:
        m2 = -lag
        grad[:n - m2] += w * (1.0 - h[m2:])
        grad[m2:] -= w * h[:n - m2]
    return grad

def topk_grad(h, k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    kk = min(k, len(corr))
    top_idx = np.argsort(corr)[::-1][:kk]
    obj = np.mean(corr[top_idx]) * dx
    grad = np.zeros(n)
    w = dx / kk
    for idx in top_idx:
        lag = int(idx) - (n - 1)
        if -n < lag < n:
            _lag_grad(grad, h, lag, w, n)
    return obj, grad

def topk_descent(h, target, k, n_steps, lr, max_time, time_fn):
    n = len(h)
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    for _ in range(n_steps):
        if time_fn() < max_time:
            break
        _, grad = topk_grad(h_cur, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        step = lr
        for _ in range(15):
            h_try = project_h(h_cur - step * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_cur - 1e-15:
                h_cur = h_try
                c5_cur = c5_try
                break
            step *= 0.7
        else:
            break
    return h_cur, c5_cur

def smooth_max_obj(x, beta, n, dx):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    s = np.sum(np.exp(np.clip(beta * (corr - m), -500, 500)))
    return (m + np.log(s) / beta) * dx

def smooth_max_grad(x, beta, n, dx):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    e = np.exp(np.clip(beta * (corr - m), -500, 500))
    w = e / np.sum(e)
    g = np.zeros(n)
    for idx in range(corr.shape[0]):
        if w[idx] < 1e-12:
            continue
        wi = w[idx] * dx
        lag = idx - (n - 1)
        if -n < lag < n:
            _lag_grad(g, x, lag, wi, n)
    return g

def do_slsqp(h, beta, target, maxiter, time_limit, time_fn):
    n = len(h)
    dx = 2.0 / n
    def obj(x):
        return smooth_max_obj(x, beta, n, dx)
    def grad(x):
        return smooth_max_grad(x, beta, n, dx)
    def con(x):
        return x.sum() - target
    def con_j(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con, 'jac': con_j}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, jac=grad,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_h(res.x, target)
        return h_new, compute_c5(h_new)
    except Exception:
        return h.copy(), compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(200, 0.5)
        best_h[:100] = 0.75
        best_h[100:] = 0.25
        best_h = project_h(best_h, 100.0)

    best_c5 = compute_c5(best_h)
    n = len(best_h)
    target = n / 2.0

    # Phase 1: SLSQP warmup
    if time_left() > 120:
        h1, c5_1 = do_slsqp(best_h, 300.0, target, 3000, 90, time_left)
        if c5_1 < best_c5:
            best_h, best_c5 = h1, c5_1

    # Phase 2: ILS with topk gradient descent
    n_ils = n
    target_ils = n_ils / 2.0
    ils_end = 0.55
    no_improve = 0
    cycle = 0

    while time_left() > budget_s * (1.0 - ils_end):
        cycle += 1
        # Adaptive perturbation scale
        scale = 0.015 + 0.025 * min(no_improve * 0.1, 4.0)
        
        # Occasionally do a resolution-change perturbation
        if cycle % 15 == 0 and time_left() > budget_s * 0.3:
            n_lo = n_ils // 2
            x_lo = np.linspace(0, 2, n_lo)
            h_lo = np.interp(x_lo, np.linspace(0, 2, n_ils), best_h)
            h_lo = project_h(h_lo, n_lo / 2.0)
            h_lo, _ = topk_descent(h_lo, n_lo / 2.0, 5, 40, 0.06,
                                   budget_s * 0.05, time_left)
            x_hi = np.linspace(0, 2, n_ils)
            h_pert = project_h(np.interp(x_hi, x_lo, h_lo), target_ils)
        else:
            h_pert = project_h(best_h + np.random.randn(n_ils) * scale, target_ils)

        # Multi-scale topk descent
        h_cur = h_pert
        c5_cur = compute_c5(h_cur)
        k_list = [max(5, n_ils // 3), max(3, n_ils // 6), max(2, n_ils // 12), 3, 2]
        for k in k_list:
            if time_left() < budget_s * (1.0 - ils_end) + 5:
                break
            h_new, c5_new = topk_descent(h_cur, target_ils, k, 60, 0.06,
                                         budget_s * (1.0 - ils_end) - 5, time_left)
            if c5_new < c5_cur:
                h_cur = h_new
                c5_cur = c5_new

        if c5_cur < best_c5 - 1e-12:
            best_h = h_cur.copy()
            best_c5 = c5_cur
            no_improve = 0
            n_ils = n
            target_ils = n_ils / 2.0
        else:
            no_improve += 1

    # Phase 3: SLSQP polish with increasing beta
    for beta in [1000, 5000, 20000, 60000]:
        if time_left() < 40:
            break
        h_new, c5_new = do_slsqp(best_h, beta, target, 6000, 35, time_left)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 4: Fine ILS with small perturbations
    while time_left() > 15:
        h_pert = project_h(best_h + np.random.randn(n) * 0.008, target)
        h_new, c5_new = topk_descent(h_pert, target, 4, 40, 0.04, 8, time_left)
        if c5_new < best_c5 - 1e-12:
            best_h, best_c5 = h_new, c5_new

    # Phase 5: Final SLSQP
    if time_left() > 20:
        h_new, c5_new = do_slsqp(best_h, 80000, target, 8000, 15, time_left)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    return best_h, float(best_c5), len(best_h)
