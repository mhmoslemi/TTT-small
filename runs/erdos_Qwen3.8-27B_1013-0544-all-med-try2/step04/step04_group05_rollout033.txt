```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    """Project h onto {x in [0,1]^n : sum(x) = target_sum}"""
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            # Need to find where to add/remove
            if diff > 0:
                # Add to positions with highest value (closest to 1)
                idx = np.argsort(h)[::-1]
                for i in idx:
                    add = min(diff, 1.0 - h[i])
                    h[i] += add
                    diff -= add
                    if diff < 1e-14:
                        break
            else:
                idx = np.argsort(h)
                for i in idx:
                    rem = min(-diff, h[i])
                    h[i] -= rem
                    diff += rem
                    if diff > -1e-14:
                        break
            break
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_subgrad(h, lags):
    n = len(h)
    grad = np.zeros(n)
    for lag in lags:
        if lag == 0:
            grad += (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += (1.0 - h[:-lag])
            grad[:n - lag] -= h[lag:]
        else:
            m = -lag
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad

def smooth_max_val(corr, beta):
    m = np.max(corr)
    x_shift = np.clip(beta * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(corr)) / beta

def subgrad_optimize(h, target, time_fn, max_iters, n_lags, step_init, momentum, 
                     perturb_every, perturb_scale, max_step_halves=8):
    """Improved subgradient with backtracking."""
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    vel = np.zeros(n)
    step = step_init
    no_improve = 0
    
    for it in range(max_iters):
        if time_fn() < 3:
            break
        
        corr = np.correlate(best, 1.0 - best, mode="full")
        top_idx = np.argsort(corr)[::-1][:n_lags]
        lags = (top_idx - (n - 1)).astype(int)
        g = compute_subgrad(best, lags)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        
        vel = momentum * vel + (1.0 - momentum) * (g / gn)
        vn = np.linalg.norm(vel)
        if vn < 1e-14:
            break
        direction = -vel / vn
        
        # Backtracking line search
        s = step
        found = False
        for _ in range(max_step_halves):
            candidate = project_h(best + s * direction, target)
            c5c = compute_c5(candidate)
            if c5c < best_c5 - 1e-15:
                best = candidate
                best_c5 = c5c
                no_improve = 0
                found = True
                break
            s *= 0.5
        
        if found:
            step = s  # Remember successful step
        else:
            no_improve += 1
            step = max(s, step_init * 0.1)  # Don't let step get too small
            
            if no_improve >= perturb_every:
                # Structured perturbation
                scale = perturb_scale * (1.0 + 0.5 * np.random.random())
                if np.random.random() < 0.3:
                    # Block perturbation
                    block_size = max(1, n // 20)
                    start = np.random.randint(0, n - block_size)
                    p = np.zeros(n)
                    p[start:start+block_size] = np.random.randn(block_size) * scale
                else:
                    p = np.random.randn(n) * scale
                candidate = project_h(best + p, target)
                c5c = compute_c5(candidate)
                if c5c < best_c5:
                    best = candidate
                    best_c5 = c5c
                    vel = np.zeros(n)
                    no_improve = 0
                    step = step_init
                elif no_improve > perturb_every * 3:
                    # Try even larger perturbation
                    p = np.random.randn(n) * perturb_scale * 3
                    candidate = project_h(best + p, target)
                    c5c = compute_c5(candidate)
                    if c5c < best_c5:
                        best = candidate
                        best_c5 = c5c
                        vel = np.zeros(n)
                        no_improve = 0
                        step = step_init
                    else:

def slsqp_polish(h, target, time_fn, max_iter=20000):
    n = len(h)
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    best = h.copy()
    best_c5 = compute_c5(best)
    
    betas = [200, 800, 3000, 10000]
    for beta in betas:
        if time_fn() < 10:
            break
        def obj_fn(h_in):
            c = np.correlate(h_in, 1.0 - h_in, mode="full")
            return smooth_max_val(c, beta) * dx
        
        def con_fn(h_in):
            return h_in.sum() - target
        
        def jac_fn(h_in):
            return np.ones(n)
        
        try:
            res = minimize(obj_fn, best, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': con_fn, 'jac': jac_fn}],
                          options={'maxiter': max_iter, 'ftol': 1e-20})
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = free.sum()
        if cnt == 0:
            if diff > 0:
                idx = np.argsort(h)[::-1]
                for i in idx:
                    add = min(diff, 1.0 - h[i])
                    h[i] += add
                    diff -= add
                    if diff < 1e-14:
                        break
            else:
                idx = np.argsort(h)
                for i in idx:
                    rem = min(-diff, h[i])
                    h[i] -= rem
                    diff += rem
                    if diff > -1e-14:
                        break
            break
        h[free] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_subgrad(h, lags):
    n = len(h)
    grad = np.zeros(n)
    for lag in lags:
        if lag == 0:
            grad += (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += (1.0 - h[:-lag])
            grad[:n - lag] -= h[lag:]
        else:
            m = -lag
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad

    m = np.max(corr)

    return m + np.log(np.sum(np.exp(x_shift)) / len(corr)) / beta
```
