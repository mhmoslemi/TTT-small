import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def con_fun(h, ts):
    return h.sum() - ts

def con_jac(h, ts):
    return np.ones_like(h)

    dx = 2.0 / n
    ts = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))

    try:
        res = minimize(obj_smooth, h0, args=(dx, beta), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, ts)
    except:
        return h0

            la = -lag

            grad[j] -= h[j + la]
    h_new = project_feasible(h - lr * grad / top_k, ts)

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        mask = (h < 1.0 - 1e-12) if diff > 0 else (h > 1e-12)
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def con_fun(h, ts):
    return h.sum() - ts

def con_jac(h, ts):
    return np.ones_like(h)

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def slsqp_run(h0, n, beta, maxiter=3000, ftol=1e-14):
    dx = 2.0 / n
    ts = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts,)}]
    try:
        res = minimize(obj_smooth, h0, args=(dx, beta), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, ts)
    except Exception:
        return h0

def focused_gradient(h, n, top_k=3, lr=0.003):
    ts = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    top = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for idx in top:
        lag = idx - (n - 1)
        if lag >= 0:
            j = np.arange(n - lag)
            grad[j] += (1.0 - h[j + lag])
            grad[j + lag] -= h[j]
        else:
            la = -lag
            j = np.arange(n - la)
            grad[j + la] += (1.0 - h[j])
            grad[j] -= h[j + la]
    h_new = project_feasible(h - lr * grad / top_k, ts)
    return h_new, compute_c5(h_new), compute_c5(h)
