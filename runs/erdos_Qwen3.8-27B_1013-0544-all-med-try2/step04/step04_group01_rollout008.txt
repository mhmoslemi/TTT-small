                nm = (c_val + delta).max() * dx
                imp = best_c - nm
                if imp > best_improve:
                    best_improve = imp
                    best_i = i
                    best_dir = 1
            if h[i] - d >= -1e-12:
                delta = np.zeros(2 * n - 1)
                delta[n - 1 - i: 2 * n - 1 - i] -= d * (1.0 - h)
                delta[i: n + i] += d * h[::-1]
                nm = (c_val + delta).max() * dx
                imp = best_c - nm
                if imp > best_improve:
                    best_improve = imp
                    best_i = i
                    best_dir = -1
        if best_i < 0:
            n_reject += 1
            continue
        n_reject = 0
        if best_dir == 1:
            delta = np.zeros(2 * n - 1)
            delta[n - 1 - best_i: 2 * n - 1 - best_i] += d * (1.0 - h)
            delta[best_i: n + best_i] -= d * h[::-1]
        else:
            delta = np.zeros(2 * n - 1)
            delta[n - 1 - best_i: 2 * n - 1 - best_i] -= d * (1.0 - h)
            delta[best_i: n + best_i] += d * h[::-1]
        c_val += delta
        h[best_i] += best_dir * d
        best_c = c_val.max() * dx
    return h, best_c

def greedy_cd_pairs(h, n, d, deadline, max_rounds=5000, radius=15):
    dx = 2.0 / n
    c_val = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    best_c = c_val.max() * dx
    h = h.copy()
    n_reject = 0
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        if n_reject > 300:
            break
        best_improve = 1e-13
        best_move = None
        for i in range(n):
            j_lo = max(0, i - radius)
            j_hi = min(n, i + radius + 1)
            for j in range(j_lo, j_hi):
                if i == j:
                    continue
                # Try h[i]+=d, h[j]-=d
                if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                    delta = np.zeros(2 * n - 1)
                    delta[n - 1 - i: 2 * n - 1 - i] += d * (1.0 - h)
                    delta[i: n + i] -= d * h[::-1]
                    delta[n - 1 - j: 2 * n - 1 - j] -= d * (1.0 - h)
                    delta[j: n + j] += d * h[::-1]
                    nm = (c_val + delta).max() * dx
                    imp = best_c - nm
                    if imp > best_improve:
                        best_improve = imp
                        best_move = (i, j, 1)
        if best_move is None:
            n_reject += 1
            continue
        n_reject = 0
        i, j, sgn = best_move
        delta = np.zeros(2 * n - 1)
        delta[n - 1 - i: 2 * n - 1 - i] += sgn * d * (1.0 - h)
        delta[i: n + i] -= sgn * d * h[::-1]
        delta[n - 1 - j: 2 * n - 1 - j] -= sgn * d * (1.0 - h)
        delta[j: n + j] += sgn * d * h[::-1]
        c_val += delta
        h[i] += sgn * d
        h[j] -= sgn * d
        best_c = c_val.max() * dx
    return h, best_c

def optimize_point(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    # SLSQP stages
    for a in [500, 2000, 10000, 50000]:
        if time.time() > deadline - 10:
            break
        ht = slsqp(best_h, n, a, 1500)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht
    # Greedy CD with decreasing steps
    for d in [0.1, 0.05, 0.02, 0.01, 0.005]:
        if time.time() > deadline - 15:
            break
        ht, c = greedy_cd(best_h, n, d, min(time.time() + 60, deadline - 15))
        if c < best_c:
            best_c = c
            best_h = ht
        best_h = project_feasible(best_h, n)
    # Pair CD
    if time.time() < deadline - 10:
        ht, c = greedy_cd_pairs(best_h, n, 0.01, min(time.time() + 40, deadline - 10),
                                max_rounds=3000, radius=20)
        if c < best_c:
            best_c = c
            best_h = ht
    # Fine CD
    for d in [0.002, 0.001, 0.0005]:
        if time.time() > deadline - 5:
            break
        ht, c = greedy_cd(best_h, n, d, min(time.time() + 40, deadline - 5))
        if c < best_c:
            best_c = c
            best_h = ht
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 4

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(120, 0.5)
        n_init = 120

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: Optimize at initial n
    alloc1 = min(250, (deadline - t0) * 0.35)
    h_opt, c_opt = optimize_point(best_h, n_init, t0 + alloc1, rng)
    if c_opt < best_c:
        best_c = c_opt
        best_h = h_opt

    # Phase 2: Try other n values
    n_candidates = [150, 250, 300]
    if n_init not in n_candidates:
        n_candidates.insert(0, n_init)
    for n_try in n_candidates:
        if n_try == n_init or time.time() > deadline - 80:
            continue
        remaining = deadline - time.time() - 60
        alloc = min(80, remaining * 0.4)
        if alloc < 30:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        h_opt, c_opt = optimize_point(h_interp, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Random restarts from best
    if time.time() < deadline - 50:
        n = len(best_h)
        for scale in [0.03, 0.06, 0.1, 0.15]:
            if time.time() > deadline - 25:
                break
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(60, (deadline - time.time() - 25) * 0.6)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_point(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Final fine-tuning
    if time.time() < deadline - 5:
        n = len(best_h)
        for d in [0.001, 0.0005, 0.0002]:
            if time.time() > deadline - 2:
                break
            best_h, best_c = greedy_cd(best_h, n, d, deadline - 1)
        if time.time() < deadline - 3:
            best_h, best_c = greedy_cd_pairs(best_h, n, 0.001, deadline - 1,
                                             max_rounds=2000, radius=10)
        best_h = project_feasible(best_h, len(best_h))
        best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
