import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_lag(h, lag, n):
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif 0 < lag < n:
        g[lag:] += 1.0 - h[:-lag]
        g[:n - lag] -= h[lag:]
    elif 0 < -lag < n:
        m = -lag
        g[:n - m] += 1.0 - h[m:]
        g[m:] -= h[:n - m]
    return g

def subgrad(h, top_m=8, temp=500.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = np.max(corr)
    idx = np.argsort(corr)[::-1][:top_m]
    w = np.exp(temp * (corr[idx] - mx))
    w /= w.sum()
    g = np.zeros(n)
    for i, k in enumerate(idx):
        g += w[i] * grad_lag(h, int(k) - (n - 1), n)
    return mx * dx, g * dx

def _slsqp_obj(x, beta_box):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    mx = np.max(corr)
    se = np.sum(np.exp(beta_box[0] * (corr - mx)))
    return (mx + np.log(se) / beta_box[0]) * dx

def _slsqp_grad(x, beta_box):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    mx = np.max(corr)
    xe = np.clip(beta_box[0] * (corr - mx), -500, 500)
    ev = np.exp(xe)
    sw = np.sum(ev)
    wts = ev / sw
    g = np.zeros(n)
    active = np.where(wts > 1e-8)[0]
    for k in active:
        wt = wts[k] * dx
        lag = int(k) - (n - 1)
        if lag == 0:
            g += wt * (1.0 - 2.0 * x)
        elif 0 < lag < n:
            g[lag:] += wt * (1.0 - x[:-lag])
            g[:n - lag] -= wt * x[lag:]
        elif 0 < -lag < n:
            m = -lag
            g[:n - m] += wt * (1.0 - x[m:])
            g[m:] -= wt * x[:n - m]
    return g

def slsqp_polish(h0, beta, maxiter, target, time_fn):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    beta_box = [beta]
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_slsqp_obj, h0, method='SLSQP', bounds=bounds,
                       jac=lambda x: _slsqp_grad(x, beta_box),
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_h(res.x, target)
        return h_new
    except Exception:
        return h0

    elif 0 < lag < n:

import numpy as np
from scipy.optimize import minimize
import time

_slq = {'beta': 1.0, 'target': 0.0}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _f(x):
    n = len(x)
    dx = 2.0 / n
    b = _slq['beta']
    corr = np.correlate(x, 1.0 - x, mode="full")
    mx = np.max(corr)
    se = np.sum(np.exp(np.clip(b * (corr - mx), -500, 500)))
    return (mx + np.log(se) / b) * dx

def _g(x):
    n = len(x)
    dx = 2.0 / n
    b = _slq['beta']
    corr = np.correlate(x, 1.0 - x, mode="full")
    mx = np.max(corr)
    ev = np.exp(np.clip(b * (corr - mx), -500, 500))
    wts = ev / np.sum(ev)
    g = np.zeros(n)
    active = np.where(wts > 1e-8)[0]
    for k in active:
        wt = wts[k] * dx
        lag = int(k) - (n - 1)
        if lag == 0:
            g += wt * (1.0 - 2.0 * x)
        elif 0 < lag < n:
            g[lag:] += wt * (1.0 - x[:-lag])
            g[:n - lag] -= wt * x[lag:]
        elif 0 < -lag < n:
            m = -lag
            g[:n - m] += wt * (1.0 - x[m:])
            g[m:] -= wt * x[:n - m]
    return g

def _c(x):
    return x.sum() - _slq['target']

def _cj(x):
    return np.ones(len(x))

def run_slsqp(h0, target, t0, budget, min_left, betas, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _c, 'jac': _cj}]
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    _slq['target'] = target
    for beta in betas:
        if (time.time() - t0) > budget - min_left:
            break
        _slq['beta'] = float(beta)
        try:
            res = minimize(_f, best_h, method='SLSQP', bounds=bounds,
                          jac=_g, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-16})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        except Exception:
            pass
    return best_h, best_c5

def subgrad_step(h, top_m, temp, eta, target):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = np.max(corr)
    idx = np.argsort(corr)[::-1][:top_m]
    w = np.exp(temp * (corr[idx] - mx))
    w /= w.sum()
    g = np.zeros(n)
    for i in range(top_m):
        lag = int(idx[i]) - (n - 1)
        if lag == 0:
            g += w[i] * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            g[lag:] += w[i] * (1.0 - h[:-lag])
            g[:n - lag] -= w[i] * h[lag:]
        elif 0 < -lag < n:
            m = -lag
            g[:n - m] += w[i] * (1.0 - h[m:])
            g[m:] -= w[i] * h[:n - m]
    return project_h(h - eta * g * dx, target)

def run_subgrad(h0, target, t0, budget, min_left, top_m, temp, eta0, n_steps):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    h = h0.copy()
    for it in range(n_steps):
        if (time.time() - t0) > budget - min_left:
            break
        eta = eta0 / np.sqrt(it + 1.0)
        h = subgrad_step(h, top_m, temp, eta, target)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5
