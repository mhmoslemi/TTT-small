from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-15) & (h < 1.0 - 1e-15)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def resample_fn(h, n_new):
    if len(h) == n_new:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, len(h), endpoint=False)
    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def sobj(h_in, beta, n):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, beta) * (2.0 / n)

def sum_eq(x, t):
    return np.sum(x) - t

def sum_eq_j(x, t):
    return np.ones(x.shape)

def slsqp_opt(h0, n, target, beta, maxiter=8000, ftol=1e-15):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_j, 'args': (target,)}]
    try:
        r = minimize(sobj, h0, args=(beta, n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def lag_gradient(cur, lag, n):
    grad = np.zeros(n)
    if lag >= 0 and lag < n:
        grad[:n - lag] += (1.0 - cur[lag:])
        grad[lag:] -= cur[:n - lag]
    elif -lag < n:
        la = -lag
        grad[la:] += (1.0 - cur[:n - la])
        grad[:n - la] -= cur[la:]
    return grad

def hot_potato(h, target, n, max_iters, t0, budget, step_init=0.015, top_k=3):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    step = step_init
    stall = 0
    for it in range(max_iters):
        if time.time() - t0 > budget:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        top_idx = np.argsort(corr)[::-1][:top_k]
        grad = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            grad += lag_gradient(cur, lag, n)
        grad /= top_k
        gn = np.linalg.norm(grad)
        if gn < 1e-14:

        trial = project_h(cur - (step / gn) * grad, target)
        c_try = compute_c5(trial)
        if c_try < best_c5 - 1e-15:

            best_c5 = c_try
            stall = 0

def binary_swap(h_bin, n, t0, budget, max_iters=300000):
    h = h_bin.copy()
    cur_max = float(np.max(compute_lags(h)))
    best_h = h.copy()
    best_max = cur_max
    rng = np.random.default_rng(54321)

        b = zeros[rng.integers(len(zeros))]
