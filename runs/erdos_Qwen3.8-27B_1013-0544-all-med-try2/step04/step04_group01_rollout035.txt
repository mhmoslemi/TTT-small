def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(c.max())

def compute_c5_all(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_fn(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def sum_con(x, n):
    return np.sum(x) - n / 2.0

def slsqp_opt(h0, n, alpha, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h0, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h0.copy()

def subgrad_sub(h, n, s):
    """Compute subgradient of c_s with respect to h."""
    dx = 2.0 / n
    g = np.zeros(n)
    for j in range(n):
        if 0 <= j + s < n:
            g[j] += dx * (1 - h[j + s])
        if 0 <= j - s < n:
            g[j] -= dx * h[j - s]
    return g

def subgrad_descend(h, n, deadline, top_m=5, max_iter=30000):
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = 0.008
    momentum = np.zeros(n)
    beta = 0.9

    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = compute_c5_all(h, n)
        c_max = c.max()
        top_idx = np.argsort(c)[-top_m:]
        g = np.zeros(n)
        for k in top_idx:
            s = k - (n - 1)
            g += subgrad_sub(h, n, s)
        g /= top_m
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        momentum = beta * momentum + (1 - beta) * g
        mn = np.linalg.norm(momentum)
        if mn < 1e-14:
            break
        improved = False
        for scale in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            h_try = h - scale * momentum / mn
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-13:
                h = h_try
                best_c = c_try
                best_h = h.copy()
                lr = min(lr * 1.03, 0.06)
                improved = True
                break
        if not improved:
            lr = max(lr * 0.85, 0.0003)
    return best_h, best_c

def coord_refine(h, n, deadline):
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    for d in [0.005, 0.002, 0.001, 0.0005]:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_arr = compute_c5_all(h, n)
            c_max = c_arr.max()
            top_k = np.where(c_arr >= c_max - 0.005)[0]
            s_set = set(int(k - (n - 1)) for k in top_k)
            for i in range(n):
                if time.time() > deadline:
                    break
                for sign in [1, -1]:
                    hi = h[i] + sign * d
                    if not (0.0 <= hi <= 1.0):
                        continue
                    partners = [(i+1)%n, (i+n//2)%n, (i+3*n//4)%n, (i-1+n)%n]
                    for j in partners:
                        if j == i:
                            continue
                        hj = h[j] - sign * d
                        if not (0.0 <= hj <= 1.0):
                            continue
                        h[i], h[j] = hi, hj
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                            break
                        else:
                            h[i], h[j] = h[i]-sign*d, h[j]+sign*d
    return h, best_c

def thue_morse(n):
    t = np.zeros(n)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t

def optimize_one(h0, n, deadline):
    h = project_feasible(h0.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alphas = [2000, 10000, 50000, 200000, 1000000, 5000000]
    for a in alphas:
        if time.time() > deadline - 20:
            break
        h_try = slsqp_opt(best_h, n, a, maxiter=4000)
        c_try = compute_c5(h_try, n)
        if c_try < best_c - 1e-14:
            best_c = c_try
            best_h = h_try
    if time.time() < deadline - 15:
        for tm in [3, 5, 10]:
            if time.time() > deadline - 10:
                break
            h_sg, c_sg = subgrad_descend(best_h, n, min(deadline - 10, time.time() + 30), top_m=tm, max_iter=8000)
            if c_sg < best_c:
                best_c = c_sg
                best_h = h_sg
    if time.time() < deadline - 5:
        h_try = slsqp_opt(best_h, n, 10000000, maxiter=5000)
        c_try = compute_c5(h_try, n)
        if c_try < best_c - 1e-14:
            best_c = c_try
            best_h = h_try
    if time.time() < deadline - 1:
        best_h, best_c = coord_refine(best_h, n, deadline - 0.5)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c
