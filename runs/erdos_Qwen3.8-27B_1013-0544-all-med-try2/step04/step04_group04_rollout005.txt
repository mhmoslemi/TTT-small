    _g_penalty[0] = 50.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_penalty_obj, h, jac=_penalty_grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
        h_new = project_h(res.x, n / 2.0)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, compute_c5(h)

def slsqp_run(h, beta, maxiter):
    n = len(h)
    _g_n[0] = n
    _g_beta[0] = beta
    _g_target[0] = n / 2.0
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _slsqp_con, 'jac': _slsqp_con_jac}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_h(res.x, n / 2.0)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, compute_c5(h)

def adam_subgrad(h, target, n_iter, beta, lr_init):
    n = len(h)
    best_c5 = compute_c5(h)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2 = 0.85, 0.999
    no_imp = 0
    for it in range(n_iter):
        if time_left() < 3:
            break
        t += 1
        _, _, w, dx = _smooth_vals(h, beta)
        grad = _corr_grad(h, w, n, dx)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad ** 2)
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        lr_t = lr_init * np.sqrt(1 - b2 ** t) / (1 - b1 ** t)
        step = lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        found = False
        for scale in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            h_try = project_h(h - scale * step, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                h = h_try
                best_c5 = c5_try
                found = True
                break
        if found:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 150 == 0:
                sc = 0.003 * min(30, no_imp // 150 + 1)
                h_p = project_h(h + np.random.randn(n) * sc, target)
                if compute_c5(h_p) < best_c5:
                    h = h_p
                    no_imp = 0
            if no_imp > n_iter // 4:
                break
    return h, best_c5

def coord_descent(h, target, n_rounds):
    n = len(h)
    best_c5 = compute_c5(h)
    for r in range(n_rounds):
        if time_left() < 5:
            break
        perm = np.random.permutation(n)
        any_improve = False
        for i in perm:
            rv = [0.0, 1.0, 0.1, 0.9, 0.2, 0.8, 0.3, 0.7, 0.4, 0.6]
            base = h[i]
            best_val = base
            for val in rv:
                if abs(val - base) < 1e-6:
                    continue
                h[i] = val
                h = project_h(h, target)
                c5_v = compute_c5(h)
                if c5_v < best_c5 - 1e-12:
                    best_c5 = c5_v
                    best_val = val
                    any_improve = True
            h[i] = best_val
            h = project_h(h, target)
        if not any_improve:
            break
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    _g_start[0] = time.time()
    _g_budget[0] = budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(200, 0.5)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B at current resolution, increasing beta
    for beta in [50, 200, 1000, 5000, 50000, 200000]:
        if time_left() < 15:
            break
        h_new, c5_new = lbfgs_run(best_h, beta, 3000)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 2: SLSQP refinement
    for beta in [2000, 20000, 200000]:
        if time_left() < 15:
            break
        h_new, c5_new = slsqp_run(best_h, beta, 4000)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 3: Adam subgradient
    n_f = len(best_h)
    tgt = n_f / 2.0
    best_h, best_c5 = adam_subgrad(best_h, tgt, 40000, beta=5000, lr_init=0.015)

    # Phase 4: SLSQP polish
    if time_left() > 20:
        for beta in [50000, 500000]:
            if time_left() < 12:
                break
            h_new, c5_new = slsqp_run(best_h, beta, 4000)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new

    # Phase 5: Coordinate descent
    if time_left() > 30:
        tgt = len(best_h) / 2.0
        best_h, best_c5 = coord_descent(best_h, tgt, 30)

    # Phase 6: Increase resolution
    if time_left() > 50:
        n_new = min(1000, 2 * len(best_h))
        if n_new > len(best_h):
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_new)
            h_res = np.interp(x_new, x_old, best_h)
            h_res = project_h(h_res, n_new / 2.0)
            for beta in [500, 5000, 100000]:
                if time_left() < 15:
                    break
                h_t, c5_t = lbfgs_run(h_res, beta, 2000)
                if c5_t < compute_c5(h_res):
                    h_res = h_t
            c5_res = compute_c5(h_res)
            if c5_res < best_c5:
                best_h, best_c5 = h_res, c5_res

    # Phase 7: Final SLSQP + subgradient
    if time_left() > 15:
        h_new, c5_new = slsqp_run(best_h, 500000, 5000)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
    if time_left() > 10:
        tgt = len(best_h) / 2.0
        best_h, best_c5 = adam_subgrad(best_h, tgt, 15000, beta=20000, lr_init=0.008)
    if time_left() > 10:
        h_new, c5_new = slsqp_run(best_h, 1000000, 3000)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Final projection to ensure feasibility
    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
