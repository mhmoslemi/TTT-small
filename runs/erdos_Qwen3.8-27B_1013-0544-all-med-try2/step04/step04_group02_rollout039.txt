    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    parent = None
    if 'initial_h_values' in dir() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float).copy()
    if parent is None:
        parent = np.full(200, 0.5)
    pn = len(parent)

    best_h = parent.copy()
    best_c5 = compute_c5(parent)
    best_n = pn

    # Phase 1: LP iterations (35% budget)
    d1 = t0 + budget_s * 0.35
    h1, c1 = lp_phase(parent.copy(), pn, d1, max_iters=400, seed=seed)
    if c1 < best_c5:
        best_h, best_c5, best_n = h1, c1, pn

    # Phase 2: Subgradient polish (10%)
    d2 = t0 + budget_s * 0.45
    h2, c2 = subgrad_polish(best_h.copy(), best_n, d2, seed + 1)
    if c2 < best_c5:
        best_h, best_c5 = h2, c2

    # Phase 3: LP round 2 (15%)
    d3 = t0 + budget_s * 0.60
    h3, c3 = lp_phase(best_h.copy(), best_n, d3, max_iters=200, seed=seed + 10)
    if c3 < best_c5:
        best_h, best_c5 = h3, c3

    # Phase 4: SLSQP refine (10%)
    d4 = t0 + budget_s * 0.70
    h4, c4 = slsqp_refine(best_h.copy(), best_n, d4, seed + 20)
    if c4 < best_c5:
        best_h, best_c5 = h4, c4

    # Phase 5: Subgradient again (10%)
    d5 = t0 + budget_s * 0.80
    h5, c5v = subgrad_polish(best_h.copy(), best_n, d5, seed + 30)
    if c5v < best_c5:
        best_h, best_c5 = h5, c5v

    # Phase 6: LP round 3 (8%)
    d6 = t0 + budget_s * 0.88
    h6, c6 = lp_phase(best_h.copy(), best_n, d6, max_iters=80, seed=seed + 40)
    if c6 < best_c5:
        best_h, best_c5 = h6, c6

    # Phase 7: Try other resolutions (8%)
    d7 = t0 + budget_s * 0.97
    for dn in [-40, 40, -80, 80, -20, 20]:
        if time.time() > d7:
            break
        nt = best_n + dn
        if nt < 100 or nt > 700:
            continue
        xo = np.linspace(0, 2, best_n, endpoint=False)
        xn = np.linspace(0, 2, nt, endpoint=False)
        h0 = np.interp(xn, xo, best_h)
        h0 = project_feasible(h0, nt / 2.0)
        h7a, c7a = lp_phase(h0, nt, min(d7, time.time() + 20), max_iters=40, seed=seed + nt)
        h7b, c7b = subgrad_polish(h7a, nt, min(d7, time.time() + 10), seed + nt)
        if c7b < best_c5:
            best_h, best_c5, best_n = h7b, c7b, nt

    # Final polish
    if time.time() < t0 + budget_s * 0.99:
        best_h, best_c5 = subgrad_polish(best_h, best_n, t0 + budget_s * 0.99, seed + 99)

    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
