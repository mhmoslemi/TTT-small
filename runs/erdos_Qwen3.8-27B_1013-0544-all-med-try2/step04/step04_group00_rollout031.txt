```python
def solve_1d_minmax(g, e, delta_min, delta_max, n_iter=60):
    """Find delta that minimizes max_k (g[k] + delta * e[k]) for delta in [delta_min, delta_max]"""
    lo = delta_min
    hi = delta_max
    for _ in range(n_iter):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        f1 = np.max(g + m1 * e)
        f2 = np.max(g + m2 * e)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    delta = (lo + hi) / 2
    return delta
```

**Approach: Direct min-max via iterative linearization**

The problem is: min_h max_k g_k(h) s.t. h ∈ [0,1]^n, ∑h = n/2

This is a minimax problem. A classical approach is:
- Geometric programming / Dinkelbach's method
- Or: introduce t, minimize t s.t. g_k(h) ≤ t for all k

The constraint g_k(h) ≤ t is non-convex (bilinear). But we can linearize it:
- At current h*, g_k(h) ≈ g_k(h*) + ∇g_k(h*)^T (h - h*)
- The linearized constraint is: g_k(h*) + ∇g_k(h*)^T (h - h*) ≤ t
- This is linear in (h, t)!
- So the linearized problem is a linear program!

This is the **cutting-plane / outer approximation** method:
1. Start with h_0, t_0 = max_k g_k(h_0)
2. Linearize all g_k at h_0
3. Solve the LP: min t s.t. g_k(h_0) + ∇g_k(h_0)^T (h - h_0) ≤ t, 0≤h≤1, ∑h=n/2
4. Let h_1 be the solution
5. Check if t_1 = max_k g_k(h_1) (the true objective)
6. If t_1 > t_0 + ε, add the violated constraint and re-solve
7. Repeat

This is a well-known technique for non-convex minimax problems. The LP is small (n+1 variables, O(n²) constraints) and can be solved quickly.

```python
def compute_sensitivity_matrix(h, n):
    dx = 2.0 / n
    m = 2 * n - 1
    E = np.zeros((n, m))
    ones = 1.0 - h
    for k in range(m):
        s = k - (n - 1)
        if s >= 0:
            # i+s valid when i < n-s
            E[:n-s, k] = ones[s:]  # (1-h[i+s]) for i=0..n-s-1
            E[s:, k] -= h[:n-s]    # -h[i-s] for i=s..n-1
        else:

            E[t:, k] = ones[:n-t]  # (1-h[i+s]) = (1-h[i-t]) for i=t..n-1
            E[:n-t, k] -= h[t:]    # -h[i-s] = -h[i+t] for i=0..n-t-1
    E *= dx
    return E
```

```python
            E[t:, k] = ones_h[:n-t]
            E[:n-t, k] -= h[t:]

def ternary_1d(g, e, lo, hi, n_iter=40):

            hi = m2
        else:
            lo = m1

def binary_ls(h, n, deadline, max_sweeps=500):
    """Binary local search with incremental evaluation."""
    g = compute_g(h, n)
    E = compute_E(h, n)
    m = 2 * n - 1
    target = int(n // 2)
    
    # Threshold to binary
    h_bin = np.zeros(n)
    idx = np.argsort(h)[::-1]
    h_bin[idx[:target]] = 1.0
    g = compute_g(h_bin, n)
    E = compute_E(h_bin, n)
    best_c = g.max()
    
    ones = np.where(h_bin == 1)[0].tolist()
    zeros = np.where(h_bin == 0)[0].tolist()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        # Try all (i,j) swaps: i in zeros, j in ones
        best_gain = 0
        best_i = -1
        best_j = -1
        for i in zeros:
            for j in ones:
                # h[i]: 0->1, h[j]: 1->0
                new_max = np.max(g + E[i] - E[j])
                gain = best_c - new_max
                if gain > best_gain:
                    best_gain = gain
                    best_i = i
                    best_j = j
        
        if best_i >= 0 and best_gain > 1e-12:
            h_bin[best_i] = 1.0
            h_bin[best_j] = 0.0
            g += E[best_i] - E[best_j]
            E = compute_E(h_bin, n)
            best_c = max(best_c, g.max())  # Actually best_c should be g.max()
            best_c = g.max()
            ones.remove(best_j)
            ones.append(best_i)
            zeros.remove(best_i)
            zeros.append(best_j)
            improved = True
        
        if not improved:
            break
    
    return h_bin, best_c

def slsqp_refine(h, n, deadline, alpha_list=None):
    if alpha_list is None:
        alpha_list = [10000, 50000, 200000, 1000000]
    
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha_list[0] * (g - m)  # Use first alpha
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha_list[0]
    
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for alpha in alpha_list:
        if time.time() > deadline:
            break
        def obj_local(x, a=alpha):
            g = np.correlate(x, 1 - x, mode='full') * dx
            m_val = g.max()
            sh = a * (g - m_val)
            sh_max = sh.max()
            return m_val + (sh_max + np.log(np.exp(sh - sh_max).sum())) / a
        
        def cons_fun(x):
            return x.sum() - n / 2.0
        
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': cons_fun}]
        try:
            res = minimize(obj_local, best_h.copy(), method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 500, 'ftol': 1e-14})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    
    return best_h, best_c
```
