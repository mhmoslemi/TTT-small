import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smoothmax_obj_and_grad(h, n, alpha, M):
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    g_max = g.max()
    g_shift = alpha * (g - g_max)
    w = np.exp(g_shift)
    w_sum = w.sum()
    obj = g_max + np.log(w_sum) / alpha
    # Penalty for constraint
    sum_dev = h.sum() - n / 2.0
    obj += M * sum_dev ** 2
    
    # Gradient of smoothmax
    grad = np.zeros(n)
    for m in range(2*n-1):
        d = m - (n-1)
        wm = w[m] / w_sum
        if wm < 1e-15:
            continue
        j_lo = max(0, d)
        j_hi = min(n, d + n)
        if j_lo < j_hi:
            grad[j_lo:j_hi] += wm * (1 - h[j_lo-d:j_hi-d])
        j_lo2 = max(0, -d)
        j_hi2 = min(n, n - d)
        if j_lo2 < j_hi2:
            grad[j_lo2:j_hi2] -= wm * h[j_lo2+d:j_hi2+d]
    
    # Gradient of penalty
    grad += 2 * M * sum_dev * np.ones(n)
    
    return obj, grad

def lbfgs_smoothmax(h, n, alpha, deadline, M=10000, maxiter=500):
    h = project_feasible(h.copy(), n)

        return smoothmax_obj_and_grad(x, n, alpha, M)[1]

    return project_feasible(h, n)

def _pair_effect(sens_i, sens_j, g, h, i, j, n):
    s = np.zeros(2 * n - 1)
    for m in range(2 * n - 1):
        dm = m - (n - 1)
        si = 0.0
        if 0 <= i - dm < n:
            si += (1.0 - h[i - dm])
        if 0 <= i + dm < n:
            si -= h[i + dm]
        sj = 0.0
        if 0 <= j - dm < n:
            sj += (1.0 - h[j - dm])
        if 0 <= j + dm < n:
            sj -= h[j + dm]
        s[m] = si - sj
    return s

def local_search(h, n, deadline, rng, max_iter=20000):
    h = h.copy()
    best_c = compute_c5(h, n)
    g = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        top_lag = np.argmax(g)
        d = top_lag - (n - 1)
        sens = np.zeros(n)
        lo1 = max(0, d)
        hi1 = min(n, d + n)
        if lo1 < hi1:
            sens[lo1:hi1] += (1.0 - h[lo1 - d:hi1 - d])
        lo2 = max(0, -d)
        hi2 = min(n, n - d)
        if lo2 < hi2:
            sens[lo2:hi2] -= h[lo2 + d:hi2 + d]
        
        if it % 5 == 0:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
        else:
            i = np.argmin(sens)
            j = np.argmax(sens)
            if sens[j] - sens[i] < 1e-10:
                stall += 1
                if stall > 200:
                    break
                continue
        
        stall = 0
        s = _pair_effect(sens[i], sens[j], g, h, i, j, n)
        d_lo = max(-h[i], h[j] - 1.0)
        d_hi = min(1.0 - h[i], h[j])
        if d_hi - d_lo < 1e-14:
            continue
        
        candidates = [d_lo, d_hi]
        top_k = np.argsort(g)[::-1][:12]
        for a in range(len(top_k)):
            for b in range(a + 1, len(top_k)):
                sa = s[top_k[a]]
                sb = s[top_k[b]]
                if abs(sa - sb) > 1e-14:
                    dc = (g[top_k[b]] - g[top_k[a]]) / (sa - sb)
                    if d_lo + 1e-12 < dc < d_hi - 1e-12:
                        candidates.append(dc)
        
        best_delta = 0.0
        best_val = g.max()
        for dc in candidates:
            dc = float(np.clip(dc, d_lo, d_hi))
            val = float(np.max(g + dc * s))
            if val < best_val - 1e-14:
                best_val = val
                best_delta = dc
        
        if best_delta != 0.0:
            h[i] += best_delta
            h[j] -= best_delta
            h = project_feasible(h, n)
            g = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
            c = float(g.max())
            if c < best_c:
                best_c = c
    
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 8
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)
    
    best_h = project_feasible(h0.copy(), n0)
    best_c = compute_c5(best_h, n0)
    
    # Phase 1: L-BFGS-B with increasing alpha (60% of budget)
    alphas = [50, 200, 1000, 5000, 20000, 80000, 300000, 1000000, 3000000, 10000000]
    p1_end = t0 + 0.55 * (budget_s - 10)
    for idx, alpha in enumerate(alphas):
        if time.time() > p1_end:
            break
        phase_dl = min(p1_end, dl - 60)
        maxit = max(30, 250 - idx * 18)
        M = 3000.0 + 500.0 * idx
        h = lbfgs_solve(best_h.copy(), n0, alpha, phase_dl, M=M, maxiter=maxit)
        c = compute_c5(h, n0)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    # Phase 2: Try different n values (20% of budget)
    p2_end = t0 + 0.75 * (budget_s - 10)
    for n_try in [200, 300, 350, 400, 500]:
        if time.time() > p2_end:
            break
        if time.time() > dl - 90:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_init = np.interp(x_new, x_old, best_h)
        h_init = project_feasible(h_init, n_try)
        
        for alpha in [5000, 50000, 500000]:
            if time.time() > dl - 90:
                break
            h = lbfgs_solve(h_init, n_try, alpha, dl - 90, M=5000.0, maxiter=80)
            c = compute_c5(h, n_try)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    
    # Phase 3: Local search (15% of budget)
    p3_end = min(t0 + 0.92 * (budget_s - 10), dl - 20)
    if time.time() < p3_end - 10:
        nf = len(best_h)
        h, c = local_search(best_h.copy(), nf, p3_end, rng, max_iter=50000)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    # Phase 4: Final L-BFGS-B polish
    if time.time() < dl - 10:
        nf = len(best_h)
        h = lbfgs_solve(best_h.copy(), nf, 10000000, dl - 5, M=20000.0, maxiter=100)
        c = compute_c5(h, nf)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
