import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def ecd_optimize(h, n, deadline, grid_size=101):
    """Exact coordinate descent with grid evaluation."""
    best_h = h.copy()
    best_c = float(np.max(np.correlate(h, 1-h, mode='full')))
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full')  # current correlation
    grid = np.linspace(0, 1, grid_size)
    
    for sweep in range(100000):
        if time.time() > deadline:
            break
        c_max_old = np.max(c)
        
        for i in range(n):
            old = h[i]
            # Compute B vector: B[k] for k=0..2n-2
            # B[k] = (1-h[i+d]) * [0<=i-d<n] - h[i+d] * [0<=i+d<n]
            # where d = k-(n-1)
            
            B = np.zeros(2*n-1)
            # Term 1: (1-h[i-d]) where d = k-(n-1), so i-d = i-k+(n-1)
            # Valid: 0 <= i-k+(n-1) < n → i-(n-1) <= k <= i
            k1_lo = max(0, i - (n-1))
            k1_hi = min(2*n-2, i)
            if k1_lo <= k1_hi:
                # i-d = i - (k-(n-1)) = i - k + n - 1
                idx = np.arange(k1_lo, k1_hi+1)
                j = i - idx + (n-1)  # these are the values of i-d
                B[k1_lo:k1_hi+1] += (1.0 - h[j])
            
            # Term 2: -h[i+d] where d = k-(n-1), so i+d = i+k-(n-1)
            # Valid: 0 <= i+k-(n-1) < n → (n-1-i) <= k <= (n-1-i)+(n-1)
            k2_lo = max(0, (n-1) - i)
            k2_hi = min(2*n-2, (n-1) - i + (n-1))
            if k2_lo <= k2_hi:
                idx = np.arange(k2_lo, k2_hi+1)
                j = i + idx - (n-1)  # these are the values of i+d
                B[k2_lo:k2_hi+1] -= h[j]
            
            # Evaluate f(t) = max_k (c[k] + (t-old)*B[k]) for t in grid
            # = max_k ((c[k] - old*B[k]) + t*B[k])
            m = c - old * B
            # Vectorized: for each t in grid, f(t) = max(m + B*t)
            F = m[None, :] + grid[:, None] * B[None, :]
            f_vals = F.max(axis=1)
            t_new = grid[np.argmin(f_vals)]
            
            if abs(t_new - old) > 1e-10:
                c += (t_new - old) * B
                h[i] = t_new
        
        # Project after sweep
        h = project_feasible(h, n)
        c = np.correlate(h, 1-h, mode='full')
        c_max = np.max(c)
        if c_max < best_c:
            best_c = c_max
            best_h = h.copy()
    
    return best_h, best_c * dx
