import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_fft(h, n):
    """Fast C5 computation using FFT."""
    dx = 2.0 / n
    H = np.fft.rfft(h, 2*n)
    OH = np.fft.rfft(1.0 - h, 2*n)
    corr = np.fft.irfft(H * np.conj(OH), 2*n)
    return float(np.max(corr)) * dx

def compute_corr_fft(h, n):
    """Return full correlation via FFT."""
    H = np.fft.rfft(h, 2*n)
    OH = np.fft.rfft(1.0 - h, 2*n)
    return np.fft.irfft(H * np.conj(OH), 2*n)

def project_feasible(h, n):
    """Project h onto {0≤h≤1, sum(h)=n/2}."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def pnorm_smooth_max(g, p):
    """Smooth max using p-norm."""
    # For numerical stability
    norm = np.sum(np.abs(g) ** p) ** (1.0/p)
    return norm

def compute_c5_fft(h, n):
    dx = 2.0 / n
    H = np.fft.rfft(h, 2*n)
    OH = np.fft.rfft(1.0 - h, 2*n)
    corr = np.fft.irfft(H * np.conj(OH), 2*n)
    return float(np.max(corr)) * dx

    H = np.fft.rfft(h, 2*n)
    OH = np.fft.rfft(1.0 - h, 2*n)
    return np.fft.irfft(H * np.conj(OH), 2*n)

def compute_c5_fft(h, n):
    dx = 2.0 / n
    corr = np.fft.irfft(np.fft.rfft(h, 2*n) * np.conj(np.fft.rfft(1.0-h, 2*n)), 2*n)
    return float(np.max(corr)) * dx

def compute_corr(h, n):
    return np.fft.irfft(np.fft.rfft(h, 2*n) * np.conj(np.fft.rfft(1.0-h, 2*n)), 2*n)

def project(h, n):
    t = n/2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        d = t - h.sum()
        if abs(d) < 1e-12: break
        if d > 0:
            r = 1.0 - h; s = r.sum()
            if s < 1e-14: break
            h += d/s * r
        else:
            s = h.sum()
            if s < 1e-14: break
            h += d/s * h
    return np.clip(h, 0.0, 1.0)

    bounds = [(0.0,1.0)]*n
