def lag_gradient(h, n, s):
    """Gradient of c[k] w.r.t. h, where s = k - (n-1)."""
    dx = 2.0 / n
    g = np.zeros(n)
    lo = max(0, s)
    hi = min(n - 1, s + n - 1)
    if lo > hi:
        return g
    # For m in [lo, hi]: g[m] = dx * (1 - 2 * h[s + n - 1 - m])
    idx = np.arange(lo, hi + 1)
    partners = s + n - 1 - idx
    g[idx] = dx * (1.0 - 2.0 * h[partners])
    return g

def subgradient_subgradient(h, n, alpha=100, n_active=5):
    """Compute weighted subgradient of max c[k]."""
    c = get_corr(h, n)
    c_max = c.max()
    # Weights for active lags
    weights = np.exp(alpha * (c - c_max))
    weights /= weights.sum()
    # Only use top n_active lags for efficiency
    top_idx = np.argsort(c)[-n_active:]
    g = np.zeros(n)
    for k in top_idx:
        s = k - (n - 1)
        g += weights[k] * lag_gradient(h, n, s)
    return g

import numpy as np

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

    return np.correlate(h, 1 - h, mode='full') * dx

def lag_gradient(h, n, s):
    dx = 2.0 / n
    g = np.zeros(n)
    lo = max(0, s)
    hi = min(n - 1, s + n - 1)
    if lo > hi:
        return g
    idx = np.arange(lo, hi + 1)
    partners = s + n - 1 - idx
    g[idx] = dx * (1.0 - 2.0 * h[partners])
    return g

    return float(np.sum(x) - n / 2.0)

def _sum_jac(x, n):
    return np.ones(n)

    dx = 2.0 / n
    hx = np.clip(x, 0, 1)
    c = np.correlate(hx, 1 - hx, mode='full') * dx
    m = c.max()
    return m + (1.0 / alpha) * np.log(np.sum(np.exp(alpha * (c - m))))

        res = minimize(_slsqp_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})

        return h.copy()

        if no_improve > 60000:
            break
    return best_h, best_c

def coordinate_polish(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    c = get_corr(best_h, n)
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    step_idx = 0
    stall = 0
    rng = np.random.RandomState(999)
    while time.time() < deadline:
        k_star = int(np.argmax(c))
        s_star = k_star - (n - 1)
        g = lag_gradient(h, n, s_star)
        if g.max() - g.min() < 1e-12:
            break
        i_src = int(np.argmax(g))
        i_dst = int(np.argmin(g))
        d = steps[step_idx]
        d = min(d, h[i_src] * 0.99, (1.0 - h[i_dst]) * 0.99)
        if d < 1e-10:
            step_idx = min(step_idx + 1, len(steps) - 1)
            stall += 1
            if stall > 400:
                i_r1 = rng.randint(n)
                i_r2 = rng.randint(n)
                if i_r1 == i_r2 or h[i_r1] < 1e-4 or h[i_r2] > 1 - 1e-4:
                    break
                dd = 0.001
                h[i_r1] -= dd
                h[i_r2] += dd
                nc = get_corr(h, n)
                if nc.max() < best_c:
                    best_c = float(nc.max())
                    best_h = h.copy()
                    c = nc
                    stall = 0
                    step_idx = 0
                else:
                    h[i_r1] += dd
                    h[i_r2] -= dd
                    if step_idx >= len(steps) - 1:
                        break
                continue
        h[i_src] -= d
        h[i_dst] += d
        new_c = get_corr(h, n)
        new_max = float(new_c.max())
        if new_max < best_c - 1e-15:
            best_c = new_max
            best_h = h.copy()
            c = new_c
            stall = 0
            step_idx = max(0, step_idx - 1)
        else:
            h[i_src] += d
            h[i_dst] -= d
            stall += 1
            if stall > 400:
                step_idx = min(step_idx + 1, len(steps) - 1)
                if step_idx >= len(steps) - 1:
                    break
                stall = 0
            else:
                step_idx = min(step_idx + 1, len(steps) - 1)
    return best_h, best_c

def multi_lag_polish(h, n, deadline, num_lags=4):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    step_idx = 0
    stall = 0
    while time.time() < deadline:
        c = get_corr(h, n)
        top_k = np.argsort(c)[-num_lags:]
        g = np.zeros(n)
        for k in top_k:
            s = int(k) - (n - 1)
            g += lag_gradient(h, n, s)
        g /= float(len(top_k))
        if g.max() - g.min() < 1e-13:
            break
        i_src = int(np.argmax(g))
        i_dst = int(np.argmin(g))
        d = steps[step_idx]
        d = min(d, h[i_src] * 0.99, (1.0 - h[i_dst]) * 0.99)
        if d < 1e-10:
            step_idx = min(step_idx + 1, len(steps) - 1)
            stall += 1
            if stall > 300:
                break
            continue
        h[i_src] -= d
        h[i_dst] += d
        nc = get_corr(h, n)
        if nc.max() < best_c - 1e-15:
            best_c = float(nc.max())
            best_h = h.copy()
            stall = 0
            step_idx = max(0, step_idx - 1)
        else:
            h[i_src] += d
            h[i_dst] -= d
            stall += 1
            if stall > 300:
                step_idx = min(step_idx + 1, len(steps) - 1)
                if step_idx >= len(steps) - 1:
                    break
                stall = 0
            else:
                step_idx = min(step_idx + 1, len(steps) - 1)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 250
        h_init = np.full(n, 0.5)
    h_init = project_feasible(h_init, n)
    best_h = h_init.copy()
    best_c = compute_c5(best_h, n)
    # Phase 1: SLSQP with log-sum-exp smoothing
    alphas = [500, 2000, 10000, 50000, 200000, 1000000, 5000000]
    for a in alphas:
        if time.time() > deadline * 0.22:
            break
        ht = slsqp_round(best_h, n, a, maxiter=1500)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c, best_h = c, ht
    # Phase 2: Subgradient descent (deterministic + randomized)
    sub_budget = (deadline - time.time()) * 0.30
    sub_end = time.time() + sub_budget
    configs = [(3, 5.0, False), (5, 3.0, False), (8, 2.0, True),
               (5, 1.5, True), (3, 0.8, False), (5, 4.0, True)]
    for n_top, eta0, rand in configs:
        if time.time() > sub_end:
            break
        alloc = min(45, sub_end - time.time())
        if alloc < 15:
            break
        ht, c = subgrad_descent(best_h, n, time.time() + alloc,
                                n_top=n_top, eta0=eta0, randomized=rand, seed=seed)
        if c < best_c:
            best_c, best_h = c, ht
    # Phase 3: Multi-lag polish
    if time.time() < deadline * 0.75:
        m_end = min(deadline * 0.75, time.time() + 120)
        ht, c = multi_lag_polish(best_h, n, m_end, num_lags=4)
        if c < best_c:
            best_c, best_h = c, ht
    # Phase 4: Coordinate polish
    if time.time() < deadline - 20:
        ht, c = coordinate_polish(best_h, n, deadline - 10)
        if c < best_c:
            best_c, best_h = c, ht
    # Phase 5: Final SLSQP + polish
    if time.time() < deadline - 12:
        ht = slsqp_round(best_h, n, 10000000, maxiter=1500)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c, best_h = c, ht
        if time.time() < deadline - 4:
            ht, c = coordinate_polish(best_h, n, deadline - 2)
            if c < best_c:
                best_c, best_h = c, ht
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
