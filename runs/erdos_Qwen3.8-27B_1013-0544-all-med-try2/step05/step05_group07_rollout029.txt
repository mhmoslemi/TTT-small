```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_coef(h, n, i, j):
    """Coefficient vector for pair swap: h[i]+=delta, h[j]-=delta"""
    coef = np.zeros(2 * n - 1)
    # Positive lags d>=0: coef[n-1+d], d=0..n-1
    p1 = np.zeros(n); p1[:n - i] = 1.0 - h[i:]
    p2 = np.zeros(n); p2[:i + 1] = -h[i::-1]
    p3 = np.zeros(n); p3[:n - j] = -(1.0 - h[j:])
    p4 = np.zeros(n); p4[:j + 1] = h[j::-1]
    pos_part = p1 + p2 + p3 + p4
    coef[n - 1:] = pos_part
    # Negative lags d=-d', d'=0..n-1: coef[n-1-d']
    n1 = np.zeros(n); n1[:i + 1] = 1.0 - h[i::-1]
    n2 = np.zeros(n); n2[:n - i] = -h[i:]
    n3 = np.zeros(n); n3[:j + 1] = -(1.0 - h[j::-1])
    n4 = np.zeros(n); n4[:n - j] = h[j:]
    neg_part = n1 + n2 + n3 + n4
    coef[:n] = neg_part[::-1]
    return coef

def pair_swap_optimize(h, n, deadline, rng):
    h = project_feasible(h.copy(), n)
    c = compute_c(h, n).copy()
    best_h = h.copy()
    best_c = c.max()
    fail = 0
    step = 0
    recompute_interval = 500

    while time.time() < deadline:
        step += 1
        # Periodic recompute to avoid drift
        if step % recompute_interval == 0:
            c = compute_c(h, n).copy()

        c_max_val = c.max()
        active_idx = int(np.argmax(c))
        k = active_idx - (n - 1)

        # Compute A_k[i] and B_k[j]
        A = np.zeros(n)
        B = np.zeros(n)
        if k >= 0:
            A[:n - k] = 1.0 - h[k:n]
            A[k:] -= h[:n - k]
            B[:n - k] = -(1.0 - h[k:n])
            B[k:] += h[:n - k]
        else:
            kp = -k
            A[kp:] = 1.0 - h[:n - kp]
            A[:n - kp] -= h[kp:n]
            B[kp:] = -(1.0 - h[:n - kp])
            B[:n - kp] += h[kp:n]

        # Mask infeasible
        A_m = A.copy()
        B_m = B.copy()
        A_m[h >= 1.0 - 1e-10] = 1e9
        B_m[h <= 1e-10] = 1e9

        # Find best pair (i,j) minimizing A[i]+B[j] with i!=j
        i_order = np.argsort(A_m)
        j_order = np.argsort(B_m)
        best_val = 1e18
        best_i = -1
        best_j = -1

        for ii in range(min(20, n)):
            i = int(i_order[ii])
            if A_m[i] > 1e8:
                break
            for jj in range(min(20, n)):
                j = int(j_order[jj])
                if i == j or B_m[j] > 1e8:
                    continue
                val = A_m[i] + B_m[j]
                if val < best_val:
                    best_val = val
                    best_i = i
                    best_j = j

        if best_i < 0 or best_val >= 0:
            fail += 1
            if fail > 200:
                h = project_feasible(best_h + 0.05 * rng.randn(n), n)
                c = compute_c(h, n).copy()
                fail = 0
            else:
                h = project_feasible(h + 0.003 * rng.randn(n), n)
                c = compute_c(h, n).copy()
            continue

        fail = 0
        coef = compute_coef(h, n, best_i, best_j)

        # Find safe delta
        delta_bound = min(1.0 - h[best_i], h[best_j])
        pos_mask = coef > 1e-12
        if pos_mask.any():
            ratios = (c_max_val - c[pos_mask]) / coef[pos_mask]
            valid = ratios > 0
            if valid.any():
                delta_bound = min(delta_bound, float(ratios[valid].min()))

        if coef[active_idx] >= -1e-12:
            h = project_feasible(h + 0.003 * rng.randn(n), n)
            c = compute_c(h, n).copy()
            continue

        delta = 0.95 * max(delta_bound, 1e-12)
        h[best_i] += delta
        h[best_j] -= delta
        h = np.clip(h, 0.0, 1.0)
        c += delta * coef

        new_max = c.max()
        if new_max < best_c:
            best_c = new_max
            best_h = h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c(best_h, n).max()
    return best_h, best_c

def softmax_polish(h, n, alpha, deadline):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        val += 1e6 * (x.sum() - n / 2.0) ** 2
        return val

    def grad(x):
        c = np.correlate(x, 1 - x, mode='full')
        m_star = int(np.argmax(c))
        kk = m_star - (n - 1)
        g = np.zeros(n)
        if kk >= 0:
            g[kk:] += (1.0 - x[:n - kk])
            g[:n - kk] -= x[kk:]
        else:
            g[:n + kk] += (1.0 - x[-kk:])
            g[-kk:] -= x[:n + kk]
        g *= dx
        g += 2e6 * (x.sum() - n / 2.0)
        return g

    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=grad,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': 500, 'ftol': 1e-16, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h
```
