import numpy as np
from scipy.optimize import linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def lp_linearize(h, n, max_iter=200, deadline=None):
    dx = 2.0 / n
    n_half = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    
    for it in range(max_iter):
        if deadline and time.time() > deadline:
            break
        
        # Compute overlap vector
        overlap = np.correlate(h, 1 - h, mode='full') * dx
        cur_c = overlap.max()
        
        # Compute gradient of each overlap_k with respect to h
        # d(overlap_k)/dh[j] = -dx * (h[j+k] + h[j-k]) for the auto part
        # But actually: overlap_k = (n/2 - auto_k) * dx
        # d(overlap_k)/dh[j] = -dx * (h[j+k] + h[j-k])
        # where h[x]=0 if x out of [0,n)
        
        # For the LP, we need constraint: overlap_k(new_h) ≤ t
        # Linearization: overlap_k(h) - dx * sum_j (h[j+k]+h[j-k]) * (new_h[j] - h[j]) ≤ t
        # => dx * sum_j g_k[j] * new_h[j] + t ≥ overlap_k(h) + dx * sum_j g_k[j] * h[j]
        # where g_k[j] = h[j+k] + h[j-k]
        # => -dx * sum_j g_k[j] * new_h[j] - t ≤ -(overlap_k(h) + dx * sum_j g_k[j] * h[j])
        
        # Build the constraint matrix
        # Number of shifts: 2n-1
        n_shifts = 2 * n - 1
        
        # For efficiency, compute g_k for all k at once
        # g_k[j] = h[j+k] + h[j-k]
        # This is: circular-like shifts of h
        
        # Let me compute the RHS for each shift k:
        # rhs_k = overlap_k(h) + dx * sum_j g_k[j] * h[j]
        # = overlap_k(h) + dx * 2 * auto_k  (as shown earlier)
        # = (n/2 - auto_k)*dx + 2*dx*auto_k
        # = n*dx/2 + dx*auto_k
        
        auto_all = np.correlate(h, h, mode='full')  # length 2n-1
        rhs = n_half * dx + dx * auto_all  # = (n/2 + auto_k) * dx
        
        # Build A_ub: for each shift k, the row is [-dx*g_k[0], ..., -dx*g_k[n-1], -1]
        # g_k[j] = h[j+k] + h[j-k] (with 0 for out of bounds)
        
        # Efficient way to build G matrix:
        # For shift s (s from -(n-1) to n-1), G[s+n-1, j] = h[j+s] + h[j-s]
        # h[j+s] is a left shift of h by s (with zero padding)
        # h[j-s] is a right shift of h by s (with zero padding)
        
        G = np.zeros((n_shifts, n))
        for idx, s in enumerate(range(-(n-1), n)):
            if s >= 0:
                G[idx, :n-s] += h[s:]      # h[j+s] for j in [0, n-s-1]
                G[idx, s:] += h[:n-s]      # h[j-s] for j in [s, n-1]
            else:
                G[idx, -s:] += h[:n+s]     # h[j+s] for j in [-s, n-1]
                G[idx, :n+s] += h[-s:]     # h[j-s] for j in [0, n+s-1]
        
        # A_ub row for shift k: [-dx * G[k,0], ..., -dx * G[k,n-1], -1]
        A_ub = np.zeros((n_shifts, n + 1))
        A_ub[:, :n] = -dx * G
        A_ub[:, n] = -1.0
        b_ub = -rhs
        
        # A_eq: sum(new_h) = n/2
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n_half])
        
        # Objective: minimize -t (i.e., maximize t)
        c_obj = np.zeros(n + 1)
        c_obj[n] = -1.0
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=[(0, 1)] * n + [(-np.inf, np.inf)],
                      method='highs')
        
        if not res.success:
            break
        
        new_h = res.x[:n]
        new_c = compute_c5(new_h, n)
        
        if new_c <= cur_c + 1e-12:
            h = new_h
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            # Linearization overestimates; try a convex combination
            for alpha in [0.5, 0.25, 0.75]:
                combo = (1 - alpha) * best_h + alpha * new_h
                combo = project_feasible(combo, n)
                combo_c = compute_c5(combo, n)
                if combo_c < best_c:
                    best_c = combo_c
                    best_h = combo.copy()
                    h = combo.copy()
                    cur_c = combo_c
                    break
            else:
                break
    
    return best_h, best_c

def efficient_binary_search(h_init, n, deadline, max_sweeps=1000):
    """Binary local search focusing on active shifts."""
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    m_range = np.arange(-(n - 1), n)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        # Find active shifts (top few)
        corr_vals = corr * dx
        top_k = np.argsort(-corr_vals)[:5]  # top 5 shifts
        improved = False
        
        z_list = sorted(zeros)
        for i in z_list:
            if time.time() > deadline:
                break
            o_list = sorted(ones)
            for j in o_list:
                # Compute delta
                im = i + m_range
                imn = i - m_range
                jm = j + m_range
                jmn = j - m_range
                delta = np.zeros(2 * n - 1)
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                # Cross terms
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < 2*n-1:
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2*n-1:
                    delta[k2] += 1.0
                
                new_corr = corr + delta
                nc = new_corr.max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr = new_corr
                    cur_c = nc
                    ones.discard(j)
                    ones.add(i)
                    zeros.discard(i)
                    zeros.add(j)
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            break
        
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    
    return best_h, best_c

import numpy as np
from scipy.optimize import linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def _build_g_matrix(h, n):
    n_shifts = 2 * n - 1
    G = np.zeros((n_shifts, n))
    for idx, s in enumerate(range(-(n - 1), n)):
        if s >= 0:
            G[idx, :n - s] += h[s:]
            G[idx, s:] += h[:n - s]
        else:
            G[idx, -s:] += h[:n + s]
            G[idx, :n + s] += h[-s:]
    return G

def lp_linearize(h, n, max_iter=500, deadline=None):
    dx = 2.0 / n
    n_half = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    stall = 0
    for it in range(max_iter):
        if deadline and time.time() > deadline:
            break
        auto_all = np.correlate(h, h, mode='full')
        overlap = (n_half - auto_all) * dx
        cur_c = overlap.max()
        G = _build_g_matrix(h, n)
        rhs = (n_half + auto_all) * dx
        n_shifts = 2 * n - 1
        A_ub = np.zeros((n_shifts, n + 1))
        A_ub[:, :n] = -dx * G
        A_ub[:, n] = -1.0
        b_ub = -rhs
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n_half])
        c_obj = np.zeros(n + 1)
        c_obj[n] = -1.0
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=[(0.0, 1.0)] * n + [(-np.inf, np.inf)],
                      method='highs')
        if not res.success:
            break
        new_h = np.clip(res.x[:n], 0.0, 1.0)
        new_h = project_feasible(new_h, n)
        new_c = compute_c5(new_h, n)
        if new_c <= cur_c + 1e-10:
            h = new_h
            if new_c < best_c - 1e-9:
                best_c = new_c
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            improved = False
            for alpha in [0.5, 0.25, 0.75, 0.1]:
                combo = (1 - alpha) * best_h + alpha * new_h
                combo = project_feasible(combo, n)
                combo_c = compute_c5(combo, n)
                if combo_c < best_c - 1e-9:
                    best_c = combo_c
                    best_h = combo.copy()
                    h = combo.copy()
                    stall = 0
                    improved = True
                    break
            if not improved:
                stall += 1
        if stall > 15:
            break
    return best_h, best_c

def binary_local_search(h_init, n, deadline, max_sweeps=2000, n_restarts=3, seed=42):
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    best_c = np.inf
    best_h = h.copy()

    rng = np.random.RandomState(seed)
    current_h = h.copy()
    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        if restart > 0:
            for _ in range(rng.randint(5, 15)):
                oi = np.where(current_h == 1)[0]
                zi = np.where(current_h == 0)[0]
                if len(oi) > 0 and len(zi) > 0:
                    a = rng.choice(oi)
                    b = rng.choice(zi)
                    current_h[a], current_h[b] = 0.0, 1.0
        h = current_h.copy()
        corr = np.correlate(h, 1 - h, mode='full').copy()
        cur_c = corr.max() * dx
        ones = set(np.where(h == 1)[0].tolist())
        zeros = set(np.where(h == 0)[0].tolist())
        for sweep in range(max_sweeps):
            if time.time() > deadline:
                break
            improved = False
            z_list = sorted(zeros)
            for i in z_list:
                if time.time() > deadline:
                    break
                o_list = sorted(ones)
                for j in o_list:
                    im = i + m_range
                    imn = i - m_range
                    jm = j + m_range
                    jmn = j - m_range
                    delta = np.zeros(2 * n - 1)
                    msk = (im >= 0) & (im < n)
                    delta[msk] += 1.0 - h[im[msk]]
                    msk = (imn >= 0) & (imn < n)
                    delta[msk] -= h[imn[msk]]
                    msk = (jm >= 0) & (jm < n)
                    delta[msk] -= 1.0 - h[jm[msk]]
                    msk = (jmn >= 0) & (jmn < n)
                    delta[msk] += h[jmn[msk]]
                    delta[n - 1] -= 2.0
                    k1 = n - 1 + (j - i)
                    if 0 <= k1 < 2 * n - 1:
                        delta[k1] += 1.0
                    k2 = n - 1 + (i - j)
                    if 0 <= k2 < 2 * n - 1:
                        delta[k2] += 1.0
                    new_corr = corr + delta
                    nc = new_corr.max() * dx
                    if nc < cur_c - 1e-15:
                        h[i], h[j] = 1.0, 0.0
                        corr = new_corr
                        cur_c = nc
                        ones.discard(j)
                        ones.add(i)
                        zeros.discard(i)
                        zeros.add(j)
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
            current_h = h.copy()
    return best_h, best_c
