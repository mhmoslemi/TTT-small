from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c)) * dx

    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def eval_swap(hb, c_full, n, i, j):
    """Evaluate C5 after swapping hb[i]:0->1, hb[j]:1->0.
    Returns (new_C5_value_before_dx, affected_k_lo, affected_k_hi)."""
    # Affected range
    k_lo = max(0, n-1-i, i-n+1, n-1-j, j-n+1)
    k_hi = min(2*n-2, n-1+i, i+n-1, n-1+j, j+n-1)

        if 0 <= q < n:

        q2 = i - k + n - 1
        if 0 <= q2 < n:

        if 0 <= q3 < n:
            dk[k-k_lo] -= (1 - hb[q3])
        q4 = j - k + n - 1
        if 0 <= q4 < n:
            dk[k-k_lo] += hb[q4]
    
    new_max_affected = dk.max()
    # Unaffected max
    if k_lo > 0:
        new_max_unaffected = c_full[:k_lo].max()
    elif k_hi < 2*n-2:
        new_max_unaffected = c_full[k_hi+1:].max()
    else:
        new_max_unaffected = 0

                hb[best_j] = 0.0

                best_c_raw = c_full.max()
                best_c = best_c_raw * (2.0/n)
                improved = True
                break
        
        if not improved:
            break
    
    return hb, best_c
