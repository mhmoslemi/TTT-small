            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 5000:
            break
        
        # Compute weighted gradient
        sorted_idx = np.argsort(c)[::-1][:top_k]
        c_vals = c[sorted_idx]
        w = np.exp((c_vals - c_vals.max()) / 0.3)
        w /= w.sum()
        g = np.zeros(n)
        for i, k in enumerate(sorted_idx):
            g += w[i] * lag_gradient(h, n, int(k))
        
        # Best source and destination
        i_src = int(np.argmax(g))
        i_dst = int(np.argmin(g))
        if i_src == i_dst:
            break
        
        # Line search: move from i_src to i_dst
        # h[i_src] decreases, h[i_dst] increases
        max_move = min(h[i_src], 1 - h[i_dst])
        if max_move < 1e-12:
            break
        
        # Binary search for the step that minimizes the new C5
        lo, hi = 0.0, max_move * 0.99
        best_step = 0.0
        best_new_c = curr
        for _ in range(15):
            mid = (lo + hi) / 2
            if mid < 1e-14:
                break
            h[i_src] -= mid
            h[i_dst] += mid
            new_c = compute_c5(h, n)
            if new_c < best_new_c:
                best_new_c = new_c
                best_step = mid
                lo = mid
            else:
                h[i_src] += mid
                h[i_dst] -= mid
                hi = mid
        
        if best_step > 0:
            h[i_src] -= best_step
            h[i_dst] += best_step
        else:
            break
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    h = h.copy()
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(3000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter=3000):
    dx = 2.0 / n
    def obj(x):
        hx = np.clip(x, 0, 1)
        c = np.correlate(hx, 1 - hx, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def lag_gradient(h, n, k):
    s = k - (n - 1)
    g = np.zeros(n)
    lo = max(0, -s)
    hi = min(n - 1, n - 1 - s)
    idx = np.arange(lo, hi + 1)
    g[idx] += (1.0 - h[idx + s])
    lo2 = max(0, lo + s)
    hi2 = min(n - 1, hi + s)
    idx2 = np.arange(lo2, hi2 + 1)
    g[idx2] -= h[idx2 - s]
    return g

def pair_polish(h, n, deadline, top_k=5, max_pairs=8):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    h = h.copy()
    stall = 0
    rng = np.random.RandomState(98765)

    while time.time() < deadline:
        c = get_corr(h, n)
        curr = float(c.max())
        if curr < best_c - 1e-15:
            best_c = curr
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 8000:
            break

        # Weighted multi-lag gradient
        sorted_idx = np.argsort(c)[::-1][:top_k]
        c_vals = c[sorted_idx]
        w = np.exp((c_vals - c_vals.max()) / 0.25)
        w /= w.sum()
        g = np.zeros(n)
        for i, k in enumerate(sorted_idx):
            g += w[i] * lag_gradient(h, n, int(k))

        # Try top candidate pairs
        improved = False
        # Rank sources and destinations
        src_rank = np.argsort(g)[::-1]
        dst_rank = np.argsort(g)

        for attempt in range(max_pairs):
            if attempt < len(src_rank):
                i_src = int(src_rank[attempt])
            else:
                i_src = int(rng.randint(n))
            if attempt < len(dst_rank):
                i_dst = int(dst_rank[attempt])
            else:
                i_dst = int(rng.randint(n))
            if i_src == i_dst:
                continue

            max_move = min(h[i_src], 1.0 - h[i_dst])
            if max_move < 1e-12:
                continue

            # Binary search for best step
            lo_s, hi_s = 0.0, max_move * 0.999
            best_step = 0.0
            best_new_c = curr
            for _ in range(14):
                mid = (lo_s + hi_s) / 2.0
                if mid < 1e-14:
                    break
                h[i_src] -= mid
                h[i_dst] += mid
                new_c = compute_c5(h, n)
                if new_c < best_new_c - 1e-16:
                    best_new_c = new_c
                    best_step = mid
                    lo_s = mid
                else:
                    h[i_src] += mid
                    h[i_dst] -= mid
                    hi_s = mid

            if best_step > 1e-12:
                h[i_src] -= best_step
                h[i_dst] += best_step
                improved = True
                break
            else:
                # Restore
                h[i_src] += 0
                h[i_dst] -= 0

        if not improved:
            # Escape: small random perturbation
            if stall > 2000:
                idx1 = int(rng.randint(n))
                idx2 = int(rng.randint(n))
                if idx1 != idx2 and h[idx1] > 1e-6 and h[idx2] < 1 - 1e-6:
                    move = rng.uniform(0.001, 0.005)
                    move = min(move, h[idx1], 1.0 - h[idx2])
                    h[idx1] -= move
                    h[idx2] += move
                    h = project_feasible(h, n)
                    stall = 0
                    top_k = min(n, top_k + 2)
            else:
                stall += 100
                if stall > 12000:
                    break
        else:
            top_k = max(1, top_k - 1) if best_c < compute_c5(best_h, n) else top_k

    return best_h, best_c

def slipsq_phase(h, n, deadline):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    alphas = [300, 1500, 8000, 40000, 200000, 800000, 3000000]
    for a in alphas:
        if time.time() > deadline - 5:
            break
        maxit = max(500, min(5000, int(150000 / n)))
        ht = slsqp_opt(h, n, a, maxiter=maxit)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht
            h = ht
        else:
            h = ht
    return best_h, best_c

    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

            best_h = h.copy()

    # Phase 1: Optimize from parent (40% budget)
    if h_parent is not None:
        t1 = t0 + (deadline - t0) * 0.38

    if best_h is not None:
        x_best = np.linspace(0, 2, len(best_h))

            remaining = deadline - time.time()

            best_h = project_feasible(best_h, n)

        best_h = project_feasible(best_h, n)
