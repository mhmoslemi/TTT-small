import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            r = 1.0 - h
            t = r.sum()
            if t < 1e-15:
                break
            h = np.clip(h + (d / t) * r, 0.0, 1.0)
        else:
            t = h.sum()
            if t < 1e-15:
                break
            h = np.clip(h + (d / t) * h, 0.0, 1.0)
    return h

def logsumexp_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    smax = shifted.max()
    if smax > 500:
        return float(m + (smax + np.log(np.sum(np.exp(shifted - smax)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    return float(np.sum(x) - n / 2.0)

    return np.ones(n)

                        continue
                    # Try making h[i] and h[j] more similar
                    if h[i] > h[j]:
                        delta = d
                    else:
                        delta = -d
                    hi = h[i] + delta
                    hj = h[j] - delta
                    if 0 <= hi <= 1 and 0 <= hj <= 1:

                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
