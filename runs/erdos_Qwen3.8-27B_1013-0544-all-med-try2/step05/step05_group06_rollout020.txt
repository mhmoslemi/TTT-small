    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:

            h = h + diff / s * h

    dm = np.arange(2 * n - 1) - (n - 1)
    F = np.zeros((n, 2 * n - 1))
    for m_idx in range(2 * n - 1):
        d = dm[m_idx]
        j_lo = max(0, d)
        j_hi = min(n, n + d)
        if j_lo < j_hi:
            F[j_lo:j_hi, m_idx] += (1.0 - h[j_lo - d:j_hi - d])
        j_lo2 = max(0, -d)
        j_hi2 = min(n, n - d)
        if j_lo2 < j_hi2:
            F[j_lo2:j_hi2, m_idx] -= h[j_lo2 + d:j_hi2 + d]
    return F

def update_F(F, dm, idx, delta, n):
    nm = 2 * n - 1
    m_idx = np.arange(nm)
    v1 = (idx + dm >= 0) & (idx + dm < n)
    F[idx + dm[v1], m_idx[v1]] -= delta
    v2 = (idx - dm >= 0) & (idx - dm < n)
    F[idx - dm[v2], m_idx[v2]] -= delta

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1 - x, mode='full') * dx
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    return m - (sm - np.log(np.exp(sh - sm).sum())) / alpha

def smooth_max_grad(x, n, alpha):
    g = compute_g(x, n)
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    w = np.exp(sh - sm)
    w /= w.sum()
    F = compute_F_vec(x, n)
    return F @ w

def slsqp_phase(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    alphas = [200, 500, 1000, 5000, 20000]

        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(smooth_max_obj, h, jac=lambda x: smooth_max_grad(x, n, alpha),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 150, 'ftol': 1e-10})
            h_new = project_feasible(res.x, n)
            c = compute_c5(h_new, n)
            if c < best_c:
                best_c = c
                best_h = h_new.copy()
                h = h_new.copy()

        else:
            stall = 0
    return best_h, best_c

def make_block_pattern(n, block_size):
    h = np.zeros(n)
    for start in range(0, n, 2 * block_size):
        h[start:start + block_size] = 1.0
    return project_feasible(h, n)

def make_periodic(n, pattern):
    p = np.array(pattern, dtype=float)
    h = np.tile(p, n // len(p) + 1)[:n]
    return project_feasible(h, n)

def perturb_restart(h, n, rng, scale=0.05):
    h_new = h + rng.uniform(-scale, scale, n)
    return project_feasible(h_new, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP with smooth max (12% budget)
    p1 = t0 + 0.12 * (budget_s - 5)
    h = best_h.copy()
    h, c = slsqp_phase(h, n, p1)
    if c < best_c:
        best_c = c
        best_h = h.copy()

    # Phase 2: Main pair descent (55% budget)
    p2 = t0 + 0.67 * (budget_s - 5)
    h, c = pair_descent(best_h.copy(), n, p2, max_sweeps=800, K=8)
    if c < best_c:
        best_c = c
        best_h = h.copy()

    # Phase 3: Perturb and restart (15% budget)
    p3 = t0 + 0.82 * (budget_s - 5)
    for attempt in range(5):
        if time.time() > p3:
            break
        h_try = perturb_restart(best_h, n, rng, scale=0.03 * (attempt + 1))
        h_try, c_try = pair_descent(h_try, n, p3, max_sweeps=100, K=6)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try.copy()

    # Phase 4: Try alternative starting points (8% budget)
    p4 = t0 + 0.90 * (budget_s - 5)
    patterns = [
        make_block_pattern(n, n // 6),
        make_block_pattern(n, n // 5),
        make_periodic(n, [1, 1, 1, 0, 0, 0]),
        make_periodic(n, [1, 1, 0, 0, 1, 0]),
        make_periodic(n, [1, 0, 1, 1, 0, 0]),
    ]
    for pat in patterns:
        if time.time() > p4:
            break
        h_try, c_try = pair_descent(pat, n, p4, max_sweeps=80, K=6)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try.copy()

    # Phase 5: Final pair descent polish (10% budget)
    if time.time() < dl - 3:
        h, c = pair_descent(best_h.copy(), n, dl - 2, max_sweeps=300, K=10)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 6: Quick SLSQP polish if time remains
    if time.time() < dl - 30:
        h, c = slsqp_phase(best_h.copy(), n, dl - 25)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if time.time() < dl - 3:
            h, c = pair_descent(best_h.copy(), n, dl - 2, max_sweeps=100, K=8)
            if c < best_c:
                best_c = c
                best_h = h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
