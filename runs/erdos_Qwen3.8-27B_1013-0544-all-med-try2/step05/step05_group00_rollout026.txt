def compute_c5(h, n):

    c = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

    dx = 2.0 / n
    best_count = np.inf
    best_h = h_init.copy()
    j_range = np.arange(n)
    rng = np.random.RandomState(seed)

            break

        else:
            cur = best_h.copy()

                    b = rng.choice(zi)
                    cur[a], cur[b] = 0.0, 1.0

        cur_max = corr.max()

            if time.time() > deadline:
                break
            # Find critical lags
            critical = np.where(corr >= cur_max - 2.0)[0]
            
            # For each zero position i, find valid one positions j
            ones_mask = (cur == 1)
            zeros_mask = (cur == 0)
            improved = False
            
            # Try each zero position
            for i in np.where(zeros_mask)[0]:
                if time.time() > deadline:
                    break
                if not zeros_mask[i]:
                    continue  # might have been changed

                valid_j = ones_mask.copy()
                feasible = True
                for k_idx in critical:

                    # A_k(i) = -h[i+lag] - h[i-lag]

                    if 0 <= i - lag < n:
                        a_val -= cur[i - lag]
                    T_k = cur_max - corr[k_idx] - a_val  # = cur_max - corr[k_idx] + h[i+lag] + h[i-lag]

                    idx2 = j_range - lag

                    B[v1] += cur[idx1[v1]]
                    B[v2] += cur[idx2[v2]]
                    
                    # Need B < T_k (strict)

                    # if j-lag==i, add 1 to delta (so B effectively decreases by 1)

                    if new_max < cur_max:

                    else:

                        cur[i] = 0.0
                        cur[j] = 1.0
            
            if not improved:

    return best_h, float(best_count * dx)

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def slsqp_active(h, n, maxiter, deadline):
    """SLSQP using only the top few critical lags for a tighter objective."""
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        # Use top-3 lags for a piecewise-linear-ish objective
        top3 = np.argsort(g)[-3:]
        m = g.max()
        # Smooth max over top lags with moderate alpha
        alpha = 500.0
        sh = alpha * (g[top3] - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 8
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    best_h = project_feasible(h0, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: aggressive binary search (70% of budget)
    p1 = t0 + 0.70 * (budget_s - 8)
    hb, cb = binary_local_search(best_h, n, p1, max_sweeps=100000, n_restarts=10, seed=seed)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    # Phase 2: SLSQP continuous polish from binary solution
    p2 = t0 + 0.85 * (budget_s - 8)
    if time.time() < p2:
        for alpha in [2000, 10000, 50000, 200000, 1000000]:
            if time.time() > p2:
                break
            h = best_h.copy()
            h = slsqp_smooth(h, n, alpha, 3000, p2)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        if time.time() < p2:
            h = slsqp_active(best_h.copy(), n, 3000, p2)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    # Phase 3: binary search from continuous solution (re-binarize + refine)
    p3 = dl - 5
    if time.time() < p3:
        # Re-binarize: take top n/2 values
        sorted_h = np.sort(best_h)[::-1]
        thresh = sorted_h[n // 2]
        h_bin = (best_h >= thresh).astype(float)
        # Ensure exactly n/2 ones
        n_ones = int(h_bin.sum())
        if n_ones > n // 2:
            excess = np.where(h_bin == 1)[0]
            for idx in excess[n_ones - n // 2:]:
                h_bin[idx] = 0.0
        elif n_ones < n // 2:
            deficit = np.where(h_bin == 0)[0]
            for idx in deficit[:n // 2 - n_ones]:
                h_bin[idx] = 1.0
        hb2, cb2 = binary_local_search(h_bin, n, p3, max_sweeps=20000, n_restarts=3, seed=seed + 100)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2.copy()
    # Final: one more SLSQP polish if time
    if time.time() < dl - 3:
        h = slsqp_smooth(best_h.copy(), n, 5000000, 2000, dl - 2)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
