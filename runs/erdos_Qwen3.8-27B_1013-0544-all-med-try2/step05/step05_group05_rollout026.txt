    # We want c[i] for i = 0,...,n-1
    # c[i] = sum_{k=max(0,i-(n-1))}^{min(i, n-1)} a[k]*b[i-k]
    # = sum_{k=0}^{i} a[k]*b[i-k] (for i ≤ n-1, and b is length n so b[i-k] is valid for k≥i-(n-1)≥0 when i≤n-1)
    
    # So c[i] = sum_{k=0}^{i} a[k]*b[i-k] for i < n
    # This is exactly the first n elements of the full convolution!
    
    a = 1.0 - h
    b = w  # length n
    c = np.fft.ifft(np.fft.fft(a, 2*n) * np.fft.fft(b, 2*n)).real[:n]
    # c[i] = sum_{k=0}^{i} a[k]*b[i-k] = sum_{k=0}^{i} (1-h[k])*w[i-k]
    # But we want f[i] = sum_{k=0}^{i} w[k]*(1-h[i-k])
    # Let j = i-k: f[i] = sum_{j=0}^{i} w[i-j]*(1-h[j])
    # = sum_{j=0}^{i} (1-h[j])*w[i-j] = c[i] ✓
    
    # Second term: s[i] = sum_{k=0}^{n-1-i} w[k]*h[i+k]
    # Let j = i+k: s[i] = sum_{j=i}^{n-1} w[j-i]*h[j]
    # This is a "reverse convolution"
    # s[i] = sum_{j=i}^{n-1} h[j]*w[j-i]
    # Let's define h_rev[j] = h[n-1-j], w_rev[j] = w[n-1-j]
    # Then s[i] = sum_{j=i}^{n-1} h[j]*w[j-i]
    # = sum_{k=0}^{n-1-i} h[i+k]*w[k]
    # This is the (n-1-i)-th element of the full convolution of h_rev and w:
    # (h_rev * w)[m] = sum_{k} h_rev[k]*w[m-k]
    # (h_rev * w)[n-1-i] = sum_{k} h[n-1-k]*w[n-1-i-k]
    # Let j = n-1-k: = sum_{j} h[j]*w[n-1-i-(n-1-j)] = sum_{j} h[j]*w[j-i]
    # = sum_{j: 0≤j≤n-1, 0≤j-i≤n-1} h[j]*w[j-i] = sum_{j=i}^{n-1} h[j]*w[j-i] ✓
    
    h_rev = h[::-1]
    s_full = np.fft.ifft(np.fft.fft(h_rev, 2*n) * np.fft.fft(w, 2*n)).real
    s = s_full[n-1::-1]  # s[i] = s_full[n-1-i]
    
    grad = c - s
    return grad

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

            if s < 1e-14:
