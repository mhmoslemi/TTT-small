def project_feasible(h, n):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c)), c

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_step(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.x is not None and res.x is not h:
            return project_feasible(res.x, n)
    except:
        pass
    return h.copy()

def focused_search(h, n, deadline):
    best_c = compute_c5(h, n)[0]
    best_h = h.copy()
    for delta in [0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002]:
        if time.time() > deadline:
            break
        for _round in range(500):
            if time.time() > deadline:
                break
            c_val, c_arr = compute_c5(h, n)
            top_idx = np.argsort(c_arr)[::-1][:3]
            improved = False
            for idx in top_idx:
                lag = abs(idx - (n - 1))
                if lag == 0:
                    mids = np.argsort(np.abs(h - 0.5))
                    lows = mids[:5]
                    highs = mids[-5:]
                    for i in lows:
                        if time.time() > deadline:
                            return best_h, best_c
                        for j in highs:
                            if i == j:
                                continue
                            if h[i] < 0.5:
                                h[i] -= delta; h[j] += delta
                            else:
                                h[i] += delta; h[j] -= delta
                            if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                                c_new = compute_c5(h, n)[0]
                                if c_new < best_c - 1e-14:
                                    best_c = c_new; best_h = h.copy(); improved = True
                                if h[i] < 0.5:
                                    h[i] += delta; h[j] -= delta
                                else:
                                    h[i] -= delta; h[j] += delta
                            else:
                                if h[i] < 0.5:
                                    h[i] += delta; h[j] -= delta
                                else:
                                    h[i] -= delta; h[j] += delta
                elif lag < n:
                    for i in range(n - lag):
                        if time.time() > deadline:
                            return best_h, best_c
                        j = i + lag
                        contrib = h[i] * (1.0 - h[j])
                        if contrib < 0.003:
                            continue
                        hi = h[i] - delta; hj = h[j] + delta
                        if 0 <= hi and hj <= 1:

                            c_new = compute_c5(h, n)[0]

                            else:

            if not improved:

def binary_local_search(h_cont, n, deadline, rng):
    h = (h_cont > 0.5).astype(float)
    n_ones = int(h.sum())
    target = n // 2
    if n_ones > target:
        idx = np.where(h == 1)[0]
        order = idx[np.argsort(h_cont[idx])]
        h[order[:n_ones - target]] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        order = idx[np.argsort(-h_cont[idx])]
        h[order[:target - n_ones]] = 1
    best_c = compute_c5(h, n)[0]
    best_h = h.copy()
    for _round in range(3000):
        if time.time() > deadline:
            break
        c_val, c_arr = compute_c5(h, n)
        worst_idx = np.argmax(c_arr)
        worst_lag = abs(worst_idx - (n - 1))
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        n_tries = min(100, len(ones) * len(zeros))
        for _t in range(n_tries):
            if time.time() > deadline:
                return best_h, best_c
            if worst_lag > 0 and worst_lag < n and rng.rand() < 0.4:
                cands = [(i, i + worst_lag) for i in range(n - worst_lag)
                         if h[i] == 1 and h[i + worst_lag] == 0]
                if not cands:
                    cands = [(i, i + worst_lag) for i in range(n - worst_lag)
                             if h[i] == 0 and h[i + worst_lag] == 1]
                if cands:
                    i, j = cands[rng.randint(len(cands))]
                else:
                    i = rng.choice(ones); j = rng.choice(zeros)
            else:
                i = rng.choice(ones); j = rng.choice(zeros)
            if i == j:
                continue
            h[i], h[j] = h[j], h[i]
            c_new = compute_c5(h, n)[0]
            if c_new < best_c - 1e-14:
                best_c = c_new; best_h = h.copy()
                improved = True
            else:
                h[i], h[j] = h[j], h[i]
        if not improved and _round > 100:
            break
    return best_h, best_c

def parametric_try(n, deadline, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    best_c = float('inf')
    best_h = None
    n_harm = 5
    for trial in range(4):
        if time.time() > deadline:
            break
        params = np.zeros(2 * n_harm)
        for m in range(n_harm):
            params[2*m] = 0.4 / (m + 1)
            params[2*m+1] = rng.uniform(0, 2*np.pi)
        def obj(p):
            hh = np.full(n, 0.5)
            for m in range(n_harm):
                hh += p[2*m] * np.sin(2*np.pi*(m+1)*x + p[2*m+1])
            hh = project_feasible(hh, n)
            return compute_c5(hh, n)[0]
        try:
            res = minimize(obj, params, method='Nelder-Mead',
                          options={'maxiter': 3000, 'xatol': 1e-9, 'fatol': 1e-10})
            if res.fun < best_c:
                best_c = res.fun
                hh = np.full(n, 0.5)
                for m in range(n_harm):
                    hh += res.x[2*m] * np.sin(2*np.pi*(m+1)*x + res.x[2*m+1])
                best_h = project_feasible(hh, n)
        except:
            pass
    return best_h, best_c
