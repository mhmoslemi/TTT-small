    c = np.correlate(h, 1 - h, mode='full') * dx

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:

                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:

            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_c_at_lag(h, n, k_lag):
    """Gradient of correlation at a specific lag w.r.t. h."""
    # c[k] = sum_i h[i] * (1 - h[i+k]) over valid i
    # where the lag in np.correlate is k - (n-1)
    s = k_lag - (n - 1)
    g = np.zeros(n)
    # For each i where both i and i+s are valid:
    # d/dh[i] of h[i]*(1-h[i+s]) = (1-h[i+s]) - h[i] (no wait)
    # Actually c[s] = sum_i h[i] * (1 - h[i+s]) for valid i

    #              term where j=i+s gives -h[j-s]

    lo2 = max(0, lo + s)

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c)), c

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_c_at_lag(h, n, k_lag):
    s = k_lag - (n - 1)
    g = np.zeros(n)
    lo = max(0, -s)
    hi = min(n - 1, n - 1 - s)
    if lo > hi:
        return g
    idx = np.arange(lo, hi + 1)
    g[idx] += (1.0 - h[idx + s])
    lo2 = max(0, lo + s)
    hi2 = min(n - 1, hi + s)
    if lo2 <= hi2:
        idx2 = np.arange(lo2, hi2 + 1)
        g[idx2] -= h[idx2 - s]
    return g

def slsqp_active(h, n, active_lags, alpha=20000, maxiter=400):
    dx = 2.0 / n
    al = list(active_lags)
    def obj(x):
        h_x = np.clip(x, 0, 1)
        c = np.correlate(h_x, 1 - h_x, mode='full') * dx
        ca = c[al]
        m = ca.max()
        shifted = alpha * (ca - m)
        mx = shifted.max()
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: float(np.sum(x) - n / 2.0),
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def slsqp_full(h, n, alpha=10000, maxiter=400):

    def obj(x):

        c = np.correlate(h_x, 1 - h_x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: float(np.sum(x) - n / 2.0),
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:

def active_set_refine(h, n, deadline, num_active=3, max_rounds=50):

def line_search_cd(h, n, deadline, num_top=3, max_rounds=5000):

def anneal_gradient(h, n, deadline, num_top=3, T0=0.002, cooling=0.997):
    best_c, _ = compute_c5(h, n)
    best_h = h.copy()
    curr_c = best_c
    T = T0
    rng = np.random.RandomState(42)
    while time.time() < deadline and T > 1e-9:
        c = get_corr(h, n)
        if float(c.max()) < best_c - 1e-15:
            best_c = float(c.max())
            best_h = h.copy()
        top_k = np.argsort(c)[-num_top:]
        g = np.zeros(n)
        for k in top_k:
            g += grad_c_at_lag(h, n, int(k))
        if g.max() - g.min() < 1e-14:
            break
        if rng.random() < 0.75:
            hi_thresh = np.percentile(g, 75)
            lo_thresh = np.percentile(g, 25)
            src_cand = np.where((g >= hi_thresh) & (h > 0.01))[0]
            dst_cand = np.where((g <= lo_thresh) & (h < 0.99))[0]
            if len(src_cand) == 0 or len(dst_cand) == 0:
                i_src = int(np.argmax(g))
                i_dst = int(np.argmin(g))
            else:
                i_src = int(src_cand[rng.randint(len(src_cand))])
                i_dst = int(dst_cand[rng.randint(len(dst_cand))])
        else:
            i_src = int(rng.randint(n))
            i_dst = int(rng.randint(n))
        if i_src == i_dst or h[i_src] < 0.01 or h[i_dst] > 0.99:
            T *= cooling
            continue
        d = (0.001 + T * 50.0) * (1.0 + rng.random() * 0.5)
        d = min(d, h[i_src] * 0.9, (1.0 - h[i_dst]) * 0.9)
        if d < 1e-12:
            T *= cooling
            continue
        h[i_src] -= d
        h[i_dst] += d
        new_c = float(get_corr(h, n).max())
        delta = new_c - curr_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            curr_c = new_c
        else:
            h[i_src] += d
            h[i_dst] -= d
        T *= cooling
    return best_h, best_c
