def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def soft_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    # Soft max approximation
    shifted = alpha * (c - m)
    sm = shifted.max()
    soft = m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    # Constraint penalty
    penalty = 1e6 * (x.sum() - n/2.0)**2
    return soft + penalty

def soft_max_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    # Soft max gradient
    probs = np.exp(shifted - sm) / np.exp(shifted - sm).sum()
    # d(soft_max)/dx = dx * (probs[k] * d cor[k]/dx)
    # d cor[k]/dx[i] = (1-x[i+k]) - x[i] (for appropriate k)
    # This is complex; use the argmax gradient as approximation
    m_star = int(np.argmax(c))
    k = m_star - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - x[:n-k])
        g[:n-k] += -x[k:]
    else:
        g[:n+k] += (1.0 - x[-k:])
        g[-k:] += -x[:n+k]
    g *= dx
    # Add penalty gradient
    g += 2e6 * (x.sum() - n/2.0)
    return g

def kkt_step(h, n, n_active=5):
    """Take a KKT-informed step to reduce the max correlation."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    # Find top n_active lags
    top_idx = np.argsort(c)[-n_active:]
    # Compute gradients for active lags
    grads = []
    for idx in top_idx:
        k = idx - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            g[k:] += (1.0 - h[:n-k])
            g[:n-k] += -h[k:]
        else:
            g[:n+k] += (1.0 - h[-k:])
            g[-k:] += -x[:n+k]
        grads.append(g)
    # Equal weights (simple) or proportional to c values
    weights = c[top_idx]
    weights = weights / weights.sum()
    direction = sum(w * g for w, g in zip(weights, grads))
    direction *= dx
    # Project direction onto feasible set (reduce sum to maintain constraint)
    # subtraction makes sum decrease, addition increases
    # We want to maintain sum(h) = n/2
    mean_dir = direction.mean()
    direction = direction - mean_dir  # zero-mean direction preserves sum
    return direction

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def soft_max_obj_grad(h, n, alpha):
    dx = 2.0 / n
    c_raw = np.correlate(h, 1 - h, mode='full')
    c = c_raw * dx
    c_max = c.max()
    shifted = alpha * (c - c_max)
    sm = shifted.max()
    log_sum_exp = sm + np.log(np.exp(shifted - sm).sum())
    p = np.exp(shifted - log_sum_exp)
    
    obj = c_max + log_sum_exp / alpha
    obj += 1e4 * (h.sum() - n / 2.0) ** 2
    
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        w = p[idx]
        if w < 1e-10:
            continue
        k = idx - (n - 1)
        if k >= 0:
            g[k:] += w * (1.0 - h[:n-k])
            g[:n-k] -= w * h[k:]
        else:
            m = -k
            g[:n-m] += w * (1.0 - h[m:])
            g[m:] -= w * h[:n-m]
    g *= dx
    g += 2e4 * (h.sum() - n / 2.0)
    return obj, g

def run_lbfgs(h, n, alpha, maxiter, deadline):
    def obj(x):
        return soft_max_obj_grad(x, n, alpha)[0]
    def grad(x):
        return soft_max_obj_grad(x, n, alpha)[1]
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10,
                                'maxcor': 20})
        return project_feasible(res.x, n)
    except Exception:
        return h

def kkt_descent(h, n, deadline, n_active=10):
    """KKT-informed steepest descent with line search."""
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.01
    for it in range(50000):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full') * dx
        c_max = c.max()
        if c_max < best_c:
            best_c = c_max
            best_h = h.copy()
        # Top n_active lags
        top = np.argsort(c)[-n_active:]
        # Compute weighted gradient
        g = np.zeros(n)
        for idx in top:
            k = idx - (n - 1)
            if k >= 0:
                g[k:] += (1.0 - h[:n-k])
                g[:n-k] -= h[k:]
            else:
                m = -k
                g[:n-m] += (1.0 - h[m:])
                g[m:] -= h[:n-m]
        g *= dx
        # Zero-mean to preserve sum constraint
        g -= g.mean()
        # Line search
        improved = False
        for trial in range(10):
            h_try = project_feasible(h - lr * g, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c:
                h = h_try
                best_c = c_try
                best_h = h.copy()
                lr = min(lr * 1.1, 0.5)
                improved = True
                break
            lr *= 0.7
        if not improved:
            # Random restart
            h = project_feasible(best_h + 0.02 * np.random.randn(n), n)
            lr = 0.01
    return best_h, best_c

The total objective (soft-max approximated by the max) is:
max_k f_k = max_k [sum_{i: j=i or j=i+k} (terms with h[j])]

1. Find the active lag k*
2. For each j, optimize h[j] to minimize f_{k*}(h) (keeping other h fixed)
3. Iterate

For fixed k*, f_{k*} = sum_i h[i+k*]*(1-h[i]) (for k* ≥ 0)

The terms involving h[j]:
- If j ≤ n-1-k*: h[j+k*] * (1-h[j]) contributes
- If j ≥ k*: h[j] * (1-h[j-k*]) contributes

This gives h[j] = ... wait, h[j] doesn't appear in its own derivative! The derivative w.r.t. h[j] only involves h[j-k*] and h[j+k*].

So f_{k*} as a function of h[j] alone:
f_{k*} = ... + h[j+k*]*(1-h[j]) [if j ≤ n-1-k*] + h[j]*(1-h[j-k*]) [if j ≥ k*] + ...

= ... + h[j+k*] - h[j+k*]*h[j] + h[j] - h[j]*h[j-k*] + ...

As a quadratic in h[j]:

= ... + h[j] * (1 - h[j+k*] - h[j-k*]) + ...

This is LINEAR in h[j]! (The h[j]² term doesn't appear.)

So for fixed k* and fixed other values, f_{k*} is linear in h[j]. The optimum is at h[j] = 0 if the coefficient is negative, or h[j] = 1 if positive.

This means the coordinate descent for the max problem will always push values to 0 or 1 (binary)! That's why binary solutions are natural.

This explains why the continuous solution is hard to find: the coordinate descent naturally pushes to binary, but the optimal continuous solution exists in between.

So the key is to use the SOFT-MAX (not just max) to get a truly continuous objective, and then optimize it.

With the soft-max, the objective is:
SoftMax(c) = c_max + (1/alpha) * log(sum_k exp(alpha * (c_k - c_max)))

This is smooth, and the gradient is a weighted average of all the f_k gradients. The weighted average can have a non-trivial minimum in the interior (not just at 0 or 1).

So my approach is correct: use the soft-max with a carefully chosen alpha, and optimize with L-BFGS-B.

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def _sm_compute(x, n, alpha):
    dx = 2.0 / n
    c_raw = np.correlate(x, 1 - x, mode='full')
    c = c_raw * dx
    c_max = c.max()
    shifted = alpha * (c - c_max)
    sm = shifted.max()
    log_sum_exp = sm + np.log(np.exp(shifted - sm).sum())
    p = np.exp(shifted - log_sum_exp)
    obj = c_max + log_sum_exp / alpha
    obj += 1e4 * (x.sum() - n / 2.0) ** 2
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        w = p[idx]
        if w < 1e-10:
            continue
        k = idx - (n - 1)
        if k >= 0:
            g[k:] += w * (1.0 - x[:n-k])
            g[:n-k] -= w * x[k:]
        else:
            m = -k
            g[:n-m] += w * (1.0 - x[m:])
            g[m:] -= w * x[:n-m]
    g *= dx
    g += 2e4 * (x.sum() - n / 2.0)
    return obj, g

    return _sm_compute(x, n, alpha)[0]

    return _sm_compute(x, n, alpha)[1]

    p2 = t0 + 0.60 * (budget_s - 3)

    if c_new < best_c:
        best_c = c_new
        best_h = h_new.copy()

    # Phase 3: Multi-restart L-BFGS-B (20%)
    p3 = t0 + 0.80 * (budget_s - 3)
    restart_alphas = [50000, 100000, 50000, 200000, 50000, 100000,
                      30000, 50000, 100000, 500000,
                      20000, 50000, 100000, 200000, 50000, 500000]
    for i in range(16):
        if time.time() > p3 - 3:
            break
        pert = 0.04 / (1 + 0.15 * i)
        h_try = project_feasible(best_h + pert * rng.randn(n), n)
        h_new = run_lbfgs(h_try, n, restart_alphas[i], 1500, p3)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()

    # Phase 4: Binary local search (10%)
    p4 = t0 + 0.90 * (budget_s - 3)
    sorted_idx = np.argsort(best_h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n // 2]] = 1.0
    hb, cb = binary_local_search(h_bin, n, p4, rng)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Phase 5: Final KKT + L-BFGS polish (10%)
    h_new, c_new = kkt_descent(best_h.copy(), n, deadline, 5, rng)
    if c_new < best_c:
        best_c = c_new
        best_h = h_new.copy()
    if time.time() < deadline - 5:
        h_new = run_lbfgs(best_h.copy(), n, 200000, 2000, deadline)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
