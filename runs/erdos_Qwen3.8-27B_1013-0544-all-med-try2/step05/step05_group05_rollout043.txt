import time

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_g(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binary_round(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:target]] = 1.0
    return h_b

def compute_A(h, n, lags):
    K = len(lags)
    A = np.zeros((K, n))
    for k in range(K):

            A[k, :n-s] += (1.0 - h[s:])
            A[k, s:] -= h[:n-s]
        elif s < 0:
            l = -s
            A[k, l:] += (1.0 - h[:n-l])
            A[k, :n-l] -= h[l:]
    return A

∂g(s)/∂h[p] = dx * [∂/∂h[p] Σ_x h[x]*(1-h[x+s])]

import numpy as np
import time

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_g(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binary_round(h, n):
    target = n // 2
    idx = np.argsort(h, kind='stable')[::-1]
    h_b = np.zeros(n)
    h_b[idx[:target]] = 1.0
    return h_b

def compute_A(h, n, lags):
    K = len(lags)
    A = np.zeros((K, n))
    for k in range(K):
        s = int(lags[k])
        if s > 0:
            A[k, :n-s] += (1.0 - h[s:])
            if n - s > 0:
                A[k, s:] -= h[:n-s]
        elif s < 0:
            l = -s
            A[k, l:] += (1.0 - h[:n-l])
            if n - l > 0:
                A[k, :n-l] -= h[l:]
    return A

def blls_fast(h, n, deadline, K=12, max_rounds=50000):
    h = h.copy()
    g = compute_g(h, n)
    best_c = g.max()
    best_h = h.copy()
    stagnant = 0
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        m = g.max()
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        g_no0 = g.copy()
        g_no0[n-1] = -1e9
        top_idx = np.argsort(g_no0)[::-1][:K]
        lags = top_idx - (n-1)
        g_K = g[top_idx]
        A = compute_A(h, n, lags)
        A_z = A[:, zeros]
        A_o = A[:, ones]
        abs_lags = np.abs(lags).astype(int)
        
        # Vectorized scoring
        base = (g_K[None,:,None] - A_o[:,:,None] + A_z[None,:,:]).max(axis=1)
        dists = np.abs(zeros[None,:] - ones[:,None])
        corr = np.zeros_like(dists, dtype=bool)
        for al in abs_lags:
            corr |= (dists == al)
        scores = base + corr
        
        min_val = scores.min()
        if min_val < m - 1e-15:
            ii, jj = np.unravel_index(scores.argmin(), scores.shape)
            h[ones[ii]] = 0.0
            h[zeros[jj]] = 1.0
            g = compute_g(h, n)
            m2 = g.max()
            if m2 < best_c - 1e-14:
                best_c = m2
                best_h = h.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            stagnant += 1
            if stagnant > 50:
                break
    
    return best_h, best_c

def proj_subgrad(h, n, deadline, max_iter=100000):

    best_h = h.copy()

import numpy as np
import time

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_g(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binary_round(h, n):
    target = n // 2
    idx = np.argsort(h, kind='stable')[::-1]
    h_b = np.zeros(n)
    h_b[idx[:target]] = 1.0
    return h_b

def compute_A(h, n, lags):
    K = len(lags)
    A = np.zeros((K, n))
    for k in range(K):
        s = int(lags[k])
        if s > 0:
            if n - s > 0:
                A[k, :n-s] += (1.0 - h[s:])
                A[k, s:] -= h[:n-s]
        elif s < 0:
            l = -s
            if n - l > 0:
                A[k, l:] += (1.0 - h[:n-l])
                A[k, :n-l] -= h[l:]
    return A

    best_c = g.max()
    best_h = h.copy()
    stagnant = 0

        if time.time() > deadline:

        m = g.max()
        ones = np.where(h == 1)[0]
