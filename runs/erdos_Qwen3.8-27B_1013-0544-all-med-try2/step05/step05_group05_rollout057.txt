def slp_step(h0, n, max_lags=500):
    """One step of successive linear programming"""
    corr = compute_corr(h0, n)  # all 2n-1 values
    m = corr.max()
    dx = 2.0 / n
    num_lags = 2*n - 1
    
    # Build LP
    # Variables: h_0,...,h_{n-1}, t (total n+1)
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    A_ub = np.zeros((num_lags + 2*n, n + 1))
    b_ub = np.zeros(num_lags + 2*n)
    
    for ki in range(num_lags):
        lag = ki - (n - 1)
        # Compute gradient of g_ki at h0
        grad_k = np.zeros(n)
        if lag >= 0:
            if n - lag > 0:
                grad_k[:n-lag] = dx * (1.0 - h0[lag:])
            if lag > 0:
                grad_k[lag:] -= dx * h0[:n-lag]
        else:
            to = -lag
            if n - to > 0:
                grad_k[:n-to] = dx * (1.0 - h0[to:])
            grad_k[to:] -= dx * h0[:n-to]
        
        rhs = grad_k @ h0 - corr[ki]
        A_ub[ki, :n] = grad_k
        A_ub[ki, n] = -1.0
        b_ub[ki] = rhs
    
    # Upper bounds: h_i <= 1
    for i in range(n):
        row = num_lags + i
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0
    
    # Lower bounds: -h_i <= 0
    for i in range(n):
        row = num_lags + n + i
        A_ub[row, i] = -1.0
        b_ub[row] = 0.0
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n/2.0])
    
    from scipy.optimize import linprog
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  method='highs', options={'maxiter': 10000})
    
    if res.success:
        h_new = res.x[:n]

    num_lags = 2 * n - 1
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = compute_corr(h, n)
        # Build LP
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_ub = np.zeros((num_lags + 2 * n, n + 1))
        b_ub = np.zeros(num_lags + 2 * n)
        
        for ki in range(num_lags):
            lag = ki - (n - 1)
            grad_k = np.zeros(n)
            if lag >= 0:
                if n - lag > 0:
                    grad_k[:n - lag] = dx * (1.0 - h[lag:])
                if lag > 0:
                    grad_k[lag:] -= dx * h[:n - lag]
            else:
                to = -lag
                if n - to > 0:
                    grad_k[:n - to] = dx * (1.0 - h[to:])
                grad_k[to:] -= dx * h[:n - to]
            rhs = grad_k @ h - corr[ki]
            A_ub[ki, :n] = grad_k
            A_ub[ki, n] = -1.0
            b_ub[ki] = rhs
        
        I = np.eye(n)
        A_ub[num_lags:num_lags+n] = I
        b_ub[num_lags:num_lags+n] = 1.0

        b_ub[num_lags+n:num_lags+2*n] = 0.0

        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])

                      method='highs', options={'maxiter': 5000})

            break
        h_new = res.x[:n]
        # Trust region
        step = h_new - h
        max_step = np.max(np.abs(step))
        if max_step > trust_region:
            h_new = h + (trust_region / max_step) * step
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            h = h_new
            best_c = c_new

        elif c_new < best_c:
            h = h_new  # accept even small improvements
        else:

            h_half = project_feasible(h + 0.5 * (h_new - h), n)
            c_half = compute_c5(h_half, n)
            if c_half < best_c:
                h = h_half
                best_c = c_half
                best_h = h.copy()
            else:
                break  # No improvement, stop
    
    return best_h, best_c

import numpy as np
from scipy.optimize import linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + d / s * room, 0, 1)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + d / s * h, 0, 1)
    return h

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def build_lp(h, n, corr):
    dx = 2.0 / n
    num_lags = 2 * n - 1
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((num_lags + 2 * n, n + 1))

        A_ub[ki, :n] = grad_k
        A_ub[ki, n] = -1.0
        b_ub[ki] = rhs
    I = np.eye(n)
    A_ub[num_lags:num_lags + n] = I

    stagnant = 0

        if time.time() > deadline:
            break
        corr = compute_corr(h, n)
        c_obj, A_ub, b_ub, A_eq, b_eq = build_lp(h, n, corr)
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      method='highs', options={'maxiter': 5000})
        if not res.success:
            break
        h_raw = res.x[:n]
        step = h_raw - h
        mx = np.max(np.abs(step))
        if mx < 1e-14:
            break
        if mx > tr:

        else:

        h_new = project_feasible(h_new, n)

def subgrad_polish(h0, n, deadline, max_iter=50000):

    best_h = h.copy()

    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = compute_corr(h, n)
        m = corr.max()
        active = np.where(corr >= m - 1e-4 * m)[0]
        grad = np.zeros(n)
        dx = 2.0 / n
        for ki in active:
            lag = ki - (n - 1)
            if lag >= 0:
                if n - lag > 0:
                    grad[:n - lag] += dx * (1.0 - h[lag:])
                if lag > 0:
                    grad[lag:] -= dx * h[:n - lag]
            else:
                to = -lag
                if n - to > 0:
                    grad[:n - to] += dx * (1.0 - h[to:])
                grad[to:] -= dx * h[:n - to]
        grad /= len(active)
        ng = np.linalg.norm(grad)

        grad /= ng
        found = False

            ct = compute_c5(ht, n)

                found = True
                break
        if not found:
            lr *= 0.8
            if lr < 1e-14:
                break
        else:
            c = compute_c5(h, n)
            if c < best_c:
