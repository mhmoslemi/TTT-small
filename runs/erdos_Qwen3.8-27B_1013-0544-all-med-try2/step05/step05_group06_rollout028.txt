import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_gradient_lexp(h, n, alpha):
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    g_max = g.max()
    exp_g = np.exp(alpha * (g - g_max))
    w = exp_g / exp_g.sum()
    u = 1.0 - 2.0 * h
    grad = np.zeros(n)
    for j in range(n):
        offset = n - 1 - j
        grad[j] = dx * np.dot(u, w[offset:offset+n])
    return grad, g.max()

def binary_local_search(h, n, deadline, max_sweeps=200):
    """1-swap binary local search."""
    h = np.round(h)
    h = np.clip(h, 0, 1)
    # Ensure sum = n/2
    n_ones = int(h.sum())
    target = n // 2
    if n_ones > target:
        ones = np.where(h == 1)[0]
        rng = np.random.RandomState(0)
        for _ in range(n_ones - target):
            idx = rng.choice(len(ones))
            h[ones[idx]] = 0
            ones = np.delete(ones, idx)
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        rng = np.random.RandomState(0)
        for _ in range(target - n_ones):
            idx = rng.choice(len(zeros))
            h[zeros[idx]] = 1
            zeros = np.delete(zeros, idx)
    
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    ones = sorted(np.where(h == 1)[0].tolist())
    zeros = sorted(np.where(h == 0)[0].tolist())
    
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        # Try all 1-swap moves
        for o in ones:
            if time.time() > deadline:
                break
            for z in zeros:
                h[o] = 0
                h[z] = 1
                c = compute_c5(h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    best_h = h.copy()
                    ones.remove(o)
                    zeros.remove(z)
                    ones.append(o)  # o is now 0, z is now 1
                    # Actually need to update properly
                    # o becomes 0, z becomes 1
                    ones = sorted(set(ones) - {o} | {z})
                    zeros = sorted(set(zeros) - {z} | {o})
                    improved = True
                    break
                else:
                    h[o] = 1
                    h[z] = 0
            if improved:
                break
        if not improved:
            stall += 1
            if stall > 3:
                break
        else:
            stall = 0
    return best_h, best_c

    target = n // 2
    n_ones = int(h.sum())
    if n_ones > target:

        h[ones[n_ones-target:]] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        h[zeros[:target-n_ones]] = 1
    
    best_c = compute_c5(h, n)

        if time.time() > deadline:
            break
        improved = False
        for oi, o in enumerate(ones):
            if time.time() > deadline:
                break
            # Precompute for this o:
            idx_o_plus = o + lags  # o+l for each lag l
            idx_o_minus = o - lags  # o-l for each lag l
            v4 = (idx_o_plus >= 0) & (idx_o_plus < n)
            v1 = (idx_o_minus >= 0) & (idx_o_minus < n)
            t4 = np.zeros(n_)
            t4[v4] = h[idx_o_plus[v4]]
            t1 = np.zeros(n_)
            t1[v1] = h[idx_o_minus[v1]]
