from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def softmax_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    val = np.log(np.sum(np.exp(shifted - mx)))
    return float(m + val / alpha)

def sum_fun(x, n):
    return float(x.sum() - n / 2.0)

def sum_jac(x, n):
    return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac, 'args': (n,)}]
    try:
        res = minimize(softmax_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except:
        return h.copy()

    best_h = h.copy()

    dx = 2.0 / n

            # For j where j-shift >= 0: grad[j] += -h[j-shift]
            upper = slice(0, n - shift)

            grad[upper] += (1 - h[s:n])

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def softmax_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    val = np.log(np.sum(np.exp(shifted - mx)))
    return float(m + val / alpha)

def sum_fun(x, n):
    return float(x.sum() - n / 2.0)

def sum_jac(x, n):
    return np.ones(n)

def slsqp_step(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac, 'args': (n,)}]
    try:
        res = minimize(softmax_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def active_shift_gradient(h, n, shift):
    grad = np.zeros(n)
    if shift >= 0:
        s = shift
        if s == 0:
            return np.zeros(n)
        for j in range(s, n):
            grad[j] += (1.0 - h[j - s])
        for j in range(n - s):
            grad[j] -= h[j + s]
    else:
        s = -shift
        for j in range(n - s):
            grad[j] += (1.0 - h[j + s])
        for j in range(s, n):
            grad[j] -= h[j - s]
    return grad

def gradient_local_search(h, n, deadline, max_iters=1000000, recompute_every=200):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = 0.02
    momentum = 0.3
    vel = np.zeros(n)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    k_max = np.argmax(c)
    shift = k_max - (n - 1)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if it % recompute_every == 0:
            c = np.correlate(h, 1 - h, mode='full') * dx
            k_max = np.argmax(c)
            shift = k_max - (n - 1)
        grad = active_shift_gradient(h, n, shift)
        grad -= grad.mean()
        vel = momentum * vel - lr * grad
        h_new = np.clip(h + vel, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.02, 0.15)
        else:
            h = h_new
            lr = max(lr * 0.97, 0.002)
        if it % 10000 == 0 and it > 0:
            ht = slsqp_step(h, n, 50000, maxiter=500)
            c_s = compute_c5(ht, n)
            if c_s < best_c:
                best_c = c_s
                best_h = ht.copy()
                h = ht
                vel = np.zeros(n)
    return best_h, best_c

def pair_local_search(h, n, deadline, deltas, radius=80, max_passes=100):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _ in range(max_passes):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
            if not improved:
                break
    return h, best_c

def optimize_at_n(h_start, n, deadline, rng):

    best_h = h.copy()

    alpha_schedule = [500, 2000, 10000, 50000, 200000, 1000000, 5000000]
    for a in alpha_schedule:

            best_c = c
            best_h = ht.copy()
    if time.time() < deadline - 8:

    if time.time() < deadline - 4:

        if c < best_c:
            best_c = c
            best_h = ht.copy()
    if time.time() < deadline - 2:
        best_h, best_c = gradient_local_search(best_h, n, deadline - 1)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c
