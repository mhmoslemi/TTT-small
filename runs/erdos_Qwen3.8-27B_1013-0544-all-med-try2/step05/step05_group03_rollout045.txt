        best_c = c
        best_h = h.copy()

    # Phase 4: Local search again (8% budget)
    p4 = t0 + 0.71 * (budget_s - 5)
    h, c = local_search(best_h, n, p4)
    if c < best_c:
        best_c = c
        best_h = h.copy()

    # Phase 5: SA from perturbed best (15% budget)
    p5 = t0 + 0.86 * (budget_s - 5)
    hp = best_h.copy()
    ones = list(np.where(hp == 1)[0])
    zeros = list(np.where(hp == 0)[0])
    for _ in range(5):
        ii = rng.randint(len(ones))
        jj = rng.randint(len(zeros))
        hp[ones[ii]], hp[zeros[jj]] = 0.0, 1.0
        ones[ii], zeros[jj] = zeros[jj], ones[ii]
    h, c = sa_binary(hp, n, p5, seed=seed + 200, T0=0.004, cool=0.99995, Tmin=1e-10)
    if c < best_c:
        best_c = c
        best_h = h.copy()

    # Phase 6: Final local search (5% budget)
    h, c = local_search(best_h, n, dl - 3)
    if c < best_c:
        best_c = c
        best_h = h.copy()

    # Phase 7: Quick SA finish if time remains
    if time.time() < dl - 2:
        h, c = sa_binary(best_h, n, dl - 1, seed=seed + 999, T0=0.001, cool=0.99999, Tmin=1e-11)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if time.time() < dl - 0.5:
            h, c = local_search(best_h, n, dl)
            if c < best_c:
                best_c = c
                best_h = h.copy()

    return best_h, best_c, n
