        j = ones_list[np.random.randint(len(ones_list))]
        hb[i], hb[j] = 1.0, 0.0
        new_c = compute_c5(hb, n)
        delta = new_c - cur_c
        if delta < -1e-15 or (T > 1e-12 and np.random.random() < np.exp(-delta / T)):
            cur_c = new_c
            best_hb = hb.copy() if cur_c < best_c else best_hb
            best_c = min(best_c, cur_c)
            ones_set.add(i)
            ones_set.discard(j)
            zeros_set.discard(i)
            zeros_set.add(j)
            if it_count % 500 == 0:
                ones_list = list(ones_set)
                zeros_list = list(zeros_set)
        else:
            hb[i], hb[j] = 0.0, 1.0
    return best_hb, best_c

def make_random_init(n, seed):
    rng = np.random.RandomState(seed)
    return (rng.random(n) < 0.5).astype(float)

def make_fourier_init(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    n_freq = rng.randint(2, 5)
    for _ in range(n_freq):
        f = rng.randint(1, max(2, n // 4))
        p = rng.uniform(0, 2 * np.pi)
        h += np.cos(2 * np.pi * f * np.arange(n) / n + p)
    h = np.clip(h / n_freq + 0.5, 0, 1)
    return to_binary(h, n)

def make_xor_init(n, seed):
    rng = np.random.RandomState(seed)
    a = rng.randint(1, n // 2)
    b = rng.randint(1, n // 2)
    h = ((np.arange(n) * a + b * (np.arange(n) ** 2) * 0) % n < n // 2).astype(float)
    return to_binary(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)

    best_h = None
    best_c = float('inf')

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        h_parent = None
        n_parent = 250

    n_candidates = sorted(set([n_parent] + [x for x in range(80, 450, 10) if x % 2 == 0]))
    n_candidates.sort(key=lambda x: abs(x - n_parent))
    n_candidates = n_candidates[:10]

    total_budget = budget_s - 15
    weights = []
    for n in n_candidates:
        if n == n_parent:
            weights.append(4.0)
        elif abs(n - n_parent) <= 50:
            weights.append(2.0)
        else:
            weights.append(1.0)
    tw = sum(weights)
    allocs = [w * total_budget / tw for w in weights]

    for ni, n in enumerate(n_candidates):
        t_alloc = allocs[ni]
        t_start_n = time.time()
        t_end_n = min(t_start_n + t_alloc, deadline)
        if t_start_n >= deadline:
            break

        n_restarts = max(2, int(t_alloc / 45))
        n_restarts = min(n_restarts, 10)

        for r in range(n_restarts):
            r_frac = (r + 1) / n_restarts
            t_end_r = t_start_n + (t_end_n - t_start_n) * r_frac
            if time.time() >= t_end_r - 2:
                continue

            if r == 0 and n == n_parent and h_parent is not None:
                h0 = to_binary(h_parent, n)
            elif r == 1 and n == n_parent and h_parent is not None:
                h0 = to_binary(1.0 - h_parent, n)
            else:
                s = seed * 1000 + ni * 100 + r
                method = r % 4
                if method == 0:
                    h0 = make_random_init(n, s)
                elif method == 1:
                    h0 = make_fourier_init(n, s)
                elif method == 2:
                    h0 = make_xor_init(n, s)
                else:
                    h0 = make_random_init(n, s + 7777)

            # SA phase
            sa_dl = t_start_n + (t_end_r - t_start_n) * 0.55
            if time.time() < sa_dl:
                hb, c = sa_optimize(h0, n, sa_dl)
                if c < best_c:
                    best_c = c
                    best_h = hb.copy()

            # Greedy phase
            gr_dl = t_start_n + (t_end_r - t_start_n) * 0.85
            if time.time() < gr_dl:
                hb, c = greedy_swap(hb, n, gr_dl, max_sweeps=80)
                if c < best_c:
                    best_c = c
                    best_h = hb.copy()

            # Perturb + SA + Greedy
            if time.time() < t_end_r - 1:
                p = hb.copy()
                for _ in range(5):
                    oz = np.where(p == 0)[0]
                    on = np.where(p == 1)[0]
                    a2 = oz[np.random.randint(len(oz))]
                    b2 = on[np.random.randint(len(on))]
                    p[a2], p[b2] = 1.0, 0.0
                hb2, c2 = sa_optimize(p, n, min(t_end_r - 5, time.time() + 15))
                hb2, c2 = greedy_swap(hb2, n, t_end_r - 0.5, max_sweeps=20)
                if c2 < best_c:
                    best_c = c2
                    best_h = hb2.copy()

    # Final polish on best solution
    if best_h is not None and time.time() < deadline - 5:
        n_best = len(best_h)
        hb, c = greedy_swap(best_h.copy(), n_best, deadline - 2, max_sweeps=50)
        if c < best_c:
            best_c = c
            best_h = hb

    if best_h is None:
        n = n_parent if n_parent % 2 == 0 else 250
        best_h = to_binary(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    return best_h, best_c, len(best_h)
