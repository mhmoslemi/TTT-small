```python
        if abs(diff) < 1e-14:

def compute_A_matrix(h, n):
    """A[p, k] = h[p+k] + h[p-k] (0 for out of bounds), k = -(n-1)..n-1"""
    m = 2 * n - 1
    k_vals = np.arange(-(n-1), n)  # shape (m,)

    pk = np.arange(n)[:, None] + k_vals[None, :]

    A[mask_pmk] += h[pmk[mask_pmk]]
```

```python
def vectorized_binary_search(h, n, deadline, n_restarts=5, seed=42):
    m = 2 * n - 1
    target = n // 2
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    
    # Initial binary from h
    h = np.where(h >= 0.5, 1.0, 0.0)
    # Ensure exactly n/2 ones
    idx = np.argsort(h)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    
    corr = np.correlate(h, 1 - h, mode='full')
    cur_max = corr.max()
    best_c = cur_max * dx
    best_h = h.copy()
    
    A = compute_A_matrix(h, n)
    
    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        if restart > 0:
            # Perturb: swap some random pairs
            for _ in range(rng.randint(3, 10)):
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                a = rng.choice(ones)
                b = rng.choice(zeros)
                h[a], h[b] = 0.0, 1.0
                update_A_after_swap(A, n, a, b)
            corr = np.correlate(h, 1 - h, mode='full')
            cur_max = corr.max()
        
        # Local search
        no_improve = 0
        while time.time() < deadline:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            n_ones = len(ones)
            n_zeros = len(zeros)
            
            if n_ones == 0 or n_zeros == 0:
                break
            
            # For each zero i, compute delta for all one j's
            # delta[j, k] = A[ones[j], k] - A[i, k]
            # new_corr = corr + delta
            # We want to minimize max(new_corr)
            
            best_improve = -1e-10
            best_i = -1
            best_j = -1
            best_new_corr = None
            
            for i in zeros:
                # delta for all j: shape (n_ones, m)
                deltas = A[ones, :] - A[i, :]  # (n_ones, m)
                new_maxes = (corr[None, :] + deltas).max(axis=1)  # (n_ones,)
                best_j_local = new_maxes.argmin()
                improve = cur_max - new_maxes[best_j_local]
                if improve > best_improve:
                    best_improve = improve
                    best_i = i
                    best_j = ones[best_j_local]
                    best_new_corr = corr + deltas[best_j_local]
            
            if best_improve <= 0:
                break
            
            # Make the swap
            h[best_i], h[best_j] = 1.0, 0.0
            corr = best_new_corr
            cur_max = corr.max()
            update_A_after_swap(A, n, best_i, best_j)
            
            if cur_max * dx < best_c:
                best_c = cur_max * dx
                best_h = h.copy()
    
    return best_h, best_c
```

Δc_vs(k) = h[j+k] + h[j-k] - h'[i+k] - h'[i-k]  for all k (including k=0)

And A[j,k] - A[i,k] = h[j+k] + h[j-k] - h[i+k] - h[i-k] (all before values)

The difference (correction) is: h[i+k] + h[i-k] - h'[i+k] - h'[i-k]

This is non-zero only when i+k or i-k equals i or j:
- i+k = i (k=0): h[i] - h'[i] = 0 - 1 = -1
- i+k = j (k=j-i): h[j] - h'[j] = 1 - 0 = +1
- i-k = i (k=0): h[i] - h'[i] = 0 - 1 = -1
- i-k = j (k=i-j): h[j] - h'[j] = 1 - 0 = +1

So:
- k=0: correction = -1 + -1 = -2
- k=j-i: correction = +1
- k=i-j: correction = +1
- other k: correction = 0

Therefore:
Δc_vs(k) = A[j,k] - A[i,k] + correction(k)

where correction(k) = -2 if k=0, +1 if k=±(j-i), 0 otherwise.

Let me verify for k=j-i:
A[j,k]-A[i,k] + 1 = (h[2j-i] + 0 - 1 - h[2i-j]) + 1 = h[2j-i] - h[2i-j] ✓

For k=0:
A[j,0]-A[i,0] + (-2) = 2 + (-2) = 0 ✓

For generic k:
A[j,k]-A[i,k] + 0 = h[j+k]+h[j-k]-h[i+k]-h[i-k] ✓

```python
            best_j = -1
```

```python
        #   From h[i] in the first factor: (1-h[i-k]) if 0<=i-k<n, else 1 (since h outside = 0, 1-h = 1)
        #   From h[i] in the second factor (through j-k=i, j=i+k): -h[i+k] if 0<=i+k<n, else 0
        # So dc/dh[i] = (1-h[i-k] if 0<=i-k<n else 1) - (h[i+k] if 0<=i+k<n else 0)
        
        # Update
        h_new = h - lr * sg / max(1, np.abs(sg).max())
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c
```
