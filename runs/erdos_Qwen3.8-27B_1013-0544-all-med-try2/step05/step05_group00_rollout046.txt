                    score += h[j+m]
                if 0 <= j-m < n:
                    score += h[j-m]

                    good_j.append((score, j))

            good_i.sort(reverse=True)

    best_c = cur_c
    best_h = h.copy()
    kicks = 0

    while time.time() < deadline:
        max_val = corr.max()
        peaks = np.where(corr >= max_val - 1e-10)[0]
        found = False
        for pk in peaks:
            m = int(pk - (n - 1))
            good_i = []
            for i in ones:
                sc = 0
                if 0 <= i + m < n:
                    sc += (1 - h[i + m])
                if 0 <= i - m < n:
                    sc += (1 - h[i - m])
                if sc > 0:
                    good_i.append((sc, i))
            good_j = []
            for j in zeros:
                sc = 0
                if 0 <= j + m < n:
                    sc += h[j + m]
                if 0 <= j - m < n:
                    sc += h[j - m]
                if sc > 0:
                    good_j.append((sc, j))
            if not good_i or not good_j:
                continue
            good_i.sort(reverse=True)
            good_j.sort(reverse=True)
            limit = min(len(good_i), 30)
            limitj = min(len(good_j), 30)
            for si, i in good_i[:limit]:
                for sj, j in good_j[:limitj]:
                    h[i], h[j] = 0.0, 1.0
                    nc = compute_corr(h, n)
                    ncv = float(nc.max()) * dx
                    if ncv < cur_c - 1e-15:
                        cur_c = ncv
                        corr = nc
                        ones.discard(i); ones.add(j)
                        zeros.discard(j); zeros.add(i)
                        found = True
                        if cur_c < best_c:
                            best_c = cur_c
                            best_h = h.copy()
                        break
                    else:
                        h[i], h[j] = 1.0, 0.0
                if found:
                    break
            if found:
                break
        if not found:
            kicks += 1
            if kicks > max_kicks:
                break
            n_kick = rng.randint(2, 5)
            ol = list(ones); zl = list(zeros)
            for _ in range(n_kick):
                if not ol or not zl:
                    break
                i = ol[rng.randint(len(ol))]
                j = zl[rng.randint(len(zl))]
                h[i], h[j] = 0.0, 1.0
                ones.discard(i); ones.add(j)
                zeros.discard(j); zeros.add(i)
                ol.remove(i); zl.remove(j)
            corr = compute_corr(h, n)
            cur_c = float(corr.max() * dx)
    return best_h, best_c

def projected_gradient(h, n, steps=500, lr0=0.02, deadline=None):
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    for it in range(steps):
        if deadline and time.time() > deadline:
            break
        corr = compute_corr(h, n)
        k_star = int(np.argmax(corr))
        shift = k_star - (n - 1)
        grad = np.zeros(n)
        for t in range(n):
            g = 0.0
            if 0 <= t + shift < n:
                g += (1.0 - h[t + shift])
            if 0 <= t - shift < n:
                g -= h[t - shift]
            grad[t] = g
        lr = lr0 / (1.0 + it * 0.01)
        h = h - lr * dx * grad
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, n)
    return h

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq',
                                 'fun': lambda x: x.sum() - n / 2.0}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project_feasible(res.x, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0

    # Phase 1: Binary local search from initial
    p1 = t0 + 0.30 * budget_s
    hb = binarize(best_h, n)
    hb, cb = peak_directed_search(hb, n, p1, seed=seed, max_kicks=300)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 2: Continuous refinement of binary
    p2 = t0 + 0.40 * budget_s
    hcont = project_feasible(hb + 0.08 * rng.randn(n), n)
    hcont = projected_gradient(hcont, n, steps=300, lr0=0.015, deadline=p2 - 10)
    cc = compute_c5(hcont, n)
    if cc < best_c:
        best_c, best_h = cc, hcont.copy()
    if time.time() < p2 - 5:
        hcont2 = slsqp_smooth(hcont, n, 50000, 500, p2 - 2)
        cc2 = compute_c5(hcont2, n)
        if cc2 < best_c:
            best_c, best_h = cc2, hcont2.copy()

    # Phase 3: Re-binarize and search again
    p3 = t0 + 0.60 * budget_s
    hb3 = binarize(best_h, n)
    hb3, cb3 = peak_directed_search(hb3, n, p3, seed=seed + 100, max_kicks=500)
    if cb3 < best_c:
        best_c, best_h = cb3, hb3.copy()

    # Phase 4: Try different resolution
    p4 = t0 + 0.80 * budget_s
    for nh in [200, 300, 400]:
        if time.time() > p4 - 60:
            break
        xn = np.linspace(0, 2, nh)
        xo = np.linspace(0, 2, n)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        hbh = binarize(hh, nh)
        hbh, cbh = peak_directed_search(hbh, nh, min(p4 - 30, time.time() + 80),
                                         seed=seed + nh, max_kicks=200)
        if cbh < best_c:
            best_c, best_h, n = cbh, hbh.copy(), nh
            if time.time() < p4 - 20:
                hci = project_feasible(hbh + 0.05 * rng.randn(nh), nh)
                hci = projected_gradient(hci, nh, steps=200, lr0=0.01, deadline=p4 - 15)
                cci = compute_c5(hci, nh)
                if cci < best_c:
                    best_c, best_h, n = cci, hci.copy(), nh

    # Phase 5: Final extended search
    p5 = dl - 5
    hb5 = binarize(best_h, n)
    hb5, cb5 = peak_directed_search(hb5, n, p5, seed=seed + 999, max_kicks=800)
    if cb5 < best_c:
        best_c, best_h = cb5, hb5.copy()

    # Final polish
    if time.time() < dl - 3:
        hf = project_feasible(best_h + 0.03 * rng.randn(n), n)
        hf = projected_gradient(hf, n, steps=100, lr0=0.005, deadline=dl - 2)
        cf = compute_c5(hf, n)
        if cf < best_c:
            best_c, best_h = cf, hf.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
