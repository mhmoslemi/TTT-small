```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def project_feasible(h, n):
    """Project h onto {sum(h)=n/2, 0<=h<=1} using water-filling"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            # Need to increase sum: add to elements with room
            room = 1.0 - h
            if room.sum() < 1e-14:
                break
            h += (diff / room.sum()) * room
        else:
            if h.sum() < 1e-14:
                break
            h += (diff / h.sum()) * h
    return np.clip(h, 0.0, 1.0)

def compute_gradient_smooth(h, n, alpha):
    """Compute gradient of smooth max objective (1/alpha)*log(sum exp(alpha*c_k*dx))"""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Softmax weights
    shifted = alpha * corr - (alpha * corr).max()
    w = np.exp(shifted)
    w /= w.sum()
    # Gradient
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        g = 0.0
        for m in range(2 * n - 1):
            k = m - (n - 1)  # lag
            wm = w[m]
            # First term: (1-h[i+k]) if 0 <= i+k < n
            if 0 <= i + k < n:
                g += wm * (1.0 - h[i + k])
            # Second term: -h[i-k] if 0 <= i-k < n
            if 0 <= i - k < n:
                g -= wm * h[i - k]
        grad[i] = g
    return grad * alpha  # gradient of (1/alpha)*log(sum exp(alpha*c))

def smooth_objective(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    m = (alpha * corr).max()
    return (1.0 / alpha) * (m + np.log(np.sum(np.exp(alpha * corr - m))))

def projected_gradient_descent(h, n, alpha, lr, deadline, momentum=0.9):
    """Projected gradient descent with momentum"""
    h = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    it = 0
    while time.time() < deadline:
        grad = compute_gradient_smooth(h, n, alpha)
        v = momentum * v - lr * grad
        h_new = h + v
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        it += 1
        # Adaptive learning rate
        if it % 500 == 0 and lr > 0.001:
            lr *= 0.95
    return best_h, best_c

def pair_search_local(h, n, deadline, deltas, max_passes=200):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        passes = 0
        while improved and passes < max_passes and time.time() < deadline:
            improved = False
            passes += 1
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
    return h, best_c

def slsqp_optimize(h, n, alpha, maxiter=3000):
    def obj(x):
        return smooth_objective(x, n, alpha)
    def obj_grad(x):
        return compute_gradient_smooth(x, n, alpha)
    def sum_con(x):
        return np.sum(x) - n / 2.0
    def sum_con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def optimize_full(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    # Phase 1: SLSQP with increasing alpha
    alphas = [100, 500, 2000, 10000, 50000, 200000]
    for a in alphas:
        if time.time() > deadline - 20:
            break
        ht = slsqp_optimize(best_h, n, a)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht.copy()
    
    # Phase 2: Projected gradient with momentum
    if time.time() < deadline - 30:
        ht, ct = projected_gradient_descent(best_h, n, 50000, 0.01, time.time() + 30)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
    
    # Phase 3: Local pair search
    if time.time() < deadline - 5:
        hd = [0.01, 0.005, 0.002, 0.001, 0.0005]
        best_h, best_c = pair_search_local(best_h, n, time.time() + (deadline - time.time()) * 0.5, hd)
    
    # Phase 4: Final SLSQP polish
    if time.time() < deadline - 3:
        ht = slsqp_optimize(best_h, n, 1000000, maxiter=5000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
    
    # Phase 5: Final pair search
    if time.time() < deadline - 1:
        hd = [0.001, 0.0005, 0.0002, 0.0001]
        best_h, best_c = pair_search_local(best_h, n, deadline, hd)
    
    return project_feasible(best_h, n), best_c
```

```python
def cutting_plane(h0, n, max_iter=200, tol=1e-10):
    """Outer approximation (cutting plane) for min max c_k(h)"""
    from scipy.optimize import linprog
    h = h0.copy()
    for it in range(max_iter):
        corr = get_correlations(h, n)
        # Linearize each c_k around h
        # c_k(h') ≈ c_k(h) + sum_i dc[k,i] * (h'[i] - h[i])
        # dc[k,i] = (1-h[i-k])*[valid] - h[i+k]*[valid]
        # This is the same as what we computed for the gradient, but for each k separately
        # Actually, dc[k,i] = ∂c_k/∂h[i]
        # Let me compute the full Jacobian J[k, i] = ∂c_k/∂h[i]
        # J[k, i] = (1-h[i-lag_k])*[0<=i-lag_k<n] - h[i+lag_k]*[0<=i+lag_k<n]
        # where lag_k = (k - (n-1))
        
        # Build the LP:
        # min t [i.e., minimize [0,...,0,1] · [h', t]]
        # s.t. for each k: sum_i J[k,i] * h'[i] - t ≤ corr[k] - sum_i J[k,i]*h[i]
        #      sum h'[i] = n/2
        #      0 ≤ h'[i] ≤ 1
        #      t is free (or t ≥ 0)
        
        # Variables: [h'(0), ..., h'(n-1), t], total n+1 variables
        # Objective: minimize t → c_obj = [0,...,0,1]
        
        # Inequality constraints: A_ineq @ x ≤ b_ineq
        # For each k: sum_i J[k,i]*h'[i] - t ≤ corr[k] - sum_i J[k,i]*h[i]
        # i.e., J[k,:] @ h' - t ≤ corr[k] - J[k,:] @ h
        # i.e., J[k,:] @ h' + [-1]*t ≤ corr[k] - J[k,:] @ h
        
        # Equality: sum h' = n/2 → [1,...,1, 0] @ x = n/2
        # Bounds: 0 ≤ h'[i] ≤ 1, t ≥ 0 (or unbounded below... but t should be positive)
        
        nf = 2 * n - 1
        lags = np.arange(nf) - (n - 1)
        
        # Compute J[k, i] for all k, i
        # J[k, i] = (1-h[i-lag_k])*[0<=i-lag_k<n] - h[i+lag_k]*[0<=i+lag_k<n]
        J = np.zeros((nf, n))
        for k_idx in range(nf):
            k = lags[k_idx]
            for i in range(n):
                t1 = i - k
                t2 = i + k
                val = 0.0
                if 0 <= t1 < n:
                    val += (1.0 - h[t1])
                if 0 <= t2 < n:
                    val -= h[t2]
                J[k_idx, i] = val
        
        # RHS: corr[k] - J[k,:] @ h
        rhs = corr - J @ h
        
        # Build constraint matrix
        A_ineq = np.zeros((nf, n + 1))
        A_ineq[:, :n] = J
        A_ineq[:, n] = -1.0  # coefficient of t
        
        b_ineq = rhs
        
        # Equality constraint
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = [n / 2.0]
        
        # Objective
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0  # minimize t
        
        # Bounds: h' in [0,1], t in [0, inf)
        bounds = [(0, 1)] * n + [(0, None)]
        
        res = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq,
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        
        if not res.success:
            break
        
        h_new = res.x[:n]
        t_new = res.x[n]
        
        # Check convergence
        if abs(compute_c5(h_new, n) - compute_c5(h, n)) < tol:
            h = h_new
            break
        
        h = h_new
    
    return h, compute_c5(h, n)
```
