```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_F(h, n):
    dm = np.arange(2 * n - 1) - (n - 1)
    F = np.zeros((n, 2 * n - 1))
    for m_idx in range(2 * n - 1):
        d = dm[m_idx]
        i_lo = max(0, d)
        i_hi = min(n, n + d)
        if i_lo < i_hi:
            F[i_lo:i_hi, m_idx] += (1.0 - h[i_lo - d:i_hi - d])
        j_lo = max(0, -d)
        j_hi = min(n, n - d)
        if j_lo < j_hi:
            F[j_lo:j_hi, m_idx] -= h[j_lo + d:j_hi + d]
    return F

def update_F(F, i, j, delta, n, dm):
    m_idx = np.arange(2 * n - 1)
    v1 = (i + dm >= 0) & (i + dm < n)
    F[i + dm[v1], m_idx[v1]] -= delta
    v2 = (i - dm >= 0) & (i - dm < n)
    F[i - dm[v2], m_idx[v2]] -= delta
    v3 = (j + dm >= 0) & (j + dm < n)
    F[j + dm[v3], m_idx[v3]] += delta
    v4 = (j - dm >= 0) & (j - dm < n)
    F[j - dm[v4], m_idx[v4]] += delta

def pair_descent(h, n, deadline, max_sweeps=500, K=8, random_pairs=False, rng=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dm = np.arange(2 * n - 1) - (n - 1)
    m_idx = np.arange(2 * n - 1)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        F = compute_F(h, n)
        g = compute_g(h, n)
        current_max = g.max()
        top_idx = np.argsort(g)[::-1][:K]
        improved = 0
        if random_pairs and rng is not None:
            pairs = [(rng.randint(n), rng.randint(n)) for _ in range(n * n // 4)]
            # filter i != j
            pairs = [(i,j) for i,j in pairs if i != j]
        else:
            pairs = [(i, j) for i in range(n) for j in range(i+1, n)]
        for i, j in pairs:
            if time.time() > deadline:
                break
            s = F[i] - F[j]
            # Quick skip: if the max possible change is tiny
            s_max = np.abs(s).max()
            if s_max < 1e-12:
                continue
            d_lo = max(-h[i], h[j] - 1.0)
            d_hi = min(1.0 - h[i], h[j])
            if d_hi - d_lo < 1e-14:
                continue
            candidates = [d_lo, d_hi]
            for m1 in range(K):
                for m2 in range(K):
                    if top_idx[m1] == top_idx[m2]:
                        continue
                    ds = s[top_idx[m2]] - s[top_idx[m1]]
                    if abs(ds) > 1e-15:
                        dc = (g[top_idx[m1]] - g[top_idx[m2]]) / ds
                        if d_lo < dc < d_hi:
                            candidates.append(dc)
            best_delta = 0.0
            best_val = current_max
            for d in candidates:
                d = float(np.clip(d, d_lo, d_hi))
                val = float(np.max(g + d * s))
                if val < best_val - 1e-13:
                    best_val = val
                    best_delta = d
            if best_delta != 0.0:
                h[i] += best_delta
                h[j] -= best_delta
                g += best_delta * s
                update_F(F, i, j, best_delta, n, dm)
                current_max = g.max()
                top_idx = np.argsort(g)[::-1][:K]
                improved += 1
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if improved == 0:
            stall += 1
            if stall > 3:
                break
        else:
            stall = 0
    return best_h, best_c

def slsqp_optimize(h, n, alpha, deadline, maxiter=200):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_polish(h, n, deadline):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    ones = np.where(h_bin == 1)[0].tolist()
    zeros = np.where(h_bin == 0)[0].tolist()
    no_improve = 0
    while time.time() < deadline and no_improve < 10:
        improved = False
        # 1-swap: move a 1 to a 0 position
        for z in zeros[:]:
            for o in ones[:]:
                h_bin[z] = 1.0
                h_bin[o] = 0.0
                c = compute_c5(h_bin, n)
                if c < best_c - 1e-14:
                    best_c = c
                    ones.remove(o)
                    ones.append(z)
                    zeros.remove(z)
                    zeros.append(o)
                    improved = True
                    no_improve = 0
                else:
                    h_bin[z] = 0.0
                    h_bin[o] = 1.0
                if improved:
                    break
            if improved:
                break
        # 2-swap: swap two 1s with two 0s
        if not improved:
            no_improve += 1
            for z1 in zeros[:]:
                for z2 in zeros[:]:
                    if z1 >= z2:
                        continue
                    for o1 in ones[:]:
                        for o2 in ones[:]:
                            if o1 >= o2:
                                continue
                            h_bin[z1] = 1.0
                            h_bin[z2] = 1.0
                            h_bin[o1] = 0.0
                            h_bin[o2] = 0.0
                            c = compute_c5(h_bin, n)
                            if c < best_c - 1e-14:
                                best_c = c
                                ones = [x for x in ones if x not in (o1, o2)]
                                ones.extend([z1, z2])
                                zeros = [x for x in zeros if x not in (z1, z2)]
                                zeros.extend([o1, o2])
                                improved = True
                                no_improve = 0
                                break
                            else:
                                h_bin[z1] = 0.0
                                h_bin[z2] = 0.0
                                h_bin[o1] = 1.0
                                h_bin[o2] = 1.0
                        if improved:
                            break
                    if improved:
                        break
                if improved:
                    break
        if not improved:
            break
    return h_bin, best_c
```

```python
def slsqp_optimize(h, n, alpha, deadline, maxiter=200):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_polish(h, n, deadline, max_rounds=200):

        neg = d < -1e-14
        if pos.any():
            t_max = min(t_max, ((1.0 - h[pos]) / d[pos]).min())
        if neg.any():
            t_max = min(t_max, (-h[neg] / d[neg]).min())
        if t_max < 1e-14:
            break
        delta_g = F @ d
        ts = np.linspace(0.0, min(t_max, 2.0), 150)
        vals = np.array([np.max(g + tv * delta_g) for tv in ts])
        idx_min = np.argmin(vals)
        if vals[idx_min] >= g_max - 1e-13:
            stall += 1
            tol *= 0.6
            if stall > 30 or tol < 1e-14:
                break
            continue
        best_t = ts[idx_min]
        lo = ts[max(0, idx_min - 1)]
        hi = ts[min(len(ts) - 1, idx_min + 1)]
        ts2 = np.linspace(lo, hi, 200)
        vals2 = np.array([np.max(g + tv * delta_g) for tv in ts2])
        idx2 = np.argmin(vals2)
        if vals2[idx2] < vals[idx_min]:
            best_t = ts2[idx2]
        h_new = h + best_t * d
        h_new = project_feasible(h_new, n)
        g_new = compute_g(h_new, n)
        c_new = float(g_new.max())
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            g = g_new
            F = compute_F(h, n)
            stall = 0
        else:
            stall += 1
            if stall > 40:
                break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 5
    dl = t0 + total
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    # Phase 1: SLSQP warmup
    p1 = t0 + 0.08 * total
    h = best_h.copy()
    for alpha in [1000, 10000, 100000, 1000000]:
        if time.time() > p1:
            break
        h = slsqp_optimize(h, n, alpha, p1, maxiter=80)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    # Phase 2: Subgradient method
    p2 = t0 + 0.30 * total
    h, c = subgradient_opt(best_h.copy(), n, p2, max_iter=6000)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    # Phase 3: Pair descent
    p3 = t0 + 0.65 * total
    h, c = pair_descent(best_h.copy(), n, p3, K=10, max_sweeps=400, reheat_every=12)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    # Phase 4: SLSQP high-alpha
    p4 = t0 + 0.78 * total
    if time.time() < p4:
        h = slsqp_optimize(best_h.copy(), n, 10000000, p4, maxiter=200)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    # Phase 5: Binary polish
    p5 = t0 + 0.85 * total
    if time.time() < p5 - 5:
        hb, cb = binary_polish(best_h.copy(), n, p5 - 5, max_rounds=150)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    # Phase 6: Final subgradient
    p6 = dl - 20
    if time.time() < p6:
        h, c = subgradient_opt(best_h.copy(), n, p6, max_iter=4000)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    # Phase 7: Final pair descent
    if time.time() < dl - 3:
        h, c = pair_descent(best_h.copy(), n, dl - 2, K=15, max_sweeps=200, reheat_every=20)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
