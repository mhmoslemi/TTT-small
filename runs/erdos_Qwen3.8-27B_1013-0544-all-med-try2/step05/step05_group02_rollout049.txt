def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def get_corr_full(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            if room.sum() < 1e-14:
                break
            h += (diff / room.sum()) * room
        else:
            if h.sum() < 1e-14:
                break
            h += (diff / h.sum()) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    """Gradient of correlation at a specific lag (0-indexed in full mode)."""
    s = lag - (n - 1)  # actual shift
    g = np.zeros(n)
    if s >= 0:
        # c[lag] = sum_{n=0}^{n-1-s} h[n+s]*(1-h[n])
        valid = np.arange(0, n - s)
        g[valid] += (1.0 - h[valid])  # from (1-h[n]) term, contributes to h[n]
        g[valid + s] -= h[valid]  # from h[n+s] term
    else:
        # c[lag] = sum_{n=-s}^{n-1} h[n+s]*(1-h[n])
        valid = np.arange(-s, n)
        g[valid] += (1.0 - h[valid])
        g[valid + s] -= h[valid]
    return g

def soft_max_objective(h, n, alpha):
    """Compute soft-max approximation of max correlation and its gradient."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    # soft max = m + log(sum(exp(shifted)))/alpha
    # Use log-sum-exp trick
    mx = shifted.max()
    logsumexp = mx + np.log(np.sum(np.exp(shifted - mx)))
    val = m + logsumexp / alpha
    return val

def slsqp_refine(h, n, alpha, maxiter=1000):
    def _scon(x):
        return float(np.sum(x) - n / 2.0)
    def _scon_jac(x):
        return np.ones(len(x))
    
    def obj(x):
        hx = np.clip(x, 0, 1)
        dx = 2.0 / n
        c = np.correlate(hx, 1.0 - hx, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        lse = mx + np.log(np.sum(np.exp(shifted - mx)))
        return m + lse / alpha
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _scon, 'jac': _scon_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except:
        return h.copy()

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def get_corr_full(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            if room.sum() < 1e-14:
                break
            h += (diff / room.sum()) * room
        else:
            if h.sum() < 1e-14:
                break
            h += (diff / h.sum()) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    s = lag - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        idx1 = np.arange(s, n)
        g[idx1] += (1.0 - h[idx1 - s])
        idx2 = np.arange(0, n - s)
        g[idx2] -= h[idx2 + s]
    else:
        sp = -s
        idx1 = np.arange(0, n - sp)
        g[idx1] += (1.0 - h[idx1 + sp])
        idx2 = np.arange(sp, n)
        g[idx2] -= h[idx2 - sp]
    return g

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(len(x))

def slsqp_refine(h, n, alpha, maxiter=1500):
    def obj(x):
        hx = np.clip(x, 0, 1)
        dx = 2.0 / n
        c = np.correlate(hx, 1.0 - hx, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        lse = mx + np.log(np.sum(np.exp(shifted - mx)))
        return m + lse / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def directed_polish(h, n, deadline, top_k=3, T_init=0.005):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    rng = np.random.RandomState(999)
    base_steps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    step_idx = 0
    stall = 0
    max_stall = 80
    T = T_init
    c = get_corr_full(h, n)
    while time.time() < deadline:
        curr_max = float(c.max())
        top_indices = np.argsort(c)[-top_k:]
        g = np.zeros(n)
        for k in top_indices:
            g += lag_gradient(h, n, int(k))
        if g.max() - g.min() < 1e-14:
            break
        i_src = int(np.argmax(g))
        i_dst = int(np.argmin(g))
        if i_src == i_dst:
            break
        d = base_steps[step_idx]
        d = min(d, h[i_src] * 0.99, (1.0 - h[i_dst]) * 0.99)
        if d < 1e-12:
            step_idx = min(step_idx + 1, len(base_steps) - 1)
            stall += 1
            if stall > max_stall:
                break
            continue
        h[i_src] -= d
        h[i_dst] += d
        new_c = get_corr_full(h, n)
        new_max = float(new_c.max())
        delta = new_max - curr_max
        if delta < 0 or rng.random() < np.exp(-max(delta, 0.0) / max(T, 1e-10)):
            if new_max < best_c - 1e-15:
                best_c = new_max
                best_h = h.copy()
            c = new_c
            stall = 0
            step_idx = max(0, step_idx - 1)
            T = max(T * 0.997, 1e-7)
        else:
            h[i_src] += d
            h[i_dst] -= d
            stall += 1
            step_idx = min(step_idx + 1, len(base_steps) - 1)
            if stall > max_stall:
                break
    return best_h, best_c

def iterated_local_search(h, n, deadline, num_restarts, rng):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for r in range(num_restarts):
        if time.time() > deadline - 10:

        scale = 0.03 + 0.07 * (r / max(1, num_restarts - 1))

    return best_h, best_c
