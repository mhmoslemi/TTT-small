    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:

            s = room.sum()
            if s < 1e-14:

            if s < 1e-14:

    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def analytic_grad(h, n, alpha):
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    m = g.max()
    sh = alpha * (g - m)
    sh -= sh.max()
    w = np.exp(sh)
    wsum = w.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    w /= wsum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - n + 1
        if s >= 0:
            if n - s > 0:
                grad[0:n-s] += wk * (1.0 - h[s:n])
            if s < n:
                grad[s:n] -= wk * h[0:n-s]
        else:
            ss = -s
            grad[ss:n] += wk * (1.0 - h[0:n-ss])
            if n - ss > 0:
                grad[0:n-ss] -= wk * h[ss:n]
    return grad * dx

def projected_grad(h, n, deadline, alpha=200000, max_iter=50000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.5
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = analytic_grad(h, n, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        step = lr / gn
        found = False
        for _ in range(30):
            h_new = project_feasible(h - step * g, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                h = h_new
                found = True
                break
            step *= 0.5
        if not found:
            lr *= 0.85
            stall += 1
            if lr < 1e-14 or stall > 100:
                break
        else:
            stall = 0
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
            lr = min(lr * 1.03, 30.0)
    return best_h, best_c

def slsqp_step(h, n, alpha, maxiter, deadline):
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def con_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def sa_binary(hb, n, deadline, seed, T_init=0.04, T_min=0.0005, cooling=0.9998):
    """Simulated annealing on binary representation."""
    rng = np.random.RandomState(seed)
    best_hb = hb.copy()
    best_c = compute_c5(hb, n)
    cur_c = best_c
    T = T_init
    ones = np.where(hb == 1)[0].tolist()
    zeros = np.where(hb == 0)[0].tolist()
    no_improve = 0
    max_no_improve = 50000
    
    while time.time() < deadline and T > T_min:
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        hb[i], hb[j] = 1.0, 0.0
        new_c = compute_c5(hb, n)
        delta = new_c - cur_c
        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
            cur_c = new_c
            ones.remove(j)
            ones.append(i)
            zeros.remove(i)
            zeros.append(j)
            if cur_c < best_c:
                best_c = cur_c
                best_hb = hb.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            hb[i], hb[j] = 0.0, 1.0
            no_improve += 1
        T *= cooling
        if no_improve > max_no_improve:
            no_improve = 0
            if not (time.time() < deadline):
                break
    
    return best_hb, best_c

def greedy_binary(hb, n, deadline, max_sweeps=200):
    """Greedy local search on binary."""
    best_c = compute_c5(hb, n)
    ones = np.where(hb == 1)[0].tolist()
    zeros = np.where(hb == 0)[0].tolist()
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for idx_i in range(len(zeros)):
            i = zeros[idx_i]
            best_j = -1
            best_cj = best_c
            for idx_j in range(len(ones)):
                j = ones[idx_j]
                hb[i], hb[j] = 1.0, 0.0
                c = compute_c5(hb, n)
                if c < best_cj:
                    best_cj = c
                    best_j = j
                hb[i], hb[j] = 0.0, 1.0
            if best_j >= 0:
                i = zeros[idx_i]
                hb[i], hb[best_j] = 1.0, 0.0
                best_c = best_cj
                ones.remove(best_j)
                ones.append(i)
                zeros.remove(i)
                improved = True
                break
        if not improved:
            break
    return hb, best_c

def two_swap_search(hb, n, deadline, max_rounds=50):
    """Try 2-swap moves to escape local optima."""
    best_hb = hb.copy()
    best_c = compute_c5(hb, n)
    ones = np.where(hb == 1)[0].tolist()
    zeros = np.where(hb == 0)[0].tolist()
    rng = np.random.RandomState(123)
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        # Try a random 2-swap
        i1, i2 = rng.choice(zeros, size=2, replace=False)
        j1, j2 = rng.choice(ones, size=2, replace=False)
        hb[i1] = 1; hb[i2] = 1; hb[j1] = 0; hb[j2] = 0
        c = compute_c5(hb, n)
        if c < best_c:
            best_c = c
            best_hb = hb.copy()
            ones = np.where(hb == 1)[0].tolist()
            zeros = np.where(hb == 0)[0].tolist()
        else:
            hb[i1] = 0; hb[i2] = 0; hb[j1] = 1; hb[j2] = 1
    return best_hb, best_c
