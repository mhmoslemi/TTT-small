import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def local_search(hb, n, deadline, max_sweeps=10000):
    best_c = compute_c5(hb, n)
    one_idx = np.where(hb == 1)[0].copy()
    zero_idx = np.where(hb == 0)[0].copy()
    n_ones = len(one_idx)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for zi in range(n_ones):
            if time.time() > deadline:
                break
            i = zero_idx[zi]
            best_j_pos = -1
            best_ct = best_c
            for oi in range(n_ones):
                j = one_idx[oi]
                hb[i] = 1.0
                hb[j] = 0.0
                c = compute_c5(hb, n)
                if c < best_ct:
                    best_ct = c
                    best_j_pos = oi
                else:
                    hb[i] = 0.0
                    hb[j] = 1.0
            if best_j_pos >= 0:
                j = one_idx[best_j_pos]
                hb[i] = 1.0
                hb[j] = 0.0
                best_c = best_ct
                one_idx[best_j_pos] = i
                zero_idx[zi] = j
                improved = True
        if not improved:
            break
    return hb, best_c
