    mask_a = (i_mk >= 0) & (i_mk < n)
    a = np.zeros(len(k_vals))
    a[mask_a] = 1.0 - h[i_mk[mask_a]]
    i_pk = i + k_vals
    mask_b = (i_pk >= 0) & (i_pk < n)
    b = np.zeros(len(k_vals))
    b[mask_b] = h[i_pk[mask_b]]
    coef = a - b
    base = S - h[i] * a - (1.0 - h[i]) * b
    lo, hi = 0.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        vals = base + coef * mid
        max_val = vals.max()
        active = vals >= max_val - 1e-14
        if coef[active].min() > 1e-14:
            hi = mid
        elif coef[active].max() < -1e-14:
            lo = mid
        else:
            break
    return max(0.0, min(1.0, (lo + hi) * 0.5))


def coordinate_descent(h, max_sweeps=200, tol=1e-10, random_order=False, rng=None):
    n = len(h)
    target_sum = n / 2.0
    for sweep in range(max_sweeps):
        max_change = 0.0
        order = np.random.permutation(n) if (random_order and rng is not None) else np.arange(n)
        for i in order:
            t_opt = find_optimal_coord(h, i, n)
            change = abs(t_opt - h[i])
            if change > max_change:
                max_change = change
            h[i] = t_opt
        h = project_h(h, target_sum)
        if max_change < tol:
            break
    return h


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_points = len(h_init)
    else:
        n_points = 70
        h_init = np.full(n_points, 0.5)

    target_sum = n_points / 2.0
    best_h = project_h(h_init.copy(), target_sum)
    best_c5 = compute_c5(best_h)

    # Phase 1: Thorough coordinate descent
    h_cd = coordinate_descent(best_h.copy(), max_sweeps=400, tol=1e-12)
    h_cd = project_h(h_cd, target_sum)
    c5 = compute_c5(h_cd)
    if c5 < best_c5:
        best_h, best_c5 = h_cd.copy(), c5

    # Phase 2: Diverse restarts + coordinate descent
    attempt = 0
    while time_left() > 25:
        attempt += 1
        h0 = best_h.copy()
        strat = attempt % 8

        if strat == 0:
            scale = 0.02 + 0.07 * np.exp(-attempt / 35.0)
            h0 += rng.randn(n_points) * scale
        elif strat == 1:
            blen = rng.randint(4, max(5, n_points // 4))
            bs = rng.randint(0, n_points - blen)
            h0[bs:bs + blen] = h0[bs:bs + blen][::-1]
        elif strat == 2:
            blen = rng.randint(5, max(6, n_points // 3))
            bs = rng.randint(0, n_points - blen)
            h0[bs:bs + blen] = rng.random(blen)
        elif strat == 3:
            mask = rng.random(n_points) < 0.25
            h0[mask] = rng.random(int(mask.sum()))
        elif strat == 4:
            corr = np.correlate(h0, 1.0 - h0, mode="full")
            top_lags = np.argsort(corr)[-5:]
            for lag_idx in top_lags:
                lag = lag_idx - (n_points - 1)
                for j in range(n_points):
                    i1 = j - lag
                    if 0 <= i1 < n_points:
                        h0[j] -= 0.03 * (1.0 - h0[i1])
                    i2 = j + lag
                    if 0 <= i2 < n_points:
                        h0[j] -= 0.03 * h0[i2]
        elif strat == 5:
            temp = 0.15 + 0.35 * np.exp(-attempt / 25.0)
            h0 = (1 - temp) * h0 + temp * rng.random(n_points)
        elif strat == 6:
            n_swaps = rng.randint(3, 8)
            for _ in range(n_swaps):
                i, j = rng.choice(n_points, 2, replace=False)
                h0[i], h0[j] = h0[j], h0[i]
            h0 += rng.randn(n_points) * 0.015
        else:
            # Multi-scale: perturb at different block sizes
            for bl in [3, 7, 15]:
                if bl < n_points:
                    bs = rng.randint(0, n_points - bl)
                    h0[bs:bs + bl] += rng.randn(bl) * 0.04

        h0 = project_h(h0, target_sum)
        h_new = coordinate_descent(h0, max_sweeps=150, tol=1e-11, random_order=True, rng=rng)
        h_new = project_h(h_new, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h, best_c5 = h_new.copy(), c5

    # Phase 3: Try different n_points by interpolation
    if time_left() > 50:
        for n_try in [55, 60, 65, 75, 80, 85, 90, 100, 110, 120]:
            if time_left() < 20:
                break
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_try = np.interp(x_new, x_old, best_h)
            h_try = project_h(h_try, n_try / 2.0)
            h_try = coordinate_descent(h_try, max_sweeps=250, tol=1e-11)
            h_try = project_h(h_try, n_try / 2.0)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_h, best_c5 = h_try.copy(), c5_try

    # Phase 4: SLSQP polish on best
    if time_left() > 20:
        try:
            n = len(best_h)
            dx = 2.0 / n
            bounds = [(0.0, 1.0)] * n

            def obj_slsqp(h):
                corr = np.correlate(h, 1.0 - h, mode="full")
                top5 = np.sort(corr)[-5:]
                return (top5.mean() + 0.5 * (top5.max() - top5.mean())) * dx

            def con_eq(h):
                return h.sum() - n / 2.0

            res = minimize(obj_slsqp, best_h, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': con_eq,
                                         'jac': lambda x: np.ones(len(x))}],
                           options={'maxiter': 3000, 'ftol': 1e-15})
            h_test = project_h(res.x, n / 2.0)
            c5_test = compute_c5(h_test)
            if c5_test < best_c5:
                best_h, best_c5 = h_test.copy(), c5_test
        except Exception:
            pass

    # Phase 5: Final ultra-fine coordinate descent
    if time_left() > 5:
        h_final = coordinate_descent(best_h.copy(), max_sweeps=1000, tol=1e-13,
                                     random_order=True, rng=rng)
        h_final = project_h(h_final, len(best_h) / 2.0)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_h, best_c5 = h_final, c5_final

    return best_h, best_c5, len(best_h)
