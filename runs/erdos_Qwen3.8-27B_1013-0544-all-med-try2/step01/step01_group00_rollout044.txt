import time

    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def lse_value(h, T):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return m + T * np.log(np.sum(np.exp((corr - m) / T)))

def lse_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    corr = corr_raw * dx
    m = np.max(corr)
    exp_vals = np.exp((corr - m) / T)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        k = j - (n - 1)
        w = weights[j]
        if abs(w) < 1e-11:
            continue

        i_end = min(n, n - k)
        if i_end > i_start:

            grad[idx] += w * dx * (1.0 - h[idx + k])
        # -h[i-k] for valid i
        i_start2 = max(0, k)
        i_end2 = min(n, n + k)
        if i_end2 > i_start2:
            idx2 = np.arange(i_start2, i_end2)
            grad[idx2] -= w * dx * h[idx2 - k]

import numpy as np

import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def lse_value(h, T):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return m + T * np.log(np.sum(np.exp((corr - m) / T)))


def lse_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    corr = corr_raw * dx
    m = np.max(corr)
    exp_vals = np.exp((corr - m) / T)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        k = j - (n - 1)
        w = weights[j]
        if abs(w) < 1e-11:
            continue
        i_start = max(0, -k)
        i_end = min(n, n - k)
        if i_end > i_start:
            idx = np.arange(i_start, i_end)
            grad[idx] += w * dx * (1.0 - h[idx + k])
        i_start2 = max(0, k)
        i_end2 = min(n, n + k)
        if i_end2 > i_start2:
            idx2 = np.arange(i_start2, i_end2)
            grad[idx2] -= w * dx * h[idx2 - k]
    return grad


def project_set(h, S):

        else:

    return np.sum(v[:n]) - n * 0.5


def t_eq_jac(v):

    j = np.zeros(len(v))

    return j


def optimize_slsqp_t(h0, max_iter, ftol):

    c5_cur = compute_c5(h0)
    v0 = np.append(h0.copy(), c5_cur + 0.01)
