At j in [p, n-p): g[j] = (1-h[j+p]) - h[j-p]
At j in [0, p): g[j] = (1-h[j+p])
At j in [n-p, n): g[j] = -h[j-p]

This matches my derivation for k = -p. ✓

import numpy as np
import time

def compute_c5(h):

    return np.max(corr) * (2.0 / len(h))

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def grad_corr_single(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n - k])
        if k > 0:
            g[:n - k] -= h[k:]
    else:
        p = -k
        g[:n - p] += (1.0 - h[p:])
        if p > 0:
            g[p:] -= h[:n - p]
    return g

def compute_weighted_grad(h, alpha, top_k=20):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m_max = np.max(corr)
    w = np.exp(alpha * (corr - m_max))
    sw = np.sum(w)
    if sw < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= sw
        if top_k < len(w):
            top_idx = np.argsort(w)[-top_k:]
            w2 = np.zeros_like(w)
            w2[top_idx] = w[top_idx]
            w2 /= np.sum(w2)
            w = w2
    grad = np.zeros(n)
    for m in range(len(w)):
        if w[m] > 1e-10:
            k = m - (n - 1)
            grad += w[m] * grad_corr_single(h, k)
    return grad * (2.0 / n), corr

def binarize_push(h, target, strength):
    mid = np.median(h[h > 0.01])
    h2 = h.copy()
    h2[h2 < mid] = h2[h2 < mid] * (1.0 - strength)
    h2[h2 >= mid] = h2[h2 >= mid] + (1.0 - h2[h2 >= mid]) * strength
    return project_box_sum(h2, target)

    lr = lr_init

    stall = 0

    while time.time() < deadline:
