import numpy as np
from scipy.optimize import minimize
import time

_g_dx = None
_g_nhalf = None
_g_target = None

def project_feasible(x, target_sum):
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_grad_lag(h, L):
    n = len(h)
    omh = 1.0 - h
    grad = np.zeros(n)
    if L == 0:
        grad[:] = omh[:] - h[:]
    elif L > 0:
        grad[L:] += omh[:n - L]
        grad[:n - L] -= h[L:]
    else:
        L2 = -L
        grad[:n - L2] += omh[L2:]
        grad[L2:] -= h[:n - L2]
    return grad

def slsqp_obj(x):
    return compute_c5(x, _g_dx)

def slsqp_jac(x):
    n = len(x)
    n_half = _g_nhalf
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    tol = max(0.05, c_max * 0.002)
    active = np.where(corr >= c_max - tol)[0]
    if len(active) < 1:
        active = np.array([int(np.argmax(corr))])
    if len(active) > 20:
        active = np.argsort(corr)[-20:]
    grad = np.zeros(n)
    for idx in active:
        L = int(idx) - n_half
        grad += compute_grad_lag(x, L)
    grad /= float(len(active))
    return grad

def eq_fun(x):
    return np.sum(x) - _g_target

def eq_jac(x):
    return np.ones(len(x))

def slsqp_optimize(h_start, n_points, maxiter=3000):
    res = minimize(slsqp_obj, h_start, jac=slsqp_jac, method='SLSQP',
                  bounds=[(0.0, 1.0)] * n_points,
                  constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],
                  options={'maxiter': maxiter, 'ftol': 1e-14})
    if res.x is not None:
        return project_feasible(res.x, _g_target)
    return h_start

def projected_gradient(h, n_points, dx, max_iters=3000, time_check=None):
    global _g_dx, _g_nhalf, _g_target
    _g_dx = dx
    _g_nhalf = n_points - 1
    _g_target = n_points / 2.0
    step_size = 0.003
    momentum = np.zeros(n_points)
    mu = 0.65
    best = compute_c5(h, dx)
    h_cur = h.copy()
    for _ in range(max_iters):
        if time_check is not None and time_check() < 10.0:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        c_max = float(np.max(corr))
        tol = max(0.01, c_max * 0.0008)
        active = np.where(corr >= c_max - tol)[0]
        if len(active) < 2:
            active = np.argsort(corr)[-4:]
        elif len(active) > 15:
            active = np.argsort(corr)[-15:]
        grad = np.zeros(n_points)
        for idx in active:
            L = int(idx) - _g_nhalf
            grad += compute_grad_lag(h_cur, L)
        grad /= float(len(active))

        h_new = project_feasible(h_cur + momentum, _g_target)

            h_cur = h_new
        else:

            h_cur = h_new
