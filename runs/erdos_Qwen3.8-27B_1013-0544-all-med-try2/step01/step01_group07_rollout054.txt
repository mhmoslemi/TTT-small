    """Compute value and gradient of soft-max objective"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")

    shifted = t * (corr - max_c)
    exp_shifted = np.exp(shifted)

        t = 40.0 * (1.0 - frac) + 3.0
        val, grad = compute_soft_max_grad(h, t, dx)
        vel = momentum * vel - step_size * grad
        h_new = project_feasible(h + vel, target_sum)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            step_size = min(0.03, step_size * 1.02)
        else:
            step_size = max(0.001, step_size * 0.93)
            vel *= 0.5
        h = h_new
        momentum = 0.85 * (1.0 - 0.3 * frac)

    # Phase 2: Local search - pair/triple coord descent + perturbations
    rng = np.random.RandomState(seed + 777)
    p2_end = t0 + 750
    no_improve = 0
    while elapsed() < p2_end - t0:
        r = rng.random()
        if r < 0.55:
            i1, i2 = rng.choice(n, 2, replace=False)
            h = pair_coord_descent(h, i1, i2, dx)
            c5 = compute_c5(h, dx)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
                no_improve = 0
            else:
                no_improve += 1
        elif r < 0.75:
            i1, i2, i3 = rng.choice(n, 3, replace=False)
            h = triple_coord_descent(h, i1, i2, i3, dx)
            c5 = compute_c5(h, dx)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
                no_improve = 0
            else:
                no_improve += 1
        else:
            n_pert = rng.randint(2, 6)
            idx = rng.choice(n, n_pert, replace=False)
            h_pert = h.copy()
            h_pert[idx] += rng.uniform(-0.25, 0.25, n_pert)
            h_pert = project_feasible(h_pert, target_sum)
            c5_pert = compute_c5(h_pert, dx)
            c5_cur = compute_c5(h, dx)
            temp = max(0.0005, 0.005 * np.exp(-no_improve / 300.0))
            delta = c5_pert - c5_cur
            if delta < 0 or rng.random() < np.exp(-delta / temp):
                h = h_pert
                if c5_pert < best_c5:
                    best_h = h.copy()
                    best_c5 = c5_pert
                    no_improve = 0

    # Phase 3: Try n=320
    if elapsed() < budget_s - 180:
        n2 = 320
        dx2 = 2.0 / n2
        ts2 = n2 / 2.0
        x_old = np.linspace(0, 2, n + 1)[:-1]
        x_new = np.linspace(0, 2, n2 + 1)[:-1]
        h2 = np.interp(x_new, x_old, best_h)
        h2 = project_feasible(h2, ts2)
        c5_2 = compute_c5(h2, dx2)
        if c5_2 < best_c5:
            best_h = h2.copy()
            best_c5 = c5_2
            best_n = n2
        vel2 = np.zeros(n2)
        ss2 = 0.005
        p3_end = min(t0 + budget_s - 120, t0 + 920)
        it2 = 0
        while elapsed() < p3_end - t0:
            it2 += 1
            t2 = max(8.0, 40.0 * np.exp(-it2 / 1500.0))
            val2, grad2 = compute_soft_max_grad(h2, t2, dx2)
            vel2 = 0.83 * vel2 - ss2 * grad2
            h2_new = project_feasible(h2 + vel2, ts2)
            c5_2 = compute_c5(h2_new, dx2)
            if c5_2 < best_c5:
                best_h = h2_new.copy()
                best_c5 = c5_2
                best_n = n2
                ss2 = min(0.02, ss2 * 1.02)
            else:
                ss2 = max(0.001, ss2 * 0.94)
                vel2 *= 0.5
            h2 = h2_new

        # Coord descent on n2
        p3c_end = min(t0 + budget_s - 60, t0 + 1020)
        while elapsed() < p3c_end - t0:
            i1, i2 = rng.choice(n2, 2, replace=False)
            h2 = pair_coord_descent(h2, i1, i2, dx2)
            c5_2 = compute_c5(h2, dx2)
            if c5_2 < best_c5:
                best_h = h2.copy()
                best_c5 = c5_2
                best_n = n2

    # Phase 4: Final polish on best solution
    if elapsed() < budget_s - 20:
        nf = best_n
        dxf = 2.0 / nf
        tsf = nf / 2.0
        hf = best_h.copy()
        velf = np.zeros(nf)
        ssf = 0.003
        p4_end = min(t0 + budget_s - 5, t0 + budget_s - 5)
        itf = 0
        while elapsed() < p4_end - t0:
            itf += 1
            tf = max(2.0, 15.0 * np.exp(-itf / 2000.0))
            valf, gradf = compute_soft_max_grad(hf, tf, dxf)
            velf = 0.8 * velf - ssf * gradf
            hf_new = project_feasible(hf + velf, tsf)
            c5f = compute_c5(hf_new, dxf)
            if c5f < best_c5:
                best_h = hf_new.copy()
                best_c5 = c5f
                best_n = nf
                ssf = min(0.01, ssf * 1.03)
            else:
                ssf = max(0.0005, ssf * 0.95)
                velf *= 0.5
            hf = hf_new

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    final_sum = best_h.sum()
    if abs(final_sum - best_n / 2.0) > 1e-5:
        best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / best_n)
    return best_h, best_c5, best_n
