```python
from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_h(h, target):
    """Project h onto {0<=h<=1, sum(h)=target}."""
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    tol = 1e-12
    while abs(diff) > tol:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
    return h

def coord_descent(h, n_sweeps=100, random_order=False, seed=42):
    """Coordinate descent for the Erdős problem."""
    n = len(h)
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h.copy()
    
    for sweep in range(n_sweeps):
        if random_order:
            order = rng.permutation(n)
        else:
            order = range(n)
        
        for i in order:
            # Compute B_k for all k
            # k ranges from -(n-1) to n-1, stored at index k+(n-1)
            B = np.zeros(2*n-1)
            
            # First factor: h[i] * (1-h[i+k]) for valid k
            # Valid: 0 <= i+k < n → -i <= k <= n-1-i
            k1_min = max(-(n-1), -i)
            k1_max = min(n-1, n-1-i)
            if k1_min <= k1_max:
                ks = np.arange(k1_min, k1_max+1)
                B[ks + (n-1)] += (1.0 - h[i+ks]) * dx
            
            # Second factor: h[i-k] * (1-h[i]) for valid k
            # Valid: 0 <= i-k < n → i-(n-1) <= k <= i
            k2_min = max(-(n-1), i-n+1)
            k2_max = min(n-1, i)
            if k2_min <= k2_max:
                ks = np.arange(k2_min, k2_max+1)
                B[ks + (n-1)] -= h[i-ks] * dx
            
            # Current correlation
            corr = np.correlate(h, 1.0-h, mode='full') * dx
            # A_k = corr_k - h[i]*B_k
            A = corr - h[i] * B
            
            # Minimize max_k (A_k + B_k * t) over t in [0,1]
            res = minimize_scalar(lambda t: np.max(A + B*t), bounds=(0.0, 1.0), method='bounded',
                                   options={'xatol': 1e-8})
            h[i] = res.x
        
        # Project to satisfy sum constraint

    return h, compute_c5(h)
```

```python
def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_h(h, target):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h
```

```python
            # Also k in [-(n-1), n-1]
            k1_lo = max(-(n-1), p - n + 1)
            k1_hi = min(n-1, p)
            if k1_lo <= k1_hi:
                ks = np.arange(k1_lo, k1_hi + 1)
                js = ks + (n - 1)  # index into out
                B[js] += (1.0 - h[p - ks]) * dx
            
            # Second term: -h[p+k] for valid k where 0 <= p+k < n
            # p+k >= 0 → k >= -p; p+k <= n-1 → k <= n-1-p
            # Also k in [-(n-1), n-1]
            k2_lo = max(-(n-1), -p)
            k2_hi = min(n-1, n - 1 - p)
            if k2_lo <= k2_hi:
                ks = np.arange(k2_lo, k2_hi + 1)
                js = ks + (n - 1)
                B[js] -= h[p + ks] * dx
            
            # A[j] = corr[j] - h[p] * B[j]
            A = corr - old_hp * B
            
            # Minimize max(A + B*t) over t in [0, 1]
            res = minimize_scalar(lambda t: np.max(A + B * t), bounds=(0.0, 1.0),
                                  method='bounded', options={'xatol': 1e-10})
            new_hp = res.x
            delta = new_hp - old_hp
            if abs(delta) > 1e-10:
                h[p] = new_hp
                corr += delta * B
        
        # Project to satisfy sum constraint
        h = project_h(h, n * 0.5)
        # Recompute correlation after projection
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
    
    return h, compute_c5(h)
```

```python
def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_h(h, target):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    dx = 2.0 / n

                B[ks + (n-1)] -= h[p + ks] * dx
```
