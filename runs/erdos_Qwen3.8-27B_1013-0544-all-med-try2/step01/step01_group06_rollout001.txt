    grad = np.zeros(n)
    active = np.where(w > 1e-8)[0]
    for idx in active:
        k = idx - (n - 1)
        grad += w[idx] * grad_corr_single(h, k)
    return grad * (2.0 / n)

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_descent_ls(h0, target, deadline, alpha, momentum=0.8):
    best = h0.astype(float).copy()
    best_c5 = compute_c5(best)
    v = np.zeros_like(best)
    lr = 0.04
    stall = 0
    while time.time() < deadline:
        grad = compute_grad_smooth(best, alpha)
        v = momentum * v - lr * grad
        accepted = False
        for _ in range(25):
            trial = project_box_sum(best + v, target)
            c5_t = compute_c5(trial)
            if c5_t < best_c5 - 1e-15:
                best = trial
                best_c5 = c5_t
                accepted = True
                stall = 0
                break
            v *= 0.6
        if not accepted:
            stall += 1
            lr *= 0.75
            v *= 0.4
            if stall > 60 or lr < 1e-8:
                break
        else:
            if lr < 0.12:
                lr = min(lr * 1.02, 0.12)
    return best, best_c5

def sa_escape(h0, target, deadline, rng, n_iters=4000, T_init=0.004):
    best = h0.astype(float).copy()
    best_c5 = compute_c5(best)
    n = len(best)
    curr = best.copy()
    curr_c5 = best_c5
    for i in range(n_iters):
        if time.time() > deadline:
            break
        T = T_init * (0.02 ** (i / max(n_iters - 1, 1)))
        bs = rng.randint(3, max(4, n // 20))
        start = rng.randint(0, max(n - bs, 1))
        bsum = curr[start:start+bs].sum()
        curr[start:start+bs] += rng.randn(bs) * 0.12
        curr[start:start+bs] = project_box_sum(curr[start:start+bs], bsum)
        curr = project_box_sum(curr, target)
        c5_new = compute_c5(curr)
        delta = c5_new - curr_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            curr_c5 = c5_new
            if c5_new < best_c5:
                best = curr.copy()
                best_c5 = c5_new
    return best, best_c5

def resample(h, n_old, n_new):
    x_old = np.arange(n_old) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    return np.interp(x_new, x_old, h, period=1.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.zeros(200)
        h_init[:100] = 1.0
    n0 = len(h_init)
    target0 = n0 / 2.0
    h_init = project_box_sum(h_init, target0)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Optimize at reduced resolution n=80 (lower dim, faster per-iter)
    n_red = 80
    h_red = resample(h_init, n0, n_red)
    h_red = project_box_sum(h_red, n_red / 2.0)
    dl1 = t0 + budget_s * 0.22
    h_red, c5_red = grad_descent_ls(h_red, n_red / 2.0, dl1, alpha=50, momentum=0.75)

    h_up = resample(h_red, n_red, n0)
    h_up = project_box_sum(h_up, target0)
    c5_up = compute_c5(h_up)
    if c5_up < best_c5:
        best_h, best_c5 = h_up, c5_up

    # Phase 2: Direct gradient descent at full resolution
    dl2 = t0 + budget_s * 0.45
    best_h, best_c5 = grad_descent_ls(best_h, target0, dl2, alpha=80, momentum=0.82)

    # Phase 3: SA escape
    dl3 = t0 + budget_s * 0.60
    h_sa, c5_sa = sa_escape(best_h, target0, dl3, rng, n_iters=5000, T_init=0.004)
    if c5_sa < best_c5:
        best_h, best_c5 = h_sa, c5_sa

    # Phase 4: Re-optimize after SA
    dl4 = t0 + budget_s * 0.75
    best_h, best_c5 = grad_descent_ls(best_h, target0, dl4, alpha=150, momentum=0.85)

    # Phase 5: Another SA with lower temperature
    dl5 = t0 + budget_s * 0.88
    h_sa2, c5_sa2 = sa_escape(best_h, target0, dl5, rng, n_iters=4000, T_init=0.002)
    if c5_sa2 < best_c5:
        best_h, best_c5 = h_sa2, c5_sa2

    # Phase 6: Final high-alpha polish
    best_h, best_c5 = grad_descent_ls(best_h, target0, t0 + budget_s - 5, alpha=300, momentum=0.9)

    # Try n=100 path independently
    if time.time() < t0 + budget_s - 5:
        n100 = 100
        h100 = resample(best_h, n0, n100)
        h100 = project_box_sum(h100, 50.0)
        h100, c5_100 = grad_descent_ls(h100, 50.0, t0 + budget_s - 5, alpha=120, momentum=0.85)
        if c5_100 < best_c5:
            best_h = resample(h100, n100, n0)
            best_h = project_box_sum(best_h, target0)
            best_c5 = compute_c5(best_h)

    best_h = project_box_sum(best_h, target0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n0
