import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def project_feasible(h, target):
    lo = -h.max() - 0.5
    hi = 1.0 - h.min() + 0.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)


def lse_obj(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    return m + temp * np.log(np.sum(np.exp((corr - m) / temp)))


def sum_eq(h):
    return float(np.sum(h) - len(h) * 0.5)


def sum_eq_jac(h):
    return np.ones(len(h))


def corr_grad(h, k):
    n = len(h)
    lag = k - (n - 1)
    grad = np.zeros(n)
    i = np.arange(n)
    if lag >= 0:
        mask1 = (i >= lag) & (i <= n - 1)
        grad[mask1] += (1.0 - h[i[mask1] - lag])
        mask2 = (i >= 0) & (i <= n - 1 - lag)
        grad[mask2] += (-h[i[mask2] + lag])
    else:
        la = -lag
        mask1 = (i >= la) & (i <= n - 1)
        grad[mask1] += 1.0
        mask2 = (i >= la) & (i <= n - 1)
        grad[mask2] += (-h[i[mask2] - la])
        mask3 = (i >= 0) & (i <= n - 1 - la)
        grad[mask3] += (-h[i[mask3] + la])
    return grad * (2.0 / n)
