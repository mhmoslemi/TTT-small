import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def smooth_obj(h, alpha, penalty):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.exp(alpha * (corr - m))
    val = (m + np.log(np.sum(shifted)) / alpha) * (2.0 / n)
    sm = np.sum(h) - n / 2.0
    return val + penalty * sm * sm

def smooth_obj_grad(h, alpha, penalty):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    w /= np.sum(w)
    grad = np.zeros(n)

    # c_k = sum_i h[i] * (1 - h[i+k]) for valid i

        else:

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def smooth_obj(h, alpha, penalty):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.exp(alpha * (corr - m))
    val = (m + np.log(np.sum(shifted)) / alpha) * (2.0 / n)
    sm = np.sum(h) - n / 2.0
    return val + penalty * sm * sm

def smooth_obj_grad(h, alpha, penalty):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    ws = np.sum(w)
    if ws < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= ws
    grad = np.zeros(n)
    n_corr = len(corr)
    for idx in range(n_corr):
        wk = w[idx]
        if wk < 1e-14:
            continue
        k = idx - (n - 1)
        if k == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:-k] += wk * (1.0 - h[k:])
            grad[k:] -= wk * h[:-k]
        else:
            p = -k
            grad[:-p] += wk * (1.0 - h[p:])
            grad[p:] -= wk * h[:n - p]
    grad *= (2.0 / n)
    sm = np.sum(h) - n / 2.0
    grad += 2.0 * penalty * sm
    return grad

def project_simplex_box(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def interp_to(h_old, n_new):
    n_old = len(h_old)
    x_old = np.arange(n_old) / n_old
    x_new = np.arange(n_new) / n_new
    return np.interp(x_new, x_old, h_old)

def optimize_lbfgsb(h0, alpha, deadline, penalty=1e5, maxiter=400):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    def obj(x):
        return smooth_obj(x, alpha, penalty)

    def jac(x):
        return smooth_obj_grad(x, alpha, penalty)

    try:
        result = minimize(
            obj, best_h, jac=jac,
            method='L-BFGS-B', bounds=bounds,
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8, 'maxfun': 5000}
        )
    except Exception:
        return best_h, best_c5

    if time.time() > deadline:
        return best_h, best_c5

    h_new = np.clip(result.x, 0.0, 1.0)
    h_new = project_simplex_box(h_new, n / 2.0)
    c5_new = compute_c5(h_new)
    if c5_new < best_c5:
        return h_new, c5_new
    return best_h, best_c5

def targeted_perturb(h, rng, n_trials=20, deadline=None):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(n_trials):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = np.argmax(corr) - (n - 1)
        perturb_idx = np.random.randint(0, n, size=max(3, n // 20))
        h_pert = h.copy()
        h_pert[perturb_idx] += rng.randn(len(perturb_idx)) * 0.05
        h_pert = project_simplex_box(h_pert, n / 2.0)
        c5_pert = compute_c5(h_pert)
        if c5_pert < best_c5:
            best_h = h_pert
            best_c5 = c5_pert
    return best_h, best_c5

    # Phase 2: Targeted perturbation at current resolution
    deadline2 = t0 + budget_s * 0.38

    # Phase 3: Resolution increase to 400
    deadline3 = t0 + budget_s * 0.72
    if time.time() < deadline3 - 30:
        h_interp = interp_to(best_h, 400)
        h_interp = project_simplex_box(h_interp, 200.0)
        for alpha in [40, 100, 250]:
            if time.time() > deadline3:
                break
            h_opt, c5 = optimize_lbfgsb(h_interp, alpha, deadline3, penalty=1e6, maxiter=350)
            if c5 < best_c5:
                best_h, best_c5 = h_opt, c5
                h_interp = best_h.copy()

    # Phase 4: Try resolution 600 if time permits
    deadline4 = t0 + budget_s * 0.90
    if time.time() < deadline4 - 20 and best_c5 < 0.3815:
        h_interp = interp_to(best_h, 600)
        h_interp = project_simplex_box(h_interp, 300.0)
        for alpha in [60, 200]:
            if time.time() > deadline4:
                break
            h_opt, c5 = optimize_lbfgsb(h_interp, alpha, deadline4, penalty=1e6, maxiter=250)
            if c5 < best_c5:
                best_h, best_c5 = h_opt, c5

    # Phase 5: Final high-alpha polish

    # Final perturb

    # Ensure feasibility
