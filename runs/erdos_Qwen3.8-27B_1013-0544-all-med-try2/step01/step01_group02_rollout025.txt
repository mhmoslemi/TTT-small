To decrease C5 = max_k C_k, I want to find a direction d such that d·∇C_k* < 0 for the active k*. The subgradient g = ∇C_k* (or average of active ones). So I want d·g < 0.

    j = j_candidates[rng.randint(len(j_candidates))]

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def sum_constraint(h, n):
    return np.sum(h) - n / 2.0

def sum_constraint_jac(h, n):
    return np.ones(n)

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

            if 0 <= j - k < n:
                g -= h[j - k]

def informed_pair(h, n, rng):
    g = compute_subgradient(h, n)
    g_sorted_idx = np.argsort(g)
    half = max(1, n // 3)
    i = g_sorted_idx[rng.randint(half)]
    j = g_sorted_idx[rng.randint(half, n)]
    if i == j:
        j = (j + 1) % n
    return int(i), int(j)

def do_2point_move(h, n, rng, xatol=1e-12, use_informed=True):
    if use_informed and rng.random() < 0.7:
        i, j = informed_pair(h, n, rng)
    else:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            j = (j + 1) % n
    d_lo = max(-h[i], h[j] - 1.0)
    d_hi = min(1.0 - h[i], h[j])
    if d_hi - d_lo < 1e-12:
        return h, compute_c5(h, n), False
    res = minimize_scalar(eval_2point, args=(h, n, i, j),
                         bounds=(d_lo, d_hi), method='bounded',
                         options={'xatol': xatol})
    c_cur = compute_c5(h, n)
    if res.fun < c_cur - 1e-15:
        h[i] += res.x
        h[j] -= res.x
        h = np.clip(h, 0, 1)
        return h, compute_c5(h, n), True
    return h, c_cur, False

def slsqp_optimize(h, n, alpha, maxiter=10000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac, 'args': (n,)}]
    try:

                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_out = project_feasible(res.x, n)
        return h_out
    except Exception:
