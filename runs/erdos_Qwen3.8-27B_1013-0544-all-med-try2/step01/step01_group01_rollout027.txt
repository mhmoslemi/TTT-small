import time

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -50, 50)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if mask.any():
            h[mask] += diff / mask.sum()
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def sa_binary(h0, n_swaps=50000, T0=0.003, alpha=0.9998):
    """Simulated annealing on binary sequences."""
    n = len(h0)
    h = (h0 >= 0.5).astype(float)
    # Fix weight
    while h.sum() > n / 2:
        idx = np.where(h == 1)[0]
        h[idx[np.random.randint(len(idx))]] = 0
    while h.sum() < n / 2:
        idx = np.where(h == 0)[0]
        h[idx[np.random.randint(len(idx))]] = 1
    
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    T = T0
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for _ in range(n_swaps):
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        
        h[i], h[j] = 0.0, 1.0
        nc = compute_c5(h)
        d = nc - c5

                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
        
        T *= alpha
    
    return best_h, best_c5

    """Reconstruct h from Fourier coefficients."""
    h = np.full(n, 0.5)
    for m, (a, b) in enumerate(params, 1):
        theta = 2.0 * np.pi * m * np.arange(n) / n
        h += a * np.cos(theta) + b * np.sin(theta)
    return np.clip(h, 0.0, 1.0)

    h = fourier_h(params, n)
    h = project_h(h, n / 2.0)
    if h.sum() != n / 2.0:  # check feasibility
        return compute_c5(h) + 0.1 * abs(h.sum() - n / 2.0)
    return compute_c5(h)

import numpy as np
from scipy.optimize import minimize, Bounds, differential_evolution
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -50, 50)
    return float(m + np.log(np.sum(np.exp(shifted))) / beta)

def obj_smooth_max(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return smooth_max(corr, beta)

def sum_eq(x):
    return float(x.sum()) - len(x) / 2.0

def sum_eq_jac(x):
    return np.ones(len(x))

def project_h(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.any():
            h[mask] += diff / mask.sum()
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def sa_binary(h0, n_swaps, T0, alpha):
    n = len(h0)
    h = (h0 >= 0.5).astype(float)
    while h.sum() > n / 2:
        idx = np.where(h == 1)[0]
        h[idx[np.random.randint(len(idx))]] = 0
    while h.sum() < n / 2:
        idx = np.where(h == 0)[0]
        h[idx[np.random.randint(len(idx))]] = 1
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    T = T0
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for _ in range(n_swaps):
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        nc = compute_c5(h)
        d = nc - c5
        if d < 0 or (T > 1e-9 and np.random.random() < np.exp(-d / T)):
            c5 = nc
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
        T *= alpha
    return best_h, best_c5

def fourier_h(params, n):
    h = np.full(n, 0.5)
    for m in range(len(params) // 2):
        a = params[2 * m]
        b = params[2 * m + 1]
        theta = 2.0 * np.pi * (m + 1) * np.arange(n) / n
        h += a * np.cos(theta) + b * np.sin(theta)
    return np.clip(h, 0.0, 1.0)

def fourier_obj_flat(params, n):
    h = fourier_h(params, n)
    h = project_h(h, n / 2.0)
    return compute_c5(h)

def slsqp_optimize(h0, n, beta, maxiter, ftol):
    bounds = Bounds(np.zeros(n), np.ones(n))
    constraints = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac}]
    res = minimize(obj_smooth_max, h0, args=(beta,), method='SLSQP',
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_h(res.x, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    else:

        h_init = np.full(n, 0.5)

    target_sum = n / 2.0
    best_h = project_h(h_init, target_sum)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - start_time

    defremaining():
        return budget_s - elapsed()

    # Phase 1: Binary SA - find good combinatorial structure
    if remaining() > 150:

        h_bin, c5_bin = sa_binary(best_h, n_swaps, 0.004, 0.9997)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

    # Phase 2: SLSQP polish with increasing smoothness
    for beta in [40, 80, 160, 320, 640, 1280]:

    # Phase 3: Fourier global search (low-dimensional)

        elif strategy == 4:

            pert = np.random.randn(half) * 0.03

        else:

            h0 = np.random.random(n) * 0.6 + 0.2

                res = minimize(obj_smooth_max, h0, args=(200,), method='COBYLA',

                if c5 < best_c5:

    # Phase 5: Final high-precision polish
    if remaining() > 15:
        for beta in [500, 1000, 2000]:
            if remaining() < 8:

            h_new = slsqp_optimize(best_h, n, beta, 3000, 1e-16)

            if c5 < best_c5:

                best_h = h_new

    # Final projection to ensure feasibility
    best_h = project_h(best_h, target_sum)
