        h = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        m = np.max(corr)
        return m + eps * np.log(np.sum(np.exp((corr - m) / eps)))

    def eq_con(x):
        return float(np.sum(x)) - target_sum

    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': eq_con}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            h_out = project_feasible(res.x, target_sum)
            return h_out, compute_c5(h_out, dx)
    except Exception:
        pass
    return None, None

def slsqp_direct(h0, n, dx, target_sum, maxiter=2000):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        return float(np.max(corr)) * dx

    def eq_con(x):
        return float(np.sum(x)) - target_sum

    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': eq_con}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            h_out = project_feasible(res.x, target_sum)
            return h_out, compute_c5(h_out, dx)
    except Exception:
        pass
    return None, None

def gradient_refine(h, n, dx, target_sum, budget_s, rng):
    n_half = n - 1
    step_size = 0.002
    momentum = np.zeros(n)
    best_c = compute_c5(h, dx)
    best_h = h.copy()
    t_end = time.time() + budget_s
    while time.time() < t_end - 1.0:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        tol = max(1e-4, c_max * 0.001)
        active = np.where(corr >= c_max - tol)[0]
        if len(active) < 2:
            active = np.argsort(corr)[-3:]
        elif len(active) > 12:
            active = np.argsort(corr)[-12:]
        grad = np.zeros(n)
        for idx in active:
            L = int(idx) - n_half
            if L == 0:
                grad[:] += (1.0 - h) - h
            elif L > 0:
                grad[L:] += (1.0 - h)[:n - L]
                grad[:n - L] -= h[L:]
            else:
                L2 = -L
                grad[:n - L2] += (1.0 - h)[L2:]
                grad[L2:] -= h[:n - L2]
        grad /= float(len(active))
        momentum = 0.5 * momentum - step_size * grad
        h_new = project_feasible(h + momentum, target_sum)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c:
            best_c = c5_new
            best_h = h_new.copy()
        else:
            step_size *= 0.92
            if step_size < 1e-9:
                h = best_h + rng.normal(0, 0.01, n)
                h = project_feasible(h, target_sum)
                step_size = 0.001
                momentum = np.zeros(n)
        h = h_new
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')

    def tleft():
        return (start_time + budget_s - 5.0) - time.time()

    n_init = 70
    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except NameError:
        pass

    # Determine which n values to try
    n_list = [n_init]
    for nv in [64, 66, 70, 72, 76, 80, 84, 88, 96, 100]:
        if nv not in n_list and nv % 2 == 0:
            n_list.append(nv)
    n_list = sorted(set(n_list))

    # Allocate time per n value
    time_per_n = (budget_s - 120) / len(n_list)

    for n in n_list:
        if tleft() < 80.0:
            break
        if n % 2 != 0:
            continue
        target_sum = n / 2.0
        dx = 2.0 / n
        n_half = n - 1

        # Build starting points
        starts = []
        if n == n_init and h_init is not None:
            starts.append(project_feasible(h_init, target_sum))

        # Binary-like starts
        for s in range(8):
            h_bin = make_binary_start(n, rng, s % 6)
            starts.append(project_feasible(h_bin, target_sum))

        # Perturbed versions of current best (if compatible)
        if best_h is not None and len(best_h) == n:
            for sc in [0.03, 0.08, 0.15]:
                starts.append(project_feasible(best_h + rng.normal(0, sc, n), target_sum))

        # Limit starts to fit in time
        max_starts = max(4, int(time_per_n / 25))
        starts = starts[:max_starts]

        for h0 in starts:
            if tleft() < 40.0:
                break
            # Smooth SLSQP
            h_res, c5_res = slsqp_smooth(h0, n, dx, target_sum, eps=0.005, maxiter=1500)
            if h_res is not None and c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res.copy()
            if tleft() < 40.0:
                break
            # Direct SLSQP from best result
            if h_res is not None:
                h_res2, c5_res2 = slsqp_direct(h_res, n, dx, target_sum, maxiter=1000)
                if h_res2 is not None and c5_res2 < best_c5:
                    best_c5 = c5_res2
                    best_h = h_res2.copy()

    # Final gradient refinement on best
    if best_h is not None and tleft() > 30.0:
        n_best = len(best_h)
        dx_best = 2.0 / n_best
        best_h, best_c5 = gradient_refine(best_h, n_best, dx_best, n_best / 2.0,
                                          min(tleft() - 5, 200), rng)

    # Final SLSQP polish
    if best_h is not None and tleft() > 25.0:
        n_best = len(best_h)
        dx_best = 2.0 / n_best
        h_res, c5_res = slsqp_smooth(best_h, n_best, dx_best, n_best / 2.0,
                                      eps=0.002, maxiter=2000)
        if h_res is not None and c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()
        if tleft() > 15.0:
            h_res, c5_res = slsqp_direct(best_h, n_best, dx_best, n_best / 2.0, maxiter=1500)
            if h_res is not None and c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res.copy()

    if best_h is None:
        n = n_init
        dx = 2.0 / n
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h, dx)

    n_final = len(best_h)
    dx_final = 2.0 / n_final
    best_h = project_feasible(best_h, n_final / 2.0)
    best_c5 = compute_c5(best_h, dx_final)
    return (best_h, best_c5, n_final)
