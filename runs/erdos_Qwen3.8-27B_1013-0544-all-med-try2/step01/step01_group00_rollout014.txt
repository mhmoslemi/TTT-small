        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5


def interp_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_h(h_new, n_new * 0.5)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    h = project_h(h, n * 0.5)
    c5_best = compute_c5(h)
    h_best = h.copy()

    # Phase 1: LSE optimization with decreasing temperature at current resolution
    temps = [0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.001, 0.0005]
    for temp in temps:
        if time_left() < 50:
            break
        h_new, c5_new = optimize_lse(h_best, temp, mu=50.0, max_iter=200)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new
        if time_left() < 30:
            break
        h_new, c5_new = optimize_lse(h_best, temp, mu=500.0, max_iter=100)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new

    # Phase 2: Subgradient on max correlation
    if time_left() > 100:
        h_new, c5_new = subgrad_max(h_best, max_iter=400, alpha0=0.003)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new

    # Phase 3: T-variable SLSQP polish
    if time_left() > 120:
        h_new, c5_new = optimize_t_var(h_best, max_iter=400, ftol=1e-11)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new

    # Phase 4: Resolution increase to n=200
    if time_left() > 200:
        n_new = 200
        h_cur = interp_h(h_best, n_new)
        for temp in [0.03, 0.01, 0.005, 0.002]:
            if time_left() < 60:
                break
            h_new, c5_new = optimize_lse(h_cur, temp, mu=100.0, max_iter=200)
            if c5_new < c5_best:
                h_best = h_new.copy()
                c5_best = c5_new
                h_cur = h_new
        if time_left() > 80:
            h_new, c5_new = optimize_t_var(h_cur, max_iter=300, ftol=1e-11)
            if c5_new < c5_best:
                h_best = h_new.copy()
                c5_best = c5_new

    # Phase 5: Resolution increase to n=240
    if time_left() > 200:
        n_new = 240
        h_cur = interp_h(h_best, n_new)
        for temp in [0.02, 0.01, 0.005]:
            if time_left() < 60:
                break
            h_new, c5_new = optimize_lse(h_cur, temp, mu=100.0, max_iter=200)
            if c5_new < c5_best:
                h_best = h_new.copy()
                c5_best = c5_new
                h_cur = h_new
        if time_left() > 80:
            h_new, c5_new = optimize_t_var(h_cur, max_iter=300, ftol=1e-11)
            if c5_new < c5_best:
                h_best = h_new.copy()
                c5_best = c5_new

    # Phase 6: Random perturbations at best resolution
    for i in range(6):
        if time_left() < 50:
            break
        scale = 0.015 + 0.005 * i
        h_p = h_best + rng.randn(len(h_best)) * scale
        h_p = project_h(h_p, len(h_best) * 0.5)
        h_new, c5_new = optimize_lse(h_p, temp=0.01, mu=100.0, max_iter=150)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new
        if time_left() < 40:
            break
        h_new, c5_new = optimize_lse(h_best, temp=0.005, mu=200.0, max_iter=100)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new

    # Phase 7: Final subgradient + t-var polish
    if time_left() > 60:
        h_new, c5_new = subgrad_max(h_best, max_iter=300, alpha0=0.002)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new
    if time_left() > 60:
        h_new, c5_new = optimize_t_var(h_best, max_iter=500, ftol=1e-12)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new

    # Final cleanup
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_h(h_best, len(h_best) * 0.5)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, len(h_best)
