def compute_grad(h, p, n, dx):
    J = np.arange(n)  # (n,)
    K = np.arange(2*n-1) - (n-1)  # (2n-1,)
    # diff1[j, i] = j - K[i]
    diff1 = J[:, None] - K[None, :]  # (n, 2n-1)
    valid1 = (diff1 >= 0) & (diff1 < n)
    # For each j, T1[j] = sum_i p[i] * (1-h[j-K[i]]) * valid1[j,i]
    # We need (1-h[diff1]) where valid
    one_minus_h = 1.0 - h
    # Build the matrix: M1[j,i] = p[i] * (1-h[diff1[j,i]]) if valid else 0
    # Index: diff1[j,i] gives the index into h
    safe_diff1 = np.where(valid1, diff1, 0)
    M1 = valid1 * p[None, :] * one_minus_h[safe_diff1]
    T1 = M1.sum(axis=1)
    
    diff2 = J[:, None] + K[None, :]
    valid2 = (diff2 >= 0) & (diff2 < n)
    safe_diff2 = np.where(valid2, diff2, 0)
    M2 = valid2 * p[None, :] * h[safe_diff2]
    T2 = M2.sum(axis=1)
    
    return dx * (T1 - T2)

    target = n / 2.0

        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target

            break

def optimize_coordinate(h, j, dx, max_iter=20):
    """Minimize C5 by optimizing h[j] with others fixed."""
    n = len(h)
    # Compute the coefficients of h[j] in each c_k
    # c_k = sum_i h[i]*(1-h[i+k])
    # Terms with h[j]: h[j]*(1-h[j+k]) [if j+k valid] and -h[j-k]*h[j] [if j-k valid]
    # So c_k = A_k * h[j] + B_k where:
    # A_k = (1-h[j+k]) [if 0<=j+k<n] + (-h[j-k]) [if 0<=j-k<n]
    # B_k = c_k - A_k * h[j]
    
    # First compute all c_k
    corr = np.correlate(h, 1-h, mode='full')
    
    # Compute A_k for each k
    # k ranges from -(n-1) to n-1, indexed as i = k+(n-1)
    A = np.zeros(2*n-1)
    for k in range(-(n-1), n):
        i = k + (n-1)
        a = 0.0
        if 0 <= j + k < n:
            a += (1.0 - h[j+k])
        if 0 <= j - k < n:
            a += (-h[j-k])
        A[i] = a
    
    # B_k = c_k - A_k * h[j]
    B = corr - A * h[j]
    
    # Now minimize max_k (A_k * t + B_k) * dx for t in [0,1]
    # This is a 1D convex problem: min_t max_k (A_k * t + B_k)
    # Can be solved by: the optimal t is where the maximum is as low as possible
    # Binary search on t, checking if all (A_k*t + B_k) <= threshold
    
    # Actually, for 1D: f(t) = max_k (A_k*t + B_k) is a convex piecewise linear function
    # Its minimum is at a vertex (intersection of two lines) or at a boundary
    
    # Simple approach: golden section search or just evaluate at many points
    # Since it's 1D and convex, bisection on the derivative works
    
    # Let's just use a fine grid + refinement
    best_t = h[j]
    best_val = np.max(corr) * dx
    
    # Golden section
    gr = (np.sqrt(5) - 1) / 2  # golden ratio ^ -1
    a, b = 0.0, 1.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = np.max(A * c + B) * dx
    fd = np.max(A * d + B) * dx
    
    for _ in range(60):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = np.max(A * c + B) * dx
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = np.max(A * d + B) * dx
    
    best_t = (a + b) / 2
    h[j] = best_t
    new_c5 = np.max(A * best_t + B) * dx
    if new_c5 < best_val:
        return h, new_c5
    else:
        h[j] = best_t  # still update since it's the 1D optimum
        return h, new_c5

Δc_k = (new_h[j]*(1-h[j+k]) - old_h[j]*(1-h[j+k])) + (h[j-k]*(1-new_h[j]) - h[j-k]*(1-old_h[j]))
= (new_h[j] - old_h[j]) * (1-h[j+k]) + h[j-k] * (old_h[j] - new_h[j])
= (new_h[j] - old_h[j]) * (1-h[j+k]-h[j-k])

So yes, c_k changes linearly with h[j]. My formulation is correct.

But wait, I need to double-check: when I write c_k = A_k * h[j] + B_k, this is only valid if the formula for c_k indeed has h[j] appearing linearly. And it does, because c_k is bilinear in h (product of two entries of h), and for a fixed h[j], the other entries are constants.

Actually, c_k = sum_i h[i] * (1-h[i+k]) is NOT linear in h[j] in general! It's:
c_k = h[j]*(1-h[j+k]) + sum_{i≠j, i+k≠j} h[i]*(1-h[i+k])

The first term is h[j] - h[j]*h[j+k], which is linear in h[j] (since h[j+k] is fixed).
The second sum also includes the term where i+k = j, i.e., i = j-k: h[j-k]*(1-h[j]) = h[j-k] - h[j-k]*h[j], which is also linear in h[j].

So c_k = (1-h[j+k] - h[j-k]) * h[j] + (sum of other terms not involving h[j])

This is indeed linear in h[j]. 

So the coordinate descent approach is valid. And since f(t) = max_k (A_k*t + B_k) is convex (max of lines), golden section search works perfectly.

    n = len(h)
    best_c5 = compute_c5(h, dx)
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            old_hj = h[j]
            # Compute A and B
            corr = np.correlate(h, 1-h, mode='full')
            A = np.zeros(2*n-1)
            for k in range(-(n-1), n):
                i = k + (n-1)
                a = 0.0
                if 0 <= j + k < n:
                    a += 1.0 - h[j+k]
                if 0 <= j - k < n:
                    a += -h[j-k]
                A[i] = a
            B = corr - A * old_hj
            
            # Golden section on t in [0,1]
            gr = (np.sqrt(5) - 1) / 2
            a, b = 0.0, 1.0
            c = b - gr * (b - a)
            d = a + gr * (b - a)
            fc = np.max(A * c + B)
            fd = np.max(A * d + B)

                    b = d; d = c; fd = fc
                    c = b - gr * (b - a)

        best_h = pop[best_idx].copy()

    num_rounds = 8
    gens_per_round = 15
    for r in range(num_rounds):
        if elapsed() > phase1_end - 5:
            break
        F = 0.8 - 0.5 * r / float(num_rounds)
        cr = 0.8 - 0.3 * r / float(num_rounds)
        for g in range(gens_per_round):
            if elapsed() > phase1_end - 5:
                break
            pop, best_idx, c5_pop = de_step(pop, best_idx, n, F, cr, rng)
        if c5_pop[best_idx] < best_c5:
            best_c5 = c5_pop[best_idx]
            best_h = pop[best_idx].copy()
        # Coordinate descent on best
        h_cd, c5_cd = coordinate_descent(best_h, dx, max_sweeps=100,
                                         time_check=elapsed, time_limit=phase1_end - 5)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
            pop[best_idx] = h_cd
            c5_pop[best_idx] = c5_cd

    # Phase 2: Smooth L-BFGS polish at n=90
    phase2_end = start_time + budget_s * 0.60
    if elapsed() < phase2_end - 5:
        h_sp = smooth_polish(best_h, dx, n, time_check=elapsed, time_limit=phase2_end - 5)
        c5_sp = compute_c5(h_sp, dx)
        if c5_sp < best_c5:
            best_c5 = c5_sp
            best_h = h_sp
        # More coordinate descent
        h_cd2, c5_cd2 = coordinate_descent(best_h, dx, max_sweeps=200,
                                           time_check=elapsed, time_limit=phase2_end - 5)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2

    # Phase 3: Upsample to 2n and optimize
    phase3_end = start_time + budget_s * 0.90
    if elapsed() < phase3_end - 30:
        n2 = 2 * n
        dx2 = 2.0 / n2
        target2 = n2 / 2.0
        h_up = np.repeat(best_h, 2)
        h_up = project_feasible(h_up, n2)
        c5_up = compute_c5(h_up, dx2)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()

        # DE at n2
        pop2_size = 60
        pop2 = de_population(h_up, n2, target2, pop2_size, seed + 100)
        c5_pop2 = np.array([compute_c5(pop2[i], dx2) for i in range(pop2_size)])
        best_idx2 = int(np.argmin(c5_pop2))
        if c5_pop2[best_idx2] < best_c5:
            best_c5 = c5_pop2[best_idx2]
            best_h = pop2[best_idx2].copy()

        rng2 = np.random.RandomState(seed + 200)
        for r2 in range(5):
            if elapsed() > phase3_end - 20:
                break
            F2 = 0.7 - 0.3 * r2 / 5.0
            cr2 = 0.7
            for g2 in range(12):
                if elapsed() > phase3_end - 20:
                    break
                pop2, best_idx2, c5_pop2 = de_step(pop2, best_idx2, n2, F2, cr2, rng2)
            if c5_pop2[best_idx2] < best_c5:
                best_c5 = c5_pop2[best_idx2]
                best_h = pop2[best_idx2].copy()
            h_cd3, c5_cd3 = coordinate_descent(best_h, dx2, max_sweeps=80,
                                               time_check=elapsed, time_limit=phase3_end - 20)
            if c5_cd3 < best_c5:
                best_c5 = c5_cd3
                best_h = h_cd3
                pop2[best_idx2] = h_cd3
                c5_pop2[best_idx2] = c5_cd3

        # Smooth polish at n2
        if elapsed() < phase3_end - 10:
            h_sp2 = smooth_polish(best_h, dx2, n2, time_check=elapsed, time_limit=phase3_end - 10)
            c5_sp2 = compute_c5(h_sp2, dx2)
            if c5_sp2 < best_c5:
                best_c5 = c5_sp2
                best_h = h_sp2

    # Phase 4: Final coordinate descent polish
    n_f = len(best_h)
    dx_f = 2.0 / n_f
    if elapsed() < budget_s - 10:
        h_final, c5_final = coordinate_descent(best_h, dx_f, max_sweeps=300,
                                               time_check=elapsed, time_limit=budget_s - 10)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    # Final feasibility check
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h, 2.0 / len(best_h))

    return (best_h, best_c5, len(best_h))
