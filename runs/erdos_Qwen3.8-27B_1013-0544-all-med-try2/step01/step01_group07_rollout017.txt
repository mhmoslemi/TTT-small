    """Compute gradient of smooth_max_obj with respect to h"""
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    max_c = np.max(corr)
    exp_shifted = np.exp(t * (corr - max_c))
    weights = exp_shifted / exp_shifted.sum()
    
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if weights[idx] < 1e-10:
            continue
        lag = idx - (n - 1)

                grad[j] += w * (-h[j + l])

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_obj(h, t, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    return (max_c + np.log(np.sum(np.exp(shifted))) / t) * dx

def top_lag_gradient(h, n_top=1):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    top_idx = np.argsort(corr)[-n_top:]
    grad = np.zeros(n)
    for idx in top_idx:
        lag = idx - (n - 1)
        if lag >= 0:
            grad[:n-lag] += (1.0 - h[lag:])
            grad[lag:] += (-h[:n-lag])
        else:
            l = -lag
            grad[l:] += (1.0 - h[:n-l])
            grad[:n-l] += (-h[l:])
    return grad / n_top

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

            no_improve = 0
            h = best_h.copy()
