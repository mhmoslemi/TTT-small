```python
def _coord_descent_fast(h, n_half, target_sum, max_sweeps, tleft_fn):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0-h, mode='full').copy()
    
    for sweep in range(max_sweeps):
        if tleft_fn() < 5.0:
            break
        for i in range(n):
            old_val = h[i]
            # d[k] = derivative of corr[k] w.r.t. h[i]
            d = np.zeros(2*n-1)
            for k in range(2*n-1):
                L = k - n_half
                val = 0.0
                if 0 <= i+L < n:
                    val += (1.0 - h[i+L])
                if 0 <= i-L < n:
                    val -= h[i-L]
                d[k] = val
            
            # A[k] = corr[k] - old_val * d[k]
            A = corr - old_val * d
            
            # Ternary search
            a, b = 0.0, 1.0
            for _ in range(40):
                m1 = a + (b-a)/3.0
                m2 = b - (b-a)/3.0
                if np.max(A + m1 * d) < np.max(A + m2 * d):
                    b = m2
                else:
                    a = m1
            new_val = (a+b)/2.0
            
            # Update corr incrementally
            delta = new_val - old_val
            CORQ = delta * d
            corr += CORQ
            h[i] = new_val
        
        h = project_feasible(h, target_sum)
        # Recalculate corr after projection
        corr = np.correlate(h, 1.0-h, mode='full')
    
    return h
```

Wait, the incremental update `corr += delta * d` assumes that the derivative d is constant over the range of t. But d depends on the OTHER coordinates which don't change during the inner loop. So d IS constant for a given i! The derivative of corr[k] w.r.t. h[i] is indeed d[k] (linear in h[i]). So the incremental update is exact.

But wait, after I change h[i], the derivative for the NEXT coordinate j will be different (because h[i] has changed). But that's fine - I compute d fresh for each coordinate.

However, there's a subtlety: when I compute `corr += delta * d`, this updates corr to reflect the new h. Then for the next coordinate j, I use the updated corr. This is correct because corr should always reflect the current h.

But I also need to update d for coordinate j, which depends on h[i] (the new value). Since I recompute d from scratch for each j, this is handled correctly.

- For each coordinate i:
  1. Compute d (derivative of corr w.r.t. h[i]) - O(n) since only 2n-1 lags, and for each lag only 2 terms

  3. Ternary search - O(n * 40)
  4. Update corr and h - O(n)

import time

def project_feasible(x, target_sum):
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _coord_gradient(h, i, n_half):
    """Compute d = dcorr/dh[i] as a vector of length 2*n-1"""
    n = n_half + 1
    d = np.zeros(2*n - 1)
    for k in range(2*n - 1):
        L = k - n_half
        if 0 <= i + L < n:
            d[k] += (1.0 - h[i + L])
        if 0 <= i - L < n:
            d[k] -= h[i - L]
    return d

def _coord_descent(h, n_half, target_sum, max_sweeps, tleft_fn):
    n = n_half + 1
    h = h.copy()

            break
        corr = np.correlate(h, 1.0 - h, mode='full').copy()
        for i in range(n):
            old_val = h[i]
            d = _coord_gradient(h, i, n_half)
            A = corr - old_val * d
            # Ternary search
            a, b = 0.0, 1.0

                h[i] = new_val
