    n = len(h)

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h, dx):
    return float(np.max(compute_corr(h)) * dx)

def project_feasible(x, s):
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        y = np.clip(x - mid, 0.0, 1.0)
        if y.sum() > s:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(x - lam, 0.0, 1.0)

def grad_corr_m(h, m):
    n = len(h)
    offset = m - (n - 1)
    grad = np.zeros(n)
    if offset >= 0:
        idx1 = np.arange(0, n - offset)
        grad[idx1] += (1.0 - h[idx1 + offset])
        idx2 = np.arange(offset, n)
        grad[idx2] -= h[idx2 - offset]
    else:
        f = -offset
        idx1 = np.arange(f, n)
        grad[idx1] += (1.0 - h[idx1 - f])
        idx2 = np.arange(0, n - f)
        grad[idx2] -= h[idx2 + f]
    return grad

def subgrad_descent(h, target_sum, dx, n_iters, step_init=0.005, top_k=5, momentum=0.5):
    n = len(h)
    h_curr = h.copy()
    v = np.zeros(n)
    step = step_init
    best_h = h_curr.copy()
    best_c5 = compute_c5(h_curr, dx)
    stall = 0
    for it in range(n_iters):
        corr = compute_corr(h_curr)
        c5_curr = np.max(corr) * dx
        k = min(top_k, len(corr))
        top_idx = np.argsort(corr)[-k:]
        top_vals = corr[top_idx]
        max_val = top_vals[-1]
        w = np.exp(8.0 * (top_vals - max_val))
        w /= w.sum()
        grad = np.zeros(n)
        for wt, mi in zip(w, top_idx):
            grad += wt * grad_corr_m(h_curr, mi)
        grad *= dx
        v = momentum * v + (1.0 - momentum) * grad
        for _ in range(25):
            h_try = project_feasible(h_curr - step * v, target_sum)
            c5_try = compute_c5(h_try, dx)
            if c5_try < c5_curr - 1e-10:
                break
            step *= 0.5
        else:
            step *= 0.5
            h_try = project_feasible(h_curr - step * v, target_sum)
            c5_try = compute_c5(h_try, dx)
        if c5_try < best_c5 - 1e-8:
            best_c5 = c5_try
            best_h = h_try.copy()
            h_curr = h_try
            step = min(step * 1.4, step_init * 5.0)
            stall = 0
        else:
            stall += 1
            h_curr = h_try
            step = max(step * 0.7, 1e-6)
            if stall > 80:
                v *= 0.5
                stall = 0
    return best_h, best_c5

def smooth_gradient(h, t):
    n = len(h)
    corr = compute_corr(h)
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    w = np.exp(shifted)
    w_sum = w.sum()
    w /= w_sum
    grad = np.zeros(n)
    sig = np.where(w > 1e-6)[0]
    for mi in sig:
        grad += w[mi] * grad_corr_m(h, mi)
    return grad

def smooth_pgd(h, target_sum, dx, t, n_iters, step_init=0.005, momentum=0.3):
    n = len(h)
    h_curr = h.copy()
    v = np.zeros(n)
    step = step_init
    best_h = h_curr.copy()
    best_c5 = compute_c5(h_curr, dx)
    stall = 0

        max_c = np.max(corr)
        c5_curr = max_c * dx
        obj_curr = (max_c + (1.0 / t) * np.log(np.sum(np.exp(t * (corr - max_c))))) * dx
        grad = smooth_gradient(h_curr, t) * dx
        v = momentum * v + (1.0 - momentum) * grad
        for _ in range(25):

            corr_try = compute_corr(h_try)
            max_try = np.max(corr_try)
            obj_try = (max_try + (1.0 / t) * np.log(np.sum(np.exp(t * (corr_try - max_try))))) * dx
            if obj_try < obj_curr - 1e-12:
                break
            step *= 0.5
        else:
            step *= 0.5
            h_try = project_feasible(h_curr - step * v, target_sum)
            obj_try = np.inf
        c5_try = compute_c5(h_try, dx)
        if c5_try < best_c5 - 1e-8:
            best_c5 = c5_try

            h_curr = h_try

            h_curr = h_try
            step = max(step * 0.75, 1e-6)

def sa_refine(h, target_sum, dx, n_iters=400, t_start=0.005):

    best_h = h_curr.copy()
    best_c5 = compute_c5(h_curr, dx)
    temp = t_start
    for it in range(n_iters):
        n_pert = np.random.randint(1, 6)
        idx = np.random.choice(n, n_pert, replace=False)
        h_try = h_curr.copy()
        h_try[idx] += np.random.uniform(-0.15, 0.15, n_pert)
        h_try = project_feasible(h_try, target_sum)

        c5_curr = compute_c5(h_curr, dx)
        d = c5_try - c5_curr
        if d < 0 or np.random.random() < np.exp(-d / max(temp, 1e-10)):
            h_curr = h_try
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
        temp *= 0.997
    return best_h, best_c5

def make_structured_init(n, style=0):
    x = np.linspace(0, 2, n, endpoint=False)
    if style == 0:

    h = np.clip(h, 0, 1)
    return h
