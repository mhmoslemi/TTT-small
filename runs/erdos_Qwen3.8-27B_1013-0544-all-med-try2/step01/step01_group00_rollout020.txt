    """Project h onto {0<=h<=1, sum(h)=target}"""
    h = h.copy()
    # Use bisection on Lagrange multiplier
    lo, hi = -2.0, 2.0
    for _ in range(60):

        projected = np.clip(h - mid, 0, 1)
        s = np.sum(projected)
        if s > target:
            lo = mid
        else:
            hi = mid

    diff = target - np.sum(projected)

        mask = (projected > 0) & (projected < 1)
        if np.sum(mask) > 0:
            projected[mask] += diff / np.sum(mask)
            projected = np.clip(projected, 0, 1)

    if abs(diff) > 1e-12:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            proj[mask] += diff / cnt
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def obj_and_grad(v, temp, target, mu):
    n = len(v)
    dx = 2.0 / n
    c = np.correlate(v, 1.0 - v, mode='full') * dx
    M = np.max(c)
    exp_t = np.exp((c - M) / temp)
    s = np.sum(exp_t)
    softmax = exp_t / s
    lse = M + temp * np.log(s)
    pen = mu * (np.sum(v) - target) ** 2
    val = lse + pen
    m_idx = np.arange(n)[:, None]
    k_vals = np.arange(2 * n - 1) - (n - 1)
    d1 = m_idx - k_vals[None, :]
    mask1 = (d1 >= 0) & (d1 < n)
    d1_safe = np.clip(d1, 0, n - 1)
    t1 = np.where(mask1, 1.0 - v[d1_safe], 0.0)
    d2 = m_idx + k_vals[None, :]
    mask2 = (d2 >= 0) & (d2 < n)
    d2_safe = np.clip(d2, 0, n - 1)
    t2 = np.where(mask2, -v[d2_safe], 0.0)
    A = t1 + t2
    grad_lse = (A.T @ softmax) * dx
    grad_pen = 2.0 * mu * (np.sum(v) - target)
    grad = grad_lse + grad_pen
    return val, grad


def obj_func(v, temp, target, mu):
    return obj_and_grad(v, temp, target, mu)[0]


def jac_func(v, temp, target, mu):
    return obj_and_grad(v, temp, target, mu)[1]


def run_lbfgsb(h0, temp, target, mu, max_iter, maxls):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj_func, h0, args=(temp, target, mu), jac=jac_func,
                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-15,
                            'gtol': 1e-10, 'maxls': maxls})
    return res.x


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    def time_left():
        return budget_s - (time.time() - t0)

    rng = np.random.RandomState(seed)
    h = np.array(initial_h_values, dtype=float)
    n0 = len(h)
    target0 = n0 * 0.5
    h = project_to_sum(h, target0)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    mu = 5000.0
    temps = [0.05, 0.03, 0.02, 0.015, 0.01, 0.007, 0.005, 0.003, 0.002,
             0.0015, 0.001, 0.0007, 0.0005, 0.0003, 0.0002, 0.0001]

    # Phase 1: optimize at original resolution
    h_cur = h.copy()
    for temp in temps:
        if time_left() < 80:
            break
        h_new = run_lbfgsb(h_cur, temp, target0, mu, 800, 80)
        h_new = project_to_sum(h_new, target0)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_cur = h_new.copy()

    # Phase 2: random restarts at original resolution
    for i in range(8):
        if time_left() < 60:
            break
        scale = 0.02 + 0.005 * i
        h_pert = h_cur + rng.randn(n0) * scale
        h_pert = project_to_sum(h_pert, target0)
        for temp in [0.01, 0.005, 0.002]:
            if time_left() < 30:
                break
            h_new = run_lbfgsb(h_pert, temp, target0, mu, 500, 60)
            h_new = project_to_sum(h_new, target0)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_cur = h_new.copy()
                h_pert = h_new + rng.randn(n0) * 0.01
                h_pert = project_to_sum(h_pert, target0)

    # Phase 3: increase to 2x resolution
    n2 = n0 * 2
    if time_left() > 200:
        x_old = np.linspace(0, 1, n0)
        x_new = np.linspace(0, 1, n2)
        h_interp = np.interp(x_new, x_old, best_h)
        target2 = n2 * 0.5
        h_interp = project_to_sum(h_interp, target2)
        h_cur2 = h_interp.copy()
        temps2 = [0.03, 0.02, 0.01, 0.007, 0.005, 0.003, 0.002, 0.001,
                  0.0007, 0.0005, 0.0003, 0.0002, 0.0001]
        for temp in temps2:
            if time_left() < 50:
                break
            h_new = run_lbfgsb(h_cur2, temp, target2, mu, 1000, 80)
            h_new = project_to_sum(h_new, target2)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_cur2 = h_new.copy()

        # Random restarts at 2x resolution
        for i in range(5):
            if time_left() < 40:
                break
            scale = 0.015 + 0.005 * i
            h_pert = h_cur2 + rng.randn(n2) * scale
            h_pert = project_to_sum(h_pert, target2)
            for temp in [0.005, 0.002, 0.001]:
                if time_left() < 25:
                    break
                h_new = run_lbfgsb(h_pert, temp, target2, mu, 600, 60)
                h_new = project_to_sum(h_new, target2)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h_cur2 = h_new.copy()

    # Phase 4: try 3x resolution if time allows
    n3 = n0 * 3
    if time_left() > 150:
        x_old = np.linspace(0, 1, len(best_h))
        x_new = np.linspace(0, 1, n3)
        h_interp = np.interp(x_new, x_old, best_h)
        target3 = n3 * 0.5
        h_interp = project_to_sum(h_interp, target3)
        h_cur3 = h_interp.copy()
        temps3 = [0.02, 0.01, 0.005, 0.003, 0.002, 0.001, 0.0005, 0.0003, 0.0001]
        for temp in temps3:
            if time_left() < 40:
                break
            h_new = run_lbfgsb(h_cur3, temp, target3, mu, 800, 60)
            h_new = project_to_sum(h_new, target3)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_cur3 = h_new.copy()

    # Phase 5: final polish at best resolution with very low temp
    if time_left() > 30:
        n_final = len(best_h)
        target_final = n_final * 0.5
        h_polish = best_h.copy()
        for temp in [0.00005, 0.00002, 0.00001]:
            if time_left() < 10:
                break
            h_new = run_lbfgsb(h_polish, temp, target_final, mu * 2, 500, 60)
            h_new = project_to_sum(h_new, target_final)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_polish = h_new.copy()

    # Final feasibility check
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, len(best_h) * 0.5)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
