from scipy.optimize import linprog

def solve_group_lp(j_indices, A, b, fixed_sum, m):
    # Minimize t
    # Subject to: A @ x + b <= t  →  A @ x - [1,1,...,1] @ [x,t] <= -b
    #             sum(x) = fixed_sum
    #             0 <= x <= 1
    #             t free (but t >= 0 implicitly)
    
    # Variables: [x_1, ..., x_m, t]
    # Objective: [0, ..., 0, 1]
    # Inequalities: A @ x - t <= -b  →  [A, -1] @ [x, t] <= -b
    #              sum(x) <= fixed_sum (equality)
    #              -sum(x) <= -fixed_sum (equality)
    # Bounds: 0 <= x_i <= 1, -inf <= t <= inf
    
    c_obj = np.zeros(m + 1)
    c_obj[-1] = 1.0
    
    A_ineq = np.hstack([A, -np.ones((len(j_indices), 1))])
    b_ineq = -b
    
    A_eq = np.zeros((1, m + 1))
    A_eq[0, :m] = 1.0
    b_eq = fixed_sum
    
    bounds = [(0, 1)] * m + [(None, None)]
    
    res = linprog(c_obj, A_ineq=A_ineq, b_ineq=b_ineq, 
                  A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if res.success:
        return res.x[:m]
    return None

def group_lp_optimize(h, group_idx, fixed_sum):

    n = len(h)

    # Current correlation
    corr = compute_corr(h)

    A = np.zeros((m_full, m))
