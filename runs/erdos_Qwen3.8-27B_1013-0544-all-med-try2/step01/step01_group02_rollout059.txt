    target_sum = n / 2.0

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def thue_morse(n, seed=0):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    n_ones = int(h.sum())
    target = n // 2
    rng = np.random.RandomState(seed)
    if n_ones > target:
        idx = np.where(h == 1)[0]
        flip = rng.choice(idx, size=n_ones - target, replace=False)
        h[flip] = 0.0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        flip = rng.choice(idx, size=target - n_ones, replace=False)
        h[flip] = 1.0
    return h

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def make_sum_con(x, n):
    return np.sum(x) - n / 2.0

def mip_solve(n, time_limit=120):
    dx = 2.0 / n
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    constraints = [h.sum() == n / 2, t >= 0]
    for L in range(-(n - 1), n):
        if L >= 0:
            valid_i = list(range(0, n - L))
        else:
            valid_i = list(range(-L, n))
        z = cp.Variable(len(valid_i))
        for idx, i in enumerate(valid_i):
            j = i + L
            constraints.append(z[idx] <= h[i])
            constraints.append(z[idx] <= 1 - h[j])
            constraints.append(z[idx] >= h[i] - h[j])
            constraints.append(z[idx] >= 0)
        constraints.append(z.sum() * dx <= t)
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver='CBC', time_limit=time_limit)
        if h.value is not None:
            h_sol = np.round(h.value).astype(float)
            n_ones = int(h_sol.sum())
            target = n // 2
            if n_ones != target:
                rng2 = np.random.RandomState(42)
                if n_ones > target:
                    idx1 = np.where(h_sol == 1)[0]
                    flip = rng2.choice(idx1, size=n_ones - target, replace=False)
                    h_sol[flip] = 0
                else:
                    idx0 = np.where(h_sol == 0)[0]
                    flip = rng2.choice(idx0, size=target - n_ones, replace=False)
                    h_sol[flip] = 1
            return h_sol
    except Exception:
        pass
    return None

def sa_binary(h, n, rng, max_time, start_time=None):
    if start_time is None:
        start_time = time.time()
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    cur_c5 = best_c5
    T = 0.08
    while time.time() - start_time < max_time:
        T = max(1e-8, T * 0.998)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5(h, n)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    return best_h, best_c5
