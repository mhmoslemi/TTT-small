import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-13) & (h < 1.0 - 1e-13)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    s = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / beta

def sml_obj(h, beta, dx):
    c = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(c, beta) * dx

def sum_fun(x, target):
    return np.sum(x) - target

def sum_jac(x, target):
    return np.ones(x.shape)

def slsqp_opt(h0, n, target, beta, bounds, maxiter, ftol):
    cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac, 'args': (target,)}]
    try:
        r = minimize(sml_obj, h0, args=(beta, 2.0 / n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        if r.fun < sml_obj(h0, beta, 2.0 / n) + 1e-12:
            return project_h(r.x, target)
    except Exception:
        pass
    return h0

def subgrad_refine(h, target, n, n_steps, step, t0, budget):
    best_c5 = compute_c5(h)
    mom = np.zeros(n)
    stall = 0
    for it in range(n_steps):
        if budget - (time.time() - t0) < 3:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_idx = np.argsort(corr)[::-1][:4]
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if lag >= 0:
                grad[lag:] += (1.0 - h[:n - lag])
                grad[:n - lag] -= h[lag:]
            else:
                la = -lag
                grad[:n - la] += (1.0 - h[la:])
                grad[la:] -= h[:n - la]
        grad *= dx
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        mom = 0.88 * mom - step * grad
        trial = project_h(h + mom, target)
        c_try = compute_c5(trial)
        if c_try < best_c5 - 1e-15:
            h = trial
            best_c5 = c_try
            stall = 0
        else:
            stall += 1
            mom *= 0.4
            trial = project_h(h + mom, target)
            c_try = compute_c5(trial)
            if c_try < best_c5 - 1e-15:
                h = trial
                best_c5 = c_try
                stall = 0
            else:
                mom = np.zeros(n)
        if stall > 300:
            break
    return h, best_c5

def coord_refine(h, target, n, t0, budget, max_sweeps=500):
    best_c5 = compute_c5(h)
    for sw in range(max_sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 1:
                break
            vals = [max(0.0, h[j] - 0.08), max(0.0, h[j] - 0.03),
                    min(1.0, h[j] + 0.03), min(1.0, h[j] + 0.08),
                    0.0, 1.0, h[j] * 0.5, 1.0 - h[j] * 0.5]
            for v in vals:
                if v < 0 or v > 1 or abs(v - h[j]) < 1e-6:
                    continue
                new_h = h.copy()
                new_h[j] = v
                new_h = project_h(new_h, target)
                c_try = compute_c5(new_h)
                if c_try < best_c5 - 1e-14:
                    h = new_h
                    best_c5 = c_try
                    improved = True
                    break
        if not improved and sw > 30:
            break
    return h, best_c5

def binary_local_search(h_start, n, target, t0, budget, max_rounds=400):
    h = (h_start > 0.5).astype(float)
    n_ones = int(h.sum())
    tgt = int(target)
    if n_ones > tgt:
        ones = np.where(h == 1)[0]
        np.random.shuffle(ones)
        for k in range(n_ones - tgt):
            h[ones[k]] = 0.0
    elif n_ones < tgt:
        zeros = np.where(h == 0)[0]
        np.random.shuffle(zeros)
        for k in range(tgt - n_ones):
            h[zeros[k]] = 1.0
    best_c5 = compute_c5(h)
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    stall = 0
    for rnd in range(max_rounds):
        if budget - (time.time() - t0) < 5:
            break
        improved = False
        n_tries = min(400, len(ones) * len(zeros))
        for _ in range(n_tries):
            if budget - (time.time() - t0) < 3:
                break
            ii = np.random.randint(len(ones))
            jj = np.random.randint(len(zeros))
            i, j = ones[ii], zeros[jj]
            h[i], h[j] = 0.0, 1.0
            c_try = compute_c5(h)
            if c_try < best_c5 - 1e-15:
                best_c5 = c_try
                ones[ii] = j
                zeros[jj] = i
                improved = True
                stall = 0
            else:
                h[i], h[j] = 1.0, 0.0
        if not improved:
            stall += 1
            if stall > 60:
                break
        else:
            stall = 0
    return h, best_c5

def gen_perturbed(h, n, scale, rng):
    p = h + rng.randn(n) * scale
    return project_h(p, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(200, 0.5)

    n = len(h_init)
    target = n / 2.0
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Binary local search from initial
    remaining = budget_s - (time.time() - t0)
    if remaining > 80:
        h_bin, c_bin = binary_local_search(h_init, n, target, t0, budget_s, max_rounds=300)
        if c_bin < best_c5:
            best_h, best_c5 = h_bin.copy(), c_bin

    # Phase 2: SLSQP refinement with increasing smoothness
    remaining = budget_s - (time.time() - t0)
    if remaining > 120:
        bounds = Bounds(np.zeros(n), np.ones(n))
        cur = best_h.copy()
        for beta in [200, 500, 1500, 4000]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 25:
                break
            cur = slsqp_opt(cur, n, target, beta, bounds, 5000, 1e-14)
            c_cur = compute_c5(cur)
            if c_cur < best_c5:
                best_h, best_c5 = cur.copy(), c_cur

    # Phase 3: Subgradient refinement
    remaining = budget_s - (time.time() - t0)
    if remaining > 80:
        cur, c_cur = subgrad_refine(best_h, target, n, 15000, 0.015, t0, budget_s)
        if c_cur < best_c5:
            best_h, best_c5 = cur.copy(), c_cur

    # Phase 4: Fine coordinate descent
    remaining = budget_s - (time.time() - t0)
    if remaining > 40:
        best_h, best_c5 = coord_refine(best_h, target, n, t0, budget_s, max_sweeps=300)

    # Phase 5: Multi-start perturbation + re-optimize
    remaining = budget_s - (time.time() - t0)
    if remaining > 50:
        bounds = Bounds(np.zeros(n), np.ones(n))
        for att in range(8):
            remaining = budget_s - (time.time() - t0)
            if remaining < 12:
                break
            scale = 0.015 / (att + 1)
            hp = gen_perturbed(best_h, n, scale, rng)
            hp = slsqp_opt(hp, n, target, 500, bounds, 3000, 1e-13)
            hp, c_hp = subgrad_refine(hp, target, n, 3000, 0.01, t0, budget_s)
            c_hp = compute_c5(hp)
            if c_hp < best_c5:
                best_h, best_c5 = hp.copy(), c_hp

    # Phase 6: Final polish
