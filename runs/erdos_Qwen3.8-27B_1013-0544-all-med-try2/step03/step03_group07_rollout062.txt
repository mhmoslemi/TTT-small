The key insight I want to add: **use a "saddle point" formulation**. Instead of just minimizing the smooth max, I can use the fact that:

Let me implement this:

```python
def variational_lag_descent(h, max_iters):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)
    best_x = x.copy()
    best_v = compute_c5(x)
    
    # Maintain a weight distribution over lags
    w = np.ones(2*n-1) / (2*n-1)  # uniform initially
    beta = 10.0  # temperature parameter
    
    for t in range(max_iters):
        lr = 0.1 / (1 + 0.01 * t)
        
        # Compute correlations
        corr = compute_corr(x) * dx
        
        # Update weights (exponential weighting / multiplicative weights)
        w = w * np.exp(beta * corr)
        w = w / w.sum()
        
        # Compute gradient of expected overlap
        grad = np.zeros(n)
        for k in range(2*n-1):
            if w[k] < 1e-10:
                continue
            s = k - (n-1)
            if s >= 0:
                grad[s:] += w[k] * (1 - x[:n-s])
                grad[:n-s] += w[k] * (-x[s:])
            elif s < 0:
                tt = -s
                grad[:n-tt] += w[k] * (1 - x[tt:])
                grad[tt:] += w[k] * (-x[:n-tt])
            else:
                grad += w[k] * (1 - 2*x)
        grad *= dx
        
        x_new = project_feasible(x - lr * grad, target)
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x = x_new
        
        # Gradually increase beta to focus on worst lags
        beta *= 1.001
    
    return best_x, best_v
```

This is essentially a no-regret algorithm for the minimax problem. It should converge to a saddle point.

from scipy.interpolate import interp1d
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):

    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

        h = np.clip(h, 0.0, 1.0)

        if abs(diff2) > 1e-12:

            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta):

    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(beta * shifted)
    sw = np.sum(exps)
    w = exps / sw

    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        a = abs(s)
        if a >= n:
            continue
        if a <= n - 1:
            idx1 = np.arange(a, n)
            grad[idx1] += wk * (1.0 - h[idx1 - a])
        if n - 1 - a >= 0:
            idx2 = np.arange(0, n - a)
            grad[idx2] += -wk * h[idx2 + a]
    grad *= dx
    return val, grad

def _obj(x, beta):
    v, _ = smoothmax_val_grad(x, beta)
    return v

def _grad(x, beta):
    _, g = smoothmax_val_grad(x, beta)
    return g

def _con(x, target):
    return x.sum() - target

def _con_jac(x):
    return np.ones(len(x))

def slsqp_refine(h, maxiter, beta):
    n = len(h)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: _con(x, target), 'jac': _con_jac}]
    try:
        res = minimize(lambda x: _obj(x, beta), h, jac=lambda x: _grad(x, beta),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h.copy(), target)

def coord_descent(h, n_sweeps=10, time_limit=None, t0=None):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)
    best_x = x.copy()
    best_v = compute_c5(x)
    corr = compute_corr(x)
    
    for sweep in range(n_sweeps):
        if time_limit and t0 and (time.time() - t0) > time_limit * 0.8:
            break
        for i in range(n):
            # Compute d_{k,i} for all k
            d = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                s = k - (n - 1)
                a = abs(s)
                if a >= n:
                    continue
                term1 = 0.0
                term2 = 0.0
                if a <= i:
                    term1 = 1.0 - x[i - a]
                if i <= n - 1 - a:
                    term2 = x[i + a]
                d[k] = term1 - term2
            
            # Solve 1D min-max: min_{delta} max_k (corr[k] + d[k]*delta)
            lo = -x[i]
            hi = 1.0 - x[i]
            
            if hi - lo < 1e-15:
                continue
            
            # Ternary search (convex function)
            best_delta = 0.0
            best_val = np.max(corr)
            a_ = lo
            b_ = hi
            for _ in range(40):
                m1 = a_ + (b_ - a_) / 3
                m2 = b_ - (b_ - a_) / 3
                v1 = np.max(corr + d * m1)
                v2 = np.max(corr + d * m2)
                if v1 < v2:
                    b_ = m2
                else:
                    a_ = m1
                mid = (a_ + b_) / 2
                vm = np.max(corr + d * mid)
            
            # Find optimal delta
            opt_delta = (a_ + b_) / 2
            opt_val = np.max(corr + d * opt_delta)
            
            if opt_val < best_v:
                x[i] = x[i] + opt_delta
                x[i] = np.clip(x[i], 0.0, 1.0)
                # Project to maintain sum
                x = project_feasible(x, target)
                corr = compute_corr(x)
                v = compute_c5(x)
                if v < best_v:
                    best_v = v
                    best_x = x.copy()
    
    return best_x, best_v

        h_cd, c_cd = coord_descent(h_best, 5, min(tl() - 3, 25), t0)
        if c_cd < best_c5:
            h_best, best_c5 = h_cd, c_cd

    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)
    return h_best, best_c5, len(h_best)
