        # f_k(h0) + grad_k · (h - h0) <= t/dx
        # => grad_k · h <= t/dx - f_k(h0) + grad_k · h0

        active = corr > 0.9 * corr.max()

import time

# Helper functions (top-level)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def con_fun(h, target_sum, n):
    return h.sum() - target_sum

def con_jac(h, target_sum, n):
    return np.ones(n)

            break

        n_corr = len(corr)

        b_ub = np.zeros(n_corr)

            g = np.zeros(n)
            if lag >= 0:

                if m > 0:
                    idx = np.arange(m)
                    g[idx + la] += (1.0 - h[idx])
                    g[idx] -= h[idx + la]

            A_ub[k_w, n] = -1.0 / dx

        c_obj[n] = 1.0
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

from scipy.optimize import minimize, Bounds, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = h.copy()
    lo, hi = -10.0, 10.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h + mid, 0, 1))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2, 0, 1)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0, 1)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def con_fun(h, target_sum, n):
    return h.sum() - target_sum

def con_jac(h, target_sum, n):
    return np.ones(n)

def run_sca(h_init, n, dx, max_iter, t_start, time_limit):
    h = h_init.copy()
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        n_corr = len(corr)
        A_ub = np.zeros((n_corr, n + 1))
        b_ub = np.zeros(n_corr)
        
        for k_w in range(n_corr):
            lag = k_w - (n - 1)
            g = np.zeros(n)
            if lag >= 0:
                m = n - lag
                if m > 0:
                    idx = np.arange(m)
                    g[idx] += (1.0 - h[idx + lag])
                    g[idx + lag] -= h[idx]
            else:
                la = -lag
                m = n - la
                if m > 0:
                    idx = np.arange(m)
                    g[idx + la] += (1.0 - h[idx])
                    g[idx] -= h[idx + la]
            A_ub[k_w, :n] = g
            A_ub[k_w, n] = -1.0 / dx
            b_ub[k_w] = g @ h - corr[k_w]
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target_sum])
        bounds = [(0, 1)] * n + [(0, None)]
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True})
        except Exception:
            break
        
        if not res.success:
            break
        
        h_new = np.clip(res.x[:n], 0, 1)
        h_new = project_feasible(h_new, target_sum)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5 - 1e-13:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new
            no_improve = 0
        else:
            no_improve += 1
            improved = False
            for alpha in [0.5, 0.25, 0.1]:
                h_step = (1 - alpha) * h + alpha * h_new
                h_step = project_feasible(h_step, target_sum)
                c5_step = compute_c5(h_step)
                if c5_step < best_c5 - 1e-13:
                    h = h_step
                    best_h = h_step.copy()
                    best_c5 = c5_step
                    improved = True
                    break
            if not improved and no_improve >= 10:
                break
    
    return best_h, best_c5

    # Phase 3: SLSQP from best
    if remaining() > 40:

        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum, n)}]
        for beta in [50, 200, 1000, 5000]:

            except Exception:

            if c_s < best_c5:

    if remaining() > 20:

        if c_s < best_c5:
            best_h, best_c5 = h_s, c5
    
    best_h = project_feasible(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)

            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_t = np.interp(x_new, x_old, best_h)
            h_t = project_feasible(h_t, ts_t)
            h_s, c_s = run_sca(h_t, n_try, dx_t, 100, t0, t0 + budget_s * 0.85)
            if c_s < best_c5:
                best_h, best_c5 = h_s, c_s
                n, dx, target_sum = n_try, dx_t, ts_t

    # Phase 7: Final SLSQP + SCA
    if remaining() > 20:
        bnds = Bounds(np.zeros(n), np.ones(n))
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum, n)}]
        for beta in [10000, 50000, 200000]:
            if remaining() < 12:
                break
            try:
                res = minimize(obj_smooth, best_h, args=(dx, beta), method='SLSQP',
                               bounds=bnds, constraints=cons,
                               options={'maxiter': 20000, 'ftol': 1e-16})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new, c5
            except Exception:
                pass

    if remaining() > 15:
        h_s, c_s = run_sca(best_h, n, dx, 300, t0, t0 + budget_s * 0.97)
        if c_s < best_c5:
            best_h, best_c5 = h_s, c5

    best_h = project_feasible(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
