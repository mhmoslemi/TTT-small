    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_c5(h, dx, top_m, beta_temp):
    n = len(h)
    corr_raw = np.correlate(h, 1.0 - h, mode="full")
    corr_val = corr_raw * dx
    top_idx = np.argsort(corr_val)[::-1][:top_m]
    vals = corr_val[top_idx]
    weights = np.exp((vals - vals[0]) * beta_temp)
    weights /= weights.sum()
    grad = np.zeros(n)
    center = n - 1
    for t_idx, w in zip(top_idx, weights):
        lag = int(t_idx) - center
        grad += w * grad_for_lag(h, lag, n, dx)
    c5 = float(np.max(corr_val))
    return grad, c5

        return h_old.copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h_old)
