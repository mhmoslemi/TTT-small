        h_new = h_cur - lr * v
        h_new = np.clip(h_new, 0.0, 1.0)
        diff = target - h_new.sum()
        free = (h_new > 1e-10) & (h_new < 1.0 - 1e-10)
        if free.sum() > 0:
            h_new[free] += diff / free.sum()
        h_new = np.clip(h_new, 0.0, 1.0)
        h_cur = h_new
    return h_cur


def slsqp_polish(h, beta, maxiter, top_k=None):
    n = len(h)
    target = n / 2.0

    def obj(x):
        return smoothmax_grad(x, beta, top_k)[0]

    def objg(x):
        return smoothmax_grad(x, beta, top_k)[1]

    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = ({'type': 'eq',
             'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)})
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)


def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='linear', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)


def basin_pgd(h_best, best_c5, n_hops, beta_list, n_iters, lr, momentum,
              noise_base, noise_decay, seed, t0, budget_s, rng):
    n = len(h_best)
    target = n / 2.0
    for hop in range(n_hops):
        if budget_s - (time.time() - t0) < 40:
            break
        noise = noise_base * (noise_decay ** hop)
        h_pert = h_best + rng.randn(n) * noise
        h_pert = project_feasible(h_pert, target)
        h_local = h_pert
        for beta in beta_list:
            if budget_s - (time.time() - t0) < 25:
                break
            h_local = pgd_solve(h_local, beta, n_iters, top_k=15, lr=lr, momentum=momentum)
        c_local = compute_c5(h_local)
        if c_local < best_c5:
            best_c5 = c_local
            h_best = h_local.copy()
    return h_best, best_c5


def block_perturb(h, rng, block_frac):
    n = len(h)
    h_pert = h.copy()
    target = n / 2.0
    blen = max(3, int(n * block_frac))
    start = rng.randint(0, n - blen + 1)
    idx = np.arange(start, start + blen)
    h_pert[idx] = np.clip(h_pert[idx] + rng.randn(blen) * 0.15, 0.0, 1.0)
    return project_feasible(h_pert, target)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def remaining():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_cur = np.array(initial_h_values, dtype=float).copy()
    else:
        h_cur = np.full(200, 0.5)

    best_c5 = compute_c5(h_cur)
    h_best = h_cur.copy()

    # Phase 1: PGD continuation
    n = len(h_cur)
    betas = [8, 20, 50, 100, 200, 400]
    for beta in betas:
        if remaining() < 50:
            break
        ni = 800 if beta <= 50 else 500
        lr = 0.08 if beta <= 50 else 0.04
        h_new = pgd_solve(h_cur, beta, ni, top_k=20, lr=lr, momentum=0.6)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            h_best = h_new.copy()
        h_cur = h_new

    # Phase 2: SLSQP polish
    if remaining() > 80:
        h_slq = slsqp_polish(h_best, 400, 300, top_k=15)
        c_slq = compute_c5(h_slq)
        if c_slq < best_c5:
            best_c5 = c_slq
            h_best = h_slq

    # Phase 3: Basin hopping with PGD
    h_best, best_c5 = basin_pgd(
        h_best, best_c5, n_hops=12,
        beta_list=[150, 400], n_iters=400, lr=0.04, momentum=0.6,
        noise_base=0.03, noise_decay=0.72, seed=seed+1,
        t0=t0, budget_s=budget_s, rng=rng)

    # Phase 4: Block perturbation exploration
    n = len(h_best)
    for bp_hop in range(8):
        if remaining() < 40:
            break
        h_pert = block_perturb(h_best, rng, 0.1 + 0.05 * (bp_hop % 3))
        h_local = h_pert
        for beta in [200, 500]:
            if remaining() < 20:
                break
            h_local = pgd_solve(h_local, beta, 350, top_k=12, lr=0.03, momentum=0.6)
        c_local = compute_c5(h_local)
        if c_local < best_c5:
            best_c5 = c_local
            h_best = h_local.copy()

    # Phase 5: Try different resolutions
    for n_try in [250, 300, 400]:
        if remaining() < 100:
            break
        h_interp = interp_to_n(h_best, n_try)
        h_c = h_interp
        for beta in [30, 100, 300]:
            if remaining() < 40:
                break
            h_c = pgd_solve(h_c, beta, 500, top_k=20, lr=0.06, momentum=0.6)
        if remaining() > 40:
            h_c = slsqp_polish(h_c, 400, 200, top_k=15)
        c_try = compute_c5(h_c)
        if c_try < best_c5:
            best_c5 = c_try
            h_best = h_c.copy()

    # Phase 6: Fine basin hopping at best resolution
    n = len(h_best)
    h_best, best_c5 = basin_pgd(
        h_best, best_c5, n_hops=8,
        beta_list=[300, 600], n_iters=300, lr=0.03, momentum=0.6,
        noise_base=0.012, noise_decay=0.75, seed=seed+100,
        t0=t0, budget_s=budget_s, rng=rng)

    # Phase 7: SLSQP final squeeze
    if remaining() > 35:
        h_fin = slsqp_polish(h_best, 600, 500, top_k=None)
        c_fin = compute_c5(h_fin)
        if c_fin < best_c5:
            best_c5 = c_fin
            h_best = h_fin

    # Phase 8: Very fine basin hop
    if remaining() > 50:
        n = len(h_best)
        h_best, best_c5 = basin_pgd(
            h_best, best_c5, n_hops=5,
            beta_list=[500, 800], n_iters=250, lr=0.02, momentum=0.6,
            noise_base=0.008, noise_decay=0.8, seed=seed+200,
            t0=t0, budget_s=budget_s, rng=rng)

    # Final polish
    if remaining() > 20:
        h_fin2 = pgd_solve(h_best, 1000, 300, top_k=None, lr=0.01, momentum=0.6)
        c_fin2 = compute_c5(h_fin2)
        if c_fin2 < best_c5:
            best_c5 = c_fin2
            h_best = h_fin2

    return h_best, best_c5, len(h_best)
