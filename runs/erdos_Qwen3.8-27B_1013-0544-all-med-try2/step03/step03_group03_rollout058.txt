                    best_c5 = c5

    # Phase 5: Final high-precision SLSQP
    if tl() > 50:
        for beta in [100000, 400000, 1000000]:
            if tl() < 15:
                break
            h_new, c5 = slsqp_solve(best_h, dx, target_sum, n, beta, 15000, 1e-16)
            if c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = c5

    # Phase 6: Micro-polish targeting worst single lag
    if tl() > 15:
        h = best_h.copy()
        stalled = 0
        for it in range(5000):
            if tl() < 5:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            k_worst = np.argmax(corr)
            lag = k_worst - (n - 1)
            grad = subgrad_for_lag(h, lag, n)
            lr = 0.0006 / (1.0 + 0.003 * it)
            h_new = project_feasible(h - lr * grad, target_sum)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                h = h_new
                best_h = h_new.copy()
                best_c5 = c_new
                stalled = 0
            else:
                stalled += 1
                if stalled > 800:
                    break

    best_h = project_feasible(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
