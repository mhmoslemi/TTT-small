def binary_local_search(h, target, max_sweeps=50, n_random=500):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    assert abs(h.sum() - target) < 1e-6
    
    corr = np.correlate(h, 1.0 - h, mode="full")
    best_c5 = np.max(corr) * dx
    cur_c5 = best_c5
    
    zeros = np.where(h < 0.5)[0].tolist()
    ones = np.where(h >= 0.5)[0].tolist()
    
    for sweep in range(max_sweeps):
        improved = False
        for _ in range(n_random):
            if len(zeros) == 0 or len(ones) == 0:
                break
            j = zeros[np.random.randint(len(zeros))]
            k = ones[np.random.randint(len(ones))]
            
            # Try flip: j: 0->1, k: 1->0
            h[j] = 1.0
            h[k] = 0.0
            new_corr = np.correlate(h, 1.0 - h, mode="full")
            new_c5 = np.max(new_corr) * dx
            
            if new_c5 < cur_c5 - 1e-12:
                cur_c5 = new_c5
                corr = new_corr
                improved = True
                if cur_c5 < best_c5:
                    best_c5 = cur_c5

                zeros.remove(j)
                ones.append(j)
                ones.remove(k)
                zeros.append(k)
            else:
                h[j] = 0.0
                h[k] = 1.0
        
        if not improved:
            # Try simulated annealing: accept with some probability
            # Or break
            break
    
    return h, best_c5

        h_fin3, c5_fin3 = local_search(best_h, target, max_sweeps=200, n_trials=300, rng=rng)
        if c5_fin3 < best_c5:
            best_h, best_c5 = h_fin3.copy(), c5_fin3

    # Ensure final solution is valid
    final_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(final_h)
    if abs(final_h.sum() - target) > 1e-6:
        final_h = project_h(final_h, target)
        final_c5 = compute_c5(final_h)

    return final_h, final_c5, int(n)
