def compute_c5(h):
    n = len(h)
    dx = 2.0 / n

    return float(np.max(corr) * dx)


def corr_full(h):
    return np.correlate(h, 1.0 - h, mode="full")


def project_feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_weighted_subgrad(h, weights):
    n = len(h)
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        w = weights[i]
        if w < 1e-12:
            continue
        lag = i - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[lag:] += w * (1.0 - h[:-lag])
            grad[:n - lag] -= w * h[lag:]
        elif -n < lag < 0:
            m = -lag
            grad[m:] += w * (1.0 - h[:n - m])
            grad[:n - m] -= w * h[m:]
    return grad
