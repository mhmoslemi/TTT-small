    dx = 2.0 / n

import numpy as np
from scipy.optimize import minimize, Bounds
import time

_opt_state = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj_and_grad(h, beta, target_sum, pw):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    w_sum = np.sum(exps)
    obj = (m + np.log(w_sum / (2 * n - 1)) / beta) * dx + pw * (h.sum() - target_sum) ** 2
    weights = exps / w_sum
    grad = np.zeros(n)
    for lag_idx in range(2 * n - 1):
        w = weights[lag_idx] * beta * dx
        if w < 1e-13:
            continue
        lag = lag_idx - (n - 1)
        if lag >= 0:
            k = lag
            if k < n:
                grad[:n - k] += w * (1.0 - h[k:])
                grad[k:] -= w * h[:n - k]
        else:
            k = -lag
            if k < n:
                grad[k:] += w * (1.0 - h[:n - k])
                grad[:n - k] -= w * h[k:]
    grad += 2.0 * pw * (h.sum() - target_sum)
    return obj, grad

def _opt_obj(u):
    return lse_obj_and_grad(u, _opt_state['beta'], _opt_state['target'], _opt_state['pw'])[0]

def _opt_jac(u):
    return lse_obj_and_grad(u, _opt_state['beta'], _opt_state['target'], _opt_state['pw'])[1]

def do_lbfgsb(h_init, n, target_sum, beta, pw, maxiter):

    _opt_state['target'] = target_sum
    _opt_state['pw'] = pw
    bounds = Bounds(np.zeros(n), np.ones(n))
    try:
        res = minimize(_opt_obj, h_init, method='L-BFGS-B', jac=_opt_jac,
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-15})
        return project_h(res.x, target_sum)
    except Exception:
        return h_init

def subgrad_apply(h, top_lags, target_sum, step):
    n = len(h)
    grad = np.zeros(n)
    for lag in top_lags:
        if lag >= 0:
            k = lag
            if k < n:
                grad[:n - k] += (1.0 - h[k:])
                grad[k:] -= h[:n - k]
        else:
            k = -lag
            if k < n:
                grad[k:] += (1.0 - h[:n - k])
                grad[:n - k] -= h[k:]
    return project_h(h - step * grad, target_sum)
