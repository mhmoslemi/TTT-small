**New idea: "Uniform reduction" via Frank-Wolfe**

The Frank-Wolfe (conditional gradient) method works for minimizing a convex function over a convex set. While our problem is not convex (due to the bilinear terms), we can use a variant:

At each iteration:
1. Compute the subgradient g of the current objective
2. Solve: min_{z in feasible set} g · z (this gives the "vertex" to move towards)
3. Update: h ← h + α * (z - h)

The feasible set is {h : ∑h = n/2, 0 ≤ h ≤ 1}. The linear minimization over this set is easy:
- Sort g in ascending order
- Set z[i] = 0 for the smallest g values, z[i] = 1 for the largest, and adjusted for the sum constraint

This is essentially the "greedy" approach: reduce h where the gradient is positive (increasing C₅) and increase h where the gradient is negative.

Let me implement this:

```python

    # Sort g, set z to 0 where g is negative, 1 where g is positive
    # Adjust to meet sum constraint
    indices = np.argsort(g)
    z = np.zeros(n)
    remaining = target
    for i in indices:
        # Add as much as possible
        z[i] = 1.0

    # If remaining < 0, we need to decrease some ones

        k = int(diff)

        ones_sorted = ones[np.argsort(-g[ones])]  # Most positive g first
        k = int(-diff)
        z[ones_sorted[:k]] = 0.0
        if -diff - k > 0:
            z[ones_sorted[k]] = 1.0 - (-diff - k)
    
    return z
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_c5(h, n, num_lags=20):
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    # Use threshold to select near-max lags
    threshold = max_corr - 0.005 * (max_corr + 1e-10)
    indices = np.where(corr >= threshold)[0]
    if len(indices) > num_lags:
        indices = indices[np.argsort(corr[indices])[::-1][:num_lags]]
    
    grad = np.zeros(n)
    for idx in indices:
        lag = int(idx) - (n - 1)
        if lag >= 0:
            grad[:n - lag] += (1.0 - h[lag:])
            grad[lag:] -= h[:n - lag]
        else:
            la = -lag
            grad[la:] += (1.0 - h[:n - la])
            grad[:n - la] -= h[la:]
    
    return grad / max(len(indices), 1), corr

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def sml_obj(h_in, beta, dx):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, beta) * dx

def sum_eq(x_in, target):
    return np.sum(x_in) - target

def sum_eq_jac(x_in, target):
    return np.ones(x_in.shape)

def do_slsqp(h0, n, target, beta, bounds, maxiter, ftol):
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        r = minimize(sml_obj, h0, args=(beta, 2.0 / n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def nesterov_subgrad(h0, n, target, n_iter, step_init, t0, budget):
    """Nesterov-accelerated subgradient descent."""
    h = h0.copy()
    h_prev = h0.copy()
    y = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = step_init
    stall = 0
    
    for it in range(n_iter):
        if time.time() - t0 > budget - 5:
            break
        g, _ = subgradient_c5(y, n, num_lags=30)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        g /= g_norm
        
        # Nesterov step
        t_k = 2.0 / (it + 2)
        h_new = project_h(y - step * g, target)
        y = h_new + (t_k - 1) / (t_k + 1) * (h_new - h_prev)
        y = np.clip(y, 0, 1)
        h_prev = h
        h = h_new
        
        c_cur = compute_c5(h)
        if c_cur < best_c5 - 1e-15:
            best_h = h.copy()
            best_c5 = c_cur
            stall = 0
        else:
            stall += 1
            if stall % 50 == 0:

    return best_h, best_c5
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad(h, n, num_lags=30):
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    thr = mc - 0.003 * (mc + 1e-10)
    idx = np.where(corr >= thr)[0]
    if len(idx) > num_lags:
        idx = idx[np.argsort(corr[idx])[::-1][:num_lags]]
    g = np.zeros(n)
    for i in idx:
        lag = int(i) - (n - 1)
        if lag >= 0:
            g[:n - lag] += (1.0 - h[lag:])
            g[lag:] -= h[:n - lag]
        else:
            la = -lag
            g[la:] += (1.0 - h[:n - la])
            g[:n - la] -= h[la:]
    return g / max(len(idx), 1), corr

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(np.clip(beta * (x - m), -500, 500))) / len(x)) / beta

    return smooth_max(np.correlate(h_in, 1.0 - h_in, mode="full"), beta) * dx

    return np.sum(x) - t

                     bounds=bounds, constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})

    rem = budget_s - (time.time() - t0)
    if rem > 25:
        nf = len(best_h)
        tf = nf / 2.0
        bf = Bounds(np.zeros(nf), np.ones(nf))
        best_h = do_slsqp(best_h, nf, tf, 5000, bf, 8000, 1e-15)
        best_c5 = compute_c5(best_h)
        rem = budget_s - (time.time() - t0)
        if rem > 15:
            best_h, best_c5 = nesterov_subgrad(best_h, nf, tf, 5000, 0.015, t0, budget_s)
        rem = budget_s - (time.time() - t0)
        if rem > 10:
            best_h, best_c5 = coord_descent(best_h, tf, nf, 50, t0, budget_s)
        rem = budget_s - (time.time() - t0)
        if rem > 5:
            best_h, best_c5 = pair_descent(best_h, tf, nf, 20, t0, budget_s)

    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
