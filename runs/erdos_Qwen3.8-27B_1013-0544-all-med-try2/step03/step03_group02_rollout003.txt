def corr_obj_and_grad(x, n, beta):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    val = smooth_max_val(c, beta)
    w = smooth_max_grad(c, beta)
    g = np.zeros(n)
    for k in range(len(c)):
        if w[k] < 1e-12:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            g[:n - lag] += w[k] * (1.0 - x[lag:]) * dx
            g[lag:] += w[k] * (-x[:n - lag]) * dx
        else:
            la = -lag
            g[la:] += w[k] * (1.0 - x[:n - la]) * dx
            g[:n - la] += w[k] * (-x[la:]) * dx
    return val, g

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-13) & (h < 1.0 - 1e-13)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    old_n = len(h)
    x_old = np.linspace(0.0, 2.0, old_n)
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h)

def slsqp_opt(h0, n, target, beta, bounds, maxiter):
    def obj(x):
        return corr_obj_and_grad(x, n, beta)[0]
    def obj_grad(x):
        return corr_obj_and_grad(x, n, beta)[1]
    def sum_eq(x):
        return np.sum(x) - target
    def sum_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_jac}]
    try:
        r = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(r.x, target)
    except Exception:
        return h0

def subgrad_active(h, n_act):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = corr * dx
    c_max = np.max(c)
    w = np.exp(-40.0 * (c_max - c))
    w = w / w.sum()
    h_grad = np.zeros(n)
    top = np.argsort(c)[::-1][:n_act]
    for idx in top:
        wi = w[idx]
        if wi < 1e-10:
            continue
        lag = int(idx) - (n - 1)
        if lag >= 0:
            h_grad[:n - lag] += wi * (1.0 - h[lag:]) * dx
            h_grad[lag:] += wi * (-h[:n - lag]) * dx
        else:
            la = -lag
            h_grad[la:] += wi * (1.0 - h[:n - la]) * dx
            h_grad[:n - la] += wi * (-h[la:]) * dx
    return h_grad, c_max

def subgrad_adaptive(cur, target, n, n_iter, t0, budget):
    best = compute_c5_val(cur)
    step = 0.008
    stall = 0
    for it in range(n_iter):
        if budget - (time.time() - t0) < 2:
            break
        g, c_max = subgrad_active(cur, 20)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        alpha = step
        accepted = False
        for _ in range(25):
            trial = project_h(cur - alpha * g, target)
            c_try = compute_c5_val(trial)
            if c_try < best - 1e-15:
                cur = trial
                best = c_try
                accepted = True
                break
            alpha *= 0.5
        if accepted:
            stall = 0
            step = min(step * 1.05, 0.03)
        else:
            stall += 1
            step *= 0.8
            if stall > 150:
                cur = project_h(cur + np.random.randn(n) * 0.008, target)
                best = compute_c5_val(cur)
                stall = 0
                step = 0.005
    return cur, best

def paired_coord(cur, target, n, sweeps, t0, budget):
    best = compute_c5_val(cur)
    rng = np.random.RandomState(42)
    for sw in range(sweeps):
        if budget - (time.time() - t0) < 1:
            break
        improved = False
        for _ in range(n):
            if budget - (time.time() - t0) < 0.5:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for d in [0.03, -0.03, 0.015, -0.015, 0.005, -0.005]:
                trial = cur.copy()
                trial[i] += d
                trial[j] -= d
                trial = project_h(trial, target)
                c_try = compute_c5_val(trial)
                if c_try < best - 1e-15:
                    cur = trial
                    best = c_try
                    improved = True
                    break
        if not improved and sw > 3:
            break
    return cur, best

def coord_fine(cur, target, n, sweeps, t0, budget):
    best = compute_c5_val(cur)
    steps = [0.01, -0.01, 0.005, -0.005, 0.002, -0.002, 0.001, -0.001, 0.0005, -0.0005]
    for sw in range(sweeps):
        if budget - (time.time() - t0) < 1:
            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 0.3:
                break
            for d in steps:
                nv = cur[j] + d
                if nv < 0 or nv > 1:
                    continue
                trial = cur.copy()
                trial[j] = nv
                trial = project_h(trial, target)
                c_try = compute_c5_val(trial)
                if c_try < best - 1e-15:
                    cur = trial
                    best = c_try
                    improved = True
                    break
        if not improved and sw > 2:
            break
    return cur, best

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(140, 0.5)

    best_h = h_parent.copy()
    best_c5 = compute_c5_val(best_h)
    best_n = len(h_parent)

    # Main optimization at parent resolution and nearby
    resolutions = sorted(set([len(h_parent)] + [280, 320, 360]))
    resolutions = [r for r in resolutions if r <= 450]

    for n_pts in resolutions:
        remaining = budget_s - (time.time() - t0)
        if remaining < 80:
            break
        target = n_pts / 2.0
        bounds = Bounds(np.zeros(n_pts), np.ones(n_pts))

        cur = project_h(resample(best_h, n_pts), target)
        cur_c5 = compute_c5_val(cur)

        # SLSQP with gradient - increasing beta
        for beta in [40, 100, 300, 800, 2000]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 8:
                break
            mi = max(800, int(remaining * 1.5))
            cur = slsqp_opt(cur, n_pts, target, beta, bounds, mi)
            cur_c5 = compute_c5_val(cur)
            if cur_c5 < best_c5:
                best_h, best_c5, best_n = cur.copy(), cur_c5, n_pts

        # Subgradient
        remaining = budget_s - (time.time() - t0)
        if remaining > 25:
            cur, cur_c5 = subgrad_adaptive(cur, target, n_pts, 4000, t0, budget_s)
            if cur_c5 < best_c5:
                best_h, best_c5, best_n = cur.copy(), cur_c5, n_pts

        # Paired coordinate
        remaining = budget_s - (time.time() - t0)
        if remaining > 15:
            cur, cur_c5 = paired_coord(cur, target, n_pts, 25, t0, budget_s)
            if cur_c5 < best_c5:
                best_h, best_c5, best_n = cur.copy(), cur_c5, n_pts

        # Fine coordinate
        remaining = budget_s - (time.time() - t0)
        if remaining > 10:
            cur, cur_c5 = coord_fine(cur, target, n_pts, 25, t0, budget_s)
            if cur_c5 < best_c5:
                best_h, best_c5, best_n = cur.copy(), cur_c5, n_pts

        # Perturbation restarts
        remaining = budget_s - (time.time() - t0)
        if remaining > 40:
            for att in range(6):
                remaining = budget_s - (time.time() - t0)
                if remaining < 10:
                    break
                scale = 0.025 / (att + 1)
                hp = project_h(cur + np.random.randn(n_pts) * scale, target)
                hp = slsqp_opt(hp, n_pts, target, 400, bounds, 800)
                cp = compute_c5_val(hp)
                if cp < cur_c5:
                    cur, cur_c5 = hp, cp
                if cur_c5 < best_c5:
                    best_h, best_c5, best_n = cur.copy(), cur_c5, n_pts

    # Final high-precision polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 15 and best_h is not None:
        nf = len(best_h)
        tf = nf / 2.0
        bf = Bounds(np.zeros(nf), np.ones(nf))
        for beta in [1500, 4000, 10000]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 4:
                break
            best_h = slsqp_opt(best_h, nf, tf, beta, bf, 8000)
            best_c5 = compute_c5_val(best_h)

        remaining = budget_s - (time.time() - t0)
        if remaining > 8:
            best_h, best_c5 = subgrad_adaptive(best_h, tf, nf, 5000, t0, budget_s)

        remaining = budget_s - (time.time() - t0)
        if remaining > 5:
            best_h, best_c5 = paired_coord(best_h, tf, nf, 30, t0, budget_s)

        remaining = budget_s - (time.time() - t0)
        if remaining > 3:
            best_h, best_c5 = coord_fine(best_h, tf, nf, 60, t0, budget_s)

    # Final constraint enforcement
    best_h = project_h(best_h, best_n / 2.0)
    best_c5 = compute_c5_val(best_h)
    return best_h, best_c5, best_n
