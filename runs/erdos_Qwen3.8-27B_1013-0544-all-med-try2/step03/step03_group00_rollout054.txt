```python
def projected_gradient_step(h, grad, lr, target):
    h_new = h - lr * grad
    # Project onto {0 <= h <= 1, sum(h) = target}
    # This requires finding a threshold
    # First clip, then adjust
    h_new = np.clip(h_new, 0, 1)
    # Now adjust to meet sum constraint
    # Water-filling: find alpha such that sum(clip(h_new + alpha, 0, 1)) = target
    # Binary search on alpha
    lo, hi = -1, 1
    for _ in range(50):
        mid = (lo + hi) / 2
        s = np.clip(h_new + mid, 0, 1).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    h_new = np.clip(h_new + (lo + hi) / 2, 0, 1)
    # Final correction
    diff = target - h_new.sum()
    if abs(diff) > 1e-15:
        free = (h_new > 1e-12) & (h_new < 1 - 1e-12)
        if free.sum() > 0:
            h_new[free] += diff / free.sum()
            h_new = np.clip(h_new, 0, 1)
    return h_new
```

```python
        best_jac['fun_count'] += 1

        h_pert = project_feasible(h_pert, len(h_best) / 2.0)
        if hop % 5 == 2:
            h_pert = block_perturb(h_pert, 3, seed + hop * 13)
        elif hop % 5 == 4:
            h_pert = make_sinusoidal(h_pert, 3, seed + hop * 17)
        h_loc, c5_loc = pg_optimize(h_pert, 300, 2000, 0.002, 0.92, 15,
                                    len(h_pert) / 2.0, t0, budget_s)
        if c5_loc < best_c5:
            best_c5 = c5_loc
            h_best = h_loc.copy()
            n_best = len(h_loc)

    # Phase 3: SLSQP polish on best
    if remaining() > 60:
        h_sls = slsqp_solve(h_best, 500, 500, top_k=15)
        c5_sls = compute_c5(h_sls)
        if c5_sls < best_c5:
            best_c5 = c5_sls
            h_best = h_sls.copy()

    # Phase 4: Try higher resolutions
    for n_try in [300, 400]:
        if remaining() < 120:
            break
        h_interp = interp_to_n(h_best, n_try)
        t_try = n_try / 2.0
        for beta in [50, 150, 300, 500]:
            if remaining() < 40:
                break
            h_try, c5_try = pg_optimize(h_interp, beta, 3000, 0.003 / (n_try / 200.0),
                                        0.92, max(10, 2000 // n_try), t_try, t0, budget_s)
            h_interp = h_try
        c5_final = compute_c5(h_interp)
        if c5_final < best_c5:
            best_c5 = c5_final
            h_best = h_interp.copy()
            n_best = n_try

    # Phase 5: More basin hopping at best resolution
    rng2 = np.random.RandomState(seed + 1000)
    for hop in range(20):
        if remaining() < 50:
            break
        noise_scale = 0.015 * (0.9 ** hop)
        h_pert = h_best + rng2.randn(len(h_best)) * noise_scale
        h_pert = project_feasible(h_pert, len(h_best) / 2.0)
        h_loc, c5_loc = pg_optimize(h_pert, 400, 2500, 0.0015, 0.93, 12,
                                    len(h_pert) / 2.0, t0, budget_s)
        if c5_loc < best_c5:
            best_c5 = c5_loc
            h_best = h_loc.copy()
            n_best = len(h_loc)

    # Phase 6: Final SLSQP + high beta PG
    if remaining() > 30:
        h_fin = slsqp_solve(h_best, 700, 400, top_k=12)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            h_best = h_fin.copy()

    if remaining() > 20:
        h_fin2, c5_fin2 = pg_optimize(h_best, 800, 3000, 0.0005, 0.94, 10,
                                      len(h_best) / 2.0, t0, budget_s)
        if c5_fin2 < best_c5:
            best_c5 = c5_fin2
            h_best = h_fin2.copy()

    # Final validation
    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)

    return h_best, best_c5, len(h_best)
```
