    return m + np.log(np.sum(np.exp(ac))) / alpha

        phase_deadline = min(deadline - 12, t0 + cum_frac * total_time)
        h_res, c_res = optimize_n(h_try, n, phase_deadline, rng)
        if c_res < best_c:
            best_c = c_res
            best_h = (h_res, n)
    if best_h is None:
        n = n_init
        best_h = (project_feasible(np.full(n, 0.5), n), n)
    bh, n = best_h
    remaining = deadline - time.time()
    if remaining > 10:
        bh, best_c = pair_search(bh, n, deadline,
                                 [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
    if remaining > 5 and compute_c5(bh, n) < best_c:
        pass
    bh = project_feasible(bh, n)
    best_c = compute_c5(bh, n)
    return bh, best_c, n
