        if lag == 0:
            grad[:] += wt * (1.0 - 2.0 * h) * dx
        elif lag > 0:
            grad[lag:] += wt * (1.0 - h[:n-lag]) * dx
            grad[:n-lag] -= wt * h[lag:] * dx
        else:
            L = -lag
            grad[L:] += wt * (1.0 - h[:n-L]) * dx
            grad[:n-L] -= wt * h[L:] * dx
    return grad

def smooth_max_obj(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    x_shift = np.clip(beta * (corr - m), -500, 500)
    return (m + np.log(np.sum(np.exp(x_shift)) / len(corr)) / beta) * dx

def resample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n0 = 250
        x = np.linspace(0, 2, n0)
        h0 = 0.5 + 0.15 * np.cos(np.pi * x)

    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n

    best_h = project_h(h0.copy(), target)
    best_c5 = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_c5 = best_c5

    # Phase 1: Adam subgradient descent (55% of budget)
    phase1_frac = 0.55
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    step_ct = 0
    patience = 0
    best_p1 = cur_c5

    while time.time() - t_start < budget_s * (phase1_frac + 1e-6):
        step_ct += 1
        elapsed = time.time() - t_start
        frac = min(1.0, elapsed / (budget_s * phase1_frac))

        top_k = max(2, int(7 - 5 * frac))
        beta = 60 + 500 * frac ** 1.5
        lr = 0.012 * (1.0 - 0.35 * frac)

        grad = subgrad_weighted(cur_h, dx, beta, top_k)
        # Add binary push penalty
        binary_push = 0.005 * (1.0 - 2.0 * cur_h)
        grad = grad + binary_push

        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break

        m_adam = 0.85 * m_adam + 0.15 * grad
        v_adam = 0.999 * v_adam + 0.001 * (grad ** 2)
        m_hat = m_adam / (1 - 0.85 ** step_ct)
        v_hat = v_adam / (1 - 0.999 ** step_ct)
        update = m_hat / (np.sqrt(v_hat) + 1e-8)

        new_h = project_h(cur_h - lr * update, target)
        new_c5 = compute_c5(new_h)

        if new_c5 < best_c5 - 1e-15:
            best_h = new_h.copy()
            best_c5 = new_c5
            cur_h = new_h
            cur_c5 = new_c5
            patience = 0
        elif new_c5 < cur_c5 + 0.002 * np.exp(-2.0 * frac):
            cur_h = new_h
            cur_c5 = new_c5
            patience += 1
        else:
            patience += 2

        if patience > 300:
            scale = 0.015 * np.exp(-patience / 3000.0)
            pert = np.random.normal(0, scale, n)
            h_try = project_h(cur_h + pert, target)
            c5_try = compute_c5(h_try)
            if c5_try < cur_c5 + scale:
                cur_h = h_try
                cur_c5 = c5_try
            patience = 0

    # Phase 2: SLSQP with smooth max (20% of budget)
    phase2_end = t_start + budget_s * 0.80
    bounds = Bounds(np.zeros(n), np.ones(n))

    for beta in [200, 500, 1000, 2000]:
        if time.time() > phase2_end:
            break
        def obj_fn(h_arr, b=beta):
            return smooth_max_obj(h_arr, dx, b)

        def con_fn(h_arr):
            return h_arr.sum() - target

        def con_jac(h_arr):
            return np.ones(n)

        cons = [{'type': 'eq', 'fun': con_fn, 'jac': con_jac}]
        try:
            res = minimize(obj_fn, best_h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 400, 'ftol': 1e-15})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        except Exception:
            pass

    # Phase 3: Fine subgradient refinement (20% of budget)
    phase3_end = t_start + budget_s * 0.97
    cur_h = best_h.copy()
    cur_c5 = best_c5

    while time.time() < phase3_end:
        grad = subgrad_weighted(cur_h, dx, 600, 2)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        step = 0.004
        improved = False
        for _ in range(50):
            h_trial = project_h(cur_h - step * grad, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < cur_c5 - 1e-15:
                cur_h = h_trial
                cur_c5 = c5_trial
                improved = True
                break
            step *= 0.5
        if improved and cur_c5 < best_c5:
            best_h = cur_h.copy()
            best_c5 = cur_c5
        elif not improved:
            # Try perturbation in a few random directions
            found = False
            for _ in range(5):
                d = np.random.normal(0, 0.002, n)
                h_try = project_h(cur_h + d, target)
                c5_try = compute_c5(h_try)
                if c5_try < cur_c5:
                    cur_h = h_try
                    cur_c5 = c5_try
                    found = True
                    break
            if not found:
                # Slightly larger perturbation
                d = np.random.normal(0, 0.008, n)
                h_try = project_h(cur_h + d, target)
                c5_try = compute_c5(h_try)
                if c5_try < cur_c5 + 0.0001:
                    cur_h = h_try
                    cur_c5 = c5_try

    # Quick check at n=200
    if time.time() - t_start < budget_s * 0.95:
        n_alt = 200
        target_alt = n_alt / 2.0
        dx_alt = 2.0 / n_alt
        h_alt = project_h(resample(best_h, n_alt), target_alt)
        c5_alt = compute_c5(h_alt)
        # Brief Adam
        m_a = np.zeros(n_alt)
        v_a = np.zeros(n_alt)
        sc = 0
        t_alt_end = min(time.time() + 20, t_start + budget_s * 0.95)
        while time.time() < t_alt_end:
            sc += 1
            grad_a = subgrad_weighted(h_alt, dx_alt, 300, 3)
            gn_a = np.linalg.norm(grad_a)
            if gn_a < 1e-15:
                break
            m_a = 0.85 * m_a + 0.15 * grad_a
            v_a = 0.999 * v_a + 0.001 * (grad_a ** 2)
            mh = m_a / (1 - 0.85 ** sc)
            vh = v_a / (1 - 0.999 ** sc)
            h_alt = project_h(h_alt - 0.008 * mh / (np.sqrt(vh) + 1e-8), target_alt)
            c5_alt = compute_c5(h_alt)
        if c5_alt < best_c5:
            best_h = h_alt
            best_c5 = c5_alt
            n = n_alt

    return best_h, best_c5, len(best_h)
