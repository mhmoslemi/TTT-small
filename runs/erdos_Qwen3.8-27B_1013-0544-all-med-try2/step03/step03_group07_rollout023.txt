        alpha = alpha0 / (1.0 + 0.01 * t)  # diminishing step size
        if t == 0:

        else:

            y = x + mu * (x - x_prev)

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_c5_core(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() > 0:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
            diff2 = target_sum - h.sum()
            if abs(diff2) > 1e-12 and free.sum() > 0:
                h[free] += diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def topk_gradient(h, top_idx, dx):
    n = len(h)
    grad = np.zeros(n)
    for ki in top_idx:
        s = ki - (n - 1)
        if s >= 0:
            i = np.arange(s, n)
            grad[i] += (1.0 - h[i - s])
            j = np.arange(0, n - s)
            grad[j] += (-h[j + s])
        elif s < 0:
            t = -s
            i = np.arange(0, n - t)
            grad[i] += (1.0 - h[i + t])
            j = np.arange(t, n)
            grad[j] += (-h[j - t])
        else:
            grad[:] += (1.0 - 2.0 * h)
    return grad / len(top_idx) * dx

def nesterov_pg(h0, n_iter, k, alpha0, decay=0.999):
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    x = h0.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5_core(x)
    for t in range(n_iter):
        alpha = alpha0 * (decay ** t)
        y = x.copy() if t == 0 else x + ((t - 1) / (t + 2)) * (x - x_prev)
        y = project_feasible(y, target)
        corr = np.correlate(y, 1.0 - y, mode="full")
        kk = min(k, len(corr))
        top_idx = np.argpartition(corr, -kk)[-kk:]
        grad = topk_gradient(y, top_idx, dx)
        x_new = project_feasible(y - alpha * grad, target)
        v = compute_c5_core(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x_prev, x = x, x_new
    return best_x, best_v

def slsqp_refine(h, beta, maxiter=200):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        s = corr - m
        return (m + np.log(np.sum(np.exp(np.clip(s * beta, -500, 500)))) / beta) * dx
    def gradient(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        s = corr - m
        exps = np.exp(np.clip(s * beta, -500, 500))
        w = exps / exps.sum()
        g = np.zeros(n)
        for ki in range(len(corr)):
            wk = w[ki]
            if wk < 1e-10:
                continue
            sci = ki - (n - 1)
            if sci >= 0:
                i = np.arange(sci, n)
                g[i] += wk * (1.0 - x[i - sci])
                j = np.arange(0, n - sci)
                g[j] += wk * (-x[j + sci])
            elif sci < 0:
                t = -sci
                i = np.arange(0, n - t)
                g[i] += wk * (1.0 - x[i + t])
                j = np.arange(t, n)
                g[j] += wk * (-x[j - t])
            else:
                g[:] += wk * (1.0 - 2.0 * x)
        return g * dx
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=gradient, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h.copy(), target)

                d[ki] = d_ki

            candidates = [0.0, 1.0]

                        if 0.0 < tc < 1.0:

def interp_resample(h, n_new):

    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

        h_new = slsqp_refine(h_best, beta, maxiter=300)

        if c_new < best_c5:
            h_best, best_c5 = h_new, c_new

    # Phase 2: Coordinate descent (exact 1D minimization per coordinate)
    if rem() > 60:
        h_cd, c_cd = coordinate_descent(h_best, n_sweeps=15)
        if c_cd < best_c5:
            h_best, best_c5 = h_cd, c_cd

    # Phase 3: Nesterov PG with small k (focus on worst lags)
    if rem() > 50:
        h_new, c_new = nesterov_pg(h_best, 3000, k=6, alpha0=0.04, decay=0.999)
        if c_new < best_c5:
            h_best, best_c5 = h_new, c_new

    # Phase 4: SLSQP after coordinate descent
    if rem() > 40:
        h_new = slsqp_refine(h_best, 1500, maxiter=250)
        c_new = compute_c5_core(h_new)
        if c_new < best_c5:
            h_best, best_c5 = h_new, c_new

    # Phase 5: More coordinate descent
    if rem() > 50:
        h_cd, c_cd = coordinate_descent(h_best, n_sweeps=20)
        if c_cd < best_c5:
            h_best, best_c5 = h_cd, c_cd

    # Phase 6: Random restarts with diverse scales
    for trial in range(12):
        if rem() < 55:
            break
        scale = 0.002 + 0.006 * (trial % 4)
        pert = np.random.randn(n) * scale
        h_pert = project_feasible(h_best + pert, n / 2.0)
        h_new = slsqp_refine(h_pert, 800, maxiter=200)
        c_new = compute_c5_core(h_new)
        if c_new < best_c5:
            h_best, best_c5 = h_new, c_new
            h_cd, c_cd = coordinate_descent(h_best, n_sweeps=10)
            if c_cd < best_c5:
                h_best, best_c5 = h_cd, c_cd

    # Phase 7: Nesterov PG with larger k
    if rem() > 60:
        h_new, c_new = nesterov_pg(h_best, 4000, k=24, alpha0=0.02, decay=0.9995)
        if c_new < best_c5:
            h_best, best_c5 = h_new, c_new
            h_cd, c_cd = coordinate_descent(h_best, n_sweeps=10)
            if c_cd < best_c5:
                h_best, best_c5 = h_cd, c_cd

    # Phase 8: Try different resolutions
    for n_try in [300, 500]:
        if rem() < 140:
            break
        h_i = interp_resample(h_best, n_try)
        h_new = slsqp_refine(h_i, 500, maxiter=200)
        c_new = compute_c5_core(h_new)
        if c_new < best_c5:
            h_best, best_c5 = h_new, c_new
            h_cd, c_cd = coordinate_descent(h_best, n_sweeps=10)
            if c_cd < best_c5:
                h_best, best_c5 = h_cd, c_cd

    # Phase 9: Final coordinate descent + SLSQP polish
    if rem() > 50:
        h_cd, c_cd = coordinate_descent(h_best, n_sweeps=25)
        if c_cd < best_c5:
            h_best, best_c5 = h_cd, c_cd
    if rem() > 35:
        h_new = slsqp_refine(h_best, 3000, maxiter=300)
        c_new = compute_c5_core(h_new)
        if c_new < best_c5:
            h_best, best_c5 = h_new, c_new
        h_cd, c_cd = coordinate_descent(h_best, n_sweeps=15)
        if c_cd < best_c5:
            h_best, best_c5 = h_cd, c_cd

    # Final feasibility
    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5_core(h_best)
    return h_best, best_c5, len(h_best)
