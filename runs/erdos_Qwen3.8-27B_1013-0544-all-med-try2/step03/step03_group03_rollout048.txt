    """Solve min_t max_k (A[k]*t + B[k]) for t in [t_low, t_high]."""

    t_grid = np.linspace(t_low, t_high, n_eval)
    # values[k, j] = A[k] * t_grid[j] + B[k]

    f_vals = values.max(axis=0)
